   But this changes the problem each time we update a coordinate, making it more complex.
   
   Alternatively, we can do:
   - Fix all a_j for j ≠ i
   - Choose a_i to minimize max_k (a*a)_k (no sum constraint)
   - Then re-normalize
   
   But the re-normalization changes everything, so this isn't clean.
   
   I think coordinate descent doesn't work well here because of the sum constraint and the non-negativity constraint interacting in complex ways.

Let me go back to the LP approach, which is more straightforward.

**Revised Plan:**

1. **Initialization**: Use parent sequence, try a few profile variations at n=1024 and n=2048.

2. **LP-based alternating minimization** (main workhorse):
   - Given current a (normalized, sum=1):
     - Compute c = a*a, M = max(c)
     - Solve LP: max sum(b) s.t. a*b <= M (or factor*M), b >= 0
     - Get b, normalize to b_norm
     - Line search for optimal t in blend a_new = (1-t)*a + t*b_norm
     - Update a = a_new
     - Repeat

3. **Gradient polish** (smooth max + Adam):
   - Use log-sum-exp approximation of max
   - Adam optimizer in log space (simplex constraint handled by softmax)

4. **Multi-restart**: Perturb the best solution and try again.

Let me now write the actual code. I'll make the LP refinement the main focus, with careful attention to:
- The LP M parameter (try M = 1.0 * peak and M = 0.98 * peak)
- Line search with golden section
- Proper handling of the normalization

"""
Step function optimizer using intensive LP-based alternating minimization
with adaptive M, golden-section line search, and smooth-max Adam polishing.
Targets score < 1.5027 by aggressively flattening the self-convolution peak.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = (a * b)[k]"""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M):
    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0"""
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _smooth_max_weights(c, beta):
    """Compute softmax weights for smooth max approximation."""
    c = np.asarray(c, dtype=np.float64).ravel()
    m = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= m - 1e-10
        cnt = mask.sum()
        if cnt == 0:
            w[np.argmax(c)] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - m) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = ex.sum()
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[np.argmax(c)] = 1.0
        s = 1.0
    return ex / s


def _cross_corr(w, p):
    """Compute cross-correlation: (w * p_reversed) trimmed to length n."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - z.max()
    e = np.exp(z)
    p = e + floor
    p = p / p.sum()
    return p


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def _adam_logit(p0, deadline, rng, lr=0.04, beta_frac=0.06, beta_decay=0.94,
                no_imp_limit=500):
    """Adam optimization in log-space with smooth max objective."""
    p0 = np.maximum(p0, 1e-12).ravel()
    p0 = p0 / p0.sum()
    n = p0.size
    z = np.log(p0)
    z -= z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    p = p0.copy()
    c = _full_convolve(p)
    maxc = c.max()
    beta = max(1e-4, beta_frac * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    
    while time.monotonic() < deadline:
        w = _smooth_max_weights(c, beta)
        g = _cross_corr(w, p)
        
        # Gradient in log space (Fisher metric)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_step = 0
            continue
        
        t_step += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1**t_step)
        vh = v / (1 - b2**t_step)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_step = 0
            continue
        
        z = np.clip(z, -50, 50)
        z -= z.mean()
        p = _softmax(z)
        c = _full_convolve(p)
        maxc = c.max()
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if t_step % 20 == 0:
            beta = max(5e-5, beta * beta_decay)
        
        if no_imp > no_imp_limit:
            # Reset with noise
            ns = 0.05
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p = _softmax(z)
            c = _full_convolve(p)
            maxc = c.max()
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0)
            v.fill(0)
            t_step = 0
            no_imp = 0
    
    return best_p, best_score


def _lp_refine_intensive(p, deadline, m_factors=None, max_iters=200):
    """Intensive LP refinement with adaptive M and line search."""
    if m_factors is None:
        m_factors = [1.0, 0.99, 0.98, 0.97, 0.95]
    
    p = _normalize(p)
    if p is None:
        return p, np.inf
    
    best_p = p.copy()
    best_score = _score_norm(p)
    n = p.size
    
    c = _full_convolve(p)
    M_cur = float(c.max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Try different M factors
        best_this_iter = None
        best_this_score = best_score
        
        for mf in m_factors:
            if time.monotonic() > deadline:
                break
            M = mf * M_cur
            b = _lp_solve(p, M)
            if b is None:
                continue
            s_b = b.sum()
            if s_b < 1e-10:
                continue
            b_norm = b / s_b
            
            # Line search
            best_t = 0
            best_sc = best_score
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                p_try = (1 - t) * p + t * b_norm
                p_try = np.maximum(p_try, 1e-15)
                p_try = p_try / p_try.sum()
                sc = _score_norm(p_try)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
            
            if best_t > 0 and best_sc < best_this_score:
                best_this_score = best_sc
                best_this_iter = (1 - best_t) * p + best_t * (b / s_b)
        
        if best_this_iter is None:
            # No improvement, try smaller M factors
            break
        
        p = _normalize(best_this_iter)
        if p is None:
            break
        sc = _score_norm(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
        
        c = _full_convolve(p)
        M_cur = float(c.max())
    
    return best_p, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exp(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t-c1)/s)**2) + np.exp(-0.5 * ((t-c2)/s)**2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = p0.ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return _normalize(np.interp(x1, x0, p0))
