    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    return e / s


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_h(a, weights, active_indices):
    """Compute h_i = sum_k weights[k] * a[k-i] for valid i."""
    n = len(a)
    h = np.zeros(n)
    for idx, k in enumerate(active_indices):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            h[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
    return h


def _z_objective(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        return float(c_top.max() + beta * np.log(np.sum(np.exp((c_top - c_top.max()) / beta))))
    else:
        cm = float(c.max())
        return float(cm + beta * np.log(np.sum(np.exp((c - cm) / beta))))


def _z_gradient(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        w_top = np.exp((c_top - c_top.max()) / beta)
        w_top = w_top / w_top.sum()
        w = np.zeros(len(c))
        w[idx] = w_top
    else:
        cm = float(c.max())
        w = np.exp((c - cm) / beta)
        w = w / w.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, max_wall_time, top_k=0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())
    beta_starts = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]
    per_level = max_wall_time / len(betas)
    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _z_objective, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 20, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass
    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.04, top_k=0, n_max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)
    if top_k == 0:
        adaptive_k = True
        cur_k = 3
    else:
        adaptive_k = False
        cur_k = top_k
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        use_k = cur_k if adaptive_k else top_k
        grad = _z_gradient(z, beta, n, use_k)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 100 == 0:
            beta = max(1e-14, beta * 0.85)
            if adaptive_k and t % 500 == 0:
                cur_k = min(cur_k + 2, 50)
        if t % 2000 == 0:
            lr = max(0.002, lr * 0.96)
    return best_a, best_score


def _kkt_multiplicative(a_init, deadline, max_iter=100000):
    """KKT balancing with multiplicative updates and adaptive parameters."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    momentum = np.zeros(n)
    eta = 0.03
    beta_m = 0.85
    stall_count = 0
    last_improve_score = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Adaptive threshold: start loose, tighten over time
        if it < 50:
            thresh_frac = 0.90
        elif it < 200:
            thresh_frac = 0.95
        elif it < 1000:
            thresh_frac = 0.98
        elif it < 5000:
            thresh_frac = 0.99
        else:
            thresh_frac = 0.995

        active = np.where(c >= thresh_frac * cm)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute h with uniform weights
        h = np.zeros(n)
        for k in active:
            k = int(k)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                h[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
        h /= len(active)

        mu = float(np.dot(h, a) / a.sum())
        excess = h - mu
        max_excess = np.abs(excess).max()
        if max_excess < 1e-15:
            break
        excess_norm = excess / max_excess

        momentum = beta_m * momentum + (1 - beta_m) * excess_norm

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * momentum
        log_a = np.clip(log_a, -60, 60)
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            eta = min(eta * 1.002, 0.08)
        else:
            a = a_new  # still update to explore
            stall_count += 1
            eta *= 0.85
            if stall_count > 200:
                # Reset momentum and try fresh
                momentum = np.zeros(n)
                stall_count = 0

        # Periodic re-optimization: if stalled a lot, try a different approach
        if stall_count > 500:
            # Try a small perturbation
            rng_local = np.random.default_rng(it)
            z = np.log(np.maximum(a, 1e-300))
            z += 0.02 * rng_local.normal(size=n)
            z -= z.mean()
            a = _softmax_fn(z)
            momentum = np.zeros(n)
            eta = 0.05
            stall_count = 0

        # Convergence check
        if it % 5000 == 0 and it > 1000:
            if best_score - last_improve_score < 1e-12:
                break
            last_improve_score = best_score

    return best_a, best_score


def _lp_direction(a, active_indices, n_active_max=80):
    """
    Solve the linearized LP to find the optimal direction.
    Minimize T subject to c_k(a) + 2*(a_shifted_k . delta) <= T, a+delta >= 0, sum(delta)=0.
    Returns the direction delta (or None if infeasible).
    """
    n = len(a)
    c = _full_convolve(a, a)

    # Limit active set size
    if len(active_indices) > n_active_max:
        c_vals = c[active_indices]
        top_local = np.argsort(c_vals)[-n_active_max:]
        active_indices = active_indices[top_local]

    m = len(active_indices)
    # Variables: delta_0, ..., delta_{n-1}, T  (n+1 vars)
    # Constraints:
    #   For each k in A: 2*sum_i a_{k-i} * delta_i - T <= -c_k  (m constraints)
    #   For each i: -delta_i <= a_i  (n constraints)
    #   sum delta_i <= 0  (1 constraint)
    #   -sum delta_i <= 0  (1 constraint)
    # Objective: minimize T  ->  c_obj = [0,...,0, 1]

    # Build constraint matrix (sparse would be ideal, but let's try dense for small m)
    # Total rows: m + n + 2
    # Total cols: n + 1

    A_ub = np.zeros((m + n + 2, n + 1))
    b_ub = np.zeros(m + n + 2)

    # Convolution constraints
    for idx, k in enumerate(active_indices):
        k = int(k)
        c_val = float(c[k])
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            A_ub[idx, lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        A_ub[idx, n] = -1.0  # -T
        b_ub[idx] = -c_val

    # Non-negativity: -delta_i <= a_i
    I = np.eye(n)
    A_ub[m:m + n, :n] = -I
    b_ub[m:m + n] = a

    # Sum constraints
    A_ub[m + n, :n] = 1.0
    A_ub[m + n + 1, :n] = -1.0
    b_ub[m + n] = 0.0
    b_ub[m + n + 1] = 0.0

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize T

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs')
        if res.success:
            delta = res.x[:n]
            return delta, float(res.x[n])
    except Exception:
        pass
    return None, None


def _lp_polish(a_init, deadline, max_iters=100):
    """Use LP-based direction finding for polishing."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    step_scale = 0.5

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set: top peaks within 99% of max
        thresh = 0.99 * cm
        active = np.where(c >= thresh)[0]
        if len(active) > 80:
            active = active[np.argsort(c[active])[-80:]]

        delta, T_opt = _lp_direction(a, active)
        if delta is None:
            break

        # Line search: find best step size alpha in [0, 1]
        best_alpha = 0.0
        best_sc = _score_norm(a)
        for alpha_try in [step_scale, step_scale * 0.5, step_scale * 0.25, step_scale * 0.1]:
            a_try = a + alpha_try * delta
            if np.any(a_try < -1e-10):
                continue
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try / a_try.sum()
            sc_try = _score_norm(a_try)
            if np.isfinite(sc_try) and sc_try < best_sc:
                best_sc = sc_try
                best_alpha = alpha_try

        if best_alpha > 0:
            a_new = a + best_alpha * delta
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            sc_new = _score_norm(a_new)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                step_scale = min(step_scale * 1.05, 0.8)
            else:
                step_scale *= 0.7
        else:
            step_scale *= 0.7
            if step_scale < 0.01:
                break

    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_hann_profile(n):
    t = np.arange(n) / (n - 1)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_kaiser_profile(n, beta=5.0):
    from scipy.special import iv
    t = np.arange(n) / (n - 1)
    x = 2.0 * t - 1.0
    a = iv(0, beta * np.sqrt(np.maximum(1 - x * x, 0))) / iv(0, beta)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_blackman_profile(n):
    t = np.arange(n) / (n - 1)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: profile init, L-BFGS continuation,
    Adam, multiplicative KKT balancing, LP direction finding.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Phase 0: Initial candidates
    print("=== Phase 0: Initial candidates ===")
    phase0_end = min(deadline, t_start + 30.0)
    n_search = 4096
    candidates = []

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                if len(parent) != n_search:
                    parent = _interp_profile(parent, n_search)
                sc = _score_norm(parent)
                candidates.append((sc, parent.copy(), "parent"))
                update_best(parent, sc, "parent")
                print(f"  Parent: n={n_search}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    profiles = []
    for alpha in [0.0005, 0.001, 0.002, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02]:
        profiles.append(_make_exp_profile(n_search, alpha))
    profiles.append(_make_hann_profile(n_search))
    profiles.append(_make_kaiser_profile(n_search, 4.0))
    profiles.append(_make_kaiser_profile(n_search, 6.0))
    profiles.append(_make_kaiser_profile(n_search, 8.0))
    profiles.append(_make_blackman_profile(n_search))
    for a1, a2 in [(-1.0, 0.3), (-0.5, 0.5), (0.5, -0.3), (-1.5, 0.7), (1.0, -0.5)]:
        profiles.append(_make_quadratic_profile(n_search, a1, a2))

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # Phase 1: L-BFGS on top candidates
    print("=== Phase 1: L-BFGS continuation ===")
    phase1_end = min(deadline, t_start + 400.0)
    count = 0
    seen = set()
    for sc, p, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= 3:
            break
        if sc > best_score * 1.01:
            break
        if label in seen:
            continue
        seen.add(label)
        remaining = phase1_end - time.monotonic()
        per_time = max(10.0, min(130.0, remaining / max(1, 3 - count)))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  L-BFGS on {label}, budget={per_time:.0f}s")
        p_best, sc_best = _run_lbfgs(p, dl, max_wall_time=per_time, top_k=0)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1
    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: Adam polish
    print("=== Phase 2: Adam polish ===")
    phase2_end = min(deadline, t_start + 750.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + remaining * 0.6)
        print(f"  Adam pass 1, budget={phase2_end - time.monotonic():.0f}s")
        p_best, sc_best = _adam_polish(best_seq, dl, lr=0.035, top_k=0, n_max_iter=60000)
        update_best(p_best, sc_best, "adam1")
        if time.monotonic() < phase2_end - 20:
            dl2 = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.5)
            p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.015, top_k=10, n_max_iter=30000)
            update_best(p_best2, sc_best2, "adam2")
    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: KKT multiplicative balancing
    print("=== Phase 3: KKT multiplicative balancing ===")
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + remaining)
        print(f"  KKT balance, budget={remaining:.0f}s")
        p_best, sc_best = _kkt_multiplicative(best_seq, dl, max_iter=200000)
        update_best(p_best, sc_best, "kkt")
        # Second KKT pass with different params
        if time.monotonic() < phase3_end - 30:
            dl2 = min(phase3_end - 5, time.monotonic() + min(120.0, (phase3_end - time.monotonic()) * 0.5))
            p_best2, sc_best2 = _kkt_multiplicative(best_seq, dl2, max_iter=100000)
            update_best(p_best2, sc_best2, "kkt2")
    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: LP-based direction finding
    print("=== Phase 4: LP direction finding ===")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        remaining = phase4_end - time.monotonic()
        dl = min(phase4_end, time.monotonic() + remaining)
        print(f"  LP polish, budget={remaining:.0f}s")
        p_best, sc_best = _lp_polish(best_seq, dl, max_iters=80)
        update_best(p_best, sc_best, "lp")
    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Multi-restart with perturbation
    print("=== Phase 5: Multi-restart ===")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = len(best_seq)
        for i in range(3):
            if time.monotonic() > phase5_end - 30:
                break
            noise_levels = [0.03, 0.06, 0.1]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i] * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            per_restart = min(80.0, (phase5_end - time.monotonic()) / max(1, 3 - i))
            dl = min(phase5_end - 5, time.monotonic() + per_restart * 0.5)
            p_m, sc_m = _run_lbfgs(p_start, dl, max_wall_time=per_restart * 0.5, top_k=5)
            update_best(p_m, sc_m, f"restart{i}_lbfgs")
            if time.monotonic() < phase5_end - 10:
                dl2 = min(phase5_end - 5, time.monotonic() + per_restart * 0.4)
                p_a, sc_a = _kkt_multiplicative(p_m, dl2, max_iter=50000)
                update_best(p_a, sc_a, f"restart{i}_kkt")
    print(f"  Phase 5 done: best={best_score:.10f}")

    # Phase 6: Final polish
    print("=== Phase 6: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            dl = min(deadline - 5, time.monotonic() + min(40.0, remaining * 0.5))
            p_best, sc_best = _kkt_multiplicative(best_seq, dl, max_iter=50000)
            update_best(p_best, sc_best, "final_kkt")
        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 3, time.monotonic() + min(30.0, deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _lp_polish(best_seq, dl2, max_iters=30)
            update_best(p_best2, sc_best2, "final_lp")

    # Final validation
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.003)
            final_score = _score_fn(final_seq)
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    fallback = _make_exp_profile(4096, 0.003)
    print(f"  FALLBACK: n=4096, score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
