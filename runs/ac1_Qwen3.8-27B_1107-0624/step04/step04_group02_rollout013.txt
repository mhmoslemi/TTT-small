def _next_pow2(n):
    if n <= 1:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    grad = np.zeros(n)
    for k_val in active:

        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k_val - hi:k_val - lo + 1][::-1]
    grad /= len(active)
    return grad, cm, active

    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * cm, a
    log_s = float(np.log(s))
    obj = 2.0 * n * (cm + beta * log_s)
    return obj, a


def _logsumexp_grad(z, n, beta):

    a /= a.sum()
    c = _full_convolve(a, a)
    cm = float(c.max())

def _subgradient_optimize(a_init, deadline, max_wall, alpha=0.003, n_iter=5000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = 1.0 / (4.0 * n)

    for it in range(n_iter):
        if time.monotonic() > deadline - 1.0:
            break

        grad, cm, active = _compute_subgradient_max(a, alpha)
        obj = 2.0 * n * cm

        step = lr / (1.0 + 0.1 * it)

        for backtrack in range(20):
            a_trial = a - step * grad
            a_trial = _project_simplex(a_trial)
            c_trial = _full_convolve(a_trial, a_trial)
            cm_trial = float(c_trial.max())
            obj_trial = 2.0 * n * cm_trial
            if obj_trial < obj - 0.5 * step * step * 0:
                a = a_trial
                obj = obj_trial
                break
            step *= 0.5

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

def _local_search_structured(a_init, deadline, rng, n_trials=5000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    t_best = time.monotonic()

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        if time.monotonic() - t_best > 30.0:
            a = best_a.copy()
            t_best = time.monotonic()

        mode = rng.integers(0, 3)
        a_trial = a.copy()

        if mode == 0:
            # Smooth bump perturbation
            width = rng.integers(10, max(20, n // 8))
            center = rng.integers(0, n)
            t = np.arange(n)
            bump = np.exp(-(t - center) ** 2 / (2.0 * width ** 2))
            scale = rng.uniform(0.005, 0.03)
            a_trial += scale * bump * rng.choice([-1, 1])
        elif mode == 1:
            # Polynomial perturbation
            t = np.linspace(0, 1, n)
            poly = rng.normal(size=4) @ np.stack([t**0, t**1, t**2, t**3], axis=1)
            scale = rng.uniform(0.002, 0.01)
            a_trial += scale * poly
        else:
            # Small block multiplicative perturbation
            bs = rng.integers(5, max(10, n // 20))
            start = rng.integers(0, n - bs)
            a_trial[start:start + bs] *= rng.uniform(0.9, 1.1, size=bs)

        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
            t_best = time.monotonic()

    return best_a, best_score


def _make_profile(n, profile_type, **kwargs):

    if profile_type == 'uniform':

    elif profile_type == 'linear':
        slope = kwargs.get('slope', 1.0)
        a = 1.0 + slope * u
    elif profile_type == 'quadratic':
        a1 = kwargs.get('a1', 0.5)
        a2 = kwargs.get('a2', 0.3)
        a = 1.0 + a1 * u + a2 * u ** 2
    elif profile_type == 'cubic':
        a1 = kwargs.get('a1', 0.5)
        a3 = kwargs.get('a3', -0.2)
        a = 1.0 + a1 * u + a3 * u ** 3
    elif profile_type == 'quartic':
        a1 = kwargs.get('a1', 0.3)
        a4 = kwargs.get('a4', -0.4)
        a = 1.0 + a1 * u + a4 * u ** 4
    elif profile_type == 'gaussian':
        width = kwargs.get('width', 0.3)
        a = np.exp(-u ** 2 / (2.0 * width ** 2))
    elif profile_type == 'cosine':
        alpha = kwargs.get('alpha', 1.0)
        a = 0.5 * (1.0 + np.cos(np.pi * alpha * u))
    elif profile_type == 'sine_sq':
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-12)
    elif profile_type == 'cos_sq':
        a = np.cos(np.pi * u / 2.0) ** 2
        a = np.maximum(a, 1e-12)
    elif profile_type == 'power':
        p = kwargs.get('p', 2.0)
        a = (1.0 - np.abs(u)) ** p
        a = np.maximum(a, 1e-12)
    elif profile_type == 'fourier':
        K = kwargs.get('K', 5)
        coeffs = kwargs.get('coeffs', None)
        if coeffs is None:
            coeffs = np.ones(K) / np.arange(1, K + 1)
        a = np.ones(n)
        for k in range(1, K + 1):
            a += coeffs[k - 1] * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 1e-12)
    elif profile_type == 'twisted':
        freq = kwargs.get('freq', 3.0)
        amp = kwargs.get('amp', 0.3)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _polynomial_optimize(a_init, deadline, max_wall, degree=6):
    n = len(a_init)
    t = np.linspace(0.0, 1.0, n)
    features = np.stack([t**k for k in range(degree + 1)], axis=1)

    z_init = np.log(np.maximum(a_init, 1e-15))

    def obj_fn(coefs):
        z = features @ coefs
        z = z - z.max()
        a = np.exp(z)
        a /= a.sum()
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = 0.01 * max_c0
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * float(np.log(s)))

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    try:
        res = minimize(obj_fn, z_init, method='L-BFGS-B',
                       options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-10})
        if res.x is not None:
            z = features @ res.x
            a = np.exp(z - z.max())
            a /= a.sum()
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
    except Exception:
        pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 1e-15, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_work = 4096
    if parent is not None:
        n_work = len(parent)
        sc_p = _score_norm(parent)
        print(f"Parent: n={n_work}, score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        parent = _make_profile(n_work, 'cos_sq')
        sc_p = _score_norm(parent)
        print(f"No parent, using cos_sq n={n_work}, score={sc_p:.10f}")
        update_best(parent, sc_p, "init")

    # Phase 0: Diverse initialization (20s)
    phase0_end = min(deadline, t_start + 20.0)
    candidates = [(best_score, best_seq.copy())]

    inits = [
        _make_profile(n_work, 'cos_sq'),
        _make_profile(n_work, 'sine_sq'),
        _make_profile(n_work, 'gaussian', width=0.35),
        _make_profile(n_work, 'gaussian', width=0.25),
        _make_profile(n_work, 'quadratic', a1=0.8, a2=0.2),
        _make_profile(n_work, 'quartic', a1=0.5, a4=-0.3),
        _make_profile(n_work, 'twisted', freq=2.0, amp=0.2),
        _make_profile(n_work, 'twisted', freq=4.0, amp=0.15),
        _make_profile(n_work, 'power', p=3.0),
        _make_profile(n_work, 'linear', slope=0.8),
    ]
    for p in inits:
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p.copy()))
        update_best(p, sc, "init")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: best={best_score:.10f}, n_cands={len(candidates)}")

    # Phase 1: L-BFGS logsumexp on top candidates (500s)
    phase1_end = min(deadline, t_start + 520.0)
    n_to_opt = 3
    per_cand = max(60.0, (phase1_end - time.monotonic()) / max(1, n_to_opt))
    count = 0
    for sc, p in candidates:
        if time.monotonic() > phase1_end - 5.0 or count >= n_to_opt:
            break
        dl = min(phase1_end - 3.0, time.monotonic() + per_cand)
        p_b, sc_b = _run_lbfgs_ls(p, dl, per_cand, beta_start=0.03)
        update_best(p_b, sc_b, f"lbfgs_{count}")
        count += 1
    print(f"Phase 1: best={best_score:.10f}")

    # Phase 2: Subgradient optimization (200s)
    phase2_end = min(deadline, t_start + 720.0)
    if time.monotonic() < phase2_end:
        dl = phase2_end - 3.0
        p_b, sc_b = _subgradient_optimize(best_seq, dl, 180.0, alpha=0.002, n_iter=3000)
        update_best(p_b, sc_b, "subgrad")
    print(f"Phase 2: best={best_score:.10f}")

    # Phase 3: Fine L-BFGS with small beta (300s)
    phase3_end = min(deadline, t_start + 1020.0)
    if time.monotonic() < phase3_end:
        dl = phase3_end - 3.0
        max_w = phase3_end - time.monotonic()
        p_b, sc_b = _run_lbfgs_ls(best_seq, dl, max_w, beta_start=0.005, n_betas=6)
        update_best(p_b, sc_b, "lbfgs_fine")
    print(f"Phase 3: best={best_score:.10f}")

    # Phase 4: Structured local search (200s)
    phase4_end = min(deadline, t_start + 1220.0)
    if time.monotonic() < phase4_end:
        dl = phase4_end - 3.0
        p_b, sc_b = _local_search_structured(best_seq, dl, rng, n_trials=8000)
        update_best(p_b, sc_b, "local")
    print(f"Phase 4: best={best_score:.10f}")

    # Phase 5: Restart + re-optimize (250s)
    phase5_end = min(deadline, t_start + 1470.0)
    if time.monotonic() < phase5_end:
        n_restarts = 4
        per_r = max(30.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 5:
                break
            noise = 0.02 * (i + 1)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean() + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_s = np.exp(z)
            p_s /= p_s.sum()
            dl = min(phase5_end - 3, time.monotonic() + per_r)
            p_b, sc_b = _run_lbfgs_ls(p_s, dl, per_r, beta_start=0.01, n_betas=5)
            update_best(p_b, sc_b, f"restart{i}")
    print(f"Phase 5: best={best_score:.10f}")

    # Phase 6: Try larger n (200s)
    phase6_end = min(deadline, t_start + 1670.0)
    if time.monotonic() < phase6_end and len(best_seq) < 8192:
        n_big = min(8192, len(best_seq) * 2)
        p_big = _interp_profile(best_seq, n_big)
        sc_big = _score_norm(p_big)
        update_best(p_big, sc_big, f"interp_n{n_big}")
        if time.monotonic() < phase6_end - 30:
            dl = min(phase6_end - 5, time.monotonic() + 120.0)
            p_b, sc_b = _run_lbfgs_ls(p_big, dl, 100.0, beta_start=0.02, n_betas=5)
            update_best(p_b, sc_b, f"lbfgs_n{n_big}")
    print(f"Phase 6: best={best_score:.10f}")

    # Phase 7: Final polish (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_seq is not None:
        # Poly optimize
        if remaining > 60:
            dl = min(deadline - 5, time.monotonic() + min(60, remaining / 3))
            p_b, sc_b = _polynomial_optimize(best_seq, dl, min(50, remaining / 3), degree=5)
            update_best(p_b, sc_b, "poly")

        # Subgradient fine
        if remaining > 40:
            dl = min(deadline - 5, time.monotonic() + min(40, remaining / 3))
            p_b, sc_b = _subgradient_optimize(best_seq, dl, min(35, remaining / 3), alpha=0.001, n_iter=5000)
            update_best(p_b, sc_b, "subgrad_fine")

        # Local search final
        if remaining > 20:
            dl = min(deadline - 3, time.monotonic() + max(10, remaining / 3))
            p_b, sc_b = _local_search_structured(best_seq, dl, rng, n_trials=5000)
            update_best(p_b, sc_b, "final_local")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile(4096, 'cos_sq')
            s = float(final_seq.sum())
        final_seq = final_seq / s
        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        return [float(x) for x in final_seq]

    fallback = _make_profile(4096, 'cos_sq')
    print(f"FALLBACK: n=4096, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
