"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive L-BFGS, exact subgradient with line search, Nesterov acceleration,
and multi-restart with diverse initializations.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Compute the full score."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:

    return w, cm

    # h[m] = sum_k w[k] * a[k-m]
    # This equals (w * a_flipped)[m + n - 1] where a_flipped = a[::-1]
    # Because (w * a_flipped)[j] = sum_k w[k] * a_flipped[j-k] = sum_k w[k] * a[n-1-(j-k)]
    # = sum_k w[k] * a[n-1-j+k]
    # We want h[m] = sum_k w[k] * a[k-m]
    # So n-1-j+k = k-m => n-1-j = -m => j = m + n - 1
    h = w_conv[n - 1:2 * n - 1]
    return h


def _smooth_objective(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _smooth_gradient(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    h = _conv_w_with_a(w, a)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _effective_support(w):
    """1 / entropy-based measure of effective support."""
    p = w + 1e-300
    log_p = np.log(p)
    H = -np.sum(p * log_p)
    return float(np.exp(H))


def _adaptive_beta(w, beta_max, beta_min):
    """Adapt beta based on effective support."""
    eff = _effective_support(w)
    # If effective support is small, we can use smaller beta
    # If effective support is large, use larger beta
    target_ratio = eff / 10.0  # Normalize
    ratio = min(1.0, max(0.1, target_ratio))
    return beta_max * ratio ** 0.5


def _lbfgs_adaptive(a_init, deadline, max_wall_time=None):
    """L-BFGS with adaptive beta and more aggressive schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    scale = max_c0

    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))

    # More aggressive beta schedule
    betas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    betas = [b * scale for b in betas]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)

        # First, check if we should skip to a smaller beta
        a_curr = _softmax_fn(z)
        c_curr = _full_convolve(a_curr, a_curr)
        w_curr, _ = _compute_weights(c_curr, beta)
        eff = _effective_support(w_curr)
        if eff < 2.0:
            # Already very peaked, skip to smaller beta
            continue

        try:
            result = minimize(
                _smooth_objective, z, args=(beta,),
                method='L-BFGS-B', jac=_smooth_gradient,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
                # Always update z for continuity even if score didn't improve
                # (the smoothed objective might have improved even if the true score didn't)
        except Exception:
            pass

    return best_a, best_score


def _nesterov_adaptive(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    """Nesterov with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    scale = max_c

    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * scale)

        stall_count = 0
        while time.monotonic() < deadline:
            t += 1
            grad_y = _smooth_gradient(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _smooth_gradient(z, beta)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 100 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                    if stall_count > 5:
                        break  # Move to next beta level

            if t % 200 == 0:
                mom = min(0.999, mom + 0.0003)
                # Adaptive learning rate
                a_curr = _softmax_fn(z)
                c_curr = _full_convolve(a_curr, a_curr)
                max_c = float(c_curr.max())
                scale = max_c

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        scale = max_c
        lr = max(0.002, lr * 0.75)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_line_search(a_init, deadline, n_iter=5000):
    """Exact subgradient method with backtracking line search."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    # Initial learning rate for line search
    step = 0.01

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Use the exact max (argmax) for the subgradient
        k_star = int(np.argmax(c))
        # Subgradient: grad[m] = 2*a[k-m] (if 0 <= k-m < n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        grad = np.zeros(n)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_star - m]
        # Scale by 2n for the objective
        grad = 2.0 * n * grad

        # Normalize gradient for step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad_dir = -grad / grad_norm

        # Backtracking line search
        a_curr_score = best_score  # Use best score as reference
        z_curr = z.copy()
        
        # Try different step sizes
        for trial in range(10):
            s_try = step / (2 ** trial)
            z_try = z_curr + s_try * grad_dir
            z_try = z_try - z_try.mean()
            z_try = np.clip(z_try, -50, 50)
            a_try = _softmax_fn(z_try)
            sc_try = _score_norm_fn(a_try)
            if np.isfinite(sc_try) and sc_try < a_curr_score:
                z = z_try
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                step = min(0.1, step * 1.2)  # Increase step if successful
                break
        else:
            step = step * 0.5

        # Also try small exact steps in the subgradient direction
        if it % 50 == 0:
            # Try a direct step
            z_small = z - 0.001 * grad / (grad_norm + 1e-15)
            z_small = z_small - z_small.mean()
            z_small = np.clip(z_small, -50, 50)
            a_small = _softmax_fn(z_small)
            sc_small = _score_norm_fn(a_small)
            if np.isfinite(sc_small) and sc_small < best_score:
                best_score = sc_small
                best_a = a_small.copy()
                z = z_small

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01):
    """Adam optimizer with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    scale = max_c
    beta = max(1e-12, beta_frac * scale)

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        h = _conv_w_with_a(w, a)
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        if t % 200 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())
            scale = max_c
            beta = max(1e-12, beta_frac * scale * (0.85 ** (t // 200)))

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinusoidal_profile(n, freq, amp):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width, center=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _truncated_gauss_subgradient(a_init, deadline, n_iter=10000):
    """
    Subgradient method using a truncated Gaussian weight around the max.
    More aggressive than pure argmax subgradient but smoother than full softmax.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.003
    momentum = np.zeros(n)
    mom_coeff = 0.95

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Truncated Gaussian weight: only lags within 0.5% of max
        threshold = cm * 0.995
        active = c >= threshold
        if not np.any(active):
            active[int(np.argmax(c))] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / np.sum(active)
        W = float(np.dot(w, c))
        h = _conv_w_with_a(w, a)
        grad = 4.0 * n * a * (h - W)

        # Momentum
        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        if it % 50 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = lr * 0.95

    return best_a, best_score
