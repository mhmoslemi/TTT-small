Mirror descent with log-sum-exp continuation and LP acceleration
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csc_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _fft_cross_corr(w, a):
    """Compute cross-correlation: h[j] = sum_m w[m+j]*a[m] for j=0..n-1"""
    n = len(a)
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation via FFT: corr[j] = sum_m w[m+j]*a[m]
    # = sum_m a[m] * w[m+j]
    # This is the 'valid' part of correlate(w, a)
    # Using: correlate(w, a, 'full')[n-1 : n-1+n]
    # = ifft(fft(w) * conj(fft(a))) with appropriate slicing
    # Actually: np.correlate(w, a, mode='full')[k] = sum_m w[m] * a[m+k-n_a+1] ... 
    # Let me just use the direct FFT formula.

    # h[j] = sum_m a[m] * w[m+j] for m=0..n-1

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s

    a_new = a * update
    s = a_new.sum()
    if s < 1e-300:

    a_new /= s
    return a_new

"""
Mirror descent with log-sum-exp continuation and LP acceleration
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovations over previous approach:
- Efficient multiplicative weight updates (mirror descent on simplex)
- Sparse LP-based direction finding for acceleration
- Adaptive step size with backtracking
- Aggressive continuation schedule
- Multi-scale profile exploration
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _fft_cross_corr(w, a):
    """Compute h[j] = sum_m w[m+j]*a[m] for j=0..len(a)-1 using FFT."""
    n = len(a)
    w_rev = w[::-1]
    corrfull = _fft_convolve(a, w_rev)
    offset = len(w) - 1
    return corrfull[offset:offset + n]

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    zm = np.max(z)
    z = z - zm
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_gaussian_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2.0 * width ** 2))
    return a / a.sum()

def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_u_shape(n, power):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = u ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_sinusoidal_profile(n, freq, phase):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.sin(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _mirror_descent_core(a_init, deadline, beta_schedule, eta_base, max_iter_per_beta=10000):
    """
    Core mirror descent with log-sum-exp continuation.
    Returns (best_a, best_score, iterations_done).
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    total_iters = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline - 3.0:
            break

        level_deadline = min(deadline - 3.0, time.monotonic() + 60.0)
        if len(beta_schedule) > 4:

        it_in_level = 0

            if time.monotonic() > level_deadline:

            c = _fft_convolve(a, a)

            c_shifted = (c - c_max) / beta
            w = np.exp(c_shifted)

            if w_sum < 1e-300:
                w = np.zeros_like(c)

            else:
                w = w / w_sum

            sc_curr = _score_norm(a)

                log_update = -eta * g
                log_update = np.clip(log_update, -30.0, 30.0)

                s = a_trial.sum()
                if s < 1e-300:
                    break
                a_trial = a_trial / s
                a = a_trial

            it_in_level += 1
            total_iters += 1

            # Periodically check true objective
            if it_in_level % 100 == 0:
                sc_check = _score_norm(a)
                if sc_check < best_score:
                    best_score = sc_check
                    best_a = a.copy()

    return best_a, best_score, total_iters

def _lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. H@b <= M, b >= 0
    where H[k,j] = a_{k-j} if 0<=k-j<=n-1
    Returns b (normalized to sum=1) or None if infeasible.
    """
    n = len(a)
    num_constraints = 2 * n - 1

    # Build sparse constraint matrix H
    # H[k, j] = a[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []

    for k in range(num_constraints):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])

    H_sparse = coo_matrix((vals, (rows, cols)), shape=(num_constraints, n)).tocsc()

    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)

    # Inequality: H @ b <= M * ones
    b_vec = np.full(num_constraints, M)

    # Bounds: b >= 0
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=H_sparse, b_ub=b_vec, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            b_sol = np.maximum(result.x, 0)
            s = b_sol.sum()
            if s > 1e-10:
                return b_sol / s
    except Exception:
        pass
    return None

        candidates.append((sc, p.copy()))
        if sc < best_score:
            update_best(p, sc, f"rand{i}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, top candidates: {[f'{c[0]:.6f}' for c in candidates[:5]]}")

    # --- Phase 1: Mirror descent with continuation (700s) ---
    phase1_end = min(deadline, t_start + 730.0)
    n_optimize = 3
    per_candidate_time = (phase1_end - time.monotonic()) / max(1, n_optimize)

    for ci in range(min(n_optimize, len(candidates))):
        if time.monotonic() > phase1_end - 10.0:
            break
        sc_init, p_init = candidates[ci]
        if sc_init > best_score * 1.05:
            break

        c0 = _fft_convolve(p_init, p_init)
        max_c0 = float(c0.max())
        beta_sched = [max_c0 * b for b in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]

        dl = min(phase1_end - 5.0, time.monotonic() + per_candidate_time)
        p_best, sc_best, n_iter = _mirror_descent_core(p_init, dl, beta_sched, eta_base=0.8)
        update_best(p_best, sc_best, f"mirrorsym_{ci}_it{n_iter}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: LP acceleration (400s) ---
    phase2_end = min(deadline, t_start + 1130.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        # Only do LP if n is manageable
        if len(best_seq) <= 4096:
            p_best, sc_best = _lp_accelerated_step(best_seq, phase2_end - 5.0, num_lp_solves=15)
            update_best(p_best, sc_best, "lp_accel")
        else:
            # For larger n, just do more mirror descent
            c0 = _fft_convolve(best_seq, best_seq)
            max_c0 = float(c0.max())
            beta_sched = [max_c0 * b for b in [0.003, 0.001, 0.0005, 0.0002, 0.0001]]
            p_best, sc_best, _ = _mirror_descent_core(best_seq, phase2_end - 5.0, beta_sched, eta_base=0.5)
            update_best(p_best, sc_best, "mirror_large_n")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Fine polishing mirror descent (300s) ---
    phase3_end = min(deadline, t_start + 1430.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        c0 = _fft_convolve(best_seq, best_seq)
        max_c0 = float(c0.max())
        beta_sched = [max_c0 * b for b in [0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]]
        p_best, sc_best, _ = _mirror_descent_core(best_seq, phase3_end - 5.0, beta_sched, eta_base=0.3, max_iter_per_beta=20000)
        update_best(p_best, sc_best, "fine_mirror")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Local search + coordinate descent (200s) ---
    phase4_end = min(deadline, t_start + 1630.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        half_time = (phase4_end - time.monotonic()) / 2.0
        p_best, sc_best = _local_search(best_seq, time.monotonic() + half_time, rng, n_trials=3000)
        update_best(p_best, sc_best, "local_search")

        if time.monotonic() < phase4_end - 5.0:
            p_best2, sc_best2 = _coordinate_descent(best_seq, phase4_end - 5.0, rng, n_rounds=200)
            update_best(p_best2, sc_best2, "coord_descent")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart and different n (160s) ---
    phase5_end = min(deadline, t_start + 1790.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        # Try a few restarts with noise
        n_restarts = 3
        per_restart = (phase5_end - time.monotonic()) / max(1, n_restarts + 1)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10.0:
                break
            noise = 0.02 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            c0 = _fft_convolve(p_start, p_start)
            max_c0 = float(c0.max())
            beta_sched = [max_c0 * b for b in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
            dl = min(phase5_end - 5.0, time.monotonic() + per_restart)
            p_best, sc_best, _ = _mirror_descent_core(p_start, dl, beta_sched, eta_base=0.5)
            update_best(p_best, sc_best, f"restart{i}")

        # Try different n
        if time.monotonic() < phase5_end - 20.0:
            for target_n in [2048, 8192]:
                if target_n == len(best_seq):
                    continue
                p_interp = _interp_profile(best_seq, target_n)
                sc_interp = _score_norm(p_interp)
                update_best(p_interp, sc_interp, f"interp_n{target_n}")

                if sc_interp < best_score * 1.02 and time.monotonic() < phase5_end - 15.0:
                    c0 = _fft_convolve(p_interp, p_interp)
                    max_c0 = float(c0.max())
                    beta_sched = [max_c0 * b for b in [0.01, 0.005, 0.002, 0.001, 0.0005]]
                    dl = min(phase5_end - 10.0, time.monotonic() + 30.0)
                    p_best, sc_best, _ = _mirror_descent_core(p_interp, dl, beta_sched, eta_base=0.5)
                    update_best(p_best, sc_best, f"opt_n{target_n}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_linear_profile(2048, 1.5)
            s = float(final_seq.sum())

        # Return normalized (score is scale-invariant, but let's be safe)
        final_seq = final_seq / s
        sc_final = _score_norm(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Evaluation check failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(4096, 1.5)
    return [float(x) for x in fallback]
