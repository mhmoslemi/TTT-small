    if s < 1e-300 or not np.isfinite(s):

    return e / s


def _score_fn(p):
    n = len(p)
    c = _full_convolve(p, p)
    return float(2.0 * n * c.max())


def _score_raw(a):
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    p = a / s
    return _score_fn(p)


def _convolve_grad(p, k):
    """Compute gradient of (p*p)_k with respect to p. Returns 2*(M_k p)."""
    n = len(p)
    g = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * p[k - hi:k - lo + 1][::-1]
    return g


def _topk_subgradient(p, K, tau):
    """Compute soft subgradient of max(p*p) using top-K lags."""
    n = len(p)
    c = _full_convolve(p, p)
    cm = float(c.max())
    if K >= len(c):
        top_idx = np.arange(len(c))
        top_vals = c
    else:
        top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]
    top_shifted = top_vals - top_vals.max()
    w = np.exp(top_shifted / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(w)
        w[top_shifted.argmax()] = 1.0
    else:
        w = w / w_sum
    g = np.zeros(n)
    for j in range(len(top_idx)):
        if w[j] < 1e-15:
            continue
        k = int(top_idx[j])
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * w[j] * p[k - hi:k - lo + 1][::-1]
    return g, c, cm


def _z_obj_topk(z, beta, n, K):
    """Objective in logit space with top-K logsumexp smoothing."""
    p = _softmax_fn(z)
    c = _full_convolve(p, p)
    if K >= len(c):
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)
    else:
        top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]
        tmax = float(top_vals.max())
        c_norm = (top_vals - tmax) / beta
        log_sum = float(np.log(np.sum(np.exp(c_norm))))
        return 2.0 * n * (tmax + beta * log_sum)


def _z_grad_topk(z, beta, n, K):
    """Gradient in logit space with top-K logsumexp smoothing."""
    p = _softmax_fn(z)
    c = _full_convolve(p, p)
    if K >= len(c):
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        w = e / e.sum()
    else:
        top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]
        tmax = float(top_vals.max())
        c_norm = (top_vals - tmax) / beta
        e = np.exp(c_norm)
        w_full = np.zeros(len(c))
        w_full[top_idx] = e / e.sum()
        w = w_full
    w_conv = _full_convolve(w, p[::-1])
    h = w_conv[n - 1:2 * n - 1]
    grad = 2.0 * n * p * (h - float(np.dot(w, p)))
    if not np.all(np.isfinite(grad)):
        return np.zeros(n)
    return grad


def _lbfgs_continue(z_init, n, deadline, beta_start, beta_end, n_levels=12,
                    top_k=0, max_iter_per_level=200):
    """L-BFGS with beta continuation from beta_start to beta_end."""
    z = z_init.copy()
    p = _softmax_fn(z)
    best_p = p.copy()
    best_score = _score_fn(p)
    c0 = _full_convolve(p, p)
    max_c0 = float(c0.max())
    if max_c0 < 1e-15:
        return best_p, best_score
    beta_start_abs = beta_start * max_c0
    beta_end_abs = beta_end * max_c0
    n_levels = max(2, n_levels)
    ratios = np.linspace(0, 1, n_levels)
    betas = [beta_start_abs * (beta_end_abs / beta_start_abs) ** r for r in ratios]
    per_level_time = max(1.0, (deadline - time.monotonic()) / n_levels)
    for lvl, beta in enumerate(betas):
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level_time)
        try:
            res = minimize(
                _z_obj_topk, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_grad_topk,
                options={'maxiter': max_iter_per_level, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxls': 50, 'maxfun': 2000}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z = res.x
                p_new = _softmax_fn(z)
                sc = _score_fn(p_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_p = p_new.copy()
        except Exception:
            pass
    p_final = _softmax_fn(z)
    sc = _score_fn(p_final)
    if sc < best_score:
        best_score = sc
        best_p = p_final.copy()
    return best_p, best_score


def _nesterov_subgrad(p_init, deadline, lr_init=0.5, K=15, tau_init=0.005,
                      tau_decay=0.998, momentum_init=0.85, momentum_grow=1.0002):
    """Nesterov-accelerated subgradient with top-K soft active set."""
    n = len(p_init)
    z = np.log(np.maximum(p_init, 1e-15))
    z = z - z.mean()
    v = np.zeros(n)
    mu = momentum_init
    tau = tau_init
    best_p = p_init.copy()
    best_score = _score_fn(p_init)
    t = 0.0
    it = 0
    failed = 0
    while time.monotonic() < deadline:
        it += 1
        t += 1.0
        p = _softmax_fn(z)
        g, c, cm = _topk_subgradient(p, K, tau)
        v = mu * v + (1.0 - mu) * g
        step = lr_init / (1.0 + 0.01 * t) ** 0.6
        z = z - step * v
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        tau = max(1e-8, tau * tau_decay)
        mu = min(0.9999, mu * momentum_grow)
        if it % 100 == 0:
            sc = _score_fn(p)
            sc_new = _score_fn(_softmax_fn(z))
            if sc_new < best_score:
                best_score = sc_new
                best_p = _softmax_fn(z).copy()
                failed = 0
            else:
                failed += 1
                if failed > 500:
                    lr_init *= 0.9
                    failed = 250
    p_final = _softmax_fn(z)
    sc = _score_fn(p_final)
    if sc < best_score:
        best_score = sc
        best_p = p_final.copy()
    return best_p, best_score


def _kkt_polish(p_init, deadline, tol=1e-12, max_iter=5000):
    """KKT-based active set polishing. At optimum, weighted sum of active grads is constant."""
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_score = _score_fn(p)
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(p, p)
        cm = float(c.max())
        threshold = cm - max(1e-10, cm * 1e-6)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            break
        if len(active) > 50:
            active = np.argpartition(c, -50)[-50:]
        w = c[active]
        w = w - w.max()
        w = np.exp(w / (0.01 * cm + 1e-15))
        w = w / w.sum()
        h = np.zeros(n)
        for j in range(len(active)):
            k = int(active[j])
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                h[lo:hi + 1] += w[j] * p[k - hi:k - lo + 1][::-1]
        W = float(np.dot(h, p))
        diff = h - W
        diff_norm = float(np.abs(diff).max())
        if diff_norm < tol:
            break
        z = np.log(np.maximum(p, 1e-15))
        z = z - z.mean()
        step_size = 0.05 * diff_norm / (cm + 1e-15)
        step_size = min(step_size, 0.1)
        z = z - step_size * diff / (diff_norm + 1e-15)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        p_new = _softmax_fn(z)
        sc = _score_fn(p_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_new.copy()
            p = p_new
        else:
            z = np.log(np.maximum(p, 1e-15))
            z = z - z.mean()
            z = z - 0.01 * step_size * diff / (diff_norm + 1e-15)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_new = _softmax_fn(z)
            sc = _score_fn(p_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p_new.copy()
                p = p_new
    return best_p, best_score


def _adam_optimize(p_init, deadline, lr=0.03, K=10, beta_init=0.01,
                   beta_decay=0.97, n_max_iter=100000):
    """Adam optimization in logit space with top-K logsumexp smoothing."""
    n = len(p_init)
    z = np.log(np.maximum(p_init, 1e-15))
    z = z - z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    c = _full_convolve(_softmax_fn(z), _softmax_fn(z))
    beta = beta_init * float(c.max())
    best_p = p_init.copy()
    best_score = _score_fn(p_init)
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        grad = _z_grad_topk(z, beta, n, K)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        if t % 50 == 0:
            p_new = _softmax_fn(z)
            sc = _score_fn(p_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p_new.copy()
        if t % 200 == 0:
            beta = max(1e-15, beta * beta_decay)
        if t % 5000 == 0:
            if best_score < 1.51:
                lr = max(0.002, lr * 0.95)
    p_final = _softmax_fn(z)
    sc = _score_fn(p_final)
    if sc < best_score:
        best_score = sc
        best_p = p_final.copy()
    return best_p, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_power_profile(n, p):
    a = (np.arange(n) + 1.0) ** p
    return a / a.sum()


def _make_two_phase(n, ratio, smooth):
    """Two-phase profile: high for first 'ratio' fraction, low for rest, with smooth transition."""
    t = np.linspace(0.0, 1.0, n)
    a = np.where(t < ratio, 1.0, 0.1)
    idx = min(int(ratio * n), n - 1)
    w = max(1, int(smooth * n))
    if idx - w >= 0:
        for j in range(w):
            frac = j / w
            a[idx - w + j] = 0.1 + 0.9 * (1 - frac) ** 2
    if idx + w < n:
        for j in range(w):
            frac = j / w
            a[idx + j] = 1.0 - 0.9 * frac ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _symmetrize_anti(p):
    """Create an anti-symmetric (increasing) version."""
    n = len(p)
    t = np.linspace(0, 1, n)
    a = 1.0 + 1.5 * (2.0 * t - 1.0)
    a = np.maximum(a, 0.05)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses homotopy L-BFGS, Nesterov subgradient, KKT polishing,
    Adam refinement, and multi-restart exploration.
    """
    global height_sequence_1
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            el = time.monotonic() - t_start
            print(f"  [{el:7.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # === Phase 0: Initialization ===
    print("=== Phase 0: Initialization ===")
    n_primary = 4096
    candidates = []
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_fn(parent)
                candidates.append((sc, parent.copy(), "parent"))
                _update(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    if parent is not None and len(parent) != n_primary:
        p_int = _interp_profile(parent, n_primary)
        sc = _score_fn(p_int)
        candidates.append((sc, p_int, "parent_interp"))
        _update(p_int, sc, "parent_interp")

    # Generate diverse profiles
    profiles = []
    for alpha in [0.0005, 0.001, 0.002, 0.004, 0.008, 0.015, 0.03]:
        profiles.append(_make_exp_profile(n_primary, alpha))
    for p in [-0.3, 0.0, 0.3, 0.7, 1.0, 1.5]:
        profiles.append(_make_power_profile(n_primary, p))
    for ratio, smooth in [(0.3, 0.05), (0.4, 0.1), (0.5, 0.15), (0.6, 0.1), (0.7, 0.05)]:
        profiles.append(_make_two_phase(n_primary, ratio, smooth))
    profiles.append(_symmetrize_anti(n_primary))
    # Random smooth profiles
    for _ in range(5):
        x = np.linspace(0, 1, 100)
        coeffs = rng.normal(size=6)
        y = sum(c * np.cos(2 * np.pi * (i + 1) * x) for i, c in enumerate(coeffs))
        y = 1.0 + 0.5 * y / (np.abs(y).max() + 1e-10)
        y = np.maximum(y, 0.01)
        a = np.interp(np.linspace(0, 1, n_primary), x, y)
        profiles.append(a / a.sum())

    phase0_end = min(deadline, t_start + 15.0)
    for idx, pr in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_fn(pr)
        candidates.append((sc, pr, f"prof{idx}"))
        _update(pr, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    if best_seq is None:
        best_seq = _make_exp_profile(n_primary, 0.003)
        best_score = _score_fn(best_seq)

    # === Phase 1: Homotopy L-BFGS on top candidates ===
    print("=== Phase 1: Homotopy L-BFGS ===")
    phase1_end = min(deadline, t_start + 500.0)
    top_candidates = candidates[:3]
    n_tried = 0
    for sc, p, label in top_candidates:
        if time.monotonic() > phase1_end:
            break
        if sc > best_score * 1.05:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        per_time = min(150.0, remaining / max(1, 3 - n_tried))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  L-BFGS homotopy on {label} (n={len(p)}), budget={per_time:.0f}s")
        n_p = len(p)
        z = np.log(np.maximum(p, 1e-15))
        z = z - z.mean()
        # Start with large beta for convergence, decrease
        p_best, sc_best = _lbfgs_continue(
            z, n_p, dl, beta_start=0.5, beta_end=0.001,
            n_levels=10, top_k=0, max_iter_per_level=150
        )
        _update(p_best, sc_best, f"homotopy_{label}")
        # Continue with smaller beta
        if time.monotonic() < phase1_end - 10:
            dl2 = min(phase1_end - 5, time.monotonic() + 60.0)
            z2 = np.log(np.maximum(p_best, 1e-15))
            z2 = z2 - z2.mean()
            p_best2, sc_best2 = _lbfgs_continue(
                z2, n_p, dl2, beta_start=0.001, beta_end=0.0001,
                n_levels=5, top_k=5, max_iter_per_level=300
            )
            _update(p_best2, sc_best2, f"homotopy2_{label}")
        n_tried += 1
    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Nesterov subgradient ===
    print("=== Phase 2: Nesterov subgradient ===")
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + remaining)
        print(f"  Nesterov on best (n={len(best_seq)}), budget={remaining:.0f}s")
        p_best, sc_best = _nesterov_subgrad(
            best_seq, dl, lr_init=0.3, K=10,
            tau_init=0.003, tau_decay=0.999,
            momentum_init=0.8, momentum_grow=1.0003
        )
        _update(p_best, sc_best, "nesterov1")
        if time.monotonic() < phase2_end - 30:
            dl2 = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.7)
            p_best2, sc_best2 = _nesterov_subgrad(
                best_seq, dl2, lr_init=0.1, K=5,
                tau_init=0.001, tau_decay=0.9995,
                momentum_init=0.9, momentum_grow=1.0002
            )
            _update(p_best2, sc_best2, "nesterov2")
    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Adam refinement ===
    print("=== Phase 3: Adam refinement ===")
    phase3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + remaining * 0.6)
        print(f"  Adam on best, budget={remaining*.6:.0f}s")
        p_best, sc_best = _adam_optimize(
            best_seq, dl, lr=0.04, K=8,
            beta_init=0.005, beta_decay=0.96
        )
        _update(p_best, sc_best, "adam1")
        if time.monotonic() < phase3_end - 20:
            dl2 = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) * 0.5)
            p_best2, sc_best2 = _adam_optimize(
                best_seq, dl2, lr=0.015, K=5,
                beta_init=0.001, beta_decay=0.98
            )
            _update(p_best2, sc_best2, "adam2")
    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: KKT polishing ===
    print("=== Phase 4: KKT polishing ===")
    phase4_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        remaining = phase4_end - time.monotonic()
        dl = min(phase4_end, time.monotonic() + remaining)
        print(f"  KKT polish, budget={remaining:.0f}s")
        p_best, sc_best = _kkt_polish(best_seq, dl, tol=1e-13, max_iter=10000)
        _update(p_best, sc_best, "kkt1")
        if time.monotonic() < phase4_end - 10:
            dl2 = min(phase4_end - 5, time.monotonic() + 60.0)
            z3 = np.log(np.maximum(best_seq, 1e-15))
            z3 = z3 - z3.mean()
            p_best2, sc_best2 = _lbfgs_continue(
                z3, len(best_seq), dl2, beta_start=0.0005, beta_end=0.00005,
                n_levels=4, top_k=3, max_iter_per_level=200
            )
            _update(p_best2, sc_best2, "lbfgs_post_kkt")
    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Multi-restart ===
    print("=== Phase 5: Multi-restart ===")
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_restarts = 5
        per_restart = min(40.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        noise_levels = [0.02, 0.05, 0.1, 0.15, 0.25]
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            n_b = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i] * rng.normal(size=n_b)
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            dl = min(phase5_end - 5, time.monotonic() + per_restart * 0.5)
            p_m, sc_m = _lbfgs_continue(
                z, n_b, dl, beta_start=0.01, beta_end=0.001,
                n_levels=5, top_k=5, max_iter_per_level=100
            )
            _update(p_m, sc_m, f"restart{i}_lbfgs")
            if time.monotonic() < phase5_end - 5:
                dl_a = min(phase5_end - 5, time.monotonic() + per_restart * 0.4)
                p_a, sc_a = _adam_optimize(
                    p_m, dl_a, lr=0.02, K=5,
                    beta_init=0.002, beta_decay=0.97
                )
                _update(p_a, sc_a, f"restart{i}_adam")
    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Try different lengths ===
    print("=== Phase 6: Different lengths ===")
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = [n for n in [2048, 8192, 16384] if n != n_curr]
        for tn in target_ns[:2]:
            if time.monotonic() > phase6_end - 30:
                break
            p_int = _interp_profile(best_seq, tn)
            sc_int = _score_fn(p_int)
            _update(p_int, sc_int, f"interp_n{tn}")
            remaining = phase6_end - time.monotonic()
            if remaining < 30:
                break
            dl = min(phase6_end - 10, time.monotonic() + min(80.0, remaining * 0.5))
            z_int = np.log(np.maximum(p_int, 1e-15))
            z_int = z_int - z_int.mean()
            p_best, sc_best = _lbfgs_continue(
                z_int, tn, dl, beta_start=0.01, beta_end=0.0005,
                n_levels=6, top_k=5, max_iter_per_level=100
            )
            _update(p_best, sc_best, f"lbfgs_n{tn}")
            if time.monotonic() < phase6_end - 20:
                dl2 = min(phase6_end - 10, time.monotonic() + 40.0)
                p_a, sc_a = _adam_optimize(
                    p_best, dl2, lr=0.02, K=5,
                    beta_init=0.001, beta_decay=0.97
                )
                _update(p_a, sc_a, f"adam_n{tn}")
    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final aggressive polish ===
    print("=== Phase 7: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            dl = min(deadline - 10, time.monotonic() + min(50.0, remaining * 0.5))
            p_best, sc_best = _kkt_polish(best_seq, dl, tol=1e-14, max_iter=5000)
            _update(p_best, sc_best, "final_kkt")
        if time.monotonic() < deadline - 10:
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                dl2 = min(deadline - 5, time.monotonic() + min(30.0, remaining2 * 0.6))
                z_f = np.log(np.maximum(best_seq, 1e-15))
                z_f = z_f - z_f.mean()
                p_best2, sc_best2 = _lbfgs_continue(
                    z_f, len(best_seq), dl2, beta_start=0.0003, beta_end=0.00003,
                    n_levels=3, top_k=3, max_iter_per_level=300
                )
                _update(p_best2, sc_best2, "final_lbfgs")
    print(f"  Phase 7 done: best={best_score:.10f}")

    # === Final validation ===
    print("=== Final ===")
    if best_seq is not None:
        final_arr = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_arr.sum())
        if s > 0.01:
            final_score = _score_raw(final_arr)
        else:
            final_arr = _make_exp_profile(4096, 0.003)
            final_score = _score_raw(final_arr)
        print(f"  FINAL: score={final_score:.10f}, n={len(final_arr)}")
        try:
            seq_list = [float(x) for x in final_arr]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  Eval check: {e}")
        return [float(x) for x in final_arr]
    else:
        fallback = _make_exp_profile(4096, 0.003)
        print(f"  FALLBACK: n=4096, score={_score_raw(fallback):.10f}")
        return [float(x) for x in fallback]
