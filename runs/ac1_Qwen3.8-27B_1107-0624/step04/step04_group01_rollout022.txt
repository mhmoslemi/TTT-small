```python
"""
LP-based best response + Adam gradient polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses scipy HiGHS solver for the LP subproblems.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _flat_target(n):
    """First n coefficients of sqrt(1 + x + ... + x^(2n-2)), normalized."""
    m = 2 * n - 2
    q = np.zeros(m + 1)
    q[0] = 1.0
    for k in range(1, m + 1):
        s = np.dot(q[1:k+1], q[k-1::-1]) if k <= m else np.dot(q[1:m+1], q[k-1:k-m-1:-1])
        # More carefully:
        # s = sum_{i=1}^{min(k,m)} q[i] * q[k-i]
        lo = 1
        hi = min(k, m)
        s = 0.0
        for i in range(lo, hi + 1):
            s += q[i] * q[k - i]
        q[k] = (1.0 - s) / (2.0 * q[0])
    result = q[:n].copy()
    result = np.maximum(result, 1e-15)
    result = result / result.sum()
    return result


def _lp_best_response(a, M):
    """Solve max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix A: A[k,j] = a[k-j] if 0 <= k-j < n else 0
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return a.copy()


def _lp_iterations(a, max_iters, deadline, t_start=0.5, t_end=0.1):
    """Run LP-based best response iterations."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()  # normalize
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        
        g = _lp_best_response(a, M)
        s_g = float(g.sum())
        
        if s_g <= 1.0 + 1e-8:
            # No improvement possible
            break
        
        g_norm = g / s_g
        
        # Adaptive step size
        t = t_start if it < 10 else max(t_end, t_start * (t_end / t_start) ** (it / 50))
        
        a_new = (1 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        elif sc < _score_norm(a) + 1e-6:
            # Accept even if not better than best (to keep moving)
            pass
        else:
            # Try smaller step
            t2 = t * 0.5
            a_new2 = (1 - t2) * a + t2 * g_norm
            a_new2 = np.maximum(a_new2, 0)
            a_new2 = a_new2 / a_new2.sum()
            sc2 = _score_norm(a_new2)
            if sc2 < _score_norm(a):
                a = a_new2
            else:
                # Reduce step further
                t = t2
                a = a  # don't update
                continue
        
        a = a_new
        a = a / a.sum()
        
        if it % 50 == 0:
            print(f"    LP iter {it}: score={_score_norm(a):.10f}, best={best_score:.10f}, sum_g={s_g:.4f}")
    
    return best_a, best_score
```
