def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)

        return np.inf
    c = _full_convolve(a, a)

LP-guided optimization with gradient polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions,

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c))

def _score_full(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Objective: minimize -sum(g) (to maximize sum(g))
    c_obj = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A.toarray(), b_ub=np.full(2 * n - 1, M),
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        return result.x
    return None

def _lp_iteration(f, t=0.1, max_lp_time=10.0):
    """One LP-guided iteration."""
    n = len(f)
    c = _full_convolve(f, f)
    M = float(np.max(c))
    
    # Solve LP
    start = time.monotonic()
    g = _solve_lp(f, M)
    elapsed = time.monotonic() - start
    
    if g is None or elapsed > max_lp_time:
        return f, M
    
    # Normalize g to have same sum as f
    s_g = float(np.sum(g))
    if s_g < 1e-15:
        return f, M
    g = g / s_g * float(np.sum(f))
    
    # Update
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    f_new = f_new / np.sum(f_new)
    
    return f_new, M

def _adam_optimize(a_init, deadline, lr=0.01, k_top=5, total_iters=50000):
    """Adam with top-K smoothed gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    ir = 0
    while ir < total_iters:
        if time.monotonic() > deadline - 1:
            break
        ir += 1
        t += 1
        
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        
        # Top-K
        k_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_actual:]
        
        # Gradient: (4*n/K) * sum_{k in top} a_curr[k-j] for valid j
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            j_min = max(0, k_val - (n - 1))
            j_max = min(n - 1, k_val)
            if j_min <= j_max:
                grad[j_min:j_max + 1] += a_curr[k_val - j_max:k_val - j_min + 1][::-1]
        grad *= 4.0 * n / k_actual
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

                best_a = a_new.copy()

def _make_profiles(n):
    t = np.linspace(0.0, 1.0, n)
    profiles = []
    # Linear
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Raised cosine
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Power
    for p in [1.0, 2.0, 3.0, 5.0]:
        u = np.abs(2.0 * t - 1.0)
        a = (1.0 - u) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Gaussian
    for w in [0.2, 0.3, 0.4]:
        u = t - 0.5
        a = np.exp(-u**2 / (2.0 * w**2))
        profiles.append(a / a.sum())
    return profiles

"""
LP-guided optimization with multi-scale gradient polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Combines linear programming for direction finding, L-BFGS with logsumexp
continuation, adaptive top-K Adam, and intensive local search.
Explores multiple sequence lengths (2048-8192) for optimal attack surface.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c))


def _score_full(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _convolve_topk_gradient(a, top_indices, n):
    grad = np.zeros(n)
    for k_val in top_indices:
        k_val = int(k_val)
        j_min = max(0, k_val - (n - 1))
        j_max = min(n - 1, k_val)
        if j_min <= j_max:
            indices = np.arange(k_val - j_max, k_val - j_min + 1)
            grad[j_min:j_max + 1] += a[indices][::-1]
    return grad


def _solve_lp_for_direction(f, M, time_limit=10.0):
    n = len(f)
    if n > 4096:
        return None
    t0 = time.monotonic()
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                         bounds=bounds, method='highs',
                         options={'max_time': time_limit, 'presolve': True})
        if time.monotonic() - t0 > time_limit * 1.5:
            return None
        if result.status == 0 and result.x is not None:
            return result.x
    except Exception:
        pass
    return None


def _lbfgs_optimize(a_init, deadline, beta_multipliers=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if beta_multipliers is None:
        beta_multipliers = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    betas = [max_c0 * b for b in beta_multipliers]
    remaining = max(10.0, deadline - time.monotonic())
    per_level = max(5.0, remaining / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 2:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj_fn(z_):

            cm_ = float(np.max(c_))
            shifted = (c_ - cm_) / beta
            s_ = np.sum(np.exp(shifted))
            if s_ < 1e-300 or not np.isfinite(s_):

            return 2.0 * n * (cm_ + beta * np.log(s_))

        def grad_fn(z_):

            c_ = _full_convolve(a_, a_)
            cm_ = float(np.max(c_))
            shifted = (c_ - cm_) / beta
            e_ = np.exp(shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(obj_fn, z, jac=grad_fn, method='L-BFGS-B',

                                       'gtol': 1e-9, 'maxls': 40, 'maxfun': 5000})
            if result.x is not None and np.all(np.isfinite(result.x)):

                a_new = _softmax(z)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
        except Exception:
            pass
    return best_a, best_score


def _adam_topk(a_init, deadline, lr=0.01, k_top=5, max_iters=100000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr_schedule = lr
    ir = 0
    while ir < max_iters:
        if time.monotonic() > deadline - 1:
            break
        ir += 1
        t_count += 1
        a_curr = _softmax(z)
        c = _full_convolve(a_curr, a_curr)
        k_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_actual:]
        top_idx = top_idx[top_idx < len(c)]
        grad = _convolve_topk_gradient(a_curr, top_idx, n)
        grad *= 4.0 * n / k_actual
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t_count)
        vh = v / (1 - b2 ** t_count)
        z = z - lr_schedule * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()
        if ir % 15 == 0:
            a_new = _softmax(z)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        if ir % 3000 == 0:
            lr_schedule *= 0.85
            if lr_schedule < 0.001:
                lr_schedule = 0.001
    return best_a, best_score


def _local_search(a_init, deadline, rng, n_trials=5000, max_block=12):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    for it in range(n_trials):
        if time.monotonic() > deadline - 1:
            break
        block_size = int(rng.integers(2, min(max_block, n // 4)))
        start = int(rng.integers(0, n - block_size))
        a_trial = a.copy()
        a_trial[start:start + block_size] *= rng.uniform(0.8, 1.2, size=block_size)
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()
        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    return best_a, best_score


def _make_profiles(n):

    profiles = []

        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)

        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for p in [1.0, 1.5, 2.0, 3.0, 5.0]:
        u = np.abs(2.0 * t - 1.0)
        a = (1.0 - u) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for w in [0.15, 0.25, 0.35, 0.45]:
        u = t - 0.5
        a = np.exp(-u ** 2 / (2.0 * w ** 2))
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    while time.monotonic() < deadline - 5:

        c = _full_convolve(a, a)
        M = float(np.max(c))

        if g is None:

        s_g = float(np.sum(g))
        if s_g < 1e-15:
            break
        g = g / s_g * float(np.sum(a))
        improved = False
        for t_frac in [0.1, 0.05, 0.02, 0.01]:
            a_new = (1.0 - t_frac) * a + t_frac * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break
        if not improved:
            break
    return best_a, best_score, n_iters

        parent = _make_profiles(n_parent)[12]
        update_best(parent, _score_norm(parent), "default")

    # Phase 0: Diverse initialization (25s)
    phase0_end = min(deadline, t_start + 25.0)
    best_n = len(best_seq)
    for n_try in [best_n, max(1024, best_n // 2), min(8192, best_n * 2)]:
        if time.monotonic() > phase0_end:
            break
        profiles = _make_profiles(n_try)
        for idx, p in enumerate(profiles):
            if time.monotonic() > phase0_end:
                break
            sc = _score_norm(p)
            update_best(p, sc, f"p{n_try}_{idx}")
        if parent is not None and n_try != len(parent):
            p = _interp_profile(parent, n_try)
            sc = _score_norm(p)
            update_best(p, sc, f"interp{n_try}")
    print(f"Phase 0: best={best_score:.10f} n={len(best_seq)}")

    # Phase 1: LP-guided optimization (250s)
    phase1_end = min(deadline, t_start + 275.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        lp_dl = min(phase1_end - 5, time.monotonic() + 240.0)
        p_lp, sc_lp, n_iters = _lp_phase(best_seq, lp_dl, max_lp_time=10.0)
        update_best(p_lp, sc_lp, f"lp({n_iters})")
        if time.monotonic() < phase1_end - 10:
            lp_dl2 = min(phase1_end - 5, time.monotonic() + 100.0)
            p_lp2, sc_lp2, n_iters2 = _lp_phase(best_seq, lp_dl2, max_lp_time=8.0)
            update_best(p_lp2, sc_lp2, f"lp2({n_iters2})")
    print(f"Phase 1: best={best_score:.10f}")

    # Phase 2: L-BFGS continuation (300s)
    phase2_end = min(deadline, t_start + 575.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        dl = min(phase2_end - 3, time.monotonic() + 290.0)
        p_best, sc_best = _lbfgs_optimize(best_seq, dl,
                                          beta_multipliers=[0.03, 0.015, 0.008,
                                                            0.004, 0.002, 0.001,
                                                            0.0005, 0.0002])
        update_best(p_best, sc_best, "lbfgs_main")
    print(f"Phase 2: best={best_score:.10f}")

    # Phase 3: Adam top-K (250s)
    phase3_end = min(deadline, t_start + 825.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        dl1 = min(phase3_end - 30, time.monotonic() + 120.0)
        p_a1, sc_a1 = _adam_topk(best_seq, dl1, lr=0.012, k_top=5)
        update_best(p_a1, sc_a1, "adam_k5")
        if time.monotonic() < phase3_end - 20:
            dl2 = min(phase3_end - 10, time.monotonic() + 100.0)
            p_a2, sc_a2 = _adam_topk(best_seq, dl2, lr=0.006, k_top=3)
            update_best(p_a2, sc_a2, "adam_k3")
        if time.monotonic() < phase3_end - 10:
            dl3 = min(phase3_end - 5, time.monotonic() + 60.0)
            p_a3, sc_a3 = _adam_topk(best_seq, dl3, lr=0.004, k_top=8)
            update_best(p_a3, sc_a3, "adam_k8")
    print(f"Phase 3: best={best_score:.10f}")

    # Phase 4: Local search (150s)
    phase4_end = min(deadline, t_start + 975.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        dl = min(phase4_end - 3, time.monotonic() + 140.0)
        p_ls, sc_ls = _local_search(best_seq, dl, rng, n_trials=4000, max_block=10)
        update_best(p_ls, sc_ls, "local4")
    print(f"Phase 4: best={best_score:.10f}")

    # Phase 5: Multi-restart L-BFGS (250s)
    phase5_end = min(deadline, t_start + 1225.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 5
        per_restart = max(25.0, (phase5_end - time.monotonic()) / n_restarts)
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 5:
                break
            noise = 0.015 + 0.015 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax(z)
            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            p_sr, sc_sr = _lbfgs_optimize(p_start, dl,
                                          beta_multipliers=[0.02, 0.01, 0.005,
                                                            0.002, 0.001])
            update_best(p_sr, sc_sr, f"restart{i}")
    print(f"Phase 5: best={best_score:.10f}")

    # Phase 6: Try different n (250s)
    phase6_end = min(deadline, t_start + 1475.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        target_ns = sorted(set([min(8192, n_curr * 2), max(2048, n_curr // 2), n_curr + 1024]))
        for target_n in target_ns:
            if time.monotonic() > phase6_end - 30:
                break
            if target_n == n_curr or target_n < 512 or target_n > 16384:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")
            if time.monotonic() < phase6_end - 30:
                dl = min(phase6_end - 20, time.monotonic() + 80.0)
                p_nb, sc_nb = _lbfgs_optimize(p_interp, dl,
                                              beta_multipliers=[0.02, 0.01,
                                                                0.005, 0.002])
                update_best(p_nb, sc_nb, f"lbfgs_n{target_n}")
    print(f"Phase 6: best={best_score:.10f}")

    # Phase 7: Final intensive polishing (remaining ~300s)
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        if remaining > 80:
            dl = min(deadline - 5, time.monotonic() + remaining * 0.35)
            p_f1, sc_f1 = _lbfgs_optimize(best_seq, dl,
                                          beta_multipliers=[0.005, 0.002, 0.001,
                                                            0.0005, 0.0002, 0.0001])
            update_best(p_f1, sc_f1, "final_lbfgs")
        if time.monotonic() < deadline - 50:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.25)
            p_f2, sc_f2 = _adam_topk(best_seq, dl, lr=0.003, k_top=2, max_iters=50000)
            update_best(p_f2, sc_f2, "final_adam")
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 5, time.monotonic() + remaining * 0.2)
            p_f3, sc_f3 = _local_search(best_seq, dl, rng, n_trials=4000, max_block=8)
            update_best(p_f3, sc_f3, "final_local")
        # Final LP pass if n is manageable
        if time.monotonic() < deadline - 25 and len(best_seq) <= 4096:
            a = best_seq.copy()
            for _ in range(5):
                if time.monotonic() > deadline - 10:
                    break
                c = _full_convolve(a, a)
                M = float(np.max(c))
                g = _solve_lp_for_direction(a, M, time_limit=8.0)
                if g is None:
                    break
                s_g = float(np.sum(g))
                if s_g < 1e-15:
                    break
                g = g / s_g * float(np.sum(a))
                improved = False
                for t_frac in [0.05, 0.02, 0.01]:
                    a_new = (1.0 - t_frac) * a + t_frac * g
                    a_new = np.maximum(a_new, 0)
                    s_new = float(np.sum(a_new))
                    if s_new < 1e-15:
                        continue
                    a_new = a_new / s_new
                    sc = _score_norm(a_new)
                    if sc < best_score:
                        a = a_new
                        update_best(a, sc, "final_lp")
                        improved = True
                        break
                if not improved:
                    break

    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profiles(4096)[12]
            s = float(final_seq.sum())
        final_seq = final_seq / s
        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, elapsed={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final_seq]

    fallback = _make_profiles(4096)[12]
    sc_fb = _score_full(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
