import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_iterate(f, M, max_time=30):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    n = len(f)
    g = cp.Variable(n)
    
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[j] if 0 <= k-j < n else 0
    # This is a "convolution-like" matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g_j contributes to (f*g)_k for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j]
        k_start = j
        k_end = min(j + n, 2*n-1)
        for k in range(k_start, k_end):
            A[k, j] = f[k - j]
    
    constraints = [g >= 0, A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except Exception as e:
        print(f"LP solve failed: {e}")
    return None

"""
Hybrid LP + smooth optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact LP-based iterated minimax, L-BFGS with fine beta continuation,
and coordinate descent polish. Multi-restart with diverse initializations.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f, max_time=60):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            g_val = g_val / g_val.sum()
            return g_val, float(t.value)
    except Exception as e:
        print(f"  LP failed: {e}")
    return None, None


def _lp_iterate(f, n_iters=100, deadline=None, step_size=0.3):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(f, f)
        M = float(c.max())
        g, t_val = _lp_minimax(f)
        if g is None:
            break
        f_new = (1 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
        if it % 20 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_continuation(a_init, deadline, beta_list=None):

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005, 0.00003, 0.00002]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:

        remaining = n_betas - idx

        maxiter = max(200, int(time_per * 30))

        try:

            pass

    return best_a, best_score


def _single_beta_lbfgs(a_init, beta, deadline, maxiter=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    try:
        result = minimize(
            _logsumexp_obj, z, args=(beta, n),
            method='L-BFGS-B', jac=_logsumexp_grad,
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                     'maxls': 50, 'maxfun': maxiter * 5}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            s_new = _score_norm_fn(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score


def _adaptive_gd(a_init, deadline, rng, lr_init=0.2, max_iter=30000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    c = _full_convolve(a_init, a_init)
    beta = 0.05 * float(c.max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-8, 0.03 * cm)
        vals = np.clip((c - cm) / beta, -50, 50)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 2000 == 0:
            lr = max(0.01, lr * 0.9)
        if t % 10000 == 0:
            z = z + 0.03 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=50):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        improved = False
        order = np.arange(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local_score = _score_norm_fn(a)
            for delta in [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u**2)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 2048
        parent = _make_profiles(n_init)[3]
        update_best(parent, _score_norm_fn(parent), "default")

    n_parent = len(parent)

    # Phase 1: LP iterations at moderate n (600s)
    phase1_end = min(deadline, t_start + 550.0)
    print(f"\nPhase 1: LP iterations (target: n={n_parent})")

    n_lp = min(n_parent, 2048)
    if n_parent > n_lp:
        a_lp = _interp_profile(parent, n_lp)
    else:
        a_lp = parent.copy()
    sc_lp_init = _score_norm_fn(a_lp)
    update_best(a_lp, sc_lp_init, f"lp_init_n{n_lp}")

    # Run LP with adaptive step size
    a_cur = a_lp.copy()
    for lp_round in range(5):
        if time.monotonic() > phase1_end:
            break
        step = 0.5 / (lp_round + 1)
        a_cur, sc_cur = _lp_iterate(a_cur, n_iters=30, deadline=phase1_end, step_size=step)
        update_best(a_cur, sc_cur, f"lp_round{lp_round}_n{n_lp}")

    # Interpolate LP result back to parent size if needed
    if n_parent > n_lp:
        a_rec = _interp_profile(a_cur, n_parent)
        sc_rec = _score_norm_fn(a_rec)
        update_best(a_rec, sc_rec, f"lp_interp_n{n_parent}")
    else:
        a_rec = a_cur.copy()

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: L-BFGS fine continuation at full n (500s)
    phase2_end = min(deadline, t_start + 1050.0)
    print(f"\nPhase 2: L-BFGS continuation (n={len(best_seq)})")

    n_curr = len(best_seq)
    a_start = best_seq.copy()

    # Try multiple n values for L-BFGS
    n_candidates = sorted(set([n_curr, 2048, 4096]))
    n_candidates = [x for x in n_candidates if x <= 4096]

    n_restarts = 3
    per_restart_time = (phase2_end - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase2_end - 30:
            break

        if r == 0:
            a_cur = best_seq.copy()
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.05 * (r + 1)
            z = z + noise * rng.normal(size=len(z))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)

        per_dl = min(phase2_end - 10, time.monotonic() + per_restart_time)
        p_best, sc_best = _run_lbfgs_continuation(a_cur, per_dl)
        update_best(p_best, sc_best, f"lbfgs_r{r}_n{len(a_cur)}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Try larger n with interpolation + optimization (250s)
    phase3_end = min(deadline, t_start + 1300.0)
    print(f"\nPhase 3: Larger n exploration")

    n_for_explore = 4096
    if n_for_explore > len(best_seq):
        p_interp = _interp_profile(best_seq, n_for_explore)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_for_explore}")

        if time.monotonic() < phase3_end - 60:
            per_dl = min(phase3_end - 20, time.monotonic() + 200.0)
            p_best, sc_best = _run_lbfgs_continuation(p_interp, per_dl)
            update_best(p_best, sc_best, f"lbfgs_n{n_for_explore}")

    # Also try n=2048 if best is already at 4096
    if len(best_seq) == 4096 and time.monotonic() < phase3_end - 30:
        p_2048 = _interp_profile(best_seq, 2048)
        sc_2048 = _score_norm_fn(p_2048)
        update_best(p_2048, sc_2048, "downsample_n2048")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Adaptive GD + very fine L-BFGS (200s)
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 4: Fine optimization")

    if time.monotonic() < phase4_end and best_seq is not None:
        # Adaptive GD
        per_dl = min(phase4_end - 60, time.monotonic() + 120.0)
        p_best, sc_best = _adaptive_gd(best_seq, per_dl, rng, lr_init=0.1)
        update_best(p_best, sc_best, "adaptive_gd")

        # Very fine L-BFGS with tiny beta
        if time.monotonic() < phase4_end - 20:
            n_final = len(best_seq)
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            beta_tiny = 1e-5 * max_c_final
            per_dl2 = min(phase4_end - 10, time.monotonic() + 70.0)
            p_best2, sc_best2 = _single_beta_lbfgs(best_seq, beta_tiny, per_dl2, maxiter=10000)
            update_best(p_best2, sc_best2, "tiny_beta_lbfgs")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish (remaining time)
    phase5_end = deadline
    print(f"\nPhase 5: Coordinate descent polish")

    if time.monotonic() < phase5_end and best_seq is not None:
        per_dl = min(phase5_end - 15, time.monotonic() + (phase5_end - time.monotonic()) * 0.7)
        p_best, sc_best = _coord_descent(best_seq, per_dl, max_rounds=30)
        update_best(p_best, sc_best, "coord_descent")

        # One more tiny-beta L-BFGS
        if time.monotonic() < phase5_end - 10:
            n_final = len(best_seq)
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            beta_ultra = 5e-6 * max_c_final
            per_dl2 = min(phase5_end - 5, time.monotonic() + 30.0)
            p_best2, sc_best2 = _single_beta_lbfgs(best_seq, beta_ultra, per_dl2, maxiter=5000)
            update_best(p_best2, sc_best2, "ultra_tiny_beta")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
