def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(p, beta):
    c = _full_convolve(p)
    w = _smooth_max_weights(c, beta)
    n = len(p)

    fw = np.fft.rfft(w, N)

    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _build_conv_matrix_sparse(a):
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows_list.append(np.full(len(idx), k, dtype=np.int32))
            cols_list.append(idx.astype(np.int32))
            vals_list.append(a[k - idx].astype(np.float64))
    if not rows_list:
        return csr_matrix((2 * n - 1, n))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

            noise_scale = 0.02 + 0.025 * i
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            per_a = min(25.0, p4_end - time.monotonic() - 35)
            if per_a < 10:
                break
            a_r, s_r = _adam_logit(p_s, time.monotonic() + per_a, rng, lr=0.04)
            record(a_r, s_r, f"R{i}")

            if time.monotonic() < p4_end - 25 and best_seq is not None:
                a_l, s_l = _lp_polish(best_seq, min(p4_end - 15, time.monotonic() + 25.0),
                                       max_iters=30)
                record(a_l, s_l, f"RL{i}")

    # Phase 5: Final intensive polish (1450s - deadline)
    print("Phase 5: Final polish")
    if best_seq is not None:
        # LP with very tight M to force convergence
        p5_end = deadline - 15.0
        a5, s5 = _lp_polish(best_seq, p5_end, max_iters=500, m_shrink=0.997, m_min=0.90)
        record(a5, s5, "LP5")

        if time.monotonic() < p5_end - 10:
            a6, s6 = _adam_logit(best_seq, p5_end - 5, rng, lr=0.01,
                                 beta_frac=0.03, beta_decay=0.97, no_imp_reset=200)
            record(a6, s6, "Adam5")

    # Final validation and output
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            t_f = np.linspace(0, 1, 1024)
            final = 1.0 + 1.5 * (t_f - 0.5)
            final = final / final.sum()

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        result = [float(x) for x in final]
    else:
        t_f = np.linspace(0, 1, 1024)
        final = 1.0 + 1.5 * (t_f - 0.5)
        final = final / final.sum()
        print(f"FALLBACK n=1024")
        result = [float(x) for x in final]

    return result
