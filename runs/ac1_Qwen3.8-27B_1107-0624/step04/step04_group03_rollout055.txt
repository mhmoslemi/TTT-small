"""
LP-guided optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Key innovation: Uses linear programming to find descent directions (following
state-of-the-art LP approach of max sum(g) s.t. g>=0, ||f*g||_inf <= M),
combined with adaptive mixing, gradient-based polishing (Nesterov, Adam),
and multiple restarts for robustness.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _make_profile(n, ptype='sine_sq', **kw):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if ptype == 'linear':
        a = 1.0 + kw.get('slope', 1.5) * (t - 0.5)
    elif ptype == 'quadratic':
        a = 1.0 + kw.get('a1', 1.0) * u + kw.get('a2', 0.2) * u ** 2
    elif ptype == 'cubic':
        a = 1.0 + kw.get('a1', 0.8) * u + kw.get('a3', -0.4) * u ** 3
    elif ptype == 'sine_sq':
        a = np.sin(np.pi * t) ** 2
    elif ptype == 'cos_sq':
        a = np.cos(np.pi * u / 2.0) ** 2
    elif ptype == 'uniform':
        a = np.ones(n)
    elif ptype == 'exponential':
        a = np.exp(-kw.get('exp', 2.0) * t)
    elif ptype == 'bell':
        a = np.exp(-kw.get('width', 2.0) * (t - 0.5) ** 2)
    else:
        a = 1.0 + 0.5 * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _build_conv_matrix(f):
    """Build sparse convolution matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    n_cons = 2 * n - 1
    total_nnz = n * (n + 1)
    row_idx = np.empty(total_nnz, dtype=np.int32)
    col_idx = np.empty(total_nnz, dtype=np.int32)

        col_idx[offset:offset + L] = j
        data[offset:offset + L] = f[:L]
        offset += L
    A = csr_matrix((data, (row_idx, col_idx)), shape=(n_cons, n))

        g = result.x
        s = g.sum()
        if s > 1e-15:
            g = g / s
            # Check if g is too close to f (no progress)
            corr = float(np.dot(g, f) / (np.linalg.norm(g) * np.linalg.norm(f) + 1e-30))
            if corr > 0.9999:
                return None
            return g
    return None


def _lp_direction_reduced(f, M, n_active=200, timeout=30):
    """Solve LP with only active constraints (near-max lags)."""
    n = len(f)
    conv = _full_convolve(f, f)
    cm = float(conv.max())
    threshold = cm * 0.95
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    active = active[:n_active]

    n_cons = len(active)
    total_nnz = sum(min(n, 2 * n - 1 - k) for k in active)
    row_idx = np.empty(total_nnz, dtype=np.int32)
    col_idx = np.empty(total_nnz, dtype=np.int32)
    data = np.empty(total_nnz, dtype=np.float64)
    offset = 0
    for i, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        L = j_end - j_start
        if L > 0:
            row_idx[offset:offset + L] = i
            col_idx[offset:offset + L] = np.arange(j_start, j_end, dtype=np.int32)
            data[offset:offset + L] = f[k - np.arange(j_start, j_end)]
            offset += L

    A = csr_matrix((data[:offset], (row_idx[:offset], col_idx[:offset])),
                   shape=(n_cons, n))
    c_obj = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': timeout,
                                                   'presolve': True})
    except Exception:
        return None
    if result.success and result.x is not None:
        g = result.x
        s = g.sum()
        if s > 1e-15:
            g = g / s
            corr = float(np.dot(g, f) / (np.linalg.norm(g) * np.linalg.norm(f) + 1e-30))
            if corr > 0.9999:
                return None
            return g
    return None


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.015, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)
        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)
            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)
            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()
            z = z_new
            y = y_new
            if t % 50 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
            if t % 100 == 0:
                mom = min(0.999, mom + 0.0005)
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.002, lr * 0.7)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.03, beta_frac=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            continue
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        if t % 200 == 0:
            a = _softmax_fn(z)
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _active_set_polish(a_init, deadline):
    """Subgradient method focusing on active (near-max) lags with adaptive step."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    lr = 0.003
    momentum = np.zeros(n)
    mom_coeff = 0.9
    no_improve = 0

    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        active = c >= cm * 0.998
        if not np.any(active):
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / np.sum(active)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -50, 50)

        a_curr = _softmax_fn(z)
        sc = _score_norm_fn(a_curr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                momentum = np.zeros(n)
                lr *= 0.7
                no_improve = 0
                if lr < 0.0001:
                    break
    return best_a, best_score


def _lbfgs_refine(a_init, deadline, max_wall_time=60):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    betas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in betas]
    per_level = max(1.0, max_wall_time / len(betas))

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _objective_smooth, z, args=(beta,),
                method='L-BFGS-B', jac=_gradient_smooth,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-9,
                         'maxls': 20, 'maxfun': 800}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _try_mix(f, g, current_score):
    """Try mixing f toward g with decreasing step sizes. Return best result."""
    best_f = f.copy()
    best_sc = current_score
    for t in [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.03]:
        f_try = (1.0 - t) * f + t * g
        f_try = np.maximum(f_try, 0.0)
        s = f_try.sum()
        if s < 1e-15:
            continue
        f_try = f_try / s
        sc = _score_norm_fn(f_try)
        if sc < best_sc:
            best_sc = sc
            best_f = f_try.copy()
    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    LP-guided optimizer: uses linear programming to find descent directions,
    combined with adaptive mixing, gradient-based polishing, and restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 4096
        parent = _make_profile(n, 'sine_sq')
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using sine_sq profile n={n}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    # --- Phase 0: Quick exploration of different profiles (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    profiles_to_try = [
        ('sine_sq', {}),
        ('cos_sq', {}),
        ('linear', {'slope': 2.0}),
        ('quadratic', {'a1': 1.0, 'a2': 0.3}),
        ('bell', {'width': 1.5}),
        ('exponential', {'exp': 1.5}),
    ]
    for ptype, kw in profiles_to_try:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(len(best_seq), ptype, **kw)
        # Interpolate to match length if needed
        if len(p) != len(best_seq):
            x0 = np.linspace(0, 1, len(p))
            x1 = np.linspace(0, 1, len(best_seq))
            p = np.interp(x1, x0, p)
            p = np.maximum(p, 1e-12) / p.sum()
        sc = _score_norm_fn(p)
        if sc < best_score:
            update_best(p, sc, f"profile_{ptype}")

    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: LP-guided iterations (main phase, ~1000s) ---
    phase1_end = min(deadline, t_start + 1030.0)
    f = best_seq.copy()
    f = f / f.sum()
    current_score = _score_norm_fn(f)
    lp_fail_count = 0
    it_count = 0

    while time.monotonic() < phase1_end:
        it_count += 1
        conv = _full_convolve(f, f)
        M = float(conv.max())

        # Try full LP first, fall back to reduced LP
        g = _lp_direction(f, M, timeout=40)
        if g is None and it_count > 3:
            # Try reduced LP (active constraints only)
            g = _lp_direction_reduced(f, M, n_active=300, timeout=30)

        if g is None:
            lp_fail_count += 1
            if lp_fail_count > 5:
                # Add small noise to escape
                z = np.log(np.maximum(f, 1e-15))
                z = z + 0.02 * rng.normal(size=len(f))
                z = np.clip(z, -50, 50)
                f = _softmax_fn(z)
                lp_fail_count = 0
            continue
        else:
            lp_fail_count = 0

        # Try mixing
        f_new, new_score = _try_mix(f, g, current_score)
        if new_score < current_score:
            f = f_new
            current_score = new_score
            update_best(f, current_score, f"lp_it{it_count}")
        else:
            # No improvement from mixing; reduce step or perturb
            lp_fail_count += 1
            if lp_fail_count > 8:
                # Perturb and try again
                z = np.log(np.maximum(f, 1e-15))
                z = z + 0.01 * rng.normal(size=len(f))
                z = np.clip(z, -50, 50)
                f = _softmax_fn(z)
                current_score = _score_norm_fn(f)
                lp_fail_count = 0
                if current_score > best_score:
                    f = best_seq.copy()
                    f = f / f.sum()
                    current_score = best_score

    print(f"Phase 1 done: best={best_score:.10f}, lp_iters={it_count}")

    # --- Phase 2: Nesterov gradient polishing (300s) ---
    phase2_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
        per_time = min(180.0, phase2_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _nesterov_optimize(best_seq, min(phase2_end - 5, time.monotonic() + per_time),
                                                  beta_fracs, lr_init=0.012, mom_init=0.9)
            update_best(p_best, sc_best, "nesterov_p2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam polishing (200s) ---
    phase3_end = min(deadline, t_start + 1530.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(120.0, phase3_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _adam_optimize(best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                                             lr=0.02, beta_frac=0.003)
            update_best(p_best, sc_best, "adam_p3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: L-BFGS refinement (150s) ---
    phase4_end = min(deadline, t_start + 1680.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(100.0, phase4_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _lbfgs_refine(best_seq, min(phase4_end - 5, time.monotonic() + per_time),
                                            max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_p4")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Active set polish + final Nesterov (remaining) ---
    phase5_end = deadline
    if best_seq is not None and time.monotonic() < phase5_end:
        # Active set
        per_time = min(60.0, phase5_end - time.monotonic() - 30)
        if per_time > 15:
            p_best, sc_best = _active_set_polish(best_seq, min(phase5_end - 25, time.monotonic() + per_time))
            update_best(p_best, sc_best, "active_set_p5")

        # Final tiny Nesterov
        if time.monotonic() < phase5_end - 15:
            per_time2 = min(40.0, phase5_end - time.monotonic() - 5)
            if per_time2 > 10:
                beta_final = [1e-5, 5e-6, 2e-6, 1e-6]
                p_best, sc_best = _nesterov_optimize(best_seq, min(phase5_end - 2, time.monotonic() + per_time2),
                                                      beta_final, lr_init=0.004, mom_init=0.95)
                update_best(p_best, sc_best, "final_nesterov")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Final validation and output ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            # Scale to have sum = 1 (safe for evaluator)
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(4096, 'sine_sq')

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(4096, 'sine_sq')
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
