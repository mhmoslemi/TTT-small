  2. LP best-response refinement, mainly at n=1024, with resampling.
  3. Adam optimization in logit/softmax space using a smoothed max surrogate
     for the self-convolution peak.
  4. Multi-scale escalation to larger lengths, where discretization can improve
     the score.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_unit(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    c = _full_convolve(a)
    if c.size == 0:
        return np.inf
    mc = float(np.max(c))
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc)


def _normalize_unit(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return 0.0, np.array([], dtype=np.float64)
    mval = float(np.max(c))
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1e-8
    x = (c - mval) / beta
    x = np.minimum(x, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _gradient_from_weights(w, p):
    n = len(p)
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return 2.0 * h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _resample_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == 0:
        return None
    if n0 == n_new:
        return _normalize_unit(p)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    y = np.interp(x1, x0, p)
    return _normalize_unit(y)


def _profile_uniform(n):
    return np.ones(n, dtype=np.float64) / float(n)


def _profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_exp(n, lam, increasing=False):
    t = np.linspace(0.0, 1.0, n)
    if increasing:
        t = 1.0 - t
    a = np.exp(-lam * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_cosine(n, freq, amp):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / float(n))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_asym_gauss(n, center, width, skew):
    t = np.linspace(0.0, 1.0, n)
    x = (t - center) / width
    a = np.exp(-0.5 * x * (x + skew))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _profile_beta(n, alpha, beta):
    eps = 1e-6
    t = np.linspace(eps, 1.0 - eps, n)
    loga = alpha * np.log(t) + beta * np.log(1.0 - t)
    loga = loga - float(np.max(loga))
    a = np.exp(loga)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _profile_power_center(n, pw):
    t = np.linspace(0.0, 1.0, n)
    a = np.power(2.0 * np.minimum(t, 1.0 - t) + 1e-6, pw)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _profile_rand_smooth(n, rng, n_modes, amp):
    if n <= 2:
        return _profile_uniform(n)
    half = n // 2 + 1
    Z = np.zeros(half, dtype=np.complex128)
    max_mode = min(n_modes, half - 1)
    for m in range(1, max_mode + 1):
        amp_m = rng.normal(scale=1.0) / np.sqrt(float(m))
        phase = 2.0 * np.pi * rng.random()
        Z[m] = amp_m * np.exp(1j * phase)
    z = np.fft.irfft(Z, n)
    z = z * amp
    z = z - float(np.mean(z))
    return _softmax_from_z(z)


def _fallback_profile(n=1024):
    return _profile_linear(n, 1.5)


def _build_conv_matrix(a):
    n = len(a)
    diag_list = []
    offs = []
    for d in range(n - 1, -1, -1):
        diag_list.append(np.full(n, a[d], dtype=np.float64))
        offs.append(-d)
    return diags(diag_list, offsets=offs, shape=(2 * n - 1, n), format='csr')


def _lp_best_response(a, M):
    n = len(a)
    if n <= 0 or n > 1200:
        return None
    if not np.isfinite(M) or M <= 0.0:
        return None
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method='highs'
        )
    except Exception:
        return None
    if res.success:
        x = np.asarray(res.x, dtype=np.float64).copy()
        x = np.maximum(x, 0.0)
        x[x < 1e-15] = 0.0
        return x
    return None


def _lp_refine(a0, deadline, max_iters=10000):
    a = _normalize_unit(a0)
    if a is None:
        return None, np.inf

    best = a.copy()
    best_s = _score_unit(best)
    n = best.size
    c = _full_convolve(best)
    M = float(np.max(c))
    stall = 0

    t_choices = (1.0, 0.8, 0.6, 0.45, 0.35, 0.25, 0.2, 0.15, 0.1, 0.05)

    for it in range(max_iters):
        now = time.monotonic()
        if now > deadline - 2.0:
            break

        b = _lp_best_response(a, M)
        if b is None:
            break

        bn = _normalize_unit(b)
        if bn is None:
            break

        if float(np.max(np.abs(bn - a))) < 1e-9:
            break

        best_t = 0.35
        best_sc = best_s

        for tv in t_choices:
            if tv >= 1.0:
                a_try = bn
            else:
                a_try = (1.0 - tv) * a + tv * bn
            a_try = _normalize_unit(a_try)
            if a_try is None:
                continue
            sc = _score_unit(a_try)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_t = tv

        if not np.isfinite(best_sc) or best_sc >= best_s - 1e-15:
            stall += 1
            if stall >= 5:
                break
            continue

        a = _normalize_unit((1.0 - best_t) * a + best_t * bn)
        if a is None:
            break

        best = a.copy()
        best_s = best_sc
        c = _full_convolve(a)
        M = float(np.max(c))
        stall = 0

        if it % 20 == 0:
            print(f"  LP iter {it}: score={best_s:.12f} n={n}")

    return best, best_s


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.95,
                   stall_limit=300):
    p = np.asarray(p0, dtype=np.float64).ravel()
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-30:
        return None, np.inf
    p = p / s
    n = p.size
    if n == 0:
        return None, np.inf

    z = np.log(np.maximum(p, 1e-12))
    z = z - float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    c = _full_convolve(p)
    maxc = float(np.max(c))
    beta = max(1e-9, beta_frac * maxc)
    beta_min = max(1e-12, 1e-7 * maxc)

    best_p = p.copy()
    best_s = float(2.0 * n * maxc)
    best_z = z.copy()
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        vp = _gradient_from_weights(w, p)
        vp = np.maximum(vp, 0.0)

        if not np.all(np.isfinite(vp)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)

        c = _full_convolve(p)
        maxc = float(np.max(c))
        sc = float(2.0 * n * maxc)

        if not np.isfinite(sc):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        if sc < best_s - 1e-15:
            best_s = sc
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.95)

        if t % 30 == 0:
            if c.size > 2:
                c_no_max = c.copy()
                c_no_max[int(np.argmax(c))] = -np.inf
                second = float(np.max(c_no_max))
                if np.isfinite(second):
                    gap = maxc - second
                else:
                    gap = 0.0
            else:
                gap = 0.0

            if gap < 5.0 * beta:
                beta = max(beta_min, beta * 0.75)
            else:
                beta = max(beta_min, beta * beta_decay)

            if no_imp > max(50, stall_limit // 2):
                beta = min(max(1e-9, 0.04 * maxc), beta * 1.35)

        if no_imp > stall_limit:
            if best_s < 1.505:
                ns = 0.018
            elif best_s < 1.515:
                ns = 0.035
            elif best_s < 1.53:
                ns = 0.06
            else:
                ns = 0.12

            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)

            c = _full_convolve(p)
            maxc = float(np.max(c))
            beta = max(1e-9, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t0 = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 1800.0
    if not np.isfinite(total) or total <= 30.0:
        total = max(30.0, total if np.isfinite(total) else 1800.0)
    deadline = t0 + max(20.0, total - 18.0)

    try:
        seed_i = int(seed)
    except Exception:
        seed_i = 42
    rng = np.random.default_rng(seed_i % (2**63 - 1))

    cands = []
    idx = 0

    # Parent, if available.
    try:
        par = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        par = None

    if par is not None:
        par = np.clip(par, 0.0, 1000.0)
        s = float(np.sum(par))
        if np.isfinite(s) and s > 1e-12:
            par = par / s
            pn = par.size
            if 1 <= pn <= 4096:
                sc = _score_unit(par)
                if np.isfinite(sc):
                    cands.append((sc, idx, par, pn, 'parent'))
                    idx += 1
            for tn in (512, 1024, 2048, 4096):
                if pn != tn:
                    pi = _resample_profile(par, tn)
                    if pi is not None:
                        sc = _score_unit(pi)
                        if np.isfinite(sc):
                            cands.append((sc, idx, pi, tn, f'parent_{tn}'))
                            idx += 1

    # Profile initialization at multiple scales.
    for n in (512, 1024, 2048, 4096):
        profs = []

        profs.append(('unif', _profile_uniform(n)))

        for eta in (0.5, 1.0, 1.5, 2.0, 3.0):
            profs.append((f'lin{eta}', _profile_linear(n, eta)))

        profs.append(('q1', _profile_quadratic(n, 1.0, -0.5)))
        profs.append(('q2', _profile_quadratic(n, 1.5, -1.0)))
        profs.append(('q3', _profile_quadratic(n, 0.5, 0.5)))
        profs.append(('q4', _profile_quadratic(n, -1.0, 1.5)))
        profs.append(('q5', _profile_quadratic(n, -1.5, 2.0)))

        profs.append(('c1', _profile_cubic(n, 1.0, -0.5)))
        profs.append(('c2', _profile_cubic(n, 1.5, -1.0)))
        profs.append(('c3', _profile_cubic(n, 2.0, -1.2)))
        profs.append(('c4', _profile_cubic(n, -1.0, 0.5)))
        profs.append(('c5', _profile_cubic(n, -2.0, 1.2)))

        for lam in (0.5, 1.0, 2.0, 4.0):
            profs.append((f'dec{lam}', _profile_exp(n, lam, False)))
            profs.append((f'inc{lam}', _profile_exp(n, lam, True)))

        for freq in (1, 2, 3):
            for amp in (0.3, 0.6, 0.9):
                profs.append((f'cos{freq}_{amp}', _profile_cosine(n, freq, amp)))

        profs.append(('tb1', _profile_two_bump(n, 0.3, 0.7, 0.12)))
        profs.append(('tb2', _profile_two_bump(n, 0.25, 0.75, 0.18)))
        profs.append(('tb3', _profile_two_bump(n, 0.2, 0.8, 0.25)))
        profs.append(('tb4', _profile_two_bump(n, 0.35, 0.65, 0.10)))
        profs.append(('tb5', _profile_two_bump(n, 0.15, 0.85, 0.30)))

        profs.append(('ag1', _profile_asym_gauss(n, 0.35, 0.2, 0.5)))
        profs.append(('ag2', _profile_asym_gauss(n, 0.4, 0.25, 1.0)))
        profs.append(('ag3', _profile_asym_gauss(n, 0.3, 0.15, 0.8)))
        profs.append(('ag4', _profile_asym_gauss(n, 0.6, 0.2, -0.5)))
        profs.append(('ag5', _profile_asym_gauss(n, 0.55, 0.25, -1.0)))

        beta_params = (
            (0.5, 0.5), (1.0, 0.5), (0.5, 1.0), (2.0, 0.5), (0.5, 2.0),
            (1.0, 2.0), (2.0, 1.0), (1.0, 0.0), (0.0, 1.0), (2.0, 0.0),
            (0.0, 2.0), (-0.2, 0.5), (0.5, -0.2), (0.8, 0.2), (0.2, 0.8)
        )
        for al, be in beta_params:
            profs.append((f'b{al}_{be}', _profile_beta(n, al, be)))

        for pw in (0.25, 0.5, 1.0, 1.5, 2.0):
            profs.append((f'pw{pw}', _profile_power_center(n, pw)))

        t = np.linspace(0.0, 1.0, n)
        for curv in (-1.0, -0.5, 0.5, 1.0, 2.0):
            u = 2.0 * t - 1.0
            a = 1.0 + curv * u * u
            a = np.maximum(a, 1e-12)
            profs.append((f'cq{curv}', a / a.sum()))

        for nm in (2, 4, 8, 16):
            for amp in (0.5, 1.0, 2.0):
                profs.append((f'r{nm}_{amp}', _profile_rand_smooth(n, rng, nm, amp)))

        for label, p in profs:
            p = _normalize_unit(p)
            if p is None:
                continue
            sc = _score_unit(p)
            if np.isfinite(sc):
                cands.append((sc, idx, p, n, label))
                idx += 1

    if len(cands) == 0:
        fb = _fallback_profile(1024)
        return [float(x) for x in fb]

    cands.sort()
    sc0, _, p0, n0, label0 = cands[0]
    best_p = np.asarray(p0, dtype=np.float64).copy()
    best_s = float(sc0)
    best_n = int(n0)
    cands = None

    print(f"Init best: score={best_s:.12f} n={best_n} label={label0}")

    # --- Phase 1: Adam polish at the best initial scale ---
    now = time.monotonic()
    phase1_end = min(deadline - 1000.0, max(now + 30.0, t0 + 0.18 * total))
    if phase1_end > now + 15.0:
        p1, s1 = _adam_optimize(
            best_p,
            phase1_end,
            rng,
            lr=0.05,
            beta_frac=0.09,
            beta_decay=0.94,
            stall_limit=max(150, int(0.20 * best_n))
        )
        if p1 is not None and np.isfinite(s1) and s1 < best_s:
            best_s = float(s1)
            best_p = np.asarray(p1, dtype=np.float64).copy()
            best_n = best_p.size
            print(f"After init Adam: score={best_s:.12f} n={best_n}")

    # --- Phase 2: Strong LP refinement at 1024, with resampling if needed ---
    now = time.monotonic()
    p1024 = _resample_profile(best_p, 1024)
    if p1024 is not None:
        sc1024 = _score_unit(p1024)
        if np.isfinite(sc1024) and sc1024 < best_s:
            best_s = float(sc1024)
            best_p = p1024.copy()
            best_n = 1024
            print(f"Resampled to 1024: score={best_s:.12f}")

    phase2_end = min(deadline - 900.0, now + 550.0)
    if phase2_end > now + 60.0:
        p1024 = _resample_profile(best_p, 1024)
        if p1024 is not None:
            p_lp, s_lp = _lp_refine(p1024, phase2_end, max_iters=20000)
            if p_lp is not None and np.isfinite(s_lp) and s_lp < best_s:
                best_s = float(s_lp)
                best_p = np.asarray(p_lp, dtype=np.float64).copy()
                best_n = best_p.size
                print(f"After LP: score={best_s:.12f} n={best_n}")

            now = time.monotonic()
            phase2b_end = min(deadline - 800.0, now + 150.0)
            if phase2b_end > now + 20.0:
                p2b, s2b = _adam_optimize(
                    best_p,
                    phase2b_end,
                    rng,
                    lr=0.025,
                    beta_frac=0.04,
                    beta_decay=0.96,
                    stall_limit=250
                )
                if p2b is not None and np.isfinite(s2b) and s2b < best_s:
                    best_s = float(s2b)
                    best_p = np.asarray(p2b, dtype=np.float64).copy()
                    best_n = best_p.size
                    print(f"After post-LP Adam: score={best_s:.12f} n={best_n}")

    # --- Phase 3: More Adam at the best current scale ---
    now = time.monotonic()
    phase3_end = min(deadline - 600.0, now + 250.0)
    if phase3_end > now + 30.0:
        p3, s3 = _adam_optimize(
            best_p,
            phase3_end,
            rng,
            lr=0.03,
            beta_frac=0.05,
            beta_decay=0.95,
            stall_limit=max(250, int(0.15 * best_n))
        )
        if p3 is not None and np.isfinite(s3) and s3 < best_s:
            best_s = float(s3)
            best_p = np.asarray(p3, dtype=np.float64).copy()
            best_n = best_p.size
            print(f"After mid Adam: score={best_s:.12f} n={best_n}")

    # --- Phase 4: Escalate to 2048 ---
    now = time.monotonic()
    if now < deadline - 450.0:
        p2048 = _resample_profile(best_p, 2048)
        if p2048 is not None:
            sc2048 = _score_unit(p2048)
            if np.isfinite(sc2048) and sc2048 < best_s:
                best_s = float(sc2048)
                best_p = p2048.copy()
                best_n = 2048
                print(f"Resampled to 2048: score={best_s:.12f}")

            phase4_end = min(deadline - 400.0, now + 450.0)
            if phase4_end > now + 60.0:
                p4, s4 = _adam_optimize(
                    p2048,
                    phase4_end,
                    rng,
                    lr=0.028,
                    beta_frac=0.05,
                    beta_decay=0.95,
                    stall_limit=400
                )
                if p4 is not None and np.isfinite(s4) and s4 < best_s:
                    best_s = float(s4)
                    best_p = np.asarray(p4, dtype=np.float64).copy()
                    best_n = best_p.size
                    print(f"After 2048 Adam: score={best_s:.12f} n={best_n}")

    # --- Phase 5: Escalate to 4096 and use remaining time ---
    now = time.monotonic()
    if now < deadline - 80.0:
        p4096 = _resample_profile(best_p, 4096)
        if p4096 is not None:
            sc4096 = _score_unit(p4096)
            if np.isfinite(sc4096) and sc4096 < best_s:
                best_s = float(sc4096)
                best_p = p4096.copy()
                best_n = 4096
                print(f"Resampled to 4096: score={best_s:.12f}")

            final_end = deadline - 35.0
            if final_end > now + 45.0:
                p5, s5 = _adam_optimize(
                    p4096,
                    final_end,
                    rng,
                    lr=0.025,
                    beta_frac=0.045,
                    beta_decay=0.96,
                    stall_limit=500
                )
                if p5 is not None and np.isfinite(s5) and s5 < best_s:
                    best_s = float(s5)
                    best_p = np.asarray(p5, dtype=np.float64).copy()
                    best_n = best_p.size
                    print(f"After final 4096 Adam: score={best_s:.12f} n={best_n}")

    # --- Final normalization and optional real evaluator check ---
    final = _normalize_unit(best_p)
    if final is None:
        final = _fallback_profile(1024)

    final = np.clip(final, 0.0, 1000.0)
    s = float(np.sum(final))
    if np.isfinite(s) and s > 0.01:
        final = final / s

    print(f"FINAL internal score: {_score_unit(final):.12f} n={final.size}")

    try:
        eval_sc = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(eval_sc):
            print(f"FINAL evaluate_sequence: {eval_sc:.12f}")
    except Exception:
        pass

    return [float(x) for x in final]
