def _normalize(a):

def _project_simplex(v, target_sum=1.0):

    v = np.asarray(v, dtype=np.float64).copy()

    n = a.size
    # (a*a)_k = sum_i a_i * a_{k-i}
    # d(a*a)_k / da_i = 2 * a_{k-i} (for valid k-i)

    # = (w * a_reversed)[i + n - 1] (full convolution)

    # This is the cross-correlation of w and a at lag i:

    # Actually, let me just compute it directly for clarity.

    # = 2 * (w * a)[i] in the "cross-correlation" sense

    # g[i] = 2 * sum_j w[i+j] * a[j] = 2 * (w ⋆ a)[i]

def _lp_solve_given_a(a, T):
    """Solve LP: max sum(b) s.t. (a*b)_k <= T, b >= 0."""
    n = a.size
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, T)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    c = _full_convolve(a)
    M = c.max()

            break

        T = M * 0.99
        b = _lp_solve_given_a(a, T)

            # Try with current peak

            best_a = a_new.copy()
        
        a = a_new
        c = _full_convolve(a)
        M = c.max()
        
        if M < best_score * 1.001 / (2 * n):
            # Can't improve much more with this approach
            break
    
    return best_a, best_score


def _multiplicative_flatten(a0, deadline, eta0=0.1, n_target_reduction=0.99):
    """Multiplicative update to flatten the convolution."""
    n = a0.size
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    eta = eta0
    
    c = _full_convolve(a)
    target = c.max() * n_target_reduction
    
    for it in range(10000):
        if time.monotonic() >= deadline:
            break
        
        c = _full_convolve(a)
        maxc = c.max()
        score = 2.0 * n * maxc
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Compute the "excess" at each position
        excess = np.maximum(c - target, 0)
        
        # Compute force on each position
        # force[i] = sum_k excess[k] * a[k-i] / sum(a)
        N = _next_pow2(2 * n - 1)
        fe = np.fft.rfft(excess, N)
        fa = np.fft.rfft(a[::-1], N)
        force = np.fft.irfft(fe * fa, N)[:2*n-1]
        # force[i + n - 1] = sum_j excess[j] * a[n-1-(j-i-n+1)] = sum_j excess[j] * a[i-n+1+j]
        # Hmm, let me just compute it as: force[i] = sum_k excess[k] * a[k-i] for valid k-i
        # This is cross_correlation(excess, a)[i]
        # = full_convolve(excess, a_flipped)[i + n - 1]
        # where a_flipped[j] = a[n-1-j]
        # full_convolve(excess, a_flipped)[m] = sum_j excess[j] * a_flipped[m-j]
        # = sum_j excess[j] * a[n-1-m+j]
        # We want: sum_k excess[k] * a[k-i]
        # Let j = k: sum_j excess[j] * a[j-i]
        # = full_convolve(excess, a)[j-i + ...] 
        # Actually: full_convolve(f, g)[m] = sum_j f[j] * g[m-j]
        # So full_convolve(excess, a)[i] = sum_j excess[j] * a[i-j]
        # That's not quite what we want. We want sum_k excess[k] * a[k-i].
        # = full_convolve(excess, a)[2*(n-1) - i] (by symmetry of convolution)
        # No, that's for the full convolution.
        # Let me just use: g[i] = sum_k excess[k] * a[k-i] = (excess ⋆ a)[i]
        # where ⋆ is cross-correlation.
        # Cross-correlation can be computed as: ifft(conj(fft(excess)) * fft(a))
        
        fe2 = np.fft.rfft(excess, N)
        fa2 = np.fft.rfft(a, N)
        cross_corr = np.fft.irfft(np.conj(fe2) * fa2, N)[:n]
        # cross_corr[i] = sum_k excess[k] * a[k-i] (circular)
        # For linear, it's approximately the same when n is large
        
        force = 2.0 * cross_corr
        
        if force.max() < 1e-20:
            break
        
        # Multiplicative update
        factor = np.exp(-eta * force / force.max())
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Decrease target
        target = max(target * 0.999, maxc * 0.95)
        eta = max(0.01, eta * 0.999)
    
    return best_a, best_score


def _fourier_optimize(a0, deadline, n_harmonics=10):
    """Optimize a small number of Fourier coefficients."""
    n = a0.size
    a = a0.copy()
    a = a / a.sum()
    
    # Get current Fourier coefficients
    af = np.fft.rfft(a)
    
    # We'll optimize the first n_harmonics of the real part
    # Parameterize: a_i = (1 + sum_{m=1}^M c_m cos(2*pi*m*i/n) + sum_{m=1}^M d_m sin(2*pi*m*i/n)) / n
    # subject to a_i >= 0
    M = n_harmonics
    
    def objective(params):
        c = params[:M]
        d = params[M:]
        t = np.arange(n)
        a_try = np.ones(n)
        for m in range(1, M + 1):
            a_try += c[m-1] * np.cos(2*np.pi*m*t/n) + d[m-1] * np.sin(2*np.pi*m*t/n)
        if a_try.min() < -0.5:
            return 1e6
        a_try = np.maximum(a_try, 1e-10)
        a_try = a_try / a_try.sum()
        return _score(a_try)
    
    # Initial point: small perturbations
    x0 = np.zeros(2 * M)
    # Initialize from current sequence
    for m in range(1, M + 1):
        x0[m-1] = 2.0 * np.sum(a * np.cos(2*np.pi*m*np.arange(n)/n)) * n
        x0[M+m-1] = 2.0 * np.sum(a * np.sin(2*np.pi*m*np.arange(n)/n)) * n
    # Scale down
    x0 = x0 * 0.1
    
    result = minimize(objective, x0, method='Nelder-Mead',
                      options={'maxiter': 200, 'xatol': 1e-6, 'fatol': 1e-8})
    
    if result.fun < _score(a):
        c = result.x[:M]
        d = result.x[M:]
        t = np.arange(n)
        a_new = np.ones(n)
        for m in range(1, M + 1):
            a_new += c[m-1] * np.cos(2*np.pi*m*t/n) + d[m-1] * np.sin(2*np.pi*m*t/n)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        return a_new, _score(a_new)
    
    return a, _score(a)
