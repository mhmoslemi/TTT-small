"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP fixed-point, game-theoretic QP iterations, Adam smoothed-max,
and multi-scale search. Uses FFT-based convolution and efficient Qa computation.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from numpy.lib.stride_tricks import sliding_window_view
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_primal(a, M, n):
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_fixed_point(a, deadline, max_iters=50, use_line_search=True):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())

        b = _lp_primal(a, M, n)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        if use_line_search:
            best_t = 0.5
            best_sc = _score_norm(a)
            for t_candidate in [0.3, 0.5, 0.7, 0.9, 1.0]:
                a_t = (1.0 - t_candidate) * a + t_candidate * b
                a_t = np.maximum(a_t, 1e-15)
                a_t = a_t / a_t.sum()
                sc_t = _score_norm(a_t)
                if sc_t < best_sc:
                    best_sc = sc_t
                    best_t = t_candidate
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
        else:
            a = b
            a = a / a.sum()

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _qa_product(a, y):
    """Compute (Q a)_j = sum_l a[l] * y[j+l] where y has length 2n-1."""
    n = len(a)
    Y = sliding_window_view(y, n)  # shape (n, n), Y[j, l] = y[j+l]
    return Y @ a


def _game_iterate(a, deadline, max_iters=200, init_temp=0.3, temp_decay=0.99):
    """Game-theoretic iteration: alternate between computing lag weights
    and solving the QP min a^T Q a s.t. a in simplex."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    temp = init_temp
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        if M < 1e-15:
            break

        # Compute soft weights on lags
        x = (c - M) / (temp * M)
        y = np.exp(x)
        s = y.sum()
        if s < 1e-30:
            y = np.zeros_like(c)
            y[int(np.argmax(c))] = 1.0
        else:
            y = y / s

        # Solve QP via projected gradient
        a_qp = a.copy()
        # Estimate step size
        eta = 0.4 / max(1.0, float(y.max()) * n * 0.1)
        for step in range(30):
            grad = 2.0 * _qa_product(a_qp, y)
            a_qp = a_qp - eta * grad
            a_qp = np.maximum(a_qp, 0.0)
            ss = a_qp.sum()
            if ss < 1e-15:
                a_qp = a.copy()
                break
            a_qp = a_qp / ss

        # Mix and update
        t_mix = 0.3 + 0.4 * min(1.0, it / 20.0)
        a_new = (1.0 - t_mix) * a + t_mix * a_qp
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score_norm(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        # Decay temperature periodically
        if it % 30 == 0:
            temp = max(0.05, temp * temp_decay)
            if no_imp > 30:
                temp = min(1.0, temp * 2.0)  # increase temp to explore

    return best_a, best_score


def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None,
                     beta_decay=0.95, no_imp_limit=200):
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.1 * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_correlate(w, a)

        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.003, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        if t_adam % 50 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-4, beta * 0.5)

        if no_imp > no_imp_limit:
            ns = 0.08 if best_score < 1.51 else 0.15
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.1 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, profile_type, param1=1.0, param2=0.0, param3=0.0):
    if profile_type == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (t - 0.5)
    elif profile_type == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (2 * t - 1) ** 2
    elif profile_type == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + param1 * u + param2 * u ** 3
    elif profile_type == 'exp':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-param1 * t / n)
    elif profile_type == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + param1 * np.cos(2 * np.pi * param2 * t / n)
    elif profile_type == 'sine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + param1 * np.sin(2 * np.pi * param2 * t / n)
    elif profile_type == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + param1
    elif profile_type == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - param1) / param3) ** 2) + np.exp(-0.5 * ((t - param2) / param3) ** 2)
    elif profile_type == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / param1) ** 2)
    elif profile_type == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + param1
    elif profile_type == 'edge_heavy':
        t = np.arange(n, dtype=np.float64) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** param1
    elif profile_type == 'asym_linear':
        t = np.linspace(0, 1, n)
        a = (1.0 + param1 * (1.0 - t)) / (1.0 + param1 * t)
    elif profile_type == 'power':
        t = np.linspace(0.01, 1, n)
        a = t ** param1
    elif profile_type == 'sawtooth':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + param1 * (2.0 * np.abs(t / n * param2 - np.floor(t / n * param2 + 0.5)) - 1.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _coord_descent(a, deadline, max_sweeps=20):
    """Coordinate descent: try scaling each element by various factors."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Shuffle coordinate order
        order = np.arange(n)
        np.random.shuffle(order)
        for j in order:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            # Binary search for optimal scaling
            lo, hi = 0.1, 10.0
            best_f = 1.0
            best_sc = best_score
            for _ in range(8):
                mid = (lo + hi) / 2.0
                a[j] = old_val * mid
                a[j] = max(a[j], 1e-15)
                sc = _score_norm(a)
                if sc < best_sc:
                    best_sc = sc
                    best_f = mid
                if sc <= best_score:
                    lo = mid
                else:
                    hi = mid
            if best_f != 1.0:
                a[j] = max(old_val * best_f, 1e-15)
                a = a / a.sum()
                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()
                    improved = True
            else:
                a[j] = old_val
        if not improved:
            break

    return best_a, best_score
