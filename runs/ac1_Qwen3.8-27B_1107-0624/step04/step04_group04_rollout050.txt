"""
Optimization of 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multiplicative (exponential) gradient descent with adaptive weighted active set
2. Gerchberg-Saxton style frequency-domain spectrum flattening
3. L-BFGS with logsumexp smoothing for fine-tuning

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _make_bell_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    return a / a.sum()


def _make_chebyshev_profile(n, order):
    x = np.cos(np.pi * np.linspace(0, 1, n))
    a = np.abs(np.cos(order * np.arccos(np.clip(x, -1, 1)))) + 0.1
    return a / a.sum()


def _mult_grad_step(a, beta, top_k_frac=0.3, eta=0.1):
    """
    One multiplicative gradient step.
    Returns (a_new, score, h, active_weights)
    """
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())

    # Compute softmax weights over the convolution peaks
    if top_k_frac < 1.0 and top_k_frac > 0.0:
        k = max(1, int(top_k_frac * len(c)))
        idx = np.argpartition(c, -k)[-k:]
        c_top = c[idx]
        c_shift = (c_top - c_top.max()) / beta
        e = np.exp(c_shift)
        s = e.sum()
        w = np.zeros(len(c))
        if s > 0 and np.isfinite(s):
            w[idx] = e / s
        else:
            w[idx[0]] = 1.0
    else:
        c_shift = (c - cm) / beta
        e = np.exp(c_shift)
        s = e.sum()
        if s > 0 and np.isfinite(s):
            w = e / s
        else:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0

    # h[j] = sum_k w[k] * a[k-j] (with zero padding)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]

    # Weighted mean of h
    h_bar = float(np.dot(a, h))

    # Multiplicative update: a_new[j] = a[j] * exp(-eta * (h[j] - h_bar) / scale)
    scale = float(np.abs(h - h_bar).max()) + 1e-30
    log_update = -eta * (h - h_bar) / scale
    log_update = np.clip(log_update, -5.0, 5.0)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        a_new = a.copy()
    else:
        a_new = a_new / s_new

    sc = _score_norm(a_new)
    return a_new, sc, h, h_bar, w, c, cm


def _mult_descent(a_init, deadline, eta0=0.5, decay=0.9995, top_k_frac=0.3,
                  beta0=0.1, beta_decay=0.999, max_iter=500000):
    """
    Multiplicative exponential gradient descent with cooling.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    eta = eta0
    beta = beta0 * max(_full_convolve(a, a).max(), 1e-10)

    stagnant = 0
    patience = 5000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_new, sc, h, h_bar, w, c, cm = _mult_grad_step(a, beta, top_k_frac, eta)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
        else:
            stagnant += 1
            # If stagnant for a while, reduce eta and increase top_k
            if stagnant == patience:
                eta *= 0.7
                beta = max(1e-10, beta * 0.5)
                stagnant = 0
            elif stagnant == patience * 2:
                top_k_frac = min(1.0, top_k_frac * 1.3)
                stagnant = 0
            elif stagnant == patience * 3:
                break

        # Cooling
        eta *= decay
        beta = max(1e-12, beta * beta_decay)

        if it % 50000 == 0 and it > 0:
            elapsed_info = ""
            print(f"    iter {it}: score={best_score:.10f}, eta={eta:.6f}, "
                  f"top_k_frac={top_k_frac:.2f}, beta={beta:.2e}")

    return best_a, best_score


def _gs_flatten(a_init, deadline, n_iter=50000, alpha=0.3, proj_alpha=0.5):
    """
    Gerchberg-Saxton style: flatten the power spectrum in frequency domain.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    L = _next_pow2(2 * n - 1)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # FFT of zero-padded sequence
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        # Power spectrum
        power = np.abs(F) ** 2
        mean_power = power.mean()

        # Flatten: reduce peaks in frequency domain
        # We want to make |F|^2 more uniform
        excess = np.maximum(power - mean_power, 0)
        if excess.max() < 1e-15:
            break

        # Scale down the excess
        reduction = alpha * excess / (excess.max() + 1e-30)
        new_power = power - reduction
        new_power = np.maximum(new_power, 0)

        # Reconstruct with original phases
        phase = F / (np.abs(F) + 1e-30)
        F_new = np.sqrt(new_power) * phase
        a_recon = np.fft.ifft(F_new).real[:n]

        # Project: non-negative, sum=1
        # Use soft projection: blend with uniform to maintain non-negativity
        a_pos = np.maximum(a_recon, 0)
        s = a_pos.sum()
        if s < 1e-15:
            a_pos = np.ones(n) / n
        else:
            a_pos = a_pos / s

        # Blend to maintain continuity
        a_new = (1 - proj_alpha) * a + proj_alpha * a_pos
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step
            alpha = max(0.01, alpha * 0.95)

        if it % 5000 == 0 and it > 0:
            print(f"    GS iter {it}: score={best_score:.10f}, alpha={alpha:.4f}")

    return best_a, best_score


def _z_objective(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)


def _z_gradient(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[idx[0]] = 1.0
        else:
            w = np.zeros_like(c)
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, max_wall_time=None, top_k=0, beta_multiplier=0.01):

        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.03, top_k=0, n_max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.005 * max_c)

    cur_k = 3 if top_k == 0 else top_k
    adaptive = (top_k == 0)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        use_k = cur_k if adaptive else top_k
        grad = _z_gradient(z, beta, n, use_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 200 == 0:
            beta = max(1e-14, beta * 0.9)
            if adaptive and t % 1000 == 0:
                cur_k = min(cur_k + 3, 40)

        if t % 5000 == 0:
            if best_score < 1.51:
                lr = max(0.002, lr * 0.95)

    return best_a, best_score


def _direct_kkt_balance(a_init, deadline, max_iter=10000, threshold_frac=0.99):
    """
    Direct KKT balancing: at optimum, h[j] should be constant for all positive j.
    Iteratively enforce this using multiplicative updates.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    step = 0.1
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = cm * threshold_frac
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            break
        if len(active) == len(c):
            threshold_frac = max(0.5, threshold_frac * 0.9)
            continue

        # h[j] = (1/|A|) * sum_{k in A} a[k-j]
        a_rev = a[::-1]
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

        # KKT: h[j] should be constant for positive j
        pos_mask = a > 1e-10
        if not np.any(pos_mask):
            break
        h_bar = float(np.dot(a[pos_mask], h[pos_mask]) / a[pos_mask].sum())

        # Multiplicative correction
        diff = h - h_bar
        scale = float(np.abs(diff).max()) + 1e-30
        log_update = -step * diff / scale
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            # Slightly increase threshold to include more peaks
            threshold_frac = max(0.95, threshold_frac + 0.001)
        else:
            step *= 0.7
            if step < 1e-6:
                break

        if it % 2000 == 0 and it > 0:
            print(f"    KKT iter {it}: score={best_score:.10f}, "
                  f"|A|={len(active)}, step={step:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 0: Initial candidates from parent and profiles
    Phase 1: Multiplicative gradient descent (main workhorse)
    Phase 2: GS frequency-domain flattening
    Phase 3: L-BFGS fine-tuning
    Phase 4: Direct KKT balancing
    Phase 5: Multi-scale + final polish
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initial candidates ---
    print("=== Phase 0: Initial candidates ===")
    phase0_end = min(deadline, t_start + 20.0)

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm(parent)
                print(f"  Parent: n={len(parent)}, normalized score={sc:.10f}")
                update_best(parent, sc, "parent")
    except Exception as e:
        print(f"  No parent available: {e}")

    n_search = 4096
    if parent is not None:
        n_search = len(parent)

    # Generate diverse profiles
    candidates = []
    profiles = [
        _make_exp_profile(n_search, 0.002),
        _make_exp_profile(n_search, 0.005),
        _make_uniform(n_search),
        _make_bell_profile(n_search, 0.2),
        _make_bell_profile(n_search, 0.3),
        _make_chebyshev_profile(n_search, 5),
        _make_chebyshev_profile(n_search, 10),
    ]

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    # Interpolated parent
    if parent is not None and len(parent) != n_search:
        p_int = _interp_profile(parent, n_search)
        sc = _score_norm(p_int)
        candidates.append((sc, p_int, "parent_interp"))
        update_best(p_int, sc, "parent_interp")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}")

    # Use parent if available, else best profile
    if parent is not None:
        a_work = parent.copy()
    else:
        a_work = candidates[0][1].copy()

    # --- Phase 1: Multiplicative gradient descent ---
    print("=== Phase 1: Multiplicative gradient descent ===")
    phase1_end = min(deadline, t_start + 900.0)
    remaining1 = phase1_end - time.monotonic()

    if remaining1 > 30:
        # Run with multiple parameter settings
        settings = [
            {'eta0': 0.8, 'decay': 0.9998, 'top_k_frac': 0.2, 'beta0': 0.05},
            {'eta0': 0.5, 'decay': 0.9997, 'top_k_frac': 0.3, 'beta0': 0.1},
            {'eta0': 0.3, 'decay': 0.9999, 'top_k_frac': 0.5, 'beta0': 0.2},
        ]

        per_setting = remaining1 / len(settings)
        for si, st in enumerate(settings):
            if time.monotonic() > phase1_end - 10:
                break
            dl = min(phase1_end - 10, time.monotonic() + per_setting)
            print(f"  Setting {si}: eta0={st['eta0']}, top_k_frac={st['top_k_frac']}")
            p_best, sc_best = _mult_descent(
                a_work, dl,
                eta0=st['eta0'], decay=st['decay'],
                top_k_frac=st['top_k_frac'], beta0=st['beta0']
            )
            update_best(p_best, sc_best, f"mult_s{si}")
            a_work = p_best.copy()

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: GS frequency-domain flattening ---
    print("=== Phase 2: GS frequency flattening ===")
    phase2_end = min(deadline, t_start + 1100.0)
    remaining2 = phase2_end - time.monotonic()

    if remaining2 > 30 and best_seq is not None:
        dl = min(phase2_end - 5, time.monotonic() + remaining2 * 0.7)
        p_gs, sc_gs = _gs_flatten(best_seq, dl, n_iter=30000, alpha=0.2, proj_alpha=0.3)
        update_best(p_gs, sc_gs, "gs")

        # Also try GS from a different starting point
        if time.monotonic() < phase2_end - 20:
            p_gs2, sc_gs2 = _gs_flatten(a_work, min(phase2_end - 5, time.monotonic() + 60.0),
                                         n_iter=10000, alpha=0.4, proj_alpha=0.5)
            update_best(p_gs2, sc_gs2, "gs2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: L-BFGS fine-tuning ---
    print("=== Phase 3: L-BFGS fine-tuning ===")
    phase3_end = min(deadline, t_start + 1450.0)
    remaining3 = phase3_end - time.monotonic()

    if remaining3 > 30 and best_seq is not None:
        # Multiple L-BFGS runs with different top_k
        for tk in [0, 5, 10, 20]:
            if time.monotonic() > phase3_end - 15:
                break
            per_lbfgs = min(80.0, (phase3_end - time.monotonic()) * 0.5)
            dl = min(phase3_end - 10, time.monotonic() + per_lbfgs)
            p_l, sc_l = _run_lbfgs(best_seq, dl, max_wall_time=per_lbfgs, top_k=tk)
            update_best(p_l, sc_l, f"lbfgs_k{tk}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Adam polish ---
    print("=== Phase 4: Adam polish ===")
    phase4_end = min(deadline, t_start + 1650.0)
    remaining4 = phase4_end - time.monotonic()

    if remaining4 > 30 and best_seq is not None:
        dl = min(phase4_end - 5, time.monotonic() + remaining4 * 0.6)
        p_a, sc_a = _adam_polish(best_seq, dl, lr=0.02, top_k=0, n_max_iter=60000)
        update_best(p_a, sc_a, "adam1")

        if time.monotonic() < phase4_end - 20:
            dl2 = min(phase4_end - 5, time.monotonic() + 120.0)
            p_a2, sc_a2 = _adam_polish(best_seq, dl2, lr=0.008, top_k=10, n_max_iter=30000)
            update_best(p_a2, sc_a2, "adam2")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Direct KKT balancing ---
    print("=== Phase 5: KKT balancing ===")
    phase5_end = min(deadline, t_start + 1750.0)
    remaining5 = phase5_end - time.monotonic()

    if remaining5 > 20 and best_seq is not None:
        dl = min(phase5_end - 5, time.monotonic() + remaining5 * 0.7)
        p_k, sc_k = _direct_kkt_balance(best_seq, dl, max_iter=20000, threshold_frac=0.99)
        update_best(p_k, sc_k, "kkt_direct")

        # Follow up with short mult descent
        if time.monotonic() < phase5_end - 10:
            dl2 = min(phase5_end - 5, time.monotonic() + 60.0)
            p_m, sc_m = _mult_descent(best_seq, dl2, eta0=0.2, decay=0.9999,
                                       top_k_frac=0.5, beta0=0.3)
            update_best(p_m, sc_m, "mult_final")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different lengths ---
    print("=== Phase 6: Different lengths ===")
    phase6_end = min(deadline, t_start + 1790.0)
    if time.monotonic() < phase6_end - 10 and best_seq is not None:
        n_curr = len(best_seq)
        for tn in [2048, 8192]:
            if tn == n_curr or time.monotonic() > phase6_end - 15:
                continue
            p_int = _interp_profile(best_seq, tn)
            sc_int = _score_norm(p_int)
            update_best(p_int, sc_int, f"interp_n{tn}")
            # Quick optimization at this length
            if time.monotonic() < phase6_end - 15:
                dl = min(phase6_end - 5, time.monotonic() + 30.0)
                p_opt, sc_opt = _mult_descent(p_int, dl, eta0=0.5, decay=0.9998,
                                               top_k_frac=0.3, beta0=0.1)
                update_best(p_opt, sc_opt, f"opt_n{tn}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            # Normalize to make sum reasonable (the evaluator handles any sum)
            # But we want to return values that give the best score
            # The score is scale-invariant, so we can return as-is
            final_score = _score_full(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.003)
            final_score = _score_full(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence if feasible
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 8192:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
            else:
                # For longer sequences, just report our computed score
                print(f"  (sequence too long for evaluate_sequence check, n={len(seq_list)})")
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_exp_profile(4096, 0.003)
    print(f"  FALLBACK: n=4096, score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
