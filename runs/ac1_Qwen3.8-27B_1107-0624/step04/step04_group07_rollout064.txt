"""
Multi-strategy search for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Core ideas:
  - Normalize to the simplex (sum=1, a>=0), so the objective is 2*n*max(conv).
  - Soft-max smoothing of the max gives a differentiable surrogate whose gradient
    is an FFT cross-correlation: grad_j = 4n*sum_k w_k a_{k-j}, w_k = softmax(conv).
  - Projected gradient descent on the simplex with backtracking line search and
    beta annealing, plus periodic noise restarts.
  - LP fixed-point polish (max sum(b) s.t. conv(a,b)_k <= M, b>=0) blended by line
    search, alternating with the gradient phase.
  - Multi-scale escalation (interpolate best to larger n and re-optimize) and random
    restarts to escape local minima.
All heavy ops use FFT (O(n log n)). No filesystem/network IO.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _cross_correlate(w, a):
    """g[j] = sum_k w[k]*a[k-j]  (0<=j<n, zero outside) via FFT cross-correlation."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    res = np.fft.irfft(fw * np.conj(fa), N)
    return res[:n]


def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _softmax_grad(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    mx = x.max()
    ex = np.exp(x - mx)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s) + mx), w


def _project_simplex(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _line_search(a, grad, max_lr, base_score, n_trials=10):
    best_a = a
    best_sc = base_score
    best_lr = max_lr
    lr = max_lr
    for _ in range(n_trials):
        a_t = _project_simplex(a - lr * grad)
        sc = _score(a_t)
        if sc < best_sc:
            best_sc = sc
            best_a = a_t
            best_lr = lr
        lr *= 0.5
    return best_a, best_sc, best_lr


def _gradient_descent(a, deadline, rng, beta_init=None, beta_min=5e-4,
                      beta_decay=0.99, lr_init=0.05, no_imp_limit=120):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    c = _full_convolve(a)
    if beta_init is None:
        beta_init = 0.05 * max(1e-6, float(c.max()))
    beta = beta_init
    lr = lr_init
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = _full_convolve(a)
        sm_val, w = _softmax_grad(c, beta)
        g = _cross_correlate(w, a)
        grad = 4.0 * n * (g - sm_val * a)
        grad = grad - grad.mean()
        if not np.all(np.isfinite(grad)):
            a = best_a.copy()
            continue
        base = _score(a)
        a_new, sc_new, lr_new = _line_search(a, grad, max(lr, 1e-7), base, n_trials=8)
        if sc_new < best_score - 1e-12:
            a = a_new
            best_score = sc_new
            best_a = a.copy()
            no_imp = 0
            lr = min(1.0, max(lr_new * 1.2, lr * 0.9))
        else:
            no_imp += 1
            lr = max(lr_init * 0.1, lr * 0.6)
        if it % 10 == 0:
            beta = max(beta_min, beta * beta_decay)
        if no_imp > no_imp_limit:
            ns = 0.06
            a = np.maximum(best_a + ns * rng.normal(size=n), 1e-15)
            a = a / a.sum()
            beta = max(beta_min * 5.0, beta * 1.5)
            lr = lr_init
            no_imp = 0
    return best_a, best_score


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        if j0 <= j1:
            idx = np.arange(j0, j1 + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_fixed_point(a, deadline, max_iters=25):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        M = float(c.max())
        A = _build_conv_matrix(n, a)
        res = linprog(-np.ones(n), A_ub=A, b_ub=M * np.ones(2 * n - 1),
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            break
        b = np.maximum(res.x, 0.0)
        sb = float(b.sum())
        if sb < 1e-10:
            break
        b = b / sb
        base = _score(a)
        a_t, sc_t, _ = _line_search(a, b - a, max_lr=2.0, base_score=base, n_trials=6)
        a = a_t
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / max(1e-6, p1)) ** 2)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / max(1e-6, p3)) ** 2) + np.exp(-0.5 * ((t - p2) / max(1e-6, p3)) ** 2)
    elif ptype == 'ramp_sq':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t ** 2
    elif ptype == 'ramp_sqrt':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * np.sqrt(t)
    elif ptype == 'mod_cos':
        t = np.linspace(0, 1, n)
        env = 1.0 - p2 * abs(2 * t - 1)
        a = np.maximum(0.05, env) * (1.0 + p1 * np.cos(2 * np.pi * p3 * t))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
