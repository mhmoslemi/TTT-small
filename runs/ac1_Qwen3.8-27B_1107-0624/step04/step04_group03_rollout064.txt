def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):

        return np.inf
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):

    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    return w, cm

def _make_profile(n, kind, param1, param2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'linear':
        a = 1.0 + param1 * (t - 0.5)
    elif kind == 'quadratic':
        a = 1.0 + param1 * u + param2 * u ** 2
    elif kind == 'sine':
        a = 1.0 + param1 * np.cos(2 * np.pi * t)
    elif kind == 'sine2':
        a = (1.0 + param1 * np.cos(2 * np.pi * t)) ** 2
    elif kind == 'gaussian':
        a = np.exp(-u ** 2 / (2 * max(0.05, param1) ** 2))
    elif kind == 'cubic':
        a = 1.0 + param1 * u + param2 * u ** 3
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-10)
    return p / p.sum()
