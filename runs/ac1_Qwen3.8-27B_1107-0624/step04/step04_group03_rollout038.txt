def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_exact(a):

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)

def _conv_self(a):

def _subgradient_polish(a_init, deadline, lr_init=0.01, mom_init=0.7):

    n = len(a_init)
    a = a_init.copy()

        t += 1

        if not np.all(np.isfinite(grad)):
            break
        # Barzilai-Borwein step size
        if prev_grad is not None and prev_a is not None:
            s = a - prev_a
            y = grad - prev_grad
            ys = float(np.dot(y, s))
            yy = float(np.dot(y, y))
            if yy > 1e-30 and abs(ys) > 1e-30:
                lr_bb = ys / yy
                if lr_bb > 0 and np.isfinite(lr_bb):
                    lr = 0.5 * lr + 0.5 * min(lr_bb, 10.0 * lr)
        prev_grad = grad.copy()
        prev_a = a.copy()
        # Momentum step
        momentum = mom * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new > 1e-10:

        else:
            a_new = a
        a = a_new

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_exact(a)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _coordinate_search(a_init, deadline, n_perturb=20, n_trials=5000):

    temp = 0.001

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        t += 1
        # Perturb a random subset of coordinates
        k = rng.integers(max(3, n // 200), min(n // 10, 200))
        idx = rng.choice(n, size=k, replace=False)
        # Multiplicative perturbation
        factors = np.exp(rng.normal(0, 0.15, size=k))
        a_pert = a.copy()
        a_pert[idx] *= factors
        a_pert = np.maximum(a_pert, 0.0)
        s = float(a_pert.sum())
        if s < 1e-10:

        a_pert = a_pert / s

            a = a_pert
            if sc_pert < best_score:
                best_score = sc_pert
                best_a = a.copy()

        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _make_exponential_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinusoidal_profile(n, freq, phase=0.0, modulate=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + modulate * np.sin(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_modulated(n, center, width, amp):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse initialization -> L-BFGS continuation ->
    Nesterov -> exact subgradient with BB step size -> coordinate search
    -> self-similarity exploitation -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_exact(parent):.10f}")
    else:
        n = 4096
        parent = _make_exponential_profile(n, 2.0)
        print(f"No parent, using exponential profile n={n}")

    # --- Phase 0: Diverse initialization (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    sc_parent = _score_exact(parent)
    candidates.append((sc_parent, parent.copy(), "parent"))
    update_best(parent, sc_parent, "parent")

    diverse_profiles = [
        _make_exponential_profile(n, 1.0),
        _make_exponential_profile(n, 2.0),
        _make_exponential_profile(n, 3.0),
        _make_exponential_profile(n, -1.5),
        _make_sinusoidal_profile(n, 2, 0.0, 0.4),
        _make_sinusoidal_profile(n, 3, np.pi/4, 0.3),
        _make_sinusoidal_profile(n, 5, 0.0, 0.25),
        _make_quadratic_profile(n, 1.5, 0.3),
        _make_quadratic_profile(n, -1.0, 0.5),
        _make_gaussian_modulated(n, 0.3, 0.15, 0.5),
        _make_gaussian_modulated(n, 0.7, 0.15, 0.5),
    ]
    for idx, p in enumerate(diverse_profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_exact(p)
        candidates.append((sc, p, f"prof{idx}"))

    # Random smooth profiles
    for _ in range(5):
        if time.monotonic() > phase0_end:
            break
        z_rand = rng.normal(0, 0.5, size=n).cumsum()
        z_rand -= z_rand.mean()
        z_rand = np.tanh(z_rand)
        p_rand = np.exp(z_rand * 2.0)
        p_rand = p_rand / p_rand.sum()
        sc = _score_exact(p_rand)
        candidates.append((sc, p_rand, "rand"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")

    # --- Phase 1: L-BFGS on best candidate (200s) ---
    phase1_end = min(deadline, t_start + 230.0)
    if best_seq is not None:
        per_time = min(180.0, phase1_end - time.monotonic())
        if per_time > 10:
            p_best, sc_best = _lbfgs_run(best_seq, min(phase1_end, time.monotonic() + per_time),
                                         max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_p1")

    # Also try second-best if different size
    if time.monotonic() < phase1_end - 30:
        for sc, p, label in candidates:
            if time.monotonic() > phase1_end - 10:
                break
            if p is best_seq or sc > best_score * 1.01:
                continue
            per_time = min(40.0, phase1_end - time.monotonic() - 5)
            if per_time > 10 and len(p) == len(best_seq):
                p_best, sc_best = _lbfgs_run(p, min(phase1_end - 5, time.monotonic() + per_time),
                                              max_wall_time=per_time)
                update_best(p_best, sc_best, f"lbfgs_{label}")
                break

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov with aggressive continuation (400s) ---
    phase2_end = min(deadline, t_start + 630.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                      5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
        p_best, sc_best = _nesterov_optimize(best_seq, phase2_end, beta_fracs,
                                              lr_init=0.015, mom_init=0.9)
        update_best(p_best, sc_best, "nesterov_p2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Exact subgradient with BB step size (400s) ---
    phase3_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(300.0, phase3_end - time.monotonic())
        if per_time > 30:
            p_best, sc_best = _subgradient_polish(best_seq,
                                                   min(phase3_end, time.monotonic() + per_time),
                                                   lr_init=0.005, mom_init=0.7)
            update_best(p_best, sc_best, "subgrad_p3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Coordinate search (250s) ---
    phase4_end = min(deadline, t_start + 1280.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(200.0, phase4_end - time.monotonic())
        if per_time > 30:
            p_best, sc_best = _coordinate_search(best_seq,
                                                  min(phase4_end, time.monotonic() + per_time),
                                                  n_trials=3000)
            update_best(p_best, sc_best, "coord_p4")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Self-similarity + Nesterov restarts (200s) ---
    phase5_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        # Self-similarity
        if time.monotonic() < phase5_end - 60:
            p_best, sc_best = _self_similarity_improve(best_seq,
                                                        min(phase5_end - 60, time.monotonic() + 80))
            update_best(p_best, sc_best, "selfsim_p5")

        # Nesterov restarts
        if time.monotonic() < phase5_end - 20:
            n_best = len(best_seq)
            for i in range(3):
                if time.monotonic() > phase5_end - 10:
                    break
                noise_level = 0.03 + 0.02 * i
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                z = z + noise_level * rng.normal(size=n_best)
                z = np.clip(z, -60, 60)
                z = z - z.mean()
                p_start = _softmax_fn(z)
                per_dl = min(phase5_end - 5, time.monotonic() + 40)
                p_best, sc_best = _nesterov_optimize(p_start, per_dl,
                                                      [0.005, 0.002, 0.001, 0.0005, 0.0002],
                                                      lr_init=0.01, mom_init=0.85)
                update_best(p_best, sc_best, f"restart{i}_p5")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final polishing (remaining time) ---
    phase6_end = deadline
    if best_seq is not None and time.monotonic() < phase6_end:
        remaining = phase6_end - time.monotonic()
        # Final subgradient
        if remaining > 50:
            per_time = min(150.0, remaining - 30)
            p_best, sc_best = _subgradient_polish(best_seq,
                                                   min(phase6_end - 30, time.monotonic() + per_time),
                                                   lr_init=0.003, mom_init=0.8)
            update_best(p_best, sc_best, "final_subgrad")

        # Final Nesterov with tiny beta
        if time.monotonic() < phase6_end - 20:
            per_time = min(80.0, phase6_end - time.monotonic() - 10)
            if per_time > 15:
                p_best, sc_best = _nesterov_optimize(best_seq,
                                                      min(phase6_end - 10, time.monotonic() + per_time),
                                                      [1e-5, 5e-6, 2e-6, 1e-6],
                                                      lr_init=0.005, mom_init=0.95)
                update_best(p_best, sc_best, "final_nesterov")

        # Final coordinate search
        if time.monotonic() < phase6_end - 10:
            per_time = min(60.0, phase6_end - time.monotonic() - 5)
            if per_time > 15:
                p_best, sc_best = _coordinate_search(best_seq,
                                                      min(phase6_end - 5, time.monotonic() + per_time),
                                                      n_trials=1000)
                update_best(p_best, sc_best, "final_coord")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_exponential_profile(4096, 2.0)

        sc_final = _score_exact(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned inf/NaN, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_exponential_profile(4096, 2.0)
    print(f"FALLBACK: n=4096, score={_score_exact(fallback):.10f}")
    return [float(x) for x in fallback]
