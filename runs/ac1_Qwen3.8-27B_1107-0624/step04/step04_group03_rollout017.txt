    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

def _make_windows(n):
    """Generate diverse initial profiles."""
    profiles = []
    # Uniform
    profiles.append(np.ones(n) / n)
    # Linear
    for s in [0.5, 1.0, 1.5, 2.0]:
        t = np.linspace(0, 1, n)
        a = 1.0 + s * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    # Raised cosine
    t = np.linspace(0, 1, n)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        a = 0.5 * (1 + np.cos(np.pi * (2*t - 1) * alpha))
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    # Gaussian
    t = np.linspace(0, 1, n)
    for sigma in [0.15, 0.25, 0.35, 0.5]:
        a = np.exp(-((t - 0.5)**2) / (2 * sigma**2))
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    # V-shape (high at edges)
    for depth in [0.3, 0.5, 0.7]:
        t = np.linspace(0, 1, n)
        a = 1.0 - depth * np.abs(2*t - 1)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    # Inverted V (high in middle, low at edges) - same as cosine
    # Quartic
    t = np.linspace(0, 1, n)
    u = 2*t - 1
    a = 1.0 - 0.3*u**2 + 0.1*u**4
    a = np.maximum(a, 1e-10)
    profiles.append(a / a.sum())
    return profiles

def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _lbfgs_obj(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_sh = (c - cm) / beta
    ls = float(np.log(np.sum(np.exp(c_sh))))
    return 2.0 * n * (cm + beta * ls)

def _lbfgs_grad(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_sh = (c - cm) / beta
    w = np.exp(c_sh)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros_like(z)
    w = w / ws
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1 : 2*n-1]
    grad = 4.0 * n * beta * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _run_lbfgs(a_init, deadline, max_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    mc0 = float(c0.max())
    
    if max_time is None:
        max_time = max(1.0, deadline - time.monotonic())
    
    betas = [b * mc0 for b in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]]
    per_level = max_time / len(betas)
    
    for beta in betas:
        if time.monotonic() >= deadline:
            break
        dl = min(deadline, time.monotonic() + per_level)
        try:
            res = minimize(_lbfgs_obj, z, args=(beta,),
                          method='L-BFGS-B', jac=_lbfgs_grad,
                          options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                z = res.x
                a_new = _softmax_fn(z)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_sc

def _lp_direction(a, time_limit=30.0):
    """Solve LP to find direction b maximizing sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    # Build constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j < n
    # A is (2n-1) x n
    # We can build it as a Toeplitz-like matrix
    # Row k: a[k], a[k-1], ..., a[k-n+1] (padded with 0s)
    
    # Using scipy's toeplitz: toeplitz(col, row)
    # col = first column of A = [a[0], 0, 0, ..., 0, a[1], a[2], ..., a[n-1], 0, ...]
    # Wait, A[k, j] = a[k-j] if 0 <= k-j < n
    # A[0, 0] = a[0], A[0, 1] = a[-1] = 0, ...
    # A[1, 0] = a[1], A[1, 1] = a[0], A[1, 2] = a[-1] = 0
    # A[k, j] = a[k-j] for j <= k, and 0 for j > k (when k-j < 0)
    # Also 0 for k-j >= n
    
    # First column (j=0): A[k, 0] = a[k] if 0 <= k < n, else 0
    # So first_col = [a[0], a[1], ..., a[n-1], 0, ..., 0] (length 2n-1)
    
    # First row (k=0): A[0, j] = a[0-j] = a[-j] = 0 for j >= 1
    # So first_row = [a[0], 0, 0, ..., 0] (length n)
    
    # Actually, toeplitz(col, row) gives a matrix where col is the first column
    # and row is the first row. Let me verify:
    # M[i,j] = col[i+j] if i+j < len(col), else row[len(row)-1-i-j] ... 
    # This is confusing. Let me just build it directly.
    
    # Actually, the matrix A has the structure:
    # Row 0: [a[0], 0, 0, ..., 0]
    # Row 1: [a[1], a[0], 0, ..., 0]
    # Row 2: [a[2], a[1], a[0], ..., 0]
    # ...
    # Row n-1: [a[n-1], a[n-2], ..., a[0]]
    # Row n: [0, a[n-1], a[n-2], ..., a[0]]
    # ...
    # Row 2n-2: [0, 0, ..., a[n-1]]
    
    # This is a lower-triangular Toeplitz matrix (first part) followed by an upper part.
    # It's the convolution matrix!
    
    # I'll use scipy.linalg.toeplitz or build it manually
    # For efficiency with the LP, I'll use the sparse structure
    
    from scipy.sparse import csc_matrix
    from scipy.sparse.linalg import aslinearoperator
    
    # Build sparse matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            if 0 <= k-j < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[k-j])
    
    A_sparse = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: minimize -sum(b) s.t. A@b <= M, b >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    start_time = time.monotonic()
    result = linprog(c_obj, A_ub=A_sparse, b_ub=b_ub, bounds=(0, None), method='highs')
    elapsed = time.monotonic() - start_time
    
    if result.success and result.x is not None:
        b = np.maximum(result.x, 0)
        if b.sum() > 1e-15:
            return b / b.sum(), M, elapsed
        return None, M, elapsed
    return None, M, elapsed

def _lp_refine(a, deadline, max_lp_calls=200):
    """Iteratively apply LP direction to reduce max convolution."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    c = _full_convolve(a, a)
    M = float(c.max())
    
    call_count = 0
    stale_count = 0
    
    while time.monotonic() < deadline and call_count < max_lp_calls:
        if time.monotonic() > deadline:
            break
        
        b, M_curr, lp_time = _lp_direction(a, time_limit=30.0)
        call_count += 1
        
        if b is None:
            stale_count += 1
            if stale_count > 5:
                break
            continue
        
        # Try different step sizes
        best_step_a = None
        best_step_sc = _score_norm(a)
        
        for t in [0.1, 0.05, 0.02, 0.01]:
            a_trial = (1-t) * a + t * b
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = _score_norm(a_trial)
            if sc < best_step_sc:
                best_step_sc = sc
                best_step_a = a_trial
        
        if best_step_a is not None:
            a = best_step_a
            if a is not best_a:
                best_a = a.copy()
                best_sc = best_step_sc
            stale_count = 0
        else:
            stale_count += 1
            if stale_count > 10:
                break
    
    return best_a, best_sc

def _nesterov_projected(a_init, deadline, lr_init=0.01, mom_init=0.9, 
                         beta_init_frac=0.01, n_beta_levels=10):
    """Nesterov accelerated projected gradient on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    # Use extrapolation: y = a + mom*(a - a_prev)
    a_prev = a.copy()
    y = a.copy()
    
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    c = _full_convolve(a, a)
    max_c = float(c.max())
    
    beta = beta_init_frac * max_c
    lr = lr_init
    mom = mom_init
    
    total_iters = 0
    beta_schedule = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for beta_frac in beta_schedule:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)
        
        while time.monotonic() < deadline:
            total_iters += 1
            
            # Compute gradient at y
            a_y = y / y.sum() if y.sum() > 1e-15 else a
            c_y = _full_convolve(a_y, a_y)
            cm = float(c_y.max())
            c_sh = (c_y - cm) / beta
            w = np.exp(c_sh)
            ws = w.sum()
            if ws < 1e-300:
                break
            w = w / ws
            W = float(np.dot(w, c_y))
            
            # Gradient: grad[j] = 4*n*beta*sum_m w[j+m]*a_y[m] - 4*n*beta*W*a_y[j]
            # Wait, for the projected method, we compute gradient of the smoothed objective
            # f(a) = 2*n*(max_c + beta*log(sum exp((c-cm)/beta)))
            # df/da[j] = 4*n*beta * sum_{m=0}^{n-1} w[j+m] * a[m] - 4*n*beta*W ... 
            # Hmm, I need to be more careful.
            
            # Actually, for the unconstrained gradient (before projection):
            # df/da[j] = 2*n*beta * sum_k w[k] * dc[k]/da[j]
            # = 2*n*beta * sum_k w[k] * 2*a[k-j] * [0<=k-j<n]
            # = 4*n*beta * sum_{m=0}^{n-1} w[j+m] * a[m]
            
            # But we also need to account for the constraint sum(a) = 1.
            # The gradient on the tangent space is: grad - (mean(grad)) * 1
            # Or more precisely, for the projection method, we just use the full gradient
            # and the projection will handle the constraint.
            
            # Compute sum_{m=0}^{n-1} w[j+m] * a[m] for j = 0, ..., n-1
            # This is the cross-correlation of w and a
            a_rev = a_y[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n-1 : 2*n-1]  # h[j] = sum_m w[j+m]*a[m]
            
            grad = 4.0 * n * beta * h
            
            # Nesterov step
            a_new = y - lr * grad
            # Project onto simplex
            a_new = _project_simplex(a_new)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / s
            
            # Update momentum
            y = a_new + mom * (a_new - a)
            y = _project_simplex(y)
            
            a_prev = a
            a = a_new
            
            if total_iters % 50 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                        if total_iters % 200 == 0:
                    mom = min(0.999, mom + 0.001)
                
                # Update max_c for beta
                c_new = _full_convolve(a, a)
                max_c = float(c_new.max())
            
            if total_iters % 1000 == 0:
                lr = max(0.001, lr * 0.8)
        
        # Update max_c at end of each beta level
        c_new = _full_convolve(a, a)
        max_c = float(c_new.max())
        lr = max(0.001, lr * 0.7)
    
    # Final check
    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc
