"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact LP solves for direction finding, combined with sub-gradient polish.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _build_lp_matrix(f, n):
    """Build sparse constraint matrix A for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    k ranges from 0 to 2n-2, j ranges from 0 to n-1.
    """
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def _solve_lp_direction(f, deadline):
    """Solve the LP to find the optimal direction g.
    Returns g (normalized to sum=1) or None if LP fails.
    """
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    
    if M < 1e-15:
        return f.copy()
    
    A = _build_lp_matrix(f, n)
    b = M * np.ones(2 * n - 1)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    try:
        if time.monotonic() > deadline:
            return None
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, 
                        method='highs', options={'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            s = float(g.sum())
            if s > 1e-10:
                return g / s
        return None
    except Exception as e:
        return None

def _make_gaussian_profile(n, sigma_frac):
    center = (n - 1) / 2.0
    t = np.arange(n) - center
    a = np.exp(-0.5 * (t / (sigma_frac * n)) ** 2)
    return a / a.sum()

    a = 1.0 + beta * np.cos(2.0 * np.pi * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _subgrad_step(a, beta, top_k, step_size):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        z = (c_top - c_top.max()) / beta
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[idx[0]] = 1.0
        else:
            w = np.zeros(len(c))
            w[idx] = e / s
    else:
        z = (c - cm) / beta
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    active_idx = np.where(w > 1e-10)[0]
    h = np.zeros(n)
    for ki in active_idx:
        k = int(ki)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            h[j_start:j_end + 1] += w[ki] * a[k - j_end:k - j_start + 1][::-1]
    z_log = np.log(np.maximum(a, 1e-15))
    z_log = z_log - z_log.mean()
    grad_log = h / np.maximum(a, 1e-10)
    z_log = z_log - step_size * grad_log
    z_log = np.clip(z_log, -60, 60)
    z_log = z_log - z_log.mean()
    a_new = _softmax_fn(z_log)
    return a_new, h

    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a_init)
    t_blend = 0.3
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if it % lp_every == 0:
            g = _lp_direction(a, deadline)
            if g is None:
                g = a.copy()
            a_new = (1.0 - t_blend) * a + t_blend * g
            s_new = float(a_new.sum())
            if s_new > 1e-10:
                a_new = a_new / s_new
                a_new = np.maximum(a_new, 0.0)
                s_new2 = float(a_new.sum())
                if s_new2 > 1e-10:
                    a_new = a_new / s_new2
                    sc = _score_norm_fn(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                    a = a_new
        else:
            beta = 0.01 * float(_full_convolve(a, a).max()) + 1e-12
            a_candidate, h = _subgrad_step(a, beta, 15, 0.02)
            sc = _score_norm_fn(a_candidate)
            if np.isfinite(sc) and sc < _score_norm_fn(a):
                a = a_candidate
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        if it % 20 == 0:
            t_blend = max(0.1, t_blend * 0.95)
    return best_a, best_score


def _kkt_refine(a_init, deadline, max_iter=3000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a_init)
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        if cm < 1e-15:
            break
        threshold = cm * (0.999 - it * 1e-6)
        threshold = max(threshold, cm * 0.95)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        h = np.zeros(n)
        for ki in active:
            k = int(ki)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                h[j_start:j_end + 1] += a[k - j_end:k - j_start + 1][::-1]
        h /= len(active)
        W = float(np.dot(h, a))
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        active_spread = float(c[active].max() - c[active].min()) + 1e-15
        step = 0.005 * active_spread / (cm + 1e-15)
        step = min(step, 0.03)
        z = z - step * (h - W) / (np.abs(h - W).max() + 1e-15)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_new = _softmax_fn(z)
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.5
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - step * (h - W) / (np.abs(h - W).max() + 1e-15)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
    return best_a, best_score

    def update_best(p, sc, label):
        nonlocal best_seq, best_score
        if p is not None and len(p) > 0 and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:

            best_score = float(sc)

    print("=== Phase 0: Initial candidates ===")
