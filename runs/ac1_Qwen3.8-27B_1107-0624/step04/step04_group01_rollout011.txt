LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find descent directions, combined with 
Adam optimization and multi-scale approaches.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(a):
    """Build the matrix A such that A @ b = conv(a, b)."""
    n = len(a)
    # conv(a, b)[k] = sum_m A[k, m] * b[m]
    # From earlier analysis:
    # For output index k (0 to 2n-2), lag = k - (n-1)
    # A[k, m] = a[m - lag] if 0 <= m - lag < n, else 0
    
    # Build as sparse matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m_start = max(0, lag)
        m_end = min(n, n + lag)
        for m in range(m_start, m_end):
            val = a[m - lag]
            if val > 1e-15:  # Skip near-zero entries
                rows.append(k)
                cols.append(m)
                vals.append(val)
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def _lp_solve(a, M):

    Solve: max sum(b) s.t. conv(a,b) <= M for all lags, b >= 0.
    Returns optimal b or None.

    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n

    except Exception:

def _lp_active_set(a, M, max_constraints=50):
    """
    Solve LP using only the most active constraints.
    """
    n = len(a)
    c = _full_convolve(a, a)
    
    # Find the most active constraints (closest to M)
    # Sort by c value descending
    order = np.argsort(c)[::-1]
    active_indices = order[:max_constraints]
    
    # Build reduced constraint matrix
    A_full = _build_conv_matrix(a)
    A_reduced = A_full[active_indices, :]
    
    c_obj = -np.ones(n)
    b_ub_reduced = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_reduced, b_ub=b_ub_reduced, bounds=bounds, method='highs')
        if result.success:
            b_sol = result.x
            # Verify it satisfies all constraints
            c_check = _full_convolve(a, b_sol)
            if c_check.max() <= M * 1.001:
                return b_sol
            # If not, fall back to full LP
            return None
    except Exception:
        pass
    return None

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1

    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]
