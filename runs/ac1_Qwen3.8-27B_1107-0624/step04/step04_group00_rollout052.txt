Multi-scale continuation optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy:
1. Multi-resolution: optimize at n=128, upsample to 256, 512, 1024, 2048, 4096.
2. At each scale, use L-BFGS with adaptive log-sum-exp continuation (decreasing beta).
3. Final phase: adaptive gradient descent (Adam) + peak-equalization refinement.
Key insight: optimal sequence is asymmetric (ramp-like), balancing b[0] (energy)
against off-center peaks. Multi-scale approach provides warm starts that help
L-BFGS converge to better local minima.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # h_i = sum_k w[k] * a[k-i] for valid k-i in [0, n-1]
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    # grad_z = a * (2*h - 2*dot(a,h))
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z

    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(20, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.05 * cm:
        return max(1e-10, 0.05 * cm)
    elif gap > 0.5 * cm:
        return max(1e-10, 0.01 * cm)
    else:
        return max(1e-10, 0.3 * gap)


def _run_lbfgs_continuation(a_init, deadline, rng, n_betas=15):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Adaptive continuation: start with large beta, decrease
    beta_start = 0.3 * max_c0
    beta_end = 1e-6 * max_c0

    # Create logarithmic schedule
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 180.0)
        level_deadline = time.monotonic() + time_budget

        # More iterations for smaller beta
        maxiter = 500 if beta > 0.1 * max_c0 else (2000 if beta > 0.01 * max_c0 else 5000)
        maxfun = maxiter * 5

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                z_new = np.clip(z_new, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_adam_gd(a_init, deadline, rng, lr_init=0.1, max_iter=100000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)
    best_beta = beta

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)

        # Adaptive beta
        if it % 100 == 0:
            beta = _adaptive_beta(c)
            if beta < best_beta * 0.5:
                best_beta = beta

        cm = float(c.max())
        if cm < 1e-30:
            break

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Decay learning rate
        if t % 2000 == 0:
            lr = max(0.01, lr * 0.9)

        # Kick if stuck
        if t % 10000 == 0:
            z = z + 0.03 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _peak_equalize(a, deadline, rng, max_rounds=200):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find top peaks
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        peak_vals = c[peak_indices]

        # For each peak, find top contributors and try to reduce them
        improved = False
        for pk in peak_indices[:3]:
            # Contributions: a[i] * a[pk-i] for valid i
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            # Sort by contribution
            order = np.argsort(prod)[::-1][:8]
            for idx_pos in order[:4]:
                idx = valid_i[idx_pos]
                # Try reducing this index
                for delta in [0.005, 0.01, 0.02, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try adding to small values
            small_idx = np.argsort(a_cur)[:5]
            for idx in small_idx:
                for delta in [0.005, 0.01, 0.02]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and rnd > 20:
            break

    return best_a, best_score


def _make_ramp(n, slope=1.5, curvature=0.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t + curvature * (t - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    # Various ramps
    for s in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(_make_ramp(n, s))
    # Quadratic shapes
    for s, c in [(1.5, 0.5), (2.0, -0.3), (1.0, 1.0), (2.5, 0.3)]:
        a = 1.0 + s * t + c * (t - 0.5) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        # Default ramp
        a0 = _make_ramp(256, 1.5)
        update_best(a0, _score_norm_fn(a0), "default_ramp")
        best_seq = a0
        best_score = _score_norm_fn(a0)

    # --- Phase 1: Multi-scale optimization from scratch (60% of time) ---
    phase1_end = t_start + (budget_s - 25.0) * 0.60
    print(f"\nPhase 1: Multi-scale optimization (deadline t={phase1_end - t_start:.0f}s)")

    # Start from best ramp profile at n=128
    n_start = 128
    a_cur = None
    sc_cur = np.inf

    # Try multiple initial profiles at n=128 and pick the best after quick optimization
    profiles_128 = _make_profiles(n_start)
    for idx, prof in enumerate(profiles_128[:4]):
        sc = _score_norm_fn(prof)
        if sc < sc_cur:
            sc_cur = sc
            a_cur = prof.copy()

    print(f"  Best initial profile: score_norm={sc_cur:.10f}")

    # Quick L-BFGS at n=128
    quick_deadline = min(phase1_end, t_start + 20.0)
    a_cur, sc_cur = _run_lbfgs_continuation(a_cur, quick_deadline, rng, n_betas=8)
    print(f"  After n=128 optimization: score_norm={sc_cur:.10f}")

    # Multi-scale up
    scales = [256, 512, 1024, 2048, 4096]
    time_alloc = {256: 25, 512: 40, 1024: 70, 2048: 150, 4096: 350}

    for target_n in scales:
        if time.monotonic() > phase1_end - 10:
            break

        # Upsample
        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm_fn(a_target)
        print(f"  Upsampled to n={target_n}: score_norm={sc_target:.10f}")

        # Optimize
        alloc = time_alloc.get(target_n, 60)
        opt_deadline = min(phase1_end - 5, time.monotonic() + alloc)
        a_opt, sc_opt = _run_lbfgs_continuation(a_target, opt_deadline, rng, n_betas=12)
        print(f"  After n={target_n} optimization: score_norm={sc_opt:.10f}")

        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    # Track multi-scale result
    if sc_cur < best_score:
        update_best(a_cur, sc_cur, "multi_scale")
    else:
        # If parent was better, keep it
        a_cur = best_seq.copy()
        sc_cur = best_score

    print(f"\n  Phase 1 complete: best_score={best_score:.10f}")

    # --- Phase 2: Continue from best with more L-BFGS (20% of time) ---
    phase2_end = t_start + (budget_s - 25.0) * 0.80
    print(f"\nPhase 2: L-BFGS continuation from best (deadline t={phase2_end - t_start:.0f}s)")

    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        n_best = len(best_seq)

        # Run L-BFGS with fine continuation from best
        opt_deadline = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()))
        a_p2, sc_p2 = _run_lbfgs_continuation(best_seq, opt_deadline, rng, n_betas=20)
        update_best(a_p2, sc_p2, "phase2_lbfgs")

        # Also try a perturbed restart
        if time.monotonic() < phase2_end - 30:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.1 * rng.normal(size=n_best)
            z = z + noise
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_pert = _softmax_fn(z)
            opt_dl2 = min(phase2_end - 5, time.monotonic() + 60.0)
            a_p2b, sc_p2b = _run_lbfgs_continuation(a_pert, opt_dl2, rng, n_betas=10)
            update_best(a_p2b, sc_p2b, "phase2_restart")

    print(f"\n  Phase 2 complete: best_score={best_score:.10f}")

    # --- Phase 3: Adam GD + Peak surgery (15% of time) ---
    phase3_end = t_start + (budget_s - 25.0) * 0.95
    print(f"\nPhase 3: Adam GD + Peak surgery (deadline t={phase3_end - t_start:.0f}s)")

    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        # Adam GD
        gd_deadline = min(phase3_end - 30, time.monotonic() + (phase3_end - time.monotonic()) * 0.5)
        a_gd, sc_gd = _run_adam_gd(best_seq, gd_deadline, rng, lr_init=0.05)
        update_best(a_gd, sc_gd, "adam_gd")

        # Peak surgery
        if time.monotonic() < phase3_end - 10:
            ps_deadline = min(phase3_end - 5, time.monotonic() + 40.0)
            a_ps, sc_ps = _peak_equalize(best_seq, ps_deadline, rng)
            update_best(a_ps, sc_ps, "peak_surgery")

        # Another Adam round with smaller lr
        if time.monotonic() < phase3_end - 5:
            gd_dl2 = min(phase3_end - 3, time.monotonic() + 30.0)
            a_gd2, sc_gd2 = _run_adam_gd(best_seq, gd_dl2, rng, lr_init=0.01)
            update_best(a_gd2, sc_gd2, "adam_gd_fine")

    print(f"\n  Phase 3 complete: best_score={best_score:.10f}")

    # --- Phase 4: Final polish with tiny beta L-BFGS ---
    print(f"\nPhase 4: Final polish")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        # Very small beta L-BFGS
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        beta_tiny = 1e-5 * max_c_final

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny, n_final),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 20000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 50000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    print(f"  Tiny-beta L-BFGS: {best_score:.10f}")
        except Exception as ex:
            print(f"  Tiny-beta L-BFGS failed: {ex}")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
