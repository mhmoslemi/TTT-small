```python
def _full_convolve(a, b):
    """Full linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:

    fa = np.fft.rfft(a, N)
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Score assuming sum(a)=1."""
    n = len(a)

def _softmax_fn(z):
```

```python
"""
Multi-scale LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses coarse-to-fine grid optimization, L-BFGS with adaptive logsumexp continuation,
LP-based direction finding at intermediate scales, and multi-restart local search.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, ptype, slope=1.5, a1=1.0, a2=0.3, a3=-0.3, w=0.3, p=2.0, floor=0.1, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if ptype == 'linear':
        a = 1.0 + slope * u
    elif ptype == 'quad':
        a = 1.0 + a1 * u + a2 * u**2
    elif ptype == 'cubic':
        a = 1.0 + a1 * u + a3 * u**3
    elif ptype == 'gauss':
        a = np.exp(-u**2 / (2.0 * w**2))
    elif ptype == 'power':
        a = (1.0 - np.abs(u))**p
    elif ptype == 'edge':
        a = np.abs(u)**p + floor
    elif ptype == 'cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * u * alpha))
    elif ptype == 'mono':
        a = t**p + 0.01
    elif ptype == 'double_edge':
        a = np.abs(u)**p + 0.05 * (1.0 - u**2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_to(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = len(p)
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _smoothed_obj(z, n, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shift = (c - cm) / beta
    e = np.exp(c_shift)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _smoothed_grad(z, n, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shift = (c - cm) / beta
    e = np.exp(c_shift)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    g = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(z)
    return g


def _topk_smoothed_obj(z, n, k_top, alpha_ls):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    k_actual = min(k_top, len(c))
    top_vals = np.sort(c)[::-1][:k_actual]
    topk_avg = float(top_vals.mean())
    c_shift = (c - cm) / alpha_ls
    e = np.exp(c_shift)
    s = e.sum()
    if s < 1e-300:
        ls = cm
    else:
        ls = cm + alpha_ls * np.log(s)
    combined = 0.5 * topk_avg + 0.5 * ls
    return 2.0 * n * combined


def _lbfgs_continuation(a_init, deadline, rng, beta_scale=1.0, n_levels=8):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Adaptive beta schedule: start coarse, end fine
    ratios = np.linspace(0.15, 0.0003, n_levels)
    betas = ratios * max_c0 * beta_scale

    for li, beta in enumerate(betas):
        if time.monotonic() > deadline - 3.0:
            break

        remaining = deadline - time.monotonic()
        frac_remaining = (n_levels - li) / n_levels
        level_time = max(10.0, remaining * 0.5)
        level_deadline = min(deadline - 2.0, time.monotonic() + level_time)

        maxfun = 5000 if n <= 2048 else 3000
        maxiter = 1500 if n <= 2048 else 800

        try:
            result = minimize(
                _smoothed_obj, z, args=(n, beta),
                jac=_smoothed_grad, args=(n, beta),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x.copy()
                a_new = _softmax_fn(z)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _build_conv_matrix(a):
    n = len(a)
    i_idx = np.arange(n)
    k_idx = np.arange(n)
    rows_arr = (i_idx[:, None] + k_idx[None, :]).ravel()
    cols_arr = np.repeat(i_idx, n)
    vals_arr = np.tile(a, n)
    mask = rows_arr < (2 * n - 1)
    from scipy.sparse import csr_matrix
    return csr_matrix((vals_arr[mask], (rows_arr[mask], cols_arr[mask])), shape=(2 * n - 1, n))


def _lp_refine(a_init, deadline, max_iters=8):

    n = len(a_init)
    if n > 2048:
        return a_init, _score_norm(a_init)

    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        A_mat = _build_conv_matrix(a)

        b_var = cp.Variable(n, nonneg=True)
        obj = cp.Maximize(cp.sum(b_var))
        constraints = [A_mat @ b_var <= M]
        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                break

        if b_var.value is None:
            break

        b = np.asarray(b_var.value, dtype=np.float64)
        b = np.maximum(b, 0)
        s_b = b.sum()
        if s_b < 1e-15:
            break
        b = b / s_b

        best_t = 0.0
        best_sc = best_score
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1 - t_frac) * a + t_frac * b
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t_frac

        if best_t > 0:
            a = (1 - best_t) * a + best_t * b
            a = a / a.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
        else:
            break

        print(f"    LP iter {it}: M={M:.8f}, t={best_t:.2f}, score={best_score:.10f}")

    return best_a, best_score


def _local_search(a_init, deadline, rng, n_trials=8000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 2.0:
            break

        block_size = int(rng.integers(3, max(4, min(50, n // 6))))
        start = int(rng.integers(0, n - block_size))

        a_trial = a.copy()
        pert = rng.uniform(0.82, 1.18, size=block_size)
        a_trial[start:start + block_size] *= pert
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < 0.003 and sc < best_score * 1.003:
            a = a_trial.copy()

    return best_a, best_score


def _adam_topk(a_init, deadline, rng, lr=0.012, k_top=5):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    for it in range(50000):
        if time.monotonic() > deadline - 2.0:
            break

        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        k_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[::-1][:k_actual]

        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a_curr[k_val - hi:k_val - lo + 1][::-1]
        grad /= k_actual
        grad *= 4.0 * n

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 20 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if t % 1000 == 0:
            lr = max(0.003, lr * 0.95)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_p:.10f}")
        best_seq = parent.copy()
        best_score = sc_p
    else:
        best_seq = _make_profile(2048, 'linear', slope=1.5)
        best_score = _score_norm(best_seq)
        print(f"Init: n=2048, score={best_score:.10f}")

    # === Phase 1: Multi-scale coarse-to-fine (350s) ===
    phase1_end = min(deadline, t_start + 350.0)
    print(f"\n=== Phase 1: Multi-scale (budget {phase1_end - t_start:.0f}s) ===")

    scales = [64, 128, 256, 512, 1024, 2048, 4096]
    current = _interp_to(best_seq, scales[0])

    for i, n_s in enumerate(scales):
        if time.monotonic() > phase1_end - 10.0:
            break

        current = _interp_to(current, n_s)
        remaining = phase1_end - 10.0 - time.monotonic()
        n_remaining = len(scales) - i
        time_per = max(15.0, remaining / n_remaining)
        scale_dl = min(phase1_end - 5.0, time.monotonic() + time_per)

        a_best, sc_best = _lbfgs_continuation(current, scale_dl, rng, beta_scale=1.0, n_levels=6)
        current = a_best
        if sc_best < _score_norm(_interp_to(best_seq, n_s)):
            best_seq = a_best
            best_score = sc_best
            print(f"  Scale n={n_s}: score={sc_best:.10f} (improved)")
        else:
            print(f"  Scale n={n_s}: score={sc_best:.10f}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Fine L-BFGS at n=4096 (350s) ===
    phase2_end = min(deadline, t_start + 700.0)
    print(f"\n=== Phase 2: Fine optimization (budget {phase2_end - t_start:.0f}s) ===")

    if len(best_seq) != 4096:
        best_seq = _interp_to(best_seq, 4096)
        best_score = _score_norm(best_seq)

    a_best, sc_best = _lbfgs_continuation(best_seq, phase2_end - 5.0, rng, beta_scale=0.5, n_levels=10)
    if sc_best < best_score:
        best_seq = a_best
        best_score = sc_best
    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: LP refinement at intermediate scale (250s) ===
    phase3_end = min(deadline, t_start + 950.0)
    print(f"\n=== Phase 3: LP refinement (budget {phase3_end - t_start:.0f}s) ===")

    if time.monotonic() < phase3_end - 30:
        n_lp = 1024
        a_lp = _interp_to(best_seq, n_lp)
        a_lp_best, sc_lp = _lp_refine(a_lp, phase3_end - 20.0, max_iters=6)

        if sc_lp < _score_norm(a_lp) - 1e-8:
            a_up = _interp_to(a_lp_best, 4096)
            sc_up = _score_norm(a_up)
            print(f"  LP improved n={n_lp}: {sc_lp:.10f} -> scaled to n=4096: {sc_up:.10f}")
            if sc_up < best_score:
                best_seq = a_up
                best_score = sc_up
            # Further optimize the improved version
            if time.monotonic() < phase3_end - 10:
                a_best2, sc_best2 = _lbfgs_continuation(best_seq, phase3_end - 5.0, rng, beta_scale=0.3, n_levels=6)
                if sc_best2 < best_score:
                    best_seq = a_best2
                    best_score = sc_best2
                    print(f"  After post-LP LBFGS: {best_score:.10f}")

    print(f"Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Top-K Adam (200s) ===
    phase4_end = min(deadline, t_start + 1150.0)
    print(f"\n=== Phase 4: Adam top-K (budget {phase4_end - t_start:.0f}s) ===")

    if time.monotonic() < phase4_end - 20:
        a_best, sc_best = _adam_topk(best_seq, phase4_end - 5.0, rng, lr=0.01, k_top=3)
        if sc_best < best_score:
            best_seq = a_best
            best_score = sc_best
            print(f"  Adam top3: {sc_best:.10f}")

        if time.monotonic() < phase4_end - 30:
            a_best2, sc_best2 = _adam_topk(best_seq, phase4_end - 5.0, rng, lr=0.008, k_top=7)
            if sc_best2 < best_score:
                best_seq = a_best2
                best_score = sc_best2
                print(f"  Adam top7: {sc_best2:.10f}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Multi-restart (250s) ===
    phase5_end = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 5: Multi-restart (budget {phase5_end - t_start:.0f}s) ===")

    if time.monotonic() < phase5_end - 20:
        n_restarts = 5
        per_restart = max(20.0, (phase5_end - time.monotonic()) / n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break

            noise = 0.015 + 0.015 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()

            if i % 3 == 0:
                z = z + noise * rng.normal(size=len(best_seq))
            elif i % 3 == 1:
                blk = len(best_seq) // 8
                for b_i in range(8):
                    z[b_i * blk:(b_i + 1) * blk] += noise * rng.normal()
            else:
                freq = int(rng.integers(2, 12))
                t_arr = np.linspace(0, 2 * np.pi * freq, len(best_seq))
                z = z + noise * np.sin(t_arr + rng.uniform(0, 2 * np.pi))

            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            a_best, sc_best = _lbfgs_continuation(p_start, dl, rng, beta_scale=0.6, n_levels=7)
            if sc_best < best_score:
                best_seq = a_best
                best_score = sc_best
                print(f"  Restart {i}: {sc_best:.10f}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Local search (150s) ===
    phase6_end = min(deadline, t_start + 1550.0)
    print(f"\n=== Phase 6: Local search (budget {phase6_end - t_start:.0f}s) ===")

    if time.monotonic() < phase6_end - 10:
        a_best, sc_best = _local_search(best_seq, phase6_end - 5.0, rng, n_trials=15000)
        if sc_best < best_score:
            best_seq = a_best
            best_score = sc_best
            print(f"  Local search: {sc_best:.10f}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final fine-tuning ===
    print(f"\n=== Phase 7: Final fine-tuning ===")
    remaining = max(10.0, deadline - time.monotonic())

    if remaining > 30:
        dl = min(deadline - 10, time.monotonic() + remaining * 0.5)
        a_best, sc_best = _lbfgs_continuation(best_seq, dl, rng, beta_scale=0.3, n_levels=6)
        if sc_best < best_score:
            best_seq = a_best
            best_score = sc_best
            print(f"  Final LBFGS: {sc_best:.10f}")

    if time.monotonic() < deadline - 8:
        a_best, sc_best = _local_search(best_seq, deadline - 5.0, rng, n_trials=5000)
        if sc_best < best_score:
            best_seq = a_best
            best_score = sc_best
            print(f"  Final local: {sc_best:.10f}")

    # Final output
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile(4096, 'linear', slope=1.5)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}, norm_score={_score_norm(final_seq):.10f}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval: {ex}")

        return [float(x) for x in final_seq]

    fb = _make_profile(4096, 'linear', slope=1.5)
    print(f"FALLBACK: score={_score(fb):.10f}")
    return [float(x) for x in fb]
```
