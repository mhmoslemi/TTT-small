"""
Optimized search for low-peak-autocorrelation non-negative sequences.
Uses softmax parameterization with log-sum-exp smoothing, L-BFGS-B with
continuation, natural gradient with adaptive beta, and direct subgradient
polishing. Multi-restart with structured perturbations.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _subgradient_at_max(a, c):
    """Compute subgradient of max(conv(a,a)) w.r.t. a at the argmax."""
    n = len(a)
    k_max = int(np.argmax(c))
    # dc_k/da_i = a_{i-k} + a_{i+k} (with zero-padding for out-of-bounds)
    # We need: grad_i = a_{i-k_max} + a_{i+k_max} for valid indices
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        j1 = i - k_max
        if 0 <= j1 < n:
            val += a[j1]
        j2 = i + k_max
        if 0 <= j2 < n:
            val += a[j2]
        grad[i] = val
    return grad


def _subgradient_at_max_fast(a, c):
    """Fast version using FFT-based approach."""
    n = len(a)
    k_max = int(np.argmax(c))
    # We want grad_i = a[(i-k_max) mod n] + a[(i+k_max) mod n] for circular
    # But we need linear convolution, so:
    # dc_k/da_i = sum over valid j of a_j * a_{j+k} where the derivative w.r.t. a_i
    # gives: a_{i-k} (from a_i * a_{i-k} term) + a_{i+k} (from a_{i+k} * a_i term)
    # Valid when 0 <= i-k < n and 0 <= i+k < n
    grad = np.zeros(n)
    if 0 <= k_max < n:
        # a_{i-k_max} is valid for i in [k_max, n-1]
        grad[k_max:] += a[:n-k_max]
        # a_{i+k_max} is valid for i in [0, n-1-k_max]
        grad[:n-k_max] += a[k_max:]
    return grad


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):
    return _gradient_smooth(z, beta)


def _run_lbfgs_continuation(a_init, deadline, betas=None):
    """Run L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if betas is None:
        c0 = _full_convolve(a_init, a_init)
        max_c0 = float(c0.max())
        betas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        betas = [b * max_c0 for b in betas]

    remaining = deadline - time.monotonic()
    per_level = max(1.0, remaining / len(betas))

    for idx, beta in enumerate(betas):
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass

    return best_a, best_score


def _adaptive_nesterov(a_init, deadline, lr_init=0.01, mom_init=0.9):

    n = len(a_init)

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0

    # Start with relatively large beta, decrease adaptively
    beta = max_c * 0.01
    beta_min = max_c * 1e-8

    while time.monotonic() < deadline:
        t += 1

        a_y = _softmax_fn(y)
        c_y = _full_convolve(a_y, a_y)
        w, cm_y = _compute_weights(c_y, beta)
        
        # Adaptive beta: decrease if effective active set is small
        eff_size = 1.0 / float(np.dot(w, w))
        if eff_size < 5:

        elif eff_size > 20:
            beta = min(max_c * 0.1, beta * 1.1)

        W = float(np.dot(w, c_y))
        a_y_rev = a_y[::-1]
        w_conv = _full_convolve(w, a_y_rev)

        y = y_new

def _make_cosine_profile(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * t / 2.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_tapered_profile(n, alpha):
    """Profile with raised cosine taper at edges."""
    t = np.linspace(0.0, 1.0, n)
    # Raised cosine window
    w = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = 1.0 + alpha * (w - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    best_a = a_init.copy()

    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-15, beta_frac * max_c)

    stagnant = 0

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)

        a_rev = a[::-1]
