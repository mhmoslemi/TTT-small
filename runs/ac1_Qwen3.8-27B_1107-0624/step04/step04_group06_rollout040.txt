from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    c = _convolve(a)
    return 2.0 * n * c.max()


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def _subgradient_exact(a, k_peak):
    """Compute subgradient of 2n*max(b) at peak k. g_k = 2*a[k-k_peak] if valid."""
    n = len(a)
    g = np.zeros(n)
    # g[k] = 2 * a[k_peak - k] if 0 <= k_peak - k < n
    idx = k_peak - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    g[valid] = 2.0 * a[idx[valid]]
    return g


def _nesterov_subgrad(a, deadline, lr0=0.01, momentum=0.9, n_restarts=5, restart_interval=200):
    """Nesterov-accelerated subgradient method with exact peak."""
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(a)
    
    lr = lr0
    m = 0.0  # momentum velocity
    prev_a = a.copy()
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a)
        k_peak = int(np.argmax(c))
        score = 2.0 * n * c[k_peak]
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Subgradient
        g = _subgradient_exact(a, k_peak)
        
        # Center the gradient (to maintain sum=1 constraint)
        g = g - g.mean()
        
        # Nesterov step
        m = momentum * m - lr * g
        a_new = a + m
        
        # Project onto simplex (a >= 0, sum a = 1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        
        # Decay learning rate
        lr = lr0 / (it + 1) ** 0.5
        
        # Periodic restart
        if (it + 1) % restart_interval == 0:
            a = best_a + 0.01 * np.random.randn(n)
            a = _normalize(a)
            if a is None:
                a = best_a.copy()
            m = 0.0
            lr = lr0 * 0.5
    
    return best_a, best_score

def _lbfgs_optimize(a_init, deadline, beta_init=1e-3):
    n = len(a_init)
    a0 = _normalize(a_init)
    
    # Initial beta for smooth max
    b0 = _convolve(a0)
    M0 = b0.max()
    beta = beta_init * M0 if beta_init < 1 else beta_init
    
    best_a = a0.copy()
    best_score = _score(a0)
    
    def objective_and_grad(z):
        # a = softmax(z)
        z_max = z.max()
        e = np.exp(z - z_max)
        a = e / e.sum()
        
        # b = conv(a, a)
        b = _convolve(a)
        
        # Smooth max
        b_shifted = (b - b.max()) / beta
        e_b = np.exp(b_shifted)
        s = b.max() + beta * np.log(e_b.sum())
        w = e_b / e_b.sum()
        
        # Gradient of smooth max w.r.t. a: g_k = 2 * (w * a_reversed)[n-1+k]
        N = _next_pow2(2*n-1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        c = np.fft.irfft(fw * fa_rev, N)
        g_a = 2.0 * c[n-1:n-1+n]
        
        # Gradient w.r.t. z: grad_z_k = a_k * (g_a_k - sum a_l * g_a_l)
        mean_g = np.dot(a, g_a)
        grad_z = a * (g_a - mean_g)
        
        # Objective (scaled)
        val = 2.0 * n * s
        
        return val, grad_z
    
    # Optimize using L-BFGS-B
    z0 = np.log(a0) - np.log(a0).mean()
    
    result = minimize(objective_and_grad, z0, jac=True, method='L-BFGS-B',
                      options={'maxiter': 10000, 'gtol': 1e-8, 'maxfun': 5000,
                              'disp': False, 'iprint': 0})
    
    a_opt = np.exp(result.x - result.x.max())
    a_opt = a_opt / a_opt.sum()
    
    return a_opt, _score(a_opt)

This is a 1D optimization in `delta`. I can solve it by bisection.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a):
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)


def _score(a):
    n = a.size
    c = _convolve(a)
    return 2.0 * n * c.max()


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def _smooth_max_grad(b, beta):
    """Returns (smooth_max_value, weights) for the smooth max of b."""
    m = b.max()
    x = (b - m) / beta
    # Clip for numerical stability
    x = np.clip(x, -100, 100)
    e = np.exp(x)
    s = e.sum()
    val = m + beta * np.log(s)
    w = e / s
    return val, w


def _grad_smooth_max(a, w, beta):
    """Gradient of smooth_max(conv(a,a)) w.r.t. a.
    Result: 2 * cross_corr(w, a)
    """
    n = a.size
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    c = np.fft.irfft(fw * fa_rev, N)
    return 2.0 * c[n-1:n-1+n]


def _lbfgs_step(a, beta, max_iters=200, timeout=60.0):
    """One round of L-BFGS optimization on the log-softmax parameterization."""
    n = a.size
    t_start = time.monotonic()
    t_end = t_start + timeout
    
    z = np.log(np.maximum(a, 1e-15))
    z -= z.mean()
    
    best_z = z.copy()
    best_a = a.copy()
    best_val = _score(a)
    
    class TimeoutException(Exception):
        pass
    
    def fun_and_grad(z_vec):
        if time.monotonic() > t_end:
            raise TimeoutException()
        
        # Softmax
        z_max = z_vec.max()
        e = np.exp(z_vec - z_max)
        a_vec = e / e.sum()
        
        # Convolution
        b = _convolve(a_vec)
        
        # Smooth max
        sm, w = _smooth_max_grad(b, beta)
        
        # Gradient w.r.t. a
        g_a = _grad_smooth_max(a_vec, w, beta)
        
        # Objective (scaled)
        val = 2.0 * n * sm
        
        # Gradient w.r.t. z
        mean_g = np.dot(a_vec, g_a)
        grad_z = a_vec * (g_a - mean_g)
        
        return val, grad_z
    
    try:
        result = minimize(fun_and_grad, z, jac=True, method='L-BFGS-B',
                         options={'maxiter': max_iters, 'gtol': 1e-10,
                                  'maxls': 30, 'disp': False})
        z_final = result.x
    except (TimeoutException, Exception):
        z_final = best_z
    
    # Compute final a
    z_max = z_final.max()
    a_final = np.exp(z_final - z_max)
    a_final = a_final / a_final.sum()
    
    # Also track best during optimization
    sc = _score(a_final)
    if sc < best_val:
        best_val = sc
        best_a = a_final
    
    return best_a, best_val

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimized step function coefficient search minimizing 2n*max(conv(a,a))/(sum a)^2.
Strategy: (1) Adam polish from parent, (2) L-BFGS continuation with decreasing 
smooth-max parameter, (3) LP-based monotone descent, (4) pairwise coordinate 
descent, (5) multi-scale exploration at n=2048, (6) Nesterov subgradient with 
random restarts and simulated annealing for escaping local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a):
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = a.size
    c = _convolve(a)
    return 2.0 * n * float(c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-30:
        return np.ones_like(z) / z.size
    return e / s
