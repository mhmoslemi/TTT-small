"""
LP-based peak reduction combined with smoothed optimization for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: Use linear programming to find directions that reduce the
autocorrelation peak, combined with fine-grained smoothed max optimization.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix, csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lp_find_direction(f, M, n):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k.
    Returns the optimal g (unnormalized).
    """
    n_vars = n
    n_cons = 2 * n - 1
    
    # Build sparse constraint matrix
    # Row k: sum_j f[j] * g[k-j] <= M (for valid j, k-j)
    rows = []
    cols = []
    vals = []
    
    for k in range(n_cons):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = coo_matrix((np.array(vals), (np.array(rows), np.array(cols))),
                   shape=(n_cons, n_vars)).tocsr()
    
    c_obj = -np.ones(n_vars)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n_vars
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None


def _lp_iteration(f, t=0.1, deadline=None):
    """
    One LP-based iteration: find g via LP, blend with f.
    Returns new f and the max of the new autocorrelation.
    """
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    
    g = _lp_find_direction(f, M, n)
    if g is None or g.sum() < 1e-10:
        return f, M, False
    
    # Rescale g to have same sum as f
    sum_f = float(f.sum())
    g = g * (sum_f / float(g.sum()))
    
    # Blend
    f_new = (1 - t) * f + t * g
    
    # Check improvement
    c_new = _full_convolve(f_new, f_new)
    M_new = float(c_new.max())
    
    return f_new, M_new, M_new < M


def _smoothed_obj(z, beta, n):
    """Smoothed max objective in log-parameterization."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    # Log-sum-exp for smooth max
    c_shifted = (c - cm) / beta
    lse = cm + beta * float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * lse


def _smoothed_grad(z, beta, n):
    """Gradient of smoothed max objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    w = np.exp(c_shifted)
    w = w / w.sum()
    W = float(np.dot(w, c))
    
    # Gradient: d/dz_j [2n * lse] = 2n * (d lse / da) * (da / dz)
    # d lse / da_k = sum_j w_j * (da_k/da_j) ... this is getting complex
    # Let me use the formula: grad_z = 4n * a * (h - W) where h_j = sum_k w_k * a[k-j]
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    grad = 4.0 * n * a * (h - W)
    
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_smoothed_lbfgs(a_init, deadline, beta_start, beta_end, n_betas=15):
    """L-BFGS with continuation on beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    # Log-spaced beta values
    log_betas = np.logspace(np.log10(beta_start), np.log10(beta_end), n_betas)
    
    for idx, beta in enumerate(log_betas):
        if time.monotonic() > deadline:
            break
        
        remaining = n_betas - idx
        per_beta_time = (deadline - time.monotonic()) / max(1, remaining)
        per_beta_time = min(per_beta_time, 90.0)
        level_dl = time.monotonic() + per_beta_time
        
        try:
            result = minimize(
                _smoothed_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_smoothed_grad,
                options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _coord_descent(a, deadline, max_rounds=200):
    """
    Coordinate descent: at each step, find the peak, then try adjusting
    individual coordinates to reduce it.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        k_star = int(np.argmax(c))
        
        # Find top contributors to the peak
        valid_i = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        if len(valid_i) == 0:
            break
        contributions = a_cur[valid_i] * a_cur[k_star - valid_i]
        # Sort by contribution, take top 15
        top_frac = min(15, len(valid_i))
        top_local = np.argsort(contributions)[::-1][:top_frac]
        top_idx = valid_i[top_local]
        
        improved = False
        for idx in top_idx:
            # Try reducing this coordinate
            for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score * (1 - 1e-8):
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Also try increasing small coordinates (redistribute)
        if not improved:
            small_idx = np.argsort(a_cur)[:min(10, n)]
            for idx in small_idx:
                for delta in [0.005, 0.01, 0.02]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_score * (1 - 1e-8):
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Try random perturbations to escape
            for _ in range(5):
                idx = np.random.randint(n)
                delta = np.random.uniform(-0.1, 0.1)
                z_trial = z.copy()
                z_trial[idx] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            
            if not improved:
                break
    
    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.1, max_iter=100000):
    """Adam optimization on the log-parameterized simplex."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = _full_convolve(a_init, a_init)
    cm = float(c.max())
    beta = 0.01 * cm  # Start with moderate smoothing
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Adaptive beta
        sorted_c = np.sort(c)[::-1]
        k = min(20, len(sorted_c) - 1)
        gap = sorted_c[0] - sorted_c[k]
        if gap < 0.05 * cm:
            beta = max(1e-10, 0.1 * cm)
        elif gap > 0.5 * cm:
            beta = max(1e-10, 0.01 * cm)
        else:
            beta = max(1e-10, 0.5 * gap)
        
        w, _ = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        a = _softmax_fn(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Decay learning rate
        if t % 2000 == 0:
            lr = max(0.005, lr * 0.9)
        
        # Periodic kick
        if t % 5000 == 0:
            z = z + 0.03 * np.random.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _make_initial(n):
    """Create a good initial sequence."""
    t = np.linspace(0, 1, n)
    # Try several profiles and pick the best
    candidates = []
    
    # Linear
    for slope in [0.5, 1.0, 2.0, -0.5, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # Quadratic
    for a1, a2 in [(1.0, 0.5), (2.0, -0.5), (-1.0, 0.5), (0.5, 1.0)]:
        u = 2*t - 1
        a = 1.0 + a1*u + a2*u**2
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        candidates.append(a / a.sum())
    
    # Cosine
    for p in [0.5, 1.0, 2.0]:
        a = np.maximum(np.cos(np.pi*(2*t-1)/2), 1e-10)**p
        candidates.append(a / a.sum())
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    best = min(candidates, key=lambda x: _score_norm(x))
    return best

"""
LP-based peak reduction combined with subgradient, smoothed L-BFGS,
coordinate descent, and Adam optimization for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.

Key innovation: Use linear programming to find directions that reduce
the autocorrelation peak (maximizing sum(g) s.t. (f*g)_k <= M),
combined with fast subgradient iterations and fine-grained local search.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix, csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _lp_find_direction(f, M, n):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k in [0, 2n-2].
    Returns optimal g or None on failure.
    """
    n_vars = n
    n_cons = 2 * n - 1

    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    count = 0

    for k in range(n_cons):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows[count] = k
            cols[count] = i
            vals[count] = f[k - i]
            count += 1

    rows = rows[:count]
    cols = cols[:count]
    vals = vals[:count]

    A = coo_matrix((vals, (rows, cols)), shape=(n_cons, n_vars)).tocsr()
    c_obj = -np.ones(n_vars)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n_vars

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 45})
        if res.success:
            return np.maximum(res.x, 0.0)
        elif res.status == 9:  # aborted due to time limit
            print("    LP aborted (time limit)")
    except Exception as e:
        print(f"    LP error: {e}")

    return None


def _subgradient_step(f, alpha):
    """
    Fast O(n) subgradient step: find peak position, subtract shifted f,
    project to non-negative and normalize.
    """
    n = len(f)
    c = _full_convolve(f, f)
    k_star = int(np.argmax(c))

    # Subgradient direction: (S_{k_star} + S_{-k_star}) f
    sf = np.zeros(n)
    # (S_{k_star} f)_i = f[i + k_star] if in range
    if k_star < n:
        sf[:n - k_star] += f[k_star:]
    # (S_{-k_star} f)_i = f[i - k_star] if in range
    if k_star >= 0:
        sf[k_star:] += f[:n - k_star]

    g = f - 2.0 * alpha * sf
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s < 1e-15:
        return f
    return g / s


def _smoothed_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    lse = cm + beta * float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * lse


def _smoothed_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    w = np.exp(c_shifted)
    w = w / w.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_smoothed_lbfgs(a_init, deadline, beta_start, beta_end, n_betas=12):
    """L-BFGS with log-spaced beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    log_betas = np.logspace(np.log10(beta_start), np.log10(beta_end), n_betas)

    for idx, beta in enumerate(log_betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        per_beta_time = (deadline - time.monotonic()) / max(1, remaining)
        per_beta_time = min(per_beta_time, 80.0)
        level_dl = time.monotonic() + per_beta_time

        try:
            result = minimize(
                _smoothed_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_smoothed_grad,
                options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': 2500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):

                s_new = _score_norm(a_new)

                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:

    return best_a, best_score

def _adam_optimize(a_init, deadline, lr=0.1, max_iter=50000):
    """Adam with adaptive beta on log-parameterized simplex."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        sorted_c = np.sort(c)[::-1]
        k = min(20, len(sorted_c) - 1)
        gap = sorted_c[0] - sorted_c[k]
        if gap < 0.05 * cm:
            beta = max(1e-10, 0.1 * cm)
        elif gap > 0.5 * cm:
            beta = max(1e-10, 0.01 * cm)
        else:
            beta = max(1e-10, 0.5 * gap)

        w, _ = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a = _softmax_fn(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 2000 == 0:
            lr = max(0.003, lr * 0.92)
        if t % 5000 == 0:
            z = z + 0.03 * np.random.default_rng(t).normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _run_lp_subgradient_combo(f, deadline, n_subgrad_per_lp=500, alpha_start=0.01):
    """
    Alternate between blocks of subgradient steps and LP steps.
    Subgradient is fast (O(n log n) per step via FFT for conv),
    LP is slow but gives big jumps.
    """
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm(f)
    a_cur = f.copy()
    alpha = alpha_start
    stuck = 0

    while time.monotonic() < deadline:
        # Block of subgradient steps
        block_end = min(deadline, time.monotonic() + 5.0)
        for _ in range(n_subgrad_per_lp):
            if time.monotonic() > block_end:
                break
            a_cur = _subgradient_step(a_cur, alpha)
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                stuck = 0
            else:
                stuck += 1

        # One LP step
        if time.monotonic() >= deadline:
            break
        c = _full_convolve(a_cur, a_cur)
        M = float(c.max())
        g = _lp_find_direction(a_cur, M, n)
        if g is not None and g.sum() > 1e-10:
            g = g / g.sum()
            best_t = 0
            best_sc = 2.0 * n * M
            for t in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                a_trial = (1 - t) * a_cur + t * g
                c_trial = _full_convolve(a_trial, a_trial)
                sc_trial = 2.0 * n * float(c_trial.max())
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_t = t
            if best_t > 0:
                a_cur = (1 - best_t) * a_cur + best_t * g
                sc = _score_norm(a_cur)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()
                    stuck = 0
        else:
            a_cur = best_a.copy()

        # Decay alpha
        alpha = max(0.001, alpha * 0.95)

        if stuck > 1000:
            # Restart from best
            a_cur = best_a.copy()
            alpha = alpha_start * 0.5
            stuck = 0

    return best_a, best_score


def _make_initial(n):
    """Generate a good initial normalized sequence."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    candidates = []

    for slope in [0.5, 1.0, 1.5, 2.0, -0.5, -1.0, -1.5, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())

    for a1, a2 in [(1.0, 0.5), (2.0, -0.5), (-1.0, 0.5), (0.5, 1.0), (1.5, -0.3)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())

    for rate in [1.0, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        candidates.append(a / a.sum())

    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-10) ** p
        candidates.append(a / a.sum())

    candidates.append(np.ones(n) / n)

    best = min(candidates, key=lambda x: _score_norm(x))
    return best


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    phase5_end = min(deadline, t_start + 1550.0)
    print("Phase 5: Adam")

    if time.monotonic() < phase5_end:
        per_dl = phase5_end - 10
        p_best, sc_best = _adam_optimize(best_seq, per_dl, lr=0.05)
        update_best(p_best, sc_best, "adam_p5")

        # Second Adam with lower lr
        if time.monotonic() < phase5_end - 30:
            per_dl2 = phase5_end - 15
            p_best2, sc_best2 = _adam_optimize(best_seq, per_dl2, lr=0.02)
            update_best(p_best2, sc_best2, "adam_slow_p5")

    # === Phase 6: Final LP + coordinate descent (100s) ===
    phase6_end = deadline
    print("Phase 6: Final refinement")

    if time.monotonic() < phase6_end - 20:
        # Quick LP round
        a_cur = best_seq.copy()
        n_cur = len(a_cur)
        stuck = 0
        for it in range(30):
            if time.monotonic() > phase6_end - 40:
                break
            c = _full_convolve(a_cur, a_cur)
            M = float(c.max())
            g = _lp_find_direction(a_cur, M, n_cur)
            if g is None or g.sum() < 1e-10:
                break
            g = g / g.sum()
            best_t = 0
            best_sc = 2.0 * n_cur * M
            for t in [0.01, 0.02, 0.05, 0.1]:
                a_trial = (1 - t) * a_cur + t * g
                sc_trial = 2.0 * n_cur * float(_full_convolve(a_trial, a_trial).max())
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_t = t
            if best_t > 0:
                a_cur = (1 - best_t) * a_cur + best_t * g
                stuck = 0
            else:
                stuck += 1
                if stuck > 5:
                    break
        update_best(a_cur, _score_norm(a_cur), "final_lp")

        if time.monotonic() < phase6_end - 15:
            p_best, sc_best = _coord_descent(best_seq, phase6_end - 5, max_rounds=150)
            update_best(p_best, sc_best, "final_coord")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_initial(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_initial(256)
    print(f"FALLBACK: score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
