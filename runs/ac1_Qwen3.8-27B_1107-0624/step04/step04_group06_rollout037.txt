"""
Optimization for minimizing 2n*max(autocorrelation)/sum^2 using:
- Multiple diverse initializations with anti-palindromic bias
- Smooth-max gradient descent in logit space (Adam)
- Targeted LP refinement with per-lag bounds
- Peak-flooding multiplicative updates
- Multi-scale search (n=1024, 2048)
- Adaptive restart strategy
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_and_grad(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_gradient(w, p):
    """Compute gradient of smooth_max((p*p)[k]) w.r.t. p."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_init_frac=0.06,
                   beta_decay=0.94, no_imp_limit=250, cooldown=0.0):

    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
   _imp_count = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_grad(c, beta)
        g = _conv_gradient(w, p)

        # Project gradient to simplex tangent space
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz

        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)

            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

        z = np.clip(z_new, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)

        score = float(2.0 * n * maxc)

                best_z = z.copy()
                no_imp = 0
                _imp_count += 1
            else:

            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

            beta = max(1e-4, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_solve_targeted(a, M_target, t_step=0.4):
    """
    LP with per-lag bounds derived from first-order approximation.
    Minimize peak of new autocorrelation by solving:
      maximize sum(b)
      s.t. (a*b)[k] <= c_k for all k, b >= 0
    where c_k accounts for current (a*a)[k] and desired reduction.
    """
    n = len(a)
    c_aa = _full_convolve(a)
    one_minus_t = 1.0 - t_step
    denom = 2.0 * t_step * one_minus_t

    # c_k = (M_target - (1-t)^2 * (a*a)[k]) / (2t(1-t))
    c_k = (M_target - one_minus_t**2 * c_aa) / denom
    # Ensure feasibility: clip to small positive
    c_k = np.maximum(c_k, 1e-10)

    # Build constraint matrix: (a*b)[k] = sum_i a[k-i] * b[i]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_ub, b_ub=c_k, bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t_start=0.35, max_solves=60, m_shrink=0.995):
    """LP refinement loop with targeted per-lag bounds."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_start
    n_solves = 0
    stall = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        # Target: slightly reduce M
        M_target = M * m_shrink
        b = _lp_solve_targeted(a, M_target, t_step=t)
        if b is None:
            # Fallback: use simpler uniform bound
            b = _lp_solve_targeted(a, M, t_step=t)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5:
                # Try larger M_target
                m_shrink = min(1.0, m_shrink + 0.005)
                stall = 0

        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.15, t * 0.93)

    return best_a, best_score


def _peak_flood(a, deadline, eta=0.04, n_iter_max=8000):
    """Multiplicative peak-flooding updates."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Identify top peaks
        threshold = M * 0.92
        hot_mask = c >= threshold
        n_hot = int(hot_mask.sum())
        if n_hot == 0:
            hot_mask[int(np.argmax(c))] = True

        # Compute heat efficiently using FFT
        # heat[i] = sum_{k: hot} a[k-i] (cross-correlation with hot mask)
        hot_vec = np.zeros(2 * n - 1)
        hot_vec[hot_mask] = 1.0
        N = _next_pow2(2 * n - 1)
        fh = np.fft.rfft(hot_vec, N)
        fa = np.fft.rfft(a, N)
        # Cross-correlation: heat[i] = sum_k hot_vec[k] * a[k-i]
        # This is the convolution of hot_vec with reversed a
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fh * fa_rev, N)
        heat = h[n - 1:2 * n - 1]
        heat = np.maximum(heat, 0)

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.005, eta * 0.85)
            else:
                eta = min(0.12, eta * 1.08)

    return best_a, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_quadratic(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_hump(n, pos1, pos2, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - pos1) / width) ** 2) + 0.7 * np.exp(-0.5 * ((t - pos2) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_anti_pal(n, strength=1.0):
    """Anti-palindromic: first half high, second half low (or vice versa)."""
    a = np.zeros(n)
    for i in range(n):
        if i < n // 2:
            a[i] = 1.0 + strength * (2.0 * i / n - 0.5)
        else:
            a[i] = 1.0 - strength * (2.0 * i / n - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_sine(n, freq, amp, phase=0.0):
    t = np.arange(n)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-12)
    return p1 / p1.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _try_update(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s
        sc = _score_norm(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {sc:.10f} n={p.size} [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")
    if parent is not None:
        print(f"Parent score: {_score_norm(parent):.10f}")

    # --- Phase 1: Quick Adam on parent (120s) ---
    phase1_end = t_start + 120.0
    if parent is not None:
        print("Phase 1: Adam on parent...")
        p1, sc1 = _adam_optimize(parent, phase1_end, rng, lr=0.04,
                                  beta_init_frac=0.05, beta_decay=0.94,
                                  no_imp_limit=200)
        _try_update(p1, 'adam_parent')
    else:
        p1 = _make_profile_linear(1024, 2.0)
        _try_update(p1, 'linear_init')

    # --- Phase 2: Try alternative initializations (150s) ---
    phase2_end = t_start + 270.0
    print(f"Phase 2: Alt inits, score={best_score:.10f}...")
    if time.monotonic() < phase2_end:
        n_try = 1024
        alts = [
            ('lin2.0', _make_profile_linear(n_try, 2.0)),
            ('lin3.5', _make_profile_linear(n_try, 3.5)),
            ('quad1.5', _make_profile_quadratic(n_try, 1.5)),
            ('quad2.5', _make_profile_quadratic(n_try, 2.5)),
            ('anti_pal1', _make_profile_anti_pal(n_try, 1.0)),
            ('anti_pal1.5', _make_profile_anti_pal(n_try, 1.5)),
            ('two_hump', _make_profile_two_hump(n_try, 0.3, 0.7, 0.15)),
            ('cos1_1.0', _make_profile_cosine(n_try, 1, 1.0)),
            ('sin1_0.8', _make_profile_sine(n_try, 1, 0.8, np.pi/4)),
            ('lin_neg', _make_profile_linear(n_try, -2.5)),  # decreasing
        ]
        per_alt = max(10.0, (phase2_end - time.monotonic()) / len(alts) - 2)
        for label, pa in alts:
            if time.monotonic() > phase2_end - 15:
                break
            # Quick evaluation
            sc_direct = _score_norm(pa)
            if sc_direct < best_score:
                _try_update(pa, f'init_{label}')
            # Quick Adam
            per = min(per_alt, phase2_end - time.monotonic() - 5)
            if per < 8:
                break
            p_alt, sc_alt = _adam_optimize(pa, time.monotonic() + per, rng,
                                            lr=0.05, beta_init_frac=0.07,
                                            beta_decay=0.93, no_imp_limit=120)
            _try_update(p_alt, f'adam_{label}')

    # --- Phase 3: LP refinement on best (200s) ---
    phase3_end = t_start + 470.0
    print(f"Phase 3: LP refine, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase3_end:
        p3, sc3 = _lp_refine(best_seq, phase3_end, t_start=0.35,
                              max_solves=50, m_shrink=0.993)
        _try_update(p3, 'lp1')
        # Follow-up Adam
        if time.monotonic() < phase3_end - 15:
            p3a, _ = _adam_optimize(best_seq, time.monotonic() + 40, rng,
                                     lr=0.03, beta_init_frac=0.04,
                                     beta_decay=0.95, no_imp_limit=100)
            _try_update(p3a, 'post_lp_adam')

    # --- Phase 4: Peak flooding (150s) ---
    phase4_end = t_start + 620.0
    print(f"Phase 4: Peak flood, score={best_score:.10f}...")
    if best_seq is not None and time.monotonic() < phase4_end:
        p4, sc4 = _peak_flood(best_seq, phase4_end, eta=0.03)
        _try_update(p4, 'flood1')

    # --- Phase 5: Second LP pass (150s) ---
    phase5_end = t_start + 770.0
    print(f"Phase 5: LP pass 2, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase5_end:
        p5, sc5 = _lp_refine(best_seq, phase5_end, t_start=0.3,
                              max_solves=40, m_shrink=0.995)
        _try_update(p5, 'lp2')

    # --- Phase 6: Scale to n=2048 (300s) ---
    phase6_end = t_start + 1070.0
    print(f"Phase 6: Scale to 2048, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 1024 and time.monotonic() < phase6_end:
        n_new = 2048
        p_int = _interp_profile(best_seq, n_new)
        sc_int = _score_norm(p_int)
        print(f"  Interpolated n={n_new}: {sc_int:.10f}")
        _try_update(p_int, f'interp_{n_new}')

        # Adam at new scale
        per = min(120.0, phase6_end - time.monotonic() - 50)
        if per > 30:
            p6, sc6 = _adam_optimize(p_int, time.monotonic() + per, rng,
                                      lr=0.03, beta_init_frac=0.04,
                                      beta_decay=0.95, no_imp_limit=200)
            _try_update(p6, f'adam_{n_new}')

        # LP at new scale
        if time.monotonic() < phase6_end - 20 and best_seq.size <= 2048:
            p6l, sc6l = _lp_refine(best_seq, time.monotonic() + min(60, phase6_end - time.monotonic() - 10),
                                    t_start=0.25, max_solves=15, m_shrink=0.996)
            _try_update(p6l, f'lp_{n_new}')

    # --- Phase 7: Try n=4096 if time permits (150s) ---
    phase7_end = t_start + 1220.0
    print(f"Phase 7: Try n=4096, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase7_end:
        n_new = 4096
        p_int = _interp_profile(best_seq, n_new)
        sc_int = _score_norm(p_int)
        print(f"  Interpolated n={n_new}: {sc_int:.10f}")
        _try_update(p_int, f'interp_{n_new}')

        per = min(80.0, phase7_end - time.monotonic() - 20)
        if per > 20:
            p7, sc7 = _adam_optimize(p_int, time.monotonic() + per, rng,
                                      lr=0.02, beta_init_frac=0.03,
                                      beta_decay=0.95, no_imp_limit=150)
            _try_update(p7, f'adam_{n_new}')

    # --- Phase 8: Final intensive polish (remaining) ---
    phase8_end = deadline
    print(f"Phase 8: Final polish, score={best_score:.10f}...")
    if best_seq is not None and time.monotonic() < phase8_end:
        remaining = phase8_end - time.monotonic()

        # Adam
        if remaining > 60:
            per = remaining * 0.35
            p8a, _ = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                     lr=0.015, beta_init_frac=0.03,
                                     beta_decay=0.96, no_imp_limit=80)
            _try_update(p8a, 'final_adam1')

        # LP if small enough
        if time.monotonic() < phase8_end - 30 and best_seq.size <= 2048:
            per = min(60.0, phase8_end - time.monotonic() - 20)
            if per > 20:
                p8l, _ = _lp_refine(best_seq, time.monotonic() + per,
                                     t_start=0.2, max_solves=15, m_shrink=0.997)
                _try_update(p8l, 'final_lp')

        # Peak flood
        if time.monotonic() < phase8_end - 20:
            per = min(40.0, phase8_end - time.monotonic() - 10)
            if per > 15:
                p8f, _ = _peak_flood(best_seq, time.monotonic() + per, eta=0.015)
                _try_update(p8f, 'final_flood')

        # Final Adam
        if time.monotonic() < phase8_end - 5:
            per = phase8_end - time.monotonic() - 5
            if per > 10:
                p8b, _ = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                         lr=0.01, beta_init_frac=0.02,
                                         beta_decay=0.97, no_imp_limit=50)
                _try_update(p8b, 'final_adam2')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 2.0)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(1024, 2.0)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
