import time

def _full_convolve(a):
    n = len(a)
    if n <= 512:
        return np.convolve(a, a)
    k = 1
    target = 2*n - 1
    while k < target:
        k <<= 1
    fa = np.fft.rfft(a, k)
    c = np.fft.irfft(fa * fa, k)[:2*n-1]

- If 0 <= k-i < n and k ≠ 2i: c[k] = 2*a[i]*a[k-i] + (independent part)
- If k = 2i and 0 <= k < 2n-1: c[k] = a[i]^2 + (independent part)
- Otherwise: c[k] = (independent part) [no dependence on a[i]]

When we change a[i] to x:
- Case 1 (valid, k≠2i): c_new[k] = 2*x*a[k-i] + ind[k], so change = 2*(x-a[i])*a[k-i]
- Case 2 (k=2i): c_new[k] = x^2 + ind[k], so change = x^2 - a[i]^2
- Case 3: no change

Now, the "independent part" ind[k] is the same as c[k] minus the a[i] contribution.

For the score: score = 2n * max_k c_new[k] / S(x)^2 where S(x) = S_old - a[i] + x.

To find the optimal x, I need to minimize:
f(x) = 2n * max_k (c[k] + Δc_k(x)) / (S_old - a[i] + x)^2

where Δc_k(x) is as above.

This is a 1D minimization problem. The function f(x) for x >= 0 is:
- The numerator max_k (c[k] + Δc_k(x)) is convex in x (max of affine/convex functions)
- The denominator (S_old - a[i] + x)^2 is convex in x
- The ratio of two convex functions is not necessarily convex

But empirically, it should be unimodal for x near the current value. So I can use golden section search.

For each coordinate i, I'll:
1. Compute the "base" convolution without the a[i] contribution: ind[k] = c[k] - contribution_of_a[i]_to_c[k]

    print(f"Started. Parent n={parent.size if parent is not None else 0}, budget={budget_s}s")

    # Phase 1: Diverse initialization (0-80s)
    phase1_end = t_start + 80.0
    candidates = []

    for n in [512, 768, 1024, 1536]:
        profs = [
            ('sqrt', _make_profile_sqrt(n)),
            ('sine1', _make_profile_sine(n)),
            ('lin0.5', _make_profile_linear(n, 0.5)),
            ('lin1.0', _make_profile_linear(n, 1.0)),
            ('lin1.5', _make_profile_linear(n, 1.5)),
            ('lin2.0', _make_profile_linear(n, 2.0)),
            ('lin2.5', _make_profile_linear(n, 2.5)),
            ('lin3.0', _make_profile_linear(n, 3.0)),
            ('quad0.5', _make_profile_quadratic(n, 0.5)),
            ('quad1.0', _make_profile_quadratic(n, 1.0)),
            ('quad1.5', _make_profile_quadratic(n, 1.5)),
            ('quad2.0', _make_profile_quadratic(n, 2.0)),
            ('gauss0.3', _make_profile_gaussian(n, 0.3)),
            ('gauss0.4', _make_profile_gaussian(n, 0.4)),
            ('gauss0.5', _make_profile_gaussian(n, 0.5)),
            ('cubic1', _make_profile_cubic(n, 0.5, -0.3)),
            ('cubic2', _make_profile_cubic(n, 0.8, -0.5)),
            ('cubic3', _make_profile_cubic(n, 1.2, -0.7)),
            ('pow0.5', _make_profile_power(n, 0.5)),
            ('pow0.7', _make_profile_power(n, 0.7)),
            ('pow1.0', _make_profile_power(n, 1.0)),
            ('thump', _make_profile_two_hump(n, 0.5, 0.2, 0.5)),
        ]
        for label, p in profs:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        pn = parent.size
        if pn >= 64:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))
        for n_try in [512, 768, 1024, 1536, 2048]:
            if n_try != pn:
                pi = _interp_profile(parent, n_try)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n_try, 'parent_interp'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # Phase 2: Adam polish top candidates (80s-400s)
    phase2_end = t_start + 400.0
    polished = []
    count = 0
    per_time = 35.0
    for sc, p, n, label in candidates:
        if time.monotonic() > phase2_end or count >= 6:
            break
        per_end = min(phase2_end, time.monotonic() + per_time)
        p_b, sc_b = _adam_logit(p, per_end, rng, lr=0.05, beta0=0.03,
                                beta_decay=0.955, no_imp_limit=500)
        _update(p_b, sc_b, f'adam_{label}_n{n}')
        polished.append((sc_b, p_b, n, label + '_pol'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best so far: {best_score:.10f}")

    # Phase 3: Second round Adam on best (400s-700s)
    phase3_end = t_start + 700.0
    if best_seq is not None:
        per = min(phase3_end - time.monotonic(), 250.0)
        if per > 20:
            p_s, sc_s = _adam_logit(best_seq, time.monotonic() + per, rng,
                                     lr=0.03, beta0=0.01, beta_decay=0.965,
                                     no_imp_limit=1500)
            _update(p_s, sc_s, 'adam_refine')

    # Phase 4: Mirror descent (700s-1100s)
    phase4_end = t_start + 1100.0
    if best_seq is not None:
        per = min(phase4_end - time.monotonic(), 350.0)
        if per > 20:
            p_m, sc_m = _mirror_descent_weighted(best_seq, time.monotonic() + per,
                                                  lr_init=0.03, top_k=3)
            _update(p_m, sc_m, 'mirror')

        # Adam after mirror
        if time.monotonic() < phase4_end - 10:
            per2 = min(60.0, phase4_end - time.monotonic() - 10)
            if per2 > 10:
                p_a2, sc_a2 = _adam_logit(best_seq, time.monotonic() + per2, rng,
                                           lr=0.02, beta0=0.005, beta_decay=0.975,
                                           no_imp_limit=2000)
                _update(p_a2, sc_a2, 'post_mirror_adam')

    # Phase 5: Coordinate descent (1100s-1500s)
    phase5_end = t_start + 1500.0
    if best_seq is not None:
        per = min(phase5_end - time.monotonic(), 350.0)
        if per > 10:
            p_cd, sc_cd = _coord_descent(best_seq, time.monotonic() + per, max_cycles=300)
            _update(p_cd, sc_cd, 'coord_descent')

        # Mirror after coord
        if time.monotonic() < phase5_end - 15:
            per2 = min(60.0, phase5_end - time.monotonic() - 15)
            if per2 > 10:
                p_m2, sc_m2 = _mirror_descent_weighted(best_seq, time.monotonic() + per2,
                                                        lr_init=0.01, top_k=2)
                _update(p_m2, sc_m2, 'mirror_cd')

    # Phase 6: Multi-restart (1500s-1750s)
    phase6_end = deadline - 30.0
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(20.0, (phase6_end - time.monotonic()) / (n_restarts + 1))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 15:
                break
            per_r_eff = min(per_r, phase6_end - time.monotonic() - 15)
            if per_r_eff < 15:
                break

            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.02 + 0.02 * i
            z += ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = np.exp(z)
            p_s = p_s / p_s.sum()

            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per_r_eff * 0.7, rng,
                                     lr=0.03, beta0=0.01, beta_decay=0.96,
                                     no_imp_limit=800)
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

            if time.monotonic() < phase6_end - 20:
                per_cd_r = min(20.0, phase6_end - time.monotonic() - 20)
                if per_cd_r > 8:
                    p_cdr, sc_cdr = _coord_descent(best_seq, time.monotonic() + per_cd_r,
                                                    max_cycles=50)
                    _update(p_cdr, sc_cdr, f'coord_r{i}')

    # Phase 7: Final polish (last 30s)
    if time.monotonic() < deadline - 3 and best_seq is not None:
        per = deadline - time.monotonic() - 3
        if per > 10:
            p_f, sc_f = _mirror_descent_weighted(best_seq, time.monotonic() + per * 0.5,
                                                  lr_init=0.005, top_k=2)
            _update(p_f, sc_f, 'final_mirror')

        if time.monotonic() < deadline - 3:
            per2 = deadline - time.monotonic() - 3
            if per2 > 8:
                p_f2, sc_f2 = _coord_descent(best_seq, time.monotonic() + per2, max_cycles=100)
                _update(p_f2, sc_f2, 'final_coord')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size} t={time.monotonic()-t_start:.1f}s")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL verify: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
