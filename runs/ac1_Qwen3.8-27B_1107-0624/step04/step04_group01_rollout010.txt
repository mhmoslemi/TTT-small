"""
Hybrid optimization combining Adam with adaptive line search, power-iteration
subroutine, diverse multi-restart, and slow beta-continuation for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin1":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-4.0 * t ** 2)
        return a / a.sum()
    elif kind == "zadoff":
        k = np.arange(n)
        phase = np.pi * k * (k + 1) / n
        a = 1.0 + 0.3 * np.cos(phase)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "multi_sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.sin(2 * np.pi * 7 * t) + 0.1 * np.sin(2 * np.pi * 13 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    for _ in range(n_modes):
        freq = rng.integers(1, n // 3)
        amp = rng.uniform(0.02, 0.12) * rng.choice([-1, 1])
        t = np.arange(n)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n)
        z = z + amp * 0.5 * np.sin(2.0 * np.pi * freq * t / n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_local(a, rng, window_size=32):
    """Perturb a random local window."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    start = rng.integers(0, max(1, n - window_size))
    w = min(window_size, n - start)
    z[start:start + w] += 0.15 * rng.standard_normal(w)
    z = z - z.mean()
    return _softmax(z)


def _compute_grad(a, c, w, n):
    """
    Compute gradient of 2n * <w, conv(a,a)> w.r.t. a.
    Result: 4n * v where v[k] = sum_j w[k+j]*a[j]
    v = convolve(a, w_rev)[n-1:2n-1][::-1]
    """
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    h = conv_aw[n - 1:2 * n - 1][::-1]
    return 4.0 * n * h


def _line_search(a, z, grad_z, lr_init, beta, n, max_backtrack=15):
    """Backtracking line search in z-space for the soft objective."""
    a_curr = _softmax(z)
    c_curr = _full_convolve(a_curr, a_curr)
    cm_curr = float(c_curr.max())
    f_curr = 2.0 * n * beta * np.log(float(np.sum(np.exp((c_curr - cm_curr) / beta))))
    
    lr = lr_init
    for _ in range(max_backtrack):
        z_trial = z - lr * grad_z
        z_trial = np.clip(z_trial, -50.0, 50.0)
        z_trial = z_trial - z_trial.mean()
        a_trial = _softmax(z_trial)
        c_trial = _full_convolve(a_trial, a_trial)
        cm_trial = float(c_trial.max())
        f_trial = 2.0 * n * beta * np.log(float(np.sum(np.exp((c_trial - cm_trial) / beta))))
        
        if f_trial < f_curr - 1e-12 or not np.isfinite(f_curr):
            return z_trial, lr
        lr *= 0.5
    return z - lr * grad_z, lr


def _adam_optimize(a_init, deadline, beta_schedule, max_iter=500000, lr=0.01,
                   use_ls=False, ls_every=50, ls_lr_init=0.005):
    """Adam with optional line search, in log-space with beta-continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    stall_count = 0
    prev_best = best_score

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad_z = _compute_grad(a, c, w, n)
            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            if use_ls and t % ls_every == 0:
                z, current_lr = _line_search(a, z, mh / (np.sqrt(vh) + eps),
                                              lr_init=ls_lr_init, beta=beta, n=n)
            else:
                update = current_lr * mh / (np.sqrt(vh) + eps)
                update_norm = np.linalg.norm(update)
                if update_norm > 0.5:
                    update = update * (0.5 / update_norm)
                z = z - update
                z = np.clip(z, -50.0, 50.0)
                z = z - z.mean()

            if t % 50 == 0:
                sc = _score_norm(_softmax(z))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = _softmax(z).copy()
                    stall_count = 0
                else:
                    stall_count += 1

            if stall_count > 200:
                # Small random kick to escape plateau
                z = z + 0.02 * np.random.standard_normal(n)
                z = z - z.mean()
                stall_count = 0
                current_lr = max(0.002, current_lr * 0.8)

            if it_in_phase % 5000 == 0:
                current_lr = max(0.001, current_lr * 0.95)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _power_iter_step(a, beta, n_steps=10):
    """Power iteration: a <- max(w * cross_corr, 0) normalized."""
    n = len(a)
    for _ in range(n_steps):
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        # v[k] = sum_j w[k+j]*a[j]
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        v = conv_aw[n - 1:2 * n - 1][::-1]
        v_pos = np.maximum(v, 0.0)
        s_v = float(v_pos.sum())
        if s_v < 1e-15:
            break
        v_norm = v_pos / s_v
        a = 0.7 * a + 0.3 * v_norm
        a = a / a.sum()
    return a


def _lp_direction(a, M):
    """
    Solve LP: maximize sum(g) subject to g>=0, A@b <= M.
    A[k,j] = a[j] if valid, else 0.
    Returns g (normalized to sum=1) or None.
    """
    n = len(a)
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = a[j] if 0 <= k-j <= n-1
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        A[k, j_min:j_max + 1] = a[j_min:j_max + 1]
    
    # LP: min -sum(g) subject to A@g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            s = float(g.sum())
            if s > 0.01:
                return g / s
    except Exception:
        pass
    return None


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initialization, Adam with line search,
    power iteration, LP-guided directions, and fine polishing to minimize
    2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        parent = _make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")

    # --- Phase 0: Exploration (0-40s) ---
    phase0_end = min(deadline, t_start + 40.0)
    
    profiles = ["sin1", "sin2", "tapered", "gaussian", "zadoff", "multi_sine"]
    for kind in profiles:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(n_main, kind)
        sc = _score(p)
        update_best(p, sc, f"init_{kind}")

    # Perturbed parent
    for sigma in [0.02, 0.05, 0.1]:
        if time.monotonic() > phase0_end:
            break
        p = _perturb(parent, sigma, rng)
        update_best(p, _score(p), f"perturb_{sigma}")

    # Fourier perturbation
    p = _perturb_fourier(parent, rng, n_modes=3)
    update_best(p, _score(p), "perturb_fourier")

    # Quick power iteration from parent
    c_p = _full_convolve(parent, parent)
    beta_pi = 0.02 * float(c_p.max())
    p_pi = _power_iter_step(parent, beta_pi, n_steps=5)
    update_best(p_pi, _score(p_pi), "power_iter_init")

    print(f"Phase 0 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 1: Main optimization (40s-500s) ---
    phase1_end = min(deadline, t_start + 500.0)
    
    c_best = _full_convolve(best_seq, best_seq)
    max_cb = float(c_best.max())
    
    # Long beta schedule - slower decay for better convergence
    beta_schedule_main = [
        0.04 * max_cb, 0.03 * max_cb, 0.02 * max_cb, 0.015 * max_cb,
        0.01 * max_cb, 0.008 * max_cb, 0.006 * max_cb, 0.004 * max_cb,
        0.003 * max_cb, 0.002 * max_cb
    ]
    
    remaining = phase1_end - time.monotonic() - 10
    if remaining > 10:
        p_b, sc_b = _adam_optimize(
            best_seq, time.monotonic() + remaining,
            beta_schedule_main, max_iter=400000, lr=0.012,
            use_ls=True, ls_every=100, ls_lr_init=0.004
        )
        update_best(p_b, _score(p_b), "main_adam_ls")

    # Power iteration refinement
    if time.monotonic() < phase1_end - 20:
        c_cur = _full_convolve(best_seq, best_seq)
        beta_cur = 0.005 * float(c_cur.max())
        p_pi = _power_iter_step(best_seq, beta_cur, n_steps=20)
        sc_pi = _score(p_pi)
        update_best(p_pi, sc_pi, "power_iter_main")
        
        # Adam from power iter result
        if sc_pi < best_score + 0.01:
            remaining2 = phase1_end - time.monotonic() - 5
            if remaining2 > 10:
                c_pi = _full_convolve(p_pi, p_pi)
                bs_pi = [0.003 * float(c_pi.max()), 0.001 * float(c_pi.max()),
                         0.0005 * float(c_pi.max())]
                p_b2, sc_b2 = _adam_optimize(
                    p_pi, time.monotonic() + remaining2,
                    bs_pi, max_iter=100000, lr=0.008
                )
                update_best(p_b2, _score(p_b2), "adam_from_pi")

    print(f"Phase 1 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 2: LP-guided + diverse restarts (500s-1000s) ---
    phase2_end = min(deadline, t_start + 1000.0)
    
    # Try LP direction a few times
    for lp_iter in range(3):
        if time.monotonic() > phase2_end - 60:
            break
        c_b = _full_convolve(best_seq, best_seq)
        M_val = float(c_b.max())
        g = _lp_direction(best_seq, M_val)
        if g is not None:
            # Blend toward LP solution
            for t_blend in [0.1, 0.2, 0.3]:
                p_blend = (1 - t_blend) * best_seq + t_blend * g
                p_blend = np.maximum(p_blend, 0)
                p_blend = p_blend / p_blend.sum()
                sc_blend = _score(p_blend)
                update_best(p_blend, sc_blend, f"lp_blend_{t_blend}_{lp_iter}")
        
        # Continue with Adam from improved point
        if time.monotonic() < phase2_end - 30:
            c_ib = _full_convolve(best_seq, best_seq)
            bs = [0.002 * float(c_ib.max()), 0.001 * float(c_ib.max()),
                  0.0005 * float(c_ib.max())]
            rem = min(60.0, phase2_end - time.monotonic() - 5)
            if rem > 5:
                p_b, sc_b = _adam_optimize(
                    best_seq, time.monotonic() + rem, bs, max_iter=50000, lr=0.006
                )
                update_best(p_b, _score(p_b), f"lp_adam_{lp_iter}")

    # Diverse restarts
    n_restarts = 6
    per_restart = (phase2_end - time.monotonic() - 20) / max(1, n_restarts)
    
    for i in range(n_restarts):
        if time.monotonic() > phase2_end - 15:
            break
        
        if i % 4 == 0:
            p_start = _perturb(best_seq, 0.03 + 0.01 * i, rng)
        elif i % 4 == 1:
            p_start = _perturb_fourier(best_seq, rng, n_modes=3)
        elif i % 4 == 2:
            p_start = _perturb_local(best_seq, rng, window_size=64)
        else:
            p_start = _power_iter_step(best_seq, 0.005 * float(_full_convolve(best_seq, best_seq).max()), n_steps=3)
            p_start = _perturb(p_start, 0.03, rng)

        c_s = _full_convolve(p_start, p_start)
        bs = [0.01 * float(c_s.max()), 0.005 * float(c_s.max()),
              0.002 * float(c_s.max()), 0.001 * float(c_s.max())]
        
        p_b, sc_b = _adam_optimize(
            p_start, time.monotonic() + max(5.0, per_restart),
            bs, max_iter=80000, lr=0.008
        )
        update_best(p_b, _score(p_b), f"restart_{i}")

    print(f"Phase 2 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 3: Fine polishing (1000s-1500s) ---
    phase3_end = min(deadline, t_start + 1500.0)
    
    if best_seq is not None and time.monotonic() < phase3_end:
        c_f = _full_convolve(best_seq, best_seq)
        max_cf = float(c_f.max())
        
        # Very fine Adam
        beta_fine = [
            0.001 * max_cf, 0.0005 * max_cf, 0.0003 * max_cf,
            0.0002 * max_cf, 0.0001 * max_cf, 0.00005 * max_cf
        ]
        rem = phase3_end - time.monotonic() - 10
        if rem > 10:
            p_b, sc_b = _adam_optimize(
                best_seq, time.monotonic() + rem,
                beta_fine, max_iter=400000, lr=0.004,
                use_ls=True, ls_every=200, ls_lr_init=0.002
            )
            update_best(p_b, _score(p_b), "fine_adam")

        # Ultra fine
        if time.monotonic() < phase3_end - 30:
            c_uf = _full_convolve(best_seq, best_seq)
            max_cuf = float(c_uf.max())
            beta_ultra = [0.00003 * max_cuf, 0.00001 * max_cuf, 0.000005 * max_cuf]
            rem2 = phase3_end - time.monotonic() - 5
            if rem2 > 10:
                p_b2, sc_b2 = _adam_optimize(
                    best_seq, time.monotonic() + rem2,
                    beta_ultra, max_iter=200000, lr=0.001
                )
                update_best(p_b2, _score(p_b2), "ultra_fine")

        # Power iteration at very low beta
        if time.monotonic() < phase3_end - 10:
            c_pi = _full_convolve(best_seq, best_seq)
            p_pi = _power_iter_step(best_seq, 0.0005 * float(c_pi.max()), n_steps=10)
            update_best(p_pi, _score(p_pi), "power_fine")

    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 4: Try different n (1500s-1750s) ---
    phase4_end = min(deadline, t_start + 1750.0)
    
    if best_seq is not None and time.monotonic() < phase4_end - 30:
        for target_n in [n_main + 256, n_main - 256, n_main + 512]:
            if time.monotonic() > phase4_end - 20:
                break
            if target_n < 100:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_int = _score(p_interp)
            update_best(p_interp, sc_int, f"interp_n{target_n}")
            
            if sc_int < best_score + 0.005:
                c_int = _full_convolve(p_interp, p_interp)
                bs_int = [0.002 * float(c_int.max()), 0.001 * float(c_int.max()),
                          0.0005 * float(c_int.max()), 0.0002 * float(c_int.max())]
                rem = min(40.0, phase4_end - time.monotonic() - 5)
                if rem > 10:
                    p_b, sc_b = _adam_optimize(
                        p_interp, time.monotonic() + rem,
                        bs_int, max_iter=100000, lr=0.005
                    )
                    update_best(p_b, _score(p_b), f"opt_n{target_n}")

    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 5: Final polish (1750s - deadline) ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        c_fin = _full_convolve(best_seq, best_seq)
        max_cfin = float(c_fin.max())
        beta_final = [0.00002 * max_cfin, 0.00001 * max_cfin]
        rem = deadline - time.monotonic() - 3
        if rem > 5:
            p_b, sc_b = _adam_optimize(
                best_seq, time.monotonic() + rem,
                beta_final, max_iter=100000, lr=0.001
            )
            update_best(p_b, _score(p_b), "final_polish")

    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
