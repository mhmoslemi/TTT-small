"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses spectral initialization, logsumexp continuation with L-BFGS,
top-K peak targeting with accelerated gradient, active-set refinement,
and multi-restart strategy with diverse profiles.
"""
import numpy as np
from scipy.optimize import minimize
import time
import math


def _next_pow2(n):
    if n <= 1:
        return 2
    return 1 << (n - 1).bit_length()


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _autocorr_gradient_weighted(w, a, n):
    """Compute g[k] = 2 * sum_j w[j] * a[j-k] for k=0..n-1 using FFT."""
    a_rev = a[::-1]
    conv = _fft_convolve(a_rev, w)
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _logsumexp_obj_and_grad(z, n, beta):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    if beta < 1e-15:
        obj = 2.0 * n * cm
        idx = int(np.argmax(c))
        g_a = np.zeros(n)
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        if lo <= hi:
            g_a[lo:hi + 1] = 2.0 * a[idx - hi:idx - lo + 1][::-1]
    else:
        c_shift = (c - cm) / beta
        e = np.exp(c_shift)
        w = e / e.sum()
        obj = 2.0 * n * (cm + beta * math.log(max(e.sum(), 1e-300)))
        g_a = _autocorr_gradient_weighted(w, a, n)
        g_a *= n
    a_dot_g = float(a @ g_a)
    grad_z = a * (g_a - a_dot_g)
    return obj, grad_z


def _topk_obj_and_grad(z, n, K, alpha):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    k_actual = min(K, len(c))
    top_idx = np.argsort(c)[-k_actual:]
    top_vals = c[top_idx]
    topk_sum = float(top_vals.sum())
    obj = 2.0 * n * topk_sum
    w = np.zeros(len(c))
    for idx in top_idx:
        w[int(idx)] = 1.0 / k_actual
    g_a = _autocorr_gradient_weighted(w, a, n) * n
    a_dot_g = float(a @ g_a)
    grad_z = a * (g_a - a_dot_g)
    return obj, grad_z


def _lbfgs_phase(a_init, deadline, beta_list, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if max_wall_time is not None:
        per_level = max(5.0, max_wall_time / max(1, len(beta_list)))
    else:
        per_level = 40.0

    for beta in beta_list:
        if time.monotonic() > deadline - 2.0:
            break
        level_dl = min(deadline, time.monotonic() + per_level)

        def _obj(z_):
            return _logsumexp_obj_and_grad(z_, n, beta)[0]

        def _grad(z_):
            return _logsumexp_obj_and_grad(z_, n, beta)[1]

        def _obj_grad(z_):
            o, g = _logsumexp_obj_and_grad(z_, n, beta)
            return o, g

        try:
            res = minimize(
                _obj_grad, z, jac=True,
                method='L-BFGS-B',
                options={'maxiter': 800, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': 3000, 'iprint': 0}

            if res.x is not None and np.all(np.isfinite(res.x)):

                a_new = _softmax(z_new)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:

def _local_search(a_init, deadline, rng, n_trials=10000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

        if time.monotonic() > deadline - 1.0:
            break
        block_size = int(rng.integers(3, min(30, n // 3)))
        start = int(rng.integers(0, n - block_size))
        a_trial = a.copy()
        pert = rng.uniform(0.9, 1.1, size=block_size)
        a_trial[start:start + block_size] *= pert
        a_trial = np.maximum(a_trial, 1e-15)

def _make_hann(n):
    t = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    return a / a.sum()


def _make_linear(n, slope=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, a1=1.0, a2=0.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n, order=1):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, order + 1):
        a *= 0.5 * (1.0 + np.cos(np.pi * (2 * t - 1) ** k))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_spectral(n, n_peaks=3, peak_width=0.1):
    """Generate sequence from a designed power spectrum (flat with peaks)."""
    N = n
    freqs = np.fft.rfftfreq(N)
    power = np.ones(N // 2 + 1)
    for k in range(1, n_peaks * 2 + 1):
        center = k / (N // 2)
        power *= 1.0 + 0.5 * np.exp(-((freqs * N - k * 5) ** 2) / (2.0 * (peak_width * N) ** 2))
    spectrum = np.sqrt(power) / np.sqrt(power.sum())
    signal = np.fft.irfft(spectrum * np.sqrt(N), N)
    signal = np.abs(signal) + 1e-8
    return signal / signal.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
