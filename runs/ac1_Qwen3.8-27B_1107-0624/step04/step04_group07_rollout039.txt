                    beta = max(1e-4, beta * 0.7)
            else:
                beta = max(1e-4, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = 0.05 + 0.03 * (no_imp // no_imp_limit)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.1 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score

def _lp_fixed_point(a, deadline, max_iters=30, use_line_search=True):
    """LP-based fixed-point iteration."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        M = float(c.max())
        
        # Build the constraint matrix
        # conv(b, a)[k] <= M for all k
        # conv(b, a)[k] = sum_i b[i] * a[k-i] (where a[j] = 0 for j outside [0,n-1])
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            # conv(b,a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} b[i]*a[k-i]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[k - idx].tolist())
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        b = np.maximum(res.x, 0.0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b
        
        if use_line_search:
            # Golden section search
            lo, hi = 0.0, 1.0
            best_t = 0.5
            best_sc = best_score
            for _ in range(15):
                t1 = lo + 0.382 * (hi - lo)
                t2 = lo + 0.618 * (hi - lo)
                a1 = (1 - t1) * a + t1 * b
                a1 = np.maximum(a1, 1e-15)
                a1 = a1 / a1.sum()
                sc1 = _score(a1)
                
                a2 = (1 - t2) * a + t2 * b
                a2 = np.maximum(a2, 1e-15)
                a2 = a2 / a2.sum()
                sc2 = _score(a2)
                
                if sc1 < sc2:
                    hi = t2
                    if sc1 < best_sc:
                        best_sc = sc1
                        best_t = t1
                else:
                    lo = t1
                    if sc2 < best_sc:
                        best_sc = sc2
                        best_t = t2
            
            a = (1 - best_t) * a + best_t * b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
        else:
            a = b
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _coord_descent(a, deadline, max_sweeps=30):
    """Coordinate descent: try adjusting each coordinate."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            old_val = a[j]
            # Try multiplying by various factors
            for factor in [0.5, 0.7, 1.0/0.7, 2.0]:
                a[j] = max(old_val * factor, 1e-15)
                sc = _score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    # Continue trying to improve at this coordinate
                else:
                    a[j] = old_val
                    break
            a[j] = old_val
        if not improved:
            break
    
    return best_a, best_score

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new

def _make_profile(n, profile_type, p1=1.0, p2=0.0, p3=0.0):
    if profile_type == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5) * 2
    elif profile_type == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif profile_type == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif profile_type == 'exp_decay':
        t = np.arange(n) / max(1, n - 1)
        a = np.exp(-p1 * t) + 0.01
    elif profile_type == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif profile_type == 'sine':
        t = np.arange(n)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif profile_type == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif profile_type == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.05 + abs(t - 0.5) * 2.0) ** p1
    elif profile_type == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2) + 0.01
    elif profile_type == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / p3) ** 2) + np.exp(-0.5 * ((t - p2) / p3) ** 2) + 0.01
    elif profile_type == 'power':
        t = np.linspace(0, 1, n)
        a = t ** p1 + (1 - t) ** p1
    elif profile_type == 'step':
        # Two-level: high at edges, low in middle
        t = np.linspace(0, 1, n)
        a = np.where(np.abs(t - 0.5) > p1, 1.0, p2)
    elif profile_type == 'triangle':
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * t - 1) ** p1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines fixed-gradient Adam smooth-max (with corrected FFT size), LP fixed-point
with golden section line search, coordinate descent, and multi-scale escalation.
Key fix: uses N >= 3n-2 for cross-correlation FFT to avoid circular aliasing.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    n = len(a)
    c = _full_convolve(a)
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _grad_smooth_max(a, beta):
    """
    Compute gradient of smooth_max(conv(a,a)) w.r.t. a.
    grad[j] = 2 * sum_{i=0}^{n-1} w[j+i] * a[i]
    where w = softmax((c - max(c))/beta) and c = conv(a,a).
    Uses FFT with N >= 3n-2 to avoid circular aliasing.
    """
    n = len(a)
    c = _full_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # Cross-correlation: grad[j] = 2 * sum_i w[j+i]*a[i]
    # = 2 * (conv(w, flip(a)))[j + n - 1]
    # conv length = (2n-1) + n - 1 = 3n-2, so N >= 3n-2
    N = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h[n - 1:2 * n - 1]

    smooth_val = mval + beta * float(np.log(s)) if s > 0 else mval
    return grad, c, smooth_val


def _adam_smooth_max(a, deadline, rng, lr=0.04, beta_init=None,
                     beta_decay=0.96, no_imp_limit=150):
    """Adam optimizer in log-space for smooth max of convolution."""
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(0.05 * maxc, 0.01)

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    it = 0

    while time.monotonic() < deadline:
        it += 1
        grad, c, _ = _grad_smooth_max(a, beta)

        # Project gradient onto tangent space of simplex
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()

        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c = _full_convolve(a_new)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            a = a_new
            if score < best_score - 1e-13:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 80 == 0:
                    lr = max(0.002, lr * 0.92)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        # Adaptive beta: shrink based on gap
        if it % 15 == 0:
            c_sorted = np.sort(c)[::-1]
            if c_sorted.size >= 2:
                gap = c_sorted[0] - c_sorted[1]
                beta = max(1e-4, beta * beta_decay)
                if gap < beta * 0.3:
                    beta = max(1e-4, beta * 0.6)

        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = 0.04 + 0.02 * min(5, no_imp // no_imp_limit)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.08 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _build_lp_matrix(n, a):
    """Build sparse matrix A where A @ b = conv(b, a)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_fixed_point(a, deadline, max_iters=25, use_line_search=True):
    """LP-based fixed-point iteration with optional golden section line search."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())

        A = _build_lp_matrix(n, a)
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        b = np.maximum(res.x, 0.0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        if use_line_search:
            # Golden section search in [0, 1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618
            lo, hi = 0.0, 1.0
            best_t = 0.5
            best_sc = best_score

            for _ in range(18):
                t1 = lo + (1.0 - gr) * (hi - lo)
                t2 = lo + gr * (hi - lo)
                a1 = (1.0 - t1) * a + t1 * b
                a1 = np.maximum(a1, 1e-15)
                a1 = a1 / a1.sum()
                sc1 = _score(a1)

                a2 = (1.0 - t2) * a + t2 * b
                a2 = np.maximum(a2, 1e-15)
                a2 = a2 / a2.sum()
                sc2 = _score(a2)

                if sc1 < sc2:
                    hi = t2
                    if sc1 < best_sc:
                        best_sc = sc1
                        best_t = t1
                else:
                    lo = t1
                    if sc2 < best_sc:
                        best_sc = sc2
                        best_t = t2

            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
        else:
            a = b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _coord_descent(a, deadline, max_sweeps=40):
    """Coordinate descent with multiple trial factors per coordinate."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    factors = [0.3, 0.5, 0.7, 1.4, 2.0, 3.0]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            old_val = a[j]
            for factor in factors:
                a[j] = max(old_val * factor, 1e-15)
                sc = _score(a)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
                    break
            a[j] = old_val
        if not improved:
            break

    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, profile_type, p1=1.0, p2=0.0, p3=0.0):
    t = np.linspace(0, 1, n)
    if profile_type == 'linear':

    elif profile_type == 'quadratic':
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif profile_type == 'cubic':
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif profile_type == 'exp_decay':
        a = np.exp(-p1 * t) + 0.01
    elif profile_type == 'cosine':
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t)

        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t)
    elif profile_type == 'U_shape':
        a = (2 * t - 1) ** 2 + p1
    elif profile_type == 'edge_heavy':
        a = 1.0 / (0.05 + abs(t - 0.5) * 2.0) ** p1
    elif profile_type == 'gauss':
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2) + 0.01
    elif profile_type == 'two_gauss':

    elif profile_type == 'power':

        a = np.where(np.abs(t - 0.5) > p1, 1.0, p2)
    elif profile_type == 'triangle':
        a = 1.0 - np.abs(2 * t - 1) ** p1
    elif profile_type == 'w_shape':
        a = 1.0 + p1 * np.cos(4 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score(p)
            candidates.append((sc, p, n, label))

    # Add parent variants
    if parent is not None:
        for n in [256, 512, 1024, 2048]:
            pi = _interp(parent, n)
            sc = _score(pi)
            candidates.append((sc, pi, n, 'parent'))
        if 0 < pn <= 4096:
            sc = _score(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][0] < best_score:
        best_seq = np.asarray(candidates[0][1], dtype=np.float64).copy()
        best_score = float(candidates[0][0])

    # === Phase 2: Adam smooth-max on top candidates ===
    print("Phase 2: Adam optimization...")
    phase2_end = t_start + 500.0

    # Select diverse top candidates (different profiles/sizes)
    to_optimize = []
    seen_labels = set()
    for (sc, p, n, label) in candidates:
        key = label
        if key not in seen_labels:
            seen_labels.add(key)
            to_optimize.append((sc, p, n, label))
        if len(to_optimize) >= 8:
            break

    polished = []
    for i, (sc, p, n, label) in enumerate(to_optimize):
        if time.monotonic() > phase2_end - 20:
            break
        per = min(60.0, phase2_end - time.monotonic() - 10)
        if per < 15:
            break
        a_ad, sc_ad = _adam_smooth_max(p, time.monotonic() + per, rng,
                                       lr=0.04, beta_decay=0.96, no_imp_limit=120)
        sc_ad = _score(a_ad)
        if sc_ad < best_score:
            best_seq = a_ad.copy()
            best_score = sc_ad
            print(f"  BEST: {best_score:.10f} n={a_ad.size} [Adam_{label}]")
        polished.append((sc_ad, a_ad, n, label + '_ad'))

    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f}")

    # === Phase 3: LP fixed-point polishing ===
    print("Phase 3: LP fixed-point...")
    phase3_end = t_start + 1000.0

    for (sc, p, n, label) in polished[:5]:
        if time.monotonic() > phase3_end - 40:
            break
        # For LP, use n <= 1024
        if n > 1024:
            p_lp = _interp(p, 512)
            n_lp = 512
        else:
            p_lp = p.copy()
            n_lp = n

        per = min(150.0, phase3_end - time.monotonic() - 30)
        if per < 25:
            break

        a_lp, sc_lp = _lp_fixed_point(p_lp, time.monotonic() + per, max_iters=20)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  BEST: {best_score:.10f} n={a_lp.size} [LP_{label}]")

        # Adam polish after LP (short)
        if time.monotonic() < phase3_end - 30 and n_lp <= 1024:
            per2 = min(50.0, phase3_end - time.monotonic() - 20)
            if per2 > 15:
                a_post, sc_post = _adam_smooth_max(a_lp, time.monotonic() + per2, rng,
                                                   lr=0.02, beta_decay=0.97, no_imp_limit=60)
                sc_post = _score(a_post)
                if sc_post < best_score:
                    best_seq = a_post.copy()
                    best_score = sc_post
                    print(f"  BEST: {best_score:.10f} n={a_post.size} [AdamPostLP_{label}]")

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale...")
    phase4_end = t_start + 1400.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score(p_int)
            print(f"  Interp n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Adam on interpolated
            if time.monotonic() < phase4_end - 50:
                per = min(120.0, phase4_end - time.monotonic() - 30)
                if per > 20:
                    a_e, sc_e = _adam_smooth_max(p_int, time.monotonic() + per, rng,
                                                 lr=0.025, beta_decay=0.97, no_imp_limit=200)
                    sc_e = _score(a_e)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

    # === Phase 5: Coordinate descent ===
    print("Phase 5: Coordinate descent...")
    phase5_end = deadline - 60.0

    if best_seq is not None and time.monotonic() < phase5_end:
        # Use a truncated version if too large for coord descent speed
        n_cd = min(best_seq.size, 1024)
        if n_cd < best_seq.size:
            p_cd = _interp(best_seq, n_cd)
        else:
            p_cd = best_seq.copy()
        per = min(phase5_end - time.monotonic() - 20, 300.0)
        if per > 30:
            a_cd, sc_cd = _coord_descent(p_cd, time.monotonic() + per, max_sweeps=30)
            sc_cd = _score(a_cd)
            if sc_cd < best_score:
                # If we improved at smaller n, try to upscale
                if n_cd < best_seq.size:
                    a_cd_up = _interp(a_cd, best_seq.size)
                    sc_cd_up = _score(a_cd_up)
                    if sc_cd_up < best_score:
                        best_seq = a_cd_up.copy()
                        best_score = sc_cd_up
                    else:
                        best_seq = a_cd.copy()
                        best_score = sc_cd
                else:
                    best_seq = a_cd.copy()
                    best_score = sc_cd
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [coord]")

    # === Phase 6: Final Adam polish ===
    print("Phase 6: Final polish...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 15:
            n_final = min(best_seq.size, 2048)
            if n_final < best_seq.size:
                p_final = _interp(best_seq, n_final)
            else:
                p_final = best_seq.copy()
            a_fp, sc_fp = _adam_smooth_max(p_final, time.monotonic() + per, rng,
                                           lr=0.01, beta_decay=0.98, no_imp_limit=40)
            sc_fp = _score(a_fp)
            if sc_fp < best_score:
                best_seq = a_fp.copy()
                best_score = sc_fp
                print(f"  BEST: {best_score:.10f} n={best_seq.size} [final]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0, 0)

        exact_sc = _score(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
