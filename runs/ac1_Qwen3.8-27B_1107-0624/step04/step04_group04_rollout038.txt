Uses aggressive Adam annealing, coordinate descent polishing,

import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_smooth_grad(a, beta, n, top_k=0):
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w[idx[0]] = 1.0
        else:
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return w, h, W, c, cm


def _z_gradient(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    w, h, W, c, cm = _compute_smooth_grad(a, beta, n, top_k)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

    c = _full_convolve(a, a)
    cm = float(c.max())

        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:

        log_sum = float(np.log(np.sum(np.exp(c_shifted))))

        grad = _z_gradient(z, beta, n, use_k)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            c = _full_convolve(a, a)
            max_c = float(c.max())
        if t % 80 == 0:
            beta = max(1e-14, beta * 0.82)
            if adaptive_k and t % 400 == 0:
                cur_k = min(cur_k + 3, 40)
        if t % 1500 == 0:
            if best_score < 1.51:
                lr = max(0.002, lr * 0.95)
            c = _full_convolve(best_a, best_a)
            max_c = float(c.max())
            beta = max(1e-14, 0.005 * max_c)
    return best_a, best_score


def _coord_descent_polish(a_init, deadline, max_sweeps=500, top_k=25):
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a)
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        active_idx = np.argsort(c)[-top_k:]
        active_c = c[active_idx]
        active_min = float(active_c.min())
        for j in range(n):
            ai = a[j]
            alphas = np.zeros(len(active_idx))
            betas = np.zeros(len(active_idx))
            for idx in range(len(active_idx)):
                k = int(active_idx[idx])
                diff = k - j
                if 0 <= diff < n:
                    if k == 2 * j:
                        alphas[idx] = 0.0
                        betas[idx] = c[k] - ai * ai
                    else:
                        alphas[idx] = 2.0 * a[diff]
                        betas[idx] = c[k] - alphas[idx] * ai
                else:
                    alphas[idx] = 0.0
                    betas[idx] = c[k]
            xlo = 0.0
            xhi = max(ai * 5.0, cm * 2.0)
            for _ in range(40):
                xm1 = xlo + (xhi - xlo) / 3.0
                xm2 = xhi - (xhi - xlo) / 3.0
                f1 = np.max(alphas * xm1 + betas + (0.0 if True else 0.0))
                if k == 2 * j:
                    pass
                vals1 = alphas * xm1 + betas
                for idx in range(len(active_idx)):
                    if int(active_idx[idx]) == 2 * j:
                        vals1[idx] = xm1 * xm1 + betas[idx]
                f1 = float(np.max(vals1))
                vals2 = alphas * xm2 + betas
                for idx in range(len(active_idx)):
                    if int(active_idx[idx]) == 2 * j:
                        vals2[idx] = xm2 * xm2 + betas[idx]
                f2 = float(np.max(vals2))
                if f1 < f2:
                    xhi = xm2
                else:
                    xlo = xm1
            xnew = (xlo + xhi) / 2.0
            a[j] = xnew
        s = a.sum()
        if s > 1e-10:
            a = a / s
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            a = best_a.copy()
            if sweep > 20 and best_score - _score_norm_fn(a) < 1e-12:
                break
    return best_a, best_score


def _kkt_balance_fast(a_init, deadline, max_iter=5000, top_k=15):
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a)
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        active = np.argsort(c)[-top_k:]
        h = np.zeros(n)
        for k_idx in active:
            k_val = int(k_val) if False else int(k_idx)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]
        h /= len(active)
        W = float(np.dot(h, a))
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        active_spread = float(c[active].max() - c[active].min()) + 1e-15
        step = 0.02 * active_spread / (cm + 1e-15)
        step = min(step, 0.08)
        denom = np.abs(h - W).max() + 1e-15
        z = z - step * (h - W) / denom
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a_new = _softmax_fn(z)
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.5
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - step * (h - W) / denom
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
        if it > 50 and (best_score - sc if np.isfinite(sc) else 0) < 1e-11:
            break
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t - np.pi)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, sigma):
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    global height_sequence_1
    t_start = time.monotonic()

    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Phase 0: Get parent
    print("=== Phase 0: Initialization ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")
        n_init = 4096
        parent = _make_raised_cosine(n_init)
        sc = _score_norm_fn(parent)
        update_best(parent, sc, "raised_cosine")

    n_parent = len(parent)

    # Phase 1: L-BFGS on parent
    print("=== Phase 1: L-BFGS on parent ===")
    phase1_end = min(deadline, t_start + 200.0)
    p_best, sc_best = _run_lbfgs(parent, phase1_end, max_wall_time=min(190.0, phase1_end - time.monotonic()), top_k=0)
    update_best(p_best, sc_best, "lbfgs_parent")

    # Phase 2: Aggressive Adam
    print("=== Phase 2: Aggressive Adam ===")
    phase2_end = min(deadline, t_start + 700.0)
    remaining = phase2_end - time.monotonic()
    if remaining > 30 and best_seq is not None:
        dl = min(phase2_end, time.monotonic() + remaining)
        p_best, sc_best = _adam_optimize(best_seq, dl, lr=0.06, top_k=0, n_max_iter=150000)
        update_best(p_best, sc_best, "adam1")
        if time.monotonic() < phase2_end - 30:
            dl2 = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
            p_best2, sc_best2 = _adam_optimize(best_seq, dl2, lr=0.02, top_k=10, n_max_iter=80000)

    # Phase 3: Coordinate descent polishing
    print("=== Phase 3: Coordinate descent ===")
    phase3_end = min(deadline, t_start + 1100.0)

        remaining = phase3_end - time.monotonic()
        if remaining > 20:
            p_best, sc_best = _coord_descent_polish(best_seq, min(phase3_end, time.monotonic() + remaining), max_sweeps=300, top_k=20)

    # Phase 4: KKT balancing
    print("=== Phase 4: KKT balancing ===")
