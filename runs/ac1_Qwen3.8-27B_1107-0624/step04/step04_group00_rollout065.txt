"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based updates, multiplicative weights, L-BFGS continuation,
subgradient momentum, and coordinate descent for fine polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_direct(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s > 1e-15:
        return a / s
    return a.copy()


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_score(a, n, beta):
    c = _full_convolve(a, a)
    cm = float(c.max())
    s = float(np.sum(np.exp((c - cm) / beta)))
    return 2.0 * n * (cm + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    w = np.exp((c - cm) / beta)
    s = w.sum()
    if s < 1e-300:
        return np.zeros(n)
    w = w / s
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 4.0 * n * a * h


def _smoothed_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    s = float(np.sum(np.exp((c - cm) / beta)))
    return 2.0 * n * (cm + beta * np.log(s))


def _lp_solve(a, M, time_limit=30):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            ks = np.full(len(js), k)
            vs = a[js]
            rows.extend(ks.tolist())
            cols.extend(js.tolist())
            vals.extend(vs.tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    if res.success and res.x is not None:
        if res.x.sum() > 1e-10:
            return res.x
    return None


def _mw_step(a, beta, alpha):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-15:
        return a.copy()
    w = np.exp((c - cm) / beta)
    s = w.sum()
    if s < 1e-300:
        return a.copy()
    w = w / s
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    a_new = a * np.exp(-alpha * h)
    a_new = np.maximum(a_new, 1e-30)
    return _normalize(a_new)


def _coord_descent(a, deadline):
    n = len(a)
    a = a.copy()
    c = _full_convolve(a, a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    iteration = 0
    while time.monotonic() < deadline:
        i = iteration % n
        iteration += 1

        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo > k_hi:
            continue
        k_range = np.arange(k_lo, k_hi + 1)
        partner = k_range - i
        if partner.min() < 0 or partner.max() >= n:
            continue

        a_i = a[i]
        c_aff = c[k_range]

        a_i_contrib = 2.0 * a_i * a[partner].astype(np.float64)
        mask_2i = (k_range == 2 * i)
        if mask_2i.any():
            idx2 = np.where(mask_2i)[0][0]
            a_i_contrib[idx2] = a_i * a_i

        base = c_aff - a_i_contrib

        all_mask = np.zeros(2 * n - 1, dtype=bool)
        all_mask[k_range] = True
        if (~all_mask).any():
            unaff_max = float(c[~all_mask].max())
        else:
            unaff_max = 0.0

        a_lo = 0.0
        a_hi = max(3.0 * a_i, 6.0 / n)
        partner_arr = a[partner].astype(np.float64)
        base_arr = base.astype(np.float64)
        has_2i = mask_2i.any()
        idx2_val = int(np.where(mask_2i)[0][0]) if has_2i else -1

        c_pt = a_hi - gr * (a_hi - a_lo)
        d_pt = a_lo + gr * (a_hi - a_lo)

        cn = base_arr + 2.0 * c_pt * partner_arr
        if has_2i:
            cn = cn.copy()
            cn[idx2_val] = base_arr[idx2_val] + c_pt * c_pt
        fc = max(unaff_max, float(cn.max()))

        dn = base_arr + 2.0 * d_pt * partner_arr
        if has_2i:
            dn = dn.copy()
            dn[idx2_val] = base_arr[idx2_val] + d_pt * d_pt
        fd = max(unaff_max, float(dn.max()))

        for _ in range(35):
            if fc < fd:
                a_hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = a_hi - gr * (a_hi - a_lo)
                cn = base_arr + 2.0 * c_pt * partner_arr
                if has_2i:
                    cn = cn.copy()
                    cn[idx2_val] = base_arr[idx2_val] + c_pt * c_pt
                fc = max(unaff_max, float(cn.max()))
            else:
                a_lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = a_lo + gr * (a_hi - a_lo)
                dn = base_arr + 2.0 * d_pt * partner_arr
                if has_2i:
                    dn = dn.copy()
                    dn[idx2_val] = base_arr[idx2_val] + d_pt * d_pt
                fd = max(unaff_max, float(dn.max()))

        optimal_ai = (a_lo + a_hi) * 0.5
        optimal_ai = max(0.0, optimal_ai)

        if abs(optimal_ai - a_i) > 1e-14:
            a[i] = optimal_ai
            c_new = base_arr + 2.0 * optimal_ai * partner_arr
            if has_2i:
                c_new = c_new.copy()
                c_new[idx2_val] = base_arr[idx2_val] + optimal_ai * optimal_ai
            c[k_range] = c_new
            new_max = max(unaff_max, float(c_new.max()))
            new_score = 2.0 * n * new_max
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    best_a = _normalize(best_a)
    best_score = _score_norm(best_a)
    return best_a, best_score


def _subgrad_momentum(a_init, deadline):
    n = len(a_init)
    a = _normalize(a_init.copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm(a)
    gamma = 0.8
    eta = 0.003
    t = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        indices = np.arange(n)
        partners = k_star - indices
        valid = (partners >= 0) & (partners < n)
        g = np.zeros(n)
        g[valid] = 2.0 * a[partners[valid]]

        v = gamma * v + g
        a = a - eta * v
        a = np.maximum(a, 1e-15)
        a = _normalize(a)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        t += 1
        if t % 200 == 0:
            eta = max(1e-6, eta * 0.97)
        if t % 3000 == 0:
            v *= 0.5

    return best_a, best_score


def _lbfgs_phase(a_init, deadline):
    n = len(a_init)
    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    n_betas = 15
    log_betas = np.linspace(np.log(0.08 * cm0), np.log(1e-5 * cm0), n_betas)

    for idx in range(n_betas):
        if time.monotonic() > deadline:
            break
        beta = log_betas[idx]
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        level_dl = time.monotonic() + min(time_per, 100.0)

        maxiter = 3000
        maxfun = 8000
        if time_per > 60:
            maxiter = 5000
            maxfun = 15000

        try:
            res = minimize(_smoothed_obj, z, args=(beta, n), jac=_smoothed_grad,
                           method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxls': 50, 'maxfun': maxfun})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _make_profiles(n, rng):
    t = np.linspace(0, 1, n)
    profiles = []
    for slope in [1.0, 2.0, 3.0, -1.5, -2.5]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(_normalize(a))
    for rate in [2.0, 4.0, -2.0, -4.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(_normalize(a))
    for p in [1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * (2 * t - 1) / 2), 1e-10) ** p
        profiles.append(_normalize(a))
    for freq in [2, 3, 5]:
        for amp in [0.4, 0.6]:
            a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
            a = np.maximum(a, 1e-10)
            profiles.append(_normalize(a))
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return _normalize(p)
