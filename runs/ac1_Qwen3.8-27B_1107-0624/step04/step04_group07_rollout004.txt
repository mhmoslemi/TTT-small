import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _build_conv_matrix(n, a):
    """Build sparse matrix A such that (A @ b)[k] = (a * b)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_solve(a, n, M):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _subgradient_step(a, lr):
    """One Nesterov-style subgradient step on the simplex."""
    n = len(a)
    c = _full_convolve(a)
    m_star = int(np.argmax(c))
    # Gradient of (a*a)[m_star] w.r.t. a
    # grad[j] = 2 * a[m_star - j] (with a[x] = 0 for x outside [0,n-1])
    grad = np.zeros(n)
    for j in range(n):
        idx = m_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    # Projected subgradient step
    a_new = a - lr * grad
    # Project onto simplex {x >= 0, sum(x) = 1}
    a_new = _project_simplex(a_new)
    return a_new

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

def _coord_descent(a, deadline, max_sweeps=20):
    """Coordinate descent: try scaling individual elements."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            old = a[j]
            for f in [0.7, 0.85, 1.15, 1.3]:
                a[j] = max(old * f, 1e-15)
                a[j] = a[j]  # Keep sum approximately 1 (we'll renorm at end)
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old
                    break
            a[j] = old
        if not improved:
            break
    
    return best_a, best_score

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new
