def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    a_rev = a[::-1]
    conv = _full_convolve(a_rev, w)
    return conv[n-1:2*n-1]

        # This is equivalent to: conv(a, b)[k] <= M for all k
        # But conv(a,b) has length 2n-1.

        # But scipy linprog doesn't support matrix-free...

        rows = []
        cols = []
        data = []
        for j in range(n):

            # Wait, c_k = sum_i a_i * a_{k-i}. a_j appears when i=j (a_j*a_{k-j}) or when k-i=j (a_{k-j}*a_j)
            # So a_j appears in c_k for k where k-j is valid (0<=k-j<n) OR k-j is valid for the other factor
            # Actually: c_k includes the term a_j * a_{k-j} (from i=j) if 0<=k-j<n
            # AND the term a_{k-j} * a_j (from i=k-j) if 0<=k-j<n
            # These are the same condition: 0<=k-j<n, i.e., j<=k<=j+n-1
            # So a_j appears in c_k for k in [j, j+n-1]
            # And the coefficient is a_{k-j} (from first term) + a_{k-j} (from second term) = 2*a_{k-j}
            # Wait no. The constraint is sum_j b_j * (coefficient of b_j in c_k) <= M
            # c_k = sum_i sum_j delta(i+j, k) * a_i * b_j ... no, that's not right.
            
            # Let me think more carefully. c = conv(a, b) where a and b are the vectors.
            # c_k = sum_i a_i * b_{k-i}
            # The constraint is c_k <= M for all k.
            # So: sum_i a_i * b_{k-i} <= M
            # In matrix form: A[k, j] = a_{k-j} if 0<=k-j<n else 0
            # A @ b <= M * ones

"""
L-BFGS-B based optimization with smooth max (log-sum-exp), multi-restart,
LP-guided refinement, and multi-scale exploration to minimize
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements over previous Adam-based approach: quasi-Newton optimizer,
better beta continuation, LP refinement, and symmetric constraint exploitation.

# Global state for L-BFGS-B objective function
_g_n = 0
_g_beta = 1.0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _cross_corr_w_a(w, a):
    n = len(a)
    a_rev = a[::-1]
    conv = _full_convolve(a_rev, w)
    return conv[n - 1:2 * n - 1]


def _compute_fg(z):
    n = _g_n
    beta = _g_beta
    a = _softmax(z)
    S = float(a.sum())
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        soft_max = cm
    else:
        w = e / s
        soft_max = cm + beta * np.log(s)
    h = _cross_corr_w_a(w, a)
    F = 2.0 * n * soft_max / (S * S)
    dF_da = 4.0 * n * (h * S - soft_max) / (S ** 3)
    grad = a * dF_da
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    if not np.isfinite(F):
        F = 1e10
    return F, grad

    return _compute_fg(z)
