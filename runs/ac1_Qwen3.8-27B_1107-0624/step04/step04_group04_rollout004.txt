    n = len(a)
    a = a / a.sum()
    best = a.copy()
    best_score = score_fn(a)
    
    while time.monotonic() < deadline:
        c = convolve(a, a)
        cm = c.max()
        threshold = cm * (1 - active_frac)
        active = np.where(c >= threshold)[0]
        
        # Compute pressure
        g = np.zeros(n)
        for k in active:
            # g_j += a_{k-j} if valid
            # j ranges from max(0, k-(n-1)) to min(n-1, k)
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            # a_{k-j} for j in [lo, hi] is a[k-hi : k-lo+1] reversed
            g[lo:hi+1] += a[k-hi:k-lo+1][::-1]

        ratio = target / (g + 1e-30)
        a = a * np.power(ratio, step)
        a = a / a.sum()
        
        sc = score_fn(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()

def _subgradient_line_search(a_init, deadline, n_iter=5000):

    Uses the subgradient at the argmax position.
    """

    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.1
    momentum = np.zeros(n)
    beta_mom = 0.9

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        # Subgradient of max(c) at k_star: grad_c_j = 2*a[k_star - j] if valid
        grad = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        grad[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Momentum
        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        direction = -momentum

        # Project: remove mean to maintain normalization in log-space
        direction -= direction.mean()

        # Backtracking line search
        best_z = z.copy()
        best_sc = _score_norm_fn(a)
        for attempt in range(10):
            z_trial = z + lr * direction
            z_trial = np.clip(z_trial, -80, 80)

            a_trial = _softmax_fn(z_trial)

                best_z = z_trial.copy()
                best_sc = sc_trial
                break
            lr *= 0.5

        z = best_z
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)

        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if it % 100 == 0:

                return 2.0 * n * (c_top.max() + beta * np.log(s))

                s = float(e.sum())

                return 2.0 * n * (cm + beta * np.log(s))

        def grad(z_in):

    return a / a.sum()


def _make_bimodal_profile(n, sep=0.3, width=0.1):
    """Two Gaussian bumps."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5 + sep) / width) ** 2) + np.exp(-((t - 0.5 - sep) / width) ** 2)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: peak-shaving multiplicative updates,
    KKT balancing with FFT pressure, subgradient line search,
    L-BFGS logsumexp, and Adam refinement.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {sc:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initialization ---
    print("=== Phase 0: Initialization ===")
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    n_main = 4096

    if parent is None:
        parent = _make_exp_profile(n_main, 0.001)
        sc = _score_norm_fn(parent)
        update_best(parent, sc, "init")

    # If parent has different length, interpolate
    if len(parent) != n_main:
        parent_interp = _interp_profile(parent, n_main)
        sc = _score_norm_fn(parent_interp)
        update_best(parent_interp, sc, "parent_interp")
        parent = parent_interp

    print(f"  Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: Peak-shaving multiplicative updates (main new method) ---
    print("=== Phase 1: Peak-shaving ===")
    phase1_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase1_end:
        remaining = phase1_end - time.monotonic()
        dl = min(phase1_end, time.monotonic() + remaining)

        # Try different parameter settings
        for init_f, min_f, init_s, cool_r in [
            (0.1, 0.001, 0.5, 0.99),
            (0.05, 0.001, 0.3, 0.995),
            (0.15, 0.002, 0.7, 0.99),
        ]:
            if time.monotonic() > phase1_end - 30:
                break
            per_run = min(100.0, (phase1_end - time.monotonic()) / 3)
            dl_run = min(phase1_end - 5, time.monotonic() + per_run)
            a_norm = best_seq / best_seq.sum()
            p_best, sc_best = _peak_shave(a_norm, dl_run,
                                          init_frac=init_f, min_frac=min_f,
                                          init_step=init_s, cool_rate=cool_r)
            update_best(p_best, sc_best, f"peak_f{init_f}_s{init_s}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: KKT balancing with FFT pressure ---
    print("=== Phase 2: KKT balancing ===")
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + remaining)
        a_norm = best_seq / best_seq.sum()
        p_best, sc_best = _active_set_kkt(a_norm, dl, max_iter=50000)
        update_best(p_best, sc_best, "kkt_fft")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: L-BFGS polishing ---
    print("=== Phase 3: L-BFGS ===")
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + remaining)
        a_norm = best_seq / best_seq.sum()
        # Run with different top_k values
        for tk in [0, 3, 10]:
            if time.monotonic() > phase3_end - 30:
                break
            per_run = min(80.0, (phase3_end - time.monotonic()) / 3)
            dl_run = min(phase3_end - 5, time.monotonic() + per_run)
            p_best, sc_best = _run_lbfgs(a_norm, dl_run, top_k=tk)
            update_best(p_best, sc_best, f"lbfgs_k{tk}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Subgradient line search ---
    print("=== Phase 4: Subgradient ===")
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        remaining = phase4_end - time.monotonic()
        dl = min(phase4_end, time.monotonic() + remaining)
        a_norm = best_seq / best_seq.sum()
        p_best, sc_best = _subgradient_line_search(a_norm, dl, n_iter=10000)
        update_best(p_best, sc_best, "subgrad")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Adam refinement ---
    print("=== Phase 5: Adam refine ===")
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        remaining = phase5_end - time.monotonic()
        dl = min(phase5_end, time.monotonic() + remaining)
        a_norm = best_seq / best_seq.sum()
        p_best, sc_best = _adam_refine(a_norm, dl, lr=0.02, n_max_iter=80000)
        update_best(p_best, sc_best, "adam_refine")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Multi-restart from best ---
    print("=== Phase 6: Multi-restart ===")
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = len(best_seq)
        for i in range(3):
            if time.monotonic() > phase6_end - 30:
                break
            noise_levels = [0.03, 0.08, 0.15]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i] * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_restart = min(50.0, (phase6_end - 30 - time.monotonic()) / (3 - i))
            # Quick peak shave + L-BFGS
            dl_ps = min(phase6_end - 20, time.monotonic() + per_restart * 0.5)
            p_m, sc_m = _peak_shave(p_start, dl_ps, init_frac=0.05,
                                     min_frac=0.005, init_step=0.2, cool_rate=0.99)
            update_best(p_m, sc_m, f"r{i}_ps")
            if time.monotonic() < phase6_end - 10:
                dl_lb = min(phase6_end - 10, time.monotonic() + per_restart * 0.4)
                p_m2, sc_m2 = _run_lbfgs(p_m, dl_lb, top_k=5)
                update_best(p_m2, sc_m2, f"r{i}_lbfgs")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Try different lengths ---
    print("=== Phase 7: Different lengths ===")
    phase7_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase7_end and best_seq is not None:
        n_curr = len(best_seq)
        for tn in [2048, 8192]:
            if time.monotonic() > phase7_end - 40:
                break
            if tn == n_curr:
                continue
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")
            if time.monotonic() < phase7_end - 20:
                dl = min(phase7_end - 10, time.monotonic() + min(60.0,
                         (phase7_end - 10 - time.monotonic()) * 0.5))
                p_best, sc_best = _active_set_kkt(p_interp, dl, max_iter=10000)
                update_best(p_best, sc_best, f"kkt_n{tn}")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Final polish ---
    print("=== Phase 8: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_norm = best_seq / best_seq.sum()
            dl = min(deadline - 5, time.monotonic() + min(40.0, remaining * 0.4))
            p_best, sc_best = _peak_shave(a_norm, dl, init_frac=0.01,
                                           min_frac=0.0005, init_step=0.05, cool_rate=0.999)
            update_best(p_best, sc_best, "final_ps")

            if time.monotonic() < deadline - 10:
                dl2 = min(deadline - 5, time.monotonic() + min(30.0,
                        deadline - time.monotonic() - 5))
                p_best2, sc_best2 = _adam_refine(best_seq / best_seq.sum(), dl2,
                                                  lr=0.005, n_max_iter=20000)
                update_best(p_best2, sc_best2, "final_adam")

    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.001)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate check failed: {e}")

        return [float(x) for x in final_seq]

    fallback = _make_exp_profile(4096, 0.001)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
