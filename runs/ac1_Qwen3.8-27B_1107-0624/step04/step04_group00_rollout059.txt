"""
Coordinate descent optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses O(n) incremental convolution updates per coordinate step, with pair-balancing
heuristic for the 1D subproblem. Multiple restarts and block refinement.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_delta(a, conv, i, dv):
    """Compute the change in conv when a[i] is increased by dv. Returns new conv."""
    n = len(a)
    conv_new = conv.copy()
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    idx_lo = k_lo - i
    idx_hi = k_hi - i
    conv_new[k_lo:k_hi + 1] += 2.0 * dv * a[idx_lo:idx_hi + 1]
    two_i = 2 * i
    if k_lo <= two_i <= k_hi:
        conv_new[two_i] += dv * dv
    return conv_new


def _coord_descent_fast(a, deadline, verbose=False, max_sweeps=5000):
    """Fast cyclic coordinate descent with pair-balancing heuristic."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a /= s
    conv = _full_convolve(a, a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    stall = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_improved = False

        for i in range(n):
            v = a[i]
            if v < 1e-10:
                continue

            c_max = float(conv.max())
            k_star = int(np.argmax(conv))

            # Sensitivity of each lag to a[i]
            k_other = k_star - i
            if 0 <= k_other < n:
                rate = 2.0 * (v if k_other == i else a[k_other])
            else:
                rate = 0.0

            # Build candidates
            candidates = []
            if abs(rate) > 1e-15:
                # Pair-balancing: equalize top 2 lags
                # Find second max
                k_star2 = int(np.argmax(np.where(np.arange(2*n-1) == k_star, -1, conv)))
                c_second = float(conv[k_star2])

                # dv to equalize k_star and k_star2
                k2_other = k_star2 - i
                if 0 <= k2_other < n:
                    rate2 = 2.0 * (v if k2_other == i else a[k2_other])
                else:
                    rate2 = 0.0

                denom = rate - rate2
                if abs(denom) > 1e-15:
                    dv_bal = (c_second - c_max) / denom
                    dv_bal = float(np.clip(dv_bal, -v * 0.95, v * 20.0))
                    candidates.append(dv_bal)

                # Fractional steps
                if rate > 0:
                    for f in [0.01, 0.05, 0.1, 0.2, 0.4, 0.7]:
                        candidates.append(-v * f)
                else:
                    for f in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0]:
                        candidates.append(v * f)

            # Also try small perturbations in both directions
            candidates.append(-v * 0.005)
            candidates.append(v * 0.005)
            candidates.append(-v * 0.5)

            best_dv = 0.0
            best_sc = 2.0 * n * c_max
            best_conv_new = None

            for dv in candidates:
                if v + dv < -1e-12:
                    continue
                new_sum = 1.0 + dv
                if new_sum <= 1e-10:
                    continue
                conv_new = _conv_delta(a, conv, i, dv)
                cm = float(conv_new.max())
                sc = 2.0 * n * cm / (new_sum * new_sum)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_dv = dv
                    best_conv_new = conv_new

            if best_dv != 0.0 and best_conv_new is not None:
                a[i] = v + best_dv
                conv = best_conv_new
                sweep_improved = True

        # Periodic recompute to avoid drift
        if sweep % 100 == 99:
            a = a / a.sum()
            conv = _full_convolve(a, a)

        # Track best
        cur_sc = 2.0 * n * float(conv.max()) / (a.sum() ** 2)
        if cur_sc < best_score - 1e-13:
            best_score = cur_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and sweep % 200 == 0:
            print(f"      CD sweep {sweep}: best={best_score:.10f}, cur={cur_sc:.10f}")

        if stall > 30 and not sweep_improved:
            break

    return best_a, best_score


def _block_descent(a, deadline, block_size=8, max_sweeps=2000):
    """Block coordinate descent: update groups of coordinates together."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    a /= s
    conv = _full_convolve(a, a).copy()
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    stall = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_improved = False

        for block_start in range(0, n, block_size):
            block = list(range(block_start, min(block_start + block_size, n)))
            # For each coordinate in the block, do a coordinate step
            for i in block:
                v = a[i]
                if v < 1e-10:
                    continue
                c_max = float(conv.max())
                k_star = int(np.argmax(conv))
                k_other = k_star - i
                if 0 <= k_other < n:
                    rate = 2.0 * (v if k_other == i else a[k_other])
                else:
                    rate = 0.0

                if abs(rate) < 1e-15:
                    continue

                # Try a few candidates
                candidates = []
                if rate > 0:
                    for f in [0.02, 0.05, 0.1, 0.2, 0.5]:
                        candidates.append(-v * f)
                else:
                    for f in [0.02, 0.05, 0.1, 0.3, 1.0]:
                        candidates.append(v * f)

                best_dv = 0.0
                best_sc = 2.0 * n * c_max
                best_cnew = None

                for dv in candidates:
                    if v + dv < -1e-12:
                        continue
                    ns = 1.0 + dv
                    if ns <= 1e-10:
                        continue
                    cnew = _conv_delta(a, conv, i, dv)
                    sc = 2.0 * n * float(cnew.max()) / (ns * ns)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_dv = dv
                        best_cnew = cnew

                if best_dv != 0.0 and best_cnew is not None:
                    a[i] = v + best_dv
                    conv = best_cnew
                    sweep_improved = True

        if sweep % 50 == 49:
            a = a / a.sum()
            conv = _full_convolve(a, a)

        cur_sc = 2.0 * n * float(conv.max()) / (a.sum() ** 2)
        if cur_sc < best_score - 1e-13:
            best_score = cur_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 20 and not sweep_improved:
            break

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    # Power-law edge decay
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = np.power(1 - np.abs(u), p)
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())

    # Linear ramp (asymmetric)
    for slope in [0.5, 1.0, 1.5, 2.0, -0.5, -1.0, -1.5, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())

    # Cosine window
    for width in [0.5, 0.7, 0.8, 0.9, 1.0]:
        a = np.maximum(np.cos(np.pi * u / (2 * width)), 0) ** 2
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Bimodal (two peaks)
    for gap in [0.2, 0.3, 0.4, 0.5]:
        a = np.exp(-((t - 0.5 + gap / 2) ** 2) / 0.01) + np.exp(-((t - 0.5 - gap / 2) ** 2) / 0.01)
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()
