from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def _solve_lp(a, M):
    """Solve the LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0"""
    n = len(a)
    n_constraints = 2 * n - 1
    
    # Build constraint matrix A (sparse)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(n_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=np.full(n_constraints, M), bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

import numpy as np

from scipy.sparse import csr_matrix
import time

# Helper functions (top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm

def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)

        return np.zeros_like(z)
    return grad

def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _solve_lp(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0"""
    n = len(a)
    n_k = 2 * n - 1
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(n_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(n_k, n))
    c_obj = -np.ones(n)
    b_ub = np.full(n_k, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'time_limit': 30})
    
    if result.success:
        return result.x
    return None

                    best_a = a_new.copy()

"""
LP-based coordinate descent with smooth optimization for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Uses the key insight that fixing one sequence and optimizing the other
is a linear program, enabling direct global optimization steps.
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _build_conv_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid, else 0.
    Shape: (2n-1, n)"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp(a, M, time_limit=25):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns b or None if failed."""
    n = len(a)
    n_k = 2 * n - 1

    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(n_k, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': time_limit,
                                                   'presolve': True})
        if result.success or result.status in [1, 3]:
            if result.x is not None and np.all(np.isfinite(result.x)):
                return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def _active_set_subgradient(a):
    """Compute subgradient using active set (near-max lags). Returns (grad, max_val)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())

    # Use lags within 0.5% of max as active set
    threshold = cm * 0.998
    active = c >= threshold
    if not np.any(active):
        active[int(np.argmax(c))] = True

    w = np.zeros_like(c)
    w[active] = 1.0 / float(np.sum(active))

    # Gradient: d/da_i [sum_k w_k (a*a)[k]] = 2 * (a * flip(w))_i restricted to [0,n)
    w_flip = w[::-1]
    grad_conv = _full_convolve(a, w_flip)
    grad = 2.0 * grad_conv[:n]

    return grad, cm

def _interp_profile(p0, n_new):

                else:
                    consecutive_fail += 1
            else:
                consecutive_fail += 1

            if consecutive_fail > 5:
                # Do a smooth gradient step to escape
                z = np.log(np.maximum(a, 1e-15))
                z = z - z.mean()
                beta = 0.005 * M
                grad = _gradient_smooth(z, beta)
                z = z - 0.02 * grad
                z = z - z.mean()
                z = np.clip(z, -50, 50)
                a_smooth = _softmax_fn(z)
                sc_smooth = _score_norm_fn(a_smooth)
                if sc_smooth < _score_norm_fn(a):
                    a = a_smooth
                    M = float(_full_convolve(a, a).max())
                    _try_update(a, sc_smooth, f"escape_{lp_count}")
                consecutive_fail = 0
                t_step = 0.1

        print(f"  Phase 1 done: {lp_count} LP steps, best={best_score:.10f}")

        # Update best if LP improved
        sc_a = _score_norm_fn(a)
        _try_update(a, sc_a, "lp_final")

    # --- Phase 2: Nesterov polishing (300s) ---
    phase2_end = min(deadline, t_start + 1020.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
        a_opt, sc_opt = _nesterov_optimize(best_seq, phase2_end, beta_fracs,
                                           lr_init=0.01, mom_init=0.92)
        _try_update(a_opt, sc_opt, "nesterov_p2")

    # --- Phase 3: Adam with very small beta (250s) ---
    phase3_end = min(deadline, t_start + 1270.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        a_opt, sc_opt = _adam_optimize(best_seq, phase3_end, lr=0.02, beta_frac=0.002)
        _try_update(a_opt, sc_opt, "adam_p3")

    # --- Phase 4: Active set subgradient (200s) ---
    phase4_end = min(deadline, t_start + 1470.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        a_opt, sc_opt = _active_set_optimize(best_seq, phase4_end, lr_init=0.002)
        _try_update(a_opt, sc_opt, "active_p4")

    # --- Phase 5: Multiple restarts with LP + smooth (250s) ---
    phase5_end = min(deadline, t_start + 1720.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        a0 = best_seq.copy()
        n_rest = 6
        per_restart = (phase5_end - time.monotonic()) / max(1, n_rest)

        for i in range(n_rest):
            if time.monotonic() > phase5_end - 15:
                break
            z = np.log(np.maximum(a0, 1e-15))
            z = z - z.mean()
            noise = rng.normal(size=len(z))
            z = z + 0.03 * noise
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a_start = _softmax_fn(z)

            per_dl = min(phase5_end - 10, time.monotonic() + per_restart - 5)

            # Quick LBFGS
            a_opt, sc_opt = _lbfgs_optimize(a_start, per_dl, max_wall_time=per_dl - time.monotonic() - 5)
            _try_update(a_opt, sc_opt, f"restart_{i}")

    # --- Phase 6: Final fine polish (remaining) ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            # Very fine active set
            fine_end = min(deadline - 5, time.monotonic() + remaining * 0.6)
            a_opt, sc_opt = _active_set_optimize(best_seq, fine_end, lr_init=0.001)
            _try_update(a_opt, sc_opt, "final_active")

        if time.monotonic() < deadline - 5:
            # Final Nesterov with tiny beta
            fine_end2 = deadline - 5
            a_opt, sc_opt = _nesterov_optimize(best_seq, fine_end2,
                                               [1e-5, 5e-6, 2e-6],
                                               lr_init=0.003, mom_init=0.95)
            _try_update(a_opt, sc_opt, "final_nest")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(4096, "linear")

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_profile(4096, "linear")
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
