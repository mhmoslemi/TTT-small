"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines exact sparse LP direction finding (via scipy.linprog + CSR) with
Adam-style sub-gradient polishing using an adaptive active set (softmax over
top-k convolution peaks), KKT balancing polishing, multi-restart in logit
space, and multi-length exploration. All helpers are top-level; no closures
or lambdas are used. Returns the best non-negative sequence found.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / float(len(z))
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(np.max(c)) / (s * s))


def _score_norm_fn(a):
    """Score under the assumption sum(a) == 1 (normalization invariant)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(np.max(c)))


def _normalize_nonneg(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, None)
    s = float(p.sum())
    if s < 1e-15:
        return None
    return p / s


def _make_gaussian_profile(n, sigma_frac):
    center = (n - 1) / 2.0
    t = np.arange(n, dtype=np.float64) - center
    a = np.exp(-0.5 * (t / (sigma_frac * n)) ** 2)
    return a / a.sum()


def _make_cosine_profile(n, beta):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + beta * np.cos(2.0 * np.pi * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, sep_frac, width_frac):
    c1 = (n - 1) * sep_frac
    c2 = (n - 1) * (1.0 - sep_frac)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - c1) / (width_frac * n)) ** 2) + \
        np.exp(-0.5 * ((t - c2) / (width_frac * n)) ** 2)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exponential_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _make_random_smooth_profile(n, seed):
    rng = np.random.default_rng(seed)
    base = rng.uniform(0.5, 1.5, size=n)
    # Smooth with a moving average to reduce high-frequency content
    kernel = np.exp(-0.5 * (np.linspace(-3, 3, 15)) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(base, kernel, mode='same')
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = float(p0.sum())
        if s < 1e-15:
            return None
        return p0 / s
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, np.maximum(p0, 0.0))
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _build_lp_matrix(f, n):
    """Sparse matrix A with A[k, j] = f[k - j] if 0 <= k - j < n else 0."""
    max_row = 2 * n - 1
    row_list = []
    col_list = []
    val_list = []
    for k in range(max_row):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        n_entries = j_end - j_start + 1
        if n_entries > 0:
            row_list.extend([k] * n_entries)
            col_list.extend(range(j_start, j_end + 1))
            val_list.extend(f[k - j_start:k - j_start + n_entries][::-1])
    A = csr_matrix((val_list, (row_list, col_list)), shape=(max_row, n))
    return A


def _lp_direction(f, deadline):
    """Solve the LP to find the best non-negative direction g (sum=1) such that
    (f * g)_k <= M for all k. Returns g normalized to sum 1, or None on failure."""
    n = len(f)
    c = _full_convolve(f, f)
    M = float(np.max(c))
    if M < 1e-15:
        return f.copy()
    remaining = deadline - time.monotonic()
    if remaining < 2.0:
        return None
    try:
        A = _build_lp_matrix(f, n)
        b = M * np.ones(2 * n - 1, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        result = linprog(
            c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
            options={'presolve': True, 'time_limit': max(1.0, remaining - 2.0)}
        )
        if result.success and result.x is not None:
            g = np.asarray(result.x, dtype=np.float64)
            s = float(g.sum())
            if s > 1e-10 and np.all(np.isfinite(g)):
                return g / s
    except Exception:
        pass
    return None


def _active_weights(c, beta, top_k):
    """Return weight vector w over convolution positions (softmax over top-k)."""
    L = len(c)
    w = np.zeros(L, dtype=np.float64)
    if top_k > 0 and top_k < L:
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        z = (c_top - c_top.max()) / beta
        e = np.exp(z)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w[idx[0]] = 1.0
        else:
            w[idx] = e / s
    else:
        cm = float(c.max())
        z = (c - cm) / beta
        e = np.exp(z)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    return w


def _weighted_h(a, w):
    """h_j = sum_k w_k * a[k - j] for valid indices (the sub-gradient direction)."""
    n = len(a)
    h = np.zeros(n, dtype=np.float64)
    active_idx = np.where(w > 1e-12)[0]
    for ki in active_idx:
        k = int(ki)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            h[j_start:j_end + 1] += w[ki] * a[k - j_end:k - j_start + 1][::-1]
    return h


def _subgrad_step(a, beta, top_k, step_size):
    n = len(a)
    c = _full_convolve(a, a)
    w = _active_weights(c, beta, top_k)
    h = _weighted_h(a, w)
    z_log = np.log(np.maximum(a, 1e-15))
    z_log = z_log - float(z_log.mean())
    grad_log = h / np.maximum(a, 1e-10)
    z_log = z_log - step_size * grad_log
    z_log = np.clip(z_log, -60.0, 60.0)
    z_log = z_log - float(z_log.mean())
    return _softmax_fn(z_log), h


def _adam_subgrad(a_init, deadline, lr=0.04, top_k=15, beta_init=0.01, n_max_iter=20000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - float(z.mean())
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m_vec = np.zeros(n, dtype=np.float64)
    v_vec = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    c0 = _full_convolve(a_init, a_init)
    beta = max(1e-12, beta_init * float(c0.max()))
    cur_k = top_k
    for _ in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w = _active_weights(c, beta, cur_k)
        h = _weighted_h(a, w)
        grad_log = 2.0 * h / np.maximum(a, 1e-10)
        if not np.all(np.isfinite(grad_log)):
            grad_log = np.zeros(n, dtype=np.float64)
        m_vec = b1 * m_vec + (1.0 - b1) * grad_log
        v_vec = b2 * v_vec + (1.0 - b2) * (grad_log * grad_log)
        mh = m_vec / (1.0 - b1 ** t)
        vh = v_vec / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z = z - float(z.mean())
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 50 == 0:
            beta = max(1e-14, beta * 0.92)
            if cur_k < 50 and t % 200 == 0:
                cur_k = min(cur_k + 3, 50)
        if t % 500 == 0 and best_score < 1.51:
            lr = max(0.005, lr * 0.95)
    return best_a, best_score


def _lp_iterate(a_init, deadline, lp_every=3, max_iters=400):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a_init)
    t_blend = 0.3
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if it % lp_every == 0:
            g = _lp_direction(a, deadline)
            if g is None:
                g = a.copy()
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            s_new = float(a_new.sum())
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc):
                    a = a_new
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
        else:
            c = _full_convolve(a, a)
            beta = 0.01 * float(c.max()) + 1e-12
            a_candidate, _ = _subgrad_step(a, beta, 15, 0.02)
            sc = _score_norm_fn(a_candidate)
            sc_cur = _score_norm_fn(a)
            if np.isfinite(sc) and sc < sc_cur:
                a = a_candidate
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        if it % 20 == 0:
            t_blend = max(0.1, t_blend * 0.95)
    return best_a, best_score


def _kkt_refine(a_init, deadline, max_iter=5000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a_init)
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        if cm < 1e-15:
            break
        threshold = cm * max(0.95, 0.999 - it * 1e-6)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))], dtype=np.int64)
        h = np.zeros(n, dtype=np.float64)
        for ki in active:
            k = int(ki)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                h[j_start:j_end + 1] += a[k - j_end:k - j_start + 1][::-1]
        h /= float(len(active))
        W = float(np.dot(h, a))
        z = np.log(np.maximum(a, 1e-15))
        z = z - float(z.mean())
        active_spread = float(c[active].max() - c[active].min()) + 1e-15
        step = 0.005 * active_spread / (cm + 1e-15)
        step = min(step, 0.03)
        denom = float(np.abs(h - W).max()) + 1e-15
        z = z - step * (h - W) / denom
        z = np.clip(z, -60.0, 60.0)
        z = z - float(z.mean())
        a_new = _softmax_fn(z)
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.5
            z = np.log(np.maximum(a, 1e-15))
            z = z - float(z.mean())
            z = z - step * (h - W) / denom
            z = np.clip(z, -60.0, 60.0)
            z = z - float(z.mean())
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search:
      Phase 0: gather diverse initial profiles + parent, pick best.
      Phase 1: LP-guided direction iteration (sparse LP for exact direction).
      Phase 2: Adam sub-gradient + KKT polishing on the best sequence.
      Phase 3: multi-restart with logit noise + short polish.
      Phase 4: explore alternate lengths via interpolation + short polish.
    Returns the best non-negative sequence as a list of floats.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(15.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Phase 0: initial candidates
    print("=== Phase 0: Initial candidates ===")
    phase0_end = min(deadline, t_start + 35.0)

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_norm = _normalize_nonneg(parent_raw)
            if parent_norm is not None:
                parent = parent_norm
                sc = _score_norm_fn(parent)
                if sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print("  Parent: n=%d, score=%.10f" % (len(parent), sc))
    except Exception as e:
        print("  No parent available: %s" % str(e))

    # Decide base length: use parent length if reasonable, else 2048
    if parent is not None and 512 <= len(parent) <= 4096:
        n_base = len(parent)
    else:
        n_base = 2048

    initial_profiles = []
    for sf in (0.2, 0.3, 0.4, 0.5, 0.7):
        p = _make_gaussian_profile(n_base, sf)
        sc = _score_norm_fn(p)
        initial_profiles.append((sc, p))
    for beta in (0.3, 0.5, 0.7, 1.0):
        p = _make_cosine_profile(n_base, beta)
        sc = _score_norm_fn(p)
        initial_profiles.append((sc, p))
    for sep, wf in ((0.2, 0.1), (0.25, 0.15), (0.3, 0.12), (0.2, 0.18)):
        p = _make_bimodal_profile(n_base, sep, wf)
        sc = _score_norm_fn(p)
        initial_profiles.append((sc, p))
    for slope in (-2.0, -1.0, 0.5, 1.0, 2.0):
        p = _make_linear_profile(n_base, slope)
        sc = _score_norm_fn(p)
        initial_profiles.append((sc, p))
    for alpha in (0.001, 0.003, 0.005, 0.01, 0.02):
        p = _make_exponential_profile(n_base, alpha)
        sc = _score_norm_fn(p)
        initial_profiles.append((sc, p))
    for sd in (1, 2, 3, 4, 5):
        p = _make_random_smooth_profile(n_base, int(seed) + sd)
        sc = _score_norm_fn(p)
        initial_profiles.append((sc, p))

    # Track top candidate among profiles
    top_sc = np.inf
    top_p = None
    for sc, p in initial_profiles:
        if np.isfinite(sc) and sc < top_sc:
            top_sc = sc
            top_p = p
        if time.monotonic() > phase0_end:
            break

    if top_p is not None:
        if top_sc < best_score:
            best_score = top_sc
            best_seq = top_p.copy()
        # Keep top_p as a working seed for LP phase (might be better shape even if parent was numerically slightly better)
        lp_seed = top_p
    else:
        lp_seed = (parent if parent is not None else _make_gaussian_profile(n_base, 0.4))

    if parent is not None and len(parent) != n_base:
        p_interp = _interp_profile(parent, n_base)
        if p_interp is not None:
            sc = _score_norm_fn(p_interp)
            if sc < best_score:
                best_score = sc
                best_seq = p_interp.copy()
            if sc < top_sc:
                top_sc = sc
                lp_seed = p_interp

    print("  Phase 0 done: best=%.10f" % best_score)

    # Phase 1: LP-guided optimization
    print("=== Phase 1: LP-guided optimization ===")
    phase1_end = min(deadline, t_start + 850.0)
    if (best_seq is not None and lp_seed is not None
            and time.monotonic() < phase1_end - 30.0):
        # Use a moderate length for the LP (manageable sparse matrix)
        n_lp = len(lp_seed)
        if n_lp > 2048:
            p_lp = _interp_profile(lp_seed, 2048)
            if p_lp is not None:
                lp_seed = p_lp
                n_lp = 2048
        if n_lp < 256:
            n_lp = len(lp_seed)

        remaining = phase1_end - time.monotonic()
        dl = min(phase1_end, time.monotonic() + remaining)
        print("  LP iterate: n=%d, score=%.10f, budget=%.1fs"
              % (n_lp, _score_norm_fn(lp_seed), remaining))
        p_best, sc_best = _lp_iterate(lp_seed, dl, lp_every=3, max_iters=300)
        if np.isfinite(sc_best) and sc_best < best_score:
            best_score = sc_best
            best_seq = p_best.copy()
            print("  LP improved: %.10f (n=%d)" % (sc_best, len(p_best)))

        # If LP ran at a reduced length, also lift back to best_seq length and polish
        if (best_seq is not None and len(p_best) != len(best_seq)) and \
           time.monotonic() < phase1_end - 20.0:
            p_lift = _interp_profile(p_best, len(best_seq))
            if p_lift is not None:
                sc_lift = _score_norm_fn(p_lift)
                if sc_lift < best_score:
                    best_score = sc_lift
                    best_seq = p_lift.copy()
                    print("  Lifted improved: %.10f (n=%d)"
                          % (sc_lift, len(best_seq)))

    print("  Phase 1 done: best=%.10f" % best_score)

    # Phase 2: Polishing (Adam sub-gradient + KKT)
    print("=== Phase 2: Polishing ===")
    phase2_end = min(deadline, t_start + 1450.0)
    if (best_seq is not None and time.monotonic() < phase2_end - 60.0):
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + min(remaining * 0.45, 220.0))
        print("  Adam polish: n=%d, budget=%.1fs" % (len(best_seq), dl - time.monotonic()))
        p_best, sc_best = _adam_subgrad(
            best_seq, dl, lr=0.04, top_k=15, beta_init=0.01, n_max_iter=15000)
        if np.isfinite(sc_best) and sc_best < best_score:
            best_score = sc_best
            best_seq = p_best.copy()
            print("  Adam1: %.10f" % sc_best)

        if time.monotonic() < phase2_end - 50.0:
            dl2 = min(phase2_end - 15.0,
                      time.monotonic() + min(120.0, (phase2_end - time.monotonic()) * 0.5))
            p_best2, sc_best2 = _kkt_refine(best_seq, dl2, max_iter=4000)
            if np.isfinite(sc_best2) and sc_best2 < best_score:
                best_score = sc_best2
                best_seq = p_best2.copy()
                print("  KKT: %.10f" % sc_best2)

        if time.monotonic() < phase2_end - 30.0:
            dl3 = min(phase2_end - 15.0,
                      time.monotonic() + min(70.0, (phase2_end - time.monotonic()) * 0.5))
            p_best3, sc_best3 = _adam_subgrad(
                best_seq, dl3, lr=0.015, top_k=30, beta_init=0.005, n_max_iter=8000)
            if np.isfinite(sc_best3) and sc_best3 < best_score:
                best_score = sc_best3
                best_seq = p_best3.copy()
                print("  Adam2: %.10f" % sc_best3)

    print("  Phase 2 done: best=%.10f" % best_score)

    # Phase 3: Multi-restart in logit space
    print("=== Phase 3: Multi-restart ===")
    phase3_end = min(deadline, t_start + 1650.0)
    if (best_seq is not None and time.monotonic() < phase3_end - 60.0):
        n_best = len(best_seq)
        noise_levels = (0.03, 0.05, 0.08, 0.12, 0.18, 0.25)
        n_restarts = len(noise_levels)
        per_restart = max(8.0, (phase3_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 20.0:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - float(z.mean())
            z = z + noise_levels[i] * rng.normal(size=n_best)
            z = np.clip(z, -50.0, 50.0)
            z = z - float(z.mean())
            p_start = _softmax_fn(z)
            dl = min(phase3_end - 15.0, time.monotonic() + per_restart)
            remaining_here = dl - time.monotonic()
            if remaining_here < 5.0:
                continue
            # Short Adam polish on the perturbed start
            p_m, sc_m = _adam_subgrad(
                p_start, dl, lr=0.03, top_k=10, beta_init=0.01, n_max_iter=5000)
            if np.isfinite(sc_m) and sc_m < best_score:
                best_score = sc_m
                best_seq = p_m.copy()
                print("  Restart %d: %.10f" % (i, sc_m))

    print("  Phase 3 done: best=%.10f" % best_score)

    # Phase 4: Explore alternate lengths
    print("=== Phase 4: Length exploration ===")
    phase4_end = min(deadline, t_start + 1780.0)
    if (best_seq is not None and time.monotonic() < phase4_end - 40.0):
        current_n = len(best_seq)
        target_ns = (1024, 2048, 4096, 8192)
        for tn in target_ns:
            if time.monotonic() > phase4_end - 35.0:
                break
            if tn == current_n:
                continue
            p_interp = _interp_profile(best_seq, tn)
            if p_interp is None:
                continue
            sc_interp = _score_norm_fn(p_interp)
            if np.isfinite(sc_interp) and sc_interp < best_score:
                best_score = sc_interp
                best_seq = p_interp.copy()
                print("  Interp n=%d: %.10f" % (tn, sc_interp))
            remaining = phase4_end - time.monotonic()
            if remaining < 25.0:
                continue
            dl = min(phase4_end - 20.0, time.monotonic() + min(60.0, remaining * 0.6))
            p_opt, sc_opt = _adam_subgrad(
                p_interp, dl, lr=0.03, top_k=15, beta_init=0.01, n_max_iter=6000)
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = sc_opt
                best_seq = p_opt.copy()
                print("  Opt n=%d: %.10f" % (tn, sc_opt))

    print("  Phase 4 done: best=%.10f" % best_score)

    # Final: ensure we return a valid non-negative sequence
    if best_seq is None or len(best_seq) == 0 or (not np.all(np.isfinite(best_seq))):
        # Fall back to parent if possible, else a simple profile
        if parent is not None:
            best_seq = parent.copy()
        else:
            best_seq = _make_gaussian_profile(2048, 0.4)

    final_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_gaussian_profile(2048, 0.4)

    final_score = _score_fn(final_seq)
    print("=== Final ===")
    print("  FINAL score=%.10f, n=%d" % (final_score, len(final_seq)))

    # Optional sanity check with the provided evaluator if sequence is not too large
    try:
        seq_list = [float(x) for x in final_seq]
        if len(seq_list) <= 8192:
            sc_eval = float(evaluate_sequence(seq_list))
            print("  evaluate_sequence: %.10f" % sc_eval)
    except Exception as e:
        print("  evaluate_sequence check skipped: %s" % str(e))

    return [float(x) for x in final_seq]
