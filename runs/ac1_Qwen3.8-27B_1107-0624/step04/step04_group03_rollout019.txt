def _next_pow2(n):

            if t % 200 == 0:
                mom = min(0.9995, mom + 0.0002)

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.6)

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _mirror_descent(a_init, deadline, lr_init=0.1, cooling=0.9997):
    """Mirror descent (exponentiated gradient) with adaptive step."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    lr = lr_init
    t = 0
    prev_score = best_score

    while time.monotonic() < deadline:
        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Active set: top 5% of lags
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[min(int(0.05 * len(c)), len(c) - 1)]
        active = c >= threshold * 0.999
        if not active.any():
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient in a-space
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        grad_a = 2.0 * w_conv[n - 1:2 * n - 1]

        # Mirror descent: a_i <- a_i * exp(-lr * grad_i)
        # In log space: z_i <- z_i - lr * grad_i
        z = np.log(np.maximum(a, 1e-30))
        z = z - lr * grad_a / a  # This is the mirror descent update in dual
        # Actually, mirror descent for entropy regularization:
        # a_i <- a_i * exp(-lr * grad_i / sum(a))
        # Simpler: just do multiplicative weights
        a_new = a * np.exp(-lr * grad_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        # Check improvement
        if t % 10 == 0:
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                # Revert and reduce lr
                lr *= 0.9
                a = a  # keep old
        else:
            a = a_new

        lr *= cooling
        if lr < 1e-6:
            lr = 1e-6

    return best_a, best_score


def _sinusoidal_perturbation(a_init, deadline, n_freqs=50):
    """Try adding sinusoidal perturbations and optimize amplitude."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    # Generate candidate frequencies
    freqs = np.arange(1, n_freqs + 1)
    # Also try some incommensurate frequencies
    extra_freqs = np.linspace(0.5, 5.0, 20)
    all_freqs = np.concatenate([freqs.astype(float), extra_freqs])

    t = np.arange(n, dtype=np.float64)

    for f in all_freqs:
        if time.monotonic() >= deadline:
            break

        # Generate sinusoidal basis (both sin and cos)
        angle = 2.0 * np.pi * f * t / n
        for basis in [np.sin(angle), np.cos(angle)]:
            if time.monotonic() >= deadline:
                break

            # Normalize basis to unit sum for clean perturbation
            b_sum = basis.sum()
            if abs(b_sum) < 1e-10:
                b = basis  # zero-mean, just scale
            else:
                b = basis - basis.mean()  # make zero-mean

            b_norm = np.linalg.norm(b)
            if b_norm < 1e-10:
                continue
            b = b / b_norm

            # Line search for optimal amplitude (and sign)
            # Try a range of amplitudes
            best_alpha = 0.0
            best_sc = best_score

            for sign in [1.0, -1.0]:
                for alpha_base in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                    alpha = sign * alpha_base
                    a_trial = best_a + alpha * b
                    a_trial = np.maximum(a_trial, 1e-15)
                    a_trial = a_trial / a_trial.sum()
                    sc = _score_norm(a_trial)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_alpha = alpha

            if best_alpha != 0.0:
                # Refine with finer search around best_alpha
                for refinement in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.5, 0.3, 0.2]:
                    alpha_ref = best_alpha * refinement
                    a_trial = best_a + alpha_ref * b
                    a_trial = np.maximum(a_trial, 1e-15)
                    a_trial = a_trial / a_trial.sum()
                    sc = _score_norm(a_trial)
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best_alpha = alpha_ref

                if best_sc < best_score:
                    best_a = (best_a + best_alpha * b)
                    best_a = np.maximum(best_a, 1e-15)
                    best_a = best_a / best_a.sum()
                    best_score = best_sc

    return best_a, best_score


def _adam_fine(a_init, deadline, lr=0.02, beta_frac=0.001):
    """Adam optimizer with very fine beta for final polishing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-15, beta_frac * max_c)

    stall_count = 0
    last_improve_t = 0

    max_iter = 200000
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 100 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                last_improve_t = t
                stall_count = 0
            elif t - last_improve_t > 500:
                stall_count += 1
                if stall_count > 10:
                    break
            # Slowly decrease beta
            beta = max(1e-15, beta * 0.95)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())
            beta = max(1e-15, max(beta_frac * max_c * 0.1, beta))

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score
