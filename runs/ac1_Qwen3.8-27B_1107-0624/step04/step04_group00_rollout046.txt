def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):

    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:

    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    """Score = 2*n*max(conv(a,a)), assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Score = 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

        eta = (score - z_star) / (gg + 1e-15)
        eta = min(eta, 1.0)
        # Clip to avoid too-large steps
        eta = max(eta, 1e-12)
        
        # Projected step
        a_new = _project_simplex(y - eta * g)
        
        # Check actual score of a_new
        c_new = _full_convolve(a_new, a_new)
        score_new = 2.0 * n * float(c_new.max())
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall_count = 0
        elif t % 10000 == 0:
            stall_count += 1
        
        # Update
        v = a_new - a
        a = a_new
        
        # Periodic restart to escape local minima
        if stall_count > restart_interval and t % restart_interval == 0:
            # Random perturbation
            pert = 0.005 * np.random.randn(n)
            a = _project_simplex(a + pert)
            v = np.zeros(n)
            stall_count = 0
            
            # Reset beta (reheat)
            c = _full_convolve(a, a)
            beta = _adaptive_beta(c)
        
        # Progress reporting
        if t % 50000 == 0:
            print(f"    t={t}, score={score_new:.10f}, best={best_score:.10f}, eta={eta:.2e}, beta={beta:.2e}")
    
    # Final evaluation
    sc = _score_normalized(best_a)
    return best_a, sc


def _make_diverse_inits(n, rng):
    """Generate diverse initial sequences on the simplex."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp (increasing)
    t = np.linspace(0.01, 1.0, n)
    a = t / t.sum()
    inits.append(a)
    
    # Linear ramp (decreasing)
    a = (1.0 - 0.99 * t)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    inits.append(a)
    
    # Quadratic (peak in center)
    u = np.linspace(-1, 1, n)
    a = np.maximum(1.0 - u**2, 1e-6)
    a = a / a.sum()
    inits.append(a)
    
    # Quadratic (peak off-center)
    a = np.maximum(1.0 - (u - 0.3)**2, 1e-6)
    a = a / a.sum()
    inits.append(a)
    
    # Exponential (right-heavy)
    a = np.exp(2.0 * t)
    a = a / a.sum()
    inits.append(a)
    
    # Exponential (left-heavy)
    a = np.exp(-2.0 * t)
    a = a / a.sum()
    inits.append(a)
    
    # Two-peak
    c1, c2 = 0.3, 0.7
    sigma = 0.1
    a = np.exp(-((t - c1)**2) / (2*sigma**2)) + np.exp(-((t - c2)**2) / (2*sigma**2))
    a = a / a.sum()
    inits.append(a)
    
    # Sine-based
    a = np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * t), 1e-6)
    a = a / a.sum()
    inits.append(a)
    
    # Cosine window (asymmetric)
    a = np.maximum(np.cos(np.pi * (t - 0.3)), 1e-6) ** 2
    a = a / a.sum()
    inits.append(a)
    
    # Random smooth (low-frequency)
    k = 5
    coeffs = rng.normal(size=k + 1)
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(np.pi * i * x)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    inits.append(a)
    
    return inits


def _interp_to_length(a, n_new):
    """Interpolate sequence a to length n_new."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new
