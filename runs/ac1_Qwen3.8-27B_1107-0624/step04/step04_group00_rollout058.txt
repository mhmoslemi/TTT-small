```python
def lp_update(f, M, n_solver='highs'):
    n = len(f)
    # LP: max sum(b) s.t. conv(f,b)_k <= M for all k, b >= 0
    # Variables: b_0, ..., b_{n-1}
    # Objective: maximize sum(b) => minimize -sum(b)
    # Constraints: conv(f,b)_k <= M for k = 0, ..., 2n-2
    #              b_j >= 0
    
    from scipy.optimize import linprog
    # c = -1 (minimize -sum(b))
    c = -np.ones(n)
    
    # A_ub: (2n-1) x n matrix where A_ub[k,j] = f_{k-j} if 0 <= k-j < n, else 0
    # b_ub = M * ones(2n-1)
    
    # This matrix is large. Let's use sparse.
    from scipy.sparse import lil_matrix, csr_matrix
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    A = A.tocsr()
    b_ub = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    return res
```

```python
def _build_conv_mat(f):
    """Build the (2n-1) x n convolution matrix where A[k,j] = f[k-j] if valid."""
    n = len(f)
    # For each row k, the non-zero entries are at columns j where 0 <= k-j < n
    # i.e., j in [k-n+1, k] ∩ [0, n-1]
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end+1)
            idx = k - js  # values f[j_start...j_end] indexed by (k - j)
            # idx ranges from k-j_start to k-j_end
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[idx].tolist())
    return coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
```

For small δ, the last term is negligible, and conv(f,f) + 2·conv(f,δ) ≈ conv(f,g) + conv(g,f) (approximately, when g ≈ f).

            s = parent_raw.sum()
            if s > 0.01:

    except Exception:

        if time.monotonic() > phase1_end:
            break
        
        c = _full_convolve(f, f)
        M = float(c.max())
        
        # Get top K peak indices
        top_k = np.argsort(c)[::-1][:K]
        top_k = top_k.tolist()
        
        # Solve reduced LP
        b_opt, sum_b = _reduced_lp_solve(f, M, top_k)
        if b_opt is None:
            print(f"  LP failed at iter {iteration}")
            break
        
        if sum_b <= 1.0 + 1e-8:
            print(f"  LP converged: sum_b={sum_b:.10f} at iter {iteration}")
            break
        
        g = b_opt / sum_b

        best_t = 0.1

        for t_val in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t_val) * f + t_val * g
            f_trial = np.maximum(f_trial, 0.0)
            s_tr = f_trial.sum()
            if s_tr < 1e-10:
                continue
            f_trial = f_trial / s_tr
            sc = _score_norm(f_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        
        f_new = (1.0 - best_t) * f + best_t * g

        f = f_new

            no_improve_count = 0
