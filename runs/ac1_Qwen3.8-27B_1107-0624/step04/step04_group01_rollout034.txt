Uses L-BFGS-B with beta-continuation on log-sum-exp smoothed max,

import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_obj_grad(z, beta):
    """
    Compute smooth objective and gradient w.r.t. z (log-space).
    Objective: 2*n * M_beta(conv(a,a)) where a = softmax(z), sum(a)=1.
    """

    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * cm
    else:
        w = e / s
        obj = 2.0 * n * (cm + beta * np.log(s))

    # dc_i/da_m = 2*a[i-m] if 0 <= i-m < n else 0
    # grad_a[m] = 2 * sum_i w_i * 2 * a[i-m] * 1[0<=i-m<n]

    # dM/da[m] = sum_i w_i * dc[i]/da[m] = 2*sum_i w_i * a[i-m] * 1[0<=i-m<n]
    # = 2 * (cross-correlation of w and a)[m] where cross-corr is at positive lag m

    # Compute h[m] = sum_i w[i] * a[i-m] where 0<=i-m<n, i.e., m<=i<m+n

    # Wait: sum_k a[k]*dM/da[k] = 2*sum_k a[k]*h[k] = 2*dot(a,h)

    # = sum_j a[j]*sum_i w[i]*a[i-j] = sum_j a[j]*h[j] = dot(a,h)

    # d obj/dz = a * (4n*h - 4n*W) = 4n*a*(h-W)

def _lbfgs_optimize(a_init, deadline, beta_schedule, max_iter=5000):
    """L-BFGS-B optimization in log-space with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        state = {'beta': beta}
        
        def fun(z):
            obj, grad = _smooth_obj_grad(z, state['beta'])
            return obj, grad
        
        try:
            res = minimize(fun, z, method='L-BFGS-B', 
                          options={'maxiter': max_iter, 'maxfun': max_iter*2,
                                   'ftol': 1e-12, 'gtol': 1e-10,
                                   'maxls': 50})
            z = res.x
        except Exception as e:
            print(f"  L-BFGS-B error: {e}")
            break
        
        a = _softmax(z)
        z = z - z.mean()
        
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _fista_optimize(a_init, deadline, beta_schedule, max_iter=50000):
    """FISTA (accelerated proximal gradient) in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        z_k = z.copy()
        z_prev = z.copy()
        t_k = 1.0
        
        # Estimate Lipschitz constant
        # For smooth approximation, L depends on beta and n
        # Use backtracking
        L = 100.0 / beta
        
        for it in range(max_iter):
            if time.monotonic() > deadline:
                break
            
            a_k = _softmax(z_k)
            obj, grad = _smooth_obj_grad(z_k, beta)
            
            if not np.all(np.isfinite(grad)):
                break
            
            # Backtracking for Lipschitz constant
            while True:
                z_try = z_k - grad / L
                a_try = _softmax(z_try)
                obj_try = _smooth_obj_grad(z_try, beta)[0]
                if obj_try <= obj + np.dot(grad, z_try - z_k) + 0.5 * L * np.dot(z_try - z_k, z_try - z_k):
                    break
                L *= 2.0
                if L > 1e10:
                    break
            
            z_new = z_k - grad / L
            
            # FISTA extrapolation
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
            z_k = z_new + ((t_k - 1.0) / t_next) * (z_new - z_prev)
            z_k = np.clip(z_k, -50.0, 50.0)
            z_k = z_k - z_k.mean()
            z_prev = z_new
            t_k = t_next
            
            if it % 200 == 0:
                sc = _score_norm(_softmax(z_k))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = _softmax(z_k).copy()
    
    a_final = _softmax(z_k)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _subgradient_polish(a_init, deadline, n_iter=50000):
    """Direct subgradient method with Nesterov's averaging."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    # Nesterov's accelerated subgradient method
    x = z.copy()
    y = z.copy()
    step = 1.0
    base_step = 0.5
    k = 0
    
    for k in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        a_y = _softmax(y)
        c = _full_convolve(a_y, a_y)
        cm = float(c.max())
        
        # Subgradient at the argmax
        idx = int(np.argmax(c))
        # d max_c / da[m] = 2*a[idx-m] if 0 <= idx-m < n else 0
        m_offset = idx
        if 0 <= m_offset < n:
            sg_a = np.zeros(n)
            sg_a[:n - m_offset] = 2.0 * a_y[m_offset:]
            # Wait: dc[idx]/da[m] = 2*a[idx-m] if 0<=idx-m<n
            # So sg_a[m] = 2*a_y[idx-m] if 0<=idx-m<n, i.e., idx-m in [0,n-1]
            # m in [idx-(n-1), idx] intersect [0, n-1]
            lo = max(0, idx - (n-1))
            hi = min(n-1, idx)
            sg_a[lo:hi+1] = 2.0 * a_y[idx - np.arange(lo, hi+1)]
        else:
            sg_a = np.zeros(n)
        
        # Objective gradient: 2n * sg_a
        grad = 2.0 * n * sg_a
        
        # Convert to z-space
        W = float(np.dot(sg_a, a_y))
        grad_z = a_y * (grad - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Step size
        step = base_step / np.sqrt(k + 1)
        
        x = y - step * grad_z
        x = np.clip(x, -50.0, 50.0)
        x = x - x.mean()
        
        # Update y (Nesterov)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (k-1)**2)) / 2.0 if k > 0 else 1.0
        y = x + ((t_k - 1.0) / t_k) * (x - y)
        y = np.clip(y, -50.0, 50.0)
        y = y - y.mean()
        
        if k % 1000 == 0:
            sc = _score_norm(_softmax(y))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = _softmax(y).copy()
    
    a_final = _softmax(y)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS-B (quasi-Newton) with beta-continuation on log-sum-exp smoothed max,
FISTA (accelerated gradient), Adam, and Nesterov-accelerated subgradient polishing.
Key improvements over previous: quasi-Newton phase for faster convergence,
more aggressive beta schedule, direct subgradient final polish, and larger-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_obj_grad(z, beta):
    """Compute smooth objective and gradient w.r.t. z (log-space)."""
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * cm
    else:
        w = e / s
        obj = 2.0 * n * (cm + beta * np.log(s))

    a_rev = a[::-1]
    wc = _full_convolve(a_rev, w)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)

    return obj, grad


def _lbfgs_phase(a_current, beta_schedule, deadline, max_iter_per_beta=300):
    """L-BFGS-B continuation in log-space."""
    n = len(a_current)
    z = np.log(np.maximum(a_current, 1e-15))
    z = z - z.mean()

    best_a = a_current.copy()
    best_score = _score_norm(a_current)

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break

        def fun(z):
            return _smooth_obj_grad(z, beta)

        z_init = z.copy()
        try:
            res = minimize(fun, z_init, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_beta,
                                    'maxfun': max_iter_per_beta * 3,
                                    'ftol': 1e-15, 'gtol': 1e-12,
                                    'maxls': 40, 'm': 20})
            z = res.x
        except Exception:
            pass

        a = _softmax(z)
        z = z - z.mean()
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score, z


def _adam_phase(a_init, deadline, beta_schedule, max_iter=50000, lr=0.01):
    """Adam optimizer in log-space with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            obj, grad = _smooth_obj_grad(z, beta)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 300 == 0:
                sc = _score_norm(_softmax(z))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = _softmax(z).copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_phase(a_init, deadline, beta_schedule, max_iter=30000):
    """FISTA (accelerated proximal gradient) in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        z_k = z.copy()
        z_prev = z.copy()
        t_k = 1.0
        L = 50.0 / beta

        for it in range(max_iter):
            if time.monotonic() > deadline:
                break

            obj, grad = _smooth_obj_grad(z_k, beta)
            if not np.all(np.isfinite(grad)):
                break

            # Backtracking line search
            z_new = z_k - grad / L
            for _ in range(20):
                obj_new = _smooth_obj_grad(z_new, beta)[0]
                bound = obj + np.dot(grad, z_new - z_k) + 0.5 * L * np.dot(z_new - z_k, z_new - z_k)
                if obj_new <= bound + 1e-12:
                    break
                L *= 2.0
                z_new = z_k - grad / L

            z_new = np.clip(z_new, -50.0, 50.0)
            z_new = z_new - z_new.mean()

            # FISTA update
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            z_prev = z_k
            z_k = z_new + ((t_k - 1.0) / t_next) * (z_new - z_prev)
            z_k = np.clip(z_k, -50.0, 50.0)
            z_k = z_k - z_k.mean()
            t_k = t_next

            if it % 500 == 0:
                sc = _score_norm(_softmax(z_k))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = _softmax(z_k).copy()

        z = z_k.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_polish(a_init, deadline, base_step=0.3):
    """Direct subgradient method with Nesterov's accelerated averaging."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    x = z.copy()
    y = z.copy()
    k = 0

    while time.monotonic() < deadline:
        a_y = _softmax(y)
        c = _full_convolve(a_y, a_y)
        cm = float(c.max())
        idx = int(np.argmax(c))

        # Subgradient of max c_k w.r.t. a
        sg_a = np.zeros(n)
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        if lo <= hi:
            sg_a[lo:hi + 1] = 2.0 * a_y[idx - np.arange(lo, hi + 1)]

        # Convert to z-space gradient
        W = float(np.dot(sg_a, a_y))
        grad_z = 4.0 * n * a_y * (sg_a / 2.0 - W)

        if not np.all(np.isfinite(grad_z)):
            break

        step = base_step / np.sqrt(k + 1)
        x = y - step * grad_z
        x = np.clip(x, -50.0, 50.0)
        x = x - x.mean()

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        if k > 0:
            y = x + ((t_k - 1.0) / t_k) * (x - y)
        else:
            y = x.copy()
        y = np.clip(y, -50.0, 50.0)
        y = y - y.mean()

        k += 1
        if k % 2000 == 0:
            sc = _score_norm(_softmax(y))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = _softmax(y).copy()

    a_final = _softmax(y)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimization: L-BFGS-B continuation -> FISTA -> Adam -> subgradient.
    Uses quasi-Newton for fast initial convergence, accelerated methods for middle,
    and direct subgradient for final polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        _update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")

    # --- Phase 1: L-BFGS-B continuation (0-450s) ---
    phase1_end = min(deadline, t_start + 450.0)

    c_init = _full_convolve(a0, a0)
    max_c0 = float(c_init.max())

    # Geometric beta schedule: 20 steps from 0.05*max to 0.0002*max
    beta_start = 0.05 * max_c0
    beta_end = 0.0002 * max_c0
    n_betas = 20
    ratios = (beta_end / beta_start) ** (1.0 / (n_betas - 1))
    beta_schedule_1 = [beta_start * (ratios ** i) for i in range(n_betas)]

    print(f"Phase 1: L-BFGS-B, {n_betas} betas, [{beta_start:.5f} -> {beta_end:.8f}]")
    a_curr, _, _ = _lbfgs_phase(a0, beta_schedule_1, phase1_end, max_iter_per_beta=200)
    _update_best(a_curr, _score(a_curr), "lbfgs_p1")

    print(f"Phase 1 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 2: L-BFGS-B fine (450s-750s) ---
    phase2_end = min(deadline, t_start + 750.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        a_f = best_seq / best_seq.sum()
        c_f = _full_convolve(a_f, a_f)
        max_cf = float(c_f.max())

        beta_start2 = 0.0002 * max_cf
        beta_end2 = 0.000005 * max_cf
        n_betas2 = 15
        ratios2 = (beta_end2 / beta_start2) ** (1.0 / (n_betas2 - 1))
        beta_schedule_2 = [beta_start2 * (ratios2 ** i) for i in range(n_betas2)]

        print(f"Phase 2: L-BFGS-B fine, [{beta_start2:.8f} -> {beta_end2:.10f}]")
        a_curr2, _, _ = _lbfgs_phase(a_f, beta_schedule_2, phase2_end, max_iter_per_beta=300)
        _update_best(a_curr2, _score(a_curr2), "lbfgs_p2")

    print(f"Phase 2 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 3: FISTA (750s-1050s) ---
    phase3_end = min(deadline, t_start + 1050.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        a_p3 = best_seq / best_seq.sum()
        c_p3 = _full_convolve(a_p3, a_p3)
        max_cp3 = float(c_p3.max())

        beta_fista = [0.0005*max_cp3, 0.0002*max_cp3, 0.0001*max_cp3,
                      0.00005*max_cp3, 0.00002*max_cp3, 0.00001*max_cp3,
                      0.000005*max_cp3, 0.000002*max_cp3]
        per_beta = max(1.0, (phase3_end - time.monotonic() - 10) / len(beta_fista))
        max_iter_f = max(2000, int(per_beta / 0.03))

        print(f"Phase 3: FISTA, {len(beta_fista)} betas")
        p_best, sc_best = _fista_phase(a_p3, phase3_end, beta_fista, max_iter=max_iter_f)
        _update_best(p_best, _score(p_best), "fista")

    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 4: Adam ultra-fine (1050s-1400s) ---
    phase4_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        a_p4 = best_seq / best_seq.sum()
        c_p4 = _full_convolve(a_p4, a_p4)
        max_cp4 = float(c_p4.max())

        beta_adam = [0.0001*max_cp4, 0.00005*max_cp4, 0.00002*max_cp4,
                     0.00001*max_cp4, 0.000005*max_cp4, 0.000002*max_cp4,
                     0.000001*max_cp4, 0.0000005*max_cp4]
        remaining = phase4_end - time.monotonic() - 10
        max_iter_a = max(10000, int(remaining / 0.04))

        print(f"Phase 4: Adam ultra-fine, {len(beta_adam)} betas")
        p_best, sc_best = _adam_phase(a_p4, phase4_end, beta_adam,
                                      max_iter=max_iter_a, lr=0.005)
        _update_best(p_best, _score(p_best), "adam_ultra")

    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 5: Subgradient polishing (1400s-1650s) ---
    phase5_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase5_end - 10:
        a_p5 = best_seq / best_seq.sum()
        print(f"Phase 5: Subgradient polishing")
        p_best, sc_best = _subgradient_polish(a_p5, phase5_end, base_step=0.2)
        _update_best(p_best, _score(p_best), "subgrad")

    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 6: Multi-restart (1650s-1750s) ---
    phase6_end = min(deadline, t_start + 1750.0)
    if best_seq is not None and time.monotonic() < phase6_end - 20:
        n_restarts = 4
        per_restart = (phase6_end - time.monotonic() - 10) / n_restarts
        sigmas = [0.02, 0.04, 0.06, 0.03]

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            p_start = _perturb(best_seq / best_seq.sum(), sigmas[i], rng)
            c_r = _full_convolve(p_start, p_start)
            max_cr = float(c_r.max())
            br = [0.001*max_cr, 0.0005*max_cr, 0.0002*max_cr, 0.0001*max_cr, 0.00005*max_cr]

            p_best, sc_best = _adam_phase(p_start, time.monotonic() + per_restart,
                                          br, max_iter=5000, lr=0.01)
            _update_best(p_best, _score(p_best), f"restart{i}")

    print(f"Phase 6 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 7: Try larger n (1750s-deadline) ---
    if best_seq is not None and len(best_seq) < 4096 and time.monotonic() < deadline - 25:
        target_n = 4096
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score(p_interp)
        _update_best(p_interp, sc_interp, f"interp_n{target_n}")

        remaining = deadline - time.monotonic() - 15
        if remaining > 20:
            c_int = _full_convolve(p_interp, p_interp)
            max_ci = float(c_int.max())
            bs_int = [0.005*max_ci, 0.002*max_ci, 0.001*max_ci,
                      0.0005*max_ci, 0.0002*max_ci, 0.0001*max_ci]
            p_best, sc_best = _adam_phase(p_interp, time.monotonic() + remaining,
                                          bs_int, max_iter=8000, lr=0.008)
            _update_best(p_best, _score(p_best), f"opt_n{target_n}")

    print(f"Phase 7 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
