import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _subgradient_polish(a_init, deadline, rng, n_steps=5000):
    """Subgradient method with decreasing step size, targeting the max lag."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)

    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))

    step_scale = 0.5 / (n ** 0.5)
    decay = 0.9998

    for t in range(n_steps):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        # Get top few lags for a smoothed subgradient
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[-top_k:]
        top_vals = c[top_idx]
        # Soft weighting
        w = top_vals - top_vals.min()
        w = w ** 8  # Sharp weighting
        if w.sum() < 1e-30:
            w = np.zeros(top_k)
            w[-1] = 1.0
        else:
            w = w / w.sum()

        # Compute smoothed subgradient
        grad = np.zeros(n)
        for w_k, k_val in zip(w, top_idx):
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                # d conv(a,a)[k]/da[i] = 2*a[k-i] for valid i
                idx = k_val - np.arange(lo, hi + 1)
                grad[lo:hi + 1] += 2.0 * w_k * a[idx]

        grad *= 4.0 * n
        # Center the gradient (for sum constraint)
        grad -= grad.mean()

        step = step_scale / (1.0 + 0.01 * t)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Accept with small probability (simulated annealing)
            if sc < best_sc * 1.002 and rng.random() < 0.05:
                a = a_new.copy()

    return best_a, best_sc


def _targeted_local_search(a_init, deadline, rng, n_trials=10000):
    """Local search that targets the dominant lag of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))

    for t in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        # Recompute if needed
        if t % 50 == 0:
            c = _full_convolve(a, a)
            k_star = int(np.argmax(c))

        # Strategy 1: Reduce the product at the critical lag
        # Find the pair (i, k_star - i) with largest product
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            idx = k_star - np.arange(lo, hi + 1)
            products = a[lo:hi + 1] * a[idx]
            max_prod_idx_local = int(np.argmax(products))
            i1 = lo + max_prod_idx_local
            i2 = int(k_star - i1)

            # Try moving mass from the larger of a[i1], a[i2] to a random other position
            if a[i1] >= a[i2]:
                src = i1
            else:
                src = i2

            # Choose destination (not near the critical lag)
            for _attempt in range(5):
                dst = int(rng.integers(0, n))
                if abs(dst - src) < 3:
                    continue
                # Check if dst would create a new peak
                break

            # Move a small fraction of mass
            frac = rng.uniform(0.02, 0.15)
            a_trial = a.copy()
            move_amt = a[src] * frac
            a_trial[src] -= move_amt
            a_trial[dst] += move_amt
            a_trial = np.maximum(a_trial, 0.0)
            a_trial /= a_trial.sum()

            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                continue

        # Strategy 2: Random block perturbation (less targeted)
        if t % 3 == 0:
            block_size = int(rng.integers(2, min(15, n // 10)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            pert = rng.uniform(0.9, 1.1, size=block_size)
            a_trial[start:start + block_size] *= pert
            a_trial = np.maximum(a_trial, 0.0)
            a_trial /= a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
            elif sc < best_sc * 1.003 and rng.random() < 0.03:
                a = a_trial.copy()

    return best_a, best_sc


def _paired_coord_descent(a_init, deadline, rng, n_rounds=200):
    """Coordinate descent on pairs that sum to the critical lag."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))

    for rnd in range(n_rounds):
        if time.monotonic() > deadline - 1.0:
            break

        if rnd % 20 == 0:
            c = _full_convolve(a, a)
            k_star = int(np.argmax(c))

        # Work on the critical lag
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo > hi:
            continue

        # For each pair (i, k_star - i), try to rebalance
        for i in range(lo, hi + 1):
            j = k_star - i
            if i == j:
                continue
            if i > j:
                # Already handled
                continue
            # Current product
            prod = a[i] * a[j]
            if prod < 1e-15:
                continue

            # Try: reduce a[i] a bit, increase a[j] a bit (or vice versa)
            # The optimal split for fixed sum s = a[i]+a[j] to minimize a[i]*a[j]
            # is to make one as small as possible (i.e., push to zero)
            s = a[i] + a[j]
            # Try extreme: put all mass in one
            a_trial = a.copy()
            a_trial[i] = s
            a_trial[j] = 0.0
            a_trial = np.maximum(a_trial, 0.0)
            a_trial /= a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

            # Try partial: 80/20 split
            a_trial2 = a.copy()
            a_trial2[i] = 0.8 * s
            a_trial2[j] = 0.2 * s
            a_trial2 = np.maximum(a_trial2, 0.0)
            a_trial2 /= a_trial2.sum()
            sc2 = _score_norm(a_trial2)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a_trial2.copy()
                a = a_trial2.copy()

        # Also try to reduce mass at the critical lag by moving to less critical lags
        if rnd % 5 == 0:
            # Find a lag with low convolution value
            c_low = _full_convolve(a, a)
            # Find lags where conv is in the bottom quartile
            threshold = np.percentile(c_low, 25)
            low_lags = np.where(c_low < threshold)[0]
            if len(low_lags) > 0:
                dst_lag = low_lags[int(rng.integers(0, len(low_lags)))]
                # Move mass from a random element to a position that contributes to dst_lag
                src = int(rng.integers(0, n))
                dst_options = [dst_lag - src]
                dst = dst_options[0]
                if 0 <= dst < n and dst != src:
                    a_trial = a.copy()
                    move = a[src] * rng.uniform(0.05, 0.2)
                    a_trial[src] -= move
                    a_trial[dst] += move
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial /= a_trial.sum()
                    sc = _score_norm(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()

    return best_a, best_sc


def _run_lbfgs_continuation(a_init, deadline, rng, max_wall_time=None, beta_scale=1.0):
    """L-BFGS with logsumexp continuation - improved version."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # More aggressive schedule with more levels
    beta_starts = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.00075,
                   0.0004, 0.0002, 0.0001, 0.00005, 0.000025, 0.000012, 0.000006]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(3.0, max_wall_time / len(betas))
    else:
        per_level = 40.0

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 100, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _mirror_descent(a_init, deadline, rng, lr_init=0.01, n_steps=20000):
    """Mirror descent with KL divergence for the non-negative simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)

    lr = lr_init
    decay = 0.9997

    for t in range(n_steps):
        if time.monotonic() > deadline - 1.0:
            break
        if t % 1000 == 0:
            lr = max(0.0005, lr * decay)

        c = _full_convolve(a, a)
        # Use top-3 weighted subgradient
        top_k = min(3, len(c))
        top_idx = np.argsort(c)[-top_k:]
        top_vals = c[top_idx]
        w = top_vals - top_vals.min()
        w = w ** 6
        if w.sum() < 1e-30:
            w = np.zeros(top_k)
            w[-1] = 1.0
        else:
            w = w / w.sum()

        grad = np.zeros(n)
        for w_k, k_val in zip(w, top_idx):
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                idx = k_val - np.arange(lo, hi + 1)
                grad[lo:hi + 1] += 2.0 * w_k * a[idx]

        grad *= 4.0 * n
        grad -= grad.mean()

        # Mirror descent update (multiplicative)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
        elif sc < best_sc * 1.001 and rng.random() < 0.1:
            a = a_new.copy()

    return best_a, best_sc


def _make_symmetric_profile(n, base):
    """Create a symmetric profile from a half-profile."""
    half = base[:n // 2 + 1]
    a = np.concatenate([half, half[-2::-1]])
    if len(a) != n:
        a = a[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def _make_windowed(n, window_type='optimal', params=None):
    """Create various windowed profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0  # [-1, 1]

    if window_type == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * u))
    elif window_type == 'gaussian':
        w = 0.3 if params is None else params
        a = np.exp(-u**2 / (2.0 * w**2))
    elif window_type == 'kaiser':
        alpha = 5.0 if params is None else params
        # Approximate Kaiser window using Bessel-like
        a = np.cos(np.pi * u * (1.0 - np.sqrt(max(0, 1 - u**2))) / 2.0) ** (2.0 / alpha)
        a = np.maximum(a, 0.001)
    elif window_type == 'tapered':
        # Taper at edges only
        taper = 0.2
        a = np.ones(n)
        k = int(n * taper)
        for i in range(k):
            a[i] = 0.5 * (1 + np.cos(np.pi * i / k))
            a[n-1-i] = a[i]
    elif window_type == 'bell':
        # super-Gaussian
        p = 4.0 if params is None else params
        a = np.exp(-u**2 * p)
    elif window_type == 'parabola':
        a = 1.0 - u**2
        a = np.maximum(a, 0.001)
    elif window_type == 'flat_taper':
        # Flat center, tapered edges
        taper = 0.15
        a = np.ones(n)
        k = int(n * taper)
        for i in range(k):
            a[i] = i / k
            a[n-1-i] = a[i]
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-12)
    return a / a.sum()

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Paired coordinate descent (100s) ---
    phase5_end = min(deadline, t_start + 980.0)
    if best_seq is not None:
        dl = phase5_end - 5.0
        p_best, sc_best = _paired_coord_descent(best_seq, dl, rng, n_rounds=100)
        update_best(p_best, sc_best, "pair_cd")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Another L-BFGS round with tighter betas (200s) ---
    phase6_end = min(deadline, t_start + 1180.0)
    if best_seq is not None:
        dl = phase6_end - 5.0
        p_best, sc_best = _run_lbfgs_continuation(best_seq, dl, rng,
                                                   max_wall_time=phase6_end - time.monotonic(),
                                                   beta_scale=0.3)
        update_best(p_best, sc_best, "lbfgs_tight")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Multi-restart (300s) ---
    phase7_end = min(deadline, t_start + 1480.0)
    if best_seq is not None:
        n_restarts = 6
        per_restart = max(15.0, (phase7_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break

            if i == 0:
                # Small noise restart
                noise = 0.02
            elif i == 1:
                noise = 0.05
            elif i == 2:
                noise = 0.1
            elif i == 3:
                # Structural restart: zero out random 10% and redistribute
                p_start = best_seq.copy()
                n_zero = max(1, len(p_start) // 10)
                zero_idx = rng.choice(len(p_start), size=n_zero, replace=False)
                p_start[zero_idx] = 0.0
                p_start /= p_start.sum()
                z = np.log(np.maximum(p_start, 1e-15))
                z = z - z.mean()
                z = z + 0.03 * rng.normal(size=len(p_start))
                z = np.clip(z, -50, 50)
                z = z - z.mean()
                p_start = _softmax_fn(z)
                dl = min(phase7_end - 5, time.monotonic() + per_restart)
                p_best, sc_best = _run_lbfgs_continuation(p_start, dl, rng,
                                                           max_wall_time=per_restart, beta_scale=0.5)
                update_best(p_best, sc_best, f"restart_sparse_{i}")
                continue
            else:
                noise = 0.03 + 0.03 * (i - 3)

            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase7_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _run_lbfgs_continuation(p_start, dl, rng,
                                                       max_wall_time=per_restart, beta_scale=0.7)
            # Polish
            if time.monotonic() < phase7_end - 3:
                dl2 = min(phase7_end - 3, time.monotonic() + per_restart * 0.3)
                p_best2, sc_best2 = _subgradient_polish(p_best, dl2, rng, n_steps=2000)
                if sc_best2 < sc_best:
                    p_best, sc_best = p_best2, sc_best2
            update_best(p_best, sc_best, f"restart_{i}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Final intensive polishing (remaining time) ---
    if best_seq is not None:
        remaining = max(30.0, deadline - time.monotonic())
        n_polish = max(3, int(remaining / 50))
        per_polish = remaining / n_polish

        for i in range(n_polish):
            if time.monotonic() > deadline - 5:
                break

            if i % 3 == 0:
                # Fine L-BFGS
                dl = min(deadline - 5, time.monotonic() + per_polish)
                p_best, sc_best = _run_lbfgs_continuation(best_seq, dl, rng,
                                                           max_wall_time=per_polish, beta_scale=0.1)
                update_best(p_best, sc_best, f"final_lbfgs_{i}")
            elif i % 3 == 1:
                # Subgradient
                dl = min(deadline - 5, time.monotonic() + per_polish)
                p_best, sc_best = _subgradient_polish(best_seq, dl, rng, n_steps=5000)
                update_best(p_best, sc_best, f"final_subgrad_{i}")
            else:
                # Targeted local search
                dl = min(deadline - 5, time.monotonic() + per_polish)
                p_best, sc_best = _targeted_local_search(best_seq, dl, rng, n_trials=5000)
                update_best(p_best, sc_best, f"final_ls_{i}")

    # --- Try different lengths ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_curr = len(best_seq)
        # Try interpolating to 2x length
        target_n = n_curr * 2
        if target_n <= 8192:
            x0 = np.linspace(0, 1, n_curr)
            x1 = np.linspace(0, 1, target_n)
            p_interp = np.interp(x1, x0, best_seq)
            p_interp = np.maximum(p_interp, 0.0)
            p_interp /= p_interp.sum()
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if sc_interp < best_score * 1.05 and time.monotonic() < deadline - 20:
                dl = min(deadline - 10, time.monotonic() + 60.0)
                p_best, sc_best = _run_lbfgs_continuation(p_interp, dl, rng,
                                                           max_wall_time=60.0, beta_scale=1.0)
                update_best(p_best, sc_best, f"lbfgs_n{target_n}")

        # Try 0.5x length (in case smaller is better)
        half_n = n_curr // 2
        if half_n >= 1024 and half_n != n_curr:
            x0 = np.linspace(0, 1, n_curr)
            x1 = np.linspace(0, 1, half_n)
            p_half = np.interp(x1, x0, best_seq)
            p_half = np.maximum(p_half, 0.0)
            p_half /= p_half.sum()
            sc_half = _score_norm(p_half)
            update_best(p_half, sc_half, f"half_n{half_n}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_windowed(4096, 'raised_cosine')
            s = float(final_seq.sum())
        # Normalize to sum = 1 (scale doesn't matter, but keep it clean)
        final_seq = final_seq / s

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator
        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned inf, checking...")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_windowed(4096, 'raised_cosine')
    sc_fb = _score_norm(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
