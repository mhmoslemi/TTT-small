"""
Iterated Local Search (ILS) with multi-strategy optimization for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Combines L-BFGS logsumexp smoothing, subgradient corrections, KKT balancing,
and diverse structured perturbations to escape local minima.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_topk_smooth(a, beta, n, top_k):
    """Compute top-k smoothed objective value and gradient components."""
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_max_top = c_top.max()
        c_norm = (c_top - c_max_top) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[idx[0]] = 1.0
        else:
            w = np.zeros(len(c))
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros(len(c))
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return w, h, W, c, cm


def _z_gradient_smooth(z, beta, n, top_k):
    """Gradient of smoothed objective in logit space."""
    a = _softmax_fn(z)
    w, h, W, c, cm = _compute_topk_smooth(a, beta, n, top_k)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _z_objective_smooth(z, beta, n, top_k):
    """Smoothed objective value in logit space."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_max_top = c_top.max()
        log_sum = float(np.log(np.sum(np.exp((c_top - c_max_top) / beta))))
        return 2.0 * n * (c_max_top + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_continuation(a_init, deadline, max_wall_time=None, top_k=0):
    """L-BFGS with beta continuation on a normalized sequence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        try:
            result = minimize(
                _z_objective_smooth, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient_smooth,
                options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 30, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.04, top_k=0, n_max_iter=50000):
    """Adam optimizer in logit space with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.01 * max_c)

    adaptive_k = (top_k == 0)
    cur_k = 3 if adaptive_k else top_k

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        use_k = cur_k if adaptive_k else top_k
        grad = _z_gradient_smooth(z, beta, n, use_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 100 == 0:
            beta = max(1e-14, beta * 0.85)
            if adaptive_k and t % 400 == 0:
                cur_k = min(cur_k + 2, 40)

        if t % 2000 == 0 and best_score < 1.52:
            lr = max(0.003, lr * 0.96)

    return best_a, best_score


def _subgradient_step(a, top_k=20, lr=0.01):
    """One subgradient step using minimum-norm subgradient approximation."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())

    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
    else:
        threshold = cm * 0.999
        idx = np.where(c >= threshold)[0]
        if len(idx) == 0:
            idx = np.array([int(np.argmax(c))])

    # Compute average subgradient over active set
    a_rev = a[::-1]
    g = np.zeros(n)
    for k_idx in idx:
        k_val = int(k_idx)
        # grad of conv[k] w.r.t. a[i] is a[k-i] + a[k-i] (both terms)
        # but in logit space, d(conv[k])/d(z[i]) = 2*a[i]*a[k-i] (approximately)
        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        if lo <= hi:
            g[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]

    g /= len(idx)

    # Project gradient to be orthogonal to all-ones (for sum constraint in logit space)
    g = g - g.mean()

    # Step in logit space
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z - lr * g / (np.linalg.norm(g) + 1e-15)
    z = np.clip(z, -80, 80)
    z = z - z.mean()
    return _softmax_fn(z)


def _kkt_balance_a(a_init, deadline, max_iter=5000):
    """KKT-based balancing with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Adaptive active set: start with tight threshold, widen if needed
        threshold_frac = 0.999 - min(0.004, it * 0.00001)
        threshold = cm * threshold_frac
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > 200:
            # Keep only top 200
            top_idx = np.argsort(c)[-200:]
            active = top_idx

        # Compute h_i = average of a[k-i] over active set
        h = np.zeros(n)
        for k_idx in active:
            k_val = int(k_idx)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]
        h /= len(active)

        W = float(np.dot(h, a))

        # Adaptive step size
        active_vals = c[active]
        active_spread = float(active_vals.max() - active_vals.min()) + 1e-15
        step = 0.02 * active_spread / (cm + 1e-15)
        step = min(step, 0.03)

        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

        # Direction in logit space
        direction = h - W
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        direction = direction / dir_norm

        z_new = z - step * direction
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()
        a_new = _softmax_fn(z_new)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                z_try = z - step * shrink * direction
                z_try = np.clip(z_try, -80, 80)
                z_try = z_try - z_try.mean()
                a_try = _softmax_fn(z_try)
                sc_try = _score_norm_fn(a_try)
                if np.isfinite(sc_try) and sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break

        # Convergence check
        if it > 50 and np.linalg.norm(direction) < 1e-10:
            break

    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_periodic_profile(n, p, pattern_type="smooth"):
    """Create a periodic profile with period p."""
    if pattern_type == "smooth":
        t = np.linspace(0, 2 * np.pi, p, endpoint=False)
        x = 1.0 + 0.3 * np.cos(t) + 0.1 * np.cos(2 * t)
    elif pattern_type == "alternating":
        x = np.ones(p)
        x[1::2] = 0.7
    elif pattern_type == "chirp":
        t = np.linspace(0, 2 * np.pi, p, endpoint=False)
        x = 1.0 + 0.2 * np.cos(t ** 2)
    else:
        x = np.ones(p)
    a = np.tile(x, n // p + 1)[:n]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_zadoff_chu_real(n):
    """Real-valued Zadoff-Chu inspired sequence."""
    if n % 2 == 0:
        n -= 1
    t = np.arange(n)
    phase = -np.pi * t * (t + 1) / n
    x = 1.0 + np.cos(phase)
    # Pad or trim to desired length
    if len(x) < n:
        x = np.concatenate([x, np.ones(n - len(x))])
    else:
        x = x[:n]
    x = np.maximum(x, 1e-12)
    return x / x.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _perturb_sinusoidal(a, rng, n_freqs=5, max_amp=0.1):
    """Add sinusoidal perturbations in logit space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    for _ in range(n_freqs):
        freq = rng.integers(1, n // 10 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, max_amp)
        t = np.arange(n)
        z += amp * np.sin(2 * np.pi * freq * t / n + phase)
    z = np.clip(z, -60, 60)
    z = z - z.mean()
    return _softmax_fn(z)


def _perturb_block_swap(a, rng, n_blocks=8, block_frac=0.1):
    """Swap random blocks of the sequence."""
    n = len(a)
    a_copy = a.copy()
    block_size = max(1, int(n * block_frac / n_blocks))
    for _ in range(n_blocks):
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) < block_size:
            continue
        a_copy[i:i + block_size], a_copy[j:j + block_size] = a_copy[j:j + block_size].copy(), a_copy[i:i + block_size].copy()
    s = a_copy.sum()
    if s < 1e-15:
        return a
    return a_copy / s


def _perturb_gaussian(a, rng, sigma=0.05):
    """Gaussian perturbation in logit space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + sigma * rng.normal(size=n)
    z = np.clip(z, -60, 60)
    z = z - z.mean()
    return _softmax_fn(z)


def _perturb_interp(a, rng, n_target=None):
    """Interpolate with a random profile."""
    n = len(a)
    if n_target is None:
        n_target = n
    t = np.linspace(0, 1, n)
    # Random smooth profile via random Fourier series
    a_rand = np.ones(n)
    for k in range(rng.integers(1, 5)):
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        a_rand += amp * np.cos(2 * np.pi * freq * t + phase)
    a_rand = np.maximum(a_rand, 1e-12)
    a_rand = a_rand / a_rand.sum()
    alpha = rng.uniform(0.1, 0.4)
    a_new = (1 - alpha) * a + alpha * a_rand
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _local_optimize(a, deadline, phase="fast"):
    """Run a local optimization sequence."""
    if time.monotonic() >= deadline:
        return a, _score_norm_fn(a)

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    if phase == "fast":
        # Quick Adam
        dl = min(deadline, time.monotonic() + 30.0)
        a2, s2 = _adam_optimize(a, dl, lr=0.03, top_k=0, n_max_iter=10000)
        if s2 < best_score:
            best_a, best_score = a2, s2
        # Quick KKT
        if time.monotonic() < deadline - 5:
            dl2 = min(deadline, time.monotonic() + 20.0)
            a3, s3 = _kkt_balance_a(best_a, dl2, max_iter=2000)
            if s3 < best_score:
                best_a, best_score = a3, s3
    elif phase == "medium":
        # L-BFGS
        dl = min(deadline, time.monotonic() + 60.0)
        a2, s2 = _lbfgs_continuation(a, dl, max_wall_time=min(55.0, deadline - time.monitonic()), top_k=0)
        if s2 < best_score:
            best_a, best_score = a2, s2
        # Adam
        if time.monotonic() < deadline - 10:
            dl2 = min(deadline, time.monotonic() + 40.0)
            a3, s3 = _adam_optimize(best_a, dl2, lr=0.02, top_k=5, n_max_iter=20000)
            if s3 < best_score:
                best_a, best_score = a3, s3
        # KKT
        if time.monotonic() < deadline - 5:
            dl3 = min(deadline, time.monotonic() + 15.0)
            a4, s4 = _kkt_balance_a(best_a, dl3, max_iter=3000)
            if s4 < best_score:
                best_a, best_score = a4, s4
    else:  # "full"
        # Full L-BFGS
        dl = min(deadline, time.monotonic() + 120.0)
        a2, s2 = _lbfgs_continuation(a, dl, max_wall_time=min(115.0, deadline - time.monotonic()), top_k=0)
        if s2 < best_score:
            best_a, best_score = a2, s2
        # Adam
        if time.monotonic() < deadline - 20:
            dl2 = min(deadline, time.monotonic() + 80.0)
            a3, s3 = _adam_optimize(best_a, dl2, lr=0.02, top_k=0, n_max_iter=40000)
            if s3 < best_score:
                best_a, best_score = a3, s3
        # KKT
        if time.monotonic() < deadline - 10:
            dl3 = min(deadline, time.monotonic() + 30.0)
            a4, s4 = _kkt_balance_a(best_a, dl3, max_iter=5000)
            if s4 < best_score:
                best_a, best_score = a4, s3 if s4 > s3 else s4
                best_a = a4
                best_score = s4
        # Subgradient polish
        if time.monotonic() < deadline - 5:
            a5 = best_a.copy()
            sc5 = best_score
            for _ in range(200):
                if time.monotonic() > deadline - 5:
                    break
                a5 = _subgradient_step(a5, top_k=30, lr=0.005)
                sc5_check = _score_norm_fn(a5)
                if sc5_check < sc5:
                    sc5 = sc5_check
            if sc5 < best_score:
                best_a, best_score = a5, sc5

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    ILS with multi-strategy local optimization and diverse perturbations.
    """
    global height_sequence_1

    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Initial candidates ---
    print("=== Phase 0: Initial candidates ===")
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 1e-10, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                candidates.append((sc, parent.copy(), len(parent), "parent"))
                _update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    n_base = 4096

    if parent is not None and len(parent) != n_base:
        p_interp = _interp_profile(parent, n_base)
        sc = _score_norm_fn(p_interp)
        candidates.append((sc, p_interp, n_base, "parent_interp"))
        _update_best(p_interp, sc, "parent_interp")

    # Diverse profiles
    profiles = []
    for alpha in [0.001, 0.003, 0.005, 0.01, 0.02]:
        profiles.append(_make_exp_profile(n_base, alpha))
    for slope in [-2.0, -1.0, 0.0, 1.0, 2.0]:
        profiles.append(_make_linear_profile(n_base, slope))
    for a1, a2 in [(-1.0, 0.5), (1.0, -0.5), (-0.5, 1.0), (0.5, -1.0)]:
        profiles.append(_make_quadratic_profile(n_base, a1, a2))
    # Periodic profiles
    for p in [2, 3, 4, 5, 7, 8]:
        profiles.append(_make_periodic_profile(n_base, p, "smooth"))
    profiles.append(_make_zadoff_chu_real(n_base))

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_base, f"prof{idx}"))
        _update_best(p, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0: {len(candidates)} candidates, best={best_score:.10f}")

    # --- Phase 1: L-BFGS on top candidates ---
    print("=== Phase 1: L-BFGS continuation ===")
    phase1_end = min(deadline, t_start + 400.0)

    for sc, p, pn, label in candidates[:3]:
        if time.monotonic() > phase1_end:
            break
        if sc > best_score * 1.05:
            break
        remaining = phase1_end - time.monotonic()
        per_time = max(30.0, min(120.0, remaining / 3))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  L-BFGS on {label} (n={pn}), budget={per_time:.0f}s")
        p_best, sc_best = _lbfgs_continuation(p, dl, max_wall_time=per_time, top_k=0)
        _update_best(p_best, sc_best, f"lbfgs_{label}")
        # Follow with Adam
        if time.monotonic() < phase1_end - 10:
            dl2 = min(phase1_end, time.monotonic() + 30.0)
            p_best2, sc_best2 = _adam_optimize(p_best, dl2, lr=0.02, top_k=3, n_max_iter=15000)
            _update_best(p_best2, sc_best2, f"adam_{label}")

    print(f"  Phase 1: best={best_score:.10f}")

    # --- Phase 2: ILS (Iterated Local Search) ---
    print("=== Phase 2: ILS ===")
    phase2_end = min(deadline, t_start + 1200.0)

    if best_seq is not None:
        current = best_seq.copy()
        current_score = best_score

        n_ils = 0
        # Perturbation strategies cycle
        perturb_strategies = ["sinusoidal", "gaussian", "block_swap", "interp", "sinusoidal", "gaussian"]
        current_strategy_idx = 0

        while time.monotonic() < phase2_end - 30:
            n_ils += 1
            strategy = perturb_strategies[current_strategy_idx % len(perturb_strategies)]
            current_strategy_idx += 1

            # Choose perturbation strength based on stagnation
            amp_factor = 1.0 + 0.5 * (n_ils % 5)

            if strategy == "sinusoidal":
                perturbed = _perturb_sinusoidal(current, rng, n_freqs=3, max_amp=0.05 * amp_factor)
            elif strategy == "gaussian":
                perturbed = _perturb_gaussian(current, rng, sigma=0.03 * amp_factor)
            elif strategy == "block_swap":
                perturbed = _perturb_block_swap(current, rng, n_blocks=4, block_frac=0.08)
            elif strategy == "interp":
                perturbed = _perturb_interp(current, rng)
            else:
                perturbed = _perturb_sinusoidal(current, rng, n_freqs=5, max_amp=0.08 * amp_factor)

            # Local optimization (fast for ILS iterations)
            dl_local = min(phase2_end - 10, time.monotonic() + 45.0)
            p_opt, sc_opt = _local_optimize(perturbed, dl_local, phase="fast")
            _update_best(p_opt, sc_opt, f"ils{n_ils}_{strategy}")

            # Accept if better (or with probability for diversity)
            if sc_opt < current_score:
                current = p_opt.copy()
                current_score = sc_opt
            elif rng.random() < 0.1:
                # Occasionally accept equal or slightly worse for diversity
                current = p_opt.copy()

            if n_ils % 10 == 0:
                print(f"  ILS iter {n_ils}: current={current_score:.10f}, best={best_score:.10f}")

    print(f"  Phase 2: best={best_score:.10f}")

    # --- Phase 3: Deep optimization on best ---
    print("=== Phase 3: Deep optimization ===")
    phase3_end = min(deadline, t_start + 1500.0)
    if best_seq is not None:
        dl = min(phase3_end, time.monotonic() + (phase3_end - time.monotonic()))
        p_best, sc_best = _local_optimize(best_seq, dl, phase="full")
        _update_best(p_best, sc_best, "deep")

    print(f"  Phase 3: best={best_score:.10f}")

    # --- Phase 4: Try different lengths ---
    print("=== Phase 4: Different lengths ===")
    phase4_end = min(deadline, t_start + 1700.0)
    if best_seq is not None:
        n_curr = len(best_seq)
        for tn in [2048, 8192]:
            if time.monotonic() > phase4_end - 30:
                break
            if tn == n_curr:
                continue
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            _update_best(p_interp, sc_interp, f"interp_n{tn}")

            if time.monotonic() < phase4_end - 30:
                per_opt = min(60.0, (phase4_end - time.monotonic()) * 0.5)
                dl = min(phase4_end - 10, time.monotonic() + per_opt)
                p_best, sc_best = _lbfgs_continuation(p_interp, dl, max_wall_time=per_opt, top_k=0)
                _update_best(p_best, sc_best, f"lbfgs_n{tn}")

                if time.monotonic() < phase4_end - 15:
                    dl2 = min(phase4_end - 10, time.monotonic() + 20.0)
                    p_best2, sc_best2 = _adam_optimize(p_best, dl2, lr=0.02, top_k=5, n_max_iter=10000)
                    _update_best(p_best2, sc_best2, f"adam_n{tn}")

    print(f"  Phase 4: best={best_score:.10f}")

    # --- Phase 5: Final polish ---
    print("=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        # Final Adam with very small lr and large top_k
        if remaining > 30:
            dl = min(deadline - 5, time.monotonic() + min(60.0, remaining * 0.6))
            p_best, sc_best = _adam_optimize(best_seq, dl, lr=0.005, top_k=20, n_max_iter=20000)
            _update_best(p_best, sc_best, "final_adam")

        # Final KKT
        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 5, time.monotonic() + min(30.0, deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _kkt_balance_a(best_seq, dl2, max_iter=5000)
            _update_best(p_best2, sc_best2, "final_kkt")

        # One more subgradient pass
        if time.monotonic() < deadline - 5:
            a5 = best_seq.copy()
            for _ in range(100):
                if time.monotonic() > deadline - 5:
                    break
                a5 = _subgradient_step(a5, top_k=50, lr=0.002)
                sc5 = _score_norm_fn(a5)
                if sc5 < best_score:
                    best_score = sc5
                    best_seq = a5.copy()

    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.003)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check: {e}")

        return [float(x) for x in final_seq]

    fallback = _make_exp_profile(4096, 0.003)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
