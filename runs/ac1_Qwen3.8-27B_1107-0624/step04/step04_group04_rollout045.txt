3. Geometric top-k annealing with adaptive thresholds
4. Multi-length exploration with interpolation
5. Simulated annealing restarts for escaping local minima
6. Custom second-order correction for final polishing
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_topk_weights(c, top_k, beta):
    """Compute softmax weights over the top-k peaks of the convolution."""
    n = len(c)
    if top_k >= n:
        idx = np.arange(n)
        c_top = c
    else:
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
    c_max = c_top.max()
    c_norm = (c_top - c_max) / beta
    e = np.exp(c_norm)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(n)
        w[idx[np.argmax(c_top)]] = 1.0
        return w, c_max + 0.0
    w = np.zeros(n)
    w[idx] = e / s
    smoothed_max = c_max + beta * np.log(s)
    return w, smoothed_max


def _z_objective_v2(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    _, smoothed_max = _compute_topk_weights(c, top_k, beta)
    return 2.0 * n * smoothed_max


def _z_gradient_v2(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, _ = _compute_topk_weights(c, top_k, beta)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - float(np.dot(w, c)))
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _make_hann_window(n):
    return np.hanning(n) / np.hanning(n).sum()


def _make_hamming_window(n):
    return np.hamming(n) / np.hamming(n).sum()


def _make_blackman_window(n):
    return np.blackman(n) / np.blackman(n).sum()


def _make_kaiser_window(n, beta):
    from scipy.signal import get_window
    w = get_window(('kaiser', beta), n)
    return w / w.sum()


def _make_cazac_like(n, rng):
    """Generate a random sequence, project to non-negative, normalize."""
    # Use random phases to create a "CAZAC-like" amplitude pattern
    # The idea: take a random complex sequence with flat spectrum,
    # take its magnitude, then normalize.
    z = rng.standard_normal(n) + 1j * rng.standard_normal(n)
    Z = np.fft.fft(z)
    # Make it have approximately flat magnitude spectrum
    phase = np.angle(Z)
    Z_flat = np.exp(1j * phase)
    a = np.abs(np.fft.ifft(Z_flat).real)
    # Or use the real part directly
    a2 = np.real(np.fft.ifft(Z_flat))
    a2 = np.maximum(a2, 0)
    if a2.sum() < 1e-10:
        a2 = a
    return a2 / a2.sum()


def _make_random_taper(n, rng, taper_power):
    """Random shape with a taper: a[i] = base[i] * window[i]."""
    base = np.ones(n) + 0.1 * rng.standard_normal(n)
    base = np.maximum(base, 0.1)
    t = np.linspace(0, 1, n)
    window = np.sin(np.pi * t) ** taper_power
    a = base * window
    return a / a.sum()


def _make_bell_profile(n, center, width):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    return a / a.sum()


def _peak_shave(a_init, deadline, max_iter=50000, top_k_init=3, top_k_max=50):
    """
    Peak shaving optimizer:
    - Identifies top-k peaks in convolution
    - Computes the "pressure" (gradient contribution) from each peak
    - Moves mass from high-pressure to low-pressure positions
    - Uses line search to ensure monotone improvement
    """
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score_norm_fn(a)
    best_cmax = float(_full_convolve(a, a).max())

    top_k = top_k_init
    stall_count = 0
    plate_flag = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        if cm < best_cmax:
            best_cmax = cm
            best_a = a.copy()
            best_score = _score_norm_fn(a)
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive top-k: grow when stalled
        if stall_count > 50 and top_k < top_k_max:
            top_k = min(top_k_max, top_k + 2)
            stall_count = 0

        # Compute weighted pressure from top-k peaks
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        # Softmax weights over top-k (temperature proportional to spread)
        spread = c_top.max() - c_top.min() + 1e-15
        temp = max(spread * 0.5, 1e-10)
        w_top = np.exp((c_top - c_top.max()) / temp)
        w_top = w_top / w_top.sum()

        # Compute pressure: h[i] = sum_k w_k * a[k-i]
        h = np.zeros(n)
        for j, k_val in enumerate(idx):
            k_int = int(k_val)
            lo = max(0, k_int - (n - 1))
            hi = min(n - 1, k_int)
            if lo <= hi:
                h[lo:hi + 1] += w_top[j] * a[k_int - hi:k_int - lo + 1][::-1]

        # Direction: reduce high-pressure, increase low-pressure
        h_mean = float(h.mean())
        d = -(h - h_mean)

        # Ensure zero sum (it should be already by construction, but clip)
        d = d - d.mean()

        # Step size with line search
        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-15:
            break

        d = d / d_norm  # normalize direction
        c0 = cm
        step_size = 0.1 / d_norm * 0.01  # initial small step
        step_size = min(step_size, 0.5)

        improved = False
        for ls_iter in range(20):
            t = step_size
            a_trial = a + t * d
            if np.any(a_trial < 0):
                # Find max valid step
                neg = a_trial < 0
                if np.any(neg):
                    t_max = -np.min(a[neg] / d[neg])
                    t = max(0, t_max * 0.9)
                    if t < 1e-12:
                        break
                    a_trial = a + t * d

            c_trial = _full_convolve(a_trial, a_trial)
            cm_trial = float(c_trial.max())

            # Also check the smoothed objective (top-k)
            if cm_trial < c0 * (1 - 1e-10):
                a = a_trial
                improved = True
                break
            else:
                t *= 0.5
                step_size *= 0.5

        if not improved:
            # Try in log-space with smaller step
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            # Gradient in log-space
            a_cur = _softmax_fn(z)
            h2 = np.zeros(n)
            for j, k_val in enumerate(idx):
                k_int = int(k_val)
                lo = max(0, k_int - (n - 1))
                hi = min(n - 1, k_int)
                if lo <= hi:
                    h2[lo:hi + 1] += w_top[j] * a_cur[k_int - hi:k_int - lo + 1][::-1]
            h2_mean = float(h2.mean())
            grad_z = a_cur * (h2 - h2_mean)
            grad_norm = float(np.linalg.norm(grad_z))
            if grad_norm < 1e-15:
                break
            for ls_iter in range(10):
                step = 0.01 / grad_norm
                z_trial = z - step * grad_z
                z_trial = np.clip(z_trial, -80, 80)
                z_trial = z_trial - z_trial.mean()
                a_trial = _softmax_fn(z_trial)
                c_trial = _full_convolve(a_trial, a_trial)
                cm_trial = float(c_trial.max())
                if cm_trial < c0 * (1 - 1e-10):
                    z = z_trial
                    a = a_trial
                    improved = True
                    break
                step *= 0.5

        if not improved and stall_count > 200:
            # Perturb to escape local minimum
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z += 0.01 * np.random.default_rng(it).standard_normal(n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a = _softmax_fn(z)
            stall_count = 0

    return best_a, best_score


def _lbfgs_v2(a_init, deadline, max_wall_time=None, top_k=10):
    """L-BFGS with geometric beta annealing in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())

    # Geometric beta schedule
    n_levels = 8
    beta_start = 0.05 * max(max_c0, 1e-10)
    beta_end = 0.0005 * max(max_c0, 1e-10)
    ratios = np.exp(np.linspace(np.log(beta_end / beta_start), 0, n_levels))
    betas = beta_start * ratios / ratios[-1]
    # Actually let's do: beta_i = beta_start * (beta_end/beta_start)^(i/(n-1))
    betas = beta_start * (beta_end / beta_start) ** (np.arange(n_levels) / (n_levels - 1))
    # Reverse so we start large
    betas = betas[::-1]

    per_level = max_wall_time / n_levels

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        try:
            result = minimize(
                _z_objective_v2, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient_v2,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_v2(a_init, deadline, lr=0.03, top_k_init=5, top_k_max=30, n_max_iter=100000):
    """Adam optimizer with adaptive top-k and geometric beta annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = 0.02 * max(max_c, 1e-10)
    cur_k = top_k_init

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        grad = _z_gradient_v2(z, beta, n, cur_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptive schedule
        if t % 200 == 0:
            beta = max(1e-15, beta * 0.9)
        if t % 500 == 0:
            cur_k = min(top_k_max, cur_k + 1)
        if t % 5000 == 0 and best_score < 1.51:
            lr = max(0.002, lr * 0.95)

    return best_a, best_score


def _second_order_polish(a_init, deadline, max_iter=5000):
    """
    Second-order polish using the curvature of the objective at the active set.
    At the optimum, the Hessian of the smoothed objective should be positive definite
    on the constraint subspace. We use this to take a better step.
    """
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Active set: top peaks within 0.5% of max
        threshold = cm * 0.9995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            break

        # Compute the "balanced" gradient direction
        w = np.zeros(len(c))
        c_top = c[active]
        temp = max(c_top.max() - c_top.min(), 1e-10) * 0.3
        w[active] = np.exp((c_top - c_top.max()) / temp)
        w[active] /= w[active].sum()

        # h[i] = sum_k w[k] * a[k-i]
        h = np.zeros(n)
        for k_val in active:
            k_int = int(k_val)
            lo = max(0, k_int - (n - 1))
            hi = min(n - 1, k_int)
            if lo <= hi:
                h[lo:hi + 1] += w[k_int] * a[k_int - hi:k_int - lo + 1][::-1]

        W = float(np.dot(h, a))
        residual = h - W  # Should be ~0 at optimum

        # Second-order correction: estimate local curvature
        # The Hessian of c[k] w.r.t. a is related to the "shifted" a values
        # For the smoothed max, the Hessian involves the covariance of h
        # Simplified: use the variance of residual as a curvature proxy
        var_res = float(np.var(residual))
        if var_res < 1e-20:
            break

        # Step in log-space with curvature-aware sizing
        step_size = 0.005 / (var_res + 1e-10) ** 0.5
        step_size = min(step_size, 0.1)

        z_step = z - step_size * residual / (np.abs(residual).max() + 1e-15)
        z_step = np.clip(z_step, -80, 80)
        z_step = z_step - z_step.mean()
        a_new = _softmax_fn(z_step)
        sc_new = _score_norm_fn(a_new)

        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            z = z_step
        else:
            # Halve step
            z_half = z - step_size * 0.5 * residual / (np.abs(residual).max() + 1e-15)
            z_half = np.clip(z_half, -80, 80)
            z_half = z_half - z_half.mean()
            a_half = _softmax_fn(z_half)
            sc_half = _score_norm_fn(a_half)
            if np.isfinite(sc_half) and sc_half < best_score:
                best_score = sc_half
                best_a = a_half.copy()
                a = a_half
                z = z_half

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining peak shaving, L-BFGS, Adam,
    and second-order polishing. Uses diverse initialization and
    multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 0: Diverse initialization ---
    print("=== Phase 0: Diverse initialization ===")
    n_search = 4096
    candidates = []

    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                candidates.append((sc, parent.copy(), "parent"))
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # Window functions
    for name, fn in [("hann", _make_hann_window), ("hamming", _make_hamming_window),
                     ("blackman", _make_blackman_window)]:
        p = fn(n_search)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, name))
        update_best(p, sc, name)

    # Kaiser windows with different beta
    for kb in [2.0, 4.0, 6.0, 8.0, 12.0, 20.0]:
        p = _make_kaiser_window(n_search, kb)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"kaiser_{kb}"))
        update_best(p, sc, f"kaiser_{kb}")

    # CAZAC-like
    for i in range(4):
        p = _make_cazac_like(n_search, rng)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"cazac_{i}"))
        update_best(p, sc, f"cazac_{i}")

    # Bell profiles
    for center, width in [(0.5, 0.2), (0.5, 0.3), (0.5, 0.4), (0.5, 0.5), (0.3, 0.4), (0.7, 0.4)]:
        p = _make_bell_profile(n_search, center, width)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"bell_{center}_{width}"))
        update_best(p, sc, f"bell_{center}_{width}")

    # Random tapers
    for tp in [0.5, 1.0, 1.5, 2.0]:
        p = _make_random_taper(n_search, rng, tp)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"taper_{tp}"))
        update_best(p, sc, f"taper_{tp}")

    # Exponential profiles
    for alpha in [0.001, 0.002, 0.005, 0.01]:
        r = 1.0 - alpha
        p = r ** np.arange(n_search)
        p = p / p.sum()
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"exp_{alpha}"))
        update_best(p, sc, f"exp_{alpha}")

    # Interpolate parent to n_search
    if parent is not None and len(parent) != n_search:
        p_interp = _interp_profile(parent, n_search)
        sc = _score_norm_fn(p_interp)
        candidates.append((sc, p_interp, "parent_interp"))
        update_best(p_interp, sc, "parent_interp")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")
    for sc, p, name in candidates[:5]:
        print(f"    {name}: {sc:.10f}")

    # --- Phase 1: Peak shaving on top candidates ---
    print("=== Phase 1: Peak shaving ===")
    phase1_end = min(deadline, t_start + 400.0)
    n_ps = min(3, len(candidates))
    for i in range(n_ps):
        if time.monotonic() > phase1_end:
            break
        sc, p, name = candidates[i]
        if sc > best_score * 1.05:
            break
        remaining = phase1_end - time.monotonic()
        per_time = max(30.0, remaining / max(1, n_ps - i))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  Peak shave {name} (n={len(p}), score={sc:.8f}), budget={per_time:.1f}s")
        p_best, sc_best = _peak_shave(p, dl, max_iter=30000, top_k_init=3, top_k_max=40)
        update_best(p_best, sc_best, f"ps_{name}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: L-BFGS refinement ---
    print("=== Phase 2: L-BFGS refinement ===")
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + min(200.0, remaining))
        print(f"  L-BFGS on best (n={len(best_seq)}), budget={dl - time.monotonic():.1f}s")
        p_best, sc_best = _lbfgs_v2(best_seq, dl, max_wall_time=max(10.0, dl - time.monotonic()), top_k=20)
        update_best(p_best, sc_best, "lbfgs_v2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam polish ---
    print("=== Phase 3: Adam polish ===")
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + remaining)
        print(f"  Adam on best (n={len(best_seq)}), budget={remaining:.1f}s")
        p_best, sc_best = _adam_v2(best_seq, dl, lr=0.03, top_k_init=5, top_k_max=30, n_max_iter=80000)
        update_best(p_best, sc_best, "adam_v2")

        # Second pass with smaller lr
        if time.monotonic() < phase3_end - 20:
            dl2 = min(phase3_end - 5, time.monotonic() + (phase3_end - time.monotonic()) * 0.4)
            p_best2, sc_best2 = _adam_v2(best_seq, dl2, lr=0.01, top_k_init=10, top_k_max=40, n_max_iter=40000)
            update_best(p_best2, sc_best2, "adam_v2b")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Second-order polish ---
    print("=== Phase 4: Second-order polish ===")
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        remaining = phase4_end - time.monotonic()
        dl = min(phase4_end, time.monotonic() + min(150.0, remaining))
        print(f"  2nd order polish, budget={dl - time.monotonic():.1f}s")
        p_best, sc_best = _second_order_polish(best_seq, dl, max_iter=10000)
        update_best(p_best, sc_best, "second_order")

        # Follow up with peak shaving
        if time.monotonic() < phase4_end - 10:
            dl2 = min(phase4_end - 5, time.monotonic() + min(60.0, phase4_end - time.monotonic()))
            p_best2, sc_best2 = _peak_shave(best_seq, dl2, max_iter=10000, top_k_init=5, top_k_max=30)
            update_best(p_best2, sc_best2, "ps_post")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart with simulated annealing ---
    print("=== Phase 5: Multi-restart ===")
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = min(50.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break
            # Annealed perturbation
            noise = 0.03 * (1 + i * 0.3)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.standard_normal(n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            # Quick peak shave then Adam
            per_ps = per_restart * 0.5
            per_adam = per_restart * 0.45
            p_m, sc_m = _peak_shave(p_start, min(phase5_end - 5, time.monotonic() + per_ps),
                                    max_iter=8000, top_k_init=3, top_k_max=20)
            update_best(p_m, sc_m, f"r{i}_ps")
            if time.monotonic() < phase5_end - 5:
                p_a, sc_a = _adam_v2(p_m, min(phase5_end - 5, time.monotonic() + per_adam),
                                     lr=0.02, top_k_init=5, top_k_max=20, n_max_iter=15000)
                update_best(p_a, sc_a, f"r{i}_adam")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different lengths ---
    print("=== Phase 6: Different lengths ===")
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = [2048, 8192]
        target_ns = [tn for tn in target_ns if tn != n_curr]

        for tn in target_ns:
            if time.monotonic() > phase6_end - 40:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            remaining = phase6_end - time.monotonic()
            if remaining < 30:
                break
            per_opt = min(80.0, remaining * 0.5)
            dl = min(phase6_end - 10, time.monotonic() + per_opt)
            p_best, sc_best = _peak_shave(p_interp, dl, max_iter=10000, top_k_init=3, top_k_max=30)
            update_best(p_best, sc_best, f"ps_n{tn}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final intensive polish ---
    print("=== Phase 7: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        # Peak shave with very small steps
        if remaining > 30:
            dl = min(deadline - 5, time.monotonic() + min(60.0, remaining * 0.4))
            p_best, sc_best = _peak_shave(best_seq, dl, max_iter=20000, top_k_init=1, top_k_max=10)
            update_best(p_best, sc_best, "final_ps")

        # Second order
        if time.monotonic() < deadline - 20:
            dl2 = min(deadline - 5, time.monotonic() + min(30.0, (deadline - time.monotonic()) * 0.4))
            p_best2, sc_best2 = _second_order_polish(best_seq, dl2, max_iter=5000)
            update_best(p_best2, sc_best2, "final_2nd")

        # Very small Adam steps
        if time.monotonic() < deadline - 10:
            dl3 = min(deadline - 3, time.monotonic() + min(20.0, deadline - time.monotonic() - 3))
            p_best3, sc_best3 = _adam_v2(best_seq, dl3, lr=0.005, top_k_init=1, top_k_max=5, n_max_iter=10000)
            update_best(p_best3, sc_best3, "final_adam")

    # --- Final validation and return ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_hann_window(4096)
            final_score = _score_fn(final_seq)

        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")

        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 10000:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check failed: {e}")

        return [float(x) for x in final_seq]

    fallback = _make_hann_window(4096)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
