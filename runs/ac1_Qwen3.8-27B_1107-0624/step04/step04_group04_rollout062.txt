```python
from scipy.sparse import csr_matrix

    c = np.convolve(a, a)

def _construct_conv_matrix(f):
    """Construct the (2n-1) x n convolution matrix A where A @ g = conv(f, g)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for j in range(2 * n - 1):
        k_lo = max(0, j - n + 1)
        k_hi = min(j, n - 1)
        for k in range(k_lo, k_hi + 1):
            rows.append(j)
            cols.append(k)
            vals.append(f[j - k])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_direction(f, M, timeout=60):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = _construct_conv_matrix(f)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(objective, constraints)
    try:

            return g.value
    except Exception:

        pass
    return None
```

```python
"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding combined with logsumexp-smoothed local optimization.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smooth_obj_and_grad(a, beta, n, top_k=0):
    """Compute smoothed objective and gradient in logit space."""
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w[idx[np.argmax(c_top)]] = 1.0
        else:
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    # Objective: 2*n*W (approximately, since W is the softmax-weighted sum)
    obj = 2.0 * n * W
    # Gradient w.r.t. a: 2*n * h (since dW/da_i = h_i)
    grad_a = 2.0 * n * h
    return obj, grad_a, w, c, cm


def _z_objective_grad(z, beta, n, top_k=0):
    """Objective and gradient in logit space."""
    a = _softmax_fn(z)
    obj, grad_a, w, c, cm = _compute_smooth_obj_and_grad(a, beta, n, top_k)
    # grad_z_i = sum_j grad_a_j * da_j/dz_i = grad_a_i * a_i - a_i * sum_j(grad_a_j * a_j)
    grad_z = grad_a * a - a * np.dot(grad_a, a)
    return obj, grad_z


def _adam_smooth(a_init, deadline, lr=0.05, beta_init=None, top_k=0, n_max_iter=100000):
    """Adam optimizer with adaptive beta for the smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    if beta_init is None:
        beta = max(1e-10, 0.01 * max_c)
    else:
        beta = beta_init

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t_count += 1
        a = _softmax_fn(z)
        obj, grad = _z_objective_grad(z, beta, n, top_k)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t_count)
        vh = v / (1 - b2 ** t_count)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t_count % 100 == 0:
            beta = max(1e-14, beta * 0.85)
        if t_count % 2000 == 0 and best_score < 1.51:
            lr = max(0.002, lr * 0.95)

    return best_a, best_score


def _lbfgs_smooth(a_init, deadline, beta_init=None, top_k=0):
    """L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    max_wall_time = deadline - time.monotonic()
    if max_wall_time <= 0:
        return best_a, best_score

    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]
    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj_fn(z_local):
            obj, _ = _z_objective_grad(z_local, beta, n, top_k)
            return obj

        def grad_fn(z_local):
            _, grad = _z_objective_grad(z_local, beta, n, top_k)
            return grad

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 30, 'maxfun': 1000,
                         'max_time': max(1, level_deadline - time.monotonic())}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _lp_direction(f, M, max_time=30):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Uses HiGHS."""
    n = len(f)
    # Build constraint matrix A where A @ g = conv(f, g)
    # A has shape (2n-1, n), A[j,k] = f[j-k] if 0 <= j-k < n
    rows = []
    cols = []
    vals = []
    for j in range(2 * n - 1):
        k_lo = max(0, j - n + 1)
        k_hi = min(j, n - 1)
        for k in range(k_lo, k_hi + 1):
            rows.append(j)
            cols.append(k)
            vals.append(f[j - k])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, 
                  method='highs', options={'time_limit': max_time})
    
    if res.success and res.x is not None:
        g = np.clip(res.x, 0, None)
        if g.sum() > 1e-10:
            return g
    return None


def _lp_iteration(f, deadline, max_iters=50, lp_timeout=20):
    """LP-guided iteration: repeatedly find LP direction and mix."""
    n = len(f)
    f_current = f.copy()
    best_f = f.copy()
    best_score = _score_norm_fn(f)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        c = _full_convolve(f_current, f_current)
        M = float(c.max())
        
        # Solve LP
        t0 = time.monotonic()
        g = _lp_direction(f_current, M, max_time=lp_timeout)
        lp_time = time.monotonic() - t0
        
        if g is None:
            break
        
        # Rescale g to have same sum as f_current
        g_sum = g.sum()
        f_sum = f_current.sum()
        if g_sum < 1e-10:
            break
        g = g * (f_sum / g_sum)
        
        # Line search for mixing parameter
        best_t = 0
        best_new_score = best_score
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            f_new = (1 - t) * f_current + t * g
            sc = _score_norm_fn(f_new)
            if np.isfinite(sc) and sc < best_new_score:
                best_new_score = sc
                best_t = t
        
        if best_t == 0:
            # Try very small t
            for t in [0.005, 0.001, 0.0005]:
                f_new = (1 - t) * f_current + t * g
                sc = _score_norm_fn(f_new)
                if np.isfinite(sc) and sc < best_new_score:
                    best_new_score = sc
                    best_t = t
        
        if best_t > 0:
            f_current = (1 - best_t) * f_current + best_t * g
            if best_new_score < best_score:
                best_score = best_new_score
                best_f = f_current.copy()
            print(f"    LP it {it}: t={best_t}, score={best_score:.10f}, M={M:.6f}, lp_time={lp_time:.1f}s")
        else:
            break
    
    return best_f, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    # Uniform
    profiles.append(np.ones(n) / n)
    # Exponential decay
    for alpha in [0.001, 0.003, 0.008, 0.02, 0.05]:
        a = (1 - alpha) ** np.arange(n)
        profiles.append(a / a.sum())
    # U-shaped
    t = np.linspace(0, 1, n)
    for sharp in [2, 4, 8, 16]:
        a = t ** sharp + (1 - t) ** sharp
        profiles.append(a / a.sum())
    # Cosine bump
    for freq in [1, 2, 3]:
        a = 1 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 1e-6)
        profiles.append(a / a.sum())
    # Linear
    for slope in [-2, -1, 1, 2]:
        a = 1 + slope * (t - 0.5)
        a = np.maximum(a, 1e-6)
        profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
```

If the active set is S (|S| = m points), we want to minimize:

This is a quadratic function of a (since h_j(a) = a^T T_j a where T_j is a symmetric matrix). We can compute the Hessian and use Newton's method or quasi-Newton methods.

The Hessian of (1/m) sum_{j in S} h_j(a) is:
(2/m) sum_{j in S} T_j

where T_j is the "convolution matrix" at lag j: T_j[i,k] = 1 if j-i = k, i.e., k = j-i.

So T_j is a permutation matrix (shift matrix). The Hessian is:
H = (2/m) sum_{j in S} T_j

This is a banded matrix (each T_j is a shift, so the sum is banded with bandwidth max(S) - min(S)).

For a "tight" active set (all j in S are close together), the Hessian is approximately a constant (all the shifts are similar), and Newton's method would be very effective.

For a "spread out" active set, the Hessian has more structure, but it's still banded.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_active_grad(a, c, epsilon=0.002):

        if lo <= hi:

    grad *= 2.0 / m
    return grad, m, cm

def _lbfgs_smooth(a_init, deadline, top_k=0):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    max_wall_time = deadline - time.monotonic()
    if max_wall_time <= 1:
        return best_a, best_score

    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]
    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj_fn(z_local):
            a_l = _softmax_fn(z_local)
            c_l = _full_convolve(a_l, a_l)
            cm_l = float(c_l.max())
            if top_k > 0 and top_k < len(c_l):
                idx_l = np.argsort(c_l)[-top_k:]
                c_top_l = c_l[idx_l]
                c_norm_l = c_top_l - c_top_l.max()
                log_sum = float(np.log(np.sum(np.exp(c_norm_l / beta))))
                return 2.0 * n * (c_top_l.max() + beta * log_sum)
            else:
                c_shifted_l = (c_l - cm_l) / beta
                log_sum = float(np.log(np.sum(np.exp(c_shifted_l))))
                return 2.0 * n * (cm_l + beta * log_sum)

        def grad_fn(z_local):
            a_l = _softmax_fn(z_local)
            c_l = _full_convolve(a_l, a_l)
            cm_l = float(c_l.max())
            if top_k > 0 and top_k < len(c_l):
                idx_l = np.argsort(c_l)[-top_k:]
                w_l = np.zeros_like(c_l)
                c_top_l = c_l[idx_l]
                c_norm_l = (c_top_l - c_top_l.max()) / beta
                e_l = np.exp(c_norm_l)
                s_l = e_l.sum()
                if s_l < 1e-300 or not np.isfinite(s_l):
                    w_l[idx_l[np.argmax(c_top_l)]] = 1.0
                else:
                    w_l[idx_l] = e_l / s_l
            else:
                c_shifted_l = (c_l - cm_l) / beta
                e_l = np.exp(c_shifted_l)
                s_l = float(e_l.sum())
                if s_l < 1e-300 or not np.isfinite(s_l):
                    idx_l = int(np.argmax(c_l))
                    w_l = np.zeros_like(c_l)
                    w_l[idx_l] = 1.0
                else:
                    w_l = e_l / s_l
            W_l = float(np.dot(w_l, c_l))
            a_rev_l = a_l[::-1]
            w_conv_l = _full_convolve(w_l, a_rev_l)
            h_l = w_conv_l[n - 1:2 * n - 1]
            grad_a_l = 2.0 * n * h_l
            grad_z_l = grad_a_l * a_l - a_l * np.dot(grad_a_l, a_l)
            if not np.all(np.isfinite(grad_z_l)):
                return np.zeros_like(z_local)
            return grad_z_l

        try:
            result = minimize(
                obj_fn, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10,

            if result.x is not None and np.all(np.isfinite(result.x)):

def _solve_lp(f, M, max_time=25):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    # Build sparse constraint matrix
    total_rows = 2 * n - 1
    nnz_estimate = 0
    for j in range(total_rows):
        k_lo = max(0, j - n + 1)
        k_hi = min(j, n - 1)
        nnz_estimate += (k_hi - k_lo + 1)

    if nnz_estimate > 50000000:
        return None  # Too large

    rows = np.empty(nnz_estimate, dtype=np.int32)
    cols = np.empty(nnz_estimate, dtype=np.int32)
    vals = np.empty(nnz_estimate, dtype=np.float64)
    ptr = 0
    for j in range(total_rows):
        k_lo = max(0, j - n + 1)
        k_hi = min(j, n - 1)
        cnt = k_hi - k_lo + 1
        rows[ptr:ptr + cnt] = j
        cols[ptr:ptr + cnt] = np.arange(k_lo, k_hi + 1)
        vals[ptr:ptr + cnt] = f[j - k_hi:j - k_lo + 1][::-1]
        ptr += cnt

    A = csr_matrix((vals[:ptr], (rows[:ptr], cols[:ptr])), shape=(total_rows, n))

    c_obj = -np.ones(n)
    b_ub = np.full(total_rows, M)
    bounds = [(0.0, None)] * n

    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': max_time, 'presolve': True})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            if g.sum() > 1e-10 and np.all(np.isfinite(g)):
                return g
    except Exception:
        pass

    if time.monotonic() - t0 < max_time * 0.5:
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': max_time - (time.monotonic() - t0)})
            if res.success and res.x is not None:
                g = np.clip(res.x, 0, None)
                if g.sum() > 1e-10 and np.all(np.isfinite(g)):
                    return g
        except Exception:
            pass

    return None


def _lp_iteration(f, deadline, max_iters=30, lp_timeout=20):
    """LP-guided iteration: find LP direction, line search, update."""
    n = len(f)
    f_current = f.copy()
    best_f = f.copy()
    best_score = _score_norm_fn(f)

    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break

        c = _full_convolve(f_current, f_current)
        M = float(c.max())

        t0 = time.monotonic()
        g = _solve_lp(f_current, M, max_time=lp_timeout)
        lp_time = time.monotonic() - t0

        if g is None:
            break

        g_sum = g.sum()
        f_sum = f_current.sum()
        if g_sum < 1e-10:
            break
        g = g * (f_sum / g_sum)

        # Line search
        best_t = 0.0
        best_new_score = _score_norm_fn(f_current)
        for t_val in [0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]:
            f_new = (1.0 - t_val) * f_current + t_val * g
            sc = _score_norm_fn(f_new)
            if np.isfinite(sc) and sc < best_new_score:
                best_new_score = sc
                best_t = t_val

        if best_t > 0:
            f_current = (1.0 - best_t) * f_current + best_t * g
            if best_new_score < best_score:
                best_score = best_new_score
                best_f = f_current.copy()
            print(f"    LP it {it}: t={best_t:.4f}, score={best_score:.10f}, "
                  f"M={M:.8f}, sum_g={g_sum:.4f}, lp_time={lp_time:.1f}s")
        else:
            break

    return best_f, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    profiles.append(np.ones(n) / n)
    for alpha in [0.001, 0.003, 0.008, 0.02, 0.05]:
        a = (1.0 - alpha) ** np.arange(n)
        profiles.append(a / a.sum())
    t = np.linspace(0, 1, n)
    for sharp in [2, 4, 8, 16, 32]:
        a = t ** sharp + (1.0 - t) ** sharp
        profiles.append(a / a.sum())
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t)
        a = np.maximum(a, 1e-6)
        profiles.append(a / a.sum())
    for slope in [-2.0, -1.0, 1.0, 2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-6)
        profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _random_profile(n, rng, sharpness=1.0):
    """Generate a random non-negative profile."""
    z = rng.normal(size=n) * sharpness
    a = np.exp(z)
    return a / a.sum()
