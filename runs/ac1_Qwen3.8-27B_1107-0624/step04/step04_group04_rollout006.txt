"""
Water-filling optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative gradient updates with FFT-accelerated subgradient computation,
adaptive active set management, backtracking line search, and L-BFGS polishing.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_sum1(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_sg(a, active_indices):
    n = len(a)
    c_len = 2 * n - 1
    ind = np.zeros(c_len)
    ind[active_indices] = 1.0
    N = _next_pow2(c_len + n - 1)
    F = np.fft.rfft(ind, N)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad, N)
    corr = np.fft.irfft(F * np.conj(A), N)[:n]
    sg = (2.0 / len(active_indices)) * corr
    return sg


def _water_filling_iter(a, active_indices, eta):
    sg = _compute_sg(a, active_indices)
    lam = float(np.dot(a, sg) / a.sum())
    log_a = np.log(np.maximum(a, 1e-30))
    log_a -= eta * (sg - lam)
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new, sg, lam


def _wf_optimize(a_init, deadline, eta_init=0.3, top_start=3, top_end=100, thresh=0.005):
    n = len(a_init)
    a = a_init.copy()
    a /= a.sum()
    best_a = a.copy()
    best_sc = _score_sum1(a)
    eta = eta_init
    no_improve = 0
    patience = 300

    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        sc = 2.0 * n * cm

        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            eta *= 0.6
            if eta < 1e-7:
                break
            no_improve = 0

        progress = min(1.0, float(it) / 5000.0)
        n_top = max(3, int(top_start + progress * (top_end - top_start)))

        thr_val = cm * (1.0 - thresh)
        active = np.where(c >= thr_val)[0]
        if len(active) < n_top:
            idx = np.argpartition(c, -n_top)[-n_top:]
            active = np.sort(idx)
        elif len(active) > 500:
            idx = np.argpartition(c, -500)[-500:]
            active = np.sort(idx)

        a_new, sg, lam = _water_filling_iter(a, active, eta)
        c_new = _full_convolve(a_new, a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < sc + 1e-14:
            a = a_new
            eta = min(0.8, eta * 1.02)
        else:
            eta *= 0.5
            if eta < 1e-8:
                break

        if it % 1000 == 0:
            print(f"    WF it={it}, score={sc:.10f}, eta={eta:.6f}, |A|={len(active)}")

    return best_a, best_sc


def _logsumexp_obj(z, beta, n, top_k):
    a = np.exp(z - z.max())
    a /= a.sum()
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        ct_shift = ct - ct.max()
        return 2.0 * n * (ct.max() + beta * np.log(np.sum(np.exp(ct_shift / beta))))
    else:
        cs = c - c.max()
        return 2.0 * n * (c.max() + beta * np.log(np.sum(np.exp(cs / beta))))


def _logsumexp_grad(z, beta, n, top_k):
    a = np.exp(z - z.max())
    a /= a.sum()
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        ct_shift = ct - ct.max()
        w_t = np.exp(ct_shift / beta)
        w_t /= w_t.sum()
        w = np.zeros(len(c))
        w[idx] = w_t
    else:
        cs = c - c.max()
        w = np.exp(cs / beta)
        w /= w.sum()
    W = float(np.dot(w, c))
    n_conv = n
    w_pad = np.zeros(_next_pow2(2 * n - 1))
    w_pad[:len(w)] = w
    a_rev = a[::-1]
    N = _next_pow2(2 * n - 1)
    Wr = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a_rev, N)
    h_full = np.fft.irfft(Wr * Ar, N)[:2 * n - 1]
    h = h_full[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _lbfgs_polish(a_init, deadline, top_k=10, max_wall=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-20))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_sum1(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    betas = [b * max(max_c0, 1e-10) for b in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]

    if max_wall is None:
        max_wall = max(1.0, deadline - time.monotonic())
    per_beta = max_wall / len(betas)

    for beta in betas:
        if time.monotonic() > deadline - 2:
            break
        level_dl = min(deadline - 2, time.monotonic() + per_beta)
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 20, 'maxfun': 800}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = np.exp(result.x - result.x.max())
                a_new /= a_new.sum()
                sc_new = _score_sum1(a_new)
                if np.isfinite(sc_new) and sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_sc


def _adam_polish(a_init, deadline, lr=0.03, top_k=5, max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-20))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_sum1(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    beta = 0.01 * float(_full_convolve(a_init, a_init).max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        grad = _logsumexp_grad(z, beta, n, top_k)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros_like(z)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if it % 200 == 0:
            a = np.exp(z - z.max())
            a /= a.sum()
            sc = _score_sum1(a)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            beta = max(1e-15, beta * 0.85)

    if t % 200 != 0:
        a = np.exp(z - z.max())
        a /= a.sum()
        sc = _score_sum1(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind, param):
    if kind == 'exp':
        a = np.exp(-param * np.arange(n))
    elif kind == 'linear':
        t = np.linspace(0, 1, n)
        a = np.maximum(1.0 + param * (t - 0.5), 0.01)
    elif kind == 'quadratic':
        t = np.linspace(0, 1, n)
        u = 2.0 * t - 1.0
        a = np.maximum(1.0 + param * u ** 2, 0.01)
    elif kind == 'power':
        a = (np.arange(n) + 1.0) ** param
    elif kind == 'bimodal':
        t = np.linspace(0, 1, n)
        a = 1.0 - param * np.cos(2 * np.pi * t)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
