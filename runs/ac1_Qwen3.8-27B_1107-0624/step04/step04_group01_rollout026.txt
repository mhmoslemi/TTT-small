import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = fft(a, N)
    fb = fft(b, N)
    r = ifft(fa * fb, N)[:out_n].real
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_objective(z, beta, n):
    """Compute smooth objective and gradient."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = c.max()
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Smooth objective value (up to additive constant)
    obj = 2.0 * n * (cm + beta * np.log(s))
    
    # Gradient
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1 : 2*n-1]
    Wc = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - Wc) / beta
    
    return obj, grad

Multi-phase optimization using L-BFGS-B with log-sum-exp smoothing,
asymmetric initialization, multi-restart, and coordinate descent polishing
to minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_obj_grad(z, beta):
    """Compute smooth objective and gradient for softmax parameterization."""
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    obj = 2.0 * n * (cm + beta * np.log(s))
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1 : 2*n-1]
    Wc = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - Wc) / beta
    
    return obj, grad

        p = _make_asymmetric(n_main, kind, rng)
        update_best(p, _score(p), f"init_{kind}")

    # Perturbed parent (asymmetric)
    if parent is not None:
        for sigma in [0.05, 0.1, 0.2]:
            if time.monotonic() > phase1_end:
                break
            z = np.log(np.maximum(parent, 1e-15)) + sigma * rng.standard_normal(n_main)
            z = z - z.mean()
            p = _softmax(z)
            update_best(p, _score(p), f"perturb_s{sigma}")

    # Random Fourier init
    for nh in [10, 25]:
        if time.monotonic() > phase1_end:
            break
        coeffs = rng.standard_normal(2 * nh) * 0.3
        t_grid = np.arange(n_main, dtype=np.float64)
        a = np.ones(n_main) * 0.5
        for k in range(nh):
            a = a + coeffs[2 * k] * np.cos(2 * np.pi * (k + 1) * t_grid / n_main)
            a = a + coeffs[2 * k + 1] * np.sin(2 * np.pi * (k + 1) * t_grid / n_main)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
        update_best(a, _score(a), f"fourier_nh{nh}")

    print(f"Phase 1 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 2: Main L-BFGS-B optimization (60s-500s) ============
    phase2_end = min(deadline, t_start + 500.0)
    print(f"Phase 2: L-BFGS-B main, deadline={phase2_end - t_start:.1f}s")

    if best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        max_cb = float(c_b.max())
        # Beta schedule: start moderate, decrease
        beta_schedule = [
            0.03 * max_cb, 0.015 * max_cb, 0.008 * max_cb,
            0.004 * max_cb, 0.002 * max_cb
        ]
        for i, beta in enumerate(beta_schedule):
            if time.monotonic() > phase2_end - 30:
                break
            per_deadline = min(phase2_end - 10, time.monotonic() + (phase2_end - 60 - time.monotonic()) / max(1, len(beta_schedule) - i - 1))
            a_opt, sc_opt, _ = _lbfgsb_phase(best_seq, per_deadline, beta, max_iter=2000)
            update_best(a_opt, sc_opt, f"lbfgs_b{i}")

    print(f"Phase 2 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 3: Adam refinement (500s-800s) ============
    phase3_end = min(deadline, t_start + 800.0)
    print(f"Phase 3: Adam refinement, deadline={phase3_end - t_start:.1f}s")

    if best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        max_cb = float(c_b.max())
        for beta in [0.001 * max_cb, 0.0005 * max_cb, 0.0002 * max_cb]:
            if time.monotonic() > phase3_end - 20:
                break
            a_opt, sc_opt = _adam_phase(best_seq, time.monotonic() + (phase3_end - time.monotonic() - 10) / 3,
                                        beta, lr=0.008, max_iter=50000)
            update_best(a_opt, sc_opt, f"adam_b{beta:.6f}")

    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 4: Multi-restart with diverse perturbations (800s-1200s) ============
    phase4_end = min(deadline, t_start + 1200.0)
    print(f"Phase 4: Multi-restart, deadline={phase4_end - t_start:.1f}s")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        n_restarts = 6
        time_per = (phase4_end - time.monotonic() - 20) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break
            # Diverse perturbation strategies
            if i % 3 == 0:
                sigma = 0.08 + 0.04 * (i // 3)
                z = np.log(np.maximum(best_seq, 1e-15)) + sigma * rng.standard_normal(n_main)
            elif i % 3 == 1:
                # Fourier perturbation
                z = np.log(np.maximum(best_seq, 1e-15))
                freq = rng.integers(1, n_main // 8)
                amp = 0.1 * rng.uniform(0.5, 2.0)
                t = np.arange(n_main)
                z = z + amp * np.cos(2 * np.pi * freq * t / n_main + rng.uniform(0, 2 * np.pi))
            else:
                # Block shift
                z = np.log(np.maximum(best_seq, 1e-15))
                block = n_main // 4
                shift = rng.integers(1, block + 1)
                z = np.roll(z, shift)

            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax(z)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            beta = 0.01 * max_cs

            deadline_this = time.monotonic() + time_per
            a_opt, sc_opt, _ = _lbfgsb_phase(p_start, deadline_this, beta, max_iter=1500)
            update_best(a_opt, sc_opt, f"restart{i}")

            # Quick Adam from the LBFGS result
            if time.monotonic() < deadline_this - 2:
                a_opt2, sc_opt2 = _adam_phase(a_opt, deadline_this - 1, 0.003 * max_cs, lr=0.005, max_iter=10000)
                update_best(a_opt2, sc_opt2, f"restart{i}_adam")

    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 5: Coordinate descent polishing (1200s-1550s) ============
    phase5_end = min(deadline, t_start + 1550.0)
    print(f"Phase 5: Coord descent, deadline={phase5_end - t_start:.1f}s")

    if best_seq is not None and time.monotonic() < phase5_end - 30:
        n_cd = len(best_seq)
        # For large n, do partial coordinate descent (random subset per sweep)
        n_sweeps = 3 if n_cd <= 2048 else 2
        a_cd, sc_cd = _coord_descent(best_seq, phase5_end - 10, n_sweeps=n_sweeps)
        update_best(a_cd, sc_cd, "coord_desc")

    print(f"Phase 5 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 6: Fine L-BFGS-B polish (1550s-1700s) ============
    phase6_end = min(deadline, t_start + 1700.0)
    print(f"Phase 6: Fine polish, deadline={phase6_end - t_start:.1f}s")

    if best_seq is not None and time.monotonic() < phase6_end - 10:
        c_b = _full_convolve(best_seq, best_seq)
        max_cb = float(c_b.max())
        for beta in [0.0003 * max_cb, 0.0001 * max_cb]:
            if time.monotonic() > phase6_end - 5:
                break
            a_opt, sc_opt, _ = _lbfgsb_phase(best_seq, phase6_end - 5, beta, max_iter=2000)
            update_best(a_opt, sc_opt, f"fine_lbfgs_b{beta:.6f}")

    print(f"Phase 6 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 7: Try different n (1700s-1760s) ============
    phase7_end = min(deadline, t_start + 1760.0)
    print(f"Phase 7: Try other n, deadline={phase7_end - t_start:.1f}s")

    if best_seq is not None and time.monotonic() < phase7_end - 20:
        for target_n in [4096, 1024]:
            if time.monotonic() > phase7_end - 10:
                break
            n0 = len(best_seq)
            if n0 == target_n:
                continue
            x0 = np.linspace(0, 1, n0)
            x1 = np.linspace(0, 1, target_n)
            p_interp = np.interp(x1, x0, best_seq)
            p_interp = np.maximum(p_interp, 1e-12)
            p_interp = p_interp / p_interp.sum()
            sc_interp = _score(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            # Quick optimize
            if time.monotonic() < phase7_end - 15 and sc_interp < best_score + 0.05:
                c_int = _full_convolve(p_interp, p_interp)
                max_ci = float(c_int.max())
                a_opt, sc_opt, _ = _lbfgsb_phase(p_interp, phase7_end - 10, 0.005 * max_ci, max_iter=1000)
                update_best(a_opt, sc_opt, f"opt_n{target_n}")

    print(f"Phase 7 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 8: Ultra-fine final polish (1760s - deadline) ============
    print(f"Phase 8: Ultra-fine polish, deadline={deadline - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 5:
        c_f = _full_convolve(best_seq, best_seq)
        max_cf = float(c_f.max())
        a_opt, sc_opt = _adam_phase(best_seq, deadline - 3, 0.0001 * max_cf, lr=0.002, max_iter=50000)
        update_best(a_opt, sc_opt, "ultra_adam")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
