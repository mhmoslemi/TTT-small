        # Actually when i = k-i (i.e., k=2i), the term is a[i]*a[i] = a[i]^2, and d/d(a[i]) = 2a[i].
        # But in the sum, this term only appears once (j=i=k-i). So dC[k]/da[i] = 2a[i] when k=2i.
        # Otherwise (k != 2i): dC[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i]... no that's wrong too.
        # Let me just enumerate for small case.
        # a = [a0, a1, a2, a3], k=2: C[2] = a0*a2 + a1*a1 + a2*a0
        # dC[2]/da[0] = a2 + a2 = 2*a2. Hmm, a[k-0] = a[2], and the other contribution is a[0-2+...] 
        # Actually: C[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0]
        # dC[2]/da[0]: terms where a[0] appears: a[0]*a[2] (derivative = a[2]) and a[2]*a[0] (derivative = a[2])
        # So dC[2]/da[0] = a[2] + a[2] = 2*a[2] = 2*a[k-i] = 2*a[2]. ✓
        # dC[2]/da[1]: terms where a[1] appears: a[1]*a[1] (derivative = 2*a[1])
        # 2*a[k-i] = 2*a[1]. ✓
        # dC[2]/da[2]: = 2*a[0] = 2*a[k-i] = 2*a[0]. ✓
        # dC[2]/da[3]: a[3] doesn't appear. 2*a[k-i] = 2*a[-1] = 0. ✓
        # So the formula is: dC[k]/da[i] = 2*a[k-i] if 0<=k-i<n, else 0.
        # Wait, that's only true when both j=i and j=k-i are in [0,n-1], which means 0<=k-i<n AND 0<=i<n (always true).
        # But if k-i < 0 or k-i >= n, then a[k-i] = 0, so the formula still works!
        # EXCEPT: we need to check that the "other" term is also valid.
        # When j=i: we need 0 <= k-i < n (for a[k-i] to be valid).
        # When j=k-i: we need 0 <= k-i < n (same condition!) AND 0 <= k-i < n (for j to be valid, which is 0 <= k-i < n).
        # So both contributions require 0 <= k-i < n, and each gives a[k-i].
        # Total: 2*a[k-i] if 0 <= k-i < n, else 0.
        if 0 <= j1 < n:
            val = 2.0 * a[j1]
        grad[i] = val
    return grad


def _subgradient_topk(a, topk=5):
    """Compute weighted subgradient using top-K active lags."""
    n = len(a)
    c = _full_convolve(a)
    c_sorted_idx = np.argsort(c)[::-1]
    top_idx = c_sorted_idx[:topk]
    top_vals = c[top_idx]
    # Normalize weights
    w = top_vals / top_vals.sum()
    # Compute weighted subgradient
    grad = np.zeros(n)
    for idx, wt in zip(top_idx, w):
        k = int(idx)
        g = _cross_correlate_shifted(a, k)
        grad += wt * g
    return grad, c


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _nesterov_topk(a, deadline, topk=10, lr_init=0.01, momentum_init=1.0,
                   patience=200, restart_factor=0.5):
    """Nesterov-accelerated top-K subgradient with simplex projection."""
    n = len(a)
    a = _project_simplex(a.copy())
    a_prev = a.copy()
    momentum = momentum_init
    lr = lr_init
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # Nesterov extrapolation
        y = a + momentum * (a - a_prev)
        y = _project_simplex(y)

        # Compute top-K subgradient at y
        grad, c_y = _subgraduate_topk(y, topk)

        # Update
        a_new = y - lr * grad
        a_new = _project_simplex(a_new)

        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Update momentum (Nesterov)
        if it % 50 == 0:
            momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
            momentum = min(momentum, 20.0)  # Cap to avoid divergence
            if no_imp > patience:
                # Restart from best
                a = best_a.copy()
                a_prev = a.copy()
                momentum = momentum_init
                lr = lr_init * restart_factor
                no_imp = 0
            elif no_imp > patience // 2:
                lr = max(0.001, lr * 0.95)

        a_prev = a
        a = a_new

    return best_a, best_score


def _adam_smooth_max(a, deadline, rng, lr=0.03, beta_init=None,
                     beta_decay=0.97, no_imp_limit=300):
    """Adam optimizer with smooth-max objective in log-space."""
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        # Smooth max weights
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        # Gradient: cross-correlation of w with a
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        g = np.fft.irfft(fw * fa_rev, N)[n - 1:2 * n - 1]

        # Center the gradient (constraint: sum(a)=1 handled by log-space centering)
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-4, beta * 0.5)

        if no_imp > no_imp_limit:
            ns = 0.06 + 0.04 * (no_imp % 5)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_fixed_point(a, deadline, max_iters=30):
    """LP fixed-point iteration with line search."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())

        A = _build_conv_matrix(n, a)
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        b = np.maximum(res.x, 0.0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        # Line search
        best_t = 0.5
        best_sc = _score(a)
        for t_candidate in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = (1.0 - t_candidate) * a + t_candidate * b
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = t_candidate

        a = (1.0 - best_t) * a + best_t * b
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Check convergence
        if it > 0:
            diff = np.abs(a - best_a).max()
            if diff < 1e-10:
                break

    return best_a, best_score


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'exp':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-p1 * t / n)
    elif ptype == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + p1
    elif ptype == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / p3) ** 2) + np.exp(-0.5 * ((t - p2) / p3) ** 2)
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n, dtype=np.float64) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'poly3':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t + p2 * t**2 + p3 * t**3
    elif ptype == 'poly4':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t + p2 * t**2 + p3 * t**3 + 0.5 * t**4
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _poly_optimize(n, coeffs_init, deadline, method='Nelder-Mead'):
    """Optimize polynomial coefficients to minimize score."""
    from scipy.optimize import minimize

    def objective(coeffs):
        t = np.linspace(0, 1, n)
        a = np.zeros(n)
        for i, c in enumerate(coeffs):
            a += c * t**i
        if np.any(a < 0):
            return 1e10
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        return _score(a)

    try:
        res = minimize(objective, coeffs_init, method='Nelder-Mead',
                       options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10,
                                'adaptive': True})
        if time.monotonic() > deadline:
            return None
        if res.fun < 1e9:
            t = np.linspace(0, 1, n)
            a = np.zeros(n)
            for i, c in enumerate(res.x):
                a += c * t**i
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            return a, res.fun
    except Exception:
        pass
    return None


def _subgraduate_topk(a, topk=5):
    """Compute weighted subgradient using top-K active lags."""
    n = len(a)
    c = _full_convolve(a)
    k = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    w = top_vals / top_vals.sum()

    grad = np.zeros(n)
    for idx, wt in zip(top_idx, w):
        kk = int(idx)
        # dC[kk]/da[i] = 2*a[kk-i] if 0<=kk-i<n else 0
        valid = (kk - np.arange(n))
        mask = (valid >= 0) & (valid < n)
        g = np.zeros(n)
        g[mask] = 2.0 * a[valid[mask]]
        grad += wt * g

    return grad, c

    print("Phase 6: Multi-scale escalation...")
    phase6_end = t_start + 1650.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 768:
            target_ns = [1024, 1536, 2048]
        elif n_cur <= 1024:
            target_ns = [1536, 2048]
        elif n_cur <= 1536:
            target_ns = [2048, 3072]
        else:
            target_ns = [3072, 4096] if n_cur < 2048 else []

        for tn in target_ns:
            if time.monotonic() > phase6_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score(p_int)
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Refine at new scale
            if time.monotonic() < phase6_end - 60:
                per = min(60.0, phase6_end - time.monotonic() - 30)
                if per > 20:
                    a_e, sc_e = _adam_smooth_max(p_int, time.monotonic() + per, rng,
                                                 lr=0.02, beta_decay=0.97, no_imp_limit=200)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

                    # Nesterov at new scale
                    if time.monotonic() < phase6_end - 30:
                        per2 = min(40.0, phase6_end - time.monotonic() - 20)
                        if per2 > 15:
                            a_ne, sc_ne = _nesterov_topk(a_e, time.monotonic() + per2,
                                                          topk=10, lr_init=0.015)
                            if sc_ne < best_score:
                                best_seq = a_ne.copy()
                                best_score = sc_ne
                                print(f"  BEST: {best_score:.10f} n={tn} [Nest_esc]")

    # === Phase 7: Random restarts ===
    print("Phase 7: Random restarts...")
    phase7_end = deadline - 60.0

    if best_seq is not None and time.monotonic() < phase7_end:
        n_best = best_seq.size
        n_restarts = 6
        per_r = max(15.0, (phase7_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase7_end:
                break
            per = min(per_r, phase7_end - time.monotonic() - 5)
            if per < 10:
                break

            noise_scale = 0.04 + 0.03 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            # Quick Nesterov
            a_r, sc_r = _nesterov_topk(a_s, time.monotonic() + per * 0.5,
                                        topk=8, lr_init=0.02)
            if time.monotonic() < phase7_end - 10 and per * 0.3 > 5:
                # Then Adam
                a_r2, sc_r2 = _adam_smooth_max(a_r, time.monotonic() + per * 0.3, rng,
                                               lr=0.03, beta_decay=0.96, no_imp_limit=100)
                if sc_r2 < sc_r:
                    a_r, sc_r = a_r2, sc_r2

            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i}]")

    # === Phase 8: Final polish ===
    print("Phase 8: Final polish...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 15:
            n_final = best_seq.size
            if n_final <= 2048:
                a_fp, sc_fp = _lp_fixed_point(best_seq, time.monotonic() + per, max_iters=20)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={n_final} [final_LP]")
            # Always do a quick Adam polish
            if time.monotonic() < deadline - 5:
                per2 = deadline - time.monotonic() - 5
                if per2 > 10:
                    a_ap, sc_ap = _adam_smooth_max(best_seq, time.monotonic() + per2, rng,
                                                   lr=0.01, beta_decay=0.98, no_imp_limit=80)
                    if sc_ap < best_score:
                        best_seq = a_ap.copy()
                        best_score = sc_ap
                        print(f"  BEST: {best_score:.10f} n={n_final} [final_Adam]")

    # Final validation
    print(f"\nFinal best: {best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(1024, 'linear', 2.0, 0, 0)

        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(1024, 'linear', 2.0, 0, 0)
    print("FALLBACK n=1024")
    return [float(x) for x in fb]
