import numpy as np
from scipy.optimize import minimize
import time


def _npow2(n):
    return 1 << max(1, (n - 1).bit_length())


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _npow2(out_n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:out_n]


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max())


def _smooth_obj_grad(z, beta):
    a = _softmax(z)
    n = len(a)
    c = _fft_conv(a, a)
    cm = float(c.max())
    c_shift = (c - cm) / beta
    e = np.exp(c_shift)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    obj = 2.0 * n * (cm + beta * np.log(s))
    W = float(w @ c)
    w_rev = w[::-1]
    aw = _fft_conv(a, w_rev)
    cross = aw[2 * n - 2 : n - 2 : -1]
    H = a * cross
    grad = 4.0 * n * (H - W * a)
    if not np.all(np.isfinite(grad)):
        return float(obj), np.zeros(n)
    return float(obj), grad


def _lbfgs_run(z0, deadlines, beta_scales):
    n = len(z0)
    best_z = z0.copy()
    a0 = _softmax(z0)
    c0 = _fft_conv(a0, a0)
    best_score = _score_norm(a0)
    max_c0 = float(c0.max())
    for dl, bs in zip(deadlines, beta_scales):
        if time.monotonic() >= dl:
            break
        beta = max(1e-12, bs * max_c0)
        try:
            res = minimize(
                lambda z: _smooth_obj_grad(z, beta)[0],
                best_z, method='L-BFGS-B',
                jac=lambda z: _smooth_obj_grad(z, beta)[1],
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 40}
            )
            if np.all(np.isfinite(res.x)):

                sc = _score_norm(a_new)

                    best_score = sc
                    best_z = res.x.copy()
                    max_c0 = float(_fft_conv(a_new, a_new).max())
        except Exception:
            pass
    return best_z, best_score

        if not np.all(np.isfinite(grad)):
            continue
        momentum = mom_c * momentum - lr * grad
        z = z + momentum
        z -= z.mean()
        z = np.clip(z, -60, 60)
        if it % 200 == 0:
            a_c = _softmax(z)
            sc = _score_norm(a_c)
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
            else:
                lr *= 0.9
    return best_z, best_score


def _poly_score(coeffs, n):
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    for d, c_ in enumerate(coeffs[1:], 1):
        a += c_ * t ** d
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return _score_norm(a), a


def _poly_optimize(n, deadline, deg=5):
    def poly_obj(x):
        a = np.ones(n)
        for d in range(1, deg + 1):
            t = np.linspace(0, 1, n)
            a += x[d - 1] * t ** d
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        return _score_norm(a)
    best_x = np.zeros(deg)
    best_sc = _score_norm(np.ones(n) / n)
    for trial in range(6):
        if time.monotonic() >= deadline:
            break
        if trial == 0:
            x0 = np.zeros(deg)
        else:
            x0 = np.array([0.0, 0.3 * (trial - 1), 0.0, -0.2 * (trial - 1), 0.1 * (trial - 1)])[:deg]
        try:
            res = minimize(poly_obj, x0, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.fun < best_sc:
                best_sc = res.fun
                best_x = res.x.copy()
        except Exception:
            pass
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    for d in range(1, deg + 1):
        a += best_x[d - 1] * t ** d
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return _score_norm(a), a


def _make_linear(n, slope):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quad(n, c1, c2):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    a = 1.0 + c1 * u + c2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()
