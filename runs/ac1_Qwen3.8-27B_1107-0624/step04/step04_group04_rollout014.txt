import numpy as np
from scipy.optimize import linprog, minimize
import time
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    # Assumes sum(a) = sqrt(2n), so score = max(conv(a,a))
    n = len(a)
    c = _full_convolve(a, a)
    return float(c.max())

def _normalize(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a * (np.sqrt(2.0 * n) / s)

def _solve_lp(f, M, n):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    # Constraint matrix: A[k,j] = f[j] if 0 <= k-j < n, else 0
    # This is a (2n-1) x n matrix
    
    # Build the constraint matrix efficiently
    # A[k, j] = f[k-j] if 0 <= k-j < n
    # For each k, the valid j range is max(0, k-n+1) to min(n-1, k)
    
    # Use scipy sparse matrix for efficiency
    from scipy import sparse
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    b_rhs = M * np.ones(2*n - 1)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_rhs, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    return res.x

"""
Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based alternation, active-set Adam, and L-BFGS continuation.
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy import sparse
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _norm_to_sqrt2n(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return a.copy()
    return a * (np.sqrt(2.0 * n) / s)


def _build_conv_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[j] if 0 <= k-j < n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[js].tolist())
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp_for_g(f, M, n, time_limit=30.0):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    t0 = time.monotonic()
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_alternation(f_init, deadline, max_iters=100, step_start=0.3, step_decay=0.95):
    """LP-based alternating optimization."""
    n = len(f_init)
    f = _norm_to_sqrt2n(f_init.copy())
    best_f = f.copy()
    best_score = _score_fn(best_f)
    
    step = step_start
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        c = _full_convolve(f, f)
        M = float(c.max())
        
        g = _solve_lp_for_g(f, M, n, time_limit=20.0)
        if g is None:
            step *= 0.7
            continue
        
        sum_g = float(g.sum())
        if sum_g < 1e-10:
            break
        
        # Rescale g to same normalization
        g = _norm_to_sqrt2n(g)
        
        # Update
        f_new = (1 - step) * f + step * g
        f_new = np.maximum(f_new, 0)
        
        sc = _score_fn(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            print(f"    LP iter {it}: score={sc:.10f}, step={step:.3f}, M={M:.6f}")
        else:
            # Try smaller step

                best_score = sc2
                best_f = f_new2.copy()
                f = f_new2
                print(f"    LP iter {it} (half step): score={sc2:.10f}")
        
        step *= step_decay
    
    return best_f, best_score

            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                a_new = _norm_to_sqrt2n(a_new)
                sc = _score_fn(a_new)

                    best_score = sc
                    best_f = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_f, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
