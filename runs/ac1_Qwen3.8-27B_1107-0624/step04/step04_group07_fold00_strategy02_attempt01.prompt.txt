def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

def _next_pow2(n):
    k = 1
    while k < n:

    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]

def _smooth_max_weights(c, beta):

def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _coord_descent(a, deadline, max_sweeps=50):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.5, 0.75, 1.25, 2.0]:
                a[j] = max(old_val * factor, 1e-15)
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
                    break
            a[j] = old_val
        if not improved:
            break

    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, profile_type, param1=1.0, param2=0.0, param3=0.0):
    if profile_type == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (t - 0.5)
    elif profile_type == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (2 * t - 1) ** 2
    elif profile_type == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + param1 * u + param2 * u ** 3
    elif profile_type == 'exp':
        t = np.arange(n)
        a = np.exp(-param1 * t / n)
    elif profile_type == 'cosine':
        t = np.arange(n)
        a = 1.0 + param1 * np.cos(2 * np.pi * param2 * t / n)
    elif profile_type == 'sine':
        t = np.arange(n)
        a = 1.0 + param1 * np.sin(2 * np.pi * param2 * t / n)
    elif profile_type == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + param1
    elif profile_type == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - param1) / param3) ** 2) + np.exp(-0.5 * ((t - param2) / param3) ** 2)
    elif profile_type == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / param1) ** 2)
    elif profile_type == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + param1
    elif profile_type == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** param1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

Approach and Rationale
We propose a method based on optimizing the sequence in the Fourier domain because the self-convolution in the time domain is computationally expensive to compute directly for large n. The key observation is that the maximum of the self-convolution corresponds to the maximum of the modulus squared of the Fourier transform. By optimizing in the Fourier domain, we can separate the optimization of frequency components and the time domain representation, potentially leading to faster convergence and lower objective values.

The objective function is: 2 * n * max(|FFT(a)|²) / (sum(a))², where a is the sequence. The strategy involves:
1. Generating initial candidates in the Fourier domain.
2. Projecting the Fourier domain candidates back to the time domain.
3. Iteratively refining the candidates with a combination of gradient-based optimization and constraint handling.

This method avoids the expensive time-domain self-convolution computation by operating primarily in the Fourier domain, where the modulus squared is a smooth function and can be optimized more efficiently.

Mathematical Specification
Given a sequence a of non-negative coefficients, the self-convolution is computed via FFT. Let F = FFT(a, N) where N is the smallest power of two >= 2*n-1. Then the self-convolution points are the real parts of the inverse FFT of |F|², but note that |F|² is not circular convolution and must be truncated to the time domain length.

The objective is to minimize:
   score = (2.0 * n) * max(|FFT(a)|²).real / ((sum(a))²)

But since we are optimizing in the Fourier domain, we use the complex FFT magnitude as a vector to be optimized. The constraints are:
   - All elements in the time domain must be non-negative.
   - The sequence must have a minimum sum of at least 0.01 after normalization.

We will use:
   • A sequence representation in the Fourier domain (complex frequencies).
   • A smooth optimization: to avoid the non-differentiability of the max(|FFT(a)|²), we will use a logarithmic barrier approach.

The score is evaluated by projecting back to the time domain and computing the maximum of the self-convolution.
