def _proj_simplex(v):

    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    max_c = c.max()

def _grad_soft_max(p, c, beta):

    n = p.size
    x = (c - c.max()) / beta
    ex = np.exp(np.minimum(x, 0.0))

    h = np.fft.irfft(fw * fp, N)

"""
Optimized step function search combining aggressive LP best-response iteration,
adaptive smooth-max Adam in logit space, hard-max subgradient steps, and
multi-restart with interpolated escalation. Targets score < 1.5027 by
allocating more time to LP refinement and using adaptive step-size strategies
that respond to convergence signals.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(p):
    n = p.size
    c = _full_convolve(p)
    return 2.0 * n * float(c.max())


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _grad_soft_max(p, c, beta):
    n = p.size
    cm = float(c.max())
    if beta < 1e-13:
        mask = c >= cm - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros_like(c)
            w[mask] = 1.0 / cnt
    else:
        x = (c - cm) / beta
        ex = np.exp(np.minimum(x, 0.0))
        ss = float(ex.sum())
        if not np.isfinite(ss) or ss <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / ss
    return 2.0 * _cross_corr(w, p), w


def _proj_simplex(v):
    n = v.size
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_conv_matrix(a):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_best_response(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(p0, deadline, max_solves=300):
    n = p0.size
    a = np.maximum(p0, 0.0)
    s = a.sum()
    if s < 1e-10:
        return p0, _score_norm(p0)
    a = a / s
    c = _full_convolve(a)
    M = float(c.max())
    score = 2.0 * n * M
    best = a.copy()
    best_score = score

    t = 0.25
    stall = 0
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_best_response(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        # Adaptive line search
        current_score = 2.0 * n * float(_full_convolve(a).max())
        best_sc = current_score
        best_t = 0.0
        for tc in [0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.5, 0.6, 0.7]:
            a_try = (1.0 - tc) * a + tc * b
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try = a_try / st
            sc_try = 2.0 * n * float(_full_convolve(a_try).max())
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_t = tc

        if best_t == 0.0:
            stall += 1
            t = max(0.05, t * 0.8)
            if stall > 15:
                break
        else:
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            c = _full_convolve(a)
            M = float(c.max())
            sc = 2.0 * n * M
            if sc < best_score - 1e-13:
                best_score = sc
                best = a.copy()
                stall = 0
                t = min(0.5, best_t * 1.1)
            else:
                stall += 1
                t = max(0.05, best_t * 0.8)
                if stall > 15:
                    break

    return best, best_score


def _adam_logit(p0, deadline, rng, lr=0.04, beta_init=None, beta_decay=0.95,
                no_imp_limit=300, use_hard_switch=False):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    if beta_init is None:
        beta = max(1e-4, 0.06 * maxc)
    else:
        beta = beta_init

    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        grad, w = _grad_soft_max(p, c, beta)
        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.003, lr * 0.92)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        # Adaptive beta: decrease when peaks are close to flat
        if t_adam % 20 == 0:
            top_k = min(20, c.size)
            sorted_c = np.sort(c)[::-1][:top_k]
            gap = sorted_c[0] - sorted_c[-1]
            if gap < beta * 0.1:
                beta = max(5e-5, beta * beta_decay)
            else:
                beta = max(5e-5, beta * 0.98)

        if no_imp > no_imp_limit:
            ns = 0.04 if best_score < 1.51 else 0.08
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, 0.06 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _subgradient_project(p0, deadline, lr_init=0.01, max_iters=5000):
    """Hard max subgradient with projection onto simplex."""
    p = np.maximum(p0, 1e-12)
    p = p / p.sum()
    n = p.size
    c = _full_convolve(p)
    best_p = p.copy()
    best_score = 2.0 * n * float(c.max())
    lr = lr_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        cm = float(c.max())
        score = 2.0 * n * cm
        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.9
                stall = 0
                if lr < 1e-6:
                    break

        # Subgradient of max
        tol = max(1e-10, 1e-6 * cm)
        mask = c >= cm - tol
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros_like(c)
            w[mask] = 1.0 / cnt
        grad = 2.0 * _cross_corr(w, p)

        # Projected gradient step
        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)
        p_new = p - lr * gz
        p_new = _proj_simplex(p_new)
        p = p_new
        c = _full_convolve(p)

    return best_p, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return np.maximum(p0, 0.0) / max(p0.sum(), 1e-10)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, np.maximum(p0, 0.0))
    s = p1.sum()
    if s < 1e-10:
        return None
    return p1 / s

                if per > 15:
                    p_e, sc_e = _adam_logit(p_int, time.monotonic() + per, rng,
                                             lr=0.03, beta_init=None, beta_decay=0.95, no_imp_limit=100)
                    _try_update(p_e, sc_e, f'escalate_n{tn}')

    # Phase 7: Final LP polish (1700s - 1785s)
    phase7_end = deadline - 5.0
    if time.monotonic() < phase7_end and best_seq is not None and best_seq.size <= 2048:
        remaining = phase7_end - time.monotonic()
        if remaining > 20:
            print(f"  Phase 7: Final LP, remaining={remaining:.0f}s")
            p_f, sc_f = _lp_refine(best_seq, time.monotonic() + remaining * 0.7, max_solves=50)
            _try_update(p_f, sc_f, 'final_LP')

        if time.monotonic() < phase7_end - 5:
            per = phase7_end - time.monotonic() - 5
            if per > 10:
                p_f2, sc_f2 = _adam_logit(best_seq, time.monotonic() + per, rng,
                                           lr=0.01, beta_init=2e-4, beta_decay=0.98, no_imp_limit=50)
                _try_update(p_f2, sc_f2, 'final_adam')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
            # Scale back to sum=1 for the normalized score
            exact_sc = _score_norm(final)
            # But evaluate_sequence computes 2*n*max_b/sum_a^2, so if sum=1, score = 2*n*max_b
            # We return the sequence as-is; evaluate_sequence will compute the score.
            # Actually, we should return the sequence such that evaluate_sequence gives the right score.
            # Since evaluate_sequence normalizes by sum^2, returning a/sum(a) gives the same score.
            print(f"FINAL: score_norm={exact_sc:.10f} n={final.size} sum={s:.6f}")
            try:
                eval_sc = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(eval_sc):
                    print(f"EVAL: {eval_sc:.10f}")
            except Exception as e:
                print(f"EVAL error: {e}")
            return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
