Actually, I think the key insight is:

Wait, but `|A(f)|^2` for a length-n sequence is a trigonometric polynomial of degree `n-1` with `2n-1` real coefficients. We want it to be constant (= `1/(2n-1) * N` for some normalization). A constant is a degree-0 polynomial, which is a subset of degree `n-1` polynomials. So there exist sequences `a` with `|A(f)|^2 = constant`.

But wait: `|A(f)|^2 = A(f) * conj(A(f))`. For a real sequence, `conj(A(f)) = A(-f) = A(2π-f)`. So `|A(f)|^2 = A(f) * A(-f)`.

So the non-negativity constraint is the binding one. We can't achieve a perfectly flat spectrum with a non-negative sequence (for non-trivial n).

"""
FISTA-based convex optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Uses β-continuation with softmax smoothing
of the max, accelerated gradient (FISTA) on the simplex, and multi-scale
search (different n values). Exploits convexity of the problem.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad(a, c, w):
    """
    Compute gradient of sum_k w_k * conv(a,a)[k] with respect to a.
    grad[i] = 2 * sum_k w_k * a[k-i] (where a[j]=0 for j out of [0,n-1])
    This equals 2 * conv(w, reversed(a)) shifted appropriately.
    """
    n = len(a)
    a_rev = a[::-1]
    # conv(w, a_rev) has length len(w) + n - 1 = (2n-1) + n - 1 = 3n-2
    # We need the middle part: indices from n-1 to n-1 + (2n-2) = 3n-3
    wc = _full_convolve(w, a_rev)
    # The gradient component for a[i] is 2 * sum_k w[k] * a[k-i]
    # = 2 * (w * a_rev)[i + (n-1)] ... let me think more carefully
    # 
    # g_k(a) = sum_i a[i]*a[k-i] (with 0 <= i < n, 0 <= k-i < n)
    # dg_k/da[i] = a[k-i] (if 0 <= k-i < n)
    # 
    # sum_k w[k] * dg_k/da[i] = sum_k w[k] * a[k-i] * 1[0<=k-i<n]
    # Let j = k-i, so k = i+j:
    # = sum_j w[i+j] * a[j] * 1[0<=j<n] * 1[0<=i+j<2n-1]
    # = (w_shifted * a)[i] where w_shifted is w adjusted for bounds
    #
    # This is equivalent to: (w * a)[i] in the "cross-correlation" sense
    # which is conv(w, reverse(a))[i + n - 1]
    
    # Actually let me verify: conv(w, a_rev)[m] = sum_j w[j] * a_rev[m-j] = sum_j w[j] * a[n-1-(m-j)]
    # = sum_j w[j] * a[n-1-m+j]
    # Let i = n-1-m+j, j = m-n+1+i:
    # = sum_i w[m-n+1+i] * a[i]
    # 
    # We want sum_j w[i+j] * a[j]. If m-n+1+i = i+j, then j = m-n+1. That's not right.
    
    # Let me use a different approach. Define h[i] = sum_k w[k]*a[k-i] for valid k.
    # This is the cross-correlation of w and a: (w ⊛ a)[i] = sum_k w[k]*a[k-i]
    # = conv(w, a_reversed)[i + len(a) - 1]
    # where a_reversed = a[::-1]
    
    # conv(w, a_rev)[m] = sum_j w[j] * a_rev[m-j]
    # a_rev[p] = a[n-1-p]
    # = sum_j w[j] * a[n-1-(m-j)]
    # Let p = n-1-(m-j), j = m-n+1+p
    # = sum_p w[m-n+1+p] * a[p]
    
    # We want h[i] = sum_k w[k] * a[k-i] (with k-i in [0,n-1])
    # Let p = k-i, k = i+p:
    # h[i] = sum_p w[i+p] * a[p] (with i+p in [0, 2n-2], p in [0, n-1])
    
    # From above: conv(w, a_rev)[m] = sum_p w[m-n+1+p] * a[p]
    # Setting m-n+1 = i, i.e., m = i + n - 1:
    # conv(w, a_rev)[i + n - 1] = sum_p w[i + p] * a[p] = h[i] ✓
    
    h = wc[n - 1: 2 * n - 1]  # indices n-1 to 2n-2, length n
    return 2.0 * h

def _fista_optimize(a_init, deadline, beta_schedule, max_iter_per_phase=50000):
    """
    FISTA on the simplex with β-continuation.
    Minimizes 2n * beta * logsumexp(conv(a,a)/beta) over a in simplex.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    a_prev = a.copy()
    t_kk = 1.0
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    L_base = 4.0 * n  # Lipschitz constant of the gradient of sum_k w_k * g_k(a)
                      # Each g_k is a quadratic form with eigenvalues at most 2, 
                      # and the max eigenvalue of Q_k is 2 (for the zero-lag)
                      # Actually, for the smoothed problem, L = 2n/beta * max_eigenvalue
                      # max eigenvalue of Q_k is at most 2 (since Q_k is a permutation-like matrix)
                      # So L = 4n/beta
    
    global_iter = 0
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        L = 4.0 * n / beta  # Lipschitz constant of the smoothed gradient
        step = 1.0 / L
        
        it = 0
        while it < max_iter_per_phase:
            if time.monotonic() > deadline:
                break
            it += 1
            global_iter += 1
            
            # Compute convolution and weights
            c = _full_convolve(a, a)
            c_max = c.max()
            w = np.exp((c - c_max) / beta)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = w / w_sum
            
            # Compute gradient
            grad = _compute_grad(a, c, w)
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_kk * t_kk)) / 2.0
            y = a_prev + ((t_kk - 1.0) / t_new) * (a_prev - a)
            
            a_new = _proj_simplex(y - step * grad)
            
            # Check convergence
            if np.linalg.norm(a_new - a) < 1e-12:
                a = a_new
                break
            
            a_prev = a.copy()
            a = a_new
            t_kk = t_new
            
            # Track best
            if global_iter % 100 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    # Final check
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

"""
FISTA-based convex optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Exploits convexity of the problem using
accelerated gradient (FISTA) with softmax-smoothed max on the simplex,
β-continuation (homotopy), multi-scale n exploration, and final polishing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_smoothed(a_init, beta, deadline, max_iter, L_override=None):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    if L_override is not None:
        L = L_override
    else:
        L = 4.0 * n / beta

    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    cm = float(c_init.max())
    best_score = 2.0 * n * cm
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score

def _adam_fine(a_init, deadline, beta_schedule, lr=0.008, max_iter=200000):

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):

    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * a * (2.0 * h / (2.0 * n) - W / (2.0 * n)) * 2.0

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
