"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative a.
Uses multi-phase optimization: diverse non-symmetric initializations,
exact-gradient projected gradient descent with line search, LP best-response
refinement, and multi-restart with adaptive perturbation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, issparse
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _project_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho_val = rho[-1]
    theta = (cssv[rho_val] - s) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    a = a / s
    c = _convolve_fft(a)
    return float(2.0 * len(a) * c.max())


def _peak_gradient(a, k, n):
    g = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    for i in range(i_min, i_max + 1):
        g[i] = a[k - i]
    return g


def _top_peak_gradient(a, top_m=3):
    n = len(a)
    c = _convolve_fft(a)
    max_val = float(c.max())
    idx = np.argsort(c)[::-1][:top_m]
    weights = c[idx]
    w_sum = float(weights.sum())
    if w_sum <= 0:
        return np.zeros(n), max_val
    weights = weights / w_sum
    g = np.zeros(n)
    for j in range(len(idx)):
        k = int(idx[j])
        g += weights[j] * _peak_gradient(a, k, n)
    return g, max_val

def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_step(a, M=None, max_iters=5):
    n = len(a)
    if M is None:
        c = _convolve_fft(a)
        M = float(c.max())
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = np.full(2 * n - 1, M * 1.000001)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds,
                      method='highs', options={'time_limit': 30})
    except Exception:
        return a, _score(a)
    if not res.success:
        return a, _score(a)
    b = np.maximum(res.x, 0.0)
    s_b = float(b.sum())
    if s_b < 1e-10:
        return a, _score(a)
    b = b / s_b
    best_a = a.copy()
    best_sc = _score(a)
    for t in [0.3, 0.5, 0.7, 0.2, 0.4, 0.6]:
        a_try = (1 - t) * a + t * b
        a_try = np.maximum(a_try, 0)
        a_try = _project_simplex(a_try, 1.0)
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc


def _lp_refine_loop(a, deadline, max_solves=20, t_init=0.4):
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    c = _convolve_fft(a)
    M = float(c.max())
    t = t_init
    stall = 0

    for solve_i in range(max_solves):
        if time.monotonic() > deadline:
            break
        a_new, sc_new = _lp_step(a, M=M, max_iters=1)
        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_a = a_new.copy()
            stall = 0
            a = a_new
            c = _convolve_fft(a)
            M = float(c.max())
        else:
            stall += 1
            if stall > 5:
                break
        t = max(0.1, t * 0.85)

    return best_a, best_score


def _adam_logit_refine(p0, deadline, rng, lr=0.03, beta_init=0.05,
                       beta_decay=0.95, no_imp_limit=300):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n = p0.size
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_init * maxc)
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        mval = maxc
        x = (c - mval) / beta
        ex = np.exp(np.minimum(x, 0.0))
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        g_raw = h[n - 1:2 * n - 1]

        dot = float(np.dot(p, g_raw))
        gz = p * (g_raw - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        p = e / e.sum()
        c = _convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.003, lr * 0.93)

        if t_adam % 20 == 0:
            beta = max(5e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.03 if best_score < 1.51 else 0.08
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            e = np.exp(z)
            p = e / e.sum()
            c = _convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_init * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _make_power_ramp(n, alpha):
    t = np.arange(n, dtype=np.float64) + 1.0
    a = t ** alpha
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_exp_ramp(n, lam):
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(lam * t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_double_exp(n, lam1, lam2):
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(lam1 * t) + 0.3 * np.exp(-lam2 * (t - 0.5))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_log_ramp(n):
    t = np.arange(n, dtype=np.float64) + 1.0
    a = np.log(t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_sqrt_ramp(n):
    return _make_power_ramp(n, 0.5)


def _make_inv_ramp(n, eps=0.01):
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 / (t + eps)
    a = np.maximum(a, 0)
    return a / a.sum()


def _make_bimodal(n, c1, c2, w):
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_score = np.inf

    def update(a, label=""):
        nonlocal best_a, best_score
        if a is None or a.size == 0:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        a = np.maximum(a, 0)
        s = float(a.sum())
        if s < 0.01:
            return
        a = a / s
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  BEST: {best_score:.10f} n={a.size} [{label}] t={elapsed:.0f}s")

    print(f"Started. Budget={budget_s}s")

    # Phase 1: Generate and score diverse initializations (0-90s)
    phase1_end = t_start + 90.0
    inits = []
    n_main = 1024

    # Power ramps
    for alpha in [0.2, 0.4, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0, 5.0]:
        inits.append((_make_power_ramp(n_main, alpha), f"pow_{alpha}"))
    # Exp ramps
    for lam in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0, 12.0]:
        inits.append((_make_exp_ramp(n_main, lam), f"exp_{lam}"))
    # Double exp
    for l1 in [1.0, 2.0, 4.0]:
        for l2 in [2.0, 4.0, 8.0]:
            inits.append((_make_double_exp(n_main, l1, l2), f"dexp_{l1}_{l2}"))
    # Log, sqrt, inverse
    inits.append((_make_log_ramp(n_main), "log"))
    inits.append((_make_sqrt_ramp(n_main), "sqrt"))
    inits.append((_make_inv_ramp(n_main), "inv"))
    # Bimodal
    for c1, c2, w in [(0.2, 0.8, 0.15), (0.1, 0.7, 0.1), (0.3, 0.9, 0.2),
                       (0.15, 0.6, 0.12), (0.25, 0.75, 0.08)]:
        inits.append((_make_bimodal(n_main, c1, c2, w), f"bi_{c1}_{c2}"))

    # Try parent
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000)
            ps = float(parent_raw.sum())
            if ps > 0.01:
                p_norm = parent_raw / ps
                inits.append((p_norm, "parent"))
                if p_norm.size != n_main:
                    x0 = np.linspace(0, 1, p_norm.size)
                    x1 = np.linspace(0, 1, n_main)
                    p_int = np.interp(x1, x0, p_norm)
                    p_int = np.maximum(p_int, 0)
                    ps2 = p_int.sum()
                    if ps2 > 0.01:
                        inits.append((p_int / ps2, "parent_int"))
    except Exception:
        pass

    # Random non-symmetric inits
    for _ in range(5):
        a = rng.random(n_main) ** 2
        a = a / a.sum()
        inits.append((a, f"rnd_{_}"))

    print(f"Phase 1: {len(inits)} initializations")
    scored_inits = []
    for a, label in inits:
        sc = _score(a)
        scored_inits.append((sc, a, label))
        update(a, label)

    scored_inits.sort(key=lambda x: x[0])
    for i, (sc, a, lab) in enumerate(scored_inits[:5]):
        print(f"  Top {i+1}: {sc:.10f} ({lab})")

    # Phase 2: Gradient descent on top candidates (90s-450s)
    phase2_end = t_start + 450.0
    top_k = min(6, len(scored_inits))
    per_init = max(30.0, (phase2_end - time.monotonic()) / top_k - 5)

    polished = []
    for i in range(top_k):
        if time.monotonic() > phase2_end - 30:
            break
        sc0, a0, lab = scored_inits[i]
        per = min(per_init, phase2_end - time.monotonic() - 10)
        if per < 15:
            break
        a_g, sc_g = _grad_descent_phase(a0, time.monotonic() + per,
                                         base_lr=0.05, top_m=3,
                                         lp_every=50, label=lab)
        update(a_g, f"grad_{lab}")
        polished.append((sc_g, a_g, lab))

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best: {best_score:.10f}")

    # Phase 3: LP refinement on best (450s-900s)
    phase3_end = t_start + 900.0
    if best_a is not None and time.monotonic() < phase3_end:
        per_lp = min(120.0, phase3_end - time.monotonic() - 10)
        if per_lp > 20:
            a_lp, sc_lp = _lp_refine_loop(best_a, time.monotonic() + per_lp,
                                            max_solves=30, t_init=0.4)
            update(a_lp, "LP_phase3")

        # Adam refinement
        if time.monotonic() < phase3_end - 20:
            per_adam = min(60.0, phase3_end - time.monotonic() - 10)
            if per_adam > 20:
                a_ad, sc_ad = _adam_logit_refine(best_a, time.monotonic() + per_adam,
                                                  rng, lr=0.03, beta_init=0.05,
                                                  beta_decay=0.95, no_imp_limit=200)
                update(a_ad, "adam_phase3")

        # Gradient after LP
        if time.monotonic() < phase3_end - 10 and best_a is not None:
            per_g2 = min(60.0, phase3_end - time.monotonic() - 5)
            if per_g2 > 15:
                a_g2, sc_g2 = _grad_descent_phase(best_a, time.monotonic() + per_g2,
                                                   base_lr=0.02, top_m=3,
                                                   lp_every=30, label="post_lp")
                update(a_g2, "grad_post_lp")

    # Phase 4: Multi-restart with perturbation (900s-1500s)
    phase4_end = t_start + 1500.0
    if best_a is not None and time.monotonic() < phase4_end:
        n_restarts = 8
        per_r = max(20.0, (phase4_end - time.monotonic()) / (n_restarts + 1) - 5)

        for ri in range(n_restarts):
            if time.monotonic() > phase4_end - 30:
                break
            # Perturb
            noise_scale = 0.02 + 0.02 * ri
            z = np.log(np.maximum(best_a, 1e-12))
            z -= z.mean()
            z = z + noise_scale * rng.normal(size=best_a.size)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            e = np.exp(z)
            p_pert = e / e.sum()

            # Gradient optimize
            per_g = min(per_r * 0.6, phase4_end - time.monotonic() - 30)
            if per_g < 10:
                break
            a_r, sc_r = _grad_descent_phase(p_pert, time.monotonic() + per_g,
                                             base_lr=0.04, top_m=3,
                                             lp_every=40, label=f"r{ri}")
            update(a_r, f"restart{ri}")

            # LP polish
            if time.monotonic() < phase4_end - 20 and best_a is not None:
                per_lp_r = min(20.0, phase4_end - time.monotonic() - 10)
                if per_lp_r > 10:
                    a_lr, sc_lr = _lp_refine_loop(best_a, time.monotonic() + per_lp_r,
                                                    max_solves=10, t_init=0.3)
                    update(a_lr, f"LP_r{ri}")

    # Phase 5: Try different lengths (1500s-1700s)
    phase5_end = t_start + 1700.0
    if best_a is not None and time.monotonic() < phase5_end:
        n_cur = best_a.size
        for n_try in [2048, 512]:
            if n_try == n_cur or time.monotonic() > phase5_end - 40:
                continue
            x0 = np.linspace(0, 1, n_cur)
            x1 = np.linspace(0, 1, n_try)
            a_int = np.interp(x1, x0, best_a)
            a_int = np.maximum(a_int, 0)
            s_int = a_int.sum()
            if s_int > 0.01:
                a_int = a_int / s_int
                update(a_int, f"interp_n{n_try}")
                per_g3 = min(40.0, phase5_end - time.monotonic() - 20)
                if per_g3 > 15:
                    a_ng, sc_ng = _grad_descent_phase(a_int, time.monotonic() + per_g3,
                                                       base_lr=0.04, top_m=3,
                                                       lp_every=30, label=f"n{n_try}")
                    update(a_ng, f"grad_n{n_try}")

    # Phase 6: Final intensive polish (1700s-1788s)
    if best_a is not None and time.monotonic() < deadline - 3:
        # LP final
        per = min(40.0, deadline - time.monotonic() - 15)
        if per > 15:
            a_f, sc_f = _lp_refine_loop(best_a, time.monotonic() + per,
                                          max_solves=15, t_init=0.2)
            update(a_f, "final_LP")

        # Gradient final
        if time.monotonic() < deadline - 5 and best_a is not None:
            per2 = min(20.0, deadline - time.monotonic() - 3)
            if per2 > 8:
                a_fg, sc_fg = _grad_descent_phase(best_a, time.monotonic() + per2,
                                                   base_lr=0.01, top_m=5,
                                                   lp_every=20, label="final")
                update(a_fg, "final_grad")

    # Final validation and return
    if best_a is not None:
        final = np.clip(best_a, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_power_ramp(1024, 1.0)
        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    # Fallback
    fb = _make_power_ramp(1024, 1.0)
    print(f"FALLBACK n=1024 score={_score(fb):.10f}")
    return [float(x) for x in fb]
