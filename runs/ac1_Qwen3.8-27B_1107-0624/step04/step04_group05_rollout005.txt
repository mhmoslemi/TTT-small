from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c

def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)_k = (convolve(a,b))_k"""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _lp_best_response(a, M):
    """Solve LP: max sum(b) s.t. convolve(a,b) <= M, b >= 0"""
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))

    fp = np.fft.rfft(p[::-1], N)

"""
Optimized step function search combining LP best-response iteration, Adam in logit space,
structured harmonic restarts, multi-length exploration, and adaptive polishing.
Targets score < 1.5027 by aggressive peak flattening via alternating LP/Adam.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)

def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_best_response(a, M):
    n = len(a)
    if n > 4096:
        return None
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

        b_norm = b / s_b

        # Line search over step sizes

        best_tt = t

    phase1_end = t_start + 40.0
    candidates = []

    # Generate diverse initial candidates at n=512 and n=1024
    for n in [512, 1024]:
        profs = [
            _make_profile_linear(n, 1.0),
            _make_profile_linear(n, 1.5),
            _make_profile_linear(n, 2.0),
            _make_profile_linear(n, 2.5),
            _make_profile_exponential(n, 0.3),
            _make_profile_exponential(n, 0.7),
            _make_profile_exponential(n, 1.2),
            _make_profile_two_bump(n, 0.3, 0.7, 0.12),
            _make_profile_two_bump(n, 0.25, 0.75, 0.2),
            _make_profile_cubic(n, 1.0, -0.5),
            _make_profile_cubic(n, 1.5, -0.8),
            _make_profile_cosine(n, 1, 0.6),
            _make_profile_cosine(n, 2, 0.4),
            _make_profile_cosine(n, 3, 0.3),
        ]
        for i, p in enumerate(profs):
            sc = _score_norm(p)
            candidates.append((sc, p, n, f'prof{i}_n{n}'))

    if parent is not None:
        sc = _score_norm(parent)
        candidates.append((sc, parent.copy(), pn, 'parent'))
        for n_try in [512, 1024, 2048]:
            if pn != n_try:
                pi = _interp_profile(parent, n_try)
                if pi is not None:
                    sc_i = _score_norm(pi)
                    candidates.append((sc_i, pi, n_try, f'parent_interp_n{n_try}'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # ============ Phase 2: Adam polish top candidates (40s-200s) ============
    phase2_end = t_start + 200.0
    for sc, p, n, label in candidates[:6]:
        if time.monotonic() > phase2_end - 20:
            break
        per = min(25.0, phase2_end - time.monotonic() - 15)
        if per < 10:
            break
        p_b, sc_b = _adam_logit(p, time.monotonic() + per, rng,
                                lr=0.05, beta_frac=0.08,
                                beta_decay=0.92, no_imp_limit=300)
        _update(p_b, sc_b, f'adam_{label}')

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # ============ Phase 3: Intensive LP on best (200s-600s) ============
    phase3_end = t_start + 600.0
    if best_seq is not None and best_seq.size <= 2048:
        n_lp = best_seq.size
        print(f"  LP refine n={n_lp}, score={best_score:.10f}")

        # Long LP run
        per = min(250.0, phase3_end - time.monotonic() - 30)
        if per > 30:
            p_lp, sc_lp = _lp_refine(best_seq, time.monotonic() + per,
                                      t_init=0.3, max_solves=100, multi_M=True)
            _update(p_lp, sc_lp, 'LP_main')

        # Alternating LP+Adam
        if time.monotonic() < phase3_end - 30:
            per2 = min(120.0, phase3_end - time.monotonic() - 20)
            if per2 > 30:
                p_alt, sc_alt = _alternating_lp_adam(best_seq, time.monotonic() + per2,
                                                      rng, max_rounds=5)
                _update(p_alt, sc_alt, 'alternating')

    # ============ Phase 4: Multi-length exploration (600s-900s) ============
    phase4_end = t_start + 900.0
    if time.monotonic() < phase4_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024]
        elif n_cur <= 1024:
            target_ns = [512, 2048]
        else:
            target_ns = [1024]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Explore n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            # Adam on interpolated
            per = min(40.0, phase4_end - time.monotonic() - 25)
            if per > 15:
                p_e, sc_e = _adam_logit(p_int, time.monotonic() + per, rng,
                                         lr=0.04, beta_frac=0.07,
                                         beta_decay=0.93, no_imp_limit=250)
                _update(p_e, sc_e, f'adam_n{tn}')

                # Quick LP if improved
                if sc_e < best_score + 0.01 and tn <= 2048:
                    per_lp = min(30.0, phase4_end - time.monotonic() - 15)
                    if per_lp > 10:
                        p_lpe, sc_lpe = _lp_refine(p_e, time.monotonic() + per_lp,
                                                    t_init=0.2, max_solves=15,
                                                    multi_M=True)
                        _update(p_lpe, sc_lpe, f'LP_n{tn}')

    # ============ Phase 5: Multi-restart with structured perturbations (900s-1500s) ============
    phase5_end = t_start + 1500.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        z_best = np.log(np.maximum(best_seq, 1e-12))
        z_best -= z_best.mean()

        n_restarts = 6
        per_r = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts + 1))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 20:
                break
            per_adam = min(per_r * 0.55, phase5_end - time.monotonic() - 30)
            if per_adam < 10:
                break

            # Structured perturbation: combination of noise and harmonics
            ns = 0.02 + 0.02 * i
            z = z_best + ns * rng.normal(size=n_best)
            # Add 1-2 harmonics
            for _ in range(rng.integers(1, 3)):
                freq = rng.integers(1, max(2, n_best // 6))
                amp = ns * rng.uniform(0.3, 0.8)
                phase = rng.uniform(0, 2 * np.pi)
                t_arr = np.arange(n_best)
                z = z + amp * np.cos(2.0 * np.pi * freq * t_arr / n_best + phase)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p_s = _softmax_from_z(z)

            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per_adam, rng,
                                     lr=0.035, beta_frac=0.06,
                                     beta_decay=0.94, no_imp_limit=200)
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

            # LP polish on restart result
            if best_seq.size <= 2048 and time.monotonic() < phase5_end - 25:
                per_lp = min(25.0, phase5_end - time.monotonic() - 15)
                if per_lp > 10:
                    p_lp3, sc_lp3 = _lp_refine(best_seq, time.monotonic() + per_lp,
                                                t_init=0.15, max_solves=12,
                                                multi_M=True)
                    _update(p_lp3, sc_lp3, f'LP_restart{i}')

    # ============ Phase 6: Final intensive polish (1500s to deadline) ============
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_final = best_seq.size
        print(f"  Final polish phase. n={n_final}, score={best_score:.10f}")

        # Long LP
        if n_final <= 2048:
            per = min(120.0, deadline - time.monotonic() - 30)
            if per > 20:
                p_f, sc_f = _lp_refine(best_seq, time.monotonic() + per,
                                        t_init=0.1, max_solves=50, multi_M=True)
                _update(p_f, sc_f, 'final_LP')

        # Final Adam with very small LR
        if time.monotonic() < deadline - 10:
            per2 = deadline - time.monotonic() - 8
            if per2 > 10:
                p_f2, sc_f2 = _adam_logit(best_seq, time.monotonic() + per2, rng,
                                           lr=0.01, beta_frac=0.03,
                                           beta_decay=0.97, no_imp_limit=100,
                                           noise_scale=0.01)
                _update(p_f2, sc_f2, 'final_adam')

        # One more LP if time
        if n_final <= 2048 and time.monotonic() < deadline - 5:
            per3 = deadline - time.monotonic() - 3
            if per3 > 8:
                p_f3, sc_f3 = _lp_refine(best_seq, time.monotonic() + per3,
                                          t_init=0.05, max_solves=8, multi_M=False)
                _update(p_f3, sc_f3, 'final_LP2')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL internal: score={exact_sc:.10f} n={final.size}")

        # Verify with evaluate_sequence
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"FINAL evaluate: {eval_sc:.10f}")
            if np.isfinite(eval_sc) and eval_sc < best_score:
                best_score = eval_sc
        except Exception as e:
            print(f"  eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
