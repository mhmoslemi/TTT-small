        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.95)

        if t % 50 == 0:
            beta = max(1e-6, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-6, beta * 0.5)

        if no_imp > patience:
            sz = np.log(np.maximum(best_p, 1e-15))
            sz -= float(np.mean(sz))
            z = sz + restart_noise * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            exp_z = np.exp(z - z.max())
            p = exp_z / exp_z.sum()
            c = _full_convolve(p)
            beta = max(1e-4, beta_frac * float(c.max()))
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_imp = 0
            lr = base_lr
            restart_noise *= 0.85

    return best_p, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_polish(a, deadline, t_init=0.3, max_solves=30, m_shrink=0.998):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    t_step = t_init
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            b = _lp_solve(a, M * 1.01)
            if b is None:
                break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b
        a_new = (1.0 - t_step) * a + t_step * b
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        score_new = _score(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M_cur = float(c.max())
        target_M = best_score / (2.0 * n)
        M = max(target_M * 1.002, M_cur * m_shrink)
        t_step = max(0.1, t_step * 0.93)

    return best_a, best_score


def _peak_flood(a, deadline, eta=0.04, n_iter_max=3000):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        M = float(c.max())
        score = float(2.0 * n * M)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        threshold = M * 0.92
        hot_mask = c >= threshold
        if not hot_mask.any():
            hot_mask[int(np.argmax(c))] = True

        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        if it % 200 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.005, eta * 0.85)
            else:
                eta = min(0.15, eta * 1.03)

    return best_a, best_score
