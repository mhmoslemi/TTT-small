"""
Optimization combining: (1) subgradient descent with exact max and momentum,
(2) aggressive LP fixed-point iteration, (3) multi-scale exploration at n=1024

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def _subgradient_step(a, c, n_peaks=3):
    """Compute subgradient of max_k (a*a)_k at current a, using top n_peaks."""
    n = len(a)
    # Find top peaks
    sorted_idx = np.argsort(c)[::-1]
    top_k = sorted_idx[:n_peaks]
    top_vals = c[top_k]
    
    # Weights: softmax of peak values (higher peaks get more weight)
    # Use a soft normalization
    max_v = top_vals[0]
    w = (top_vals / max_v) ** 4  # Emphasize the maximum
    w = w / w.sum()
    
    # Compute subgradient: g[j] = sum_l w_l * 2 * a[k_l - j] (if valid)
    g = np.zeros(n)
    for l, k in enumerate(top_k):
        if w[l] < 1e-10:
            continue
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            partner = k - np.arange(j_start, j_end + 1)
            g[j_start:j_end + 1] += w[l] * 2.0 * a[partner]
    
    return g


def _project_simplex(v):
    """Project vector v onto the probability simplex (non-negative, sum=1)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _subgradient_descent(a0, deadline, n_peaks=3, lr_init=0.01, momentum=0.7,
                         n_restarts=5, noise=0.01):
    """Subgradient descent with momentum and restarts on the simplex."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = _normalize(a)
    if a is None:
        return a0, _score_norm(a0)
    
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    
    c = _full_convolve(a)
    v_mom = np.zeros(n)
    lr = lr_init
    no_imp = 0
    rng = np.random.default_rng(42)
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        g = _subgradient_step(a, c, n_peaks)
        
        # Momentum
        v_mom = momentum * v_mom + (1 - momentum) * g
        
        # Step
        a_new = a - lr * v_mom
        a_new = _project_simplex(a_new)
        
        # Check if valid

            a_new = a.copy()

            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            lr = min(lr * 1.02, 0.05)  # Slightly increase

            a_new = a.copy()  # Revert
            no_imp += 1
            if no_imp % 50 == 0:
                lr = max(0.001, lr * 0.9)
        
        a = a_new
        c = c_new

            a = _project_simplex(a)
            v_mom = np.zeros(n)
            lr = lr_init
            no_imp = 0
            c = _full_convolve(a)
    
    return best_a, best_score


def _make_conv_matrix(a):
    """Build the sparse convolution matrix A where (A @ b)[k] = (a*b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(a, M):
    """Solve the LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    A = _make_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _build_conv_matrix(a):
    n = len(a)
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    pos = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        idx = np.arange(j_start, j_end + 1)
        rows[pos:pos + length] = k
        cols[pos:pos + length] = idx
        vals[pos:pos + length] = a[k - idx]
        pos += length
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

                if sc_half < best_score - 1e-14:
                    best_score = sc_half
                    best_a = a_half.copy()
                    a = a_half
                    c = _full_convolve(a)
                    M = float(c.max()) * 0.9998
                    improved = True

        if not improved:
            # Use dual for subgradient step if available
            if use_dual and dual is not None and float(dual.sum()) > 1e-10:
                g = _subgrad_dual(a, dual)
                if g is not None:
                    v_mom = 0.7 * v_mom + 0.3 * g
                    a_try = _project_simplex(a - 0.005 * v_mom)
                    c_try = _full_convolve(a_try)
                    sc_try = float(2.0 * n * c_try.max())
                    if sc_try < best_score - 1e-14:
                        best_score = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        c = c_try
                        M = float(c.max())
                    else:
                        M = float(c.max())
            else:
                M = float(c.max())

        t = max(0.1, t * 0.985)

    return best_a, best_score


def _alternating(a0, deadline, n_cycles=20):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_norm(a)

    for cycle in range(n_cycles):
        if time.monotonic() > deadline:
            break
        # Subgradient phase
        sg_dead = min(time.monotonic() + 15, deadline - 30)
        a, sc = _subgrad_line_search(a, sg_dead, n_peaks=3, lr_init=0.004,
                                     momentum=0.75, noise=0.004)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        # LP phase
        if a.size <= 2048:
            lp_dead = min(time.monotonic() + 8, deadline - 10)
            a, sc = _lp_fixed_point(a, lp_dead, t_init=0.25, max_solves=3, use_dual=True)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    return _normalize(p1)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = np.inf

    def _update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc):
            sc = _score_norm(p)
            if sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} t={time.monotonic()-t_start:.0f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={parent.size}, score={_score_norm(parent):.10f}")
        a = parent.copy()
    else:
        a = np.ones(1024) / 1024
        print("No parent, using uniform n=1024")

    _update(a, _score_norm(a))

    # Phase 1: Subgradient line search (100s)
    print(f"\nPhase 1: Subgradient (best={best_score:.10f})")
    p1_end = t_start + 100.0
    a, sc = _subgrad_line_search(a, p1_end, n_peaks=3, lr_init=0.01,
                                 momentum=0.7, noise=0.01)
    _update(a, sc)

    # Phase 2: LP fixed point (400s) - main workhorse
    print(f"\nPhase 2: LP fixed point (best={best_score:.10f})")
    p2_end = t_start + 500.0
    a, sc = _lp_fixed_point(best_seq, p2_end, t_init=0.35, max_solves=200, use_dual=True)
    _update(a, sc)

    # Phase 3: Alternating subgradient + LP (250s)
    print(f"\nPhase 3: Alternating (best={best_score:.10f})")
    p3_end = t_start + 750.0
    a, sc = _alternating(best_seq, p3_end, n_cycles=30)
    _update(a, sc)

    # Phase 4: More LP with tighter M (250s)
    print(f"\nPhase 4: Tight LP (best={best_score:.10f})")
    p4_end = t_start + 1000.0
    a, sc = _lp_fixed_point(best_seq, p4_end, t_init=0.2, max_solves=100, use_dual=True)
    _update(a, sc)

    # Phase 5: Multi-scale to n=2048 (350s)
    print(f"\nPhase 5: Multi-scale n=2048 (best={best_score:.10f})")
    p5_end = t_start + 1350.0
    if time.monotonic() < p5_end and best_seq is not None and best_seq.size <= 1024:
        n_new = 2048
        a_int = _interp_profile(best_seq, n_new)
        if a_int is not None:
            sc_int = _score_norm(a_int)
            print(f"  Interpolated n={n_new}: {sc_int:.10f}")
            _update(a_int, sc_int)
            # Subgradient at new scale
            sg_end = min(p5_end - 100, time.monotonic() + 150)
            if time.monotonic() < sg_end:
                a, sc = _subgrad_line_search(a_int, sg_end, n_peaks=5, lr_init=0.004,
                                             momentum=0.7, noise=0.004)
                _update(a, sc)
            # LP at new scale
            if time.monotonic() < p5_end - 10:
                a, sc = _lp_fixed_point(best_seq, p5_end, t_init=0.2,
                                        max_solves=20, use_dual=True)
                _update(a, sc)

    # Phase 6: Final intensive polish (remaining)
    print(f"\nPhase 6: Final polish (best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0})")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        # Subgradient
        if remaining > 80:
            a, sc = _subgrad_line_search(best_seq, time.monotonic() + remaining * 0.3,
                                         n_peaks=3, lr_init=0.002, momentum=0.8, noise=0.002)
            _update(a, sc)
        # LP if feasible
        if time.monotonic() < deadline - 30 and best_seq.size <= 2048:
            lp_time = min(80.0, deadline - time.monotonic() - 20)
            if lp_time > 20:
                a, sc = _lp_fixed_point(best_seq, time.monotonic() + lp_time,
                                        t_init=0.15, max_solves=15, use_dual=True)
                _update(a, sc)
        # Final subgradient
        if time.monotonic() < deadline - 5:
            a, sc = _subgrad_line_search(best_seq, deadline - 5,
                                         n_peaks=3, lr_init=0.001, momentum=0.85, noise=0.001)
            _update(a, sc)

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = np.ones(1024) / 1024
        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fb = np.ones(1024) / 1024
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
