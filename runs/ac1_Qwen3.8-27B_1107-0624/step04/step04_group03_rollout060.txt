def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Compute 2*n*max(conv) for normalized (sum=1) sequence."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Compute full score = 2*n*max(conv)/sum^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights(c, beta):
    """Compute softmax weights on convolution values."""
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        # Find all positions within a tiny epsilon of max
        w[np.isclose(c, cm, atol=beta*0.1)] = 1.0
        w /= w.sum()
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    """Compute gradient of smooth objective w.r.t. z."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    """Smooth objective value."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta):
    return _gradient_smooth(z, beta)


def _make_profile_inv_sqrt(n):
    """a[j] = 1/sqrt(j+1), normalized to sum=1."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(j)
    return a / a.sum()


def _make_profile_inv_sqrt_sym(n):
    """a[j] = 1/sqrt((j+1)*(n-j)), normalized."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(j * (n + 1 - j))
    return a / a.sum()


def _make_profile_power(n, alpha):
    """a[j] = (j+1)^(-alpha), normalized."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** (-alpha)
    return a / a.sum()


def _make_profile_cos(n, power=2):
    """a[j] = cos^power(pi*j/(2*n)), normalized."""
    t = np.linspace(0, np.pi / 2, n)
    a = np.cos(t) ** power
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_profile_triangle(n):
    """Triangular profile peaking at center."""
    t = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_profile_uniform(n):
    return np.ones(n) / n


def _make_profile_vshape(n, sharpness=2.0):
    """V-shaped profile: high at edges, low in middle."""
    t = np.linspace(0, 1, n)
    a = np.abs(2.0 * t - 1.0) ** sharpness + 0.1
    return a / a.sum()


def _run_lbfgs(a_init, deadline, max_wall_time=300.0):

    z = np.log(np.maximum(a_init, 1e-15))

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Beta schedule: start relatively large, decrease geometrically

    betas = [b * max_c0 for b in betas]

    n_betas = len(betas)
    per_beta_time = max_wall_time / n_betas

    for i, beta in enumerate(betas):

            if result.x is not None and np.all(np.isfinite(result.x)):

                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass

    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_schedule, lr_init=0.02, mom_init=0.9,
                        lr_decay=0.85, mom_increment=0.001):
    """Nesterov-accelerated gradient descent with smooth max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = lr_init
    mom = mom_init
    t = 0

    for beta in beta_schedule:
        if time.monotonic() >= deadline:
            break

        # Inner loop for this beta
        iterations_at_beta = 0
        while time.monotonic() < deadline:
            t += 1
            iterations_at_beta += 1

            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)
                if not np.all(np.isfinite(grad_y)):
                    break

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -60, 60)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 100 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

            if t % 200 == 0:
                mom = min(0.999, mom + mom_increment)

            # Adaptive: if not improving, reduce learning rate
            if iterations_at_beta > 500:
                break

        # Update for next beta
        lr = max(0.001, lr * lr_decay)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_init=0.01, beta_decay=0.9,
                   min_beta=1e-7, eval_interval=100):
    """Adam optimizer with decreasing beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(min_beta, beta_init * max_c)

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            # Decay beta
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())
            beta = max(min_beta, beta_decay * beta)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _direct_subgradient(a_init, deadline, n_iter=100000, threshold=0.01):
    """Direct subgradient method on the actual max (no smoothing).
    Uses averaging over near-max positions."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.002
    momentum = np.zeros(n)
    mom_coeff = 0.95

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Average over positions within threshold of max
        tol = threshold * cm
        active = c >= (cm - tol)
        n_active = np.sum(active)
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        W = float(np.dot(w, c))

        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -60, 60)

        if it % 200 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                mom_coeff = min(0.99, mom_coeff + 0.005)
            elif it % 1000 == 0:
                lr = lr * 0.8

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

def _next_pow2(n):

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

def _softmax_fn(z):

                mom = min(0.999, mom + mom_increment)

        lr = max(0.001, lr * lr_decay)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac_init=0.01, beta_decay=0.88,
                   min_beta_frac=1e-7, eval_interval=80):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac_init * max_c)

    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())
            beta = max(1e-12, min_beta_frac * max_c, beta_decay * beta)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _direct_subgradient(a_init, deadline, threshold=0.01, lr_init=0.003, mom_coeff_init=0.93):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = lr_init
    momentum = np.zeros(n)
    mom_coeff = mom_coeff_init
    stall_count = 0
    last_best_t = 0

    for it in range(500000):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        tol = threshold * cm
        active = c >= (cm - tol)
        n_active = int(np.sum(active))
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        W = float(np.dot(w, c))

        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -60, 60)

        if it % 150 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                last_best_t = it
                mom_coeff = min(0.995, mom_coeff + 0.003)
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 10:
                    lr = lr * 0.7
                    stall_count = 0

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse init -> L-BFGS -> Nesterov -> Adam ->
    direct subgradient -> restarts -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # --- Try to get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score_fn(parent):.10f}")

    # --- Phase 0: Evaluate diverse initial profiles ---
    phase0_end = min(deadline, t_start + 25.0)
    n_values = [4096, 8192]
    all_candidates = []

    for n in n_values:
        profiles = []
        profiles.append((f"inv_sqrt_n{n}", _make_profile_inv_sqrt(n)))
        profiles.append((f"inv_sqrt_sym_n{n}", _make_profile_inv_sqrt_sym(n)))
        profiles.append((f"power04_n{n}", _make_profile_power(n, 0.4)))
        profiles.append((f"power06_n{n}", _make_profile_power(n, 0.6)))
        profiles.append((f"cos2_n{n}", _make_profile_cos(n, 2)))
        profiles.append((f"triangle_n{n}", _make_profile_triangle(n)))
        profiles.append((f"uniform_n{n}", _make_profile_uniform(n)))
        profiles.append((f"vshape2_n{n}", _make_profile_vshape(n, 2.0)))
        profiles.append((f"vshape15_n{n}", _make_profile_vshape(n, 1.5)))

        if parent is not None:
            profiles.append((f"parent_n{n}", _interp_profile(parent, n)))

        for label, p in profiles:
            if time.monotonic() > phase0_end:
                break
            sc = _score_norm_fn(p)
            all_candidates.append((sc, p, n, label))
            print(f"  Init: {label}: {sc:.10f}")

    all_candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 best: {all_candidates[0][3]} = {all_candidates[0][0]:.10f}")

    best_seq = all_candidates[0][1].copy()
    best_score = all_candidates[0][0]
    print(f"  [{time.monotonic() - t_start:.1f}s] Initial best: {best_score:.10f}")

    # --- Phase 1: L-BFGS on top candidates ---
    phase1_end = min(deadline, t_start + 260.0)
    tried_n = set()
    for sc, p, n, label in all_candidates[:6]:
        if time.monotonic() > phase1_end - 20:
            break
        if n in tried_n:
            continue
        tried_n.add(n)
        per_time = min(80.0, phase1_end - time.monotonic() - 5)
        if per_time < 15:
            break
        print(f"  Phase1: L-BFGS {label} (n={n}, {per_time:.0f}s)")
        p_b, sc_b = _run_lbfgs(p, min(phase1_end - 5, time.monotonic() + per_time),
                               max_wall_time=per_time)
        if np.isfinite(sc_b) and sc_b < best_score:
            best_score = sc_b
            best_seq = p_b.copy()
            print(f"    [{time.monotonic() - t_start:.1f}s] NEW: {sc_b:.10f} [{label}]")

    print(f"Phase 1 done: {best_score:.10f}")

    # --- Phase 2: Nesterov (main workhorse) ---
    phase2_end = min(deadline, t_start + 660.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        n_b = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mc_b = float(c_b.max())
        beta_sched = [f * mc_b for f in
                      [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]]
        per_time = min(380.0, phase2_end - time.monotonic() - 10)
        if per_time > 30:
            print(f"  Phase2: Nesterov (n={n_b}, {per_time:.0f}s)")
            p_b, sc_b = _nesterov_optimize(
                best_seq, min(phase2_end - 10, time.monotonic() + per_time),
                beta_sched, lr_init=0.02, mom_init=0.9, lr_decay=0.82)
            if np.isfinite(sc_b) and sc_b < best_score:
                best_score = sc_b
                best_seq = p_b.copy()
                print(f"    [{time.monotonic() - t_start:.1f}s] NEW: {sc_b:.10f} [nesterov]")

    print(f"Phase 2 done: {best_score:.10f}")

    # --- Phase 3: Adam polishing ---
    phase3_end = min(deadline, t_start + 960.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(280.0, phase3_end - time.monotonic() - 10)
        if per_time > 30:
            print(f"  Phase3: Adam ({per_time:.0f}s)")
            p_b, sc_b = _adam_optimize(
                best_seq, min(phase3_end - 10, time.monotonic() + per_time),
                lr=0.04, beta_frac_init=0.005, beta_decay=0.85, min_beta_frac=1e-8)
            if np.isfinite(sc_b) and sc_b < best_score:
                best_score = sc_b
                best_seq = p_b.copy()
                print(f"    [{time.monotonic() - t_start:.1f}s] NEW: {sc_b:.10f} [adam]")

    print(f"Phase 3 done: {best_score:.10f}")

    # --- Phase 4: Multiple restarts ---
    phase4_end = min(deadline, t_start + 1360.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_b = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mc_b = float(c_b.max())
        n_restarts = 5
        per_restart = min(75.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break

            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()

            if i % 3 == 0:
                noise = 0.04 + 0.015 * i
                z = z + noise * rng.normal(size=n_b)
            elif i % 3 == 1:
                freq = rng.integers(1, 6)
                amp = 0.08
                t_idx = np.linspace(0, 2 * np.pi * freq, n_b)
                z = z + amp * np.sin(t_idx + rng.uniform(0, 2 * np.pi))
            else:
                # Blend with a different profile
                alt = all_candidates[min(i + 2, len(all_candidates) - 1)][1]
                alt_interp = _interp_profile(alt, n_b)
                blend = 0.6 * best_seq + 0.4 * alt_interp
                z = np.log(np.maximum(blend, 1e-15))
                z = z - z.mean()

            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase4_end - 5, time.monotonic() + per_restart)
            short_beta = [f * mc_b for f in [0.004, 0.002, 0.001, 0.0005, 0.0002, 1e-4]]
            p_b, sc_b = _nesterov_optimize(
                p_start, per_dl, short_beta, lr_init=0.015, mom_init=0.85, lr_decay=0.8)
            if np.isfinite(sc_b) and sc_b < best_score:
                best_score = sc_b
                best_seq = p_b.copy()
                print(f"    [{time.monotonic() - t_start:.1f}s] NEW: {sc_b:.10f} [restart{i}]")

    print(f"Phase 4 done: {best_score:.10f}")

    # --- Phase 5: Direct subgradient ---
    phase5_end = min(deadline, t_start + 1610.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        per_time = min(220.0, phase5_end - time.monotonic() - 10)
        if per_time > 20:
            print(f"  Phase5: Direct subgradient ({per_time:.0f}s)")
            p_b, sc_b = _direct_subgradient(
                best_seq, min(phase5_end - 10, time.monotonic() + per_time),
                threshold=0.005, lr_init=0.002, mom_coeff_init=0.93)
            if np.isfinite(sc_b) and sc_b < best_score:
                best_score = sc_b
                best_seq = p_b.copy()
                print(f"    [{time.monotonic() - t_start:.1f}s] NEW: {sc_b:.10f} [subgrad]")

    print(f"Phase 5 done: {best_score:.10f}")

    # --- Phase 6: Try other length + final fine polish ---
    phase6_end = deadline
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        target_n = 8192 if n_curr <= 4096 else 4096

        if target_n != n_curr and time.monotonic() < phase6_end - 50:
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Phase6: interp to n={target_n}: {sc_interp:.10f}")
            if sc_interp < best_score * 1.005:
                per_time = min(60.0, phase6_end - time.monotonic() - 20)
                if per_time > 15:
                    c_i = _full_convolve(p_interp, p_interp)
                    mc_i = float(c_i.max())
                    bs = [f * mc_i for f in [0.005, 0.002, 0.001, 0.0005]]
                    p_b, sc_b = _nesterov_optimize(
                        p_interp, min(phase6_end - 20, time.monotonic() + per_time),
                        bs, lr_init=0.015, mom_init=0.9)
                    if np.isfinite(sc_b) and sc_b < best_score:
                        best_score = sc_b
                        best_seq = p_b.copy()
                        print(f"    [{time.monotonic() - t_start:.1f}s] NEW: {sc_b:.10f} [n{target_n}]")

        # Final tiny-beta polish
        if time.monotonic() < phase6_end - 15:
            per_time = min(50.0, phase6_end - time.monotonic() - 5)
            if per_time > 10:
                c_f = _full_convolve(best_seq, best_seq)
                mc_f = float(c_f.max())
                tiny_beta = [f * mc_f for f in [2e-6, 1e-6, 5e-7, 2e-7]]
                p_b, sc_b = _nesterov_optimize(
                    best_seq, min(phase6_end - 5, time.monotonic() + per_time),
                    tiny_beta, lr_init=0.004, mom_init=0.97, lr_decay=0.92)
                if np.isfinite(sc_b) and sc_b < best_score:
                    best_score = sc_b
                    best_seq = p_b.copy()
                    print(f"    [{time.monotonic() - t_start:.1f}s] NEW: {sc_b:.10f} [final]")

    print(f"Phase 6 done: {best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile_inv_sqrt(4096)

        sc_final = _score_fn(final_seq)
        print(f"FINAL internal: {sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"FINAL evaluator: {sc_eval:.10f}")
            else:
                print(f"Evaluator inf, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile_inv_sqrt(4096)
    print(f"FALLBACK inv_sqrt n=4096: {_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
