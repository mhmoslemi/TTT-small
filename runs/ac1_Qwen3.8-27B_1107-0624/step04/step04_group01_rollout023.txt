    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _perturb_fourier(a, rng, n_modes=3):
    """Add small random Fourier modes perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.01 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _try_lbfgs_schedule(a_start, deadline, min_time_remaining=10.0):
    """Run a full L-BFGS beta-continuation schedule from a_start."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.3, 0.15, 0.07, 0.035, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale max_iter with available time (rough: ~5ms per iteration for n=2048)
        est_iter_time = 0.005 * (n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (est_iter_time * 8))))

        a_new, success, niter = _lbfgs_optimize(current_a, beta, n, deadline, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-12:
            current_a = a_new
            current_sc = sc

    return current_a

def propose_candidate(seed=42, budget_s=1800):
    """
    L-BFGS based optimizer with log-sum-exp smoothing and beta-continuation
    for minimizing 2*n*max(conv(a,a))/sum(a)^2. Includes subgradient polishing
    and multi-restart diversification.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}, score={_score(a0):.10f}")
        update_best(a0, _score(a0), "init_uniform")

    # --- Phase 1: Main L-BFGS optimization from parent ---
    print(f"\nPhase 1: Main L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a_main = _try_lbfgs_schedule(a0, deadline, min_time_remaining=300.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "main_lbfgs")

    # --- Phase 2: Subgradient polishing ---
    print(f"\nPhase 2: Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        sg_deadline = min(deadline - 200.0, time.monotonic() + 120.0)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=3000, lr_init=0.03)
        update_best(a_sg, sc_sg, "subgrad_polish")

    # --- Phase 3: Fine L-BFGS with very small beta ---
    print(f"\nPhase 3: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        n = len(best_seq)
        fine_betas = max_cb * np.array([0.0003, 0.00015, 0.00008, 0.00004, 0.00002])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - 60 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(2000, max(50, int(remaining / 0.04)))
            a_new, success, niter = _lbfgs_optimize(best_seq, beta, n, deadline, max_iter)
            sc = _score(a_new)
            update_best(a_new, sc, f"fine_{i}")
            if sc < _score(best_seq):
                best_seq = a_new

    # --- Phase 4: Multi-restart with diverse perturbations ---
    print(f"\nPhase 4: Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        n_restarts = min(4, max(1, int(remaining / 150)))

        for i in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break

            # Different perturbation strategies
            if i == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=2)
            elif i == 2:
                p_start = _perturb(best_seq, 0.01, rng)
            else:
                p_start = _perturb_fourier(best_seq, rng, n_modes=4)

            sc_start = _score(p_start)
            print(f"  Restart {i}: start_score={sc_start:.10f}")

            a_r = _try_lbfgs_schedule(p_start, deadline, min_time_remaining=30.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            # Quick subgradient polish on the result
            if sc_r < best_score + 0.01 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _subgrad_polish(a_r, deadline - 20, n_max_steps=500, lr_init=0.02)
                update_best(a_r2, sc_r2, f"restart_{i}_sg")

    # --- Phase 5: Final L-BFGS + subgradient cycle ---
    print(f"\nPhase 5: Final cycle (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # One more fine L-BFGS
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_final = mb * 0.00005
        if time.monotonic() < deadline - 3:
            a_fin, _, _ = _lbfgs_optimize(best_seq, beta_final, n, deadline - 3, max_iter=1000)
            update_best(a_fin, _score(a_fin), "final_lbfgs")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2, n_max_steps=200, lr_init=0.01)
            update_best(a_fin2, sc_fin2, "final_sg")

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
