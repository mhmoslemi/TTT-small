    n = len(v)

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)

    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    beta_starts = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]
    
    per_level = max(1.0, (deadline - time.monotonic()) / len(betas))
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_end = min(deadline, time.monotonic() + per_level)
        
        def obj(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if top_k > 0 and top_k < len(c_in):
                idx = np.argsort(c_in)[-top_k:]
                c_top = c_in[idx]
                c_norm = (c_top - c_top.max()) / beta
                log_sum = float(np.log(np.sum(np.exp(c_norm))))
                return 2.0 * n * (c_top.max() + beta * log_sum)
            else:
                c_shifted = (c_in - cm) / beta
                e = np.exp(c_shifted)
                log_sum = float(np.log(np.sum(e)))
                return 2.0 * n * (cm + beta * log_sum)
        
        def grad(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if top_k > 0 and top_k < len(c_in):
                idx = np.argsort(c_in)[-top_k:]
                c_top = c_in[idx]
                c_norm = (c_top - c_top.max()) / beta
                e = np.exp(c_norm)
                s = e.sum()
                if s < 1e-300:
                    w = np.zeros(len(c_in))
                    w[idx[0]] = 1.0
                else:
                    w = np.zeros(len(c_in))
                    w[idx] = e / s
            else:
                c_shifted = (c_in - cm) / beta
                e = np.exp(c_shifted)
                s = float(e.sum())
                if s < 1e-300:
                    idx = int(np.argmax(c_in))
                    w = np.zeros(len(c_in))
                    w[idx] = 1.0
                else:
                    w = e / s
            
            W = float(np.dot(w, c_in))
            a_rev = a_in[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_in * (h - W)
            if not np.all(np.isfinite(g)):
                g = np.zeros(n)
            return g
        
        try:
            result = minimize(obj, z, method='L-BFGS-B', jac=grad,
                              options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 30})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _coordinate_descent(a_init, deadline, group_size=64, n_rounds=1000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        idx = rng.integers(0, n, size=min(group_size, n))
        idx = np.unique(idx)
        
        a_sub = a[idx].copy()
        
        def sub_obj(x):
            x = np.maximum(x, 0.0)
            xs = x.sum()
            if xs < 1e-15:
                return 1e10
            a_trial = a.copy()
            a_trial[idx] = x
            a_trial = _simplex_project(a_trial)
            return _score_norm(a_trial)
        
        try:
            res = minimize(sub_obj, a_sub, method='Nelder-Mead',
                           options={'maxiter': 50 * len(idx), 'xatol': 1e-12, 'fatol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_trial = a.copy()
                a_trial[idx] = np.maximum(res.x, 0.0)
                a_trial = _simplex_project(a_trial)
                sc = _score_norm(a_trial)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
        except Exception:
            pass
    
    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_poisson_profile(n, lam):
    t = np.arange(n) / n
    a = np.exp(-lam * t) * (1.0 + 0.3 * np.sin(2.0 * np.pi * t))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    global height_sequence_1
    
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Phase 0: Initialization ---
    print("=== Phase 0: Initialization ===")
    phase0_end = min(deadline, t_start + 30.0)
    
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm(parent)
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
                update_best(parent, sc, "parent")
    except Exception as e:
        print(f"  No parent: {e}")
    
    n_main = 4096
    candidates = []
    
    if parent is not None and len(parent) == n_main:
        candidates.append((parent.copy(), "parent"))
    
    # Generate diverse initializations
    for alpha in [0.0003, 0.0005, 0.001, 0.002, 0.003, 0.005]:
        p = _make_exp_profile(n_main, alpha)
        candidates.append((p, f"exp_{alpha}"))
    
    for lam in [1.0, 2.0, 3.0, 5.0]:
        p = _make_poisson_profile(n_main, lam)
        candidates.append((p, f"pois_{lam}"))
    
    if parent is not None and len(parent) != n_main:
        candidates.append((_interp_profile(parent, n_main), "parent_interp"))
    
    # Score all candidates
    scored_candidates = []
    for p, label in candidates:
        sc = _score_norm(p)
        scored_candidates.append((sc, p, label))
        update_best(p, sc, f"init_{label}")
    
    scored_candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: {len(scored_candidates)} candidates, best={best_score:.10f}")
    
    # --- Phase 1: LP-based direction finding on best candidate ---
    print("=== Phase 1: LP direction finding ===")
    phase1_end = min(deadline, t_start + 500.0)
    
    if scored_candidates:
        best_init = scored_candidates[0][1]
        p_best, sc_best = _lp_iterate(best_init, phase1_end, top_k_start=20, top_k_end=150,
                                      n_iters=300, blend_range=(0.05, 0.1, 0.2, 0.3, 0.5))
        update_best(p_best, sc_best, "lp_main")
    
    # Also try LP on second-best candidate
    if len(scored_candidates) > 1 and time.monotonic() < phase1_end - 30:
        second = scored_candidates[1][1]
        dl2 = min(phase1_end - 5, time.monotonic() + min(120.0, (phase1_end - time.monotonic()) * 0.5))
        p_b2, sc_b2 = _lp_iterate(second, dl2, top_k_start=20, top_k_end=100,
                                  n_iters=100, blend_range=(0.05, 0.1, 0.2, 0.3))
        update_best(p_b2, sc_b2, "lp_second")
    
    print(f"  Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: Second LP pass (finer) ---
    print("=== Phase 2: LP refinement ===")
    phase2_end = min(deadline, t_start + 800.0)
    
    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end - 5, time.monotonic() + remaining)
        p_best, sc_best = _lp_iterate(best_seq, dl, top_k_start=50, top_k_end=300,
                                      n_iters=200, blend_range=(0.02, 0.05, 0.1, 0.2))
