"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Strategy (different from prior art, more global-focused):
  * Exact FFT-based subgradient of max(conv(a,a)) ("peak pressure").
  * Adam in log-space with adaptive smooth-max for local polish.
  * Multiplicative-weights update (MWU) as a distinct global reshaper.
  * LP bisection that keeps the direction (a) fixed while solving a linear
     feasibility (max sum(b) s.t. conv(a,b) <= M) to get validated blends.
  * Targeted mass-swap hill climbing to escape plateaus.
  * Multi-start over diverse parametric inits + annealed random restarts.
  * Adaptive time budgeting so a valid best sequence is always returned.
"""
import time

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        return np.ones_like(e) / float(e.size)
    return p / s


def _log_from_p(p, floor=1e-12):
    return np.log(np.maximum(np.asarray(p, dtype=np.float64), floor))


def _pressure_from_weights(w, p):
    """Exact subgradient of max-pooled convolution w.r.t. p (peak pressure).

    pressure[j] = sum_k w[k] * (p[k-j] + p[j-k]), out-of-range treated as 0.
    Computed with two linear convolutions via a single padded FFT size.
    """
    w = np.asarray(w, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(3 * n + 2)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    fp_rev = np.fft.rfft(p[::-1], N)
    pw = np.fft.irfft(fw * fp, N)
    pwr = np.fft.irfft(fw * fp_rev, N)
    termB = pw[:n]            # B[j] = sum_i p[i] w[j-i]
    termA = pwr[n - 1:2 * n - 1]  # A[j] = sum_i p[i] w[i+j]
    pr = termA + termB
    return np.where(np.isfinite(pr), pr, 0.0)


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.06, beta_min=1e-5,
                beta_decay=0.94, restart_patience=180):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = _log_from_p(p0)
    z -= float(np.mean(z))
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc / max(n, 1))

    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max(c, beta)
        g = _pressure_from_weights(w, p)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        g = g / gn
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            z = _log_from_p(best_p)
            z -= float(np.mean(z))
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        if not np.all(np.isfinite(z)):
            z = _log_from_p(best_p)
            z -= float(np.mean(z))
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        if np.isfinite(score):
            if score < best_score - 1e-10:
                best_score = score
                best_p = p.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = _log_from_p(best_p)
            z -= float(np.mean(z))
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 20 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_imp > restart_patience:
            if best_score < 1.51:
                ns = 0.03
            elif best_score < 1.53:
                ns = 0.08
            else:
                ns = 0.15
            z = _log_from_p(best_p)
            z = z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc / max(n, 1))
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _mwu_update(p0, deadline, eta_init=0.1, max_iter=200000):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    rng = np.random.default_rng(7)

    best_p = p.copy()
    best_score = _score_norm(p)
    c = _full_convolve(p)
    eta = eta_init
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        M = float(c.max())
        score = float(2.0 * n * M)
        if score < best_score - 1e-10:
            best_score = score
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1

        tol = 0.01 * M
        peak_mask = c >= M - tol
        npk = int(peak_mask.sum())
        if npk == 0:
            peak_mask[int(np.argmax(c))] = True
            npk = 1
        w = peak_mask.astype(np.float64) / npk
        pressure = _pressure_from_weights(w, p)
        pressure = np.maximum(pressure, 0.0)
        pm = float(pressure.max())
        if pm < 1e-15:
            break
        pressure = pressure / pm

        p = p * np.exp(-eta * pressure)
        p = np.maximum(p, 1e-12)
        p = p / p.sum()
        c = _full_convolve(p)

        if it % 50 == 0 and it > 0:
            if no_imp > 60:
                eta = max(0.01, eta * 0.9)
            elif no_imp == 0:
                eta = min(0.3, eta * 1.05)
        if no_imp > 600:
            p = best_p * (1.0 + 0.05 * rng.normal(size=n))
            p = np.maximum(p, 1e-12)
            p = p / p.sum()
            c = _full_convolve(p)
            eta = eta_init
            no_imp = 0

    return best_p, best_score


def _lp_feasible_direction(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    res = linprog(-np.ones(n), A_ub=A, b_ub=M * np.ones(2 * n - 1),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_bisection(a0, deadline, max_solves=24):
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    c = _full_convolve(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    curM = float(c.max())
    M_hi = curM * 1.2
    M_lo = curM * 0.9
    t_blend = 0.3

    solves = 0
    while time.monotonic() < deadline and solves < max_solves:
        solves += 1
        M_mid = (M_lo + M_hi) / 2.0
        b = _lp_feasible_direction(a, M_mid)
        if b is None or float(b.sum()) < 1e-10:
            M_lo = M_mid
            continue
        b_norm = b / float(b.sum())
        a_new = (1.0 - t_blend) * a + t_blend * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc = _score_norm(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        curM = float(c.max())
        if M_mid >= curM * 1.001:
            M_lo = curM * 0.99
        else:
            M_hi = M_mid
        t_blend = max(0.1, t_blend * 0.96)
    return best_a, best_score


def _targeted_swap(a0, deadline, max_steps=40000):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    rng = np.random.default_rng(11)
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)

    for it in range(max_steps):
        if time.monotonic() > deadline:
            break
        pk = int(np.argmax(c))
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()

        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        iidx = np.arange(js, je + 1)
        prod = a[iidx] * a[pk - iidx]
        i_src = iidx[int(np.argmax(prod))]
        # choose a low-pressure target far from the peak partners
        partner_hi = max(0, pk)
        cand = rng.integers(0, n)
        tries = 0
        while abs(cand - i_src) < 3 and tries < 8:
            cand = rng.integers(0, n)
            tries += 1
        frac = 0.02
        amt = a[i_src] * frac
        if (a[i_src] - amt) <= 0:
            amt = a[i_src] * 0.5
        a[i_src] -= amt
        a[cand] += amt
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        c = _full_convolve(a)

    return best_a, best_score


def _init_constant(n):
    return np.ones(n) / n


def _init_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-9)
    return a / a.sum()


def _init_cosh(n, alpha):
    t = np.linspace(-1.0, 1.0, n)
    a = np.cosh(alpha * t)
    a = np.maximum(a, 1e-9)
    return a / a.sum()


def _init_power(n, p):
    t = np.linspace(1e-3, 1.0, n)
    a = np.power(t, p) + np.power(1.0 - t, p)
    a = np.maximum(a, 1e-9)
    return a / a.sum()


def _init_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-9)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    y = np.interp(x1, x0, p0)
    y = np.maximum(y, 1e-9)
    return y / y.sum()


def _find_parent():
    names = ["height_sequence_1", "height_sequence", "height_sequence_0",
             "best_sequence", "parent_sequence"]
    g = globals()
    for nm in names:
        if nm in g:
            try:
                arr = np.asarray(g[nm], dtype=np.float64).ravel()
                if arr.size > 0:
                    arr = np.clip(arr, 0.0, 1000.0)
                    if float(arr.sum()) > 0.01:
                        return arr / arr.sum()
            except Exception:
                pass
    return None

        pass

    # --- MWU global reshape (18%) ---
    phase3_end = t0 + (deadline - t0) * 0.66
    try:
        pm, _ = _mwu_update(best_seq, phase3_end, eta_init=0.12)
        _update(pm, "mwu")
        # re-polish after mwu
        pp2, _ = _adam_logit(best_seq, min(phase3_end, time.monotonic() + 30),
                             rng, lr=0.03, beta_frac=0.05, restart_patience=120)
        _update(pp2, "mwu_polish")
    except Exception:
        pass

    # --- LP bisection refinement (14%) ---
    phase4_end = t0 + (deadline - t0) * 0.80
    if best_seq.size <= 2048:
        try:
            pl, _ = _lp_bisection(best_seq, phase4_end, max_solves=20)
            _update(pl, "lp_bisect")
        except Exception:
            pass

    # --- Targeted swap hill climb (8%) ---
    phase5_end = t0 + (deadline - t0) * 0.88
    try:
        ps, _ = _targeted_swap(best_seq, phase5_end)
        _update(ps, "swap")
    except Exception:
        pass

    # --- Final Adam polish (remaining) ---
    try:
        pf, _ = _adam_logit(best_seq, deadline, rng,
                            lr=0.015, beta_frac=0.03, beta_decay=0.96,
                            restart_patience=80)
        _update(pf, "final_adam")
    except Exception:
        pass

    # --- Optional larger-n interpolation (only if cheap & time left) ---
    if best_seq is not None and best_seq.size <= 1024 and time.monotonic() < deadline - 40:
        try:
            big = _interp_profile(best_seq, 2048)
            _update(big, "interp_2048")
            pb, _ = _adam_logit(big, min(deadline, time.monotonic() + 25),
                                rng, lr=0.03, beta_frac=0.05, restart_patience=100)
            _update(pb, "adam_2048")
        except Exception:
            pass

    # --- Final validation & return ---
    if best_seq is None:
        best_seq = _init_linear(1024, 2.0)
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _init_linear(1024, 2.0)
        s = float(final.sum())
    final = final / s
    exact = _score_norm(final)
    print("FINAL score=%.10f  n=%d" % (exact, final.size))
    try:
        ev = float(evaluate_sequence([float(x) for x in final]))
        print("EVAL score=%.10f" % ev)
    except Exception:
        pass
    return [float(x) for x in final]
