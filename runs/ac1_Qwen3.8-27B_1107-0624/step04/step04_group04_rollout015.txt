Let me also think about a completely different approach: **coordinate descent**.

For coordinate descent, we fix all $a_i$ except $a_j$ and optimize over $a_j$. The objective is:
$f(a) = \max_k (a*a)_k$

When we change $a_j$ to $a_j + \delta$, the change in $c_k$ is:
$\Delta c_k = 2\delta \cdot a_{k-j} + \delta \cdot (\text{if } k = 2j: \delta)$ ... 

Actually: $c_k' = c_k + 2\delta \cdot a_{k-j} + \delta^2 \cdot [k = 2j]$ (where $[k=2j]$ is 1 if $k=2j$ and 0 otherwise, and $a_{k-j} = 0$ if out of range).

This is a quadratic in $\delta$ for each $k$. We want to choose $\delta$ to minimize $\max_k (c_k + 2\delta a_{k-j} + \delta^2 [k=2j])$.

This is a 1D minimization for each coordinate update. It can be solved exactly!

For each coordinate $j$:
- We have $c_k' = c_k + 2\delta a_{k-j}$ for $k \neq 2j$
- $c_{2j}' = c_{2j} + 2\delta a_j + \delta^2$

We want to minimize $\max_k c_k'$ over $\delta$ (with the constraint $a_j + \delta \geq 0$, i.e., $\delta \geq -a_j$).

This is a 1D piecewise-linear (or piecewise-quadratic) minimization that can be solved by finding the intersection points.

However, this is $O(n)$ per coordinate update (to evaluate the max), and $O(n)$ coordinates, so $O(n^2)$ per full sweep. For $n = 4096$, that's about $16 \times 10^6$ operations per sweep, which is fast.

Actually, the 1D problem can be solved more efficiently. The function $\phi(\delta) = \max_k (c_k + 2\delta a_{k-j} + \delta^2 [k=2j])$ is the max of linear (and one quadratic) functions of $\delta$. It's convex in $\delta$. We can find its minimum by:
1. Binary search on the subgradient (which is $2a_{k^*} + 2\delta[k^*=2j]$ where $k^* = \arg\max$)
2. Or, note that the minimum occurs where the two largest "active" constraints are equal.

This is doable but might be slow due to the overhead of finding the argmax each time.

"""
Subgradient + Spectral optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses soft subgradient with top-K peaks, momentum, Polyak step size,
spectral equalization, and L-BFGS polishing.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_size(n):
    return _next_pow2(2 * n - 1)


def _full_convolve(a, b, N=None):
    n, m = len(a), len(b)
    out_n = n + m - 1
    if N is None:
        N = _fft_size(max(n, m))
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score assuming sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return v / s


def _subgradient_step(a, N, K, beta):
    """One subgradient step: compute soft subgradient and return (g, f_val, top_idx)."""
    n = len(a)
    c = _full_convolve(a, a, N)
    f_val = float(c.max())
    
    # Find top-K peaks
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))
    top_vals = c[top_idx]
    
    # Softmax weights
    max_tv = top_vals.max()
    w = np.exp((top_vals - max_tv) / beta)
    w /= w.sum()
    
    # Compute soft subgradient via FFT
    m = np.arange(N // 2 + 1)
    phi = -2.0 * np.pi * np.outer(m, top_idx.astype(np.float64)) / N
    H = np.exp(1j * phi) @ w
    
    A = np.fft.rfft(a, N)
    G = H * np.conj(A)
    g_full = np.fft.irfft(G, N)
    g = 2.0 * g_full[:n]
    
    return g, f_val, top_idx, w

    best_score = _score_norm(a)
    
    v = np.zeros(n)  # momentum
    alpha = 0.9  # momentum coefficient

    avg_count = 1

        if g_norm_sq > 1e-30:
            eta = polar_k * max(f_val - f_lower, 0) / g_norm_sq
        else:
            eta = 1e-6
        eta = min(eta, 0.5)  # cap step size
        
        # Momentum update
        v = alpha * v + g
        a = a - eta * v
        a = _simplex_project(a)

        a_avg += (a - a_avg) / avg_count

def _spectral_equalize(a, deadline, n_iter=200):
    """
    Spectral equalization: iteratively equalize FFT magnitudes.
    """
    n = len(a)
    N = _fft_size(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        
        # Target magnitude: equal for all non-DC
        total_power = float(np.sum(np.abs(A)**2))  # = N * sum(a^2)
        dc_power = float(np.abs(A[0])**2)
        non_dc_power = total_power - dc_power
        target_mag = np.sqrt(non_dc_power / (N // 2))  # equal magnitude for N/2 non-DC bins
        
        # Set magnitudes, keep phases
        A_new = A.copy()
        for m in range(1, N // 2 + 1):
            if np.abs(A[m]) > 1e-15:
                A_new[m] = target_mag * A[m] / np.abs(A[m])
            else:
                A_new[m] = target_mag * np.exp(2j * np.pi * np.random.rand())
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = _simplex_project(a_new)
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Blend: move towards equalized version
        a = 0.5 * a + 0.5 * a_new
        a = _simplex_project(a)
    
    return best_a, best_score

"""
Subgradient-based PASTR minimization for non-negative sequences.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Soft subgradient with adaptive top-K peaks
2. Momentum + Polyak step size  
3. Spectral equalization restarts
4. L-BFGS final polish
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_n(n):
    """Return FFT size >= 2n-1 (power of 2)."""
    return _next_pow2(2 * n - 1)


def _convolve_fft(a, b, N):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]


def _score(a):
    """Compute score = 2*n*max(conv(a,a)) (assuming sum(a)=1)."""
    n = len(a)
    N = _fft_n(n)
    c = _convolve_fft(a, a, N)
    return 2.0 * n * float(c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return v / s


def _compute_soft_grad(a, N, K, beta):
    """
    Compute soft subgradient of max(conv) and related quantities.
    Returns: (g, f_val, c) where g is the gradient direction,
    f_val is max(c), c is the full convolution.
    """
    n = len(a)
    c = _convolve_fft(a, a, N)
    f_val = float(c.max())
    
    # Top-K indices
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))
    top_vals = c[top_idx]
    
    # Softmax weights
    shift = top_vals.max()
    w = np.exp((top_vals - shift) / beta)
    w /= w.sum()
    
    # Compute gradient via FFT
    # g_j = 2 * sum_i w_i * a_{k_i - j}
    m = np.arange(N // 2 + 1)
    phi = -2.0 * np.pi * np.outer(m, top_idx.astype(np.float64)) / N
    H = np.exp(1j * phi) @ w
    
    A = np.fft.rfft(a, N)
    G = H * np.conj(A)
    g = 2.0 * np.fft.irfft(G, N)[:n]
    
    return g, f_val, c

                 avg_every=50, spectral_every=2000):

    Subgradient descent with adaptive parameters.
    """

    N = _fft_n(n)
    a = a0 / a0.sum()
    
    if beta0 is None:
        c0 = _convolve_fft(a, a, N)
        beta0 = 0.01 * float(c0.max())
    
    f_lower = 1.0 / (2.0 * n - 1.0)
    
    best_a = a.copy()
    best_score = _score(a)

    a_avg = a.copy()
    count = 1
    
    t = 0
    last_print = time.monotonic()
    last_spectral = 0
    
    while time.monotonic() < deadline:
        t += 1

        K = max(3, int(K0 + (K1 - K0) * frac))
        beta = max(beta1, beta0 * (beta1 / beta0) ** frac)
        momentum = momentum0 + (momentum1 - momentum0) * frac
        polar_eff = polar * (0.5 + 0.5 * (1 - frac))  # start aggressive, gentle
        
        g, f_val, c = _compute_soft_grad(a, N, K, beta)
        
        # Polyak step
        g2 = float(np.dot(g, g))
        if g2 > 1e-30:
            eta = polar_eff * max(f_val - f_lower, 1e-15) / g2
        else:
            eta = 1e-8
        eta = min(eta, 0.3)
        
        # Update
        v = momentum * v + g
        a = _project_simplex(a - eta * v)
        
        # Average
        a_avg += (a - a_avg) / count
        count += 1
        
        # Periodic checks
        if t % avg_every == 0:
            sc = _score(a)

                best_score = sc
                best_a = a.copy()
            sc_avg = _score(a_avg)

                best_score = sc_avg
                best_a = a_avg.copy()

                print(f"      subgrad t={t}: cur={sc:.8f}, avg={sc_avg:.8f}, best={best_score:.8f}")
