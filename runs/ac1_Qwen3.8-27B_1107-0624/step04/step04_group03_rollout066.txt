def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_fft(c, a, tol_frac=0.0):
    n = len(a)
    c_max = float(c.max())
    if tol_frac > 0:
        active = (c >= c_max * (1.0 - tol_frac)).astype(np.float64)
    else:
        active = (c == c_max).astype(np.float64)
    if active.sum() == 0:
        active[int(np.argmax(c))] = 1.0
    a_rev = a[::-1]

    A = np.fft.rfft(active, N)
    B = np.fft.rfft(a_rev, N)
    conv_result = np.fft.irfft(A * B, N)
    grad = conv_result[n - 1:n - 1 + n].copy()

def _smooth_grad_z(z, beta):

    dot_ga_a = float(np.dot(grad_a, a))

    return grad_z
