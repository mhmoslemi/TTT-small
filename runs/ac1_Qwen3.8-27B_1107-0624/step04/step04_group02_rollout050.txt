        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        if lo <= hi:
            grad[lo:hi + 1] = 2.0 * a_curr[k_val - hi:k_val - lo + 1][::-1]
        # Scale for the full objective (2n * max_c)
        grad *= 2.0 * n
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        # Softmax parameterization gradient: dz/ds where s = log-softmax(z)
        p = a_curr
        grad_z = grad - np.dot(grad, p) * p
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        if t % 50 == 0:
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = _score_norm(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            if t % 500 == 0:
                print(f"    Adam iter {t}: score={sc:.10f}, best={best_sc:.10f}")
        # Decay learning rate
        if t % 2000 == 0:
            lr *= 0.7
            if lr < 0.001:
                break
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimizer: LP-based iteration, Adam refinement, local search,
    multi-restart with diverse profiles.
    """
    global t_lp_start
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) ***")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                sc_p = _score_norm(parent)
                print(f"Parent: n={n_parent}, score={sc_p:.10f}")
                update_best(parent, sc_p)
    except Exception:
        pass

    if parent is None:
        n_parent = 4096
        parent = _make_profile(n_parent, 0)
        sc_p = _score_norm(parent)
        print(f"Default uniform: n={n_parent}, score={sc_p:.10f}")
        update_best(parent, sc_p)

    n = len(parent)

    # --- Phase 0: Quick profile sweep to find good starting point (30s) ---
    print("Phase 0: Profile sweep")
    phase0_end = min(deadline, t_start + 30.0)
    profiles_to_try = [
        (n, 0, 1.0), (n, 1, 0.5), (n, 1, 1.0), (n, 1, 1.5),
        (n, 2, 0.5), (n, 2, 1.0), (n, 3, 1.0),
        (n, 4, 0.2), (n, 4, 0.3), (n, 5, 1.5), (n, 5, 2.0),
        (n, 6, 0.5), (n, 6, 1.0),
    ]
    for pn, pt, pm in profiles_to_try:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(pn, pt, pm)
        sc = _score_norm(p)
        update_best(p, sc)

    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: LP-based optimization on best found (700s) ---
    print("Phase 1: LP-based optimization")
    t_lp_start = time.monotonic()
    phase1_end = min(deadline, t_start + 700.0)
    a_work = best_seq.copy()
    a_best, sc_best, n_iters = _lp_phase(a_work, phase1_end, rng)
    update_best(a_best, sc_best)
    print(f"Phase 1 done: {n_iters} iters, best={best_score:.10f}")

    # --- Phase 2: Continue LP with smaller time limit per solve (300s) ---
    print("Phase 2: LP continuation")
    t_lp_start = time.monotonic()
    phase2_end = min(deadline, t_start + 1000.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        a_work = best_seq.copy()
        a_best2, sc_best2, n_iters2 = _lp_phase(a_work, phase2_end, rng)
        update_best(a_best2, sc_best2)
    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam refinement (200s) ---
    print("Phase 3: Adam refinement")
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        a_best3, sc_best3 = _adam_refine(best_seq, phase3_end, rng, lr=0.02)
        update_best(a_best3, sc_best3)
    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Local search (150s) ---
    print("Phase 4: Local search")
    phase4_end = min(deadline, t_start + 1350.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        a_best4, sc_best4 = _local_search(best_seq, phase4_end, rng)
        update_best(a_best4, sc_best4)
    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart with LP (200s) ---
    print("Phase 5: Multi-restart")
    phase5_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 4
        per_restart = max(15.0, (phase5_end - time.monotonic()) / n_restarts)
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            noise_level = 0.02 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n)
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = np.exp(z)
            p_start = p_start / p_start.sum()
            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            t_lp_start = time.monotonic()
            a_r, sc_r, _ = _lp_phase(p_start, dl, rng, max_iters=200)
            update_best(a_r, sc_r)
    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final intensive LP + local (remaining) ---
    print("Phase 6: Final polishing")
    if best_seq is not None and time.monotonic() < deadline - 20:
        # One more LP round
        t_lp_start = time.monotonic()
        dl_lp = min(deadline - 50, time.monotonic() + 100.0)
        a_f1, sc_f1, _ = _lp_phase(best_seq, dl_lp, rng, max_iters=200)
        update_best(a_f1, sc_f1)

        # Final local search
        if best_seq is not None and time.monotonic() < deadline - 10:
            dl_ls = deadline - 10
            a_f2, sc_f2 = _local_search(best_seq, dl_ls, rng)
            update_best(a_f2, sc_f2)

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Final output ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, 0)
        final_score = _score_full(final)
        print(f"FINAL: score={final_score:.10f}, n={len(final)}, sum={s:.6f}")
        # Validate
        if len(final) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                print(f"Evaluator validates: {sc_eval:.10f}")
            except Exception as ex:
                print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, 0)
    print(f"FALLBACK: score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
