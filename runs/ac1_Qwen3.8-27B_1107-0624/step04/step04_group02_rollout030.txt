"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. LP-based Frank-Wolfe (conditional gradient) method using cvxpy
2. Peak-targeting local search that identifies responsible element pairs
3. Coarse-to-fine multi-scale optimization
4. Improved logsumexp continuation with more aggressive beta schedule
5. Bimodal and W-shaped profile initializations

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Score assuming sum=1 (so score = 2n * max_conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, separation, width):
    """Two Gaussian bumps."""
    t = np.linspace(0.0, 1.0, n)
    center1 = 0.5 - separation / 2.0
    center2 = 0.5 + separation / 2.0
    a = np.exp(-(t - center1) ** 2 / (2 * width ** 2)) + np.exp(-(t - center2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_profile(n, freq, depth):
    """W-shaped profile with `freq` oscillations."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_triangular_weight(n):
    """a[i] proportional to (i+1)*(n-1-i), giving flat-ish convolution."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bspline_profile(n, order=4):
    """B-spline based profile (smooth, compact-ish support)."""
    t = np.linspace(0.0, 1.0, n)
    # Use a high-order B-spline shape: (u^k * (1-u)^k) type
    u = t * 0.9 + 0.05  # Slightly inset to avoid exact zeros
    u = np.clip(u, 0.01, 0.99)
    a = u ** (order - 1) * (1.0 - u) ** (order - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _build_conv_matrix(n, f):
    """Build sparse matrix A where A @ g = convolve(f, g). Shape: (2n-1, n)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_fw_step(f, M, timeout=60.0):
    """
    Solve LP: max sum(g) s.t. convolve(f, g) <= M, g >= 0.
    Returns optimal g or None if timeout/failure.
    """
    n = len(f)
    try:
        A = _build_conv_matrix(n, f)
        g = cp.Variable(n, non_neg=True)
        constraints = [A @ g <= M]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)
        
        t0 = time.monotonic()
        # Try multiple solvers
        solved = False
        for solver in [cp.CBC, cp.GLPK, cp.ECOS]:
            try:
                prob.solve(solver=solver, threads=1)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
            if time.monotonic() - t0 > timeout:
                break
        
        if solved and g.value is not None:
            return np.maximum(g.value, 0.0)
    except Exception:
        pass
    return None


def _fw_iteration(f, M, step_size, rng):
    """One Frank-Wolfe iteration using LP."""
    n = len(f)
    g = _lp_fw_step(f, M, timeout=80.0)
    if g is None:
        return f, M, False
    
    # Rescale g to match normalization of f
    s_f = f.sum()
    s_g = g.sum()
    if s_g < 1e-15:
        return f, M, False
    g = g * (s_f / s_g)
    
    # Update
    f_new = (1.0 - step_size) * f + step_size * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f, M, False
    f_new = f_new / s * s_f
    
    c_new = _full_convolve(f_new, f_new)
    M_new = float(c_new.max())
    return f_new, M_new, True


def _lp_fw_method(f_init, deadline, rng, max_iters=20):
    """LP-based Frank-Wolfe method."""
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    f = f / s * np.sqrt(2.0 * n)  # Normalize so sum = sqrt(2n)
    
    c = _full_convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    best_score = 2.0 * n * best_M / (2.0 * n)  # = best_M since sum=sqrt(2n)
    
    print(f"  FW start: M={M:.10f}, score={best_score:.10f}")
    
    # Step size schedule: start large, decrease
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        # Adaptive step size
        t = 2.0 / (it + 2.0)
        t = min(t, 0.5)
        
        f_new, M_new, success = _fw_iteration(f, M, t, rng)
        
        if success and M_new < best_M * 1.001:
            best_M = M_new
            best_f = f_new.copy()
            best_score = 2.0 * n * best_M / (2.0 * n)
            print(f"  FW iter {it}: M={M_new:.10f}, score={best_score:.10f}, t={t:.3f}")
        
        if success:
            f = f_new.copy()
            M = M_new
        else:
            # LP failed, try smaller step
            break
    
    # Convert back to sum=1 normalization
    best_f_norm = best_f / best_f.sum()
    return best_f_norm, best_score


def _run_lbfgs_continuation(a_init, deadline, rng, max_wall_time=None, beta_scale=1.0, min_beta_frac=0.0001):
    """L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    # More aggressive beta schedule
    beta_starts = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0007, 0.0003, 0.0001, min_beta_frac]
    betas = [b * max_c0 * beta_scale for b in beta_starts]
    
    if max_wall_time is not None:
        per_level = max(3.0, max_wall_time / len(betas))
    else:
        per_level = 30.0
    
    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        
        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g
        
        try:
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _adam_smoothed(a_init, deadline, rng, lr=0.02, k_top=5):
    """Adam with top-K smoothed gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    def compute_topk_grad(a_curr, k):
        c = _full_convolve(a_curr, a_curr)
        k_actual = min(k, len(c))
        top_idx = np.argsort(c)[-k_actual:]
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a_curr[k_val - hi:k_val - lo + 1][::-1]
        grad /= k_actual
        grad *= 4.0 * n
        return grad
    
    max_iters = 50000
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        t += 1
        a_curr = _softmax_fn(z)
        grad = compute_topk_grad(a_curr, k_top)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()
        
        if t % 5 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        # Decay learning rate
        if t % 2000 == 0:
            lr = max(0.001, lr * 0.95)
    
    return best_a, best_score


def _peak_targeted_search(a_init, deadline, rng, n_trials=20000):
    """Local search targeting the highest convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        
        # Find top K lags
        K = min(5, len(c))
        top_lags = np.argsort(c)[-K:]
        max_lag = int(top_lags[-1])
        
        # Strategy 1: Reduce elements contributing to the top lag
        if it % 3 == 0:
            # Find the pair with largest product at the top lag
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min <= i_max:
                pairs = np.arange(i_min, i_max + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                idx = pairs[int(np.argmax(products))]
                
                a_trial = a.copy()
                frac = rng.uniform(0.2, 0.6)
                delta = a_trial[idx] * frac
                a_trial[idx] -= delta
                a_trial[idx] = max(0.0, a_trial[idx])
                
                # Add to a "safe" position (low contribution to top lags)
                # Compute contribution score for each position
                contrib = np.zeros(n)
                for kl in top_lags:
                    kl = int(kl)
                    lo = max(0, kl - (n - 1))
                    hi = min(n - 1, kl)
                    for j in range(lo, hi + 1):
                        partner = kl - j
                        if 0 <= partner < n:
                            contrib[j] += a[partner]
                
                # Pick from lowest-contribution positions
                safe = np.argsort(contrib)[:max(1, n // 5)]
                add_idx = int(safe[rng.integers(len(safe))])
                a_trial[add_idx] += delta
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm_fn(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
        
        # Strategy 2: Random block multiplicative perturbation
        elif it % 3 == 1:
            block_size = int(rng.integers(2, min(16, n // 8)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
        
        # Strategy 3: Pair swap (exchange mass between two positions)
        else:
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            # Swap a fraction of mass
            frac = rng.uniform(0.1, 0.4)
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] -= amt1
            a_trial[i2] -= amt2
            a_trial[i1] = max(0.0, a_trial[i1])
            a_trial[i2] = max(0.0, a_trial[i2])
            a_trial[i1] += amt2
            a_trial[i2] += amt1
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
    
    return best_a, best_sc

def _multi_scale_optimize(a_init, deadline, rng):
    """Optimize at multiple scales: downsample, optimize, upsample, refine."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score_norm_fn(a_init)
    
    # Coarse scale
    n_coarse = max(128, n // 8)
    a_coarse = _interp_profile(a_init, n_coarse)
    sc_coarse = _score_norm_fn(a_coarse)
    
    # Optimize at coarse scale
    dl_coarse = min(deadline - 60, time.monotonic() + 120.0)
    a_c_best, sc_c_best = _run_lbfgs_continuation(a_coarse, dl_coarse, rng, max_wall_time=120.0)
    
    # Upsample and refine
    if sc_c_best < best_sc / (n_coarse / n):  # Rough comparison
        a_up = _interp_profile(a_c_best, n)
        sc_up = _score_norm_fn(a_up)
        if sc_up < best_sc:
            best_sc = sc_up
            best_a = a_up.copy()
    
    # Fine-scale refinement
    if time.monotonic() < deadline - 30:
        dl_fine = deadline - 20
        a_best, sc_best = _run_lbfgs_continuation(best_a, dl_fine, rng, max_wall_time=100.0, beta_scale=0.3)
        if sc_best < best_sc:
            best_sc = sc_best
            best_a = a_best.copy()
    
    return best_a, best_sc

        _make_raised_cosine(n_try, 2.0),
    ]
    
    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, best_candidate={candidates[0][2] if candidates else 'none'}")
    
    # --- Phase 1: LP-based Frank-Wolfe (500s) ---
    phase1_end = min(deadline, t_start + 540.0)
    
    # Use the best starting point for FW
    fw_start = candidates[0][1].copy() if candidates else parent.copy()
    # Also try the parent
    if parent is not None:
        sc_p = _score_norm_fn(parent)
        if sc_p < _score_norm_fn(fw_start):
            fw_start = parent.copy()
    
    print(f"Phase 1: LP Frank-Wolfe, starting score={_score_norm_fn(fw_start):.10f}")
    a_fw, sc_fw = _lp_fw_method(fw_start, phase1_end, rng, max_iters=15)
    update_best(a_fw, sc_fw, "fw_lp")
    
    # --- Phase 2: L-BFGS continuation on best (300s) ---
    phase2_end = min(deadline, t_start + 840.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("Phase 2: L-BFGS continuation")
        dl = phase2_end - 5.0
        a_best, sc_best = _run_lbfgs_continuation(best_seq, dl, rng, max_wall_time=290.0, beta_scale=1.0)
        update_best(a_best, sc_best, "lbfgs_main")
        
        # Second pass with smaller beta
        if time.monotonic() < phase2_end - 20:
            dl2 = phase2_end - 5.0
            a_best2, sc_best2 = _run_lbfgs_continuation(best_seq, dl2, rng, max_wall_time=60.0, beta_scale=0.4, min_beta_frac=0.00005)
            update_best(a_best2, sc_best2, "lbfgs_fine")
    
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: Adam smoothed (200s) ---
    phase3_end = min(deadline, t_start + 1040.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: Adam top-K")
        dl = phase3_end - 5.0
        a_best, sc_best = _adam_smoothed(best_seq, dl, rng, lr=0.012, k_top=3)
        update_best(a_best, sc_best, "adam_k3")
        
        if time.monotonic() < phase3_end - 15:
            dl2 = phase3_end - 5.0
            a_best2, sc_best2 = _adam_smoothed(best_seq, dl2, rng, lr=0.008, k_top=7)
            update_best(a_best2, sc_best2, "adam_k7")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Peak-targeted local search (250s) ---
    phase4_end = min(deadline, t_start + 1290.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Peak-targeted local search")
        dl = phase4_end - 5.0
        a_best, sc_best = _peak_targeted_search(best_seq, dl, rng, n_trials=30000)
        update_best(a_best, sc_best, "peak_search")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Multi-restart (200s) ---
    phase5_end = min(deadline, t_start + 1490.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Multi-restart")
        n_restarts = 4
        per_restart = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            noise = 0.02 + 0.015 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            a_best, sc_best = _run_lbfgs_continuation(p_start, dl, rng, max_wall_time=per_restart, beta_scale=0.6)
            update_best(a_best, sc_best, f"restart{i}")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Try different n (150s) ---
    phase6_end = min(deadline, t_start + 1640.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        
        # Try larger n
        for target_n in [n_curr * 2]:
            if target_n > 8192 or target_n <= n_curr:
                continue
            if time.monotonic() > phase6_end - 60:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interp to n={target_n}: score={sc_interp:.10f}")
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")
            
            if time.monotonic() < phase6_end - 30:
                dl = min(phase6_end - 20, time.monotonic() + 90.0)
                a_best, sc_best = _run_lbfgs_continuation(p_interp, dl, rng, max_wall_time=90.0, beta_scale=0.5)
                update_best(a_best, sc_best, f"lbfgs_n{target_n}")
        
        # Try the best profile family at n=4096 if current is different
        if n_curr != 4096 and time.monotonic() < phase6_end - 30:
            p4096 = _interp_profile(best_seq, 4096)
            sc4096 = _score_norm_fn(p4096)
            update_best(p4096, sc4096, "interp_n4096")
    
    print(f"Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Final intensive polishing (remaining time) ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 7: Final polishing, remaining={remaining:.0f}s")
        
        # 7a: Very fine L-BFGS
        if remaining > 60:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.4)
            a_best, sc_best = _run_lbfgs_continuation(best_seq, dl, rng,
                                                       max_wall_time=remaining * 0.4,
                                                       beta_scale=0.2, min_beta_frac=0.00003)
            update_best(a_best, sc_best, "final_lbfgs")
        
        # 7b: More local search
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _peak_targeted_search(best_seq, dl, rng, n_trials=20000)
            update_best(a_best, sc_best, "final_local")
        
        # 7c: Final Adam with tiny lr
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.25)
            a_best, sc_best = _adam_smoothed(best_seq, dl, rng, lr=0.003, k_top=2)
            update_best(a_best, sc_best, "final_adam")
    
    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(4096)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        # Validate with evaluator
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_triangular_weight(4096)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
