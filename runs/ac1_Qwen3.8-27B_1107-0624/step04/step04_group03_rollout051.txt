"""
Active-set subgradient optimizer with equalization penalty, spectral initialization,
and structured restarts for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Key improvements over previous: exact subgradient with constraint equalization,
GS-based initialization, adaptive step size, and focused polishing.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights_smooth(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _active_set_weights(c, frac=0.01):
    """Get weights concentrated on the top `frac` fraction of lags."""
    cm = float(c.max())
    threshold = cm - frac * cm
    active = c >= threshold
    if not np.any(active):
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / np.sum(active)
    return w, cm


def _gradient_exact(z, active_frac=0.01):
    """Exact subgradient using active set."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _active_set_weights(c, active_frac)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return grad, c, cm, w


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights_smooth(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _gradient_with_equalization(z, active_frac=0.01, eq_weight=1.0):
    """Subgradient plus equalization penalty gradient."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    
    # Active set with tighter threshold
    threshold = cm - active_frac * cm
    active = c >= threshold
    if not np.any(active):
        active[int(np.argmax(c))] = True
    
    # Standard subgradient weights (uniform over active)
    w_sub = np.zeros_like(c)
    w_sub[active] = 1.0 / np.sum(active)
    
    # Equalization weights: proportional to (c[k] - mean_active) for active k
    c_active = c[active]
    mean_active = float(c_active.mean())
    spread = c[active] - mean_active
    # Normalize so max weight is 1
    max_spread = float(np.max(np.abs(spread))) if max_spread < 1e-30 else 1.0
    w_eq = np.zeros_like(c)
    w_eq[active] = spread / max_spread
    # Shift so weights sum to 0 (pure equalization direction)
    w_eq[active] -= np.mean(w_eq[active])
    
    # Compute gradient from subgradient
    W_sub = float(np.dot(w_sub, c))
    a_rev = a[::-1]
    w_conv_sub = _full_convolve(w_sub, a_rev)
    h_sub = w_conv_sub[n - 1:2 * n - 1]
    grad_sub = 4.0 * n * a * (h_sub - W_sub)
    
    # Compute gradient from equalization
    w_conv_eq = _full_convolve(w_eq, a_rev)
    h_eq = w_conv_eq[n - 1:2 * n - 1]
    W_eq = float(np.dot(w_eq, c))
    grad_eq = 4.0 * n * a * (h_eq - W_eq)
    
    grad = grad_sub + eq_weight * grad_eq
    return grad, c, cm, active


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _run_lbfgs(a_init, deadline, max_wall_time=200.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    betas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    betas = [b * max_c0 for b in betas]
    per_level = max_wall_time / len(betas)

    def obj(z):
        return _objective_smooth(z, betas[0]) if False else _objective_smooth(z, beta)

    def grd(z):
        return _gradient_smooth(z, beta)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _lambda_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_lambda_grad,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 20, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass

    return best_a, best_score


def _lambda_obj(z, beta):
    return _objective_smooth(z, beta)


def _lambda_grad(z, beta):
    return _gradient_smooth(z, beta)


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.015, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)
        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)
            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)
            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()
            z = z_new
            y = y_new
            if t % 100 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
            if t % 50 == 0:

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.003, lr * 0.75)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _adam_optimize(a_init, deadline, lr=0.03, beta_frac=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights_smooth(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            continue
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if t % 100 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _make_linear_profile(n, slope=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1=1.0, a2=0.2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, power=2.0):
    t = np.linspace(1e-6, 1.0, n)
    a = t ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gs_profile(n, n_iter=200, seed=42):
    """Gerchberg-Saxton for flat spectrum design."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()

    N = _next_pow2(2 * n - 1)
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        # Target: flat magnitude (except DC)
        mag = np.abs(A)
        target_mag = np.median(mag[1:]) * np.ones(N // 2 + 1)
        target_mag[0] = mag[0]  # Keep DC
        # Blend: keep some of the original phase, set magnitude to target
        A_new = target_mag * (A / np.maximum(np.abs(A), 1e-30))
        a_new = np.fft.irfft(A_new, N)[:n]
        # Project onto probability simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s
        else:
            a = np.ones(n) / n
    return a


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _structured_noise(a, rng, mode=0):
    """Generate structured (low-frequency) noise for restarts."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    freqs = np.fft.rfftfreq(n)
    if mode == 0:
        # Very low frequency
        mask = freqs < 2.0 / n
    elif mode == 1:
        mask = freqs < 5.0 / n
    elif mode == 2:
        mask = freqs < 10.0 / n
    else:
        mask = np.ones(n, dtype=bool)
    
    noise_spec = np.zeros(n // 2 + 1, dtype=np.complex128)
    noise_spec[mask.astype(bool)[:n//2+1]] = rng.normal(size=np.sum(mask.astype(bool)[:n//2+1])) + 1j * rng.normal(size=np.sum(mask.astype(bool)[:n//2+1]))
    noise = np.fft.irfft(noise_spec, n)
    noise = noise / (np.std(noise) + 1e-10)
    z = z + 0.05 * noise
    z = z - z.mean()
    z = np.clip(z, -50, 50)
    return _softmax_fn(z)


def _fine_polish(a_init, deadline, lr=0.001, active_frac=0.005):
    """Very fine polishing with exact subgradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    momentum = np.zeros(n)
    mom_coeff = 0.9

    while time.monotonic() < deadline:
        grad, c, cm, active = _gradient_with_equalization(z, active_frac, 0.3)
        if not np.all(np.isfinite(grad)):
            break
        momentum = mom_coeff * momentum - lr * grad
        z_new = z + momentum
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -50, 50)
        z = z_new

        if np.random.random() < 0.1:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: spectral init -> L-BFGS warmup -> Nesterov continuation
    -> active-set subgradient with equalization -> fine polish -> structured restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 4096
        parent = _make_linear_profile(n, 1.5)
        print(f"No parent, using linear profile n={n}")

    # --- Phase 0: Initialization (60s) ---
    phase0_end = min(deadline, t_start + 60.0)
    candidates = []

    sc_parent = _score_norm_fn(parent)
    candidates.append((sc_parent, parent.copy(), "parent"))
    update_best(parent, sc_parent, "parent")

    # Try different analytic profiles
    profiles = [
        (_make_linear_profile(n, 1.0), "linear_1.0"),
        (_make_linear_profile(n, 2.0), "linear_2.0"),
        (_make_quadratic_profile(n, 1.5, 0.3), "quad"),
        (_make_power_profile(n, 1.5), "power_1.5"),
        (_make_power_profile(n, 2.5), "power_2.5"),
    ]
    for p, label in profiles:
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, label))

    # Try GS initialization
    if time.monotonic() < phase0_end - 10:
        gs = _make_gs_profile(n, n_iter=100, seed=seed)
        sc_gs = _score_norm_fn(gs)
        candidates.append((sc_gs, gs, "gs"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")
    for sc, p, label in candidates[:3]:
        print(f"  {label}: {sc:.10f}")

    # --- Phase 1: L-BFGS warmup (200s) ---
    phase1_end = min(deadline, t_start + 260.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        per_time = min(180.0, phase1_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, min(phase1_end - 5, time.monotonic() + per_time),
                                          max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_warmup")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov with smooth max (400s) ---
    phase2_end = min(deadline, t_start + 660.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                      5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
        p_best, sc_best = _nesterov_optimize(best_seq, phase2_end, beta_fracs,
                                              lr_init=0.012, mom_init=0.9)
        update_best(p_best, sc_best, "nesterov")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Active-set subgradient with equalization (400s) ---
    phase3_end = min(deadline, t_start + 1060.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(380.0, phase3_end - time.monotonic())
        if per_time > 30:
            p_best, sc_best = _subgradient_equality(best_seq, min(phase3_end - 5, time.monotonic() + per_time),
                                                     lr_init=0.003, active_frac=0.01, eq_weight=0.5)
            update_best(p_best, sc_best, "subgrad_eq")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Adam polishing (200s) ---
    phase4_end = min(deadline, t_start + 1260.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(180.0, phase4_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _adam_optimize(best_seq, min(phase4_end - 5, time.monotonic() + per_time),
                                              lr=0.02, beta_frac=0.003)
            update_best(p_best, sc_best, "adam_polish")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Fine polishing (150s) ---
    phase5_end = min(deadline, t_start + 1410.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        per_time = min(130.0, phase5_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _fine_polish(best_seq, min(phase5_end - 5, time.monotonic() + per_time),
                                            lr=0.0008, active_frac=0.005)
            update_best(p_best, sc_best, "fine_polish")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Structured restarts (250s) ---
    phase6_end = deadline
    if best_seq is not None and time.monotonic() < phase6_end:
        n_restarts = 4
        per_restart = min(50.0, (phase6_end - time.monotonic()) / max(1, n_restarts) - 5)

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            p_start = _structured_noise(best_seq, rng, mode=i % 3)
            per_dl = min(phase6_end - 5, time.monotonic() + per_restart)

            # Quick Nesterov on the restart
            beta_short = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
            p_best, sc_best = _nesterov_optimize(p_start, per_dl, beta_short,
                                                  lr_init=0.01, mom_init=0.85)
            update_best(p_best, sc_best, f"restart{i}")

            # Quick subgradient
            if time.monotonic() < phase6_end - 20:
                per_sg = min(20.0, phase6_end - time.monotonic() - 10)
                if per_sg > 5:
                    p_best, sc_best = _subgradient_equality(p_best, min(phase6_end - 5, time.monotonic() + per_sg),
                                                             lr_init=0.002, active_frac=0.01, eq_weight=0.5)
                    update_best(p_best, sc_best, f"restart_sg{i}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(4096, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, using internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_linear_profile(4096, 1.5)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
