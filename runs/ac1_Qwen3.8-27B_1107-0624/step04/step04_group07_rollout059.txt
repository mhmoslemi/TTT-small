import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    n = len(a)
    c = _full_convolve(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_min_peak(a, n, method='glop'):
    """Solve: min t s.t. conv(a,b)_k <= t, sum(b)=1, b>=0, t>=0.
    Returns (b_optimal, t_optimal) or (None, None)."""
    A_conv = _build_conv_matrix(n, a)
    n_row = 2 * n - 1

    rows = []
    cols = []
    vals = []
    for k in range(n_row):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            for j in idx:
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)

    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_row, n + 1))
    b_ub = np.zeros(n_row)

    A_eq = csr_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=method)
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            return None, None

    if res.success and res.x is not None:
        b_opt = np.maximum(res.x[:n], 0.0)
        s = float(b_opt.sum())
        if s > 1e-10:
            b_opt = b_opt / s
        return b_opt, float(res.x[n])
    return None, None


def _lp_fixed_point(a, deadline, max_iters=60, use_acceleration=True):
    """LP fixed-point: iterate b = argmin max_k conv(a,b)_k s.t. sum(b)=1, b>=0."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    prev_a = None
    momentum = 0.85

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b, t_opt = _solve_lp_min_peak(a, n)
        if b is None or float(b.sum()) < 1e-10:
            break

        b = b / b.sum()

        if use_acceleration and prev_a is not None:
            # Nesterov-style: y = a + momentum * (a - prev_a), then blend toward b
            y = a + momentum * (a - prev_a)
            y = np.maximum(y, 1e-15)
            y = y / y.sum()
            a_new = 0.5 * y + 0.5 * b
        else:
            # Simple convex combination with adaptive step
            alpha = max(0.3, 1.0 - 0.01 * it)
            a_new = (1.0 - alpha) * a + alpha * b

        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        prev_a = a.copy()
        a = a_new

        # Decay momentum
        if use_acceleration:
            momentum = max(0.3, momentum * 0.99)

    return best_a, best_score


def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None,
                     beta_decay=0.95, no_imp_limit=200):
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_correlate(w, a)

        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)

        if no_imp > no_imp_limit:
            ns = 0.06 if best_score < 1.51 else 0.12
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _lbfgs_refine(a, deadline, beta=1e-4, max_iter=200):
    """L-BFGS-B refinement on smooth-max objective in log-space."""
    n = len(a)
    a0 = np.maximum(a, 1e-15).copy()
    a0 = a0 / a0.sum()
    z0 = np.log(a0)
    z0 -= z0.mean()

    def objective_and_grad(z):
        z_mod = np.clip(z, -50, 50)
        z_mod = z_mod - z_mod.max()
        a_tmp = np.exp(z_mod)
        a_tmp = a_tmp / a_tmp.sum()
        a_tmp = np.maximum(a_tmp, 1e-15)
        a_tmp = a_tmp / a_tmp.sum()

        c = _full_convolve(a_tmp)
        sm, w = _smooth_max_weights(c, beta)
        score = 2.0 * n * sm

        # Gradient
        g_cross = _cross_correlate(w, a_tmp)
        dot = float(np.dot(a_tmp, g_cross))
        grad_a = 2.0 * (g_cross - dot)
        grad_z = a_tmp * grad_a - a_tmp * float(np.dot(a_tmp, grad_a))
        grad_z -= grad_z.mean()

        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)

        return score, grad_z

    obj_fun = lambda z: objective_and_grad(z)[0]
    jac_fun = lambda z: objective_and_grad(z)[1]

    try:
        res = minimize(obj_fun, z0, jac=jac_fun, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'gtol': 1e-10,
                                'ftol': 1e-12, 'maxfun': max_iter * 3})
        if res.success or res.fun < obj_fun(z0):
            z_res = res.x
            z_res = np.clip(z_res, -50, 50)
            z_res -= z_res.mean()
            a_res = np.exp(z_res - z_res.max())
            a_res = a_res / a_res.sum()
            a_res = np.maximum(a_res, 1e-15)
            a_res = a_res / a_res.sum()
            sc = _score_norm(a_res)
            return a_res, sc
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    return a0, _score_norm(a0)


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'exp':
        t = np.arange(n)
        a = np.exp(-p1 * t / n)
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / p3) ** 2) + np.exp(-0.5 * ((t - p2) / p3) ** 2)
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'chirp':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * t + p2 * t ** 2
    elif ptype == 'sawtooth':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * np.abs(2 * t - 1) ** p2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

        if n_cur <= 1024:
            target_ns = [2048]
        elif n_cur <= 2048:
            target_ns = [4096]

        for tn in target_ns:
            if time.monotonic() > phase5_end - 80:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score_norm(p_int)
            print(f"  Interpolated to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Adam on interpolated
            if time.monotonic() < phase5_end - 40:
                per = min(80.0, phase5_end - time.monotonic() - 20)
                if per > 20:
                    a_e, sc_e = _adam_smooth_max(p_int, time.monotonic() + per, rng,
                                                 lr=0.03, beta_decay=0.96, no_imp_limit=200)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

    # === Phase 6: Random restarts ===
    print("Phase 6: Random restarts...")
    phase6_end = deadline - 100.0

    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(10.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 10:
                break

            # Structured perturbation: mix with a random profile
            ns = 0.04 + 0.04 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            a_r, sc_r = _adam_smooth_max(a_s, time.monotonic() + per, rng,
                                         lr=0.04, beta_decay=0.94, no_imp_limit=120)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i}]")

            # Quick LP polish
            if n_best <= 1024 and time.monotonic() < phase6_end - 10:
                per2 = min(15.0, phase6_end - time.monotonic() - 5)
                if per2 > 8:
                    a_rl, sc_rl = _lp_fixed_point(a_r, time.monotonic() + per2,
                                                  max_iters=10, use_acceleration=True)
                    if sc_rl < best_score:
                        best_seq = a_rl.copy()
                        best_score = sc_rl
                        print(f"  BEST: {best_score:.10f} n={n_best} [LP_restart{i}]")

    # === Phase 7: Final polish ===
    print("Phase 7: Final polish...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 10:
            if best_seq.size <= 1024:
                a_fp, sc_fp = _lp_fixed_point(best_seq, time.monotonic() + min(per, 60),
                                              max_iters=30, use_acceleration=True)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_LP]")
                # L-BFGS after LP
                if time.monotonic() < deadline - 15:
                    a_fl, sc_fl = _lbfgs_refine(best_seq, deadline - 15,
                                                beta=5e-5, max_iter=80)
                    if sc_fl < best_score:
                        best_seq = a_fl.copy()
                        best_score = sc_fl
                        print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_LBFOPS]")
            else:
                a_fp, sc_fp = _adam_smooth_max(best_seq, time.monotonic() + min(per, 60), rng,
                                               lr=0.015, beta_decay=0.96, no_imp_limit=80)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_Adam]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0, 0)

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
