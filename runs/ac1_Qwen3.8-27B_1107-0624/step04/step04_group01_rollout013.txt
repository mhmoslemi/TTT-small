import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score without normalization: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_step(a, M=None):
    """
    One LP step: find b >= 0 maximizing sum(b) subject to conv(a,b) <= M.
    Returns b (unnormalized) and M.
    """
    n = len(a)
    if M is None:
        c = _full_convolve(a, a)
        M = float(c.max())
    
    # Build constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max 1^T b s.t. A b <= M*1, b >= 0
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        # LP failed, return a copy of a
        return a.copy(), M
    
    b = res.x
    # Numerical cleanup
    b = np.maximum(b, 0)
    return b, M

LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the "best response" LP method: at each step, solve an LP to find
the direction b that keeps conv(a,b) bounded, then move towards b.
Combined with gradient-based polishing and multi-restart.

    while k < n:

    out_n = n + m - 1
    N = _next_pow2(out_n)

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _build_conv_matrix(a):
    """Build sparse matrix A such that A @ b = conv(a, b)."""
    n = len(a)
    # A[k,j] = a[k-j] for 0<=k-j<n
    # For each i in [0,n-1], a[i] appears at (j+i, j) for j=0,...,n-1
    i_vals = np.arange(n)
    j_vals = np.tile(np.arange(n), n)
    k_vals = j_vals + np.repeat(i_vals, n)
    a_vals = np.repeat(a, n)
    return sparse.coo_matrix((a_vals, (k_vals, j_vals)), shape=(2*n-1, n)).tocsr()

def _lp_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b (unnormalized)."""
    n = len(a)
    A = _build_conv_matrix(a)
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 30, 'presolve': True})
    
    if res.status != 0:
        return None
    
    b = np.maximum(res.x, 0)
    if b.sum() < 1e-15:
        return None
    return b

def _lp_optimize(a_init, deadline, max_iter=5000, t_init=0.3):

def _adam_polish(a_init, deadline, max_iter=50000, lr=0.005):
    """Adam optimization in log-space with log-sum-exp max approximation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    
    beta_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    # Scale beta by the current max
    c_init = _full_convolve(a_init, a_init)
    cm_init = float(c_init.max())
    beta_schedule = [b * cm_init for b in beta_schedule]
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        for _ in range(max_iter):
            if time.monotonic() > deadline:
                break
            t += 1
            
            # Softmax
            z_max = np.max(z)
            e = np.exp(z - z_max)
            s = e.sum()
            a = e / s if s > 1e-300 else np.ones(n) / n
            
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e_c = np.exp(c_shifted)
            s_c = float(e_c.sum())
            if s_c < 1e-300 or not np.isfinite(s_c):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e_c / s_c
            
            # Gradient: d/dz [2n * sum(w * c)] = 4n * a * (h - W)
            # where h[m] = sum_i w[i] * a[i-m] (cross-correlation of w and a)
            # and W = dot(w, c)
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1**t)
            vh = v_buf / (1.0 - b2**t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            max_update = 0.3
            update_norm = np.linalg.norm(update)
            if update_norm > max_update:
                update = update * (max_update / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 200 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    # Final evaluation
    z_max = np.max(z)
    e = np.exp(z - z_max)
    a_final = e / e.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
