"""
Active-set projected gradient with FFT-based computation for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.

Key innovations over previous approach:
- FFT-based gradient (O(n log n) vs O(n*|A|) for large active sets)
- Exact backtracking line search for guaranteed monotonic decrease
- Adaptive active set with hysteresis (grow slowly, shrink slowly)
- Periodic exploration via structured perturbations
- Multi-scale refinement (optimize at 4096, interpolate to 8192, re-optimize)
"""
import numpy as np
import time

# --- Core computation functions ---

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]

def _self_convolve(a):
    return _fft_convolve(a, a)

def _cross_corr_fft(w, a):
    """g[i] = sum_k w[k] * a[k-i] for i=0..n-1, using FFT."""
    n = len(a)
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)
    return conv[n-1 : 2*n-1]

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _score_normalized(a):
    """Score for a with sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    c = _self_convolve(a)
    return float(2.0 * n * c.max())

def _grad_a_smooth(a, active_set):
    """Gradient of (1/|A|)*sum_{k in A} c_k w.r.t. a, via FFT."""
    n = len(a)
    w = np.zeros(2*n - 1)
    w[active_set] = 1.0 / len(active_set)
    return _cross_corr_fft(w, a)

def _grad_z_smooth(a, active_set):
    """Gradient in logit space."""
    grad_a = _grad_a_smooth(a, active_set)
    center = np.dot(grad_a, a)
    return a * (grad_a - center)

def _grad_z_smooth_beta(a, active_set, beta):
    """Weighted gradient with temperature beta (emphasizes higher peaks)."""
    n = len(a)
    c = _self_convolve(a)
    c_active = c[active_set]
    w_active = np.exp((c_active - c_active.max()) / beta)
    w_active = w_active / w_active.sum()
    w = np.zeros(2*n - 1)
    w[active_set] = w_active
    grad_a = _cross_corr_fft(w, a)
    center = np.dot(grad_a, a)
    return a * (grad_a - center)

def _active_set_optimize(a_init, deadline, verbose=True):
    """
    Active-set projected gradient with backtracking line search.
    Works in logit space (softmax parameterization ensures non-negativity).
    """
    n = len(a_init)
    a = a_init / a_init.sum()
    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    # Active set management
    update_interval = 20
    grow_tol = 0.02  # grow threshold: include if c >= M*(1-grow_tol)
    shrink_tol = 0.05  # shrink: remove if c < M*(1-shrink_tol)
    active_set = None
    
    step_size = 0.05
    min_step = 1e-7
    perturbation_interval = 200
    no_improvement_count = 0
    last_improve_score = best_score
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        c = _self_convolve(a)
        M = float(c.max())
        score = 2.0 * n * M
        
        # Update active set periodically
        if it % update_interval == 1 or active_set is None:
            if active_set is None:
                threshold = M * (1 - grow_tol)
                active_set = np.where(c >= threshold)[0]
            else:
                # Hysteresis: grow with grow_tol, shrink with shrink_tol
                grow_mask = c >= M * (1 - grow_tol)
                shrink_mask = c < M * (1 - shrink_tol)
                # Remove indices that dropped below shrink threshold
                current_mask = np.zeros(2*n-1, dtype=bool)
                current_mask[active_set] = True
                # New active set: (current OR grow) AND NOT (shrink AND not newly grown)
                new_mask = (current_mask | grow_mask) & ~(shrink_mask & ~grow_mask)
                # Simpler: keep if above shrink threshold OR was just added
                new_mask = grow_mask | (current_mask & ~shrink_mask)
                active_set = np.where(new_mask)[0]
        
        if len(active_set) == 0:
            active_set = np.array([int(np.argmax(c))])
        
        # Compute gradient
        grad_z = _grad_z_smooth(a, active_set)
        
        # If gradient is too small, perturb
        if np.linalg.norm(grad_z) < 1e-12:
            break
        
        # Backtracking line search
        alpha = step_size
        improved = False
        for _ in range(30):
            z_trial = z - alpha * grad_z
            z_trial = np.clip(z_trial, -60, 60)
            z_trial -= z_trial.mean()
            a_trial = _softmax(z_trial)
            sc_trial = _score_normalized(a_trial)
            if sc_trial < score * (1 - 1e-12) - 1e-15:
                z = z_trial
                a = a_trial
                sc = sc_trial
                improved = True
                # Increase step if we took the full step
                if _ == 0:
                    step_size = min(0.5, step_size * 1.3)
                break
            alpha *= 0.5
        
        if improved:
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            if sc < last_improve_score - 1e-8:
                last_improve_score = sc
                no_improvement_count = 0
        else:
            no_improvement_count += 1
            step_size *= 0.5
        
        # Periodic perturbation to escape local minima
        if it % perturbation_interval == 0 and no_improvement_count > 100:
            noise = 0.05 * np.random.randn(n)
            z += noise
            z -= z.mean()
            a = _softmax(z)
            step_size = 0.03
            no_improvement_count = 0
            if verbose:
                print(f"    [it {it}] Perturbation, current score={score:.10f}, best={best_score:.10f}")

    return best_a, best_score

def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()

def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_poly_profile(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i, c in enumerate(coeffs):
        a += c * t**i
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

import numpy as np
from scipy.special import i0
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def _self_convolve(a):
    return _fft_convolve(a, a)


def _cross_corr_fft(w, a):
    """Compute g[i] = sum_k w[k] * a[k-i] for i=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)
    return conv[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score_normalized(a):
    """Score for a with sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    c = _self_convolve(a)
    return float(2.0 * n * c.max())


def _compute_grad_z(a, c, active_set, beta):
    """Compute gradient in logit space given self-convolution and active set.
    beta controls concentration: beta->0 gives uniform on argmax, large beta spreads."""
    n = len(a)
    c_active = c[active_set]
    if beta is not None and beta > 1e-15:
        w_active = np.exp((c_active - c_active.max()) / beta)
        w_active = w_active / w_active.sum()
    else:
        w_active = np.ones(len(active_set)) / len(active_set)
    w = np.zeros(2 * n - 1)
    w[active_set] = w_active
    grad_a = _cross_corr_fft(w, a)
    center = np.dot(grad_a, a)
    return a * (grad_a - center)


def _active_set_optimize(a_init, deadline, beta_start=0.01, beta_end=0.0001,
                         verbose=True, momentum_decay=0.85, max_momentum=500):

    Active-set projected gradient with backtracking line search and momentum.
    Works in logit space (softmax ensures non-negativity).
    """
    n = len(a_init)
    a = a_init / a_init.sum()
    z = np.log(np.maximum(a, 1e-30))
    z -= z.mean()

    best_a = a.copy()
    best_score = _score_normalized(a)

    # Active set management parameters
    update_interval = 15
    grow_tol = 0.03
    shrink_tol = 0.06
    active_set = None
    c = _self_convolve(a)
    M = float(c.max())
    active_set = np.where(c >= M * (1 - grow_tol))[0]

    # Temperature schedule
    total_budget = max(1.0, deadline - time.monotonic())
    it = 0
    step_size = 0.03
    min_step = 1e-8
    max_step = 0.3
    v = np.zeros(n)  # momentum
    no_improve = 0
    perm_interval = 300
    perm_done = 0

    while time.monotonic() < deadline:
        it += 1
        c = _self_convolve(a)
        M = float(c.max())
        score = 2.0 * n * M

        # Update active set with hysteresis
        if it % update_interval == 0:
            grow_mask = c >= M * (1 - grow_tol)
            shrink_mask = c < M * (1 - shrink_tol)
            current = np.zeros(2 * n - 1, dtype=bool)
            current[active_set] = True
            # Keep if above shrink threshold; add if above grow threshold
            new_mask = grow_mask | (current & ~shrink_mask)
            active_set = np.where(new_mask)[0]
            if len(active_set) == 0:
                active_set = np.array([int(np.argmax(c))])

        # Temperature schedule: decrease over time
        frac = min(1.0, it / (0.5 * total_budget / 0.002))
        beta = beta_start * (beta_end / beta_start) ** min(frac, 1.0)

        # Compute gradient
        grad_z = _compute_grad_z(a, c, active_set, beta)

        # Momentum
        if it <= max_momentum:
            v = momentum_decay * v + (1 - momentum_decay) * grad_z
        else:
            v = grad_z

        # Backtracking line search
        alpha = step_size
        improved = False
        best_trial_score = score
        best_trial_z = z
        best_trial_a = a
        for ls in range(25):
            z_trial = z - alpha * v
            z_trial = np.clip(z_trial, -60, 60)
            z_trial -= z_trial.mean()
            a_trial = _softmax(z_trial)
            sc_trial = _score_normalized(a_trial)
            if sc_trial < best_trial_score:
                best_trial_score = sc_trial
                best_trial_z = z_trial
                best_trial_a = a_trial
            if sc_trial < score * (1 - 1e-10) - 1e-14:
                improved = True
                z = z_trial
                a = a_trial
                step_size = min(max_step, alpha * 1.4)
                break
            alpha *= 0.5

        if not improved and best_trial_score < score:
            z = best_trial_z
            a = best_trial_a
            improved = True
            step_size = min(max_step, alpha * 2.0)
            sc = best_trial_score
        elif improved:
            sc = score * (1 - 1e-10)
        else:
            sc = score
            step_size *= 0.5

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic perturbation to escape local minima
        if it % perm_interval == 0 and no_improve > 80:
            strength = 0.03 + 0.02 * perm_done
            z += strength * np.random.randn(n)
            z -= z.mean()
            a = _softmax(z)
            v = np.zeros(n)
            step_size = 0.02
            no_improve = 0
            perm_done += 1

def _smoothed_objective(z, beta, n, top_k):
    """Logsumexp-smoothed objective in logit space."""
    a = _softmax(z)
    c = _self_convolve(a)
    if top_k and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_max = c_top.max()
        log_sum = float(np.log(np.sum(np.exp((c_top - c_max) / beta))))
        return 2.0 * n * (c_max + beta * log_sum)
    c_max = float(c.max())
    log_sum = float(np.log(np.sum(np.exp((c - c_max) / beta))))
    return 2.0 * n * (c_max + beta * log_sum)


def _smoothed_gradient(z, beta, n, top_k):
    """Gradient of smoothed objective."""
    a = _softmax(z)
    c = _self_convolve(a)
    if top_k and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        w_top = np.exp((c_top - c_top.max()) / beta)
        w_top = w_top / w_top.sum()
        w = np.zeros(len(c))
        w[idx] = w_top
    else:
        c_max = float(c.max())
        w = np.exp((c - c_max) / beta)
        w = w / w.sum()
    grad_a = _cross_corr_fft(w, a)
    center = np.dot(grad_a, a)
    return 2.0 * n * a * (grad_a - center)


def _lbfgs_polish(a_init, deadline, top_k=0):
    """L-BFGS polish with logsumexp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z -= z.mean()

    best_a = a_init.copy()
    best_score = _score_normalized(a_init)

    c0 = _self_convolve(a_init)
    max_c0 = float(c0.max())
    betas = [b * max(max_c0, 1e-10) for b in [0.05, 0.02, 0.01, 0.005, 0.001]]
    per_beta = max(0.5, (deadline - time.monotonic()) / len(betas))

    for beta in betas:
        if time.monotonic() > deadline:
            break
        dl = min(deadline, time.monotonic() + per_beta)
        try:
            result = minimize(
                _smoothed_objective, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_smoothed_gradient,
                options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 20, 'maxfun': 1000, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                sc_new = _score_normalized(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_poly_profile(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i, c_val in enumerate(coeffs):
        a = a + c_val * t ** i
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_kaiser_profile(n, alpha):
    t = np.linspace(-1.0, 1.0, n)
    w = i0(alpha * np.sqrt(np.maximum(0.0, 1.0 - t ** 2))) / i0(alpha)
    return w / w.sum()


def _make_fejer_profile(n):
    k = np.arange(n)
    w = 1.0 - np.abs(k - (n - 1) / 2.0) / ((n - 1) / 2.0)
    w = np.maximum(w, 0.0)
    return w / w.sum()


def _make_hann_profile(n):
    t = np.linspace(0.0, 1.0, n)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    candidates = []

    # Parent
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_normalized(parent)
                candidates.append((sc, parent.copy(), len(parent), "parent"))
                try_update(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # Diverse profiles at multiple lengths
    for n_try in [4096, 8192]:
        if time.monotonic() > phase0_end:
            break
        profiles = []
        # Exponential
        for alpha in [0.0005, 0.001, 0.002, 0.005]:
            profiles.append(_make_exp_profile(n_try, alpha))
        # Linear
        for slope in [-1.5, -1.0, -0.5, 0.5, 1.0, 1.5]:
            profiles.append(_make_linear_profile(n_try, slope))
        # Polynomial
        for coeffs in [[0.5, -0.3], [-0.5, 0.3], [1.0, -0.8], [0.3, -0.5, 0.3]]:
            profiles.append(_make_poly_profile(n_try, coeffs))
        # Kaiser
        for alpha_k in [3.0, 5.0, 8.0, 12.0]:
            profiles.append(_make_kaiser_profile(n_try, alpha_k))
        # Fejer
        profiles.append(_make_fejer_profile(n_try))
        # Hann
        profiles.append(_make_hann_profile(n_try))

        for idx, p in enumerate(profiles):
            if time.monotonic() > phase0_end:
                break
            sc = _score_normalized(p)
            candidates.append((sc, p, n_try, f"p{idx}_n{n_try}"))
            try_update(p, sc, f"p{idx}_n{n_try}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0: {len(candidates)} candidates, best={best_score:.10f}")

    # === Phase 1: Main active-set optimization ===
    print("=== Phase 1: Active-set optimization ===")
    phase1_end = min(deadline, t_start + 700.0)
    n_top = 3
    per_candidate = (phase1_end - time.monotonic()) / max(1, n_top)

    seen_labels = set()
    count = 0
    for idx in range(min(n_top, len(candidates))):
        if time.monotonic() > phase1_end - 30:
            break
        sc, p, pn, label = candidates[idx]
        if count > 0 and sc > best_score * 1.005:
            break
        if label in seen_labels:
            continue
        seen_labels.add(label)

        remaining = phase1_end - time.monotonic()
        per_time = min(per_candidate, remaining / (n_top - count))
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  Opt {label} (n={pn}, sc={sc:.6f}), budget={per_time:.0f}s")

        p_best, sc_best = _active_set_optimize(p, dl, beta_start=0.01, beta_end=0.00005,
                                               verbose=True, momentum_decay=0.85)
        try_update(p_best, sc_best, f"opt_{label}")
        count += 1

    print(f"  Phase 1: best={best_score:.10f}")

    # === Phase 2: L-BFGS polish ===
    print("=== Phase 2: L-BFGS polish ===")
    phase2_end = min(deadline, t_start + 1000.0)

    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        if remaining > 15:
            dl = min(phase2_end, time.monotonic() + min(200.0, remaining))
            print(f"  L-BFGS polish, budget={min(200.0, remaining):.0f}s")
            p_best, sc_best = _lbfgs_polish(best_seq / best_seq.sum(), dl, top_k=0)
            try_update(p_best, sc_best, "lbfgs1")

            # Second pass with top_k
            if time.monotonic() < phase2_end - 10:
                dl2 = min(phase2_end, time.monotonic() + (phase2_end - time.monotonic() - 5))
                p_best2, sc_best2 = _lbfgs_polish(best_seq / best_seq.sum(), dl2, top_k=10)
                try_update(p_best2, sc_best2, "lbfgs2")

    print(f"  Phase 2: best={best_score:.10f}")

    # === Phase 3: Further active-set refinement ===
    print("=== Phase 3: Further refinement ===")
    phase3_end = min(deadline, t_start + 1350.0)

    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        if remaining > 20:
            dl = min(phase3_end, time.monotonic() + remaining)
            # Slight perturbation then refine
            p = best_seq / best_seq.sum()
            z = np.log(np.maximum(p, 1e-15))
            z -= z.mean()
            z += 0.015 * rng.normal(size=len(p))
            z -= z.mean()
            p_pert = _softmax(z)
            p_best, sc_best = _active_set_optimize(p_pert, dl, beta_start=0.005,
                                                   beta_end=0.00002, verbose=True,
                                                   momentum_decay=0.9, max_momentum=1000)
            try_update(p_best, sc_best, "refine_pert")

            # Direct refinement
            if time.monotonic() < phase3_end - 10:
                dl2 = min(phase3_end, time.monotonic() + (phase3_end - time.monotonic() - 5))
                p_best2, sc_best2 = _active_set_optimize(best_seq / best_seq.sum(), dl2,
                                                         beta_start=0.003, beta_end=0.00001,
                                                         verbose=False, momentum_decay=0.9)
                try_update(p_best2, sc_best2, "refine_direct")

    print(f"  Phase 3: best={best_score:.10f}")

    # === Phase 4: Multi-length exploration ===
    print("=== Phase 4: Multi-length ===")
    phase4_end = min(deadline, t_start + 1650.0)

    if best_seq is not None and time.monotonic() < phase4_end:
        n_curr = len(best_seq)
        target_ns = [2048, 4096, 8192, 16384]
        target_ns = [tn for tn in target_ns if tn != n_curr and tn <= 16384]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 40:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_normalized(p_interp)
            try_update(p_interp, sc_interp, f"interp_n{tn}")

            if time.monotonic() < phase4_end - 40:
                remaining = phase4_end - time.monotonic()
                per_opt = min(80.0, remaining * 0.35)
                if per_opt > 10:
                    dl = min(phase4_end - 10, time.monotonic() + per_opt)
                    p_best, sc_best = _active_set_optimize(p_interp, dl, beta_start=0.01,
                                                           beta_end=0.0005, verbose=False,
                                                           momentum_decay=0.85)
                    try_update(p_best, sc_best, f"opt_n{tn}")

    print(f"  Phase 4: best={best_score:.10f}")

    # === Phase 5: Final polish ===
    print("=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            dl = min(deadline - 3, time.monotonic() + min(50.0, remaining * 0.5))
            p_best, sc_best = _active_set_optimize(best_seq / best_seq.sum(), dl,
                                                   beta_start=0.002, beta_end=0.00001,
                                                   verbose=False, momentum_decay=0.92)
            try_update(p_best, sc_best, "final1")

        if time.monotonic() < deadline - 8:
            dl2 = min(deadline - 3, time.monotonic() + min(30.0, deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _lbfgs_polish(best_seq / best_seq.sum(), dl2, top_k=5)
            try_update(p_best2, sc_best2, "final_lbfgs")

    # === Return ===
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            c = _self_convolve(final_seq)
            final_score = 2.0 * len(final_seq) * float(c.max()) / (s * s)
            print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_kaiser_profile(4096, 6.0)
    print(f"  FALLBACK: Kaiser n=4096, score={_score_normalized(fallback):.10f}")
    return [float(x) for x in fallback]
