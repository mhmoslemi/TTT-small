"""
Multi-phase optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Active-set quadratic peak-shaving (solves ridge-regularized least squares on active peaks)
2. Spectral-domain reparameterization for final polish
3. Aggressive multi-length exploration with adaptive time allocation
4. Subgradient methods with exact max (no smoothing bias)
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_conv_at_k(a, k):
    """Compute gradient of conv[k] w.r.t. a. Result is 2*a[k-m] for valid m."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        # g[lo:hi+1] = 2 * a[k - hi : k - lo + 1][::-1]
        g[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _active_set_qp_step(a, top_k=20, lam=0.1):
    """
    Active-set quadratic peak reduction.
    Finds direction d that minimizes sum of (linearized peak values)^2 + lam*||d||^2
    subject to sum(d) = 0.
    Returns (direction, predicted_reduction).
    """
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    
    # Get top peaks
    if top_k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(c)[-top_k:]
    
    # Only use peaks that are significant (> 90% of max)
    threshold = cm * 0.90
    idx = idx[c[idx] >= threshold]
    
    if len(idx) == 0:
        return None, 0.0
    
    K = len(idx)
    
    # Build the system: (G^T G + lam*I) d = -G^T c_active + (mu/2)*1
    # where G's rows are gradients of each active peak
    # For efficiency, we build G implicitly
    
    # G^T G = sum_j g_j g_j^T where g_j = _grad_conv_at_k(a, k_j)
    # We'll form this matrix explicitly (n x n)
    # For n up to ~16384, this is ~100MB which might be tight
    # Instead, use iterative solver
    
    # First, compute G^T c (the right-hand side without the mu term)
    Gt_c = np.zeros(n)
    for k_val in idx:
        g_k = _grad_conv_at_k(a, int(k_val))
        Gt_c += g_k * c[int(k_val)]
    
    # Now solve (GtG + lam*I + mu*1*1^T) d = -Gt_c
    # Using the constraint sum(d) = 0, we can use the trick:
    # Add a large penalty for the constraint: lam*1*1^T with large lam
    # Or use the projection approach
    
    # Simpler approach: solve (GtG + lam*I) d = -Gt_c, then project: d = d - mean(d)*1
    # This is approximate but works well in practice
    
    # For the matrix-vector product with GtG:
    # (GtG) v = sum_j g_j (g_j . v)
    
    def matvec(v):
        result = np.zeros(n)
        for k_val in idx:
            g_k = _grad_conv_at_k(a, int(k_val))
            result += g_k * np.dot(g_k, v)
        result += lam * v
        return result
    
    # Use conjugate gradient (a few iterations)
    d = np.zeros(n)
    r = -Gt_c.copy()
    p_vec = r.copy()
    rs = np.dot(r, r)
    
    cg_iters = min(50, max(10, n // 100))
    for _ in range(cg_iters):
        Ap = matvec(p_vec)
        pAp = np.dot(p_vec, Ap)
        if pAp < 1e-30:
            break
        alpha_cg = rs / pAp
        d += alpha_cg * p_vec
        r -= alpha_cg * Ap
        rs_new = np.dot(r, r)
        if rs_new < 1e-30 * rs:
            break
        p_vec = r + (rs_new / rs) * p_vec
        rs = rs_new
    
    # Project onto sum(d) = 0
    d = d - d.mean()
    
    # Predicted reduction
    pred_reduction = 0.0
    for k_val in idx:
        g_k = _grad_conv_at_k(a, int(k_val))
        pred_reduction += (c[int(k_val)] + np.dot(g_k, d)) ** 2
    
    return d, pred_reduction


def _peak_shaving_iter(a_init, deadline, max_iter=500, top_k=30, lam_start=1.0, verbose=True):
    """
    Iterative peak shaving using active-set QP steps with line search.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a)
    
    lam = lam_start
    consecutive_fails = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        d, _ = _active_set_qp_step(a, top_k=top_k, lam=lam)
        if d is None:
            break
        
        # Line search
        best_alpha = 0.0
        best_trial_score = _score_norm_fn(a)
        alphas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
        
        for alpha in alphas:
            a_trial = a + alpha * d
            if np.any(a_trial < 0):
                # Clip and renormalize
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s * a.sum()
            sc = _score_norm_fn(a_trial)
            if np.isfinite(sc) and sc < best_trial_score:
                best_trial_score = sc
                best_alpha = alpha
        
        if best_alpha > 0:
            a_new = a + best_alpha * d
            if np.any(a_new < 0):
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-15:
                    break
                a_new = a_new / s * a.sum()
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                consecutive_fails = 0
                if verbose and it % 50 == 0:
                    print(f"    PS iter {it}: score={best_score:.10f}, alpha={best_alpha:.4f}")
            else:
                consecutive_fails += 1
                lam *= 2.0  # Increase regularization
        else:
            consecutive_fails += 1
            lam *= 1.5
        
        # Adapt top_k: increase if stuck
        if consecutive_fails > 10:
            top_k = min(top_k + 5, 80)
            lam = max(0.01, lam * 0.5)
            consecutive_fails = 0
        
        if consecutive_fails > 30:
            break
    
    return best_a, best_score

    sum_a = a_init.sum()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    lr = lr_start
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_fn(z) * sum_a  # scale back
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Get active set

            idx = np.arange(len(c))
        else:
            idx = np.argsort(c)[-top_k:]
        
        # Compute average gradient of active peaks
        g_avg = np.zeros(n)
        for k_val in idx:
            g_avg += _grad_conv_at_k(a, int(k_val))
        g_avg /= len(idx)
        
        # Transform to logit space: grad_z = grad_a / a
        grad_z = g_avg / np.maximum(a, 1e-15)
        
        # Normalize gradient
        gn = np.linalg.norm(grad_z)
        if gn > 1e-15:
            grad_z = grad_z / gn
        
        m = 0.9 * m + 0.1 * grad_z
        v = 0.999 * v + 0.001 * (grad_z * grad_z)
        mh = m / (1 - 0.9 ** t)
        vh = v / (1 - 0.999 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        a_new = _softmax_fn(z) * sum_a
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Decay learning rate
        if t % 200 == 0:
            lr *= 0.9
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: score={best_score:.10f}, lr={lr:.6f}")
    
    return best_a, best_score


def _z_objective(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_norm = c_top - c_top.max()
        log_sum = float(np.log(np.sum(np.exp(c_norm / beta))))
        return 2.0 * n * (c_top.max() + beta * log_sum)
    else:
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)


def _z_gradient(z, beta, n, top_k=0):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        w = np.zeros_like(c)
        c_top = c[idx]
        c_norm = (c_top - c_top.max()) / beta
        e = np.exp(c_norm)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w[idx[0]] = 1.0
        else:
            w[idx] = e / s
    else:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
    
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, max_wall_time=None, top_k=0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())
    
    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max(max_c0, 1e-10) for b in beta_starts]
    
    per_level = max_wall_time / len(betas)
    
    cur_z = z.copy()
    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        
        try:
            result = minimize(
                _z_objective, cur_z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 30, 'maxfun': 800}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                cur_z = result.x.copy()
                a_new = _softmax_fn(cur_z)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.05, top_k=0, n_max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.005 * max_c)
    
    adaptive_k = (top_k == 0)
    cur_k = 3 if adaptive_k else top_k
    
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_fn(z)
        use_k = cur_k if adaptive_k else top_k
        grad = _z_gradient(z, beta, n, use_k)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()
        a = _softmax_fn(z)
        
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if t % 100 == 0:
            beta = max(1e-14, beta * 0.9)
            if adaptive_k and t % 500 == 0:
                cur_k = min(cur_k + 3, 40)
        
        if t % 5000 == 0 and best_score < 1.52:
            lr = max(0.002, lr * 0.95)
    
    return best_a, best_score


def _spectral_polish(a_init, deadline, max_iter=1000, verbose=True):
    """
    Polish in frequency domain: adjust |FFT(a)|^2 to flatten the autocorrelation.
    Uses a simple iterative approach: compute autocorrelation, identify peaks,
    adjust the spectrum to reduce them.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a)
    
    N = _next_pow2(2 * n - 1)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Go to frequency domain
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1)
        
        # Compute autocorrelation (inverse FFT of power)
        autocorr = np.fft.irfft(power, N)[:2 * n - 1]
        ac_max = float(autocorr.max())
        
        # Identify peaks in autocorrelation
        num_peaks = min(20, 2 * n - 1)
        peak_idx = np.argsort(autocorr)[-num_peaks:]
        
        # For each peak, compute what frequency adjustment would reduce it
        # The peak at lag k corresponds to freq component exp(-2*pi*i*k*f/N)
        # We want to reduce power at frequencies that contribute to the peaks
        
        # Simple approach: reduce power where the "peak response" is highest
        # peak_response[f] = sum_k I(k is peak) * |exp(-2*pi*i*k*f)|^2 = sum_k I(k is peak) (always 1)
        # That's not helpful. Instead, use the actual contribution.
        
        # Better: compute the gradient of max(autocorr) w.r.t. power
        # autocorr[k] = sum_f power[f] * exp(2*pi*i*k*f/N) / N
        # d(autocorr[k])/d(power[f]) = exp(2*pi*i*k*f/N) / N
        # For real: = cos(2*pi*k*f/N) / N
        # We want to reduce autocorr at peaks, so reduce power where cos is positive at peaks
        
        freqs = np.arange(len(power))
        # Gradient of sum of peak autocorrelations w.r.t. power
        grad_power = np.zeros(len(power))
        for k_val in peak_idx:
            k_int = int(k_val)
            grad_power += np.cos(2.0 * np.pi * k_int * freqs / N) / N
        
        # Reduce power in direction of gradient (where it increases peaks)
        # Increase power where gradient is negative (reduces peaks)
        step_size = 0.1 * ac_max / (np.max(np.abs(grad_power)) + 1e-30)
        
        power_new = power - step_size * grad_power
        power_new = np.maximum(power_new, 0)
        
        # Reconstruct
        A_new = phase * np.sqrt(power_new)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s * a.sum()
        
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        
        if verbose and it % 200 == 0:
            print(f"    SP iter {it}: score={best_score:.10f}")
    
    return best_a, best_score


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_hann_profile(n, asym=0.0):
    """Hann-like window with optional asymmetry."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = a + asym * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_ramp_profile(n, p):
    """Power-law ramp: a_i = (i+1)^p, normalized."""
    a = (np.arange(n) + 1.0) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining:
    - L-BFGS with logsumexp continuation
    - Active-set QP peak shaving (novel)
    - Subgradient with exact max
    - Adam with adaptive top-k
    - Spectral-domain polish
    - Multi-length exploration
    """
    global height_sequence_1
    
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # === Phase 0: Initial candidates ===
    print("=== Phase 0: Initial candidates ===")
    candidates = []
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                candidates.append((sc, parent.copy(), len(parent), "parent"))
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")
    
    # Generate diverse profiles
    n_base = 4096
    profiles = []
    for alpha in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01]:
        profiles.append(_make_exp_profile(n_base, alpha))
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        profiles.append(_make_ramp_profile(n_base, p))
    for asym in [0.0, 0.2, 0.5, 1.0, -0.5, -1.0]:
        profiles.append(_make_hann_profile(n_base, asym))
    
    # If we have a parent at different length, interpolate
    if parent is not None and len(parent) != n_base:
        profiles.append(_interp_profile(parent, n_base))
    
    for idx, p in enumerate(profiles):
        if time.monotonic() > t_start + 15:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_base, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")
    
    # === Phase 1: L-BFGS on top candidate (quick) ===
    print("=== Phase 1: L-BFGS ===")
    phase1_end = min(deadline, t_start + 200.0)
    if time.monotonic() < phase1_end and best_seq is not None:
        remaining = phase1_end - time.monotonic()
        dl = min(phase1_end, time.monotonic() + min(180.0, remaining))
        n_best = len(best_seq)
        print(f"  L-BFGS on best (n={n_best}), budget={dl - time.monotonic():.1f}s")
        p_best, sc_best = _run_lbfgs(best_seq, dl, max_wall_time=dl - time.monotonic(), top_k=0)
        update_best(p_best, sc_best, "lbfgs1")
    
    print(f"  Phase 1 done: best={best_score:.10f}")
    
    # === Phase 2: Active-set QP peak shaving ===
    print("=== Phase 2: Peak shaving (QP) ===")
    phase2_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        remaining = phase2_end - time.monotonic()
        dl = min(phase2_end, time.monotonic() + min(300.0, remaining))
        n_best = len(best_seq)
        print(f"  Peak shaving on best (n={n_best}), budget={dl - time.monotonic():.1f}s")
        # Use normalized sequence (sum=1) for the QP
        a_norm = best_seq / best_seq.sum()
        p_best, sc_best = _peak_shaving_iter(a_norm, dl, max_iter=500, top_k=30, lam_start=0.5)
        update_best(p_best, sc_best, "peak_shave")
    
    # Quick Adam to clean up after peak shaving
    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        dl2 = min(phase2_end - 5, time.monotonic() + 60.0)
        p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.03, top_k=5, n_max_iter=15000)
        update_best(p_best2, sc_best2, "adam_after_ps")
    
    print(f"  Phase 2 done: best={best_score:.10f}")
    
    # === Phase 3: Subgradient with exact max ===
    print("=== Phase 3: Subgradient ===")
    phase3_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        dl = min(phase3_end, time.monotonic() + min(200.0, remaining))
        a_norm = best_seq / best_seq.sum()
        p_best, sc_best = _subgradient_step(a_norm, dl, max_iter=3000, top_k=10, lr_start=0.02)
        update_best(p_best, sc_best, "subgrad")
    
    print(f"  Phase 3 done: best={best_score:.10f}")
    
    # === Phase 4: More peak shaving rounds ===
    print("=== Phase 4: Extended peak shaving ===")
    phase4_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        remaining = phase4_end - time.monotonic()
        dl = min(phase4_end, time.monotonic() + min(250.0, remaining))
        a_norm = best_seq / best_seq.sum()
        p_best, sc_best = _peak_shaving_iter(a_norm, dl, max_iter=500, top_k=50, lam_start=0.1)
        update_best(p_best, sc_best, "peak_shave2")
        
        # Follow with Adam
        if time.monotonic() < phase4_end - 20:
            dl2 = min(phase4_end - 10, time.monotonic() + 80.0)
            p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.02, top_k=10, n_max_iter=20000)
            update_best(p_best2, sc_best2, "adam4")
    
    print(f"  Phase 4 done: best={best_score:.10f}")
    
    # === Phase 5: Spectral polish ===
    print("=== Phase 5: Spectral polish ===")
    phase5_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        remaining = phase5_end - time.monotonic()
        dl = min(phase5_end, time.monotonic() + min(150.0, remaining))
        a_norm = best_seq / best_seq.sum()
        p_best, sc_best = _spectral_polish(a_norm, dl, max_iter=500)
        update_best(p_best, sc_best, "spectral")
    
    print(f"  Phase 5 done: best={best_score:.10f}")
    
    # === Phase 6: Multi-length exploration ===
    print("=== Phase 6: Multi-length ===")
    phase6_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = [2048, 8192]
        if n_curr not in target_ns:
            target_ns.insert(0, n_curr if n_curr not in [2048, 8192] else 2048)
        
        for tn in [2048, 8192]:
            if time.monotonic() > phase6_end - 30:
                break
            if tn == n_curr:
                continue
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")
            
            remaining = phase6_end - time.monotonic()
            if remaining < 30:
                break
            per_opt = min(100.0, remaining * 0.6)
            dl = min(phase6_end - 10, time.monotonic() + per_opt)
            p_best, sc_best = _run_lbfgs(p_interp, dl, max_wall_time=per_opt, top_k=5)
            update_best(p_best, sc_best, f"lbfgs_n{tn}")
            
            if time.monotonic() < phase6_end - 20:
                dl2 = min(phase6_end - 10, time.monotonic() + 50.0)
                p_best2, sc_best2 = _peak_shaving_iter(p_best / p_best.sum(), dl2,
                                                       max_iter=200, top_k=40, lam_start=0.3)
                update_best(p_best2, sc_best2, f"ps_n{tn}")
    
    print(f"  Phase 6 done: best={best_score:.10f}")
    
    # === Phase 7: Multi-restart with noise ===
    print("=== Phase 7: Multi-restart ===")
    phase7_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase7_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = min(50.0, (phase7_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 15:
                break
            noise_levels = [0.03, 0.08, 0.15]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i] * rng.normal(size=n_best)
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase7_end - 5, time.monotonic() + per_restart)
            # Quick L-BFGS
            per_lbfgs = per_restart * 0.5
            p_m, sc_m = _run_lbfgs(p_start, min(phase7_end - 5, time.monotonic() + per_lbfgs),
                                    max_wall_time=per_lbfgs, top_k=5)
            update_best(p_m, sc_m, f"r{i}_lbfgs")
            if time.monotonic() < phase7_end - 10:
                p_a, sc_a = _peak_shaving_iter(p_m / p_m.sum(),
                                               min(phase7_end - 5, time.monotonic() + per_restart * 0.4),
                                               max_iter=100, top_k=30, lam_start=0.5)
                update_best(p_a, sc_a, f"r{i}_ps")
    
    print(f"  Phase 7 done: best={best_score:.10f}")
    
    # === Phase 8: Final intensive polish ===
    print("=== Phase 8: Final polish ===")
    if time.monotonic() < deadline and best_seq is not None:
        remaining = deadline - time.monotonic()
        
        # Adam with very small lr
        if remaining > 30:
            dl = min(deadline - 10, time.monotonic() + min(60.0, remaining * 0.4))
            p_best, sc_best = _adam_polish(best_seq, dl, lr=0.005, top_k=15, n_max_iter=30000)
            update_best(p_best, sc_best, "final_adam")
        
        # Peak shaving
        if time.monotonic() < deadline - 20:
            dl2 = min(deadline - 10, time.monotonic() + min(50.0, (deadline - time.monotonic()) * 0.4))
            a_norm = best_seq / best_seq.sum()
            p_best2, sc_best2 = _peak_shaving_iter(a_norm, dl2, max_iter=300, top_k=40, lam_start=0.1)
            update_best(p_best2, sc_best2, "final_ps")
        
        # One more L-BFGS
        if time.monotonic() < deadline - 10:
            dl3 = min(deadline - 5, time.monotonic() + min(30.0, deadline - time.monotonic() - 5))
            p_best3, sc_best3 = _run_lbfgs(best_seq, dl3, max_wall_time=max(5.0, dl3 - time.monotonic()), top_k=10)
            update_best(p_best3, sc_best3, "final_lbfgs")
    
    print(f"  Phase 8 done: best={best_score:.10f}")
    
    # === Final validation and return ===
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_exp_profile(4096, 0.002)
            final_score = _score_fn(final_seq)
        
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        
        # Quick validation
        try:
            seq_list = [float(x) for x in final_seq[:min(len(final_seq), 4096)]]
            if len(seq_list) <= 4096:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence (truncated to {len(seq_list)}): {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence check: {e}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_exp_profile(4096, 0.002)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
