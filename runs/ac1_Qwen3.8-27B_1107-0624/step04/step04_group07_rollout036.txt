"""
Optimization of step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses LP fixed-point, L-BFGS on smooth-max, and multi-scale escalation.
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    p = 1
    while p < n:
        p <<= 1
    return p

def _full_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _cross_corr(w, a):
    """Compute cross-correlation h[j] = sum_k w[k]*a[k-j] for j=0..n-1."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n-1:2*n-1]

def _smooth_max_obj_grad(z, n, beta):
    """
    Objective and gradient in log-space.
    z is the log-parameter (unconstrained).
    Returns (f, grad) where f = 2n * smooth_max(conv, beta) and grad = df/dz.
    """
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s
    c = _full_convolve(a)
    mval = c.max()
    x = (c - mval) / beta
    ex = np.exp(x)
    sm = ex.sum()
    w = ex / sm
    fval = 2.0 * n * (mval + beta * np.log(sm))
    
    # Gradient wrt a: g = 2 * cross_corr(w, a)
    g = 2.0 * _cross_corr(w, a)
    
    # Gradient wrt z: grad_z = a * (g - dot(a, g))
    dot = np.dot(a, g)
    grad_z = a * (g - dot)
    
    return fval, grad_z

def _build_conv_matrix(n):
    """Build the sparse matrix A such that A @ b = a * b (full convolution)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            # A[k, j] = a[k-j] for the LP (but we build it parameter-free here)
            vals.extend(np.ones(len(idx)))  # We'll multiply by a later
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_primal(a, M, n):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_end = min(n - 1, k)
        if j_start <= j_end:

            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())

def _lp_fixed_point(a, deadline, max_iters=200):
    """LP fixed-point iteration with line search."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        M = float(c.max())
        
        b = _lp_primal(a, M, n)
        if b is None:
            break
        s_b = b.sum()
        if s_b < 1e-10:
            break
        b = b / s_b
        
        # Line search
        best_t = 0.5
        best_sc = _score(a)
        for t_c in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = (1.0 - t_c) * a + t_c * b
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = t_c
        a = (1.0 - best_t) * a + best_t * b

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
        elif it > 20 and sc >= best_score * (1 - 1e-8):
            # Converged
            break
    
    return best_a, best_score

def _coord_descent(a, deadline, max_sweeps=10):

def _interp(a, n_new):

    a_new = a_new / a_new.sum()
    return a_new

"""
Multi-strategy optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Combines LP fixed-point, L-BFGS on smooth-max in log-space, adaptive
coordinate descent, and multi-scale escalation.

import numpy as np

def _next_pow2(n):
    if n <= 1:
        return 1
    p = 1
    while p < n:
        p <<= 1
    return p


def _full_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _cross_corr(w, a):
    """h[j] = sum_k w[k]*a[k-j] for j=0..n-1 (valid range)."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _smooth_max_and_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _obj_grad_lbfgs(z, n, beta):
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _full_convolve(a)
    sm, w = _smooth_max_and_weights(c, beta)
    fval = 2.0 * n * sm
    g = 2.0 * _cross_corr(w, a)
    dot = float(np.dot(a, g))
    grad_z = a * (g - dot)
    return fval, grad_z

        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # === Phase 1: Generate candidates ===
    print("Phase 1: Candidates...")
    candidates = []

    if parent is not None:
        sc_p = _score(parent)
        candidates.append((sc_p, parent.copy(), pn, 'parent'))
        for tn in [128, 256, 512, 1024, 2048]:
            if tn != pn:
                pi = _interp(parent, tn)
                sc_i = _score(pi)
                candidates.append((sc_i, pi, tn, f'par_{tn}'))

    profile_defs = [
        ('l1', 'linear', 1.0, 0), ('l1.5', 'linear', 1.5, 0),
        ('l2', 'linear', 2.0, 0), ('l2.5', 'linear', 2.5, 0),
        ('l3', 'linear', 3.0, 0), ('l4', 'linear', 4.0, 0),
        ('q1', 'quadratic', 1.0, 0), ('q2', 'quadratic', 2.0, 0),
        ('q3', 'quadratic', 3.0, 0),
        ('c1', 'cubic', 1.5, -0.5), ('c2', 'cubic', 2.0, -1.0),
        ('cos1', 'cosine', 0.5, 1), ('cos2', 'cosine', 0.4, 2),
        ('sin2', 'sine', 0.5, 2),
        ('e05', 'exp', 0.5, 0), ('e1', 'exp', 1.0, 0),
        ('U03', 'U', 0.3, 0), ('U05', 'U', 0.5, 0),
        ('p05', 'parabola', 0.5, 0),
        ('ed1', 'edge', 1.0, 0), ('ed15', 'edge', 1.5, 0),
    ]

    for ns in [128, 256, 512, 1024]:
        for (label, pt, p1, p2) in profile_defs:
            p = _make_profile(ns, pt, p1, p2)
            sc = _score(p)
            candidates.append((sc, p, ns, label))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")
    if candidates[0][0] < best_score:
        best_seq = candidates[0][1].copy()
        best_score = candidates[0][0]

    # === Phase 2: LP fixed-point ===
    print("Phase 2: LP fixed-point...")
    phase2_end = t_start + 450.0
    polished = []
    count = 0
    for (sc, p, n, label) in candidates:
        if time.monotonic() > phase2_end or count >= 10:
            break
        if n > 1024:
            continue
        alloc = max(25.0, min(70.0, 450.0 * (256.0 / float(n))))
        per_end = min(phase2_end, time.monotonic() + alloc)
        a_lp, sc_lp = _lp_fixed_point(p, per_end, max_iters=150)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  BEST: {best_score:.10f} n={n} [LP_{label}]")
        polished.append((sc_lp, a_lp, n, label + '_lp'))
        count += 1
    polished.sort(key=lambda x: x[0])
    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: L-BFGS refinement ===
    print("Phase 3: L-BFGS...")
    phase3_end = t_start + 1000.0
    for (sc, p, n, label) in polished[:5]:
        if time.monotonic() > phase3_end - 40:
            break
        per = min(150.0, phase3_end - time.monotonic() - 15)
        if per < 30:
            break
        a_lb, sc_lb = _lbfgs_smooth_max(p, time.monotonic() + per, n_beta=18)
        if sc_lb < best_score:
            best_seq = a_lb.copy()
            best_score = sc_lb
            print(f"  BEST: {best_score:.10f} n={n} [LB_{label}]")
        # LP after L-BFGS
        if n <= 1024 and time.monotonic() < phase3_end - 40:
            per2 = min(50.0, phase3_end - time.monotonic() - 15)
            if per2 > 20:
                a_lp2, sc_lp2 = _lp_fixed_point(a_lb, time.monotonic() + per2, max_iters=50)
                if sc_lp2 < best_score:
                    best_seq = a_lp2.copy()
                    best_score = sc_lp2
                    print(f"  BEST: {best_score:.10f} n={n} [LP_{label}_postLB]")

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale...")
    phase4_end = t_start + 1400.0
    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        else:
            target_ns = [4096]
        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score(p_int)
            print(f"  n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")
            if time.monotonic() < phase4_end - 60:
                per = min(100.0, phase4_end - time.monotonic() - 30)
                if per > 25:
                    a_e, sc_e = _lbfgs_smooth_max(p_int, time.monotonic() + per, n_beta=15)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [LB_esc]")
            if tn <= 1024 and time.monotonic() < phase4_end - 40:
                per2 = min(50.0, phase4_end - time.monotonic() - 20)
                if per2 > 15:
                    a_le, sc_le = _lp_fixed_point(p_int, time.monotonic() + per2, max_iters=30)
                    if sc_le < best_score:
                        best_seq = a_le.copy()
                        best_score = sc_le
                        print(f"  BEST: {best_score:.10f} n={tn} [LP_esc]")

    # === Phase 5: Random restarts + L-BFGS ===
    print("Phase 5: Restarts...")
    phase5_end = deadline - 100.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        n_restarts = 6
        per_r = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 15:
                break
            nsigma = 0.04 + 0.06 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + nsigma * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()
            a_r, sc_r = _lbfgs_smooth_max(a_s, time.monotonic() + per, n_beta=12)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i}]")
            if n_best <= 1024 and time.monotonic() < phase5_end - 20:
                per2 = min(25.0, phase5_end - time.monotonic() - 10)
                if per2 > 10:
                    a_rl, sc_rl = _lp_fixed_point(a_r, time.monotonic() + per2, max_iters=20)
                    if sc_rl < best_score:
                        best_seq = a_rl.copy()
                        best_score = sc_rl
                        print(f"  BEST: {best_score:.10f} n={n_best} [LP_r{i}]")

    # === Phase 6: Final polish ===
    print("Phase 6: Final polish...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 15:
            if best_seq.size <= 1024:
                a_fp, sc_fp = _lp_fixed_point(best_seq, time.monotonic() + per * 0.6, max_iters=40)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} [final_LP]")
            else:
                a_fp, sc_fp = _lbfgs_smooth_max(best_seq, time.monotonic() + per * 0.6, n_beta=10)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} [final_LB]")
            # Coordinate descent
            if time.monotonic() < deadline - 5:
                per_cd = deadline - time.monotonic() - 5
                a_cd, sc_cd = _coord_descent(best_seq, time.monotonic() + per_cd, max_sweeps=5)
                if sc_cd < best_score:
                    best_seq = a_cd.copy()
                    best_score = sc_cd
                    print(f"  BEST: {best_score:.10f} [final_CD]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0)
        exact_sc = _score(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")
        return [float(x) for x in final]

    fb = _make_profile(512, 'linear', 1.5, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
