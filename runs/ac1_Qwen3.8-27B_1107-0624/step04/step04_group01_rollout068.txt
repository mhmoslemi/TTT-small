"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy (improving on prior Adam-only approach):
  * Work in softmax log-space z -> p=softmax(z) (p sums to 1).
  * Objective = 2*n*softplus-like logsumexp(conv(p,p)/beta); gradient via FFT
    cross-correlation, exact subgradient as beta -> 0.
  * Phase A: diverse initializations at a few lengths (parent n, n/2, 2n).
  * Phase B: L-BFGS-B (quasi-Newton, uses curvature) with beta-continuation,
    which escapes plateaus better than Adam.
  * Phase C: multi-restart (Gaussian / Fourier / block perturbations), mixing
    L-BFGS-B and Adam, plus a direct projected "peak-shave" step on p.
  * Phase D: ultra-low-beta L-BFGS-B polish + light sparsification (snap small
    entries to 0) which can lower the dominant lag further.
  * Phase E: try a longer length by upsampling the best profile, then polish.
L-BFGS-B with curvature info + targeted peak-shaving + sparsify is the main
new ingredient over the previous pure-Adam pipeline.
"""
import time
import functools
import numpy as np
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_general(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_prob(p):
    n = len(p)
    c = _full_convolve(p, p)
    return float(2.0 * n * float(c.max()))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _sep_h(p, w):
    """h[j] = sum_k w[k]*p[k-j]  for j=0..n-1 (p out-of-range = 0)."""
    n = len(p)
    w_fl = w[::-1]
    pw = _full_convolve(p, w_fl)          # length 3n-2
    return pw[n - 1:2 * n - 1][::-1]      # h[0..n-1]


def _weights(c, beta):
    cm = float(c.max())
    sh = (c - cm) / beta
    e = np.exp(sh)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, cm, s
    w = e / s
    return w, cm, s


def _smoothed(z, beta):
    p = _softmax(z)
    n = len(p)
    c = _full_convolve(p, p)
    w, cm, s = _weights(c, beta)
    h = _sep_h(p, w)
    g_p = 4.0 * n * h
    W = float(np.dot(p, g_p))
    g_z = p * (g_p - W)
    obj = float(2.0 * n * (cm + beta * np.log(s)))
    return obj, g_z


def _obj_only(z, beta):
    return _smoothed(z, beta)[0]


def _jac_only(z, beta):
    return _smoothed(z, beta)[1]


def _lbfgsb_run(z_init, deadline, betas, max_iter):
    n = len(z_init)
    z = z_init.copy()
    for beta in betas:
        if time.monotonic() > deadline - 5.0:
            break
        obj_fn = functools.partial(_obj_only, beta=beta)
        jac_fn = functools.partial(_jac_only, beta=beta)
        try:
            res = minimize(obj_fn, z, jac=jac_fn, method="L-BFGS-B",
                           options={"maxiter": max_iter, "gtol": 1e-7,
                                    "maxfun": max_iter * 3,
                                    "ftol": 1e-11, "disp": False,
                                    "maxls": 40})
            if np.all(np.isfinite(res.x)):
                z = res.x
        except Exception:
            pass
        z = z - z.mean()
    return z


def _adam_run(z_init, deadline, betas, lr, max_iter):
    n = len(z_init)
    z = z_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur_lr = lr
    for beta in betas:
        if time.monotonic() > deadline - 5.0:
            break
        for _ in range(max_iter):
            if time.monotonic() > deadline - 5.0:
                break
            t += 1
            _, g = _smoothed(z, beta)
            if not np.all(np.isfinite(g)):
                return z
            m = b1 * m + (1.0 - b1) * g
            v = b2 * v + (1.0 - b2) * (g * g)
            mh = m / (1.0 - b1 ** t)
            vh = v / (1.0 - b2 ** t)
            up = cur_lr * mh / (np.sqrt(vh) + eps)
            u = float(np.linalg.norm(up))
            if u > 0.6:
                up = up * (0.6 / u)
            z = z - up
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if _ % 4000 == 0:
                cur_lr = max(0.001, cur_lr * 0.98)
    return z


def _direct_polish(p, deadline, max_iter=4000):
    """Projected (sum-preserving) subgradient steps directly on p, tracking
    the moving argmax lag; evaluates true score each step."""
    n = len(p)
    cur = p.copy()
    best = _score_prob(cur)
    step = 0.02
    for it in range(max_iter):
        if time.monotonic() > deadline - 5.0:
            break
        c = _full_convolve(cur, cur)
        k = int(np.argmax(c))
        idx = np.arange(n)
        off = k - idx
        h = np.zeros(n)
        valid = (off >= 0) & (off < n)
        h[valid] = cur[off[valid]]
        sub = h - np.mean(h)          # keep sum constant
        cand = cur - step * sub
        cand = np.clip(cand, 0.0, None)
        ss = cand.sum()
        if ss <= 0:
            break
        cand = cand / ss
        sc = _score_prob(cand)
        if sc < best:
            best = sc
            cur = cand
            step = min(0.05, step * 1.05)
        else:
            step = max(1e-4, step * 0.7)
    return cur, best


def _sparsify(p, frac):
    """Zero bottom-frac portion, renormalize; keep only if score improves."""
    p2 = p.copy()
    thr = np.quantile(p2, max(0.0, min(0.5, frac)))
    p2[p2 <= thr] = 0.0
    s = p2.sum()
    if s <= 1e-12:
        return p, _score_prob(p)
    p2 = p2 / s
    sc = _score_prob(p2)
    sc0 = _score_prob(p)
    if sc < sc0:
        return p2, sc
    return p, sc0


def _upsample_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    q = np.interp(x1, x0, p)
    q = np.maximum(q, 0.0)
    s = q.sum()
    if s <= 1e-12:
        return np.ones(n_new) / n_new
    return q / s

        t = np.linspace(0.0, 1.0, n)

        t = np.linspace(0.0, 1.0, n)

        z = np.log(a)

    z = np.log(np.maximum(p, 1e-15))

        per = max(6.0, rem / max(1, n_restarts))
        base = best_seq / best_seq.sum()
        nb = len(base)
        for i in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            z = _from_p_to_z(base)
            if i % 4 == 0:      # gaussian
                sg = 0.03 + 0.015 * (i % 3)
                z = z + sg * rng.standard_normal(nb)
            elif i % 4 == 1:    # fourier
                f = int(rng.integers(1, max(2, nb // 6)))
                t = np.arange(nb)
                amp = 0.05 * float(rng.uniform(0.5, 2.0))
                z = z + amp * np.cos(2 * np.pi * f * t / nb)
                z = z + amp * np.sin(2 * np.pi * f * t / nb)
            elif i % 4 == 2:    # block
                bs = int(rng.integers(max(2, nb // 80), max(3, nb // 20)))
                for _ in range(int(rng.integers(1, 4))):
                    st = int(rng.integers(0, max(1, nb - bs)))
                    z[st:st + bs] += 0.1 * rng.standard_normal(bs)
            else:               # anti-peak: push down current dominant lag region
                c = _full_convolve(base, base)
                k = int(np.argmax(c))
                mask = np.zeros(nb)
                idx = np.arange(nb)
                off = k - idx
                v = (off >= 0) & (off < nb)
                mask[v] = base[off[v]]
                z = z - 0.15 * np.log(np.maximum(mask, 1e-6)) + 0.0
            z = z - z.mean()
            p0 = _softmax(z)

            c0 = _full_convolve(p0, p0)
            mc0 = float(c0.max())
            betas = [mc0 * 0.02, mc0 * 0.01, mc0 * 0.005, mc0 * 0.002,
                     mc0 * 0.001]
            z = _adam_run(z, time.monotonic() + per * 0.45, betas,
                          lr=0.01, max_iter=20000)
            pA = _softmax(z)
            # L-BFGS continuation from there
            betas2 = [mc0 * 0.005, mc0 * 0.002, mc0 * 0.001, mc0 * 0.0005]
            z = _lbfgsb_run(z, time.monotonic() + per * 0.40, betas2, max_iter=1500)
            pB = _softmax(z)
            # peak-shave finish
            pc, _ = _direct_polish(pB / pB.sum(),
                                   time.monotonic() + per * 0.10, max_iter=800)
            for cand in (pA, pB, pc):
                try_accept(cand)
            # light sparsify
            sp, _ = _sparsify(pc / pc.sum(), frac=0.05 + 0.02 * (i % 3))
            try_accept(sp)
            print("  C[%d] n=%d score=%.6f t=%.1fs" % (i, nb,
                                                        _score_prob(pc),
                                                        time.monotonic() - t0))

    # ---- Phase D: ultra low-beta L-BFGS polish + sparsify sweep ----
    if best_seq is not None and time.monotonic() < deadline - 40:
        p = best_seq / best_seq.sum()
        z = _from_p_to_z(p)
        c = _full_convolve(p, p)
        mc = float(c.max())
        betas = [mc * 0.001, mc * 0.0005, mc * 0.0002, mc * 0.0001,
                 mc * 0.00005, mc * 0.00002]
        z = _lbfgsb_run(z, time.monotonic() + min(120.0,
                                     deadline - time.monotonic() - 30),
                        betas, max_iter=3000)
        p = _softmax(z)
        ps, _ = _direct_polish(p, time.monotonic() + 30.0, max_iter=1200)
        try_accept(ps)
        for frac in (0.0, 0.03, 0.06, 0.10):
            sp, _ = _sparsify(ps / ps.sum(), frac=frac)
            try_accept(sp)
        print("  D n=%d score=%.8f t=%.1fs" % (len(best_seq), best_score,
                                                time.monotonic() - t0))

    # ---- Phase E: try a longer length from best profile ----
    target = min(8192, len(best_seq) * 2) if best_seq is not None else 4096
    if best_seq is not None and target > len(best_seq) and \
            time.monotonic() < deadline - 40:
        pL = _upsample_profile(best_seq / best_seq.sum(), target)
        try_accept(pL)
        z = _from_p_to_z(pL)
        c = _full_convolve(pL, pL)
        mc = float(c.max())
        betas = [mc * 0.01, mc * 0.005, mc * 0.002, mc * 0.001]
        z = _lbfgsb_run(z, min(90.0, deadline - time.monotonic() - 25),
                        betas, max_iter=2000)
        pL = _softmax(z)
        ps, _ = _direct_polish(pL, time.monotonic() + 20.0, max_iter=800)
        try_accept(ps)
        print("  E n=%d score=%.8f t=%.1fs" % (target, best_score,
                                                time.monotonic() - t0))

    # ---- final validation ----
    if best_seq is not None:
        fin = np.clip(best_seq, 0.0, 1000.0)
        s = float(fin.sum())
        if s < 0.01:
            fin = np.ones(256) / 256
        sc = _score_general(fin)
        print("FINAL internal score=%.10f n=%d t=%.1fs" %
              (sc, len(fin), time.monotonic() - t0))
        try:
            ex = float(evaluate_sequence([float(x) for x in fin]))
            print("FINAL evaluator score=%.10f" % ex)
        except Exception as e:
            print("eval error: %s" % e)
        return [float(x) for x in fin]

    fb = np.ones(256) / 256
    print("FALLBACK uniform n=256 score=%.6f" % _score_general(fb))
    return [float(x) for x in fb]
