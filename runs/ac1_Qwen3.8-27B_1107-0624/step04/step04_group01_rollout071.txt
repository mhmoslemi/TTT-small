"""
LP-direction + smooth-gradient polishing for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.

The main loop repeatedly solves a small linear program whose variables are a
candidate direction on the simplex.  The LP is constrained by a dynamic subset
of the convolution coefficients, always including the center/edge lags, and
bounded variable magnitudes to keep the LP robust.  A line search based on the
exact convolution of the current sequence and the LP direction updates the
sequence monotonically.  The final phases use L-BFGS-B on a log-sum-exp
smoothed maximum and a plain Adam loop in log-space to polish the result.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog, minimize

_POLISH_STATE = {'n': 0, 'tau': 1.0}

_LINE_T_GRID = np.array(
    [1.0, 0.85, 0.75, 0.65, 0.55, 0.5, 0.45, 0.4, 0.35, 0.3,
     0.25, 0.22, 0.2, 0.18, 0.16, 0.14, 0.12, 0.1, 0.09, 0.08,
     0.07, 0.06, 0.055, 0.05, 0.045, 0.04, 0.035, 0.03, 0.025,
     0.02, 0.018, 0.015, 0.012, 0.01, 0.008, 0.006, 0.005,
     0.004, 0.003, 0.002, 0.0015, 0.001, 0.0007, 0.0005,
     0.0003, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
    dtype=np.float64
)


def _next_pow2(n):
    if n <= 2:
        return 2
    k = 1
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _score_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def _make_profile(n, kind):
    if n <= 0:
        n = 1
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    if kind == 'tapered':
        a = np.sin(np.pi * t) ** 0.5
    elif kind == 'sinusoidal':
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * t)
    elif kind == 'bimodal':
        a = 1.0 + 0.35 * np.cos(4.0 * np.pi * t)
    elif kind == 'rick':
        a = 1.0 / (1.0 + 20.0 * (t - 0.5) ** 2)
    else:
        a = np.ones(n, dtype=np.float64)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0 or n_new <= 0:
        return _make_profile(max(1, n_new), 'uniform')
    n0 = p.size
    if n0 == n_new:
        return _normalize_nonneg(p)
    x0 = np.linspace(0.0, 1.0, n0, dtype=np.float64)
    x1 = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    q = np.interp(x1, x0, p)
    return _normalize_nonneg(q)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(0, dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.full(z.size, 1.0 / z.size, dtype=np.float64)
    return e / s


def _perturb_log(a, sigma, rng):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    z = np.log(np.maximum(a, 1e-15))
    z = z - float(np.mean(z))
    if sigma > 0.0:
        z = z + sigma * rng.standard_normal(a.size)
    z = z - float(np.mean(z))
    return _softmax(z)


def _smooth_value(c, tau, n):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.inf
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return 1e100
    if tau <= 1e-15:
        val = 2.0 * n * cmax
        return val if np.isfinite(val) else 1e100
    x = (c - cmax) / tau
    np.clip(x, -745.0, 0.0, out=x)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s < 1.0:
        s = 1.0
    val = 2.0 * float(n) * (cmax + float(np.log(s)) / float(tau))
    return val if np.isfinite(val) else 1e100


def _softmax_from_c(c, tau):
    c = np.asarray(c, dtype=np.float64)
    cmax = float(np.max(c)) if c.size else 0.0
    if c.size == 0:
        return c, cmax
    if tau <= 1e-15:
        w = np.zeros_like(c, dtype=np.float64)
        tol = 1e-12 + 1e-8 * abs(cmax)
        idx = np.flatnonzero(np.abs(c - cmax) <= tol)
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
        w[idx] = 1.0 / float(idx.size)
        return w, cmax
    x = (c - cmax) / tau
    np.clip(x, -745.0, 0.0, out=x)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s < 1.0:
        s = 1.0
    return ex / s, cmax


def _smooth_grad(a, w, c, n):
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    wc = float(np.dot(w, c))
    grad = 4.0 * float(n) * a * (h - wc)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def _precompute_row_structure(n):
    full_m = 2 * n - 1
    total = n * n
    row_len = np.empty(full_m, dtype=np.int32)
    row_start = np.empty(full_m + 1, dtype=np.int32)
    col = np.empty(total, dtype=np.int32)
    fidx = np.empty(total, dtype=np.int32)
    pos = 0
    for k in range(full_m):
        l = min(k + 1, n, full_m - k)
        l = int(l)
        row_len[k] = l
        row_start[k] = pos
        start = max(0, k - n + 1)
        if l > 0:
            cols = np.arange(start, start + l, dtype=np.int32)
            col[pos:pos + l] = cols
            fidx[pos:pos + l] = k - cols
        pos += l
    row_start[full_m] = pos
    return {
        'n': n,
        'full_m': full_m,
        'row_len': row_len,
        'row_start': row_start,
        'col': col,
        'fidx': fidx,
    }


def _select_active_rows(c, n, M, top_frac, min_top=64):
    full_m = 2 * n - 1
    min_top = int(min(min_top, full_m))
    c = np.asarray(c, dtype=np.float64)
    if top_frac >= 1.0 or min_top >= full_m:
        active = np.arange(full_m, dtype=np.int64)
    else:
        k = int(np.clip(round(full_m * top_frac), min_top, full_m))
        if k >= full_m:
            active = np.arange(full_m, dtype=np.int64)
        else:
            part = np.argpartition(c, full_m - k)[full_m - k:]
            thr = float(np.min(c[part]))
            active = np.flatnonzero(c >= thr)
            max_top = int(min(full_m, max(min_top, round(full_m * max(0.25, 1.5 * top_frac)))))
            if active.size > max_top:
                k2 = max_top
                if k2 >= full_m:
                    active = np.arange(full_m, dtype=np.int64)
                else:
                    active = np.argpartition(c, full_m - k2)[full_m - k2:]
    always = np.array([0, n - 1, full_m - 1], dtype=np.int64)
    active = np.unique(np.concatenate((active, always)))
    return active


def _build_active_arrays(struct, active_rows):
    n = struct['n']
    full_m = struct['full_m']
    active_rows = np.asarray(active_rows, dtype=np.int64)
    if active_rows.size == full_m:
        return struct['row_start'], struct['col'], struct['fidx']
    lens = struct['row_len'][active_rows].astype(np.int64)
    starts = struct['row_start'][active_rows].astype(np.int64)
    row_ptr = np.empty(active_rows.size + 1, dtype=np.int32)
    row_ptr[0] = 0
    row_ptr[1:] = np.cumsum(lens, dtype=np.int64).astype(np.int32)
    total = int(row_ptr[-1]) if total > 0 else 0
    col = np.empty(total, dtype=np.int32)
    fidx = np.empty(total, dtype=np.int32)
    col_all = struct['col']
    fidx_all = struct['fidx']
    pos = 0
    for i in range(active_rows.size):
        l = int(lens[i])
        if l:
            s = int(starts[i])
            e = s + l
            col[pos:pos + l] = col_all[s:e]
            fidx[pos:pos + l] = fidx_all[s:e]
            pos += l
    return row_ptr, col, fidx


def _line_search_arrays(cf, fc, gg, M_cur):
    cf = np.asarray(cf, dtype=np.float64)
    fc = np.asarray(fc, dtype=np.float64)
    gg = np.asarray(gg, dtype=np.float64)
    if cf.size == 0:
        return 0.0, M_cur
    T = _LINE_T_GRID[:, None]
    vals = (1.0 - T) * (1.0 - T) * cf[None, :]
    vals += (2.0 * T * (1.0 - T)) * fc[None, :]
    vals += (T * T) * gg[None, :]
    vals = np.maximum(vals, 0.0)
    row = vals.max(axis=1)
    idx = int(np.argmin(row))
    best_t = float(_LINE_T_GRID[idx])
    best_val = float(row[idx])
    if best_t <= 1e-14 or best_val >= M_cur * (1.0 - 1e-12):
        return 0.0, M_cur
    lo = 0.0 if best_t < 0.02 else max(0.0, best_t * 0.5)
    hi = min(1.0, max(best_t * 2.0, best_t + 5e-3))
    if hi <= lo:
        hi = min(1.0, best_t + 0.01)
    Ts = np.linspace(lo, hi, 200, dtype=np.float64)[:, None]
    vals2 = (1.0 - Ts) * (1.0 - Ts) * cf[None, :]
    vals2 += (2.0 * Ts * (1.0 - Ts)) * fc[None, :]
    vals2 += (Ts * Ts) * gg[None, :]
    vals2 = np.maximum(vals2, 0.0)
    row2 = vals2.max(axis=1)
    idx2 = int(np.argmin(row2))
    t2 = float(Ts[idx2, 0])
    v2 = float(row2[idx2])
    if v2 < best_val:
        best_t, best_val = t2, v2
    if best_val < M_cur * (1.0 - 1e-10) and best_t > 1e-13:
        return best_t, best_val
    return 0.0, M_cur


def _lp_optimize(f_init, deadline, rng=None, verbose=True):
    f = _normalize_nonneg(f_init)
    n = int(f.size)
    if n <= 0:
        return f, np.inf

    struct = _precompute_row_structure(n)
    c = _full_convolve(f, f)
    M = float(np.max(c))

    best_f = f.copy()
    best_M = M

    obj = -np.ones(n, dtype=np.float64)

    cached_active = None
    cached_row_ptr = None
    cached_col = None
    cached_fidx = None

    top_frac = 0.03
    it = 0
    no_best = 0

    while time.monotonic() < deadline:
        if no_best > 130:
            break
        if time.monotonic() > deadline - 3.0:
            break

        if no_best == 3:
            top_frac = 0.05
        elif no_best == 8:
            top_frac = 0.08
        elif no_best == 15:
            top_frac = 0.12
        elif no_best == 25:
            top_frac = 0.18
        elif no_best == 40:
            top_frac = 0.30
        elif no_best == 60:
            top_frac = 0.50
        elif no_best >= 80:
            top_frac = 1.0

        active_rows = _select_active_rows(c, n, M, top_frac)
        if active_rows.size == 0:
            active_rows = np.array([0, n - 1, 2 * n - 2], dtype=np.int64)

        b_ub_active = np.full(active_rows.size, M + 1e-12, dtype=np.float64)

        if (cached_active is None or
                cached_active.size != active_rows.size or
                not np.array_equal(cached_active, active_rows)):
            cached_row_ptr, cached_col, cached_fidx = _build_active_arrays(struct, active_rows)
            cached_active = active_rows.copy()

        data = f[cached_fidx]
        try:
            A = sp.csr_matrix((data, cached_col, cached_row_ptr),
                              shape=(active_rows.size, n))
            fmax = float(np.max(f)) if n else 0.0
            ub = min(1.0, max(0.5, 1.05 * fmax))
            bounds = np.column_stack((np.zeros(n, dtype=np.float64), np.full(n, ub, dtype=np.float64)))
            tl = max(1.0, min(8.0, deadline - time.monotonic() - 2.0))
            res = linprog(
                obj,
                A_ub=A,
                b_ub=b_ub_active,
                bounds=bounds,
                method='highs',
                options={'time_limit': float(tl)},
            )
        except Exception:
            break

        if res is None or not res.success or res.x is None:
            no_best += 1
            if verbose and it % 20 == 0:
                print(f"    LP solve failed it={it} best_score={2.0 * n * best_M:.10f}")
            it += 1
            continue

        b = np.asarray(res.x, dtype=np.float64)
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        bsum = float(b.sum())
        if bsum < 1e-15:
            no_best += 1
            it += 1
            continue

        g = b / bsum
        old_M = M
        fc = _full_convolve(f, g)
        gg = _full_convolve(g, g)
        t, val = _line_search_arrays(c, fc, gg, old_M)

        if t > 1e-14 and val < old_M * (1.0 - 1e-10):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            s = float(f_new.sum())
            if s < 1e-15 or not np.isfinite(s):
                break
            f_new = f_new / s
            c_new = _full_convolve(f_new, f_new)
            M_new = float(np.max(c_new))
            if np.isfinite(M_new) and M_new < old_M * (1.0 - 1e-10):
                f = f_new
                c = c_new
                M = M_new
                if M < best_M * (1.0 - 1e-12):
                    best_M = M
                    best_f = f.copy()
                    no_best = 0
                else:
                    no_best += 1
            else:
                no_best += 1
        else:
            no_best += 1

        if verbose and it % 100 == 0:
            print(f"    LP it={it} best_score={2.0 * n * best_M:.10f} "
                  f"no_best={no_best} top_frac={top_frac:.3f}")
        it += 1

    return best_f, best_M


def _set_polish_state(n, tau):
    _POLISH_STATE['n'] = int(n)
    _POLISH_STATE['tau'] = float(tau)


def _polish_value(z):
    n = int(_POLISH_STATE.get('n', 0))
    tau = float(_POLISH_STATE.get('tau', 1.0))
    z = np.asarray(z, dtype=np.float64)
    if n <= 0 or z.size != n:
        return 1e100
    a = _softmax(z)
    c = _full_convolve(a, a)
    return _smooth_value(c, tau, n)


def _polish_grad(z):
    n = int(_POLISH_STATE.get('n', 0))
    tau = float(_POLISH_STATE.get('tau', 1.0))
    z = np.asarray(z, dtype=np.float64)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if z.size != n:
        return np.zeros(n, dtype=np.float64)
    a = _softmax(z)
    c = _full_convolve(a, a)
    w, _ = _softmax_from_c(c, tau)
    return _smooth_grad(a, w, c, n)


def _lbfgs_polish(f_init, deadline, verbose=True):
    f = _normalize_nonneg(f_init)
    n = int(f.size)
    if n <= 0:
        return f, np.inf

    best_f = f.copy()
    best_score = _score_sequence(f)

    z = np.log(np.maximum(f, 1e-15))
    z = z - float(np.mean(z))

    _set_polish_state(n, best_score / (2.0 * n) if n > 0 else 1.0)

    bounds_list = [(-40.0, 40.0) for _ in range(n)]

    phases = [
        (0.20, 1400),
        (0.10, 1400),
        (0.05, 1100),
        (0.02, 900),
        (0.01, 700),
        (0.005, 550),
        (0.002, 400),
        (0.001, 300),
        (0.0005, 200),
    ]

    for gamma, maxfun in phases:
        if time.monotonic() > deadline - 5.0:
            break
        m_est = best_score / (2.0 * float(n)) if n > 0 and np.isfinite(best_score) else 1e-8
        tau = max(1e-12, m_est * gamma)
        _set_polish_state(n, tau)
        try:
            res = minimize(
                _polish_value,
                z,
                jac=_polish_grad,
                method='L-BFGS-B',
                bounds=bounds_list,
                options={
                    'maxiter': 100000,
                    'maxfun': int(maxfun),
                    'gtol': 1e-12,
                    'ftol': 1e-12,
                    'disp': False,
                },
            )
            if res is not None and hasattr(res, 'x') and res.x is not None:
                x = np.asarray(res.x, dtype=np.float64)
                if x.size == n and np.all(np.isfinite(x)):
                    a = _normalize_nonneg(_softmax(x))
                    sc = _score_sequence(a)
                    if np.isfinite(sc):
                        if sc <= best_score + 0.02:
                            z = x
                        if sc < best_score * (1.0 - 1e-12):
                            best_score = sc
                            best_f = a.copy()
                            if verbose:
                                print(f"    L-BFGS gamma={gamma:.5f} score={best_score:.10f}")
        except Exception:
            pass

    a = _normalize_nonneg(_softmax(z))
    sc = _score_sequence(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_f = a.copy()

    return best_f, best_score


def _adam_polish(f_init, deadline, rng=None, verbose=True):
    f = _normalize_nonneg(f_init)
    n = int(f.size)
    if n <= 0 or time.monotonic() > deadline:
        return f, _score_sequence(f)

    z = np.log(np.maximum(f, 1e-15))
    z = z - float(np.mean(z))

    best_f = f.copy()
    c = _full_convolve(f, f)
    best_M = float(np.max(c))

    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)

    t = 0
    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    lr = 0.018
    gamma = 0.05

    no_best = 0
    pert = 0
    while time.monotonic() < deadline - 3.0:
        a = _softmax(z)
        c = _full_convolve(a, a)
        M = float(np.max(c))
        if not np.isfinite(M):
            break

        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_f = _normalize_nonneg(a)
            no_best = 0
        else:
            no_best += 1

        tau = max(1e-12, M * gamma)
        w, _ = _softmax_from_c(c, tau)
        grad = _smooth_grad(a, w, c, n)
        if not np.all(np.isfinite(grad)):
            break

        t += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        step = lr * mh / (np.sqrt(vh) + eps)

        norm = float(np.linalg.norm(step))
        if np.isfinite(norm) and norm > 1.0:
            step = step * (1.0 / norm)

        z = z - step
        z = np.clip(z, -45.0, 45.0)
        z = z - float(np.mean(z))

        gamma = max(0.0007, gamma * 0.995)
        if no_best > 300:
            gamma = max(0.0002, gamma * 0.6)
            lr = max(0.002, lr * 0.85)
            no_best = 0
            if rng is not None and pert < 3 and time.monotonic() < deadline - 120.0:
                z = z + 0.015 * (pert + 1) * rng.standard_normal(n)
                z = np.clip(z, -45.0, 45.0)
                z = z - float(np.mean(z))
                pert += 1

        if t % 500 == 0 and verbose:
            a_check = _normalize_nonneg(a)
            sc = _score_sequence(a_check)
            print(f"    Adam t={t} score={sc:.10f} gamma={gamma:.6f}")

    a = _normalize_nonneg(_softmax(z))
    sc = _score_sequence(a)
    if np.isfinite(sc) and sc < 2.0 * n * best_M:
        return a, sc
    return best_f, 2.0 * n * best_M


def propose_candidate(seed=42, budget_s=1800):
    """Search for a non-negative sequence minimizing evaluate_sequence."""
    t_start = time.monotonic()
    budget_s = max(60.0, float(budget_s))
    hard_deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng((int(seed) % 1000000) + 1)

    parent_raw = globals().get('height_sequence_1', None)
    parent_arr = None
    if parent_raw is not None:
        try:
            arr = np.asarray(parent_raw, dtype=np.float64).ravel()
            if arr.size > 0 and np.any(np.isfinite(arr)):
                parent_arr = arr
        except Exception:
            parent_arr = None

    if parent_arr is not None:
        n_parent = int(parent_arr.size)
        if 256 <= n_parent <= 2560:
            n_main = n_parent
            base = _normalize_nonneg(parent_arr)
        elif n_parent < 256:
            n_main = 2048
            base = _interp_profile(parent_arr, n_main)
        else:
            n_main = 2560
            base = _interp_profile(parent_arr, n_main)
    else:
        n_main = 2048
        base = _make_profile(n_main, 'uniform')

    best_seq = base
    best_score = _score_sequence(best_seq)

    print(f"Initial n={n_main} score={best_score:.10f}")

    # Quick diverse initializations at the main length.
    for kind in ('uniform', 'tapered', 'sinusoidal', 'bimodal'):
        p = _make_profile(n_main, kind)
        sc = _score_sequence(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p
            best_score = sc
            print(f"  init {kind}: {best_score:.10f}")

    for sigma in (0.01, 0.02, 0.04):
        p = _perturb_log(base, sigma, rng)
        sc = _score_sequence(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p
            best_score = sc
            print(f"  perturb {sigma}: {best_score:.10f}")

    if not np.isfinite(best_score) or best_score > 2.0:
        best_seq = _make_profile(256, 'uniform')
        best_score = _score_sequence(best_seq)

    # Main LP-based monotonically-improving phase.
    lp_deadline = max(
        t_start + 60.0,
        min(hard_deadline - 280.0, t_start + 0.75 * budget_s)
    )
    print(f"Running LP phase, deadline in {lp_deadline - t_start:.1f}s")
    lp_f, lp_m = _lp_optimize(best_seq, lp_deadline, rng, verbose=True)
    lp_score = _score_sequence(lp_f)
    if np.isfinite(lp_score) and lp_score < best_score:
        best_seq = lp_f
        best_score = lp_score

    # Optional second LP restart from a small perturbation if time remains.
    if time.monotonic() < hard_deadline - 420.0:
        restart = _perturb_log(best_seq, 0.012, rng)
        restart_score = _score_sequence(restart)
        if np.isfinite(restart_score):
            extra_deadline = min(hard_deadline - 420.0, time.monotonic() + 180.0)
            p2, _ = _lp_optimize(restart, extra_deadline, rng, verbose=False)
            sc2 = _score_sequence(p2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_seq = p2
                best_score = sc2
                print(f"  LP restart improved to {best_score:.10f}")

    best_seq = _normalize_nonneg(best_seq)
    best_score = _score_sequence(best_seq)
    print(f"After LP: {best_score:.10f}")

    # L-BFGS continuation polish on smoothed max convolution.
    lbf_deadline = max(time.monotonic() + 10.0, hard_deadline - 90.0)
    best_seq, best_score = _lbfgs_polish(best_seq, lbf_deadline, verbose=True)
    print(f"After L-BFGS: {best_score:.10f}")

    # Final Adam polish.
    if time.monotonic() < hard_deadline - 8.0:
        best_seq, best_score = _adam_polish(best_seq, hard_deadline - 4.0, rng, verbose=True)
        print(f"After Adam: {best_score:.10f}")

    final = _normalize_nonneg(best_seq)
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)

    s = float(final.sum())
    if not np.isfinite(s) or s < 0.01:
        final = _make_profile(256, 'uniform')
        final = np.clip(final, 0.0, 1000.0)
    else:
        final = final / s

    try:
        ev = evaluate_sequence([float(x) for x in final])
        print(f"Final evaluator score: {ev}")
        print(f"Final n={final.size}, time={time.monotonic() - t_start:.1f}s")
    except Exception as ex:
        print(f"Final evaluator error: {ex}")

    return [float(x) for x in final]
