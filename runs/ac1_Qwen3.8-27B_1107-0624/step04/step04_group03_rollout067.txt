"""
Active-set + smooth continuation optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Key innovation: exact KKT linear-system solution for the active set of near-maximal autocorrelation lags,
combined with log-sum-exp smooth continuation for robust convergence to the correct active set.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    momentum = np.zeros(n)
    mom_coeff = 0.9
    lr = lr_init
    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        active = c >= cm * 0.9995
        if not np.any(active):
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / float(np.sum(active))
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 2.0 * n * (h - W)
        momentum = mom_coeff * momentum - lr * grad
        a = a + momentum
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        if it % 200 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            else:
                lr = lr * 0.85
    return best_a, best_score


def _active_set_solve(S, n):
    S_bool = np.zeros(2 * n - 1, dtype=bool)
    S_valid = S[(S >= 0) & (S < 2 * n - 1)]
    S_bool[S_valid] = True
    if not np.any(S_bool):
        return np.ones(n) / n, False
    A = S_bool[np.arange(n)[:, None] + np.arange(n)[None, :]].astype(np.float64)
    free = np.arange(n)
    fixed = np.zeros(n)
    a_f = None
    for inner in range(30):
        if len(free) < 2:
            break
        A_ff = A[np.ix_(free, free)]
        ones_vec = np.ones(len(free))
        try:
            x = np.linalg.solve(A_ff + 1e-12 * np.eye(len(free)), ones_vec)
            mu_prime = 1.0 / float(np.sum(x))
            a_f = mu_prime * x
        except np.linalg.LinAlgError:
            break
        if a_f is None:
            break
        if np.all(a_f >= -1e-9):
            result = fixed.copy()
            result[free] = np.maximum(a_f, 0)
            s = result.sum()
            if s > 0:
                result = result / s
            return result, True
        min_pos = int(np.argmin(a_f))
        fixed[free[min_pos]] = 0.0
        free = np.delete(free, min_pos)
    result = fixed.copy()
    if a_f is not None and len(free) > 0:
        result[free] = np.maximum(a_f, 0)
    s = result.sum()
    if s > 0:
        result = result / s
    return result, False
