"""
Optimized step function search using intensive LP best-response iteration
with aggressive line search, multi-scale escalation, and subgradient polishing.
Targets score < 1.5027 by exploiting the LP fixed-point structure of the
self-convolution minimization problem.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _score(p):
    """Compute score for a probability distribution (sum=1)."""
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    """Build the sparse convolution matrix A where (A @ b)[k] = (a*b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_best_response(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iters=200, verbose=True):
    """Iterative LP best-response with line search."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)

        if s_b < 1e-12:

        best_t = 0.0

def _subgradient_refine(a, deadline, lr_init=0.01, verbose=False):
    """Subgradient descent on the max of the convolution."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_score = _score(a)
    best_a = a.copy()
    lr = lr_init
    stall = 0
    
    c = _full_convolve(a)
    
    while time.monotonic() < deadline:
        k_star = int(np.argmax(c))
        # Gradient of (a*a)[k_star] w.r.t. a[i] is a[k_star - i]
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
        
        # Projected gradient step on the simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = _score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # still accept for exploration
            stall += 1
            if stall > 100:
                lr *= 0.9
                stall = 0
                if lr < 1e-6:
                    break
        
        c = _full_convolve(a)
        
        if verbose and (stall == 0):
            print(f"    Subgrad: score={sc:.10f}, k*={k_star}")
    
    return best_a, best_score


def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.94):
    """Adam optimization in logit space with smooth-max objective."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = _full_convolve(a)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)
    
    best_score = float(2.0 * n * maxc)
    best_a = a.copy()
    best_z = z.copy()
    no_imp = 0
    
    while time.monotonic() < deadline:
        # Smooth max weights
        x = (c - maxc) / beta
        ex = np.exp(np.minimum(x, 0.0))
        s = ex.sum()
        if s <= 0 or not np.isfinite(s):
            w = np.zeros(n)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s
        
        # Cross-correlation: grad = w * a (convolved appropriately)
        # (softmax grad) g_i = sum_k w_k * a[k-i] (with bounds)
        # This is the cross-correlation of w and a
        n_full = 2 * n - 1
        N = _next_pow2(n_full)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)
        g = h[n-1:2*n-1]  # cross-correlation of w and a, length n
        
        # Logit gradient
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1*m + (1-b1)*gz
        v = b2*v + (1-b2)*(gz*gz)
        mh = m / (1 - b1**t_adam)
        vh = v / (1 - b2**t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        e = np.exp(z)
        a = e / e.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        c = _full_convolve(a)
        maxc = float(c.max())
        sc = float(2.0 * n * maxc)
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.9)
        
        if t_adam % 50 == 0:
            beta = max(5e-5, beta * beta_decay)
        
        if no_imp > 1000:
            # Restart with noise
            z = best_z + 0.05 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            a = e / e.sum()
            c = _full_convolve(a)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = 0.05
            no_imp = 0
    
    return best_a, best_score


def _make_uniform(n):
    return np.ones(n) / n


def _make_profile(n, kind, **params):
    t = np.linspace(0, 1, n)
    if kind == 'linear':
        eta = params.get('eta', 1.0)
        a = 1.0 + eta * (t - 0.5)
    elif kind == 'quadratic':
        a1 = params.get('a1', 1.0)
        a2 = params.get('a2', -0.5)
        a = 1.0 + a1 * t + a2 * t * t
    elif kind == 'cubic':
        a1 = params.get('a1', 1.0)
        a3 = params.get('a3', -0.5)
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a3 * u**3
    elif kind == 'cosine':
        freq = params.get('freq', 1)
        amp = params.get('amp', 0.5)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    elif kind == 'gaussian':
        center = params.get('center', 0.5)
        width = params.get('width', 0.2)
        a = np.exp(-0.5 * ((t - center) / width)**2)
    elif kind == 'two_bump':
        c1 = params.get('c1', 0.3)
        c2 = params.get('c2', 0.7)
        s = params.get('s', 0.15)
        a = np.exp(-0.5 * ((t-c1)/s)**2) + np.exp(-0.5 * ((t-c2)/s)**2)
    elif kind == 'u_shaped':
        amp = params.get('amp', 1.0)
        a = 1.0 + amp * (2*t - 1)**2
    elif kind == 'w_shaped':
        amp = params.get('amp', 0.8)
        a = 1.0 + amp * np.cos(4.0 * np.pi * t)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_to_length(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
