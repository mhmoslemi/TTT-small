    # Easiest: h[j] = (a * W_rev)[j + (2n-2) - 0]... no.

    # h[j] = sum_{m=0}^{n-1} a[m] * W[m+j]
    # = sum_{m=0}^{n-1} a[m] * W[j+m]
    # This is the cross-correlation of a and W at lag -j, or equivalently,
    # the convolution of a with reversed W, evaluated at position (2n-2) - j... 
    
    # Let me define: conv(a, W_rev)[p] = sum_i a[i] * W_rev[p-i] = sum_i a[i] * W[(2n-2)-(p-i)]
    # We want sum_i a[i] * W[i+j] = sum_i a[i] * W_rev[(2n-2)-(i+j)]
    # = conv(a, W_rev)[(2n-2)-(i+j) + ... ]
    
    # I'm overcomplicating this. Let me use a different approach.
    
    # Since W has length 2n-1 and a has length n, I'll pad a to length 2n-1 and compute:
    # h[j] = sum_m a[m] * W[m+j] for j = 0, ..., n-1
    # This is the same as: take the first n elements of the cross-correlation.
    
    # Cross-correlation via FFT:
    # Let N >= 2*max(n, 2n-1) = 4n-2, use N = next_pow2(4n)
    # A = fft(a, N), C = fft(W, N)
    # cross = ifft(A * conj(C), N)  # gives cross-correlation
    # cross[j] = sum_m a[m] * W[m-j] (with circular wrap)
    # We want sum_m a[m] * W[m+j] = cross[-j] = cross[N-j]
    # So h = cross[N-(n-1):N] reversed... 
    # Actually: cross[N-j] = sum_m a[m] * W[m-(N-j)] = sum_m a[m] * W[m+j] (for j < n, with W zero-padded)
    # So h[j] = cross[N-j] for j = 0, ..., n-1
    # i.e., h = cross[N-n:N][::-1] ... let me just compute it.
    
    # You know what, let me just use the existing approach from the previous program:
    # a_rev = a[::-1]
    # w_conv = _full_convolve(w, a_rev)  # w has length 2n-1, a_rev has length n
    # h = w_conv[n-1:2*n-1]  # this gives n elements
    # This computes h[j] = sum_k w[k] * a_rev[k-j] = sum_k w[k] * a[n-1-k+j]
    # Let m = n-1-k+j, so k = n-1-m+j: h[j] = sum_m w[n-1-m+j] * a[m]
    # = sum_m a[m] * W[j + (n-1-m)] ... hmm, this isn't quite what I want either.
    
    # Let me just verify: the previous program used this approach and it worked.
    # The key formula is:
    # grad_z = 4*n*a*(h - W_mean) where h = w_conv[n-1:2*n-1]
    # and W_mean = dot(w, c)
    # Let me just use this.
    
    W_mean = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    
    # For the softmax parameterization with normalization (sum a = 1):
    # grad_z = a * (g - a * dot(a, g)) where g = free gradient = 2*n*2*h = 4*n*h
    # But we also have the normalization constraint, so:
    # grad_z = 4*n*a*(h - W_mean)
    
    grad = 4.0 * n * a * (h - W_mean)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _objective_smooth(z, beta, topk_frac=0.05):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    # Add top-k penalty
    n_c = len(c)
    k = max(1, int(topk_frac * n_c))
    topk_vals = np.sort(c)[::-1][:k]
    topk_penalty = topk_vals.mean()
    return 2.0 * n * (cm + beta * log_sum) + 2.0 * n * 0.3 * topk_penalty


def _run_lbfgs(a_init, deadline, max_wall_time=None, topk_frac=0.05):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))

    # More aggressive beta schedule
    betas = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005,
             0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6]
    betas = [b * max_c0 for b in betas]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n, topk_frac),
                method='L-BFGS-B', jac=_lbfgs_grad_fn,
                options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-11,
                         'maxls': 50, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass

    return best_a, best_score


def _lbfgs_obj(z, beta, n, topk_frac):
    return _objective_smooth(z, beta, topk_frac)

def _lbfgs_grad_fn(z, beta, n, topk_frac):
    return _gradient_smooth(z, beta, topk_frac)


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9,
                        topk_frac=0.05, adaptive=True):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0
    patience = 0
    best_score_phase = best_score

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta, topk_frac)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta, topk_frac)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -60, 60)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 50 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
                    patience = 0
                    if adaptive and sc < best_score_phase:
                        best_score_phase = sc
                else:
                    patience += 1
                    if patience > 20 and adaptive:
                        # Try a smaller step
                        lr *= 0.8

            if t % 100 == 0:
                mom = min(0.999, mom + 0.0003)
                if adaptive:
                    # Gradually decrease learning rate
                    lr *= 0.995

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.001, lr * 0.6)
        mom = min(0.999, mom + 0.01)

    # Final evaluation
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01, topk_frac=0.05):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W_mean = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W_mean)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 100 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _peak_equalize(a_init, deadline, n_iter=10000):
    """Directly target the lags that achieve the maximum, trying to equalize them."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.003
    momentum = np.zeros(n)
    mom_coeff = 0.85
    patience = 0
    best_so_far = best_score

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Find lags within 0.1% of max
        threshold = cm * (1 - 0.001)
        active = c >= threshold
        n_active = np.sum(active)
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Equal weights on active lags
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        W_mean = float(np.dot(w, c))

        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W_mean)

        # Momentum step
        momentum = mom_coeff * momentum - lr * grad
        z = z + momentum
        z = z - z.mean()
        z = np.clip(z, -60, 60)

        if it % 50 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                patience = 0
                if sc < best_so_far:
                    best_so_far = sc
            else:
                patience += 1
                if patience > 30:
                    lr *= 0.85
                    patience = 0
                    if lr < 1e-6:
                        break

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sine_profile(n, power=2):
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cos_modulated_profile(n, freq=1.0, amp=0.3):
    t = np.linspace(0.0, 1.0, n)
    base = np.sin(np.pi * t) ** 2
    mod = 1.0 + amp * np.cos(2 * np.pi * freq * t)
    a = base * mod
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _structured_noise(n, rng, low_freq=5, high_freq=100):
    """Generate noise with energy concentrated in low frequencies."""
    # Generate in frequency domain, then transform back
    N = _next_pow2(n)
    freq = np.fft.rfftfreq(N)
    # Weight: more energy at low frequencies
    weight = 1.0 / (1.0 + (freq * N / low_freq) ** 2)
    # Add some high-freq component
    weight += 0.1 * np.exp(-freq * N / high_freq)
    noise_freq = weight * (rng.standard_normal(N // 2 + 1) + 1j * rng.standard_normal(N // 2 + 1))
    noise = np.fft.irfft(noise_freq, N)[:n]
    return noise / (np.linalg.norm(noise) + 1e-15)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with multi-peak smoothing, adaptive temperature,
    peak equalization, and structured restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {2*len(p)*np.max(_full_convolve(best_seq,best_seq))/(np.sum(best_seq)**2):.10f} norm: {sc:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}")
    else:
        n = 4096
        parent = _make_sine_profile(n, power=2)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using sine profile n={n}, score={sc_parent:.10f}")

    best_seq = parent.copy()
    best_score = sc_parent

    # --- Phase 0: Quick L-BFGS on parent (200s) ---
    phase0_end = min(deadline, t_start + 200.0)
    if time.monotonic() < phase0_end:
        per_time = min(180.0, phase0_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, min(phase0_end, time.monotonic() + per_time),
                                          max_wall_time=per_time, topk_frac=0.05)
            update_best(p_best, sc_best, "lbfgs_p0")

    print(f"Phase 0 done: best={best_score:.10f}, n={len(best_seq)}")

    # --- Phase 1: Nesterov with extended beta schedule (300s) ---
    phase1_end = min(deadline, t_start + 500.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        beta_fracs = [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003,
                      0.0001, 5e-5, 3e-5, 2e-5, 1e-5, 5e-6, 3e-6, 2e-6, 1e-6]
        per_time = min(280.0, phase1_end - time.monotonic())
        p_best, sc_best = _nesterov_optimize(best_seq, min(phase1_end, time.monotonic() + per_time),
                                              beta_fracs, lr_init=0.012, mom_init=0.88,
                                              topk_frac=0.05, adaptive=True)
        update_best(p_best, sc_best, "nesterov_p1")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Peak equalization (200s) ---
    phase2_end = min(deadline, t_start + 700.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        per_time = min(180.0, phase2_end - time.monotonic())
        p_best, sc_best = _peak_equalize(best_seq, min(phase2_end, time.monotonic() + per_time),
                                          n_iter=20000)
        update_best(p_best, sc_best, "peak_eq_p2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam with small beta (200s) ---
    phase3_end = min(deadline, t_start + 900.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(180.0, phase3_end - time.monotonic())
        p_best, sc_best = _adam_optimize(best_seq, min(phase3_end, time.monotonic() + per_time),
                                          lr=0.02, beta_frac=0.003, topk_frac=0.05)
        update_best(p_best, sc_best, "adam_p3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Structured restarts (250s) ---
    phase4_end = min(deadline, t_start + 1150.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        n_restarts = 5
        per_restart = min(45.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            # Structured noise (low frequency dominant)
            noise = _structured_noise(n_best, rng, low_freq=3 + 2*i, high_freq=50)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + (0.03 + 0.02 * i) * noise
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase4_end - 5, time.monotonic() + per_restart)
            beta_fracs_short = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
            p_best_r, sc_best_r = _nesterov_optimize(p_start, per_dl, beta_fracs_short,
                                                      lr_init=0.008, mom_init=0.85,
                                                      topk_frac=0.05, adaptive=True)
            # Also try peak equalization
            if time.monotonic() < per_dl:
                p_best_r2, sc_best_r2 = _peak_equalize(p_best_r, per_dl, n_iter=5000)
                if sc_best_r2 < sc_best_r:
                    sc_best_r = sc_best_r2
                    p_best_r = p_best_r2
            update_best(p_best_r, sc_best_r, f"restart_{i}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Fine L-BFGS with very small beta (150s) ---
    phase5_end = min(deadline, t_start + 1300.0)
    if best_seq is not None and time.monotonic() < phase5_end and len(best_seq) <= 4096:
        per_time = min(130.0, phase5_end - time.monotonic())
        if per_time > 20:
            # Custom L-BFGS with only tiny betas
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            c_curr = _full_convolve(best_seq, best_seq)
            max_c_curr = float(c_curr.max())
            fine_betas = [max_c_curr * b for b in [2e-5, 1e-5, 5e-6, 2e-6, 1e-6]]
            per_beta = per_time / len(fine_betas)

            for fb in fine_betas:
                if time.monotonic() >= phase5_end:
                    break
                level_dl = min(phase5_end, time.monotonic() + per_beta)
                try:
                    result = minimize(
                        _lbfgs_obj, z, args=(fb, len(best_seq), 0.03),
                        method='L-BFGS-B', jac=_lbfgs_grad_fn,
                        options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-12,
                                 'maxls': 30, 'maxfun': 800}
                    )
                    if result.x is not None and np.all(np.isfinite(result.x)):
                        a_new = _softmax_fn(result.x)
                        s_new = _score_norm_fn(a_new)
                        if np.isfinite(s_new) and s_new < best_score:
                            best_score = s_new
                            best_seq = a_new.copy()
                            z = result.x.copy()
                            print(f"  Fine LBFGS beta={fb:.2e}: score={s_new:.10f}")
                except Exception:
                    pass

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Peak equalization final (100s) ---
    phase6_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        per_time = min(80.0, phase6_end - time.monotonic())
        p_best, sc_best = _peak_equalize(best_seq, min(phase6_end, time.monotonic() + per_time),
                                          n_iter=20000)
        update_best(p_best, sc_best, "peak_eq_final")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Try different length (200s) ---
    phase7_end = min(deadline, t_start + 1600.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_curr = len(best_seq)

        # Try a few different lengths
        for target_n in [n_curr * 2, n_curr // 2, n_curr + 256]:
            if time.monotonic() > phase7_end - 20:
                break
            if target_n < 256 or target_n > 8192:
                continue

"""
Multi-phase optimizer with adaptive temperature continuation, peak equalization,
structured restarts, and L-BFGS refinement for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Key improvements: better beta schedule, peak equalization subgradient, structured
low-frequency noise for restarts, and multi-stage tightening.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W_mean = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W_mean)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj_fn(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_jac_fn(z, beta, n):
    return _gradient_smooth(z, beta)


def _peak_eq_gradient(z, threshold_frac=0.999):
    """Subgradient focusing only on lags near the maximum."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    threshold = cm * threshold_frac
    active = c >= threshold
    n_active = np.sum(active)
    if n_active == 0:
        active[int(np.argmax(c))] = True
        n_active = 1
    w = np.zeros_like(c)
    w[active] = 1.0 / n_active
    W_mean = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W_mean)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z), 1
    return grad, int(n_active)


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sine_profile(n, power=2):
    t = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cos_mod_profile(n, freq=1.0, amp=0.2):
    t = np.linspace(0.0, 1.0, n)
    base = np.sin(np.pi * t) ** 2
    mod = 1.0 + amp * np.cos(2 * np.pi * freq * t)
    a = base * mod
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quad_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _structured_noise(n, rng, low_freq=5):
    N = _next_pow2(n)
    freq = np.fft.rfftfreq(N)
    weight = 1.0 / (1.0 + (freq * N / low_freq) ** 2)
    weight += 0.05
    noise_freq = weight * (rng.standard_normal(N // 2 + 1) + 1j * rng.standard_normal(N // 2 + 1))
    noise = np.fft.irfft(noise_freq, N)[:n]
    norm = np.linalg.norm(noise)
    if norm < 1e-15:
        return rng.standard_normal(n) / np.sqrt(n)
    return noise / norm


def _run_lbfgs(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = min(200.0, max(1.0, deadline - time.monotonic()))

    betas = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005,
             0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
    betas = [b * max_c0 for b in betas]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _lbfgs_obj_fn, z, args=(beta, n),

def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):

    mom = mom_init

    patience = 0

        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -60, 60)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

def _adam_optimize(a_init, deadline, lr=0.04, beta_frac=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W_mean = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W_mean)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 100 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a_curr, a_curr)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

                best_score = sc
                best_a = a_curr.copy()
                patience = 0
            else:
                patience += 1
                if patience > 40:
                    lr *= 0.8
                    patience = 0
                    if lr < 1e-7:
                        break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS continuation -> Nesterov -> Peak equalization
    -> Adam -> Structured restarts -> Fine L-BFGS -> Final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and len(p) > 0 and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            actual_score = _score_fn(best_seq)
            print(f"  [{elapsed:6.1f}s] NEW BEST: {actual_score:.10f} (norm={sc:.10f}, n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_used = 4096
    if parent is not None:
        n_used = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_used}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
    else:
        parent = _make_sine_profile(n_used, power=2)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, sine profile n={n_used}, score={sc_parent:.10f}")

    best_seq = parent.copy()
    best_score = sc_parent

    # --- Phase 0: L-BFGS on best starting point (180s) ---
    phase0_end = min(deadline, t_start + 180.0)
    if time.monotonic() < phase0_end:
        per_time = min(160.0, phase0_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, min(phase0_end, time.monotonic() + per_time),
                                         max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_p0")

    print(f"Phase 0 done: best={best_score:.10f}, n={len(best_seq)}")

    # --- Phase 1: Nesterov with long beta schedule (350s) ---
    phase1_end = min(deadline, t_start + 530.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        beta_fracs = [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003,
                      0.0001, 5e-5, 3e-5, 2e-5, 1e-5, 5e-6, 3e-6, 2e-6, 1e-6]
        per_time = min(330.0, phase1_end - time.monotonic())
        p_best, sc_best = _nesterov_optimize(best_seq, min(phase1_end, time.monotonic() + per_time),
                                              beta_fracs, lr_init=0.012, mom_init=0.88)
        update_best(p_best, sc_best, "nesterov_p1")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Peak equalization (200s) ---
    phase2_end = min(deadline, t_start + 730.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        per_time = min(180.0, phase2_end - time.monotonic())
        p_best, sc_best = _peak_equalize(best_seq, min(phase2_end, time.monotonic() + per_time),
                                          threshold_frac=0.999)
