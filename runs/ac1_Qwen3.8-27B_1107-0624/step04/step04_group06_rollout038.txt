"""

adding multi-scale approach, Nesterov momentum, and aggressive LP refinement.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)  # numerical artifacts
    return c

def _cross_corr_w(w, p):
    """Compute g[i] = sum_k w[k]*p[k-i] for i=0,...,n-1.
    w has length 2n-1, p has length n.
    Uses FFT with sufficient padding to avoid aliasing.
    """
    n = len(p)
    # Need N >= (2n-1) + n - 1 = 3n-2
    N = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n-1:2*n-1]

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _smooth_max_weights(c, beta):
    """Return (smooth_max_value, weights) for soft max of c."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _multiplicative_v2(p0, deadline, eta=0.05, top_k_frac=0.1, beta_frac=0.05,

        beta = max(1e-5, beta_frac * M)
        _, w = _smooth_max_weights(c, beta)
        
        # Gradient (cross-correlation)
        g = _cross_corr_w(w, p)
        
        # Nesterov
        v_nest = nesterov_beta * v_nest + (1.0 - nesterov_beta) * g

        log_update -= log_update.mean()  # keep sum approximately constant
        log_update = np.clip(log_update, -5.0, 5.0)
        p = p * np.exp(log_update)
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _lp_refine(p0, deadline, t_start=0.3, max_solves=30, m_shrink=0.995):

    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = p / p.sum()
    best_p = p.copy()
    best_score = _score_norm(p)
    c = _full_convolve(p)
    M = float(c.max())
    t = t_start
    n_solves = 0
    
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(p, M)
        if b is None:
            b = _lp_solve(p, M * 1.01)
            if b is None:
                break
        
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        
        b_norm = b / s_b
        p_new = (1.0 - t) * p + t * b_norm
        p_new = np.maximum(p_new, 1e-15)
        p_new = p_new / p_new.sum()
        sc_new = _score_norm(p_new)
        
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_p = p_new.copy()
        
        p = p_new
        c = _full_convolve(p)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.005)
        t = max(0.1, t * 0.9)
    
    return best_p, best_score

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _cross_corr_w(w, p):
    """g[i] = sum_k w[k]*p[k-i], w len 2n-1, p len n. FFT size >= 3n-2."""
    n = len(p)
    N = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _smooth_max_weights(c, beta):

        else:

    x = (c - mval) / beta

        heat = _cross_corr_w(w, p)
        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max
        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)
        p = p * factor
        p = np.maximum(p, 1e-15)
        p = p / p.sum()
    return best_p, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(p0, deadline, t_start=0.3, max_solves=25, m_shrink=0.996):
    n = len(p0)
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = p / p.sum()
    best_p = p.copy()
    best_score = _score_norm(p)
    c = _full_convolve(p)
    M = float(c.max())
    t = t_start
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(p, M)
        if b is None:
            b = _lp_solve(p, M * 1.01)
            if b is None:
                break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        p_new = (1.0 - t) * p + t * b_norm
        p_new = np.maximum(p_new, 1e-15)
        p_new = p_new / p_new.sum()
        sc_new = _score_norm(p_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_p = p_new.copy()
        p = p_new
        c = _full_convolve(p)
        M_new = float(c.max())
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.005)
        t = max(0.1, t * 0.9)
    return best_p, best_score


def _local_polish(p0, deadline, n_trials=50000, rng=None):
    """Fast pairwise transfer local search."""
    if rng is None:
        rng = np.random.default_rng(0)
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size
    c = _full_convolve(p)
    best_p = p.copy()
    best_score = float(2.0 * n * c.max())
    for it in range(n_trials):
        if time.monotonic() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = 0.005 * p[i]
        if delta < 1e-15:
            continue
        p[i] -= delta
        p[j] += delta
        if p[i] < 0:
            p[i] += delta
            p[j] -= delta
            continue
        c_new = _full_convolve(p)
        sc_new = float(2.0 * n * c_new.max())
        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_p = p.copy()
            c = c_new
        else:
            p[i] += delta
            p[j] -= delta
            c = c_new if sc_new < best_score else c
    return best_p, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p0)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _make_uniform(n):
    return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s <= 0.01:
            return
        p = p / s
        sc = _score_norm(p)
        if sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    if parent is not None:
        sc = _score_norm(parent)
        print(f"Parent score: {sc:.10f}")
        _update(parent, 'parent')
    else:
        _update(_make_uniform(1024), 'uniform_1024')

    # Phase 1: Adam with fixed FFT at original n (250s)
    phase1_end = t_start + 250.0
    print(f"Phase 1: Adam (fixed FFT) n={best_seq.size}, score={best_score:.10f}")
    if time.monotonic() < phase1_end:
        p1, sc1 = _adam_logit_v2(best_seq, phase1_end, rng, lr=0.05,
                                  beta_frac=0.07, beta_decay=0.92,
                                  no_imp_limit=200, nesterov_beta=0.85)
        _update(p1, 'adam_fixed')

    # Phase 2: Multiplicative + peak flood (250s)
    phase2_end = t_start + 500.0
    print(f"Phase 2: Multiplicative+flood, score={best_score:.10f}")
    if time.monotonic() < phase2_end - 100:
        p2, sc2 = _multiplicative_v2(best_seq, time.monotonic() + 120,
                                      eta=0.08, beta_frac=0.05, nesterov_beta=0.75)
        _update(p2, 'multiplicative')
    if time.monotonic() < phase2_end - 20:
        p2f, sc2f = _peak_flood_v2(best_seq, phase2_end, eta=0.02)
        _update(p2f, 'flood')

    # Phase 3: LP refinement (200s)
    phase3_end = t_start + 700.0
    print(f"Phase 3: LP n={best_seq.size}, score={best_score:.10f}")
    if time.monotonic() < phase3_end and best_seq.size <= 2048:
        p3, sc3 = _lp_refine(best_seq, phase3_end, t_start=0.35,
                              max_solves=20, m_shrink=0.996)
        _update(p3, 'lp')

    # Phase 4: Local polish (100s)
    phase4_end = t_start + 800.0
    print(f"Phase 4: Local polish, score={best_score:.10f}")
    if time.monotonic() < phase4_end:
        p4, sc4 = _local_polish(best_seq, phase4_end, n_trials=30000, rng=rng)
        _update(p4, 'local_polish')

    # Phase 5: Scale to n=2048 (300s)
    phase5_end = t_start + 1100.0
    print(f"Phase 5: Scale to 2048, score={best_score:.10f}")
    if time.monotonic() < phase5_end:
        p_int = _interp_profile(best_seq, 2048)
        _update(p_int, 'interp_2048')
        if time.monotonic() < phase5_end - 80:
            p5, sc5 = _adam_logit_v2(p_int, min(phase5_end - 80, time.monotonic() + 180),
                                      rng, lr=0.04, beta_frac=0.06, beta_decay=0.93,
                                      no_imp_limit=150, nesterov_beta=0.85)
            _update(p5, 'adam_2048')
        if time.monotonic() < phase5_end - 20:
            p5m, sc5m = _multiplicative_v2(best_seq, phase5_end, eta=0.06,
                                            beta_frac=0.04, nesterov_beta=0.75)
            _update(p5m, 'mult_2048')

    # Phase 6: LP at 2048 (150s)
    phase6_end = t_start + 1250.0
    print(f"Phase 6: LP 2048, score={best_score:.10f}")
    if time.monotonic() < phase6_end and best_seq.size <= 2048:
        p6, sc6 = _lp_refine(best_seq, phase6_end, t_start=0.25,
                              max_solves=12, m_shrink=0.997)
        _update(p6, 'lp_2048')

    # Phase 7: Scale to n=4096 (200s)
    phase7_end = t_start + 1450.0
    print(f"Phase 7: Scale to 4096, score={best_score:.10f}")
    if time.monotonic() < phase7_end:
        p_int7 = _interp_profile(best_seq, 4096)
        _update(p_int7, 'interp_4096')
        if time.monotonic() < phase7_end - 30:
            p7, sc7 = _adam_logit_v2(p_int7, phase7_end, rng, lr=0.03,
                                      beta_frac=0.05, beta_decay=0.94,
                                      no_imp_limit=150, nesterov_beta=0.85)
            _update(p7, 'adam_4096')

    # Phase 8: Final polish (remaining)
    phase8_end = deadline
    print(f"Phase 8: Final polish, score={best_score:.10f}")
    if time.monotonic() < phase8_end and best_seq is not None:
        remaining = phase8_end - time.monotonic()
        if remaining > 80:
            per = min(remaining * 0.35, 100.0)
            p8a, sc8a = _adam_logit_v2(best_seq, time.monotonic() + per, rng,
                                        lr=0.015, beta_frac=0.03, beta_decay=0.96,
                                        no_imp_limit=80, nesterov_beta=0.9)
            _update(p8a, 'final_adam')
        if time.monotonic() < phase8_end - 40:
            per2 = min(remaining * 0.25, 80.0)
            p8m, sc8m = _multiplicative_v2(best_seq, time.monotonic() + per2,
                                            eta=0.03, beta_frac=0.02, nesterov_beta=0.8)
            _update(p8m, 'final_mult')
        if time.monotonic() < phase8_end - 20 and best_seq.size <= 2048:
            per3 = min(40.0, phase8_end - time.monotonic() - 5)
            if per3 > 15:
                p8l, sc8l = _lp_refine(best_seq, time.monotonic() + per3,
                                        t_start=0.15, max_solves=6, m_shrink=0.999)
                _update(p8l, 'final_lp')
        if time.monotonic() < phase8_end - 10:
            p8p, sc8p = _local_polish(best_seq, phase8_end, n_trials=20000, rng=rng)
            _update(p8p, 'final_local')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_uniform(1024)
        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fb = _make_uniform(1024)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
