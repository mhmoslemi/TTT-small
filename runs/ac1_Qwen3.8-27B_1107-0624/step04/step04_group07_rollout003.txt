"""
Optimized search minimizing 2*n*max(conv(a,a))/(sum(a))^2 using LP fixed-point
with over-relaxation, adaptive mirror descent, and multi-scale escalation.
Improves upon previous 1.506296 by using over-relaxation, better temperature
scheduling, and focused optimization at n=1024-2048.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _build_conv_matrix(n, a, top_frac=1.0):
    rows = []
    cols = []
    vals = []
    num_rows = int((2 * n - 1) * top_frac)
    c = _full_convolve(a)
    idx = np.argsort(c)[::-1][:num_rows]
    for k in idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(j_idx))
            cols.extend((j_idx).tolist())
            vals.extend(a[k - j_idx].tolist())
    rows_arr = np.array(rows, dtype=np.int32)
    cols_arr = np.array(cols, dtype=np.int32)
    vals_arr = np.array(vals, dtype=np.float64)
    if len(rows_arr) == 0:
        return csr_matrix((2 * n - 1, n))
    A = csr_matrix((vals_arr, (rows_arr, cols_arr)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M, n, top_frac=1.0):
    A = _build_conv_matrix(n, a, top_frac)
    non_zero_rows = A.getrowindices((A.getnnz()[0] if len(A.getnnz()) > 0 else np.array([0])))
    active = np.zeros(2 * n - 1, dtype=bool)
    active_rows = np.unique(A.getrowindices(0)) if A.nnz > 0 else np.array([], dtype=np.int32)
    active[active_rows] = True
    A_active = A[active, :]
    c_obj = -np.ones(n)
    b_ub = M * np.ones(A_active.shape[0])
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_active, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_fixed_point(a, deadline, max_iters=40, over_relax=1.3):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    top_frac = 1.0 if n <= 512 else (0.7 if n <= 1024 else 0.5)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())

        b = _lp_solve(a, M, n, top_frac)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b

        # Line search with over-relaxation
        best_t = 0.5
        best_sc = _score_norm(a)
        t_candidates = [0.3, 0.5, 0.7, 0.9, 1.0]
        if over_relax > 1.0:
            t_candidates += [1.1, 1.2, over_relax]
        for t_c in t_candidates:
            a_t = (1.0 - t_c) * a + t_c * b
            if np.any(a_t < -1e-10):
                continue
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score_norm(a_t)
            if sc_t < best_sc:
                best_sc = sc_t
                best_t = t_c

        a = (1.0 - best_t) * a + best_t * b
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            # If no improvement, try smaller step
            a = best_a.copy()

    return best_a, best_score


def _mirror_descent(a, deadline, rng, lr=0.08, beta_init=None, beta_decay=0.97,
                    momentum=0.9, no_imp_limit=300, use_top_k=False, top_k=20):

    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()

    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.05 * maxc

    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        # Compute gradient
        if use_top_k and c.size > top_k:
            top_idx = np.argsort(c)[::-1][:top_k]
            c_top = c[top_idx]
            mval = float(c_top.max())
            w_top = np.exp((c_top - mval) / beta)
            w_top = w_top / w_top.sum()
            w = np.zeros(c.size)
            w[top_idx] = w_top
        else:
            _, w = _smooth_max_weights(c, beta)

        g = _cross_correlate(w, a)
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_acc.fill(0)
            v_acc.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m_acc = b1 * m_acc + (1 - b1) * gz
        v_acc = b2 * v_acc + (1 - b2) * (gz * gz)
        mh = m_acc / (1 - b1 ** t_adam)
        vh = v_acc / (1 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z -= z.mean()

        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-11:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.005, lr * 0.95)
        else:
            z = best_z.copy()
            m_acc.fill(0)
            v_acc.fill(0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:

            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.4)

        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = 0.05 if best_score < 1.51 else 0.1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            m_acc.fill(0)
            v_acc.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _exact_coord_descent(a, deadline, max_sweeps=30):
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try several multiplicative factors
            best_local = best_score
            best_local_val = old_val
            for factor in [0.3, 0.5, 0.7, 1.3, 1.6, 2.0]:
                trial = max(old_val * factor, 1e-15)
                a[j] = trial
                # Rescale to maintain sum=1
                a_n = a / a.sum()
                sc = _score_norm(a_n)
                if sc < best_local:
                    best_local = sc
                    best_local_val = trial
                a[j] = old_val
            if best_local_val != old_val:
                a[j] = best_local_val
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        if not improved:
            break

    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + p1 * u + p2 * u ** 3
    elif ptype == 'exp':
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-p1 * t / n)
    elif ptype == 'cosine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    elif ptype == 'sine':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * np.sin(2 * np.pi * p2 * t / n)
    elif ptype == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + p1
    elif ptype == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - p1) / p3) ** 2) + np.exp(-0.5 * ((t - p2) / p3) ** 2)
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / p1) ** 2)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n, dtype=np.float64) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** p1
    elif ptype == 'sawtooth':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + p1 * (2.0 * (t / n - np.floor(t / n + 0.5)))
    elif ptype == 'bimodal_step':
        t = np.linspace(0, 1, n)
        a = np.where(t < 0.3, 1.0 + p1, np.where(t > 0.7, 1.0 + p1, 1.0 - p1 * 0.5))
    elif ptype == 'triangle':
        t = np.linspace(0, 1, n)
        a = 1.0 - p1 * np.abs(2 * t - 1)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _sort_by_score(candidates):
    return sorted(candidates, key=lambda x: x[0])

                        best_seq = a_e.copy()
                        best_score = sc_e

                z = np.clip(z, -60, 60)

                z = z + ns * rng.normal(size=n_best)
                z = np.clip(z, -60, 60)
                z -= z.mean()
                a_s = np.exp(z - z.max())
                a_s = a_s / a_s.sum()

            a_r, sc_r = _mirror_descent(a_s, time.monotonic() + per, rng,

            if sc_r < best_score:

                best_score = float(sc_r)

            # Quick LP polish on restart
            if n_best <= 1024 and time.monotonic() < phase6_end - 15:

                                                  max_iters=8, over_relax=1.2)

                        best_score = float(sc_rl)

                    best_score = float(sc_fp)

    # Final validation

        else:

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
