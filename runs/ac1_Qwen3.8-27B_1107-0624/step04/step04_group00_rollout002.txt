def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a) ~ 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_weights_exact(c, temperature):
    """Compute softmax weights over convolution values with given temperature."""
    cm = float(c.max())
    c_shifted = (c - cm) / temperature
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _obj_smoothed(z, beta, n):
    """Smoothed objective using log-sum-exp approximation of max."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    lse = cm + beta * float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * lse


def _grad_smoothed(z, beta, n):
    """Gradient of smoothed objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _subgradient_exact(a, c):
    """Compute exact subgradient of max_k c_k(a). Returns (n,) vector."""
    n = len(a)
    k_star = int(np.argmax(c))
    # gradient_j = 2 * a[k_star - j] if 0 <= k_star - j < n, else 0
    grad = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    for j in range(j_min, j_max + 1):
        grad[j] = 2.0 * a[k_star - j]
    return grad


def _subgradient_weighted(a, c, m_top=5):
    """Compute weighted subgradient using top-m peaks."""
    n = len(a)
    # Get top m indices
    cm = float(c.max())
    # Use a small temperature to weight top peaks
    temperature = max(1e-10, 0.01 * cm)
    c_shifted = (c - cm) / temperature
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s > 1e-300 and np.isfinite(s):
        w = e / s
    else:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    # Restrict to top m
    top_m_idx = np.argsort(c)[::-1][:m_top]
    w_top = np.zeros_like(w)
    total_w = sum(w[i] for i in top_m_idx)
    if total_w > 1e-15:
        for i in top_m_idx:
            w_top[i] = w[i] / total_w

    # Compute weighted gradient
    grad = np.zeros(n)
    for k in top_m_idx:
        if w_top[k] < 1e-10:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        grad[j_min:j_max + 1] += w_top[k] * 2.0 * a[k - j_max:k - j_min - 1:-1]
    return grad


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    mu = np.zeros(n)
    positive_v = v[v > 0]
    if len(positive_v) == 0:
        return np.ones(n) / n
    positive_v_sorted = np.sort(positive_v)[::-1]
    cssv = np.cumsum(positive_v_sorted)
    rho = np.nonzero(positive_v_sorted * np.arange(1, len(positive_v_sorted) + 1) > (cssv - 1))[0]
    if len(rho) > 0:
        rho = rho[0]
        mu = (cssv[rho] - 1) / (rho + 1.0)
    u = np.maximum(v - mu, 0)
    return u


def _multiplicative_weights(a_init, deadline, m_top=3, alpha_init=0.01, decay=0.9999):
    """Multiplicative weights / mirror descent for min max_k c_k(a)."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    alpha = alpha_init

    for it in range(1000000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        score = 2.0 * n * cm
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Compute subgradient
        if m_top == 1:
            g = _subgradient_exact(a, c)
        else:
            g = _subgradient_weighted(a, c, m_top=m_top)

        # Multiplicative update: a_j <- a_j * exp(-alpha * g_j)
        # To avoid underflow, normalize: a_j <- a_j * exp(-alpha * (g_j - g_max))
        g_max = float(g.max())
        if g_max < 1e-15:
            g_adj = -g
        else:
            g_adj = g - g_max
        a = a * np.exp(-alpha * g_adj)
        a_sum = float(a.sum())
        if a_sum < 1e-300 or not np.isfinite(a_sum):
            break
        a = a / a_sum

        # Ensure non-negativity
        a = np.maximum(a, 0)
        a_sum = float(a.sum())
        if a_sum > 1e-15:
            a = a / a_sum

        # Decay alpha
        alpha *= decay

    return best_a, best_score


def _run_lbfgs_continuation(a_init, deadline, rng, beta_list=None):
    """L-BFGS with fine continuation schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        # Very fine continuation: start moderate, end extremely small
        beta_frac = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                     0.005, 0.004, 0.003, 0.0025, 0.002, 0.0015,
                     0.001, 0.0008, 0.0006, 0.0005, 0.0004,
                     0.0003, 0.00025, 0.0002, 0.00015, 0.0001,
                     0.00007, 0.00005, 0.00003, 0.00002, 0.00001]
        beta_list = [b * max_c0 for b in beta_frac]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 180.0)
        level_deadline = time.monotonic() + time_per

        # More iterations for smaller beta
        if beta < 0.001 * max_c0:
            maxiter = 3000
            maxfun = 8000
        elif beta < 0.01 * max_c0:
            maxiter = 2000
            maxfun = 5000
        else:
            maxiter = 1500
            maxfun = 3000

        try:
            result = minimize(
                _obj_smoothed, z, args=(beta, n),
                method='L-BFGS-B', jac=_grad_smoothed,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 100, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x.copy()
                a_new = _softmax_fn(z)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_exact_subgradient(a_init, deadline, m_top=3, alpha_init=0.005):
    """Exact subgradient with multiplicative weights, multiple alpha settings."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    # Try different configurations
    configs = [
        (m_top, alpha_init, 0.99995),
        (1, alpha_init * 0.5, 0.9999),
        (5, alpha_init * 2, 0.99999),
        (3, alpha_init * 0.5, 0.99999),
        (1, alpha_init, 0.99995),
    ]

    time_per_config = (deadline - time.monotonic()) / len(configs)
    for m_t, a_init_val, decay in configs:
        if time.monotonic() > deadline:
            break
        config_deadline = min(deadline, time.monotonic() + time_per_config)
        p, sc = _multiplicative_weights(best_a, config_deadline, m_top=m_t,
                                         alpha_init=a_init_val, decay=decay)
        if sc < best_score:
            best_score = sc
            best_a = p.copy()

    return best_a, best_score


def _run_hybrid_polish(a_init, deadline, rng):
    """Alternate between L-BFGS (small beta) and exact subgradient."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(best_a, best_a)
    max_c0 = float(c0.max())

    # Very small beta values for L-BFGS polish
    beta_vals = [1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    beta_vals = [b * max_c0 for b in beta_vals]

    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()

    for beta in beta_vals:
        if time.monotonic() > deadline:
            break
        level_dl = deadline - (deadline - time.monotonic()) / (len(beta_vals) + 1)

        # L-BFGS step
        try:
            result = minimize(
                _obj_smoothed, z, args=(beta, n),
                method='L-BFGS-B', jac=_grad_smoothed,
                options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 100, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x.copy()
                a_new = _softmax_fn(z)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
        except Exception:
            pass

        # Subgradient step
        if time.monotonic() < deadline - 30:
            sub_dl = min(deadline - 5, time.monotonic() + (deadline - time.monotonic()) * 0.3)
            p, sc = _multiplicative_weights(best_a, sub_dl, m_top=3,
                                             alpha_init=0.002, decay=0.99999)
            if sc < best_score:
                best_score = sc
                best_a = p.copy()
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()

    return best_a, best_score


def _make_initial_profiles(n, rng):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    # Linear
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, -0.5, -1.0, -1.5]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Quadratic
    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (0.0, 1.0), (-1.0, 0.5), (1.5, -1.0)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.5]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Cosine
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-10) ** p
        profiles.append(a / a.sum())

    # Random structured
    for _ in range(5):
        a = np.exp(0.3 * rng.normal(size=n).cumsum() * 0.1)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    """Interpolate profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization:
    1. L-BFGS continuation from parent (main workhorse)
    2. Exact subgradient / multiplicative weights
    3. Hybrid polish (alternate L-BFGS + subgradient)
    4. Large-n exploration (n=8192)
    5. Final fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}, actual={_score_full(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_default = 4096
        t = np.linspace(0, 1, n_default)
        parent = 1.0 + 1.5 * (t - 0.5)
        parent = np.maximum(parent, 1e-10)
        parent = parent / parent.sum()
        update_best(parent, _score_norm(parent), "default")

    # --- Phase 1: L-BFGS continuation from parent (900s) ---
    phase1_end = min(deadline, t_start + 900.0)
    print(f"Phase 1: L-BFGS continuation from parent (deadline t={phase1_end - t_start:.0f}s)")

    a_best = best_seq.copy()
    p_best, sc_best = _run_lbfgs_continuation(a_best, phase1_end, rng)
    update_best(p_best, sc_best, "lbfgs_cont_main")

    # --- Phase 1b: Multi-restart L-BFGS (200s) ---
    phase1b_end = min(deadline, t_start + 1100.0)
    print(f"Phase 1b: Multi-restart L-BFGS (deadline t={phase1b_end - t_start:.0f}s)")

    n_restarts = 3
    per_restart = (phase1b_end - time.monotonic()) / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > phase1b_end - 10:
            break

        if r == 0:
            a_cur = best_seq.copy()
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            # Structural perturbation: shift the peak location
            noise_scale = 0.08 * (r + 1)
            noise = noise_scale * rng.normal(size=len(best_seq))
            # Add a linear tilt to encourage asymmetry
            tilt = 0.05 * (r + 1) * (np.linspace(-1, 1, len(best_seq)))
            z = z + noise + tilt
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)

        per_dl = min(phase1b_end - 5, time.monotonic() + per_restart)
        p_r, sc_r = _run_lbfgs_continuation(a_cur, per_dl, rng)
        update_best(p_r, sc_r, f"restart{r}")

    # --- Phase 2: Exact subgradient / multiplicative weights (300s) ---
    phase2_end = min(deadline, t_start + 1400.0)
    print(f"Phase 2: Exact subgradient (deadline t={phase2_end - t_start:.0f}s)")

    if time.monotonic() < phase2_end and best_seq is not None:
        p_sub, sc_sub = _run_exact_subgradient(best_seq, phase2_end, m_top=3, alpha_init=0.005)
        update_best(p_sub, sc_sub, "subgradient")

    # --- Phase 3: Hybrid polish (200s) ---
    phase3_end = min(deadline, t_start + 1600.0)
    print(f"Phase 3: Hybrid polish (deadline t={phase3_end - t_start:.0f}s)")

    if time.monotonic() < phase3_end and best_seq is not None:
        p_hyb, sc_hyb = _run_hybrid_polish(best_seq, phase3_end, rng)
        update_best(p_hyb, sc_hyb, "hybrid_polish")

    # --- Phase 4: Try n=8192 (80s) ---
    phase4_end = min(deadline, t_start + 1680.0)
    print(f"Phase 4: Try n=8192 (deadline t={phase4_end - t_start:.0f}s)")

    if time.monotonic() < phase4_end and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr <= 4096:
            n_new = 8192
            p_interp = _interp_profile(best_seq, n_new)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{n_new}")

            if time.monotonic() < phase4_end - 20:
                per_dl = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic()) * 0.8)
                p_big, sc_big = _run_lbfgs_continuation(p_interp, per_dl, rng)
                update_best(p_big, sc_big, f"lbfgs_n{n_new}")

    # --- Phase 5: Final ultra-fine polish (remaining) ---
    phase5_end = deadline
    print(f"Phase 5: Final ultra-fine polish (deadline t={phase5_end - t_start:.0f}s)")

    if time.monotonic() < phase5_end and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        # Ultra-fine L-BFGS with tiny beta
        beta_tiny = [5e-7, 2e-7, 1e-7, 5e-8, 2e-8]
        beta_tiny = [b * max_c_final for b in beta_tiny]

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        best_a_local = best_seq.copy()
        best_sc_local = best_score

        for beta in beta_tiny:
            if time.monotonic() > phase5_end - 10:
                break
            level_dl = min(phase5_end - 5, time.monotonic() + (phase5_end - time.monotonic()) * 0.15)
            try:
                result = minimize(
                    _obj_smoothed, z, args=(beta, n_final),
                    method='L-BFGS-B', jac=_grad_smoothed,
                    options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-14,
                             'maxls': 100, 'maxfun': 8000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    z = result.x.copy()
                    a_new = _softmax_fn(z)
                    s_new = _score_norm(a_new)
                    if np.isfinite(s_new) and s_new < best_sc_local:
                        best_sc_local = s_new
                        best_a_local = a_new.copy()
            except Exception:
                pass

        # Final subgradient push
        if time.monotonic() < phase5_end - 5:
            sub_dl = min(phase5_end - 2, time.monotonic() + (phase5_end - time.monotonic()) * 0.4)
            p_final, sc_final = _multiplicative_weights(best_a_local, sub_dl, m_top=1,
                                                         alpha_init=0.001, decay=0.99999)
            if sc_final < best_sc_local:
                best_sc_local = sc_final
                best_a_local = p_final.copy()

        update_best(best_a_local, best_sc_local, "final_ultra")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        final_seq = final_seq / final_seq.sum()  # Normalize to sum=1
        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, norm_score={_score_norm(final_seq):.10f}")

        # Validate with evaluator (only for reasonable sizes)
        if len(final_seq) <= 6000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fallback = 256
    t = np.linspace(0, 1, n_fallback)
    fallback = 1.0 + 1.5 * (t - 0.5)
    fallback = np.maximum(fallback, 1e-10)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n={n_fallback}, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
