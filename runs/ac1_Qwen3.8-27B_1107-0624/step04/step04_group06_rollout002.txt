def _next_pow2(n):

    return max(2, k)


def _full_convolve(a):

    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

    n = len(p)

    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        return np.ones(len(e)) / len(e)
    return p / s


def _smooth_topK(c, K, beta):
    """Smooth max over top-K values with softmax weights."""
    mval = float(c.max())
    if K >= len(c):
        K = len(c)
    # Use all values but weight by top-K
    if beta < 1e-14:
        w = np.zeros_like(c)
        top_idx = np.argsort(c)[::-1][:K]
        w[top_idx] = 1.0 / K
        return mval, w
    # Softmax over all, but only top-K contribute significantly for large beta
    x = (c - mval) / beta
    ex = np.exp(np.clip(x, -30, 0))
    # Zero out bottom (2n-1-K) to focus on top-K
    if K < len(c):
        threshold_idx = np.argsort(c)[::-1][K - 1]
        threshold_val = c[threshold_idx]
        ex[c < threshold_val - 1e-10] = 0.0
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _adam_logdomain(a, deadline, rng, lr=0.05, beta_init=None,
                    topK=None, beta_decay=0.95, no_imp_limit=200,
                    restart_ns=0.1, base_lr=None):
    """Adam in log-domain with smooth top-K max objective."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = a.size

    if topK is None:
        topK = max(20, n // 4)
    if beta_init is None:
        c0 = _full_convolve(a)
        beta_init = 0.05 * c0.max()

    z = np.log(a)
    z = z - z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = _full_convolve(a)
    beta = beta_init
    best_a = a.copy()
    best_z = z.copy()
    best_score = _score(a)
    no_imp = 0
    if base_lr is None:
        base_lr = lr

    while time.monotonic() < deadline:
        val, w = _smooth_topK(c, topK, beta)
        g = _conv_cross(w, a)

        # Gradient in z-space: p * (g - mean(p*g))
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a = _softmax_z(z)
        c = _full_convolve(a)
        sc = _score(a)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.97)

        # Adapt beta
        if t_adam % 20 == 0:
            beta = max(1e-6, beta * beta_decay)
            # If top lags are close, decrease beta faster
            sidx = np.argsort(c)[::-1]
            if len(sidx) >= 2:
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    topK = max(5, topK // 2)
                    beta = max(1e-6, beta * 0.7)

        # Restart
        if no_imp > no_imp_limit:
            z = best_z + restart_ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a = _softmax_z(z)
            c = _full_convolve(a)
            beta = 0.05 * c.max()
            topK = max(20, n // 4)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
            restart_ns *= 1.3
            restart_ns = min(restart_ns, 0.5)

    return best_a, best_score


def _active_lp_solve(a, M, topK=100):
    """LP: max sum(b) s.t. (a*b)[k] <= M for top-K lags, b >= 0."""
    n = len(a)
    c = _full_convolve(a)
    # Get top-K lags
    top_idx = np.argsort(c)[::-1][:min(topK, 2 * n - 1)]
    # Sort by value descending for consistency
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]

    rows = []
    cols = []
    vals = []
    for k in top_idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([list(top_idx).index(k)] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())

    n_cons = len(top_idx)
    A = coo_matrix((vals, (rows, cols)), shape=(n_cons, n))
    c_obj = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _active_lp_iteration(a, deadline, t_init=0.4, topK=100, max_iters=200,
                         m_shrink=0.998, t_decay=0.97):
    """Iterative LP with active set and geometric blending."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_iter = 0

    while time.monotonic() < deadline and n_iter < max_iters:
        n_iter += 1

        b = _active_lp_solve(a, M, topK)
        if b is None:
            # Try relaxed M
            b = _active_lp_solve(a, M * 1.02, topK)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b

        # Geometric blend in log-domain
        log_a = np.log(np.maximum(a, 1e-30))
        log_b = np.log(np.maximum(b_norm, 1e-30))
        log_new = (1 - t) * log_a + t * log_b
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())

        # Update M: tighten toward the LP-dual bound
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.003)
        t = max(0.1, t * t_decay)

    return best_a, best_score


def _peak_flood_v2(a, deadline, eta=0.03, topK_frac=0.1):
    """Improved peak flooding targeting top-K convolution lags."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    topK = max(10, int(topK_frac * (2 * n - 1)))

    for it in range(5000):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        sc = 2.0 * n * M
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()

        # Top-K lags
        top_idx = np.argsort(c)[::-1][:topK]
        hot_mask = np.zeros(2 * n - 1)
        hot_mask[top_idx] = 1.0

        # Heat computation via FFT
        heat = _conv_cross(hot_mask, a)
        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.02)
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        c = _full_convolve(a)

        # Adaptive eta
        if it % 50 == 0 and it > 0:
            if sc >= best_score:
                eta = max(0.005, eta * 0.92)
            else:
                eta = min(0.1, eta * 1.03)

    return best_a, best_score


def _spectral_init(n, n_eig=5):
    """Initialize using eigenvectors of the convolution midpoint matrix."""
    # The middle convolution b[n-1] = sum a[i]*a[n-1-i] is a quadratic form a^T H a
    # where H[i,j] = 1 if i+j = n-1. We want to minimize this.
    # More generally, build the "average" convolution matrix.
    # Q[i,j] = sum_k W[k] * 1{i+j is in the support at lag k}
    # For convolution: b[k] = sum_{i+j=k} a[i]*a[j]
    # So the "midpoint" constraint is b[n-1] = sum_{i=0}^{n-1} a[i]*a[n-1-i]
    # Build H as the anti-identity matrix
    H = np.zeros((n, n))
    for i in range(n):
        H[i, n - 1 - i] = 1.0

    # Find smallest eigenvectors
    # For efficiency, use a few power iterations on H^{-1} or just use the structure
    # H is a permutation matrix (anti-identity), eigenvalues are all 1 (or -1 for odd n)
    # So this doesn't help directly. Instead, let's use a different approach:
    # Minimize the maximum of all lags approximately by minimizing sum_k b[k]^2
    # This is a 4th-order form, but we can approximate by the "energy" in the
    # convolution: sum_k b[k]^2 = ||a*a||^2 = <a*a, a*a> = sum_{i,j} a[i]*a[j] * (a*a)[i+j]
    # This is quadratic: a^T Q a where Q[i,j] = (a*a)[i+j]... but that depends on a.

    # Alternative: use a weighted sum of lags
    # b[k] = a^T T_k a where T_k is the k-th "convolution matrix"
    # T_k[i,j] = 1 if i+j = k
    # Minimize sum_k w_k * b[k] = a^T (sum_k w_k T_k) a
    # For uniform weights: sum_k b[k] = (sum a)^2 = 1 (constant!)
    # So we need non-uniform weights. Use w_k proportional to... the current peak.

    # Simple approach: return a profile that de-emphasizes the middle
    # The middle lag k=n-1 has the most terms, so it tends to be largest.
    # Counteract by putting more mass at the edges.
    t = np.linspace(0, 1, n)
    # U-shaped profile: high at edges, low in middle
    a = 1.0 + 2.0 * (2 * t - 1) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    profiles = {}

    # Linear (increasing)
    profiles['lin1'] = _normalize(1.0 + 1.0 * t)
    profiles['lin2'] = _normalize(1.0 + 2.0 * t)
    profiles['lin3'] = _normalize(1.0 + 3.0 * t)
    profiles['lin4'] = _normalize(1.0 + 4.0 * t)

    # Quadratic
    profiles['quad1'] = _normalize(1.0 + 1.0 * t ** 2)
    profiles['quad2'] = _normalize(1.0 + 2.0 * t ** 2)
    profiles['quad_neg'] = _normalize(1.0 + 2.0 * (1 - t) ** 2)

    # Exponential
    for r in [0.99, 0.98, 0.97, 0.95]:
        profiles[f'exp_{r}'] = _normalize(r ** np.arange(n))
    for r in [0.99, 0.98, 0.97]:
        profiles[f'expR_{r}'] = _normalize(r ** np.arange(n, 0, -1))

    # Cosine
    for f, amp in [(1, 0.5), (1, 1.0), (1, 1.5), (2, 0.3), (2, 0.6)]:
        key = f'cos_{f}_{amp}'
        a = 1.0 + amp * np.cos(2 * np.pi * f * t)
        profiles[key] = _normalize(a)

    # Asymmetric cosine
    profiles['asym_cos'] = _normalize(1.0 + 0.8 * np.cos(2 * np.pi * t) * (2*t - 1))

    # Power law
    for alpha in [0.3, 0.5, 1.0, 1.5, 2.0]:
        a = np.arange(1, n + 1) ** (-alpha)
        profiles[f'pow_{alpha}'] = _normalize(a)

    # Spectral (U-shaped)
    profiles['spec'] = _spectral_init(n)

    return profiles


def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, label=""):
        nonlocal best_seq, best_score
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            sc = _score(p)
            if np.isfinite(sc) and sc < best_score:
                best_seq = p.copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent score: {sc_parent:.10f}")
        _update(parent, 'parent')

    # --- Phase 1: Active-set LP on parent (500s) ---
    phase1_end = t_start + 500.0

    print(f"\nPhase 2: Adam polish, best={best_score:.10f}")
    if best_seq is not None:

        if pct > 30:
            p2, sc2 = _adam_logdomain(best_seq, time.monotonic() + pct, rng,
                                      lr=0.04, beta_init=0.05 * _full_convolve(best_seq).max(),
                                      topK=max(20, best_seq.size // 4),
                                      beta_decay=0.94, no_imp_limit=150, restart_ns=0.08)
            _update(p2, 'adam1')

        # LP again after Adam
        if time.monotonic() < phase2_end - 30:
            pct = min(100.0, phase2_end - time.monotonic() - 5)
            if pct > 20:
                p2l, sc2l = _active_lp_iteration(best_seq, time.monotonic() + pct,
                                                 t_init=0.3, topK=100, max_iters=150,
                                                 m_shrink=0.9985, t_decay=0.97)
                _update(p2l, 'lp2')

    # --- Phase 3: Multi-scale n=2048 (300s) ---
    phase3_end = t_start + 1100.0

                _update(p_lp, f'final_lp{cyc}')

            if time.monotonic() < phase5_end - 10 and pct_ad > 10:

                                              restart_ns=0.05 * (cyc + 1))
                _update(p_ad, f'final_adam{cyc}')

            # Peak flood
            pct_f = per_cycle * 0.2
            if time.monotonic() < phase5_end - 10 and pct_f > 10:
                pct_f = min(pct_f, phase5_end - time.monotonic() - 5)
                p_f, sc_f = _peak_flood_v2(best_seq, time.monotonic() + pct_f,
                                           eta=0.02 - 0.005 * cyc)
                _update(p_f, f'final_flood{cyc}')

    # --- Final validation ---
    print(f"\n--- FINAL RESULTS ---")
    print(f"Best score: {best_score:.10f}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profiles(1024)['lin2']

        exact_sc = _score(final)
        print(f"Final score (recomputed): {exact_sc:.10f}, n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluate_sequence: {eval_sc:.10f}")
        except Exception as e:
            print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profiles(1024)['lin2']
    print(f"FALLBACK n=1024, score={_score(fb):.10f}")
    return [float(x) for x in fb]
