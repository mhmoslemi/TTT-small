Multi-phase optimization combining LP-based direction finding, 
L-BFGS continuation, Adam, and KKT balancing.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time

# Helper functions (all top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Assumes sum(a) = 1, returns 2n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _lp_direction(f, M, top_k_constraints=None):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for selected k.
    Returns g (normalized to sum=1) or None if LP fails.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    if top_k_constraints is not None and top_k_constraints < num_constraints:
        c_f = _full_convolve(f, f)
        # Select the top-k constraint indices (where f*f is largest)
        idx = np.argsort(c_f)[-top_k_constraints:]
    else:
        idx = np.arange(num_constraints)
    
    K = len(idx)
    
    # Build constraint matrix A (K x n) where A[k_idx, j] = f[k_idx - j] if valid
    # For constraint k: (f*g)[k] = sum_j f[k-j] * g[j] (when 0<=k-j<=n-1)
    rows = []
    cols = []
    vals = []
    
    for k in idx:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j  # index into f
            if 0 <= i <= n - 1:
                rows.append(k)  # We'll remap later
                cols.append(j)
                vals.append(f[i])
    
    # Remap rows to 0..K-1
    k_map = {int(k): i for i, k in enumerate(idx)}
    rows = [k_map[r] for r in rows]
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    b = np.full(K, M)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                return g / s
        return None
    except Exception:
        return None

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding, L-BFGS continuation with top-k logsumexp,
adaptive Adam with backtracking, QP-inspired KKT balancing, and multi-length search.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_topk_weights(c, beta, top_k):
    n_c = len(c)
    if top_k <= 0 or top_k >= n_c:
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(n_c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        idx = np.argmax(c)
    else:
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_max_top = float(c_top.max())
        c_norm = (c_top - c_max_top) / beta
        e = np.exp(c_norm)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w_top = np.zeros(top_k)
            w_top[0] = 1.0
        else:
            w_top = e / s
        w = np.zeros(n_c)
        w[idx] = w_top
        W = float(np.dot(w_top, c_top))
    return w, W, idx


def _z_gradient(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, W, _ = _compute_topk_weights(c, beta, top_k)
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
    grad = 2.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _z_objective(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        c_max_top = float(c_top.max())
        c_norm = (c_top - c_max_top) / beta
        log_s = float(np.log(np.sum(np.exp(c_norm))))
        return 2.0 * n * (c_max_top + beta * log_s)
    else:
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        log_s = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_s)


def _run_lbfgs(a_init, deadline, top_k=0, beta_mult=0.02):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta_starts = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 * (beta_mult / 0.02) for b in beta_starts]

    remaining = deadline - time.monotonic()
    per_level = remaining / len(betas)

    for beta in betas:
        if time.monotonic() > deadline - 1:
            break
        level_dl = min(deadline, time.monotonic() + per_level * 1.5)
        try:
            result = minimize(
                _z_objective, z, args=(beta, n, top_k),
                method='L-BFGS-B', jac=_z_gradient,
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 30, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(np.isfinite(a_new)):
                    sc = _score_norm_fn(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.04, top_k=5, n_max_iter=100000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, 0.02 * max_c)

    adaptive_k = (top_k == 0)
    cur_k = 5 if adaptive_k else top_k
    cur_lr = lr
    patience = 0
    best_so_far = best_score

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t_count += 1
        a = _softmax_fn(z)
        use_k = cur_k
        grad = _z_gradient(z, beta, n, use_k)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t_count)
        vh = v / (1 - b2 ** t_count)
        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()
        a_new = _softmax_fn(z_new)

            best_a = a_new.copy()
            z = z_new
            patience = 0
        else:
            patience += 1
            if patience > 5000:
                cur_lr = max(0.001, cur_lr * 0.8)
                patience = 0

        if t_count % 200 == 0:
            beta = max(1e-14, beta * 0.85)
        if adaptive_k and t_count % 1000 == 0:
            cur_k = min(cur_k + 3, 50)
        if t_count % 5000 == 0:
            if best_score < 1.52:

def _kkt_balance(a_init, deadline, max_iter=5000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    step_scale = 0.05

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = cm * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            break
        if len(active) > 100:
            top_idx = np.argsort(c)[-100:]
            active = top_idx

        h = np.zeros(n)
        for k_val in active:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                h[lo:hi + 1] += a[k_val - hi:k_val - lo + 1][::-1]
        h /= len(active)
        W = float(np.dot(h, a))
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        diff = h - W
        dmax = np.abs(diff).max()
        if dmax < 1e-15:
            break
        z = z - step_scale * diff / dmax
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a_new = _softmax_fn(z)
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step_scale *= 0.7
            if step_scale < 1e-6:
                break
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z - step_scale * diff / dmax
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:

    return best_a, best_score


def _lp_direction(f, M, n_constr):
    n = len(f)
    c_f = _full_convolve(f, f)
    total_c = len(c_f)
    if n_constr < total_c:
        idx = np.argsort(c_f)[-n_constr:]
    else:
        idx = np.arange(total_c)
    K = len(idx)

    rows_list = []
    cols_list = []
    vals_list = []
    for row_i, k in enumerate(idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            is_ = k - js
            mask = (is_ >= 0) & (is_ < n)
            js = js[mask]
            is_ = is_[mask]
            if len(js) > 0:
                rows_list.append(np.full(len(js), row_i, dtype=np.int32))
                cols_list.append(js.astype(np.int32))
                vals_list.append(f[is_])

    if not rows_list:
        return None
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    b = np.full(K, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.success and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_exp_profile(n, alpha):
    r = 1.0 - alpha
    a = r ** np.arange(n)
    return a / a.sum()


def _make_sine_profile(n, freq, ratio):
    t = np.arange(n) / n
    base = 1.0 + ratio * np.cos(2 * np.pi * freq * t)
    base = np.maximum(base, 0.01)
    return base / base.sum()


def _make_two_scale_profile(n, block_frac, scale):
    a = np.ones(n)
    n_block = int(n * block_frac)
    a[:n_block] += scale
    return a / a.sum()
