```
def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0: return np.array([], dtype=np.float64)
    N = _next_pow2(2*n - 1)
```

```
def _random_smooth_z(n, rng, sigma, modes=16):
    t = np.linspace(0.0, 1.0, n)
    add = np.zeros(n, dtype=np.float64)
    for _ in range(modes):
        f = int(rng.integers(1, 25))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(-1.0, 1.0))
        add += amp * np.sin(2.0 * np.pi * f * t + ph)
    mx = float(np.max(np.abs(add)))
    if mx > 0.0:
        add /= mx
    return add * sigma
```

```python
"""
Step-function search for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Core ideas:
- Work with normalized nonnegative p (sum=1). The score is 2*n*max(p*p).
- Optimize z in softmax(z)=p using L-BFGS-B on a smooth max of the self-convolution.
- Use a larger multiplicative peak-damping polish for exact peak sharpening.
- Use an active-set linear-program direction generator to escape local basins.
- Multi-scale continuation: interpolate the best sequence to larger n and re-optimize.
- Final sparse cleanup and validation.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix


class _OptTimeout(Exception):
    pass


_GLOBAL_STATE = {
    "best_seq": None,
    "best_score": np.inf,
    "deadline": 0.0,
}

_OPT_CONTEXT = {
    "beta": 1e-3,
    "scale": 1.0,
    "deadline": 0.0,
    "local_best_score": np.inf,
    "best_z": None,
    "timeout_x": None,
}


def _next_pow2(n):
    k = 1
    if n <= 0:
        return 1
    while k < n:
        k <<= 1
    return k


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def _score_seq(a):
    p = _normalize_nonneg(a)
    if p is None:
        return np.inf
    c = _full_convolve(p)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * p.size * m)


def _update_global_from_score(p, sc, label=""):
    p = _normalize_nonng(p)
    if p is None:
        return False
    if not np.isfinite(sc):
        return False
    old = _GLOBAL_STATE["best_score"]
    if sc < old:
        _GLOBAL_STATE["best_seq"] = p.copy()
        _GLOBAL_STATE["best_score"] = float(sc)
        if not np.isfinite(old) or (float(old) - float(sc)) > 1e-8:
            print(f"  best {float(sc):.10f} n={p.size} {label}")
        return True
    return False


def _z_from_p(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 1e-300)
    z = np.log(p)
    z = z - float(np.mean(z))
    return np.clip(z, -50.0, 50.0)


def _softmax_raw(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = np.where(np.isfinite(z), z, -50.0)
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(z)
        s = float(z.size)
    return e / s


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return 0.0, np.array([], dtype=np.float64)
    m = float(np.max(c))
    if not np.isfinite(m):
        return m, np.zeros_like(c)

    if beta <= 1e-14:
        w = np.zeros_like(c)
        tol = 1e-12 * max(1.0, abs(m))
        mask = c >= m - tol
        cnt = int(np.sum(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return m, w

    x = (c - m) / beta
    x = np.clip(x, -700.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = m + beta * float(np.log(s))
    w = ex / s
    return float(val), w


def _grad_conv_weights(w, p):
    """
    Gradient of sum_k w_k * (p*p)_k with respect to p.

    For c_k = sum_i p_i p_{k-i},
        d c_k / d p_j = 2 p_{k-j} if 0 <= k-j < n, else 0.
    Therefore the gradient is 2 * cross-correlation of p with w.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)

    need = 2 * n - 1
    if w.size != need:
        tmp = np.zeros(need, dtype=np.float64)
        m = min(w.size, need)
        if m > 0:
            tmp[:m] = w[:m]
        w = tmp

    N = _next_pow2(3 * n - 1)
    fp = np.fft.rfft(p, N)
    fw_rev = np.fft.rfft(w[::-1], N)
    d = np.fft.irfft(fp * fw_rev, N)

    idx = np.arange(2 * n - 2, n - 2, -1, dtype=np.int64)
    v = 2.0 * d[idx]
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    return v


def _opt_obj_and_grad(x):
    beta = float(_OPT_CONTEXT["beta"])
    scale = float(_OPT_CONTEXT["scale"])

    p = _softmax_raw(x)
    c = _full_convolve(p)
    if c.size == 0:
        return 0.0, np.zeros_like(x)

    val, w = _smooth_max_and_weights(c, beta)
    v = _grad_conv_weights(w, p)

    dot = float(np.dot(p, v))
    g = p * (v - dot)
    g = g - float(np.mean(g))

    if not np.all(np.isfinite(g)):
        return float(val) * scale, np.zeros_like(x)

    maxg = float(np.max(np.abs(g)))
    if not np.isfinite(maxg) or maxg > 1e8:
        g = np.clip(g, -1e8, 1e8)

    return float(val) * scale, g * scale


def _opt_callback(x):
    if time.monotonic() > float(_OPT_CONTEXT["deadline"]):
        _OPT_CONTEXT["timeout_x"] = np.asarray(x, dtype=np.float64).copy()
        raise _OptTimeout

    p = _softmax_raw(x)
    sc = _score_seq(p)
    local = float(_OPT_CONTEXT["local_best_score"])
    if np.isfinite(sc) and sc < local:
        _OPT_CONTEXT["local_best_score"] = float(sc)
        _OPT_CONTEXT["best_z"] = np.asarray(x, dtype=np.float64).copy()
        _update_global_from_score(p, sc, "lbfgs")


def _lbgfs_refine(p, deadline, beta_start=0.35, beta_end=0.003, beta_decay=0.55,
                  max_stage_secs=None, no_imp_limit=3):
    p = _normalize_nonneg(p)
    if p is None:
        return None, np.inf

    n = p.size
    z = _z_from_p(p)
    best_p = p.copy()
    best_score = _score_seq(p)
    _update_global_from_score(best_p, best_score, "lbfgs_init")

    c0 = _full_convolve(best_p)
    m0 = float(np.max(c0)) if c0.size else 1.0
    if not np.isfinite(m0) or m0 <= 0.0:
        m0 = 1.0

    beta = max(1e-12, beta_start * m0)
    beta_min = max(1e-14, beta_end * m0)

    if max_stage_secs is None:
        if n <= 1024:
            max_stage_secs = 35.0
        elif n <= 2048:
            max_stage_secs = 45.0
        elif n <= 4096:
            max_stage_secs = 55.0
        else:
            max_stage_secs = 70.0

    no_improve = 0

    while time.monotonic() < deadline - 2.0 and beta > beta_min:
        if deadline - time.monotonic() < 5.0:
            break

        stage_dead = min(deadline, time.monotonic() + max_stage_secs)
        if stage_dead - time.monotonic() < 3.0:
            break

        local_stage_start = best_score
        z = _z_from_p(best_p)

        c_best = _full_convolve(best_p)
        m_best = float(np.max(c_best)) if c_best.size else 1.0
        if not np.isfinite(m_best) or m_best <= 0.0:
            m_best = 1.0

        _OPT_CONTEXT["beta"] = beta
        _OPT_CONTEXT["scale"] = float(n) * float(n)
        _OPT_CONTEXT["deadline"] = stage_dead
        _OPT_CONTEXT["local_best_score"] = local_stage_start
        _OPT_CONTEXT["best_z"] = z.copy()
        _OPT_CONTEXT["timeout_x"] = None

        try:
            res = minimize(
                _opt_obj_and_grad,
                z,
                jac=True,
                method="L-BFGS-B",
                options={
                    "maxiter": 1000000,
                    "ftol": 1e-14,
                    "gtol": 1e-10,
                    "maxcor": 30,
                    "maxls": 30,
                    "disp": False,
                    "callback": _opt_callback,
                }
            )
            if hasattr(res, "x") and res.x is not None:
                zc = np.asarray(res.x, dtype=np.float64).ravel()
                pc = _softmax_raw(zc)
                sc = _score_seq(pc)
                if np.isfinite(sc) and sc < best_score:
                    best_p = pc.copy()
                    best_score = float(sc)
                    _update_global_from_score(best_p, best_score, "lbfgs_end")
        except _OptTimeout:
            pass
        except Exception:
            pass

        ctx_score = float(_OPT_CONTEXT["local_best_score"])
        if np.isfinite(ctx_score) and ctx_score < best_score:
            best_score = ctx_score
            bz = _OPT_CONTEXT["best_z"]
            if bz is not None and np.asarray(bz).size == n and np.all(np.isfinite(np.asarray(bz))):
                best_p = _softmax_raw(np.asarray(bz)).copy()
            _update_global_from_score(best_p, best_score, "lbfgs_ctx")

        if best_score <= local_stage_start + 1e-13:
            no_improve += 1
        else:
            no_improve = 0

        if no_improve > no_imp_limit:
            break

        beta *= beta_decay

    _update_global_from_score(best_p, best_score, "lbfgs_done")
    return best_p, best_score


def _random_smooth_z(n, rng, sigma, modes=16):
    t = np.linspace(0.0, 1.0, n)
    add = np.zeros(n, dtype=np.float64)
    for _ in range(modes):
        f = int(rng.integers(1, 25))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(-1.0, 1.0))
        add += amp * np.sin(2.0 * np.pi * f * t + ph)
    mx = float(np.max(np.abs(add)))
    if mx > 0.0:
        add /= mx
    return add * sigma


def _peak_polish(p, deadline, beta_factor=0.02, max_iter=None):
    p = _normalize_nonneg(p)
    if p is None:
        return None, np.inf

    n = p.size
    if max_iter is None:
        max_iter = max(250, int(2_500_000 / max(n, 1)))

    z = _z_from_p(p)
    best_p = p.copy()
    best_score = _score_seq(p)
    _update_global_from_score(best_p, best_score, "peak_init")

    c0 = _full_convolve(best_p)
    m0 = float(np.max(c0)) if c0.size else 1.0
    if not np.isfinite(m0) or m0 <= 0.0:
        m0 = 1.0

    beta = max(1e-15, beta_factor * m0)
    no_improve = 0
    eta_choices = (0.20, 0.10, 0.05, 0.025, 0.012, 0.006, 0.003)

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        c = _full_convolve(p)
        m = float(np.max(c))
        sc = 2.0 * n * m

        if sc < best_score - 1e-14:
            best_score = float(sc)
            best_p = p.copy()
            no_improve = 0
            _update_global_from_score(best_p, best_score, "peak")
        else:
            no_improve += 1

        if it % 80 == 0 and it > 0:
            beta = max(1e-12 * m0, beta * 0.94)

        if no_improve > 160 and it % 80 == 0:
            rng = np.random.default_rng(987654 + it)
            z = _z_from_p(best_p) + _random_smooth_z(n, rng, 0.03, modes=12)
            z = np.clip(z, -50.0, 50.0)
            z = z - float(np.mean(z))
            p = _softmax_raw(z)
            no_improve = 0
            continue

        if no_improve > 600:
            break

        val, w = _smooth_max_and_weights(c, beta)
        v = _grad_conv_weights(w, p)
        maxv = float(np.max(v))
        if not np.isfinite(maxv) or maxv <= 0.0:
            break

        h = v / maxv
        h = h - float(np.mean(h))

        improved = False
        for eta in eta_choices:
            z_new = z - eta * h
            z_new = np.clip(z_new, -50.0, 50.0)
            z_new = z_new - float(np.mean(z_new))
            p_new = _softmax_raw(z_new)
            sc_new = _score_seq(p_new)
            if np.isfinite(sc_new) and sc_new < best_score - 1e-15:
                best_score = float(sc_new)
                best_p = p_new.copy()
                z = z_new
                p = p_new
                improved = True
                _update_global_from_score(best_p, best_score, "peak")
                break

        if not improved:
            z_new = z - 1e-4 * h
            z_new = np.clip(z_new, -50.0, 50.0)
            z_new = z_new - float(np.mean(z_new))
            p_new = _softmax_raw(z_new)
            sc_new = _score_seq(p_new)
            if np.isfinite(sc_new) and sc_new < best_score - 1e-15:
                best_score = float(sc_new)
                best_p = p_new.copy()
                z = z_new
                p = p_new
                _update_global_from_score(best_p, best_score, "peak")

    _update_global_from_score(best_p, best_score, "peak_done")
    return best_p, best_score


def _lp_active_polish(p, deadline, max_solves=12, max_active=80, shrink=0.004):
    p = _normalize_nonng(p)
    if p is None:
        return None, np.inf

    n = p.size
    best_p = p.copy()
    best_score = _score_seq(p)
    _update_global_from_score(best_p, best_score, "lp_init")

    if max_active is None or max_active <= 0:
        max_active = 80
    max_solves = max(2, min(int(max_solves), 24))
    solves = 0

    while time.monotonic() < deadline and solves < max_solves:
        c = _full_convolve(p)
        m = float(np.max(c))
        sc = 2.0 * n * m
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = float(sc)
            best_p = p.copy()
            _update_global_from_score(best_p, best_score, "lp")

        if not np.isfinite(m) or m <= 1e-15:
            break

        if c.size > max_active:
            active = np.argsort(c)[::-1][:max_active]
        else:
            active = np.arange(c.size)

        m_base = m * (1.0 - shrink)
        found = False
        b = None

        for relax in range(4):
            m_try = m_base * (1.0 + 0.02 * relax)
            if not np.isfinite(m_try) or m_try <= 1e-15:
                continue

            A = lil_matrix((active.size, n), dtype=np.float64)
            for r, k in enumerate(active):
                i0 = max(0, int(k) - n + 1)
                i1 = min(n - 1, int(k))
                if i0 <= i1:
                    idx = np.arange(i0, i1 + 1, dtype=np.int64)
                    A[r, idx] = p[k - idx]

            try:
                A = A.tocsr()
                res = linprog(
                    -np.ones(n),
                    A_ub=A,
                    b_ub=m_try * np.ones(active.size),
                    bounds=[(0.0, 1.0)] * n,
                    method="highs",
                )
                if res.success and res.x is not None:
                    b = np.asarray(res.x, dtype=np.float64).ravel()
                    found = True
                    break
            except Exception:
                pass

        if not found or b is None:
            break

        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        s = float(b.sum())
        if s <= 1e-12:
            break
        b = b / s

        improved = False
        for t in (1.0, 0.85, 0.70, 0.55, 0.40, 0.25, 0.15, 0.75, 0.35):
            pn = (1.0 - t) * p + t * b
            pn = _normalize_nonng(pn)
            if pn is None:
                continue
            sc_new = _score_seq(pn)
            if np.isfinite(sc_new) and sc_new < best_score - 1e-15:
                best_score = float(sc_new)
                best_p = pn.copy()
                p = pn
                improved = True
                _update_global_from_score(pn, sc_new, "lp")
                break

        if not improved:
            if shrink > 0.0005:
                shrink *= 0.5
            if solves > 5:
                break

        solves += 1

    _update_global_from_score(best_p, best_score, "lp_done")
    return best_p, best_score


def _make_profile_uniform(n):
    return np.ones(n, dtype=np.float64)


def _make_profile_triangle(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2.0 * t - 1.0)
    return np.maximum(a, 1e-9)


def _make_profile_parab(n):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 - u * u
    return np.maximum(a, 1e-9)


def _make_profile_edge(n):
    t = np.linspace(0.0, 1.0, n)
    a = (t + 0.01) * (1.0 - t + 0.01)
    return np.maximum(a, 1e-9)


def _make_profile_sine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.35 * np.sin(np.pi * t)
    return np.maximum(a, 1e-9)


def _make_profile_power(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.power(t + 0.001, 1.6)
    return np.maximum(a, 1e-9)


def _initial_profiles(n):
    raw = [
        ("uniform", _make_profile_uniform(n)),
        ("triangle", _make_profile_triangle(n)),
        ("parab", _make_profile_parab(n)),
        ("edge", _make_profile_edge(n)),
        ("sine", _make_profile_sine(n)),
        ("power", _make_profile_power(n)),
    ]
    out = []
    for lab, a in raw:
        a = _normalize_nonneg(a)
        if a is None:
            continue
        sc = _score_seq(a)
        out.append((sc, lab, a))
    out.sort()
    return [a for _, _, a in out]


def _interp_profile(p, n_new):
    p = _normalize_nonneg(p)
    if p is None:
        return None
    n0 = p.size
    if n0 == int(n_new):
        return p
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, int(n_new))
    q = np.interp(x1, x0, p)
    q = np.where(np.isfinite(q), q, 0.0)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 1e-300:
        q = np.ones(int(n_new), dtype=np.float64) / float(n_new)
    else:
        q = q / s
    return q


def _finalize_sequence(p):
    p = _normalize_nonng(p)
    if p is None:
        return [1.0 / 1024.0] * 1024

    best = p.copy()
    best_score = _score_seq(p)

    for th in (1e-16, 1e-14, 1e-12, 1e-10, 1e-8):
        q = best.copy()
        q[q < th] = 0.0
        q = _normalize_nonng(q)
        if q is None:
            continue
        sc = _score_seq(q)
        if np.isfinite(sc) and sc < best_score - 1e-15:
            best = q
            best_score = sc

    try:
        vals = [float(x) for x in best]
        sc_eval = evaluate_sequence(vals)
        if np.isfinite(sc_eval):
            if not np.isfinite(best_score) or float(sc_eval) < float(best_score):
                best_score = float(sc_eval)
    except Exception:
        pass

    print(f"FINAL score {float(best_score):.10f} n={best.size}")
    return [float(x) for x in best]
```
