```python
"""
L-BFGS with temperature annealing on log-sum-exp, combined with LP fixed-point
polishing, peak-targeted perturbation, and multi-scale escalation for minimizing
2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _lsmax_value_grad(a, beta):
    """
    Compute log-sum-exp of conv(a,a) and its gradient w.r.t. a.
    Returns (value, gradient).
    """
    n = len(a)
    c = _full_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        val = mval
    else:
        w = ex / s
        val = mval + beta * float(np.log(s))
    
    # Gradient: g[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # Using FFT cross-correlation
    a_rev = a[::-1].copy()
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a_rev, N)
    conv_res = np.fft.irfft(fw * fa_rev, N)
    g = 2.0 * conv_res[n-1:2*n-1]
    
    return float(val), g

def _lbfgs_anneal(a_init, deadline, rng, beta_init=None, beta_min=0.01,
                   beta_decay=0.97, max_lbfgs_iters=200):
    """
    L-BFGS optimization with temperature annealing in log-space.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-12).astype(np.float64).copy()
    a = a / a.sum()
    
    # Work in log-space
    z = np.log(a)
    z -= z.mean()  # Center for numerical stability
    
    c = _full_convolve(a)
    max_c = float(c.max())
    beta = beta_init if beta_init is not None else max_c / 5.0
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Objective in z-space
    def obj_and_grad(z_flat):
        z_vec = z_flat.reshape(n)
        # Compute a = softmax(z_vec)
        z_shifted = z_vec - z_vec.max()
        ez = np.exp(z_shifted)
        a_vec = ez / ez.sum()
        a_vec = np.maximum(a_vec, 1e-15)
        a_vec = a_vec / a_vec.sum()
        
        val, grad_a = _lsmax_value_grad(a_vec, beta)
        
        # Chain rule: grad_z = a * (grad_a - dot(a, grad_a))
        dot = float(np.dot(a_vec, grad_a))
        grad_z = a_vec * (grad_a - dot)
        
        return float(val), grad_z
    
    n_anneal_steps = 30
    per_step_time = max(5.0, (deadline - time.monotonic()) / (n_anneal_steps + 5))
    
    for step in range(n_anneal_steps):
        if time.monotonic() > deadline:
            break
        
        # Current objective
        cur_val, cur_grad = obj_and_grad(z)
        
        # Run L-BFGS
        step_deadline = min(time.monotonic() + per_step_time, deadline - 5)
        
        # Count function evaluations
        nevals = [0]
        def fun(x):
            nevals[0] += 1
            if time.monotonic() > step_deadline:
                raise KeyboardInterrupt
            val, g = obj_and_grad(x)
            if not np.isfinite(val):
                return 1e10, np.zeros(n)
            return val, g
        
        try:
            res = minimize(fun, z, jac=True, method='L-BFGS-B',
                          options={'maxiter': max_lbfgs_iters, 
                                   'maxfun': 500,
                                   'gtol': 1e-6,
                                   'factr': 1e4})
        except (KeyboardInterrupt, Exception):
            continue
        
        # Update z from result
        if hasattr(res, 'x') and res.x is not None:
            z_new = res.x
            z_new = np.clip(z_new, -50, 50)
            z_new -= z_new.mean()
            
            # Check improvement
            z_shifted = z_new - z_new.max()
            ez = np.exp(z_shifted)
            a_new = ez / ez.sum()
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            
            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            
            z = z_new
        
        # Decrease temperature
        beta = max(beta_min, beta * beta_decay)
        
        # Track progress
        if step % 5 == 0:
            sc_cur = _score(np.exp(z - z.max()) / np.exp(z - z.max()).sum())
            print(f"    Anneal step {step}: beta={beta:.6f}, score={sc_cur:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

def _lp_fixed_point(a, deadline, max_iters=50):
    """LP fixed-point method (from previous approach, for polishing)."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        M = float(c.max())
        
        # Build LP: maximize sum(b) subject to conv(a*b)[k] <= M, b >= 0
        # conv(a*b)[k] = sum_i a[i]*b[k-i]
        # This is linear in b!
        # Variables: b[0], ..., b[n-1]
        # Constraints: for each k = 0,...,2n-2: sum_i a[i]*b[k-i] <= M (where b[j]=0 if j<0 or j>=n)
        
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(a[k - idx].tolist())
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'time_limit': max(1.0, deadline - time.monotonic() - 5)})
        
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s_b = float(b.sum())
            if s_b < 1e-10:
                break
            b = b / s_b
            
            # Line search
            best_t = 0.5
            best_sc = _score(a)
            for t_c in [0.3, 0.5, 0.7, 0.9, 1.0]:
                a_t = (1.0 - t_c) * a + t_c * b
                a_t = np.maximum(a_t, 1e-15)
                a_t = a_t / a_t.sum()
                sc_t = _score(a_t)
                if sc_t < best_sc:
                    best_sc = sc_t
                    best_t = t_c
            
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            break
    
    return best_a, best_score

def _coord_descent(a, deadline, max_sweeps=30):
    """Coordinate descent for fine-tuning."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.5, 0.75, 1.25, 2.0]:
                a[j] = max(old_val * factor, 1e-15)
                sc = _score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            else:
                a[j] = old_val
        if not improved:
            break
    
    return best_a, best_score

def _make_profile(n, ptype, p1=1.0, p2=0.0, p3=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2*t - 1)**2
    elif ptype == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2*t - 1
        a = 1.0 + p1*u + p2*u**3
    elif ptype == 'exp':
        t = np.arange(n)
        a = np.exp(-p1 * t / n)
    elif ptype == 'cosine':
        t = np.arange(n)
        a = 1.0 + p1 * np.cos(2*np.pi*p2*t/n)
    elif ptype == 'sine':
        t = np.arange(n)
        a = 1.0 + p1 * np.sin(2*np.pi*p2*t/n)
    elif ptype == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5)**2 + p1
    elif ptype == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5*((t-0.5)/p1)**2)
    elif ptype == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2*t - 1)**2 + p1
    elif ptype == 'edge_heavy':
        t = np.arange(n) / max(1, n-1)
        a = 1.0 / (0.01 + abs(t - 0.5)*2.0)**p1
    elif ptype == 'random':
        rng = np.random.default_rng(int(p1))
        a = rng.chisquare(df=int(p2) if p2 > 0 else 4, size=n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new
