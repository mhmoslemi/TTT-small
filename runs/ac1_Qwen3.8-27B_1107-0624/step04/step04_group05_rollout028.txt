Step function optimization using LP best-response, FFT-based gradient descent
with adaptive smooth-max, peak-targeted local search, and multi-scale search.
Targets score < 1.5027.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _gradient_fft(a, w):
    """Compute gradient vector g where g[j] = sum_k w[k]*(a[k-j] + a[j-k])."""
    n = len(a)
    N = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # Cross-correlation: sum_i w[i+j]*a[i] = ifft(FFT(w) * conj(FFT(a)))
    cross = np.fft.irfft(fw * np.conj(fa), N)[:n]
    # Convolution: sum_i w[i]*a[j-i] = ifft(FFT(w) * FFT(a))
    conv = np.fft.irfft(fw * fa, N)[:n]
    g = cross + conv
    return g


def _smooth_max_weights(c, beta):
    """Compute weights for smooth-max approximation."""
    mval = c.max()
    x = (c - mval) / beta
    x = np.minimum(x, 0.0)
    e = np.exp(x)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return mval, w
    return mval + beta * np.log(s), e / s

    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    maxc = c.max()
    if beta_init is None:

    else:

Let me think about a more aggressive strategy:

**Idea: Use the LP to define a "basin" and then do exhaustive search within that basin.**

After LP convergence, the peak positions are fixed (or nearly so). We can then:
1. Identify the top 5 peak positions.
2. For each peak position k, the peak value is sum_i a[i]*a[k-i].
3. We can try to reduce this by moving mass from "hot" positions (positions i that appear in many peak equations) to "cold" positions.

This is essentially a **coordinate descent** on the mass distribution, where each coordinate update is:
- Choose position j to reduce (a "hot" position).
- Choose position j' to increase (a "cold" position).
- Transfer a small amount of mass from j to j'.
- Accept if the peak decreases.

This is O(n^2) per step but each step is very cheap (just recompute the affected convolution values).

Actually, for n=1024, direct coordinate descent might be too slow. Let me think of a more efficient approach.

**Better idea: Use the "subgradient" of the max function directly.**

The subgradient of max(c) at a is: w (the uniform distribution over argmax positions).
The "gradient flow" is: a(t+dt) = a(t) - dt * P_a(w) where P_a is the projection onto the tangent space.

In the simplex, this is: a(t+dt) = a(t) - dt * (g - mean(g)) where g[j] = sum_{k in argmax} (1/|argmax|) * (a[k-j] + a[j-k]).

This is a **mirror descent** step in the simplex (with entropy regularization), which is:
a_new[j] = a[j] * exp(-dt * g[j]) / Z

or equivalently in logit space:
z_new = z - dt * (g - mean(g))
a_new = softmax(z_new)

This is exactly what the Adam-in-logit-space is doing, but with Adam's adaptive learning rate.

The gradient of the penalty at position k (where conv_k > M) is:

This naturally focuses on the positions that violate the constraint.

    """Optimize to minimize sum_k max(0, conv_k - M)^2."""
    a = a0.copy()
    n = len(a)

        c = _full_convolve(a)
        excess = np.maximum(c - M_target, 0)
        
        if excess.max() < 1e-10:
            break  # Feasible!
        
        # Gradient of sum excess^2
        # d/d a[j] sum_k excess_k^2 = 2 * sum_{k: c_k > M} excess_k * dc_k/da[j]
        w = excess / max(excess.sum(), 1e-20)  # normalize for stability
        g = _gradient_fft(a, w)
        g = g - g.mean()  # project to simplex
        
        # Update
        z = np.log(np.maximum(a, 1e-15))
        z = z - lr * g
        z = z - z.mean()
        a = np.exp(z - z.max())
        a = a / a.sum()
    
    return a

Step function optimization combining aggressive LP best-response, adaptive
smooth-max gradient descent in logit space, penalty-based peak targeting,
multi-restart perturbation, and multi-scale escalation. Targets score < 1.5027.

import numpy as np
from scipy.optimize import linprog

import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

    n = len(a)

            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
