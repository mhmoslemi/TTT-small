Top-K projected gradient descent with adaptive schedule for minimizing
the peak-to-mean ratio of linear autocorrelation. Combined with logsumexp
L-BFGS polishing and flattening penalty.
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _topk_projgrad(a_init, deadline, rng, K_schedule=None, lr_init=0.05, 
                    use_flatten=True, flatten_weight=0.1, use_adam=True):

    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    # Default K schedule
    if K_schedule is None:
        K_schedule = [max(1, n // 4), max(1, n // 8), max(1, n // 16), 
                      max(1, n // 32), max(1, n // 64), 5, 3, 2, 1]
    
    K_idx = 0
    iters_per_K = 500
    
    for it in range(200000):
        if time.monotonic() > deadline - 1.0:
            break

        if it % iters_per_K == 0 and K_idx < len(K_schedule) - 1:
            K_idx += 1
        K = K_schedule[K_idx]
        K = min(K, 2*n - 1)
        
        b = _fft_convolve(a, a)
        # Get top K indices
        if K >= len(b):
            top_idx = np.arange(len(b))
        else:

"""

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _topk_flatten_grad(a, K, flatten_w):
    """Compute combined gradient: top-K peak reduction + flattening penalty.
    Returns gradient vector of length n."""
    n = len(a)
    b = _fft_convolve(a, a)
    a_rev = a[::-1]
    
    # Top-K mask
    K_actual = min(K, 2 * n - 1)
    if K_actual >= 2 * n - 1:
        mask = np.ones(2 * n - 1)
    else:
        top_idx = np.argpartition(b, -K_actual)[-K_actual:]
        mask = np.zeros(2 * n - 1)
        mask[top_idx] = 1.0
    
    # Combined weight: alpha*mask + beta*b
    # alpha controls peak reduction, beta controls flattening
    alpha = 1.0
    beta = flatten_w
    combined_w = alpha * mask + beta * b
    
    # Single convolution for gradient
    conv_res = _fft_convolve(combined_w, a_rev)
    grad = 4.0 * conv_res[n - 1:2 * n - 1]
    
    return grad, b


def _topk_adam(a_init, deadline, rng, K_start=None, K_end=2, iters_per_phase=400,
               lr_init=0.03, flatten_init=0.05, flatten_growth=1.02, use_fw=False):
    """Top-K Adam with adaptive flattening and optional Frank-Wolfe steps."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    if K_start is None:
        K_start = max(1, n // 4)
    
    # K schedule: geometric from K_start to K_end
    n_phases = max(1, int(np.log(max(1, K_start / K_end)) / np.log(2)))
    K_schedule = [max(K_end, int(K_start / (2 ** i))) for i in range(n_phases + 1)]
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    
    flatten_w = flatten_init
    no_improve_count = 0
    last_best = best_score
    
    it = 0
    while time.monotonic() < deadline - 1.0:
        it += 1
        
        # Determine current K from schedule
        phase = min(it // iters_per_phase, len(K_schedule) - 1)
        K = K_schedule[phase]
        
        # Compute gradient
        grad, b = _topk_flatten_grad(a, K, flatten_w)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        step = lr * mh / (np.sqrt(vh) + eps)
        
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s
        
        # Check improvement
        if it % 5 == 0 or it < 20:
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
            
            a = a_new
        
        # Adaptive strategies
        if no_improve_count > 10:
            # Increase flattening weight
            flatten_w = min(1.0, flatten_w * flatten_growth)
            no_improve_count = 0
        
        if it % 500 == 0:
            # Reduce learning rate
            lr = max(0.002, lr * 0.92)
        
        if it % 2000 == 0:
            # Periodic Frank-Wolfe step for diversity
            if use_fw and rng.random() < 0.3:
                grad_norm = np.linalg.norm(grad)
                if grad_norm > 1e-12:
                    # FW step: move toward the vertex with most negative gradient
                    i_min = np.argmin(grad)
                    v_fw = np.zeros(n)
                    v_fw[i_min] = 1.0
                    t_fw = 2.0 / (it + 2)
                    a_fw = (1 - t_fw) * a + t_fw * v_fw
                    a_fw = np.maximum(a_fw, 0.0)
                    sf = a_fw.sum()
                    if sf > 1e-15:
                        a_fw /= sf
                    sc_fw = _score_norm(a_fw)
                    if sc_fw < best_score:
                        best_score = sc_fw
                        best_a = a_fw.copy()
                        a = a_fw
    
    return best_a, best_score


def _lbfgs_continuation(a_init, deadline, rng, max_wall_time=100, beta_scale=1.0):
    """L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _fft_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    beta_starts = [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 * beta_scale for b in beta_starts]
    
    per_level = max(3.0, max_wall_time / len(betas))
    
    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        
        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _fft_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _fft_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _fft_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                g = np.zeros_like(z_)
            return g
        
        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _local_search_focused(a_init, deadline, rng, n_trials=3000):

def _make_profile(n, profile_type, **kwargs):
    """Generate various profile shapes."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    
    if profile_type == 'linear':
        slope = kwargs.get('slope', 1.5)
        a = 1.0 + slope * u
    elif profile_type == 'quadratic':
        a1 = kwargs.get('a1', 0.8)
        a2 = kwargs.get('a2', 0.3)
        a = 1.0 + a1 * u + a2 * u ** 2
    elif profile_type == 'cubic':
        a1 = kwargs.get('a1', 0.8)
        a3 = kwargs.get('a3', -0.4)
        a = 1.0 + a1 * u + a3 * u ** 3
    elif profile_type == 'gaussian':
        width = kwargs.get('width', 0.3)
        a = np.exp(-u ** 2 / (2.0 * width ** 2))
    elif profile_type == 'cosine':
        alpha = kwargs.get('alpha', 1.0)
        a = 0.5 * (1.0 + np.cos(np.pi * u * alpha))
    elif profile_type == 'power':
        p = kwargs.get('p', 2.0)
        a = (1.0 - np.abs(u)) ** p
    elif profile_type == 'flat':
        a = np.ones(n)
    elif profile_type == 'v_shape':
        depth = kwargs.get('depth', 0.5)
        a = 1.0 - depth * np.abs(u)
    elif profile_type == 'bell':
        w = kwargs.get('w', 0.4)
        a = 1.0 / (1.0 + (u / w) ** 2)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _coordinate_descent(a_init, deadline, rng, n_sweeps=20):
    """Coordinate-wise descent: optimize each element (or small group) one at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Randomize the order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline - 1.0:
                break
            
            # Try a few values for a[i]
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # The score depends on a[i] through b[k] for k where a[i] participates
            # b[k] = sum_j a[j]*a[k-j], a[i] appears when j=i or k-j=i
            # So b[i+j] and b[j+i] are affected for all j
            # Specifically: b[k] changes by 2*a[i]*a[k-i] when we change a[i]
            # (accounting for the fact that it appears in two terms unless k=2i)
            
            # Try multiplicative changes
            for factor in [0.7, 0.85, 1.0, 1.15, 1.3]:
                if factor == 1.0:
                    continue
                a_trial = a.copy()
                a_trial[i] = old_val * factor
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial /= a_trial.sum()
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    break  # Accept first improvement
    
    return best_a, best_score

    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, 'cosine', alpha=1.0)
        print(f"No parent, using cosine profile n={n_parent}")
        update_best(parent, _score_norm(parent), "init_cosine")
    
    # --- Phase 0: Diverse initialization (40s) ---
    phase0_end = min(deadline, t_start + 40.0)
    
    n_base = n_parent
    candidates = [(best_score, best_seq.copy(), n_base)]
    
    profiles_to_try = [
        ('cosine', {'alpha': 0.8}),
        ('cosine', {'alpha': 1.2}),
        ('cosine', {'alpha': 1.5}),
        ('linear', {'slope': 1.0}),
        ('linear', {'slope': 2.0}),
        ('quadratic', {'a1': 1.0, 'a2': -0.2}),
        ('quadratic', {'a1': 0.5, 'a2': 0.5}),
        ('cubic', {'a1': 1.0, 'a3': 0.5}),
        ('power', {'p': 1.5}),
        ('power', {'p': 2.5}),
        ('bell', {'w': 0.3}),
        ('bell', {'w': 0.5}),
        ('v_shape', {'depth': 0.3}),
        ('v_shape', {'depth': 0.6}),
        ('gaussian', {'width': 0.25}),
        ('gaussian', {'width': 0.4}),
    ]
    
    for ptype, pkw in profiles_to_try:
        if time.monotonic() > phase0_end:
            break
        p = _make_profile(n_base, ptype, **pkw)
        sc = _score_norm(p)
        candidates.append((sc, p, n_base))
        update_best(p, sc, f"init_{ptype}")
    
    # Try a couple different lengths
    for alt_n in [2048, 8192]:
        if time.monotonic() > phase0_end:
            break
        for ptype in ['cosine', 'linear']:
            kw = {'alpha': 1.0} if ptype == 'cosine' else {'slope': 1.5}
            p = _make_profile(alt_n, ptype, **kw)
            sc = _score_norm(p)
            candidates.append((sc, p, alt_n))
            update_best(p, sc, f"init_n{alt_n}_{ptype}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")
    
    # --- Phase 1: Top-K ProjGrad on top candidates (500s) ---
    phase1_end = min(deadline, t_start + 540.0)
    n_to_opt = 3
    per_cand = max(30.0, (phase1_end - time.monotonic()) / n_to_opt)
    count = 0
    seen_n = set()
    
    for sc, p, pn in candidates:
        if time.monotonic() > phase1_end - 5.0 or count >= n_to_opt:
            break
        if sc > best_score * 1.05:
            break
        if pn in seen_n:
            continue
        seen_n.add(pn)
        
        dl = min(phase1_end - 5.0, time.monotonic() + per_cand)
        p_best, sc_best = _topk_adam(p, dl, rng, K_start=max(1, pn//4),
                                     K_end=2, iters_per_phase=300,
                                     lr_init=0.025, flatten_init=0.05)
        update_best(p_best, sc_best, f"topk_n{pn}")
        count += 1
    
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: L-BFGS continuation (350s) ---
    phase2_end = min(deadline, t_start + 890.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        dl = phase2_end - 5.0
        wt = max(30.0, (phase2_end - time.monotonic()))
        p_best, sc_best = _lbfgs_continuation(best_seq, dl, rng, max_wall_time=wt, beta_scale=1.0)
        update_best(p_best, sc_best, "lbfgs_main")
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: Second round Top-K with different params (250s) ---
    phase3_end = min(deadline, t_start + 1140.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        # Try with more aggressive flattening
        dl = min(phase3_end - 5.0, time.monotonic() + 120.0)
        p_best, sc_best = _topk_adam(best_seq, dl, rng, K_start=max(1, len(best_seq)//8),
                                     K_end=1, iters_per_phase=200,
                                     lr_init=0.015, flatten_init=0.2, flatten_growth=1.05,
                                     use_fw=True)
        update_best(p_best, sc_best, "topk_flatten")
        
        # Try with less flattening, more peak-focused
        if time.monotonic() < phase3_end - 30:
            dl2 = min(phase3_end - 5.0, time.monotonic() + 80.0)
            p_best2, sc_best2 = _topk_adam(best_seq, dl2, rng, K_start=5,
                                           K_end=1, iters_per_phase=150,
                                           lr_init=0.01, flatten_init=0.01)
            update_best(p_best2, sc_best2, "topk_peak")
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: L-BFGS fine (200s) ---
    phase4_end = min(deadline, t_start + 1340.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        dl = phase4_end - 5.0
        wt = max(30.0, (phase4_end - time.monotonic()))
        p_best, sc_best = _lbfgs_continuation(best_seq, dl, rng, max_wall_time=wt, beta_scale=0.4)
        update_best(p_best, sc_best, "lbfgs_fine")
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Focused local search (150s) ---
    phase5_end = min(deadline, t_start + 1490.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        dl = phase5_end - 5.0
        p_best, sc_best = _local_search_focused(best_seq, dl, rng, n_trials=5000)
        update_best(p_best, sc_best, "local_focused")
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Coordinate descent (100s) ---
    phase6_end = min(deadline, t_start + 1590.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        dl = phase6_end - 5.0
        p_best, sc_best = _coordinate_descent(best_seq, dl, rng, n_sweeps=5)
        update_best(p_best, sc_best, "coord_descent")
    print(f"Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Try different lengths (100s) ---
    phase7_end = min(deadline, t_start + 1690.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_curr = len(best_seq)
        for target_n in [2048, 4096, 8192]:
            if target_n == n_curr or time.monotonic() > phase7_end - 10:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")
            
            if sc_interp < best_score * 1.01 and time.monotonic() < phase7_end - 15:
                dl = min(phase7_end - 5, time.monotonic() + 40.0)
                p_best, sc_best = _topk_adam(p_interp, dl, rng, K_start=max(1, target_n//8),
                                             K_end=2, iters_per_phase=200,
                                             lr_init=0.02, flatten_init=0.1)
                update_best(p_best, sc_best, f"topk_n{target_n}")
    print(f"Phase 7 done: best={best_score:.10f}")
    
    # --- Phase 8: Final intensive polishing (remaining) ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = max(20.0, deadline - time.monotonic())
        
        # L-BFGS with smallest beta
        if remaining > 40:
            dl = min(deadline - 5, time.monotonic() + remaining * 0.4)
            p_best, sc_best = _lbfgs_continuation(best_seq, dl, rng,
                                                 max_wall_time=remaining * 0.4, beta_scale=0.2)
            update_best(p_best, sc_best, "final_lbfgs")
        
        # Final local search
        if time.monotonic() < deadline - 10:
            dl = min(deadline - 5, time.monotonic() + 50.0)
            p_best, sc_best = _local_search_focused(best_seq, dl, rng, n_trials=2000)
            update_best(p_best, sc_best, "final_local")
        
        # One more top-K round
        if time.monotonic() < deadline - 10:
            dl = min(deadline - 3, time.monotonic() + 30.0)
            p_best, sc_best = _topk_adam(best_seq, dl, rng, K_start=3, K_end=1,
                                         iters_per_phase=100, lr_init=0.005,
                                         flatten_init=0.3)
            update_best(p_best, sc_best, "final_topk")
    
    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile(4096, 'cosine', alpha=1.0)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        # Validate with evaluator
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval != sc_final:
                        print(f"  (internal: {sc_final:.10f})")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_profile(4096, 'cosine', alpha=1.0)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
