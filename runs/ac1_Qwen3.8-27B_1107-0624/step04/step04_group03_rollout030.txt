**Cycling Algorithm:**

The key idea: instead of using a smoothed approximation of `max(c)`, we explicitly cycle through the active lags. At each step, we pick the lag with the current maximum and take a gradient step to reduce it. This forces the maximum to "move" to a different lag, and over time, the profile flattens.

- Use momentum to smooth out the oscillations.
- Use a diminishing step size.
- Periodically evaluate and only keep the best solution.

from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1

    fa = np.fft.rfft(a, N)

    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    s = v.sum()
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad_lag(a, m):
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, m - n + 1)
    j_hi = min(n - 1, m)
    if j_lo <= j_hi:
        idx = np.arange(j_lo, j_hi + 1)
        grad[idx] = 2.0 * a[m - idx]
    return grad


def _smoothed_grad(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    w_conv = _full_convolve(w, a[::-1])
    grad = (2.0 / beta) * w_conv[n - 1:2 * n - 1]
    return grad


def _smoothed_obj(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    logsum = float(np.log(np.sum(np.exp(shifted))))
    return 2.0 * n * (cm + beta * logsum)


def _cycling_optimize(a_init, deadline, k_avg=1, lr_init=0.02, mom_init=0.85):
    """Cycling subgradient with momentum and backtracking on the simplex."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)

    c = _full_convolve(a, a)
    cm = float(c.max())

    moment = np.zeros(n)
    beta_mom = mom_init
    lr = lr_init
    min_lr = 1e-9
    max_lr = 0.1

    for it in range(500000):
        if time.monotonic() > deadline:
            break

        if k_avg == 1:
            m_star = int(np.argmax(c))
            g = _subgrad_lag(a, m_star)
        else:
            top_idx = np.argsort(c)[::-1][:k_avg]
            g = np.zeros(n)
            for m in top_idx:
                g += _subgrad_lag(a, m)
            g /= k_avg

        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        moment = beta_mom * moment - lr * g / gnorm
        a_new = _simplex_project(a + moment)

        c_new = _full_convolve(a_new, a_new)
        cm_new = float(c_new.max())

        if cm_new < cm:
            a = a_new
            c = c_new
            cm = cm_new
            sc = 2.0 * n * cm
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            lr = min(max_lr, lr * 1.05)
        else:
            lr = max(min_lr, lr * 0.7)
            moment = beta_mom * moment

        if it % 500 == 0:
            beta_mom = min(0.995, beta_mom + 0.002)
            sc_check = _score_norm(a)
            if sc_check < best_score:
                best_score = sc_check
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _nesterov_smoothed(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    """Nesterov-accelerated gradient descent on the smoothed objective."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)

    y = a.copy()
    lr = lr_init
    mom = mom_init
    t = 0

    c = _full_convolve(a, a)
    max_c = float(c.max())

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-10, beta_frac * max_c)

        iters_in_phase = 0
        while time.monotonic() < deadline:
            t += 1
            iters_in_phase += 1

            a_y = _simplex_project(y)
            grad = _smoothed_grad(a_y, beta)
            if not np.all(np.isfinite(grad)):
                grad = _smoothed_grad(a, beta)
            grad = grad - grad.mean()

            a_new = _simplex_project(a - lr * grad)
            y_new = a_new + mom * (a_new - a)

            a = a_new
            y = y_new

            if t % 40 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if t % 200 == 0:
                mom = min(0.998, mom + 0.001)
                if iters_in_phase > 5000:
                    break

        a_final = _simplex_project(a)
        c = _full_convolve(a_final, a_final)
        max_c = float(c.max())
        lr = max(0.003, lr * 0.75)
        a = a_final

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _adam_smoothed(a_init, deadline, lr=0.03, beta_frac_init=0.01, max_iter=200000):
    """Adam optimizer on the smoothed objective."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta_frac = beta_frac_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        beta = max(1e-10, beta_frac * max_c)

        grad = _smoothed_grad(a, beta)
        if not np.all(np.isfinite(grad)):
            continue
        grad = grad - grad.mean()

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        a_new = _simplex_project(a - lr * mh / (np.sqrt(vh) + eps))
        a = a_new

        if t % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta_frac *= 0.85
            c = _full_convolve(a, a)
            max_c = float(c.max())
            lr *= 0.98

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _lbfgs_phase(a_init, deadline):
    """L-BFGS with continuation on smoothed objective (using log-parametrization)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    def softmax_fn(z):
        z = z - z.max()
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300:
            return np.ones_like(z) / len(z)
        return e / s

    def obj(z):
        a = softmax_fn(z)
        n_inner = len(a)
        c = _full_convolve(a, a)
        cm = float(c.max())
        max_c_ref = 2.0
        beta = 0.01 * max_c_ref
        shifted = (c - cm) / beta
        logsum = float(np.log(np.sum(np.exp(shifted))))
        return 2.0 * n_inner * (cm + beta * logsum)

    def grad_fn(z):
        a = softmax_fn(z)
        n_inner = len(a)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = 0.01 * 2.0
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        w_conv = _full_convolve(w, a[::-1])
        h = (2.0 / beta) * w_conv[n_inner - 1:2 * n_inner - 1]
        return a * (h - h.mean())

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    betas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    betas = [b * max_c0 for b in betas]

    per_level = max(1.0, (deadline - time.monotonic()) / len(betas))

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)

        def obj_b(z):
            a = softmax_fn(z)
            n_inner = len(a)
            c = _full_convolve(a, a)
            cm = float(c.max())
            shifted = (c - cm) / beta
            logsum = float(np.log(np.sum(np.exp(shifted))))
            return 2.0 * n_inner * (cm + beta * logsum)

        def grad_b(z):
            a = softmax_fn(z)
            n_inner = len(a)
            c = _full_convolve(a, a)
            cm = float(c.max())
            shifted = (c - cm) / beta
            e = np.exp(shifted)
            s = float(e.sum())
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            w_conv = _full_convolve(w, a[::-1])
            h = (2.0 / beta) * w_conv[n_inner - 1:2 * n_inner - 1]
            return a * (h - h.mean())

        try:
            result = minimize(
                obj_b, z,
                method='L-BFGS-B', jac=grad_b,
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxls': 20, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = softmax_fn(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = softmax_fn(result.x) * 0 + result.x
        except Exception:
            pass

    return best_a, best_score


def _active_set_polish(a_init, deadline, top_frac=0.005):
    """Polish focusing on the top fraction of lags."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_norm(a)

    lr = 0.005
    moment = np.zeros(n)
    mom_coeff = 0.9
    c_len = 2 * n - 1
    n_active = max(1, int(c_len * top_frac))

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = cm - 0.001 * cm
        active = c >= threshold
        n_act = int(np.sum(active))
        if n_act < 1:
            active[int(np.argmax(c))] = True
            n_act = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        w_conv = _full_convolve(w, a[::-1])
        g = 2.0 * w_conv[n - 1:2 * n - 1]
        g = g - g.mean()

        moment = mom_coeff * moment - lr * g
        a_new = _simplex_project(a + moment)

        c_new = _full_convolve(a_new, a_new)
        if c_new.max() <= cm * 1.001:
            a = a_new
        else:
            lr = max(1e-8, lr * 0.85)

        if it % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            lr = max(1e-8, lr * 0.995)

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _make_ramp(n, steepness=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(steepness * (1.0 - t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_expon(n, rate=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_two_peak(n, pos1=0.3, pos2=0.7, width=0.1):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - pos1) / width) ** 2) + np.exp(-((t - pos2) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, a1=1.5, a2=0.3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        p = p0.copy()
    else:
        x0 = np.linspace(0.0, 1.0, n0)
        x1 = np.linspace(0.0, 1.0, n_new)
        p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining:
    - L-BFGS continuation (warm start)
    - Nesterov-accelerated smoothed gradient with adaptive beta
    - Cycling exact subgradient (key phase for fine improvement)
    - Adam with fine smoothing
    - Active-set polish
    - Multi-restart with structured perturbations
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 60.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_p = _score_norm(parent)
        print(f"Parent: n={n}, score={sc_p:.10f}")
    else:
        n = 4096
        parent = _make_ramp(n, 3.0)
        sc_p = _score_norm(parent)
        print(f"No parent, using ramp n={n}, score={sc_p:.10f}")

    update_best(parent, sc_p, "parent")

    # --- Phase 0: Quick structured candidates (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    structured = [
        (_make_ramp(n, 2.0), "ramp2"),
        (_make_ramp(n, 4.0), "ramp4"),
        (_make_expon(n, 3.0), "expon3"),
        (_make_two_peak(n, 0.25, 0.75, 0.12), "twopeak"),
        (_make_quadratic(n, 1.0, 0.2), "quad"),
    ]
    for p, lbl in structured:
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        if sc < best_score:
            update_best(p, sc, lbl)

    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: L-BFGS continuation (150s) ---
    phase1_end = min(deadline, t_start + 180.0)
    if best_seq is not None and time.monotonic() < phase1_end:
        p_best, sc_best = _lbfgs_phase(best_seq, phase1_end)
        update_best(p_best, sc_best, "lbfgs")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov with adaptive beta (300s) ---
    phase2_end = min(deadline, t_start + 480.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002,
                      0.0001, 5e-5, 2e-5, 1e-5]
        p_best, sc_best = _nesterov_smoothed(best_seq, phase2_end,
                                             beta_fracs, lr_init=0.015, mom_init=0.9)
        update_best(p_best, sc_best, "nesterov")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Cycling exact subgradient (300s) ---
    phase3_end = min(deadline, t_start + 780.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        # Try different k_avg values
        for k_avg in [1, 2, 3]:
            if time.monotonic() > phase3_end - 30:
                break
            per_dl = min(phase3_end - 10, time.monotonic() + 90.0)
            p_best, sc_best = _cycling_optimize(best_seq, per_dl,
                                                k_avg=k_avg, lr_init=0.01,
                                                mom_init=0.85 + 0.05 * (k_avg - 1))
            update_best(p_best, sc_best, f"cycling_k{k_avg}")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Adam with fine smoothing (250s) ---
    phase4_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(200.0, phase4_end - time.monotonic() - 10)
        if per_time > 20:
            p_best, sc_best = _adam_smoothed(best_seq,
                                             min(phase4_end - 5, time.monotonic() + per_time),
                                             lr=0.02, beta_frac_init=0.005)
            update_best(p_best, sc_best, "adam_fine")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Active-set polish (200s) ---
    phase5_end = min(deadline, t_start + 1230.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        for frac in [0.002, 0.005, 0.01]:
            if time.monotonic() > phase5_end - 20:
                break
            per_dl = min(phase5_end - 10, time.monotonic() + 60.0)
            p_best, sc_best = _active_set_polish(best_seq, per_dl, top_frac=frac)
            update_best(p_best, sc_best, f"active_{frac}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Multi-restart with structured perturbations (250s) ---
    phase6_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_restarts = 5
        per_restart = min(50.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break

            if i == 0:
                # Gaussian noise
                noise_level = 0.05
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z + noise_level * rng.normal(size=len(best_seq))
                p_start = _simplex_project(z)
            elif i == 1:
                # Blend with ramp
                p_start = 0.7 * best_seq + 0.3 * _make_ramp(len(best_seq), 3.0)
                p_start = _simplex_project(p_start)
            elif i == 2:
                # Blend with exponential
                p_start = 0.7 * best_seq + 0.3 * _make_expon(len(best_seq), 2.0)
                p_start = _simplex_project(p_start)
            elif i == 3:
                # Large structured perturbation
                noise = rng.normal(size=len(best_seq)) * 0.1
                p_start = _simplex_project(best_seq + noise)
            else:
                # Mirror and blend
                mirror = best_seq[::-1]
                p_start = _simplex_project(0.6 * best_seq + 0.4 * mirror)

            per_dl = min(phase6_end - 5, time.monotonic() + per_restart)
            if per_dl - time.monotonic() < 5:
                continue

            # Quick Nesterov on the perturbed start
            beta_fracs_short = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
            p_best, sc_best = _nesterov_smoothed(p_start, per_dl,
                                                 beta_fracs_short, lr_init=0.01, mom_init=0.88)
            update_best(p_best, sc_best, f"restart{i}")

            # Also try cycling on the perturbed start if time allows
            if time.monotonic() < phase6_end - 10:
                per_dl2 = min(phase6_end - 5, time.monotonic() + 20.0)
                p_best2, sc_best2 = _cycling_optimize(p_best, per_dl2,
                                                      k_avg=1, lr_init=0.005,
                                                      mom_init=0.9)
                update_best(p_best2, sc_best2, f"restart{i}_cycle")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine polish (remaining time) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 60:
            # Very fine Nesterov
            per1 = min(50.0, remaining - 40)
            beta_fracs_tiny = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
            p_best, sc_best = _nesterov_smoothed(best_seq,
                                                 min(phase7_end - 40, time.monotonic() + per1),
                                                 beta_fracs_tiny, lr_init=0.005, mom_init=0.95)
            update_best(p_best, sc_best, "final_nesterov")

        if time.monotonic() < phase7_end - 30:
            # Fine cycling
            per2 = min(40.0, phase7_end - time.monotonic() - 20)
            if per2 > 10:
                p_best, sc_best = _cycling_optimize(best_seq,
                                                    min(phase7_end - 20, time.monotonic() + per2),
                                                    k_avg=1, lr_init=0.003, mom_init=0.95)
                update_best(p_best, sc_best, "final_cycling")

        if time.monotonic() < phase7_end - 15:
            # Adam ultra-fine
            per3 = min(30.0, phase7_end - time.monotonic() - 10)
            if per3 > 5:
                p_best, sc_best = _adam_smoothed(best_seq,
                                                 min(phase7_end - 10, time.monotonic() + per3),
                                                 lr=0.005, beta_frac_init=0.001)
                update_best(p_best, sc_best, "final_adam")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_ramp(4096, 3.0)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite, internal: {sc_final:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(4096, 3.0)
    sc_fb = _score_full(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
