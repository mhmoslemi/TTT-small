"""
Refined multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive logsumexp continuation, power mean smoothing, peak-directed local search,
and length exploration. Key improvements over previous: adaptive temperature scheduling,
peak-directed perturbations, and more aggressive fine-tuning.
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _lse_objective(z, n, T):
    """Logsumexp-smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + T * np.log(s))

def _lse_gradient(z, n, T):
    """Gradient of logsumexp-smoothed objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # h[j] = sum_k w[k] * a[k-j] (cross-correlation)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    grad_a = 4.0 * n * h
    grad_z = a * (grad_a - np.dot(grad_a, a))
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z

def _power_mean_objective(z, n, p):
    """Power mean smoothed objective."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)
    pm = float(np.sum(c**p))**(1.0/p)
    return 2.0 * n * pm

def _power_mean_gradient(z, n, p):
    """Gradient of power mean smoothed objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)
    c_p = c**p
    sp = float(np.sum(c_p))
    if sp < 1e-200:
        return np.zeros_like(z)
    w = c**(p-1) / sp  # normalized weights
    # Actually for power mean: dM_p/da = 2 * S_p^{1/p-1} * sum_k c_k^{p-1} * a_{k-j}
    # So w_k = c_k^{p-1} and scale = 2 * S_p^{1/p-1}
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    scale = 4.0 * n * sp**(1.0/p - 1.0)
    grad_a = scale * h
    grad_z = a * (grad_a - np.dot(grad_a, a))
    if not np.all(np.isfinite(grad_z)):

    return grad_z

def _peak_directed_search(a_init, deadline, rng, n_trials=5000):
    """Local search that targets the maximum lag of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        k_max = int(np.argmax(c))
        
        # Find which a_i contribute most to lag k_max
        contributions = np.zeros(n)
        lo = max(0, k_max - (n-1))
        hi = min(n-1, k_max)
        for j in range(lo, hi+1):
            contributions[j] = a[j] * a[k_max - j]
        
        # Select top contributing indices
        top_frac = min(0.3, max(0.05, 10.0 / n))
        n_top = max(1, int(n * top_frac))
        top_idx = np.argsort(contributions)[-n_top:]
        
        # Delay the top contributors (reduce them)
        a_trial = a.copy()
        reduce_factor = rng.uniform(0.7, 0.95)
        a_trial[top_idx] *= reduce_factor
        a_trial = np.maximum(a_trial, 1e-15)
        
        # Redistribute to bottom contributors of the same lag
        bottom_idx = np.argsort(contributions)[:n_top]
        mass_moved = a[top_idx].sum() - a_trial[top_idx].sum()
        if mass_moved > 0:
            # Distribute to positions that don't contribute to k_max
            # or contribute less
            all_idx = np.arange(n)
            mask = np.ones(n, dtype=bool)
            mask[top_idx] = False
            candidates = all_idx[mask]
            if len(candidates) > 0:
                # Prefer positions with low contribution to k_max
                contrib_candidates = contributions[candidates]
                n_add = min(len(candidates), n_top * 2)
                add_idx = candidates[np.argsort(contrib_candidates)[:n_add]]
                a_trial[add_idx] += mass_moved / len(add_idx)
        
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()
        
        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < 0.05 and sc < best_score * 1.005:
            a = a_trial.copy()
    
    return best_a, best_score

def _run_lbfgs_lse(a_init, deadline, T_starts, max_wall_time=None):
    """L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if max_wall_time is not None:
        per_level = max(5.0, max_wall_time / len(T_starts))
    else:
        per_level = 30.0
    
    for T_frac in T_starts:
        if time.monotonic() > deadline - 2.0:
            break
        T = T_frac * max_c0
        level_deadline = min(deadline, time.monotonic() + per_level)
        
        def obj(z_):
            return _lse_objective(z_, n, T)
        
        def grad_fn(z_):
            return _lse_gradient(z_, n, T)
        
        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 30, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax(z_new)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _run_lbfgs_power_mean(a_init, deadline, p_list, max_wall_time=None):
    """L-BFGS with power mean continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

        per_level = max(5.0, max_wall_time / len(p_list))
    else:
        per_level = 30.0
    
    for p in p_list:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        
        def obj(z_):
            return _power_mean_objective(z_, n, p)
        
        def grad_fn(z_):

                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 30, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax(z_new)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()

        except Exception:
            pass
    
    return best_a, best_score

def _adam_optimize(a_init, deadline, rng, lr=0.01, T=0.01, k_mode='lse'):
    """Adam optimization with logsumexp or power mean gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    T_abs = T * max_c0
    
    for it in range(500000):
        if time.monotonic() > deadline - 1.0:
            break
        t += 1
        
        if k_mode == 'lse':
            grad = _lse_gradient(z, n, T_abs)
        else:
            grad = _power_mean_gradient(z, n, 10)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        if t % 50 == 0:
            a_new = _softmax(z)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        # Adaptive LR decay
        if t % 5000 == 0:
            lr *= 0.9
    
    return best_a, best_score

def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_gaussian_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u**2 / (2.0 * width**2))
    return a / a.sum()

def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
