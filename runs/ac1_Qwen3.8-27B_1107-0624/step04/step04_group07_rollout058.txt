"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using
Adam with Nesterov momentum, adaptive smooth-max, coordinate descent,
and multi-scale escalation.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _score(a):
    n = len(a)
    c = _full_convolve(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))

def _score_normalized(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _cross_corr(w, a):
    """Compute cross-correlation: for each j, sum_k w[k]*a[k-j] (valid range)"""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]

def _smooth_max_grad(c, beta):
    """Returns (smooth_max_value, weights) for smooth max of c with temperature beta"""
    mval = c.max()
    x = (c - mval) / beta
    ex = np.exp(x)
    s = ex.sum()
    w = ex / s
    return mval + beta * np.log(s), w

def _adam_nesterov(a, deadline, rng, lr=0.03, beta_init=None,

def _coord_descent_precise(a, deadline, max_sweeps=20):
    """Coordinate descent with ternary search for optimal value"""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            v = a[j]
            # Try to find better value via ternary search in [v*0.3, v*3]
            lo, hi = v * 0.3, v * 3.0
            for _ in range(8):  # 8 iterations of ternary search
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                a[j] = m1
                s1 = _score_normalized(a)
                a[j] = m2
                s2 = _score_normalized(a)
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1
            a[j] = (lo + hi) / 2
            s_new = _score_normalized(a)
            if s_new < best_score - 1e-12:
                best_score = s_new
                best_a = a.copy()
                improved = True
            else:
                a[j] = v
        
        if not improved:
            break
    
    return best_a, best_score

def _lp_primal(a, M, n):
    """LP to find b maximizing sum(b) subject to conv(a,b) <= M, b >= 0"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    
    if not rows:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_normalized(a)

        if s_b < 1e-10:

        best_t = 0.5

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)

    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / a_new.sum()
    return a_new

Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using
Adam with adaptive smooth-max, LP fixed-point, coordinate descent with
ternary search, and multi-scale escalation. Key improvements over previous:
- More aggressive temperature cooling for sharper max approximation
- Alternating LP/Adam iterations for better convergence
- Precise coordinate descent via ternary search
- Structured initializations inspired by low-autocorrelation sequence theory

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _cross_corr(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    w = ex / s
    return w


def _adam_optimize(a, deadline, rng, lr=0.04, beta_init=None,

def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_primal(a, M, n):
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

        if s_b < 1e-10:

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / a_new.sum()
    return a_new

    a = real_part + c
    a = np.maximum(a, 1e-10)

        # LP at new scale (only if feasible)
        if tn <= 1024 and time.monotonic() < phase4_end - 30:
            per2 = min(80.0, phase4_end - time.monotonic() - 20)
            if per2 > 20:
                a_le, sc_le = _lp_fixed_point(best_seq, time.monotonic() + per2, max_iters=25)
                if sc_le < best_score:
                    best_seq = a_le.copy()
                    best_score = sc_le
                    print(f"  NEW BEST: {best_score:.10f} n={tn} [esc_LP]")

    # === Phase 5: Coordinate descent ===
    print("Phase 5: Coordinate descent...")
    phase5_end = deadline - 60.0

    if best_seq.size <= 2048 and time.monotonic() < phase5_end:
        per = min(150.0, phase5_end - time.monotonic() - 10)
        if per > 15:
            a_cd, sc_cd = _coord_descent(best_seq, time.monotonic() + per, max_sweeps=12)
            if sc_cd < best_score:
                best_seq = a_cd.copy()
                best_score = sc_cd
                print(f"  NEW BEST: {best_score:.10f} n={best_seq.size} [coord]")

    # === Phase 6: Final polish ===
    print("Phase 6: Final polish...")
    if time.monotonic() < deadline - 15 and best_seq is not None:
        per = deadline - time.monotonic() - 15
        if per > 15:
            if best_seq.size <= 1024:
                a_fp, sc_fp = _lp_fixed_point(best_seq, time.monotonic() + per, max_iters=30)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  NEW BEST: {best_score:.10f} [final_LP]")
            else:
                a_fp, sc_fp = _adam_optimize(best_seq, time.monotonic() + per, rng, lr=0.015)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  NEW BEST: {best_score:.10f} [final_Adam]")

    # Final validation
    print(f"\nFINAL: score={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = np.ones(512) / 512.0

        exact = _score_norm(final)
        print(f"Exact score (normalized): {exact:.10f}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = np.ones(512) / 512.0
    print("FALLBACK n=512 uniform")
    return [float(x) for x in fb]
