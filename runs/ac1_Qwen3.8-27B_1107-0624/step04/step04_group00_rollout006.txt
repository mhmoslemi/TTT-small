def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

            h = w_conv[n-1:2*n-1]
            W = float(np.dot(h, a))
            return 4.0 * n * a * (h - W)
        
        try:
            result = minimize(
                obj_func, z, jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = np.exp(result.x - result.x.max())
                a_new = a_new / a_new.sum()
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception as e:
            print(f"  L-BFGS error at beta={beta:.2e}: {e}")
    
    return best_a, best_score

def _adam_polish(a_init, deadline, beta, lr=0.01):
    """Adam optimizer with small beta for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    while time.monotonic() < deadline:
        t += 1
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = _full_convolve(a, a)
        cm = c.max()
        w = np.exp((c - cm) / beta)
        w = w / w.sum()
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        W = float(np.dot(h, a))
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score
