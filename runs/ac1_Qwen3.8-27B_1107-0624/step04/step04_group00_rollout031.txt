from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

Actually, I think the most impactful change would be to use a **dual approach**: instead of minimizing the primal (the max of the convolution), we can use the fact that:

`max_k c_k = max_{p >= 0, sum(p)=1} sum_k p_k * c_k`

So our problem is:
`min_a max_{p in simplex} sum_k p_k * c_k(a)`

This is a minimax problem. We can use the "game" approach:
- Maintain a distribution `p` over the peak positions
- At each iteration, update `a` to minimize `sum_k p_k * c_k(a)` (which is smooth)
- Update `p` to put more weight on the peaks that are currently high

This is exactly what the log-sum-exp smoothing does (self-adjusting `p`), but we can do it more explicitly:

1. Compute `c = convolve(a, a)`
2. Set `p_k = c_k^alpha / sum(c_j^alpha)` for some large `alpha` (this is the "mirror" of the log-sum-exp)
3. Compute the gradient of `sum_k p_k * c_k` with respect to `z` (the log-softmax parameter)
4. Take a step in the negative gradient direction
5. Repeat

The advantage of using `c_k^alpha` vs `exp(c_k/beta)` is that `c^alpha` might be more stable when the peaks have very different magnitudes.

Actually, `exp(c_k/beta)` with small `beta` is equivalent to `c_k^(1/beta) / (sum c_j^(1/beta) * beta)` approximately (when `c_k` are all positive and `beta` is small). So they're essentially the same.

"""
Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS with log-sum-exp continuation (fine schedule)
2. Direct peak-chasing subgradient with adaptive step size
3. Coordinate descent on peak contributors
4. Multi-restart exploration
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _conv_and_peak(a):
    c = _full_convolve(a, a)
    return c, float(c.max()), int(np.argmax(c))

def _ls_obj(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    # Handle overflow
    max_shifted = float(shifted.max())
    if max_shifted > 50:
        shifted = shifted - max_shifted
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    return 2.0 * n * (cm + beta * log_sum)

def _ls_grad(z, beta, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    max_shifted = float(shifted.max())
    if max_shifted > 50:
        shifted = shifted - max_shifted
    e = np.exp(shifted)
    w = e / float(e.sum())
    W = float(np.dot(w, c))
    # h_i = sum_k w_k * a_{k-i} (valid range)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _run_lbfgs_continuation(a_init, deadline, rng, beta_schedule=None):
    """L-BFGS with fine log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_schedule is None:
        # Very fine schedule
        fractions = [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04,
                    0.03, 0.025, 0.02, 0.015, 0.012, 0.01, 0.008,
                    0.006, 0.005, 0.004, 0.003, 0.0025, 0.002,
                    0.0015, 0.001, 0.0008, 0.0006, 0.0005, 0.0004,
                    0.0003, 0.0002, 0.0001, 0.00005]
        beta_schedule = [f * max_c0 for f in fractions]
    
    for idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        
        remaining = len(beta_schedule) - idx
        time_alloc = min(120.0, (deadline - time.monotonic()) / max(1, remaining))
        level_deadline = time.monotonic() + time_alloc
        
        # For very small beta, limit iterations to save time
        if beta < 1e-5 * max_c0:
            maxiter = 3000
            maxfun = 15000
        elif beta < 1e-4 * max_c0:
            maxiter = 5000
            maxfun = 25000
        else:
            maxiter = 8000
            maxfun = 40000
        
        try:
            result = minimize(
                _ls_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_ls_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 100, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _peak_chasing(a_init, deadline, rng):
    """Direct subgradient method targeting the exact max peak."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Barzilai-Borwein step size
    z_prev = z.copy()
    a_prev = _softmax_fn(z_prev)
    c_prev, cm_prev, k_prev = _conv_and_peak(a_prev)
    
    gamma = 0.1  # Initial step size
    patience = 0
    
    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c, cm, k_star = _conv_and_peak(a)
        
        # Check improvement
        norm_score = 2.0 * n * cm
        if norm_score < best_score:
            best_score = norm_score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 200:
            # Perturb and reset
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()
            patience = 0
            continue
        
        # Compute subgradient at the peak
        # dc_k*/dz_i = 2 * a_i * (a_{k*-i} - c_{k*})
        valid = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        a_at_peak = a[k_star - valid]  # a_{k*-j} for j in valid
        # grad_i = 2 * a_i * (a_{k*-i} - cm) for 0 <= k*-i <= n-1
        idx = k_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[mask] = 2.0 * a[mask] * (a[idx[mask]] - cm)
        
        # Adaptive step size (BB)
        if np.all(np.isfinite(grad)) and np.linalg.norm(grad) > 1e-15:
            z_delta = z - z_prev
            a_delta = a - a_prev
            s_norm = np.dot(z_delta, z_delta)
            y_norm = np.dot(grad, z_delta)
            if s_norm > 1e-20 and abs(y_norm) > 1e-15:
                gamma_bb = s_norm / max(1e-15, abs(y_norm))
                gamma = 0.5 * gamma + 0.5 * np.clip(gamma_bb, 0.001, 10.0)
            
            # Backtracking
            step = gamma
            for _ in range(20):
                z_trial = z - step * grad
                z_trial = np.clip(z_trial, -80, 80)
                z_trial = z_trial - z_trial.mean()
                a_trial = _softmax_fn(z_trial)
                c_trial = _full_convolve(a_trial, a_trial)
                cm_trial = float(c_trial.max())
                if 2.0 * n * cm_trial < norm_score:
                    break
                step *= 0.5
            
            z_prev = z.copy()
            z = z - step * grad
            z = np.clip(z, -80, 80)
            z = z - z.mean()
        
        # Occasionally try a different peak
        if patience % 50 == 25:
            # Find second-highest peak and target it too
            c_sorted = np.sort(c)[::-1]
            if len(c_sorted) > 1 and c_sorted[1] > 0.9 * cm:
                # Multiple peaks are close; use smoothed approach briefly
                beta = 0.05 * cm
                w, _ = _compute_weights(c, beta)
                W = float(np.dot(w, c))
                a_rev = a[::-1]
                w_conv = _full_convolve(w, a_rev)
                h = w_conv[n-1:2*n-1]
                grad = 4.0 * n * a * (h - W)
                if np.all(np.isfinite(grad)) and np.linalg.norm(grad) > 1e-15:
                    z = z - 0.5 * step * grad / (np.linalg.norm(grad) / np.linalg.norm(a) + 1e-15)
                    z = np.clip(z, -80, 80)
                    z = z - z.mean()
    
    return best_a, best_score

def _compute_weights(c, beta):
    cm = float(c.max())
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    if max_s > 50:
        shifted = shifted - max_s
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm

def _coord_descent(a_init, deadline, rng):
    """Coordinate descent on peak-contributing elements."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c, cm, k_star = _conv_and_peak(a)
        norm_score = 2.0 * n * cm
        
        if norm_score < best_score:
            best_score = norm_score
            best_a = a.copy()
        
        # Find top contributors to the peak
        valid_i = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        contributions = a[valid_i] * a[k_star - valid_i]
        top_k = min(20, len(valid_i))
        top_idx = valid_i[np.argsort(contributions)[::-1][:top_k]]
        
        improved = False
        for idx in top_idx:
            # Try reducing this element
            for delta in [0.0005, 0.001, 0.002, 0.005, 0.01]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                c_trial = _full_convolve(a_trial, a_trial)
                sc = float(c_trial.max())
                if 2.0 * n * sc < best_score:
                    best_score = 2.0 * n * sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try increasing small elements (redistribute mass)
            small_idx = np.argsort(a)[:10]
            for idx in small_idx:
                for delta in [0.001, 0.005, 0.01]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    c_trial = _full_convolve(a_trial, a_trial)
                    sc = float(c_trial.max())
                    if 2.0 * n * sc < best_score:
                        best_score = 2.0 * n * sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Try pairwise exchanges
            top_i = top_idx[0] if len(top_idx) > 0 else 0
            bottom_i = int(np.argmin(a))
            for frac in [0.01, 0.02, 0.05]:
                delta = frac * a[top_i]
                z_trial = z.copy()
                # Move mass from top_i to bottom_i in log space
                z_trial[top_i] -= delta * n  # rough adjustment
                z_trial[bottom_i] += delta * n
                a_trial = _softmax_fn(z_trial)
                c_trial = _full_convolve(a_trial, a_trial)
                sc = float(c_trial.max())
                if 2.0 * n * sc < best_score:
                    best_score = 2.0 * n * sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if not improved:
                # Stuck, add noise
                z = z + 0.01 * rng.normal(size=n)
                z = z - z.mean()
    
    return best_a, best_score

def _multi_restart(a_init, deadline, rng, n_restarts=10):

    n = len(a_init)
    z_best = np.log(np.maximum(a_init, 1e-15))

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    time_per_restart = (deadline - time.monotonic()) / max(1, n_restarts)
    
    for r in range(n_restarts):

        # Generate perturbation
        noise_scale = 0.02 + 0.03 * (r / max(1, n_restarts))
        z_pert = z_best + noise_scale * rng.normal(size=n)
        z_pert = np.clip(z_pert, -60, 60)
        z_pert = z_pert - z_pert.mean()
        a_pert = _softmax_fn(z_pert)
        
        sc_pert = _score_norm_fn(a_pert)

"""
Multi-stage optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses fine L-BFGS continuation with log-sum-exp smoothing, direct peak-chasing
subgradient with Barzilai-Borwein steps, coordinate descent, and multi-restart.
Key improvements: finer beta schedule, longer L-BFGS runs, BB step sizes,
and explicit peak-leveling via dual weight tracking.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _conv_peak(a):
    c = _full_convolve(a, a)
    cm = float(c.max())
    k = int(np.argmax(c))
    return c, cm, k


def _ls_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    ms = float(shifted.max())
    if ms > 50:
        shifted = shifted - ms
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _ls_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    ms = float(shifted.max())
    if ms > 50:
        shifted = shifted - ms
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        return np.zeros_like(z)
    w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z_best = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _dual_weight_optimize(a_init, deadline, rng):
    """Optimize using explicit dual weights (game approach)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    alpha = 5.0  # Weight exponent for dual

    for it in range(5000):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c, cm, k_star = _conv_peak(a)
        norm_score = 2.0 * n * cm

        if norm_score < best_score - 1e-12:
            best_score = norm_score
            best_a = a.copy()

        # Update dual weights
        c_pos = np.maximum(c, 1e-30)
        w = c_pos ** alpha / float(np.sum(c_pos ** alpha))

        # Gradient of sum_k w_k * c_k with respect to z
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step
        step = 0.1 / (gnorm * np.linalg.norm(a))
        step = min(step, 0.5)

        z = z - step * grad
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        # Increase alpha slowly (concentrate on dominant peaks)
        alpha *= 1.001

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization: L-BFGS continuation, peak chasing,
    coordinate descent, dual-weight optimization, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 4096
        t = np.linspace(0, 1, n)
        parent = np.exp(1.0 * (t - 0.5))
        parent = parent / parent.sum()
        update_best(parent, _score_norm_fn(parent), "default_exp")

    n_best = len(best_seq)

    # Phase 1: L-BFGS fine continuation (800s)
    phase1_end = min(deadline, t_start + 800.0)
    print(f"Phase 1: L-BFGS continuation ({phase1_end - t_start:.0f}s)")

    p_best, sc_best = _run_lbfgs_continuation(best_seq.copy(), phase1_end, rng)
    update_best(p_best, sc_best, "lbfgs_main")
    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: Peak chasing (250s)
    phase2_end = min(deadline, t_start + 1050.0)
    print(f"Phase 2: Peak chasing ({phase2_end - t_start:.0f}s)")

    if time.monotonic() < phase2_end and best_seq is not None:
        p_best, sc_best = _peak_chasing(best_seq.copy(), phase2_end, rng)
        update_best(p_best, sc_best, "peak_chase")
    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Dual weight optimization (150s)
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"Phase 3: Dual weight ({phase3_end - t_start:.0f}s)")

    if time.monotonic() < phase3_end and best_seq is not None:
        p_best, sc_best = _dual_weight_optimize(best_seq.copy(), phase3_end, rng)
        update_best(p_best, sc_best, "dual_weight")
    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Coordinate descent (150s)
    phase4_end = min(deadline, t_start + 1350.0)
    print(f"Phase 4: Coord descent ({phase4_end - t_start:.0f}s)")

    if time.monotonic() < phase4_end and best_seq is not None:
        p_best, sc_best = _coord_descent(best_seq.copy(), phase4_end, rng)
        update_best(p_best, sc_best, "coord_desc")
    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Multi-restart (200s)
    phase5_end = min(deadline, t_start + 1550.0)
    print(f"Phase 5: Multi-restart ({phase5_end - t_start:.0f}s)")

    if time.monotonic() < phase5_end and best_seq is not None:
        p_best, sc_best = _multi_restart(best_seq.copy(), phase5_end, rng, n_restarts=20)
        update_best(p_best, sc_best, "multi_restart")
    print(f"  Phase 5 done: best={best_score:.10f}")

    # Phase 6: Final fine L-BFGS polish (remaining - 30s)
    phase6_end = deadline
    print(f"Phase 6: Final polish ({phase6_end - t_start:.0f}s)")

    if time.monotonic() < phase6_end and best_seq is not None:
        c = _full_convolve(best_seq, best_seq)
        max_c = float(c.max())
        fine_betas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                      0.00002, 0.00001, 0.000005]
        fine_betas = [b * max_c for b in fine_betas]
        p_best, sc_best = _run_lbfgs_continuation(best_seq.copy(), phase6_end, rng,
                                                   beta_schedule=fine_betas)
        update_best(p_best, sc_best, "final_polish")

        # One more peak chase round
        if time.monotonic() < phase6_end - 30:
            p_best2, sc_best2 = _peak_chasing(best_seq.copy(), phase6_end - 5, rng)
            update_best(p_best2, sc_best2, "final_peak")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(256) / 256

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite: {sc_eval}")
                # Fallback
                fallback = np.ones(4096) / 4096
                print(f"Fallback score: {_score_fn(fallback):.10f}")
                return [float(x) for x in fallback]
        except Exception as ex:
            print(f"Evaluator error: {ex}")

        return [float(x) for x in final_seq]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
