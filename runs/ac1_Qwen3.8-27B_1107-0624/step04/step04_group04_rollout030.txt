def _next_pow2(n):

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _logsumexp_smooth(a, beta, top_k=0):
    """Logsumexp-smoothed max of autocorrelation."""
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        m = c_top.max()
        return m + beta * np.log(np.sum(np.exp((c_top - m) / beta)))
    else:
        cm = c.max()
        return cm + beta * np.log(np.sum(np.exp((c - cm) / beta)))

def _logsumexp_grad(a, beta, n, top_k=0):
    """Gradient of logsumexp-smoothed max w.r.t. a."""
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        m = c_top.max()
        w = np.zeros_like(c)
        e = np.exp((c_top - m) / beta)
        w[idx] = e / e.sum()
    else:
        cm = c.max()
        e = np.exp((c - cm) / beta)
        w = e / e.sum()
    
    # Same gradient formula as before
    n = len(a)
    g = np.zeros(n)
    for k_idx in np.where(w > 1e-10)[0]:
        k = int(k_idx)
        j_lo = max(0, k - (n-1))

def _lbfgs_smooth(a_init, deadline, beta_start=None, top_k=0):
    """L-BFGS with logsumexp smoothing in logit space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_start is None:
        beta_start = 0.01 * max_c0
    
    max_wall_time = deadline - time.monotonic()
    if max_wall_time <= 0:
        return best_a, best_score
    
    # Beta schedule
    betas = [beta_start * (0.7 ** i) for i in range(15)]
    per_level = max_wall_time / len(betas)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        
        def obj(z):
            a = np.exp(z - z.max())
            a /= a.sum()
            return _logsumexp_smooth(a, beta, top_k)
        
        def grad(z):
            a = np.exp(z - z.max())
            a /= a.sum()
            g = _logsumexp_grad(a, beta, n, top_k)
            gz = a * (g - np.dot(g, a))
            return gz
        
        try:
            result = minimize(obj, z, jac=grad, method='L-BFGS-B',
                            options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-10})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = np.exp(result.x - result.x.max())
                a_new /= a_new.sum()
                sc = _score_norm_fn(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _gradient_descent_multiactive(a, c, lr, n_active_max=20):
    """
    Gradient step targeting the top n_active_max autocorrelation peaks.
    """
    n = len(a)
    idx = np.argsort(c)[-n_active_max:]
    w = np.zeros(2*n-1)
    w[idx] = 1.0 / len(idx)
    
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        j_lo = max(0, k - (n-1))
        j_hi = min(n-1, k)
        g[j_lo:j_hi+1] += w[k] * a[k - j_hi : k - j_lo + 1][::-1]
    g *= 2.0
    
    obj_grad = a * (g - np.dot(g, a))
    z = z - lr * obj_grad
    z = np.clip(z, -80, 80)
    z = z - z.mean()
    a_new = np.exp(z - z.max())
    a_new /= a_new.sum()
    return a_new

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

def _make_random_profile(n, rng, smoothness=0.9):
    """Generate a smooth random profile."""
    # Start with white noise, smooth it
    a = rng.random(n)
    for _ in range(3):
        kernel = np.array([1, 2, 3, 2, 1], dtype=float)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_bipolar_smooth(n, rng):
    """Create a smooth profile that's high at two ends and low in middle (or vice versa)."""
    t = np.linspace(0, 1, n)
    center = rng.uniform(0.3, 0.7)
    width = rng.uniform(0.1, 0.4)
    # Gaussian bump
    a = 1.0 + rng.uniform(0.5, 2.0) * np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

"""
Multi-scale optimization with FFT-based gradients, L-BFGS continuation,
Adam with adaptive smoothing, and active-set subgradient methods
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_from_weights_fft(w, a):
    """
    Compute gradient g_j = 2 * sum_k w_k * a_{k-j} using FFT.
    w has length 2n-1 (autocorrelation lags), a has length n.
    Returns g of length n.
    """
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w) + n - 1
    N = _next_pow2(out_len)
    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(W * Ar, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _grad_from_weights_direct(w, a, active_indices):
    """Direct gradient computation for sparse active sets."""
    n = len(a)
    g = np.zeros(n)
    w_dict = {int(k): w[int(k)] for k in active_indices if w[int(k)] > 1e-14}
    for k, wk in w_dict.items():
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        g[j_lo:j_hi + 1] += wk * a[k - j_hi:k - j_lo + 1][::-1]
    return 2.0 * g


def _compute_weights_full(c, beta):
    """Compute softmax weights over all autocorrelation values."""
    cm = c.max()
    e = np.exp((c - cm) / beta)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _compute_weights_topk(c, beta, top_k):
    """Compute softmax weights over top-k autocorrelation values."""
    if top_k >= len(c):
        return _compute_weights_full(c, beta)
    idx = np.argsort(c)[-top_k:]
    w = np.zeros_like(c)
    c_top = c[idx]
    m = c_top.max()
    e = np.exp((c_top - m) / beta)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w[idx[0]] = 1.0
    else:
        w[idx] = e / s
    return w


def _logsumexp_value(c, beta, top_k=0):
    """Log-sum-exp smoothed max."""
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        m = c_top.max()
        return m + beta * float(np.log(np.sum(np.exp((c_top - m) / beta))))
    else:
        cm = c.max()
        return cm + beta * float(np.log(np.sum(np.exp((c - cm) / beta))))


def _softmax_from_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _z_to_a(z):
    return _softmax_from_z(z)


def _obj_and_grad_z(z, beta, n, top_k=0):
    """Objective and gradient in logit space."""
    a = _softmax_from_z(z)
    c = _full_convolve(a, a)
    val = _logsumexp_value(c, beta, top_k)
    
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_top = c[idx]
        m = c_top.max()
        e = np.exp((c_top - m) / beta)
        s = e.sum()
        w = np.zeros(2 * n - 1)
        if s > 1e-300 and np.isfinite(s):
            w[idx] = e / s
        else:
            w[idx[0]] = 1.0
        g = _grad_from_weights_direct(w, a, idx)
    else:
        w = _compute_weights_full(c, beta)
        if n <= 2048:
            g = _grad_from_weights_fft(w, a)
        else:
            # For large n, use top-k approximation for speed
            top = min(50, 2 * n - 1)
            idx = np.argsort(c)[-top:]
            cw = np.zeros(2 * n - 1)
            cw[idx] = w[idx]
            s_tag = cw[idx].sum()
            if s_tag > 1e-14:
                g = _grad_from_weights_direct(cw, a, idx)
            else:
                g = _grad_from_weights_fft(w, a)
    
    # Transform to z-space: dObj/dz_k = a_k * (g_k - <g, a>)
    gz = a * (g - np.dot(g, a))
    if not np.all(np.isfinite(gz)):
        gz = np.zeros_like(z)
    return val, gz


def _lbfgs_optimize(a_init, deadline, beta_start=None, top_k=0, max_wall_time=None):
    """L-BFGS with beta continuation in logit space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max()) + 1e-15
    
    if beta_start is None:
        beta_start = 0.02 * max_c0
    
    if max_wall_time is None:
        max_wall_time = max(1.0, deadline - time.monotonic())
    
    n_levels = 12
    betas = [beta_start * (0.72 ** i) for i in range(n_levels)]
    per_level = max_wall_time / n_levels
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        elapsed_level = time.monotonic()
        
        def obj_fn(z):
            val, _ = _obj_and_grad_z(z, beta, n, top_k)
            return val
        
        def grad_fn(z):
            _, gz = _obj_and_grad_z(z, beta, n, top_k)
            return gz
        
        try:
            result = minimize(
                obj_fn, z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-11,
                         'maxls': 20, 'maxfun': 800,
                         'iprint': -1}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_from_z(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.04, top_k=0, beta=None,
                   beta_decay=0.93, lr_decay=0.995, lr_floor=0.003):
    """Adam optimizer with adaptive beta in logit space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    if beta is None:
        beta = 0.01 * (float(c0.max()) + 1e-15)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    cur_lr = lr
    best_t = 0
    
    while time.monotonic() < deadline:
        t += 1
        a = _softmax_from_z(z)
        
        c = _full_convolve(a, a)
        val, _ = None, 0
        
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            c_top = c[idx]
            m_c = c_top.max()
            e = np.exp((c_top - m_c) / beta)
            s = e.sum()
            w = np.zeros(2 * n - 1)
            if s > 1e-300:
                w[idx] = e / s
            else:
                w[idx[0]] = 1.0
            g = _grad_from_weights_direct(w, a, idx)
        else:
            w = _compute_weights_full(c, beta)
            if n <= 2048:
                g = _grad_from_weights_fft(w, a)
            else:
                top = min(50, 2 * n - 1)
                idx = np.argsort(c)[-top:]
                cw = np.zeros(2 * n - 1)
                cw[idx] = w[idx]
                g = _grad_from_weights_direct(cw, a, idx)
        
        gz = a * (g - np.dot(g, a))
        if not np.all(np.isfinite(gz)):
            gz = np.zeros(n)
        
        m_buf = b1 * m_buf + (1 - b1) * gz
        v_buf = b2 * v_buf + (1 - b2) * (gz * gz)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        z = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if t % 200 == 0:
            a_chk = _softmax_from_z(z)
            sc = _score_norm_fn(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                best_t = t
            beta = max(1e-16, beta * beta_decay)
            if t - best_t > 2000:
                cur_lr = max(lr_floor, cur_lr * lr_decay)
    
    a_final = _softmax_from_z(z)
    sc = _score_norm_fn(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score

    a_final = _softmax_from_z(z)
    sc = _score_norm_fn(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score


def _kkt_refine(a_init, deadline, max_iter=20000):
    """
    KKT-based refinement: at the optimum, the weighted combination of
    active gradient contributions should be constant across coordinates.
    We iteratively adjust to equalize the "pressure" from active constraints.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_from_z(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Active set: lags within 0.5% of max
        threshold = cm * 0.9995
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Weights proportional to how close to max
        c_active = c[active]
        w_active = (c_active - c_active.min()) / (c_active.max() - c_active.min() + 1e-15)
        w_active = w_active / w_active.sum()
        
        w = np.zeros(2 * n - 1)
        w[active] = w_active
        g = _grad_from_weights_direct(w, a, active)
        
        # Direction: reduce where g > mean, increase where g < mean
        g_mean = np.dot(g, a)
        direction = g - g_mean
        
        # Adaptive step size
        spread = float(np.abs(direction).max()) + 1e-15
        step = 0.02 / spread
        
        z = z - step * direction
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if it % 200 == 0:
            a_chk = _softmax_from_z(z)
            sc = _score_norm_fn(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        
        # Convergence check
        if it > 500 and (best_score - sc if np.isfinite(sc) else 0) < 1e-8:
            break
    
    a_final = _softmax_from_z(z)
    sc = _score_norm_fn(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score


def _interp_profile(p0, n_new):
    """Interpolate a normalized profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def _make_gaussian_profile(n, center, width):
    """Gaussian bump profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 + np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_arcosh_profile(n):
    """Arcoshine-like profile: high at edges, low in middle."""
    t = np.linspace(1e-6, 1 - 1e-6, n)
    a = 1.0 / np.sqrt(t * (1.0 - t))
    return a / a.sum()


def _make_flat_profile(n):
    return np.ones(n) / n


def _make_bell_profile(n, sharpness=2.0):
    """Bell curve: a_i = exp(-sharpness * (2t-1)^2)."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.exp(-sharpness * u ** 2)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Multi-scale initialization with diverse profiles
    2. L-BFGS with FFT-based gradients and beta continuation
    3. Adam with adaptive smoothing parameter
    4. Active set subgradient descent
    5. KKT-based refinement
    6. Multi-restart with structured perturbations
    7. Length exploration
    8. Final intensive polish
    """
    global height_sequence_1
    
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # === Phase 0: Initialization ===
    print("=== Phase 0: Initialization ===")
    
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc = _score_norm_fn(parent)
                update_best(parent, sc, "parent")
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent available: {e}")
    
    # Generate diverse profiles at multiple scales
    n_base = 1024
    profiles_1k = []
    profiles_1k.append(_make_flat_profile(n_base))
    profiles_1k.append(_make_arcosh_profile(n_base))
    profiles_1k.append(_make_bell_profile(n_base, 1.5))
    profiles_1k.append(_make_bell_profile(n_base, 3.0))
    profiles_1k.append(_make_gaussian_profile(n_base, 0.5, 0.2))
    profiles_1k.append(_make_gaussian_profile(n_base, 0.3, 0.15))
    profiles_1k.append(_make_gaussian_profile(n_base, 0.7, 0.15))
    
    if parent is not None:
        profiles_1k.append(_interp_profile(parent, n_base))
    
    # Score profiles
    scored_profiles = []
    for i, p in enumerate(profiles_1k):
        sc = _score_norm_fn(p)
        scored_profiles.append((sc, p, f"prof{i}"))
        update_best(p, sc, f"init_prof{i}")
    
    scored_profiles.sort(key=lambda x: x[0])
    print(f"  Phase 0 best: {best_score:.10f}")
    
    # === Phase 1: Multi-scale optimization ===
    print("=== Phase 1: Multi-scale ===")
    phase1_end = min(deadline, t_start + 150.0)
    
    scales = [128, 256, 512, 1024, 2048, 4096]
    
    # Start from the best profile at smallest scale
    if parent is not None:
        a_scale = _interp_profile(parent, 128)
    else:
        a_scale = scored_profiles[0][1][:128] / scored_profiles[0][1][:128].sum() if len(scored_profiles[0][1]) >= 128 else scored_profiles[0][1].copy()
    
    for n_s in scales:
        if time.monotonic() > phase1_end - 20:
            break
        
        per_scale = max(5.0, min(40.0, (phase1_end - time.monotonic()) / (len(scales) - scales.index(n_s))))
        
        if len(a_scale) != n_s:
            a_scale = _interp_profile(a_scale, n_s)
        
        # Quick optimization at this scale
        dl = min(phase1_end - 10, time.monotonic() + per_scale)
        try:
            p, sc = _lbfgs_optimize(a_scale, dl, beta_start=0.03 * (_full_convolve(a_scale, a_scale).max() + 1e-15), top_k=0)
            if sc < _score_norm_fn(a_scale):
                a_scale = p
            update_best(a_scale, _score_norm_fn(a_scale), f"ms_n{n_s}")
        except Exception:
            pass
    
    # Also try the best 1K profile optimized at 2048
    if time.monotonic() < phase1_end - 30:
        p2k = _interp_profile(scored_profiles[0][1], 2048)
        dl2 = min(phase1_end - 10, time.monotonic() + 40.0)
        try:
            p_opt, sc_opt = _lbfgs_optimize(p2k, dl2, top_k=0)
            update_best(p_opt, sc_opt, "ms_best1k_to2k")
            if sc_opt < _score_norm_fn(a_scale):
                a_scale = p_opt
        except Exception:
            pass
    
    # Upscale to 4096 for main optimization
    if best_seq is not None:
        a_main = _interp_profile(best_seq, 4096)
    elif a_scale is not None:
        a_main = _interp_profile(a_scale, 4096)
    else:
        a_main = _make_flat_profile(4096)
    
    sc_main = _score_norm_fn(a_main)
    update_best(a_main, sc_main, "main_4096")
    print(f"  Phase 1 done: best={best_score:.10f}")
    
    # === Phase 2: L-BFGS continuation ===
    print("=== Phase 2: L-BFGS continuation ===")
    phase2_end = min(deadline, t_start + 550.0)
    
    if best_seq is not None and time.monotonic() < phase2_end:
        n_opt = len(best_seq)
        remaining = phase2_end - time.monotonic()
        
        # First L-BFGS pass: full smoothing
        per1 = min(200.0, remaining * 0.5)
        dl1 = min(phase2_end - 5, time.monotonic() + per1)
        print(f"  L-BFGS full (n={n_opt}, {per1:.0f}s)")
        try:
            p1, sc1 = _lbfgs_optimize(best_seq, dl1, beta_start=None, top_k=0)
            update_best(p1, sc1, "lbfgs_full")
        except Exception as e:
            print(f"  L-BFGS error: {e}")
        
        # Second L-BFGS pass: top-k
        if time.monotonic() < phase2_end - 30:
            per2 = min(150.0, (phase2_end - time.monotonic()) * 0.6)
            dl2 = min(phase2_end - 5, time.monotonic() + per2)
            print(f"  L-BFGS top-20 (n={n_opt}, {per2:.0f}s)")
            try:
                p2, sc2 = _lbfgs_optimize(best_seq, dl2, beta_start=0.001 * (_full_convolve(best_seq, best_seq).max() + 1e-15), top_k=20)
                update_best(p2, sc2, "lbfgs_top20")
            except Exception as e:
                print(f"  L-BFGS2 error: {e}")
    
    print(f"  Phase 2 done: best={best_score:.10f}")
    
    # === Phase 3: Adam optimization ===
    print("=== Phase 3: Adam ===")
    phase3_end = min(deadline, t_start + 950.0)
    
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        
        # Adam pass 1: full smoothing, moderate lr
        per1 = remaining * 0.4
        dl1 = min(phase3_end - 10, time.monotonic() + per1)
        print(f"  Adam full lr=0.04 ({per1:.0f}s)")
        try:
            p1, sc1 = _adam_optimize(best_seq, dl1, lr=0.04, top_k=0)
            update_best(p1, sc1, "adam_full")
        except Exception as e:
            print(f"  Adam1 error: {e}")
        
        # Adam pass 2: top-k, smaller lr
        if time.monotonic() < phase3_end - 20:
            per2 = min(phase3_end - 15, time.monotonic() + (phase3_end - time.monotonic()) * 0.65)
            dl2 = min(phase3_end - 5, time.monotonic() + per2)
            print(f"  Adam top-10 lr=0.02 ({per2:.0f}s)")
            try:
                p2, sc2 = _adam_optimize(best_seq, dl2, lr=0.02, top_k=10)
                update_best(p2, sc2, "adam_top10")
            except Exception as e:
                print(f"  Adam2 error: {e}")
    
    print(f"  Phase 3 done: best={best_score:.10f}")
    
    # === Phase 4: Active set subgradient ===
    print("=== Phase 4: Active set descent ===")
    phase4_end = min(deadline, t_start + 1200.0)
    
    if best_seq is not None and time.monotonic() < phase4_end:
        remaining = phase4_end - time.monotonic()
        
        # Active set descent with small top-k
        per1 = remaining * 0.5
        dl1 = min(phase4_end - 10, time.monotonic() + per1)
        print(f"  Active set (n=4096, {per1:.0f}s)")
        try:
            p1, sc1 = _active_set_descent(best_seq, dl1, n_active_start=3, n_active_max=30, lr0=0.05)
            update_best(p1, sc1, "activeset")
        except Exception as e:
            print(f"  Active set error: {e}")
        
        # KKT refinement
        if time.monotonic() < phase4_end - 15:
            per2 = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic()) * 0.6)
            dl2 = min(phase4_end - 5, time.monotonic() + per2)
            print(f"  KKT refine ({per2:.0f}s)")
            try:
                p2, sc2 = _kkt_refine(best_seq, dl2, max_iter=15000)
                update_best(p2, sc2, "kkt")
            except Exception as e:
                print(f"  KKT error: {e}")
    
    print(f"  Phase 4 done: best={best_score:.10f}")
    
    # === Phase 5: Multi-restart ===
    print("=== Phase 5: Multi-restart ===")
    phase5_end = min(deadline, t_start + 1500.0)
    
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = min(80.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 20:
                break
            
            z_base = np.log(np.maximum(best_seq, 1e-15))
            z_base = z_base - z_base.mean()
            
            if i == 0:
                # Small white noise
                z_pert = z_base + 0.03 * rng.normal(size=n_best)
            elif i == 1:
                # Low-frequency noise
                noise_freq = np.fft.rfft(rng.normal(size=n_best))
                cutoff = n_best // 10
                noise_freq[cutoff:] *= 0.1
                noise = np.fft.irfft(noise_freq, n_best)
                z_pert = z_base + 0.05 * noise
            else:
                # Two-bump perturbation
                z_pert = z_base.copy()
                for _ in range(2):
                    center = rng.integers(n_best // 4, 3 * n_best // 4)
                    width = max(20, n_best // 30)
                    t = np.arange(n_best)
                    bump = 0.08 * np.exp(-((t - center) / width) ** 2)
                    z_pert += bump
            
            z_pert = np.clip(z_pert, -60, 60)
            z_pert = z_pert - z_pert.mean()
            p_start = _softmax_from_z(z_pert)
            
            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            per_lbfgs = per_restart * 0.55
            per_adam = per_restart * 0.35
            
            try:
                p_m, sc_m = _lbfgs_optimize(p_start, min(phase5_end - 5, time.monotonic() + per_lbfgs), top_k=5)
                update_best(p_m, sc_m, f"rs{i}_lbfgs")
            except Exception:
                p_m = p_start
                sc_m = _score_norm_fn(p_start)
            
            if time.monotonic() < phase5_end - 5:
                try:
                    p_a, sc_a = _adam_optimize(p_m, min(phase5_end - 5, time.monotonic() + per_adam), lr=0.03, top_k=8)
                    update_best(p_a, sc_a, f"rs{i}_adam")
                except Exception:
                    pass
    
    print(f"  Phase 5 done: best={best_score:.10f}")
    
    # === Phase 6: Try different lengths ===
    print("=== Phase 6: Length exploration ===")
    phase6_end = min(deadline, t_start + 1700.0)
    
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr != 2048:
            target_ns.append(2048)
        if n_curr != 8192:
            target_ns.append(8192)
        
        for tn in target_ns:
            if time.monotonic() > phase6_end - 50:
                break
            
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")
            
            remaining = phase6_end - time.monotonic()
            if remaining < 25:
                break
            
            per_opt = min(100.0, remaining * 0.5)
            dl = min(phase6_end - 10, time.monotonic() + per_opt)
            try:
                p, sc = _lbfgs_optimize(p_interp, dl, top_k=10)
                update_best(p, sc, f"opt_n{tn}")
            except Exception:
                pass
            
            if time.monotonic() < phase6_end - 15:
                dl2 = min(phase6_end - 10, time.monotonic() + min(50.0, (phase6_end - time.monotonic()) * 0.4))
                try:
                    p2, sc2 = _adam_optimize(p_interp, dl2, lr=0.02, top_k=10)
                    update_best(p2, sc2, f"adam_n{tn}")
                except Exception:
                    pass
    
    print(f"  Phase 6 done: best={best_score:.10f}")
    
    # === Phase 7: Final intensive polish ===
    print("=== Phase 7: Final polish ===")
    
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        
        # Extended Adam with very small lr and large top-k
        if remaining > 30:
            per1 = min(100.0, remaining * 0.35)
            dl1 = min(deadline - 10, time.monotonic() + per1)
            try:
                p1, sc1 = _adam_optimize(best_seq, dl1, lr=0.008, top_k=30, beta_decay=0.96)
                update_best(p1, sc1, "final_adam_small")
            except Exception:
                pass
        
        # Tight L-BFGS
        if time.monotonic() < deadline - 20:
            per2 = min(70.0, (deadline - time.monotonic()) * 0.4)
            dl2 = min(deadline - 5, time.monotonic() + per2)
            beta_tight = 1e-5 * (float(_full_convolve(best_seq, best_seq).max()) + 1e-15)
            try:
                p2, sc2 = _lbfgs_optimize(best_seq, dl2, beta_start=beta_tight, top_k=5)
                update_best(p2, sc2, "final_lbfgs_tight")
            except Exception:
                pass
        
        # One more KKT pass
        if time.monotonic() < deadline - 10:
            dl3 = min(deadline - 5, time.monotonic() + (deadline - time.monotonic() - 5) * 0.5)
            try:
                p3, sc3 = _kkt_refine(best_seq, dl3, max_iter=5000)
                update_best(p3, sc3, "final_kkt")
            except Exception:
                pass
    
    print(f"  Phase 7 done: best={best_score:.10f}")
    
    # === Final validation and return ===
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_flat_profile(2048)
            final_score = _score_fn(final_seq)
        
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 10000:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check: {e}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_flat_profile(2048)
    print(f"  FALLBACK: n=2048, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
