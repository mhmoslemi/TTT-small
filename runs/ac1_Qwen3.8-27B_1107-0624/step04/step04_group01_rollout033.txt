"""
Hybrid Adam optimization with top-k mean loss for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses top-k continuation, multi-restart
diversification, and spectral initializations.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_indices(c, k):
    """Return indices of top-k values in c."""
    if k >= len(c):
        return np.arange(len(c))
    # Use argpartition for efficiency
    part = np.argpartition(c, -k)[-k:]
    return part


def _compute_gradient_topk(a, c, topk_idx, k):
