```python
def active_set_lp(p, M, n_active=50):
    """
    Solve: maximize sum(g) subject to conv(p,g)[k] <= M for active k, g >= 0
    """
    n = len(p)
    c = convolve(p, p)  # current convolution
    # Find active lags
    active_k = np.argsort(c)[::-1][:n_active]
    
    # Build LP: max sum(g) s.t. A @ g <= b, g >= 0
    # A[i, j] = p[(k_i - j) mod n] ... hmm, need to be careful
    # conv(p,g)[k] = sum_j p[j] * g[k-j] (with valid indices)
    # So A[i, j] = p[(k_i - j)] if 0 <= k_i - j < n, else 0
    
    m = len(active_k)
    A = np.zeros((m, n))
    for i, k in enumerate(active_k):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[i, j] = p[k-j]
    b = np.full(m, M)
    
    # Solve LP
    from scipy.optimize import linprog
    c_obj = -np.ones(n)  # minimize -sum(g)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res.x if res.status == 0 else None
```

"""
Active-set LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: uses linear programming to find optimal redistribution directions,
combined with gradient-based polishing for final refinement.
"""
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]

def _compute_conv(a):
    return _full_convolve(a, a)

def _score_norm(a):
    """Score for normalized sequence (sum=1): 2*n*max(conv(a,a))"""
    n = len(a)
    c = _compute_conv(a)
    return 2.0 * n * float(c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _lp_solve(p, M, active_k):
    """Solve LP: max sum(g) s.t. conv(p,g)[k] <= M for k in active_k, g >= 0"""
    n = len(p)
    m = len(active_k)
    A = np.zeros((m, n))
    for i in range(m):
        k = int(active_k[i])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            idx = np.arange(lo, hi)
            A[i, lo:hi] = p[k - idx]
    b = np.full(m, M)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def _lp_iterate(p_init, deadline, n_active_list=None, max_iter_per_phase=500):
    """Iterative LP-based optimization."""
    n = len(p_init)
    p = np.maximum(p_init, 0.0).astype(np.float64)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    
    best_p = p.copy()
    best_score = _score_norm(p)
    
    if n_active_list is None:
        n_active_list = [50, 100, 200, 400]
    
    for n_active in n_active_list:
        if time.monotonic() > deadline:
            break
        
        no_improve = 0
        for it in range(max_iter_per_phase):
            if time.monotonic() > deadline:
                break
            
            c = _compute_conv(p)
            M = float(c.max())
            active_k = np.argsort(c)[::-1][:min(n_active, len(c))]
            
            g = _lp_solve(p, M, active_k)
            if g is None:
                break
            
            g_sum = float(g.sum())
            if g_sum < 1e-10:
                break
            g = g / g_sum
            
            # Check progress
            diff = float(np.abs(g - p).max())
            if diff < 1e-10:
                no_improve += 1
                if no_improve > 10:
                    break
                continue
            
            # Line search
            best_t = 0.0
            best_sc = best_score
            for t in [0.3, 0.5, 0.2, 0.1, 0.7]:
                p_trial = (1.0 - t) * p + t * g
                sc = _score_norm(p_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
            
            if best_t == 0.0:
                no_improve += 1
                if no_improve > 15:
                    break
                continue
            
            p = (1.0 - best_t) * p + best_t * g
            if best_sc < best_score:
                best_score = best_sc
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        print(f"  LP (n_active={n_active}): score={best_score:.10f}")
    
    return best_p, best_score
