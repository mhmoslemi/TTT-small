1. **Iterative water-filling**: Start with a, compute c = conv(a,a). Find the lags where c is largest. Modify a to reduce those lags while not increasing others.

2. **Gradient descent on the smoothed max**: This is what the previous solution does. The log-sum-exp or top-K average provides a smooth approximation to the max.

3. **Spectral approach**: The FFT of a has magnitude |A(f)|. The convolution c = IDFT(|A(f)|^2) (approximately, for circular convolution). For linear convolution, it's more complex but similar. We want |A(f)|^2 to be as flat as possible (white spectrum), which would make c as flat as possible.

Wait, that's not quite right for linear convolution. For circular convolution of a with itself (length N >= 2n-1), the result is IDFT(|A(f)|^2). So if we make |A(f)|^2 flat (i.e., |A(f)| = constant), the circular convolution is flat. But we need linear convolution to be flat.

For linear convolution, if we zero-pad a to length N >= 2n-1 and take the DFT, the first 2n-1 values of IDFT(|A_N(f)|^2) give the linear convolution. Making |A_N(f)|^2 flat would make the linear convolution flat (approximately, for large N).

But we can't freely choose |A(f)| - it's determined by a. The constraint is that a >= 0 and has length n (the rest is zeros in the zero-padded version).

So the problem becomes: find a non-negative sequence of length n such that the DFT of the zero-padded version has as flat a magnitude spectrum as possible.

    k = 1

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _topk_smooth_objective_grad(z, n, k_top, alpha):

    top_idx = np.argsort(c)[-k_actual:]

def _adam_topk(a_init, deadline, rng, lr=0.02, k_top=5, max_iters=100000):
    """Adam optimization with top-K smoothed gradient."""
    n = len(a_init)
    a_init = np.asarray(a_init, dtype=np.float64)
    s = a_init.sum()
    a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        t += 1
        a_curr = _softmax_fn(z)
        _, grad = _topk_smooth_objective_grad(z, n, k_top, 1.0)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if t % 50 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.95)
    
    return best_a, best_score


def _local_search(a_init, deadline, rng, n_trials=10000):
    """Local search with block perturbations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Random block perturbation
        block_size = int(rng.integers(2, min(30, n // 3)))
        start = int(rng.integers(0, n - block_size))
        
        a_trial = a.copy()
        # Multiplicative perturbation with varying intensity
        intensity = rng.uniform(0.05, 0.3)
        perturbation = 1.0 + intensity * rng.normal(size=block_size)
        perturbation = np.maximum(perturbation, 0.1)
        a_trial[start:start + block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()
        
        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif rng.random() < 0.02 and sc < best_score * 1.02:
            # Simulated annealing: occasionally accept worse
            a = a_trial.copy()
    
    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0, phase=0.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * alpha * (t - 0.5) + phase))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width, center=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) ** 2) / (2.0 * width ** 2))
    return a / a.sum()


def _make_sawtooth_nondecreasing(n, n_cycles=1, noise=0.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    # Add a non-symmetric structure
    a = 1.0 + 0.5 * np.sin(2 * np.pi * n_cycles * t)
    if noise > 0:
        a += noise * np.random.default_rng(42).normal(size=n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_dual_peak(n, pos1=0.3, pos2=0.7, width=0.15, asym=0.0):
    """Two peaks with optional asymmetry (for non-symmetric sequences)."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - pos1) ** 2) / (2 * width ** 2)) + \
        (1 + asym) * np.exp(-((t - pos2) ** 2) / (2 * width ** 2))
    return a / a.sum()


def _make_random_smooth(n, rng, smoothness=3):
    """Random smooth sequence by interpolating random points."""
    n_pts = int(smoothness) + 1
    pts = rng.uniform(0.5, 1.5, size=n_pts)
    t = np.linspace(0.0, 1.0, n)
    t_pts = np.linspace(0.0, 1.0, n_pts)
    a = np.interp(t, t_pts, pts)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
