```python
def spectral_flatten(a, n_top=10):
    """Flatten the top n_top spectral peaks."""
    A = np.fft.rfft(a)
    mag = np.abs(A)
    # Find the threshold that would reduce the top n_top peaks to the (n_top+1)-th value
    sorted_mag = np.sort(mag)[::-1]
    T = sorted_mag[n_top]  # threshold
    # Soft clip
    new_mag = np.minimum(mag, T)
    # Preserve phase
    new_A = new_mag * (A / np.abs(A))  # careful with zeros
    new_a = np.fft.irfft(new_A, len(a))
    return np.maximum(new_a, 0)
```
