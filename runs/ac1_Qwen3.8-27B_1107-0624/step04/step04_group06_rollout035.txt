Optimized collision minimization using L-BFGS with power-mean smooth-max,
adaptive homotopy, and peak-shaving heuristics.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Module-level state for objective function
_opt_state = {}

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _cross_corr(w, x):
    """Cross-correlation: result[i] = sum_k w[k] * x[k-i] (valid range)"""
    n = len(x)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fx = np.fft.rfft(x[::-1], N)
    h = np.fft.irfft(fw * fx, N)
    return h[n-1:2*n-1]

def _power_mean(c, p):
    """Compute power mean and its 'weights' (proportional to c^(p-1))."""
    c = np.maximum(c, 1e-30)
    cp = c ** p
    s = cp.sum()
    m = (s / len(c)) ** (1.0 / p)
    w = cp / s  # normalized weights
    return m, w

def _score(x):
    n = len(x)
    c = _full_convolve(x)
    return 2.0 * n * c.max()

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def _obj_grad(z):
    """Objective and gradient for L-BFGS in logit space."""
    n = _opt_state['n']
    p = _opt_state['p']
    
    x = _softmax(z)
    c = _full_convolve(x)
    m, w = _power_mean(c, p)
    f = 2.0 * n * m

    # Simpler: gw = cross_corr(w, x) / len(c)  [since w = c^p / sum(c^p)]
    # Actually, let's use the chain rule more carefully.
    # m = (sum c^p / K)^(1/p)
    # dm/dc_k = (1/p) * (sum c^p / K)^(1/p - 1) * c_k^(p-1) / K
    #         = m^(p-1) / K * c_k^(p-1) / (sum c^p / K)^(p-1) * ... 
    # Let me just use: dm/dc_k = w_k / p * p / m^(p-1) ...

Optimized collision minimization combining:
- L-BFGS with power-mean smooth-max (fast, second-order)
- Projected gradient descent on the probability simplex
- Targeted peak-shaving heuristics
- LP-based refinement
- Multi-scale and multi-initialization search

import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def _score(x):
    n = len(x)
    c = _full_convolve(x)
    return 2.0 * n * float(c.max())

def _cross_corr_valid(u, x):
    """Compute result[i] = sum_k u[k] * x[k-i] for i=0..n-1 where 0<=k-i<n."""
    n = len(x)
    n_u = len(u)

    # conv(u, x_rev)[n-1 : n-1+n]
    x_rev = x[::-1]
    N = _next_pow2(n_u + n - 1)
    fu = np.fft.rfft(u, N)
    fxr = np.fft.rfft(x_rev, N)
    h = np.fft.irfft(fu * fxr, N)
    return h[n-1:n-1+n]

    best_x = x0.copy()

            c_safe = np.maximum(c, 1e-30)
            cp = c_safe ** p
            S = cp.sum()

            return f

            cc = _cross_corr_valid(c_pm1, x)
            g = 4.0 * n * (m ** (p-1)) / K * cc

        sc = _score(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()

def _peak_shave(x, deadline, n_iter=10000):
    """Targeted peak shaving: reduce mass at positions contributing most to peaks."""
    n = len(x)
    x = x.copy()
    x = x / x.sum()
    best_x = x.copy()
    best_score = _score(x)
    rng = np.random.default_rng(42)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(x)
        M = float(c.max())
        score = 2.0 * n * M
        
        if score < best_score - 1e-14:
            best_score = score
            best_x = x.copy()
        
        # Find peak position
        peak_k = int(np.argmax(c))
        
        # Find positions contributing most to this peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        contrib = x[idx] * x[peak_k - idx]
        
        # Source: position with highest contribution
        src = idx[np.argmax(contrib)]
        
        # Target: random position that doesn't contribute to top peaks
        top_peaks = np.argsort(c)[::-1][:min(5, len(c))]
        forbidden = set()
        for pk in top_peaks:
            s = max(0, pk - n + 1)
            e = min(n - 1, pk)
            forbidden.update(range(s, e + 1))
        
        # Also forbid the source's partner
        forbidden.add(peak_k - src)
        
        valid_targets = [i for i in range(n) if i not in forbidden and i != src]
        if not valid_targets:
            valid_targets = list(range(n))
        
        tgt = valid_targets[rng.integers(0, len(valid_targets))]
        
        # Transfer small amount
        amount = x[src] * 0.01
        x[src] -= amount
        x[tgt] += amount
        x = _projected_simplex(x)
        
        # Also try reducing the partner
        partner = peak_k - src
        if 0 <= partner < n and partner != src:
            amount2 = x[partner] * 0.005
            x[partner] -= amount2
            # Distribute to random targets
            x[rng.integers(0, n)] += amount2
            x = _projected_simplex(x)
    
    return best_x, best_score

def _lp_refine(x, deadline, t_init=0.3, max_solves=30):
    """LP-based refinement (improved version)."""
    n = len(x)
    x = x.copy()
    x = x / x.sum()
    best_x = x.copy()
    best_score = _score(x)
    c = _full_convolve(x)
    M = float(c.max())
    t = t_init
    n_solves = 0
    
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        
        # Build LP: maximize sum(b) s.t. (a*b)_k <= M, b >= 0
        # (a*b)_k = sum_{j: 0<=k-j<n} a[k-j] * b[j]
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(x[k - idx].tolist())
        
        A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        res = linprog(-np.ones(n), A_ub=A, b_ub=M * np.ones(2*n-1),
                     bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            M *= 1.01
            continue
        
        b = np.maximum(res.x, 0)
        s_b = b.sum()
        if s_b < 1e-10:
            break
        
        b = b / s_b
        x_new = (1 - t) * x + t * b
        x_new = _projected_simplex(x_new)
        
        sc = _score(x_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_x = x_new.copy()
        
        x = x_new
        c = _full_convolve(x)
        M = max(float(c.max()) * 0.998, best_score / (2*n) * 1.002)
        t = max(0.1, t * 0.93)
    
    return best_x, best_score

    k = 1

def _full_convolve(a):

            best_score = score
            best_x = x.copy()
        peak_k = int(np.argmax(c))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        contrib = x[idx] * x[peak_k - idx]
        src = int(idx[np.argmax(contrib)])
        # Find safe target
        top_peaks = np.argsort(c)[::-1][:3]
        forbidden = set()
        for pk in top_peaks:
            s2 = max(0, pk - n + 1)
            e2 = min(n - 1, pk)
            forbidden.update(range(s2, e2 + 1))
        forbidden.add(src)
        valid = [i for i in range(n) if i not in forbidden]
        if not valid:
            continue
        tgt = int(valid[rng.integers(0, len(valid))])
        amount = x[src] * 0.005
        x[src] -= amount
        x[tgt] += amount
        x = _projected_simplex(x)
    return best_x, best_score

def _lp_refine(x, deadline, t_init=0.25, max_solves=30):
    """LP-based refinement: alternate between fixing x (get M) and fixing M (optimize via LP)."""
    n = len(x)
    x = np.asarray(x, dtype=np.float64).ravel().copy()
    s = x.sum()
    x = x / s
    best_x = x.copy()
    best_score = _score(x)
    c = _full_convolve(x)
    M = float(c.max())
    t = t_init
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                rows.extend([k] * len(idx))
                cols.extend(idx.tolist())
                vals.extend(x[k - idx].tolist())
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=M * np.ones(2 * n - 1),
                         bounds=[(0, None)] * n, method='highs')
        except Exception:
            break
        if not res.success:
            M *= 1.005
            continue
        b = np.maximum(res.x, 0.0)
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b
        x_new = (1.0 - t) * x + t * b
        x_new = _projected_simplex(x_new)
        sc = _score(x_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_x = x_new.copy()
        x = x_new
        c = _full_convolve(x)
        M = max(float(c.max()) * 0.997, best_score / (2.0 * n) * 1.003)
        t = max(0.08, t * 0.92)
    return best_x, best_score

def _make_linear(n, slope=2.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    return a / a.sum()

def _make_cosine(n, freq=1, amp=0.8):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_quadratic(n, curv=2.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + curv * (t - 0.5) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_exponential(n, rate=3.0):
    t = np.linspace(0, 1, n)
    a = np.exp(rate * t)
    return a / a.sum()

def propose_candidate(seed=42, budget_s=1800):
    """Main search: multi-phase optimization combining L-BFGS, projected gradient,
    peak shaving, LP refinement, and multi-scale exploration."""
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0

    best_seq = None
    best_score = np.inf

    def _attempt(x, label):
        nonlocal best_seq, best_score
        if x is None:
            return
        x = np.asarray(x, dtype=np.float64).ravel()
        x = np.maximum(x, 0.0)
        s = float(x.sum())
        if s < 0.01:
            return
        x = x / s
        sc = _score(x)
        if sc < best_score:
            best_score = sc
            best_seq = x.copy()
            print(f"  BEST: {best_score:.10f} n={x.size} [{label}] t={time.monotonic()-t0:.0f}s")

    print(f"Budget={budget_s}s, effective deadline={deadline - t0:.0f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={parent.size}, score={_score(parent):.10f}")
        _attempt(parent, "parent")

    # Phase 1: L-BFGS coarse-to-fine on parent (0-400s)
    p1_end = t0 + 400.0
    print("Phase 1: L-BFGS coarse-to-fine on parent...")
    x_start = parent if parent is not None else _make_linear(1024, 2.0)
    n = x_start.size
    x1, sc1 = _lbfgs_optimize(x_start, n, [5, 10, 20, 50, 100], min(p1_end, deadline), maxiter_per_p=5000)
    _attempt(x1, "lbfgs_coarse")

    # Phase 2: L-BFGS fine (400-700s)
    p2_end = t0 + 700.0
    print(f"Phase 2: L-BFGS fine, best={best_score:.10f}")
    if time.monotonic() < p2_end:
        x2, sc2 = _lbfgs_optimize(best_seq, best_seq.size, [50, 100, 200, 400],
                                   min(p2_end, deadline), maxiter_per_p=10000)
        _attempt(x2, "lbfgs_fine")

    # Phase 3: Projected gradient (700-900s)
    p3_end = t0 + 900.0
    print(f"Phase 3: Projected gradient, best={best_score:.10f}")
    if time.monotonic() < p3_end:
        x3, sc3 = _proj_grad(best_seq, min(p3_end, deadline), p=30, max_iter=100000)
        _attempt(x3, "proj_grad")

    # Phase 4: Peak shaving (900-1100s)
    p4_end = t0 + 1100.0
    print(f"Phase 4: Peak shaving, best={best_score:.10f}")
    if time.monotonic() < p4_end:
        x4, sc4 = _peak_shave(best_seq, min(p4_end, deadline), n_iter=200000)
        _attempt(x4, "peak_shave")

    # Phase 5: LP refinement (1100-1350s)
    p5_end = t0 + 1350.0
    print(f"Phase 5: LP refinement, best={best_score:.10f}")
    if time.monotonic() < p5_end and best_seq.size <= 2048:
        x5, sc5 = _lp_refine(best_seq, min(p5_end, deadline), t_init=0.2, max_solves=25)
        _attempt(x5, "lp_refine")

    # Phase 6: Multi-scale exploration (1350-1550s)
    p6_end = t0 + 1550.0
    print(f"Phase 6: Multi-scale, best={best_score:.10f}")
    if time.monotonic() < p6_end:
        for n_try in [512, 2048]:
            if time.monotonic() > p6_end - 40:
                break
            x0 = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, len(best_seq)), best_seq)
            x0 = np.maximum(x0, 1e-10)
            x0 = x0 / x0.sum()
            per = min(80.0, p6_end - time.monotonic() - 10)
            if per < 20:
                break
            x6, sc6 = _lbfgs_optimize(x0, n_try, [10, 20, 50, 100, 200],
                                       time.monotonic() + per, maxiter_per_p=3000)
            _attempt(x6, f"scale_n{n_try}")

    # Phase 7: Alternative inits (1550-1700s)
    p7_end = t0 + 1700.0
    print(f"Phase 7: Alt inits, best={best_score:.10f}")
    if time.monotonic() < p7_end and best_seq.size <= 2048:
        n_use = best_seq.size
        inits = [
            _make_linear(n_use, 3.0),
            _make_exponential(n_use, 4.0),
            _make_cosine(n_use, 1, 1.2),
            _make_quadratic(n_use, 3.0),
        ]
        per_init = max(10.0, (p7_end - time.monotonic()) / len(inits))
        for i, xin in enumerate(inits):
            if time.monotonic() > p7_end - 5:
                break
            # Blend with current best
            x_blend = 0.3 * xin + 0.7 * best_seq
            x_blend = _projected_simplex(x_blend)
            x7, sc7 = _lbfgs_optimize(x_blend, n_use, [20, 50, 100, 200],
                                       min(time.monotonic() + per_init, p7_end - 5),
                                       maxiter_per_p=2000)
            _attempt(x7, f"alt_{i}")

    # Phase 8: Final intensive polish (1700-deadline)
    p8_end = deadline
    print(f"Phase 8: Final polish, best={best_score:.10f}")
    if time.monotonic() < p8_end - 5 and best_seq is not None:
        remaining = p8_end - time.monotonic() - 5
        if remaining > 10:
            x8, sc8 = _lbfgs_optimize(best_seq, best_seq.size, [200, 500, 1000],
                                       time.monotonic() + remaining, maxiter_per_p=5000)
            _attempt(x8, "final_polish")
        if time.monotonic() < p8_end - 3:
            x8b, sc8b = _peak_shave(best_seq, time.monotonic() + min(30, p8_end - time.monotonic() - 3),
                                     n_iter=50000)
            _attempt(x8b, "final_shave")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_linear(1024, 2.0)
        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fb = _make_linear(1024, 2.0)
    print(f"FALLBACK: n=1024, score={_score(fb):.10f}")
    return [float(x) for x in fb]
