```python
"""
Multi-phase optimizer combining soft-max L-BFGS, Nesterov acceleration,
direct subgradient peak shaving, and spectral refinement for minimizing
2*n*max(convolve(a,a))/sum(a)^2.
Key improvements: direct subgradient peak shaving, multi-peak balancing,
better asymmetric initialization, fixed time management, and aggressive
beta continuation schedules.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _gradient_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _objective_smooth(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_obj(z, beta, n):
    return _objective_smooth(z, beta)


def _lbfgs_grad(z, beta, n):
    return _gradient_smooth(z, beta)


def _run_lbfgs(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))

    betas = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    betas = [b * max_c0 for b in betas]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _lbfgs_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass

    return best_a, best_score


def _nesterov_optimize(a_init, deadline, beta_fracs, lr_init=0.02, mom_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())

    lr = lr_init
    mom = mom_init
    t = 0

    for beta_frac in beta_fracs:
        if time.monotonic() >= deadline:
            break
        beta = max(1e-12, beta_frac * max_c)

        while time.monotonic() < deadline:
            t += 1
            grad_y = _gradient_smooth(y, beta)
            if not np.all(np.isfinite(grad_y)):
                grad_y = _gradient_smooth(z, beta)

            z_new = z - lr * grad_y
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -50, 50)

            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()

            z = z_new
            y = y_new

            if t % 50 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

            if t % 200 == 0:
                mom = min(0.999, mom + 0.0003)

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        lr = max(0.002, lr * 0.7)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        if t % 200 == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            beta = max(1e-12, beta * 0.85)
            c_new = _full_convolve(a, a)
            max_c = float(c_new.max())

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _peak_shaving(a_init, deadline, lr_init=0.001, top_k=5):
    """Direct subgradient method on the nonsmooth max-convolution objective."""
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    best_a = a.copy()
    best_score = _score_fn(a)

    lr = lr_init
    momentum = np.zeros(n)
    mom_coeff = 0.7

    stagnant = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Find top-k peaks
        top_indices = np.argsort(c)[-top_k:]
        top_vals = c[top_indices]
        top_max = top_vals[0]

        # Weight: emphasize peaks close to the maximum
        weights = np.exp((top_vals - top_max) / (0.01 * top_max + 1e-15))
        weights = weights / weights.sum()

        # Compute combined subgradient
        grad_c = np.zeros(n)
        for wi, ki in zip(weights, top_indices):
            k = int(ki)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        grad_c[i] += wi * 2.0 * a[i]
                    else:
                        grad_c[i] += wi * (a[j] + a[i])

        # Full subgradient of score = 2n * cm / s^2
        s2 = s * s
        df = 2.0 * n * grad_c / s2 - 4.0 * n * cm / (s2 * s) * a

        # Project to maintain sum constraint
        df = df - df.mean()

        # Momentum step
        momentum = mom_coeff * momentum - lr * df
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        new_s = float(a_new.sum())
        if new_s > 1e-10:
            a_new = a_new * s / new_s
        else:
            a_new = a.copy()

        sc = _score_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
        else:
            stagnant += 1
            momentum = mom_coeff * momentum
            a_new2 = a + momentum * 0.5
            a_new2 = np.maximum(a_new2, 0.0)
            ns2 = float(a_new2.sum())
            if ns2 > 1e-10:
                a_new2 = a_new2 * s / ns2
            sc2 = _score_fn(a_new2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()
                a = a_new2
                stagnant = 0
            else:
                lr *= 0.95
                mom_coeff = max(0.3, mom_coeff * 0.99)
            if stagnant > 200:
                lr = max(1e-6, lr * 0.5)
                mom_coeff = 0.5
                momentum *= 0.5
                stagnant = 0

        # Periodic print
        if int((time.monotonic() % 30) * 10) == 0 and int(time.monotonic() % 30) != int(time.monotonic() % 30 - 0.1):
            pass

    return best_a, best_score


def _peak_shaving_fft(a_init, deadline, lr_init=0.0005, top_k=10):
    """Faster peak shaving using FFT-based gradient computation."""
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    best_a = a.copy()
    best_score = _score_fn(a)

    lr = lr_init
    momentum = np.zeros(n)
    mom_coeff = 0.8
    stagnant = 0
    last_print = 0.0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Find top-k peaks
        top_indices = np.argsort(c)[-top_k:]
        top_vals = c[top_indices]
        top_max = top_vals[0]

        # Weight by proximity to max
        scale = max(0.005 * top_max, 1e-15)
        weights = np.exp((top_vals - top_max) / scale)
        weights = weights / weights.sum()

        # Compute combined gradient using FFT
        # dc_k/da[i] = a[(k-i) mod n] + a[(i-k) mod n] (circular approx)
        # For linear conv, this is approximate when k is not near edges
        grad_c = np.zeros(n)
        for wi, ki in zip(weights, top_indices):
            k = int(ki)
            # a[k-i] for i in range(n) is a reversed and shifted
            # Use circular shift as approximation
            if k <= n - 1:
                # a[k-i] = reverse(a)[i-k]... let's just do it directly
                shift_fwd = np.zeros(n)
                shift_bwd = np.zeros(n)
                # a[j] where j = k-i, valid when 0<=k-i<n, i.e., k-n<i<=k
                # shift_fwd[i] = a[k-i] if 0<=k-i<n else 0
                for i in range(max(0, k - n + 1), min(k + 1, n)):
                    j = k - i
                    if 0 <= j < n:
                        shift_fwd[i] = a[j]
                # shift_bwd[i] = a[i-k] if 0<=i-k<n else 0  (same as a[(i) mod n] shifted)
                # Actually shift_bwd[i] = a[i-k] for valid i, which is same as shift_fwd but with sign
                # dc_k/da[i] = a[k-i] + a[i-k] (when both valid and k-i != i-k, i.e., 2k != 2i)
                # a[i-k] = a[k-i] when the sequence is symmetric, but in general:
                # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i < k+n
                for i in range(max(0, k), min(k + n, n)):
                    j = i - k
                    if 0 <= j < n:
                        shift_bwd[i] = a[j]

                # dc_k/da[i] = a[k-i] + a[i-k] (with the 2x when k-i = i-k, i.e., i=k/2)
                g = shift_fwd + shift_bwd
                # When i = k/2 (if k is even), we've double-counted, need to fix
                if k % 2 == 0:
                    mid = k // 2
                    if 0 <= mid < n:
                        # The term a[mid]*a[mid] appears in both shift_fwd and shift_bwd
                        # But actually dc_k/da[i] for i=k/2 is 2*a[k/2], not a[k/2]+a[k/2]=2*a[k/2]
                        # So it's already correct!
                        pass
                grad_c += wi * g

        # Score gradient: f = 2n*cm/s^2
        s2 = s * s
        df = 2.0 * n * grad_c / s2 - 4.0 * n * cm / (s2 * s) * a

        # Project to maintain sum
        df = df - df.mean()

        # Momentum
        momentum = mom_coeff * momentum - lr * df
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        new_s = float(a_new.sum())
        if new_s > 1e-10:
            a_new = a_new * s / new_s
        else:
            a_new = a.copy()

        sc = _score_fn(a_new)
        now = time.monotonic()
        if now - last_print > 30:
            print(f"    peak_shaving_fft: score={best_score:.10f}, lr={lr:.2e}, stagnant={stagnant}")
            last_print = now

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
        else:
            stagnant += 1
            if stagnant % 20 == 0:
                lr *= 0.9
            if stagnant > 500:
                lr = max(1e-8, lr * 0.5)
                mom_coeff = max(0.1, mom_coeff * 0.8)
                momentum *= 0.3
                stagnant = 0

    return best_a, best_score


def _multi_peak_balance(a_init, deadline):
    """Iteratively balance the top peaks by direct perturbation."""
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    best_a = a.copy()
    best_score = _score_fn(a)
    last_print = 0.0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_min_top = float(np.partition(c, -10)[-10])  # 10th largest

        # If top peaks are well-separated, just reduce the max
        # If they're close, try to balance them
        top_10 = np.partition(c, -10)[-10:]
        spread = (top_10[0] - top_10[-1]) / (top_10[0] + 1e-15)

        if spread < 0.01:
            # Peaks are balanced; use uniform weight on top 10
            top_indices = np.argsort(c)[-10:]
            w = np.zeros(len(c))
            w[top_indices] = 1.0 / 10.0
        else:
            # Focus more on the highest peaks
            top_indices = np.argsort(c)[-5:]
            vals = c[top_indices]
            w = np.zeros(len(c))
            exp_w = np.exp(-(vals - vals[0]) / (0.01 * vals[0] + 1e-15))
            w[top_indices] = exp_w / exp_w.sum()

        # Compute gradient direction
        grad_c = np.zeros(n)
        for ki in np.where(w > 0)[0]:
            ki = int(ki)
            wi = w[ki]
            for i in range(n):
                j1 = ki - i
                j2 = i - ki
                contrib = 0.0
                if 0 <= j1 < n:
                    contrib += a[j1]
                if 0 <= j2 < n:
                    contrib += a[j2]
                grad_c[i] += wi * contrib

        s2 = s * s
        cm_val = float(c.max())
        df = 2.0 * n * grad_c / s2 - 4.0 * n * cm_val / (s2 * s) * a
        df = df - df.mean()

        # Adaptive step
        lr = 0.0001
        for trial in range(3):
            a_new = a - lr * df
            a_new = np.maximum(a_new, 0.0)
            ns = float(a_new.sum())
            if ns > 1e-10:
                a_new = a_new * s / ns
            sc = _score_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                break
            lr *= 0.5

        else:
            # Couldn't improve, reduce step
            a_new = a - lr * df
            a_new = np.maximum(a_new, 0.0)
            ns = float(a_new.sum())
            if ns > 1e-10:
                a_new = a_new * s / ns
            sc = _score_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

        now = time.monotonic()
        if now - last_print > 30:
            print(f"    multi_peak_balance: score={best_score:.10f}")
            last_print = now

    return best_a, best_score


def _make_asymmetric_profile(n, type_id=0):
    t = np.linspace(0.0, 1.0, n)
    if type_id == 0:
        # Exponential decay
        a = np.exp(-3.0 * t)
    elif type_id == 1:
        # Power law decay
        a = (1.0 - t + 0.01) ** 2.0
    elif type_id == 2:
        # Skewed Gaussian
        a = np.exp(-20.0 * (t - 0.3) ** 2)
    elif type_id == 3:
        # Rising then flat
        a = np.minimum(1.0, 3.0 * t)
    elif type_id == 4:
        # Cosine bell, shifted
        a = np.maximum(0.1, np.cos(np.pi * (t - 0.3)) ** 2)
    elif type_id == 5:
        # Inverse linear
        a = 1.0 / (t + 0.1)
    else:
        a = np.exp(-2.0 * t) * (1.0 + 0.5 * np.sin(4.0 * np.pi * t))
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_symmetric_profile(n, type_id=0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if type_id == 0:
        a = 1.0 + 0.5 * u ** 2
    elif type_id == 1:
        a = 1.0 - 0.5 * u ** 2
    elif type_id == 2:
        a = np.exp(-2.0 * u ** 2)
    elif type_id == 3:
        a = 1.0 + 0.3 * np.cos(3.0 * np.pi * u)
    else:
        a = (1.0 + 0.5 * u) ** 3
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS continuation -> Nesterov -> direct subgradient
    peak shaving -> Adam polishing -> fine restarts -> final peak shaving.
    Key innovation: direct nonsmooth subgradient peak shaving for local refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_fn(parent)
        print(f"Parent: n={n}, score={sc_parent:.10f}")
    else:
        n = 4096
        parent = _make_asymmetric_profile(n, 0)
        sc_parent = _score_fn(parent)
        print(f"No parent, using asymmetric profile n={n}, score={sc_parent:.10f}")

    # --- Phase 0: Quick evaluation of diverse starting points (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    sc_par = _score_norm_fn(parent)
    candidates.append((sc_par, parent.copy(), len(parent), "parent"))
    update_best(parent, sc_par, "parent")

    n_target = len(parent)
    for i in range(12):
        if time.monotonic() > phase0_end:
            break
        p = _make_asymmetric_profile(n_target, i)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_target, f"asym_{i}"))

    for i in range(6):
        if time.monotonic() > phase0_end:
            break
        p = _make_symmetric_profile(n_target, i)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_target, f"sym_{i}"))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, n={len(best_seq)}")
    for i, (sc, p, pn, label) in enumerate(candidates[:5]):
        print(f"    {i}: {sc:.10f} [{label}]")

    # --- Phase 1: L-BFGS continuation on best candidate (250s) ---
    phase1_end = min(deadline, t_start + 280.0)
    if best_seq is not None:
        per_time = min(200.0, phase1_end - time.monotonic())
        if per_time > 20:
            p_best, sc_best = _run_lbfgs(best_seq, phase1_end, max_wall_time=per_time)
            update_best(p_best, sc_best, "lbfgs_phase1")

    # Try second best if time allows
    if time.monotonic() < phase1_end - 50:
        for sc, p, pn, label in candidates[1:3]:
            if time.monotonic() > phase1_end - 20:
                break
            per_time = min(60.0, phase1_end - time.monotonic() - 10)
            if per_time > 20:
                p_best, sc_best = _run_lbfgs(p, min(phase1_end, time.monotonic() + per_time),
                                             max_wall_time=per_time)
                update_best(p_best, sc_best, f"lbfgs_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov with aggressive continuation (300s) ---
    phase2_end = min(deadline, t_start + 580.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        beta_fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                      5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
        p_best, sc_best = _nesterov_optimize(best_seq, phase2_end, beta_fracs,
                                             lr_init=0.012, mom_init=0.9)
        update_best(p_best, sc_best, "nesterov_phase2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Direct subgradient peak shaving (350s) ---
    phase3_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        per_time = min(300.0, phase3_end - time.monotonic())
        if per_time > 30:
            p_best, sc_best = _peak_shaving_fft(best_seq, min(phase3_end - 10,
                                                              time.monotonic() + per_time),
                                                lr_init=0.0003, top_k=8)
            update_best(p_best, sc_best, "peak_shave_phase3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Adam fine polishing (250s) ---
    phase4_end = min(deadline, t_start + 1180.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        per_time = min(200.0, phase4_end - time.monotonic())
        if per_time > 30:
            p_best, sc_best = _adam_optimize(best_seq, min(phase4_end - 10,
                                                           time.monotonic() + per_time),
                                             lr=0.02, beta_frac=0.003)
            update_best(p_best, sc_best, "adam_phase4")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-peak balancing (200s) ---
    phase5_end = min(deadline, t_start + 1380.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        per_time = min(180.0, phase5_end - time.monotonic())
        if per_time > 30:
            p_best, sc_best = _multi_peak_balance(best_seq, min(phase5_end - 10,
                                                                time.monotonic() + per_time))
            update_best(p_best, sc_best, "balance_phase5")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Nesterov restarts (150s) ---
    phase6_end = min(deadline, t_start + 1530.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = len(best_seq)
        n_restarts = 3
        per_restart = min(45.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            noise_level = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase6_end - 5, time.monotonic() + per_restart)
            beta_fracs_short = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
            p_best, sc_best = _nesterov_optimize(p_start, per_dl, beta_fracs_short,
                                                 lr_init=0.008, mom_init=0.85)
            update_best(p_best, sc_best, f"restart{i}")

            # Quick peak shave on restart result
            if time.monotonic() < phase6_end - 15 and sc_best < best_score * 1.001:
                per_dl2 = min(phase6_end - 5, time.monotonic() + 20)
                if per_dl2 > 5:
                    p_bs, sc_bs = _peak_shaving_fft(p_best, per_dl2, lr_init=0.0002, top_k=5)
                    update_best(p_bs, sc_bs, f"restart{i}_shave")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final fine peak shaving (remaining) ---
    phase7_end = deadline
    if best_seq is not None and time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        if remaining > 30:
            # Final Nesterov with tiny beta
            per_time = min(80.0, remaining * 0.4)
            if per_time > 20:
                beta_fracs_final = [5e-6, 2e-6, 1e-6, 5e-7]
                p_best, sc_best = _nesterov_optimize(best_seq, min(phase7_end,
                                                                   time.monotonic() + per_time),
                                                     beta_fracs_final, lr_init=0.003, mom_init=0.95)
                update_best(p_best, sc_best, "final_nesterov")

            # Final peak shaving
            if time.monotonic() < phase7_end - 15:
                per_time2 = min(100.0, phase7_end - time.monotonic() - 10)
                if per_time2 > 20:
                    p_best, sc_best = _peak_shaving_fft(best_seq, phase7_end - 5,
                                                       lr_init=0.0001, top_k=10)
                    update_best(p_best, sc_best, "final_peak_shave")

            # Final multi-peak balance
            if time.monotonic() < phase7_end - 10:
                per_time3 = min(60.0, phase7_end - time.monotonic() - 5)
                if per_time3 > 10:
                    p_best, sc_best = _multi_peak_balance(best_seq, phase7_end - 2)
                    update_best(p_best, sc_best, "final_balance")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if
