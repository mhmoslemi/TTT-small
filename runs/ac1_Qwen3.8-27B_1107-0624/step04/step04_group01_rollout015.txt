"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp continuation, diverse initializations,
multi-restart exploration, and coordinate descent polishing.
Key improvements over previous approach: L-BFGS second-order info,
power-sum objective variant, structured initializations, and adaptive
coordinate descent for final polish.
"""
import numpy as np
import time
import scipy.optimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_h(w, a, N):
    """Compute h[j] = sum_{i=0}^{n-1} w[i+j] * a[i] for j=0..n-1 using FFT."""
    n = len(a)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # Cross-correlation: h = IFFT(conj(FFT(a)) * FFT(w))
    H = np.fft.irfft(np.conj(A) * W, N)[:n]
    return H


def _lse_objective_grad(z, beta, N):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. z."""
    a = _softmax(z)
    n = len(a)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    h = _compute_h(w, a, N)
    grad = 2.0 * a * (h - W)
    obj = cm + beta * np.log(s / beta) if s > 0 else cm
    return obj, grad


def _pow_objective_grad(z, p, N):
    """Power-sum smoothed objective and gradient."""
    a = _softmax(z)
    n = len(a)
    c = _fft_convolve(a, a)
    c_clipped = np.maximum(c, 0)
    cp = c_clipped ** p
    S = float(cp.sum())
    if S < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = cp / S
    W = float(np.dot(w, c))
    h = _compute_h(w, a, N)
    grad = 2.0 * a * (h - W) * p * (S ** (-1.0 / p))
    obj = S ** (1.0 / p)
    return obj, grad


def _lbfgs_optimize(z_init, deadline, beta_schedule, max_iter_per_phase=5000, lr_scale=1.0):
    """L-BFGS optimization with beta continuation in log-sum-exp smoothing."""
    n = len(z_init)
    N = _next_pow2(2 * n - 1)
    z = z_init.copy()
    best_z = z.copy()
    best_obj = np.inf

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        def fun(z_cur):
            obj, grad = _lse_objective_grad(z_cur, beta, N)
            if not np.all(np.isfinite(grad)):
                return obj, np.zeros_like(z_cur)
            return obj, grad

        try:
            res = scipy.optimize.minimize(
                fun, z, method='L-BFGS-B', jac=True,
                options={'maxiter': max_iter_per_phase, 'maxls': 50,
                         'm': 30, 'ftol': 1e-12, 'gtol': 1e-10}
            )
            z = res.x
            obj_val = float(res.fun)
            if np.isfinite(obj_val) and obj_val < best_obj:
                best_obj = obj_val
                best_z = z.copy()
            print(f"    L-BFGS beta={beta:.2e}: obj={obj_val:.8f}, iters={res.nit}, status={res.status} ({res.message})")
        except Exception as ex:
            print(f"    L-BFGS error at beta={beta:.2e}: {ex}")

    # Evaluate final score
    a_final = _softmax(best_z)
    sc = _score(a_final)
    return best_z, sc


def _lbfgs_pow_optimize(z_init, deadline, p_schedule, max_iter_per_phase=5000):
    """L-BFGS optimization with power-sum continuation."""
    n = len(z_init)
    N = _next_pow2(2 * n - 1)
    z = z_init.copy()

        def fun(z_cur):
            obj, grad = _pow_objective_grad(z_cur, p, N)
            if not np.all(np.isfinite(grad)):
                return obj, np.zeros_like(z_cur)
            return obj, grad

        try:
            res = scipy.optimize.minimize(
                fun, z, method='L-BFGS-B', jac=True,
                options={'maxiter': max_iter_per_phase, 'maxls': 50,
                         'm': 30, 'ftol': 1e-12, 'gtol': 1e-10}
            )
            z = res.x
            obj_val = float(res.fun)
            if np.isfinite(obj_val) and obj_val < best_obj:
                best_obj = obj_val
                best_z = z.copy()
            print(f"    L-BFGS pow p={p:.0f}: obj={obj_val:.8f}, iters={res.nit}")
        except Exception as ex:
            print(f"    L-BFGS pow error at p={p:.0f}: {ex}")

    a_final = _softmax(best_z)
    sc = _score(a_final)
    return best_z, sc


def _coord_descent_polish(a, deadline, max_passes=5):
    """Coordinate descent: optimize each coordinate using scalar minimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(a.sum())
    a = a / s  # normalize
    best_score = _score(a)
    c = _fft_convolve(a, a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        # Only optimize coordinates where a[i] is "significant"
        threshold = 0.001
        active = np.where(a > threshold)[0]
        # Shuffle active indices for diversity
        rng_cd = np.random.default_rng(42 + p)
        idx = active.copy()
        rng_cd.shuffle(idx)

        for i in idx:
            if time.monotonic() > deadline - 10:
                break
            x0 = a[i]
            if x0 < 1e-10:
                continue

            def obj_1d(x):
                a[i] = x
                s_new = s + (x - x0)
                if s_new < 1e-10:
                    a[i] = x0
                    return np.inf
                # Only recompute affected c[k]
                # c[k] changes for k where 0 <= k-i < n (i.e., k in [i, i+n-1])
                # and k where 0 <= k-(i) < n from the other side
                # Actually c[k] = sum_j a[j]*a[k-j], changing a[i] affects:
                # c[k] += (x - x0) * (a[k-i] + a[k-i]) for k where valid, k-i != i
                # c[2i] += (x-x0)*(2*a[i]_old + x - x0) ... this is getting complex
                # Just recompute fully for simplicity (n=2048 is fast enough)
                c_new = _fft_convolve(a, a)
                a[i] = x0  # restore
                return float(2.0 * n * c_new.max() / (s_new * s_new))

            # Use bounded scalar minimization
            try:
                res = scipy.optimize.minimize_scalar(
                    obj_1d, bounds=(0.0, 2.0 * x0 * n),
                    method='bounded', options={'xatol': 1e-10, 'maxiter': 50}
                )
                if res.success and res.x < x0 * 1.01 or (res.x > 0 and res.fun < obj_1d(x0) - 1e-12):
                    a[i] = res.x
                    s = float(a.sum())
                    a = a / s
                    s = 1.0
                    new_sc = _score(a)
                    if new_sc < best_score - 1e-12:
                        best_score = new_sc
                        improved = True
            except Exception:
                pass

        if not improved:
            break
        print(f"    CD pass {p+1}: score={best_score:.10f}")

    a = a / a.sum()
    return a, _score(a)


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        center = 0.5
        a = np.exp(-((t - center) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_peak":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered_sinusoidal":
        t = np.linspace(0, 1, n)
        envelope = np.sin(np.pi * t)
        mod = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        a = envelope * mod
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_z(z, sigma, rng):
    """Perturb in z-space."""
    z_new = z + sigma * rng.standard_normal(len(z))
    z_new = z_new - z_new.mean()
    return z_new


def _perturb_structured_z(z, rng, mode):
    """Apply structured perturbation in z-space."""
    n = len(z)
    z_new = z.copy()
    if mode == "fourier_low":
        freq = rng.integers(1, max(2, n // 8))
        amp = 0.1 * rng.uniform(0.5, 2.0)
        t = np.arange(n)
        z_new = z_new + amp * np.cos(2.0 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    elif mode == "fourier_mid":
        freq = rng.integers(n // 8, max(n // 8 + 1, n // 3))
        amp = 0.05 * rng.uniform(0.5, 2.0)
        t = np.arange(n)
        z_new = z_new + amp * np.cos(2.0 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    elif mode == "shift":
        shift = rng.integers(1, max(2, n // 16))
        z_new = np.roll(z_new, shift)
    elif mode == "scale_local":
        center = rng.integers(0, n)
        width = rng.integers(n // 32, n // 8)
        amp = 0.1 * rng.uniform(0.5, 2.0) * rng.choice([-1, 1])
        t = np.arange(n)
        gaussian = np.exp(-(t - center) ** 2 / (2 * width ** 2))
        z_new = z_new + amp * gaussian
    else:
        z_new = z_new + 0.05 * rng.standard_normal(n)
    z_new = z_new - z_new.mean()
    return z_new


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: L-BFGS with log-sum-exp continuation,
    power-sum variant, diverse initializations, multi-restart,
    and coordinate descent polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            t_el = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={t_el:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        parent = _make_profile(n_main, "uniform")
        sc0 = _score(parent)
        print(f"No parent, using uniform n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "init_uniform")

    # --- Phase 1: Quick exploration of initializations (0-60s) ---
    phase1_end = min(deadline, t_start + 60.0)
    print(f"\n=== Phase 1: Quick exploration ===")

    for kind in ["cosine", "gaussian", "sin_mod", "tapered_sinusoidal", "two_peak"]:
        if time.monotonic() > phase1_end:
            break
        p = _make_profile(n_main, kind)
        sc = _score(p)
        update_best(p, sc, f"init_{kind}")

    # --- Phase 2: L-BFGS continuation from parent (60s - 500s) ---
    phase2_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 2: L-BFGS continuation from parent ===")

    if parent is not None:
        z_parent = np.log(parent)
        z_parent = z_parent - z_parent.mean()

        # Compute initial max for beta
        c_init = _fft_convolve(parent, parent)
        max_c_init = float(c_init.max())

        beta_schedule = [
            0.05 * max_c_init, 0.02 * max_c_init, 0.01 * max_c_init,
            0.005 * max_c_init, 0.002 * max_c_init, 0.001 * max_c_init,
            0.0005 * max_c_init
        ]

        per_time = phase2_end - time.monotonic() - 10
        if per_time > 30:
            z_best, sc_best = _lbfgs_optimize(
                z_parent, time.monotonic() + per_time,
                beta_schedule, max_iter_per_phase=3000
            )
            a_best = _softmax(z_best)
            update_best(a_best, sc_best, "lbfgs_parent")

    # --- Phase 3: Power-sum optimization (500s - 700s) ---
    phase3_end = min(deadline, t_start + 700.0)
    print(f"\n=== Phase 3: Power-sum optimization ===")

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        z_best = np.log(np.maximum(best_seq, 1e-15))
        z_best = z_best - z_best.mean()
        p_schedule = [8, 16, 32, 64, 128, 256]
        per_time = phase3_end - time.monotonic() - 10
        if per_time > 20:
            z_pow, sc_pow = _lbfgs_pow_optimize(
                z_best, time.monotonic() + per_time,
                p_schedule, max_iter_per_phase=2000
            )
            a_pow = _softmax(z_pow)
            update_best(a_pow, sc_pow, "pow_sum")

        # Then switch back to LSE for fine-tuning
        if time.monotonic() < phase3_end - 20:
            c_pw = _fft_convolve(best_seq, best_seq)
            max_cp = float(c_pw.max())
            beta_fine = [0.002 * max_cp, 0.001 * max_cp, 0.0005 * max_cp, 0.0002 * max_cp]
            z_cur = np.log(np.maximum(best_seq, 1e-15))
            z_cur = z_cur - z_cur.mean()
            z_res, sc_res = _lbfgs_optimize(
                z_cur, time.monotonic() + (phase3_end - time.monotonic() - 10),
                beta_fine, max_iter_per_phase=2000
            )
            a_res = _softmax(z_res)
            update_best(a_res, sc_res, "lse_after_pow")

    # --- Phase 4: Multi-restart (700s - 1300s) ---
    phase4_end = min(deadline, t_start + 1300.0)
    print(f"\n=== Phase 4: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        n_restarts = 10
        per_restart = (phase4_end - time.monotonic() - 30) / n_restarts
        z_base = np.log(np.maximum(best_seq, 1e-15))
        z_base = z_base - z_base.mean()

        modes = ["fourier_low", "fourier_mid", "shift", "scale_local", "random"]
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break

            mode = modes[i % len(modes)]
            if mode == "random":
                sigma = 0.03 + 0.03 * (i % 3)
                z_start = _perturb_z(z_base, sigma, rng)
            else:
                z_start = _perturb_structured_z(z_base, rng, mode)

            a_start = _softmax(z_start)
            sc_start = _score(a_start)

            c_s = _fft_convolve(a_start, a_start)
            max_cs = float(c_s.max())
            beta_sched = [0.02 * max_cs, 0.01 * max_cs, 0.005 * max_cs,
                          0.002 * max_cs, 0.001 * max_cs]

            per_t = min(per_restart, phase4_end - time.monotonic() - 15)
            if per_t < 15:
                break
            z_res, sc_res = _lbfgs_optimize(
                z_start, time.monotonic() + per_t,
                beta_sched, max_iter_per_phase=2000
            )
            a_res = _softmax(z_res)
            update_best(a_res, sc_res, f"restart_{i}_{mode}")

    # --- Phase 5: Fine polish (1300s - 1650s) ---
    phase5_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 5: Fine polish ===")

    if best_seq is not None and time.monotonic() < phase5_end - 20:
        c_best = _fft_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        beta_ultra = [0.0005 * max_cb, 0.0002 * max_cb, 0.0001 * max_cb,
                      0.00005 * max_cb, 0.00002 * max_cb]

        z_cur = np.log(np.maximum(best_seq, 1e-15))
        z_cur = z_cur - z_cur.mean()

        per_time = phase5_end - time.monotonic() - 10
        if per_time > 20:
            z_res, sc_res = _lbfgs_optimize(
                z_cur, time.monotonic() + per_time,
                beta_ultra, max_iter_per_phase=3000
            )
            a_res = _softmax(z_res)
            update_best(a_res, sc_res, "ultra_fine_lse")

        # Power-sum with very high p
        if time.monotonic() < phase5_end - 20:
            z_cur2 = np.log(np.maximum(best_seq, 1e-15))
            z_cur2 = z_cur2 - z_cur2.mean()
            p_ultra = [256, 512, 1024]
            per_t2 = phase5_end - time.monotonic() - 10
            if per_t2 > 15:
                z_res2, sc_res2 = _lbfgs_pow_optimize(
                    z_cur2, time.monotonic() + per_t2,
                    p_ultra, max_iter_per_phase=2000
                )
                a_res2 = _softmax(z_res2)
                update_best(a_res2, sc_res2, "ultra_pow")

    # --- Phase 6: Coordinate descent polish (1650s - 1770s) ---
    phase6_end = min(deadline, t_start + 1770.0)
    print(f"\n=== Phase 6: Coordinate descent ===")

    if best_seq is not None and time.monotonic() < phase6_end - 5:
        a_cd, sc_cd = _coord_descent_polish(
            best_seq, time.monotonic() + (phase6_end - time.monotonic() - 3),
            max_passes=3
        )
        update_best(a_cd, sc_cd, "coord_descent")

        # One more L-BFGS pass after CD
        if time.monotonic() < phase6_end - 15 and best_seq is not None:
            c_fin = _fft_convolve(best_seq, best_seq)
            max_cf = float(c_fin.max())
            beta_final = [0.00005 * max_cf, 0.00002 * max_cf]
            z_fin = np.log(np.maximum(best_seq, 1e-15))
            z_fin = z_fin - z_fin.mean()
            per_t = phase6_end - time.monotonic() - 5
            if per_t > 10:
                z_res, sc_res = _lbfgs_optimize(
                    z_fin, time.monotonic() + per_t,
                    beta_final, max_iter_per_phase=2000
                )
                a_res = _softmax(z_res)
                update_best(a_res, sc_res, "final_lbfgs")

    # --- Phase 7: Try larger n if time permits (1770s - deadline) ---
    print(f"\n=== Phase 7: Larger n exploration ===")
    if best_seq is not None and len(best_seq) < 3072 and time.monotonic() < deadline - 30:
        target_n = max(3072, len(best_seq) * 2)
        target_n = min(target_n, 4096)
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score(p_interp)
        print(f"  Interpolated to n={target_n}: score={sc_interp:.10f}")
        update_best(p_interp, sc_interp, f"interp_n{target_n}")

        if time.monotonic() < deadline - 30 and best_seq is not None:
            c_int = _fft_convolve(best_seq, best_seq)
            max_ci = float(c_int.max())
            beta_int = [0.002 * max_ci, 0.001 * max_ci, 0.0005 * max_ci]
            z_int = np.log(np.maximum(best_seq, 1e-15))
            z_int = z_int - z_int.mean()
            per_t = deadline - time.monotonic() - 15
            if per_t > 20:
                z_res, sc_res = _lbfgs_optimize(
                    z_int, time.monotonic() + per_t,
                    beta_int, max_iter_per_phase=2000
                )
                a_res = _softmax(z_res)
                update_best(a_res, sc_res, f"optimize_n{target_n}")

    # --- Final validation ---
    print(f"\n=== Final ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
