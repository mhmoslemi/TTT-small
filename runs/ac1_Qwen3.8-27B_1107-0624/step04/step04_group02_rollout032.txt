"""

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinusoidal_profile(n, freq, amp):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_product_profile(n, slope, freq, amp):
    t = np.linspace(0.0, 1.0, n)
    a_linear = 1.0 + slope * (t - 0.5)
    a_sin = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    a = a_linear * a_sin
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _coordinate_descent(a_init, deadline, max_passes=200, tol=1e-14):
    """
    Exact coordinate descent: for each coordinate i, find the optimal value
    using closed-form solution given the active lag.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score_norm_fn(a)
    c = _full_convolve(a, a)
    max_c = float(c.max())

    for p in range(max_passes):
        if time.monotonic() > deadline - 2.0:
            break

        improved = False
        max_change = 0.0

        for i in range(n):
            v = a[i]
            a[i] = 0.0
            a_no_i = a

            if v > 0:
                s_o = 1.0 - v
            else:
                s_o = 1.0

            c_base = _full_convolve(a_no_i, a_no_i)

            # Compute g_k(x) = c_base[k] + contribution from a[i]=x
            # For k where 0 <= k-i <= n-1: g_k(x) = c_base[k] + 2*x*a_no_i[k-i]
            # For k = 2i (if valid, i.e., 2*i < 2n-1, always true for i < n):
            #   actually k=2i means the self-term: g_{2i}(x) = c_base[2i] + x^2
            #   (because when j=i, k-j = 2i-i = i, so a[i]*a[i] = x^2)
            #   But c_base[2i] already has a[i]=0, so it's correct.

            # Vectorized computation of the "slope" for each lag:
            # slope[k] = 2*a_no_i[k-i] if 0 <= k-i <= n-1, else 0
            # For k = 2i: there's an additional x^2 term

            lo = 0
            hi = n - 1
            # k range: 0 to 2n-2
            # k-i range: -i to 2n-2-i
            # valid k-i: 0 to n-1 => k from i to i+n-1

            slope = np.zeros(2 * n - 1)
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            # k-i ranges from k_lo-i to k_hi-i
            idx_lo = k_lo - i
            idx_hi = k_hi - i
            slope[k_lo:k_hi + 1] = 2.0 * a_no_i[idx_lo:idx_hi + 1]

            # For k = 2*i: add the quadratic term indicator
            k_self = 2 * i
            has_self = k_self < 2 * n - 1

            # Find optimal x by trying the closed form for the currently active lag
            # and checking a few neighboring lags
            current_max_idx = int(np.argmax(c_base + slope * v + (v * v if has_self and k_self < len(c_base) else 0.0)))

            best_x = v
            best_f = 2.0 * n * (c_base + slope * v + (v * v if has_self and k_self < len(c_base) else 0.0)).max()

            # Try closed-form solutions for top few candidate lags
            g_current = c_base + slope * v
            if has_self:
                g_current[k_self] += v * v
            top_kags = np.argpartition(g_current, -min(5, len(g_current)))[-min(5, len(g_current)):]

            for k_star in top_kags:
                k_star = int(k_star)
                slope_val = slope[k_star]
                cb = c_base[k_star]

                if k_star == k_self and has_self:
                    # g(x) = cb + x^2
                    # f(x) = 2n*(cb + x^2)/(s_o + x)^2
                    # optimal: x* = cb / s_o
                    x_candidate = cb / s_o if s_o > 1e-15 else 0.0
                elif slope_val > 1e-15:
                    # g(x) = cb + 2*slope_val/2 * x = cb + slope_val * x
                    # f(x) = 2n*(cb + slope_val*x)/(s_o+x)^2
                    # optimal: x* = s_o - cb/slope_val
                    x_candidate = s_o - cb / slope_val
                else:
                    x_candidate = 0.0

                x_candidate = max(0.0, x_candidate)
                # Clamp to reasonable range
                x_candidate = min(x_candidate, 10.0 * s_o + 1.0)

                # Evaluate f at x_candidate
                g_eval = c_base + slope * x_candidate
                if has_self:
                    g_eval[k_self] += x_candidate * x_candidate
                f_val = 2.0 * n * g_eval.max()
                # Note: since we're working with sum=1, the score is just 2n*max_c
                # But we need to check that sum is still 1: s_o + x_candidate should = 1
                # If v was part of the normalized sequence, s_o + x = 1 automatically

                if f_val < best_f:
                    best_f = f_val
                    best_x = x_candidate

            # Also try x=0 and x=2*v as sanity checks
            for x_try in [0.0, 2.0 * v if v > 0 else 0.0]:
                x_try = max(0.0, x_try)
                g_eval = c_base + slope * x_try
                if has_self:
                    g_eval[k_self] += x_try * x_try
                f_val = 2.0 * n * g_eval.max()
                if f_val < best_f:
                    best_f = f_val
                    best_x = x_try

            # Apply update
            if abs(best_x - v) > tol:
                a[i] = best_x
                max_change = max(max_change, abs(best_x - v))
                if max_change > 1e-10:
                    improved = True

        # Re-normalize to ensure sum = 1
        s = a.sum()
        if s > 1e-15:
            a = a / s

        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if not improved or max_change < tol:
            break

    return best_a, best_score


def _active_set_equalization(a_init, deadline, tol=1e-12, max_iters=100):
    """
    Try to equalize the top active lags using a Newton-like approach.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break

        c = _full_convolve(a, a)
        max_c = float(c.max())
        # Find active set: lags within 1% of max
        threshold = max_c * 0.999
        active_mask = c >= threshold
        active_idx = np.where(active_mask)[0]

        if len(active_idx) == 0:
            break
        if len(active_idx) == 1:
            # Single active lag - coordinate descent handles this
            break

        # Compute gradient of c[k] w.r.t. a for active lags
        # d(c[k])/da[i] = a[k-i] + a[i-(k-i)] ... simplified:
        # For each active k, the gradient w.r.t. a[i] is:
        #   a[k-i] if 0 <= k-i <= n-1, plus a[i-k+n-1]... 
        # Actually: d(c[k])/da[i] = 2*a[k-i] (if valid and k-i != i)
        #                                      a[k-i] (if valid and k-i == i, i.e., k=2i)
        # Simpler: use the full formula d(c[k])/da[i] = a[(k-i) mod (2n-1)] ... 
        # Let's use the FFT-based approach for efficiency

        # Gradient accumulation: for each active lag k, add weight * d(c[k])/da
        # We want to reduce the spread among active lags
        # Use: for each active k, try to reduce c[k] by reducing a in directions
        # where d(c[k])/da is large, and increase where it's small.

        # Simple approach: for each active lag k, compute the "pressure" p_k = c[k] - mean(active)
        # Then update a[i] -= step * sum_k p_k * d(c[k])/da[i]
        
        active_c = c[active_idx]
        mean_active = float(active_c.mean())
        pressures = active_c - mean_active  # positive for above-mean, negative for below

        # Compute total gradient: sum_k pressures[k] * d(c[k])/da[i]
        grad = np.zeros(n)
        for k in active_idx:
            k = int(k)
            p = pressures[list(active_idx).index(k)]
            # d(c[k])/da[i]: 
            # For i in [max(0,k-n+1), min(k, n-1)]:
            #   d(c[k])/da[i] = a[k-i] (from j=i term)
            #   + a[i] (from j=k-i term, if k-i != i and 0<=k-i<=n-1)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                # j ranges from lo to hi, k-j ranges from k-lo to k-hi
                vals = a[k - hi:k - lo + 1][::-1]  # a[k-lo], a[k-lo-1], ..., a[k-hi]
                # Actually: for j in [lo, hi], k-j in [k-hi, k-lo]
                # d(c[k])/da[j] = a[k-j] + (a[j] if j != k-j else 0)
                # Let's just do it element by element for correctness
                for j in range(lo, hi + 1):
                    ki = k - j
                    if 0 <= ki <= n - 1:
                        if ki == j:
                            grad[j] += p * 2.0 * a[j]
                        else:
                            grad[j] += p * (a[ki] + a[j])
                    else:
                        grad[j] += p * a[j]  # only one term

        # Update: a -= lr * grad, then project to non-negative and renormalize
        lr = 0.01
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            sc_new = _score_norm_fn(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                lr *= 0.5
                if lr < 1e-10:
                    break
        else:
            break

    return best_a, best_score


def _run_lbfgs_adaptive(a_init, deadline, max_wall_time=None):
    """
    L-BFGS with adaptive logsumexp + top-K blend smoothing.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Adaptive beta schedule
    beta_starts = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(5.0, max_wall_time / len(betas))
    else:
        per_level = 60.0

    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            # Blend: top-3 average + logsumexp
            top3 = np.sort(c_)[-3:]
            top3_avg = float(top3.mean())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                ls = cm_
            else:
                ls = cm_ + beta * float(np.log(s_))
            # Blend top3_avg and logsumexp
            blended = 0.4 * top3_avg + 0.6 * ls
            return 2.0 * n * blended

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            # Top-3 weights
            top3_idx = np.argsort(c_)[-3:]
            top3_vals = c_[top3_idx]
            top3_range = top3_vals.max() - top3_vals.min()
            if top3_range > 1e-15:
                w_top = (top3_vals - top3_vals.min()) / top3_range
            else:
                w_top = np.ones(3) / 3.0
            w = np.zeros_like(c_)
            w[top3_idx] = w_top
            # Logsumexp weights
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ > 1e-300:
                w_ls = e_ / s_
            else:
                w_ls = np.zeros_like(c_)
                w_ls[int(np.argmax(c_))] = 1.0
            # Blend
            w_blend = 0.4 * w + 0.6 * w_ls

            # Gradient of blended objective w.r.t. a
            W = float(np.dot(w_blend, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w_blend, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            # Also need gradient of top3_avg (piecewise linear, use subgradient)
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-12,
                         'maxls': 100, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score

                x[ii] = 0.0
                a[idx] = x
                c_base_block = _full_convolve(a, a)
                a[idx] = x
                x[ii] = x_save

                # Compute slope for this coordinate
                slope_local = np.zeros(2 * n - 1)
                k_lo = max(0, gi)
                k_hi = min(2 * n - 2, gi + n - 1)
                idx_lo = k_lo - gi
                idx_hi = k_hi - gi
                slope_local[k_lo:k_hi + 1] = 2.0 * a[idx_lo:idx_hi + 1]
                k_self = 2 * gi
                has_self = k_self < 2 * n - 1

                # Find best x[ii]
                g_curr = c_base_block + slope_local * x_save
                if has_self:
                    g_curr[k_self] += x_save * x_save
                cur_f = 2.0 * n * g_curr.max()

                best_x = x_save
                best_f = cur_f

                # Top lags
                top_idxs = np.argpartition(g_curr, -3)[-3:]
                for k_star in top_idxs:
                    k_star = int(k_star)
                    sv = slope_local[k_star]
                    cb = c_base_block[k_star]
                    if k_star == k_self and has_self:
                        xc = cb / s_o if s_o > 1e-15 else 0.0
                    elif sv > 1e-15:
                        xc = s_o - cb / sv
                    else:
                        xc = 0.0
                    xc = max(0.0, min(xc, 5.0))
                    g_eval = c_base_block + slope_local * xc
                    if has_self:
                        g_eval[k_self] += xc * xc
                    f_val = 2.0 * n * g_eval.max()
                    if f_val < best_f:
                        best_f = f_val
                        best_x = xc

                x[ii] = best_x
                max_delta = max(max_delta, abs(best_x - x_save))

            a[idx] = x
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s

            if max_delta > 1e-10:
                improved = True

        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if not improved:
            break

    return best_a, best_score


def _fast_scoring_batch(a_list):
    """Score multiple sequences efficiently."""
    n = len(a_list[0])
    results = []
    for a in a_list:
        c = _full_convolve(a, a)
        results.append(float(2.0 * n * c.max()))
    return results


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse initialization, L-BFGS, coordinate descent,
    active-set equalization, block coordinate descent, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            t_elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [t={t_elapsed:.0f}s]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                print(f"Parent loaded: n={len(parent)}")
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent score: {sc_parent:.10f}")
        update_best(parent, sc_parent)
    else:
        parent = _make_linear_profile(4096, 1.2)
        update_best(parent, _score_norm_fn(parent))
        print(f"No parent, using default n=4096")

    n_base = len(parent)

    # --- Phase 1: Diverse initialization (30s) ---
    phase1_end = min(deadline, t_start + 30.0)
    candidates = []

    profiles = [
        _make_linear_profile(n_base, 0.5),
        _make_linear_profile(n_base, 1.0),
        _make_linear_profile(n_base, 1.5),
        _make_linear_profile(n_base, 2.0),
        _make_linear_profile(n_base, 3.0),
        _make_quadratic_profile(n_base, 0.5, 0.5),
        _make_quadratic_profile(n_base, 1.0, -0.3),
        _make_quadratic_profile(n_base, 1.5, 0.2),
        _make_sinusoidal_profile(n_base, 1, 0.3),
        _make_sinusoidal_profile(n_base, 2, 0.2),
        _make_sinusoidal_profile(n_base, 3, 0.15),
        _make_product_profile(n_base, 1.0, 1, 0.3),
        _make_product_profile(n_base, 1.5, 2, 0.2),
    ]

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase1_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p))
        update_best(p, sc)

    # Try different n
    for sn in [2048, 8192]:
        if time.monotonic() > phase1_end:
            break
        for slope in [0.8, 1.2, 1.5]:
            p = _make_linear_profile(sn, slope)
            sc = _score_norm_fn(p)
            candidates.append((sc, p))
            update_best(p, sc)

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}, {len(candidates)} candidates")

    # --- Phase 2: L-BFGS on top candidates (300s) ---
    phase2_end = min(deadline, t_start + 330.0)
    n_to_opt = 3
    per_cand = max(30.0, (phase2_end - time.monotonic()) / max(1, n_to_opt))
    count = 0
    seen_n = set()

    for sc, p in candidates:
        if time.monotonic() > phase2_end - 5.0:
            break
        if count >= n_to_opt:
            break
        if sc > best_score * 1.03:
            break
        pn = len(p)
        if pn in seen_n:
            continue
        seen_n.add(pn)

        dl = min(phase2_end - 5.0, time.monotonic() + per_cand)
        p_best, sc_best = _run_lbfgs_adaptive(p, dl, max_wall_time=per_cand)
        update_best(p_best, sc_best)
        count += 1

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Coordinate descent (300s) ---
    phase3_end = min(deadline, t_start + 630.0)
    if best_seq is not None:
        dl = phase3_end - 5.0
        p_best, sc_best = _coordinate_descent(best_seq, dl, max_passes=500)
        update_best(p_best, sc_best)

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Active set equalization (150s) ---
    phase4_end = min(deadline, t_start + 780.0)
    if best_seq is not None:
        dl = phase4_end - 5.0
        p_best, sc_best = _active_set_equalization(best_seq, dl)
        update_best(p_best, sc_best)

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: More coordinate descent (200s) ---
    phase5_end = min(deadline, t_start + 980.0)
    if best_seq is not None:
        dl = phase5_end - 5.0
        p_best, sc_best = _coordinate_descent(best_seq, dl, max_passes=1000, tol=1e-15)
        update_best(p_best, sc_best)

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: L-BFGS refinement (200s) ---
    phase6_end = min(deadline, t_start + 1180.0)
    if best_seq is not None:
        dl = min(phase6_end - 5.0, time.monotonic() + 180.0)
        p_best, sc_best = _run_lbfgs_adaptive(best_seq, dl, max_wall_time=180.0)
        update_best(p_best, sc_best)

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Block coordinate descent (150s) ---
    phase7_end = min(deadline, t_start + 1330.0)
    if best_seq is not None:
        dl = phase7_end - 5.0
        p_best, sc_best = _block_coordinate_descent(best_seq, dl, block_size=4, max_passes=50)
        update_best(p_best, sc_best)

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Multi-n exploration (200s) ---
    phase8_end = min(deadline, t_start + 1530.0)
    if best_seq is not None:
        for target_n in [2048, 8192]:
            if time.monotonic() > phase8_end - 20:
                break
            if target_n == len(best_seq):
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp)

            if time.monotonic() < phase8_end - 10:
                dl = min(phase8_end - 5, time.monotonic() + 60.0)
                p_best, sc_best = _coordinate_descent(p_interp, dl, max_passes=200)
                update_best(p_best, sc_best)

    print(f"Phase 8 done: best={best_score:.10f}")

    # --- Phase 9: Final intensive coordinate descent (remaining) ---
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic())
        # Split between coordinate descent and L-BFGS
        half = remaining / 2.0

        dl1 = min(deadline - half - 5, time.monotonic() + half)
        p_best, sc_best = _coordinate_descent(best_seq, dl1, max_passes=2000, tol=1e-16)
        update_best(p_best, sc_best)

        if time.monotonic() < deadline - 10:
            dl2 = min(deadline - 5, time.monotonic() + (deadline - time.monotonic() - 5))
            p_best2, sc_best2 = _run_lbfgs_adaptive(best_seq, dl2, max_wall_time=(deadline - time.monotonic()))
            update_best(p_best2, sc_best2)

        # Final coordinate descent squeeze
        if time.monotonic() < deadline - 5:
            dl3 = deadline - 5
            p_best3, sc_best3 = _coordinate_descent(best_seq, dl3, max_passes=5000, tol=1e-16)
            update_best(p_best3, sc_best3)

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_linear_profile(4096, 1.5)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        t_total = time.monotonic() - t_start
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}, time={t_total:.0f}s")

        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(4096, 1.5)
    return [float(x) for x in fallback]
