"""
Optimization for minimum peak autocorrelation of non-negative sequences.
Combines LP-based refinement (with both sum-max and min-max objectives),
subgradient mirror descent, adaptive M scheduling, symmetric constraint,
and final coordinate descent polish.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    maxc = float(c.max())
    s = float(p.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return None
    return a / s


def _symmetrize(a):
    n = len(a)
    return (a + a[::-1]) / 2.0


def _make_uniform(n):
    return np.ones(n) / n


def _make_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_hyperbolic(n, alpha):
    t = np.linspace(-1.0, 1.0, n)
    a = 1.0 / (1.0 + alpha * t * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, width):
    center = (n - 1) / 2.0
    t = np.arange(n)
    a = np.exp(-((t - center) ** 2) / (2.0 * width * width))
    a += np.exp(-((t - 0) ** 2) / (2.0 * (width * 0.3) ** 2))
    a += np.exp(-((t - (n - 1)) ** 2) / (2.0 * (width * 0.3) ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _build_conv_matrix(a, n):
    """Build the cross-convolution matrix: (A @ b)[k] = sum_i a[i]*b[k-i]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_sum_max(a, M):
    """LP: maximize sum(b) subject to A(b)<=M, b>=0."""
    n = len(a)
    A = _build_conv_matrix(a, n)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_min_max(a, M):
    """LP: maximize t subject to A(b)<=M, b_j>=t, b>=0, sum(b)=1."""
    n = len(a)
    A = _build_conv_matrix(a, n)
    # Variables: b[0..n-1], t
    # Objective: maximize t -> minimize -t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    # A_ub: A @ b <= M (constraints on b only, t has no upper bound constraint from A)
    A_full = np.hstack([A.toarray(), np.zeros((2 * n - 1, 1))])
    b_ub = M * np.ones(2 * n - 1)
    # b_j - t >= 0 -> -b_j + t <= 0
    I_b = -np.eye(n)
    ones_t = np.ones(n)
    A_extra = np.hstack([I_b, ones_t[:, None]])
    b_extra = np.zeros(n)
    A_ub_full = np.vstack([A_full, A_extra])
    b_ub_full = np.concatenate([b_ub, b_extra])
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x[:n], 0.0)
    return None


def _lp_refine(a, deadline, use_minmax=False, m_tol=1.0005, t_sched='decay',
               t_init=0.3, t_min=0.05, t_decay=0.97, max_iter=None):
    """LP-based refinement with adaptive M."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = _normalize(a)
    if a is None:
        return a, np.inf
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    s = float(a.sum())
    M_norm = M / (s * s)  # normalized peak
    t = t_init
    it = 0

    while time.monotonic() < deadline:
        if max_iter is not None and it >= max_iter:
            break
        it += 1

        # Solve LP
        if use_minmax:
            b = _lp_min_max(a, M)
        else:
            b = _lp_sum_max(a, M)

        if b is None:
            # Try with slightly larger M
            M = M * 1.005
            b = _lp_sum_max(a, M) if not use_minmax else _lp_min_max(a, M)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _full_convolve(a_new)
        max_new = float(c_new.max())
        score_new = float(2.0 * n * max_new)

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        c = c_new
        M = float(c.max())

        # Update t
        if t_sched == 'decay':
            t = max(t_min, t * t_decay)
        else:
            t = t_init

    return best_a, best_score


def _subgradient_descent(a, deadline, lr_init=0.1, lr_decay=0.999, max_iter=None):
    """Subgradient descent with multiplicative (mirror) updates on log-scale."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = _normalize(a)
    if a is None:
        return a, np.inf
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    lr = lr_init
    it = 0
    no_imp = 0

    while time.monotonic() < deadline:
        if max_iter is not None and it >= max_iter:
            break
        it += 1

        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                lr = max(0.001, lr * 0.5)
                no_imp = 0

        # Find all near-peak positions
        threshold = M * 0.99
        peaks = np.where(c >= threshold)[0]

        # Compute subgradient: average of 2*S_k(a) over peak positions
        grad = np.zeros(n)
        for k in peaks:
            # (S_k a)[j] = a[k-j] if 0 <= k-j < n
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                grad[j_start:j_end + 1] += a[k - np.arange(j_start, j_end + 1)]

        grad /= len(peaks)

        # Multiplicative update: a[j] *= exp(-lr * grad[j] / a[j] * a[j])
        # Simplified: a[j] *= exp(-lr * grad[j])
        # But this can make values zero. Use: a_new = a * (1 - lr * grad)
        # Or better: work in log space.
        log_a = np.log(np.maximum(a, 1e-30))
        # Gradient of score w.r.t. log_a[j] = a[j] * grad[j]
        log_grad = a * grad
        log_a -= lr * log_grad
        log_a -= np.mean(log_a)  # re-center to maintain scale
        log_a = np.clip(log_a, -30, 30)
        a = np.exp(log_a)
        a = a / a.sum()

        c = _full_convolve(a)
        lr = lr * lr_decay

    return best_a, _score(best_a)


def _coordinate_descent(a, deadline, n_iter=2000):
    """Coordinate descent: try moving mass from peak-contributing positions."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = _normalize(a)
    if a is None:
        return a, np.inf
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    c = _full_convolve(a)
    rng = np.random.default_rng(777)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        peak_k = int(np.argmax(c))
        score = float(2.0 * n * M)

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()

        # Find top contributors to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        if i_start > i_end:
            continue
        partners = peak_k - np.arange(i_start, i_end + 1)
        contrib = a[i_start:i_end + 1] * a[partners]
        order = np.argsort(contrib)[::-1]
        top_positions = i_start + order[:3]

        # Try moving a small amount from the top contributor to a random other position
        src = top_positions[0]
        # Choose target: a position far from the peak's partner set
        partner_set = set()
        for k_val in range(max(0, peak_k - 3), min(2 * n - 1, peak_k + 4)):
            ps = max(0, k_val - n + 1)
            pe = min(n - 1, k_val)
            if ps <= pe:
                partner_set.update((k_val - np.arange(ps, pe + 1)).tolist())
        valid_targets = [x for x in range(n) if x not in partner_set and x != src]
        if not valid_targets:
            valid_targets = [x for x in range(n) if x != src]
        if not valid_targets:
            continue

        tgt = valid_targets[rng.integers(0, len(valid_targets))]
        # Optimal fraction to move (line search with a few candidates)
        best_frac = 0.0
        best_move_score = score
        for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.15, 0.2]:
            amount = a[src] * frac
            if amount < 1e-15:
                continue
            a_test = a.copy()
            a_test[src] -= amount
            a_test[tgt] += amount
            a_test = np.maximum(a_test, 1e-15)
            a_test = a_test / a_test.sum()
            c_test = _full_convolve(a_test)
            sc_test = float(2.0 * n * c_test.max())
            if sc_test < best_move_score:
                best_move_score = sc_test
                best_frac = frac

        if best_frac > 0:
            amount = a[src] * best_frac
            a[src] -= amount
            a[tgt] += amount
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            c = _full_convolve(a)
        else:
            # Try a small random perturbation
            idx1, idx2 = rng.integers(0, n, 2)
            if idx1 == idx2:
                continue
            amount = a[idx1] * 0.02
            a[idx1] -= amount
            a[idx2] += amount
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            c = _full_convolve(a)

    return best_a, _score(best_a)


def _polish_combination(a, deadline):
    """Alternate between LP and subgradient for final polish."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = _normalize(a)
    if a is None:
        return a, np.inf
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        remaining = deadline - time.monotonic()

        # LP step (short)

        if lp_time > 5:

                                 use_minmax=(it % 3 == 0),
                                 t_init=0.15, t_min=0.05, t_decay=0.98)
            if sc2 < best_score:
                best_score = sc2
                best_a = a2.copy()
                a = a2.copy()

        # Subgradient step (short)

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))

        a_sym = _normalize(_symmetrize(best_seq))
        if a_sym is not None:
            _update(a_sym, 'final_sym')

        # Final LP
        if time.monotonic() < phase6_end - 10:
            a6, sc6 = _lp_refine(best_seq, phase6_end - 5, use_minmax=False,
                                  t_init=0.08, t_min=0.02, t_decay=0.997)
            _update(a6, 'final_lp')

            a6s = _normalize(_symmetrize(a6))
            if a6s is not None:
                _update(a6s, 'final_lp_sym')

    # --- Final validation ---
    print(f"\n=== FINAL RESULT ===")
    print(f"Best score: {best_score:.10f}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_uniform(1024)

        exact_sc = _score(final)
        print(f"Exact score: {exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"Eval score: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_uniform(1024)
    print(f"FALLBACK n=1024, score={_score(fb):.6f}")
    return [float(x) for x in fb]
