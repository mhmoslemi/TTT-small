"""
Optimized step function search combining aggressive LP refinement with golden-section
line search, dual-guided updates, adaptive smooth-max Adam, and multi-scale escalation.
Targets score < 1.5027 by iteratively flattening the self-convolution peak.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-13:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M):
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_solve_dual(a, M):
    """Solve LP and return both primal and dual solutions."""
    n = len(a)
    A_ub = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success:
            b = np.maximum(res.x, 0.0)
            lam = np.maximum(np.asarray(res.ineqlam, dtype=np.float64).ravel(), 0.0)
            return b, lam
    except Exception:
        pass
    return None, None


def _line_search_blend(a, b_norm, current_score):
    """Find best blend factor t in (0,1) minimizing score."""
    best_t = 0.3
    best_sc = current_score
    # Coarse search
    for tc in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8]:
        a_try = (1.0 - tc) * a + tc * b_norm
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()
        sc_try = _score_norm(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_t = tc
    # Refine around best_t
    lo = max(0.01, best_t - 0.1)
    hi = min(0.99, best_t + 0.1)
    for tc in np.linspace(lo, hi, 11):
        a_try = (1.0 - tc) * a + tc * b_norm
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()
        sc_try = _score_norm(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_t = tc
    a_new = (1.0 - best_t) * a + best_t * b_norm
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new, best_sc, best_t


def _lp_refine(a, deadline, max_iters=100, aggressive=False):
    """LP-based refinement with line search and optional aggressive M."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        m_factor = 0.9 if aggressive else 1.0
        M_use = M * m_factor

        b, lam = _lp_solve_dual(a, M_use)
        if b is None:
            b = _lp_solve(a, M)
            lam = None
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b

        a_new, sc_new, t_used = _line_search_blend(a, b_norm, current_score)

        if sc_new < current_score - 1e-13:
            current_score = sc_new
            current_best = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 8:
                break

        a = a_new
        c = _full_convolve(a)
        M = float(c.max())

    return current_best, current_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=500, noise_scale=None):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.005, lr * 0.92)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 25 == 0:
            beta = max(5e-5, beta * beta_decay)
            if c.size >= 2:
                top5 = np.sort(c)[::-1][:5]
                gap = top5[0] - top5[-1]
                if gap < beta * 0.05:
                    beta = max(5e-5, beta * 0.7)

        if no_imp > no_imp_limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.04
                elif best_score < 1.53:
                    ns = 0.08
                else:
                    ns = 0.15
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a1 * t + a2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_asym(n, center, width, skew):
    t = np.linspace(0.0, 1.0, n)
    x = (t - center) / width
    a = np.exp(-0.5 * x * (x + skew))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _dual_guided_update(a, lam, M, alpha=0.1):
    """Use dual variables to construct a shrinkage direction."""
    n = len(a)
    c = _full_convolve(a)
    # For each position j, compute blame from dual
    # blame[j] = sum_k lam[k] * a[k-j]
    # This is the cross-correlation of lam with reversed a
    N = _next_pow2(2 * n - 1)
    flam = np.fft.rfft(lam[:2*n-1], N)
    fa_rev = np.fft.rfft(a[::-1], N)
    blame = np.fft.irfft(flam * fa_rev, N)
    blame = blame[n-1:2*n-1]

    # Shrink positions with high blame
    max_blame = blame.max()
    if max_blame < 1e-15:
        return a.copy()
    blame_norm = blame / max_blame
    # Reduce values where blame is high
    a_new = a * (1.0 - alpha * blame_norm)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Started. Parent n={pn}, budget={budget_s}s")

    # Phase 1: Quick initialization
    phase1_end = t_start + 30.0
    n_base = 1024
    init_candidates = []

    if parent is not None:
        if pn == n_base:
            init_candidates.append(('parent', parent.copy(), n_base))
        elif pn > 0 and pn <= 4096:
            pi = _interp_profile(parent, n_base)
            if pi is not None:
                init_candidates.append(('parent_interp', pi, n_base))
            if pn != n_base:
                init_candidates.append(('parent_orig', parent.copy(), pn))

    if not init_candidates:
        init_candidates.append(('linear', _make_profile_linear(n_base, 1.5), n_base))

    # Pick the best initial candidate
    best_init = None
    best_init_score = np.inf
    for label, p, n in init_candidates:
        sc = _score_norm(p)
        if sc < best_init_score:
            best_init_score = sc
            best_init = p.copy()
        _update(p, sc, f'init_{label}_n{n}')

    a = best_init.copy()
    n_cur = a.size
    print(f"Phase 1 done. Best init: {best_init_score:.10f} n={n_cur}")

    # Phase 2: Intensive LP refinement (30s - 700s)
    phase2_end = t_start + 700.0
    print("Phase 2: LP refinement")

    while time.monotonic() < phase2_end - 30:
        batch_deadline = min(time.monotonic() + 60, phase2_end - 30)
        a, sc = _lp_refine(a, batch_deadline, max_iters=50, aggressive=False)
        _update(a, sc, 'LP')

        # Check for stagnation - if no improvement in score, try aggressive
        if sc >= best_score - 1e-10:
            # Try aggressive LP (smaller M)
            m_factor = 0.85
            c = _full_convolve(a)
            M = float(c.max()) * m_factor
            b = _lp_solve(a, M)
            if b is not None:
                s_b = float(b.sum())
                if s_b > 1e-10:
                    b_norm = b / s_b
                    a_try, sc_try, _ = _line_search_blend(a, b_norm, sc)
                    if sc_try < sc - 1e-10:
                        a = a_try
                        sc = sc_try
                        _update(a, sc, 'LP_aggr')
                    else:
                        # Try dual-guided update
                        _, lam = _lp_solve_dual(a, float(_full_convolve(a).max()))
                        if lam is not None and lam.max() > 0:
                            a_dual = _dual_guided_update(a, lam, M, alpha=0.05)
                            sc_dual = _score_norm(a_dual)
                            if sc_dual < sc - 1e-10:
                                a = a_dual
                                sc = sc_dual
                                _update(a, sc, 'dual_guided')
                        else:
                            break

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # Phase 3: Adam polishing (700s - 1100s)
    phase3_end = t_start + 1100.0
    if time.monotonic() < phase3_end:
        print("Phase 3: Adam polishing")
        for round_i in range(3):
            if time.monotonic() > phase3_end - 30:
                break
            per = min(120.0, phase3_end - time.monotonic() - 20)
            if per < 20:
                break
            p_ad, sc_ad = _adam_logit(a, time.monotonic() + per, rng,
                                      lr=0.04 - 0.01 * round_i,
                                      beta_frac=0.06 - 0.01 * round_i,
                                      beta_decay=0.94,
                                      no_imp_limit=300)
            _update(p_ad, sc_ad, f'adam_r{round_i}')
            if sc_ad < _score_norm(a) - 1e-10:
                a = p_ad.copy()

    print(f"Phase 3 done. Best: {best_score:.10f}")

    # Phase 4: LP refinement round 2 (1100s - 1350s)
    phase4_end = t_start + 1350.0
    if time.monotonic() < phase4_end and best_seq is not None:
        print("Phase 4: LP round 2")
        a = best_seq.copy()
        a, sc = _lp_refine(a, phase4_end - 10, max_iters=30, aggressive=False)
        _update(a, sc, 'LP_r2')

    # Phase 5: Try larger n (1350s - 1600s)
    phase5_end = t_start + 1600.0
    if time.monotonic() < phase5_end and best_seq is not None and best_seq.size <= 1024:
        print("Phase 5: Try n=2048")
        p_int = _interp_profile(best_seq, 2048)
        if p_int is not None:
            sc_int = _score_norm(p_int)
            _update(p_int, sc_int, 'interp_2048')
            if time.monotonic() < phase5_end - 30:
                per = min(phase5_end - 30, time.monotonic() + 180)
                p_ad, sc_ad = _adam_logit(p_int, per, rng,
                                          lr=0.03, beta_frac=0.05,
                                          beta_decay=0.95, no_imp_limit=200)
                _update(p_ad, sc_ad, 'adam_2048')
                if sc_ad < best_score:
                    # Do a few LP iters on n=2048
                    if time.monotonic() < phase5_end - 20:
                        a2048, sc2048 = _lp_refine(p_ad, phase5_end - 10, max_iters=8, aggressive=False)
                        _update(a2048, sc2048, 'LP_2048')

    # Phase 6: Multi-restart (1600s - 1780s)
    phase6_end = deadline - 20.0
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = best_seq.size
        print(f"Phase 6: Multi-restart n={n_best}")
        n_restarts = 4
        per_r = max(20.0, (phase6_end - time.monotonic()) / (n_restarts + 1))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            per_adam = min(per_r * 0.6, phase6_end - time.monotonic() - 20)
            if per_adam < 10:
                break

            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.03 + 0.03 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per_adam, rng,
                                     lr=0.03, beta_frac=0.05, beta_decay=0.95,
                                     no_imp_limit=int(0.2 * n_best))
            _update(p_r, sc_r, f'restart{i}')

            if best_seq.size <= 1024 and time.monotonic() < phase6_end - 15:
                per_lp = min(30.0, phase6_end - time.monotonic() - 10)
                if per_lp > 15:
                    p_lp, sc_lp = _lp_refine(best_seq, time.monotonic() + per_lp,
                                              max_iters=10, aggressive=True)
                    _update(p_lp, sc_lp, f'LP_re{i}')

    # Phase 7: Final polish
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 10 and best_seq.size <= 1024:
            p_f, sc_f = _lp_refine(best_seq, time.monotonic() + per * 0.5,
                                    max_iters=8, aggressive=False)
            _update(p_f, sc_f, 'final_LP')
        if time.monotonic() < deadline - 3:
            per2 = deadline - time.monotonic() - 3
            if per2 > 5:
                p_f2, sc_f2 = _adam_logit(best_seq, time.monotonic() + per2, rng,
                                           lr=0.01, beta_frac=0.03, beta_decay=0.97,
                                           no_imp_limit=50)
                _update(p_f2, sc_f2, 'final_adam')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
