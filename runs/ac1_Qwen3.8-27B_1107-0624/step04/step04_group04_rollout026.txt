Uses active-set subgradient method with FFT-based convolution,

import numpy as np

import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    """Score assuming sum(a) = 1 (normalized). Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Compute beta*logsumexp(c/beta) as a smooth approximation to max(c)."""
    cm = c.max()
    return cm + beta * np.log(np.sum(np.exp((c - cm) / beta)))

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

"""
Active-set subgradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based exact subgradient computation, adaptive active set with temperature,
dual exponential weights, and simplex projection. Explores multiple sequence lengths.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    n = len(v)
    s = v.sum()
    if s < 1e-300:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    ws = w.sum()
    if ws < 1e-300:
        return np.ones(n) / n
    return w / ws


def _compute_subgrad(p, c, A):
    """
    Compute subgradient of max(c_k) for k in active set A.
    d_j = sum_{k in A} w_k * 2 * p_{k-j} (approximate, ignoring k=2j correction)
    Computed via FFT cross-correlation: d_j = (p * flip(w))_{2n-2-j}
    """
    n = len(p)
    conc = len(c)
    w = np.zeros(conc)
    w[A] = 1.0 / len(A)
    w_flip = w[::-1]
    conv_pw = _full_convolve(p, w_flip)
    # d_j = conv_pw[2n-2-j] for j=0,...,n-1
    start = 2 * n - 2
    end = start - n + 1
    d = conv_pw[end + 1:start + 1][::-1]
    d *= 2.0
    return d


def _compute_subgrad_weighted(p, c, A, weights):
    """Subgradient with non-uniform weights on active set."""
    n = len(p)
    conc = len(c)
    w = np.zeros(conc)
    w[A] = weights / weights.sum()
    w_flip = w[::-1]
    conv_pw = _full_convolve(p, w_flip)
    start = 2 * n - 2
    end = start - n + 1
    d = conv_pw[end + 1:start + 1][::-1]
    d *= 2.0
    return d


def _peak_shaving(p_init, deadline, n_inner_iters=50000, verbose=True):
    """
    Active-set subgradient method with adaptive temperature and step size.
    p_init should be normalized (sum=1, all >= 0).
    """
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_score = _score_norm_fn(p)
    
    c = _full_convolve(p, p)
    max_c = float(c.max())
    
    # Temperature: fraction of max to include in active set
    # Higher temp = more peaks included = more global direction
    temp = 0.05
    step_size = 0.1
    
    for it in range(n_inner_iters):
        if time.monotonic() > deadline:
            break
        
        # Find active set with temperature
        threshold = max_c * (1.0 - temp)
        A = np.where(c >= threshold)[0]
        
        if len(A) == 0:
            A = np.array([np.argmax(c)])
        
        # Adaptive weights: proportional to (c_k - threshold)^2 + 1
        c_A = c[A]
        spread = c_A - c_A.min()
        weights = spread ** 2 + 1.0
        
        # Compute subgradient
        d = _compute_subgrad_weighted(p, c, A, weights)
        
        # Center the direction (to maintain sum=1 after projection)
        d = d - d.mean()
        
        # Adaptive step size with backtracking
        alpha = step_size
        for _bt in range(10):
            p_try = p - alpha * d
            p_try = _project_simplex(p_try)
            c_try = _full_convolve(p_try, p_try)
            max_try = float(c_try.max())
            if max_try <= max_c + 1e-12:
                break
            alpha *= 0.5
        
        p_new = p - alpha * d
        p_new = _project_simplex(p_new)
        
        c = _full_convolve(p_new, p_new)
        max_c = float(c.max())
        score = 2.0 * n * max_c
        
        if score < best_score:
            best_score = score
            best_p = p_new.copy()
            p = p_new
        else:
            # Rejected step - reduce step size
            step_size *= 0.9
            p = best_p.copy()
            c = _full_convolve(p, p)
            max_c = float(c.max())
        
        # Adaptive temperature: decrease over time (tighten active set)
        if it % 200 == 0 and it > 0:
            temp = max(0.005, temp * 0.92)
        
        # Occasionally reset temperature for exploration
        if it % 2000 == 1000:
            temp = min(0.15, temp * 3.0)
        
        # Adapt step size based on progress
        if it % 500 == 0:
            if score < best_score * 0.999:
                step_size = min(0.5, step_size * 1.1)
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - (deadline - n_inner_iters * 0.01)
            print(f"    iter {it}, score={score:.10f}, best={best_score:.10f}, "
                  f"temp={temp:.4f}, step={alpha:.4f}, |A|={len(A)}")
    
    return best_p, best_score


def _dual_exponential_weights(p_init, deadline, n_iters=30000, verbose=True):
    """
    Dual approach: maintain weights on peaks, solve inner QP approximately.
    Alternates between updating dual weights and primal variables.
    """
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_score = _score_norm_fn(p)
    
    c = _full_convolve(p, p)
    conc = len(c)
    
    # Dual variables (exponential weights on peaks)
    beta_dual = 5.0  # temperature for dual update
    lam = np.ones(conc) / conc
    
    step_size = 0.05
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(p, p)
        max_c = float(c.max())
        
        # Update dual weights (exponential tilting toward peaks)
        c_shifted = (c - c.max()) / max(beta_dual * max_c, 1e-10)
        lam = np.exp(c_shifted)
        lam /= lam.sum()
        
        # Compute weighted subgradient: grad of sum_k lam_k * c_k
        w = lam
        w_flip = w[::-1]
        conv_pw = _full_convolve(p, w_flip)
        start = 2 * n - 2
        end = start - n + 1
        d = conv_pw[end + 1:start + 1][::-1]
        d *= 2.0
        
        # Center for sum constraint
        d = d - d.mean()
        
        # Update with backtracking
        alpha = step_size
        p_new = p
        for _bt in range(8):
            p_try = p - alpha * d
            p_try = _project_simplex(p_try)
            c_try = _full_convolve(p_try, p_try)
            if float(c_try.max()) <= max_c + 1e-12:
                p_new = p_try
                break
            alpha *= 0.5
        else:
            p_new = _project_simplex(p - alpha * d)
        
        c_new = _full_convolve(p_new, p_new)
        score_new = 2.0 * n * float(c_new.max())
        
        if score_new < best_score:
            best_score = score_new
            best_p = p_new.copy()
            p = p_new
        
        # Adaptive beta_dual: decrease for more focus on peaks

def _logsumexp_smooth_optimize(p_init, deadline, verbose=True):

        z = z - z.max()
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300:
            return np.ones_like(z) / len(z)
        return e / s
    
    def _obj(z, beta):
        p = _softmax(z)
        c = _full_convolve(p, p)
        cm = c.max()
        ls = float(np.log(np.sum(np.exp((c - cm) / beta))))
        return float(2.0 * n * (cm + beta * ls))
    
    def _grad(z, beta):
        p = _softmax(z)
        c = _full_convolve(p, p)
        cm = c.max()
        w = np.exp((c - cm) / beta)
        w /= w.sum()
        W = float(np.dot(w, c))
        w_flip = w[::-1]
        conv_pw = _full_convolve(p, w_flip)
        start = 2 * n - 2
        end = start - n + 1
        h = conv_pw[end + 1:start + 1][::-1]
        h *= 2.0
        grad = 4.0 * n * p * (h - W)
        return grad
    
    from scipy.optimize import minimize
    
    z = np.log(np.maximum(p_init, 1e-15))
    z = z - z.mean()
    
    best_p = p_init.copy()
    best_score = _score_norm_fn(p_init)
    
    c0 = _full_convolve(p_init, p_init)

def _make_centered_profile(n, decay):
    """Gaussian-like profile centered in the middle."""
    center = (n - 1) / 2.0
    x = np.arange(n) - center
    p = np.exp(-decay * x ** 2 / n)
    return p / p.sum()


def _make_cosine_profile(n, order):
    """Cosine-modulated profile."""
    x = np.arange(n)
    t = 2.0 * np.pi * x / n
    p = 1.0 + 0.8 * np.cos(t * order)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _make_dual_lobe(n, ratio):
    """Two-lobe profile."""
    center = (n - 1) / 2.0
    offset = center * ratio
    x = np.arange(n)
    p1 = np.exp(-((x - (center - offset)) ** 2) / (n * 0.1))
    p2 = np.exp(-((x - (center + offset)) ** 2) / (n * 0.1))
    p = p1 + p2
    return p / p.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

                _update_best(parent, sc)
                print(f"  Parent: n={len(parent)}, score={sc:.10f}")
    except Exception as e:
        print(f"  No parent: {e}")
    
    # Generate diverse profiles at multiple lengths
    for n in [4096, 8192]:
        profiles = []
        for decay in [1.0, 2.0, 3.0, 5.0, 8.0]:
            profiles.append(_make_centered_profile(n, decay))
        for order in [1, 2, 3, 4]:
            profiles.append(_make_cosine_profile(n, order))
        for ratio in [0.2, 0.3, 0.5]:
            profiles.append(_make_dual_lobe(n, ratio))
        # Uniform
        profiles.append(np.ones(n) / n)
        
        for idx, p in enumerate(profiles):
            sc = _score_norm_fn(p)
            candidates.append((sc, p, n, f"prof{idx}_n{n}"))
            _update_best(p, sc)
    
    # Interpolate parent to n=8192
    if parent is not None:
        for tn in [8192, 16384]:
            if len(parent) != tn:
                p_interp = _interp_profile(parent, tn)
                sc = _score_norm_fn(p_interp)
                candidates.append((sc, p_interp, tn, f"parent_interp_n{tn}"))
                _update_best(p_interp, sc)
    
    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 0 done: best={best_score:.10f}, {len(candidates)} candidates")
    
    # --- Phase 1: Active-set peak shaving on top candidates ---
    print("=== Phase 1: Peak shaving ===")
    phase1_end = min(deadline, t_start + 600.0)
    
    # Pick top 2-3 candidates by score, prefer different n values
    seen_n = set()
    selected = []
    for sc, p, pn, label in candidates:
        if len(selected) >= 3:
            break
        if sc > best_score * 1.05 and len(selected) > 0:
            break
        if pn in seen_n:
            continue
        seen_n.add(pn)
        selected.append((sc, p, pn, label))
    
    if not selected:
        selected = [candidates[0]]
    
    for sc, p, pn, label in selected:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        per_time = min(remaining, 300.0)
        dl = min(phase1_end, time.monotonic() + per_time)
        print(f"  Peak shaving on {label} (n={pn}), budget={per_time:.0f}s")
        
        p_best, sc_best = _peak_shaving(p, dl, n_inner_iters=100000, verbose=True)
        _update_best(p_best, sc_best)
    
    print(f"  Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: Dual exponential weights ---
    print("=== Phase 2: Dual exponential weights ===")
    phase2_end = min(deadline, t_start + 900.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        remaining = phase2_end - time.monotonic()
        if remaining > 30:
            dl = min(phase2_end, time.monotonic() + remaining * 0.7)
            print(f"  Dual EW on best (n={len(best_seq)}), budget={remaining*0.7:.0f}s")
            p_best, sc_best = _dual_exponential_weights(best_seq, dl, n_iters=50000, verbose=True)
            _update_best(p_best, sc_best)
            
            # Also try on parent if different n
            if parent is not None and len(parent) != len(best_seq):
                if time.monotonic() < phase2_end - 20:
                    dl2 = min(phase2_end - 10, time.monotonic() + 100.0)
                    p_best2, sc_best2 = _dual_exponential_weights(parent, dl2, n_iters=20000, verbose=False)
                    _update_best(p_best2, sc_best2)
    
    print(f"  Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: L-BFGS logsumexp polish ---
    print("=== Phase 3: L-BFGS polish ===")
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        if remaining > 30:
            dl = min(phase3_end - 5, time.monotonic() + min(200.0, remaining * 0.6))
            print(f"  L-BFGS on best (n={len(best_seq)}), budget={min(200,remaining*0.6):.0f}s")
            p_best, sc_best = _logsumexp_smooth_optimize(best_seq, dl, verbose=True)
            _update_best(p_best, sc_best)
    
    print(f"  Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Restart from best with perturbation + peak shaving ---
    print("=== Phase 4: Perturbed restarts ===")
    phase4_end = min(deadline, t_start + 1500.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = len(best_seq)
        n_restarts = 3
        remaining = phase4_end - time.monotonic()
        per_restart = max(10.0, remaining / (n_restarts + 1))
        
        noise_levels = [0.02, 0.05, 0.1]
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = noise_levels[i % len(noise_levels)] * rng.normal(size=n_best)
            z = z + noise
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = np.exp(z)
            p_start /= p_start.sum()
            
            dl = min(phase4_end - 5, time.monotonic() + per_restart * 0.7)
            p_best, sc_best = _peak_shaving(p_start, dl, n_inner_iters=30000, verbose=False)
            _update_best(p_best, sc_best)
            
            if time.monotonic() < phase4_end - 10:
                dl2 = min(phase4_end - 5, time.monotonic() + per_restart * 0.3)
                p_best, sc_best = _peak_shaving(p_best, dl2, n_inner_iters=10000, verbose=False)
                _update_best(p_best, sc_best)
    
    print(f"  Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Try different length ---
    print("=== Phase 5: Different length ===")
    phase5_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_curr = len(best_seq)
        target_ns = [8192] if n_curr != 8192 else [4096]
        
        for tn in target_ns:
            if time.monotonic() > phase5_end - 30:
                break
            p_interp = _interp_profile(best_seq, tn)
            _update_best(p_interp, _score_norm_fn(p_interp))
            
            remaining = phase5_end - time.monotonic()
            if remaining < 30:
                break
            dl = min(phase5_end - 10, time.monotonic() + min(120.0, remaining * 0.5))
            p_best, sc_best = _peak_shaving(p_interp, dl, n_inner_iters=30000, verbose=False)
            _update_best(p_best, sc_best)
    
    print(f"  Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Final intensive polish ---
    print("=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Intensive peak shaving with slowly decreasing temperature
            dl = min(deadline - 5, time.monotonic() + min(120.0, remaining * 0.5))
            p_best, sc_best = _peak_shaving(best_seq, dl, n_inner_iters=20000, verbose=True)
            _update_best(p_best, sc_best)
            
            if time.monotonic() < deadline - 10:
                remaining2 = deadline - time.monotonic()
                if remaining2 > 20:
                    dl2 = min(deadline - 5, time.monotonic() + min(60.0, remaining2 * 0.5))
                    p_best2, sc_best2 = _dual_exponential_weights(best_seq, dl2, n_iters=20000, verbose=False)
                    _update_best(p_best2, sc_best2)
    
    print(f"  Phase 6 done: best={best_score:.10f}")
    
    # --- Final validation ---
    print("=== Final ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_score = _score_fn(final_seq)
        else:
            final_seq = _make_centered_profile(4096, 3.0)
            final_score = _score_norm_fn(final_seq)
        
        print(f"  FINAL: score={final_score:.10f}, n={len(final_seq)}")
        
        # Quick validation
        try:
            seq_list = [float(x) for x in final_seq]
            if len(seq_list) <= 16384:
                sc_eval = float(evaluate_sequence(seq_list))
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  eval check: {e}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_centered_profile(4096, 3.0)
    print(f"  FALLBACK: n=4096, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
