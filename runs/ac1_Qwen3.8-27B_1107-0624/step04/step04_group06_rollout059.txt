"""
Optimization of step function heights to minimize 2*n*max(conv)/(sum)^2.
Uses: (1) bottleneck-exact subgradient with Adam in log-domain,
(2) alternating LP refinement, (3) multi-scale search, (4) coordinate
descent on identified bottleneck positions.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score_fn(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    c = _full_conv(a)
    return float(2.0 * a.size * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def _bottleneck_grad(a, c, top_frac=0.05):
    """Compute weighted gradient of top-fraction convolution entries."""
    n = len(a)
    M = float(c.max())
    cmin = float(c.min())
    # Use entries within top_frac of the range
    threshold = cmin + (M - cmin) * (1.0 - top_frac)
    mask = c >= threshold
    if not mask.any():
        mask[int(np.argmax(c))] = True
    w = c[mask] - cmin + 1e-12
    w = w / w.sum()
    indices = np.where(mask)[0]

    g = np.zeros(n)
    for wi, ki in zip(w, indices):
        # dC[ki]/da[j] = a[ki-j]*1_{valid} + a[j-ki]*1_{valid}
        j1_start = max(0, ki - n + 1)
        j1_end = min(ki, n - 1)
        if j1_start <= j1_end:
            partners1 = ki - np.arange(j1_start, j1_end + 1)
            g[j1_start:j1_end + 1] += wi * a[partners1]
        j2_start = max(ki, 0)
        j2_end = min(ki + n - 1, n - 1)
        if j2_start <= j2_end:
            partners2 = np.arange(j2_start, j2_end + 1) - ki
            g[j2_start:j2_end + 1] += wi * a[partners2]
    return g


def _adam_bottleneck(p0, deadline, rng, lr=0.08, top_frac=0.05,
                     no_imp_limit=200, restart_noise=0.1):
    """Optimize using exact bottleneck subgradient with Adam in log-domain."""
    p0 = np.asarray(p0, dtype=np.float64).ravel().copy()
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_conv(p)
    best_p = p0.copy()
    best_z = z.copy()
    best_score = _score_fn(p0)
    no_imp = 0
    global_no_imp = 0

    while time.monotonic() < deadline:
        g = _bottleneck_grad(p, c, top_frac)

        # Gradient in log-domain: dC/dz[j] = a[j]*(g[j] - dot(g,a))
        dot = float(np.dot(p, g))
        gz = p * (g - dot)

        # Center the gradient (remove mean to preserve sum invariance)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)

        # Adaptive top_frac: increase when converging, decrease when stuck
        if no_imp > 50:
            top_frac_use = max(0.01, top_frac * 0.8)
        else:
            top_frac_use = top_frac

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))

        p = np.exp(z - z.max())
        p = p / p.sum()
        c = _full_conv(p)
        sc = _score_fn(p)

        if np.isfinite(sc):
            if sc < best_score - 1e-14:
                best_score = sc
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
                global_no_imp = 0
                if sc < 1.51:
                    lr = max(0.005, lr * 0.99)
                    top_frac = max(0.02, top_frac * 0.97)
            else:
                no_imp += 1
                global_no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.003, lr * 0.95)

        if global_no_imp > no_imp_limit:
            # Restart with noise
            z = best_z + restart_noise * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = np.exp(z - z.max())
            p = p / p.sum()
            c = _full_conv(p)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            no_imp = 0
            global_no_imp = 0
            top_frac = 0.05
            if best_score < 1.51:
                restart_noise = 0.04
            elif best_score < 1.55:
                restart_noise = 0.08
            else:
                restart_noise = 0.15

    return best_p, best_score


def _lp_solve_given_a(a, M):
    """Solve: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, t_start=0.3, max_solves=30, m_shrink=0.998):
    """Iterative LP refinement: find b with (a*b) <= M, mix with a."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_fn(a)
    c = _full_conv(a)
    M = float(c.max())
    t = t_start
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve_given_a(a, M)
        if b is None:
            b = _lp_solve_given_a(a, M * 1.005)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_fn(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_conv(a)
        M_new = float(c.max())

        # Tighten M
        target_M = best_score / (2.0 * n)
        M = max(M_new * m_shrink, target_M * 1.002)
        t = max(0.1, t * 0.93)

    return best_a, best_score


def _alternating_lp(a, deadline, max_rounds=20, m_shrink=0.999):

def _coordinate_descent(a, deadline, n_steps=300):
    """Targeted coordinate descent: move mass away from bottleneck contributors."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()

    n = a.size
    best_a = a.copy()
    best_score = _score_fn(a)

    for it in range(n_steps):

            break

        M = float(c.max())
        sc = _score_fn(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        # Find top 3 bottleneck lags
        top_k = np.argsort(c)[::-1][:3]
        for ki in top_k:
            if time.monotonic() > deadline:
                break
            # Find the two most contributing positions
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start > j_end:
                continue
            contrib = np.zeros(j_end - j_start + 1)
            for jj in range(j_end - j_start + 1):
                i = j_start + jj
                contrib[jj] = a[i] * a[ki - i] if 0 <= ki - i < n else 0.0
            top2 = np.argsort(contrib)[::-1][:2]
            if len(top2) < 2:
                continue
            i1 = j_start + top2[0]
            i2 = j_start + top2[1]

            # Move mass from i1 to a random position far from bottleneck
            valid = [x for x in range(n) if x != i1 and abs(x - i1) > n // 4]
            if not valid:
                valid = [x for x in range(n) if x != i1]
            if not valid:
                continue
            target = valid[rng.integers(0, len(valid))]
            frac = 0.005 + 0.01 * rng.random()
            amount = a[i1] * frac
            a[i1] -= amount
            a[target] += amount
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    return best_a, best_score
