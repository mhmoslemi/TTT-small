Phase 2: L-BFGS with log-sum-exp smoothing (efficient local optimization)

Phase 4: Ultra-fine polishing with very small smoothing parameter

from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp(x):
    m = np.max(x)
    if not np.isfinite(m):
        return m
    return m + np.log(np.sum(np.exp(x - m)))


def _build_lp_indices(n):

    try:

def _compute_grad_z(a, beta, n):
    c = _full_convolve(a, a)
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv_aw = np.fft.irfft(fa * fw, N)[:len(a_rev) + len(w) - 1]
    grad_a = 2.0 * conv_aw[n - 1:2 * n - 1]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return grad_z

        update_best(a0, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")
        update_best(a0, _score(a0), "uniform_init")

    n = n_main
    print(f"Target n={n}, starting score={best_score:.10f}")

    # Precompute LP structure
    lp_rows, lp_cols = _build_lp_indices(n)
    print(f"LP structure: {len(lp_rows)} non-zeros in ({2*n-1}x{n}) matrix")

    # ============ Phase 1: LP-based optimization (0 - 400s) ============
    phase1_end = min(deadline, t_start + 400.0)
    print(f"\n=== Phase 1: LP-based optimization (until t={phase1_end-t_start:.0f}s) ===")

    f = a0.copy()
    prev_sc = _score_norm(f)
    lp_count = 0

    while time.monotonic() < phase1_end - 10:
        c_f = _full_convolve(f, f)
        M = float(c_f.max())
        if M < 1e-10:
            break

        t_lp_start = time.monotonic()
        b_dir = _lp_direction(f, M, n, lp_rows, lp_cols)
        t_lp_end = time.monotonic()

        f_new, sc_new = _line_search_lp(f, b_dir, n)
        sc_new_actual = _score_norm(f_new)

        if sc_new_actual < prev_sc - 1e-10:
            f = f_new
            prev_sc = sc_new_actual
            lp_count += 1
            update_best(f, _score(f), f"lp_{lp_count}")
            if lp_count % 10 == 0:
                print(f"  LP iter {lp_count}: score={_score(f):.10f}, M={M:.6f}, LP_time={t_lp_end-t_lp_start:.3f}s, total_t={time.monotonic()-t_start:.1f}s")
        else:
            # No improvement, try smaller step or break
            break

    print(f"Phase 1 done: {lp_count} LP iters, score={_score(f):.10f}, t={time.monotonic()-t_start:.1f}s")
    update_best(f, _score(f), "after_lp")
    if best_seq is not None:
        f = best_seq.copy()

    # ============ Phase 2: L-BFGS + Adam polishing (400s - 800s) ============
    phase2_end = min(deadline, t_start + 800.0)
    print(f"\n=== Phase 2: L-BFGS/Adam polishing (until t={phase2_end-t_start:.0f}s) ===")

    if best_seq is not None:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())

        # L-BFGS with moderate beta
        beta1 = 0.02 * max_cb
        per_time = min(60.0, phase2_end - time.monotonic() - 20)
        if time.monotonic() < phase2_end - 10:
            print(f"  L-BFGS beta={beta1:.6f}...")
            p_best, sc_best = _lbgfs_polish(best_seq, beta1, time.monotonic() + per_time, max_iter=300)
            update_best(p_best, _score(p_best), "lbgfs_b1")

        # Adam with beta continuation
        if time.monotonic() < phase2_end - 30:
            beta_schedule_adam = [0.02 * max_cb, 0.01 * max_cb, 0.005 * max_cb, 0.002 * max_cb, 0.001 * max_cb]
            per_time = phase2_end - time.monotonic() - 15
            p_best, sc_best = _adam_polish(best_seq, time.monotonic() + per_time, beta_schedule_adam, max_iter=300000, lr=0.01)
            update_best(p_best, _score(p_best), "adam_polish")

    print(f"Phase 2 done: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 3: Multi-restart with diverse initializations (800s - 1400s) ============
    phase3_end = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 3: Multi-restart (until t={phase3_end-t_start:.0f}s) ===")

    if best_seq is not None and time.monotonic() < phase3_end:
        restarts = [
            ("perturb_small", lambda: _perturb(best_seq, 0.03, rng)),
            ("perturb_med", lambda: _perturb(best_seq, 0.08, rng)),
            ("fourier_perturb", lambda: _perturb_fourier(best_seq, rng, 5)),
            ("fourier_perturb2", lambda: _perturb_fourier(best_seq, rng, 10)),
            ("symmetrized_perturb", lambda: _symmetrize(_perturb(best_seq, 0.05, rng))),
            ("fourier_init", lambda: _make_profile(n, "fourier", rng)),
            ("power_spec", lambda: _make_profile(n, "power_spectrum", rng)),
            ("cosine", lambda: _make_profile(n, "cosine")),
            ("bimodal", lambda: _make_profile(n, "bimodal")),
            ("exp_taper", lambda: _make_profile(n, "exponential_taper")),
        ]

        n_restarts = len(restarts)
        per_restart = (phase3_end - time.monotonic() - 20) / max(1, n_restarts)
        per_restart = max(10.0, min(per_restart, 80.0))

        for i, (name, gen_func) in enumerate(restarts):
            if time.monotonic() > phase3_end - 15:
                break
            if per_restart < 10:
                break

            p_start = gen_func()
            p_start = np.clip(p_start, 1e-10, None)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            sc_start = _score(p_start)

            if sc_start < best_score:
                update_best(p_start, sc_start, f"restart_{name}_init")
                p_start = best_seq.copy()

            # Short LP phase
            f_r = p_start.copy()
            prev_sc_r = _score_norm(f_r)
            lp_r_count = 0
            lp_time_budget = min(per_restart * 0.4, 15.0)
            lp_deadline_r = time.monotonic() + lp_time_budget

            while time.monotonic() < lp_deadline_r:
                c_f = _full_convolve(f_r, f_r)
                M_r = float(c_f.max())
                if M_r < 1e-10:
                    break
                b_r = _lp_direction(f_r, M_r, n, lp_rows, lp_cols)
                f_r_new, _ = _line_search_lp(f_r, b_r, n)
                sc_r = _score_norm(f_r_new)
                if sc_r < prev_sc_r - 1e-10:
                    f_r = f_r_new
                    prev_sc_r = sc_r
                    lp_r_count += 1
                else:
                    break

            update_best(f_r, _score(f_r), f"{name}_lp")

            # Adam polish
            if time.monotonic() < phase3_end - 10:
                c_r = _full_convolve(f_r, f_r)
                max_cr = float(c_r.max())
                bs_r = [0.015 * max_cr, 0.008 * max_cr, 0.004 * max_cr, 0.001 * max_cr]
                remaining_r = min(per_restart * 0.6, phase3_end - time.monotonic() - 5)
                if remaining_r > 5:
                    p_best_r, sc_best_r = _adam_polish(f_r, time.monotonic() + remaining_r, bs_r, max_iter=100000, lr=0.008)
                    update_best(p_best_r, _score(p_best_r), f"{name}_adam")

            print(f"  Restart {i+1}/{n_restarts} ({name}): best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    print(f"Phase 3 done: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 4: Ultra-fine polishing (1400s - 1750s) ============
    phase4_end = min(deadline, t_start + 1750.0)
    print(f"\n=== Phase 4: Ultra-fine polishing (until t={phase4_end-t_start:.0f}s) ===")

    if best_seq is not None and time.monotonic() < phase4_end:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())

        # Very fine Adam
        beta_fine = [0.001 * max_cb, 0.0005 * max_cb, 0.0002 * max_cb, 0.0001 * max_cb]
        per_time_fine = (phase4_end - time.monotonic() - 20) * 0.6
        if per_time_fine > 10:
            p_best, sc_best = _adam_polish(best_seq, time.monotonic() + per_time_fine, beta_fine, max_iter=300000, lr=0.004)
            update_best(p_best, _score(p_best), "fine_adam")

        # L-BFGS with very small beta
        if time.monotonic() < phase4_end - 20:
            beta_ult = 0.0005 * max_cb
            per_time2 = min(60.0, phase4_end - time.monotonic() - 10)
            p_best2, sc_best2 = _lbgfs_polish(best_seq, beta_ult, time.monotonic() + per_time2, max_iter=200)
            update_best(p_best2, _score(p_best2), "ult_lbgfs")

        # Coordinate descent polish
        if time.monotonic() < phase4_end - 10:
            a_cd = best_seq.copy()
            coord_deadline = min(time.monotonic() + (phase4_end - time.monotonic() - 5) * 0.3, phase4_end - 5)
            cd_improvements = 0
            for _ in range(200):
                if time.monotonic() > coord_deadline:
                    break
                improved = False
                for j in range(0, n, 1):
                    if time.monotonic() > coord_deadline:
                        break
                    save_val = a_cd[j]
                    # Try a few perturbations
                    for factor in [0.9, 0.95, 1.05, 1.1]:
                        a_try = a_cd.copy()
                        a_try[j] = max(0.0, save_val * factor)
                        a_try = a_try / a_try.sum()
                        sc_try = _score_norm(a_try)
                        if sc_try < _score_norm(a_cd) - 1e-12:
                            a_cd = a_try
                            improved = True
                            cd_improvements += 1
                            break
                if not improved:
                    break
            update_best(a_cd, _score(a_cd), "coord_desc")
            print(f"  Coord descent: {cd_improvements} improvements")

    print(f"Phase 4 done: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 5: Final polish and validation (1750s - deadline) ============
    print(f"\n=== Phase 5: Final polish ===")

    if best_seq is not None and time.monotonic() < deadline - 5:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())

        # One more round of very fine optimization
        beta_vf = [0.0002 * max_cf, 0.0001 * max_cf, 0.00005 * max_cf]
        per_time = deadline - time.monotonic() - 3
        if per_time > 5:
            p_best, sc_best = _adam_polish(best_seq, time.monotonic() + per_time * 0.7, beta_vf, max_iter=100000, lr=0.002)
            update_best(p_best, _score(p_best), "final_vfine")

        # Final L-BFGS
        if time.monotonic() < deadline - 3:
            p_best2, sc_best2 = _lbgfs_polish(best_seq, 0.0001 * max_cf, time.monotonic() + max(1.0, deadline - time.monotonic() - 2), max_iter=100)
            update_best(p_best2, _score(p_best2), "final_lbgfs")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
