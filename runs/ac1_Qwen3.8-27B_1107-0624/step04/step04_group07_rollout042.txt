"""
Coordinate descent + Adam optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: exact 1D coordinate optimization with vectorized evaluation,
combined with top-k subgradient Adam in log-space. Exploits the fact that
for non-negative sequences, the convolution's peak structure allows efficient
per-coordinate optimization.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    c = _full_convolve(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _cross_correlate_topk(c, a, k_top):
    """Compute top-k subgradient: average of dc/dx at top-k lags."""
    n = len(a)
    if k_top <= 1:
        k_star = int(np.argmax(c))
        grad = np.zeros(2 * n - 1)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        grad[lo:hi + 1] = 2.0 * a[k_star - lo:k_star - lo + (hi - lo + 1)]
        return grad
    
    # Top-k weights
    k_top = min(k_top, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k_top]
    top_c = c[top_idx]
    # Soft weighting within top-k
    w = top_c / top_c.sum()
    
    N = _next_pow2(2 * n - 1)
    # Build the gradient via cross-correlation with weighted impulse
    w_impulse = np.zeros(N)
    for ii, kk in enumerate(top_idx):
        w_impulse[kk] = w[ii]
    
    fw = np.fft.rfft(w_impulse, N)
    fa = np.fft.rfft(a, N)
    h = np.fft.irfft(fw * fa, N)
    # Extract the relevant portion: h[n-1:2n-1] gives dc[k]/da[j] = 2*a[k-j]
    # Actually, the cross-correlation gives sum_k w[k]*a[k-j] for valid k
    # We need: grad[j] = sum_k w[k] * 2*a[k-j] (with bounds)
    # FFT approach: convolve w_impulse with a
    result = h[:2 * n - 1]
    return 2.0 * result


def _adam_topk(a, deadline, rng, k_top=10, lr=0.1, max_iters=5000,
               no_imp_limit=300, restart_prob=0.3):

    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z -= z.mean()
    
    b1, b2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    a_cur = np.exp(z - z.max())
    a_cur = a_cur / a_cur.sum()
    c = _full_convolve(a_cur)
    best_score = float(2.0 * n * c.max())
    best_z = z.copy()
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        # Compute top-k subgradient
        g = _cross_correlate_topk(c, a_cur, k_top)
        
        # Softmax gradient: grad_z = a * (g - a.dot(g))
        dot = float(np.dot(a_cur, g))
        grad_z = a_cur * (g - dot)
        grad_z -= grad_z.mean()  # Remove mean for reparameterization invariance
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -40, 40)
        z_new -= z_new.mean()
        
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        
        c_new = _full_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())

            best_score = sc_new
            best_z = z_new.copy()
            z = z_new
            a_cur = a_new
            c = c_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:

    best_score = _score(a)
    best_a = a.copy()
    
    c_full = _full_convolve(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        s_total = float(a.sum())
        
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            x0 = a[j]
            if x0 < 1e-14:
                # Try a small positive value
                x_trial = s_total / n
                a[j] = x_trial
                a_n = a / a.sum()
                sc = _score(a_n)
                if sc < best_score:
                    best_score = sc
                    best_a = a_n.copy()
                    a = a_n.copy()
                    improved = True
                else:
                    a[j] = x0
                continue
            
            # Grid of x values (log-spaced for wide coverage)
            x_min = x0 * 1e-3
            x_max = x0 * x_mult
            x_vals = np.geomspace(x_min, x_max, n_grid)
            # Add the current value and a few linear points
            x_vals = np.concatenate([np.array([0.0]), x_vals, np.array([x0])])
            x_vals = np.sort(x_vals)
            
            # Compute c_b = conv(b, b) where b = a with b[j] = 0
            b = a.copy()
            b[j] = 0.0
            # Convolution: c_b[k] = c_full[k] - x0*(a[k-j] + a[j-k]) + (x0*x0 if k=2j)
            # More precisely: c_full[k] includes terms with a[j]
            # c_full[k] = c_b[k] + 2*x0*a[k-j] (for k≠2j) or c_b[2j] + x0² (for k=2j)
            # So: c_b[k] = c_full[k] - 2*x0*a[k-j] for k≠2j
            #        c_b[2j] = c_full[2j] - x0²
            
            # Compute d[k] = 2*a[k-j] for k≠2j, 0 for k=2j
            d = np.zeros(2 * n - 1)
            lo = j
            hi = min(2 * n - 2, j + n - 1)
            d[lo:hi + 1] = 2.0 * a[0:hi + 1 - lo]
            if 0 <= 2 * j < 2 * n - 1:
                d[2 * j] = 0.0
            
            # c_b from c_full
            c_b = c_full.copy()
            # Subtract the linear terms: c_full[k] = c_b[k] + d[k]*x0 (for k≠2j)
            # For k=2j: c_full[2j] = c_b[2j] + x0²
            if 0 <= 2 * j < 2 * n - 1:
                c_b[2 * j] = c_full[2 * j] - x0 * x0
            else:
                c_b = c_full - d * x0
            if 0 <= 2 * j < 2 * n - 1:
                c_b[2 * j] = c_full[2 * j] - x0 * x0
                # For k≠2j: c_b[k] = c_full[k] - d[k]*x0
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[2 * j] = False
                c_b[mask] = c_full[mask] - d[mask] * x0
            c_b = np.maximum(c_b, 0.0)
            
            # Evaluate score for each x in x_vals (vectorized)
            # c[k](x) = c_b[k] + d[k]*x for k≠2j, c_b[2j] + x² for k=2j
            s_base = s_total - x0
            s_x = s_base + x_vals  # shape (n_grid+2,)
            
            # Linear part: shape (n_x, 2n-1)
            c_lin = c_b[None, :] + d[None, :] * x_vals[:, None]
            if 0 <= 2 * j < 2 * n - 1:
                c_lin[:, 2 * j] = c_b[2 * j] + x_vals ** 2
            
            max_c = c_lin.max(axis=1)
            with np.errstate(divide='ignore', invalid='ignore'):
                scores_x = 2.0 * n * max_c / (s_x ** 2)
            scores_x[~np.isfinite(scores_x)] = np.inf
            
            best_idx = scores_x.argmin()
            best_x = x_vals[best_idx]
            best_sc_x = scores_x[best_idx]
            
            if best_sc_x < best_score - 1e-13:
                a[j] = best_x
                s_total = s_base + best_x
                # Update c_full
                a_n = a / a.sum()
                best_score = _score(a_n)
                if best_score < best_sc_x * 1.001:  # consistency check
                    best_a = a_n.copy()
                    c_full = _full_convolve(a_n)
                    a = a_n.copy()
                    s_total = float(a.sum())
                    improved = True
        
        if not improved:
            # Try a few random coordinates as perturbation
            for _ in range(5):
                if time.monotonic() > deadline:
                    break
                j_rand = rng.integers(n) if hasattr(rng, 'integers') else np.random.randint(n)
                a[j_rand] *= (0.5 + np.random.random())
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    c_full = _full_convolve(a)
                    improved = True
    
    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, ptype, p1=1.0, p2=0.0):
    if ptype == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (t - 0.5)
    elif ptype == 'exp':
        t = np.arange(n, dtype=float)
        a = np.exp(p1 * t / n)
    elif ptype == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + p1 * (2 * t - 1) ** 2
    elif ptype == 'sigmoid':
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-p1 * (t - 0.5)))
    elif ptype == 'power':
        t = np.linspace(0.01, 1.0, n)
        a = t ** p1
    elif ptype == 'cosine':
        t = np.arange(n, dtype=float)
        a = 1.0 + p1 * np.cos(2 * np.pi * p2 * t / n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: coordinate descent (exact 1D) + Adam top-k + escalation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent sequence
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    print(f"Parent: n={parent.size if parent is not None else 0}, budget={budget_s}s")
    
    # === Phase 1: Initialization ===
    print("Phase 1: Initialization...")
    
    candidates = []
    
    if parent is not None:
        sc_p = _score(parent)
        candidates.append((sc_p, parent.copy(), parent.size, 'parent'))
        print(f"  Parent score: {sc_p:.10f}, n={parent.size}")
        # Interpolated versions
        for n_try in [512, 1024, 2048]:
            if n_try != parent.size:
                pi = _interp(parent, n_try)
                sc_i = _score(pi)
                candidates.append((sc_i, pi, n_try, 'parent_interp'))
    
    # Generate diverse initializations
    profiles = [
        ('lin1.0', 'linear', 1.0, 0),
        ('lin1.5', 'linear', 1.5, 0),
        ('lin2.0', 'linear', 2.0, 0),
        ('lin3.0', 'linear', 3.0, 0),
        ('exp0.5', 'exp', 0.5, 0),
        ('exp1.0', 'exp', 1.0, 0),
        ('exp2.0', 'exp', 2.0, 0),
        ('exp3.0', 'exp', 3.0, 0),
        ('quad1', 'quadratic', 1.0, 0),
        ('quad2', 'quadratic', 2.0, 0),
        ('sig3', 'sigmoid', 3.0, 0),
        ('sig5', 'sigmoid', 5.0, 0),
        ('sig8', 'sigmoid', 8.0, 0),
        ('pow0.5', 'power', 0.5, 0),
        ('pow1.0', 'power', 1.0, 0),
        ('pow1.5', 'power', 1.5, 0),
        ('pow2.0', 'power', 2.0, 0),
        ('cos05_1', 'cosine', 0.5, 1),
        ('cos08_2', 'cosine', 0.8, 2),
    ]
    
    for n in [256, 512, 1024]:
        for (label, pt, p1, p2) in profiles:
            p = _make_profile(n, pt, p1, p2)
            sc = _score(p)
            candidates.append((sc, p, n, label))
    
    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best init: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")
    
    for sc, seq, n, label in candidates[:3]:
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    
    # === Phase 2: Coordinate Descent on best candidates ===
    print("Phase 2: Coordinate descent...")
    phase2_end = t_start + 500.0
    polished = []
    
    # Sort by (score, n) - prefer smaller n for speed in CD
    candidates_cd = sorted(candidates, key=lambda x: (x[0], x[2]))
    
    count = 0
    for sc, p, n, label in candidates_cd:
        if time.monotonic() > phase2_end or count >= 8:
            break
        if n > 2048:
            continue
        
        # Allocate time based on n
        time_alloc = max(30.0, min(120.0, 500.0 * (256.0 / float(n))))
        per_end = min(phase2_end, time.monotonic() + time_alloc)
        
        a_cd, sc_cd = _coord_descent_vec(p, per_end, max_sweeps=300, n_grid=35, x_mult=10.0)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  BEST: {best_score:.10f} n={a_cd.size} [CD_{label}_n{n}]")
        polished.append((sc_cd, a_cd, n, label + '_cd'))
        count += 1
    
    polished.sort(key=lambda x: x[0])
    print(f"  After Phase 2: best={best_score:.10f}")
    
    # === Phase 3: Adam refinement ===
    print("Phase 3: Adam top-k refinement...")
    phase3_end = t_start + 1000.0
    
    for sc, p, n, label in polished[:5]:
        if time.monotonic() > phase3_end - 50:
            break
        per = min(150.0, phase3_end - time.monotonic() - 20)
        if per < 30:
            break
        
        # Try different k_top values
        for k_top in [5, 10, 20]:
            if time.monotonic() > phase3_end - 30:
                break
            per_k = min(50.0, phase3_end - time.monotonic() - 10)
            if per_k < 15:
                break
            a_ad, sc_ad = _adam_topk(p, time.monotonic() + per_k, rng,
                                     k_top=k_top, lr=0.08, no_imp_limit=200)
            if sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
                print(f"  BEST: {best_score:.10f} n={a_ad.size} [Adam_k{k_top}_{label}]")
    
    # === Phase 4: CD + Adam interleaving ===
    print("Phase 4: CD + Adam interleaving...")
    phase4_end = t_start + 1400.0
    
    if best_seq is not None:
        n_cur = best_seq.size
        cur_a = best_seq.copy()
        
        num_rounds = 6
        per_round = (phase4_end - time.monotonic()) / max(1, num_rounds)
        
        for rnd in range(num_rounds):
            if time.monotonic() > phase4_end - 30:
                break
            per = min(per_round, phase4_end - time.monotonic() - 20)
            if per < 20:
                break
            
            # Adam step
            per_a = per * 0.5
            a_ad, sc_ad = _adam_topk(cur_a, time.monotonic() + per_a, rng,
                                     k_top=10, lr=0.06, no_imp_limit=150)
            if sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
                cur_a = a_ad.copy()
            
            # CD step
            per_c = phase4_end - time.monotonic() - 20
            if per_c > 15:
                a_cd, sc_cd = _coord_descent_vec(cur_a, time.monotonic() + per_c,
                                                 max_sweeps=100, n_grid=35, x_mult=8.0)
                if sc_cd < best_score:
                    best_score = sc_cd
                    best_seq = a_cd.copy()
                    cur_a = a_cd.copy()
                    print(f"  Round {rnd}: BEST={best_score:.10f}")
    
    # === Phase 5: Multi-scale escalation ===
    print("Phase 5: Multi-scale escalation...")
    phase5_end = t_start + 1650.0
    
    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        else:
            target_ns = []
        
        for tn in target_ns:
            if time.monotonic() > phase5_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score(p_int)
            if sc_int < best_score:
                best_score = sc_int
                best_seq = p_int.copy()
                print(f"  Escalated to n={tn}: {sc_int:.10f}")
            
            # CD on escalated
            if time.monotonic() < phase5_end - 30:
                per = min(80.0, phase5_end - time.monotonic() - 20)
                if per > 20:
                    a_e, sc_e = _coord_descent_vec(p_int, time.monotonic() + per,
                                                   max_sweeps=80, n_grid=30, x_mult=8.0)
                    if sc_e < best_score:
                        best_score = sc_e
                        best_seq = a_e.copy()
                        print(f"  CD escalated n={tn}: {sc_e:.10f}")
            
            # Adam on escalated
            if time.monotonic() < phase5_end - 20:
                per = min(50.0, phase5_end - time.monotonic() - 10)
                if per > 15:
                    a_ea, sc_ea = _adam_topk(p_int, time.monotonic() + per, rng,
                                             k_top=10, lr=0.05, no_imp_limit=100)
                    if sc_ea < best_score:
                        best_score = sc_ea
                        best_seq = a_ea.copy()
                        print(f"  Adam escalated n={tn}: {sc_ea:.10f}")
    
    # === Phase 6: Random restarts ===
    print("Phase 6: Random restarts...")
    phase6_end = deadline - 60.0
    
    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = best_seq.size
        n_restarts = 6
        per_r = max(10.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 8:
                break
            
            ns = 0.05 + 0.05 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -40, 40)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()
            
            # Quick CD
            per_cd = per * 0.6
            a_r, sc_r = _coord_descent_vec(a_s, time.monotonic() + per_cd,
                                           max_sweeps=50, n_grid=30, x_mult=6.0)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {i}: BEST={best_score:.10f}")
            
            # Quick Adam
            if time.monotonic() < phase6_end - 5:
                per_ad = phase6_end - time.monotonic() - 5
                if per_ad > 10:
                    a_ra, sc_ra = _adam_topk(a_r, time.monotonic() + per_ad, rng,
                                             k_top=10, lr=0.05, no_imp_limit=100)
                    if sc_ra < best_score:
                        best_score = sc_ra
                        best_seq = a_ra.copy()
                        print(f"  Restart {i} Adam: BEST={best_score:.10f}")
    
    # === Phase 7: Final polish ===
    print("Phase 7: Final polish...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 15:
            a_fp, sc_fp = _coord_descent_vec(best_seq, time.monotonic() + per,
                                             max_sweeps=200, n_grid=50, x_mult=10.0)
            if sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()
                print(f"  Final CD: BEST={best_score:.10f}")
    
    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0)
        
        exact_sc = _score(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")
        
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")
        
        return [float(x) for x in final]
    
    # Fallback
    fb = _make_profile(512, 'linear', 1.5, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
