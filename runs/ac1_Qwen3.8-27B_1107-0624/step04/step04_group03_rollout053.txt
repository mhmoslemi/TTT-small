    c = _full_convolve(a, a)
    w, cm = _compute_soft_weights(c, beta)
    log_sum = 0.0
    # Compute log-sum-exp of (c - cm)/beta
    cs = (c - cm) / beta
    cs_max = np.max(cs)
    log_sum = float(cs_max + np.log(np.sum(np.exp(cs - cs_max))))
    return 2.0 * n * (cm + beta * log_sum)


def _gradient_soft(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_soft_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)

    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):

    return grad

    profiles.append(p / p.sum())
    
    # Sine
    p = np.sin(np.pi * t) + 0.1
    profiles.append(p / p.sum())
    
    # Cosine
    p = np.cos(np.pi * t) ** 2 + 0.1
    profiles.append(p / p.sum())
    
    # Exponential
    p = np.exp(2.0 * (t - 0.5))
    profiles.append(p / p.sum())
    
    # Exponential reversed
    p = np.exp(2.0 * (0.5 - t))
    profiles.append(p / p.sum())
    
    # Power law
    p = (t + 0.01) ** 2
    profiles.append(p / p.sum())
    
    # Uniform
    p = np.ones(n)
    profiles.append(p / p.sum())
    
    return profiles


def _nesterov_loop(a_init, deadline, beta_start, beta_end, n_min_iters=100):
    """Nesterov-accelerated gradient descent with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    
    beta = beta_start * max_c
    mom = 0.9
    lr = 0.01
    t_param = 0
    y = z.copy()
    
    iters_per_beta = max(n_min_iters, n // 2)
    
    # Create geometric beta schedule
    n_beta_steps = max(10, int(np.log(beta_start / beta_end) / np.log(2)))
    beta_schedule = np.geomspace(beta_start * max_c, beta_end * max_c, n_beta_steps)
    
    for beta_idx, beta in enumerate(beta_schedule):
        if time.monotonic() >= deadline:
            break
        
        for it in range(iters_per_beta):
            if time.monotonic() >= deadline:
                break
            
            t_param += 1
            a_y = _softmax_fn(y)
            c_y = _full_convolve(a_y, a_y)
            w, cm = _compute_soft_weights(c_y, beta)
            W = float(np.dot(w, c_y))
            a_rev = a_y[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1: 2 * n - 1]
            grad = 4.0 * n * a_y * (h - W)
            
            if not np.all(np.isfinite(grad)):
                continue
            
            # Nesterov step
            z_new = z - lr * grad
            z_new = z_new - z_new.mean()
            z_new = np.clip(z_new, -60, 60)
            
            y_new = z_new + mom * (z_new - z)
            y_new = y_new - y_new.mean()
            
            z = z_new
            y = y_new
            
            # Check improvement
            if t_param % 100 == 0:
                a_curr = _softmax_fn(z)
                sc = _score_norm_fn(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
            
            # Adapt learning rate
            if t_param % 200 == 0:
                lr = lr * 0.95
                mom = min(0.99, mom + 0.001)
    
    # Final check
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _lbfgs_phase(a_init, deadline, max_wall_time=None):
    """L-BFGS-B optimization with continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if max_wall_time is None:
        max_wall_time = min(300.0, max(1.0, deadline - time.monotonic()))
    
    # Beta schedule: geometric from large to small
    beta_fracs = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.00075, 0.0004, 0.0002, 0.0001, 5e-5]
    betas = [b * max_c0 for b in beta_fracs]
    
    per_level = max_wall_time / len(betas)
    
    for beta in betas:
        if time.monotonic() >= deadline:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        try:
            result = minimize(
                _objective_soft, z, args=(beta, n),
                method='L-BFGS-B', jac=_gradient_soft,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 50, 'maxfun': 2000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass
    
    return best_a, best_score


def _active_set_descent(a_init, deadline, n_iter=20000):
    """
    Subgradient descent focusing only on the active (max-achieving) lags.
    Uses Barzilai-Borwein step size.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Barzilai-Borwein state
    s_prev = None
    y_prev = None
    lr = 0.001
    momentum = np.zeros(n)
    mom_coeff = 0.8
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Active set: top few lags
        # Use lags within 0.1% of max
        threshold = cm * (1.0 - 0.001)
        active = c >= threshold
        if not np.any(active):
            active[int(np.argmax(c))] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / float(np.sum(active))
        W = float(np.dot(w, c))
        
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1: 2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            continue
        
        # Barzilai-Borwein step size
        if s_prev is not None and y_prev is not None:
            sy = float(np.dot(s_prev, y_prev))
            if abs(sy) > 1e-10:
                ss = float(np.dot(s_prev, s_prev))
                yy = float(np.dot(y_prev, y_prev))
                # BB1
                lr_bb = ss / sy if sy > 0 else lr
                lr_bb = min(max(lr_bb, 1e-6), 0.1)
                lr = lr_bb
        
        # Momentum step
        momentum = mom_coeff * momentum - lr * grad
        z_new = z + momentum
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -60, 60)
        
        # Store for BB
        z_curr = z
        z = z_new
        s_prev = z - z_curr
        # y will be computed next iteration
        if y_prev is not None:
            y_prev_grad = grad  # approximate
        # Actually, let me simplify: just use adaptive lr
        
        if it % 500 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            else:
                lr = max(1e-6, lr * 0.9)
        
        # Reduce momentum over time
        if it % 1000 == 0:
            mom_coeff = max(0.5, mom_coeff - 0.05)
            lr = max(1e-6, lr * 0.95)
    
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _adam_loop(a_init, deadline, lr=0.02, beta_frac=0.005, n_max_iter=50000):
    """Adam optimizer with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)
    
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        w, cm = _compute_soft_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1: 2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            continue
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        if t % 500 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            beta = max(1e-12, beta * 0.85)
    
    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_a_score
