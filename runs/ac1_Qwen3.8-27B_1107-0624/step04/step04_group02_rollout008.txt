"""
Optimized solver using projected gradient with top-K smoothing,
L-BFGS with logsumexp continuation, deflation steps,
and coordinate-swap local search.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Sort in descending order
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    rho = np.arange(1, n + 1)
    cond = (sv * rho - (cv - target_sum)) > 0
    # Actually the condition for the water-filling is:
    # rho = max{i : sv[i] > (cv[i] - target_sum) / i}
    # which is equivalent to: sv[i] * i > cv[i] - target_sum
    # i.e., sv[i] > cv[i]/i - target_sum/i
    # or: sv[i] * i - cv[i] + target_sum > 0
    # Hmm let me just use the standard algorithm
    # From the paper by Duchi et al. (2008):
    # Sort v in descending order: v_{(1)} >= v_{(2)} >= ... >= v_{(n)}
    # Find rho = max{i : v_{(i)} > (sum_{j<=i} v_{(j)} - target_sum) / i}
    # theta = (sum_{j<=rho} v_{(j)} - target_sum) / rho
    # x_i = max(v_i - theta, 0)
    
    # Compute (cv[i] - target_sum) / i for each i
    # Find the largest i where sv[i-1] > (cv[i-1] - target_sum) / i
    thresh = (cv - target_sum) / rho
    # cond[i-1] = sv[i-1] > thresh[i-1]
    cond = sv > thresh
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_val = np.max(np.where(cond)[0]) + 1
    theta = (cv[rho_val - 1] - target_sum) / rho_val
    return np.maximum(v - theta, 0.0)


def _compute_gradient_topk(a, K):
    """Compute gradient of top-K average of conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    # Get top K indices
    if K >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -K)[-K:]
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            # Sum m from lo to hi: a[m] * a[k-m], derivative wrt a[j] is 2*a[k-j]
            # So g[j] += 2 * a[k-j] for j in [lo, hi]
            # a[k-j] for j in [lo, hi] is a[k-hi : k-lo+1][::-1]
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(top_idx)
    return g, float(c.max())


def _proj_grad_topk(a_init, deadline, K, lr_init=0.005, decay=0.999, min_lr=1e-5):
    """Projected gradient descent on top-K average with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    lr = lr_init
    momentum = 0.9
    
    # Nesterov momentum
    y = a.copy()
    
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        
        # Nesterov step
        y = a + momentum * (a - a_prev)
        
        # Compute gradient at y
        g, cmax = _compute_gradient_topk(y, K)
        
        # Take step
        a_new = y - lr * g
        a_new = _proj_simplex(a_new, 1.0)
        
        # Check improvement
        sc = _score_norm_fn(a_new)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a_prev = a
        a = a_new
        
        # Decay learning rate
        lr = max(min_lr, lr * decay)
        
        # Occasionally recompute with smaller K
        if it % 10000 == 0 and it > 0:
            K = max(1, K // 2)
    
    return best_a, best_score

"""
Optimizer combining L-BFGS with logsumexp continuation, projected gradient
with top-K smoothing, coordinate-swap local search, deflation refinement,
and multi-scale exploration. Designed to achieve sub-1.503 scores.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    rho = np.arange(1, n + 1)
    thresh = (cv - target_sum) / rho
    cond = sv > thresh
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_val = int(np.max(np.where(cond)[0])) + 1
    theta = (cv[rho_val - 1] - target_sum) / rho_val
    return np.maximum(v - theta, 0.0)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_topk(a, K):
    n = len(a)
    c = _full_convolve(a, a)
    if K >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -K)[-K:]
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(top_idx)
    return g, float(c.max())


def _compute_grad_logsumexp(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    w = e / e.sum()
    w_flip = w[::-1]
    cw = _full_convolve(a, w_flip)
    g = (2.0 / beta) * cw[2 * n - 2:n - 1:-1]
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return g, cm


def _proj_grad_logsumexp(a_init, deadline, beta_init, beta_min, n_halve=500):
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    beta = beta_init
    lr = 0.05
    momentum = 0.85

        a_new = _proj_simplex(a_new, 1.0)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:

        a = a_new

            beta = max(beta_min, beta * 0.5)

    return best_a, best_score

def _run_lbfgs_continuation(a_init, deadline, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    betas = [b * max_c0 for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(3.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_sh = (c_ - cm_) / beta
            e_ = np.exp(c_sh)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_sh = (c_ - cm_) / beta
            e_ = np.exp(c_sh)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-8,
                         'maxls': 40, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _coordinate_swap_search(a_init, deadline, rng, n_trials=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    c = _full_convolve(a, a)
    cmax = float(c.max())

    for it in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Pick two coordinates to swap mass between
        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue

        # Try small swap: move delta from k to j
        delta = a[k] * 0.05
        if delta < 1e-15:
            continue

        a_trial = a.copy()
        a_trial[j] += delta
        a_trial[k] -= delta
        if a_trial[k] < 0:
            continue

        # Quick score check
        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_score


def _block_search(a_init, deadline, rng, n_trials=5000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(n_trials):
        if time.monotonic() > deadline:
            break

        block_size = int(rng.integers(4, min(32, n // 4)))
        start = int(rng.integers(0, n - block_size))

        a_trial = a.copy()
        perturbation = rng.uniform(0.9, 1.1, size=block_size)
        a_trial[start:start + block_size] *= perturbation
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc = _score_norm_fn(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        elif sc < best_score * 1.005 and rng.random() < 0.05:
            a = a_trial.copy()

    return best_a, best_score


def _deflation_step(a_init, deadline, rng, n_steps=100):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cmax = float(c.max())
        # Find top lag
        k_star = int(np.argmax(c))

        # Compute "pressure" on each coordinate from the worst lag
        # a_j contributes a_{k_star - j} to c_{k_star}
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        pressure = np.zeros(n)
        pressure[lo:hi + 1] = a[k_star - hi:k_star - lo + 1][::-1]

        # Reduce high-pressure coordinates, increase low-pressure ones
        alpha = 0.01 / (step + 1)
        a_new = a.copy()
        a_new[lo:hi + 1] -= alpha * pressure[lo:hi + 1]
        a_new = np.maximum(a_new, 0.0)

        # Redistribute the removed mass to low-pressure coordinates
        removed = (a[lo:hi + 1] - a_new[lo:hi + 1]).sum()
        if removed > 0:
            # Add to coordinates outside the active range
            mask = np.zeros(n, dtype=bool)
            mask[lo:hi + 1] = False
            if mask.any():
                targets = np.where(mask)[0]
                # Distribute proportionally to current values (or uniformly if all zero)
                t_vals = a[targets]
                if t_vals.sum() > 0:
                    frac = t_vals / t_vals.sum()
                else:
                    frac = np.ones(len(targets)) / len(targets)
                a_new[targets] += removed * frac

        # Renormalize
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        else:
            a_new = a

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def _make_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = np.exp(-u ** 2 / (2.0 * width ** 2))
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
