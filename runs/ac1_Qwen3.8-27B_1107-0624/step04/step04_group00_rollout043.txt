4. Stationarity: 
   - ∂L/∂t = 1 - sum_k μ_k = 0, so sum_k μ_k = 1.
   - ∂L/∂a_i = sum_k μ_k * a_{k-i} - ν_i - ν_0 = 0 (for valid k-i).

From stationarity: sum_k μ_k * a_{k-i} = ν_0 + ν_i for all i.

For indices where a_i > 0 (ν_i = 0): sum_k μ_k * a_{k-i} = ν_0.
For indices where a_i = 0 (ν_i >= 0): sum_k μ_k * a_{k-i} <= ν_0.

So the condition is: there exist weights μ_k >= 0 (sum to 1) such that:
- For all i with a_i > 0: sum_k μ_k * a_{k-i} = ν_0 (a constant).
- For all i with a_i = 0: sum_k μ_k * a_{k-i} <= ν_0.

This means: the weighted average of the "correlations" at the active lags is constant (equal to ν_0) at all positions where a_i > 0.

This is a nice characterization! It means that at the optimum, the "influence" of the active peaks is uniform across all non-zero positions of a.

This gives us a check: given a candidate optimum a*, compute C, find the active set, and check if there exist weights that make the weighted correlation uniform.

For the optimization, we can use this to guide the search:
- If the weighted correlation is not uniform, move mass from positions with high weighted correlation to positions with low weighted correlation.

def _project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1} using water-filling."""
    n = len(v)
    w = np.sort(v)[::-1]
    cv = np.cumsum(w)
    rho = np.where(w * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_topk(c, a, K):
    """
    Compute average subgradient over top-K peaks.
    g[i] = (1/K) * sum_{j} a[k_j - i] for valid indices.

    for k in top_idx:
        k = int(k)

        if i_min <= i_max:

def _make_inits(n, rng):
    """Generate diverse initial sequences (all normalized to sum=1)."""
    t = np.linspace(0, 1, n)
    inits = []

def _optimize_mirror_descent(a_init, deadline, K start=5, K_end=15, mom=0.85):

    Mirror descent (exponentiated gradient) with adaptive K and momentum.
    Works in log-space implicitly through multiplicative updates.
    """

    z = np.log(np.maximum(a_init, 1e-15))
    z -= z.mean()
    best_a = a_init.copy()
    best_obj = float('inf')
    best_obj = _obj(a_init)

    momentum = np.zeros(n)
    it = 0
    last_improve = 0
    base_alpha = 0.5

    while time.monotonic() < deadline:
        it += 1
        a = np.exp(z - z.max())
        a /= a.sum()

        c = _full_convolve(a, a)
        obj = 2.0 * n * c.max()

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            last_improve = it

        # Adaptive K: start small, grow over time
        progress = min(1.0, it / 50000)
        K = max(K_start, int(K_start + (K_end - K_start) * progress))

        g = _subgradient_topk(c, a, K)

        # Adaptive step size
        alpha = base_alpha / (1 + 0.01 * progress)

        # Multiplicative update with momentum
        momentum = mom * momentum - alpha * g
        z = z + momentum
        z = np.clip(z, -100, 100)
        z -= z.mean()

        # Periodic restart of momentum if stuck
        if it - last_improve > 5000:
            momentum *= 0.5
            # Small random kick
            kick_scale = 0.01 * (1 + it / 50000)
            z = z + kick_scale * np.random.default_rng(it).standard_normal(n)
            z -= z.mean()
            last_improve = it

def _optimize_projected(a_init, deadline, lr_init=0.05, K=5):
    """
    Projected gradient descent with Barzilai-Borwein step size.
    """
    n = len(a_init)
    a = a_init.copy()
    a /= a.sum()
    best_a = a.copy()
    best_obj = _obj(a)

    s_prev = None
    g_prev = None
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() < deadline:
        it += 1
        c = _full_convolve(a, a)
        obj = 2.0 * n * c.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        g = _subgradient_topk(c, a, K)

        # Barzilai-Borwein step size
        if s_prev is not None and g_prev is not None:
            y = g - g_prev
            sy = float(np.dot(s_prev, y))
            ss = float(np.dot(s_prev, s_prev))
            yy = float(np.dot(y, y))
            if sy > 1e-20:
                alpha = ss / sy
            elif yy > 1e-20:
                alpha = yy / max(ss, 1e-20)
            else:
                alpha = lr_init
            alpha = float(np.clip(alpha, 1e-6, 1.0))
        else:
            alpha = lr_init

        # Decay alpha slowly
        alpha *= max(0.5, 1.0 - 1e-6 * it)

        a_new = a - alpha * g
        a_new = _project_simplex(a_new)

        s_prev = a_new - a
        c_new = _full_convolve(a_new, a_new)
        g_new = _subgradient_topk(c_new, a_new, K)
        g_prev = g
        a = a_new

        # If stuck, try a larger random step
        if stuck_count > 2000:
            scale = 0.005 + 0.001 * (stuck_count // 2000)
            noise = np.random.default_rng(42 + it).standard_normal(n)
            a_trial = a + scale * noise
            a_trial = _project_simplex(a_trial)
            obj_trial = _obj(a_trial)
            if obj_trial < best_obj:
                a = a_trial
                best_obj = obj_trial
                best_a = a_trial.copy()
            stuck_count = 0

    return best_a, best_obj


def _optimize_hybrid(a_init, deadline, K):

    """
    n = len(a_init)
    a = a_init.copy()
    a /= a.sum()
    best_a = a.copy()
    best_obj = _obj(a)

    it = 0
    last_improve = 0
    mode = 0  # 0 = multiplicative, 1 = additive

    while time.monotonic() < deadline:
        it += 1
        c = _full_convolve(a, a)
        obj = 2.0 * n * c.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            last_improve = it

        g = _subgradient_topk(c, a, K)

        # Switch modes periodically
        if it % 10000 == 0:
            mode = 1 - mode

        if mode == 0:
            # Multiplicative (mirror descent)
            alpha = 0.3 / (1 + 0.005 * (it // 10000))
            z = np.log(np.maximum(a, 1e-15))
            z -= z.mean()
            z -= alpha * g / np.maximum(a, 1e-10)
            z = np.clip(z, -80, 80)
            a_new = np.exp(z - z.max())
            a_new /= a_new.sum()
        else:
            # Additive (projected gradient)
            alpha = 0.02 / (1 + 0.005 * (it // 10000))
            a_new = a - alpha * g
            a_new = _project_simplex(a_new)

        a = a_new

        # Random restart if stuck
        if it - last_improve > 20000:
            scale = 0.02
            noise = np.random.default_rng(100 + it).standard_normal(n)
            a_trial = a + scale * noise
            a_trial = _project_simplex(a_trial)
            obj_trial = _obj(a_trial)
            if obj_trial < best_obj:
                a = a_trial
                best_obj = obj_trial
                best_a = a_trial.copy()
                last_improve = it
            else:
                a = best_a.copy()
                last_improve = it

    return best_a, best_obj


def _peak_targeted(a_init, deadline, n_reductions=100):
    """
    Targeted peak reduction: identify top contributors to the peak
    and iteratively reduce them while compensating elsewhere.
    """
    n = len(a_init)
    a = a_init.copy()
    a /= a.sum()
    best_a = a.copy()
    best_obj = _obj(a)

    for phase in range(n_reductions):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        obj = 2.0 * n * c.max()

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Find top peaks
        top_peaks = np.argsort(c)[-5:]

        for k in top_peaks:
            k = int(k)
            # Find contributing indices
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            contribs = a[idx] * a[k - idx]
            total = contribs.sum()
            if total < 1e-15:
                continue

            # Top contributors
            top_contrib = np.argsort(contribs)[-3:]

            for tc in top_contrib:
                i = int(idx[tc])
                partner = k - i

                # Try reducing a[i] (and or a[partner])
                for frac in [0.01, 0.02, 0.05]:
                    reduction = a[i] * frac
                    if reduction < 1e-15:
                        continue

                    # Redistribute to positions with low contribution to all top peaks
                    # Simple: add to the minimum element
                    min_pos = int(np.argmin(a))
                    a_trial = a.copy()
                    a_trial[i] -= reduction
                    a_trial[min_pos] += reduction
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial /= s

                    obj_trial = _obj(a_trial)
                    if obj_trial < best_obj:
                        best_obj = obj_trial
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        break

    return best_a, best_obj


def _kkt_detect_active(a):
    """
    Detect the active set and check KKT conditions.
    Returns (active_lags, kkt_residual).
    """
    n = len(a)
    c = _full_convolve(a, a)
    cm = c.max()
    # Active set: lags within 1% of max
    tolerance = 0.01 * cm
    active = np.where(c >= cm - tolerance)[0]
    return active, c, cm


def _kkt_refine(a, deadline, n_iter=500):
    """
    Use KKT conditions to refine: find weights w_k for active lags
    such that sum_k w_k * a[k-i] is as constant as possible.
    """
    n = len(a)
    a_best = a.copy()
    best_obj = _obj(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = c.max()
        obj = 2.0 * n * cm

        if obj < best_obj:
            best_obj = obj
            a_best = a.copy()

        # Get top-K lags
        K = min(20, 2 * n - 1)
        top_k = np.argsort(c)[-K:]

        # Compute: for each position i, h[i] = (1/K) sum_{k in top} a[k-i]
        h = np.zeros(n)
        for k in top_k:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_min <= i_max:
                h[i_min:i_max + 1] += a[k - i_max:k - i_min + 1][::-1]
        h /= K

        # We want h to be as constant as possible where a > 0
        # Gradient of variance of h (restricted to a>0) w.r.t. a
        # Simpler: just use the spread of h as a measure
        mask = a > 1e-10
        if mask.sum() < 2:
            break
        h_mean = h[mask].mean()
        h_spread = h[mask].max() - h[mask].min()

        if h_spread < 1e-10:
            break

        # Move mass from high-h positions to low-h positions
        h_high = h > h_mask + 0.1 * h_spread if False else h > h_mean + 0.1 * h_spread
        h_low = h < h_mask - 0.1 * h_spread if False else h < h_mean - 0.1 * h_spread

        # Actually this approach is too crude. Let's just use the subgradient.
        g = _subgradient_topk(c, a, K)
        alpha = 0.01 * (1 + h_spread)
        a_new = a - alpha * g
        a_new = _project_simplex(a_new)

        obj_new = _obj(a_new)
        if obj_new < obj:
            a = a_new
        else:
            alpha *= 0.5
            a_new = a - alpha * g
            a_new = _project_simplex(a_new)
            if _obj(a_new) < obj:
                a = a_new

    return a_best, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimization: mirror descent, projected gradient,
    hybrid, peak-targeted, and KKT refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_a = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_a, best_score
        if p is not None:
            s = float(p.sum())
            if s > 0.01:
                p_norm = p / s
                actual_sc = _score(p_norm)
                if np.isfinite(actual_sc) and actual_sc < best_score:
                    best_a = p_norm.copy()
                    best_score = float(actual_sc)
                    print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        n_default = 2048
        parent = np.ones(n_default) / n_default
        update_best(parent, _score(parent), "uniform_default")

    n_parent = len(parent)

    # --- Phase 1: Try different n and initializations (120s) ---
    phase1_end = min(deadline, t_start + 120.0)
    print("Phase 1: Initialization search")

    # Try the parent at its native n
    candidates = []
    candidates.append((n_parent, parent, "parent"))

    # Try a few other n values
    for n_try in [1024, 2048, 3072, 4096, 5120, 6144]:
        if n_try == n_parent:
            continue
        p_interp = _interp_profile(parent, n_try)
        sc = _score(p_interp)
        candidates.append((n_try, p_interp, f"interp_n{n_try}"))
        update_best(p_interp, sc, f"init_interp{n_try}")

    # Also try random inits at parent n
    for i, init in enumerate(_make_inits(n_parent, rng)[:5]):
        sc = _score(init)
        update_best(init, sc, f"init_smooth{i}")
        candidates.append((n_parent, init, f"smooth{i}"))

    # Pick the best initial for main optimization
    best_init = best_a.copy() if best_a is not None else parent
    n_best = len(best_init)
    print(f"  Phase 1 best: score={best_score:.10f}, n={n_best}")

    # --- Phase 2: Mirror descent (500s) ---
    phase2_end = min(deadline, t_start + 620.0)
    print(f"Phase 2: Mirror descent (n={n_best})")

    if time.monotonic() < phase2_end:
        per_dl = min(phase2_end - 5, time.monotonic() + 480.0)
        a_best, obj_best = _optimize_mirror_descent(
            best_init, per_dl, K_start=3, K_end=10, mom=0.8
        )
        update_best(a_best, _score(a_best), "mirror_descent")
        best_init = a_best

    # --- Phase 3: Projected gradient (400s) ---
    phase3_end = min(deadline, t_start + 1020.0)
    print(f"Phase 3: Projected gradient (n={n_best})")

    if time.monotonic() < phase3_end:
        per_dl = min(phase3_end - 5, time.monotonic() + 380.0)
        a_best, obj_best = _optimize_projected(
            best_init, per_dl, lr_init=0.02, K=5
        )
        update_best(a_best, _score(a_best), "projected_grad")
        best_init = a_best

    # --- Phase 4: Hybrid optimization (300s) ---
    phase4_end = min(deadline, t_start + 1320.0)
    print(f"Phase 4: Hybrid (n={n_best})")

    if time.monotonic() < phase4_end:
        per_dl = min(phase4_end - 5, time.monotonic() + 280.0)
        a_best, obj_best = _optimize_hybrid(best_init, per_dl, K=5)
        update_best(a_best, _score(a_best), "hybrid")
        best_init = a_best

    # --- Phase 5: Peak targeted reduction (200s) ---
    phase5_end = min(deadline, t_start + 1520.0)
    print(f"Phase 5: Peak targeted (n={n_best})")

    if time.monotonic() < phase5_end:
        per_dl = min(phase5_end - 5, time.monotonic() + 180.0)
        a_best, obj_best = _peak_targeted(best_init, per_dl, n_reductions=200)
        update_best(a_best, _score(a_best), "peak_targeted")
        best_init = a_best

    # --- Phase 6: KKT refinement (150s) ---
    phase6_end = min(deadline, t_start + 1670.0)
    print(f"Phase 6: KKT refinement (n={n_best})")

        a_best, obj_best = _kkt_refine(best_init, per_dl, n_iter=2000)

        best_init = a_best

        per_dl = min(phase7_end - 10, time.monotonic() + (phase7_end - time.monotonic()) * 0.6)
        a_best, _ = _optimize_projected(best_a, per_dl, lr_init=0.005, K=3)
        update_best(a_best, _score(a_best), "final_fine")

        if time.monotonic() < phase7_end - 5:
            per_dl2 = min(phase7_end - 5, time.monotonic() + (phase7_end - time.monotonic()) * 0.4)
            a_best2, _ = _optimize_mirror_descent(best_a, per_dl2, K_start=2, K_end=5, mom=0.7)
            update_best(a_best2, _score(a_best2), "final_mirror")

    if time.monotonic() < deadline - 60 and best_a is not None:

        if n_new <= 8192:
            p_interp = _interp_profile(best_a, n_new)
            sc = _score(p_interp)
            update_best(p_interp, sc, f"try_n{n_new}")
            if sc < best_score:
                per_dl = min(deadline - 30, time.monotonic() + 50.0)
                a_best, _ = _optimize_projected(p_interp, per_dl, lr_init=0.01, K=5)
                update_best(a_best, _score(a_best), f"opt_n{n_new}")

        final_seq = np.clip(best_a, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(256) / 256

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

                    if sc_eval < best_score:
                        best_score = sc_eval
