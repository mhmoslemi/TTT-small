Uses multi-scale warm start, adaptive log-sum-exp smoothing with L-BFGS-B,

import sys

N_MAIN = 4096
_N_SMALL = 512

_FFT_SELF = 1 << (2 * N_MAIN - 1).bit_length()
_FFT_GRAD = 1 << (3 * N_MAIN - 1).bit_length()

_FFT_SELF_S = 1 << (2 * _N_SMALL - 1).bit_length()
_FFT_GRAD_S = 1 << (3 * _N_SMALL - 1).bit_length()

_BOUNDS = [(-40.0, 40.0)] * (N_MAIN - 1)

_best_score = float("inf")
_best_z = None
_search_deadline = 0.0


class SearchTimeout(Exception):

def _sanitized_z(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -40.0, 40.0)
    return x


def _probs_from_z(z, n):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -40.0, 40.0)
    m = max(0.0, float(np.max(z))) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(n, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _self_conv(p, fft_size):
    f = np.fft.rfft(p, fft_size)
    c = np.fft.irfft(f * f, fft_size).real
    n = len(p)
    c = c[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _grad_p_from_weights(p, w, n, fft_size):
    """Compute gradient of sum_k w_k * c_k with respect to p."""
    conv = _full_conv_padded(p[::-1], w, fft_size)
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _full_conv_padded(a, b, fft_size):
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size).real


def _score_p(p, n):
    c = _self_conv(p, _FFT_SELF if n == N_MAIN else _FFT_SELF_S)
    return 2.0 * float(n) * float(np.max(c))


def _update_best(x, score):
    global _best_score, _best_z
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_z = np.array(x, copy=True)


def _score_z(z):
    p = _probs_from_z(z, N_MAIN)
    s = _score_p(p, N_MAIN)
    _update_best(z, s)
    return s


# ---------------------------------------------------------------------------
# Objective and gradient for L-BFGS-B
# ---------------------------------------------------------------------------

def _objective_and_grad(x, alpha, n, fft_self, fft_grad):
    if time.monotonic() >= _search_deadline:
        raise SearchTimeout

    x = _sanitized_z(x)
    p = _probs_from_z(x, n)
    c = _self_conv(p, fft_self)

    hard = 2.0 * float(n) * float(np.max(c))
    _update_best(x, hard)

    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        J = cmax
        w = np.zeros_like(c)
        idx = int(np.argmax(c))
        w[idx] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s

    gp = _grad_p_from_weights(p, w, n, fft_grad)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    return float(J) * 2.0 * float(n), gz

def _optimize_z(z0, alpha, max_fun, deadline, n=N_MAIN):
    global _search_deadline
    if minimize is None:
        return

    _search_deadline = float(deadline)
    x0 = _sanitized_z(z0)

    if n == N_MAIN:
        fft_self = _FFT_SELF
        fft_grad = _FFT_GRAD
        bounds = _BOUNDS
    else:
        fft_self = _FFT_SELF_S
        fft_grad = _FFT_GRAD_S
        bounds = [(-40.0, 40.0)] * (n - 1)

    max_fun = int(max(50, max_fun))
    max_iter = int(max(30, max_fun // 2))

    try:
        minimize(
            _objective_and_grad,
            x0,
            args=(float(alpha), n, fft_self, fft_grad),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxfun": max_fun,
                "maxiter": max_iter,
                "ftol": 1e-14,
                "gtol": 1e-9,
                "maxls": 20,
                "disp": False,
            },
        )
    except SearchTimeout:
        return
    except Exception:
        return

def _density_to_z(f, n):
    f = np.asarray(f, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 1e-12)
    logf = np.log(f)
    z = logf[1:] - logf[0]
    return np.clip(z, -40.0, 40.0)


def _logistic_z(beta, n=N_MAIN):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    u = beta * (x - 0.5)
    u = np.clip(u, -50.0, 50.0)
    f = 2.0 / (1.0 + np.exp(-u))
    return _density_to_z(f, n)


def _linear_z(rho, n=N_MAIN):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = 1.0 + rho * (2.0 * x - 1.0)
    f = np.maximum(f, 1e-8)
    return _density_to_z(f, n)


def _power_z(a, reverse=False, n=N_MAIN):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    if reverse:
        f = np.power(1.0 - x + 1e-4, a)
    else:
        f = np.power(x + 1e-4, a)
    f = np.maximum(f, 1e-12)
    return _density_to_z(f, n)


def _exp_z(beta, n=N_MAIN):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = np.exp(beta * x)
    return _density_to_z(f, n)


def _gaussian_z(sigma, center=0.5, n=N_MAIN):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return _density_to_z(f, n)


def _bimodal_z(sep, width, n=N_MAIN):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = np.exp(-((x - (0.5 - sep)) / width) ** 2) + np.exp(-((x - (0.5 + sep)) / width) ** 2)
    return _density_to_z(f, n)


def _qresidue_z(n=N_MAIN):
    """Quadratic residue-based initialization."""
    f = np.ones(n, dtype=np.float64)
    p = 1
    while p < n:
        p *= 2
    p = 2 * p + 1  # next prime-ish (not guaranteed prime but ok for init)
    # Use a simpler construction: mark positions that are quadratic residues mod p
    for i in range(n):
        r = (i * i) % p
        if r < n:
            f[r] += 2.0
    return _density_to_z(f, n)


def _random_z(rng, n=N_MAIN):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    t = x - 0.5
    z = np.zeros(n - 1, dtype=np.float64)
    z += rng.uniform(-3.0, 3.0) * t
    z += rng.uniform(-1.5, 1.5) * t * t
    for fq in (1, 2, 3, 4, 6, 8, 12):
        amp = rng.normal(0.0, 2.0 / (fq + 0.5))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.sin(2.0 * np.pi * fq * x + ph)
    return np.clip(z, -40.0, 40.0)


def _poly_random_z(rng, n=N_MAIN):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    t = x - 0.5
    deg = rng.integers(3, 8)
    coef = rng.normal(scale=1.5, size=deg + 1)
    val = np.ones(n, dtype=np.float64)
    f = np.ones(n, dtype=np.float64)
    for k in range(deg, 0, -1):
        val = val * t + coef[k]
    f += val
    f = np.maximum(f, 1e-8)
    return _density_to_z(f, n)


def _harmonic_z(rng, n=N_MAIN):
    """Initialize with a few sine components."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = np.ones(n, dtype=np.float64)
    n_harm = rng.integers(2, 6)
    for _ in range(n_harm):
        freq = rng.integers(1, n // 4)
        amp = rng.uniform(0.1, 0.6)
        phase = rng.uniform(0, 2 * np.pi)
        f += amp * np.sin(2 * np.pi * freq * x + phase)
    f = np.maximum(f, 1e-8)
    return _density_to_z(f, n)

def _make_random_candidates(rng, count, n=N_MAIN):

def _upsample_z(z_small, n_small, n_large):
    """Upsample a probability vector from n_small to n_large."""
    p_small = _probs_from_z(z_small, n_small)
    old_x = np.linspace(0.0, 1.0, n_small)
    new_x = np.linspace(0.0, 1.0, n_large)
    p_large = np.interp(new_x, old_x, p_small)
    p_large = np.maximum(p_large, 0.0)
    if p_large.sum() < 1e-12:
        return np.zeros(n_large - 1, dtype=np.float64)
    p_large /= p_large.sum()
    return _density_to_z(p_large, n_large)

    for i, z in enumerate(cands):
        if n == N_MAIN:
            scores[i] = _score_z(z)
        else:
            p = _probs_from_z(z, n)
            fft_s = _FFT_SELF_S
            scores[i] = 2.0 * float(n) * float(np.max(_self_conv(p, fft_s)))
    order = np.argsort(scores)
    return [cands[int(j)] for j in order]


# ---------------------------------------------------------------------------
# Subgradient polishing
# ---------------------------------------------------------------------------

def _polish_subgradient(z0, deadline, rng, n=N_MAIN):
    z = _sanitized_z(z0)
    p = _probs_from_z(z, n)
    score = _score_p(p, n)
    if n == N_MAIN:
        _update_best(z, score)
    fft_s = _FFT_SELF if n == N_MAIN else _FFT_SELF_S
    fft_g = _FFT_GRAD if n == N_MAIN else _FFT_GRAD_S

    noise = 0.10
    fails = 0
    topk = 6

    while time.monotonic() < deadline:
        if time.monotonic() >= deadline:
            break

        c = _self_conv(p, fft_s)
        cmax = float(np.max(c))

        # Adaptive top-k
        topk = min(len(c), max(4, int(4 + (cmax - 1.0 / (2 * n - 1)) * 200)))
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros_like(c)
        wv = c[idx] + 1e-15
        # Use softmax weighting on the top peaks
        wu = 5000.0 * (wv - wv.max())
        we = np.exp(wu)
        w[idx] = we / we.sum()

        gp = _grad_p_from_weights(p, w, n, fft_g)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        accepted = False
        if gnorm > 1e-14:
            direction = gz / gnorm
            # Adaptive line search
            best_step_score = score
            best_step_z = z
            best_step_p = p
            for step in (0.5, 0.3, 0.2, 0.1, 0.05, 0.025, 0.0125):
                zt = z - step * direction
                zt = _sanitized_z(zt)
                pt = _probs_from_z(zt, n)
                st = _score_p(pt, n)
                if st < best_step_score - 1e-13:
                    best_step_score = st
                    best_step_z = zt
                    best_step_p = pt
            if best_step_score < score - 1e-13:
                z = best_step_z
                p = best_step_p
                score = best_step_score
                if n == N_MAIN:
                    _update_best(z, score)
                accepted = True

        if not accepted:
            # Try random perturbations
            best_local_score = score
            best_local_z = z
            best_local_p = p
            n_tries = 5
            for _ in range(n_tries):
                zt = z + rng.normal(0.0, noise, n - 1)
                zt = _sanitized_z(zt)
                pt = _probs_from_z(zt, n)
                st = _score_p(pt, n)
                if st < best_local_score - 1e-13:
                    best_local_score = st
                    best_local_z = zt
                    best_local_p = pt
            if best_local_score < score - 1e-13:
                z = best_local_z
                p = best_local_p
                score = best_local_score
                if n == N_MAIN:
                    _update_best(z, score)
                accepted = True

        if accepted:
            fails = 0
            noise = min(0.3, noise * 1.2)
            topk = min(topk + 1, 20)
        else:
            fails += 1
            noise = max(0.01, noise * 0.75)
            if fails > 80:
                break


# ---------------------------------------------------------------------------
# Alpha selection
# ---------------------------------------------------------------------------

def _choose_alpha(score):
    if not np.isfinite(score):
        return 5.0e4
    if score > 1.90:
        return 5.0e4
    if score > 1.75:
        return 1.5e5
    if score > 1.60:
        return 5.0e5
    if score > 1.52:
        return 1.5e6
    if score > 1.505:
        return 4.0e6
    return 1.0e7


def _adaptive_maxfun(rem, run_idx):
    if rem > 900.0:
        base = 700
    elif rem > 450.0:
        base = 500
    elif rem > 180.0:
        base = 350
    else:
        base = 180
    if run_idx < 4:
        base = int(base * 1.3)
    elif run_idx > 60:
        base = int(base * 0.7)
    return int(max(60, base))


# ---------------------------------------------------------------------------
# Small-n pre-optimization
# ---------------------------------------------------------------------------

def _small_n_preoptimize(rng, deadline_pre):
    """Optimize at small n to get a good initial shape."""
    n = _N_SMALL
    best_z_small = None
    best_score_small = float("inf")

    # Generate candidates at small n
    cands = _make_structured_candidates(n)
    cands.extend(_make_random_candidates(rng, 40, n))

    scores = np.empty(len(cands))
    for i, z in enumerate(cands):
        p = _probs_from_z(z, n)
        c = _self_conv(p, _FFT_SELF_S)
        scores[i] = 2.0 * n * float(np.max(c))
        if scores[i] < best_score_small:
            best_score_small = scores[i]
            best_z_small = z.copy()

    order = np.argsort(scores)
    top_cands = [cands[int(j)] for j in order[:15]]

    # L-BFGS-B on top candidates
    for z0 in top_cands:
        rem = deadline_pre - time.monotonic()
        if rem < 5.0:
            break
        alpha = _choose_alpha(best_score_small)
        max_fun = min(200, int(rem / 0.01))
        _optimize_z(z0, alpha, max_fun, deadline_pre, n=n)

    if best_z_small is None:
        best_z_small = np.zeros(n - 1, dtype=np.float64)

    # Re-evaluate
    p = _probs_from_z(best_z_small, n)
    c = _self_conv(p, _FFT_SELF_S)
    final_score = 2.0 * n * float(np.max(c))
    print(f"Small-n pre-opt: score={final_score:.6f}, time={time.monotonic()-start_ref:.1f}s")
    return best_z_small


# ---------------------------------------------------------------------------
# Main search
# ---------------------------------------------------------------------------

start_ref = 0.0


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_z, _search_deadline, start_ref

    start_time = time.monotonic()
    start_ref = start_time
    budget_s = float(budget_s)
    end_time = start_time + max(1.0, budget_s) - 2.0

    _best_score = float("inf")
    _best_z = None
    _search_deadline = start_time

    rng = np.random.default_rng(seed)

    print(f"Starting search with budget={budget_s:.0f}s, time={start_time:.1f}")

    # Phase 0: Small-n pre-optimization (use ~5% of budget)
    if budget_s > 300.0 and minimize is not None:
        pre_budget = min(60.0, 0.05 * budget_s)
        deadline_pre = start_time + pre_budget
        z_small = _small_n_preoptimize(rng, deadline_pre)
        z_upsampled = _upsample_z(z_small, _N_SMALL, N_MAIN)
        _score_z(z_upsampled)
        print(f"After upsample: best={_best_score:.6f}")
    else:
        z_upsampled = None

    # Phase 1: Generate and score diverse initializations
    struct_cands = _make_structured_candidates(N_MAIN)
    random_count = 120 if budget_s > 600 else 60
    random_cands = _make_random_candidates(rng, random_count, N_MAIN)
    struct_cands = _add_base_candidate(struct_cands)

    if z_upsampled is not None:
        struct_cands.insert(0, z_upsampled)

    struct_sorted = _sort_by_score(struct_cands, N_MAIN)
    random_sorted = _sort_by_score(random_cands, N_MAIN)

    print(f"Best initial score: {_best_score:.6f}")

    # Phase 2: Main optimization loop
    if budget_s > 400.0:
        main_deadline = start_time + budget_s - 5.0 - max(90.0, 0.10 * budget_s)
    else:
        main_deadline = start_time + max(10.0, budget_s - 15.0)

    max_struct_runs = min(len(struct_sorted), 30)
    max_rand_runs = min(40, len(random_sorted)) if budget_s > 300 else min(15, len(random_sorted))

    run_idx = 0
    last_best = _best_score
    stagnant = 0
    total_runs = 0

    while time.monotonic() < main_deadline:
        rem = main_deadline - time.monotonic()
        if rem < 10.0:
            break

        # Choose next starting point
        if run_idx < max_struct_runs:
            z0 = struct_sorted[run_idx]
        elif run_idx < max_struct_runs + max_rand_runs:
            z0 = random_sorted[run_idx - max_struct_runs]
        else:
            # Perturb the current best
            if _best_z is None:
                z0 = rng.normal(0.0, 1.0, N_MAIN - 1)
            else:
                frac = max(0.0, min(1.0, rem / max(1.0, main_deadline - start_time)))
                noise_scale = 0.05 + 0.40 * frac
                if stagnant > 20:
                    noise_scale *= 2.5
                z0 = _best_z + rng.normal(0.0, min(2.0, noise_scale), N_MAIN - 1)

        run_idx += 1
        total_runs += 1
        alpha = _choose_alpha(_best_score)
        max_fun = _adaptive_maxfun(rem, run_idx)
        _optimize_z(z0, alpha, max_fun, main_deadline)

        if _best_score < last_best - 1e-12:
            stale = last_best - _best_score
            last_best = _best_score
            stagnant = 0
            elapsed = time.monotonic() - start_time
            if total_runs % 5 == 0:
                print(f"run {total_runs}: best={_best_score:.6f} delta={stale:.6f} elapsed={elapsed:.1f}s")
        else:
            stagnant += 1

        # Periodic: try a fresh random restart if very stagnant
        if stagnant > 40 and rem > 60.0:
            z0_fresh = _random_z(rng, N_MAIN)
            _score_z(z0_fresh)
            if _best_score < last_best - 1e-6:
                stagnant = 0
                last_best = _best_score

    print(f"Main phase done: best={_best_score:.6f}, runs={total_runs}, elapsed={time.monotonic()-start_time:.1f}s")

    # Phase 3: Final polishing
    if _best_z is not None:
        polish_deadline = max(time.monotonic() + 10.0, end_time - 5.0)
        _polish_subgradient(_best_z, polish_deadline, rng, N_MAIN)
        print(f"After polishing: best={_best_score:.6f}")

        # One more L-BFGS-B pass with very high alpha
        if minimize is not None and _best_z is not None:
            rem = end_time - 3.0 - time.monotonic()
            if rem > 30.0:
                z0 = _best_z + rng.normal(0.0, 0.03, N_MAIN - 1)
                max_fun = max(100, int(min(500, rem / 0.008)))
                _optimize_z(z0, 1.0e7, max_fun, end_time - 3.0)
            # Another polish
            if time.monotonic() < end_time - 8.0 and _best_z is not None:
                _polish_subgradient(_best_z, end_time - 3.0, rng, N_MAIN)

    print(f"Final best score: {_best_score:.6f}")

    # Construct output sequence
    if _best_z is None:
        z = np.zeros(N_MAIN - 1, dtype=np.float64)
    else:
        z = _sanitized_z(_best_z)

    p = _probs_from_z(z, N_MAIN)
    p = np.clip(p, 0.0, 1000.0)
    seq = [float(v) for v in p]

    try:
        score = float(evaluate_sequence(seq))
    except Exception:
        score = float("inf")

    if not np.isfinite(score) or score > 2.000001:
        seq = [1.0 / float(N_MAIN)] * N_MAIN
        try:
            score = float(evaluate_sequence(seq))
        except Exception:
            score = 2.0

    # Safety check against inherited sequence
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            base_score = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(base_score) and base_score < score:
                seq = [float(v) for v in base]
                score = base_score
    except Exception:
        pass

    print(f"FINAL: score={score:.6f} length={len(seq)} elapsed={time.monotonic()-start_time:.1f}s")
    return seq
