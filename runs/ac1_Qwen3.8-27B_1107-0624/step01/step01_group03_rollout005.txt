def gradient_step(s, beta=5.0, eta=0.3):
    """One multiplicative gradient step. Returns new s."""
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = conv.max()
    if max_conv <= 0:
        return s.copy()
    # Soft weighting emphasizing peaks
    w = np.exp(beta * (conv - max_conv) / max_conv)
    grad = compute_gradient_fft(s, w)
    grad = np.maximum(grad, 0.0)
    grad_max = grad.max()
    if grad_max < 1e-15:
        return s.copy()
    s_new = s * np.exp(-eta * grad / grad_max)
    s_new = np.maximum(s_new, 1e-15)
    s_new = s_new / s_new.sum()
    return s_new


def lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M, b >= 0. Returns b or None."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def local_search(s, deadline, rng, use_lp=True):
    """Run local optimization from initial sequence s. Returns (best_s, best_score)."""
    n = len(s)
    s = s.copy() / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)

    eta = 0.3
    beta = 5.0
    no_improve = 0
    patience = 150

    for it in range(20000):
        if time.monotonic() > deadline:
            break

        cur_score = fast_score(s)

        # Main gradient step
        if it % 50 == 0 and use_lp and n <= 512:
            # Occasional LP-guided step
            conv = np.convolve(s, s)
            M = conv.max()
            b = lp_direction(s, M, time_limit=2.0)
            if b is not None and b.sum() > 1e-12:
                b = b / b.sum()
                # Line search over blend
                best_t = 0.0
                best_ts = cur_score
                for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    cand = ((1 - t) * s + t * b)
                    cand = np.maximum(cand, 0.0)
                    cand = cand / cand.sum()
                    ts = fast_score(cand)
                    if ts < best_ts:
                        best_ts = ts
                        best_t = t
                if best_t > 0:
                    s = ((1 - best_t) * s + best_t * b)
                    s = np.maximum(s, 0.0) / s.sum()
                    if best_ts < best_score - 1e-12:
                        best_score = best_ts
                        best_s = s.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    continue

        # Standard gradient step
        s_new = gradient_step(s, beta=beta, eta=eta)
        score_new = fast_score(s_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_s = s_new.copy()
            no_improve = 0
            s = s_new
        elif score_new < cur_score:
            s = s_new
            no_improve += 1
        else:
            no_improve += 1

        # Adaptive parameter adjustment
        if no_improve == patience:
            eta *= 0.75
            no_improve = 0
            if eta < 0.02:
                # Restart with perturbation
                s = s + 0.02 * rng.standard_normal(n)
                s = np.maximum(s, 0.01)
                s = s / s.sum()
                eta = 0.3
                beta = rng.choice([3.0, 5.0, 8.0, 12.0, 20.0])
                no_improve = 0
        elif no_improve == patience // 2:
            beta = min(beta * 1.3, 50.0)

        # Periodically increase beta (focus more on peaks)
        if it % 1000 == 999:
            beta = min(beta + 1.0, 30.0)

    return best_s, best_score


def try_parameterized(n, rng):
    """Generate diverse parameterized starting points."""
    results = []

    # Uniform
    s = np.ones(n) / n
    results.append((fast_score(s), s.copy()))

    # Linear ramps
    for beta in np.linspace(-0.8, 0.8, 17):
        x = np.arange(n) / max(n - 1, 1)
        s = 1.0 + beta * (2.0 * x - 1.0)
        s = np.maximum(s, 0.05)
        s = s / s.sum()
        results.append((fast_score(s), s.copy()))

    # Exponential
    for alpha in [-3.0, -2.0, -1.0, 1.0, 2.0, 3.0]:
        s = np.exp(alpha * np.arange(n) / max(n - 1, 1))
        s = s / s.sum()
        results.append((fast_score(s), s.copy()))

    # Two-level
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        k = max(1, int(frac * n))
        for ratio in [0.5, 1.5, 2.0, 3.0]:
            s = np.zeros(n)
            s[:k] = ratio
            s[k:] = 1.0
            s = s / s.sum()
            results.append((fast_score(s), s.copy()))

    # Quadratic / Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.15, 0.3, 0.5]:
            x = np.arange(n) / max(n - 1, 1)
            s = np.exp(-(x - center) ** 2 / (2 * width ** 2))
            s = np.maximum(s, 0.01)
            s = s / s.sum()
            results.append((fast_score(s), s.copy()))

    # Cosine window
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        x = np.arange(n) / max(n - 1, 1)
        s = np.cos(alpha * np.pi * x) ** 2
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append((fast_score(s), s.copy()))

    # Random
    for _ in range(6):
        s = rng.random(n) + 0.1
        s = s / s.sum()
        results.append((fast_score(s), s.copy()))

    results.sort(key=lambda x: x[0])
    return results


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(s,s))/(sum(s))^2.
    Uses FFT-based gradient descent with adaptive parameters, LP-guided corrections,
    diverse initialization, and multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0

    global_best_score = float('inf')
    global_best_seq = np.ones(256) / 256.0

    # Try to load height_sequence_1
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}")
    except Exception:
        pass

    # Sequence lengths to try (smaller n tends to give better scores due to the n factor)
    lengths = [128, 192, 256, 384, 512, 768]

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")

        # Get candidate starting points
        candidates = try_parameterized(n, rng)

        # Add resized height_sequence_1
        if hseq is not None:
            x_old = np.linspace(0.0, 1.0, hseq.size)
            x_new = np.linspace(0.0, 1.0, n)
            h_r = np.interp(x_new, x_old, hseq)
            h_r = np.maximum(h_r, 0.0)
            if h_r.sum() > 0.01:
                h_r = h_r / h_r.sum()
                candidates.insert(0, (fast_score(h_r), h_r))

        # Also try direct height_sequence_1 if length matches
        if hseq is not None and hseq.size == n:
            candidates.insert(0, (fast_score(hseq / hseq.sum()), hseq / hseq.sum()))

        # Sort and select top candidates
        candidates.sort(key=lambda x: x[0])

        # Run local search on top candidates
        num_cands = min(4, len(candidates))
        iters_budget = remaining / num_cands

        for i in range(num_cands):
            if time.monotonic() > deadline - 20:
                break
            sc0, s0 = candidates[i]
            print(f"  Candidate {i}: score={sc0:.6f}")

            # Allocate time proportional to quality
            time_per_cand = max(10.0, iters_budget * (1.0 - i * 0.15))
            cand_deadline = time.monotonic() + time_per_cand

            use_lp = (n <= 384)  # LP only for smaller n
            best_s, best_sc = local_search(s0, cand_deadline, rng, use_lp=use_lp)

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** New global best: {best_sc:.6f} (n={n})")

        # Random restarts if time remains
        restarts = 0
        while time.monotonic() < deadline - 30 and restarts < 3:
            s_r = rng.random(n) + 0.1
            s_r = s_r / s_r.sum()
            best_s, best_sc = local_search(s_r, time.monotonic() + 30, rng, use_lp=False)
            restarts += 1
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Restart {restarts}: new best {best_sc:.6f}")

    # Final refinement: run additional iterations on the best found
    if time.monotonic() < deadline - 10:
        print(f"\n=== Final refinement, score={global_best_score:.6f} ===")
        n = len(global_best_seq)
        best_s, best_sc = local_search(global_best_seq, deadline - 5, rng,
                                       use_lp=(n <= 384))
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()
            print(f"  ** Refined: {best_sc:.6f}")

    # Verify with actual evaluator
    final_score = float(evaluate_sequence(global_best_seq.tolist()))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(global_best_seq)}")

    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
