Hybrid multiplicative + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast convolution-based subgradient, LP direction finding, and random restarts.

import numpy as np
import time
from scipy.optimize import linprog

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = res.x
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None

    if ws < 1e-15:

        w[np.argmax(c)] = 1.0

    s = a_new.sum()

def make_window(n, kind='cosine'):
    """Generate various window functions."""
    if kind == 'cosine':
        return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))
    elif kind == 'gaussian':
        c = (n - 1) / 2.0
        sigma = 0.25 * n
        return np.exp(-0.5 * ((np.arange(n) - c) / sigma) ** 2)
    elif kind == 'parabolic':
        c = (n - 1) / 2.0
        return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 0.5))
    elif kind == 'uniform':
        return np.ones(n)
    elif kind == 'lorentzian':
        c = (n - 1) / 2.0
        return 1.0 / (1.0 + ((np.arange(n) - c) / (0.3 * n)) ** 2)
    elif kind == 'super_gaussian':
        c = (n - 1) / 2.0
        sigma = 0.3 * n
        return np.exp(-((np.arange(n) - c) / sigma) ** 4)
    elif kind == 'boundary_boost':
        # Boost boundaries to fill in the convolution ramp
        base = 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))
        boost = 0.3 * (1.0 - base)
        return base + boost
    else:
        return np.ones(n)
