import numpy as np

from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(n):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g) for given a.
    Actually, we need A such that for fixed a, A @ g gives conv(a, g).
    conv(a, g)[k] = sum_j a[k-j] * g[j] (where valid)
    So A[k, j] = a[k-j] if 0 <= k-j < n, else 0."""
    A = np.zeros((2 * n - 1, n))
    a = None  # We'll pass a separately
    return A


def build_conv_matrix_for_a(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve_multi_peak(a, M, top_k=1):
    """
    Solve: max sum(g) s.t. convolve(a,g) <= M for top_k peaks, g >= 0.
    Returns g or None.
    """
    n = len(a)
    c_conv = np.convolve(a, a, mode='full')
    
    # Get top_k peak positions
    sorted_idx = np.argsort(c_conv)[::-1][:top_k]
    
    # Build constraint matrix for only the top_k constraints
    A_rows = np.zeros((top_k, n))
    for ki, k in enumerate(sorted_idx):
        for j in range(n):
            if 0 <= k - j < n:
                A_rows[ki, j] = a[k - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_rows, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_solve_all(a, M):
    """Solve LP with all constraints."""
    n = len(a)
    A = build_conv_matrix_for_a(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)

def compute_objective(a):

    n = len(a)

def lp_min_peak_direction(a, peak_fraction=0.95, max_active=50):

def subgradient_optimize(a, deadline, rng, num_peaks=5):
    """Projective subgradient method targeting multiple peaks."""
    n = len(a)

    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)

    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(best_a, best_a, mode='full')
        
        # Find top peaks weighted by their excess
        M = c.max()
        # Weight peaks by how much they exceed the mean
        mean_c = c.mean()
        weights = np.maximum(c - mean_c, 0)
        
        if weights.sum() < 1e-15:
            break
        
        weights /= weights.sum()
        
        # Compute weighted gradient: grad[i] = sum_k w[k] * a[k-i]
        grad = np.zeros(n)
        for k in range(2*n-1):

                    if 0 <= j < n:
                        grad[i] += weights[k] * best_a[j]
        
        # Projected step
        a_new = best_a - lr * (grad - grad.mean())
        a_new = np.clip(a_new, 1e-10, None)
        a_new /= a_new.sum()
        
        sc = compute_objective(a_new)
        if sc < best_score - 1e-10:
            best_a = a_new
            best_score = sc
            stall = 0
            lr = min(0.3, lr * 1.01)
        else:
            stall += 1

                a_new /= a_new.sum()
                best_a = a_new

def coordinate_descent(a, deadline, rng, passes=5):
    """Coordinate descent with 1D optimization per coordinate."""
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_score = compute_objective(best_a)
    
    for p in range(passes):
        if time.monotonic() > deadline:

        order = rng.permutation(n)
        improved_any = False
        
        for i in order:
            if time.monotonic() > deadline:
                return best_a, best_score
            
            orig = best_a[i]
            # Try several values
            candidates = [orig * f for f in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7]]
            candidates.append(0.0)
            candidates.append(a.max() * 0.5)

                best_a[i] = cv
                sc = compute_objective(best_a)

                if best_sc < best_score:
                    best_score = best_sc
        
        # Golden section refinement on changed coordinates
        if not improved_any:
            break
    
    return best_a, best_score


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_flat_with_taper(n, taper=0.1):
    """Flat center with tapered edges."""
    a = np.ones(n)
    t = int(taper * n)
    for i in range(t):
        a[i] = 0.5 * (1 + np.cos(np.pi * i / t))
        a[n-1-i] = a[i]
    return a

"""

import numpy as np

from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def lp_min_peak_direction(a, peak_fraction=0.95, max_active=40):
    """
    Solve: min t s.t. conv(a,g)[k] <= t for k in active peaks, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, inf).
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    if M < 1e-15:
        return None, np.inf

    # Active set
    threshold = peak_fraction * M
    active = np.where(c >= threshold)[0]
    if len(active) > max_active:
        active = np.argsort(c)[::-1][:max_active]
    m = len(active)
    if m == 0:
        return None, np.inf

    # Variables: [g_0, ..., g_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Inequality constraints: conv(a,g)[k] - t <= 0 for each active k
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for idx, k in enumerate(active):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_ub[idx, j] = a[kj]
        A_ub[idx, n] = -1.0

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            if g.sum() < 1e-15:
                return None, np.inf
            return g, res.x[n]
    except Exception:
        pass
    return None, np.inf

def subgradient_multi_peak(a, deadline, rng, num_peaks=5):
    """Projective subgradient method targeting multiple peaks simultaneously."""
    n = len(a)
    a = np.clip(a, 1e-10, None).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    lr = 0.08
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(best_a, best_a, mode='full')
        M = c.max()

        # Weight peaks by excess above median
        med = np.median(c)
        weights = np.maximum(c - med, 0)
        if weights.sum() < 1e-15:
            break
        weights /= weights.sum()

        # Weighted gradient: grad[i] = sum_k w[k] * a[k-i]
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] > 1e-8:
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        grad[i] += weights[k] * best_a[j]

        # Projected step onto {a >= 0, sum(a) = 1}
        a_new = best_a - lr * (grad - grad.mean())
        a_new = np.clip(a_new, 1e-10, None)
        a_new /= a_new.sum()

        sc = compute_objective(a_new)
        if sc < best_score - 1e-10:
            best_a = a_new
            best_score = sc
            stall = 0
            lr = min(0.25, lr * 1.01)
        else:
            stall += 1
            lr = max(0.003, lr * 0.97)
            if stall > 40:
                mask = rng.random(n) < 0.1
                a_new = best_a.copy()
                a_new[mask] *= rng.uniform(0.8, 1.2, max(1, mask.sum()))
                a_new = np.clip(a_new, 1e-10, None)
                a_new /= a_new.sum()
                best_a = a_new
                stall = 0
            if stall > 400:
                break

    return best_a, best_score


def coordinate_descent(a, deadline, rng, max_passes=20):
    """Coordinate descent: optimize one element at a time via grid search."""
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_score = compute_objective(best_a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved_any = False

        for i in order:
            if time.monotonic() > deadline:
                return best_a, best_score

            orig = best_a[i]
            # Candidate values: scale factors + 0 + reference
            refs = [best_a.max() * 0.3, best_a.mean() * 1.5]
            candidates = [orig * f for f in [0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0]]
            candidates.append(0.0)
            candidates.extend(refs)

            best_val = orig
            best_sc = best_score

            for cv in candidates:
                best_a[i] = cv
                sc = compute_objective(best_a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_val = cv

            best_a[i] = best_val
            if best_val != orig:
                improved_any = True
                if best_sc < best_score:
                    best_score = best_sc

        if not improved_any:
            break

    return best_a, best_score


def make_cosine(n):
    """Raised cosine (Hann) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_flat_taper(n, taper_frac=0.15):
    """Flat center with cosine-tapered edges."""
    a = np.ones(n)
    t = max(1, int(taper_frac * n))
    for i in range(t):
        val = 0.5 * (1.0 + np.cos(np.pi * i / t))
        a[i] = val
        a[n - 1 - i] = val
    return a


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_spectral(n, rng):
    """Spectral initialization: inverse DFT of flat magnitude spectrum."""
    N = 4 * n
    phase = rng.random(N // 2 + 1) * 2 * np.pi
    phase[0] = 0.0
    spec = np.exp(1j * phase)
    a = np.fft.irfft(spec, n=N)[:n]
    a = np.abs(a)
    if a.sum() < 1e-10:
        a = make_cosine(n)
    return a + 1e-6

    for idx, (name, n, init_f, method, weight) in enumerate(configs):
        if time.monotonic() > deadline:
            break

        time_alloc = available_time * weight / total_weight
        config_deadline = min(start_time + (budget_s - 120.0 - available_time) + time_alloc * (idx + 1) * max(1, len(configs)) / len(configs), deadline)
        
        # Simpler: just allocate proportional time
        elapsed = time.monotonic() - start_time
        remaining_total = deadline - time.monotonic()
        configs_remaining = len(configs) - idx
        time_per_config = remaining_total / configs_remaining
        config_deadline = min(time.monotonic() + time_per_config * (weight / 1.5), deadline)

        if config_deadline - time.monotonic() < 15:
            continue

        try:
            a = init_f(n)
        except Exception:
            a = make_cosine(n)

        a = np.clip(a, 1e-10, None).astype(np.float64)
        if a.sum() < 1e-10:
            a = make_cosine(n)
        a = a / a.sum()

        remaining = deadline - time.monotonic()
        print(f"\n[{idx+1}/{len(configs)}] {name} (n={n}, {method}, ~{config_deadline - time.monotonic():.0f}s)")

        t0 = time.monotonic()
        if method == 'lp':
            a_opt, score = optimize_lp_direct(a, config_deadline, rng, verbose=True)
        elif method == 'subgrad':
            a_opt, score = subgradient_multi_peak(a, config_deadline, rng)
        else:
            a_opt, score = optimize_lp_direct(a, config_deadline, rng, verbose=False)

        dt = time.monotonic() - t0
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: {best_score:.6f} (n={len(a_opt)}, t={dt:.1f}s)")
        else:
            print(f"  => {score:.6f} (best: {best_score:.6f}, t={dt:.1f}s}")

    # ===== FINAL POLISHING =====
    if best_seq is not None:
        print(f"\n{'='*60}")
        print(f"FINAL POLISHING: score={best_score:.6f}, n={len(best_seq)}")
        print(f"{'='*60}")

        # Phase 1: LP with tight active set
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t1 = min(remaining * 0.35, deadline - 30)
            print(f"  Phase 1: LP polish ({t1 - time.monotonic():.0f}s)")
            best_seq, score = optimize_lp_direct(best_seq, t1, rng, initial_step=0.1, verbose=False)
            if score < best_score:
                best_score = score
            print(f"  After Phase 1: {best_score:.6f}")

        # Phase 2: Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 30 and len(best_seq) <= 1024:
            t2 = min(time.monotonic() + remaining * 0.3, deadline - 20)
            print(f"  Phase 2: Coord descent ({t2 - time.monotonic():.0f}s)")
            best_seq, score = coordinate_descent(best_seq, t2, rng, max_passes=5)
            if score < best_score:
                best_score = score
            print(f"  After Phase 2: {best_score:.6f}")

        # Phase 3: Fine LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t3 = min(time.monotonic() + remaining * 0.4, deadline - 10)
            print(f"  Phase 3: Fine LP ({t3 - time.monotonic():.0f}s)")
            best_seq, score = optimize_lp_direct(best_seq, t3, rng, initial_step=0.05, verbose=False)
            if score < best_score:
                best_score = score
            print(f"  After Phase 3: {best_score:.6f}")

        # Phase 4: Final coord
        remaining = deadline - time.monotonic()
        if remaining > 10 and len(best_seq) <= 768:
            t4 = min(time.monotonic() + remaining * 0.5, deadline - 2)
            print(f"  Phase 4: Final coord ({t4 - time.monotonic():.0f}s)")
            best_seq, score = coordinate_descent(best_seq, t4, rng, max_passes=15)
            if score < best_score:
                best_score = score
            print(f"  After Phase 4: {best_score:.6f}")

    if best_seq is None:
        print("WARNING: No valid sequence found, using default.")
        best_seq = make_cosine(512)
        best_score = compute_objective(best_seq)

    # Ensure evaluator compatibility
    best_seq = np.clip(best_seq, 0.0, None)
    if best_seq.sum() < 0.01:
        best_seq = best_seq / best_seq.sum() * 1.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"DONE in {elapsed:.1f}s | Final score={best_score:.6f} | n={len(best_seq)}")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
