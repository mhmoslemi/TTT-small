"""
Search for a non-negative step-function sequence minimizing
    f(seq) = 2*n*max(convolve(seq, seq)) / sum(seq)^2
using adaptive-temperature gradient descent and LP cutting-plane refinement.
Focuses on lengths near 384 where the known best was found.

import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def mixed_normalized(n, lam):
    p = lam * greedy_normalized(n) + (1.0 - lam) * np.ones(int(n), dtype=np.float64)
    return normalize_nonneg(p)


def perturb_greedy(n, rng, sigma):
    p = greedy_normalized(n)
    noise = np.exp(rng.normal(0.0, float(sigma), int(n)))
    return normalize_nonneg(p * noise)


def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    if n <= 768:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(a.size)
        s = float(a.sum())
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]


def adam_optimize(p, budget_s, seed=0, alpha_start=10.0, alpha_end=80.0):
    """
    Adam optimizer for min max_k((p*p)_k) subject to sum(p)=1, p>=0.
    Uses log-softmax parameterization and adaptive temperature.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 0.5:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = 0.04
    elif n < 256:
        lr = 0.025
    elif n < 1024:
        lr = 0.015
    else:
        lr = 0.01

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + budget_s

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        # Adaptive temperature: start high, increase further
        alpha = alpha_start + (alpha_end - alpha_start) * frac * frac

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Smoothed max gradient
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0.0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient of smoothed max w.r.t. p
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (2.0 * n) * C  # gradient of 2n * smoothed_max
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def lp_improve(p, budget_s, max_rows=None):
    """
    Cutting-plane LP method:
    Given f (normalized to sum=1), minimize max(f*f) by:
    1. Set M = max(f*f)
    2. Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0
    3. Blend f and g, renormalize
    4. Repeat
    """
    if _scipy_opt is None:
        return p, actual_score_fft(p)

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s < 20.0:
        return p, actual_score_fft(p)

    if max_rows is None:
        if n > 2000:
            max_rows = 800
        elif n > 500:
            max_rows = 1000
        else:
            max_rows = 1200

    max_elements = 5000000
    while max_rows * n > max_elements:
        max_rows = max(100, int(max_rows * 0.8))
    max_rows = max(100, min(int(max_rows), max_elements // max(1, n)))

    f = p.copy()  # sum = 1
    cur_M = float(np.max(np.convolve(f, f))) if n <= 768 else float(
        np.max(np.fft.irfft(np.fft.rfft(f, next_power_of_two(2*n-1))**2, next_power_of_two(2*n-1))[:2*n-1].real))

    best_f = f.copy()
    best_score = float(2.0 * n * cur_M)

    end = time.monotonic() + budget_s

    # Initial rows: uniform sampling
    r0 = min(max_rows // 2, 200)
    rows = list(np.unique(np.linspace(0, 2 * n - 2, r0, dtype=int)))
    row_set = set(rows)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.array(rows, dtype=int)
        
        # Build constraint matrix: (f*g)_k <= cur_M
        A = np.zeros((row_arr.size, n), dtype=np.float64)
        for ii in range(row_arr.size):
            k = row_arr[ii]
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
        
        b = np.full(row_arr.size, cur_M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(3.0, min(60.0, rem * 0.2))
        res = None
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": tl, "dual_feasibility_tolerance": 1e-8, "primal_feasibility_tolerance": 1e-8}
            )
        except Exception:
            break

        if res is None or not res.success or res.x is None:
            break

        g = np.clip(res.x, 0.0, None)
        sg = float(g.sum())
        if sg <= 1e-12:
            break

        # Check violations on all constraints
        if n <= 768:
            conv = np.convolve(f, g)
        else:
            L2 = next_power_of_two(2 * n - 1)
            conv = np.fft.irfft(np.fft.rfft(f, L2) * np.fft.rfft(g, L2), L2)[:2*n-1].real
        conv = np.maximum(conv, 0.0)

        tol = 1e-10
        violations = conv - cur_M
        add_needed = np.any(violations > tol)

        if add_needed:
            added = 0
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                for ix in order:
                    if added >= min(256, max_rows - len(rows)):
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            if added == 0:
                if float(np.max(violations)) > 10.0 * tol + 1e-12:
                    break
            else:
                continue

        # Rescale g to sum=1
        g = g / sg

        # Try blending
        accepted = None
        accepted_M = np.inf
        for step in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn / sn
            if n <= 768:
                M_new = float(np.max(np.convolve(fn, fn)))
            else:
                L3 = next_power_of_two(2 * n - 1)
                M_new = float(np.max(np.fft.irfft(np.fft.rfft(fn, L3)**2, L3)[:2*n-1].real))
            if M_new < accepted_M:
                accepted_M = M_new
                accepted = fn.copy()

        if accepted is None:
            break
        if accepted_M >= cur_M * (1 - 1e-10):
            break

        f = accepted
        cur_M = accepted_M
        score = 2.0 * n * cur_M
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if it % 3 == 0:
            print("  LP iter %d: M=%.8f score=%.6f rows=%d" % (it, cur_M, best_score, len(rows)))

    return normalize_nonneg(best_f), best_score


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = actual_score_fft(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
        best_by_n[int(a.size)] = (float(score), a.copy())

    # Load known best
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                record(arr)
                print("Loaded height_sequence_1: n=%d score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    # Quick baselines at various lengths
    for n in [256, 320, 384, 448, 512, 640, 768]:
        if time.monotonic() >= deadline:
            break
        record(greedy_normalized(n))
        record(power_normalized(n, 0.45))
        record(power_normalized(n, 0.55))
        record(mixed_normalized(n, 0.5))

    print("After baselines: best_score=%.6f (n=%d)" % (best_score, best_p.size if best_p is not None else 0))

    # Phase 1: Adam optimization at the best length and nearby
    # Focus on the length that gave the best result
    if best_p is not None:
        best_n = int(best_p.size)
    else:
        best_n = 384

    target_lengths = sorted(set([best_n, best_n - 32, best_n + 32, 384, 512]))
    target_lengths = [n for n in target_lengths if n >= 64]

    for n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break

        # Use best known at this length or greedy
        rec = best_by_n.get(n)
        base = rec[1] if rec is not None else greedy_normalized(n)

        alloc = min(rem * 0.3, 120.0)
        p_out, f_out = adam_optimize(base, alloc, seed=seed + n, alpha_start=10.0, alpha_end=80.0)
        record(p_out, f_out)
        print("Adam n=%d: score=%.6f" % (n, f_out))

        # Second pass with different seed
        rem = deadline - time.monotonic()
        if rem > 30.0:
            base2 = perturb_greedy(n, rng, 0.1)
            rec2 = best_by_n.get(n)
            if rec2 is not None:
                base2 = rec2[1]
            alloc2 = min(rem * 0.25, 90.0)
            p_out2, f_out2 = adam_optimize(base2, alloc2, seed=seed + n + 1000, alpha_start=20.0, alpha_end=100.0)
            record(p_out2, f_out2)

    # Phase 2: LP/cutting-plane refinement
    if _scipy_opt is not None and best_p is not None:
        lp_lengths = sorted(set([int(best_p.size), 384, 512]))
        lp_lengths = [n for n in lp_lengths if n <= 2000]
        
        for n in lp_lengths:
            rem = deadline - time.monotonic()
            if rem < 40.0:
                break
            rec = best_by_n.get(n)
            base = rec[1] if rec is not None else greedy_normalized(n)
            alloc = min(rem * 0.4, 180.0)
            print("LP n=%d, alloc=%.0fs" % (n, alloc))
            p_out, f_out = lp_improve(base, alloc)
            record(p_out, f_out)
            print("LP n=%d: score=%.6f" % (n, f_out))

            # Follow up with Adam on the LP result

"""
Search for a non-negative step-function sequence minimizing
    f(seq) = 2*n*max(convolve(seq, seq)) / sum(seq)^2
using adaptive-temperature gradient descent, L-BFGS refinement,
and LP cutting-plane block coordinate descent. Focuses on lengths
near 384 where the known best was found, with aggressive refinement.
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def mixed_normalized(n, lam):
    p = lam * greedy_normalized(n) + (1.0 - lam) * np.ones(int(n), dtype=np.float64)
    return normalize_nonneg(p)


def perturb_greedy(n, rng, sigma):
    p = greedy_normalized(n)
    noise = np.exp(rng.normal(0.0, float(sigma), int(n)))
    return normalize_nonneg(p * noise)


def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    if n <= 768:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(a.size)
        s = float(a.sum())
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]


def compute_conv(p, L):
    Pg = np.fft.rfft(p, L)
    c = np.fft.irfft(Pg * Pg, L)[:2 * len(p) - 1].real
    return np.maximum(c, 0.0)


def smoothed_max_and_grad(p, alpha, L):
    c = compute_conv(p, L)
    n = p.size
    maxc = float(c.max())
    if maxc <= 0.0:
        return 0.0, np.zeros(n)
    z = alpha * (c / maxc - 1.0)
    expz = np.exp(z)
    sz = float(expz.sum())
    if not np.isfinite(sz) or sz <= 0.0:
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = expz / sz
    W = np.fft.rfft(w[::-1], L)
    Cfull = np.fft.irfft(np.fft.rfft(p, L) * W, L)
    C = Cfull[n - 1:2 * n - 1][::-1].real
    C = np.maximum(C, 0.0)
    obj = 2.0 * n * maxc
    grad_p = (2.0 * n) * C
    return obj, grad_p


def adam_optimize(p, budget_s, seed=0, alpha_start=10.0, alpha_end=100.0, lr=None):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 0.5:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if lr is None:
        if n < 64:
            lr = 0.04
        elif n < 256:
            lr = 0.025
        elif n < 1024:
            lr = 0.015
        else:
            lr = 0.01

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + budget_s
    last_improve = start
    no_improve_patience = max(10.0, budget_s * 0.15)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = alpha_start + (alpha_end - alpha_start) * frac * frac

        obj, grad_p = smoothed_max_and_grad(p, alpha, L)
        maxc = float(compute_conv(p, L).max())
        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > no_improve_patience:
            break

        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def lbfgs_smoothed_optimize(p, budget_s, alpha=30.0, max_iter=500):
    """Use L-BFGS-B on the log-sum-exp smoothed objective with log parameterization."""
    if _scipy_opt is None:
        return p, actual_score_fft(p)

    p0 = np.asarray(p, dtype=np.float64).ravel()
    n = p0.size
    p0 = normalize_nonneg(p0)
    budget_s = float(budget_s)
    if budget_s < 2.0:
        return p0, actual_score_fft(p0)

    x0 = np.log(p0 + 1e-300)
    x0 -= x0.mean()
    L = next_power_of_two(3 * n - 2)

    def objective_and_grad(x):
        x = np.clip(x, -50.0, 50.0)
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            return 1e10, np.zeros(n)
        p = e / se
        obj, grad_p = smoothed_max_and_grad(p, alpha, L)
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)
        return float(obj), grad_x

    start = time.monotonic()
    best_x = x0.copy()
    best_f = actual_score_fft(p0)

    try:
        res = _scipy_opt.minimize(
            lambda x: objective_and_grad(x)[0],
            x0,
            jac=lambda x: objective_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                     'iprint': -1, 'gtol': 1e-8}
        )
        if res.x is not None:
            x_new = res.x
            x_new = np.clip(x_new, -50.0, 50.0)
            x_new = x_new - x_new.max()
            e_new = np.exp(x_new)
            se_new = float(e_new.sum())
            if np.isfinite(se_new) and se_new > 0:
                p_new = e_new / se_new
                f_new = actual_score_fft(p_new)
                if f_new < best_f:
                    best_f = f_new
                    best_x = x_new.copy()
    except Exception:
        pass

    # Convert back
    x_final = best_x
    x_final = np.clip(x_final, -50.0, 50.0)
    x_final = x_final - x_final.max()
    e_f = np.exp(x_final)
    se_f = float(e_f.sum())
    if not np.isfinite(se_f) or se_f <= 0:
        p_final = p0
    else:
        p_final = e_f / se_f

    f_final = actual_score_fft(p_final)
    if f_final < best_f:
        best_f = f_final
    return p_final, best_f


def lp_improve(p, budget_s, max_rows=None):
    """
    Cutting-plane LP block coordinate descent:
    Fix f, solve max sum(g) s.t. (f*g)_k <= M, g>=0, then blend.
    """
    if _scipy_opt is None:
        return p, actual_score_fft(p)

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s < 15.0:
        return p, actual_score_fft(p)

        if n > 1500:
            max_rows = 800
        else:
            max_rows = 1100

    max_elements = 5000000
    while max_rows * n > max_elements:
        max_rows = max(100, int(max_rows * 0.8))
    max_rows = max(100, min(int(max_rows), max_elements // max(1, n)))

    f = p.copy()
    if n <= 768:
        cur_M = float(np.max(np.convolve(f, f)))
    else:
        L2 = next_power_of_two(2 * n - 1)
        cur_M = float(np.max(np.fft.irfft(np.fft.rfft(f, L2)**2, L2)[:2*n-1].real))

    best_f_arr = f.copy()
    best_score = float(2.0 * n * cur_M)

    end = time.monotonic() + budget_s
    r0 = min(max_rows // 3, 200)
    rows = list(np.unique(np.linspace(0, 2 * n - 2, r0, dtype=int)))
    row_set = set(rows)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.array(rows, dtype=int)
        ok = (rows_arr >= 0) & (rows_arr <= 2 * n - 2)
        rows_arr = rows_arr[ok]
        if rows_arr.size == 0:
            break

        A = np.zeros((rows_arr.size, n), dtype=np.float64)
        for ii in range(rows_arr.size):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(rows_arr.size, cur_M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 2.0)
        tl = max(3.0, min(50.0, rem * 0.25))
        res = None
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": tl}
            )
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.clip(res.x, 0.0, None)
        sg = float(g.sum())
        if sg <= 1e-12:
            break

        # Check full convolution violation
        if n <= 768:
            conv = np.convolve(f, g)
        else:
            L3 = next_power_of_two(2 * n - 1)
            conv = np.fft.irfft(np.fft.rfft(f, L3) * np.fft.rfft(g, L3), L3)[:2*n-1].real
        conv = np.maximum(conv, 0.0)

        tol = 1e-10
        violations = conv - cur_M
        add_needed = np.any(violations > tol)

        if add_needed:
            added = 0
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                for ix in order:
                    if added >= min(200, max_rows - len(rows)):
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            if added == 0:
                if float(np.max(violations)) > 10.0 * tol + 1e-12:
                    break
            else:
                continue

        g = g / sg

        accepted = None
        accepted_M = np.inf
        for step in (1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())

            fn = fn / sn
            if n <= 768:

        if it % 5 == 0:

            best_state = record_candidate(p_out, f_out, best_state)
            print("Final Adam: %.6f" % f_out)

        rem = deadline - time.monotonic()
        if rem > 10.0 and _scipy_opt is not None:
            p_out, f_out = lbfgs_smoothed_optimize(best_state['p'], min(rem - 3.0, 50.0),
                                                   alpha=100.0, max_iter=200)
            best_state = record_candidate(p_out, f_out, best_state)
            print("Final L-BFGS: %.6f" % f_out)

    # Validate
    bp = best_state['p']
    if bp is not None:
        seq = scale_output(bp, 10.0)
        try:
            gscore = evaluate_sequence(seq)
            print("Global eval: %.6f" % gscore)
            if np.isfinite(gscore) and gscore < best_state['score']:
                best_state['score'] = gscore
                best_state['p'] = bp.copy()
        except Exception as e:
            print("Eval error: %s" % str(e))

    if best_state['p'] is None:
        best_state['p'] = greedy_normalized(384)

    print("FINAL: score=%.6f n=%d" % (best_state['score'], best_state['p'].size))
    return scale_output(best_state['p'], 10.0)
