"""
Minimize J = 2n * max(convolve(a,a)) / (sum(a))^2 over non-negative sequences a.
Uses FFT-accelerated gradient descent on the simplex with adaptive softmax
smoothing of the max function. Tries multiple sequence lengths and seeds.
"""

import time
import math
import numpy as np


def _next_pow2(m):
    """Next power of 2 >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find largest rho such that u[rho] * (rho+1) > cssv[rho] - target
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fft_conv_square(a, M):
    """Compute linear convolution a*a via FFT, returning length 2n-1."""
    fa = np.fft.rfft(a, M)
    c = np.fft.irfft(fa * fa, M)
    return c[:2 * a.size - 1]


def _softmax_weights(b, tau):
    """Compute softmax weights from b with temperature tau."""
    b_max = b.max()
    if b_max <= 0:
        return np.ones_like(b) / b.size
    z = (b - b_max) / max(tau, 1e-12)
    w = np.exp(z)
    s = w.sum()
    if s <= 0 or not np.isfinite(s):
        # Fallback: uniform over top values
        top_val = b_max
        mask = b >= top_val * 0.99
        w = mask.astype(np.float64)
        s = w.sum()
        return w / s
    return w / s


def _compute_gradient(a, w, M):
    """Compute gradient of sum_k w_k * b_k w.r.t. a, where b = conv(a,a).
    
    Returns F where F_j = 2 * sum_m w_{j+m} * a_m (for valid indices).
    Equivalent to 2 * cross_corr(w, a)[j].
    """
    # Zero-pad a and w to length M
    a_pad = np.zeros(M)
    a_pad[:a.size] = a
    w_pad = np.zeros(M)
    w_pad[:w.size] = w
    
    # F_j = 2 * sum_m a_m * w_{j+m}
    # = 2 * ifft(conj(FFT(a_pad)) * FFT(w_pad))[:n]
    fa = np.fft.rfft(a_pad)
    fw = np.fft.rfft(w_pad)
    F = np.fft.irfft(np.conj(fa) * fw, M)[:a.size]
    return 2.0 * F


def _run_descent(a_init, M, t_end, rng, verbose=False):
    """Run FFT-accelerated gradient descent on the simplex.
    
    Args:
        a_init: initial point (will be projected to simplex)
        M: FFT length (power of 2, >= 2n-1)
        t_end: absolute time to stop
        rng: numpy random generator
        verbose: print progress
    
    Returns:
        (best_a, best_obj) where best_obj = max(b) for the best a found
    """
    n = a_init.size
    a = _proj_simplex(np.abs(a_init.astype(np.float64)), 1.0)
    
    # State
    tau = 0.0  # Will be initialized below
    lr = 0.01
    momentum_buf = np.zeros(n)
    momentum = 0.85
    
    best_a = a.copy()
    best_mc = np.inf
    best_obj = np.inf
    
    step = 0
    last_print = 0.0
    last_eval = 0.0
    
    # Initial b and tau
    b = _fft_conv_square(a, M)
    mc = b.max()
    best_mc = mc
    tau = mc / 10.0  # effective support ~= 10 lags initially
    
    while time.monotonic() < t_end - 0.5:
        step += 1
        
        # Compute gradient
        w = _softmax_weights(b, tau)
        F = _compute_gradient(a, w, M)
        
        # Normalized gradient direction
        gnorm = np.linalg.norm(F)
        if gnorm > 1e-15:
            g = F / gnorm
        else:
            g = F
        
        # Momentum update
        momentum_buf = momentum * momentum_buf - lr * g
        a_new = a + lr * momentum_buf
        
        # Project onto simplex
        a_new = _proj_simplex(a_new, 1.0)
        
        # Evaluate new point
        b = _fft_conv_square(a_new, M)
        mc_new = b.max()
        
        if mc_new < best_mc:
            best_mc = mc_new
            best_a = a_new.copy()
        
        # Accept or reject (simple: always accept, but reduce lr if stuck)
        a = a_new
        
        # Adaptive parameters
        if step % 100 == 0:
            # Adapt tau: maintain effective support ~ K active lags
            # Sort b descending, find tau such that softmax has ~K "active"
            # Simple heuristic: tau = mc / K where K decreases over time
            K = max(3, int(20 * (0.99 ** (step // 100))))
            tau = mc / K
            
            # Adapt momentum
            momentum = min(0.95, 0.85 + 0.01 * (step // 1000))
            
            # Adapt lr
            if step % 1000 == 0:
                lr = max(0.001, lr * 0.9)
        
        # Periodic restart with noise

def _make_seeds(n, rng):
    """Generate diverse initial seeds of length n."""
    seeds = []
    # Uniform
    seeds.append(np.ones(n) / n)
    # Raised cosine
    x = np.arange(n) / (n - 1)
    seeds.append(np.clip(np.cos(np.pi * (x - 0.5)) ** 2, 1e-6, None))
    # Gaussian centered
    sigma = 0.25
    x = (np.arange(n) - n / 2) / (sigma * n)
    seeds.append(np.exp(-0.5 * x ** 2))
    # Gaussian offset
    x = (np.arange(n) - 0.3 * n) / (0.15 * n)
    seeds.append(np.exp(-0.5 * x ** 2))
    # Sine (zero at edges)
    seeds.append(np.clip(np.sin(np.pi * (np.arange(n) + 0.5) / n), 1e-6, None))
    # Linear ramp
    seeds.append(np.linspace(0.01, 1.0, n))
    # Quadratic bump
    x = 2.0 * np.arange(n) / n - 1.0
    seeds.append(np.clip(1.0 - x ** 2, 1e-6, None))
    # Two-bump
    x1 = (np.arange(n) - 0.3 * n) / (0.1 * n)
    x2 = (np.arange(n) - 0.7 * n) / (0.1 * n)
    seeds.append(np.exp(-0.5 * x1 ** 2) + np.exp(-0.5 * x2 ** 2))
    # Random
    seeds.append(rng.dirichlet(np.ones(n)))
    # Fejer-like (triangular squared)
    mid = n / 2
    tri = np.clip(1.0 - np.abs(np.arange(n) - mid) / mid, 0, None)
    seeds.append(np.clip(tri ** 2, 1e-6, None))
    
    # Normalize all
    seeds = [_proj_simplex(s, 1.0) for s in seeds]
    return seeds

import time

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fft_conv_sq(a, M):
    fa = np.fft.rfft(a, M)
    c = np.fft.irfft(fa * fa, M)
    return c[:2 * a.size - 1]


def _softmax_weights(b, tau):
    b_max = b.max()
    if b_max <= 0:
        return np.ones_like(b) / b.size
    z = (b - b_max) / max(tau, 1e-15)
    # Clip z to avoid overflow
    z = np.clip(z, -50.0, 50.0)
    w = np.exp(z)
    s = w.sum()
    if s <= 0 or not np.isfinite(s):
        mask = b >= b_max * 0.999
        w = mask.astype(np.float64)
        s = w.sum()
        if s == 0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
            return w
        return w / s
    return w / s


def _compute_grad(a, w, M):
    """Gradient of sum_k w_k * b_k w.r.t. a, using FFT cross-correlation."""
    n = a.size
    a_pad = np.zeros(M)
    a_pad[:n] = a
    w_pad = np.zeros(M)
    w_pad[:w.size] = w
    fa = np.fft.rfft(a_pad)
    fw = np.fft.rfft(w_pad)
    F = np.fft.irfft(np.conj(fa) * fw, M)[:n]
    return 2.0 * F

            print(f"    step={step} mc={mc:.6e} best={best_mc:.6e} "
                  f"obj={2*n*best_mc:.6f} t_left={t_end-elapsed:.1f}s")

    return best_a, best_mc


def _make_seeds(n, rng):
    seeds = []
    # Uniform
    seeds.append(np.ones(n))
    # Raised cosine
    x = np.arange(n) / max(n - 1, 1)
    seeds.append(np.clip(np.cos(np.pi * (x - 0.5)) ** 2, 1e-8, None))
    # Gaussian centered
    xc = (np.arange(n) - n * 0.5) / (n * 0.2)
    seeds.append(np.exp(-0.5 * xc ** 2))
    # Gaussian left-shifted
    xl = (np.arange(n) - n * 0.35) / (n * 0.15)
    seeds.append(np.exp(-0.5 * xl ** 2))
    # Sine window
    seeds.append(np.clip(np.sin(np.pi * (np.arange(n) + 0.5) / n), 1e-8, None))
    # Two bumps
    x1 = (np.arange(n) - 0.3 * n) / (0.1 * n)
    x2 = (np.arange(n) - 0.7 * n) / (0.1 * n)
    seeds.append(np.exp(-0.5 * x1 ** 2) + np.exp(-0.5 * x2 ** 2))
    # Quadratic bump
    xq = 2.0 * np.arange(n) / n - 1.0
    seeds.append(np.clip(1.0 - xq ** 2, 1e-8, None))
    # Fejer kernel shape
    mid = n / 2.0
    tri = np.clip(1.0 - np.abs(np.arange(n) - mid) / mid, 0, None)
    seeds.append(np.clip(tri ** 2, 1e-8, None))
    # Random Dirichlet
    seeds.append(rng.dirichlet(np.ones(n)))
    # Power law (concentrated middle)
    seeds.append(np.clip(1.0 - np.abs(np.arange(n) - mid) / mid, 0, None) ** 4)

    seeds = [_proj_simplex(s, 1.0) for s in seeds]
    return seeds


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    t_end = t_start + max(0.0, float(budget_s)) - 2.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size >= 4 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p
    except Exception:
        pass

    # Determine lengths to try
    lengths = [512, 1024, 2048, 4096, 8192, 16384]
    if parent is not None:
        pn = parent.size
        if pn not in lengths:
            lengths.append(pn)
            lengths.sort()

    best_score = float('inf')
    best_seq = None

    # Phase 1: Quick scan (first ~15% of time)
    phase1_end = t_start + 0.15 * (budget_s - 2.0)
    per_length_quick = max(5.0, (phase1_end - t_start) / len(lengths))

    quick_best = {}  # n -> (best_a, best_mc)

    for n in lengths:
        if time.monotonic() > phase1_end - 2:
            break
        M = _next_pow2(2 * n - 1)
        seeds = _make_seeds(n, rng)

        # Add parent (resampled) if available
        if parent is not None:
            if parent.size == n:
                seeds.insert(0, _proj_simplex(parent, 1.0))
            else:
                x_old = np.linspace(0, 1, parent.size)
                x_new = np.linspace(0, 1, n)
                resampled = np.interp(x_new, x_old, parent)
                seeds.insert(0, _proj_simplex(resampled, 1.0))

        # Run on best 2-3 seeds
        n_seeds = min(3, len(seeds))
        per_seed_t = per_length_quick / n_seeds

        local_best_a = None
        local_best_mc = float('inf')

        for si in range(n_seeds):
            t_seed_end = time.monotonic() + max(2.0, per_seed_t)
            a_out, mc_out = _run_descent(seeds[si], M, t_seed_end, rng)
            if mc_out < local_best_mc:
                local_best_mc = mc_out
                local_best_a = a_out

        if local_best_a is not None:
            quick_best[n] = (local_best_a, local_best_mc)
            obj = 2.0 * n * local_best_mc
            print(f"  Quick: n={n:5d} obj={obj:.6f}")

            # Evaluate with official
            try:
                sc = float(evaluate_sequence([float(x) for x in local_best_a]))
                if sc < best_score:
                    best_score = sc
                    best_seq = local_best_a.copy()
            except Exception:
                pass

    print(f"  After quick phase: best={best_score:.6f}, t_left={t_end-time.monotonic():.1f}s")

    # Phase 2: Focused optimization on best 1-2 lengths (most of remaining time)
    if quick_best:
        sorted_lengths = sorted(quick_best.keys(), key=lambda nn: 2.0 * nn * quick_best[nn][1])
        top_n = sorted_lengths[0]
        M = _next_pow2(2 * top_n - 1)

        # Build seed list for focused phase
        seeds_focused = []
        best_quick_a, _ = quick_best[top_n]
        seeds_focused.append(best_quick_a.copy())

        # Perturbed versions
        for _ in range(3):
            pert = best_quick_a * np.exp(rng.normal(0.0, 0.06, top_n))
            pert = np.clip(pert, 0, None)
            if pert.sum() > 0:
                seeds_focused.append(_proj_simplex(pert, 1.0))

        # Fresh structured seeds
        fresh = _make_seeds(top_n, rng)
        seeds_focused.extend(fresh[:3])

        # Remove duplicates (approximately)
        unique_seeds = [seeds_focused[0]]
        for s in seeds_focused[1:]:
            if not any(np.allclose(s, u, atol=1e-6) for u in unique_seeds):
                unique_seeds.append(s)
        seeds_focused = unique_seeds

        # Time for focused phase
        t_focus_end = t_end - 30.0
        n_focused = len(seeds_focused)
        per_focus = max(5.0, (t_focus_end - time.monotonic()) / n_focused)

        for si, s in enumerate(seeds_focused):
            if time.monotonic() > t_focus_end - 5:
                break
            t_seed_end = time.monotonic() + min(per_focus, t_focus_end - time.monotonic() - 3)
            a_out, mc_out = _run_descent(s, M, t_seed_end, rng, verbose=(si == 0))
            obj = 2.0 * top_n * mc_out

            try:
                sc = float(evaluate_sequence([float(x) for x in a_out]))
            except Exception:
                sc = obj

            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Focused: new best={sc:.6f} n={top_n} seed={si} "
                      f"t_left={t_end-time.monotonic():.1f}s")
            else:
                print(f"  Focused: n={top_n} seed={si} obj={obj:.6f} (not better)")

        # Also try second-best length if time remains
        if len(sorted_lengths) > 1 and time.monotonic() < t_end - 60:
            second_n = sorted_lengths[1]
            M2 = _next_pow2(2 * second_n - 1)
            sa, smc = quick_best.get(second_n, (None, float('inf')))
            if sa is not None:
                t2_end = t_end - 30.0
                a_out, mc_out = _run_descent(sa, M2, t2_end, rng)
                obj = 2.0 * second_n * mc_out
                try:
                    sc = float(evaluate_sequence([float(x) for x in a_out]))
                except Exception:
                    sc = obj
                if sc < best_score:
                    best_score = sc
                    best_seq = a_out.copy()
                    print(f"  Second length: new best={sc:.6f} n={second_n}")

    # Phase 3: Final refinement on the best candidate
    if best_seq is not None and time.monotonic() < t_end - 20:
        rn = best_seq.size
        RM = _next_pow2(2 * rn - 1)
        a_ref = _proj_simplex(best_seq.copy(), 1.0)

        b_ref = _fft_conv_sq(a_ref, RM)
        mc_ref = b_ref.max()
        tau_ref = mc_ref / 5.0
        lr_ref = 0.15
        mom_ref = np.zeros(rn)
        mom_val = 0.9
        t_ref_end = t_end - 3.0
        ref_step = 0
        ref_best_mc = mc_ref
        ref_best_a = a_ref.copy()

        while time.monotonic() < t_ref_end:
            ref_step += 1
            w_ref = _softmax_weights(b_ref, tau_ref)
            F_ref = _compute_grad(a_ref, w_ref, RM)

            fnorm = np.abs(F_ref).max()
            if fnorm < 1e-18:
                break
            g_ref = F_ref / fnorm

            mom_ref = mom_val * mom_ref + lr_ref * g_ref
            a_try = a_ref - lr_ref * mom_ref
            a_try = _proj_simplex(a_try, 1.0)

            b_ref = _fft_conv_sq(a_try, RM)
            mc_ref = b_ref.max()

            if mc_ref < ref_best_mc:
                ref_best_mc = mc_ref
                ref_best_a = a_try.copy()

            a_ref = a_try

            if ref_step % 100 == 0:
                frac = min(1.0, ref_step / 3000.0)
                tau_ref = max(1e-8, mc_ref / max(2, int(10 * (1 - frac) + 2)))
                lr_ref = max(0.01, 0.15 * (1 - 0.7 * frac))
                mom_val = min(0.95, 0.9 + 0.03 * frac)

        # Final eval
        try:
            sc_ref = float(evaluate_sequence([float(x) for x in ref_best_a]))
        except Exception:
            sc_ref = 2.0 * rn * ref_best_mc

        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = ref_best_a.copy()
            print(f"  Refinement: new best={best_score:.6f}")

    # Also evaluate parent if we haven't beaten it
    if parent is not None:
        try:
            ps = float(evaluate_sequence([float(x) for x in parent]))
            if ps < best_score:
                best_score = ps
                best_seq = parent.copy()
                print(f"  Parent wins: {best_score:.6f}")
        except Exception:
            pass

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024)

    out = np.clip(best_seq.astype(np.float64), 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.ones(out.size)

    try:
        final_sc = float(evaluate_sequence([float(x) for x in out]))
        print(f"FINAL: score={final_sc:.6f} n={out.size}")
    except Exception:
        final_sc = float('inf')
        print(f"FINAL: n={out.size} (eval failed)")

    return [float(x) for x in out]
