"""
Search for a non-negative sequence minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses L-BFGS-B with soft-max surrogate, followed by aggressive subgradient polishing
with adaptive step sizes and peak-targeted updates.
"""

import numpy as np
import time
from scipy.optimize import minimize

N = 4096
FFT_SELF = 8192  # next power of 2 >= 2*N - 1
FFT_GRAD = 32768  # next power of 2 >= 3*N - 2

best_score = float('inf')
best_a = None
budget_end = 0.0

def self_conv(a):
    fa = np.fft.rfft(a, FFT_SELF)
    c = np.fft.irfft(fa * fa, FFT_SELF)
    return c[:2*N - 1]

def compute_score(a):
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * N * np.max(c) / (s * s)

def update_best(a):
    global best_score, best_a
    sc = compute_score(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()
    return sc

def normalize(a):
    s = np.sum(a)
    if s <= 0:
        return np.ones(N) / N
    return a / s

def softmax(z):
    z = np.clip(z, -50.0, 50.0)
    m = np.max(z)
    e = np.exp(z - m)
    return e / np.sum(e)

def log_probs(a):
    a = np.clip(a, 1e-30, None)
    return np.log(a) - np.log(np.sum(a))  # log of probability vector

def seq_to_z(a):
    """Convert a positive sequence to log-differences (for softmax parameterization)."""
    a = np.clip(a, 1e-30, None)
    log_a = np.log(a)
    z = log_a[1:] - log_a[0]
    return np.clip(z, -50.0, 50.0)

def z_to_seq(z):
    """Convert log-differences back to probability vector."""
    z = np.clip(z, -50.0, 50.0)
    # p[0] = 1, p[i] = exp(z[i-1]) (unnormalized)
    log_p = np.zeros(N)
    log_p[1:] = z
    # Subtract max for numerical stability
    log_p -= np.max(log_p)
    p = np.exp(log_p)
    p /= np.sum(p)
    return p

def grad_p_from_peaks(p, c, peak_idx, peak_weights):
    """Compute gradient of weighted peak sum with respect to p."""
    n = N
    gp = np.zeros(n)
    for i, k in enumerate(peak_idx):
        w = peak_weights[i]
        # dc[k]/dp[j] = p[k-j] (if 0<=k-j<n) + p[j-k] (if 0<=j-k<n)
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:
            js = np.arange(lo1, hi1 + 1)
            gp[js] += w * p[k - js]
        lo2 = max(k, 0)
        hi2 = min(k + n - 1, n - 1)
        if lo2 <= hi2:
            js = np.arange(lo2, hi2 + 1)
            gp[js] += w * p[js - k]
    return gp

def grad_z_from_p_grad(gp, p):
    """Convert gradient w.r.t. p to gradient w.r.t. z (log-difference parameterization)."""
    # p[0] = exp(-logsumexp(z)), p[j] = exp(z[j-1] - logsumexp(z))
    # dp[j]/dz[k] = p[j] * (delta(j,k+1) - p[k+1]) for j,k >= 1
    # dp[0]/dz[k] = -p[0] * p[k+1]
    # So dz-gradient: gz[k] = sum_j gp[j] * dp[j]/dz[k]
    # = gp[k+1]*p[k+1] - p[k+1]*sum_j gp[j]*p[j]
    sdot = np.dot(gp, p)
    gz = p[1:] * (gp[1:] - sdot)
    return gz

def soft_max_surrogate_and_grad(c, alpha):
    """Compute soft-max of c and its gradient."""
    cmax = np.max(c)
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = np.sum(e)
    J = cmax + np.log(s) / alpha
    w = e / s
    return J, w

def objective_and_grad(x, alpha):

        raise ValueError("timeout")
    z = np.clip(x, -50.0, 50.0)
    p = z_to_seq(z)

    J, w = soft_max_surrogate_and_grad(c, alpha)
    gp = grad_p_from_peaks(p, c, np.where(w > 1e-10)[0], w[w > 1e-10])
    gz = grad_z_from_p_grad(gp, p)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    # Update best
    sc = compute_score(p)
    update_best(p)
    return float(J), gz
