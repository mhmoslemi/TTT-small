def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)

def _conv_sq(a):

def _adam_step(a, grad, m_buf, v_buf, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step."""
    m_buf = beta1 * m_buf + (1 - beta1) * grad
    v_buf = beta2 * v_buf + (1 - beta2) * grad**2
    m_hat = m_buf / (1 - beta1**t)
    v_hat = v_buf / (1 - beta2**t)

    return a_new, m_buf, v_buf

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a.
Uses Adam optimizer with adaptive smoothed-max gradient, multiple parametric
initializations, peak-shaving refinement, and multiple restarts.
"""

import time
import math
import numpy as np


def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target} using Euclidean projection."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    cond = sv * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n)
    rho = rho_idx[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _conv_sq_fft(a):
    """Linear convolution a*a via FFT. Returns array of length 2*n-1."""
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _conv_sq_fft_m(a, m):
    """Linear convolution a*a via FFT with precomputed size m. Returns array of length 2*n-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * a.size - 1]


def _grad_smoothed_max(c, a, L, tau):
    """Gradient of smoothed max approximation: sum_k w_k * b_k where w_k = softmax(c_k/tau) over top-L."""
    n = a.size
    total = 2 * n - 1
    L = min(L, total)
    if L == total:
        order = np.arange(total)
    else:
        order = np.argpartition(-c, L - 1)[:L]
    wc = c[order]
    w = np.exp(wc / max(tau, 1e-15))
    w_sum = w.sum()
    if w_sum <= 0:
        w = np.ones(L) / L
    else:
        w = w / w_sum
    grad = np.zeros(n, dtype=np.float64)
    for i in range(L):
        wk = w[i]
        if wk < 1e-12:
            continue
        k = int(order[i])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        grad[lo:hi + 1] += 2.0 * wk * a[k - hi:k - lo + 1][::-1]
    return grad


def _grad_exact_max(c, a, k_star):
    """Exact subgradient at the max lag k_star."""
    n = a.size
    k = int(k_star)
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def _parametric_seeds(n, base=None):
    """Generate diverse parametric seed sequences, all normalized to sum=1."""
    seeds = []
    x = np.linspace(0.0, 1.0, n)
    xi = np.linspace(-1.0, 1.0, n)

    # Uniform
    seeds.append(np.ones(n))

    # Cosine half-wave
    seeds.append(np.sin(np.linspace(0.0, math.pi, n)))

    # Raised cosine
    seeds.append(0.5 + 0.5 * np.cos(2.0 * math.pi * x))

    # Parabolic
    seeds.append(np.clip(1.0 - xi ** 2, 1e-10, None))

    # Flatter parabola
    seeds.append(np.clip(1.0 - 0.5 * xi ** 2, 1e-10, None))

    # Very flat top
    seeds.append(np.clip(1.0 - 0.2 * xi ** 2, 1e-10, None))

    # Hamming-like
    seeds.append(np.clip(0.54 + 0.46 * np.cos(2.0 * math.pi * x), 1e-10, None))

    # Hanning
    seeds.append(np.clip(0.5 + 0.5 * np.cos(2.0 * math.pi * x), 1e-10, None))

    # Gaussian bump
    for mu, sig in [(0.5, 0.3), (0.5, 0.4), (0.5, 0.25)]:
        seeds.append(np.exp(-0.5 * ((x - mu) / sig) ** 2))

    # Power shapes
    for p in [0.5, 0.7, 1.5, 2.0]:
        seeds.append(np.abs(xi) ** p + 0.1)

    # Slight non-uniform modulation
    for amp in [0.05, 0.1, 0.15, 0.2]:
        seeds.append(1.0 + amp * np.cos(2.0 * math.pi * x))
        seeds.append(1.0 - amp * xi ** 2)

    # From base if available
    if base is not None and np.size(base) == n:
        b = np.asarray(base, dtype=np.float64)
        b = np.clip(b, 0.0, None)
        if b.sum() > 0:
            seeds.append(b)

    # Normalize all
    result = []
    for s in seeds:
        s = np.array(s, dtype=np.float64)
        s = np.clip(s, 0.0, None)
        s_sum = s.sum()
        if s_sum < 1e-10:
            continue
        result.append(s / s_sum)
    return result


def _adam_update(a, g, m_buf, v_buf, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam update step. Returns new a, updated buffers."""
    m_buf = beta1 * m_buf + (1.0 - beta1) * g
    v_buf = beta2 * v_buf + (1.0 - beta2) * (g * g)
    m_hat = m_buf / (1.0 - beta1 ** t)
    v_hat = v_buf / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m_buf, v_buf

def _peak_shave(a, t_end, n_iter=500):
    """Peak-shaving: iteratively reduce the contribution to the peak lag."""
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1

    a = _proj_simplex(a, 1.0).copy()
    best_a = a.copy()
    best_mc = _conv_sq_fft_m(a, m).max()
    mom = np.zeros(n)
    momentum = 0.7
    lr = 0.03

    for it in range(n_iter):
        if time.monotonic() > t_end - 3.0:
            break

        c = _conv_sq_fft_m(a, m)
        mc = c.max()
        if mc < best_mc - 1e-15:
            best_mc = mc
            best_a = a.copy()

        # Find top-3 lags and use their combined gradient
        top3 = np.argpartition(-c, min(3, 2 * n - 2))[:min(3, 2 * n - 1)]
        vals = c[top3]
        w = np.exp(vals / 0.05)
        w = w / w.sum()

        g = np.zeros(n)
        for i, k in enumerate(top3):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            g[lo:hi + 1] += 2.0 * w[i] * a[k - hi:k - lo + 1][::-1]

        gn = np.linalg.norm(g)
        if gn > 1e-18:
            g = g / gn

        mom = momentum * mom - lr * g
        a = a + lr * mom
        a = _proj_simplex(a, 1.0)

    return best_a, best_mc


def _try_LP_refinement(a, t_end):
    """Try LP-based direction finding if cvxpy is available and time allows."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = a.size
    a = _proj_simplex(a, 1.0)
    c = _conv_sq_fft(a)
    M = c.max()

    # Use top constraints for speed
    n_con = min(100, 2 * n - 1)
    top_lags = np.argpartition(-c, n_con - 1)[:n_con]

    F_sub = np.zeros((n_con, n))
    for i, k in enumerate(top_lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        F_sub[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    b_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [F_sub @ b_var <= M * np.ones(n_con)]
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, max_iters=10000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            return None, None

    if b_var.value is None:
        return None, None

    b_val = np.clip(np.asarray(b_var.value, dtype=np.float64), 0.0, None)
    s = b_val.sum()
    if s < 1e-10:
        return None, None

    g_dir = b_val / s
    # Try blending

    best_sc = float('inf')
    for t_blend in [0.05, 0.1, 0.2, 0.4, 0.6]:

            if sc < best_sc:

                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_sc
