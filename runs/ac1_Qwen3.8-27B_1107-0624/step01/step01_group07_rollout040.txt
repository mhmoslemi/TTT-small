"""
Hybrid subgradient + LP optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast subgradient iterations with periodic LP-based direction finding,
multiple asymmetric initializations, and adaptive learning rates.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_update(a, lr, n_top_peaks=3):
    """One subgradient step. Assumes sum(a)=1, a>=0."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    
    # Find top peak positions
    peak_vals = c.max()
    # Get positions where c is close to max
    threshold = peak_vals * 0.999
    peak_idx = np.where(c >= threshold)[0]
    
    # Limit to top n_top_peaks by value
    if len(peak_idx) > n_top_peaks:
        peak_vals_at_idx = c[peak_idx]
        top_order = np.argsort(peak_vals_at_idx)[::-1][:n_top_peaks]
        peak_idx = peak_idx[top_order]
    
    # Compute subgradient
    subgrad = np.zeros(n)
    for k in peak_idx:
        # subgrad[j] += 2 * a[k-j] (when k-j is valid)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            subgrad[j] += 2.0 * a[k - j]
    
    subgrad /= len(peak_idx)
    
    # Update
    a_new = a - lr * subgrad
    a_new = project_simplex(a_new)
    return a_new

def subgradient_optimize(a, deadline, rng, verbose=False):
    """Run subgradient method until deadline."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.1
    stall = 0
    it = 0
    n_top_peaks = 3
    
    while time.monotonic() < deadline:
        it += 1
        a = subgradient_update(a, lr, n_top_peaks)
        sc = compute_score(a)
        
        if sc < best_score - 1e-10:
            best_a = a.copy()
            best_score = sc
            stall = 0
            # Slightly increase lr
            lr = min(0.3, lr * 1.01)
        else:
            stall += 1
            lr = max(0.001, lr * 0.95)
            n_top_peaks = min(10, n_top_peaks + 1) if stall % 10 == 0 else n_top_peaks
        
        if stall > 100:
            # Perturb
            noise = rng.normal(0, 0.05, n)
            a_new = best_a + noise * best_a
            a_new = project_simplex(np.maximum(a_new, 0))
            a = a_new
            stall = 0
            lr = 0.05
            n_top_peaks = 3
        
        if verbose and it % 1000 == 0:
            print(f"        SG iter {it}, score={best_score:.6f}, lr={lr:.4f}")
    
    return best_a, best_score

"""
Hybrid multiplicative-weights + LP optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based convolution, 
asymmetric initializations, and adaptive peak targeting.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def convolve_fft(a, b):
    """FFT-based full convolution."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def multiplicative_weights_step(a, lr, threshold_frac=0.95):
    """One multiplicative weights (mirror descent) step."""
    n = len(a)
    c = convolve_fft(a, a)
    peak = c.max()
    if peak < 1e-15:
        return a
    
    # Find peaks above threshold
    threshold = threshold_frac * peak
    peak_mask = c >= threshold
    peak_idx = np.where(peak_mask)[0]
    
    if len(peak_idx) == 0:
        peak_idx = np.array([int(np.argmax(c))])
    
    # Compute "pressure" at each position
    # pressure[j] = sum over peak positions k of a[k-j] (when valid)
    pressure = np.zeros(n)
    for k in peak_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For j in [lo, hi], a[k-j] is valid
        # pressure[j] += a[k-j] for j in [lo, hi]
        # This is a slice of a
        js = np.arange(lo, hi + 1)
        vals = a[k - js]  # a[k-lo], a[k-lo+1], ..., a[k-hi]
        pressure[js] += vals
    
    # Normalize pressure
    pressure /= len(peak_idx)
    
    # Multiplicative update
    a_new = a * np.exp(-lr * pressure)
    a_new = np.maximum(a_new, 0)
    if a_new.sum() < 1e-15:
        return a
    a_new /= a_new.sum()
    return a_new

def mw_optimize(a, deadline, rng, verbose=False):
    """Run multiplicative weights optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    if a.sum() < 1e-10:
        return a.copy(), np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.5
    threshold_frac = 0.95
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        a = multiplicative_weights_step(a, lr, threshold_frac)
        sc = compute_score(a)
        
        if sc < best_score - 1e-10:
            best_a = a.copy()
            best_score = sc
            stall = 0
            lr = min(2.0, lr * 1.005)
        else:
            stall += 1
            lr = max(0.01, lr * 0.97)
            if stall % 20 == 0:
                threshold_frac = max(0.5, threshold_frac - 0.02)
        
        if stall > 80:
            # Structured perturbation: cyclic shift
            shift = rng.integers(1, max(2, n // 4))
            a = np.roll(best_a, shift)
            # Smooth a bit
            kernel = np.ones(3) / 3
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, 0)
            a /= a.sum()
            stall = 0
            lr = 0.3
            threshold_frac = 0.9
        
        if verbose and it % 5000 == 0:
            print(f"        MW iter {it}, score={best_score:.6f}, lr={lr:.4f}, thresh={threshold_frac:.2f}")
    
    return best_a, best_score

def lp_direction(a, M=None):
    """Solve LP to find best direction g.
    max sum(g) s.t. conv(a,g)[k] <= M, g >= 0
    Returns normalized g or None.
    """
    n = len(a)
    if M is None:
        c = convolve_fft(a, a)
        M = c.max()
    if M < 1e-15:
        return None
    
    # Build constraint matrix
    # conv(a, g)[k] = sum_{i: valid} a[i] * g[k-i]
    # For a given g[j], it contributes to conv[k] for k = j, j+1, ..., j+n-1
    # Specifically, g[j] contributes a[k-j] * g[j] to conv[k]
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 2, j + n - 1)
        # For k in [j, j+n-1], A[k, j] = a[k-j]
        A[lo:hi+1, j] = a[:hi-lo+1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0 and res.x.sum() > 1e-15:
            return res.x / res.x.sum()
    except Exception:
        pass
    return None

def lp_optimize(a, deadline, rng, verbose=False):
    """LP-based optimization (Frank-Wolfe style)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    if a.sum() < 1e-10:
        return a.copy(), np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    step = 0.1
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = convolve_fft(a, a)
        M = c.max()
        
        g = lp_direction(a, M)
        if g is None:
            break
        
        improved = False
        for shrink in [1.0, 0.7, 0.5, 0.3, 0.1]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.maximum(a_new, 0)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break
        
        if improved:
            stall = 0
            step = min(0.4, step * 1.02)
        else:
            stall += 1
            step = max(0.01, step * 0.8)
            if stall > 30:
                break
        
        if verbose and it % 100 == 0:
            print(f"        LP iter {it}, score={best_score:.6f}, step={step:.4f}")
    
    return best_a, best_score

def combined_optimize(a, deadline, rng, verbose=False):
    """Combined MW + LP optimization."""
    n = len(a)

        if sc < best_score:
            best_a, best_score = a, sc
            print(f"        After MW polish: {best_score:.6f}")
    
    return best_a, best_score

def make_linear_ramp(n, direction='left'):
    """Linear ramp initialization."""
    if direction == 'left':

    """Exponential decay initialization."""
    if direction == 'left':
        return np.exp(-rate * np.arange(n) / n)
    else:
        return np.exp(-rate * (n - 1 - np.arange(n)) / n)

def make_cosine(n):
    """Raised cosine window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))

def make_bimodal(n):
    """Two peaks at 1/4 and 3/4 positions."""
    x = np.arange(n)
    c1, c2 = n * 0.25, n * 0.75
    sigma = n * 0.1
    return np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)

def make_asymmetric_gaussian(n, center_frac=0.3, sigma_frac=0.2):
    """Asymmetric Gaussian (off-center)."""
    x = np.arange(n)
    center = center_frac * n
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((x - center) / sigma) ** 2)

                a = best_a.copy()
            else:
                a /= a_sum
            stall = 0
            lr = 0.5
            min_ratio = 0.7
            peak_power = 2.0

        if verbose and it % 2000 == 0:
            print(f"        MW iter {it}, score={best_score:.6f}, lr={lr:.4f}, min_r={min_ratio:.2f}")

    return best_a, best_score


def build_lp_matrix_sparse(a):
    """Build sparse LP constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    vals = np.tile(a, n)
    # rows for column j are j, j+1, ..., j+n-1
    rows = np.arange(n * n).reshape(n, n) + np.arange(n).reshape(-1, 1)
    rows = rows.flatten()
    cols = np.tile(np.arange(n), n)
    vals = np.tile(a, n)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_direction(a, M=None, time_limit=20):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    if M is None:
        c = convolve_fft(a, a)
        M = c.max()
    if M < 1e-15:
        return None

    A = build_lp_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x.sum() > 1e-15:
            return res.x / res.x.sum()
    except Exception:
        pass
    return None


def lp_optimize(a, deadline, rng, verbose=False):
    """LP-based Frank-Wolfe style optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    if a.sum() < 1e-10:
        return a.copy(), np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    step = 0.1
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = convolve_fft(a, a)
        M = c.max()

        g = lp_direction(a, M, time_limit=15)
        if g is None:
            break

        improved = False
        for shrink in [1.0, 0.7, 0.5, 0.3, 0.1]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.maximum(a_new, 0)
            a_sum = a_new.sum()
            if a_sum < 1e-10:
                continue
            a_new /= a_sum
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.35, step * 1.03)
        else:
            stall += 1
            step = max(0.02, step * 0.8)
            if stall > 50:
                break

        if verbose and it % 50 == 0:
            print(f"        LP iter {it}, score={best_score:.6f}, step={step:.4f}")

    return best_a, best_score


def combined_optimize(a, deadline, rng, verbose=False):
    """
    Combined MW + LP optimization with multiple phases.
    Phase 1: MW (fast exploration)
    Phase 2: LP (precise direction finding)
    Phase 3: MW polish
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    if a.sum() < 1e-10:
        return a.copy(), np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    total_time = deadline - time.monotonic()

    # Phase 1: MW (60% of time)
    mw_deadline = time.monotonic() + total_time * 0.55
    if time.monotonic() < mw_deadline:
        a, sc = mw_optimize(a, mw_deadline, rng, verbose)
        if sc < best_score:
            best_a, best_score = a, sc
            if verbose:
                print(f"        After MW: {best_score:.6f}")

    # Phase 2: LP (25% of time) - only if n is not too large
    lp_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.45
    if n <= 1500 and time.monotonic() < lp_deadline - 10:
        a, sc = lp_optimize(best_a, lp_deadline, rng, verbose)
        if sc < best_score:
            best_a, best_score = a, sc
            if verbose:
                print(f"        After LP: {best_score:.6f}")

    # Phase 3: Final MW polish
    if time.monotonic() < deadline - 5:
        a, sc = mw_optimize(best_a, deadline, rng, verbose)
        if sc < best_score:
            best_a, best_score = a, sc
            if verbose:
                print(f"        After MW polish: {best_score:.6f}")

    return best_a, best_score


def make_linear_ramp(n, direction='left'):
    if direction == 'left':
        return np.arange(1, n + 1, dtype=np.float64)
    else:
        return np.arange(n, 0, -1, dtype=np.float64)


def make_exponential(n, rate=1.0, direction='left'):
    if direction == 'left':
        return np.exp(-rate * np.arange(n) / n)
    else:
        return np.exp(-rate * (n - 1 - np.arange(n)) / n)


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_asymmetric_gaussian(n, center_frac=0.3, sigma_frac=0.2):
    x = np.arange(n, dtype=np.float64)
    center = center_frac * n
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((x - center) / sigma) ** 2)


def make_bimodal(n):
    x = np.arange(n, dtype=np.float64)
    c1, c2 = n * 0.25, n * 0.75
    sigma = n * 0.12
    return (np.exp(-0.5 * ((x - c1) / sigma) ** 2) +
            np.exp(-0.5 * ((x - c2) / sigma) ** 2))


def make_triangle_offset(n, offset_frac=0.2):
    """Triangular window with offset center."""
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * n + offset_frac * n
    half = 0.4 * n
    return np.maximum(0, 1.0 - np.abs(x - center) / half)


def get_config_init(n, init_type):
    """Get initialization for given n and type."""
    if init_type == 'ramp_left':
        return make_linear_ramp(n, 'left')
    elif init_type == 'ramp_right':
        return make_linear_ramp(n, 'right')
    elif init_type == 'exp_left':
        return make_exponential(n, 1.0, 'left')
    elif init_type == 'exp_left2':
        return make_exponential(n, 2.0, 'left')
    elif init_type == 'exp_right':
        return make_exponential(n, 1.0, 'right')
    elif init_type == 'cosine':
        return make_cosine(n)
    elif init_type == 'asym_gauss_03':
        return make_asymmetric_gaussian(n, 0.3, 0.2)
    elif init_type == 'asym_gauss_04':
        return make_asymmetric_gaussian(n, 0.4, 0.15)
    elif init_type == 'asym_gauss_025':
        return make_asymmetric_gaussian(n, 0.25, 0.25)
    elif init_type == 'bimodal':
        return make_bimodal(n)
    elif init_type == 'tri_offset':
        return make_triangle_offset(n, 0.15)
    elif init_type == 'tri_offset_neg':
        return make_triangle_offset(n, -0.15)
    elif init_type == 'random':
        return np.abs(np.random.default_rng(int(time.monotonic() * 10000)).normal(0, 1, n))
    else:
        return np.ones(n)


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using hybrid MW+LP optimization
    with multiple diverse asymmetric initializations and adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all() and p.sum() > 0.01:
            parent = p
            p_score = compute_score(parent)
            print(f"Parent: n={p.size}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Build configuration list: (n, init_type, weight)
    configs = []

    # Parent-based configs (highest priority)
    if parent is not None:
        pn = parent.size
        configs.append((pn, 'parent', 3.0))
        configs.append((pn, 'parent_mirror', 2.5))
        configs.append((pn, 'parent_shift_q', 2.0))
        configs.append((pn, 'parent_shift_e', 2.0))
        # Truncated parent
        if pn > 256:
            configs.append((pn // 2, 'parent', 1.5))

    # Key asymmetric initializations (the insight: asymmetry breaks the barrier)
    asymmetric_configs = [
        (512, 'ramp_left', 2.0),
        (512, 'ramp_right', 2.0),
        (512, 'exp_left', 1.8),
        (512, 'exp_left2', 1.5),
        (512, 'asym_gauss_03', 1.8),
        (512, 'asym_gauss_04', 1.5),
        (512, 'asym_gauss_025', 1.3),
        (512, 'tri_offset', 1.5),
        (512, 'tri_offset_neg', 1.5),
        (512, 'bimodal', 1.2),
        (256, 'ramp_left', 1.5),
        (256, 'ramp_right', 1.5),
        (256, 'exp_left', 1.2),
        (256, 'asym_gauss_03', 1.3),
        (256, 'tri_offset', 1.2),
        (1024, 'ramp_left', 1.5),
        (1024, 'ramp_right', 1.5),
        (1024, 'asym_gauss_03', 1.5),
        (1024, 'exp_left', 1.3),
        (1024, 'tri_offset', 1.2),
        (2048, 'ramp_left', 1.0),
        (2048, 'asym_gauss_03', 1.0),
        (2048, 'exp_left', 0.8),
    ]
    configs.extend(asymmetric_configs)

    # Some symmetric for comparison
    configs.append((512, 'cosine', 0.8))
    configs.append((256, 'cosine', 0.6))
    configs.append((1024, 'cosine', 0.6))

    # Random
    configs.append((512, 'random', 0.8))
    configs.append((256, 'random', 0.6))

    total_configs = len(configs)
    total_weight = sum(w for _, _, w in configs)
    available_time = budget_s - 40.0
    print(f"Configs: {total_configs}, total weight: {total_weight:.1f}, available: {available_time:.0f}s")

    # Sort by weight descending (do most promising first)
    configs_sorted = sorted(configs, key=lambda x: -x[2])

    # Time allocation proportional to weight
    time_allocs = [available_time * w / total_weight for _, _, w in configs_sorted]

    for idx, (n, init_type, weight) in enumerate(configs_sorted):
        if time.monotonic() > deadline:
            break

        config_time = time_allocs[idx]
        config_deadline = min(time.monotonic() + config_time, deadline)

        if config_deadline <= time.monotonic() + 3:
            continue

        # Initialize
        if init_type == 'parent':
            a = parent[:n].copy()
        elif init_type == 'parent_mirror':
            a = parent[:n][::-1].copy()
        elif init_type == 'parent_shift_q':
            a = np.roll(parent[:n], n // 4).copy()
        elif init_type == 'parent_shift_e':
            a = np.roll(parent[:n], n // 2).copy()
        else:
            a = get_config_init(n, init_type)

        a = np.maximum(a, 1e-10).astype(np.float64)
        a_sum = a.sum()
        if a_sum < 1e-10:
            a = np.ones(n)
            a_sum = n
        a /= a_sum

        desc = f"n={n},{init_type}"
        remaining = deadline - time.monotonic()
        print(f"[{idx+1}/{total_configs}] {desc}, alloc~{config_time:.0f}s, remaining~{remaining:.0f}s")

        a_opt, score = combined_optimize(a, config_deadline, rng, verbose=False)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: {best_score:.6f} ({desc})")
        elif np.isfinite(score):
            print(f"  => {score:.6f} (best: {best_score:.6f})")

    # Final intensive polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\nFinal polish: {remaining:.0f}s remaining, current best={best_score:.6f}, n={len(best_seq)}")

        # Try multiple polish strategies
        polish_results = []

        # Strategy 1: Pure MW with very fine tuning
        if time.monotonic() < deadline - 10:
            t_dead = time.monotonic() + remaining * 0.4
            a_p, sc_p = mw_optimize(best_seq.copy(), t_dead, rng, verbose=True)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                print(f"  MW polish: {best_score:.6f}")

        # Strategy 2: LP refinement
        if len(best_seq) <= 1200 and time.monotonic() < deadline - 10:
            t_dead = time.monotonic() + (deadline - time.monotonic()) * 0.5
            a_p, sc_p = lp_optimize(best_seq.copy(), t_dead, rng, verbose=True)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                print(f"  LP polish: {best_score:.6f}")

        # Strategy 3: One more MW pass
        if time.monotonic() < deadline - 5:
            a_p, sc_p = mw_optimize(best_seq.copy(), deadline, rng, verbose=True)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                print(f"  Final MW: {best_score:.6f}")

    if best_seq is None:
        # Fallback
        best_seq = make_linear_ramp(256, 'left')
        best_score = compute_score(best_seq)

    # Ensure valid for evaluator
    best_seq = np.maximum(best_seq, 0)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())

    # Cap at 1000 to be safe
    best_seq = np.minimum(best_seq, 1000.0)

    elapsed_total = time.monotonic() - start_time
    final_score = compute_score(best_seq)
    print(f"\nDONE in {elapsed_total:.1f}s. Final score={final_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
