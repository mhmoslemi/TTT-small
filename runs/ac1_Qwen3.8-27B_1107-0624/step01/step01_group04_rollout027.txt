Let me also think about the gradient approach more carefully. The objective (with sum=1 normalization) is:
F(a) = 2n * max_k (a*a)[k]

The subgradient with respect to a at the point where the max is achieved at k* is:
dF(a)/da[i] = 2n * sum_j (a*a)[i,j at k*] = 2n * (a reversed, convolved with a)[i]

More precisely, if (a*a)[k] = sum_j a[j]*a[k-j], then:

So the subgradient of F at the active constraint k* is:
grad[i] = 2n * a[k*-i] (for 0 <= k*-i <= n-1, else 0)

With the sum constraint (a >= 0, sum(a) = 1), the projected gradient step is:

The projection onto the simplex is well-known.

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAS_SCIPY = True
except:
    HAS_SCIPY = False

def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def convolve_self(a):
    n = len(a)
    if n <= 1024:
        return np.convolve(a, a)
    L = next_pow2(2*n-1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A*A, L)[:2*n-1]
    return np.maximum(c, 0)

def obj_value(a):
    """Objective assuming sum(a) is already fixed."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    c = convolve_self(a_norm)
    return 2.0 * n * np.max(c)

def make_power_law(n, exponent, decay_dir='left'):
    """a[i] proportional to (i+1)^(-exponent) or (n-i)^(-exponent)."""
    idx = np.arange(n, dtype=float) + 1
    if decay_dir == 'left':
        a = idx**(-exponent)  # decays from left to right
    else:
        a = (n + 1 - idx)**(-exponent)  # decays from right to left
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s

def make_exponential(n, rate):
    idx = np.arange(n, dtype=float)
    a = np.exp(-rate * idx)
    s = a.sum()
    return a / s

def make_asymmetric_poly(n, p, q):
    """a[i] proportional to ((n-i)/n)^p * (i/n+1)^q"""
    idx = np.arange(n, dtype=float)
    x = idx / (n - 1) if n > 1 else np.array([0.0])
    a = (1 - x)**p * (x + 0.1)**q
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s
