"""
Optimizes non-negative sequence to minimize 2n*max(autocorr)/(sum)^2.
Uses projected subgradient descent with adaptive step size and line search,
combined with L-BFGS smooth-max polishing and multi-start exploration.
Explores multiple sequence lengths and diverse parametric initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize

_best_score = float('inf')
_best_seq = None


def _fft_conv(a):
    n = len(a)
    fs = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, fs)
    c = np.fft.irfft(fa * fa, fs).real[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return float('inf')
    return 2.0 * n * float(_fft_conv(a).max()) / (S * S)


def _update(a, s):
    global _best_score, _best_seq
    if np.isfinite(s) and s < _best_score:
        _best_score = float(s)
        _best_seq = a.copy()


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    r = np.where(u * np.arange(1, n + 1) > cv - 1)[0]
    if len(r) == 0:
        return np.ones(n) / n
    r = r[-1]
    th = (cv[r] - 1) / (r + 1)
    return np.maximum(v - th, 0)


def _grad_active(a, conv, topk, weights=None):
    n = len(a)
    idx = np.argpartition(conv, -topk)[-topk:]
    if weights is None:
        pv = conv[idx]
        w = pv / pv.sum()
    else:
        w = weights
    g = np.zeros(n)
    for k, wk in zip(idx, w):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        g[lo:hi] += 2.0 * wk * a[k - np.arange(lo, hi)]
    return g, idx, w


def _subgrad_optimize(a0, deadline, max_iters=50000, topk_init=5, topk_max=25, verbose=False):
    a = a0.copy()
    S = a.sum()
    if S > 0:
        a = a / S
    score = _score(a)
    _update(a, score)

    step_base = 0.15
    stale = 0
    best_local = score

    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        if t % 5000 == 0 and verbose:
            print(f"    subgrad t={t} score={score:.6f} topk={min(topk_max, topk_init + t // 300)}")

        conv = _fft_conv(a)
        topk = min(topk_max, topk_init + t // 300)
        grad, _, _ = _grad_active(a, conv, topk)

        step = step_base / (1.0 + 0.3 * np.log1p(t / 100.0))
        accepted = False
        for mult in [1.0, 0.5, 0.25, 2.0, 0.1]:
            cand = a - step * mult * grad
            cand = _proj_simplex(cand)
            sc = _score(cand)
            if sc < score - 1e-14:
                a = cand
                score = sc
                _update(a, score)
                accepted = True
                stale = 0
                if score < best_local:
                    best_local = score
                break

        if not accepted:
            stale += 1
            if stale > 200:
                # Try random perturbation to escape
                if verbose:
                    print(f"    subgrad stale at t={t}, perturbing")
                for _ in range(5):
                    pert = a + 0.02 * np.random.randn(len(a))
                    pert = _proj_simplex(pert)
                    sc = _score(pert)
                    if sc < score:
                        a = pert
                        score = sc
                        _update(a, score)
                        stale = 0
                        break
                else:
                    step_base *= 0.95
                    if step_base < 0.005:
                        break

    return a, score


def _lbfgs_optimize(a0, alpha, max_iter, deadline, verbose=False):
    n = len(a0)

    a_safe = np.maximum(a0, 1e-8)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()

    def obj_grad(z):
        if time.monotonic() > deadline:
            raise ValueError("timeout")
        z = np.clip(z, -25.0, 25.0)
        zm = z - z.max()
        a = np.exp(zm)
        a = a / a.sum()

        conv = _fft_conv(a)
        cmax = float(conv.max())
        u = alpha * (conv - cmax)
        e = np.exp(np.clip(u, -50, 50))
        s = float(e.sum())
        if s < 1e-30:
            return 2.0 * n * cmax, np.zeros(n)

        J = 2.0 * n * (cmax + np.log(s) / alpha)
        w = e / s

        # Only use top weights for efficiency
        thresh = 1e-10
        active = np.where(w > thresh)[0]
        if len(active) == 0:
            return J, np.zeros(n)
        wa = w[active]

        grad_a = np.zeros(n)
        for k, wk in zip(active, wa):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            grad_a[lo:hi] += 2.0 * wk * a[k - np.arange(lo, hi)]

        sdot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - sdot)
        grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)

        return float(J), grad_z

    try:
        result = minimize(
            obj_grad, z0, method='L-BFGS-B', jac=True,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-11, 'maxls': 20}
        )
        zf = np.clip(result.x, -25.0, 25.0)
        af = np.exp(zf - zf.max())
        af = af / af.sum()
        sc = _score(af)
        _update(af, sc)
        if verbose:
            print(f"      L-BFGS alpha={alpha:.0e}: score={sc:.6f}")
    except Exception:
        pass


def _make_inits(n, rng, base_seq=None):
    inits = []
    x = np.linspace(0.0, 1.0, n)
    # Uniform
    inits.append(np.ones(n) / n)
    # Linear
    for sl in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        for d in [1, -1]:
            f = 1.0 + sl * d * (x - 0.5)
            f = np.maximum(f, 0.01)
            inits.append(f / f.sum())
    # Quadratic / parabolic
    for a2 in [-0.5, -0.3, -0.1, 0.1, 0.3, 0.5, 1.0]:
        f = 1.0 + a2 * (2 * x - 1) ** 2
        f = np.maximum(f, 0.01)
        inits.append(f / f.sum())
    # Sinusoidal
    for freq in [1, 2, 3, 4, 5, 7, 10]:
        for amp in [0.2, 0.4, 0.7]:
            f = 1.0 + amp * np.sin(2 * np.pi * freq * x)
            f = np.maximum(f, 0.01)
            inits.append(f / f.sum())
    # Exponential
    for beta in [1, 2, 4, 8, 16, 32]:
        for d in [1, -1]:
            f = np.exp(d * beta * (x - 0.5))
            inits.append(f / f.sum())
    # Power law
    for p in [0.5, 0.75, 1.5, 2.0, 3.0]:
        for d in [1, -1]:
            base_x = x if d > 0 else 1 - x
            f = np.maximum(base_x, 1e-6) ** p
            inits.append(f / f.sum())
    # Random low-freq
    for _ in range(30):
        f = np.ones(n)
        for _k in range(rng.integers(3, 8)):
            freq = float(rng.integers(1, n // 15 + 1))
            amp = rng.normal(0, 0.4)
            ph = rng.uniform(0, 2 * np.pi)
            f += amp * np.sin(2 * np.pi * freq * x + ph)
        f = np.maximum(f, 0.01)
        inits.append(f / f.sum())
    # Random smooth (spline-like)
    for _ in range(20):
        knots = np.sort(rng.uniform(0, 1, size=rng.integers(5, 15)))
        vals = rng.uniform(0.2, 1.5, size=len(knots))
        f = np.interp(x, knots, vals)
        f = np.maximum(f, 0.01)
        inits.append(f / f.sum())
    # Base sequence
    if base_seq is not None:
        try:
            b = np.asarray(base_seq, dtype=np.float64)
            b = np.clip(b, 0, None)
            if b.sum() > 0:
                if len(b) != n:
                    ox = np.linspace(0, 1, len(b))
                    nx = np.linspace(0, 1, n)
                    b = np.interp(nx, ox, b)
                b = np.maximum(b, 0)
                b = b / b.sum()
                inits.append(b)
        except Exception:
            pass
    return inits
