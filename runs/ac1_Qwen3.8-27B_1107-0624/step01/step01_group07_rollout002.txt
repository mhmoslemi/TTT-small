def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve_with_matrix(A, M, n):
    """Solve: max sum(g) s.t. A @ g <= M, g >= 0. Returns g or None."""
    c_obj = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

    """

    Returns (a_new, score).
    """

    c = np.convolve(a, a, mode='full')
    # Softmax weights (temperature t)
    tc = t * c
    tc = tc - tc.max()
    weights = np.exp(tc)
    weights /= weights.sum()
    
    # Gradient: g_m = 2 * sum_j a[j] * weights[m+j] for m=0..n-1

    print("=== STAGE 4: Final LP ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        best_seq, best_score = targeted_lp_opt(best_seq, deadline - 10)
    
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0
    
    # Scale to ensure sum >= 0.01
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())
    
    # Verify with evaluate_sequence
    try:
        ev = evaluate_sequence([float(x) for x in best_seq])
        print(f"    evaluate_sequence says: {ev:.6f}")
        if not np.isfinite(ev):
            print("    WARNING: Invalid! Using fallback.")
            best_seq = np.ones(512)
            best_score = 2.0
    except Exception as e:
        print(f"    evaluate_sequence error: {e}")
    
    elapsed = time.monotonic() - start_time
    print(f"DONE in {elapsed:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
