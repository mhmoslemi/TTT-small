```python
import numpy as np
import time
from scipy.optimize import minimize

class _Timeout(Exception):
    pass

# Global state
_g_best_score = float('inf')
_g_best_p = None
_g_deadline = 0.0

def _check_time():
    if time.monotonic() > _g_deadline:
        raise _Timeout()

def _fft_size(n):
    return 1 << (n - 1).bit_length()

def _self_conv(p):
    """Full self-convolution of p."""
    n = len(p)
    size = _fft_size(2 * n - 1)
    fp = np.fft.rfft(p, size)
    c = np.fft.irfft(fp * fp, size)[:2*n-1]
    np.maximum(c, 0, out=c)  # numerical safety
    return c

def _grad_p(p, w):
    """
    Compute grad_p[j] = 2 * sum_k w[k] p[k-j] for j = 0,...,n-1.
    Using FFT: 2 * (full_conv(reversed_p, w))[n-1 : n-1+n]
    """
    n = len(p)
    r = p[::-1]  # reversed
    # full_conv(r, w) has length n + len(w) - 1 = n + (2n-1) - 1 = 3n-2
    size = 1 << (3 * n - 2 - 1).bit_length()
    fr = np.fft.rfft(r, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fr * fw, size)
    # We need conv[n-1 : n-1+n]
    return 2.0 * conv[n-1 : n-1+n]

def _objective_grad(z, alpha, n):
    """Compute smooth objective and gradient w.r.t. logits z (length n)."""
    _check_time()
    # Softmax
    z_shift = z - np.max(z)
    exp_z = np.exp(z_shift)
    Z = np.sum(exp_z)
    p = exp_z / Z
    
    # Self-convolution
    c = _self_conv(p)
    c_max = np.max(c)
    
    # Update best
    global _g_best_score, _g_best_p
    true_obj = 2.0 * n * c_max
    if true_obj < _g_best_score:
        _g_best_score = true_obj
        _g_best_p = p.copy()
    
    # Smooth surrogate
    u = alpha * (c - c_max)
    exp_u = np.exp(u)
    S = np.sum(exp_u)
    if S > 0 and np.isfinite(S):
        J = c_max + np.log(S) / alpha
        w = exp_u / S
    else:
        J = c_max
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
```
