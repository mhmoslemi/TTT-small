def _fft_conv(a, m):
    """Compute linear convolution a*a via FFT, returning first 2n-1 elements."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*a.size-1]


def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    css = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n+1) > (css - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    lam = (css[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _objective(a):
    """Compute J = 2*n*max(c) / S^2 where c = convolve(a,a), S = sum(a)."""
    n = a.size
    c = _fft_conv(a, _fft_len(n))
    S = a.sum()
    if S < 1e-12:
        return float('inf')
    return 2.0 * n * c.max() / (S * S)


def _fft_len(n):
    m = 1
    while m < 2*n - 1:
        m <<= 1
    return m


def _subgradient(a, c, active_set):
    """Compute subgradient of max_k c_k at a, using the given active set.
    grad_i = 2 * sum_{k in active} (1/|active|) * a[k-i] (with a[j]=0 for j out of range)"""
    n = a.size
    g = np.zeros(n)
    w = 1.0 / len(active_set)
    for k in active_set:
        # grad contribution from c_k: 2*a[k-i] for i where 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * w * a[k - np.arange(lo, hi+1)]
    return g

"""
Minimize J = 2*n*max(conv(a,a))/(sum(a))^2 for a >= 0.
Uses projected subgradient with exact active-set max, Nesterov momentum,
backtracking line search, and multiple sequence lengths. Exploits KKT
structure for Fourier-based initialization.
"""

import time
import math
import numpy as np


def _fft_len(n):
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    return m


def _conv_sq(a, m):
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * a.size - 1]


def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    if n == 0:
        return v
    sv = np.sort(v)[::-1]
    css = np.cumsum(sv)
    rho_arr = sv * np.arange(1, n + 1) - (css - target)
    rho_idx = np.nonzero(rho_arr > 0)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = rho_idx[-1]
    lam = (css[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _to_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s <= 1e-14:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _subgrad_active(a, c, active, L=1):
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    w = 1.0 / len(active)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        src = k - idx
        g[idx] += 2.0 * w * a[src]
    return g


def _fourier_init(n, mu_lags):
    exp_vals = np.exp(2j * np.pi * np.arange(n) / (2 * n))
    mu_ft = np.zeros(n, dtype=np.complex128)
    for k in mu_lags:
        fk = k % (2 * n)
        if fk < n:
            mu_ft[fk] += 1.0
        else:
            mu_ft[2 * n - fk] += 1.0
    if n > 0:
        mu_ft[0] = max(mu_ft[0].real, 1e-12)
    valid = np.abs(mu_ft) > 1e-10
    a_ft = np.zeros(n, dtype=np.complex128)
    a_ft[valid] = 1.0 / mu_ft[valid]
    a = np.fft.irfft(a_ft[:n], 2 * n)[:n] if n < 2 * n else np.fft.irfft(a_ft, 2 * n)[:n]
    a = np.real(a)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-12:
        a = np.ones(n)
    return a / a.sum()


def _optimize_one(a0, t_end, verbose=False):
    a = _to_sum1(np.abs(a0).astype(np.float64))
    n = a.size
    m = _fft_len(n)

    best_a = a.copy()
    best_M = _conv_sq(a, m).max()

    t_k = 1.0
    x_prev = a.copy()
    y = a.copy()
    eta = 0.15
    eta_max = 1.5
    eta_min = 1e-7
    tol_frac = 5e-5
    stall = 0
    stall_best = best_M

    step = 0
    last_print = 0.0

    while time.monotonic() < t_end - 0.3:
        c = _conv_sq(y, m)
        M_y = c.max()
        thr = M_y * max(tol_frac, 1e-12)
        active = np.where(c >= M_y - thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > 150:
            active = np.argsort(c)[::-1][:150]

        g = _subgrad_active(y, c, active)
        gn = np.linalg.norm(g)

        if gn < 1e-14:
            break

        x_new = _proj_simplex(y - eta * g / gn, 1.0)
        c_new = _conv_sq(x_new, m)
        M_new = c_new.max()

        if M_new > M_y * 1.005 + 1e-15:
            eta = max(eta * 0.5, eta_min)
            y = x_new.copy()
            x_prev = a.copy()
            a = x_new.copy()
            t_k = 1.0
            step += 1
            continue

        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        mom = (t_k - 1.0) / t_new
        y_new = x_new + mom * (x_new - x_prev)
        y_new = _proj_simplex(y_new, 1.0)

        if M_new < M_y * 0.995:
            eta = min(eta * 1.08, eta_max)

        a = x_new
        x_prev = a.copy()
        y = y_new
        t_k = t_new
        step += 1

        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 200:
            eta = max(eta * 0.85, eta_min)
            tol_frac *= 0.7
            if stall > 600:
                y = _to_sum1(best_a * np.exp(np.random.default_rng(step).normal(0, 0.02, n)))
                x_prev = y.copy()
                t_k = 1.0
                stall = 0

        if verbose and step % 200 == 0 and time.monotonic() - last_print > 8.0:
            last_print = time.monotonic()
            print(f"    step={step} M={best_M:.10f} eta={eta:.4f} tol={tol_frac:.2e} stall={stall}")

    return best_a, best_M


def _make_seed(n, kind, rng):
    if kind == 0:
        return np.ones(n)
    elif kind == 1:
        t = np.linspace(0, 1, n)
        return 0.1 + np.sin(np.pi * t) ** 0.7
    elif kind == 2:
        t = np.linspace(0, 1, n)
        return np.exp(-5.0 * (t - 0.5) ** 2)
    elif kind == 3:
        t = np.linspace(0, 1, n)
        return 0.05 + t ** 1.5
    elif kind == 4:
        t = np.linspace(0, 1, n)
        return 0.05 + (1.0 - t) ** 1.5
    elif kind == 5:
        t = np.linspace(0, 1, n)
        return 0.05 + (1.0 - np.cos(2 * np.pi * t)) / 2.0
    elif kind == 6:
        return np.abs(rng.standard_normal(n)) ** 0.5
    elif kind == 7:
        t = np.linspace(0, 1, n)
        return np.maximum(np.ones(n) - 2.5 * np.abs(t - 0.5), 0.05)
    else:
        return np.ones(n) * (0.5 + 0.5 * rng.random(n))
