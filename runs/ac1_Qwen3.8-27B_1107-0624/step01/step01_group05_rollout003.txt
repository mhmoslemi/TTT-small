    f = np.fft.rfft(p, fft_len)
    c = np.fft.irfft(f * f, fft_len).real
    c = c[:2*N-1]
    np.maximum(c, 0.0, out=c)
    return c

def _cross_conv(p, q, N, fft_len):
    """Compute linear convolution of p and q (each length N) using FFT."""
    fp = np.fft.rfft(p, fft_len)
    fq = np.fft.rfft(q, fft_len)
    c = np.fft.irfft(fp * fq, fft_len).real
    c = c[:2*N-1]
    np.maximum(c, 0.0, out=c)
    return c

def _grad_p_from_weights(p, w, N, fft_len):
    """
    Compute gradient of c_k = sum_j p_j * w_{k-j} (for k = i+j)
    with respect to p_i.
    Result: gp_i = 2 * sum_j p_j * w_{i+j} (by symmetry of self-conv)
    Wait, that's not right for cross-conv.
    
    For self-conv: c_k = sum_j p_j * p_{k-j}
    dc_k/dp_i = p_{k-i} + p_{i-k} = 2*p_{|k-i|} (if we use the symmetry)
    
    Actually, for the weighted version:
    If we have J = sum_k w_k * c_k where c = p*p,
    then dJ/dp_i = 2 * sum_k w_k * p_{k-i} (for valid k-i)
    = 2 * (w * p_reversed)[i + N - 1] (approximately)
    """
    # dJ/dp_i = 2 * sum_j p_j * w_{i+j}
    # This is 2 * (p_reversed * w)[N-1+i] where * is linear convolution
    pr = p[::-1]
    conv = _cross_conv(pr, w, N, fft_len)
    # conv[k] = sum_j pr[j] * w[k-j] = sum_j p[N-1-j] * w[k-j]
    # We want: gp[i] = 2 * sum_j p[j] * w[i+j]
    # = 2 * sum_j p[j] * w[i+j]
    # Let's use: (pr * w)[N-1+i] = sum_j pr[j] * w[N-1+i-j] = sum_j p[N-1-j] * w[N-1+i-j]
    # Let m = N-1-j: = sum_m p[m] * w[i+m] ✓
    h = conv[N-1:2*N-1]
    return 2.0 * h

def _probs_from_z(z, N):
    """Convert logits z (length N-1) to probability vector p (length N)."""
    z = np.clip(z, -50, 50)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p

def _score(p, N):
    """Compute the objective for probability vector p."""
    fft_len = _pow2(2 * N - 1)
    c = _self_conv(p, N, fft_len)
    return 2.0 * N * float(np.max(c))

def _density_to_z(f, N):
    """Convert a density (length N) to logits."""
    f = np.clip(f, 1e-30, None)
    logf = np.log(f)
    z = logf[1:] - logf[0]
    return np.clip(z, -50, 50)

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/(sum a)^2.
Uses multi-scale smooth-max L-BFGS in logit space with subgradient polishing,
diverse parametric initializations, and optional multi-length search.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None


def _pow2(n):
    p = 1
    while p < n:
        p <<= 1
    return p


def _self_conv(p, N, fft_len):
    f = np.fft.rfft(p, fft_len)
    c = np.fft.irfft(f * f, fft_len).real
    c = c[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _grad_p_from_weights(p, w, N, fft_len):
    pr = p[::-1]
    fp = np.fft.rfft(pr, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv = np.fft.irfft(fp * fw, fft_len).real
    h = conv[N - 1:2 * N - 1]
    return 2.0 * h


def _probs_from_z(z, N):
    z = np.clip(z, -50.0, 50.0)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _score_p(p, N, fft_len):
    c = _self_conv(p, N, fft_len)
    return 2.0 * float(N) * float(np.max(c))


def _density_to_z(f, N):
    f = np.clip(f, 1e-30, None)
    logf = np.log(f)
    z = logf[1:] - logf[0]
    return np.clip(z, -50.0, 50.0)


def _kaiser_z(beta, N):
    x = np.linspace(0, 1, N)
    if beta == 0:
        w = np.ones(N)
    else:
        from scipy.special import i0
        w = i0(beta * np.sqrt(1 - (2 * x - 1) ** 2)) / i0(beta)
    return _density_to_z(w, N)


def _blackman_z(N):
    x = np.arange(N, dtype=np.float64)
    w = 0.42 - 0.5 * np.cos(2 * np.pi * x / (N - 1)) + 0.08 * np.cos(4 * np.pi * x / (N - 1))
    w = np.maximum(w, 1e-10)
    return _density_to_z(w, N)


def _hamming_z(N):
    x = np.arange(N, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * x / (N - 1))
    w = np.maximum(w, 1e-10)
    return _density_to_z(w, N)


def _logistic_z(beta, N):
    x = np.linspace(0, 1, N)
    u = beta * (x - 0.5)
    u = np.clip(u, -50, 50)
    f = 2.0 / (1.0 + np.exp(-u))
    return _density_to_z(f, N)


def _exp_z(beta, N):
    x = np.linspace(0, 1, N)
    f = np.exp(beta * x)
    return _density_to_z(f, N)


def _power_z(a, N, reverse=False):
    x = np.linspace(0, 1, N)
    if reverse:
        f = np.power(1.0 - x + 1e-8, a)
    else:
        f = np.power(x + 1e-8, a)
    f = np.maximum(f, 1e-12)
    return _density_to_z(f, N)


def _poly_z(coef, N):
    x = np.linspace(0, 1, N)
    t = x - 0.5
    f = np.ones(N)
    for k in range(len(coef) - 1, -1, -1):
        f = f * t + coef[k]
    f = np.maximum(f, 1e-10)
    return _density_to_z(f, N)


def _random_z(rng, N):
    x = np.linspace(0, 1, N)
    t = x - 0.5
    z = np.zeros(N - 1)
    z[:N-1] += rng.uniform(-3, 3) * t[:N-1]
    z[:N-1] += rng.uniform(-1.5, 1.5) * (t[:N-1] ** 2)
    for fq in (1, 2, 3, 4, 5, 7):
        amp = rng.normal(0.0, 1.5 / (fq + 0.5))
        ph = rng.uniform(0, 2 * np.pi)
        z[:N-1] += amp * np.sin(2 * np.pi * fq * x[1:N] + ph)
    return np.clip(z, -50, 50)


class _Ctx:
    """Holds per-N constants and best state."""
    __slots__ = ('N', 'fft_self', 'fft_grad', 'X', 'best_score', 'best_z',
                 'bounds', 'deadline', 'main_deadline')

    def __init__(self, N):
        self.N = N
        self.fft_self = _pow2(2 * N - 1)
        self.fft_grad = _pow2(3 * N - 1)
        self.X = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
        self.best_score = float('inf')
        self.best_z = None
        self.bounds = [(-50.0, 50.0)] * (N - 1)
        self.deadline = 0.0
        self.main_deadline = 0.0


class SearchTimeout(Exception):
    pass


def _update_best(ctx, z, score):
    if np.isfinite(score) and score < ctx.best_score:
        ctx.best_score = float(score)
        ctx.best_z = np.array(z, copy=True)


def _score_z(ctx, z):
    p = _probs_from_z(z, ctx.N)
    s = _score_p(p, ctx.N, ctx.fft_self)
    _update_best(ctx, z, s)
    return s


def _obj_and_grad(ctx, x, alpha):
    if time.monotonic() >= ctx.deadline:
        raise SearchTimeout

    x = np.clip(x, -50, 50)
    p = _probs_from_z(x, ctx.N)
    c = _self_conv(p, ctx.N, ctx.fft_self)
    cmax = float(np.max(c))
    _update_best(ctx, x, 2.0 * ctx.N * cmax)

    u = alpha * (c - cmax)
    u = np.clip(u, -500, 0)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s

    gp = _grad_p_from_weights(p, w, ctx.N, ctx.fft_grad)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return float(J), gz


def _lbfgs_run(ctx, z0, alpha, max_fun):
    if minimize is None:
        return
    ctx.deadline = ctx.main_deadline
    x0 = np.clip(z0, -50, 50)
    try:
        minimize(_obj_and_grad, x0, args=(alpha,), method='L-BFGS-B',
                 jac=True, bounds=ctx.bounds,
                 options={'maxfun': max_fun, 'maxiter': max_fun // 3,
                          'ftol': 1e-14, 'gtol': 1e-9, 'maxls': 20, 'disp': False})
    except SearchTimeout:
        pass
    except Exception:
        pass


def _adaptive_alpha(score):
    if not np.isfinite(score):
        return 1e5
    if score > 1.9:
        return 1e5
    if score > 1.7:
        return 4e5
    if score > 1.55:
        return 1.5e6
    if score > 1.51:
        return 5e6
    return 1.2e7


def _multi_scale_optimize(ctx, z0, max_fun_total, n_scales=3):
    """Run L-BFGS with increasing alpha (multi-scale)."""
    scales = [2e5, 1e6, 5e6, 1.5e7, 3e7]
    # Pick a subset of scales based on n_scales
    if n_scales == 1:
        alphas = [scales[-1]]
    elif n_scales == 2:
        alphas = [scales[len(scales)//2], scales[-1]]
    else:
        alphas = scales[:n_scales] if n_scales <= len(scales) else scales

    for i, a in enumerate(alphas):
        if time.monotonic() >= ctx.main_deadline:
            break
        fun_alloc = max(80, max_fun_total // len(alphas))
        _lbfgs_run(ctx, z0, a, fun_alloc)
        if ctx.best_z is not None:
            z0 = ctx.best_z


def _subgradient_polish(ctx, z0, deadline, rng):
    """Subgradient method on active peaks with momentum."""
    N = ctx.N
    z = np.clip(z0, -50, 50)
    p = _probs_from_z(z, N)
    score = _score_p(p, N, ctx.fft_self)
    _update_best(ctx, z, score)

    noise = 0.1
    fails = 0
    momentum = np.zeros(N - 1)
    mu = 0.7  # momentum coefficient

    while time.monotonic() < deadline:
        c = _self_conv(p, N, ctx.fft_self)
        cmax = float(np.max(c))

        # Focus on top peaks
        topk = min(16, 2 * N - 1)
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros(2 * N - 1)
        wv = c[idx] + 1e-15
        w[idx] = wv / float(np.sum(wv))

        gp = _grad_p_from_weights(p, w, N, ctx.fft_grad)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        accepted = False
        if gnorm > 1e-15:
            direction = gz / gnorm
            momentum = mu * momentum + direction
            mnorm = float(np.linalg.norm(momentum))
            if mnorm > 0:
                direction = momentum / mnorm
            for step in (0.4, 0.2, 0.1, 0.05, 0.025):
                zt = np.clip(z - step * direction, -50, 50)
                pt = _probs_from_z(zt, N)
                st = _score_p(pt, N, ctx.fft_self)
                if st < score - 1e-13:
                    z, p, score = zt, pt, st
                    _update_best(ctx, z, score)
                    accepted = True
                    break

        if not accepted:
            # Try noisy perturbations
            best_local = score
            best_local_z = z
            for _ in range(4):
                zt = np.clip(z + rng.normal(0, noise, N - 1), -50, 50)
                pt = _probs_from_z(zt, N)
                st = _score_p(pt, N, ctx.fft_self)
                if st < best_local - 1e-13:
                    best_local = st
                    best_local_z = zt
            if best_local < score - 1e-13:
                z = best_local_z
                p = _probs_from_z(z, N)
                score = best_local
                _update_best(ctx, z, score)
                accepted = True

        if accepted:
            fails = 0
            noise = min(0.3, noise * 1.2)
        else:
            fails += 1
            noise = max(0.01, noise * 0.85)
            if fails > 150:
                break


def _make_candidates(ctx, rng, budget):
    """Generate diverse candidate logits vectors for length N."""
    N = ctx.N

    for beta in (0, 2, 4, 6, 8, 10, 12, 15, 20):

    # Logistic
    for b in (0.5, 1, 1.5, 2, 3, 4, 5, 7, 10):
        cands.append(_logistic_z(b, N))
        cands.append(_logistic_z(-b, N))

    # Exponential
    for b in (1, 2, 3, 5, 8):
        cands.append(_exp_z(b, N))
        cands.append(_exp_z(-b, N))

    # Power
    for a in (0.5, 1, 1.5, 2, 3):
        cands.append(_power_z(a, N, False))
        cands.append(_power_z(a, N, True))

    # Poly random
    n_poly = 30 if budget > 600 else 15
    for _ in range(n_poly):
        deg = int(rng.integers(3, 9))
        coef = rng.normal(0, 1.0, deg + 1)
        cands.append(_poly_z(coef, N))

    # Pure random
    n_rand = 20 if budget > 600 else 10
    for _ in range(n_rand):
        cands.append(_random_z(rng, N))

    # Base candidate from height_sequence_1 if available
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)

                if base.size != N:
                    old = np.linspace(0, 1, base.size)

                    base = np.interp(new, old, base)
                base = base / base.sum()
