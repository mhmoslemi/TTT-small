"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
and logit-space Adam optimization.
"""
import numpy as np
import time
from scipy.optimize import linprog

# Helper functions

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = np.size(a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]

def fft_convolve_cross(a, b):
    """Linear convolution of a and b using FFT."""
    na, nb = np.size(a), np.size(b)
    N = next_pow2(na + nb - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:na + nb - 1]

def score_internal(a):
    n = np.size(a)
    s = np.sum(a)
    if s < 1e-30:
        return np.inf
    b = fft_convolve(a)
    return 2.0 * n * np.max(b) / (s * s)

def softmax_stable(z):
    z = np.clip(z, -100, 100)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-30:
        return np.ones_like(z) / len(z)
    return e / s

def normalize_nonneg(a):
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s

def solve_lp_direction(f, M):
    """
    Solve: max sum(g) s.t. conv(f, g) <= M, g >= 0.
    Returns the optimal g (or None if infeasible).
    """
    n = len(f)
    # Build constraint matrix A: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Shape: (2n-1) x n
    # A[k, j] = f[k-j] if max(0, k-n+1) <= j <= min(k, n-1), i.e., 0 <= k-j < n
    # So A[k, j] = f[k-j] if j <= k and k-j < n, i.e., k-n+1 <= j <= k and j >= 0 and j < n
    
    # For efficiency, build A row by row using the structure
    # Row k: A[k, :] = [0, ..., f[k], f[k-1], ..., f[max(0,k-n+1)], 0, ...]
    # The non-zero part is at indices max(0, k-n+1) to min(k, n-1)
    
    # Actually, let's just build it directly
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = f[k - j_end : k - j_start + 1][::-1]
    
    # Maximize c^T g where c = ones(n)
    # Subject to: A g <= M * ones(2n-1), g >= 0
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = M * np.ones(2 * n - 1)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None

def lp_direction_iterate(f, t_lp, max_iters, deadline):
    """
    Iterate the LP direction method.
    f: current sequence (normalized, sum=1)
    t_lp: step size for LP direction
    Returns best f found.
    """
    n = len(f)
    f = f.copy()
    
    b = fft_convolve(f)
    M = np.max(b)
    best_M = M
    best_f = f.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = solve_lp_direction(f, M)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-15:
            break
        
        g = g / g_sum  # normalize to sum 1
        
        # Update
        f_new = (1 - t_lp) * f + t_lp * g
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new = f_new / s
        
        b_new = fft_convolve(f_new)
        M_new = np.max(b_new)
        
        if M_new < best_M * (1 - 1e-10):
            best_M = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                t_lp = t_lp * 0.95
        
        f = f_new
        M = M_new
    
    return best_f, best_M

def logit_adam_optimize(p0, deadline, rng, lr_init=0.1, beta_init=None):
    """
    Full logit-space Adam optimization.
    p0: initial nonnegative sequence (will be normalized to sum 1)
    Returns (best_p, best_score_normalized) where score = 2n*max(b) (with sum=1).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-15)
    p0 = p0 / p0.sum()
    
    z = np.log(p0)
    z = z - np.mean(z)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    p = p0.copy()
    c = fft_convolve(p)
    maxc = np.max(c)
    if beta_init is None:
        beta = 0.1 * maxc
    else:
        beta = beta_init
    
    best_p = p0.copy()
    best_score = maxc  # with sum=1, score = 2n*maxc, we just track maxc
    no_imp = 0
    lr = lr_init
    base_lr = lr
    
    while time.monotonic() < deadline:
        # Soft-max weights
        z_sm = c / beta
        z_sm = z_sm - np.max(z_sm)
        e = np.exp(np.clip(z_sm, -50, 50))
        w = e / e.sum()
        
        # Gradient of sum_k w_k * b_k w.r.t. p
        # db_k/dp_j = 2*p_{k-j} if 0 <= k-j < n
        # g_p[j] = 2 * sum_k w_k * p_{k-j} (valid)
        # = 2 * (cross_correlation of p and w)
        
        # Compute g_p efficiently
        N = next_pow2(3 * n)
        fp = np.fft.rfft(p, N)
        fw = np.fft.rfft(w, N)
        cross = np.fft.irfft(fp * np.conj(fw), N)
        g_p = 2.0 * cross[:n]
        
        # Gradient w.r.t. z (log-space)
        dot = np.dot(p, g_p)
        g_z = p * (g_p - dot)
        
        # Adam step
        if not np.all(np.isfinite(g_z)):
            z = np.log(best_p)
            z = z - np.mean(z)
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * g_z
        v = b2 * v + (1 - b2) * g_z * g_z
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        z = np.clip(z, -50, 50)
        z = z - np.mean(z)
        p = softmax_stable(z)
        
        c = fft_convolve(p)
        maxc = np.max(c)
        
        if maxc < best_score - 1e-14:
            best_score = maxc
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.01, lr * 0.98)
        
        # Anneal beta
        if t_adam % 50 == 0:
            beta = max(1e-6, beta * 0.85)
        
        # Restart
        if no_imp > 300:
            z = np.log(best_p) + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - np.mean(z)
            p = softmax_stable(z)
            c = fft_convolve(p)
            beta = max(1e-4, 0.1 * np.max(c))
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score
