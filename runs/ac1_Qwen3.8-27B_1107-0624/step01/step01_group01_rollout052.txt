def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])

def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] for the cross-correlation gradient."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def subgradient_step(a, lr, beta, reg=1e-7):
    """One step of subgradient descent on the normalized sequence."""
    n = a.size
    c = full_convolve_fft(a)
    _, w = soft_max_weights(c, beta)
    g = 2.0 * corr_weights(w, a)
    # Add regularization to prevent all-mass concentration
    g = g + reg * a
    # Project gradient to maintain sum constraint (remove mean component)
    g = g - float(np.sum(g) * a) / float(np.sum(a) ** 2)
    # Actually, simpler: just do multiplicative update
    a_new = a * np.exp(-lr * g / (a.mean() + 1e-10))

    return a_new / s

def softmax_from_z(z):

def param_space_adam(n, deadline, rng, dims=15):

However, we could use a **semidefinite relaxation**. The constraint `sum_i a[i]*a[k-i] <= M` can be written as a constraint on the outer product `a * a^T`. Specifically, `sum_i a[i]*a[k-i] = sum_{i+j=k} a[i]*a[j] = <A_k, a*a^T>` where `A_k` is a matrix with 1s at positions `(i, k-i)`.

The SDP relaxation would replace `a*a^T` with a PSD matrix `X` and add the constraint that the diagonal of X equals `a^2`... but that's still non-convex.

A simpler relaxation: use `X = a*a^T` and note that `Tr(X) = ||a||^2 <= (sum a)^2 = 1` (by Cauchy-Schwarz). So we're bounding `Tr(X)`.

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines soft-max subgradient descent, parameter-space Adam, scipy L-BFGS-B,
simulated annealing restarts, and multi-length search.

import numpy as np
import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60, 0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50, 50)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / e.size
    else:
        p = p / s
    return p

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cosine_profile(n, freq=1.0, amp=0.8, phase=0.0, base=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = base + amp * np.cos(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_det_profiles(n):
    profs = []
    for eta in (1.0, 1.5, 2.0, 2.5, 3.0):
        profs.append(make_linear_profile(n, eta))
    for freq in (1, 2, 3, 4, 5):
        for amp in (0.4, 0.7, 1.0, 1.3):
            profs.append(make_cosine_profile(n, freq, amp))
    t = np.linspace(0, 1, n)
    for rate in (0.5, 1.0, 2.0):
        a = np.exp(rate * t)
        profs.append(a / a.sum())
        a = np.exp(-rate * t)
        profs.append(a / a.sum())
    for center in (0.3, 0.25, 0.35):
        a = np.exp(-50*(t-center)**2) + np.exp(-50*(t-(1-center))**2) + 0.1
        profs.append(a / a.sum())
    # Symmetric profiles
    for amp in (0.5, 0.8, 1.0):
        a = 1.0 + amp * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    # Anti-symmetric-ish
    for amp in (0.5, 0.8, 1.0):
        a = 1.0 + amp * np.sin(np.pi * t)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    return profs

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))

def wgd_optimize(a0, deadline, rng, lr_init=0.5, beta_init=None,
                 beta_decay=0.94, lr_decay=0.999, restart_patience=200,
                 reg=1e-6, sa_temperature=0.0):
    """
    Weighted gradient descent with soft-max approximation,
    adaptive parameters, and optional simulated annealing acceptance.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    c_init = full_convolve_fft(a)
    if beta_init is None:
        beta_init = 0.08 * float(c_init.max())
    beta = max(1e-12, float(beta_init))
    lr = lr_init
    best_a = a.copy()
    best_score = internal_score(a)
    no_imp = 0
    iter_count = 0
    sa_temp = sa_temperature

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        _, w = soft_max_weights(c, beta)
        g = 2.0 * corr_weights(w, a)
        g = g + reg * a
        # Project to tangent space of simplex (remove component along a)
        a_sum = float(a.sum())
        g_proj = g - (float(np.dot(a, g)) / (a_sum * a_sum)) * a
        a_new = a * np.exp(-lr * g_proj)
        a_new = np.maximum(a_new, 1e-15)
        s_new = float(a_new.sum())
        if not np.isfinite(s_new) or s_new < 1e-30:
            a = best_a.copy()
            beta *= 2.0
            continue
        a_new = a_new / s_new
        sc = internal_score(a_new)

        if not np.isfinite(sc):
            a = best_a.copy()
            beta *= 2.0
            continue

        # Simulated annealing: occasionally accept worse solutions
        accept = False
        if sc < best_score - 1e-13:
            accept = True
        elif sa_temp > 1e-15 and np.isfinite(sc):
            # Accept with probability exp(-(sc - best_score) / sa_temp)
            delta = sc - best_score
            if delta < sa_temp * 10:
                prob = np.exp(-delta / sa_temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                no_imp = 0
            a = a_new
            sa_temp = max(1e-15, sa_temp * 0.99)
        else:
            no_imp += 1
            a = a_new
            if no_imp % 200 == 0:
                lr = max(0.005, lr * 0.93)
                sa_temp = max(1e-15, sa_temp * 0.9)

        iter_count += 1

        if iter_count % 40 == 0:
            beta = max(1e-14, beta * beta_decay)
        if iter_count % 300 == 0:
            lr = max(0.005, lr * lr_decay)

        if no_imp > restart_patience:
            pert_type = rng.integers(0, 6)
            if pert_type == 0:
                freq = rng.uniform(0.5, 4.0)
                phase = rng.uniform(0, 2 * np.pi)
                t = np.linspace(0, 1, n)
                pert = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + phase)
                a = best_a * pert
            elif pert_type == 1:
                t = np.linspace(0, 1, n)
                c1 = rng.uniform(-0.5, 0.5)
                c2 = rng.uniform(-0.3, 0.3)
                pert = 1.0 + c1 * t + c2 * t * t
                a = best_a * pert
            elif pert_type == 2:
                a = best_a * (1.0 + 0.15 * rng.normal(size=n))
            elif pert_type == 3:
                # Mirror perturbation
                a = best_a.copy()
                half = n // 2
                a[half:] = best_a[:half] * (1.0 + 0.2 * rng.normal(size=n - half))
            elif pert_type == 4:
                # Block swap
                a = best_a.copy()
                bl = max(1, n // 8)
                start = rng.integers(0, n - bl)
                a[start:start+bl] = best_a[start:start+bl] * 0.5
            else:
                # Frequency-domain perturbation
                fa = np.fft.rfft(best_a)
                fa *= (1.0 + 0.1 * rng.normal(size=fa.size) + 0.1j * rng.normal(size=fa.size))
                a = np.fft.irfft(fa, n)

            a = np.maximum(a, 1e-15)
            s = float(a.sum())
            if s > 1e-30:
                a = a / s
            else:
                a = best_a.copy()
            c_check = full_convolve_fft(a)
            beta = max(0.05 * float(c_check.max()), 1e-12)
            lr = lr_init
            no_imp = 0
            sa_temp = max(1e-15, sa_temperature)

    return best_a, best_score

def param_space_adam(n, deadline, rng, dims=16):
    """Adam in low-dimensional parameter space with rich basis."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    cols = [u]
    m = 1
    while len(cols) < dims:
        cols.append(np.cos(2 * np.pi * m * t))
        cols.append(np.sin(2 * np.pi * m * t))
        m += 1
        if len(cols) >= dims:
            break
    if len(cols) < dims:
        cols.append(u ** 2)
    if len(cols) < dims:
        cols.append(u ** 3)
    if len(cols) < dims:
        cols.append(u ** 4)

    basis = np.column_stack(cols[:dims])
    basis_dim = basis.shape[1]

    best_a = None
    best_score = np.inf
    n_restarts = 10
    elapsed = time.monotonic()
    total_time = deadline - elapsed
    if total_time <= 0:
        return None, np.inf
    time_per = total_time / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > deadline - 1.0:
            break
        t_end = time.monotonic() + max(1.0, time_per - 0.5)

        if r == 0:
            theta = np.zeros(basis_dim)
            theta[0] = 1.8
        elif r == 1:
            theta = np.zeros(basis_dim)
            theta[0] = 2.5
            if basis_dim > 2:
                theta[2] = -2.0
        elif r == 2:
            theta = np.zeros(basis_dim)
            theta[1] = 1.5
            theta[0] = 1.0
        elif r == 3:
            theta = np.zeros(basis_dim)
            theta[0] = 1.5
            if basis_dim > 4:
                theta[4] = 1.0
                theta[5] = -0.5
        else:
            theta = rng.uniform(-4, 4, basis_dim)

        b1, b2, eps = 0.9, 0.999, 1e-8
        m_adam = np.zeros(basis_dim)
        v_adam = np.zeros(basis_dim)
        adam_t = 0
        lr = 0.3
        base_lr = lr

        z = basis @ theta
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        beta = max(1e-12, 0.08 * float(c.max()))

        best_theta = theta.copy()
        best_p_local = p.copy()
        best_sc_local = float(2.0 * n * float(c.max()))
        no_imp = 0

        while time.monotonic() < t_end:
            c = full_convolve_fft(p)
            _, w = soft_max_weights(c, beta)
            g = 2.0 * corr_weights(w, p)
            g = g + 1e-6 * p
            gz = p * (g - float(np.dot(p, g)))
            gtheta = basis.T @ gz

            if not np.all(np.isfinite(gtheta)):
                theta = best_theta.copy()
                m_adam.fill(0)
                v_adam.fill(0)
                adam_t = 0
                continue

            adam_t += 1
            m_adam = b1 * m_adam + (1 - b1) * gtheta
            v_adam = b2 * v_adam + (1 - b2) * gtheta ** 2
            mh = m_adam / (1 - b1 ** adam_t)
            vh = v_adam / (1 - b2 ** adam_t)
            theta = theta - lr * mh / (np.sqrt(vh) + eps)
            theta = np.clip(theta, -10, 10)

            z = basis @ theta
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            sc = float(2.0 * n * float(c.max()))

            if np.isfinite(sc):
                if sc < best_sc_local - 1e-13:
                    best_sc_local = sc
                    best_p_local = p.copy()
                    best_theta = theta.copy()
                    no_imp = 0
                else:
                    no_imp += 1
                    if no_imp % 150 == 0:
                        lr = max(0.02, lr * 0.94)
            else:
                theta = best_theta.copy()
                m_adam.fill(0)
                v_adam.fill(0)
                adam_t = 0
                continue

            if adam_t % 50 == 0:
                beta = max(1e-14, beta * 0.82)

            if no_imp > 250:
                theta = best_theta + 0.6 * rng.normal(size=basis_dim)
                theta = np.clip(theta, -10, 10)
                m_adam.fill(0)
                v_adam.fill(0)
                adam_t = 0
                lr = base_lr
                no_imp = 0
                beta = max(1e-12, 0.08 * float(full_convolve_fft(softmax_from_z(basis @ theta)).max()))

        if best_sc_local < best_score:
            best_score = best_sc_local
            best_a = best_p_local.copy()

    return best_a, best_score

def logit_adam(a0, deadline, rng, lr=0.05, reg=1e-6, beta_factor=0.08):
    """Adam in full logit space with aggressive soft-max annealing."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-12)
    a0 = a0 / a0.sum()
    n = a0.size

    z = np.log(a0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = full_convolve_fft(a0)
    beta = max(1e-12, beta_factor * float(c.max()))

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * float(c.max()))
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        a_curr = softmax_from_z(z)
        c = full_convolve_fft(a_curr)
        _, w = soft_max_weights(c, beta)
        g = 2.0 * corr_weights(w, a_curr)
        g = g + reg * a_curr
        gz = a_curr * (g - float(np.dot(a_curr, g)))
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        z = np.clip(z, -50, 50)
        z -= float(np.mean(z))
        a_curr = softmax_from_z(z)
        c = full_convolve_fft(a_curr)
        sc = float(2.0 * n * float(c.max()))

        if np.isfinite(sc):
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_curr.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 250 == 0:
                    lr = max(0.01, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        if adam_t % 50 == 0:
            beta = max(1e-14, beta * 0.80)

        if no_imp > max(350, 3 * n):
            z = 0.85 * best_z + 0.25 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-12, beta_factor * float(full_convolve_fft(softmax_from_z(z)).max()))

    return best_a, best_score

def scipy_lbfgsb(a0, deadline, rng):
    """scipy L-BFGS-B with subgradient (active set) approximation."""
    from scipy.optimize import minimize

    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-10)
    a0 = a0 / a0.sum()
    n = a0.size

    def objective(x):
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        c = full_convolve_fft(x)
        return float(2.0 * n * float(c.max()))

    def gradient(x):
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        c = full_convolve_fft(x)
        mval = float(c.max())
        # Use soft max with small beta for smoother gradient
        _, w = soft_max_weights(c, 0.01 * mval)
        g = 2.0 * corr_weights(w, x)
        g = g - float(np.dot(x, g))
        return g

    best_a = a0.copy()
    best_score = objective(a0)
    remaining = deadline - time.monotonic()
    if remaining < 5.0:
        return best_a, best_score

    n_restarts = 3
    chunk = remaining / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > deadline - 3.0:
            break
        if r == 0:
            x0 = a0.copy()
        else:
            x0 = best_a * (1.0 + 0.05 * rng.normal(size=n))
            x0 = np.maximum(x0, 1e-10)
            x0 = x0 / x0.sum()

        t_end = time.monotonic() + max(2.0, chunk - 1.0)
        done = [False]

        def callback(xk):
            if time.monotonic() > t_end:
                done[0] = True
            return done[0]

        try:
            result = minimize(objective, x0, method='L-BFGS-B',
                              jac=gradient, callback=callback,
                              options={'maxiter': 20000, 'maxfun': 100000,
                                       'ftol': 1e-16, 'gtol': 1e-12,
                                       'maxls': 50})
            if result.x is not None:
                rx = np.maximum(result.x, 1e-10)
                rx = rx / rx.sum()
                sc = objective(rx)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = rx.copy()
        except Exception:
            pass
        a0 = best_a.copy()

    return best_a, best_score

def local_swap_search(a0, deadline, rng, max_iterations=5000):
    """Local search by trying small perturbations (swaps, rescalings)."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    best_a = a.copy()
    best_score = internal_score(a)
    no_imp = 0

    for _ in range(max_iterations):
        if time.monotonic() > deadline:
            break
        pert_type = rng.integers(0, 4)

        if pert_type == 0:
            # Scale a random block
            bl = max(1, n // 10)
            start = rng.integers(0, n - bl)
            factor = rng.uniform(0.5, 2.0)
            a_try = a.copy()
            a_try[start:start + bl] *= factor
        elif pert_type == 1:
            # Swap two elements
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            a_try = a.copy()
            a_try[i], a_try[j] = a_try[j], a_try[i]
        elif pert_type == 2:
            # Gaussian bump
            center = rng.integers(0, n)
            width = max(2, n // 10)
            amp = rng.uniform(-0.3, 0.3)
            t = np.arange(n)
            a_try = a * np.exp(amp * np.exp(-0.5 * ((t - center) / width) ** 2))
        else:
            # Small random rescale
            a_try = a * (1.0 + 0.05 * rng.normal(size=n))

        a_try = np.maximum(a_try, 1e-15)
        s = float(a_try.sum())
        if s < 1e-30:
            continue
        a_try = a_try / s
        sc = internal_score(a_try)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a_try.copy()
            a = a_try.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 500:
                # Larger perturbation
                a = best_a * (1.0 + 0.2 * rng.normal(size=n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                no_imp = 0

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search:
    Phase 1: Deterministic profiles at multiple lengths
    Phase 2: Low-dimensional parameter Adam
    Phase 3: WGD with simulated annealing
    Phase 4: Full logit Adam
    Phase 5: scipy L-BFGS-B
    Phase 6: Local swap search
    Phase 7: Extended lengths + final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    all_candidates = []

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {sc:.10f} (n={p.size}) {label}")
        if p is not None and np.isfinite(sc):
            all_candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
            all_candidates.sort(key=sort_key_func)
            if len(all_candidates) > 40:
                all_candidates = all_candidates[:40]

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            parent = normalize_nonneg(p)
            if parent is not None:
                sc = internal_score(parent)
                update_best(parent, sc, "parent")
    except Exception:
        pass

    print(f"Initial best: {best_score:.10f}")

    # Phase 1: Deterministic profiles
    print("Phase 1: Deterministic profiles")
    t_phase1_end = min(deadline, t_start + 120.0)
    lengths = [128, 192, 256, 384, 512]
    for n_len in lengths:
        if time.monotonic() > t_phase1_end:
            break
        for p0 in make_det_profiles(n_len):
            if time.monotonic() > t_phase1_end:
                break
            sc = internal_score(p0)
            update_best(p0, sc, f"det n={n_len}")

    # Parent interpolated
    if parent is not None:
        for n_len in lengths:
            if time.monotonic() > t_phase1_end:
                break
            p_int = interpolate_profile(parent, n_len)
            if p_int is not None:
                sc = internal_score(p_int)
                update_best(p_int, sc, f"par_int n={n_len}")

    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Parameter space Adam
    print("Phase 2: Param space Adam")
    t_phase2_end = min(deadline, t_start + 420.0)
    for n_len in [128, 256, 512]:
        if time.monotonic() > t_phase2_end:
            break
        t_end = min(t_phase2_end, time.monotonic() + 90.0)
        p_best, sc = param_space_adam(n_len, t_end, rng, dims=16)
        if p_best is not None:
            update_best(p_best, sc, f"param n={n_len}")
        # Also try with more dims
        if time.monotonic() < t_phase2_end:
            t_end2 = min(t_phase2_end, time.monotonic() + 60.0)
            p_best2, sc2 = param_space_adam(n_len, t_end2, rng, dims=20)
            if p_best2 is not None:
                update_best(p_best2, sc2, f"param2 n={n_len}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: WGD with SA
    print("Phase 3: WGD + SA")
    t_phase3_end = min(deadline, t_start + 900.0)
    for n_len in [128, 256, 512]:
        if time.monotonic() > t_phase3_end:
            break
        # Find best at this length
        best_at_n = None
        for sc_c, p_c in all_candidates:
            if p_c.size == n_len:
                best_at_n = p_c
                break
        if best_at_n is None and parent is not None:
            best_at_n = interpolate_profile(parent, n_len)
        if best_at_n is None:
            best_at_n = make_linear_profile(n_len, 1.5)

        # Multiple WGD runs with different params
        for trial in range(3):
            if time.monotonic() > t_phase3_end:
                break
            lr_val = 0.4 * (0.7 ** trial)
            sa_temp = 0.002 * (0.5 ** trial) if trial > 0
