def _softmax(b):
    b_shifted = b - b.max()
    e = np.exp(b_shifted)
    return e / e.sum()

def _conv_sq(a):
    n = len(a)
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

    """Compute the (unofficial) objective value: 2*n*max(c) where c = conv(a,a), assuming sum(a)=1."""
    c = _conv_sq(a)
    return 2.0 * len(a) * c.max()

def _grad_smoothed_max(b, tau, L):
    """Compute gradient of smoothed max objective w.r.t. b (softmax parameterization)."""
    n = len(b)
    a = _softmax(b)
    c = _conv_sq(a)
    
    # Top-L selection
    L = min(L, len(c))
    topL_idx = np.argpartition(-c, L-1)[:L]
    
    # Softmax weights over top-L
    c_top = c[topL_idx]
    w_top = np.exp(tau * c_top)
    w_top = w_top / w_top.sum()

    w = np.zeros_like(c)
    w[topL_idx] = w_top

    h = np.correlate(w, a, mode='full')[n-1:]

    grad = 2.0 * a * (h - C)
    
    return grad, c.max()

def _adam_update(state, grad, lr, t):
    """One step of Adam optimizer."""
    m = state['m']
    v = state['v']

    beta2 = 0.999

    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**(t+1))
    v_hat = v / (1 - beta2**(t+1))
    
    state['m'] = m
    state['v'] = v
    return m_hat / (np.sqrt(v_hat) + eps)

        tau = tau_start + (tau_end - tau_start) * (1 + math.cos(math.pi * progress)) / 2
        lr = lr_start * (1 + math.cos(math.pi * progress)) / 2 + 1e-6

Looking at this problem, I need to minimize `2*n*max(convolve(a,a))/(sum(a))^2` for non-negative sequences. The key insight is that this is scale-invariant, so I can fix `sum(a)=1` and minimize `2*n*max(conv(a,a))`.

My approach differs from the previous program by:
1. **Softmax parameterization** (b → a = softmax(b)) for unconstrained optimization
2. **Lp-norm objective** with annealing (smoother gradients than top-L smoothed max)
3. **Adam optimizer** with proper gradient via FFT-based cross-correlation
4. **Diverse restarts** including perturbed versions of the known good solution
5. **Coordinate descent polishing** phase at the end

The gradient derivation: with `a = softmax(b)`, `c = conv(a,a)`, and objective `J = (Σc[k]^p)^(1/p)`, the gradient is `∂J/∂b[m] = 2·s·a[m]·(h[m] - C)` where `h[m] = Σ_k v[k]·a[k-m]`, `C = Σ_k v[k]·c[k]`, `v[k] = c[k]^(p-1)/Σc[j]^(p-1)`, and `s = (Σc^p)^(1/p-1)`.

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 via softmax parameterization with
Adam optimization of the Lp-norm of the convolution (annealed p), followed
by coordinate-descent polishing. Multiple diverse restarts.
Gradient: dJ/db[m] = 2*s*a[m]*(h[m]-C) where h=correlate(v,a), v~c^(p-1).
"""

import time
import math
import numpy as np


def _softmax(b):
    b_shifted = b - b.max()
    e = np.exp(b_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(b), 1.0 / len(b))
    return e / s


def _conv_sq(a):
    n = len(a)
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _objective(a):
    n = len(a)
    c = _conv_sq(a)
    return 2.0 * n * c.max()


def _grad_Lp(b, p):
    n = len(b)
    a = _softmax(b)
    c = _conv_sq(a)
    c = np.maximum(c, 1e-30)
    
    powers = c ** (p - 1)
    s_p = powers.sum()
    if s_p < 1e-30:
        return np.zeros(n), 0.0
    v = powers / s_p
    
    # h[m] = sum_k v[k] * a[k-m] for 0<=k-m<n
    # This is cross-correlation of v and a
    h = np.correlate(v, a, mode='full')[n - 1:]
    
    C = np.dot(v, c)
    
    s = (np.sum(c ** p)) ** (1.0 / p - 1.0)
    
    grad = 2.0 * s * a * (h - C)
    obj = (np.sum(c ** p)) ** (1.0 / p)
    return grad, obj


def _adam_step(state, grad, lr, t):
    m = state['m']
    v = state['v']
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad ** 2
    m_hat = m / (1.0 - b1 ** (t + 1))
    v_hat = v / (1.0 - b2 ** (t + 1))
    state['m'] = m
    state['v'] = v
    return m_hat / (np.sqrt(v_hat) + eps)


def _optimize(b_init, t_end, p_start, p_end, lr0, n_target_steps, rng):
    n = len(b_init)
    b = b_init.copy().astype(np.float64)
    state = {'m': np.zeros(n), 'v': np.zeros(n)}
    
    a0 = _softmax(b)
    best_obj = _objective(a0)
    best_b = b.copy()
    t = 0
    stall_count = 0
    
    while t < n_target_steps and time.monotonic() < t_end - 2.0:
        progress = t / max(1, n_target_steps - 1)
        # Anneal p from p_start to p_end (higher p = sharper max approximation)
        p = p_start + (p_end - p_start) * progress
        # Cosine-anneal learning rate
        lr = lr0 * 0.5 * (1.0 + math.cos(math.pi * progress)) + 1e-7
        
        grad, obj = _grad_Lp(b, p)
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_b = b.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # If stalled, add perturbation
        if stall_count > 200:
            b += rng.normal(0, 0.05, n)
            stall_count = 0
        
        update = _adam_step(state, grad, lr, t)
        b = b - lr * update
        t += 1
    
    return best_b, best_obj


def _coordinate_descent(a, n_inner, t_end, rng):
    """Polish via coordinate descent on the original a (simplex)."""
    n = len(a)
    a = a.copy()
    best_obj = _objective(a)
    
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    
    for outer in range(n_inner):
        if time.monotonic() > t_end - 2.0:
            break
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        
        for idx in order:
            if time.monotonic() > t_end - 2.0:
                break
            best_a = a.copy()
            best_local = best_obj
            # Try several perturbations of this coordinate
            for delta in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                trial = a.copy()
                amt = trial[idx] + delta
                if amt < 0 or amt > 1.0:
                    continue
                # Redistribute to maintain sum=1
                remaining = trial.sum() - trial[idx]
                if remaining <= 0:
                    continue
                diff = amt - trial[idx]
                trial = trial - diff * (trial / remaining)
                trial[idx] = amt
                trial = np.maximum(trial, 0)
                s = trial.sum()
                if s < 1e-10:
                    continue
                trial /= s
                
                c = _conv_sq(trial)
                local_obj = 2.0 * n * c.max()
                if local_obj < best_local - 1e-9:
                    best_local = local_obj
                    best_a = trial.copy()
                    improved = True
            
            if improved:
                a = best_a
                best_obj = best_local
    
    return a, best_obj
