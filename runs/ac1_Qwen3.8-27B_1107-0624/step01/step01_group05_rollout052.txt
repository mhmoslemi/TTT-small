    """Solve the restricted LP and return the optimal g."""
    c = self_conv(p)
    # Top K active positions
    idx = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix: (p*g)[k] = sum_i p[i] * g[k-i]
    # For each k in idx, the constraint is: sum_i p[i] * g[k-i] <= M
    # This is a linear constraint on g.
    
    # The constraint for position k: sum_{i: 0<=i<n, 0<=k-i<n} p[i] * g[k-i] <= M
    # Let j = k-i, then j ranges from max(0, k-n+1) to min(k, n-1)
    # So: sum_j p[k-j] * g[j] <= M where p[k-j] is defined for valid j
    
    # This is: A[k,:] @ g <= M where A[k,j] = p[k-j] if 0 <= k-j < n else 0
    
    # Build A (K x n matrix)
    A = np.zeros((K, len(p)))
    for row, k in enumerate(idx):
        for j in range(len(p)):
            i = k - j
            if 0 <= i < len(p):
                A[row, j] = p[i]
    
    # LP: max 1^T g s.t. A g <= M, g >= 0
    # In linprog form: min -1^T g s.t. A g <= M, g >= 0
    from scipy.optimize import linprog
    res = linprog(-np.ones(len(p)), A_ub=A, b_ub=np.full(K, M), bounds=[(0, None)]*len(p), method='highs')
    
    if res.success:

    return None

    # A[row, j] = p[k - j] if 0 <= k-j < N else 0
    A = np.zeros((K, N))

        # j ranges from max(0, k-N+1) to min(k, N-1)

        j_end = min(k, N - 1)

            A[row, j_start:j_end+1] = p[i_vals]

    c_obj = -np.ones(N)

        if sum_g < 1e-10:
            return None
        # Normalize g to have same sum as p (=1)
        g = g / sum_g
        return g
    except Exception:
        return None

        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros(2 * N - 1)
        vals = c[idx] + 1e-15
        w[idx] = vals / float(np.sum(vals))

        gp = _grad_p_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        grad = gp - sdot
        grad = grad / float(np.linalg.norm(grad) + 1e-30)

        accepted = False
        for step in (0.04, 0.02, 0.01, 0.005, 0.002):
            pt = p - step * grad
            pt = np.maximum(pt, 0.0)
            ss = float(np.sum(pt))
            if ss < 1e-12:
                continue
            pt = pt / ss
            s = _score_p(pt)
            if s < cur - 1e-14:
                p = pt
                cur = _update_best(p)
                accepted = True
                break

        if not accepted:
            best_local = cur
            best_local_p = p
            for _ in range(4):
                z0 = _density_to_z(p)
                zt = z0 + rng.normal(0.0, noise, N - 1)
                pt = _probs_from_z(zt)
                s = _score_p(pt)
                if s < best_local:
                    best_local = s
                    best_local_p = pt.copy()
            if best_local < cur - 1e-14:
                p = best_local_p
                cur = _update_best(p)
                accepted = True

        if accepted:
            fails = 0
            noise = min(0.3, noise * 1.3)
        else:
            fails += 1
            noise = max(0.01, noise * 0.75)
            if fails > 80:
                break

    return p


def _symmetrize(p):
    """Average p with its reverse to enforce palindromic symmetry."""
    return 0.5 * (p + p[::-1])


def _make_structured_candidates():
    """Generate diverse structured starting points."""
    x = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
    cands = []

    # Uniform
    cands.append(np.ones(N) / N)

    # Cosine windows (various shapes)
    for period in (0.5, 0.7, 1.0, 1.3, 1.5, 2.0):
        f = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5) / period))
        f = np.maximum(f, 1e-10)
        cands.append(f / float(np.sum(f)))

    # Raised cosine squared
    for width in (0.3, 0.5, 0.7, 1.0):
        u = (x - 0.5) / width
        f = np.cos(np.pi * u / 2.0)**4
        f = np.where(np.abs(u) <= 1.0, f, 0.0)
        f = np.maximum(f, 1e-12)
        cands.append(f / float(np.sum(f)))

    # Exponential (various rates)
    for beta in (1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        f = np.exp(-beta * (x - 0.5)**2 / 0.05)
        f = np.maximum(f, 1e-12)
        cands.append(f / float(np.sum(f)))

    # Logistic transitions
    for beta in (2.0, 5.0, 10.0, 20.0, 40.0):
        f = 1.0 / (1.0 + np.exp(-beta * (x - 0.5)))
        f = np.maximum(f, 1e-12)
        cands.append(f / float(np.sum(f)))
        f2 = 1.0 / (1.0 + np.exp(beta * (x - 0.5)))
        f2 = np.maximum(f2, 1e-12)
        cands.append(f2 / float(np.sum(f2)))

    # Power laws
    for a in (0.5, 1.0, 1.5, 2.0, 3.0):
        f = (x + 1e-6)**a
        f = np.maximum(f, 1e-12)
        cands.append(f / float(np.sum(f)))
        f2 = ((1.0 - x) + 1e-6)**a
        f2 = np.maximum(f2, 1e-12)
        cands.append(f2 / float(np.sum(f2)))

    # "Tent" shapes
    for width in (0.2, 0.3, 0.5, 0.8):
        f = np.maximum(0.0, 1.0 - np.abs(x - 0.5) / width)
        f = np.maximum(f, 1e-12)
        cands.append(f / float(np.sum(f)))

    # "Comb" patterns (sparse peaks)
    for period in (8, 16, 32, 64):
        f = np.ones(N) * 0.01
        for i in range(0, N, period):
            f[i] = 1.0
        f = np.maximum(f, 1e-12)
        cands.append(f / float(np.sum(f)))

    # Dual-window (two peaks)
    for sep in (0.25, 0.3, 0.4):
        f = np.exp(-40.0 * (x - 0.5 + sep / 2)**2) + np.exp(-40.0 * (x - 0.5 - sep / 2)**2)
        f = np.maximum(f, 1e-12)
        cands.append(f / float(np.sum(f)))

    return cands


def _make_random_candidates(rng, count):
    """Generate random structured candidates."""
    cands = []
    x = (np.arange(N, dtype=np.float64) + 0.5) / float(N)

    for _ in range(count):
        # Random mixture of sinusoids
        f = np.ones(N)
        n_freq = rng.integers(2, 6)
        for _ in range(n_freq):
            freq = rng.uniform(0.5, 8.0)
            amp = rng.normal(0.0, 0.8)
            phase = rng.uniform(0, 2 * np.pi)
            f += amp * np.sin(2 * np.pi * freq * x + phase)
        # Random polynomial perturbation
        deg = rng.integers(2, 6)
        t = x - 0.5
        for k in range(deg, 0, -1):
            f += rng.normal(0, 0.5) * t**k
        f = np.maximum(f, 1e-8)
        f /= float(np.sum(f))
        cands.append(f)

    for _ in range(count // 2):
        # Random log-normal
        z = rng.normal(0, 1.5, N)
        f = np.exp(z)
        f = np.maximum(f, 1e-12)
        f /= float(np.sum(f))
        cands.append(f)

    return cands


def _choose_alpha(score):
    if not np.isfinite(score):
        return 1e5
    if score > 1.85:
        return 1e5
    if score > 1.70:
        return 3e5
    if score > 1.55:
        return 1e6
    if score > 1.51:
        return 3e6
    return 1e7


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_p, _search_deadline

    start = time.monotonic()
    budget_s = float(budget_s)
    end_time = start + max(1.0, budget_s) - 3.0

    _best_score = float('inf')
    _best_p = None
    rng = np.random.default_rng(seed)

    # Try to use inherited best
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size == N:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.maximum(base, 0.0)
            ss = float(np.sum(base))
            if ss > 0.01:
                base_p = base / ss
                _update_best(base_p)
                print(f"Inherited base score: {_best_score:.6f}")
                return _finalize(end_time, rng)
        elif base.size > 0:
            new_x = np.linspace(0, 1, N)
            old_x = np.linspace(0, 1, base.size)
            base = np.interp(new_x, old_x, base)
            base = np.maximum(base, 0.0)
            ss = float(np.sum(base))
            if ss > 0.01:
                base_p = base / ss
                _update_best(base_p)
                print(f"Interpolated base score: {_best_score:.6f}")
    except Exception:
        pass

    # Generate candidates
    struct_cands = _make_structured_candidates()
    n_rand = 60
    rand_cands = _make_random_candidates(rng, n_rand)

    # Also add symmetrized versions of best
    if _best_p is not None:
        struct_cands.append(_symmetrize(_best_p))
        struct_cands.append(_best_p.copy())

    # Score all candidates
    all_cands = struct_cands + rand_cands
    scores = []
    for c in all_cands:
        c = np.maximum(c, 0.0)
        ss = float(np.sum(c))
        if ss < 0.01:
            scores.append(float('inf'))
            continue
        c = c / ss
        s = _update_best(c)
        scores.append(s)

    scores = np.array(scores)
    order = np.argsort(scores)
    top_n = min(12, len(order))
    top_indices = [int(order[i]) for i in range(top_n) if np.isfinite(scores[order[i]])]

    print(f"Best initial score: {_best_score:.6f}")
    print(f"Top candidates: {[f'{scores[int(order[i])]:.6f}' for i in range(top_n)]}")

    # Phase 2: Optimize top candidates
    main_deadline = end_time - max(30.0, 0.08 * budget_s)

    run_idx = 0
    last_best = _best_score
    stagnant = 0

    while time.monotonic() < main_deadline:
        rem = main_deadline - time.monotonic()
        if rem < 15.0:
            break

        if run_idx < len(top_indices):
            p0 = all_cands[top_indices[run_idx]].copy()
        else:
            # Generate new random perturbations of best
            if _best_p is None:
                p0 = np.ones(N) / N
            else:
                frac = max(0.0, min(1.0, rem / max(1.0, main_deadline - start)))
                noise_scale = 0.03 + 0.15 * frac
                if stagnant > 20:
                    noise_scale *= 2.0
                z0 = _density_to_z(_best_p)
                zt = z0 + rng.normal(0.0, noise_scale, N - 1)
                p0 = _probs_from_z(zt)

        run_idx += 1
        p0 = np.maximum(p0, 0.0)
        p0 = p0 / float(np.sum(p0))
        _update_best(p0)

        # L-BFGS-B phase
        z0 = _density_to_z(p0)
        alpha = _choose_alpha(_best_score)
        max_fun = int(min(800, max(100, rem / 0.012)))
        _lbfgs_optimize(z0, alpha, max_fun, main_deadline)

        # LP descent phase
        if _best_p is not None and time.monotonic() < main_deadline - 10.0:
            p_lp = _best_p.copy()
            p_lp = _lp_descent(p_lp, num_iters=min(20, int(rem / 5)), K=25, deadline=main_deadline)
            _update_best(p_lp)

        # Check progress
        if _best_score < last
best - 1e-12:
            last_best = _best_score
            stagnant = 0
            elapsed = time.monotonic() - start
            print(f"  run {run_idx}: best={_best_score:.6f} elapsed={elapsed:.0f}s")
        else:
            stagnant += 1

    # Phase 3: Final polishing
    if _best_p is not None:
        # Subgradient polish
        polish_deadline = min(end_time - 5.0, time.monotonic() + max(10.0, 0.1 * budget_s))
        _subgradient_polish(_best_p.copy(), polish_deadline, rng)

        # Symmetrize and check
        p_sym = _symmetrize(_best_p)
        p_sym = np.maximum(p_sym, 0.0)
        p_sym = p_sym / float(np.sum(p_sym))
        _update_best(p_sym)

        # Another L-BFGS pass on symmetrized
        if time.monotonic() < end_time - 10.0:
            z0 = _density_to_z(_best_p)
            alpha = _choose_alpha(_best_score)
            _lbfgs_optimize(z0, alpha, 300, end_time - 5.0)

        # Final LP polish
        if time.monotonic() < end_time - 8.0:
            p_lp = _best_p.copy()
            p_lp = _lp_descent(p_lp, num_iters=10, K=20, deadline=end_time - 5.0)
            _update_best(p_lp)
            p_sym2 = _symmetrize(_best_p)
            p_sym2 = np.maximum(p_sym2, 0.0)
            p_sym2 = p_sym2 / float(np.sum(p_sym2))
            _update_best(p_sym2)

    return _finalize(end_time, rng)


def _finalize(end_time, rng):
    """Convert best probability vector to sequence and validate."""
    if _best_p is None:
        p = np.ones(N) / N
    else:
        p = _best_p.copy()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        ss = float(np.sum(p))
        if ss < 0.01:
            p = np.ones(N) / N
        else:
            p = p / ss

    seq = [float(v) for v in p]

    try:
        score = float(evaluate_sequence(seq))
    except Exception:
        score = float('inf')

    # Fallback to uniform if invalid
    if not np.isfinite(score) or score > 2.0:
        seq = [1.0 / N] * N
        try:
            score = float(evaluate_sequence(seq))
        except Exception:
            score = 2.0

    # Compare with inherited
    try:
        base = height_sequence_1
        base_arr = np.asarray(base, dtype=np.float64)
        if base_arr.size > 0:
            base_arr = np.where(np.isfinite(base_arr), base_arr, 0.0)
            base_arr = np.maximum(base_arr, 0.0)
            bs = float(evaluate_sequence([float(v) for v in base_arr]))
            if np.isfinite(bs) and bs < score:
                seq = [float(v) for v in base_arr]
                score = bs
    except Exception:
        pass

    print(f"FINAL: score={score:.6f}, length={len(seq)}")
    return seq
