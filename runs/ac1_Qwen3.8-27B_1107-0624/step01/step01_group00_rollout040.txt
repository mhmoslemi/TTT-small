"""
Minimize 2n * max(conv(a,a)) / (sum(a))^2 for non-negative sequences a.
Uses (1-x)^{-1/2} (central binomial) seeds with smoothed-subgradient descent
and periodic LP refinement.
"""

import time
import math
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - target)
    if not np.any(cond):
        return np.zeros(n)
    rho_max = np.max(rho[cond])
    theta = (cssv[rho_max-1] - target) / rho_max
    return np.maximum(v - theta, 0.0)

def _fft_conv_sq(a, m):
    """Linear convolution a*a via FFT, truncated to 2n-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:len(a)*2-1]

def _eval_fast(a):
    """Fast evaluation of 2n * max_b / S^2 (assuming a >= 0, sum > 0)."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e30
    m = 1
    while m < 2*n-1:
        m <<= 1
    c = _fft_conv_sq(a, m)
    return 2.0 * n * c.max() / (S*S)

def _gen_binomial_seed(n):
    """Generate (1-x)^{-1/2} seed: a_i = C(2i,i) / 4^i."""
    a = np.zeros(n)
    # Use recurrence: a_0 = 1, a_{k+1} = a_k * (2k+1)/(2k+2)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k-1) / (2*k)
    return a

def _smoothed_grad(c, a, L, tau):
    """Gradient of smoothed max over top-L convolution coefficients."""
    n = len(a)
    L = min(L, n)
    # Find top-L indices
    if L == n:
        order = np.arange(n)
    else:
        # Use argpartition for efficiency
        idx = np.argpartition(-c, min(L, len(c)-1))[:L]
        order = idx[np.argsort(-c[idx])]
    
    wc = c[order]
    w = np.exp(wc / max(tau, 1e-15))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(L) / L
    else:
        w = w / w_sum
    
    # Compute gradient: grad[i] = sum_k w_k * 2 * a[k-i] (valid i)
    grad = np.zeros(n)
    for k, wk in zip(order, w):
        if wk < 1e-15:
            continue
        # For lag k, grad[i] += 2*wk*a[k-i] for valid i (0 <= k-i < n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        idx = np.arange(lo, hi+1)
        src = k - idx
        grad[idx] += 2.0 * wk * a[src]
    return grad

def _run_gradient_descent(a0, t_end, lr0=0.05, L0=100, tau0=0.5, momentum0=0.9):
    """Run smoothed-subgradient descent until t_end. Return best a and score."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    # Normalize to sum = 1
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S
    
    m = 1
    while m < 2*n-1:
        m <<= 1
    
    buf = np.zeros(n)
    best_a = a.copy()
    best_mc = _fft_conv_sq(a, m).max()
    L = L0
    tau = tau0
    lr = lr0
    momentum = momentum0
    step = 0
    
    while time.monotonic() < t_end - 1.0:
        c = _fft_conv_sq(a, m)
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
        
        step += 1
        
        # Adaptive schedule
        if step % 1000 == 0:
            tau = max(0.01, tau * 0.85)
            L = max(5, int(L * 0.75))
            momentum = min(0.98, momentum + 0.002)
        
        if step % 100 == 0:
            lr = max(1e-6, lr * 0.995)
        
        # Compute gradient
        g = _smoothed_grad(c, a, L, tau)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Update with momentum
        buf = momentum * buf - lr * g
        a_new = a + lr * buf
        a_new = _proj_simplex(a_new, 1.0)
        a = a_new
    
    # Final evaluation
    final_mc = _fft_conv_sq(a, m).max()
    if final_mc < best_mc:
        best_mc = final_mc
        best_a = a.copy()
    
    score = 2.0 * n * best_mc  # since sum = 1
    return best_a, score

def _lp_refine(a, M, timeout=2.0):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)_k <= M for all k.
    Return the rescaled g (sum = 1) or None if failed."""
    n = len(a)
    # Build constraint matrix: A_ub[g] <= M for each k
    # (a*g)_k = sum_i a[i] * g[k-i] (valid indices)
    # For variable g_j, it appears in constraint k with coefficient a[k-j] (if valid)
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.full(2*n-1, M),
                        bounds=[(0, None)]*n, method='highs',
                        options={'time_limit': timeout})
        if result.status == 0:
            g = result.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:

def _proj_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target)
    if not np.any(cond):
        return np.zeros(n)
    rho = np.where(cond)[0][-1] + 1
    theta = (cssv[rho-1] - target) / rho
    return np.maximum(v - theta, 0.0)


def _conv_sq_fft(a, m):
    """Linear convolution a*a via FFT. m is next pow2 >= 2*len(a)-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*len(a)-1]


def _gen_binom(n):
    """Central binomial: C(2k,k)/4^k for k=0..n-1. Monotonically decreasing."""
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    return a


def _smoothed_grad(c, a, L, tau):
    """Subgradient of smoothed max (top-L softmax) of c, w.r.t. a.
    grad[i] = sum_{k in topL} w_k * 2 * a[k-i] for valid (k-i)."""
    n = a.size
    L = min(L, c.size)

    if L >= c.size:

        top_idx = part[np.argsort(-c[part])]

    # Softmax weights
    w_max = wc.max()
    w = np.exp((wc - w_max) / max(tau, 1e-15))
    ws = w.sum()
    if ws < 1e-30:

        if lo <= hi:

    return grad
