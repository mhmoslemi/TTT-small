"""
Hybrid optimization combining log-sum-exp gradient descent (FFT-accelerated)
with LP-based refinement to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses continuation method on smoothing parameter and multiple initializations.

import numpy as np
import time
from scipy.optimize import linprog

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_solve(a, M, weights=None):
    """Solve: max sum(g) - penalty s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M
    if weights is not None:
        b_ub = M * weights  # element-wise
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def optimize_lp_improved(a, deadline, rng, max_iters=500):
    """Improved LP-based optimization with weighted constraints."""
    n = len(a)

    best_score = compute_objective(a)
    step = 0.15
    stall = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        M = c.max()

                s = step * shrink

def gradient_step_lse(a, alpha, rng):
    """
    One gradient step using log-sum-exp smoothing.
    Returns (new_a, grad_norm).
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    
    # Compute softmax weights p_k
    # Use stable log-sum-exp
    c_scaled = alpha * c
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    p = exp_c / exp_c.sum()
    
    # Compute sum_k p_k * C_k a = cross-correlation of p and a
    # (C_k a)_j = a[k-j] if valid
    # sum_k p_k * a[k-j] = cross-correlation
    # Zero-pad a to length 2n-1
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    
    # cross-correlation: sum_k p[k] * a_pad[k-j] for valid k-j
    # This is the convolution of p with reversed a_pad, evaluated at specific positions
    # Actually, sum_k p[k] * a[k-j] where 0<=k-j<n
    # = (p * a_rev) evaluated at... 
    
    # Let me think more carefully.
    # grad_k_j = 2 * a[k-j] if 0 <= k-j < n
    # dL/da_j = sum_k p_k * 2 * a[k-j] (for valid k-j)
    # = 2 * sum_k p_k * a[k-j]
    
    # Let's define q[k] = p[k] for k = 0,...,2n-2
    # Then sum_k q[k] * a[k-j] where 0<=k-j<n
    # = sum_{i=0}^{n-1} q[i+j] * a[i]  (substituting i = k-j, so k = i+j)
    # This is (q * a_rev)[n-1-j] ... hmm, let me just compute it directly with FFT.
    
    # Actually, the simplest way: 
    # sum_k p[k] * a[k-j] = (full cross-correlation of p with a) at lag j
    # = np.correlate(p, a, mode='valid') ... no, not quite.
    
    # Let's use: cross_correlate(p, a)[j] = sum_k p[k+j] * a[k] (for k where both are valid)
    # We want sum_k p[k] * a[k-j] = cross_correlate(a, p)[j] ... 
    
    # Let me just use FFT directly.
    # sum_k p[k] * a[k-j] where 0<=k<=2n-2 and 0<=k-j<=n-1
    # = sum_{i=0}^{n-1} a[i] * p[i+j] (where i = k-j, so k = i+j)
    # We need 0<=i+j<=2n-2, i.e., 0<=i<=2n-2-j. Since i<n, this is fine for j>=0.
    # So the result at position j = sum_{i=0}^{n-1} a[i] * p[i+j]
    
    # This is the cross-correlation of a and p, but p has different length.
    # a has length n, p has length 2n-1.
    # result[j] = sum_i a[i] * p[i+j] for j = 0, 1, ..., 2n-2-n+1 = n-1
    
    # We can compute this as: reverse a, convolve with p, and extract.
    # (a_rev * p)[n-1+j] = sum_i a_rev[i] * p[n-1+j-i] = sum_i a[n-1-i] * p[n-1+j-i]
    # Let i' = n-1-i: sum_{i'} a[i'] * p[n-1+j-(n-1-i')] = sum_{i'} a[i'] * p[i'+j]
    # So (a_rev * p)[n-1+j] = sum_i a[i] * p[i+j] = what we want!
    
    a_rev = a[::-1]
    conv_ap = np.convolve(a_rev, p, mode='full')
    # Extract positions n-1, n, ..., 2n-2
    grad_L = 2.0 * conv_ap[n-1:2*n-1]
    
    # Now: df/da = 2n * [S * dL/da - 2 * L * 1] / S^3
    S = a.sum()
    L = np.log(exp_c.sum()) / alpha  # = (1/alpha) * logsumexp(alpha * c) - c_max/alpha
    # Wait, L = (1/alpha) * log(sum exp(alpha * c_k))
    # log(sum exp(alpha * c_k)) = c_max + log(sum exp(alpha * c_k - c_max)) = c_max + log(exp_c.sum())
    L = (c_max + np.log(exp_c.sum())) / alpha
    
    grad_f = 2 * n * (S * grad_L - 2 * L * np.ones(n)) / (S ** 3)
    
    return grad_f, p, c

def optimize_gradient_lse(a, deadline, rng, alpha_start=1.0, alpha_end=50.0, 
                           n_alpha_steps=20, max_iters_per_alpha=200):
    """
    Gradient descent with log-sum-exp smoothing and continuation on alpha.
    """
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf

    best_a = a.copy()
    best_score = compute_objective(a)
    
    alphas = np.geomspace(alpha_start, alpha_end, n_alpha_steps)
    
    for alpha in alphas:
        if time.monotonic() > deadline:
            break
        
        # Use Adam-like optimizer
        m = np.zeros(n)  # first moment
        v = np.zeros(n)  # second moment
        beta1, beta2 = 0.8, 0.99
        lr = 0.1
        t = 0
        
        for it in range(max_iters_per_alpha):
            if time.monotonic() > deadline:
                break
            t += 1
            
            grad, p, c = gradient_step_lse(a, alpha, rng)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            # Update
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                # Try smaller step
                a_new = a - lr * 0.1 * m_hat / (np.sqrt(v_hat) + 1e-8)
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() < 1e-10:
                    break
            a_new /= a_new.sum()
            
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new.copy()
                best_score = sc
            a = a_new
            
            # Adaptive learning rate
            if it % 20 == 0 and it > 0:
                # Check if we're making progress
                pass
        
        # After each alpha, do a quick LP refinement
        if time.monotonic() < deadline - 30:

            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))

def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)

def make_parabolic(n):
    c = (n - 1) / 2.0
    return 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)

def make_optimized_shape(n, rng):
    """Create a shape that's optimized for flat convolution."""
    # Start with a raised cosine and add some structure
    base = make_cosine(n)
    # Add a second harmonic to reduce peaks
    t = np.linspace(0, 1, n)
    shape = base + 0.1 * np.sin(2 * np.pi * t) ** 2
    return np.clip(shape, 0.01, None)

**Spectral approach:**
If we zero-pad `a` to length L >= 2n-1, then:
`DFT(conv(a,a))_k = |DFT(a_pad)_k|^2`

So `conv(a,a) = IDFT(|A|^2)` where `A = DFT(a_pad)`.

To make `conv(a,a)` flat, we want `|A|^2` to be flat, i.e., `|A_k|` constant for all k.

This is the "constant modulus" condition on the spectrum. For real signals, this means the amplitude spectrum is flat.

We can try to enforce this iteratively:
1. Start with a signal
2. Compute its spectrum
3. Replace the amplitude with a flat one (keeping the phase)
4. Take the inverse FFT
5. Truncate to length n
6. Clip to non-negative
7. Repeat

This is similar to the Gerchberg-Saxton algorithm used in optical phased arrays!

"""
Hybrid optimization combining log-sum-exp gradient descent (FFT-accelerated),
LP-based refinement, and spectral flattening (Gerchberg-Saxton) to minimize
2*n*max(conv(a,a))/(sum(a))^2. Uses continuation method on smoothing parameter
and multiple diverse initializations with adaptive time allocation.

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_solve_weighted(a, M, peak_weights):
    """Solve: max sum(g) s.t. convolve(a,g) <= M*w_k for each lag k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * peak_weights
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def compute_lse_gradient(a, alpha):
    """
    Compute gradient of 2n*logsumexp(alpha*conv(a,a))/(sum(a))^2.
    Uses FFT for efficiency. Returns (grad, L, c).
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    S = a.sum()

    # Stable log-sum-exp
    c_scaled = alpha * c
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    exp_sum = exp_c.sum()
    p = exp_c / exp_sum  # softmax weights

    # Compute dL/da = 2 * sum_k p_k * C_k * a
    # Using the identity: sum_k p_k * a[k-j] = convolution trick
    a_rev = a[::-1]
    conv_ap = np.convolve(a_rev, p, mode='full')
    # Extract positions n-1 to 2n-2
    grad_L = 2.0 * conv_ap[n - 1:2 * n - 1]

    # L = (1/alpha) * logsumexp(alpha * c)
    L = (c_max + np.log(exp_sum)) / alpha

    # Full gradient: df/da = 2n * [S * dL/da - 2*L*1] / S^3
    grad = 2.0 * n * (S * grad_L - 2.0 * L * np.ones(n)) / (S ** 3)

    return grad, L, c, p


def optimize_gradient_lse(a, deadline, rng, alpha_min=2.0, alpha_max=100.0,
                          n_alpha_steps=15):
    """
    Gradient descent with log-sum-exp smoothing and continuation on alpha.
    Uses Adam optimizer and multiplicative updates for stability.
    """
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)

    alphas = np.geomspace(alpha_min, alpha_max, n_alpha_steps)

    for alpha in alphas:
        if time.monotonic() > deadline:
            break

        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2 = 0.75, 0.95
        lr = 0.05
        t = 0
        stall = 0
        local_best = best_score.copy() if isinstance(best_score, np.ndarray) else best_score

        for it in range(500):
            if time.monotonic() > deadline:
                break
            t += 1

            grad, L, c, p = compute_lse_gradient(a, alpha)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            # Multiplicative update (maintains non-negativity)
            # a_new = a * exp(-lr * grad / scale)
            scale = np.max(np.abs(grad)) + 1e-15
            log_update = -lr * m_hat / (np.sqrt(v_hat) + 1e-10)
            # Clip to prevent extreme updates
            log_update = np.clip(log_update, -3.0, 3.0)
            a_new = a * np.exp(-lr * m_hat / (scale * 0.1 + 1e-10))
            a_new = np.clip(a_new, 0.0, None)

            if a_new.sum() < 1e-10:
                # Fallback: simple gradient step
                a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() < 1e-10:
                    break

            a_new /= a_new.sum()
            sc = compute_objective(a_new)

            if sc < best_score - 1e-13:
                best_a = a_new.copy()
                best_score = sc
                local_best = sc
                stall = 0
            else:
                stall += 1

            a = a_new

            # Adaptive LR
            if it % 10 == 0 and stall > 20:
                lr = max(0.001, lr * 0.8)

            if stall > 80:
                break

        # Periodic LP refinement
        if time.monotonic() < deadline - 20:
            rem = deadline - time.monotonic()
            if rem > 30:
                lp_deadline = time.monotonic() + min(30, rem * 0.3)
                a_lp, sc_lp = optimize_lp(a.copy(), lp_deadline, rng)
                if sc_lp < best_score:
                    best_a = a_lp.copy()
                    best_score = sc_lp
                    a = a_lp

    return best_a, best_score

def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)


def make_optimized_shape(n):
    """Custom shape designed to reduce convolution peaks."""
    t = np.linspace(0, 1, n)
    # Raised cosine with slight modification to reduce center peak
    base = 0.5 * (1.0 + np.cos(2 * np.pi * t))
    # Add a small dip in the center to reduce the peak of convolution
    center = (n - 1) / 2.0
    dip = 0.05 * np.exp(-((np.arange(n) - center) / (n * 0.15)) ** 2)
    shape = base - dip
    return np.clip(shape, 0.01, None)


def make_bell(n, width=0.6):
    """Wider bell curve."""
    t = np.linspace(-1, 1, n)
    return np.exp(-t ** 2 / (2 * width ** 2))

    continuation method for the smoothing parameter.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p
    except Exception:
        pass

    print(f"Starting hybrid search, budget={budget_s}s")
    if parent is not None:
        print(f"Parent available: n={len(parent)}, score={compute_objective(parent):.6f}")

    # Build candidate list
    candidates = []

    if parent is not None:
        candidates.append(('parent', parent.copy()))
        # Downsample parent
        for n_ds in [256, 384]:
            if len(parent) >= n_ds:
                # Take first n_ds or evenly sample
                idx = np.linspace(0, len(parent) - 1, n_ds).astype(int)
                candidates.append((f'parent_ds_{n_ds}', parent[idx].copy()))

    # Diverse initializations
    init_funcs = [
        (256, make_cosine), (384, make_cosine), (512, make_cosine),
        (640, make_cosine), (768, make_cosine), (1024, make_cosine),
        (384, make_gaussian), (512, make_gaussian), (768, make_gaussian),
        (512, make_parabolic), (768, make_parabolic),
        (384, make_optimized_shape), (512, make_optimized_shape), (768, make_optimized_shape),
        (512, make_bell), (768, make_bell),
    ]
    for n, func in init_funcs:
        try:
            a = func(n)
            candidates.append((f'{func.__name__}_{n}', a))
        except Exception:
            pass

    # Random initializations
    for n in [384, 512, 768, 1024]:
        a = np.abs(rng.normal(1.0, 0.5, n))
        candidates.append((f'rand_{n}', a))

    # Smoothed random (blur a random signal)
    for n in [512, 768]:
        a = np.abs(rng.normal(0, 1, n))
        # Apply a simple moving average
        k = max(3, n // 32)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        candidates.append((f'srand_{n}', a))

    print(f"Total candidates: {len(candidates)}")

    # Phase 1: Quick screening
    screened = []
    for name, a in candidates:
        a = np.clip(a, 1e-10, None).astype(np.float64)
        if a.sum() < 1e-10:
            continue
        a = a / a.sum()
        sc = compute_objective(a)
        screened.append((sc, name, a))

    screened.sort(key=lambda x: x[0])
    print("Screening results (top 10):")
    for sc, name, _ in screened[:10]:
        print(f"  {name}: {sc:.6f}")

    # Phase 2: Detailed optimization
    n_top = min(10, len(screened))
    # Allocate time: more to better initial scores
    available_time = deadline - time.monotonic()
    polish_reserve = min(180, available_time * 0.12)
    search_time = available_time - polish_reserve

    # Weight by inverse rank (better initial = more time)
    weights = np.array([1.0 / (i + 1) for i in range(n_top)])
    weights /= weights.sum()
    time_allocs = weights * search_time

    for rank in range(n_top):
        if time.monotonic() > deadline - polish_reserve:
            break
        sc_init, name, a_init = screened[rank]
        t_alloc = time_allocs[rank]
        config_deadline = time.monotonic() + t_alloc

        print(f"\n[{rank + 1}/{n_top}] {name} (init={sc_init:.6f}, alloc={t_alloc:.0f}s)")

        # Method 1: LP optimization (gets ~40% of time)
        lp_deadline = time.monotonic() + t_alloc * 0.4
        a_lp, sc_lp = optimize_lp(a_init.copy(), lp_deadline, rng)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  LP: {sc_lp:.6f} *** NEW BEST ***")
        else:
            print(f"  LP: {sc_lp:.6f}")

        # Method 2: Gradient LSE on best of (LP result, original)
        a_for_grad = a_lp if sc_lp < sc_init else a_init.copy()
        grad_deadline = min(config_deadline, time.monotonic() + t_alloc * 0.35)
        if time.monotonic() < grad_deadline - 5:
            a_grad, sc_grad = optimize_gradient_lse(a_for_grad, grad_deadline, rng)
            if sc_grad < best_score:
                best_score = sc_grad
                best_seq = a_grad.copy()
                print(f"  GradLSE: {sc_grad:.6f} *** NEW BEST ***")
            else:
                print(f"  GradLSE: {sc_grad:.6f}")

            # Method 3: Spectral on the gradient result
            a_for_spec = a_grad if sc_grad < sc_lp else a_lp
            spec_deadline = min(config_deadline, deadline - polish_reserve)
            if time.monotonic() < spec_deadline - 10:
                a_spec, sc_spec = optimize_spectral(a_for_spec, spec_deadline, rng)
                if sc_spec < best_score:
                    best_score = sc_spec
                    best_seq = a_spec.copy()
                    print(f"  Spec: {sc_spec:.6f} *** NEW BEST ***")
                else:
                    print(f"  Spec: {sc_spec:.6f}")

    # Phase 3: Cross-pollination - try combining top results
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\nCross-pollination phase (current best: {best_score:.6f})")
        # Try blending best with some top initializations
        for sc_c, name_c, a_c in screened[:5]:
            if time.monotonic() > deadline - 30:
                break
            n_best = len(best_seq)
            if n_best != len(a_c):
                # Resample
                idx = np.linspace(0, len(a_c) - 1, n_best).astype(int)
                a_c = a_c[idx]
            for blend in [0.3, 0.5, 0.7]:
                a_blend = blend * best_seq + (1 - blend) * a_c
                a_blend = np.clip(a_blend, 0, None)
                if a_blend.sum() < 1e-10:
                    continue
                a_blend /= a_blend.sum()
                sc_blend = compute_objective(a_blend)
                if sc_blend < best_score:
                    best_score = sc_blend
                    best_seq = a_blend.copy()
                    print(f"  Blend {blend} with {name_c}: {sc_blend:.6f} *** NEW BEST ***")
                    break

    # Phase 4: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline:
        print(f"\nFinal polish (current best: {best_score:.6f})")
        remaining = deadline - time.monotonic()

        # LP polish
        lp_final_deadline = time.monotonic() + remaining * 0.4
        a_final, sc_final = optimize_lp(best_seq.copy(), lp_final_deadline, rng)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  LP polish: {best_score:.6f} *** NEW BEST ***")

        # Gradient polish
        if time.monotonic() < deadline - 10:
            grad_final_deadline = time.monotonic() + remaining * 0.3
            a_gf, sc_gf = optimize_gradient_lse(best_seq, grad_final_deadline, rng,
                                                alpha_min=10.0, alpha_max=200.0,
                                                n_alpha_steps=10)
            if sc_gf < best_score:
                best_score = sc_gf
                best_seq = a_gf.copy()
                print(f"  Grad polish: {best_score:.6f} *** NEW BEST ***")

        # Spectral polish
        if time.monotonic() < deadline - 10:
            spec_final_deadline = deadline
            a_sf, sc_sf = optimize_spectral(best_seq, spec_final_deadline, rng)
            if sc_sf < best_score:
                best_score = sc_sf
                best_seq = a_sf.copy()
                print(f"  Spec polish: {best_score:.6f} *** NEW BEST ***")

        # One more LP pass
        if time.monotonic() < deadline - 5:
            a_ll, sc_ll = optimize_lp(best_seq, deadline, rng)
            if sc_ll < best_score:
                best_score = sc_ll
                best_seq = a_ll.copy()
                print(f"  Final LP: {best_score:.6f} *** NEW BEST ***")

    if best_seq is None:
        best_seq = make_cosine(512)
        best_score = compute_objective(best_seq)

    # Validate and format output
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.05 / best_seq.sum())

    # Final validation
    final_score = compute_objective(best_seq)
    elapsed_total = time.monotonic() - start_time
    print(f"\nDONE in {elapsed_total:.1f}s. Final score={final_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
