"""
Search for low-score non-negative step-function heights.

Strategy:
- Normalize all candidate sequences to sum 1. The score is then 2*n*max(conv(a,a)).
- Use a portfolio of starts: provided construction, two-block optimum, random smooth,

def _next_pow2(n):
    n = int(n)
    p = 1
    while p < n:
        p <<= 1
    return p


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(2, dtype=np.float64) / 2.0
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def softmax_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(2, dtype=np.float64) / 2.0
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def conv_and_max(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64), 0.0
    if n <= 1200:
        c = np.convolve(a, a)
    else:
        size = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fa * fa, size)
        c = c[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    M = float(c.max()) if c.size else 0.0
    if M < 0.0:
        M = 0.0
    return c, M

def choose_active_rows(c, max_rows):
    m = c.size
    max_rows = int(max_rows)
    if max_rows <= 0:
        return np.array([], dtype=np.int64)
    if max_rows >= m:
        return np.arange(m, dtype=np.int64)
    if max_rows == 1:
        return np.array([int(np.argmax(c))], dtype=np.int64)
    idx = np.argpartition(c, -max_rows)[-max_rows:]
    return idx.astype(np.int64)


def solve_lp_direction(f, M, rows):
    if linprog is None:
        return None

    f = np.asarray(f, dtype=np.float64).ravel()
    rows = np.asarray(rows, dtype=np.int64).ravel()
    n = f.size

    if rows.size == 0 or n == 0 or M <= 1e-300:
        return None
    if n > 2200:
        return None

    f_lp = f.copy()
    minv = float(f_lp.min())
    if minv < 1e-9:
        f_lp = np.maximum(f_lp, 1e-8)
        f_lp /= float(f_lp.sum())

    A = np.zeros((rows.size, n), dtype=np.float64)
    for rrow, k in enumerate(rows):
        k = int(k)
        if k < 0 or k >= 2 * n - 1:
            continue
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            idx = lo + np.arange(hi - lo, dtype=np.int64)
            A[rrow, lo:hi] = f_lp[k - idx]

    if not np.any(A):
        return None

    A = A / float(M)
    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.ones(rows.size, dtype=np.float64)

    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=(0.0, 1.0), method="highs")
    except Exception:
        return None

    if res is not None and getattr(res, "status", 1) == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        s = float(g.sum())
        if s > 1.0 + 1e-7:
            return g
    return None


def gradient_raw_weights(p, f):
    """Return r[j] = sum_i f[i] p[i+j] without centering or normalization."""
    f = np.asarray(f, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = f.size
    m = p.size

    if n == 0:
        return np.zeros(0, dtype=np.float64)

    if n <= 512:
        r = np.empty(n, dtype=np.float64)
        for j in range(n):
            r[j] = float(np.dot(f, p[j:j + n]))
    else:
        size = _next_pow2(n + m - 1)
        fa = np.zeros(size, dtype=np.float64)
        fa[:n] = f
        pb = np.zeros(size, dtype=np.float64)
        pb[:m] = p[::-1]
        out = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(pb, size), size)
        idx = (m - 1) - np.arange(n, dtype=np.int64)
        r = out[idx]

    return np.where(np.isfinite(r), r, 0.0)


def _center_unit(d):
    d = np.asarray(d, dtype=np.float64).ravel()
    if d.size == 0:
        return d
    d = d - float(np.mean(d))
    mx = float(np.max(np.abs(d)))
    if mx <= 1e-300:
        return np.zeros(d.size, dtype=np.float64)
    return d / mx


def smooth_max_direction(f, c, M, temp_frac):
    n = f.size
    if M <= 1e-300 or n == 0:
        return np.zeros(n, dtype=np.float64)

    temp = max(1e-12, float(temp_frac) * float(M))
    e = (c - M) / temp
    e = np.clip(e, -30.0, 0.0)
    p = np.exp(e)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / s

    r = gradient_raw_weights(p, f)
    return _center_unit(r)


def top_power_direction(f, c, top, power):
    n = f.size
    m = c.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)

    top = int(min(max(1, top), m))
    p = np.zeros(m, dtype=np.float64)

    if top == m:
        tmp = c.copy()
        s = float(tmp.sum())
        if s <= 1e-300:
            p[int(np.argmax(c))] = 1.0
        else:
            p = tmp / s
    else:
        idx = np.argpartition(c, -top)[-top:]
        vals = c[idx]
        vmax = float(vals.max())
        if vmax <= 1e-300:
            p[int(np.argmax(c))] = 1.0
        else:
            v = vals / vmax
            p[idx] = v ** float(power)
            s = float(p.sum())
            if s <= 1e-300:
                p[int(np.argmax(c))] = 1.0
            else:
                p = p / s

    r = gradient_raw_weights(p, f)
    return _center_unit(r)


def peak_direction(f, k):
    n = f.size
    k = int(k)
    d = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if hi > lo:
        idx = np.arange(lo, hi, dtype=np.int64)
        d[lo:hi] = f[k - idx]
    return _center_unit(d)


def try_gradient_additive(f, cur, d):
    n = f.size
    if d is None or n == 0:
        return None, cur

    mx = float(np.max(np.abs(d)))
    if mx <= 1e-300:
        return None, cur

    maxf = float(np.max(f))
    if maxf <= 1e-300:
        maxf = 1.0 / float(n)

    alphas = (0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 8.0)
    best_h = None
    best_sc = cur

    for alpha in alphas:
        h = f - (maxf * alpha) * d
        h = normalize_nonneg(h)
        sc = score_normalized(h)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_h = h

    return best_h, best_sc


def try_gradient_multiplicative(f, cur, d):
    n = f.size
    if d is None or n == 0:
        return None, cur

    mx = float(np.max(np.abs(d)))
    if mx <= 1e-300:
        return None, cur

    logf = np.log(np.maximum(f, 1e-300))
    betas = (0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0, 30.0)
    best_h = None
    best_sc = cur

    for beta in betas:
        h = np.exp(np.clip(logf - beta * d, -50.0, 50.0))
        h = normalize_nonneg(h)
        sc = score_normalized(h)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_h = h

    return best_h, best_sc


def try_lp_combo(f, cur, g):
    g = np.asarray(g, dtype=np.float64).ravel()
    S = float(g.sum())
    if S <= 1.0 + 1e-7:
        return None, cur

    gn = g / S
    best_h = None
    best_sc = cur

    if S > 10.0:
        ts = (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5)
    elif S > 2.0:
        ts = (0.005, 0.01, 0.03, 0.1, 0.2, 0.4, 0.7)
    else:
        ts = (0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0)

    for t in ts:
        h = (1.0 - float(t)) * f + float(t) * gn
        h = normalize_nonneg(h)
        sc = score_normalized(h)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_h = h

    return best_h, best_sc


def random_perturb(f, rng, strength):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return np.ones(2, dtype=np.float64) / 2.0

    h = f.copy()
    if n > 1:
        h = h * np.exp(rng.normal(0.0, float(strength), size=n))

        if n > 16:
            w1 = max(2, n // 200)
            w2 = max(3, n // 6)
            width = int(rng.integers(w1, w2 + 1))
            if width > n:
                width = n
            start = int(rng.integers(0, max(1, n - width + 1)))
            h[start:start + width] *= np.exp(float(rng.normal(0.0, strength * 1.5)))

        if rng.random() < 0.15:
            blend = float(rng.uniform(0.05, 0.35))
            h = (1.0 - blend) * f + blend * f[::-1]

        if rng.random() < 0.10:
            shift = int(rng.integers(1, max(2, min(n, 16))))
            h = 0.8 * f + 0.2 * np.roll(f, shift)

    return normalize_nonneg(h)


def update_global(state, seq, score):
    if not np.isfinite(score):
        return False
    seq = np.asarray(seq, dtype=np.float64).ravel()
    if seq.size == 0:
        return False

    seq = normalize_nonneg(seq)
    score = float(score)
    old = float(state["score"])

    if score < old - 1e-12:
        state["score"] = score
        state["seq"] = seq.copy()
        if (not np.isfinite(old)) or (old - score) > 1e-6:
            print(f"new best score={score:.12f} n={seq.size}", flush=True)
        return True
    return False


def smooth_score_value_grad(z, temp):
    z = np.asarray(z, dtype=np.float64).ravel()
    a = softmax_z(z)
    n = a.size

    if n == 0:
        return 1e100, np.zeros(0, dtype=np.float64)

    c, M = conv_and_max(a)
    c = np.maximum(c, 0.0)

    t = float(temp)
    if t <= 1e-12:
        t = 1e-6

    e = (c - M) / t
    e = np.clip(e, -30.0, 0.0)
    p = np.exp(e)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / s

    val = 2.0 * n * float(np.dot(p, c))
    if not np.isfinite(val):
        return 1e100, np.zeros(z.size, dtype=np.float64)

    r = gradient_raw_weights(p, a)
    g = 4.0 * n * r
    g = g - float(np.dot(g, a)) * np.ones_like(g)
    gz = a * g

    if not np.all(np.isfinite(gz)):
        gz = np.zeros(z.size, dtype=np.float64)

    return val, gz


def lowdim_refine(a0, K, rng, temps=(0.25, 0.12, 0.06, 0.03)):
    if opt_minimize is None:
        return resize_linear(a0, K)

    a = resize_linear(a0, int(max(4, K)))
    n = a.size

    for temp in temps:
        z0 = np.log(np.maximum(a, 1e-12))
        z0 = z0 - float(np.max(z0))

        try:
            cur_val, _ = smooth_score_value_grad(z0, temp)
        except Exception:
            cur_val = score_normalized(a) * 10.0

        z = z0 + rng.normal(0.0, 0.01, size=n)
        try:
            res = opt_minimize(
                smooth_score_value_grad,
                z,
                args=(temp,),
                jac=True,
                method="L-BFGS-B",
                options={
                    "maxiter": 150,
                    "maxls": 40,
                    "maxfun": 1200,
                    "ftol": 1e-10,
                    "gtol": 1e-6,
                },
            )
        except Exception:
            continue

        if res is not None and res.x is not None:
            an = normalize_nonneg(softmax_z(res.x))
            accept = False
            if np.all(np.isfinite(an)):
                try:
                    if np.isfinite(res.fun) and float(res.fun) < float(cur_val) - 1e-10:
                        accept = True
                    else:
                        new_sc = score_normalized(an)
                        cur_sc = score_normalized(a)
                        if np.isfinite(new_sc) and new_sc < cur_sc - 1e-10:
                            accept = True
                        elif (
                            np.isfinite(res.fun)
                            and abs(new_sc - cur_sc) <= 1e-8
                            and float(res.fun) < float(cur_val) + 1e-8
                        ):
                            accept = True
                except Exception:
                    accept = False

            if accept:
                a = an

    return a


def run_log_smooth(a0, temp, maxiter, noise_scale=0.0, rng=None):
    if opt_minimize is None:
        return normalize_nonneg(a0)

    a = normalize_nonneg(a0)
    n = a.size
    if n == 0:
        return a
    if n > 1200:
        return a

    z = np.log(np.maximum(a, 1e-12))
    z = z - float(np.max(z))
    if noise_scale > 0.0 and rng is not None:
        z = z + rng.normal(0.0, float(noise_scale), size=n)

    try:
        res = opt_minimize(
            smooth_score_value_grad,
            z,
            args=(float(temp),),
            jac=True,
            method="L-BFGS-B",
            options={
                "maxiter": int(maxiter),
                "maxls": 50,
                "maxfun": 2000,
                "ftol": 1e-10,
                "gtol": 1e-6,
            },
        )
        if res is not None and res.x is not None:
            an = normalize_nonneg(softmax_z(res.x))
            if np.all(np.isfinite(an)):
                return an
    except Exception:
        pass

    return a


def optimize_sequence(f, time_limit, rng, state, use_lp=True):
    f = normalize_nonneg(f)
    n = f.size
    if n < 2:
        f = np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64)
        n = 2

    c, M = conv_and_max(f)
    cur = float(2.0 * n * M)
    local_best = cur
    local_seq = f.copy()
    update_global(state, f, cur)

    deadline = time.monotonic() + max(0.0, float(time_limit))
    temp = 0.22
    no_improve = 0
    lp_bad = 0
    last_progress = time.monotonic()
    it = 0
    should_lp = bool(use_lp and n <= 1200)

    while time.monotonic() < deadline:
        it += 1

        c, M = conv_and_max(f)
        cur = float(2.0 * n * M)
        if not np.isfinite(cur):
            f = local_seq.copy()
            continue

        if cur < local_best - 1e-12:
            local_best = cur
            local_seq = f.copy()
            update_global(state, f, local_best)
            no_improve = 0
            last_progress = time.monotonic()
            temp = max(0.03, temp * 0.95)
        else:
            no_improve += 1
            if no_improve > 20:
                temp = min(0.55, temp * 1.08)

        if cur > local_best * 1.05 + 1e-12:
            f = local_seq.copy()
            no_improve = max(0, no_improve - 5)
            continue

        if no_improve > 80 and (time.monotonic() - last_progress) > max(
            20.0, 0.45 * float(time_limit)
        ):
            break

        moved = False

        # LP active-set direction.
        if (
            should_lp
            and (lp_bad < 10 or no_improve > 20)
            and ((it % (1 if n <= 512 else 2) == 0) or no_improve > 5)
        ):
            if n <= 600:
                max_rows = 2 * n - 1
            elif n <= 1000:
                max_rows = min(2 * n - 1, 900)
            else:
                max_rows = min(2 * n - 1, 650)

            rows = choose_active_rows(c, max_rows)
            g = solve_lp_direction(f, M, rows)

            if g is not None:
                h, hsc = try_lp_combo(f, cur, g)
                if h is not None:
                    f = h
                    moved = True
                    lp_bad = 0
                else:
                    lp_bad += 1
            else:
                lp_bad += 1

        if not moved:
            d = smooth_max_direction(f, c, M, temp)
            h, hsc = try_gradient_additive(f, cur, d)
            if h is not None:
                f = h
                moved = True

        if not moved:
            h, hsc = try_gradient_multiplicative(f, cur, d)
            if h is not None:
                f = h
                moved = True

        if not moved and n <= 1024:
            top = int(min(2 * n - 1, max(64, n // 2)))
            d2 = top_power_direction(f, c, top, 4.0)
            h, hsc = try_gradient_additive(f, cur, d2)
            if h is not None:
                f = h
                moved = True

        if not moved and n <= 1024:
            h, hsc = try_gradient_multiplicative(f, cur, d2)
            if h is not None:
                f = h
                moved = True

        if not moved:
            topk = min(4, int(c.size))
            if topk == c.size:
                ks = np.arange(c.size, dtype=np.int64)
            else:
                ks = np.argpartition(c, -topk)[-topk:].astype(np.int64)

            for k in ks:
                d3 = peak_direction(f, int(k))
                h, hsc = try_gradient_additive(f, cur, d3)
                if h is not None:
                    f = h
                    moved = True
                    break

        if not moved:
            strength = 0.03 + 0.01 * min(5.0, float(no_improve) / 10.0)
            f = random_perturb(f, rng, strength)

    c, M = conv_and_max(local_seq)
    sc = float(2.0 * local_seq.size * M)
    update_global(state, local_seq, sc)
    return local_seq, sc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    hard_deadline = t0 + total_budget - 35.0
    if hard_deadline < t0 + 30.0:
        hard_deadline = t0 + 30.0

    state = {"score": float("inf"), "seq": None}

    provided = load_provided_sequence()
    if provided is not None:
        if provided.size > 6000:
            provided = resize_linear(provided, 4096)
        update_global(state, provided, score_normalized(provided))

    # Guaranteed sane fallbacks.
    update_global(
        state,
        np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64),
        16.0 / 9.0,
    )
    update_global(state, np.ones(128, dtype=np.float64), score_normalized(np.ones(128)))

    # If the provided construction is available, polish it first.
    if provided is not None:
        remaining = hard_deadline - time.monotonic()
        limit = min(500.0, 0.45 * remaining)
        if limit > 20.0:
            optimize_sequence(
                provided,
                limit,
                rng,
                state,
                use_lp=(provided.size <= 1200),
            )

    # Full-sequence smooth L-BFGS annealing, if length is moderate.
    if state["seq"] is not None and state["seq"].size <= 1024:
        remaining = hard_deadline - time.monotonic()
        if remaining > 60.0:
            for temp in (0.25, 0.12, 0.06, 0.03):
                if hard_deadline - time.monotonic() < 20.0:
                    break
                seq = run_log_smooth(state["seq"], temp, 200, 0.0, rng)
                sc = score_normalized(seq)
                if np.isfinite(sc):
                    update_global(state, seq, sc)

    # Portfolio of basins and lengths.
    tasks = []
    for kind in (4, 5, 0, 1, 2, 6, 7, 8):
        tasks.append((64, kind, 8.0))
    for kind in (4, 5, 0, 1, 2, 6, 7, 8):
        tasks.append((128, kind, 10.0))
    for kind in (4, 5, 0, 1, 2, 6, 7, 8, -1):
        tasks.append((256, kind, 15.0))
    for kind in (-1, 4, 5, 0, 1, 6, 7, 8):
        tasks.append((512, kind, 35.0))
    for kind in (-1, 4, 5, 0, 1, 6):
        tasks.append((1024, kind, 50.0))
    for kind in (-1, 4, 0, 1):
        tasks.append((2048, kind, 70.0))

    if provided is not None and provided.size not in (64, 128, 256, 512, 1024, 2048):
        tasks.append((int(provided.size), -2, 40.0))

    total_task_time = 0.0
    for _, _, base in tasks:
        total_task_time += float(base)

    remaining = hard_deadline - time.monotonic()
    task_budget = max(0.0, 0.55 * remaining)
    if total_task_time > 0.0 and task_budget < total_task_time:
        scale = max(0.20, task_budget / total_task_time)
    else:
        scale = 1.0

    for N, kind, base in tasks:
        remaining = hard_deadline - time.monotonic()
        if remaining < 5.0:
            break
        limit = float(base) * scale
        if limit < 2.0:
            continue

        seq = make_initial(N, kind, rng, state["seq"], provided)
        sc = score_normalized(seq)

        # Once we have a good answer, skip obviously bad random starts.
        if (
            np.isfinite(sc)
            and np.isfinite(state["score"])
            and state["score"] < 1.70
            and sc > max(3.0, 2.0 * state["score"])
            and kind not in (-1, -2)
        ):
            continue

        optimize_sequence(seq, limit, rng, state, use_lp=(N <= 1200))

    # Another full log-smooth anneal after portfolio.
    if state["seq"] is not None and state["seq"].size <= 1024:
        remaining = hard_deadline - time.monotonic()
        if remaining > 50.0:
            for temp in (0.12, 0.06, 0.03):
                if hard_deadline - time.monotonic() < 20.0:
                    break
                seq = run_log_smooth(state["seq"], temp, 250, 0.0, rng)
                sc = score_normalized(seq)
                if np.isfinite(sc):
                    update_global(state, seq, sc)

    # Multi-resolution refinement around the best basin.
    if state["seq"] is not None:
        baseN = int(state["seq"].size)
        lengths = []
        for N in (baseN, max(64, baseN // 2), min(4096, baseN * 2), min(4096, baseN * 3), 512, 1024):
            N = int(max(2, N))
            if N <= 4096 and N not in lengths:
                lengths.append(N)

        for N in lengths:
            remaining = hard_deadline - time.monotonic()
            if remaining < 50.0:
                break
            limit = min(150.0, 0.18 * remaining)
            if limit < 20.0:
                break

            seq = resize_linear(state["seq"], N)
            optimize_sequence(seq, limit, rng, state, use_lp=(N <= 1200))

            remaining = hard_deadline - time.monotonic()
            if remaining > 40.0 and N <= 2048:
                seq2 = resize_hold(state["seq"], N)
                limit2 = min(80.0, 0.10 * remaining)
                if limit2 >= 15.0:
                    optimize_sequence(seq2, limit2, rng, state, use_lp=(N <= 1200))

    # Low-dimensional coarse refinement, then re-expand.
    if opt_minimize is not None and state["seq"] is not None:
        targetN = int(state["seq"].size)
        for K in (16, 32, 64, 128):
            remaining = hard_deadline - time.monotonic()
            if remaining < 40.0:
                break
            aK = lowdim_refine(state["seq"], K, rng)
            seq = resize_linear(aK, targetN)
            sc = score_normalized(seq)
            if np.isfinite(sc):
                update_global(state, seq, sc)

            limit = min(60.0, 0.08 * remaining)
            if limit >= 15.0:
                optimize_sequence(seq, limit, rng, state, use_lp=(targetN <= 1200))

    # Final full-sequence smooth polish.
    if state["seq"] is not None and state["seq"].size <= 1024:
        for temp in (0.12, 0.06, 0.03, 0.02):
            remaining = hard_deadline - time.monotonic()
            if remaining < 15.0:
                break
            seq = run_log_smooth(state["seq"], temp, 300, 0.0, rng)
            sc = score_normalized(seq)
            if np.isfinite(sc):
                update_global(state, seq, sc)

    # Final local polish with all remaining time.
    remaining = hard_deadline - time.monotonic()
    if state["seq"] is not None and remaining > 10.0:
        optimize_sequence(
            state["seq"],
            remaining,
            rng,
            state,
            use_lp=(state["seq"].size <= 1200),
        )

    seq = state["seq"] if state["seq"] is not None else np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64)
    seq = normalize_nonneg(seq)

    cand = []
    for x in seq:
        v = float(x)
        if not np.isfinite(v):
            v = 0.0
        if v < 0.0:
            v = 0.0
        if v > 1000.0:
            v = 1000.0
        cand.append(v)

    try:
        official = evaluate_sequence(cand)
        if np.isfinite(official):
            print(f"official_best score={official:.12f} n={len(cand)}", flush=True)
        else:
            print(f"official score non-finite, using internal score={state['score']:.12f}", flush=True)
    except Exception:
        pass

    return cand
