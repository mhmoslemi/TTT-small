    n = len(a)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = compute_objective(a)
    eta = 0.01  # initial step size
    stall = 0
    
    while time.monotonic() < deadline:
        c = compute_conv_fft(a)
        # Get top n_peaks
        top_k = np.argsort(c)[-n_peaks:]
        c_max = c[top_k[0]]  # Actually c.max()
        
        # Weighted subgradient: weight each peak by its value
        weights = c[top_k]
        weights = weights / weights.sum()
        
        subgrad = np.zeros(n)
        for w, k in zip(weights, top_k):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                indices = np.arange(lo, hi + 1)
                subgrad[indices] += w * 2 * a[k - indices]
        
        # Center the gradient (project onto sum=0 hyperplane)
        subgrad -= subgrad.mean()
        
        a_new = a - eta * subgrad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        sc = compute_objective(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.05, 0.1)  # increase step
        else:
            stall += 1
            eta *= 0.95
            a = best_a  # revert
            if stall > 100:
                # Perturbation
                noise = rng.normal(0, 0.05, n)
                a = best_a * (1 + noise)
                a = np.maximum(a, 0)
                a /= a.sum()
                stall = 0
        if stall > 300:
            break
    
    return best_a, best_score

def make_cosine(n):
    """Raised cosine (Hann) window."""
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    x = np.arange(n) - center
    return np.exp(-0.5 * (x / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_flat_top(n, edge_frac=0.2):
    """Flat-top window with smooth edges."""
    x = np.arange(n) / (n - 1.0)
    edge = edge_frac
    a = np.ones(n)
    # Smooth left edge
    mask_l = x < edge
    a[mask_l] = 0.5 * (1.0 + np.cos(np.pi * (0.5 - x[mask_l] / edge)))
    # Smooth right edge
    mask_r = x > 1.0 - edge
    a[mask_r] = 0.5 * (1.0 + np.cos(np.pi * (x[mask_r] - (1.0 - edge)) / edge))
    return a


def make_modulated(n, freq=1.0, amp=0.1):
    """Flat sequence with sinusoidal modulation."""
    base = np.ones(n)
    x = np.arange(n) / n
    return base * (1.0 + amp * np.cos(2.0 * np.pi * freq * x))


def make_super_gaussian(n, p=4.0, width=0.3):
    """Super-Gaussian window."""
    center = (n - 1) / 2.0
    x = (np.arange(n) - center) / (width * n)

                    a_opt = a_lp
                    score = score_lp

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: score={best_score:.6f} (n={n}, init={init_type})")
        else:
            print(f"  => score={score:.6f} (best: {best_score:.6f})")

    # Final polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\nFinal polish phase, current best={best_score:.6f}")
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)

        # Try subgradient again from the best
        a_p, s_p = subgradient_optimize(
            best_seq, min(start_time + remaining * 0.5, deadline), rng,
            use_momentum=True, top_k=12, sharpness=5.0
        )
        if s_p < best_score:
            best_score = s_p
            best_seq = a_p
            print(f"  After subgradient polish: {best_score:.6f}")

        # LP polish if n is small
        if n_best <= 768 and time.monotonic() < deadline - 10:
            a_lp, s_lp = lp_polish(best_seq, deadline, rng)
            if s_lp < best_score:
                best_score = s_lp
                best_seq = a_lp
                print(f"  After LP polish: {best_score:.6f}")

        # Final random restarts around best
        if time.monotonic() < deadline - 5:
            for trial in range(5):
                if time.monotonic() > deadline:
                    break
                noise = rng.normal(0, 0.01, n_best)
                a_try = best_seq * (1.0 + noise)
                a_try = np.maximum(a_try, 0.0)
                if a_try.sum() < 1e-10:
                    continue
                a_try /= a_try.sum()
                trial_deadline = min(time.monotonic() + (deadline - time.monotonic()) / (5 - trial), deadline)
                a_t, s_t = subgradient_optimize(a_try, trial_deadline, rng,
                                                 use_momentum=True, top_k=8, sharpness=4.0)
                if s_t < best_score:
                    best_score = s_t
                    best_seq = a_t
                    print(f"  Restart {trial}: new best {best_score:.6f}")

    if best_seq is None:
        # Fallback
        best_seq = make_cosine(256)
        best_score = compute_objective(best_seq)

    # Ensure valid for evaluator
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())

    # Scale so values are in reasonable range (avoid clipping at 1000)
    if best_seq.max() > 500:
        best_seq = best_seq * (500.0 / best_seq.max())

    elapsed = time.monotonic() - start_time
    final_score = compute_objective(best_seq)
    print(f"\nDONE in {elapsed:.1f}s. Final: score={final_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
