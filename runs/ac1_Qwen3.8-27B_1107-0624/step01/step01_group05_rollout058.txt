"""
Multi-scale gradient optimization with adaptive soft-max and peak-directed
polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
1. Multi-resolution: Optimize at coarser scales first, then upsample.
2. Adaptive soft-max: Dynamically tune the temperature for the max surrogate.
3. Peak-directed local search: Specifically target and reduce top convolution peaks.
4. Diverse initializations: Structured shapes + random + inherited best.
5. Population recombination: Blend top candidates for exploration.

import numpy as np
import time
from scipy.optimize import minimize

# --- Global state ---
_best_score = float("inf")
_best_p = None
_search_deadline = 0.0


class SearchTimeout(Exception):
    pass


def _pow2_at_least(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _self_conv_fft(p):
    """Compute full self-convolution of p using FFT."""
    n = len(p)
    size = _pow2_at_least(2 * n - 1)
    f = np.fft.rfft(p, size)
    c = np.fft.irfft(f * f, size).real[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _grad_from_weights(p, w):
    """Compute gradient: grad_p[j] = 2 * sum_k w[k] * p[k-j] (valid indices)."""
    n = len(p)
    # We need: for each j in [0,n), sum over k where 0<=k<2n-1 and 0<=k-j<n of w[k]*p[k-j]
    # This equals (reversed_p * w)[n-1-j] in the full convolution
    rev_p = p[::-1]
    size = _pow2_at_least(n + (2 * n - 1) - 1)
    fp = np.fft.rfft(rev_p, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fp * fw, size).real
    # Extract indices n-1 to 2n-2
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _score_p(p):
    """Compute the objective for a probability vector p (sum=1)."""
    c = _self_conv_fft(p)
    return 2.0 * float(len(p)) * float(np.max(c))


def _update_best(p, score):
    global _best_score, _best_p
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_p = np.array(p, copy=True)


def _probs_from_logit(z):
    """Convert logit vector to probability vector (z has length n-1, p[0] is reference)."""
    n = len(z) + 1
    z = np.clip(z, -50.0, 50.0)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(n, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _density_to_logit(f):
    """Convert a non-negative density vector to logit representation."""
    f = np.asarray(f, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 1e-30)
    n = len(f)
    f0 = max(f[0], 1e-30)
    z = np.log(f[1:] / f0)
    return np.clip(z, -50.0, 50.0)


def _objective_and_grad(z, alpha, n):
    """Soft-max surrogate objective and its gradient w.r.t. logit z."""
    if time.monotonic() >= _search_deadline:
        raise SearchTimeout

    p = _probs_from_logit(z)
    c = _self_conv_fft(p)

    # Track hard score
    hard = 2.0 * float(n) * float(np.max(c))
    _update_best(p, hard)

    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s

    gp = _grad_from_weights(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e8, 1e8)
    return float(J), gz


def _optimize_logit(z0, alpha, n, max_fun, deadline):
    """Run L-BFGS-B on the logit parametrization."""
    global _search_deadline
    _search_deadline = float(deadline)
    bounds = [(-50.0, 50.0)] * (n - 1)
    try:
        minimize(
            _objective_and_grad,
            np.array(z0, dtype=np.float64),
            args=(alpha, n),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxfun": int(max_fun),
                "maxiter": int(max(30, max_fun // 3)),
                "ftol": 1e-14,
                "gtol": 1e-10,
                "maxls": 20,
                "disp": False,
            },
        )
    except SearchTimeout:
        return
    except Exception:
        return


def _upsample(p, factor):
    """Upsample probability vector by block repetition with scaling."""
    n = len(p)
    new_n = n * factor
    new_p = np.empty(new_n, dtype=np.float64)
    for i in range(n):
        new_p[i * factor:(i + 1) * factor] = p[i] / factor
    return new_p


def _interpolate_upsample(p, factor):
    """Upsample via linear interpolation (better than block repetition)."""
    n = len(p)
    new_n = n * factor
    old_x = np.arange(n, dtype=np.float64)
    new_x = np.arange(new_n, dtype=np.float64) / factor
    new_p = np.interp(new_x, old_x, p)
    s = float(np.sum(new_p))
    if s > 0:
        new_p /= s
    return new_p


def _make_candidate_logistic-beta(beta, n):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    u = beta * (x - 0.5)
    f = 1.0 / (1.0 + np.exp(-np.clip(u, -50, 50)))
    f = np.maximum(f, 1e-10)
    f /= np.sum(f)

def _make_candidate_linear(rho, n):

    f = 1.0 + rho * (2.0 * x - 1.0)
    f = np.maximum(f, 1e-10)
    f /= np.sum(f)
    return _density_to_logit(f)


def _make_candidate_exp(beta, n):
    x = np.arange(n, dtype=np.float64) / n
    f = np.exp(beta * x)
    f /= np.sum(f)
    return _density_to_logit(f)


def _make_candidate_power(a, n, reverse=False):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    if reverse:

    else:

    f = np.maximum(f, 1e-10)

    return _density_to_logit(f)


def _make_candidate_poly(rng, n, deg=4):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    t = x - 0.5
    coef = rng.normal(0.0, 1.0, size=deg + 1)
    f = np.ones(n, dtype=np.float64)
    val = np.zeros(n, dtype=np.float64)
    for k in range(deg, -1, -1):
        if k == deg:
            val = val * t + coef[k]
        else:
            val = val * t + coef[k]
    f = 1.0 + 0.5 * val
    f = np.maximum(f, 1e-10)
    f /= np.sum(f)
    return _density_to_logit(f)


def _make_candidate_sinusoidal(rng, n):
    x = np.arange(n, dtype=np.float64) / n
    f = np.ones(n, dtype=np.float64)
    for _ in range(rng.integers(2, 5)):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        f += amp * np.sin(2 * np.pi * freq * x + phase)
    f = np.maximum(f, 1e-10)
    f /= np.sum(f)
    return _density_to_logit(f)


def _score_logit(z, n):
    p = _probs_from_logit(z)
    s = _score_p(p)
    _update_best(p, s)
    return s


def _peak_directed_polish(p, n, deadline, rng):
    """
    Directly manipulate the sequence to reduce the top convolution peaks.
    Uses a subgradient-descent-like approach targeting specific peaks.
    """
    global _search_deadline
    _search_deadline = float(deadline)

    z = _density_to_logit(p)
    score = _score_p(p)
    _update_best(p, score)

    # Adaptive parameters
    lr = 0.1
    topk = 6
    fails = 0
    max_fails = 200

    while time.monotonic() < deadline and fails < max_fails:
        if time.monotonic() >= _search_deadline:
            break

        p = _probs_from_logit(z)
        c = _self_conv_fft(p)
        cmax = float(np.max(c))

        # Get top-k peaks
        idx = np.argpartition(c, -topk)[-topk:]
        # Weight by peak values
        w = np.zeros_like(c)
        wv = c[idx] + 1e-15
        w[idx] = wv / float(np.sum(wv))

        # Compute gradient direction
        gp = _grad_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        if gnorm < 1e-15:
            fails += 1
            continue

        direction = gz / gnorm
        accepted = False

        # Backtracking line search
        for step in (lr, lr * 0.5, lr * 0.25, lr * 0.125, lr * 0.0625):
            zt = z - step * direction
            zt = np.clip(zt, -50.0, 50.0)
            pt = _probs_from_logit(zt)
            st = _score_p(pt)
            if st < score - 1e-14:
                z = zt
                p = pt
                score = st
                _update_best(p, score)
                accepted = True
                break

        if accepted:
            fails = 0
            lr = min(0.3, lr * 1.1)
        else:
            # Try random perturbation
            best_local = score
            best_local_z = z
            for _ in range(4):
                zt = z + rng.normal(0.0, 0.05, n - 1)
                zt = np.clip(zt, -50.0, 50.0)
                pt = _probs_from_logit(zt)
                st = _score_p(pt)
                if st < best_local:
                    best_local = st
                    best_local_z = zt
            if best_local < score - 1e-14:
                z = best_local_z
                score = best_local
                _update_best(_probs_from_logit(z), score)
                accepted = True

            if not accepted:
                fails += 1
                lr = max(0.01, lr * 0.7)

    return score

    alphas = [2e5, 5e5, 1e6, 3e6, 8e6, 1.5e7]
    for i, alpha in enumerate(alphas):
        if time.monotonic() >= phase2_deadline:
            break
        rem = phase2_deadline - time.monotonic()
        if rem < 10.0:
            break
        max_fun = int(min(500, max(50, rem / (6 - i) / 0.02)))
        # Add small perturbation for diversity in later runs
        if i > 0:
            z_current = z_current + rng.normal(0.0, 0.02, N - 1)
            z_current = np.clip(z_current, -50.0, 50.0)
        _optimize_logit(z_current, alpha, N, max_fun, phase2_deadline - 2.0)
        p_current = _probs_from_logit(z_current)
        s = _score_p(p_current)
        print(f"  Phase 2 iter {i}: alpha={alpha:.0e}, score={s:.6f}, best={_best_score:.6f}")
        z_current = _density_to_logit(_best_p) if _best_p is not None else z_current

    print(f"After phase 2: best={_best_score:.6f} (time={time.monotonic()-start_time:.0f}s)")

    # --- Phase 3: Peak-directed polishing ---
    phase3_deadline = end_time - 30.0
    if _best_p is not None and time.monotonic() < phase3_deadline:
        p_polish = _best_p.copy()
        if len(p_polish) != N:
            old_x = np.arange(len(p_polish), dtype=np.float64)
            new_x = np.arange(N, dtype=np.float64) * (len(p_polish) - 1) / (N - 1)
            p_polish = np.interp(new_x, old_x, p_polish)
            p_polish = np.maximum(p_polish, 0.0)
            p_polish /= np.sum(p_polish)
        _peak_directed_polish(p_polish, N, phase3_deadline, rng)
        print(f"After phase 3: best={_best_score:.6f} (time={time.monotonic()-start_time:.0f}s)")

    # --- Phase 4: Final short L-BFGS-B if time remains ---
    if _best_p is not None and time.monotonic() < end_time - 15.0:
        z_final = _density_to_logit(_best_p)
        rem = end_time - 15.0 - time.monotonic()
        if rem > 10.0:
            _search_deadline = end_time - 5.0
            _optimize_logit(z_final, 1e7, N, max(30, int(rem / 0.03)), end_time - 5.0)
            print(f"After phase 4: best={_best_score:.6f}")

    # --- Construct final output ---
    if _best_p is not None:
        p_out = _best_p.copy()
        p_out = np.where(np.isfinite(p_out), p_out, 0.0)
        p_out = np.clip(p_out, 0.0, 1000.0)
        s = float(np.sum(p_out))
        if s < 0.01:
            p_out = np.ones(N) / N
            s = 1.0
        # Scale so that sum(a) = 1 (the evaluate_sequence handles any positive sum)
        seq = [float(v) for v in p_out]
    else:
        seq = [1.0 / N] * N

    # Final evaluation
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    # Safety fallback
    if not np.isfinite(final_score) or final_score > 2.0:
        seq = [1.0 / N] * N
        final_score = 2.0

    # Try inherited one more time
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, None)
            if float(np.sum(base)) > 0.01:
                base_score = float(evaluate_sequence([float(v) for v in base]))
                if np.isfinite(base_score) and base_score < final_score:
                    seq = [float(v) for v in base]
                    final_score = base_score
    except Exception:
        pass

    print(f"FINAL: score={final_score:.6f}, length={len(seq)}")
    return seq
