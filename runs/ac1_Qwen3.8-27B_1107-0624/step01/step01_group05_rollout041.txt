def _fft_conv(a):
    """Self full-convolution via FFT, returns length 2N-1."""
    fa = np.fft.rfft(a, FFT_N)
    c = np.fft.irfft(fa * fa, FFT_N)
    np.maximum(c, 0.0, out=c)
    return c[:2 * N - 1]


def _fft_conv_pair(a, b):
    """Full convolution of a and b via FFT."""
    fa = np.fft.rfft(a, FFT_N_GRAD)
    fb = np.fft.rfft(b, FFT_N_GRAD)
    return np.fft.irfft(fa * fb, FFT_N_GRAD)[:2 * N - 1]

    return 2.0 * float(N) * float(np.max(c))

    # gp[j] = 2 * sum_{k=j}^{j+N-1} w[k] * p[k-j] (where p is extended with 0s)
    # = 2 * (w_shifted * p)[0] shifted... 
    # Let's just use: gp = 2 * ifft(fft(w, N2) * fft(p, N2)).real[N-1:2N-1]... no.
    
    # OK let me think step by step. We have:
    # c[k] = sum_i p[i] * p[k-i] (full conv, k=0..2N-2)
    # dc/dp[j] = 2 * p[k-j] (for fixed k, if 0 <= k-j < N)
    # So grad_obj = 2 * sum_k w[k] * p[k-j] (masking)
    # = 2 * (w_cross_conv_p)[j] where w_cross_conv_p[j] = sum_k w[k]*p[k-j]
    # This is the cross-correlation of w and p!
    # cross_corr[j] = sum_k w[k] * p[k-j]
    # = conv(w_flipped, p)[j + len(w) - len(p)]... 
    # Actually: cross_corr = ifft(fft(w) * conj(fft(p)))... no that's for DFT convention.
    # numpy.correlate(w, p, 'full')[offset] gives this.
    # But w has length 2N-1 and p has length N.
    # correlate(w, p, 'full') has length (2N-1) + N - 1 = 3N-2.
    # correlate(w, p, 'full')[2N-1 - 1 + j] = ... hmm.
    
    # Let me just do it with a direct FFT approach:
    # Define W = fft(w, size=M) and P = fft(p, size=M) where M >= 3N-2.
    # Then ifft(W * P).real gives the full convolution w*p (length M, first 3N-2 meaningful).
    # (w*p)[m] = sum_k w[k] * p[m-k]
    # We want sum_k w[k] * p[k-j] = (w*p_flipped)[j] where p_flipped[i] = p[-i]
    # (w * p_flipped)[j] = sum_k w[k] * p_flipped[j-k] = sum_k w[k] * p[k-j] (if p_flipped[i]=p[-i])
    # But p_flipped would be p[-j+k]... let me just use:
    # sum_k w[k]*p[k-j] = (w * rev_p)[j + N - 1] where rev_p[i] = p[N-1-i]
    # Because: (w * rev_p)[m] = sum_k w[k] * rev_p[m-k] = sum_k w[k] * p[N-1-m+k]
    # Set N-1-m+k = k'-j => k' = k + m - N + 1... 
    # I'm going in circles. Let me just use the padding trick:
    
    # gp[j] = 2 * sum_k w[k] * p[k-j] for 0 <= k-j < N
    # = 2 * sum_{j'} w[j+j']*p[j'] where j' = k-j, so k = j+j'
    # = 2 * (w_shifted * p)[0]... 
    
    # You know what, let me just compute it via the full convolution of w and reversed_p:
    rev_p = p[::-1]  # rev_p[i] = p[N-1-i]
    # full_conv = convolve(w, rev_p) has length (2N-1) + N - 1 = 3N-2
    # full_conv[m] = sum_k w[k] * rev_p[m-k] = sum_k w[k] * p[N-1-(m-k)] = sum_k w[k] * p[N-1-m+k]
    # We want gp[j]/2 = sum_k w[k] * p[k-j]
    # Set N-1-m+k = k-j => N-1-m = -j => m = N-1+j
    # So gp[j] = 2 * full_conv[N-1+j]
    # But we need to check that the sum is over valid k (0 <= k < 2N-1) AND p index valid (0 <= k-j < N).
    # The full_conv already handles the w validity. For p: N-1-m+k = k-j, and we need 0 <= k-j < N.
    # In the full convolution, rev_p is only non-zero for indices 0..N-1, so p[N-1-m+k] is only
    # non-zero when 0 <= N-1-m+k < N, i.e., m-N+1 <= k <= m. Combined with 0 <= k < 2N-1.
    # For m = N-1+j: valid k range is max(0, j) to min(2N-2, N-1+j). And p index = k-j ranges
    # from max(0,j)-j to min(2N-2,N-1+j)-j = max(0,0) to min(2N-2-j, N-1) = 0 to min(2N-2-j, N-1).
    # So p index ranges from 0 to min(2N-2-j, N-1), which is valid (0 to N-1). 
    
    M = 1 << (3 * N - 2).bit_length()
    fw = np.fft.rfft(w, M)
    frp = np.fft.rfft(rev_p, M)
    fc = np.fft.irfft(fw * frp, M).real
    gp = 2.0 * fc[N - 1:2 * N - 1]
    
    # Now convert gradient from p-space to z-space.
    # p[0] = exp(0)/Z, p[j] = exp(z_j)/Z where Z = 1 + sum exp(z_i)
    # dp[0]/dz_j = -p[0]*p[j]
    # dp[j]/dz_j = p[j]*(1-p[j])
    # dp[j]/dz_i = -p[j]*p[i] for i != j
    # dJ/dz_j = sum_i dJ/dp_i * dp_i/dz_j
    # = dJ/dp_0 * (-p[0]*p[j]) + dJ/dp_j * p[j]*(1-p[j]) + sum_{i!=j} dJ/dp_i * (-p[j]*p[i])
    # = -p[0]*p[j]*gp[0] + p[j]*(1-p[j])*gp[j] - p[j]*(sum_i gp_i * p_i - gp[j]*p[j])
    # = -p[0]*p[j]*gp[0] + p[j]*gp[j] - p[j]^2*gp[j] - p[j]*sum_i gp_i*p_i + p[j]^2*gp[j]
    # = -p[0]*p[j]*gp[0] + p[j]*gp[j] - p[j]*sum_i gp_i*p_i
    # = p[j] * (gp[j] - p[0]*gp[0] - sum_i gp_i*p_i)
    # = p[j] * (gp[j] - sum_i gp_i*p_i - p[0]*gp[0])
    
    # Wait, let me redo. dJ/dz_j for j >= 1:
    # dp[i]/dz_j for i=0: -p[0]*p[j]
    # dp[i]/dz_j for i=j: p[j]*(1-p[j])
    # dp[i]/dz_j for i != 0,j: -p[i]*p[j]
    
    # dJ/dz_j = gp[0]*(-p[0]*p[j]) + gp[j]*p[j]*(1-p[j]) + sum_{i not 0,j} gp[i]*(-p[i]*p[j])
    # = -p[0]*p[j]*gp[0] + p[j]*gp[j]*(1-p[j]) - p[j]*sum_{i not 0,j} gp[i]*p[i]
    # = -p[0]*p[j]*gp[0] + p[j]*gp[j] - p[j]^2*gp[j] - p[j]*(sum_i gp[i]*p[i] - gp[0]*p[0] - gp[j]*p[j])
    # = -p[0]*gp[0]*p[j] + p[j]*gp[j] - p[j]^2*gp[j] - p[j]*sum_i gp[i]*p[i] + p[j]*gp[0]*p[0] + p[j]*gp[j]*p[j]
    # = p[j]*gp[j] - p[j]*sum_i gp[i]*p[i]
    # = p[j] * (gp[j] - dot(gp, p))
    
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    return float(2.0 * N * J), gz


def _optimize_lbfgs(z0, alpha_ls, max_fun, state, deadline):
    global _deadline
    _deadline = deadline
    x0 = np.clip(np.where(np.isfinite(z0), z0, 0.0), -50.0, 50.0)
    bounds = [(-50.0, 50.0)] * (N - 1)
    try:
        minimize(
            _objective_and_grad, x0, args=(alpha_ls, state),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxfun': max_fun, 'maxiter': max(50, max_fun // 2),
                     'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 20, 'disp': False}
        )
    except Exception:
        pass


def _subgradient_polish(p0, state, deadline):
    """Active-set subgradient method on the probability simplex."""
    p = np.array(p0, copy=True)
    p = np.maximum(p, 0.0)
    s = np.sum(p)
    if s < 1e-15:
        p = np.ones(N) / N
    else:
        p /= s
    score = _score_from_conv(_fft_conv(p))
    state.update(_rotf(p), score)
    
    step = 0.05
    fails = 0
    patience = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() < deadline:
        c = _fft_conv(p)
        cmax = float(np.max(c))
        
        # Find top-K active peaks
        topk = min(16, 2 * N - 1)
        if cmax > 0:
            threshold = 0.85 * cmax
            active = np.where(c >= threshold)[0]
            if len(active) == 0:
                idx = np.argpartition(c, -topk)[-topk:]
            else:
                idx = active
        else:
            idx = np.array([0])
        
        # Weight by value
        wv = c[idx] + 1e-15
        w = wv / np.sum(wv)
        
        # Compute gradient: gp[j] = 2 * sum_k w[k] * p[k-j]
        rev_p = p[::-1]
        w_full = np.zeros(2 * N - 1)
        w_full[idx] = w
        M = 1 << (3 * N - 2).bit_length()
        fw = np.fft.rfft(w_full, M)
        frp = np.fft.rfft(rev_p, M)
        fc = np.fft.irfft(fw * frp, M).real
        gp = 2.0 * fc[N - 1:2 * N - 1]
        
        gnorm = np.linalg.norm(gp)
        
        if gnorm > 1e-14:
            direction = -gp / gnorm
            accepted = False
            for step_try in (step, step * 0.5, step * 0.25, step * 0.125, step * 0.0625):
                p_new = p + step_try * direction
                p_new = np.maximum(p_new, 0.0)
                sn = np.sum(p_new)
                if sn < 1e-15:
                    continue
                p_new /= sn
                s_new = _score_from_conv(_fft_conv(p_new))
                if s_new < score - 1e-13:
                    p = p_new
                    score = s_new
                    state.update(_rotf(p), score)
                    accepted = True
                    break
            
            if accepted:
                step = min(0.3, step * 1.08)
                fails = 0
                patience = 0
            else:
                fails += 1
                patience += 1
                step = max(0.001, step * 0.7)
                
                if patience > 80:
                    # Try random perturbation
                    best_local = score
                    best_p_local = p.copy()
                    for _ in range(5):
                        p_pert = p + rng.normal(0, 0.08, N)
                        p_pert = np.maximum(p_pert, 0)
                        sn = np.sum(p_pert)
                        if sn > 0:
                            p_pert /= sn
                            s_pert = _score_from_conv(_fft_conv(p_pert))
                            if s_pert < best_local:
                                best_local = s_pert
                                best_p_local = p_pert
                    if best_local < score - 1e-13:
                        p = best_p_local
                        score = best_local
                        state.update(_rotf(p), score)
                        fails = 0
                        patience = 0
                        step = 0.05
                    elif patience > 300:
                        break
                elif fails > 200:
                    break
        else:
            break


def _rotf(p):
    """Convert probability vector to logits (for state tracking)."""
    la = np.log(np.maximum(p, 1e-300))
    z = la[1:] - la[0]
    return np.clip(z, -50.0, 50.0)


def _project_simplex(v):
    """Project v onto probability simplex (non-negative, sum=1)."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def _generate_candidates(rng):
    """Generate diverse initial candidates."""
    cands = []
    
    # Geometric sequences (theoretically motivated)
    for lg_q in np.linspace(-8, -0.1, 30):
        q = np.exp(lg_q)
        cands.append(_geometric_z(q))
        cands.append(_power_rev_z(-lg_q / max(1, N / 1000)))  # reversed power
    
    # Power laws
    for alpha in (-3, -2, -1.5, -1, -0.5, 0.5, 1, 1.5, 2, 3, 5, 10):
        cands.append(_power_z(alpha))
        cands.append(_power_rev_z(-alpha))
    
    # Products
    for a in (-1, -0.5, 0, 0.5, 1, 2):
        for b in (-1, -0.5, 0, 0.5, 1, 2):
            cands.append(_product_z(a, b))
    
    # Gaussian modulated
    for alpha in (0, 1, 2):
        for center in (0.2, 0.3, 0.5, 0.7, 0.8):
            for width in (0.1, 0.2, 0.3, 0.5):
                cands.append(_gauss_mod_z(alpha, center * N, width * N, 1.0))
    
    # Slightly perturbed uniform
    for eps in (0.1, 0.3, 0.5, 1.0, 2.0):
        x = np.arange(N, dtype=np.float64) / N
        for sign in (1, -1):
            f = 1.0 + sign * eps * (x - 0.5)
            f = np.maximum(f, 1e-6)
            p = f / np.sum(f)
            cands.append(_rotf(p))
    
    return cands


def _candidate_from_array(arr):
    """Convert an array to logits representation."""
    arr = np.asarray(arr, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = np.sum(arr)
    if s < 1e-15:
        return np.zeros(N - 1)
    arr /= s
    if arr.size != N:
        # Resample to N
        old_x = np.linspace(0, 1, arr.size)
        new_x = np.linspace(0, 1, N)
        arr = np.interp(new_x, old_x, arr)
        arr = np.maximum(arr, 0)
        s2 = np.sum(arr)
        if s2 > 0:
            arr /= s2
    return _rotf(arr)


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _state, _deadline
    
    start = time.monotonic()
    budget_s = float(budget_s)
    end_time = start + max(1.0, budget_s) - 2.0
    
    _state = _SearchState()
    _deadline = start
    
    rng = np.random.default_rng(seed)
    
    # Start with uniform
    _score_z(np.zeros(N - 1))
    _state.update(np.zeros(N - 1), _state.best_score)
    
    print(f"Initial (uniform): score={_state.best_score:.6f}")
    
    # Try to load previous best
    try:
        prev = height_sequence_1
        prev = np.asarray(prev, dtype=np.float64)
        if prev.size > 0:
            z_prev = _candidate_from_array(prev)
            s_prev = _score_z(z_prev)
            _state.update(z_prev, s_prev)
            print(f"Previous best: score={_state.best_score:.6f}")
    except Exception:
        pass
    
    # Generate and score candidates
    cands = _generate_candidates(rng)
    
    # Add perturbations of best
    if _state.best_z is not None:
        for eps in (0.05, 0.1, 0.2, 0.5, 1.0):
            cands.append(_state.best_z + rng.normal(0, eps, N - 1))
    
    # Score all candidates
    scored = []
    for i, z in enumerate(cands):
        z = np.clip(np.where(np.isfinite(z), z, 0.0), -50.0, 50.0)
        s = _score_z(z)
        scored.append((s, z))
        if (i + 1) % 50 == 0:
            print(f"  scored {i+1}/{len(cands)} candidates, best={_state.best_score:.6f}")
    
    scored.sort(key=lambda x: x[0])
    
    # Phase 1: L-BFGS-B optimization on top candidates
    phase1_deadline = start + min(600.0, 0.4 * budget_s)
    n_opt = min(30, len(scored))
    
    for i in range(n_opt):
        if time.monotonic() >= phase1_deadline:
            break
        rem = phase1_deadline - time.monotonic()
        if rem < 15:
            break
        
        s_init, z_init = scored[i]
        alpha_ls = 5e5 if _state.best_score > 1.6 else (2e6 if _state.best_score > 1.52 else 8e6)
        
        # Adaptive maxfun
        if rem > 30:
            max_fun = int(min(600, rem / 0.05))
        else:
            max_fun = 100
        
        z_opt = _rotf(_z_to_prob(z_init) + rng.normal(0, 0.02, N))
        _optimize_lbfgs(z_opt, alpha_ls, max_fun, _state, phase1_deadline)
        
        if i % 5 == 0 or i == n_opt - 1:
            print(f"  L-BFGS {i+1}/{n_opt}: best={_state.best_score:.6f} t={time.monotonic()-start:.0f}s")
    
    # Phase 2: Subgradient polishing
    phase2_deadline = start + min(1200.0, 0.75 * budget_s)
    if _state.best_p is not None and time.monotonic() < phase2_deadline:
        _subgradient_polish(_state.best_p, _state, phase2_deadline)
        print(f"After subgradient polish: best={_state.best_score:.6f} t={time.monotonic()-start:.0f}s")
    
    # Phase 3: More L-BFGS from polished + perturbations
    phase3_deadline = end_time - 30
    if _state.best_z is not None and time.monotonic() < phase3_deadline:
        attempts = 0
        while time.monotonic() < phase3_deadline and attempts < 20:
            attempts += 1
            rem = phase3_deadline - time.monotonic()
            if rem < 10:
                break
            noise = 0.03 + 0.05 * attempts
            z_try = _state.best_z + rng.normal(0, min(noise, 0.5), N - 1)
            alpha_ls = 1e7
            max_fun = int(min(400, rem / 0.04))
            max_fun = max(80, max_fun)
            _optimize_lbfgs(z_try, alpha_ls, max_fun, _state, phase3_deadline)
        
        print(f"After phase 3: best={_state.best_score:.6f} t={time.monotonic()-start:.0f}s")
    
    # Phase 4: Final subgradient polish
    if _state.best_p is not None and time.monotonic() < end_time - 5:
        _subgradient_polish(_state.best_p, _state, end_time - 5)
        print(f"Final polish: best={_state.best_score:.6f} t={time.monotonic()-start:.0f}s")
    
    # Convert to sequence
    if _state.best_z is not None:
        p = _z_to_prob(_state.best_z)
    else:
        p = np.ones(N) / N
    
    # Scale to ensure sum >= 0.01 and values <= 1000
    p = np.clip(p, 0.0, 1000.0)
    s = np.sum(p)
    if s < 0.01:
        p = p * (0.01 / max(s, 1e-15))
        p = np.clip(p, 0.0, 1000.0)
    
    seq = [float(v) for v in p]
    
    # Verify with evaluate_sequence
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')
    
    if not np.isfinite(final_score):
        # Fallback to uniform
        seq = [1.0 / N] * N
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            final_score = 2.0
    
    # Check if previous is better
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size > 0:
            prev = np.where(np.isfinite(prev), prev, 0.0)
            prev = np.clip(prev, 0.0, 1000.0)
            prev_score = float(evaluate_sequence([float(v) for v in prev]))
            if np.isfinite(prev_score) and prev_score < final_score:
                seq = [float(v) for v in prev]
                final_score = prev_score
    except Exception:
        pass
    
    print(f"FINAL: score={final_score:.6f}, length={len(seq)}")
    return seq
