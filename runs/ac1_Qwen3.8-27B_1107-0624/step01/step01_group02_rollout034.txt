    c = np.convolve(a, a)
    return 2.0 * n * c.max()

def fft_convolve(a):

    n = len(a)

    while size < 2*n - 1:

    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

"""
Step function optimization minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses subgradient descent with momentum, FFT-accelerated convolution,
LP-based refinement, and multi-start with diverse asymmetric initializations.
Key insight: asymmetric (monotonic) sequences reduce the center convolution peak.
"""
import numpy as np
import time

try:
    from scipy.optimize import linprog
    HAS_LP = True
except Exception:
    HAS_LP = False


def normalize_nonneg(a):
    """Normalize to sum 1, non-negative."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s


def fft_convolve(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:need]


def score_norm(a):
    """Score = 2*n*max(a*a) for sum-1 nonnegative a."""
    n = len(a)
    c = fft_convolve(a)
    return 2.0 * n * float(c.max())


def exact_subgradient(a, c, top_k=None):
    """
    Exact subgradient of max_k (a*a)_k at the active set.
    g[i] = (2/|K*|) * sum_{k in K*} a[k-i] (if 0<=k-i<n)
    Equivalent to cross-correlation of a with indicator of active set.
    """
    n = len(a)
    m = len(c)
    M = c.max()
    if top_k is None:
        top_k = min(m, max(32, n // 2))
    # Select active: top-k by value, but at least all near-max
    if top_k >= m:
        active = np.arange(m)
    else:
        idx = np.argpartition(c, -top_k)[-top_k:]
        active = idx
    ka = len(active)

    # g[i] = (2/ka) * sum_{j=0}^{n-1} a[j] * P[j+i] where P = 1_{active}
    P = np.zeros(m)
    P[active] = 1.0

    # Cross-correlation via FFT: g[i] = sum_j a[j] P[j+i]
    # = (a_rev * P)[n-1+i] where a_rev = a[::-1]
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    A_rev = np.fft.rfft(a[::-1], size)
    P_f = np.fft.rfft(P, size)
    conv = np.fft.irfft(A_rev * P_f, size)
    g = conv[n - 1: n - 1 + n]
    return (2.0 / ka) * g


def lp_direction(f, c, max_rows=300):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k rows, g >= 0."""
    if not HAS_LP or len(f) > 2048:
        return None
    n = len(f)
    m = len(c)
    M = float(c.max())
    if M < 1e-15:
        return None
    k = min(max_rows, m)
    rows = np.argpartition(c, -k)[-k:]

    # Build constraint matrix: A[r, j] = f[k-r - j] if valid
    A = np.zeros((k, n))
    for r, kk in enumerate(rows):
        kk = int(kk)
        jlo = max(0, kk - n + 1)
        jhi = min(n, kk + 1)
        if jhi > jlo:
            A[r, jlo:jhi] = f[kk - np.arange(jlo, jhi)]

    A /= M
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.ones(k),
                      bounds=(0, None), method='highs',
                      options={'time_limit': 30})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, None)
            if g.sum() > 1.0 + 1e-6:
                return g
    except Exception:
        pass
    return None


def subgradient_trajectory(a, time_limit, rng, use_lp=True, lp_every=5):
    """
    Main trajectory: subgradient with momentum + optional LP refinement.
    """
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_norm(a)
    print(f"  traj start: score={best_score:.10f} n={n}", flush=True)

    deadline = time.monotonic() + time_limit
    momentum = np.zeros(n)
    mu = 0.7  # momentum coefficient
    step = 0.1
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(a)
        cur = 2.0 * n * float(c.max())

        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
            step = max(0.01, step * 0.95)
        else:
            no_improve += 1
            if no_improve > 20:
                step = min(0.5, step * 1.3)

        # Compute subgradient
        top_k = min(len(c), max(16, n // 4))
        if no_improve > 30:
            top_k = min(len(c), max(32, n // 2))
        g = exact_subgradient(a, c, top_k=top_k)

        # Center the gradient (project to sum-0 direction to preserve sum)
        g = g - g.mean()

        # Momentum update
        momentum = mu * momentum + (1 - mu) * g
        grad_norm = np.linalg.norm(momentum)
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        alpha = step / grad_norm
        # Try a few step sizes
        best_h = None
        best_sc = cur
        for t in (alpha, 2 * alpha, 0.5 * alpha, 4 * alpha):
            h = a - t * momentum
            h = np.clip(h, 0, None)
            h = normalize_nonneg(h)
            sc = score_norm(h)
            if sc < best_sc:
                best_sc = sc
                best_h = h
        if best_h is not None:
            a = best_h
        else:
            # Random escape
            a = normalize_nonneg(np.clip(a * np.exp(rng.normal(0, 0.1, n)), 0, None))

        # LP refinement periodically
        if use_lp and HAS_LP and it % lp_every == 0 and n <= 2048:
            mx = min(200, 2 * n)
            g_lp = lp_direction(a, fft_convolve(a), max_rows=mx)
            if g_lp is not None:
                Sg = g_lp.sum()
                if Sg > 1.01:
                    g_norm = g_lp / Sg
                    for t in (0.05, 0.1, 0.2, 0.4, 0.7, 1.0):
                        h = (1 - t) * a + t * g_norm
                        h = normalize_nonneg(h)
                        sc = score_norm(h)
                        if sc < best_sc:
                            best_sc = sc
                            a = h
                            best_h = h
                    if a is not best_h:
                        a = best_h

    return best_a, best_score


def make_linear_ramp(N, power=1.0, reverse=False):
    """Power-law ramp: a_i = (i+1)^power / sum."""
    x = np.arange(1, N + 1, dtype=np.float64)
    a = x ** power
    if reverse:
        a = a[::-1]
    return normalize_nonneg(a)


def make_exponential_ramp(N, alpha=1.0, reverse=False):
    """Exponential ramp."""
    x = np.arange(N, dtype=np.float64)
    a = np.exp(alpha * x / N)
    if reverse:
        a = a[::-1]
    return normalize_nonneg(a)


def make_two_block(N, frac=0.5, mass_ratio=0.5):
    """Two adjacent constant blocks."""
    L1 = max(1, int(N * frac))
    L2 = N - L1
    a = np.zeros(N)
    a[:L1] = mass_ratio / L1
    a[L1:] = (1 - mass_ratio) / L2
    return normalize_nonneg(a)


def make_triangle_asym(N, peak_pos=0.7):
    """Asymmetric triangle: rises to peak at peak_pos fraction, then falls."""
    x = np.linspace(0, 1, N)
    p = peak_pos
    a = np.where(x <= p, (x / p) if p > 0 else 0,
                 (1 - x) / (1 - p) if p < 1 else 0)
    return normalize_nonneg(np.maximum(a, 0))
