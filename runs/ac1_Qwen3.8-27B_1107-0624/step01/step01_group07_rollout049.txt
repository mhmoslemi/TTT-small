   """
   Smooth-gradient + LP hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
   Uses log-sum-exp smoothed gradient with momentum for fast convergence,
   followed by LP-based refinement for precision.
   """
   import numpy as np
   import time
   from scipy.optimize import linprog
   from scipy.fft import rfft, irfft


   def fast_conv(a, b):
       """Fast convolution using FFT."""
       n = len(a) + len(b) - 1
       size = 1 << (n - 1).bit_length()
       fa = rfft(a, size)
       fb = rfft(b, size)
       return np.real(irfft(fa * fb, size)[:n])


   def compute_objective(a):
       """Compute 2*n*max(conv(a,a))/sum(a)^2."""
       n = len(a)
       S = a.sum()
       if S < 1e-10:
           return np.inf
       c = fast_conv(a, a)
       return float(2.0 * n * c.max() / (S * S))


   def project_simplex(v):
       """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
       n = len(v)
       u = np.sort(v)[::-1]
       cv = np.cumsum(u)
       cond = u * np.arange(1, n + 1) > (cv - 1)
       if not cond.any():
           return np.ones(n) / n
       rho = np.nonzero(cond)[0][-1]
       theta = (cv[rho] - 1) / (rho + 1.0)
       return np.maximum(v - theta, 0)


   def smooth_gradient(a, T):
       """Compute smooth gradient of max(conv(a,a)) using log-sum-exp."""
       n = len(a)
       conv = fast_conv(a, a)
       M = conv.max()
       # Log-sum-exp weights
       shifted = (conv - M) / T
       # Clip for numerical stability
       shifted = np.clip(shifted, -50, 50)
       exp_vals = np.exp(shifted)
       p = exp_vals / exp_vals.sum()
       
       # grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
       size = 1 << (3 * n - 2).bit_length()
       fp = rfft(p, size)
       fa = rfft(a, size)
       corr = np.real(irfft(fp * np.conj(fa), size))[:n]
       
       grad = 2.0 * corr
       return grad, M, conv


   def optimize_smooth_grad(a, deadline, rng, T_init=0.05, T_final=0.001,
                            lr_init=0.1, beta=0.9, verbose=False):
       """Optimize using smooth gradient descent with momentum."""
       n = len(a)
       a = a.astype(np.float64).copy()
       S = a.sum()
       if S < 1e-10:
           return a, np.inf
       a = a / S
       
       best_a = a.copy()
       best_score = compute_objective(a)
       
       v = np.zeros(n)  # momentum
       lr = lr_init
       T = T_init
       stall = 0
       it = 0

       while time.monotonic() < deadline:
           it += 1
           
           # Adaptive T: decrease over time
           progress = min(1.0, it / 5000.0)
           T = T_init * (T_final / T_init) ** progress
           
           grad, M, conv = smooth_gradient(a, T)
           
           # Normalize gradient for better conditioning
           grad_norm = np.linalg.norm(grad)
           if grad_norm > 1e-15:

           v = beta * v + (1 - beta) * grad

       A = np.zeros((2 * n - 1, n))
       for j in range(n):
