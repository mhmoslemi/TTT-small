def bb_step_size(s_prev, s_curr, g_prev, g_curr):
    """Barzilai-Borwein step size."""
    ds = s_curr - s_prev
    dg = g_curr - g_prev
    denom = np.dot(dg, dg)
    if denom < 1e-30:
        return 0.01
    num1 = np.dot(ds, dg)
    num2 = np.dot(ds, ds)
    # Use the "short" BB step
    if num1 < 1e-30:
        return 0.01
    eta1 = num2 / num1  # BB1
    eta2 = num1 / denom  # BB2
    # Use the one that's positive and reasonable
    candidates = [eta1, eta2]
    for eta in candidates:
        if 1e-10 < eta < 10.0:
            return eta
    return 0.01

    w = exp_shifted / exp_shifted.sum()
    # Gradient: g[j] = sum_k w[k] * s[k-j] for valid k-j
    # Use convolution: g = conv(w, s_reversed)[n-1 : 2n-2]
    s_rev = s[::-1]
    g_full = np.convolve(w, s_rev)
    g = g_full[n-1:2*n-1]
    return g, conv

def accelerated_subgrad(s_init, max_time, tau_init, tau_final, rng, momentum_init=0.9):
    """Accelerated projected subgradient with Nesterov momentum."""
    n = len(s_init)
    s = s_init.astype(np.float64).copy()
    s = np.maximum(s, 0)
    s = s / s.sum()
    
    best_s = s.copy()
    best_sc = fast_score(s)
    
    y = s.copy()  # Nesterov extrapolation point
    v = np.zeros(n)  # momentum
    momentum = momentum_init
    eta = 0.1 / n  # initial step size
    
    s_prev = None
    g_prev = None
    
    start = time.monotonic()
    iter_count = 0
    stall = 0
    
    max_iters = int(max_time * 50000)  # rough upper bound
    
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        
        # Anneal temperature and momentum
        progress = min(1.0, it / max_iters)
        tau = tau_init * (tau_final / tau_init) ** (0.5 * progress)  # sqrt annealing
        momentum = momentum_init * (1.0 - 0.3 * progress)
        
        # Nesterov extrapolation
        if it > 0:
            t_k = 2.0 / (it + 2.0)  # Nesterov parameter
            y_new = y + (t_k - 1.0) * (y - s)
        else:
            y_new = y
        
        # Compute gradient at y
        g, conv = compute_smoothed_grad(y, tau)
        
        # BB step size
        if s_prev is not None and g_prev is not None and it >= 20:
            ds = s - s_prev
            dg = g - g_prev
            denom = np.dot(dg, dg)
            num = np.dot(ds, dg)
            if denom > 1e-30 and abs(num) > 1e-30:
                eta_bb = num / denom
                if 1e-12 < eta_bb < 10.0:
                    eta = 0.5 * eta + 0.5 * eta_bb
        
        # Update
        s_new = y - eta * g
        s_new = proj_simplex(s_new, 1.0)
        
        # Check score
        sc = fast_score(s_new)
        if sc < best_sc:
            best_sc = sc
            best_s = s_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive step size: reduce if stalled
        if stall > 5000:
            eta *= 0.9
            if eta < 1e-10:
                eta = 1.0 / n
                # Random kick
                y = y + 0.05 * rng.standard_normal(n)
                y = np.maximum(y, 0)
                y = y / y.sum()
                eta = 0.1 / n
            stall = 0
        
        # Update momentum
        if it > 0:
            v = momentum * v + (1.0 - momentum) * (s - s_prev)
        
        s_prev = s
        g_prev = g
        s = s_new
        y = y_new
        iter_count += 1
    
    return best_s, best_sc

def lp_kick(s, top_k=10, time_limit=3.0):
    """LP-based direction finding using top_k constraints."""
    n = len(s)
    conv = np.convolve(s, s)
    M = conv.max()
    
    # Get top_k peak positions
    top_idx = np.argsort(conv)[::-1][:top_k]
    
    # Build constraint matrix
    A = np.zeros((top_k, n))
    for i, k in enumerate(top_idx):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A[i, j] = s[kj]
    
    c = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'time_limit': time_limit})
        if res.success and res.x is not None and res.x.sum() > 1e-10:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def search_one(s_init, budget, rng):
    """Run a full search from one initial point."""
    n = len(s_init)
    s = s_init.copy()
    s = np.maximum(s, 0)
    s = s / s.sum()
    
    best_s = s.copy()
    best_sc = fast_score(s)
    
    start = time.monotonic()
    phase = 0
    
    while time.monotonic() - start < budget:
        remaining = budget - (time.monotonic() - start)
        
        # Subgradient phase
        sg_time = min(30.0, remaining * 0.6)
        if sg_time < 2.0:
            break
        
        s, sc = accelerated_subgrad(s, sg_time, 
                                     tau_init=0.05, tau_final=0.002,
                                     rng=rng)
        if sc < best_sc:
            best_sc = sc
            best_s = s.copy()
            print(f"      Subgrad: {sc:.8f}")
        
        # LP kick
        if remaining > 10.0:
            b = lp_kick(s, top_k=10, time_limit=min(3.0, remaining*0.2))
            if b is not None:
                b = b / b.sum()
                best_t = 0
                best_ts = best_sc
                for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    cand = (1-t) * best_s + t * b
                    cand = np.maximum(cand, 0)
                    if cand.sum() < 1e-10:
                        continue
                    cand = cand / cand.sum()
                    ts = fast_score(cand)
                    if ts < best_ts:
                        best_ts = ts
                        best_t = t
                if best_t > 0:
                    s = (1-best_t) * best_s + best_t * b
                    s = np.maximum(s, 0)
                    s = s / s.sum()
                    if fast_score(s) < best_sc:
                        best_sc = fast_score(s)
                        best_s = s.copy()
                        print(f"      LP kick: {best_sc:.8f}")
        
        phase += 1
    
    return best_s, best_sc

def generate_candidates(n, rng, hseq=None):
    """Generate diverse initial candidates."""
    candidates = []
    
    # From hseq
    if hseq is not None:
        x_old = np.linspace(0, 1, len(hseq))
        x_new = np.linspace(0, 1, n)
        h = np.interp(x_new, x_old, hseq)
        h = np.maximum(h, 0)
        if h.sum() > 0.01:
            h = h / h.sum()
            candidates.append((fast_score(h), h))
    
    # Skewed exponential
    for alpha in np.linspace(-2, 2, 9):
        x = np.arange(n) / max(1, n-1)
        s = np.exp(alpha * (x - 0.5))
        s = np.maximum(s, 1e-6)
        s = s / s.sum()
        candidates.append((fast_score(s), s))
    
    # Linear ramp
    for beta in np.linspace(-1.5, 1.5, 7):
        x = np.arange(n) / max(1, n-1)
        s = 1.0 + beta * (2*x - 1)
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        candidates.append((fast_score(s), s))
    
    # Two-block
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for ratio in [1.5, 2, 3, 4]:
            s = np.ones(n)
            k = int(frac * n)
            s[:k] = ratio
            s = s / s.sum()
            candidates.append((fast_score(s), s))
    
    # Random
    for _ in range(5):
        s = rng.random(n) + 0.01
        s = s / s.sum()
        candidates.append((fast_score(s), s))
    
    candidates.sort(key=lambda x: x[0])
    return candidates
