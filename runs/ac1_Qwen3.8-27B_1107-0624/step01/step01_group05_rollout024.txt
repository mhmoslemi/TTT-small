    fb = np.fft.rfft(b, FFT_GRAD)
    return np.fft.irfft(fa * fb, FFT_GRAD).real


def grad_from_weights(p, w):
    conv = full_conv_padded(p[::-1], w)
    h = conv[N - 1:2 * N - 1]
    return 2.0 * h


def probs_from_z(z):
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -50.0, 50.0)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def density_to_z(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 1e-15)
    f = f / float(np.sum(f))
    logf = np.log(f)
    z = logf[1:] - logf[0]
    return np.clip(z, -50.0, 50.0)


def score_p(p):
    c = self_conv(p)
    return 2.0 * float(N) * float(np.max(c))


def update_best(z, p, sc):
    global _best_score, _best_p, _best_z
    if np.isfinite(sc) and sc < _best_score:
        _best_score = float(sc)
        _best_p = np.array(p, copy=True)
        _best_z = np.array(z, copy=True)


def score_z(z):
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -50.0, 50.0)
    p = probs_from_z(z)
    s = score_p(p)
    update_best(z, p, s)
    return s


def objective_and_grad(x, alpha):
    if time.monotonic() >= _search_deadline:
        raise SearchTimeout
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -50.0, 50.0)
    p = probs_from_z(x)
    c = self_conv(p)

    hard = 2.0 * float(N) * float(np.max(c))
    update_best(x, p, hard)

    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        J = cmax
        w = np.zeros_like(c)
        idx = int(np.argmax(c))
        w[idx] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s

    gp = grad_from_weights(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    return float(J), gz


def adam_optimize(z0, deadline, alpha_start=1e5, alpha_end=1e7, lr_start=0.05, lr_end=0.001, max_iter=2000):
    z = np.where(np.isfinite(z0), z0, 0.0).copy()
    z = np.clip(z, -50.0, 50.0)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_local = score_p(probs_from_z(z))
    no_improve = 0

    while t < max_iter and time.monotonic() < deadline:
        t += 1
        frac = t / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        x = np.clip(z, -50.0, 50.0)
        p = probs_from_z(x)
        c = self_conv(p)
        cmax = float(np.max(c))
        hard = 2.0 * N * cmax
        update_best(x, p, hard)

        u = alpha * (c - cmax)
        e = np.exp(u)
        s = float(np.sum(e))
        if s > 0 and np.isfinite(s):
            w = e / s
        else:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0

        gp = grad_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gz = np.where(np.isfinite(gz), gz, 0.0)

        m = beta1 * m + (1 - beta1) * gz
        v = beta2 * v + (1 - beta2) * gz ** 2
        mhat = m / (1 - beta1 ** t)
        vhat = v / (1 - beta2 ** t)
        z = z - lr * mhat / (np.sqrt(vhat) + eps)
        z = np.clip(z, -50.0, 50.0)

        p = probs_from_z(z)
        sc = score_p(p)
        if sc < best_local - 1e-10:
            best_local = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break


def subgradient_polish(z0, deadline, rng):
    z = np.where(np.isfinite(z0), z0, 0.0).copy()
    p = probs_from_z(z)
    score = score_p(p)
    update_best(z, p, score)

    noise = 0.1
    fails = 0
    topk = 16

    while time.monotonic() < deadline and fails < 200:
        c = self_conv(p)
        cmax = float(np.max(c))

        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros_like(c)
        wv = c[idx] + 1e-15
        w[idx] = wv / float(np.sum(wv))

        gp = grad_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        accepted = False
        if gnorm > 1e-14:
            direction = gz / gnorm
            for step in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
                zt = z - step * direction
                zt = np.clip(zt, -50.0, 50.0)
                pt = probs_from_z(zt)
                st = score_p(pt)
                if st < score - 1e-12:
                    z = zt
                    p = pt
                    score = st
                    update_best(z, p, score)
                    accepted = True
                    break

        if not accepted:
            best_local_score = score
            best_local_z = z
            best_local_p = p
            for _ in range(5):
                zt = z + rng.normal(0.0, noise, N - 1)
                zt = np.clip(zt, -50.0, 50.0)
                pt = probs_from_z(zt)
                st = score_p(pt)
                if st < best_local_score - 1e-12:
                    best_local_score = st
                    best_local_z = zt
                    best_local_p = pt
            if best_local_score < score - 1e-12:
                z = best_local_z
                p = best_local_p
                score = best_local_score
                update_best(z, p, score)
                accepted = True

        if accepted:
            fails = 0
            noise = min(0.3, noise * 1.3)
        else:
            fails += 1
            noise = max(0.01, noise * 0.75)


def init_hann(beta=1.0):
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    f = 0.5 * (1.0 + np.cos(2.0 * np.pi * x * beta))
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def init_gaussian(sigma_center, width):
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    f = np.exp(-((x - sigma_center) ** 2) / (2.0 * width * width))
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def init_logistic(center, steep):
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    f = 1.0 / (1.0 + np.exp(-steep * (x - center)))
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def init_bessel():
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    t = np.pi * x * 2.4
    from scipy.special import j0
    f = j0(t)
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def init_random_fourier(rng, n_freqs=8):
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    z = np.zeros(N, dtype=np.float64)
    for _ in range(n_freqs):
        fq = rng.integers(1, 12)
        amp = rng.normal(0.0, 1.5)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.sin(2.0 * np.pi * fq * x + ph)
    f = np.exp(0.5 * z)
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def init_multi_gaussian(rng, n_bumps=3):
    f = np.ones(N, dtype=np.float64) * 0.01
    for _ in range(n_bumps):
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.03, 0.15)
        amp = rng.uniform(0.3, 1.0)
        x = (np.arange(N, dtype=np.float64) + 0.5) / N
        f += amp * np.exp(-((x - center) ** 2) / (2.0 * width * width))
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def init_freq_shaping(rng):
    nfft = 2 * N
    phase = rng.uniform(0, 2 * np.pi, nfft // 2 + 1)
    amp = np.ones(nfft // 2 + 1)
    A = np.zeros(nfft, dtype=np.complex128)
    A[:nfft // 2 + 1] = amp * np.exp(1j * phase)
    A[nfft // 2 + 1:] = np.conj(A[1:nfft // 2][::-1])
    a = np.fft.ifft(A).real
    a = np.abs(a[:N])
    a = np.maximum(a, 1e-15)
    return density_to_z(a)


def init_zadoff_chu(m=1):
    n = N
    t = np.arange(n, dtype=np.float64)
    phase = -np.pi * m * t * (t + 1) / n
    a = np.abs(np.cos(phase))
    a = np.maximum(a, 1e-15)
    return density_to_z(a)


def init_power(a, reverse=False):
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    if reverse:
        f = np.power(1.0 - x + 1e-5, a)
    else:
        f = np.power(x + 1e-5, a)
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def init_exp(beta):
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    f = np.exp(beta * (x - 0.5))
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def init_squred_sine():
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    f = np.sin(np.pi * x) ** 2
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def init_flat_top(width):
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    f = np.ones(N, dtype=np.float64) * 0.01
    mask = np.abs(x - 0.5) < width
    f[mask] = 1.0
    f = f + 0.1 * np.sin(np.pi * x) ** 2
    f = np.maximum(f, 1e-15)
    return density_to_z(f)


def make_initial_candidates(rng):
    cands = []

    # Inherited best
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, None)
            if float(np.sum(base)) > 0:
                if base.size != N:
                    old = np.linspace(0.0, 1.0, base.size)
                    new = np.linspace(0.0, 1.0, N)
                    base = np.interp(new, old, base)
                cands.append(density_to_z(base))
                cands.append(density_to_z(base[::-1]))
    except Exception:
        pass

    # Uniform
    cands.append(np.zeros(N - 1, dtype=np.float64))

    # Hann windows
    for beta in (0.5, 1.0, 1.5, 2.0, 3.0):
        cands.append(init_hann(beta))

    # Squared sine
    cands.append(init_squred_sine())

    # Gaussian
    for sc in (0.3, 0.4, 0.5, 0.6, 0.7):
        for w in (0.05, 0.1, 0.15, 0.2, 0.3):
            cands.append(init_gaussian(sc, w))

    # Logistic
    for center in (0.3, 0.4, 0.5, 0.6, 0.7):
        for steep in (2.0, 4.0, 6.0, 8.0, 12.0, 16.0):
            cands.append(init_logistic(center, steep))
            cands.append(init_logistic(1.0 - center, -steep))

    # Bessel
    cands.append(init_bessel())

    # Power law
    for a in (0.5, 1.0, 1.5, 2.0, 3.0):
        cands.append(init_power(a, False))
        cands.append(init_power(a, True))

    # Exponential
    for b in (1.0, 2.0, 3.0, 5.0, 8.0):
        cands.append(init_exp(b))
        cands.append(init_exp(-b))

    # Flat top
    for w in (0.2, 0.3, 0.4, 0.45):
        cands.append(init_flat_top(w))

    # Zadoff-Chu
    for m in (1, 2, 3, 5):
        cands.append(init_zadoff_chu(m))

    # Random Fourier
    for _ in range(20):
        cands.append(init_random_fourier(rng, n_freqs=rng.integers(3, 12)))

    # Multi-Gaussian
    for _ in range(15):
        cands.append(init_multi_gaussian(rng, n_bumps=rng.integers(2, 6)))

    # Frequency shaping
    for _ in range(10):
        cands.append(init_freq_shaping(rng))

    # Random smooth
    for _ in range(15):
        x = (np.arange(N, dtype=np.float64) + 0.5) / N
        t = x - 0.5
        z = np.zeros(N, dtype=np.float64)
        z += rng.uniform(-3.0, 3.0) * t
        z += rng.uniform(-2.0, 2.0) * t ** 3
        for fq in (1, 2, 3, 5, 7, 11):
            amp = rng.normal(0.0, 2.0 / (fq + 0.5))
            ph = rng.uniform(0.0, 2.0 * np.pi)
            z += amp * np.sin(2.0 * np.pi * fq * x + ph)
        f = np.exp(0.5 * z)
        f = np.maximum(f, 1e-15)
        cands.append(density_to_z(f))

    return cands


def choose_alpha(score):
    if not np.isfinite(score):
        return 1e5
    if score > 1.80:
        return 1e5
    if score > 1.65:
        return 3e5
    if score > 1.55:
        return 1e6
    return 3e6


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_p, _best_z, _search_deadline

    start_time = time.monotonic()
    budget_s = float(budget_s)
    end_time = start_time + max(1.0, budget_s) - 2.0

    _best_score = float("inf")
    _best_p = None
    _best_z = None

    rng = np.random.default_rng(seed)

    # Generate initial candidates
    cands = make_initial_candidates(rng)

    # Score all candidates
    print(f"Scoring {len(cands)} initial candidates...")
    scored = []
    for i, z in enumerate(cands):
        sc = score_z(z)
        scored.append((sc, z))
    scored.sort(key=lambda x: x[0])
    print(f"Top 10 initial scores: {[f'{s:.6f}' for s, _ in scored[:10]]}")
    print(f"Best initial: {_best_score:.6f}")

    # Phase 1: L-BFGS-B from top candidates
    top_k = min(30, len(scored))
    lbfgs_deadline = start_time + budget_s * 0.65
    run_idx = 0

    while time.monotonic() < lbfgs_deadline:
        rem = lbfgs_deadline - time.monotonic()
        if rem < 15.0:
            break

        if run_idx < top_k:
            z0 = scored[run_idx][1]
        else:
            if _best_z is None:
                z0 = rng.normal(0.0, 1.0, N - 1)
            else:
                frac = max(0.0, min(1.0, rem / max(1.0, lbfgs_deadline - start_time)))
                noise = 0.05 + 0.3 * frac
                if run_idx % 10 == 0:
                    noise += 0.3
                z0 = _best_z + rng.normal(0.0, noise, N - 1)

        run_idx += 1
        alpha = choose_alpha(_best_score)

        rem_fraction = rem / max(1.0, lbfgs_deadline - start_time)
        if rem > 120.0:
            maxfun = 600
        elif rem > 60.0:
            maxfun = 400
        else:
            maxfun = 250

        _search_deadline = lbfgs_deadline
        bounds = [(-50.0, 50.0)] * (N - 1)

        if minimize is not None:
            x0 = np.clip(np.where(np.isfinite(z0), z0, 0.0), -50.0, 50.0)
            try:
                minimize(
                    objective_and_grad,
                    x0,
                    args=(float(alpha),),
                    method="L-BFGS-B",
                    jac=True,
                    bounds=bounds,
                    options={
                        "maxfun": maxfun,
                        "maxiter": maxfun // 2,
                        "ftol": 1e-14,
                        "gtol": 1e-10,
                        "maxls": 20,
                        "disp": False,
                    },
                )
            except SearchTimeout:
                pass
            except Exception:
                pass

        if run_idx % 5 == 0:
            print(f"  L-BFGS run {run_idx}: best={_best_score:.6f} elapsed={time.monotonic()-start_time:.1f}s")

    print(f"After L-BFGS: best={_best_score:.6f}")

    # Phase 2: Adam fine-tuning from best
    adam_deadline = start_time + budget_s * 0.82
    if _best_z is not None and time.monotonic() < adam_deadline:
        best_z_copy = np.array(_best_z, copy=True)
        adam_optimize(best_z_copy, adam_deadline, alpha_start=5e5, alpha_end=2e6, lr_start=0.03, lr_end=0.002, max_iter=3000)
        print(f"After Adam: best={_best_score:.6f}")

    # Phase 3: Subgradient polishing
    polish_deadline = end_time - 3.0
    if _best_z is not None and time.monotonic() < polish_deadline:
        subgradient_polish(_best_z, polish_deadline, rng)
        print(f"After subgradient: best={_best_score:.6f}")

    # Final L-BFGS pass with high alpha if time remains
    if _best_z is not None and minimize is not None:
        rem = end_time - 5.0 - time.monotonic()
        if rem > 30.0:
            z0 = _best_z + rng.normal(0.0, 0.03, N - 1)
            _search_deadline = end_time - 5.0
            try:
                minimize(
                    objective_and_grad,
                    np.clip(np.where(np.isfinite(z0), z0, 0.0), -50.0, 50.0),
                    args=(5e6,),
                    method="L-BFGS-B",
                    jac=True,
                    bounds=[(-50.0, 50.0)] * (N - 1),
                    options={"maxfun": 300, "maxiter": 150, "ftol": 1e-14, "gtol": 1e-10, "display": False},
                )
            except Exception:
                pass
            # Subgradient again
            rem2 = end_time - 3.0 - time.monotonic()
            if rem2 > 10.0:
                subgradient_polish(_best_z, end_time - 3.0, rng)

    # Construct final output
    if _best_z is None:
        p = np.ones(N, dtype=np.float64) / N
    else:
        z = np.clip(np.where(np.isfinite(_best_z), _best_z, 0.0), -50.0, 50.0)
        p = probs_from_z(z)
        p = np.clip(p, 0.0, 1000.0)

    seq = [float(v) for v in p]

    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score) or final_score > 2.000001:
        seq = [1.0 / N] * N
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            final_score = 2.0

    # Check inherited best
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            bs = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(bs) and bs < final_score:
                seq = [float(v) for v in base]
                final_score = bs
    except Exception:
        pass

    print(f"FINAL: score={final_score:.6f} length={len(seq)} time={time.monotonic()-start_time:.1f}s")
    return seq
