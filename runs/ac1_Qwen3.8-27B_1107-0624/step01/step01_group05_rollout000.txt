"""
Search for a non-negative sequence of length 4096 minimizing
    2 * n * max_k (a*a)[k] / (sum a)^2
using a continuation method with L-BFGS-B on a soft-max surrogate of the
maximum self-convolution coefficient, followed by active-peak subgradient
polishing with adaptive backtracking line search.

Key improvements over previous approach:
- Continuation (homotopy) on the soft-max parameter alpha
- Backtracking line search for subgradient steps
- Multiple active peaks with exponential weighting
- Periodic subgradient interleaving with L-BFGS
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

N = 4096
FFT_SELF = 1 << (2 * N - 2).bit_length()
FFT_GRAD = 1 << (3 * N - 2).bit_length()
X_GRID = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
BOUNDS = [(-35.0, 35.0)] * (N - 1)

_best_score = float("inf")
_best_z = None
_search_deadline = 0.0


class SearchTimeout(Exception):
    pass


def _san(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    return np.clip(z, -35.0, 35.0)


def _probs(z):
    z = _san(z)
    m = float(np.max(z)) if z.size else 0.0
    ez = np.exp(z - m)
    e0 = np.exp(-m)
    Z = e0 + float(np.sum(ez))
    p = np.empty(N, dtype=np.float64)
    p[0] = e0 / Z
    p[1:] = ez / Z
    return p


def _self_conv(p):
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f * f, FFT_SELF).real[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(p):
    c = _self_conv(p)
    return 2.0 * float(N) * float(np.max(c))


def _update_best(z, s):
    global _best_score, _best_z
    if np.isfinite(s) and s < _best_score:
        _best_score = float(s)
        _best_z = np.array(z, copy=True)


def _grad_from_weights(p, w):
    """Compute 2 * sum_j p_j * w_{i+j} for i=0..N-1 using FFT."""
    rev = p[::-1]
    fa = np.fft.rfft(rev, FFT_GRAD)
    fb = np.fft.rfft(w, FFT_GRAD)
    conv = np.fft.irfft(fa * fb, FFT_GRAD).real
    return 2.0 * conv[N - 1:2 * N - 1]


def _obj_grad(x, alpha):
    if time.monotonic() >= _search_deadline:
        raise SearchTimeout
    x = _san(x)
    p = _probs(x)
    c = _self_conv(p)
    cmax = float(np.max(c))
    _update_best(x, 2.0 * N * cmax)

    u = alpha * (c - cmax)
    u = np.clip(u, -50.0, 50.0)
    e = np.exp(u)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(se) / alpha
        w = e / se

    gp = _grad_from_weights(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e5, 1e5)
    return float(J), gz


def _lbfgs(z0, alpha, maxfun, deadline):
    global _search_deadline
    if minimize is None:
        return
    _search_deadline = deadline
    x0 = _san(z0)
    try:
        minimize(
            _obj_grad, x0, args=(alpha,),
            method="L-BFGS-B", jac=True, bounds=BOUNDS,
            options={"maxfun": maxfun, "maxiter": max(50, maxfun // 3),
                     "ftol": 1e-14, "gtol": 1e-9, "maxls": 25, "disp": False}
        )
    except SearchTimeout:
        pass
    except Exception:
        pass


def _subgrad_polish(z0, deadline, rng, topk=12):
    z = _san(z0)
    p = _probs(z)
    score = _score(p)
    _update_best(z, score)

    base_step = 0.08
    fails = 0
    step = base_step

    while time.monotonic() < deadline:
        c = _self_conv(p)
        cmax = float(np.max(c))

        # Select active peaks with exponential weighting
        idx = np.argpartition(c, -topk)[-topk:]
        cv = c[idx]
        thresh = cmax - 0.05 * cmax
        active = cv[cv > thresh]
        if len(active) == 0:
            active = cv[:1]
        # Exponential weights: peaks closer to max get more weight
        scale = 50.0
        weights_k = np.exp(scale * (active - cmax) / max(cmax, 1e-15))
        weights_k /= float(np.sum(weights_k))

        w = np.zeros(2 * N - 1, dtype=np.float64)
        for i_k, k in enumerate(idx):
            if cv[i_k] > thresh:
                w[k] = weights_k[list(cv).index(cv[i_k])] if cv[i_k] in list(active) else 0

        # Redo properly
        w = np.zeros(2 * N - 1, dtype=np.float64)
        for i_k in range(len(idx)):
            if cv[i_k] > thresh:
                w[idx[i_k]] = np.exp(scale * (cv[i_k] - cmax) / max(cmax, 1e-15))
        wsum = float(np.sum(w))
        if wsum > 0:
            w /= wsum
        else:
            w[int(np.argmax(c))] = 1.0

        gp = _grad_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        accepted = False
        if gnorm > 1e-15:
            direction = gz / gnorm
            # Backtracking line search
            for _bt in range(8):
                zt = _san(z - step * direction)
                pt = _probs(zt)
                st = _score(pt)
                if st < score - 1e-13:
                    z = zt
                    p = pt
                    score = st
                    _update_best(z, score)
                    accepted = True
                    step = min(step * 1.15, 0.3)
                    break
                step *= 0.5

        if not accepted:
            # Try a few random perturbations
            best_sr = score
            best_zl = z
            for _ in range(4):
                zt = _san(z + rng.normal(0.0, 0.05, N - 1))
                pt = _probs(zt)
                st = _score(pt)
                if st < best_sr - 1e-13:
                    best_sr = st
                    best_zl = zt
            if best_sr < score - 1e-13:
                z = best_zl
                p = _probs(z)
                score = best_sr
                _update_best(z, score)
                accepted = True

        if accepted:
            fails = 0
        else:
            fails += 1
            step = base_step * (0.7 ** min(fails, 10))
            if fails > 80:
                break


def _density_to_z(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 1e-12)
    logf = np.log(f)
    z = logf[1:] - logf[0]
    return np.clip(z, -35.0, 35.0)


def _logistic_z(beta):
    x = X_GRID
    u = beta * (x - 0.5)
    f = 2.0 / (1.0 + np.exp(-np.clip(u, -50, 50)))
    return _density_to_z(f)


def _u_shape_z(gamma):
    """U-shaped: high at edges, low in middle. f(x) = 1 + gamma*(2x-1)^2"""
    x = X_GRID
    f = 1.0 + gamma * (2.0 * x - 1.0) ** 2
    return _density_to_z(f)


def _cos_bump_z(beta):
    """Raised cosine shape: f(x) = 1 + beta*cos(2*pi*x)"""
    x = X_GRID
    f = 1.0 + beta * np.cos(2.0 * np.pi * x)
    f = np.maximum(f, 1e-8)
    return _density_to_z(f)


def _two_peak_z(amp, width):
    """Two Gaussian peaks near the edges"""
    x = X_GRID
    f = np.ones(N)
    f += amp * np.exp(-((x - 0.05) ** 2) / (2 * width ** 2))
    f += amp * np.exp(-((x - 0.95) ** 2) / (2 * width ** 2))
    return _density_to_z(f)


def _structured_candidates():
    out = []
    for beta in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0):
        out.append(_logistic_z(beta))
        out.append(_logistic_z(-beta))
    for g in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        out.append(_u_shape_z(g))
    for b in (0.2, 0.4, 0.6, 0.8, 0.95):
        out.append(_cos_bump_z(b))
        out.append(_cos_bump_z(-b))
    for amp in (0.5, 1.0, 2.0, 3.0):
        for wid in (0.02, 0.03, 0.05, 0.08):
            out.append(_two_peak_z(amp, wid))
    return out


def _random_poly_z(rng):
    x = X_GRID
    t = x - 0.5
    deg = rng.integers(3, 8)
    coef = rng.normal(scale=0.8, size=deg + 1)
    f = np.ones(N, dtype=np.float64)
    val = np.ones(N, dtype=np.float64)
    for k in range(deg, 0, -1):
        val = val * t + coef[k]
    f += val
    f = np.maximum(f, 1e-8)
    return _density_to_z(f)


def _random_fourier_z(rng):
    x = X_GRID
    z = np.zeros(N - 1)
    for fq in range(1, 12):
        amp = rng.normal(0, 1.5 / fq)

        z += amp * np.sin(2 * np.pi * fq * X_GRID[1:] + ph)
        z += amp * np.cos(2 * np.pi * fq * X_GRID[1:] + ph)

def _continuation_optimize(z0, deadline, rng):
    """Run L-BFGS with increasing alpha (continuation method)."""
    alphas = [200, 500, 1500, 5000, 15000, 50000, 150000, 500000, 1500000, 5000000, 10000000]
    z = _san(z0)

    for i, alpha in enumerate(alphas):
        if time.monotonic() >= deadline - 30:
            break
        rem = deadline - time.monotonic()
        # More iterations for early (smooth) stages, fewer for later
        if i < 3:
            maxfun = min(400, int(rem / 0.02))
        elif i < 6:
            maxfun = min(300, int(rem / 0.02))
        else:
            maxfun = min(200, int(rem / 0.03))
        if maxfun < 50:
            maxfun = 50

        _lbfgs(z, alpha, maxfun, deadline - 5)
        z = _san(z)

        # Interleave subgradient steps
        if i >= 3 and time.monotonic() < deadline - 30:

            z = _san(_best_z) if _best_z is not None else z
