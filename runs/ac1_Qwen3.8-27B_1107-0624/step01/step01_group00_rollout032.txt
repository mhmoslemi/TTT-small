import time
import math
import numpy as np

def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum x = target}"""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    # Algorithm from Giftis & O'Donnell
    if np.all(v <= 0):
        return np.full(n, target / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - target)
    rho_val = np.max(np.where(cond, rho, 0))
    theta = (cssv[rho_val - 1] - target) / rho_val
    return np.maximum(v - theta, 0.0)

def _conv_sq(a):
    """Linear convolution a*a via FFT"""
    n = a.size
    m = 1
    while m < 2*n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def _objective(a):
    """Compute 2*n*max(c)/S^2"""
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _conv_sq(a)
    return 2.0 * a.size * c.max() / (S * S)

def _power_sum_obj(a, p):
    """Smoothed objective using power sum approximation of max"""
    S = a.sum()
    if S < 1e-12:
        return np.inf, np.zeros_like(a)

    # Power sum approximation: max(c) ≈ (sum c^p)^(1/p)
    cs = c ** p  # all c >= 0
    val = cs.sum() ** (1.0/p)
    obj = 2.0 * a.size * val / (S * S)
    return obj, c

import time
import math
import numpy as np

def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}"""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rv = rho[idx[-1]]
    theta = (cssv[rv - 1] - target) / rv
    return np.maximum(v - theta, 0.0)

def _fft_len(n):
    m = 1
    while m < n:
        m <<= 1
    return m

def _conv_sq(a, m=None):
    n = a.size
    if m is None:
        m = _fft_len(2 * n - 1)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]

def _grad_max_smoothed(a, c, p, m=None, m2=None):
    """Gradient of (sum c^p)^(1/p) w.r.t. a, where c = a*a.
    Returns the unnormalized gradient (multiply by 2n/S^2 for the full objective gradient)."""
    n = a.size
    if m2 is None:
        m2 = _fft_len(n + 2 * n - 2)
    w = c ** p  # weights (using p as the power for the gradient weights, not p-1)
    # Actually for the power-sum: f = (sum c^p)^(1/p)
    # df/da[i] = (1/p) * (sum c^p)^(1/p-1) * p * sum_k c[k]^(p-1) * dc[k]/da[i]
    # = (sum c^p)^(1/p-1) * sum_k c[k]^(p-1) * dc[k]/da[i]
    # Let w = c^(p-1), then df/da[i] = (sum c^p)^(1/p-1) * sum_k w[k] * dc[k]/da[i]
    # And sum_k w[k] * dc[k]/da[i] ≈ 2 * xcorr(a, w)[i]
    
    w = c ** (p - 1) if p > 1 else np.ones_like(c)
    sum_cp = np.sum(c ** p)
    scale = sum_cp ** (1.0 / p - 1.0)
    
    # Cross-correlation: xcorr[i] = sum_k a[k] * w[k+i] (linear)
    Fw = np.fft.rfft(w, m2)
    Fa = np.fft.rfft(a, m2)
    r = np.fft.irfft(np.conj(Fa) * Fw, m2)[:n]
    
    return 2.0 * scale * r

def _full_obj(a):
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _conv_sq(a)
    return 2.0 * a.size * c.max() / (S * S)

    """Compute objective and gradient in v-space, where a = softmax(v)."""
    # Stable softmax
    v_shift = v - v.max()
    exp_v = np.exp(v_shift)
    a = exp_v / exp_v.sum()
    
    # Compute objective in terms of a
    S = 1.0  # by construction
    c = _conv_sq(a)
    sum_cp = np.sum(c ** p)
    max_approx = sum_cp ** (1.0 / p)
    obj = 2.0 * a.size * max_approx
    
    # Gradient of obj w.r.t. a
    w = c ** (p - 1)
    scale = sum_cp ** (1.0 / p - 1.0)
    # grad_a[i] = 2 * scale * xcorr(a, w)[i]
    m2 = _fft_len(a.size + 2 * a.size - 2)
    Fw = np.fft.rfft(w, m2)
    Fa = np.fft.rfft(a, m2)
    r = np.fft.irfft(np.conj(Fa) * Fw, m2)[:a.size]
    grad_a = 2.0 * a.size * scale * r
    
    # Convert to v-space gradient
    dot_ga_a = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot_ga_a)
    
    return obj, grad_v

The function is a ratio of a piecewise-linear function (the max) and a quadratic (the sum squared). The critical points occur where:
1. Two of the c[k](t) are equal (the max changes)
2. The derivative of the ratio is zero

1. Compute d[k] = a[k-i] + a[i-k] (for valid indices), in O(n)

def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rv = rho[idx[-1]]
    theta = (cssv[rv - 1] - target) / rv
    return np.maximum(v - theta, 0.0)

def _next_pow2(n):
    m = 1
    while m < n:
        m <<= 1
    return m

def _conv_sq(a, m=None):
    n = a.size

    c = np.fft.irfft(fa * fa, m)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected gradient with power-sum smoothing, FISTA momentum, and local search.
Multiple diverse seeds for robustness.
"""

import time
import math
import numpy as np

def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} using the water-filling algorithm."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rv = int(rho[idx[-1]])
    theta = (cssv[rv - 1] - target) / rv
    return np.maximum(v - theta, 0.0)

def _next_pow2(n):
    m = 1
    while m < n:
        m <<= 1
    return m

def _conv_sq(a, m):
    """Linear convolution a*a (length 2n-1) using FFT of size m."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * a.size - 1]

def _xcorr(a, w, m):
    """Cross-correlation r[i] = sum_k a[k]*w[k+i] for i = 0,...,n-1."""
    n = a.size
    Fa = np.fft.rfft(a, m)
    Fw = np.fft.rfft(w, m)
    r = np.fft.irfft(np.conj(Fa) * Fw, m)
    return r[:n]

def _objective_and_grad(a, c, p, m1, m2):
    """
    Compute the power-sum smoothed objective and its gradient.
    obj = 2*n*(sum c^p)^(1/p) / S^2
    grad = d(obj)/da
    """
    n = a.size
    S = a.sum()
    c = np.maximum(c, 1e-30)
    
    cp = c ** p
    sum_cp = cp.sum()
    max_ap = sum_cp ** (1.0 / p)
    obj = 2.0 * n * max_ap / (S * S)
    
    if p == 1:
        # Top-1 subgradient
        k_max = np.argmax(c)
        w = np.zeros_like(c)
        w[k_max] = 1.0
        scale = 1.0
    else:
        w = c ** (p - 1)
        scale = sum_cp ** (1.0 / p - 1.0)
    
    r = _xcorr(a, w, m2)
    prefactor = (2.0 * n / (S * S)) * scale * 2.0
    grad = prefactor * r
    
    return obj, grad

def _make_seeds(base, n):
    """Create diverse initial seeds."""
    seeds = []
    
    # Provided sequence
    b = np.clip(base, 0.0, None)
    if b.sum() > 0:
        seeds.append(b / b.sum())
    seeds.append(base[::-1].copy() / max(base.sum(), 1e-15))
    
    # Uniform
    seeds.append(np.ones(n) / n)
    
    # Sine bump (centered)
    x = np.linspace(0, 1, n)
    seeds.append(np.sin(np.pi * x) / np.sin(np.pi * x).sum())
    
    # Quadratic bump
    y = 1.0 - ((x - 0.5) * 2) ** 2
    y = np.maximum(y, 0.01)
    seeds.append(y / y.sum())
    
    # U-shape
    u = ((x - 0.5) * 2) ** 2 + 0.1
    seeds.append(u / u.sum())
    
    # Chirp (made non-negative)
    chirp = np.abs(np.exp(1j * np.pi * x ** 2))
    # Actually this is all 1s. Let me use real part + offset
    real_chirp = np.cos(2 * np.pi * 3 * x) + 1.0
    seeds.append(real_chirp / real_chirp.sum())
    
    # Random
    rng = np.random.default_rng(42)
    r = rng.random(n)
    seeds.append(r / r.sum())
    
    # Fractal-like (1/f noise)
    freqs = np.fft.rfftfreq(n)
    power = 1.0 / (freqs + 0.1)
    phase = rng.random(len(power)) * 2 * np.pi
    sig = np.fft.irfft(power * np.exp(1j * phase), n)
    sig = np.abs(sig)
    seeds.append(sig / sig.sum())
    
    return seeds

def _run_gradient_descent(a0, t_limit, p_init, name=""):

    """

    best_a = a.copy()

    t_fista = 1.0
    momentum = 0.0
    p = p_init
    lr = 1.0 / (4.0 * n)  # initial learning rate estimate
    step = 0
    stall_count = 0
    
    while time.monotonic() < t_limit:
        # FISTA extrapolation
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        y = a + ((t_fista - 1.0) / t_new) * (a - y)

                improved = True

        # Update best
        obj_actual = 2.0 * n * _conv_sq(a, m1).max()

    # Initial perturbation scale
    base_scale = 0.5 / math.sqrt(n)
    scale = base_scale
    temp = 0.01  # annealing temperature (relative)
    
    step = 0
    while time.monotonic() < t_limit:
        # Generate perturbation
        r = rng.random()
        if r < 0.4:
            # Gaussian perturbation
            pert = rng.normal(0, scale, n)
            a_new = _proj_simplex(a + pert, 1.0)
        elif r < 0.7:
            # Block multiplication
            blen = rng.integers(5, max(6, n // 20))
            start = rng.integers(0, n - blen)
            a_new = a.copy()
            a_new[start:start+blen] *= np.exp(rng.normal(0, scale * 2, blen))
            a_new = _proj_simplex(a_new, 1.0)
        elif r < 0.9:
            # Single coordinate
            idx = rng.integers(0, n)
            a_new = a.copy()
            a_new[idx] *= np.exp(rng.normal(0, scale * 5))
            a_new = max(0, a_new[idx])
            a_new = _proj_simplex(a_new, 1.0)
        else:
            # "Jump" - larger perturbation
            pert = rng.normal(0, scale * 3, n)
            a_new = _proj_simplex(a + pert, 1.0)
        
        c_new = _conv_sq(a_new, m1)
        obj_new = 2.0 * n * c_new.max()
        
        diff = obj_new - best_obj
        if diff < 0:
            # Accept (improvement)
            accept_prob = 1.0
        else:
            # Simulated annealing: accept with probability exp(-diff/temp)
            accept_prob = math.exp(-diff / temp) if temp > 1e-10 else 0.0
        
        if rng.random() < accept_prob:
            a = a_new.copy()
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                scale *= 0.999  # slowly decrease

            # Rejected
            pass
        
        step += 1
        # Anneal
        temp *= 0.9995
        scale = max(base_scale * 0.01, scale)
        
        if verbose and step % 5000 == 0:
            print(f"  local_search step {step}: best={best_obj:.6f}, temp={temp:.4e}, scale={scale:.4e}")
    
    return best_a, best_obj
