```python
def auto_convolve(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.zeros(size)
    fa[:n] = a
    fa_fft = np.fft.rfft(fa)
    result = np.fft.irfft(fa_fft * fa_fft, size)
    return result[:2*n-1]
```

Step 3: dS/dθ_i = Σ_j (dS/da_j) * (da_j/dθ_i)

= a_i * (g_i - Σ_j g_j * a_j)

where g_i = dS/da_i.

This is the standard softmax gradient formula.

"""
Search for low-score step-function heights using gradient-based optimization
with smooth-max objective, LP-based refinement, and random restarts.

Key insights:
- The score 2*n*max(a*a)/(sum(a))^2 is scale-invariant, so we work with sum(a)=1.
- The auto-convolution peak is minimized by making the sequence "asymmetric"
  (non-palindromic), which allows the center of the auto-convolution to be
  lower than ||a||^2.
- Gradient descent with smooth-max objective provides a differentiable
  surrogate for the non-smooth max function.
- The gradient can be computed efficiently via FFT (O(n log n) per eval).
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def auto_convolve(a):
    """Compute linear auto-convolution of a using FFT. Returns array of length 2n-1."""
    n = a.size
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = a
    fa_fft = np.fft.rfft(fa)
    result = np.fft.irfft(fa_fft * fa_fft, size)
    return result[:2 * n - 1]


def normalize_nonneg(a):
    """Normalize a non-negative array to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a, dtype=np.float64) / a.size
    return a / s


def compute_score(a):
    """Compute score = 2*n*max(a*a)/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = auto_convolve(a)
    m = c.max()
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def compute_score_normalized(a):
    """Score for a normalized (sum=1) non-negative sequence."""
    c = auto_convolve(a)
    return float(2.0 * a.size * c.max())


def cross_corr_ap(a, p):
    """
    Compute z[i] = sum_j a[j] * p[i+j] for i=0..n-1.
    Uses FFT for efficiency.
    """
    n = a.size
    size = 1
    while size < len(p) + n:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = a
    fp = np.zeros(size, dtype=np.float64)
    fp[:len(p)] = p
    Fa = np.fft.rfft(fa)
    Fp = np.fft.rfft(fp)
    z = np.fft.irfft(np.conj(Fa) * Fp)
    return z[:n]


def smooth_max_value_and_grad(theta, tau):
    """
    Compute smooth-max score and gradient w.r.t. theta (softmax params).
    Returns (score, gradient).
    """
    n = theta.size
    # Softmax
    theta_max = theta.max()
    a = np.exp(theta - theta_max)
    a_sum = a.sum()
    a = a / a_sum
    
    # Auto-convolution
    c = auto_convolve(a)
    
    # Smooth-max
    tc = tau * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    exp_sum = exp_tc.sum()
    p = exp_tc / exp_sum
    
    # Smooth-max value (this approximates max(c))
    S_val = (tc_max + np.log(exp_sum)) / tau
    
    # Gradient w.r.t. a: g[i] = 2*z[i] - p[2i]*a[i]
    z = cross_corr_ap(a, p)
    g_a = 2.0 * z
    # Diagonal correction
    g_a -= p[:n] * a  # p[2i] for i=0..n-1 is p[0], p[2], p[4], ...
    # Wait, p[2i] means p at index 2i. Let me fix this.
    # p has length 2n-1. p[2i] for i=0..n-1 gives p[0], p[2], p[4], ..., p[2n-2].
    diag_indices = np.arange(0, 2 * n - 1, 2)  # 0, 2, 4, ..., 2n-2
    g_a = 2.0 * z - p[diag_indices] * a
    
    # Gradient w.r.t. theta (softmax chain rule)
    ga_dot_a = np.dot(g_a, a)
    grad_theta = a * (g_a - ga_dot_a)
    
    return S_val, grad_theta


def optimize_smooth_max(a_init, time_limit, seed=42):
    """
    Optimize a sequence using smooth-max gradient descent.
    a_init: initial normalized non-negative sequence (sum=1).

    tau_values = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.007]

        def fun_and_grad(th):
            val, grad = smooth_max_value_and_grad(th, tau)
            return val, grad
        
        try:

        # Choose active rows (top peaks)

                A[i, jlo:jhi] = a[k - idx]
        
        # Solve LP: max sum(g) s.t. A*g <= M, g >= 0
        c_obj = -np.ones(n)
        b_vec = np.full(len(rows), M)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                         bounds=(0, None), method='highs')

                Sg = g.sum()

def resize_sequence(a, N):
    """Resize a normalized sequence to length N using linear interpolation + normalization."""
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    if m == N:
        return normalize_nonneg(a)
    if m <= 1:
        return np.full(N, 1.0 / N)
    x = np.linspace(0.0, m - 1, N)
    i0 = np.clip(np.floor(x).astype(int), 0, m - 1)
    i1 = np.clip(i0 + 1, 0, m - 1)
    frac = x - i0
    out = a[i0] * (1 - frac) + a[i1] * frac
    return normalize_nonneg(np.maximum(out, 0.0))
