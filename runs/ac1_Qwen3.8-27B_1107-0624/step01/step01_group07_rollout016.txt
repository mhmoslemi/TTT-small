import numpy as np
import time
from scipy.optimize import linprog

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def lp_direct(a):
    """Solve: minimize M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (M, g)."""
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M -> c_obj = [0, ..., 0, 1]
    # Constraints: A @ g - M <= 0 for each row -> [A, -1] @ [g, M] <= 0
    #             sum(g) = 1 -> handled via equality constraint
    #             g >= 0, M >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    # Inequality: A @ g - M <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.hstack([np.ones(n), [[0.0]]])
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(0, None)]  # g_i >= 0, M >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[-1], res.x[:n]
    return None, None

def optimize_direct_lp(a, deadline, rng):
    """Direct LP-based optimization."""
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.3
    stall = 0
    
    while time.monotonic() < deadline:
        M_opt, g = lp_direct(a)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        improved = False
        for shrink in [1.0, 0.7, 0.5, 0.3, 0.1]:
            s = step * shrink
            a_new = (1.0 - s) * a + s * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-10:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break
        
        if improved:
            stall = 0
            step = min(0.8, step * 1.02)
        else:
            stall += 1
            step *= 0.8
            if stall > 100:
                # Random restart perturbation
                a = best_a.copy() * (1 + 0.1 * np.abs(rng.normal(0, 1, n)))
                a = np.clip(a, 0, None)
                a = a / a.sum()
                stall = 0
            if step < 0.001:
                break
    
    return best_a, best_score

"""
Direct LP-based optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Core idea: at each step, solve an LP to find the probability vector g minimizing
the peak of conv(a_current, g), then take an adaptive step towards g.
Multiple lengths and initializations are explored within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog

_global_best = None
_global_best_score = np.inf


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = np.convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def solve_lp_min_peak(a):
    """
    Solve: minimize M
           s.t. conv(a, g)_k <= M for all k
                sum(g) = 1
                g >= 0, M >= 0
    Returns (M_opt, g_opt) or (None, None) on failure.
    """
    n = len(a)
    A = build_conv_matrix(a)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            M_opt = float(res.x[-1])
            g_opt = np.maximum(res.x[:n], 0.0)
            return M_opt, g_opt
    except Exception:
        pass
    return None, None


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_uniform(n):
    return np.ones(n)


def optimize_lp(a_init, deadline, rng, verbose=True):
    """
    LP-based iterative optimization.
    At each step: solve LP for optimal g, then move towards g.
    """
    n = len(a_init)
    a = np.clip(a_init, 0, None).astype(np.float64)
    a_sum = a.sum()
    if a_sum < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a_sum

    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()

    step = 0.5
    stall = 0
    it = 0
    last_print = t0
    last_best_print = t0

    while time.monotonic() < deadline:
        it += 1

        M_opt, g = solve_lp_min_peak(a)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        # Try various step sizes (largest first)
        improved = False
        for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            s = step * frac
            a_new = (1.0 - s) * a + s * g
            a_new = np.clip(a_new, 0, None)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new.copy()
                best_score = sc
                a = a_new
                improved = True
                if verbose and time.monotonic() - last_best_print > 3.0:
                    last_best_print = time.monotonic()
                    print(f"      new best: {best_score:.6f} (iter {it})")
                break

        if improved:
            stall = 0
            step = min(0.95, step * 1.04)
        else:
            stall += 1
            step = max(0.01, step * 0.8)

            if stall == 40:
                noise = np.abs(rng.normal(0, 0.06, n))
                a = best_a * (1.0 + noise)
                a = np.clip(a, 0, None)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                stall = 0
            elif stall == 100:
                a = best_a * (1.0 + 0.25 * np.abs(rng.normal(0, 1, n)))
                a = np.clip(a, 0, None)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                stall = 0
            elif stall == 200:
                # Mix with random
                a = 0.5 * best_a + 0.5 * np.abs(rng.normal(0, 1, n))
                a = np.clip(a, 0, None)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                stall = 0
            elif stall > 400:
                break

        if verbose and time.monotonic() - last_print > 8.0:
            last_print = time.monotonic()
            elapsed = time.monotonic() - t0
            print(f"    n={n}, iter={it}, best={best_score:.6f}, "
                  f"M_opt={M_opt:.4e}, step={step:.3f}, t={elapsed:.0f}s")

    return best_a, best_score


def optimize_subgrad(a_init, deadline, rng, verbose=True):
    """
    Subgradient-based optimization as an alternative/complement.
    Directly uses the subgradient of max_k conv(a,a)_k.
    """
    n = len(a_init)
    a = np.clip(a_init, 0, None).astype(np.float64)
    a_sum = a.sum()
    if a_sum < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a_sum

    best_a = a.copy()
    best_score = compute_objective(a)
    t0 = time.monotonic()
    lr = 0.1
    stall = 0
    it = 0
    last_print = t0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a)
        M = c.max()
        peak_indices = np.where(c >= M - 1e-12 * M)[0]

        # Subgradient: for each peak index k*, grad_j = 2*a_{k*-j} (if valid)
        grad = np.zeros(n)
        for k in peak_indices:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        grad /= len(peak_indices)

        # Project to tangent space of simplex (subtract mean)
        grad = grad - grad.mean()

        # Gradient step (minimize, so go opposite to gradient)
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0, None)
        sn = a_new.sum()
        if sn < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / sn
        sc = compute_objective(a_new)

        if sc < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.5, lr * 1.02)
        else:
            stall += 1
            lr = max(0.001, lr * 0.9)
            if stall == 50:
                noise = np.abs(rng.normal(0, 0.05, n))
                a = best_a * (1.0 + noise)
                a = np.clip(a, 0, None)
                a = a / a.sum()
                stall = 0
            if stall > 200:
                break

        if verbose and time.monotonic() - last_print > 8.0:
            last_print = time.monotonic()
            elapsed = time.monotonic() - t0
            print(f"    [subgrad] n={n}, iter={it}, best={best_score:.6f}, "
                  f"lr={lr:.5f}, t={elapsed:.0f}s")

    return best_a, best_score
