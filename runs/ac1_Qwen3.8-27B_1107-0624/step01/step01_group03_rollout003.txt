def fast_score(s):

    """
    Compute smooth max objective and gradient w.r.t. s.
    Returns (value, grad_s) where grad_s[j] = dF/ds_j.
    """
    c = np.convolve(s, s)
    # Smooth max: F = eps * log(sum(exp(c_k/eps)))
    c_scaled = c / eps
    c_max = np.max(c_scaled)
    c_exp = np.exp(c_scaled - c_max)
    w = c_exp / c_exp.sum()  # These are the "weights" for the smooth max
    
    F = eps * (c_max + np.log(c_exp.sum()))

def adam_step(t, grad_t, m, v, t_count, lr, beta1, beta2, eps_adam):
    """Adam optimizer step. Returns updated t, m, v."""
    m = beta1 * m + (1 - beta1) * grad_t
    v = beta2 * v + (1 - beta2) * grad_t ** 2
    m_hat = m / (1 - beta1 ** t_count)
    v_hat = v / (1 - beta2 ** t_count)
    t_new = t - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
    return t_new, m, v
