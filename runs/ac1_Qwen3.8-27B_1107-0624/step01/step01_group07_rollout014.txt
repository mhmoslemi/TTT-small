The objective function is: `2 * n * max(conv(a,a)) / (sum(a))^2`

Key observations:
1. The function is scale-invariant (if we multiply all elements by c, the numerator and denominator both scale by c^2, so the ratio is unchanged).
2. We want to minimize the maximum value of the auto-convolution relative to the sum.

Let me think about what makes a good sequence:
- If all elements are equal (uniform), then conv(a,a) peaks at the center with value n*a^2, sum = n*a, so the objective is 2*n*n*a^2/(n*a)^2 = 2*n/(n) = 2. Wait, let me recalculate.

- For a raised cosine (Hann window), the auto-convolution is smoother, so the peak is lower relative to the sum.

The LP approach solves: maximize sum(g) subject to conv(a, g) <= M, g >= 0. This finds the "best" direction to move in that keeps the convolution bounded.

5. Normalize

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines multiplicative weight updates, LP-based direction finding,
and direct gradient optimization with multiple initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def compute_objective_normalized(a):
    """Assume sum(a)=1, compute 2*n*max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max()

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_direction(a, M, tol=None):
    """
    Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None

def multiplicative_update(a, eta, n_peak_frac=0.3):
    """
    Multiplicative update: reduce positions that contribute to peaks.
    a is normalized (sum=1).
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    # Select peaks above threshold
    threshold = M * (1.0 - n_peak_frac)
    peak_mask = c > threshold
    
    # Compute contribution of each a[i] to peak convolutions
    contrib = np.zeros(n)
    for i in range(n):
        js = np.arange(max(0, i - (2*n - 2)), min(2*n - 1, i + n - 1) + 1) - i
        # c[k] = sum_j a[j] * a[k-j], so a[i] contributes a[i]*a[k-i] to c[k]
        # For position i, it contributes to c[k] via a[k-i] where k-i is valid
        # contrib[i] = sum_{k in peaks} a[k-i] (where k-i is in [0,n-1])
        k_vals = np.where(peak_mask)[0]
        offsets = k_vals - i
        valid = (offsets >= 0) & (offsets < n)
        if valid.any():
            contrib[i] = a[offsets[valid]].sum()
    
    # Normalize contrib
    avg_c = contrib.mean() + 1e-15
    update = np.exp(-eta * contrib / avg_c)
    a_new = a * update
    a_new = np.clip(a_new, 1e-15, None)
    a_new /= a_new.sum()
    return a_new

def smooth_sequence(a, sigma):
    """Apply Gaussian smoothing to the sequence."""
    n = len(a)
    if sigma < 0.5:
        return a.copy()
    kernel_size = int(6 * sigma) | 1  # make odd
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / sigma ** 2)
    kernel /= kernel.sum()
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.clip(a_smooth, 0, None)
    if a_smooth.sum() > 1e-10:
        a_smooth /= a_smooth.sum()
    return a_smooth

def make_initialization(n, kind='cosine'):
    """Create various window initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        sigma = 0.25 * n
        return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif kind == 'parabolic':
        c = (n - 1) / 2.0
        w = 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)
        return np.clip(w, 0.01, None)
    elif kind == 'blackman':
        n0 = np.arange(n)
        return (0.42 - 0.5 * np.cos(2 * np.pi * n0 / (n - 1)) + 0.08 * np.cos(4 * np.pi * n0 / (n - 1)))
    elif kind == 'blackman_harris':
        n0 = np.arange(n)
        a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
        return a0 - a1 * np.cos(2 * np.pi * n0 / (n - 1)) + a2 * np.cos(4 * np.pi * n0 / (n - 1)) - a3 * np.cos(6 * np.pi * n0 / (n - 1))
    elif kind == 'flattop':
        n0 = np.arange(n)
        a0, a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
        return a0 - a1 * np.cos(2 * np.pi * n0 / (n - 1)) + a2 * np.cos(4 * np.pi * n0 / (n - 1)) - a3 * np.cos(6 * np.pi * n0 / (n - 1)) + a4 * np.cos(8 * np.pi * n0 / (n - 1))
    elif kind == 'bspline':
        # Cubic B-spline support
        center = (n - 1) / 2.0
        x = np.arange(n) - center
        # B-spline of order 3 (cubic)
        def b3(t):
            t = np.abs(t)
            result = np.zeros_like(t)
            mask = t < 0.5
            result[mask] = 1 - 2 * t[mask]**2
            mask2 = (t >= 0.5) & (t < 1.5)
            result[mask2] = 0.5 * (1.5 - t[mask2])**3
            mask3 = (t >= 1.5) & (t < 2.5)
            result[mask3] = (2.5 - t[mask3])**3 / 6.0
            return result
        # Convolve three times for cubic
        base = b3(x / max(n / 8, 1))
        return np.clip(base, 0.01, None)
    else:
        return np.ones(n) / n

def optimize_lp_iterative(a, deadline, rng, verbose=False):
    """
    LP-based iterative optimization.
    a should be normalized (sum=1).
    """
    n = len(a)
    a_work = a.copy()
    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.15
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a_work, a_work, mode='full')
        M = c.max()
        
        g = lp_direction(a_work, M)
        if g is None:
            break
        
        improved = False
        for shrink in [1.0, 0.6, 0.3, 0.1, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 1e-15, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a_work = a_new
                improved = True
                break
        
        if improved:
            stall = 0
            step = min(0.5, step * 1.05)
        else:
            stall += 1
            step = max(0.003, step * 0.7)
            if stall % 15 == 0:
                # Perturbation
                noise_scale = min(0.15, 0.02 * stall / 15)
                noise = rng.normal(0, noise_scale, n)
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-15, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    sc = compute_objective(a_new)
                    if sc < best_score:
                        best_a = a_new
                        best_score = sc
                        a_work = a_new
                        stall = 0
                    else:
                        # Simulated annealing: accept with decreasing probability
                        if rng.random() < 0.1 * np.exp(-stall / 30):
                            a_work = a_new
                        else:
                            a_work = best_a
                else:
                    break
            if stall > 100:
                break
        
        if verbose and it % 200 == 0:
            print(f"    LP iter {it}, score={best_score:.6f}, step={step:.4f}, stall={stall}")
    
    return best_a, best_score

def optimize_hybrid(a, deadline, rng, verbose=False):
    """
    Hybrid: start with multiplicative for fast convergence, then LP for fine-tuning.
    """
    n = len(a)
    a = a.copy()
    a /= a.sum()
    
    # Phase 1: Multiplicative (fast initial progress)
    phase1_time = min(60, (deadline - time.monotonic()) * 0.4)
    phase1_deadline = time.monotonic() + phase1_time
    a, score1 = optimize_multiplicative(a, phase1_deadline, rng, verbose=verbose)
    
    # Phase 2: LP-based (fine-tuning)
    remaining_deadline = deadline
    a, score2 = optimize_lp_iterative(a, remaining_deadline, rng, verbose=verbose)
    
    # Phase 3: Short multiplicative polish
    if time.monotonic() < deadline - 30:
        polish_deadline = time.monotonic() + min(30, deadline - time.monotonic() - 5)
        a, score3 = optimize_multiplicative(a, polish_deadline, rng, verbose=verbose)
        if score3 < score2:
            score2 = score3
    
    return a, score2

def local_search_refine(a, deadline, rng, verbose=False):
    """
    Coordinate descent style: try small perturbations to individual coordinates.
    """
    n = len(a)
    a_work = a.copy()
    best_a = a.copy()
    best_score = compute_objective(a)
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        # Try random small perturbations
        n_trials = min(50, max(5, int(2 * n / (it + 1))))
        for _ in range(n_trials):
            idx = rng.integers(0, n)
            scale = rng.uniform(0.5, 2.0)
            a_new = a_work.copy()
            a_new[idx] *= scale
            a_new = np.clip(a_new, 1e-15, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a_work = a_new
                break
        
        if it % 100 == 0 and verbose:
            print(f"    local iter {it}, score={best_score:.6f}")
    
    return best_a, best_score

"""
Hybrid optimization combining soft-max gradient descent, LP-based direction finding,
and multiplicative updates to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiple initializations, adaptive annealing, and efficient FFT-based gradients.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def make_initialization(n, kind='cosine'):
    """Create various window initializations, normalized to sum=1."""
    if kind == 'uniform':
        w = np.ones(n)
    elif kind == 'cosine':
        w = 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        sigma = 0.25 * n
        w = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif kind == 'parabolic':
        c = (n - 1) / 2.0
        w = np.clip(1.0 - np.abs(np.arange(n) - c) / (c + 1.0), 0.01, None)
    elif kind == 'blackman':
        n0 = np.arange(n)
        w = 0.42 - 0.5 * np.cos(2 * np.pi * n0 / (n - 1)) + 0.08 * np.cos(4 * np.pi * n0 / (n - 1))
    elif kind == 'blackman_harris':
        n0 = np.arange(n)
        a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
        w = (a0 - a1 * np.cos(2 * np.pi * n0 / (n - 1))
             + a2 * np.cos(4 * np.pi * n0 / (n - 1))
             - a3 * np.cos(6 * np.pi * n0 / (n - 1)))
    elif kind == 'flattop':
        n0 = np.arange(n)
        a0, a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
        w = (a0 - a1 * np.cos(2 * np.pi * n0 / (n - 1))
             + a2 * np.cos(4 * np.pi * n0 / (n - 1))
             - a3 * np.cos(6 * np.pi * n0 / (n - 1))
             + a4 * np.cos(8 * np.pi * n0 / (n - 1)))
    elif kind == 'sinc_like':
        center = (n - 1) / 2.0
        x = np.arange(n) - center
        w = np.sinc(x / (n * 0.15))
        w = np.clip(w, 0.01, None)
    else:
        w = np.ones(n)
    w = np.clip(w, 1e-10, None).astype(np.float64)
    w /= w.sum()
    return w


def optimize_softmax_gradient(a, deadline, rng, verbose=False):
    """
    Gradient descent using soft-max approximation of max(conv(a,a)).
    Uses FFT for efficient gradient computation with annealed temperature.
    """
    n = len(a)
    a_work = a.copy().astype(np.float64)
    a_work = np.clip(a_work, 1e-10, None)
    a_work /= a_work.sum()

    best_a = a_work.copy()
    best_score = compute_objective(a_work)

    T = 3.0
    lr = 0.01
    stall = 0
    it = 0
    size = 4 * n

    while time.monotonic() < deadline:
        it += 1
        # Anneal temperature
        T_cur = 3.0 * (1.0 + 0.002 * it)
        T_cur = min(T_cur, 80.0)

        c = np.convolve(a_work, a_work, mode='full')
        c_max = c.max()
        c_shifted = c - c_max
        weights = np.exp(T_cur * c_shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights /= wsum

        # Gradient via FFT: grad[i] = 2 * sum_k weights[k] * a[k-i]
        W_fft = np.fft.fft(weights, size=size)
        A_fft = np.fft.fft(a_work, size=size)
        grad = 2.0 * np.fft.ifft(A_fft * np.conj(W_fft)).real[:n]

        # Remove mean to reduce effect of normalization
        grad -= grad.mean()

        # Step with backtracking
        accepted = False
        for trial in range(5):
            a_new = a_work - lr * grad
            a_new = np.clip(a_new, 1e-10, None)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-10:
                lr *= 0.5
                continue
            a_new /= a_new_sum
            sc = compute_objective(a_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_new.copy()
                a_work = a_new
                accepted = True
                break
            lr *= 0.5

        if accepted:
            stall = 0
            lr = min(0.05, lr * 1.05)
        else:
            stall += 1
            if stall % 20 == 0:
                # Adaptive perturbation
                noise_scale = min(0.1, 0.01 * (stall / 20.0))
                noise = rng.normal(0, noise_scale, n)
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-10, None)
                a_new_sum = a_new.sum()
                if a_new_sum > 1e-10:
                    a_new /= a_new_sum
                    sc_new = compute_objective(a_new)
                    if sc_new < best_score:
                        best_a = a_new
                        best_score = sc_new
                        a_work = a_new
                        stall = 0
                    elif rng.random() < 0.05 * np.exp(-stall / 50.0):
                        a_work = a_new
                    else:
                        a_work = best_a.copy()
            if stall > 300:
                break

        if verbose and it % 500 == 0:
            print(f"    softmax iter {it}, T={T_cur:.1f}, score={best_score:.6f}, lr={lr:.6f}, stall={stall}")

    return best_a, best_score


def optimize_lp_iterative(a, deadline, rng, verbose=False):
    """
    LP-based iterative optimization with adaptive step size and perturbations.
    """
    n = len(a)
    a_work = a.copy().astype(np.float64)
    a_work = np.clip(a_work, 1e-10, None)
    a_work /= a_work.sum()

    best_a = a_work.copy()
    best_score = compute_objective(a_work)
    step = 0.15
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a_work, a_work, mode='full')
        M = c.max()

        g = lp_direction(a_work, M)
        if g is None:
            break

        improved = False
        for shrink in [1.0, 0.6, 0.3, 0.1, 0.05, 0.02]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 1e-10, None)
            s_sum = a_new.sum()
            if s_sum < 1e-10:
                continue
            a_new /= s_sum
            sc = compute_objective(a_new)
            if sc < best_score - 1e-13:
                best_a = a_new
                best_score = sc
                a_work = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.4, step * 1.04)
        else:
            stall += 1
            step = max(0.002, step * 0.72)
            if stall % 15 == 0:
                noise_scale = min(0.12, 0.015 * (stall / 15.0))
                noise = rng.normal(0, noise_scale, n)
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-10, None)
                s_sum = a_new.sum()
                if s_sum > 1e-10:
                    a_new /= s_sum
                    sc = compute_objective(a_new)
                    if sc < best_score:
                        best_a = a_new
                        best_score = sc
                        a_work = a_new
                        stall = 0
                    else:
                        if rng.random() < 0.08 * np.exp(-stall / 40.0):
                            a_work = a_new
                        else:
                            a_work = best_a.copy()
            if stall > 150:
                break

        if verbose and it % 200 == 0:
            print(f"    LP iter {it}, score={best_score:.6f}, step={step:.4f}, stall={stall}")

    return best_a, best_score


def optimize_multiplicative(a, deadline, rng, verbose=False):
    """
    Multiplicative weights update: reduce positions contributing to peaks.
    """
    n = len(a)
    a_work = a.copy().astype(np.float64)
    a_work = np.clip(a_work, 1e-10, None)
    a_work /= a_work.sum()

    best_a = a_work.copy()
    best_score = compute_objective(a_work)
    eta = 0.25
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a_work, a_work, mode='full')
        c_max = c.max()
        threshold = c_max * 0.7
        peak_mask = (c > threshold).astype(np.float64)

        # contrib[i] = sum_j a[j] * peak_mask[i+j]
        # Use correlate: np.correlate(peak_mask, a, mode='full')[i] = sum_j peak_mask[j+i] * a[j]
        contrib = np.correlate(peak_mask, a_work, mode='full')[:n]

        avg_c = contrib.mean() + 1e-15
        update = np.exp(-eta * contrib / avg_c)
        a_new = a_work * update
        a_new = np.clip(a_new, 1e-10, None)
        s_sum = a_new.sum()
        if s_sum < 1e-10:
            break
        a_new /= s_sum

        sc = compute_objective(a_new)
        if sc < best_score - 1e-14:
            best_a = a_new
            best_score = sc
            a_work = a_new
            stall = 0
            eta = min(0.8, eta * 1.02)
        else:
            stall += 1
            eta = max(0.01, eta * 0.88)
            if stall % 15 == 0:
                noise = rng.normal(0, 0.04, n)
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-10, None)
                s_sum = a_new.sum()
                if s_sum > 1e-10:
                    a_new /= s_sum
                    if compute_objective(a_new) < best_score:
                        best_a = a_new
                        best_score = compute_objective(a_new)
                        a_work = a_new
                        stall = 0
                    elif rng.random() < 0.15:
                        a_work = a_new
                    else:
                        a_work = best_a.copy()
            if stall > 150:
                break

        if verbose and it % 500 == 0:
            print(f"    multi iter {it}, score={best_score:.6f}, eta={eta:.4f}")

    return best_a, best_score


def optimize_hybrid(a, deadline, rng, verbose=False):
    """
    Three-phase hybrid: multiplicative (fast) -> soft-max gradient (medium) -> LP (fine-tune).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 1e-10, None)
    a /= a.sum()

    remaining_total = deadline - time.monotonic()
    if remaining_total < 10:
        return a, compute_objective(a)

    # Phase 1: Multiplicative (fast initial progress) - 25% of time
    t1 = time.monotonic() + remaining_total * 0.25
    a, _ = optimize_multiplicative(a, t1, rng)
    print(f"    Phase 1 (multi) done: score={compute_objective(a):.6f}")

    # Phase 2: Soft-max gradient - 35% of time
    t2 = time.monotonic() + (deadline - time.monotonic()) * 0.4
    a, _ = optimize_softmax_gradient(a, t2, rng)
    print(f"    Phase 2 (softmax) done: score={compute_objective(a):.6f}")

    # Phase 3: LP-based fine-tuning - remaining time
    a, score = optimize_lp_iterative(a, deadline, rng, verbose=verbose)
    print(f"    Phase 3 (LP) done: score={score:.6f}")

    return a, score


def optimize_direct_lp(a, deadline, rng, verbose=False):
    """
    Pure LP-based optimization with more aggressive exploration.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 1e-10, None)
    a /= a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.2
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(best_a, best_a, mode='full')
        M = c.max()

        g = lp_direction(best_a, M)
        if g is None:
            # Try with slightly higher M
            g = lp_direction(best_a, M * 1.01)
            if g is None:
                break

        improved = False
        for shrink in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07, 0.03]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 1e-10, None)
            s_sum = a_new.sum()
            if s_sum < 1e-10:
                continue
            a_new /= s_sum
            sc = compute_objective(a_new)
            if sc < best_score - 1e-13:
                best_a = a_new
                best_score = sc
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.35, step * 1.03)
        else:
            stall += 1
            step = max(0.003, step * 0.75)
            if stall % 10 == 0:
                # Try multiple random perturbations
                found = False
                for _ in range(3):
                    noise_scale = min(0.1, 0.02 * (stall / 10.0))
                    noise = rng.normal(0, noise_scale, n)
                    a_new = best_a * (1.0 + noise)
                    a_new = np.clip(a_new, 1e-10, None)
                    s_sum = a_new.sum()
                    if s_sum > 1e-10:
                        a_new /= s_sum
                        sc = compute_objective(a_new)
                        if sc < best_score:
                            best_a = a_new
                            best_score = sc
                            stall = 0
                            found = True
                            break
                if not found and stall > 100:
                    break
            if stall > 200:
                break

        if verbose and it % 500 == 0:
            print(f"    direct LP iter {it}, score={best_score:.6f}, step={step:.4f}, stall={stall}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimization combining soft-max gradients, LP refinement,
    and multiplicative updates with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            if p.sum() > 0.01:
                parent = p / p.sum()
    except Exception:
        pass

    print(f"Budget: {budget_s}s, deadline in {budget_s - 20:.0f}s")
    if parent is not None:
        parent_score = compute_objective(parent)
        print(f"Parent: n={len(parent)}, score={parent_score:.6f}")

    # Build search plan: list of (n, init_kind, method, time_fraction)
    # Time fractions are relative weights
    plan = []

    # Parent-based (highest priority)
    if parent is not None:
        n_p = len(parent)
        plan.append((n_p, 'parent', 'hybrid', 2.5))
        if n_p >= 512:
            plan.append((n_p, 'parent', 'direct_lp', 1.5))
            plan.append((n_p, 'parent', 'softmax', 1.0))

    # Key configurations with various methods
    configs = [
        (512, 'gaussian', 'hybrid', 1.5),
        (512, 'blackman_harris', 'hybrid', 1.2),
        (512, 'flattop', 'hybrid', 1.0),
        (1024, 'cosine', 'hybrid', 1.5),
        (1024, 'gaussian', 'hybrid', 1.2),
        (1024, 'blackman_harris', 'direct_lp', 1.0),
        (1024, 'flattop', 'softmax', 0.8),
        (2048, 'cosine', 'multiplicative', 0.8),
        (2048, 'gaussian', 'multiplicative', 0.6),
        (256, 'blackman_harris', 'direct_lp', 0.8),
        (256, 'flattop', 'hybrid', 0.6),
        (512, 'sinc_like', 'hybrid', 0.5),
        (1024, 'parabolic', 'softmax', 0.5),
    ]
    plan.extend(configs)

    total_weight = sum(w for _, _, _, w in plan)
    available = budget_s - 80.0  # reserve 80s for final polish

    print(f"Search plan: {len(plan)} configs, total weight={total_weight:.1f}, available={available:.0f}s")

    executed = 0
    for idx, (n, init_kind, method, weight) in enumerate(plan):
        if time.monotonic() > deadline:
            break

        alloc_time = available * weight / total_weight
        config_deadline = min(time.monotonic() + alloc_time, deadline)
        if config_deadline <= time.monotonic() + 5:
            continue

        remaining_global = deadline - time.monotonic()
        print(f"\n[{executed+1}] n={n}, init={init_kind}, method={method}, alloc={alloc_time:.0f}s, global_rem={remaining_global:.0f}s")

        # Initialize
        if init_kind == 'parent' and parent is not None:
            a = parent.copy()
        elif init_kind == 'random':
            a = np.abs(rng.normal(0, 1, n))
        else:
            a = make_initialization(n, init_kind)

        a = np.clip(a, 1e-10, None).astype(np.float64)
        a_sum = a.sum()
        if a_sum < 1e-10:
            a = np.ones(n)
            a_sum = n
        a /= a_sum

        init_score = compute_objective(a)
        print(f"  Init score: {init_score:.6f}")

        # Run optimization
        if method == 'hybrid':
            a_opt, score = optimize_hybrid(a, config_deadline, rng)
        elif method == 'direct_lp':
            a_opt, score = optimize_direct_lp(a, config_deadline, rng)
        elif method == 'softmax':
            a_opt, score = optimize_softmax_gradient(a, config_deadline, rng)
        elif method == 'multiplicative':
            a_opt, score = optimize_multiplicative(a, config_deadline, rng)
        else:
            a_opt, score = optimize_hybrid(a, config_deadline, rng)

        executed += 1
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: {best_score:.6f} (n={n}, init={init_kind}, method={method})")
        else:
            print(f"  => score={score:.6f} (best: {best_score:.6f})")

    # Final polish phase
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== FINAL POLISH (score={best_score:.6f}, n={len(best_seq)}) ===")

        # LP polish
        polish_time = min(180, deadline - time.monotonic() - 30)
        if polish_time > 20:
            pd = time.monotonic() + polish_time
            best_seq, s1 = optimize_lp_iterative(best_seq, pd, rng, verbose=True)
            if s1 < best_score:
                best_score = s1
            print(f"  After LP polish: {best_score:.6f}")

        # Soft-max gradient polish
        if time.monotonic() < deadline - 30:
            sm_time = min(90, deadline - time.monotonic() - 20)
            if sm_time > 15:
                sd = time.monotonic() + sm_time
                bs, s2 = optimize_softmax_gradient(best_seq, sd, rng, verbose=False)
                if s2 < best_score:
                    best_seq = bs
                    best_score = s2
                print(f"  After softmax polish: {best_score:.6f}")

        # One more short LP pass
        if time.monotonic() < deadline - 20:
            fl = deadline - 5
            best_seq, s3 = optimize_lp_iterative(best_seq, fl, rng, verbose=False)
            if s3 < best_score:
                best_score = s3
            print(f"  Final: {best_score:.6f}")

    if best_seq is None:
        best_seq = make_initialization(512, 'gaussian')
        best_score = compute_objective(best_seq)

    # Ensure validity for evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / max(best_seq.sum(), 1e-30))

    elapsed = time.monotonic() - start_time
    print(f"\nDONE in {elapsed:.1f}s. Final: score={best_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
