    """Iterative spectral peak reduction.
    
    Compute autocorrelation, find peaks, modify spectrum to reduce them.
    """

    a_work = a.copy()

        size = 1
        while size < 2 * n:
            size <<= 1

        power = fa * np.conj(fa)  # |A|^2
        r = np.fft.irfft(power, size)[:n]  # circular autocorrelation
        
        S = np.sum(a_work)
        if S < 0.01:
            break
        
        M = np.max(r)
        J_current = 2.0 * n * M / (S * S)
        
        # Find the peak lag
        m = np.argmax(r)
        
        # For each frequency bin, compute its contribution to r[m]
        # r[m] = (1/size) * sum_f |A[f]|^2 * exp(2*pi*i*f*m/size)
        f = np.arange(np.size(power))
        phases = np.exp(2j * np.pi * f * m / size)
        contributions = power * phases

        max_contrib = np.max(np.abs(real_contrib))

        J_new = compute_J(a_new)
        if J_new < J_current:

def _fft_size(n):
    """Next power of 2 >= n."""
    size = 1
    while size < n:
        size <<= 1
    return size

def convolve_fft(a):
    """Self-convolution via FFT. Length 2n-1."""
    n = len(a)
