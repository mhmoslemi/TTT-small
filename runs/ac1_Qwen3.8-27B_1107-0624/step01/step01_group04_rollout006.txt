def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 1e-300, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def perturb_greedy(n, rng, sigma):
    p = greedy_normalized(n)
    noise = np.exp(rng.normal(0.0, float(sigma), int(n)))
    return normalize_nonneg(p * noise)

def actual_score_fft(p, direct_limit=512):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())

    if n <= direct_limit:

    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:n + m - 1].real


def autocorr_fft(a):
    n = a.size
    L = next_power_of_two(2 * n - 1)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:2 * n - 1].real


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def try_global_eval(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
        val = float(val)
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def record_candidate(p, best_p, best_score, best_by_n, score=None):
    if p is None:
        return best_p, best_score
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        return best_p, best_score
    if score is None:
        score = actual_score_fft(a)
    if not np.isfinite(score):
        return best_p, best_score
    if score < best_score:
        best_p = a.copy()
        best_score = float(score)
    n = int(a.size)
    old = best_by_n.get(n)
    if old is None or score < old[0]:
        best_by_n[n] = (float(score), a.copy())
    return best_p, best_score


def adam_optimize_v2(p, budget_s, seed=0):
    """
    Improved softmax-projected gradient descent with adaptive temperature
    and cosine annealing restarts for the smoothed max-convolution objective.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 1.0:
        f = actual_score_fft(p)
        return p, f

    x = np.log(p + 1e-300)
    x -= x.mean()

    # Adaptive learning rate based on n
    if n < 64:
        lr_base = 0.08
    elif n < 256:
        lr_base = 0.05
    elif n < 1024:
        lr_base = 0.03
    elif n < 4096:
        lr_base = 0.02
    else:
        lr_base = 0.012

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(2 * n - 1)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 3.0)

    # Adaptive temperature: start high, decay
    alpha_init = 10.0
    alpha_min = 3.0

    # Cosine restart schedule
    restart_period = max(20, int(budget_s / 8.0))

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Adaptive temperature
        alpha = alpha_min + (alpha_init - alpha_min) * np.exp(-3.0 * frac)

        # Cosine annealing with restarts for learning rate
        cycle = elapsed / restart_period
        cycle_frac = cycle - np.floor(cycle)
        lr = lr_base * 0.5 * (1.0 + np.cos(np.pi * cycle_frac))
        lr = max(lr, lr_base * 0.1)

        # Compute autocorrelation
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)  # p sums to one
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Softmax weights for smoothed max
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient: d(max_c)/dp ≈ 4n * conv(p, w_reversed) for the "tail" portion
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        # Adam update
        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        # Periodic random kick to escape local minima
        if t % 200 == 0:
            kick = np.random.default_rng(t * 7919).normal(0, 0.05, n)
            x = x + kick
            x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -70.0, 70.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def contribution_peel(p, budget_s, inner_iters=50):
    """
    Novel method: iteratively identify the lag with max autocorrelation,
    compute each coordinate's contribution to that lag, and adjust
    (reduce high-contribution, increase low-contribution coordinates).
    """
    p = normalize_nonneg(p.copy())
    n = p.size
    if n < 2:
        return p, actual_score_fft(p)

    best_p = p.copy()
    best_f = actual_score_fft(p)
    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)

    L = next_power_of_two(2 * n - 1)
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() < end and it < inner_iters:
        it += 1

        # Compute autocorrelation
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = float(2.0 * n * maxc)
        if f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Find top few lags (for a more robust gradient)
        top_k = min(10, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        # Weight by how much above the median
        med_c = float(np.median(c))
        weights = np.maximum(top_vals - med_c, 0.0)
        wsum = float(weights.sum())
        if wsum <= 0:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / wsum

        # Compute gradient contribution for each coordinate
        # d conv[k]/dp_j = p[k-j] + p[j-k] (with boundary) for k >= 0
        # More precisely, for full convolution: conv[k] = sum_i p[i]*p[k-i]
        # d/dp_j conv[k] = sum_i (delta_{j,i} p[k-i] + p[i] delta_{j,k-i})
        #                  = p[k-j] (if 0<=k-j<n) + p[j-k+n-1-... ] 
        # Actually for np.convolve(p,p), conv[k] = sum_i p[i] * p[k-i] where
        # the second p is not flipped. So d/dp_j conv[k] = p[k-j] + p[j-k]
        # with appropriate zero-padding.
        
        grad = np.zeros(n, dtype=np.float64)
        P = np.fft.rfft(p, L)
        for idx_i, k in enumerate(top_indices):
            k = int(k)
            w = float(weights[idx_i])
            # conv[k] = sum_i p[i] * p[k-i]
            # d/dp_j = p[k-j] + p[j-k] (with bounds 0<=index<n)
            # Use FFT: correlate p with delta at position k
            # Actually, let's just compute directly for efficiency with top_k small
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # Contribution to conv[k] from p[j] for j in [lo, hi]:
                # p[j] appears in terms: p[j]*p[k-j] and p[k-j]*p[j]
                # If k-j == j (i.e., k even, j=k/2), it's just one term
                # Otherwise it's two terms
                # So d conv[k]/dp[j] = 2*p[k-j] (for j != k-j) or p[k-j] (for j == k-j)
                for j in range(lo, hi + 1):
                    partner = k - j
                    if 0 <= partner < n:
                        if j == partner:
                            grad[j] += w * p[partner]
                        else:
                            grad[j] += w * 2.0 * p[partner]

        if not np.all(np.isfinite(grad)) or float(np.abs(grad).max()) <= 0:
            break

        # Adjust: reduce high-contribution coords, increase low-contribution
        # Use exponential adjustment
        scale = 0.3 + 0.2 * (it / max(1, inner_iters))
        g_max = float(np.abs(grad).max())
        if g_max > 0:
            grad_normed = grad / g_max
            adjustment = np.exp(-scale * grad_normed)
            # Ensure non-negative
            adjustment = np.maximum(adjustment, 0.01)
            p_new = p * adjustment
            sn = float(p_new.sum())
            if sn > 1e-12 and np.isfinite(sn):
                p_new = p_new / sn
                f_new = actual_score_fft(p_new)
                if np.isfinite(f_new) and f_new < best_f:
                    p = p_new
                    best_f = f_new
                    best_p = p.copy()
                elif np.isfinite(f_new):
                    # Try smaller step
                    for frac in (0.5, 0.25, 0.1):
                        p_try = p * (1.0 - frac * (1.0 - adjustment))
                        p_try = np.maximum(p_try, 1e-15)
                        st = float(p_try.sum())
                        if st > 1e-12:
                            p_try = p_try / st
                            f_try = actual_score_fft(p_try)
                            if np.isfinite(f_try) and f_try < best_f:
                                p = p_try
                                st = float(p.sum())
                                p = p / st
                                best_f = f_try
                                best_p = p.copy()
                                break

    return best_p, best_f


def lp_improve_v2(p, budget_s, max_rows=None):
    """
    Improved LP cutting-plane with smart initial row selection and
    backtracking line search.
    """
    if _scipy_opt is None:
        return p, actual_score_fft(p)

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4:
        return p, actual_score_fft(p)
    budget_s = float(budget_s)
    if budget_s < 30.0:
        return p, actual_score_fft(p)

    if max_rows is None:
        if n > 4096:
            max_rows = 600
        elif n > 1500:
            max_rows = 800
        else:
            max_rows = 1000

    # Limit total matrix elements for memory
    max_elements = 8000000
    while max_rows * n > max_elements:
        max_rows = max(100, int(max_rows * 0.8))
    max_rows = max(100, min(int(max_rows), max_elements // max(1, n)))

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score_fft(f)
    if not np.isfinite(cur_score):
        return p, cur_score

    best_f = f.copy()
    best_score = float(cur_score)
    end = time.monotonic() + max(0.0, budget_s - 8.0)

    # Smart initial rows: peaks of autocorrelation + uniform
    L_init = next_power_of_two(2 * n - 1)
    F = np.fft.rfft(f, L_init)
    c_init = np.fft.irfft(F * F, L_init)[:2 * n - 1].real
    c_init = np.maximum(c_init, 0.0)

    # Take top peaks + evenly spaced
    num_uniform = min(max_rows // 2, 2 * n - 1)
    uniform_rows = set(np.linspace(0, 2 * n - 2, num_uniform, dtype=int).tolist())
    top_peaks = np.argsort(c_init)[::-1][:min(100, 2 * n - 1)]
    for k in top_peaks:
        uniform_rows.add(int(k))
    rows = list(uniform_rows)[:max_rows]
    row_set = set(rows)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        n_rows = int(rows_arr.size)

        # Build constraint matrix A (sparse-friendly but using dense for simplicity)
        A = np.zeros((n_rows, n), dtype=np.float64)
        for ii in range(n_rows):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                seg = f[k - hi:k - lo + 1][::-1]
                A[ii, lo:hi + 1] = seg

        b = np.full(n_rows, cur_score, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 2.0)
        tl = max(5.0, min(60.0, rem * 0.3))
        res = None
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None), method="highs",
                options={"time_limit": float(tl)}
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False):
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        # Check full constraint violations
        if n > 512:
            L2 = next_power_of_two(2 * n - 1)
            Pf = np.fft.rfft(f, L2)
            Pg = np.fft.rfft(g, L2)
            conv = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        else:
            conv = np.convolve(f, g)
        conv = np.maximum(conv, 0.0)
        tol = 1e-6 + 1e-6 * cur_score
        violations = conv - cur_score
        max_viol = float(np.max(violations)) if violations.size > 0 else 0.0

        if max_viol > tol:
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    if added >= min(200, max_rows - len(rows)):
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                if added > 0:
                    continue
            elif max_viol > 10.0 * tol:
                break

        # Scale g to match normalization
        g = g * (S0 / sg)

        # Backtracking line search
        accepted_score = best_score
        accepted = best_f.copy()
        for step in (1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()

        if accepted_score >= best_score - 1e-10:
            break

        f = accepted
        cur_score = accepted_score
        best_f = f.copy()
        best_score = cur_score

        if it % 3 == 0:
            print("  LP n=%d iter=%d score=%.6f rows=%d t=%.1fs" % (
                n, it, best_score, len(rows), time.monotonic() - (end - budget_s + 8.0)))

    return normalize_nonneg(best_f), best_score


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    print("Starting search at t=0, budget=%.0fs" % budget_s)

    # --- Phase 1: Quick baselines ---
    base_lengths = [16, 32, 64, 128, 256, 384, 512, 768, 1024, 1536, 2048, 3840, 4096]
    for n in base_lengths:
        if time.monotonic() >= deadline:
            break
        for kind in ["greedy", "power04", "power06", "sym"]:
            if kind == "greedy":
                p = greedy_normalized(n)
            elif kind == "power04":
                p = power_normalized(n, 0.4)
            elif kind == "power06":
                p = power_normalized(n, 0.6)
            else:
                p = symmetric_greedy(n)
            best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)

    # Load previous best if available
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                best_p, best_score = record_candidate(arr, best_p, best_score, best_by_n)
                print("Loaded height_sequence_1: n=%d, score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    print("After baselines: best_score=%.6f" % best_score)

    # --- Phase 2: Adam optimization from diverse starts ---
    jobs = [
        # (n, kind, time_alloc)
        (128, "greedy", 12),
        (192, "greedy", 15),
        (256, "greedy", 20),
        (256, "pert10", 20),
        (256, "power04", 18),
        (384, "greedy", 25),
        (384, "pert15", 25),
        (384, "power05", 22),
        (512, "greedy", 30),
        (512, "pert20", 30),
        (512, "mix50", 28),
        (512, "power04", 28),
        (768, "greedy", 35),
        (768, "pert15", 35),
        (768, "power05", 30),
        (1024, "greedy", 40),
        (1024, "pert10", 40),
        (1024, "mix75", 35),
        (1536, "greedy", 40),
        (1536, "pert10", 35),
        (2048, "greedy", 45),
        (2048, "pert05", 40),
        (3840, "greedy", 40),
        (4096, "greedy", 45),
    ]

    for n, kind, alloc in jobs:
        rem = deadline - time.monotonic()
        if rem < 20.0:
            break

        if kind == "greedy":
            base = greedy_normalized(n)
        elif kind == "pert05":
            base = perturb_greedy(n, rng, 0.05)
        elif kind == "pert10":
            base = perturb_greedy(n, rng, 0.10)
        elif kind == "pert15":
            base = perturb_greedy(n, rng, 0.15)
        elif kind == "pert20":
            base = perturb_greedy(n, rng, 0.20)
        elif kind == "power04":
            base = power_normalized(n, 0.4)
        elif kind == "power05":
            base = power_normalized(n, 0.5)
        elif kind == "mix50":
            base = mix_greedy_uniform(n, 0.5)
        elif kind == "mix75":
            base = mix_greedy_uniform(n, 0.75)
        else:
            base = greedy_normalized(n)

        alloc_use = min(float(alloc), max(8.0, rem - 5.0))
        p_out, f_out = adam_optimize_v2(base, alloc_use, seed=(int(seed) + int(n)))
        best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)

        if int(n) in best_by_n:
            print("  Adam n=%d kind=%s score=%.6f (global best=%.6f) t=%.0fs" % (
                n, kind, best_by_n[n][0], best_score, time.monotonic() - t0))

    print("After Adam phase: best_score=%.6f" % best_score)

    # --- Phase 3: Contribution peeling on best candidates ---
    peel_candidates = []
    for n in sorted(best_by_n.keys()):
        rec = best_by_n[n]
        peel_candidates.append((rec[0], n))
    peel_candidates.sort(key=lambda x: x[0])

    for sc, n in peel_candidates[:5]:
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break
        base = best_by_n[n][1]
        alloc = min(60.0, max(15.0, rem * 0.15))
        p_out, f_out = contribution_peel(base, alloc)
        best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
        print("  Peel n=%d score=%.6f (global best=%.6f) t=%.0fs" % (
            n, f_out, best_score, time.monotonic() - t0))

    # --- Phase 4: LP cutting-plane ---
    if _scipy_opt is not None:
        lp_targets = [256, 384, 512, 768, 1024, 1536, 2048, 3840]
        if best_p is not None:
            ln = int(np.asarray(best_p).size)
            if ln not in lp_targets:
                lp_targets.append(ln)
        if h is not None:
            try:
                ln = int(np.asarray(h).size)
                if ln not in lp_targets:
                    lp_targets.append(ln)
            except Exception:
                pass

        # Sort by current best score for that n
        lp_scored = []
        for n in lp_targets:
            rec = best_by_n.get(int(n))
            sc = rec[0] if rec is not None else 2.0
            lp_scored.append((sc, n))
        lp_scored.sort(key=lambda x: x[0])

        for _, n in lp_scored:
            rem = deadline - time.monotonic()
            if rem < 80.0:
                break
            rec = best_by_n.get(int(n))
            base = rec[1] if rec is not None else greedy_normalized(n)
            alloc = min(200.0, max(50.0, 0.35 * rem))
            p_out, f_out = lp_improve_v2(base, alloc)
            best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
            print("  LP n=%d score=%.6f (global best=%.6f) t=%.0fs" % (
                n, f_out, best_score, time.monotonic() - t0))

    # --- Phase 5: Final Adam polish on global best ---
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 60.0:
            # Polish from current best
            p0 = normalize_nonneg(best_p)
            p_out, f_out = adam_optimize_v2(p0, min(rem - 10.0, 180.0), seed=int(seed) + 777)
            best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
            print("  Polish1 score=%.6f t=%.0fs" % (best_score, time.monotonic() - t0))

        rem = deadline - time.monotonic()
        if rem > 40.0:
            # Polish with perturbation
            p0 = random_perturb(best_p, rng, 0.05)
            p_out, f_out = adam_optimize_v2(p0, min(rem - 5.0, 120.0), seed=int(seed) + 888)
            best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
            print("  Polish2 score=%.6f t=%.0fs" % (best_score, time.monotonic() - t0))

        # One more peel on the final best
        rem = deadline - time.monotonic()
        if rem > 30.0:
            p_out, f_out = contribution_peel(best_p, min(rem - 5.0, 60.0))
            best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
            print("  FinalPeel score=%.6f t=%.0fs" % (best_score, time.monotonic() - t0))

    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = actual_score_fft(best_p)

    print("Final best_score=%.6f, n=%d" % (best_score, best_p.size))

    # --- Validation with global evaluator ---
    candidates = [best_p]
    items = sorted(best_by_n.items(), key=lambda x: x[1][0])
    for n, (sc, p) in items[:3]:
        candidates.append(p)

    for cand in candidates:
        if deadline - time.monotonic() < 10.0:
            break
        try:
            seq = scale_output(cand, 10.0)
            gscore = try_global_eval(seq)
            if np.isfinite(gscore):
                print("  Validation: n=%d fft_score=%.6f global_score=%.6f" % (
                    cand.size, actual_score_fft(cand), gscore))
                if gscore < best_score:
                    best_score = gscore
                    best_p = cand.copy()
        except Exception:
            pass

    try:
        final_seq = scale_output(best_p, 10.0)
    except Exception:
        final_seq = scale_output(greedy_normalized(384), 10.0)

    print("DONE. Returning sequence of length %d, estimated score %.6f" % (
        len(final_seq), best_score))
    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation function."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Main search failed: %s" % str(e))
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
