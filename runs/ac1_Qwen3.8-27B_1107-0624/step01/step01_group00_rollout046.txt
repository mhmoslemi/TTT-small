    n = len(a)

    cw = c[top_idx]

    w /= w.sum()
    
    grad = np.zeros(n)

        idx = np.arange(max(0, k-n+1), min(k, n-1)+1)
        src = k - idx
        grad[idx] += 2.0 * wk * a[src]

    a_new = a * np.exp(-eta * grad)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        a_new = a
    return a_new, grad

    if best_a is not None and time.monotonic() < t_end - 5.0:
        time_left = t_end - time.monotonic() - 3.0
        print(f"Polishing best solution (n={best_n}), time_left={time_left:.1f}s")
        a, score = _polish(best_a, best_n, time_left, rng)
        if a is not None:
            try:
                official = evaluate_sequence([float(x) for x in a])
                if official < best_score:
                    best_score = official
                    best_a = a.copy()
                    print(f"  Polished: score={best_score:.6f}")
            except Exception:
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

    # If we still have time, try a few more random restarts on best_n
    if best_a is not None and time.monotonic() < t_end - 30.0:
        time_left = t_end - time.monotonic() - 10.0
        n = best_n
        # Try perturbed versions of the best
        for trial in range(5):
            if time.monotonic() > t_end - 15.0:
                break
            alloc = time_left / (5 - trial)
            a_pert = best_a * np.exp(rng.normal(0, 0.2, n))
            a_pert = np.maximum(a_pert, 0.0)
            s = a_pert.sum()
            if s > 1e-10:
                a_pert /= s
            # Quick local optimization
            a2, sc2 = _optimize_n(n, alloc, rng, a_pert)
            if a2 is not None:
                try:
                    official = evaluate_sequence([float(x) for x in a2])
                    if official < best_score:
                        best_score = official
                        best_a = a2.copy()
                        print(f"  Restart {trial}: NEW BEST score={best_score:.6f}")
                except Exception:
                    if sc2 < best_score:
                        best_score = sc2
                        best_a = a2.copy()

    # Final output
    if best_a is None:
        # Fallback: uniform sequence
        n = 512
        best_a = np.full(n, 1.0)
        try:
            best_score = evaluate_sequence([float(x) for x in best_a])
        except Exception:
            best_score = 2.0

    # Ensure valid output
    out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.full(best_n if best_n > 0 else 512, 1.0)

    try:
        final = evaluate_sequence([float(x) for x in out])
        print(f"FINAL: n={len(out)}, score={final:.6f}")
    except Exception:
        print(f"FINAL: n={len(out)}, score (local)={best_score:.6f}")

    return [float(x) for x in out]
