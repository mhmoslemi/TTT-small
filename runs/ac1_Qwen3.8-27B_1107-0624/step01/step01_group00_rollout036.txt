Minimize 2*n*max(convolve(a,a)) / (sum a)^2 for non-negative a.
Uses Nesterov-accelerated smoothed-subgradient descent with FFT-based
convolution, multiple diverse seeds, and adaptive parameter scheduling.

import time
import math
import numpy as np


def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(np.arange(1, n+1) * u > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n, dtype=np.float64)
    lam = (cssv[rho[-1]] - target) / (rho[-1] + 1.0)
    return np.maximum(v - lam, 0.0)


def _fft_len(n):
    """Next power of 2 >= n."""
    m = 1
    while m < n:
        m <<= 1
    return m


def _compute_conv(a, fft_m):
    """Compute linear convolution a*a via FFT. Returns array of length 2*len(a)-1."""
    fa = np.fft.rfft(a, fft_m)
    c = np.fft.irfft(fa * fa, fft_m)
    n = a.size
    return c[:2*n-1]


def _compute_grad(a, w, fft_m3):
    """Compute gradient 2*cross_correlate(a, w)[:n] via FFT.
    w has length 2*n-1 (the convolution length).
    """
    n = a.size
    fa = np.fft.rfft(a, fft_m3)
    fw = np.fft.rfft(w, fft_m3)
    grad = np.fft.irfft(np.conj(fa) * fw, fft_m3)[:n]
    return 2.0 * grad


def _smoothed_max_and_weights(c, top_k, temp):
    """Compute smoothed max value and softmax weights over top-k lags."""
    n = c.size
    k = min(top_k, n)
    # Find top-k indices
    if k == n:
        idx = np.arange(n)
        vals = c
    else:
        # Partial sort for efficiency
        part = np.argpartition(-c, k-1)[:k]
        idx = part
        vals = c[part]
    # Softmax
    max_val = vals.max()
    exp_vals = np.exp((vals - max_val) / temp)
    w = np.zeros(n, dtype=np.float64)
    w[idx] = exp_vals / exp_vals.sum()
    smooth_val = temp * (math.log(exp_vals.sum()) + max_val / temp)
    return smooth_val, w

def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cs = np.arange(1, n + 1) * u > (cssv - target)
    if not cs.any():
        return np.full(n, target / n, dtype=np.float64)
    rho = np.where(cs)[0][-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)

"""
Minimize 2*n*max(convolve(a,a))/(sum a)^2 for non-negative a, using
L-BFGS with softmax parameterization and a continuation (annealing) schedule
on the smoothed-max temperature. Multiple diverse seeds are optimized in
parallel; the best candidate is polished in a final phase.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


def _fft_pow2(n):
    m = 1
    while m < n:
        m <<= 1
    return m


def _softmax(z):
    z = z - z.max()
    ez = np.exp(z)
    s = ez.sum()
    if s < 1e-300:
        return np.full(z.size, 1.0 / z.size)
    return ez / s


def _conv_sq_from_a(a, m2):
    fa = np.fft.rfft(a, m2)
    c = np.fft.irfft(fa * fa, m2)
    return c[: 2 * a.size - 1]


def _grad_a_from_w(a, w, m3):
    fa = np.fft.rfft(a, m3)
    fw = np.fft.rfft(w, m3)
    g = np.fft.irfft(np.conj(fa) * fw, m3)[: a.size]
    return 2.0 * g


def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = np.arange(1, n + 1) * u > (cssv - target)
    if not cond.any():
        return np.full(n, target / n)
    rho = np.where(cond)[0][-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _make_fun_grad(n, m2, m3, top_k, temp):
    """Return (fun, grad) for L-BFGS operating on z (softmax params)."""

    def fun(z):
        a = _softmax(z)
        c = _conv_sq_from_a(a, m2)
        k = min(top_k, c.size)
        if k >= c.size:
            vals = c
            idx = np.arange(c.size)
        else:
            part = np.argpartition(-c, k - 1)[:k]
            idx = part
            vals = c[part]
        mx = vals.max()
        ev = np.exp((vals - mx) / temp)
        denom = ev.sum()
        smooth_obj = temp * (math.log(denom) + mx / temp)
        return smooth_obj

    def grad(z):
        a = _softmax(z)
        c = _conv_sq_from_a(a, m2)
        k = min(top_k, c.size)
        if k >= c.size:
            vals = c
            idx = np.arange(c.size)
        else:
            part = np.argpartition(-c, k - 1)[:k]
            idx = part
            vals = c[part]
        mx = vals.max()
        ev = np.exp((vals - mx) / temp)
        w = np.zeros(c.size, dtype=np.float64)
        w[idx] = ev / ev.sum()
        ga = _grad_a_from_w(a, w, m3)
        mg = np.dot(ga, a)
        gz = a * (ga - mg)
        return gz

    return fun, grad


def _lbfgs_single(z_init, n, m2, m3, top_k, temp, max_iter, ftol, gtol):
    fun, grad = _make_fun_grad(n, m2, m3, top_k, temp)
    res = minimize(
        fun,
        z_init,
        jac=grad,
        method="L-BFGS-B",
        options={
            "maxiter": max_iter,
            "ftol": ftol,
            "gtol": gtol,
            "maxfun": max_iter * 3,
            "maxls": 40,
        },
    )
    return res.x, res.fun, res.nit, res.success


def _continuation(z_init, n, m2, m3, t_end, n_temps, verbose=False):
    """Continuation: anneal temperature from high to low."""
    z = z_init.copy()
    # Log-spaced temperatures
    t_high = 5e-4
    t_low = 5e-7
    if n_temps > 1:
        logs = np.linspace(math.log10(t_high), math.log10(t_low), n_temps)
        temps = 10.0 ** logs
    else:
        temps = [t_high]
    # top_k schedule: start wide, narrow down
    k_start = 300
    k_end = 5
    top_ks = [max(k_end, int(k_start * (k_end / k_start) ** (i / max(1, n_temps - 1)))) for i in range(n_temps)]

    best_z = z.copy()
    best_val = float("inf")

    for i in range(n_temps):
        if time.monotonic() > t_end - 5:
            break
        temp = temps[i]
        tk = top_ks[i]
        # More iterations at higher temps (easier landscape)
        max_it = 800 if i < n_temps // 3 else 500
        z, val, nit, ok = _lbfgs_single(z, n, m2, m3, tk, temp, max_it, 1e-14, 1e-10)
        if val < best_val:
            best_val = val
            best_z = z.copy()
        if verbose:
            print(f"    T={temp:.2e} k={tk} val={val:.10f} it={nit} ok={ok}")

    return _softmax(best_z)


def _manual_descent(a_init, n, m2, m3, t_end, top_k_init, temp_init, lr_init):
    """Fallback: manual gradient descent with Nesterov momentum on the simplex."""
    a = np.abs(a_init).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    z = np.log(a + 1e-30)
    temp = temp_init
    top_k = top_k_init
    lr = lr_init
    momentum = 0.85
    v = np.zeros(n, dtype=np.float64)
    best_a = a.copy()
    best_mc = _conv_sq_from_a(a, m2).max()
    step = 0
    patience = 0

    while time.monotonic() < t_end - 2:
        a = _softmax(z)
        c = _conv_sq_from_a(a, m2)
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive
        if step % 200 == 0:
            temp = max(1e-6, temp * 0.92)
            top_k = max(5, int(top_k * 0.95))
            if patience > 100:
                lr = max(1e-7, lr * 0.7)
                patience = 0

        k = min(top_k, c.size)
        if k >= c.size:
            vals = c
            idx = np.arange(c.size)
        else:
            part = np.argpartition(-c, k - 1)[:k]
            idx = part
            vals = c[part]
        mx = vals.max()
        ev = np.exp((vals - mx) / temp)
        w = np.zeros(c.size, dtype=np.float64)
        w[idx] = ev / ev.sum()
        ga = _grad_a_from_w(a, w, m3)
        mg = np.dot(ga, a)
        gz = a * (ga - mg)

        gn = np.linalg.norm(gz)
        if gn > 1e-15:
            gz = gz / gn

        v = momentum * v - lr * gz
        z_new = z + v
        z_new = z_new - z_new.mean()  # recenter for stability
        z = z_new
        step += 1

    return best_a
