def make_cosine_mod(n, freq, eps, phase=0.0):
    """a[i] = (1 + eps*cos(2*pi*freq*i/n + phase))^2, clipped non-negative."""
    x = np.arange(n)
    base = 1.0 + eps * np.cos(2.0 * np.pi * freq * x / n + phase)
    return np.maximum(base ** 2, 0.0)


def make_multi_cosine(n, freqs, eps):
    """a[i] = (1 + sum eps_j * cos(2*pi*f_j*i/n))^2"""
    x = np.arange(n)
    base = np.ones(n)
    for f in freqs:
        base += eps * np.cos(2.0 * np.pi * f * x / n)
    return np.maximum(base ** 2, 0.0)


def run_gradient_descent(a_init, budget_s, seed, method='lse', tau_start=10.0, tau_end=0.1, lr_start=0.05, p_start=2, p_end=50):
    """Adam gradient descent with adaptive smoothing."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    # Normalize: scale so sum = n (arbitrary normalization)
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    start_time = time.monotonic()
    last_print = 0.0
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        
        # Annealing schedule (cosine)
        cos_frac = 0.5 * (1 + np.cos(np.pi * frac))
        
        if method == 'lse':
            tau = tau_end + (tau_start - tau_end) * cos_frac
            J, grad = logsumexp_obj_and_grad(a, tau)
        else:
            p = p_end + (p_start - p_end) * cos_frac
            J, grad = power_mean_obj_and_grad(a, p)
        
        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        # Learning rate with decay
        lr = lr_start / (1.0 + 0.3 * t / 500.0)
        
        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        
        # Periodic check
        if t % 500 == 0:
            local_J = compute_J(a)
            if local_J < best_J:

                best_a = a.copy()
                if time.monotonic() - last_print > 5.0:

                    last_print = time.monotonic()

                m = np.zeros(n)
                v = np.zeros(n)
                t = 0

    return best_a, min(best_J, final_J)

"""
Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Phase 1: Structured initializations + short gradient descent (power-mean smoothing).
Phase 2: Extended gradient descent with annealed smoothing.
Phase 3: Coordinate descent with exact 1D golden-section line search.
Phase 4: Peak-targeted refinement focusing on high-convolution contributions.

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def logsumexp_obj_grad(a, tau):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(np.clip(shifted, -500, 500))
    sum_exp = np.sum(exp_shifted)
    Tse = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    J = 2.0 * n * Tse / (S * S)
    # Gradient via cross-correlation
    p_flip = p[::-1]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_flip, size)
    conv = np.fft.irfft(fa * fp, size)[:2*n-1]
    cross = conv[n-1::-1][:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tse) / (S * S * S)
    return J, grad


def power_mean_obj_grad(a, p):
    """Power-mean smoothed objective and gradient."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c = np.maximum(c, 1e-30)
    cp = c ** p
    sum_cp = np.sum(cp)
    Tm = sum_cp ** (1.0 / p)
    J = 2.0 * n * Tm / (S * S)
    w = c ** (p - 1.0)
    w_flip = w[::-1]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_flip, size)
    conv = np.fft.irfft(fa * fw, size)[:2*n-1]
    cross = conv[n-1::-1][:n]
    grad_Tm = 2.0 * (sum_cp ** (1.0/p - 1.0)) * cross
    grad = (grad_Tm * 2.0 * n) / (S * S) - (4.0 * n * Tm) / (S * S * S)
    return J, grad


def make_cosine_mod(n, freq, eps, phase=0.0):
    """a[i] = (1 + eps*cos(2*pi*freq*i/n + phase))^2."""
    x = np.arange(n)
    base = 1.0 + eps * np.cos(2.0 * np.pi * freq * x / n + phase)
    return np.maximum(base ** 2, 1e-6)


def make_multi_cosine(n, freqs, eps):
    """a[i] = (1 + sum eps*cos(2*pi*f_j*i/n))^2."""
    x = np.arange(n)
    base = np.ones(n)
    for f in freqs:
        base += eps * np.cos(2.0 * np.pi * f * x / n)
    return np.maximum(base ** 2, 1e-6)


def make_sine_mod(n, freq, eps, phase=0.0):
    """a[i] = (1 + eps*sin(2*pi*freq*i/n + phase))^2."""
    x = np.arange(n)
    base = 1.0 + eps * np.sin(2.0 * np.pi * freq * x / n + phase)
    return np.maximum(base ** 2, 1e-6)


def make_asymmetric(n):
    """Asymmetric: high at one end, low at other, with cosine modulation."""
    x = np.arange(n)
    t = x / (n - 1)
    base = 1.0 + 0.8 * np.cos(np.pi * t)
    return np.maximum(base ** 2, 1e-6)


def make_double_cos(n, f1=3, f2=7, eps1=0.5, eps2=0.3):
    """Two-frequency cosine modulation."""
    x = np.arange(n)
    base = 1.0 + eps1 * np.cos(2.0 * np.pi * f1 * x / n) + eps2 * np.cos(2.0 * np.pi * f2 * x / n)
    return np.maximum(base ** 2, 1e-6)


def run_gradient_descent(a_init, budget_s, seed, method='lse', tau_start=10.0, tau_end=0.05, lr_start=0.05, p_start=2.0, p_end=100.0):
    """Adam gradient descent with adaptive smoothing parameter."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    start_time = time.monotonic()
    last_print = 0.0
    stall_count = 0
    stall_J = best_J

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        cos_frac = 0.5 * (1.0 + np.cos(np.pi * frac))

        if method == 'lse':
            tau = tau_end + (tau_start - tau_end) * cos_frac
            J, grad = logsumexp_obj_grad(a, tau)
        else:
            p = p_end + (p_start - p_end) * cos_frac
            J, grad = power_mean_obj_grad(a, p)

        if not np.isfinite(J) or not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        lr = lr_start / (1.0 + 0.5 * t / 1000.0)
        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)

        if t % 300 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall_count = 0
                stall_J = local_J
                if time.monotonic() - last_print > 8.0:
                    print(f"    GD t={t} J={best_J:.6f} frac={frac:.2f}")
                    last_print = time.monotonic()
            else:
                stall_count += 1
                if stall_count > 10:
                    a = best_a.copy()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    stall_count = 0

    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def coord_step(a, c, S, i, max_iter=35):
    """Optimize single coordinate i via golden section search. Returns new value."""
    n = len(a)
    old_val = a[i]
    S_fixed = S - old_val
    k_start = i
    k_end = min(i + n, 2 * n - 1)
    num_aff = k_end - k_start
    slopes = a[:num_aff].copy()
    base_vals = (c[k_start:k_end] - 2.0 * old_val * slopes).copy()
    k_quad = 2 * i
    has_quad = (k_start <= k_quad < k_end)
    if has_quad:
        quad_idx = k_quad - k_start
        base_quad = c[k_quad] - old_val * old_val
        base_vals[quad_idx] = base_quad
    unaff_max = 0.0
    if k_start > 0:
        unaff_max = max(unaff_max, np.max(c[:k_start]))
    if k_end < 2 * n - 1:
        unaff_max = max(unaff_max, np.max(c[k_end:]))

    def obj(x):
        if S_fixed + x < 0.01:
            return 1e15
        vals = base_vals + 2.0 * x * slopes
        if has_quad:
            vals[quad_idx] = base_quad + x * x
        m = max(unaff_max, np.max(vals))
        s = S_fixed + x
        return 2.0 * n * m / (s * s)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(2.0 * S_fixed / n, old_val * 2.5, 0.5)
    hi = min(hi, 2000.0)
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj(c1)
    f2 = obj(c2)
    for _ in range(max_iter):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
    new_val = (lo + hi) / 2.0
    return max(0.0, min(1000.0, new_val))


def apply_coord_update(a, c, S, i, new_val):
    """Apply coordinate update to a, c, S. Returns new S."""
    old_val = a[i]
    delta = new_val - old_val
    if abs(delta) < 1e-12:
        return S
    k_start = i
    k_end = min(i + n_global, 2 * n_global - 1)
    num_aff = k_end - k_start
    c[k_start:k_end] += 2.0 * delta * a[:num_aff]
    k_quad = 2 * i
    if k_start <= k_quad < k_end:
        c[k_quad] += delta * delta
    a[i] = new_val
    return S + delta


def coordinate_descent(a_init, budget_s, seed):
    """Coordinate descent with random ordering."""
    global n_global
    n = len(a_init)
    n_global = n
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()
    c = convolve_fft(a)
    S = np.sum(a)
    best_a = a.copy()
    best_J = 2.0 * n * np.max(c) / (S * S)
    sweep_count = 0
    last_print = 0.0
    no_improve_sweeps = 0

    while time.monotonic() < deadline:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline:
                break
            new_val = coord_step(a, c, S, i, max_iter=30)
            if abs(new_val - a[i]) > 1e-10:
                S = apply_coord_update(a, c, S, i, new_val)
                improved = True

        sweep_count += 1
        current_J = 2.0 * n * np.max(c) / (S * S)
        if current_J < best_J - 1e-10:
            best_J = current_J
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if time.monotonic() - last_print > 15.0:
            print(f"    CD sweep {sweep_count}: best_J={best_J:.6f} current={current_J:.6f}")
            last_print = time.monotonic()

        if no_improve_sweeps > 20:
            break

    return best_a, best_J


def peak_targeted_cd(a_init, c_init, S_init, budget_s, seed):
    """Coordinate descent focusing on coordinates that contribute to top peaks."""
    global n_global
    n = len(a_init)
    n_global = n
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()
    c = c_init.copy()
    S = S_init
    best_a = a.copy()
    best_J = 2.0 * n * np.max(c) / (S * S)
    last_print = 0.0
    iter_count = 0

    while time.monotonic() < deadline:
        # Find top peaks
        top_k = np.argsort(c)[-20:]
        # For each peak, find contributing coordinates
        priority = np.zeros(n)
        for k in top_k:
            # c[k] involves a[j]*a[k-j] for valid j
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                if j != k - j:
                    priority[j] += c[k]
                    priority[k - j] += c[k]
                else:
                    priority[j] += c[k]
        # Create priority-based ordering
        order = np.argsort(-priority)
        # Mix with some random
        for idx in range(0, n, 10):
            swap = rng.integers(idx, n)
            order[idx], order[swap] = order[swap], order[idx]

        for i in order:
            if time.monotonic() > deadline:
                break
            new_val = coord_step(a, c, S, i, max_iter=25)
            if abs(new_val - a[i]) > 1e-10:
                S = apply_coord_update(a, c, S, i, new_val)

        iter_count += 1
        current_J = 2.0 * n * np.max(c) / (S * S)
        if current_J < best_J - 1e-10:
            best_J = current_J
            best_a = a.copy()

        if time.monotonic() - last_print > 15.0:
            print(f"    PT-CD iter {iter_count}: best_J={best_J:.6f} current={current_J:.6f}")
            last_print = time.monotonic()

    return best_a, c, S, best_J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer: structured initialization, gradient descent with
    annealed smoothing, coordinate descent with exact 1D line search, and
    peak-targeted refinement.
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"n = {n}")

    # Initial score
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score: {best_global_score:.6f}")

    # Build diverse initializations
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("ones", np.ones(n)))
    inits.append(("cos1", make_cosine_mod(n, 3, 0.5)))
    inits.append(("cos2", make_cosine_mod(n, 7, 0.7, phase=0.5)))
    inits.append(("cos3", make_cosine_mod(n, 13, 0.6, phase=1.0)))
    inits.append(("multi1", make_multi_cosine(n, [3, 7], 0.4)))
    inits.append(("multi2", make_multi_cosine(n, [5, 11], 0.5)))
    inits.append(("asym", make_asymmetric(n)))
    inits.append(("dcos", make_double_cos(n, 3, 7, 0.5, 0.3)))
    inits.append(("sine1", make_sine_mod(n, 5, 0.5, phase=0.3)))
    inits.append(("rand1", np.exp(rng.normal(0, 0.4, n))))
    inits.append(("rand2", np.exp(rng.normal(0, 0.8, n))))

    # Score and sort initializations
    for idx, (name, arr) in enumerate(inits):
        arr = np.maximum(arr, 0.0)
        S = np.sum(arr)
        if S > 0:
            arr = arr * (n / S)
        J = compute_J(arr)
        inits[idx] = (name, J, arr)

    inits.sort(key=lambda x: x[1])
    print("Initializations sorted:")
    for name, J, _ in inits[:6]:
        print(f"  {name}: J = {J:.6f}")

    # Phase 1: Short GD on top 4 initializations (~200s)
    phase1_budget = min(200.0, (total_deadline - time.monotonic()) * 0.12)
    print(f"\nPhase 1: Short GD (budget={phase1_budget:.0f}s)")
    phase1_results = []
    for i, (name, init_J, init_arr) in enumerate(inits[:4]):
        if time.monotonic() > total_deadline - 300:
            break
        b = phase1_budget / 4.0
        b = min(b, total_deadline - time.monotonic() - 250)
        if b < 5:
            break
        print(f"  GD on '{name}' (budget={b:.1f}s)")
        res_a, res_J = run_gradient_descent(init_arr, b, seed=seed + i * 100,
                                             method='lm', p_start=2.0, p_end=30.0,
                                             lr_start=0.05)
        phase1_results.append((name, res_a, res_J))
        if res_J < best_global_score:
            best_global = res_a.copy()
            best_global_score = res_J
            print(f"    *** NEW BEST: {res_J:.6f} ***")

    # Phase 2: Extended GD on best Phase 1 result (~500s)
    phase2_budget = min(500.0, (total_deadline - time.monotonic()) * 0.30)
    print(f"\nPhase 2: Extended GD (budget={phase2_budget:.0f}s)")
    if phase1_results:
        best_p1_name, best_p1_a, best_p1_J = min(phase1_results, key=lambda x: x[2])
    else:
        best_p1_a, best_p1_J = inits[0][2], inits[0][1]
        best_p1_name = inits[0][0]

    # Run two GD variants: one with log-sum-exp, one with power-mean
    gd_budget = phase2_budget / 2.0
    gd_budget = min(gd_budget, total_deadline - time.monotonic() - 600)
    if gd_budget > 10:
        print(f"  LSE GD (budget={gd_budget:.1f}s)")
        res_a, res_J = run_gradient_descent(best_p1_a, gd_budget, seed=seed + 500,
                                             method='lse', tau_start=5.0, tau_end=0.05,
                                             lr_start=0.03)
        if res_J < best_global_score:
            best_global = res_a.copy()
            best_global_score = res_J
            print(f"    *** NEW BEST: {res_J:.6f} ***")

    gd_budget2 = min(phase2_budget / 2.0, total_deadline - time.monotonic() - 500)
    if gd_budget2 > 10:
        print(f"  PowerMean GD (budget={gd_budget2:.1f}s, p: 4->80)")
        res_a, res_J = run_gradient_descent(best_p1_a, gd_budget2, seed=seed + 600,
                                             method='lm', p_start=4.0, p_end=80.0,
                                             lr_start=0.03)
        if res_J < best_global_score:
            best_global = res_a.copy()
            best_global_score = res_J
            print(f"    *** NEW BEST: {res_J:.6f} ***")

    # Phase 3: Coordinate descent (~700s)
    phase3_budget = min(700.0, (total_deadline - time.monotonic()) * 0.45)
    print(f"\nPhase 3: Coordinate Descent (budget={phase3_budget:.0f}s)")
    cd_budget = min(phase3_budget, total_deadline - time.monotonic() - 200)
    if cd_budget > 30:
        res_a, res_J = coordinate_descent(best_global, cd_budget, seed=seed + 700)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"    *** NEW BEST after CD: {val:.6f} ***")
        else:
            print(f"    CD result: local={res_J:.6f}, verified={val:.6f}")

    # Phase 4: Peak-targeted refinement (~250s)
    phase4_budget = min(250.0, (total_deadline - time.monotonic()) * 0.25)
    print(f"\nPhase 4: Peak-Targeted CD (budget={phase4_budget:.0f}s)")
    pt_budget = min(phase4_budget, total_deadline - time.monotonic() - 60)
    if pt_budget > 30:
        c_cur = convolve_fft(best_global)
        S_cur = np.sum(best_global)
        res_a, res_c, res_S, res_J = peak_targeted_cd(best_global, c_cur, S_cur,
                                                        pt_budget, seed=seed + 800)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"    *** NEW BEST after PT-CD: {val:.6f} ***")
        else:
            print(f"    PT-CD result: local={res_J:.6f}, verified={val:.6f}")

    # Phase 5: Final coordinate descent polish
    remaining = total_deadline - time.monotonic() - 10.0
    print(f"\nPhase 5: Final CD polish (budget={remaining:.0f}s)")
    if remaining > 30:
        res_a, res_J = coordinate_descent(best_global, remaining, seed=seed + 900)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"    *** FINAL BEST: {val:.6f} ***")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\nFinal score: {best_global_score:.6f}")
    # Ensure valid output
    best_global = np.clip(best_global, 0.0, 1000.0)
    return [float(x) for x in best_global]
