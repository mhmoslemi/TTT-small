"""
Hybrid optimization combining improved LP-based peak flattening,
softmax-gradient multiplicative updates, and coordinate-wise local search
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def compute_conv(a):
    """Compute full auto-convolution of a using FFT."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft *= 2
    A = fft(a, N_fft)
    C = ifft(A * A).real[:N]
    # Clean up tiny negative values from numerical error
    C = np.maximum(C, 0.0)
    return C


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    C = compute_conv(a)
    return 2.0 * n * C.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M, alpha=1.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= alpha*M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, alpha * M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:

def coordinate_descent(a, deadline, rng):

        it += 1

                break

                if sc < best_score - 1e-12:
                    best_score = sc
                    old_val = new_val
                    improved = True
                else:
                    best_a[i] = old_val

            # Try adding mass at random positions with low current value
            for _ in range(10):

                i = rng.integers(0, n)

                best_score = compute_score(best_a)

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)


def make_welch(n):
    """Sequoia-inspired: try to equalize convolution values."""
    # Start with a "spread" pattern
    x = np.linspace(0, 1, n)
    return x * (1 - x) * 4  # parabolic

    """Dual parabolic: high at edges, low in middle."""
    c = (n - 1) / 2.0
    t = np.abs(np.arange(n) - c) / (c + 1.0)
    return t  # 0 at center, 1 at edges

"""
Hybrid optimization combining LP-based peak flattening, softmax-gradient
multiplicative updates, and coordinate-wise local search to minimize
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-accelerated convolution,
aggressive alpha parameters for the LP, and adaptive time allocation.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def compute_conv(a):
    """Compute full auto-convolution of a using FFT."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft *= 2
    A = fft(a, N_fft)
    C = ifft(A * A).real[:N]
    C = np.maximum(C, 0.0)
    return C


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    C = compute_conv(a)
    return 2.0 * n * C.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M, alpha):
    """Solve LP: max sum(g) s.t. conv(a,g) <= alpha*M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, alpha * M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

    kernel = np.ones(5) / 5.0
    smoothed = np.convolve(raw, kernel, mode='same')
    return smoothed + 0.05


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using hybrid LP + gradient + CD.
    Multi-phase: broad search over (n, init, method) combos, then polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 12.0

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all() and p.sum() > 0.01:
            parent = p.copy()
    except Exception:
        pass

    # Build search configurations
    search_configs = []

    if parent is not None:
        n_p = len(parent)
        search_configs.append((n_p, 'parent', 'lp'))
        search_configs.append((n_p, 'parent', 'sg'))

    sizes = [384, 512, 640, 768, 1024]
    init_names = ['cosine', 'gaussian', 'parabolic', 'dual_parabolic', 'sinc_like']

    for n in sizes:
        for name in init_names:
            method = 'lp' if n <= 1024 else 'sg'
            search_configs.append((n, name, method))

    search_configs.append((512, 'random', 'sg'))
    search_configs.append((384, 'random', 'sg'))
    search_configs.append((768, 'random', 'sg'))

    # Deduplicate
    seen = set()
    unique_configs = []
    for cfg in search_configs:
        if cfg not in seen:
            seen.add(cfg)
            unique_configs.append(cfg)
    search_configs = unique_configs

    # Time allocation
    n_configs = len(search_configs)
    weights = []
    for n, init_name, method in search_configs:
        w = 1.0
        if parent is not None and n == parent.size:
            w = 3.0
        elif n == 512:
            w = 2.2
        elif n in [384, 640]:
            w = 1.8
        elif n in [768, 1024]:
            w = 1.4
        if method == 'lp':
            w *= 1.4
        weights.append(w)

    total_weight = sum(weights)
    available_time = budget_s - 50.0
    time_allocs = []
    cum = 0.0
    for w in weights:
        cum += available_time * w / total_weight
        time_allocs.append(cum)

    print(f"Hybrid search: {n_configs} configs, budget={budget_s}s")
    if parent is not None:
        print(f"Parent: n={parent.size}, score={compute_score(parent):.6f}")

    for idx in range(n_configs):
        if time.monotonic() > deadline:
            break

        n, init_name, method = search_configs[idx]
        config_deadline = min(start_time + time_allocs[idx], deadline)

        if config_deadline <= time.monotonic() + 3:
            continue

        if init_name == 'parent' and parent is not None:
            a = parent[:n].copy()
        elif init_name == 'cosine':
            a = init_cosine(n)
        elif init_name == 'gaussian':
            a = init_gaussian(n)
        elif init_name == 'parabolic':
            a = init_parabolic(n)
        elif init_name == 'dual_parabolic':
            a = init_dual_parabolic(n)
        elif init_name == 'sinc_like':
            a = init_sinc_like(n)
        elif init_name == 'random':
            a = init_random_smooth(n, rng)
        else:
            a = init_cosine(n)

        a = np.maximum(a, 1e-15).astype(np.float64)
        if a.sum() < 1e-10:
            a = np.ones(n)
        a = a / a.sum()

        remaining = deadline - time.monotonic()
        print(f"[{idx+1}/{n_configs}] n={n}, init={init_name}, method={method}, ~{remaining:.0f}s left")

        if method == 'lp' and n <= 1500:
            a_opt, score = optimize_lp_core(a, config_deadline, rng)
        else:
            a_opt, score = optimize_softmax_grad(a, config_deadline, rng)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: {best_score:.6f} (n={n}, {init_name}/{method})")
        else:
            print(f"  => score={score:.6f} (best={best_score:.6f})")

    # === POLISH PHASE ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        polish_time = deadline - time.monotonic()
        print(f"\n=== POLISH PHASE ({polish_time:.0f}s available) ===")
        print(f"Polish start: score={best_score:.6f}, n={len(best_seq)}")

        if len(best_seq) <= 1500:
            t1 = time.monotonic() + polish_time * 0.45
            best_seq, best_score = optimize_lp_core(best_seq, min(t1, deadline - 30), rng)
            print(f"  After LP polish: {best_score:.6f}")

        if time.monotonic() < deadline - 25:
            t2 = time.monotonic() + (deadline - time.monotonic()) * 0.5
            best_seq, best_score = optimize_softmax_grad(best_seq, min(t2, deadline - 15), rng)
            print(f"  After SG polish: {best_score:.6f}")

        if time.monotonic() < deadline - 10:
            best_seq, best_score = coordinate_descent(best_seq, deadline, rng)
            print(f"  After CD polish: {best_score:.6f}")

    if best_seq is None:
        best_seq = init_cosine(256)
        best_score = compute_score(best_seq)

    final_score = compute_score(best_seq)
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_seq)}, time={time.monotonic()-start_time:.1f}s")

    return [float(x) for x in best_seq]
