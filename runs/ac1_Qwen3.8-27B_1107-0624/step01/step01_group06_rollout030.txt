import numpy as np
import time

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_J_fast(a):
    """Compute J = 2n*max(conv)/S^2 efficiently."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    conv = convolve_fft(a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv / (S * S)

def coordinate_descent(a, budget_s, seed=42):

    Coordinate descent to minimize max(conv(a,a)) / sum(a)^2.
    Since J is scale-invariant, we fix sum(a) = n and minimize max(conv).

    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n)
    else:
        a = a * n / S
    
    conv = convolve_fft(a)

For each coordinate i, the optimal delta (to minimize the new max) can be found by:
- The new max is max(conv[k] + 2*delta*a[k-i] + delta^2*[k=2i]) for k in [0, 2n-2]
- This is a convex function of delta (max of linear/quadratic functions)
- We can find the optimal delta by bisection or by checking the kink points

For each coordinate i:
1. Compute the "sensitivity" s[i] = how much conv[k*] changes per unit change in a[i], where k* is the argmax
2. Propose a step in the direction that reduces the max
3. Use line search (backtracking) to determine step size

1. Identify the active set: the set of lags k where conv[k] is within epsilon of the max
2. For each coordinate i, compute the gradient of max(conv) (or the average of gradients of the active lags)
3. Take a step in the negative gradient direction
4. Project onto a >= 0

This is essentially what the smoothed gradient descent does, but using the exact max instead of log-sum-exp.

"""
Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
Uses smoothed gradient descent (log-sum-exp) with adaptive tau, exact
subgradient method, and coordinate descent polishing. Exploits FFT-based
fast convolution and the scale invariance of the objective.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(conv)/S^2 using FFT."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    conv = convolve_fft(a)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv / (S * S)


def smoothed_value_and_grad(a, tau):
    """Compute smoothed J and gradient using log-sum-exp over convolution."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(conv)
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad


def exact_subgradient(a):
    """Compute exact subgradient of J using active set of max convolution lags."""
    n = len(a)
    conv = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    max_conv = np.max(conv)
    J = 2.0 * n * max_conv / (S * S)

    # Find active set: lags within 1% of max
    threshold = max_conv * 0.99
    active = np.where(conv >= threshold)[0]

    # Compute subgradient as weighted average over active lags
    weights = (conv[active] - threshold) / (max_conv - threshold + 1e-15)
    weights = weights / np.sum(weights)

    grad = np.zeros(n)
    for w, k in zip(weights, active):
        # grad of conv[k] w.r.t. a[i]: 2*a[k-i] for valid i, with factor 1 if k=2i
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            val = a[k - i]
            factor = 1.0 if k == 2 * i else 2.0
            grad[i] += w * factor * val

    J_grad = (2.0 * n * grad) / (S * S) - (4.0 * n * max_conv) / (S * S * S)
    return J, J_grad, max_conv


def run_smoothed_gd(a_init, budget_s, seed, tau_start, tau_end, lr_start):
    """Run Adam gradient descent with log-sum-exp smoothing."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J_fft(a)

    start_time = time.monotonic()
    stall_count = 0
    stall_threshold = best_J

    while time.monotonic() < deadline - 2.0:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start / (1.0 + 0.3 * t / 500.0)

        J_smooth, grad = smoothed_value_and_grad(a, tau)

        if not np.isfinite(J_smooth) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)

        if t % 5000 == 0:
            local_J = compute_J_fft(a)

                stall_count = 0
                stall_threshold = local_J
            else:
                stall_count += 1

                best_J = local_J
                best_a = a.copy()

            if stall_count > 3:
                # Perturbation: multiplicative noise on a random block
                w = max(10, n // 50)
                s = int(rng.integers(0, n - w))
                cand = a.copy()
                cand[s:s+w] *= np.exp(rng.normal(0, 0.15, w))
                cand = np.maximum(cand, 0.0)
                cJ = compute_J_fft(cand)

                    a = cand
                else:
                    a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall_count = 0

        # Verify with true objective every ~10s

def run_coordinate_descent(a_init, budget_s, seed):

    conv_new[k] = conv[k] + 2*delta*a[k-i] + delta^2*[k=2i]

    best_max = max_conv

    pass_num = 0
    no_improve_passes = 0

    while time.monotonic() < deadline - 5.0:
        pass_num += 1
        pass_improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline - 5.0:
                break

            a_i = a[i]
            if a_i < 1e-10:
                continue

            for frac in step_fracs:
                delta = frac * a_i
                if a_i + delta < 0:
                    continue

                # Compute new max using fast update
                # conv_new[k] = conv[k] + 2*delta*a[k-i] for k in [max(0,i), min(2n-2, i+n-1)]
                k_start = i
                k_end = min(2 * n - 2, i + n - 1)
                idx_start = k_start - i
                idx_end = k_end - i

                # Get the range of conv we need to update
                conv_range = conv[k_start:k_end + 1]
                a_range = a[idx_start:idx_end + 1]

                new_range = conv_range + 2.0 * delta * a_range

                # Add delta^2 at position k=2i if in range
                if k_start <= 2 * i <= k_end:
                    new_range[2 * i - k_start] += delta * delta

                new_max_range = float(np.max(new_range))
                # The rest of conv is unchanged, so overall max is:
                overall_max = max(new_max_range, float(np.max(conv[:k_start])) if k_start > 0 else -np.inf,
                                  float(np.max(conv[k_end+1:])) if k_end < 2*n-2 else -np.inf)

                if overall_max < max_conv - 1e-12:
                    # Accept this step
                    a[i] = a_i + delta
                    conv[k_start:k_end + 1] = new_range
                    max_conv = overall_max
                    pass_improved = True
                    break

        if max_conv < best_max:
            best_max = max_conv
            best_a = a.copy()
            no_improve_passes = 0
        else:
            no_improve_passes += 1

        if pass_num % 20 == 0 or (no_improve_passes > 30 and pass_num % 5 == 0):

            print(f"  CD pass {pass_num}: max_conv={best_max:.4f}, J_approx={J:.6f}")

        if no_improve_passes > 100:
            # Try random restart within neighborhood
            cand = best_a.copy()
            n_pert = max(20, n // 50)
            idx = rng.integers(0, n, size=n_pert)
            cand[idx] *= np.exp(rng.normal(0, 0.05, n_pert))
            cand = np.maximum(cand, 0.0)
            S_c = np.sum(cand)
            if S_c > 0.01:
                cand = cand * n / S_c
                conv_c = convolve_fft(cand)
                max_c = float(np.max(conv_c))
                if max_c < best_max:
                    a = cand.copy()
                    conv = conv_c
                    max_conv = max_c
                    best_a = cand.copy()
                    best_max = max_c
                    no_improve_passes = 0
                else:
                    a = best_a.copy()
                    conv = convolve_fft(a)
                    max_conv = best_max
            no_improve_passes = 0

    # Re-normalize
    best_a = np.maximum(best_a, 0.0)
    S = np.sum(best_a)
    if S > 0:

    return best_a, 2.0 * best_max / n

        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  Phase 5 result: {val:.6f}")
        else:
            print(f"  Phase 5: local={res_J:.6f}, verified={val:.6f}")

    # Phase 6: Final smooth GD refinement with very small tau
    elapsed = time.monotonic() - t0
    remaining = float(budget_s) - 5.0 - elapsed
    budget6 = min(remaining * 0.5, 150.0)
    if budget6 > 15:
        print(f"\nPhase 6: Final refinement tau=[0.1,0.01], budget={budget6:.0f}s")
        res_a, res_J = run_smoothed_gd(best_global, budget6, seed=seed + 5000,
                                        tau_start=0.1, tau_end=0.01, lr_start=0.005)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  Phase 6 result: {val:.6f}")
        else:
            print(f"  Phase 6: local={res_J:.6f}, verified={val:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    # Also try a few different normalizations/nearby points
    elapsed = time.monotonic() - t0
    remaining = float(budget_s) - 5.0 - elapsed
    if remaining > 10:
        # Try scaling the best
        for scale_factor in [0.95, 0.97, 1.0, 1.02, 1.05]:
            cand = best_global * scale_factor
            cand = np.clip(cand, 0, 1000).tolist()
            v = float(evaluate_sequence(cand))
            if np.isfinite(v) and v < best_global_score:
                best_global = np.array(cand)
                best_global_score = v
                print(f"  Scale {scale_factor}: {v:.6f}")

        # Try a few random perturbations of the best
        for _ in range(5):
            cand = best_global * np.exp(rng.normal(0, 0.005, n))
            cand = np.clip(cand, 0, 1000)
            v = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(v) and v < best_global_score:
                best_global = cand
                best_global_score = v
                print(f"  Perturb: {v:.6f}")

    print(f"\nFinal score: {best_global_score:.6f}")

    # Ensure valid output
    result = np.clip(best_global, 0.0, 1000.0)
    return [float(x) for x in result]
