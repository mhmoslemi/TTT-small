"""
Hybrid optimizer for minimizing 2*n*max(p*p)/(sum(p))^2 over non-negative sequences.
Uses FFT-based convolution, smooth-max L-BFGS, projected gradient on top-K peaks,
and simulated annealing for global exploration.
"""

import numpy as np
import time
import sys

try:
    from scipy.optimize import minimize
    HAS_SCIPY = True
except ImportError:
    HAS_SCIPY = False

# Global state
_best_score = float('inf')
_best_seq = None
_search_deadline = 0.0


class _Timeout(Exception):
    pass


def _fft_size(n):
    m = 2*n - 1
    return 1 << (m - 1).bit_length() if m > 1 else 2


def _self_conv(p):
    """Self-convolution of probability vector p using FFT."""
    n = len(p)
    fs = _fft_size(n)
    F = np.fft.rfft(p, fs)
    c = np.fft.irfft(F * F, fs).real[:2*n-1]
    return np.maximum(c, 0.0)


def _score(p):
    """Objective for probability vector (sum=1)."""
    n = len(p)
    c = _self_conv(p)
    return 2.0 * float(n) * float(np.max(c))


def _update_best(p, score):
    global _best_score, _best_seq
    if np.isfinite(score) and score < _best_score:
        _best_score = score
        _best_seq = p.copy()


def _grad_topk(p, c, k=16):
    """Gradient of weighted top-k peaks objective."""
    n = len(p)
    cmax = np.max(c)
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(len(c))
    vals = c[idx]
    # Soft weighting: peaks closer to max get more weight
    wv = vals - np.min(c) + 1e-12
    w[idx] = wv / np.sum(wv)
    
    # Gradient: 2 * cross-correlation of p with w
    fs = 1 << (3*n - 2).bit_length()
    Fp = np.fft.rfft(p[::-1], fs)
    Fw = np.fft.rfft(w, fs)
    conv = np.fft.irfft(Fp * Fw, fs).real
    grad = 2.0 * conv[n-1:2*n-1]
    return grad, cmax, w


def _smooth_max_obj_grad(p, alpha):
    """Smooth max objective and its gradient using log-sum-exp."""
    c = _self_conv(p)
    cmax = np.max(c)
    u = alpha * (c - cmax)
    e = np.exp(np.clip(u, -50, 50))
    s = np.sum(e)
    if s < 1e-300:
        J = cmax
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s
    
    n = len(p)
    fs = 1 << (3*n - 2).bit_length()
    Fp = np.fft.rfft(p[::-1], fs)
    Fw = np.fft.rfft(w, fs)
    conv = np.fft.irfft(Fp * Fw, fs).real
    gp = 2.0 * conv[n-1:2*n-1]
    return J, gp


def _project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _logit_forward(z, n):
    """Convert logits to probability vector."""
    z = np.clip(z, -50, 50)
    m = np.max(z)
    expz = np.exp(z - m)
    Z = np.sum(expz)
    return expz / Z


def _logit_backward_grad(p, grad_p):
    """Convert gradient w.r.t. p to gradient w.r.t. logits z."""
    # p_j = exp(z_j) / Z = softmax(z)_j
    # dz_j = p_j * (grad_p_j - sum_i p_i * grad_p_i)
    sdot = np.dot(grad_p, p)
    return p * (grad_p - sdot)


def _optimize_lbfgs(z0, n, alpha, max_iter, deadline):
    """L-BFGS-B optimization in logit space."""
    global _best_score, _best_seq, _search_deadline
    _search_deadline = deadline
    
    def obj_grad(z):
        if time.monotonic() >= _search_deadline:
            raise _Timeout
        z = np.clip(z, -50, 50)
        m = np.max(z)
        expz = np.exp(z - m)
        p = expz / np.sum(expz)
        
        c = _self_conv(p)
        hard = 2.0 * n * np.max(c)
        _update_best(p, hard)
        
        cmax = np.max(c)
        u = alpha * (c - cmax)
        e = np.exp(np.clip(u, -50, 50))
        s = np.sum(e)
        if s < 1e-300:
            J = cmax
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            J = cmax + np.log(s) / alpha
            w = e / s
        
        fs = 1 << (3*n - 2).bit_length()
        Fp = np.fft.rfft(p[::-1], fs)
        Fw = np.fft.rfft(w, fs)
        conv = np.fft.irfft(Fp * Fw, fs).real
        gp = 2.0 * conv[n-1:2*n-1]
        
        sdot = np.dot(gp, p)
        gz = p * (gp - sdot)
        gz = np.where(np.isfinite(gz), gz, 0.0)
        return float(2.0 * n * J), gz
    
    try:
        result = minimize(
            obj_grad, z0,
            method='L-BFGS-B',
            jac=True,
            bounds=[(-50, 50)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8, 'maxls': 50}
        )
        z_final = np.clip(result.x, -50, 50)
        p_final = _logit_forward(z_final, n)
        s = _score(p_final)
        _update_best(p_final, s)
    except _Timeout:
        pass
    except Exception:
        pass


def _projected_gradient(p0, n, deadline, topk=16, lr_init=0.01):
    """Projected gradient descent on the simplex."""
    global _best_score, _best_seq
    p = p0.copy()
    score = _score(p)
    _update_best(p, score)
    lr = lr_init
    fails = 0
    
    while time.monotonic() < deadline:
        c = _self_conv(p)
        grad, cmax, w = _grad_topk(p, c, topk)
        
        # Project gradient direction onto tangent space of simplex
        # (subtract mean to ensure sum of step is 0)
        grad_centered = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        accepted = False
        for _ in range(10):
            p_new = _project_simplex(p - lr * grad_centered)
            s_new = _score(p_new)
            if s_new < score - 1e-10:
                p = p_new
                score = s_new
                _update_best(p, score)
                accepted = True
                lr = min(lr * 1.5, 0.1)
                fails = 0
                break
            lr *= 0.5
        
        if not accepted:
            fails += 1
            if fails > 50:
                break
    
    return p, score


def _simulated_annealing(p0, n, deadline, n_iter_per_sec=50):
    """Simulated annealing on the simplex."""
    global _best_score, _best_seq
    p = p0.copy()
    score = _score(p)
    _update_best(p, score)
    
    T_init = 0.01
    T_min = 1e-6
    total_iter = int(n_iter_per_sec * max(1, deadline - time.monotonic()))
    
    for it in range(total_iter):
        if time.monotonic() >= deadline:
            break
        
        T = T_init * (T_min / T_init) ** (it / max(1, total_iter))
        
        # Pick two random indices and transfer mass
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        delta = np.random.uniform(-0.01, 0.01)
        if delta > p[i]:
            delta = p[i] * 0.5
        elif -delta > p[j]:
            delta = -p[j] * 0.5
        
        p_new = p.copy()
        p_new[i] -= delta
        p_new[j] += delta
        
        if p_new[i] < 0 or p_new[j] < 0:
            continue
        
        s_new = _score(p_new)
        delta_s = s_new - score
        if delta_s < 0 or np.random.random() < np.exp(-delta_s / max(T, 1e-10)):
            p = p_new
            score = s_new
            _update_best(p, score)


def _active_set_refine(p0, n, deadline):
    """Iteratively reduce active peaks using LP-like approach."""
    global _best_score, _best_seq
    p = p0.copy()
    score = _score(p)
    _update_best(p, score)
    
    while time.monotonic() < deadline:
        c = _self_conv(p)
        cmax = np.max(c)
        
        # Find active peaks (within 1% of max)
        threshold = cmax * 0.99
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Compute gradient for the active peaks
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        
        n_local = len(p)
        fs = 1 << (3*n_local - 2).bit_length()
        Fp = np.fft.rfft(p[::-1], fs)
        Fw = np.fft.rfft(w, fs)
        conv = np.fft.irfft(Fp * Fw, fs).real
        gp = 2.0 * conv[n_local-1:2*n_local-1]
        
        # Direction: reduce where gradient is high, increase where low
        # This is a "water filling" direction
        gp_centered = gp - np.mean(gp)
        direction = -gp_centered  # decrease high-convolution-contributing elements
        
        # Project and take step
        for step in [0.05, 0.02, 0.01, 0.005, 0.001]:
            p_new = _project_simplex(p + step * direction)
            s_new = _score(p_new)
            if s_new < score - 1e-10:
                p = p_new
                score = s_new
                _update_best(p, score)
                break
    
    return p, score


def _seq_from_global(name, n_target):
    """Get a sequence from a global variable, resampled to n_target if needed."""
    try:
        seq = eval(name)
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        seq = np.maximum(seq, 0.0)
        s = np.sum(seq)
        if s < 1e-10:
            return None
        seq = seq / s  # normalize to probability vector
        if len(seq) != n_target:
            x_old = np.linspace(0, 1, len(seq))
            x_new = np.linspace(0, 1, n_target)
            seq = np.interp(x_new, x_old, seq)
            seq = np.maximum(seq, 0)
            seq = seq / np.sum(seq)
        return seq
    except Exception:
        return None


def _structured_candidates(n):
    """Generate structured candidate probability vectors."""
    cands = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    cands.append(np.ones(n) / n)
    
    # Logistic shapes
    for beta in [1, 2, 3, 5, 8, 12, 20]:
        f = 1.0 / (1.0 + np.exp(-beta * (x - 0.5)))
        cands.append(f / np.sum(f))
        f2 = 1.0 / (1.0 + np.exp(beta * (x - 0.5)))
        cands.append(f2 / np.sum(f2))
    
    # Power shapes
    for a in [0.5, 1, 2, 3, 5]:
        f = x**a + 1e-10
        cands.append(f / np.sum(f))
        f2 = (1-x)**a + 1e-10
        cands.append(f2 / np.sum(f2))
    
    # Bump / U-shape
    for w in [0.1, 0.2, 0.3, 0.5]:
        f = 1.0 + (1.0/w) * (x - 0.5)**2
        cands.append(f / np.sum(f))
    
    # Sinusoidal
    for freq in [1, 2, 3, 5]:
        f = 1.0 + 0.5 * np.sin(2*np.pi*freq*x)
        f = np.maximum(f, 0.01)
        cands.append(f / np.sum(f))
    
    return cands
