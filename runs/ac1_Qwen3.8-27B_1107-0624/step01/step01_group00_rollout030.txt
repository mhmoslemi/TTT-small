"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over a >= 0.
Uses mirror descent (multiplicative weights), Nesterov gradient descent,
structured initializations (exponential ramps, modulated uniform), and
multi-length exploration with periodic re-heating to escape local minima.
"""

import time
import math
import numpy as np


def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if target <= 0:
        return np.zeros(n)
    if v.min() >= 0 and abs(v.sum() - target) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _to_sum1(a):
    s = a.sum()
    if s <= 1e-15:
        return np.full(a.size, 1.0 / a.size)
    return a / s


def _conv_sq(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _max_conv(a):
    return _conv_sq(a).max()


def _gradient_from_weights(a, w_full):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. a.
    grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j.
    Uses FFT: grad = 2 * convolve(a, w_flipped, 'full')[2n-2 : n-2 : -1]
    """
    n = a.size
    w_flip = w_full[::-1]
    m = 1
    while m < 3 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    fw = np.fft.rfft(w_flip, m)
    full_conv = np.fft.irfft(fa * fw, m)
    # Extract: indices 2n-2, 2n-3, ..., n-1
    start = 2 * n - 2
    grad = 2.0 * full_conv[start : start - n : -1]
    return grad


def _mirror_descent(a, t_end, eta0=0.3, L=80, reheat_every=2000, reheat_scale=0.1,
                    rng=None):
    """Mirror descent (multiplicative weights) for min max(c)."""
    if rng is None:
        rng = np.random.default_rng()
    n = a.size
    a = _to_sum1(np.maximum(a, 0.0))
    best_a = a.copy()
    best_mc = _max_conv(a)
    eta = eta0
    L_eff = L
    step = 0
    patience = 0

    while time.monotonic() < t_end:
        c = _conv_sq(a)
        mc = c.max()

            best_mc = mc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        Lcur = min(L_eff, 2 * n - 1)
        if patience > 500:
            Lcur = min(Lcur + 20, 2 * n - 1)
        order = np.argpartition(-c, Lcur - 1)[:Lcur]

        # Softmax weights over active lags
        active_c = c[order]
        max_ac = active_c.max()
        w = np.exp((active_c - max_ac) / 0.01)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.full(Lcur, 1.0 / Lcur)
        else:
            w = w / w_sum

        w_full[order] = w

        grad = _gradient_from_weights(a, w_full)
