def convolve_fft(a):
    """Fast self-convolution using FFT."""

    size = 1
    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

convolution objective J = 2n*max(a*a)/(sum(a))^2.

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_J_fft(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def smooth_J_and_grad(a, tau):
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    J = 2.0 * n * Tmax / (S * S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad

def lp_direction(a, n_active=200):
    """Find LP improvement direction."""
    n = len(a)
    c = convolve_fft(a)
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01 or M < 1e-10:
        return None
    
    active_idx = np.argsort(c)[::-1][:n_active]
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_idx):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    c_obj = -np.ones(n)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=np.full(len(active_idx), M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': 30})
    except Exception:
        return None
    
    if result.status != 0 or result.x is None:
        return None
    
    g = result.x
    g_sum = np.sum(g)
    if g_sum < 0.01:
        return None
    g = g * (S / g_sum)
    return g

"""
Multi-strategy optimizer combining active-set LP direction finding,
multi-scale Adam gradient descent with FFT-accelerated log-sum-exp smoothing,
and parametric initialization scanning to minimize J = 2n*max(a*a)/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * float(np.max(c)) / (S * S)


def smooth_J_and_grad(a, tau):
    """Compute smoothed objective and gradient via log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    J = 2.0 * n * Tmax / (S * S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad


def lp_direction(a, n_active=200):
    """Find LP improvement direction: max sum(g) s.t. (a*g)[k]<=M for active k, g>=0."""
    n = len(a)
    c = convolve_fft(a)
    M = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01 or M < 1e-10:
        return None

    active_idx = np.argsort(c)[::-1][:n_active]

    rows = []
    cols = []
    vals = []
    for idx in range(len(active_idx)):
        k = int(active_idx[idx])
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    if len(rows) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    ub = max(10.0 * S / n, 1.0)
    bounds = [(0.0, ub)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': 15, 'presolve': True})
    except Exception:
        return None

    if result.status != 0 or result.x is None:
        return None

    g = result.x
    g_sum = np.sum(g)
    if g_sum < 0.01:
        return None
    g = g * (S / g_sum)
    return g


def run_lp_loop(a, budget_s, seed):
    """Iterative LP-based improvement loop."""
    n = len(a)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    best_a = a.copy()
    best_J = compute_J_fft(a)
    stuck = 0
    iter_count = 0

    while time.monotonic() < deadline:
        n_act = int(rng.choice([100, 150, 200, 300, 400]))
        g = lp_direction(a, n_active=n_act)
        iter_count += 1

        if g is not None:
            t_val = float(rng.uniform(0.05, 0.25))
            a_new = np.maximum((1.0 - t_val) * a + t_val * g, 0.0)
            J_new = compute_J_fft(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                a = a_new.copy()
                stuck = 0
                if iter_count % 10 == 0:
                    print(f"    LP iter {iter_count}: J = {J_new:.6f}")
            else:
                stuck += 1
        else:
            stuck += 1

        if stuck > 40:
            a = best_a.copy()
            w = max(1, int(n // 25))
            s = int(rng.integers(0, max(1, n - w)))
            a[s:s + w] *= np.exp(rng.normal(0, 0.3, w))
            a = np.maximum(a, 0.0)
            stuck = 0

    return best_a, best_J


def run_multiscale_gd(a_init, budget_s, seed):
    """Multi-scale Adam gradient descent with adaptive tau cycling."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J_fft(a)

    start_time = time.monotonic()
    tau_schedule = [8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.03]
    tau_idx = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))

        # Progressive tau reduction
        max_idx = min(len(tau_schedule) - 1, int(frac * len(tau_schedule) * 1.5))
        tau = tau_schedule[min(tau_idx, max_idx)] * (1.0 + 0.3 * (1.0 - frac))
        tau = max(0.02, tau)
        lr = 0.012 * (1.0 - 0.4 * frac)

        J, grad = smooth_J_and_grad(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1

        if t % 30000 == 0:
            tau_idx = min(tau_idx + 1, len(tau_schedule) - 1)

        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)

        if t % 4000 == 0:

                a = np.maximum(a, 0.0)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0

    final_local = compute_J_fft(best_a)
    return best_a, min(best_J, final_local)


def make_hann(n):
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))


def make_gaussian(n, sigma_frac):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-x * x / (2.0 * sigma * sigma))


def make_flat_taper(n, frac):
    a = np.ones(n)
    taper_len = max(2, int(n * (1.0 - frac) / 2.0))
    if taper_len < n // 2:
        t = np.arange(taper_len)
        taper = 0.5 * (1.0 - np.cos(np.pi * t / (taper_len - 1)))
        a[:taper_len] = taper
        a[-taper_len:] = taper[::-1]
    return a


def make_cos_mod(n, freq, amp, sigma_frac):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    gauss = np.exp(-x * x / (2.0 * sigma * sigma))
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
    return gauss * np.maximum(mod, 0.01)


def make_super_gauss(n, sigma_frac, p):
    """Super-Gaussian: exp(-|x|^p / (2*sigma^p))."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-np.abs(x) ** p / (2.0 * sigma ** p))
