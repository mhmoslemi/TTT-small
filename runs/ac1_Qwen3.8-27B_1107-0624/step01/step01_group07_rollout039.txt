Enforces palindrome symmetry to halve the search space. Uses time-budgeted multi-n search.

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def make_symmetric(a):
    """Make a palindrome."""
    n = len(a)
    half = a[:n//2]
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, a[n//2:n//2+1], half[::-1]])

def get_half(a):
    """Get the first half of a palindrome."""
    n = len(a)
    return a[:n//2].copy()

def build_sym_conv_matrix_half(a):
    """
    Build matrix B (shape n x m) such that for a palindrome g = S @ y (m = n//2),
    the first n components of conv(a, g) are B @ y.
    Here a is already a palindrome of length n, m = n//2.
    """
    n = len(a)
    m = n // 2
    # S is the n x m matrix that maps y (half) to g (full palindrome)
    # S[j, i] = 1 if g[j] = y[i]
    S = np.zeros((n, m))
    for i in range(m):
        S[i, i] = 1.0
        S[n - 1 - i, i] = 1.0
    
    # Full conv matrix A: (2n-1) x n, A @ g = conv(a, g)
    # We only need the first n rows (due to symmetry)
    A_half = np.zeros((n, n))
    for j in range(n):
        # a[j] * g[k-j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # So A[k, j] = a[j] for k in [j, j+n-1]
        A_half[j:min(j+n, n), j] = a[j]
    
    # B = A_half @ S, shape (n, m)
    B = A_half @ S
    return B, S

def lp_solve_symmetric(a, M, method='highs'):
    """
    Solve: max sum(g) s.t. conv(a,g)[k] <= M for k=0,...,n-1, g >= 0, g is palindrome.
    Returns the full palindrome g, or None.
    """
    n = len(a)
    m = n // 2
    if m < 2:
        return None
    
    B, S = build_sym_conv_matrix_half(a)
    
    # Objective: maximize sum(g) = sum(S @ y) = (S' @ 1) @ y
    obj_coeff = S.T @ np.ones(n)  # shape (m,)
    c_obj = -obj_coeff  # minimize -sum
    
    # Constraints: B @ y <= M (n constraints), y >= 0 (m constraints, handled by bounds)
    b_ub = np.full(n, M)
    bounds = [(0.0, None)] * m
    
    try:
        res = linprog(c_obj, A_ub=B, b_ub=b_ub, bounds=bounds, method=method)
        if res.status == 0:
            y = res.x
            g = S @ y
            return g
    except Exception:
        pass
    return None

"""
Symmetric LP-based iterative optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Enforces palindrome symmetry to halve the search space, uses running-average smoothing
for the LP direction-finding step, and employs time-budgeted multi-n search with
final polish. Key improvements over prior approach: correct symmetric LP formulation,
smoothing via exponential moving average, and better time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def make_palindrome(half):
    """Build full even-length palindrome from its first half."""
    return np.concatenate([half, half[::-1]])


def symmetrize(a):
    """Symmetrize a vector by averaging with its reverse."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def build_B_matrix(a):
    """
    Build B (shape n x m) such that for palindrome g = S@y (m=n//2),
    the first n components of conv(a, g) equal B @ y.
    a must be a palindrome of even length n.
    B[k,i] = a[k-i] (if k>=i) + a[k-n+1+i] (if k+i>=n-1)
    """
    n = len(a)
    m = n // 2
    B = np.zeros((n, m))
    for i in range(m):
        # First term: a[k-i] for k = i, i+1, ..., n-1
        for k in range(i, n):
            B[k, i] += a[k - i]
        # Second term: a[k-n+1+i] for k = n-1-i, n-i, ..., n-1
        start_k = max(0, n - 1 - i)
        for k in range(start_k, n):
            idx = k - n + 1 + i
            if 0 <= idx < n:
                B[k, i] += a[idx]
    return B


def lp_direction(a, M):
    """
    Solve LP: max sum(y) s.t. B@y <= M, y >= 0.
    Returns full palindrome direction g, or None.
    """
    n = len(a)
    m = n // 2
    if m < 2:
        return None

    B = build_B_matrix(a)
    c_obj = -np.ones(m)
    b_ub = np.full(n, M)
    bounds = [(0.0, None)] * m

    try:
        res = linprog(c_obj, A_ub=B, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            y = np.maximum(res.x, 0.0)
            if y.sum() < 1e-15:
                return None
            g = make_palindrome(y)
            return g
    except Exception:
        pass
    return None


def project_simplex(v):
    """Project v onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()


def subgradient_direction(a):
    """
    Compute a (sub)gradient direction for reducing max(conv(a,a)).
    Returns a direction vector (not normalized, not projected).
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    # Active set: where c is at M (within tolerance)
    tol = 0.05 * M + 1e-15
    active = np.where(c >= M - tol)[0]
    # Subgradient: average of gradients at active peaks
    # grad_k[i] = 2*a[k-i] if 0<=k-i<n
    sg = np.zeros(n)
    for k in active:
        # grad_k[i] = 2*a[k-i] for valid i
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                sg[i] += 2.0 * a[j]
    sg /= len(active)
    return sg

    LP-based optimization with symmetry enforcement.
    a must be an even-length palindrome.

    m = n // 2
    a = symmetrize(np.clip(a, 0.0, None))
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)
    if verbose:
        print(f"    Initial: n={n}, score={best_score:.6f}")

    # EMA of a for LP input (smoothing)
    a_ema = a.copy()
    ema_beta = 0.3

    stall = 0
    it = 0
    step_mult = 1.0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')

        a_lp_input = np.clip(a_lp_input, 1e-15, None)
        a_lp_input = a_lp_input / a_lp_input.sum()

        g = lp_direction(a_lp_input, M)
        if g is None or g.sum() < 1e-15:
            # Fallback: subgradient direction
            if it > 5:
                sg = subgradient_direction(a)
                a_new = a - 0.01 * sg
                a_new = symmetrize(np.clip(a_new, 0.0, None))
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    sc = compute_objective(a_new)
                    if sc < best_score:
                        best_a = a_new
                        best_score = sc
                        a = a_new
                break
            break

        g = symmetrize(np.clip(g, 0.0, None))
        if g.sum() < 1e-15:
            break
        g = g / g.sum()

        # Line search
        improved = False
        best_t = 0.0
        best_sc = best_score
        for t in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_new = (1.0 - t) * best_a + t * g
            a_new = symmetrize(np.clip(a_new, 0.0, None))
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * best_a + best_t * g
            a_new = symmetrize(np.clip(a_new, 0.0, None))
            a_new = a_new / a_new.sum()
            best_a = a_new
            best_score = best_sc
            a = a_new
            stall = 0
            # Update EMA
            a_ema = (1.0 - ema_beta) * a_ema + ema_beta * a
        else:
            stall += 1
            if stall > 3:
                # Perturbation
                noise_scale = min(0.15, 0.05 + 0.005 * stall)
                noise = np.abs(rng.normal(0, noise_scale, m))
                half = 0.5 * best_a[:m]
                half = half * (1.0 + noise)
                half = np.clip(half, 0.0, None)
                if half.sum() > 1e-10:
                    a_new = make_palindrome(half / half.sum())
                    sc = compute_objective(a_new)
                    if sc < best_score:
                        best_a = a_new
                        best_score = sc
                        a = a_new
                        a_ema = a.copy()
                        stall = 0
                    else:
                        # Simulated annealing: accept with probability decreasing with stall
                        prob = max(0.01, 0.2 * np.exp(-stall / 20.0))
                        if rng.random() < prob:
                            a = a_new
                        else:
                            a = best_a
                else:
                    break

            if stall > 80:
                # Try a different perturbation strategy
                half = best_a[:m].copy()
                if rng.random() < 0.5:
                    # Multiply by random
                    half = half * np.abs(rng.normal(1, 0.3, m))
                else:
                    # Add random
                    half = half + np.abs(rng.normal(0, 0.1 * half.mean(), m))
                half = np.clip(half, 0.0, None)
                if half.sum() > 1e-10:
                    a_new = make_palindrome(half / half.sum())
                    sc = compute_objective(a_new)
                    if sc < best_score:
                        best_a = a_new
                        best_score = sc
                        a = a_new
                        a_ema = a.copy()
                        stall = 0
                if stall > 200:
                    break

        if verbose and it % 200 == 0:
            print(f"      iter {it}, score={best_score:.6f}, stall={stall}")

    return best_a, best_score


def make_cosine(n):
    """Raised cosine (Hann-like) palindrome window."""
    m = n // 2
    half = 0.5 * (1.0 + np.cos(np.linspace(0, np.pi / 2, m)))
    return make_palindrome(half)


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian palindrome window."""
    m = n // 2
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    idx = np.arange(m)
    half = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    return make_palindrome(half)


def make_u_shape(n, power=0.5):
    """U-shaped (high at edges) palindrome window."""
    m = n // 2
    t = np.linspace(1.0, 0.0, m)
    half = t ** power
    return make_palindrome(half)


def make_uniform(n):
    """Uniform palindrome window."""
    return np.ones(n)


def make_rand_init(n, rng):
    """Random non-negative palindrome."""
    half = np.abs(rng.normal(0, 1, n // 2))
    return make_palindrome(half)


def make_mixed_init(n, rng):
    """Mixture of different shapes."""
    choice = rng.integers(0, 4)
    if choice == 0:
        return make_cosine(n)
    elif choice == 1:
        return make_gaussian(n, sigma_frac=rng.uniform(0.15, 0.5))
    elif choice == 2:
        return make_u_shape(n, power=rng.uniform(0.3, 2.0))
    else:
        return make_rand_init(n, rng)


def resize_to_palindrome(a, target_n):
    """Resize sequence a to a palindrome of length target_n (even)."""
    target_n = 2 * (target_n // 2)
    m = target_n // 2
    if len(a) >= target_n:
        half = a[:m].copy()
    else:
        # Interpolate
        n_old = len(a)
        m_old = max(1, n_old // 2)
        old_half = a[:m_old].copy()
        if m_old == 1:
            new_half = np.full(m, old_half[0])
        else:
            x_old = np.linspace(0, 1, m_old)
            x_new = np.linspace(0, 1, m)
            new_half = np.interp(x_new, x_old, old_half)
        half = new_half
    half = np.clip(half, 0.0, None)
    if half.sum() < 1e-10:
        half = np.ones(m)
    return make_palindrome(half / half.sum())
