"""
FISTA-based optimization of step function coefficients to minimize
J(a) = 2n * max(convolve(a,a)) / (sum(a))^2 using:
- Log-sum-exp smoothing with multiplicative manifold updates
- Nesterov-accelerated proximal gradient (FISTA)
- Piecewise-linear exact subgradient descent for final refinement
- Multi-start with diverse initializations and random nudge escapes
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast aperiodic self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute exact J = 2n * max(conv) / sum^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    c = convolve_fft(a)
    return 2.0 * n * np.max(c) / (S * S)


def smooth_grad(a, tau):
    """
    Compute smoothed objective T_tau and its gradient with respect to a.
    T_tau = tau * log(sum_k exp(c_k/tau)) where c = a*a (aperiodic conv).
    Gradient: grad[m] = sum_k p_k * (a[k-m] + a[m-k])
    Also returns c (the convolution) for inspection.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    T_tau = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    # grad[m] = sum_k p[k] * a[k-m] + sum_k p[k] * a[m-k]
    # Term 1: sum_k p[k]*a[k-m] = (p * a)[m] (convolution of p and a)
    # Term 2: sum_k p[k]*a[m-k] = (p * a_rev)[m] where a_rev[j] = a[n-1-j]
    #         and we take the last n elements (reversed window of p*a_rev)

    fp = np.fft.rfft(p, size)
    conv_pa = np.fft.irfft(fp * fa, size)[:n]

    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, size)
    conv_pa_rev = np.fft.irfft(fp * fa_rev, size)
    # (p * a_rev)[n-1-m] gives term2[m]
    term2 = np.zeros(n)
    term2[:] = conv_pa_rev[n - 1 :: -1][:n]

    grad = conv_pa + term2
    return T_tau, grad, c


def project_simplex(z, target_sum):
    """Project z onto {a >= 0, sum(a) = target_sum} via Euclidean projection."""
    n = len(z)
    if target_sum <= 0:
        return np.zeros(n)
    # Binary search for lambda such that sum(max(z_i - lambda, 0)) = target_sum
    lo = np.min(z) - target_sum
    hi = np.max(z)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.maximum(z - mid, 0.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.maximum(z - lam, 0.0)


def run_fista(a_init, budget_s, seed, tau_start=None, tau_end=None, lr_schedule='inverse'):
    """
    FISTA on the multiplicative manifold (log a) with Euclidean projection.
    Minimizes smoothed f_tau(a) = 2n * T_tau(a) / S^2 where S = sum(a).
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    # Work in log-space: x = log(a + eps)
    eps = 1e-10
    a = np.maximum(a_init, eps).copy()
    x = np.log(a)
    x = x - np.mean(x)  # center for numerical stability

    best_a = a.copy()
    best_J = compute_J(a)
    t = 0

    if tau_start is None:
        tau_start = 0.5 / n
    if tau_end is None:
        tau_end = 0.01 / n

    # Adaptive step size
    alpha = 0.01
    momentum = 0.0
    best_alpha = alpha
    streak = 0

    deficit = 0.0  # for invariant tracking
    c = convolve_fft(a)
    c_max = np.max(c)
    S = np.sum(a)

    while time.monotonic() < deadline:
        t += 1
        elapsed_frac = min(1.0, (time.monotonic() - (deadline - budget_s)) / max(budget_s, 1e-6))

        # Anneal tau
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        # Compute smooth gradient at x (which corresponds to a = exp(x))
        a_curr = np.exp(x)
        T_tau, grad_a, c_curr = smooth_grad(a_curr, tau)

        if not np.isfinite(T_tau) or not np.isfinite(grad_a).all():
            x = np.log(np.maximum(best_a, eps)) - np.mean(np.log(np.maximum(best_a, eps)))
            continue

        # Chain rule: dT/da * da/dx = grad_a * a (element-wise)
        grad_x = grad_a * a_curr

        #.normalize gradient for stable steps
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 1e-12:
            grad_x = grad_x / gnorm

        # FISTA momentum update
        if t > 1:
            x_new = x + momentum * (x - x_prev)
        else:
            x_new = x.copy()

        # Take step
        x_step = x_new - alpha * grad_x

        # Project: ensure exp(x_step) has sum in a reasonable range
        # We want sum(exp(x_step)) to be close to S (maintain scale)
        # Use simplex projection on y = exp(x_step), then x_new = log(y)
        y = np.exp(x_step)
        S_target = np.sum(a_curr)  # maintain current scale
        y_proj = project_simplex(y, S_target)
        y_proj = np.maximum(y_proj, eps)
        x_candidate = np.log(y_proj)

        # Check if step improved the objective
        J_cand = 2.0 * n * T_tau / (S_target * S_target)  # smoothed proxy
        # Actually let's just use the smoothed value at the candidate
        T_cand = smooth_grad(y_proj, tau)[0]
        J_cand_abs = 2.0 * n * T_cand / (S_target * S_target)

        if J_cand_abs < (J_curr_smooth + 1e-12):
            x_prev = x
            x = x_candidate
            a = y_proj
            deficit = 0.0
            streak = 0
            # Maybe increase alpha
            if streak > 10:
                best_alpha = alpha
        else:
            streak += 1
            alpha = max(1e-6, alpha * 0.7)
            x = x_new  # undo the step, keep momentum point

        # Update momentum (FISTA)
        momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        # Standard FISTA: beta_t = (1 - t_{prev}) / t_curr where t_curr = (1+sqrt(1+4t^2))/2
        # Simplified: beta = (t-1)/(t+2) for t >= 1
        if t >= 2:
            momentum = (t - 1) / (t + 2)

        # Track smooth objective
        J_curr_smooth = 2.0 * n * T_tau / (np.sum(a) ** 2)

        # Periodically check exact objective
        if t % 200 == 0:
            J_exact = compute_J(a)
            if J_exact < best_J:
                best_J = J_exact
                best_a = a.copy()
            # Reset momentum if stuck
            if t % 2000 == 0 and best_J - J_exact > 0.05:
                pass  # keep going

        # Occasionally reset momentum to avoid cycling
        if t % 5000 == 0:
            momentum = 0.0
            alpha = best_alpha

    # Final exact check
    J_final = compute_J(a)
    if J_final < best_J:
        best_J = J_final
        best_a = a.copy()

    return best_a, best_J


def run_exact_subgrad(a_init, budget_s, seed):
    """
    Piecewise-linear exact subgradient descent on the simplex manifold.
    Minimizes f(a) = max_k (a*a)[k] subject to sum(a) = 1, a >= 0.
    Using the multiplicative update: a_new = a * exp(-alpha * P(grad))
    where P removes the mean (to maintain sum approximately).
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = np.maximum(a_init, 1e-15).copy()
    S = np.sum(a)
    a = a / S  # normalize to sum = 1

    best_a = a.copy()
    best_maxc = np.max(convolve_fft(a))
    t = 0

    while time.monotonic() < deadline:
        t += 1
        c = convolve_fft(a)
        c_max = np.max(c)

        if c_max < best_maxc:
            best_maxc = c_max
            best_a = a.copy()

        # Find active set (lags near the max)
        tol = max(1e-10, 1e-8 * c_max)
        active_mask = c >= c_max - tol
        active_idx = np.where(active_mask)[0]

        # Compute subgradient
        grad = np.zeros(n)
        # Use efficient computation
        # For each active lag k: grad[m] += a[k-m] + a[m-k]
        # Vectorized: for all active k at once
        # Term 1: sum_{k in active} a[k-m] = cross-correlation structure
        # Term 2: sum_{k in active} a[m-k]

        # Efficient: create indicator w[k] = 1 if k in active else 0
        w = np.zeros(2 * n - 1)
        w[active_idx] = 1.0

        # Term 1: (w * a)[m] for m = 0..n-1 (convolution of w and a, first n elements)
        # Term 2: (w * a_rev)[m] reversed appropriately

        size = 1
        while size < 3 * n:
            size <<= 1

        fw = np.fft.rfft(w, size)
        fa = np.fft.rfft(a, size)
        conv_wa = np.fft.irfft(fw * fa, size)[:n]  # (w*a)[m] = sum_k w[k]a[m-k] = term2

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size)
        conv_wa_rev = np.fft.irfft(fw * fa_rev, size)
        # (w * a_rev)[n-1-m] = sum_k w[k] * a_rev[n-1-m-k] = sum_k w[k] * a[m+k]
        # We want sum_k w[k] * a[k-m] = sum_k w[k] * a_rev[n-1-(k-m)] = sum_k w[k] * a_rev[n-1-m-k]
        # = (w * a_rev)[n-1-m]
        term1 = np.zeros(n)
        # (w * a_rev)[j] for j = n-1, n-2, ..., 0 (last n elements of conv, reversed)
        conv_full_len = min(size, 2 * n - 1 + n - 1)
        available = conv_wa_rev[:min(n, len(conv_wa_rev))]
        # Actually: (w*a_rev)[j] = sum_k w[k]*a_rev[j-k]
        # We need (w*a_rev)[n-1-m] for m=0,...,n-1, i.e., j = n-1, n-2, ..., 0
        term1 = conv_wa_rev[:n][::-1]

        grad = term1 + conv_wa

        # Normalize
        S = np.sum(a)
        # Gradient of f/S^2: grad/S^2 - 2*f/S^3 * 1
        # The constant part doesn't matter (removed by projection), so use grad/S^2
        grad_step = grad / (S * S)

        # Step size: adaptive
        alpha = 0.01 / np.sqrt(1.0 + t * 0.001)
        # Decay alpha over time
        alpha *= max(0.1, 1.0 - 0.3 * (t / max(1, int(budget_s * 50))))

        # Multiplicative update on the manifold
        # a_new = a * exp(-alpha * (grad_step - mean(grad_step)))
        centered = grad_step - np.mean(grad_step)
        log_factor = -alpha * centered
        # Clip to avoid overflow
        log_factor = np.clip(log_factor, -2.0, 2.0)
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 1e-15)

        # Project to simplex (sum = 1, a >= 0)
        a_new = project_simplex(a_new, 1.0)
        a_new = np.maximum(a_new, 1e-15)
        a = a_new

    J_final = compute_J(best_a)
    return best_a, J_final


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-x * x / (2.0 * sigma * sigma))


def make_cosine(n, freq=1.0):
    x = np.arange(n)
    return 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)


def make_random_positive(n, rng, sigma=0.5):
    return np.exp(rng.normal(0.0, sigma, n))


def make_taper(n):
    x = np.arange(n)
    mid = (n - 1) / 2.0
    return 1.0 - 0.5 * np.abs(x - mid) / max(mid, 1.0)


def make_edge_boost(n):
    x = np.arange(n)
    mid = (n - 1) / 2.0
    denom = max(mid, 1.0)
    return 0.3 + 0.7 * np.abs(x - mid) / denom
