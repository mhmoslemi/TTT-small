"""
Convex optimization approach to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative weights) with proper subgradients,
LP-based polishing, and diverse initializations (both symmetric and non-symmetric).
Key insight: the problem is convex (max of convex quadratics over a simplex),
so mirror descent converges to the global optimum.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_refine(a, M, deadline):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A_mat = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max(1, deadline - time.monotonic())})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def mirror_descent(a, deadline, rng):
    """Mirror descent (multiplicative weights) on the simplex.
    Minimize max(conv(a,a)) s.t. a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)

    if S < 1e-10:
        a = np.ones(n) / n
    else:

    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.1
    momentum = 0.5
    velocity = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        # Soft subgradient: weight positions by how close they are to the max
        # Use a threshold to include "near-peak" positions
        threshold = M - 0.02 * M  # 98% of max
        weights = np.maximum(c - threshold, 0)
        total_w = weights.sum()
        if total_w < 1e-15:
            # Only one peak position
            peak_idx = c.argmax()
            subgrad = np.zeros(n)
            for i in range(n):
                k = peak_idx
                if 0 <= k - i < n:
                    subgrad[i] = 2.0 * a[k - i]
        else:
            weights = weights / total_w
            subgrad = np.zeros(n)
            for i in range(n):
                # subgrad[i] = sum_k weights[k] * 2*a[k-i] (if valid)

                targets = ks - i  # k - i values

                # Actually k ranges over 0..2n-2, and we need 0 <= k-i < n
                # k from max(0, i) to min(2n-2, i+n-1)

                k_max = min(2 * n - 2, i + n - 1)
                if k_min <= k_max:

        # Momentum update

            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.5, lr * 1.02)
        else:

            for shrink in [0.5, 0.25, 0.1]:
                a_try = a * np.exp(np.clip(shrink * velocity, -50, 50))

                    a = a_try
                    stall = 0

                    if S_pert > 1e-10:

                    stall = 0

            break

    return best_a, best_score


def lp_optimize(a, deadline, rng):
    """LP-based optimization loop."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_score = compute_score(a)
    step = 0.05
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_refine(a, M, deadline)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            s = step * shrink
            a_new = (1.0 - s) * a + s * g
            a_new = np.clip(a_new, 0.0, None)
            S_new = a_new.sum()
            if S_new < 1e-10:
                continue
            a_new = a_new / S_new
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.3, step * 1.03)
        else:
            stall += 1
            step = max(0.002, step * 0.7)
            if stall > 50:
                # Random perturbation
                noise = np.abs(rng.normal(0, 0.05, n))
                a_pert = best_a * (1.0 + noise)
                a_pert = np.clip(a_pert, 0.0, None)
                S_pert = a_pert.sum()
                if S_pert > 1e-10:
                    a_pert = a_pert / S_pert
                    sc_pert = compute_score(a_pert)
                    if sc_pert < best_score:
                        best_a = a_pert
                        best_score = sc_pert
                    a = a_pert
                    stall = 0
                if stall > 150:
                    break

    return best_a, best_score


def combined_optimize(a, deadline, rng):
    """Combine mirror descent and LP optimization."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    # Phase 1: Mirror descent
    md_deadline = time.monotonic() + 0.4 * (deadline - time.monotonic())
    a, sc = mirror_descent(a, md_deadline, rng)
    best_a = a.copy()
    best_score = sc

    # Phase 2: LP refinement
    if time.monotonic() < deadline - 5:
        a, sc = lp_optimize(a, deadline, rng)
        if sc < best_score:
            best_a = a
            best_score = sc

    # Phase 3: Another round of mirror descent from LP result
    if time.monotonic() < deadline - 5:
        md_deadline2 = deadline
        a, sc = mirror_descent(a, md_deadline2, rng)
        if sc < best_score:
            best_a = a
            best_score = sc

    return best_a, best_score


def make_initialization(n, kind, rng):
    """Create various initializations."""
    if kind == 'uniform':
        return np.ones(n)
    elif kind == 'cosine':
        return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        sigma = 0.3 * n
        return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif kind == 'parabolic':
        c = (n - 1) / 2.0
        return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))
    elif kind == 'ramp_up':
        return np.linspace(0.1, 1.0, n)
    elif kind == 'ramp_down':
        return np.linspace(1.0, 0.1, n)
    elif kind == 'exp_up':
        return np.exp(1.5 * np.arange(n) / n)
    elif kind == 'exp_down':
        return np.exp(1.5 * (1 - np.arange(n) / n))
    elif kind == 'skew_cosine':
        x = np.linspace(0, 1, n)
        x = x ** 0.7  # skew
        return 0.5 * (1.0 + np.cos(np.pi * x))
    elif kind == 'skew_cosine2':
        x = np.linspace(0, 1, n)
        x = x ** 1.4  # more skew
        return 0.5 * (1.0 + np.cos(np.pi * x))
    elif kind == 'random':
        return np.abs(rng.normal(0, 1, n)) + 0.1
    elif kind == 'bimodal':
        c1 = 0.3 * n
        c2 = 0.7 * n
        s = 0.1 * n
        return np.exp(-0.5 * ((np.arange(n) - c1) / s) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / s) ** 2)
    elif kind == 'plateau':
        # Flat top with smooth edges
        x = np.arange(n) / (n - 1)
        return np.clip((x - 0.2) / 0.1, 0, 1) * np.clip((0.8 - x) / 0.1, 0, 1)
    else:
        return np.ones(n)
