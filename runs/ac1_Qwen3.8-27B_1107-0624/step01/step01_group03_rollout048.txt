    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = s
    return A

def solve_lp_direction(s, M, time_limit=3.0):
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success and res.x is not None:
            x = np.maximum(res.x, 0.0)
            if x.sum() > 1e-14:
                return x / x.sum()
    except:
        pass
    return None

def lp_optimize(s, n, rng, deadline, max_iters=200):
    best = s.copy()
    best_sc = fast_score(s)
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        M = np.max(conv)
        
        b = solve_lp_direction(s, M, time_limit=3.0)
        if b is None:
            no_imp += 1
            if no_imp > 5:
                break
            s = np.maximum(s + 0.01 * rng.standard_normal(n), 0)
            s = s / s.sum()
            continue
        
        # Line search
        best_t = 0.0
        best_ts = best_sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = np.maximum((1-t)*s + t*b, 0)
            ts = fast_score(cand)
            if ts < best_ts:
                best_ts = ts
                best_t = t
        
        if best_t == 0:
            no_imp += 1
            if no_imp > 10:
                break
            s = np.maximum(s + 0.005 * rng.standard_normal(n), 0)
            s = s / s.sum()
            continue
        
        s = np.maximum((1-best_t)*s + best_t*b, 0)
        s = s / s.sum()
        no_imp = 0
        
        cur_sc = fast_score(s)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = s.copy()
    
    return best, best_sc
