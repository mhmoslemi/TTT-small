    np.maximum(c, 0.0, out=c)
    return c

    p[1:] = expz / Z

def _self_conv(p):
    f = np.fft.rfft(p, _FFT_SELF)
    c = np.fft.irfft(f * f, _FFT_SELF).real
    c = c[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score_p(p):
    c = _self_conv(p)
    return 2.0 * N * float(np.max(c))

def _update_best(p, score):

def _probs_from_z(z):
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -40.0, 40.0)
    m = float(np.max(z))
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p

    f = np.where(np.isfinite(f), f, 0.0)

    end_time = start_time_global + max(1.0, budget_s) - 3.0
    
    _best_score = float('inf')
    _best_p = None
    
    rng = np.random.default_rng(seed)
    
    # Generate and score candidates
    candidates = _make_candidates(rng)
    
    cand_scores = []
    for z in candidates:
        p = _probs_from_z(z)
        s = _score_p(p)
        _update_best(p, s)
        cand_scores.append(s)
    
    order = np.argsort(cand_scores)
    candidates = [candidates[int(i)] for i in order]
    
    print(f"Initial best: {_best_score:.6f} from {len(candidates)} candidates")
    print(f"Top 3 initial scores: {sorted(cand_scores)[:3]}")
    
    # Phase 1: L-BFGS-B optimization (55% of time)
    phase1_end = start_time_global + 0.55 * float(budget_s)
    
    run_idx = 0
    num_cands = len(candidates)
    while time.monotonic() < phase1_end:
        rem = phase1_end - time.monotonic()
        if rem < 15:
            break
        
        if run_idx < num_cands:
            z0 = candidates[run_idx]
        else:
            if _best_p is not None:
                z_best = _density_to_z(_best_p)
                frac = max(0.0, min(1.0, rem / max(1.0, phase1_end - start_time_global)))
                noise = 0.03 + 0.25 * frac
                z0 = z_best + rng.normal(0, noise, N - 1)
            else:
                z0 = rng.normal(0, 1, N - 1)
        
        run_idx += 1
        
        # Adaptive alpha: larger when score is close to target
        if _best_score < 1.52:
            alpha = 10e6
        elif _best_score < 1.55:
            alpha = 5e6
        elif _best_score < 1.60:
            alpha = 2e6
        else:
            alpha = 5e5
        
        max_fun = max(100, min(1000, int(rem / 0.012)))
        _optimize_z(z0, alpha, max_fun, phase1_end)
        
        if run_idx % 5 == 0:
            elapsed = time.monotonic() - start_time_global
            print(f"Phase1 run {run_idx}: best={_best_score:.6f} elapsed={elapsed:.1f}s")
    
    print(f"Phase 1 done: best={_best_score:.6f} at t={time.monotonic()-start_time_global:.1f}s")
    
    # Phase 2: LP-based polishing (35% of time)
    phase2_end = start_time_global + 0.90 * float(budget_s)
    if _best_p is not None and time.monotonic() < phase2_end - 5:
        print("Starting LP polish phase...")
        f_start = _best_p.copy()
        _lp_polish(f_start, phase2_end)
        print(f"LP polish done: best={_best_score:.6f} at t={time.monotonic()-start_time_global:.1f}s")
    
    # Phase 3: Final subgradient + L-BFGS polish (remaining time)
    if _best_p is not None and time.monotonic() < end_time - 5:
        print("Starting final subgradient polish...")
        _subgradient_polish(_best_p, end_time - 10, rng)
        print(f"Subgradient done: best={_best_score:.6f}")
        
        # One more short L-BFGS if time remains
        rem = end_time - 5 - time.monotonic()
        if rem > 20:
            if _best_p is not None:
                z_best = _density_to_z(_best_p)
                z0 = z_best + rng.normal(0, 0.03, N - 1)
                _optimize_z(z0, 10e6, max(80, int(rem / 0.01)), end_time - 5)
            print(f"Final L-BFGS: best={_best_score:.6f}")
    
    # Build final sequence
    if _best_p is None:
        p = np.ones(N) / N
    else:
        p = _best_p
    p = np.clip(p, 0.0, 1000.0)
    seq = [float(v) for v in p]
    
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')
    
    if not np.isfinite(final_score) or final_score > 2.0001:
        seq = [1.0 / N] * N
        final_score = 2.0
    
    # Check against height_sequence_1
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.clip(base, 0.0, 1000.0)
            base_score = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(base_score) and base_score < final_score:
                seq = [float(v) for v in base]
                final_score = base_score
    except Exception:
        pass
    
    print(f"FINAL: score={final_score:.6f} length={len(seq)} time={time.monotonic()-start_time_global:.1f}s")
    return seq
