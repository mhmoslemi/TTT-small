import numpy as np
import time

"""
Soft Frank-Wolfe with multi-peak subgradient for minimizing
2n * max(conv(s,s)) / (sum(s))^2 over non-negative sequences.
Uses FFT-based convolution, top-K peak weighting, and adaptive step sizes.
Multiple sequence lengths and restarts are explored within the time budget.
"""

def compute_conv(s):
    """Compute full linear convolution of s with itself."""
    n = len(s)
    fft_len = 2 * n - 1
    f = 1
    while f < fft_len:
        f *= 2
    S = np.fft.rfft(s, f)
    return np.fft.irfft(S * S, f)[:fft_len]

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = compute_conv(s)
    return 2.0 * n * conv.max() / (total * total)

def subgradient_multi_peak(s, conv, K=5):
    """
    Compute a weighted subgradient using the top K peaks of the convolution.
    Returns a non-negative vector g where g[j] represents the "cost" of
    having mass at position j.
    """
    n = len(s)
    M = conv.max()
    if M < 1e-15:
        return np.ones(n) / n
    
    # Get top K peaks
    # For efficiency, use a threshold instead of full sort
    threshold = M * (1.0 - 0.05)  # peaks within 5% of max
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Limit to top K
    if len(active) > K:
        # Get the K largest
        active_vals = conv[active]
        top_k_local = np.argsort(active_vals)[::-1][:K]
        active = active[top_k_local]
    
    # Weight by peak value (higher peaks get more weight)
    w = conv[active].copy()
    w = w / w.sum()
    
    # Compute weighted subgradient
    g = np.zeros(n)
    j_idx = np.arange(n)
    for i, k in enumerate(active):
        diff = k - j_idx
        valid = (diff >= 0) & (diff < n)
        gk = np.zeros(n)
        gk[valid] = 2.0 * s[diff[valid]]
        if k % 2 == 0:
            kd = k // 2
            if 0 <= kd < n:
                gk[kd] = s[kd]
        g += w[i] * gk
    
    return g

def soft_fw(s, max_iters=500, t0=0.5, t_min=0.02, K=5, 
            deadline=None, verbose=False):
    """
    Soft Frank-Wolfe optimization.
    Moves mass from high-gradient positions to low-gradient positions.
    """
    n = len(s)
    best_s = s.copy()
    best_sc = fast_score(s)
    t = t0
    no_improve = 0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = compute_conv(s)
        g = subgradient_multi_peak(s, conv, K=K)
        
        # Soft FW: move toward 1/g distribution
        g_safe = np.maximum(g, 1e-12)
        target = 1.0 / g_safe
        target = target / target.sum()
        
        # Try the update
        s_new = (1.0 - t) * s + t * target
        s_new = np.maximum(s_new, 0.0)
        s_new = s_new / s_new.sum()
        
        sc = fast_score(s_new)
        
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_s = s_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        s = s_new
        t = max(t_min, t * 0.998)
        
        if no_improve > 100:
            # Restart with a smaller step size and some noise
            t = t_min
            s = best_s + 0.01 * np.random.standard_normal(n)
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    iter {it}, score={sc:.6f}, best={best_sc:.6f}, t={t:.4f}")
    
    return best_s, best_sc

def local_search(s, max_iters=500, deadline=None):
    """
    Simple local search: randomly perturb groups of coordinates.
    """
    n = len(s)
    best_s = s.copy()
    best_sc = fast_score(s)
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Choose a random block to perturb
        block_size = rng.integers(max(1, n // 100), max(2, n // 20))
        start = rng.integers(0, n - block_size + 1)
        
        # Perturb the block
        scale = rng.uniform(0.5, 1.5)
        s_new = s.copy()
        s_new[start:start+block_size] *= scale
        
        # Renormalize
        s_new = np.maximum(s_new, 0.0)
        s_sum = s_new.sum()
        if s_sum < 1e-10:
            continue
        s_new = s_new / s_sum
        
        sc = fast_score(s_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_s = s_new.copy()
            s = s_new
    
    return best_s, best_sc
