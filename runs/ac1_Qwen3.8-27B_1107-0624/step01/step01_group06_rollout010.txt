        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            eta = (J_s - polyak_lb) / (g_norm * g_norm)
            eta = min(eta, max_B)
            eta = max(eta, 1e-10)
        else:
            eta = 0.0

        a_new = y - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_a(a_new, n)

        beta = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        if beta > 1.0:
            y = a_new + ((beta - 1.0) / (t_prev + 1.0)) * (a_new - a)
        else:
            y = a_new.copy()

        a = a_new
        t_prev = t_prev + 1

        if iteration % 200 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 30 and iteration % 2000 == 0:
                    a = best_a.copy()
                    a = normalize_a(a * np.exp(rng.normal(0, 0.05, n)), n)
                    y = a.copy()
                    v = np.zeros(n)
                    stall = 0

    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def directed_local_search(a_init, budget_s, seed):
    """Targeted search: identify peaks, compute responsible elements, make cuts."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = normalize_a(a_init.copy(), n)
    best_a = a.copy()
    best_J = compute_J(a)
    c = convolve_fft(a)

    iters = 0
    while time.monotonic() < deadline and iters < 50000:
        iters += 1
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        count = i_hi - i_lo + 1
        if count < 2:
            break

        products = a[i_lo:i_hi + 1] * a[k_star - i_hi : k_star - i_lo + 1][::-1]
        total = products.sum()
        if total < 1e-15:
            break

        # Try top few contributing pairs
        top_pairs = np.argsort(products)[-min(5, count):][::-1]
        accepted = False
        for pi in top_pairs:
            i_idx = i_lo + int(pi)
            j_idx = k_star - i_idx
            if j_idx < 0 or j_idx >= n:
                continue

            # Reduce a[i_idx], scale up rest to maintain sum
            delta_frac = rng.uniform(0.005, 0.03)
            reduction = a[i_idx] * delta_frac
            if reduction < 1e-15:
                continue

            a_trial = a.copy()
            a_trial[i_idx] -= reduction
            # Compensate on j_idx (the pair partner) to reduce the specific product
            a_trial[j_idx] += reduction * 0.5
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize_a(a_trial, n)

            tJ = compute_J(a_trial)
            if tJ < best_J - 1e-9:
                best_J = tJ
                best_a = a_trial.copy()
                a = a_trial
                c = convolve_fft(a)
                accepted = True
                break

        if not accepted:
            # Broader random perturbation
            width = int(rng.integers(max(2, n // 500), max(3, n // 20)))
            s = int(rng.integers(0, n - width))
            a_trial = a.copy()
            a_trial[s:s + width] *= np.exp(rng.normal(0, 0.03, width))
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize_a(a_trial, n)
            tJ = compute_J(a_trial)
            if tJ < best_J:
                best_J = tJ
                best_a = a_trial.copy()
                a = a_trial
                c = convolve_fft(a)
            elif iters % 200 == 0:
                a = best_a.copy()
                c = convolve_fft(a)

    return best_a, best_J


def multiscale_refine(a_init, budget_s, seed):
    """Coarsen, optimize, refine back to full resolution."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    # Coarsen by factor 10
    block = 10
    n_c = n // block
    if n_c < 100:
        a_init, _ = run_subgradient_nesterov(a_init, budget_s, seed)
        return a_init, compute_J(a_init)

    a_c = a_init[:n_c * block].reshape(n_c, block).mean(axis=1)

    # Optimize coarse
    best_c, J_c = run_subgradient_nesterov(a_c, budget_s * 0.5, seed + 1,
                                           K_start=10, K_end=1)

    # Upsample
    a_up = np.repeat(best_c, block)[:n]
    a_up = normalize_a(a_up, n)

    # Fine-tune at full resolution
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_up, J_up = run_subgradient_nesterov(a_up, remaining, seed + 2,
                                              K_start=10, K_end=1)
    else:
        J_up = compute_J(a_up)

    return a_up, J_up


def make_all_inits(n, seed):
    rng = np.random.default_rng(seed)
    inits = {}
    inits['ones'] = np.ones(n)
    inits['gauss'] = make_gaussian(n, 0.25)
    inits['gauss_wide'] = make_gaussian(n, 0.4)
    inits['cos1'] = make_cosine(n, 1.0)
    inits['cos2'] = make_cosine(n, 2.0)
    inits['cos3'] = make_cosine(n, 3.0)
    inits['rand_logn_03'] = make_random_logn(n, rng, 0.3)
    inits['rand_logn_07'] = make_random_logn(n, rng, 0.7)
    inits['bimodal'] = make_gaussian(n, 0.9) * 0.5 + make_gaussian(n, 0.1) * 0.5
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer combining:
    1. Nesterov-accelerated top-K subgradient with Polyak step sizes
    2. Conflict-aware directed local search (peak pair elimination)
    3. Multi-scale coarsen-refine
    4. Final fine-tuning on global best
    """
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 8.0)

    # Load parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    parent = normalize_a(parent, n)
    print(f"n = {n}")

    best_global = parent.copy()
    best_global_J = float(evaluate_sequence(best_global.tolist()))
    print(f"Parent score: {best_global_J:.6f}")

    # Build and score initializations
    inits = make_all_inits(n, seed)
    inits['parent'] = parent.copy()
    scored = []
    for name, arr in inits.items():
        arr = normalize_a(arr.copy(), n)
        J = compute_J(arr)
        scored.append((J, name, arr))
    scored.sort(key=lambda x: x[0])
    print("Initializations sorted:")
    for J, name, _ in scored:
        print(f"  {name}: {J:.6f}")

    # Check if parent is better than all inits
    parent_J = computed_pJ if 'computed_pJ' in dir() else compute_J(parent)
    if parent_J < scored[0][0]:
        scored.insert(0, (parent_J, 'parent_best', parent.copy()))

    remaining = total_deadline - time.monotonic()

    # --- Phase A: Subgradient on top-3 inits (55% of time) ---
    phase_A_budget = remaining * 0.55
    per_budget_A = phase_A_budget / 3.0

    for i in range(min(3, len(scored))):
        if time.monotonic() > total_deadline - 15:
            break
        J_i, name, arr_i = scored[i]
        b = min(per_budget_A, total_deadline - time.monotonic() - 12)
        if b < 10:
            break
        print(f"\n[Phase A] Subgradient on '{name}' (budget={b:.0f}s, init_J={J_i:.4f})")
        res_a, res_J = run_subgradient_nesterov(arr_i, b, seed + 100 * i,
                                                K_start=30, K_end=1)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_J:
            best_global = res_a.copy()
            best_global_J = val
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  {name}: local={res_J:.4f}, verified={val:.4f}")

    # --- Phase B: Directed local search on global best (20% of time) ---
    remaining = total_deadline - time.monotonic()
    phase_B_budget = remaining * 0.20
    if phase_B_budget > 10:
        print(f"\n[Phase B] Directed local search (budget={phase_B_budget:.0f}s)")
        res_a, res_J = directed_local_search(best_global, phase_B_budget, seed + 500)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_J:
            best_global = res_a.copy()
            best_global_J = val
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  local={res_J:.4f}, verified={val:.4f}")

    # --- Phase C: Multi-scale on a different init (10% of time) ---
    remaining = total_deadline - time.monotonic()
    phase_C_budget = remaining * 0.12
    if phase_C_budget > 15:
        # Use the 4th or 5th best init for diversity
        init_c = scored[min(3, len(scored) - 1)][2].copy()
        # Blend with current best for warm start
        init_c = 0.5 * init_c + 0.5 * best_global
        init_c = normalize_a(init_c, n)
        print(f"\n[Phase C] Multi-scale (budget={phase_C_budget:.0f}s)")
        res_a, res_J = multiscale_refine(init_c, phase_C_budget, seed + 900)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_J:
            best_global = res_a.copy()
            best_global_J = val
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  local={res_J:.4f}, verified={val:.4f}")

    # --- Phase D: Final fine-tuning (remaining time) ---
    remaining = total_deadline - time.monotonic() - 5.0
    if remaining > 15:
        print(f"\n[Phase D] Final fine-tuning (budget={remaining:.0f}s)")
        res_a, res_J = run_subgradient_nesterov(best_global, remaining, seed + 7777,
                                                K_start=5, K_end=1,
                                                polyak_lb=1.0 + 1.0 / n)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_J:
            best_global = res_a.copy()
            best_global_J = val
            print(f"  *** FINAL BEST: {val:.6f} ***")
        else:
            print(f"  local={res_J:.4f}, verified={val:.4f}, keeping {best_global_J:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_J = min(best_global_J, final_val)

    print(f"\n=== FINAL SCORE: {best_global_J:.6f} ===")
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
