import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def next_power_of_two(m):
    L = 1
    while L < m:

def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 1e-300, 1000.0)

def greedy_normalized(n):
    """Coefficients of the formal square root of 1/(1-x), normalized."""
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)

def actual_score_fft(p, direct_limit=768):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01 * 1e-6:
        return np.inf

    if n <= direct_limit:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.inf
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))

def adam_optimize(p, budget_s, seed=0):
    """
    Softmax-projected gradient descent for a smoothed maximum-convolution
    objective.  p is interpreted as sum-one probabilities.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 0.5:
        f = actual_score_fft(p)
        return p, f

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = 0.05
    elif n < 256:
        lr = 0.03
    elif n < 1024:
        lr = 0.02
    else:
        lr = 0.015

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = 5.0 + 45.0 * np.exp(-4.0 * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)  # p sums to one
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0.0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -50.0, 50.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -80.0, 80.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def lp_improve_rows(p, budget_s, max_rows=None):
    """
    Cutting-plane version of the LP-direction method:

      maximize sum(g)
      subject to f*g <= M, g >= 0

    where f is normalized to sum sqrt(2n) and M = max(f*f).
    """
    if _scipy_opt is None:
        f = actual_score_fft(p)
        return p, f

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 2:
        n = 16
        p = normalize_nonneg(np.ones(n, dtype=np.float64))
    budget_s = float(budget_s)
    if budget_s < 25.0:
        f = actual_score_fft(p)
        return p, f

    if max_rows is None:
        if n > 4000:
            max_rows = 900
        elif n > 1500:
            max_rows = 1000
        else:
            max_rows = 1100

    # Keep dense A within a modest element count.
    max_elements = 6000000
    while max_rows * n > max_elements:
        max_rows = max(128, int(max_rows * 0.85))
    max_rows = max(128, min(int(max_rows), max_elements // max(1, n)))

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score_fft(f)
    if not np.isfinite(cur_score):
        return p, cur_score

    best_f = f.copy()
    best_score = float(cur_score)

    end = time.monotonic() + max(0.0, budget_s - 5.0)

    r0 = min(max_rows, max(80, max_rows // 2))
    rows0 = np.unique(np.linspace(0, 2 * n - 2, r0, dtype=int).astype(int))
    rows = [int(k) for k in rows0.tolist()]
    row_set = set(rows)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        A = np.zeros((int(rows_arr.size), n), dtype=np.float64)
        for ii in range(int(rows_arr.size)):
            k = int(rows_arr[ii])
            lo = k - n + 1
            if lo < 0:
                lo = 0
            hi = k if k < n else n - 1
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(int(rows_arr.size), cur_score, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(5.0, min(90.0, rem * 0.25))
        res = None
        try:
            res = _scipy_opt.linprog(
                cobj,
                A_ub=A,
                b_ub=b,
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": float(tl)},
            )
        except Exception:
            try:
                res = _scipy_opt.linprog(
                    cobj,
                    A_ub=A,
                    b_ub=b,
                    bounds=(0.0, None),
                    method="highs",
                )
            except Exception:
                break

        if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        # Check full-constraint violations.
        if n > 1024:
            L2 = next_power_of_two(2 * n - 1)
            Pf = np.fft.rfft(f, L2)
            Pg = np.fft.rfft(g, L2)
            conv = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        else:
            conv = np.convolve(f, g)
        conv = np.maximum(conv, 0.0)
        tol = 1e-7 + 1e-7 * cur_score
        violations = conv - cur_score
        add_needed = np.any(violations > tol)

        if add_needed:
            added = 0
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                for ix in order:
                    if added >= 0.5 * (max_rows - len(rows)) or added >= 256:
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            if added == 0:
                # If existing constraints are still violated too much, stop this local LP phase.
                if float(np.max(violations)) > 10.0 * tol:
                    break
            else:
                continue

        g = g * (S0 / sg)

        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.5, 0.25, 0.15, 0.10, 0.05, 0.02):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()

        if accepted is None or not np.isfinite(accepted_score):
            break
        if accepted_score >= best_score:
            break

        f = accepted
        cur_score = accepted_score
        best_f = f.copy()
        best_score = cur_score

        if it % 5 == 0:
            print("LP cut n=%d iter=%d score=%.6f rows=%d" % (n, it, best_score, len(rows)))

    final_score = actual_score_fft(best_f)
    if final_score < best_score:
        best_score = final_score
    return normalize_nonneg(best_f), best_score
