import time

def convolve_full(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    conv = convolve_full(a)
    return 2.0 * n * np.max(conv) / (S*S)

def smooth_grad(a, tau):
    """Compute smoothed J and gradient using FFT."""
    n = len(a)
    size = 1
    while size < 3*n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2*n-1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_s / sum_exp
    J = 2.0 * n * Tmax / (S*S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S*S) - (4.0 * n * Tmax) / (S**3)
    return J, grad

"""
Hybrid optimizer: FFT-accelerated smoothed gradient descent followed by
direct coordinate descent on the true (unsmoothed) objective.
The coordinate descent uses O(n) convolution updates per candidate value,
enabling many passes within the time budget. Includes adaptive restarts
and refinement to escape local minima.
"""

import numpy as np
import time


def convolve_full(a):
    """Full self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    conv = convolve_full(a)
    return float(2.0 * n * np.max(conv) / (S * S))


def smooth_J_grad(a, tau):
    """Smoothed objective and gradient via log-sum-exp + FFT."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_s / sum_exp
    J = float(2.0 * n * Tmax / (S * S))
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S ** 3)
    return J, grad


def run_gradient_phase(a_init, budget, seed):
    """Adam gradient descent on smoothed objective. Returns (best_a, best_J)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau_start = 3.0
    tau_end = 0.02
    lr_start = 0.03
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    best_J = compute_J(a)
    start_time = time.monotonic()
    stall = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start / (1.0 + 0.3 * t / 500.0)

        J_s, grad = smooth_J_grad(a, tau)
        if not np.isfinite(J_s) or not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m[:] = 0
            v[:] = 0
            t = 0
            continue

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)

        if t % 3000 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 8:
                    # Perturb to escape
                    cand = a.copy()
                    w = max(1, n // 100)
                    s = int(rng.integers(0, max(1, n - w)))
                    cand[s:s + w] *= np.exp(rng.normal(0, 0.3, w))
                    cand = np.maximum(cand, 0)
                    cS = np.sum(cand)
                    if cS > 0:
                        cand *= n / cS
                    cJ = compute_J(cand)
                    if cJ < best_J:
                        a = cand
                        best_a = cand.copy()
                        best_J = cJ
                        m[:] = 0
                        v[:] = 0
                        t = 0
                        stall = 0
                    else:
                        a = best_a.copy()
                        m[:] = 0
                        v[:] = 0
                        t = 0
                        stall = 0

    return best_a, best_J


def coord_optimize_pass(a, conv, order, verbose=False):
    """One pass of coordinate descent. Modifies a and conv in place. Returns total improvement."""
    n = len(a)
    n_full = 2 * n - 1
    total_improve = 0.0
    S = np.sum(a)
    current_max = np.max(conv)
    current_J = 2.0 * n * current_max / (S * S)

    for i in order:

        m_start = i
        m_end = min(n_full - 1, i + n - 1)

        # Unaffected max
        if m_start == 0 and m_end == n_full - 1:
            unaffected_max = 0.0
        else:
            if m_start > 0 and m_end < n_full - 1:
                unaffected_max = max(np.max(conv[:m_start]), np.max(conv[m_end + 1:]))
            elif m_start > 0:
                unaffected_max = np.max(conv[:m_start])

                unaffected_max = np.max(conv[m_end + 1:])

        # Affected slice (always length n since i in [0, n-1])

        if curr_max < 1e-30:
            continue

            candidates.extend(extra)

        best_t1 = t0
        best_J_i = 2.0 * n * curr_max / (S * S)
        best_aff = aff if best_t1 == t0 else None

            for _ in range(8):
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                for tc in (c1, c2):
                    pass  # We'll evaluate both
                # Evaluate at c1
                delta1 = c1 - t0
                if abs(c1 - t0) > 1e-12 and S + (c1 - t0) >= 0.01:
                    aff1 = aff + 2.0 * delta1 * a
                    if 2 * i <= n_full - 1:
                        aff1[i] = conv[2 * i] + delta1 * (c1 + t0)
                    max1 = max(unaffected_max, np.max(aff1))
                    J1 = 2.0 * n * max1 / ((S + delta1) ** 2)
                else:
                    J1 = best_J_i
                # Evaluate at c2
                delta2 = c2 - t0
                if abs(c2 - t0) > 1e-12 and S + (c2 - t0) >= 0.01:

                else:
                    J2 = best_J_i
                if J1 < J2:
                    hi = c2
                else:
                    lo = c1
                # Check if refinement improves

                   afft = aff + 2.0 * deltat * a

                    maxt = max(unaffected_max, np.max(afft))
                    Jt = 2.0 * n * maxt / ((S + deltat) ** 2)
                    if Jt < best_J_i:
                        best_J_i = Jt
                        best_t1 = tc
                        best_aff = afft
        elif best_t1 == t0 and t0 > 0:
            # Try small perturbations around t0
            for perturb in (t0 * 0.95, t0 * 1.05, t0 - 0.05, t0 + 0.05):
                if perturb < 0:
                    continue
                t1 = perturb
                new_S = S + (t1 - t0)
                if new_S < 0.01:
                    continue
                delta = t1 - t0
                if abs(delta) < 1e-12:
                    continue
                new_aff = aff + 2.0 * delta * a

                if J_i < best_J_i:
                    best_J_i = J_i
                    best_t1 = t1
                    best_aff = new_aff

        if best_t1 != t0:
            a[i] = best_t1
            conv[m_start:m_end + 1] = best_aff
            S += (best_t1 - t0)
            total_improve += 1

    new_S = np.sum(a)
    new_max = np.max(conv)
    new_J = 2.0 * n * new_max / (new_S * new_S)
    if verbose:
        print(f"  pass: J {current_J:.8f} -> {new_J:.8f} ({total_improve} coords changed)")
    return new_J - current_J

        if frac < 0.3:
            order = rng.permutation(n)

def run_refinement(a_init, budget, seed):
    """Final refinement: focus on high-impact coordinates. Returns (best_a, best_J)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    conv = convolve_full(a)
    best_J = 2.0 * n * np.max(conv) / (S * S)
    best_a = a.copy()

    # Identify top coordinates by "impact" (how much they contribute to max conv)
    max_pos = np.argmax(conv)
    # Coordinates near the peak of convolution have the most impact
    # The peak is at lag max_pos, so coordinates around i and max_pos - i matter most
    impact_centers = [max_pos // 2, n - 1 - max_pos // 2]

    pass_count = 0
    while time.monotonic() < deadline:
        pass_count += 1
        # Process in order of impact (close to peak first)
        order = np.arange(n)
        # Sort by distance to impact centers
        dists = np.minimum(
            np.abs(order.astype(float) - impact_centers[0]),
            np.abs(order.astype(float) - impact_centers[1])
        )
        order = order[np.argsort(dists)]

        J_before = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
        improvement = coord_optimize_pass(a, conv, order)
        J_after = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)

        if J_after < best_J - 1e-9:
            best_J = J_after
            best_a = a.copy()

        if pass_count % 30 == 0:
            print(f"  ref pass={pass_count} J={best_J:.8f}")

        if J_after >= best_J and pass_count > 20:
            break

    return best_a, best_J
