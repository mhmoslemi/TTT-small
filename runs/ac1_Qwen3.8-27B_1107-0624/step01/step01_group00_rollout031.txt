    a = np.maximum(a, 0.0)
    return _to_sum1(a)


def _compute_conv(a, M):
    """Linear convolution a*a via FFT, returns array of length 2n-1."""
    fa = np.fft.rfft(a, M)
    c = np.fft.irfft(fa * fa, M)
    return c[: 2 * len(a) - 1]


def _compute_grad_lse(b, a, tau, M_corr):
    """Gradient of log-sum-exp(b/tau) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    # Softmax weights
    bt = b / tau
    bt -= bt.max()
    eb = np.exp(bt)
    s = eb.sum()
    if s < 1e-300:
        w = np.full(len(b), 1.0 / len(b))
    else:
        w = eb / s
    # Cross-correlation: R[j] = sum_k w[k+j] * a[k]
    W = np.fft.rfft(w, M_corr)
    A = np.fft.rfft(a, M_corr)
    R = np.fft.irfft(W * np.conj(A), M_corr)[:n]
    return 2.0 * R / tau


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for the non-negative sequence minimizing the objective."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    t_end = t_start + float(budget_s) - 3.0

        b0 = np.clip(b0, 0.0, 1000.0)
        if b0.ndim == 1 and b0.size >= 4 and np.isfinite(b0).all():
            base = b0
    except Exception:
        pass

    n = base.size
    M_conv = _next_pow2(2 * n - 1)
    M_corr = _next_pow2(3 * n - 2)

    # Build diverse structured seeds
    x = np.linspace(0.0, 1.0, n)
    seeds = []
    seeds.append(_make_palindromic(_to_sum1(np.abs(base))))
    seeds.append(_make_palindromic(_to_sum1(np.abs(base[::-1]))))
    # U-shape
    seeds.append(_make_palindromic(_to_sum1((x - 0.5) ** 2 + 0.05)))
    # Cosine bump
    seeds.append(_make_palindromic(_to_sum1(np.cos(np.pi * (x - 0.5)) + 0.3)))
    # Two bumps
    bump = np.exp(-40.0 * (x - 0.2) ** 2) + np.exp(-40.0 * (x - 0.8) ** 2) + 0.1
    seeds.append(_make_palindromic(_to_sum1(bump)))
    # High-freq modulation
    mod = 1.0 + 0.4 * np.cos(4 * np.pi * x)
    seeds.append(_make_palindromic(_to_sum1(np.abs(mod))))
    # Random smooth seed
    rs = np.random.default_rng(seed + 7)
    r = rs.normal(0, 1, n)
    sm = np.convolve(r, np.ones(11) / 11, mode='same')
