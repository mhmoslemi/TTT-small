```python
import time
from scipy.optimize import linprog

def lp_step(f, M, n):
    """
    Solve: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0
    
    (f*g)(k) = sum_j f[j] * g[k-j] for valid j
    """
    # c = -1 for all variables (maximize sum = minimize -sum)
    c = -np.ones(n)
    
    # A_ub * g <= b_ub
    # (f*g)(k) = sum_j f[j] * g[k-j] <= M
    # For k = 0, 1, ..., 2n-2:
    #   sum_{j: 0<=j<n, 0<=k-j<n} f[j] * g[k-j] <= M
    
    # Let's build the matrix
    # Row k: A_ub[k, j] = f[j] if 0 <= k-j < n, i.e., k-n < j <= k
    # Actually: (f*g)(k) = sum_j f[j] * g[k-j]
    # For g index j to contribute, we need 0 <= k-j < n, i.e., k-n < j <= k
    # So for row k, column j, the entry is f[j] if max(0, k-n+1) <= j <= min(k, n-1)
    
    # Actually let me be more careful:
    # (f*g)(k) = sum_{j=0}^{n-1} f[j] * g[k-j]
    # g[k-j] is only defined for 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k
    # And j must be in [0, n-1]
    
    # So for row k, the non-zero entries are at columns j where:
    #   max(0, k-(n-1)) <= j <= min(n-1, k)
    # and the entry at (k, j) is f[j]
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        A_ub[k, j_min:j_max+1] = f[j_min:j_max+1]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
import numpy as np
import time
from scipy.optimize import linprog

# Helper functions (top-level, no closures)

def full_convolve_fft(a):
    """Full linear convolution via FFT."""
    n = a.size
    if n <= 32:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0)  # clip tiny negatives from FFT

def internal_score(a):
    """Fast score computation."""
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    return 2.0 * n * c.max() / (s * s)

def make_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix C such that (C@g)(k) = (a*g)(k)."""
    n = a.size
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n-1, k)
        C[k, j_min:j_max+1] = a[j_min:j_max+1]
    return C

def lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0.
    Returns normalized g (sum=1) and the sum S.
    """
    n = a.size
    c_obj = -np.ones(n)  # minimize -sum(g)
    C = make_conv_matrix(a)
    A_ub = C
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return None, 0
    g = res.x
    S = g.sum()
    if S < 1e-10:
        return None, 0
    return g / S, S

def softmax_seq(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def normalize_nonneg(a):
    a = np.clip(a, 0, None)
    s = a.sum()

def interpolate_profile(p, n_new):
    n_old = p.size
    if n_old == n_new:
        return p
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, p)

def lp_iterations(a, max_iters=500, t_init=0.3, t_min=0.01, deadline=None):
    """
    Iterative LP-based peaking reduction.
    Returns best sequence found.
    """
    a = normalize_nonneg(a)
    n = a.size
    M = full_convolve_fft(a).max()
    best_a = a.copy()
    best_M = M
    t = t_init
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g, S = lp_direction(a, M)
        if g is None:
            break
        
        # Try the update
        a_new = (1-t)*a + t*g
        a_new = normalize_nonneg(a_new)
        if a_new is None:
            t *= 0.5
            continue
        
        M_new = full_convolve_fft(a_new).max()
        
        if M_new < M - 1e-12:
            a = a_new
            M = M_new
            if M < best_M:
                best_M = M
                best_a = a.copy()
            t = min(0.5, t * 1.05)  # slightly increase step
        else:
            t *= 0.5
            if t < t_min:
                # Try a random perturbation
                break
    
    return best_a, best_M

            best_a = a_new.copy()

        max_c = M_new
```

```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses LP-based iterative peaking reduction, gradient-based polish with soft-max, and multi-resolution upsampling.
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_convolve_fft(a):
    """Full linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < 0:
        c = np.maximum(c, 0.0)
    return c


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-30:
        return None
    return a / s


def interpolate_profile(p, n_new):
    """Linear interpolation of a profile to a new length."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n_old = p.size
    if n_old == n_new:
        return p.copy()
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p)


def make_conv_matrix(a):
    """Build (2n-1) x n convolution matrix C where (C@g)(k) = (a*g)(k)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        C[k, j_min:j_max + 1] = a[j_min:j_max + 1]
    return C


def lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0.
    Returns (normalized_g, S) where S = sum(g_optimal), or (None, 0) on failure.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c_obj = -np.ones(n)
    C = make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
    except Exception:
        return None, 0.0

    if not res.success:
        return None, 0.0
    g = np.asarray(res.x, dtype=np.float64)
    S = float(g.sum())
    if S < 1e-10 or not np.isfinite(S):
        return None, 0.0
    return g / S, S


def corr_weights_fft(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 using FFT."""
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def lp_iterations(a, max_iters=2000, deadline=None, verbose=False):
    """
    Iterative LP-based peaking reduction.
    a: initial sequence (will be normalized to sum=1)
    Returns (best_a, best_M) where best_M = max(conv(best_a, best_a)).
    """
    a = normalize_nonneg(a)
    if a is None:
        return None, np.inf
    n = a.size
    c = full_convolve_fft(a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    t = 0.3
    stall = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g, S = lp_direction(a, M)
        if g is None:
            break

        if S <= 1.0 + 1e-10:
            stall += 1
            if stall > 50:
                break
            t = max(0.01, t * 0.7)
            continue

        stall = 0

        for attempt in range(8):
            a_new = (1.0 - t) * a + t * g
            a_new = normalize_nonneg(a_new)
            if a_new is None:
                t *= 0.5
                break
            c_new = full_convolve_fft(a_new)
            M_new = float(c_new.max())

            if M_new < M - 1e-14:
                a = a_new
                M = M_new
                if M < best_M - 1e-14:
                    best_M = M
                    best_a = a.copy()
                t = min(0.5, t * 1.08)
                break
            else:
                t *= 0.5
                if t < 0.005:
                    break

        if verbose and it % 200 == 0:
            print(f"    LP iter {it}: M={M:.10f}, S={S:.4f}, t={t:.4f}")

    return best_a, best_M


def gradient_polish(a, max_iters=5000, deadline=None, lr=0.04, verbose=False):
    """
    Gradient-based polish using soft-max approximation of max.
    a: initial sequence (normalized to sum=1)
    Returns (best_a, best_M).
    """
    a = normalize_nonneg(a)
    if a is None:
        return None, np.inf
    n = a.size
    c = full_convolve_fft(a)
    max_c = float(c.max())
    beta = max(0.05 * max_c, 1e-6)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_M = max_c
    no_imp = 0
    base_lr = lr

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Soft-max weights
        x = (c - max_c) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s_ex = ex.sum()
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # Gradient: grad[j] = sum_k w[k] * d(c[k])/d(a[j])
        # d(c[k])/d(a[j]) = a[k-j] (if valid) + a[j-k] (if valid and j != k-j)
        # Simplified: grad[j] = 2 * h[j] where h[j] = sum_i w[i+j]*a[i]
        # More precisely: grad[j] = h_fwd[j] + h_bwd[j]
        h = corr_weights_fft(w, a)
        # h_bwd[j] = sum_i a[i]*w[j-i] = conv(a, w)[j]
        N2 = 1
        while N2 < 2 * n - 1:
            N2 <<= 1
        fa_fw = np.fft.rfft(a, N2)
        fw_fw = np.fft.rfft(w, N2)
        conv_aw = np.fft.irfft(fa_fw * fw_fw, N2)[:2 * n - 1]
        grad = h + conv_aw[:n]

        # Logit-space gradient (natural gradient)
        z = np.log(a + 1e-30)
        z -= z.mean()
        gz = a * grad
        gz -= gz.mean()

        # Adam update
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z -= lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -30, 30)
        z -= z.mean()

        a_new = np.exp(z)
        a_new = a_new / a_new.sum()

        c = full_convolve_fft(a_new)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.95)

        a = a_new
        max_c = M_new

        if it % 300 == 0:
            beta = max(0.01 * max_c, 1e-8)
            if verbose:
                print(f"    Grad iter {it}: M={M:.10f}, beta={beta:.6f}, lr={lr:.5f}")

        if no_imp > 800:
            z = 0.8 * np.log(a + 1e-30) - 0.8 * np.log(a + 1e-30).mean()
            z += 0.3 * np.random.randn(n)
            z = np.clip(z, -30, 30)
            z -= z.mean()
            a = np.exp(z)
            a = a / a.sum()
            c = full_convolve_fft(a)
            max_c = float(c.max())
            beta = max(0.1 * max_c, 1e-6)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a, best_M


def make_linear_profile(n, eta):
    """Monotone linear profile 1 + eta*t, normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_profile(n, freq, amp):
    """1 + amp*cos(2*pi*freq*t), normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_quadratic_profile(n, c1, c2):
    """1 + c1*t + c2*t^2, normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + c1 * t + c2 * t * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bell_profile(n, center, width):
    """Gaussian-like profile centered at 'center' with given width."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequences minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
      1. Parametric initialization at multiple lengths
      2. LP-based iterative peaking reduction at n=128
      3. Gradient polish at n=128
      4. Upscale to n=256, LP + gradient
      5. Upscale to n=512, LP + gradient
      6. Final verification
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(a, score):
        nonlocal best_seq, best_score
        if a is not None and np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_seq = np.asarray(a, dtype=np.float64).copy()
            print(f"  [t={time.monotonic()-t_start:7.1f}s] new best: score={score:.10f}, n={a.size}")

    # === Phase 0: Initialization ===
    print("Phase 0: Initialization")

    # Try parent if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = np.maximum(parent, 0.0)
            s = parent.sum()
            if s > 0.01:
                parent = parent / s
                sc = internal_score(parent)
                try_update(parent, sc)
    except Exception:
        pass

    # Parametric families at n=128 and n=256
    for n_init in [128, 256]:
        for eta in [1.0, 1.5, 2.0, 2.5, 3.0, 4.0]:
            a = make_linear_profile(n_init, eta)
            try_update(a, internal_score(a))

        for c1, c2 in [(2.0, -1.0), (3.0, -2.0), (1.5, 0.5), (2.5, -1.5), (4.0, -3.0), (1.0, 1.0)]:
            a = make_quadratic_profile(n_init, c1, c2)
            try_update(a, internal_score(a))

        for f in [1, 2, 3, 4]:
            for amp in [0.3, 0.5, 0.7, 1.0]:
                a = make_cosine_profile(n_init, f, amp)
                try_update(a, internal_score(a))

        for center in [0.3, 0.4, 0.5, 0.6]:
            for width in [0.1, 0.15, 0.2, 0.3]:
                a = make_bell_profile(n_init, center, width)
                try_update(a, internal_score(a))

    print(f"  After init: best={best_score:.10f}")

    # === Phase 1: LP optimization at n=128 ===
    print("Phase 1: LP optimization at n=128")
    n1 = 128
    a1 = interpolate_profile(best_seq, n1) if best_seq is not None else make_linear_profile(n1, 2.0)
    a1 = normalize_nonneg(a1)

    phase1_end = min(deadline, t_start + 500.0)
    a1_best, M1 = lp_iterations(a1, max_iters=3000, deadline=phase1_end, verbose=True)
    if a1_best is not None:
        sc = 2.0 * n1 * M1
        try_update(a1_best, sc)

    # Also try from a few different starting points
    for start_eta in [1.5, 3.0]:
        if time.monotonic() > phase1_end - 60:
            break
        a_try = make_linear_profile(n1, start_eta)
        phase_sub_end = min(phase1_end, time.monotonic() + 120.0)
        a_try_best, M_try = lp_iterations(a_try, max_iters=800, deadline=phase_sub_end)
        if a_try_best is not None:
            sc = 2.0 * n1 * M_try
            try_update(a_try_best, sc)

    print(f"  After LP n=128: best={best_score:.10f}")

    # === Phase 2: Gradient polish at n=128 ===
    print("Phase 2: Gradient polish at n=128")
    phase2_end = min(deadline, t_start + 800.0)
    a2_start = best_seq if best_seq.size == n1 else interpolate_profile(best_seq, n1)
    a2_start = normalize_nonneg(a2_start)

    a2_best, M2 = gradient_polish(a2_start, max_iters=8000, deadline=phase2_end, lr=0.04)
    if a2_best is not None:
        sc = 2.0 * n1 * M2
        try_update(a2_best, sc)

    print(f"  After grad n=128: best={best_score:.10f}")

    # === Phase 3: Upscale to n=256 ===
    print("Phase 3: Optimization at n=256")
    n2 = 256
    phase3_start = time.monotonic()
    phase3_end = min(deadline, t_start + 1200.0)

    a3 = interpolate_profile(best_seq, n2)
    a3 = normalize_nonneg(a3)
    sc = internal_score(a3)
    try_update(a3, sc)

    # LP at n=256
    if time.monotonic() < phase3_end - 100:
        phase3_lp_end = min(phase3_end - 100, time.monotonic() + 250.0)
        a3_lp, M3 = lp_iterations(a3, max_iters=800, deadline=phase3_lp_end, verbose=True)
        if a3_lp is not None:
            sc = 2.0 * n2 * M3
            try_update(a3_lp, sc)

    # Gradient at n=256
    if time.monotonic() < phase3_end - 30:
        a3g = best_seq if best_seq.size == n2 else interpolate_profile(best_seq, n2)
        a3g = normalize_nonneg(a3g)
        a3_best, M3g = gradient_polish(a3g, max_iters=5000, deadline=phase3_end, lr=0.03)
        if a3_best is not None:
            sc = 2.0 * n2 * M3g
            try_update(a3_best, sc)

    print(f"  After n=256: best={best_score:.10f}")

    # === Phase 4: Upscale to n=512 ===
    print("Phase 4: Optimization at n=512")
    n4 = 512
    phase4_end = min(deadline, t_start + budget_s - 60.0)

    a4 = interpolate_profile(best_seq, n4)
    a4 = normalize_nonneg(a4)
    sc = internal_score(a4)
    try_update(a4, sc)

    # LP at n=512 (fewer iterations due to higher cost per LP)
    if time.monotonic() < phase4_end - 120:
        phase4_lp_end = min(phase4_end - 120, time.monotonic() + 300.0)
        a4_lp, M4 = lp_iterations(a4, max_iters=300, deadline=phase4_lp_end, verbose=True)
        if a4_lp is not None:
            sc = 2.0 * n4 * M4
            try_update(a4_lp, sc)

    # Gradient polish at n=512
    if time.monotonic() < phase4_end - 30:
        a4g = best_seq if best_seq.size == n4 else interpolate_profile(best_seq, n4)
        a4g = normalize_nonneg(a4g)
        a4_best, M4g = gradient_polish(a4g, max_iters=5000, deadline=phase4_end, lr=0.02)
        if a4_best is not None:
            sc = 2.0 * n4 * M4g
            try_update(a4_best, sc)

    print(f"  After n=512: best={best_score:.10f}")

    # === Phase 5: Try additional variations ===
    if time.monotonic() < deadline - 60 and best_score > 1.505:
        print("Phase 5: Additional variations")
        # Try different n values
        for n_alt in [192, 384]:
            if time.monotonic() > deadline - 40:
                break
            a_alt = interpolate_profile(best_seq, n_alt)
            a_alt = normalize_nonneg(a_alt)
            sub_end = min(deadline - 40, time.monotonic() + 80.0)
            a_alt_bp, M_alt = gradient_polish(a_alt, max_iters=3000, deadline=sub_end, lr=0.03)
            if a_alt_bp is not None:
                sc = 2.0 * n_alt * M_alt
                try_update(a_alt_bp, sc)

    # === Final verification ===
    print("Final verification")
    final_seq = best_seq
    final_score = best_score

    if final_seq is not None:
        # Re-score with exact convolution
        if final_seq.size <= 2048:
            c_exact = np.convolve(final_seq, final_seq)
            m_exact = float(c_exact.max())
            s_exact = float(final_seq.sum())
            if s_exact > 0.01:
                sc_exact = float(2.0 * final_seq.size * m_exact / (s_exact * s_exact))
                if sc_exact < final_score:
                    final_score = sc_exact

        # Call evaluator
        try:
            sc_eval = evaluate_sequence(final_seq.tolist())
            if np.isfinite(sc_eval):
                print(f"  Evaluator score: {sc_eval:.10f}")
                if sc_eval < final_score:
                    final_score = sc_eval
        except Exception as e:
            print(f"  Evaluator error: {e}")

    if final_seq is None:
        final_seq = make_linear_profile(256, 2.0)
        final_score = internal_score(final_seq)

    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0:
        final_seq = final_seq / s

    print(f"FINAL: score={final_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
```
