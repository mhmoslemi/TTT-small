    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:

import numpy as np
from scipy.signal import fftconvolve
import time

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(ind[cond])
    theta = cssv[k - 1] / k
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def compute_grad_fft(s, T):
    """Compute gradient of 2n * soft_max(conv(s,s), T) w.r.t. s using FFT."""
    n = len(s)
    conv = fftconvolve(s, s, mode='full')  # or np.convolve
    conv_max = np.max(conv)
    exp_vals = np.exp(T * (conv - conv_max))
    w_sum = np.sum(exp_vals)
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights = exp_vals / w_sum
    
    # crosscorr[j] = sum_m s[m] * weights[j+m]
    # = convolve(s[::-1], weights)[n-1 : 2n-1]
    c = fftconvolve(s[::-1], weights, mode='full')
    crosscorr = c[n - 1 : 2 * n - 1]
    
    return 2.0 * n * crosscorr

def smooth_objective(s, T):
    """Compute the smooth objective value."""
    n = len(s)
    conv = np.convolve(s, s)
    conv_max = np.max(conv)
    return 2.0 * n * (conv_max + np.log(np.sum(np.exp(T * (conv - conv_max)))) / T)

def projected_gradient_descent(s, T_schedule, max_iters, step_size=0.1):
    """Run projected gradient descent with annealing."""
    n = len(s)
    s = project_simplex(s.copy())
    best_s = s.copy()
    best_score = np.convolve(s, s).max() * 2 * n  # with sum=1
    
    for it in range(max_iters):
        T = T_schedule[min(it, len(T_schedule) - 1)]
        grad = compute_grad_fft(s, T)
        
        # Line search
        alpha = step_size
        for _ in range(20):
            s_new = project_simplex(s - alpha * grad)
            if s_new.sum() > 0:
                conv_new = np.convolve(s_new, s_new)
                score_new = 2.0 * n * conv_new.max()
                if score_new < best_score * 1.01:  # allow small increases during annealing
                    s = s_new
                    if score_new < best_score:
                        best_score = score_new
                        best_s = s.copy()
                    break
            alpha *= 0.5
        
        if alpha < 1e-10:
            break
    
    return best_s, best_score

    """Objective and gradient when s = softmax(v)."""
    n = len(v)
    # Compute s = softmax(v)
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    s = exp_v / np.sum(exp_v)
    
    # Compute conv and soft-max
    conv = np.convolve(s, s)
    conv_max = np.max(conv)
    log_sum_exp = conv_max + np.log(np.sum(np.exp(T * (conv - conv_max))))
    obj = 2.0 * n * log_sum_exp / T
    
    # Gradient w.r.t. s
    exp_vals = np.exp(T * (conv - conv_max))
    weights = exp_vals / np.sum(exp_vals)
    c = fftconvolve(s[::-1], weights, mode='full')
    crosscorr = c[n-1 : 2*n-1]
    grad_s = 2.0 * n * crosscorr
    
    # Chain rule: grad_v = s * (grad_s - s * dot(s, grad_s))
    # (This is the Jacobian of softmax transposed times grad_s)
    dot = np.dot(s, grad_s)
    grad_v = s * (grad_s - dot)
    
    return obj, grad_v

    n = len(s)
    conv = fftconvolve(s, s, 'full')
    m = np.max(conv)
    return 2.0 * n * (m + np.log(np.sum(np.exp(T * (conv - m)))) / T)

def true_score(s):
    """Compute the true objective."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

    """Compute softmax(v) numerically stable."""
    v_max = np.max(v)
    e = np.exp(v - v_max)
    return e / np.sum(e)

def obj_and_grad_v(v, T):
    """Objective and gradient w.r.t. v where s = softmax(v)."""
    n = len(v)
    s = softmax_of_v(v)
    
    conv = fftconvolve(s, s, 'full')
    conv_max = np.max(conv)
    exp_vals = np.exp(T * (conv - conv_max))
    w_sum = np.sum(exp_vals)
    
    obj = 2.0 * n * (conv_max + np.log(w_sum) / T)
    
    weights = exp_vals / w_sum
    
    # grad_s = 2n * crosscorr where crosscorr[j] = sum_m s[m]*weights[j+m]
    c = fftconvolve(s[::-1], weights, 'full')
    crosscorr = c[n-1:2*n-1]
    grad_s = 2.0 * n * crosscorr
    
    # Chain rule through softmax: grad_v = s * (grad_s - s * dot(s, grad_s))
    dot = np.dot(s, grad_s)
    grad_v = s * (grad_s - dot)
    
    return obj, grad_v

    """Numerically stable softmax."""
    v_max = np.max(v)
    e = np.exp(v - v_max)
    s = e / np.sum(e)
    return s


def compute_grad_s(s, T):
    """Compute 2n * sum_k softmax(T*conv)[k] * d(conv[k])/ds."""
    n = len(s)
    conv = fftconvolve(s, s, 'full')
    conv_max = np.max(conv)
    exp_vals = np.exp(T * (conv - conv_max))
    w_sum = np.sum(exp_vals)
    weights = exp_vals / w_sum

    crosscorr = c[n-1:2*n-1]
    return 2.0 * n * crosscorr

    # 8. Cosine modulated (to introduce aperiodicity)
    for freq in [2, 3, 5, 7, 11]:
        for amp in [0.3, 0.5, 0.7]:
            s = 1.0 + amp * np.cos(2.0 * np.pi * freq * x)
            s = np.maximum(s, 0.01)
            s = s / s.sum()
            inits.append(s.copy())

    # 9. Random
    for _ in range(15):
        s = rng.random(n) + 0.01
        s = s / s.sum()
        inits.append(s.copy())

    # 10. Random smooth (low-pass filtered random)
    for _ in range(10):
        s_raw = rng.random(n)
        # Simple smoothing
        kernel = np.ones(5) / 5.0
        s = np.convolve(s_raw, kernel, 'same')
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        inits.append(s.copy())

    # 11. Resized previous best
    if previous_s is not None:
        if previous_s.size == n:
            inits.append(previous_s.copy())
        else:
            x_old = np.linspace(0, 1, previous_s.size)
            x_new = np.linspace(0, 1, n)
            try:
                f_interp = interp1d(x_old, previous_s, kind='cubic',
                                    fill_value='extrapolate', bounds_error=False)
                s_resized = np.maximum(f_interp(x_new), 0.0)
                if s_resized.sum() > 0.01:
                    s_resized = s_resized / s_resized.sum()
                    inits.append(s_resized.copy())
            except Exception:
                pass

    # 12. "Near-optimal" shapes: flat top with smooth edges
    for edge_frac in [0.1, 0.15, 0.2, 0.3]:
        k = int(edge_frac * n)
        s = np.ones(n)
        # Smooth edges using a cosine taper
        for i in range(k):
            s[i] = 0.5 * (1.0 - np.cos(np.pi * i / k))
            s[n - 1 - i] = 0.5 * (1.0 - np.cos(np.pi * i / k))
        s = np.maximum(s, 1e-10)
        s = s / s.sum()
        inits.append(s.copy())

    return inits


def proposed_candidate_internal(seed, budget_s):
    """Main search routine."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0

    global_best_score = float('inf')
    global_best_seq = np.ones(512) / 512.0

    # Load previous best
    previous_s = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            previous_s = h / h.sum()
            print(f"Loaded previous best, length={h.size}, score={true_score(previous_s):.6f}")
    except Exception:
        pass

    # T schedules
    T_full = [30, 60, 120, 250, 500, 1000, 2000]
    T_short = [100, 300, 800, 2000]

    # Primary sequence lengths
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")

        inits = generate_initializations(n, rng, previous_s)

        # Score and sort
        scored = []
        for s in inits:
            sc = true_score(s)
            if np.isfinite(sc):
                scored.append((sc, s))
        scored.sort(key=lambda x: x[0])

        if not scored:
            continue
        print(f"  Best init: {scored[0][0]:.6f}, total inits: {len(scored)}")

        # Determine how many runs we can afford
        max_runs = min(8, len(scored))
        if n <= 1024:
            max_runs = min(10, len(scored))

        for idx in range(max_runs):
            if time.monotonic() > deadline - 30:
                break

            sc0, s0 = scored[idx]
            v0 = np.log(np.maximum(s0, 1e-15))
            v0 = v0 - np.max(v0)  # center for numerical stability

            schedule = T_full if idx < 3 else T_short
            mi = 200 if n <= 2048 else 150

            print(f"  Run {idx + 1}/{max_runs}: init={sc0:.6f}", end=" ")
            s_result = run_lbgfs(v0, schedule, deadline - 10, maxiter_per_phase=mi)
            sc_result = true_score(s_result)
            print(f"-> {sc_result:.6f}")

            if sc_result < global_best_score:
                global_best_score = sc_result
                global_best_seq = s_result.copy()
                print(f"  *** New global best: {sc_result:.6f} (n={n})")

        # Additional random restarts
        restarts = 0
        while time.monotonic() < deadline - 40 and restarts < 4:
            s_r = rng.random(n) + 0.01
            s_r = s_r / s_r.sum()
            v0 = np.log(np.maximum(s_r, 1e-15))
            v0 = v0 - np.max(v0)
            s_result = run_lbgfs(v0, T_short, deadline - 10, maxiter_per_phase=100)
            sc_result = true_score(s_result)
            restarts += 1
            if sc_result < global_best_score:
                global_best_score = sc_result
                global_best_seq = s_result.copy()
                print(f"  *** Restart {restarts}: {sc_result:.6f}")

    # Final polish on best result with very high T
    if global_best_seq is not None and time.monotonic() < deadline - 10:
        n = len(global_best_seq)
        v0 = np.log(np.maximum(global_best_seq, 1e-15))
        v0 = v0 - np.max(v0)
        print(f"\nPolishing best (n={n})...")
        s_polished = run_lbgfs(v0, [1000, 2000, 5000, 10000], deadline, maxiter_per_phase=300)
        sc_polished = true_score(s_polished)
        print(f"  Polished score: {sc_polished:.6f}")
        if sc_polished < global_best_score:
            global_best_score = sc_polished
            global_best_seq = s_polished.copy()

    # Verify
    final_score = float(evaluate_sequence(global_best_seq.tolist()))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(global_best_seq)}")

    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid optimization to minimize 2n*max(conv(s,s))/(sum(s))^2.
    Uses L-BFGS-B with softmax parameterization, log-sum-exp annealing,
    diverse initializations, and final polishing.
    """
    return proposed_candidate_internal(seed, budget_s)
