"""
Optimization of non-negative sequences minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses top-k soft-max objective, piecewise-constant global search, multi-resolution
approach, and full-space natural gradient Adam with aggressive annealing.

import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = c - mn
    return c


def topk_softmax(c, k, beta):
    """Compute top-k soft-max: value and gradient weights."""
    c = np.asarray(c, dtype=np.float64)
    n_c = c.size
    if n_c == 0:
        return 0.0, np.array([])
    k = min(k, n_c)
    # Get top-k indices
    if k >= n_c:
        top_idx = np.arange(n_c)
    else:
        # Use argpartition for efficiency
        part_idx = np.argpartition(c, n_c - k)[n_c - k:]
        top_idx = part_idx
    top_vals = c[top_idx]
    mval = float(top_vals.max())
    if beta <= 1e-14:
        w = np.zeros(n_c)
        mm = top_vals >= mval - 1e-12
        if not np.any(mm):
            top_idx[int(np.argmax(top_vals))]  # ensure at least one
            mm = np.zeros_like(top_vals)
            mm[int(np.argmax(top_vals))] = True
        w[top_idx[mm]] = 1.0 / float(np.count_nonzero(mm))
        return mval, w
    x = (top_vals - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(top_vals)
        ex[int(np.argmax(top_vals))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = np.zeros(n_c)
    w[top_idx] = ex / s
    return val, w


def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] for valid indices, using FFT."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):

    z = np.clip(z, -60.0, 60.0)

    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def make_piecewise_profile(n, K, rng):
    """Create a random piecewise-constant profile with K pieces."""
    piece_size = n // K
    remaining = n - piece_size * K
    values = rng.uniform(0.1, 2.0, K)
    a = np.zeros(n)
    pos = 0
    for k in range(K):
        sz = piece_size + (1 if k < remaining else 0)
        a[pos:pos + sz] = values[k]
        pos += sz
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_piecewise(n, K, rng):
    """Piecewise linear (trapezoidal) profile."""
    breakpoints = np.sort(rng.uniform(0, 1, K))
    breakpoints = np.clip(breakpoints, 0.01, 0.99)
    values = rng.uniform(0.2, 2.0, K + 1)
    t = np.linspace(0, 1, n)
    a = np.interp(t, np.concatenate([[0], breakpoints, [1]]),
                  np.concatenate([[values[0]], values, [values[-1]]]))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_symmetric_profile(n, half_profile):
    """Make a symmetric sequence from a half-profile."""
    h = len(half_profile)
    if n % 2 == 0:
        a = np.concatenate([half_profile, half_profile[::-1]])
    else:
        a = np.concatenate([half_profile, half_profile[-2:0:-1]])
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def logit_anneal_topk(p0, deadline, rng, lr=0.10, k_frac=0.05, reg=1e-7):
    """
    Full logit-space Adam with top-k soft-max objective.
    k_frac: fraction of convolution elements to focus on (adaptive).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.15 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    k_min = max(3, int(n * 0.02))
    k_max = 2 * n - 1
    k_current = max(k_min, int(k_frac * (2 * n - 1)))

    while time.monotonic() < deadline:
        L, w = topk_softmax(c, k_current, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.02, lr * 0.96)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Anneal beta and k
        if adam_t % 80 == 0:
            beta = max(1e-12, beta * 0.70)
        if adam_t % 200 == 0:
            k_current = max(k_min, int(k_current * 1.5))

        if no_imp > max(400, 4 * n):
            # Restart with perturbation
            scale = 0.3 + 0.4 * rng.random()
            z = best_z + scale * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.15 * maxc)
            k_current = max(k_min, int(k_frac * (2 * n - 1)))
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def logit_anneal_symmetric(p0, deadline, rng, lr=0.10, reg=1e-7):
    """Optimize only the first half, enforce symmetry."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    half_n = (n + 1) // 2

    # Work in log-space for the first half
    zh = np.log(np.maximum(p0[:half_n], 1e-12))
    zh -= float(np.mean(zh))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(half_n, dtype=np.float64)
    v = np.zeros(half_n, dtype=np.float64)
    adam_t = 0

    def z_to_p_symmetric(zh):
        ph = np.exp(np.clip(zh - zh.max(), -50, 50) + 1e-15)
        if n % 2 == 0:
            p = np.concatenate([ph, ph[::-1]])
        else:
            p = np.concatenate([ph, ph[-2:0:-1]])
        s = float(p.sum())
        return p / s

    p = z_to_p_symmetric(zh)
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.15 * maxc)

    best_p = p.copy()
    best_zh = zh.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        k = max(5, int(0.04 * (2 * n - 1)))
        L, w = topk_softmax(c, k, beta)
        gp = 2.0 * corr_weights(w, p)
        vh_grad = gp[:half_n] + reg * p[:half_n]
        dot = float(np.dot(p[:half_n], vh_grad))
        gz = p[:half_n] * (vh_grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            zh = best_zh.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh2 = v / (1.0 - b2 ** adam_t)
        zh = zh - lr * mh / (np.sqrt(vh2) + eps)

        if not np.all(np.isfinite(zh)):
            zh = best_zh.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        zh = np.clip(zh, -60.0, 60.0)
        p = z_to_p_symmetric(zh)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_zh = zh.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.02, lr * 0.96)
        else:
            zh = best_zh.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 80 == 0:
            beta = max(1e-12, beta * 0.70)

        if no_imp > max(300, 4 * half_n):
            zh = best_zh + 0.4 * rng.normal(size=half_n)
            zh = np.clip(zh, -60.0, 60.0)
            zh -= float(np.mean(zh))
            p = z_to_p_symmetric(zh)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.15 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def piecewise_optimize(n, K, deadline, rng):
    """Optimize a piecewise-constant sequence directly."""
    piece_size = n // K
    remaining = n - piece_size * K
    # Initialize with random values
    values = rng.uniform(0.3, 2.0, K)

    def vals_to_seq(vals):
        a = np.zeros(n)
        pos = 0
        for k in range(K):
            sz = piece_size + (1 if k < remaining else 0)
            a[pos:pos + sz] = vals[k]
            pos += sz
        return a

    a = vals_to_seq(values)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()

    best_values = values.copy()
    best_a = a.copy()
    best_score = internal_score(a)

    lr = 0.05
    no_imp = 0
    t_local = time.monotonic()

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        maxc = float(c.max())
        k = max(5, int(0.05 * (2 * n - 1)))
        _, w = topk_softmax(c, k, max(1e-10, 0.1 * maxc))
        gp = 2.0 * corr_weights(w, a)

        # Gradient with respect to piece values
        grad_vals = np.zeros(K)
        pos = 0
        for k_idx in range(K):
            sz = piece_size + (1 if k_idx < remaining else 0)
            grad_vals[k_idx] = float(np.sum(gp[pos:pos + sz]))
            pos += sz

        # Simple gradient step with adaptive LR
        step = lr * grad_vals
        values = values - step
        values = np.maximum(values, 1e-8)
        a_new = vals_to_seq(values)
        a_new = a_new / a_new.sum()
        score_new = internal_score(a_new)

        if np.isfinite(score_new) and score_new < best_score - 1e-12:
            best_score = score_new
            best_values = values.copy()
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                lr = max(0.01, lr * 0.95)
            # Try a small perturbation
            values = best_values + lr * 0.5 * rng.normal(size=K)
            values = np.maximum(values, 1e-8)
            a = vals_to_seq(values)
            a = a / a.sum()

        if no_imp > 500:
            # Large restart
            values = best_values + 0.5 * rng.normal(size=K)
            values = np.maximum(values, 1e-8)
            a = vals_to_seq(values)
            a = a / a.sum()
            lr = 0.05
            no_imp = 0

    return best_a, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def make_basis_mixed(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u]
    m = 1
    while len(cols) < K:
        ang = 2.0 * np.pi * m * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
        m += 1
        if len(cols) > K:
            break
    return np.column_stack(cols[:K])


def profile_from_basis(theta, basis):
    z = basis @ np.asarray(theta, dtype=np.float64)
    return softmax_from_z(z)


def param_adam_topk(theta0, basis, deadline, rng, lr=0.25, k_frac=0.05, reg=1e-6):
    """Adam in low-dim param space with top-k soft-max."""
    basis = np.asarray(basis, dtype=np.float64)
    theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
    dim = theta.size
    n = basis.shape[0]

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(dim, dtype=np.float64)
    v = np.zeros(dim, dtype=np.float64)
    adam_t = 0

    p = profile_from_basis(theta, basis)
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.15 * maxc)

    best_theta = theta.copy()
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    k_min = max(3, n // 20)
    k_current = max(k_min, int(k_frac * (2 * n - 1)))

    while time.monotonic() < deadline:
        L, w = topk_softmax(c, k_current, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gtheta = basis.T @ gz

        if not np.all(np.isfinite(gtheta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gtheta
        v = b2 * v + (1.0 - b2) * (gtheta * gtheta)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        theta = theta - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(theta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        theta = np.clip(theta, -8.0, 8.0)
        p = profile_from_basis(theta, basis)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_theta = theta.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.05, lr * 0.96)
        else:
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 80 == 0:
            beta = max(1e-12, beta * 0.70)
        if adam_t % 150 == 0:
            k_current = max(k_min, int(k_current * 1.4))

        if no_imp > max(200, 5 * dim):
            theta = best_theta + 0.5 * rng.normal(size=dim)
            theta = np.clip(theta, -8.0, 8.0)
            p = profile_from_basis(theta, basis)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.15 * maxc)
            k_current = max(k_min, int(k_frac * (2 * n - 1)))
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_theta, best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Strategies:
      A. Low-dim parametric global search (top-k soft-max)
      B. Piecewise-constant optimization
      C. Full-space logit Adam (top-k)
      D. Symmetric optimization
      E. Multi-resolution upscale
      F. Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    def local_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
        if p is not None and np.isfinite(sc):
            if len(candidates) < 24:
                candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
                candidates.sort(key=lambda x: x[0])
            else:
                worst = max(range(len(candidates)), key=lambda i: candidates[i][0])
                if float(sc) < candidates[worst][0]:
                    candidates[worst] = (float(sc), np.asarray(p, dtype=np.float64).copy())
                    candidates.sort(key=lambda x: x[0])

    # --- Initial candidates ---
    n_main = 512

    # Try getting parent sequence
    parent_arr = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pn = normalize_nonneg(parent)
            if pn is not None:
                parent_arr = pn
    except Exception:
        pass

    if parent_arr is not None:
        if parent_arr.size == n_main:
            sc = internal_score(parent_arr)
            local_update(parent_arr, sc)
        else:
            p_interp = interpolate_profile(parent_arr, n_main)
            sc = internal_score(p_interp)
            local_update(p_interp, sc)
            if parent_arr.size <= 2048:
                sc_parent = internal_score(parent_arr)
                local_update(parent_arr, sc_parent)

    # Deterministic starts
    for eta in (1.0, 1.5, 2.0, 2.5):
        p = make_linear_profile(n_main, eta)
        sc = internal_score(p)
        local_update(p, sc)

    for a1, a3 in ((1.5, -1.0), (2.0, -2.0), (1.2, -0.5), (2.5, -3.0)):
        p = make_cubic_profile(n_main, a1, a3)
        sc = internal_score(p)
        local_update(p, sc)

    print(f"initial best={best_score:.10f} n={n_main}")

    # --- Phase A: Low-dim parametric search (300s) ---
    phase_a_end = min(deadline, t_start + 300.0)
    bm = make_basis_mixed(n_main, 15)
    n_bm = bm.shape[1]

    theta_starts = []
    th0 = np.zeros(n_bm)
    th0[0] = 1.5
    theta_starts.append(th0)
    th1 = np.zeros(n_bm)
    th1[0] = 2.0
    if n_bm > 3:
        th1[2] = -1.5
    theta_starts.append(th1)
    for _ in range(20):
        theta_starts.append(rng.uniform(-3.0, 3.0, n_bm))

    for th in theta_starts:
        if time.monotonic() > phase_a_end:
            break
        per = min(phase_a_end, time.monotonic() + 8.0)
        _, p_best, sc = param_adam_topk(th, bm, per, rng, lr=0.25)
        local_update(p_best, sc)

    print(f"after phase A best={best_score:.10f}")

    # --- Phase B: Piecewise-constant optimization (150s) ---
    phase_b_end = min(deadline, t_start + 450.0)
    for K in (8, 16, 32):
        if time.monotonic() > phase_b_end:
            break
        for trial in range(4):
            if time.monotonic() > phase_b_end:
                break
            per = min(phase_b_end, time.monotonic() + 12.0)
            p_pq, sc_pq = piecewise_optimize(n_main, K, per, rng)
            local_update(p_pq, sc_pq)

    print(f"after phase B best={best_score:.10f}")

    # --- Phase C: Full-space logit Adam on top candidates (700s) ---
    phase_c_end = min(deadline, t_start + 1150.0)
    top_candidates = candidates[:12]
    for i, (sc_c, p_c) in enumerate(top_candidates):
        if time.monotonic() > phase_c_end:
            break
        p0 = p_c
        if p0.size != n_main:
            p0 = interpolate_profile(p0, n_main)
        p0 = normalize_nonneg(p0)
        if p0 is None:
            continue
        per = min(phase_c_end, time.monotonic() + 45.0)
        p_best, sc = logit_anneal_topk(p0, per, rng, lr=0.09, k_frac=0.04)
        local_update(p_best, sc)
        if i % 4 == 0:
            print(f"  phase C [{i+1}/{len(top_candidates)}] best={best_score:.10f}")

    print(f"after phase C best={best_score:.10f}")

    # --- Phase D: Symmetric optimization (200s) ---
    phase_d_end = min(deadline, t_start + 1350.0)
    for i in range(min(6, len(candidates))):
        if time.monotonic() > phase_d_end:
            break
        p0 = candidates[i][1]
        if p0.size != n_main:
            p0 = interpolate_profile(p0, n_main)
        p0 = normalize_nonneg(p0)
        if p0 is None:
            continue
        per = min(phase_d_end, time.monotonic() + 30.0)
        p_best, sc = logit_anneal_symmetric(p0, per, rng, lr=0.09)
        local_update(p_best, sc)

    print(f"after phase D best={best_score:.10f}")

    # --- Phase E: Upscale to n=1024 and optimize (250s) ---
    phase_e_end = min(deadline, t_start + 1600.0)
    n_long = 1024
    if best_seq is not None:
        p_long = interpolate_profile(best_seq, n_long)
        p_long = normalize_nonneg(p_long)
        if p_long is not None:
            per = min(phase_e_end, time.monotonic() + 60.0)
            p_best, sc = logit_anneal_topk(p_long, per, rng, lr=0.06, k_frac=0.03)
            local_update(p_best, sc)
            print(f"  n=1024 best={best_score:.10f}")

    # Also try a good deterministic profile at n=1024
    for eta in (1.5, 2.0):
        if time.monotonic() > phase_e_end:
            break
        p_long = make_linear_profile(n_long, eta)
        per = min(phase_e_end, time.monotonic() + 40.0)
        p_best, sc = logit_anneal_topk(p_long, per, rng, lr=0.06, k_frac=0.03)
        local_update(p_best, sc)

    # --- Phase F: Final polish on best candidates (200s) ---
    phase_f_end = deadline
    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    for i in range(min(5, len(candidates))):
        if time.monotonic() > phase_f_end:
            break
        p0 = candidates[i][1]
        n_p = p0.size
        if n_p > 2048:
            p0 = interpolate_profile(p0, n_main)
            n_p = n_main
        p0 = normalize_nonneg(p0)
        if p0 is None:
            continue
        per = min(phase_f_end, time.monotonic() + 30.0)
        p_best, sc = logit_anneal_topk(p0, per, rng, lr=0.04, k_frac=0.02)
        local_update(p_best, sc)

    print(f"after phase F best={best_score:.10f}")

    # --- Final exact evaluation and return ---
    if best_seq is not None:
        final_seq = best_seq
        final_score = best_score

        # Exact score for sequences up to length 2048
        if final_seq.size <= 2048:
            sc = true_score(final_seq)
            if np.isfinite(sc) and sc < final_score:
                final_score = sc

        # Check all top candidates exactly
        for sc_c, p_c in candidates[:8]:
            if p_c.size <= 2048:
                sc_c_exact = true_score(p_c)
                if np.isfinite(sc_c_exact) and sc_c_exact < final_score:
                    final_score = sc_c_exact
                    final_seq = p_c

        # Validate with evaluate_sequence if length allows
        if final_seq.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval) and sc_eval < final_score:
                    final_score = sc_eval
            except Exception:
                pass

        final_seq = normalize_nonneg(final_seq)
        if final_seq is None:
            final_seq = make_linear_profile(256, 1.5)
            final_score = internal_score(final_seq)

        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s

        print(f"FINAL best={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = make_linear_profile(512, 1.5)
    print(f"FALLBACK used, score={internal_score(fallback):.6f}")
    return [float(x) for x in fallback]
