However, longer sequences are more expensive to optimize. I'll focus on n around 256-1024.

Let me write the code:

```python
"""
Search for non-negative step-function sequences minimizing
f(seq) = 2*n*max(convolve(seq, seq)) / sum(seq)^2

Using:
1. Adaptive softmax-gradient descent with variance penalty
2. Coordinate descent polishing
3. LP cutting-plane refinement
4. Multi-length, multi-initialization strategy
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 1e-300, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def actual_score_fft(p, direct_limit=1024):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-6:
        return np.inf
    if n <= direct_limit:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.inf
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]


def compute_conv_fft(a, L=None):
    """Compute full convolution via FFT."""
    n = a.size
    if L is None:
        L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def softmax_gradient_step(p, alpha, var_weight=0.0, L=None):
    """
    Compute gradient of the smoothed objective:
    J = alpha * log(sum(exp(c/alpha))) + var_weight * var(c)
    where c = convolve(p, p) and p sums to 1.
    Returns the gradient projected onto the tangent space of the simplex.
    """
    n = p.size
    if L is None:
        L = next_power_of_two(2 * n - 1)
    
    c = compute_conv_fft(p, L)
    maxc = c.max()
    
    # Softmax weights
    z = (c - maxc) / alpha
    expz = np.exp(z)
    sz = expz.sum()
    if sz <= 0 or not np.isfinite(sz):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = expz / sz
    
    # Gradient of log-sum-exp part: d/dp_k sum_m w_m * c_m
    # c_m = sum_j p_j * p_{m-j}
    # dc_m/dp_k = p_{m-k} + p_{k-m} (with appropriate bounds)
    # = 2 * p_{m-k} if m != 2k, = p_{m-k} if m == 2k
    # So d(sum_m w_m * c_m)/dp_k = sum_m w_m * (p_{m-k} + p_{k-m})
    # = 2 * sum_m w_m * p_{m-k} (counting both m-k and k-m)
    # But we need to be careful: c_m = sum_j p_j * p_{m-j}
    # dc_m/dp_k = p_{m-k} (the term where j=k) + p_{m-k} (the term where m-j=k, i.e., j=m-k)
    # If m-j = k, then j = m-k. So the two terms are p_{m-k} and p_{m-k}, giving 2*p_{m-k}.
    # But if m-k = k (i.e., m = 2k), then it's the same term, so dc_m/dp_k = 2*p_k (since p_{m-k} = p_k).
    # Wait no: c_m = sum_j p_j * p_{m-j}. The term with j=k is p_k * p_{m-k}. The term with j=m-k is p_{m-k} * p_k.
    # These are the same term! So dc_m/dp_k = p_{m-k} (from j=k) + p_{m-k} (from j=m-k) only if k != m-k.
    # If k = m-k (m=2k), then there's only one term: p_k * p_k, and dc_m/dp_k = 2*p_k.
    # Actually no: c_m = sum_j p_j * p_{m-j}. If we differentiate w.r.t. p_k:
    # dc_m/dp_k = sum_j [delta_{jk} * p_{m-j} + p_j * delta_{m-j,k}]
    # = p_{m-k} + p_{m-k} (if both k and m-k are in [0,n-1])
    # = 2 * p_{m-k} (if k != m-k and both in range)
    # = 2 * p_k (if m = 2k, since p_{m-k} = p_k)
    
    # So d(sum_m w_m * c_m)/dp_k = 2 * sum_m w_m * p_{m-k} (for m-k in [0,n-1])
    # which is 2 * (w * p)[k] (circular... no, linear convolution)
    # = 2 * (w_rev * p)[k] where w_rev[m] = w[2n-2-m]
    
    # Let's compute: grad_p[k] = 2 * sum_{m: m-k in [0,n-1]} w[m] * p[m-k]
    # Let j = m-k, so m = j+k, j in [0,n-1]:
    # = 2 * sum_{j=0}^{n-1} w[j+k] * p[j]
    # This is the cross-correlation of w and p shifted by k.
    # Equivalently: grad_p = 2 * (w_full * p) where * is linear convolution, taking the "middle" part.
    
    # More precisely, if we define w_full of length 2n-1, then:
    # grad_p[k] = 2 * sum_j w_full[j+k] * p[j] for j = 0..n-1
    # This is the (2n-2-k)-th element of the convolution of w_full and p? No...
    # Let's just compute it directly:
    # grad_p = 2 * correlate(w_full, p) where correlate gives the linear cross-correlation
    
    # Actually, the cleanest way: 
    # grad_p[k] = 2 * (w_full reversed) convolved with p, evaluated at position... 
    
    # Let me just use FFT:
    # Define w_ext of length L: w_ext[:2n-1] = w, w_ext[2n-1:] = 0
    # p_ext of length L: p_ext[:n] = p, p_ext[n:] = 0
    # Then the "gradient" in the sense of the convolution:
    # grad_p[k] = 2 * sum_j w[j+k] * p[j]
    # = 2 * (IFFT(conj(FFT(w_ext)) * FFT(p_ext))[k]) ... no this is cross-correlation
    
    # Simpler: grad_p[k] = 2 * (w_rev * p)[k] where w_rev[m] = w[2n-2-m] and * is linear convolution
    # Because (w_rev * p)[k] = sum_j w_rev[k-j] * p[j] = sum_j w[2n-2-k+j] * p[j]
    # That doesn't seem right either.
    
    # Let me just do it the straightforward way:
    # grad_p[k] = 2 * sum_{j=0}^{n-1} p[j] * w[j+k]
    
    # This is the cross-correlation of p and w, shifted. Specifically:
    # (p corr w)[k] = sum_j p[j] * w[j+k] (for linear cross-correlation)
    # So grad_p = 2 * (p corr w)
    
    # In FFT terms: cross-correlation = IFFT(conj(FFT(p)) * FFT(w))
    # But we need to be careful about the length.
    
    # Let's compute w_rev of length 2n-1: w_rev[m] = w[2n-2-m]
    # Then (w_rev * p)[k] = sum_j w_rev[k-j] * p[j] = sum_j w[2n-2-k+j] * p[j]
    # For this to equal sum_j p[j] * w[j+k], we need 2n-2-k+j = j+k, i.e., 2n-2-k = k, i.e., k = n-1.
    # That's not right.
    
    # OK let me just compute it directly using the full convolution:
    # Define W of length 2n-1 (the convolution weights)
    # grad_p[k] = 2 * sum_{j=0}^{n-1} p[j] * W[j+k]
    # where W = w (the softmax weights over the convolution)
    
    # This is the k-th element of the "anti-convolution" of p and W.
    # Equivalently: grad_p = 2 * IFFT(FFT(p_ext, L) * FFT(W_rev_ext, L))[:n]
    # where W_rev[m] = W[2n-2-m]
    
    W = w  # length 2n-1
    W_rev = W[::-1]  # reverse
    Wp = np.fft.rfft(W_rev, L)
    Pp = np.fft.rfft(p, L)
    grad_raw = np.fft.irfft(Pp * Wp, L)[:n].real
    grad_p = 2.0 * grad_raw
    
    # Add variance penalty gradient
    if var_weight > 0:
        mean_c = c.mean()
        # d(var(c))/dp_k = 2 * sum_m (c_m - mean_c) * dc_m/dp_k / (2n-1)
        # = 2 / (2n-1) * sum_m (c_m - mean_c) * 2 * p_{m-k} (for m-k in range)
        # = 4/(2n-1) * (p corr (c - mean_c))[k]
        c_centered = c - mean_c
        c_rev = c_centered[::-1]
        cp = np.fft.rfft(c_rev, L)
        grad_var = np.fft.irfft(Pp * cp, L)[:n].real
        grad_p += var_weight * (4.0 / (2 * n - 1)) * grad_var
    
    # Project to tangent space of simplex (sum of gradient = 0)
    gp = float(np.dot(p, grad_p))
    grad_x = p * (grad_p - gp)
    
    return grad_x, maxc


def adam_optimize_v2(p, budget_s, seed=0, var_weight=0.1, alpha_start=30.0, alpha_end=3.0):
    """
    Improved Adam optimizer with variance penalty and adaptive temperature.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 0.5:
        f = actual_score_fft(p)
        return p, f

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = 0.04
    elif n < 256:
        lr = 0.025
    elif n < 1024:
        lr = 0.015
    else:
        lr = 0.01

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(2 * n - 1)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        
        # Adaptive temperature: start hot, cool down
        alpha = alpha_start + (alpha_end - alpha_start) * frac**1.5
        
        # Variance weight: increase as we cool down
        vw = var_weight * (1.0 - frac * 0.5)
        
        grad_x, maxc = softmax_gradient_step(p, alpha, vw, L)
        
        if not np.all(np.isfinite(grad_x)):
            break
            
        f_now = float(2.0 * n * maxc)  # p sums to one
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        
        # Adaptive learning rate: reduce as we approach convergence
        lr_t = lr * (1.0 - 0.5 * frac)
        step = lr_t * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -50.0, 50.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -80.0, 80.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def coordinate_descent(p, max_sweeps=50, tol=1e-8):
    """
    Coordinate descent: for each k, find optimal a_k given others fixed.
    Uses the structure that conv[m] is at most quadratic in a_k.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n < 2:
        return p
    p = normalize_nonneg(p)
    
    L = next_power_of_two(2 * n - 1)
    best_score = actual_score_fft(p)
    
    for sweep in range(max_sweeps):
        improved = False
        for k in range(n):
            # Save current value
            a_k = p[k]
            if a_k < 1e-15:
                # Try a small positive value
                p[k] = 1e-6
                c = compute_conv_fft(p, L)
                sc = float(c.max())
                if sc < best_score:
                    best_score = sc
                    improved = True
                else:
                    p[k] = a_k
                continue
            
            # Compute conv with a_k = 0
            p[k] = 0.0
            c_base = compute_conv_fft(p, L)
            
            # Now conv[m] with a_k = val:
            # c[m] = c_base[m] + val * (p[m-k] + p[k-m]) for m != 2k
            # c[2k] = c_base[2k] + val^2
            # where p[i] = 0 if i out of [0, n-1]
            
            # Build the "weight" vector w[m] = p[m-k] + p[k-m] (with bounds)
            w = np.zeros(2 * n - 1, dtype=np.float64)
            # p[m-k] for m-k in [0, n-1]: m in [k, k+n-1]
            lo = max(0, k)
            hi = min(2 * n - 2, k + n - 1)
            if lo <= hi:
                w[lo:hi+1] += p[lo-k:hi-k+1]
            # p[k-m] for k-m in [0, n-1]: m in [k-(n-1), k]
            lo2 = max(0, k - (n - 1))
            hi2 = min(2 * n - 2, k)
            if lo2 <= hi2:
                w[lo2:hi2+1] += p[k-hi2:k-lo2+1][::-1]
            
            # Special: at m = 2k, the contribution is val^2, not 2*val*p[k]
            # So w[2k] should be p[k] + p[k] = 2*p[k] from the linear part,
            # but actually the quadratic part is val^2.
            # Let me redo: c[m] = c_base[m] + val * w[m] + (val^2 if m == 2k else 0)
            # where w[m] = p[m-k] + p[k-m] (double counting at m=2k gives 2*p[k])
            # But at m=2k, the actual contribution is val^2, not 2*val*p[k].
            # So: c[2k] = c_base[2k] + val^2
            #     c[m] = c_base[m] + val * w[m] for m != 2k
            
            # To find optimal val >= 0:
            # minimize max( c_base[2k] + val^2, max_{m != 2k} (c_base[m] + val * w[m]) )
            
            # Simple approach: try a few candidate values
            candidates = [a_k]  # current value
            
            # Value that makes the quadratic part equal to the current max
            cur_max = float(c_base.max())
            if c_base[2*n-2-2*k if 2*k < 2*n-1 else 2*n-2] > 0:
                idx_2k = min(2 * k, 2 * n - 2)
                if c_base[idx_2k] + a_k * a_k > 0:
                    diff = c_base[idx_2k]  # this is c[2k] with val=0
                    # val^2 + diff = cur_max => val = sqrt(cur_max - diff)
                    if cur_max > diff:
                        candidates.append(np.sqrt(cur_max - diff))
            
            # Value that balances top linear contributors
            top_m = int(np.argmax(c_base + a_k * w))  # approximate
            for m_try in [top_m, int(np.argmax(c_base))]:
                if m_try != 2 * k and 0 <= m_try < 2 * n - 1:
                    # c_base[m] + val * w[m] = c_base[2k] + val^2
                    # val^2 - w[m]*val + (c_base[2k] - c_base[m]) = 0
                    if w[m_try] > 1e-15:
                        idx_2k = min(2 * k, 2 * n - 2)
                        b_coef = w[m_try]
                        c_coef = c_base[idx_2k] - c_base[m_try]
                        disc = b_coef * b_coef - 4 * c_coef
                        if disc >= 0:
                            v1 = (b_coef - np.sqrt(disc)) / 2.0
                            if v1 >= 0:
                                candidates.append(v1)
                            v2 = (b_coef + np.sqrt(disc)) / 2.0
                            if v2 >= 0:
                                candidates.append(v2)
            
            # Also try 0 and a_k * 0.5, a_k * 1.5
            candidates.extend([0.0, a_k * 0.5, a_k * 1.5])
            
            # Evaluate each candidate
            best_val = a_k
            best_sc = float(np.max(c_base + best_val * w))
            # Handle the quadratic at 2k
            idx_2k = min(2 * k, 2 * n - 2)
            best_sc = max(best_sc, c_base[idx_2k] + best_val * best_val)
            
            for val in candidates:
                val = max(0.0, float(val))
                sc_lin = float(np.max(c_base + val * w))
                sc_quad = c_base[idx_2k] + val * val
                sc = max(sc_lin, sc_quad)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_val = val
            
            p[k] = best_val
            if abs(best_val - a_k) > tol * max(1.0, a_k):
                improved = True
        
        p = normalize_nonneg(p)
        cur_score = float(compute_conv_fft(p, L).max())
        if cur_score < best_score:
            best_score = cur_score
        elif not improved:
            break
    
    return p


def lp_polish(p, budget_s=60.0):
    """LP-based polishing using cutting planes."""
    if _scipy_opt is None:
        return p, actual_score_fft(p)
    
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n < 2 or budget_s < 10:
        return p, actual_score_fft(p)
    
    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score_fft(f)
    
    best_f = f.copy()
    best_score = float(cur_score)
    
    end = time.monotonic() + budget_s - 3.0
    
    # Initial constraint set
    n_rows = min(200, 2 * n - 1)
    rows = list(np.unique(np.linspace(0, 2 * n - 2, n_rows, dtype=int)).astype(int))
    row_set = set(rows)
    max_rows = min(500, 2 * n - 1)
    
    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        
        # Build constraint matrix A where A[i, j] = f[rows[i] - j] if valid
        A = np.zeros((rows_arr.size, n), dtype=np.float64)
        for ii in range(rows_arr.size):
            k = int(rows_arr[ii])
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A[ii, j] = f[idx]
        
        b = np.full(rows_arr.size, cur_score * 1.001, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)
        
        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(3.0, min(30.0, rem * 0.3))
        
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": float(tl)}
            )
        except Exception:
            break
        
        if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
            break
        
        g = np.clip(np.asarray(res.x, dtype=np.float64), 0.0, None)
        sg = float(g.sum())
        if sg <= 1e-12:
            break
        
        # Check violations
        conv = compute_conv_fft(f * (g / sg) * S0)  # normalize g to sum S0
        # Actually: f*g where both sum to S0
        L2 = next_power_of_two(4 * n - 2)
        Pf = np.fft.rfft(f, L2)
        Pg = np.fft.rfft(g * S0 / sg, L2)
        conv = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        
        tol = 1e-6 + 1e-6 * cur_score
        violations = conv - cur_score
        
        if np.any(violations > tol):
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    if added >= min(100, max_rows - len(rows)):
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            continue
        
        # Accept direction g
        g = g * (S0 / sg)
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.5, 0.25, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()
        
        if accepted is None or accepted_score >= best_score:
            break
        
        f = accepted
        cur_score = accepted_score
        best_f = f.copy()
        best_score = cur_score
    
    return normalize_nonneg(best_f), best_score


def try_global_eval(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
    except Exception:
        return np.inf
    try:
        val = float(val)
    except Exception:
        return np.inf
    return val if np.isfinite(val) else np.inf


def random_perturb(p, rng, sigma):
    p = np.asarray(p, dtype=np.float64).ravel()
    noise = np.exp(rng.normal(0.0, float(sigma), p.size))
    return normalize_nonneg(p * noise)
