    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_direction_lp(a):

    Solve: min t s.t. conv(a,b) <= t, sum(b) = 1, b >= 0, t >= 0.
    Returns (b, t) or (None, None) on failure.

    A = build_conv_matrix(a)  # shape (2n-1, n)
    
    # Variables: [b_0, ..., b_{n-1}, t]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)

    # [A, -1_col] @ [b, t] <= 0

    b_ub = np.zeros(2*n-1)

    A_eq[0, :n] = 1.0

        t = res.x[n]

"""
LP-based alternating optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a direct LP formulation (min peak of conv(a,b) subject to sum(b)=1) 
with line search, sparsification, and multiple initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_direction_lp(a):
    """
    Solve: min t s.t. conv(a,b) <= t, sum(b)=1, b>=0, t>=0.
    Returns optimal b or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.empty((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None


def solve_direction_lp_max(a, M):
    """
    Solve: max sum(b) s.t. conv(a,b) <= M, b>=0.
    Returns optimal b or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a, deadline, rng, verbose=True, method='min_t'):
    """
    LP-based optimization. Alternates between finding best response and stepping.
    method='min_t': LP minimizes t (the peak) directly.
    method='max_sum': LP maximizes sum(g) subject to conv(a,g)<=M.
    """
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    stall = 0
    it = 0
    step_size = 1.0

    while time.monotonic() < deadline:
        it += 1

        if method == 'min_t':
            b = solve_direction_lp(a)
        else:
            c = np.convolve(a, a, mode='full')
            M = c.max()
            g = solve_direction_lp_max(a, M)
            if g is not None and g.sum() > 1e-15:
                b = g / g.sum()
            else:
                b = None

        if b is None:
            break
        b = np.clip(b, 0.0, None)
        if b.sum() < 1e-10:
            break
        b = b / b.sum()

        improved = False
        # Adaptive step sizes
        alphas = [step_size, 0.7, 0.5, 0.3, 0.1, 0.05]
        for alpha in alphas:
            if alpha > 1.0:
                alpha = 1.0
            a_new = (1.0 - alpha) * best_a + alpha * b
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step_size = min(1.0, step_size * 1.05)
        else:
            stall += 1
            step_size = max(0.02, step_size * 0.8)
            if stall % 30 == 0 and stall > 30:
                # Random perturbation to escape local minimum
                perturbation = rng.random(n)
                mix = rng.uniform(0.1, 0.4)
                a_pert = (1 - mix) * best_a + mix * perturbation / perturbation.sum()
                a_pert = np.clip(a_pert, 0.0, None)
                if a_pert.sum() > 1e-10:
                    a_pert = a_pert / a_pert.sum()
                    sc = compute_score(a_pert)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_pert.copy()
                        a = a_pert
                        stall = 0
                    else:
                        a = a_pert  # Move anyway to explore
            if stall > 200:
                break

        if verbose and it % 2000 == 0:
            print(f"      iter {it}, score={best_score:.7f}, step={step_size:.3f}, stall={stall}")

    return best_a, best_score


def sparsify(a, max_rounds=50):
    """Try removing low-contribution elements to improve score."""
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for _ in range(max_rounds):
        # Compute contribution of each position to the peak
        c = np.convolve(best_a, best_a, mode='full')
        peak_idx = np.argmax(c)
        peak_val = c.max()

        # For each position i, compute how much it contributes to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = best_a[i] * best_a[j]
            else:
                contrib[i] = 0.0

        # Try removing the position with lowest value (among non-peak contributors)
        # Only consider positions where value is small relative to max
        threshold = 0.05 * best_a.max()
        candidates = np.where((best_a < threshold) & (best_a > 0))[0]

        if len(candidates) == 0:
            break

        best_improvement = None
        best_new_score = best_score

        for i in candidates:
            a_trial = best_a.copy()
            a_trial[i] = 0.0
            if a_trial.sum() < 1e-10:
                continue
            sc = compute_score(a_trial)
            if sc < best_new_score:
                best_new_score = sc
                best_improvement = a_trial.copy()

        if best_improvement is not None:
            best_a = best_improvement
            best_score = best_new_score
        else:
            break

    # Trim trailing/leading zeros
    nonzero = np.where(best_a > 1e-15)[0]
    if len(nonzero) > 0:
        best_a = best_a[nonzero[0]:nonzero[-1] + 1]
        best_score = compute_score(best_a)

    return best_a, best_score


def coordinate_descent_polish(a, deadline, max_sweeps=200):
    """
    Coordinate descent: for each position, find the value that minimizes the peak.
    """
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            # Compute base conv without a[i]
            a_temp = a.copy()
            orig = a_temp[i]
            a_temp[i] = 0.0
            base = np.convolve(a_temp, a_temp, mode='full')

            # Find optimal x for position i
            # c[k] = base[k] + 2*x*a_temp[k-i] (for valid k-i, k!=2i)
            # c[2i] += x^2
            # Use golden section search on x in [0, 2*orig+0.1]
            x_lo = 0.0
            x_hi = 2.0 * max(orig, 0.01) + 0.1

            def peak_at_x(x):
                # Compute peak of convolution with a[i] = x
                # c[k] = base[k] + contribution from x
                peak = base.max()
                # For k where 0 <= k-i < n:
                j_lo = max(0, i)
                j_hi = min(2 * n - 2, i + n - 1)
                if j_lo <= j_hi:
                    # c[k] for k in [i, i+n-1] gets 2*x*a_temp[k-i] added
                    # (or x^2 when k=2i)
                    ks = np.arange(j_lo, j_hi + 1)
                    js = ks - i
                    contrib = 2.0 * x * a_temp[js]
                    # Fix the k=2i case: should be x^2 not 2*x*a_temp[i]
                    if 2 * i >= j_lo and 2 * i <= j_hi:
                        idx = 2 * i - j_lo
                        contrib[idx] = x * x  # replace
                    vals = base[ks] + contrib
                    peak = max(peak, vals.max())
                return peak

            # Golden section
            gr = (np.sqrt(5) - 1) / 2.0
            lo, hi = x_lo, x_hi
            for _ in range(30):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                f1 = peak_at_x(m1)
                f2 = peak_at_x(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_opt = (lo + hi) / 2.0

            if abs(x_opt - orig) > 1e-10:
                a[i] = x_opt
                improved = True

        # Normalize and check
        a = np.clip(a, 0.0, None)
        if a.sum() < 1e-10:
            break
        # Don't renormalize during CD (the optimal x already accounts for the scale)
        # Actually we should normalize to keep sum=1
        a = a / a.sum()
        sc = compute_score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
        elif not improved and sweep > 5:
            break

    return best_a, best_score


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.clip(1.0 - np.abs(np.arange(n) - c) / (c + 1.0), 0.0, None)


def make_ramp(n):
    """Linearly increasing sequence."""
    return np.linspace(0.1, 1.0, n)


def make_two_peak(n, gap_frac=0.3):
    """Two Gaussian peaks."""
    c1 = (1 - gap_frac) * (n - 1) / 2.0
    c2 = (1 + gap_frac) * (n - 1) / 2.0
    sigma = 0.1 * n
    return np.exp(-0.5 * ((np.arange(n) - c1) / sigma) ** 2) + \
           np.exp(-0.5 * ((np.arange(n) - c2) / sigma) ** 2)


def make_exponential(n, rate=0.5):
    """Exponentially decaying sequence."""
    return np.exp(-rate * np.arange(n) / n)

def _resize_parent(p, n):
    """Resize parent sequence to length n using interpolation."""
    if len(p) == n:
        return p.copy()
    x_old = np.linspace(0, 1, len(p))
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, p)
