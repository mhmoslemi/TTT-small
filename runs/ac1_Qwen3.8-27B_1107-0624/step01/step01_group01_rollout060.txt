```python
    a = a0.copy()
    # Nesterov momentum
    y = a.copy()
    beta = 0.9
    
    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Compute convolution of y
        c = convolve(y, y)
        # Find peak
        k_star = argmax(c)
        t_val = c[k_star]
        
        # Subgradient of max_k c[k] at y
        # grad_j = y[j - k_star] + y[j + k_star] (with boundary = 0)
        g = np.zeros(n)
        for j in range(n):
            g[j] = 0
            if j - k_star >= 0:
                g[j] += y[j - k_star]
            if j + k_star < n:
                g[j] += y[j + k_star]
        # Subtract mean to project onto tangent space of simplex
        g -= np.mean(g)
        
        # Step size
        step = 1.0 / (t + 1)
        
        # Nesterov update
        a_new = y - step * g
        a_new = project_simplex(a_new)
        beta_new = (1 + sqrt(1 + 4*beta*beta)) / 2
        y = a_new + (beta/beta_new - 1) * (a_new - a)
        a = a_new
        beta = beta_new
    
    return a
```
