```python
"""
Smooth max with temperature annealing for minimizing auto-convolution peak.
Uses FFT-based gradient computation for speed, multiplicative updates for
positivity, and multi-start strategy for robustness.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len

def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return float(2.0 * n * c.max() / (S * S))

def smooth_max_and_grad(a, T):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient_array).
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    c_max = c.max()
    c_shifted = c - c_max
    exp_vals = np.exp(c_shifted / T)
    exp_sum = exp_vals.sum()
    p = exp_vals / exp_sum
    M_S = c_max + T * np.log(exp_sum)
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j]
    N = next_fast_len(3 * n)
    fft_a = fft(a, N)
    fft_p = fft(p, N)
    h = ifft(fft_a * np.conj(fft_p)).real[:n]
    grad = 2.0 * h
    
    return M_S, grad

def optimize_smooth_max(a, deadline, rng, verbose=False):
    """Optimize using smooth max with temperature annealing."""
    n = len(a)
    a = np.clip(a, 1e-10, None).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Initial convolution for temperature reference
    c0 = np.convolve(a, a, mode='full')
    c0_max = c0.max()
    
    # Temperature schedule (geometric)
    T_start = c0_max * 0.3
    T_end = c0_max * 0.005
    n_temps = 30
    temps = np.geomspace(T_start, T_end, n_temps)
    
    # Learning rate schedule
    lr = 0.3
    lr_decay = 0.97
    iters_per_temp = 30
    
    total_iters = 0
    for T in temps:
        if time.monotonic() > deadline:
            break
        
        for it in range(iters_per_temp):
            if time.monotonic() > deadline:
                break
            
            M_S, grad = smooth_max_and_grad(a, T)
            
            # Full gradient (with S=1 normalization, which we maintain)
            S = a.sum()
            full_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M_S / S)
            
            # Multiplicative update with adaptive scaling
            scale = np.abs(full_grad).max() + 1e-10
            a_new = a * np.exp(-lr * full_grad / scale)
            a_new = np.clip(a_new, 1e-15, None)
            a_new = a_new / a_new.sum()
            
            total_iters += 1
            score = compute_score(a_new)
            if score < best_score:
                best_a = a_new.copy()
                best_score = score
            
            a = a_new
            lr *= lr_decay
        
        if verbose and total_iters % 100 < iters_per_temp:

Smooth max with temperature annealing and FFT-based optimization
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len


def convolve_full_fft(a):
    """Full convolution using FFT."""
    n = len(a)
    N = next_fast_len(2 * n - 1)
    fa = fft(a, N)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = convolve_full_fft(a)
    return float(2.0 * n * c.max() / (S * S))


def smooth_max_and_grad(a, T):
    """Compute smooth max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    c = convolve_full_fft(a)
    c_max = c.max()
    c_shifted = c - c_max
    exp_vals = np.exp(c_shifted / T)
    exp_sum = exp_vals.sum()
    p = exp_vals / exp_sum
    M_S = c_max + T * np.log(exp_sum)
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j]
    N = next_fast_len(3 * n)
    fft_a = fft(a, N)
    fft_p = fft(p, N)
    h = ifft(fft_a * np.conj(fft_p)).real[:n]
    grad = 2.0 * h
    
    return M_S, grad, p, c


def optimize_smooth_max(a, deadline, rng):
    """Optimize using smooth max with temperature annealing."""
    n = len(a)

    a = a / a.sum()

            S = a.sum()
            
            # Full gradient

```python
"""
Smooth max with temperature annealing and FFT-based subgradient optimization
for minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses multi-start coarse-to-fine
strategy with adaptive learning rates and peak-weighted subgradients.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len


def convolve_full_fft(a):
    """Full linear convolution via FFT."""
    n = len(a)
    N = next_fast_len(2 * n - 1)
    fa = fft(a, N)
    return ifft(fa * fa).real[:2 * n - 1]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = convolve_full_fft(a)
    return float(2.0 * n * c.max() / (S * S))


def smooth_max_and_grad(a, T):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a (with S=1)."""
    n = len(a)
    N = next_fast_len(3 * n)
    fft_a = fft(a, N)
    c = ifft(fft_a * fft_a).real[:2 * n - 1]
    c_max = c.max()
    c_shifted = c - c_max
    exp_vals = np.exp(c_shifted / T)
    exp_sum = exp_vals.sum()
    p = exp_vals / exp_sum
    M_S = c_max + T * np.log(exp_sum)
    fft_p = fft(p, N)
    h = ifft(fft_a * np.conj(fft_p)).real[:n]
    grad = 2.0 * h
    return M_S, grad


def optimize_smooth_max(a, deadline, rng):
    """Optimize using smooth max with geometric temperature annealing."""
    n = len(a)
    a = np.clip(a, 1e-10, None).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    c0 = convolve_full_fft(a)
    c0_max = c0.max()
    T_start = max(c0_max * 0.5, 1e-6)
    T_end = max(c0_max * 0.002, 1e-10)
    n_temps = 50
    temps = np.geomspace(T_start, T_end, n_temps)

    iters_per_temp = 40
    stall_count = 0

    for T in temps:
        if time.monotonic() > deadline:
            break
        lr = 0.4
        for _ in range(iters_per_temp):
            if time.monotonic() > deadline:
                break
            M_S, grad = smooth_max_and_grad(a, T)
            S = a.sum()
            full_grad = (2.0 * n / (S * S)) * (grad - 2.0 * M_S / S)
            scale = np.abs(full_grad).max() + 1e-10
            a_new = a * np.exp(-lr * full_grad / scale)
            a_new = np.clip(a_new, 1e-15, None)
            a_new = a_new / a_new.sum()
            score = compute_score(a_new)
            if score < best_score:
                best_a = a_new.copy()
                best_score = score
                stall_count = 0
            else:
                stall_count += 1
            a = a_new
            lr *= 0.99

        if stall_count > 200:
            noise = rng.normal(0, 0.03, n)
            a = best_a * (1.0 + noise)
            a = np.clip(a, 1e-10, None)
            a = a / a.sum()
            stall_count = 0

    return best_a, best_score


def subgradient_optimize(a, deadline, rng, max_iters=20000):
    """Peak-weighted subgradient descent with multiplicative updates."""
    n = len(a)
    a = np.clip(a, 1e-10, None).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    N = next_fast_len(3 * n)

    lr = 0.12
    stall = 0
    peak_frac = 0.97

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        fft_a = fft(a, N)
        c = ifft(fft_a * fft_a).real[:2 * n - 1]
        M = c.max()

        threshold = peak_frac * M
        pw = np.zeros(2 * n - 1)
        mask = c >= threshold
        if mask.any():
            pw[mask] = c[mask]
            pw_sum = pw.sum()
            if pw_sum > 1e-15:
                pw /= pw_sum
            else:
                pw[np.argmax(c)] = 1.0
        else:
            pw[np.argmax(c)] = 1.0

        fft_pw = fft(pw, N)
        h = ifft(fft_a * np.conj(fft_pw)).real[:n]
        grad = 2.0 * h

        ratio = grad / (2.0 * M + 1e-10)
        a_new = a * np.exp(-lr * (ratio - 1.0))
        a_new = np.clip(a_new, 1e-15, None)
        a_new = a_new / a_new.sum()

        # Compute new score
        fft_an = fft(a_new, N)
        cn = ifft(fft_an * fft_an).real[:2 * n - 1]
        score = 2.0 * n * cn.max()  # S=1

        if score < best_score - 1e-10:
            best_a = a_new.copy()
            best_score = score
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0 and stall > 400:

                noise = rng.normal(1.0, strength, n)
                a = best_a * noise
                a = np.clip(a, 1e-10, None)
                a = a / a.sum()
                lr = 0.12
                stall = 0

        lr = max(0.005, lr * 0.999)
        peak_frac = min(0.999, 0.95 + 0.0001 * it)

    return best_a, best_score
