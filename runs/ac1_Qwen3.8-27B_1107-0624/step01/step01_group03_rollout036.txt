def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def subgradient(s, alpha=None, temp=1.0):
    """Compute averaged subgradient of max_k (s*s)_k."""
    n = len(s)
    conv = np.convolve(s, s)
    M = conv.max()
    if alpha is None:
        # Softmax with temperature
        scaled = conv / (temp * M + 1e-15)
        scaled -= scaled.max()
        alpha = np.exp(scaled)
        alpha /= alpha.sum()
    # v_i = 2 * sum_j s_j * alpha[i+j]
    beta = alpha[::-1]
    full_conv = np.convolve(s, beta)
    v = 2.0 * full_conv[2*n-2 : n-2 : -1]
    return v, conv, M

def mirror_descent(s, max_iters, eta0=0.05, temp0=10.0, temp_end=0.1, 
                   nesterov_avg=True, deadline=None, verbose=False):

    n = len(s)
    s = s.copy().astype(np.float64)
    s = s / s.sum()
    best_sc = fast_score(s)
    best_s = s.copy()
    
    if nesterov_avg:
        avg_s = s.copy()
    
    for t in range(1, max_iters + 1):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal temperature
        frac = t / max_iters
        temp = temp0 * (temp_end / temp0) ** frac
        
        # Step size (decaying)
        eta = eta0 / (1.0 + 0.001 * t)
        
        v, conv, M = subgradient(s, temp=temp)
        
        # Mirror descent update
        s = s * np.exp(-eta * v)
        s = s / s.sum()
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        
        if nesterov_avg:
            avg_s = (1.0 - 1.0/t) * avg_s + (1.0/t) * s
        
        # Track best

            sc = fast_score(s)
            if sc < best_sc:

            if nesterov_avg and t % 500 == 0:

Hybrid optimization for minimizing 2n*max(conv(s,s))/(sum(s))^2.
Uses Frank-Wolvo and mirror descent with FFT-accelerated convolutions,
multiple initializations, and coordinate descent polishing.
"""

def fft_convolve(s):
    """FFT-based autocorrelation (linear convolution of s with itself)."""
    n = len(s)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    F = np.fft.rfft(s, fsize)
    return np.fft.irfft(F * F, fsize)[:size]

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = fft_convolve(s)
    return 2.0 * n * conv.max() / (total * total)

def compute_subgradient(s, conv, temp):
    """Compute smoothed subgradient of max_k conv_k."""
    n = len(s)
    M = conv.max()
    if M < 1e-15:
        return np.zeros(n), M
    scaled = conv / (temp * M + 1e-15)
    scaled -= scaled.max()
    alpha = np.exp(scaled)
    alpha /= alpha.sum()
    # v_i = 2 * sum_j s_j * alpha[i+j]
    beta = alpha[::-1]
    # Use FFT
    fsize = 1
    while fsize < n + len(alpha) - 1:
        fsize *= 2
    Fs = np.fft.rfft(s, fsize)
    Fb = np.fft.rfft(beta, fsize)
    full_conv = np.fft.irfft(Fs * Fb, fsize)[:n + len(beta) - 1]
    # v_i = 2 * full_conv[2*n-2-i] for i=0,...,n-1
    # full_conv has length n + (2n-1) - 1 = 3n-2
    # We need indices 2n-2, 2n-3, ..., n-1
    v = 2.0 * full_conv[2*n-2 : n-2 : -1]
    return v, M

def frank_wolvo(s, max_iters, deadline=None, verbose=True):
    """Frank-Wolvo with adaptive step size."""
    n = len(s)
    s = s.copy().astype(np.float64)
    s = s / s.sum()
    best_sc = fast_score(s)
    best_s = s.copy()
    
    # Nesterov averaging
    avg_s = s.copy()
    
    no_imp = 0
    for t in range(1, max_iters + 1):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = fft_convolve(s)
        M = conv.max()
        
        # Subgradient at the peak (or smoothed)
        # Use a mix: primarily the peak, with some smoothing
        temp = max(0.01, 1.0 / (t * 0.01))  # decreasing temperature
        v, M = compute_subgradient(s, conv, temp)
        
        # Frank-Wolvo: find best vertex
        i_star = np.argmin(v)
        
        # Adaptive step size
        gamma = 2.0 / (t + 2.0)
        # Occasionally use larger steps
        if t % 100 == 0:
            gamma = min(0.5, 2.0 / (t / 100 + 2.0))
        
        s_new = (1.0 - gamma) * s
        s_new[i_star] += gamma
        s_new = np.maximum(s_new, 0.0)
        s_new = s_new / s_new.sum()
        
        # Nesterov average
        avg_s = (1.0 - gamma) * avg_s + gamma * s_new
        
        s = s_new
        
        # Periodic evaluation
        if t % 200 == 0 or t == 1:
            sc = fast_score(s)
            if sc < best_sc:
                best_sc = sc
                best_s = s.copy()
                no_imp = 0
            else:
                no_imp += 1
            
            sc_avg = fast_score(np.maximum(avg_s, 0.0) / max(avg_s.sum(), 1e-15))
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_s = np.maximum(avg_s, 0.0)
                best_s = best_s / best_s.sum()
                no_imp = 0
        
        # Restart if stuck
        if no_imp > 20:
            # Perturb
            s = s + 0.01 * np.random.standard_normal(n)
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            avg_s = s.copy()
            no_imp = 0
    
    # Final
    sc = fast_score(s)
    if sc < best_sc:
        best_sc = sc
        best_s = s.copy()
    avg_s = np.maximum(avg_s, 0.0)
    if avg_s.sum() > 1e-15:
        avg_s = avg_s / avg_s.sum()
        sc_avg = fast_score(avg_s)
        if sc_avg < best_sc:
            best_sc = sc_avg
            best_s = avg_s.copy()
    
    return best_s, best_sc

        temp = temp0 * (temp_end / temp0) ** frac
        eta = eta0 / (1.0 + 0.0005 * t)

        v, M = compute_subgradient(s, conv, temp)
        
        s = s * np.exp(-eta * v)

        avg_s = (1.0 - 1.0/t) * avg_s + (1.0/t) * s
        
        if t % 200 == 0:
            sc = fast_score(s)
            if sc < best_sc:
                best_sc = sc
                best_s = s.copy()

    if sc < best_sc:

        sc_avg = fast_score(avg_s)
        if sc_avg < best_sc:

    return best_s, best_sc

def polish_pairwise(s, max_pairs=500, deadline=None):
    """-coordinate descent on random pairs."""
    n = len(s)
    s = s.copy().astype(np.float64)
    s = s / s.sum()
    best_sc = fast_score(s)
    best_s = s.copy()
    rng = np.random.default_rng(12345)
    
    for it in range(max_pairs):
        if deadline is not None and time.monotonic() > deadline:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        
        # Try a few delta values
        best_d = 0.0
        best_ds = best_sc
        for d in np.linspace(-0.5 * s[i], 0.5 * s[j], 11):
            if d == 0:
                continue
            s2 = s.copy()
            s2[i] += d
            s2[j] -= d
            if s2[i] < 0 or s2[j] < 0:
                continue
            sc2 = fast_score(s2)
            if sc2 < best_ds:
                best_ds = sc2
                best_d = d
        
        if best_d != 0:
            s[i] += best_d
            s[j] -= best_d
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            best_sc = fast_score(s)
            best_s = s.copy()
    
    return best_s, best_sc
