"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses multi-phase search: low-dim parameter optimization, full logit-space Adam,
length interpolation, and scipy-based final polish.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < -1e-9:
        c = np.maximum(c, 0.0)
    elif c.min() < 0:
        c = c - c.min()
    return c

def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = _convolve_fft(a)
    return float(2.0 * n * c.max() / (s * s))

def _true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z, floor=1e-12):
    z = np.clip(z, -50.0, 50.0)
    z = z - z.max()
    e = np.exp(z) + floor
    return e / e.sum()

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))

def _make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_gaussian_profile(n, center, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
    a = a + 1e-12
    return a / a.sum()

def _make_deterministic_profiles(n):
    profs = []
    for eta in (0.5, 1.0, 1.25, 1.5, 1.75, 2.0, 2.5):
        profs.append(_make_linear_profile(n, eta))
    for a1, a3 in ((1.5, -1.0), (2.0, -2.0), (1.0, -0.5), (1.8, -1.5), (1.2, 0.8), (2.5, -3.0)):
        profs.append(_make_cubic_profile(n, a1, a3))
    for c, w in ((0.5, 0.15), (0.5, 0.2), (0.5, 0.25), (0.4, 0.15), (0.6, 0.15)):
        profs.append(_make_gaussian_profile(n, c, w))
    # Additional shapes
    t = np.linspace(0.0, 1.0, n)
    for shape_fn in [
        lambda t: 1.0 + 2.0*t - 0.5*t*t,
        lambda t: 1.0 + 1.5*t + 0.5*t**2,
        lambda t: 1.0 + 0.8*np.sin(np.pi*t),
        lambda t: 1.0 + 1.2*t**0.5 - 0.3*t,
    ]:
        a = shape_fn(t)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    return profs

def _adam_optimize_logit(p0, deadline, rng, lr=0.06, reg=1e-7, beta_init_factor=0.3,

    Adam optimization in logit space for a non-negative simplex sequence.
    Minimizes 2*n*max(conv(p,p)) subject to p >= 0, sum(p) = 1.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    z -= float(np.mean(z))

    c = _convolve_fft(p)

    beta = max(beta_init_factor * maxc, 1e-12)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    total_iters = 0
    
    while time.monotonic() < deadline:
        # Compute soft-max weights
        mval = float(c.max())
        if beta <= 1e-14:
            w = np.zeros_like(c)
            mask = c >= mval - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                w[int(np.argmax(c))] = 1.0
            else:
                w[mask] = 1.0 / cnt
        else:
            x = (c - mval) / beta
            ex = np.exp(x)
            s = float(ex.sum())
            if not np.isfinite(s) or s <= 0:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ex / s
        
        # Gradient: d(max_c)/d(p[j]) = 2 * sum_k w[k] * p[k-j]
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)

        gp = 2.0 * h[n - 1:2 * n - 1]
        
        # Project gradient to simplex tangent: subtract mean
        gz = p * (gp + reg * p - float(np.dot(p, gp + reg * p)))
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1

        vh = v / (1 - b2 ** t_adam)

            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        z = np.clip(z, -max_theta, max_theta)
        z -= float(np.mean(z))
        p = _softmax(z)
        c = _convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        total_iters += 1
        
        if np.isfinite(score):
            if score < best_score - restart_tol:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        
        # Decay beta
        if total_iters % 50 == 0:
            beta = max(beta_min, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > restart_interval:
            # Try different restart strategies
            r = rng.random()
            if r < 0.3:
                # Small perturbation
                z = best_z + 0.3 * rng.normal(size=n)
            elif r < 0.6:
                # Medium perturbation
                z = 0.9 * best_z + 0.5 * rng.normal(size=n)
            elif r < 0.8:

                random_profile = _make_linear_profile(n, rng.uniform(0.5, 2.5))
                z = np.log(0.8 * best_p + 0.2 * random_profile + 1e-12)
                z -= np.mean(z)
            else:
                # Large perturbation
                z = 0.7 * best_z + 0.8 * rng.normal(size=n)
            
            z = np.clip(z, -max_theta, max_theta)
            z -= float(np.mean(z))
            p = _softmax(z)
            c = _convolve_fft(p)
            maxc = float(c.max())
            beta = max(0.15 * maxc, 1e-10)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
    
    return best_p, best_score

        z = basis @ theta
        return _softmax(z)

    maxc = float(c.max())

    best_score = float(2.0 * n * maxc)

        gp = 2.0 * h[n - 1:2 * n - 1]
        
        # Gradient through softmax: gz = p * (gp - sum(p*gp))

        gtheta = basis.T @ gz

            v.fill(0)
            t_adam = 0

def _make_basis_mixed(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u]
    m = 1
    while len(cols) < K:
        ang = 2.0 * np.pi * m * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
        m += 1
        if len(cols) > K:
            break
    return np.column_stack(cols[:K])

def _make_basis_poly(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u ** k for k in range(1, K + 1)]
    return np.column_stack(cols)

    intensive final polish with adaptive restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    candidates = []

    # Phase 0: Quick initial candidates at several lengths
    for nf in (128, 256, 512):
        for eta in (1.0, 1.5, 2.0, 2.5):
            p = _make_linear(nf, eta)
            sc = _internal_score(p)
            try_update(p, sc)
        for a1, a3 in ((1.5, -1.0), (2.0, -2.0), (1.0, -0.5)):
            p = _make_cubic(nf, a1, a3)
            sc = _internal_score(p)
            try_update(p, sc)
    if best_seq is not None:
        _add_to_candidates(candidates, best_seq, best_score)

    # Load parent
    parent_p = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent_p = _normalize_nonneg(parent)
            if parent_p is not None:
                sc = _internal_score(parent_p)
                try_update(parent_p, sc)
                _add_to_candidates(candidates, parent_p, sc)
    except Exception:
        pass

    print(f"Phase 0: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # Phase 1: Low-dim parameter search at n=128
    n1 = 128
    bm = _basis_mixed(n1, 15)
    bp = _basis_poly(n1, 14)
    phase1_end = min(deadline, t_start + 250.0)

    for basis in (bm, bp):
        dims = basis.shape[1]
        starts = []
        th = np.zeros(dims)
        th[0] = 1.5
        starts.append(th.copy())
        th = np.zeros(dims)
        th[0] = 2.0
        if dims > 2:
            th[2] = -2.0
        starts.append(th.copy())
        th = np.zeros(dims)
        th[0] = 1.0
        if dims > 3:
            th[3] = -1.0
        starts.append(th.copy())
        for _ in range(25):
            starts.append(rng.uniform(-4.0, 4.0, dims))

        for th in starts:
            if time.monotonic() > phase1_end:
                break
            per = min(phase1_end, time.monotonic() + 3.0)
            p_best, sc = _adam_basis(th, basis, per, rng, lr=0.3)
            try_update(p_best, sc)
            _add_to_candidates(candidates, p_best, sc)

    print(f"Phase 1: best={best_score:.10f}")

    # Phase 2: Full logit-space optimization at multiple lengths
    phase2_end = min(deadline, t_start + 1100.0)
    lengths_to_try = [128, 256, 512]
    if parent_p is not None:
        pn = parent_p.size
        if pn not in lengths_to_try:
            lengths_to_try.append(min(1024, max(128, pn)))
    lengths_to_try = sorted(set(lengths_to_try))

    time_per_len = max(40.0, (phase2_end - time.monotonic()) / len(lengths_to_try))

    for nf in lengths_to_try:
        if time.monotonic() > phase2_end:
            break
        t_len_start = time.monotonic()

        # Collect starting points
        start_points = []
        for sc_c, p_c in candidates[:10]:
            if p_c.size == nf:
                start_points.append(p_c)
            else:
                p_i = _interp_profile(p_c, nf)
                if p_i is not None:
                    start_points.append(p_i)

        # Deterministic profiles
        det_profs = _make_det_profiles(nf)
        start_points.extend(det_profs[:8])

        # Parent
        if parent_p is not None:
            if parent_p.size == nf:
                start_points.append(parent_p.copy())
            else:
                p_i = _interp_profile(parent_p, nf)
                if p_i is not None:
                    start_points.append(p_i)

        # Random smooth profiles
        for _ in range(4):
            a1 = float(rng.uniform(0.5, 3.0))
            a2 = float(rng.uniform(-1.5, 1.5))
            a3 = float(rng.uniform(-3.0, 1.0))
            start_points.append(_make_cubic(nf, a1, a3))

        # Limit number of starts based on available time
        n_starts = len(start_points)
        per_start = min(25.0, time_per_len / max(1, n_starts))
        max_starts = max(3, int(time_per_len / max(1.0, per_start)))
        start_points = start_points[:max_starts]

        for p0 in start_points:
            if time.monotonic() > phase2_end:
                break
            per = min(phase2_end, time.monotonic() + per_start)
            p_best, sc = _adam_logit(p0, per, rng, lr=0.05)
            try_update(p_best, sc)
            _add_to_candidates(candidates, p_best, sc)

        print(f"Phase 2 n={nf}: best={best_score:.10f}")

    # Phase 3: Intensive polish of the best candidate
    if best_seq is not None:
        phase3_end = deadline - 40.0
        if time.monotonic() < phase3_end:
            # Long intensive run on best
            p_best, sc = _adam_logit(
                best_seq, phase3_end, rng,
                lr=0.035, reg=5e-8,
                beta_factor=0.12,
                beta_decay=0.965,
                beta_min=1e-12,
                restart_every=700
            )
            try_update(p_best, sc)
            _add_to_candidates(candidates, p_best, sc)

            # If time remains, try second candidate
            if len(candidates) >= 2 and time.monotonic() < phase3_end - 60:
                p2 = candidates[1][1]
                if p2.size == best_seq.size:
                    p_b2, sc2 = _adam_logit(
                        p2, min(phase3_end, time.monotonic() + 90), rng,
                        lr=0.035, reg=5e-8,
                        beta_factor=0.12,
                        beta_decay=0.965
                    )
                    try_update(p_b2, sc2)

    print(f"Phase 3: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # Phase 4: scipy-based final polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        try:
            from scipy.optimize import minimize as sp_min
            p0 = best_seq.copy()
            n = p0.size
            z0 = np.log(p0 + 1e-15)
            z0 -= float(z0.mean())

            def obj_fn(z):
                pp = _softmax_stable(z)
                cc = _convolve_fft(pp)
                return float(2.0 * n * cc.max())

            def grad_fn(z):
                pp = _softmax_stable(z)
                cc = _convolve_fft(pp)
                gp, w, mv = _compute_gradient(cc, pp, 0.03 * float(cc.max()))
                vp = gp + 1e-8 * pp
                dot = float(np.dot(pp, vp))
                gz = pp * (vp - dot)
                gz -= float(np.mean(gz))
                return gz

            t_scipy_end = min(deadline, time.monotonic() + 45.0)
            n_iter_budget = max(100, int((t_scipy_end - time.monotonic()) * 20))

            res = sp_minimize(
                obj_fn, z0, method='L-BFGS-B',
                jac=grad_fn,
                options={'maxiter': n_iter_budget, 'maxfun': n_iter_budget * 5,
                         'gtol': 1e-12, 'ftol': 1e-14}
            )
            p_sp = _softmax_stable(res.x)
            sc_sp = _internal_score(p_sp)
            try_update(p_sp, sc_sp)
            print(f"Phase 4 scipy: score={sc_sp:.10f}")
        except Exception as e:
            print(f"Phase 4 scipy error: {e}")

    # Final validation and return
    if best_seq is not None:
        final_seq = _normalize_nonneg(best_seq)
        if final_seq is None:
            final_seq = _make_linear(512, 1.5)

        # Check with true convolution
        if final_seq.size <= 2048:
            sc_true = _true_score(final_seq)
            if np.isfinite(sc_true) and sc_true < best_score:
                best_score = sc_true

        # Try evaluate_sequence
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s

        print(f"FINAL: score={best_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Ultimate fallback
    fb = _make_linear(256, 1.5)
    return [float(x) for x in fb]
