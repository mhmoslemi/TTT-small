import numpy as np
import time

def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2"""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def smooth_and_grad(a, tau):
    """Smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad

def peak_subgradient(a, top_k=5):
    """Compute subgradient using top-k peaks directly (no smoothing)."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    # Get top-k peak positions
    top_indices = np.argsort(c)[-top_k:]
    # Weight them by how close they are to max
    c_max = c[top_indices[0]]
    c_min_top = c[top_indices[-1]]
    range_c = max(c_max - c_min_top, 1e-10)
    weights = (c[top_indices] - c_min_top) / range_c
    weights = weights / weights.sum()
    
    # Compute gradient: for each peak position k, grad[j] += a[k-j] (if valid)
    grad = np.zeros(n)
    for w, k in zip(weights, top_indices):
        # c[k] = sum_i a[i] * a[k-i]

        # But since a is length n, valid indices for a[k-j] are 0 <= k-j < n
        # i.e., k-n+1 <= j <= k

        hi = min(n - 1, k)

        # = from min(k, n-1) to max(0, k-n+1)
        a_mirror = a[k - np.arange(lo, hi + 1)]  # a[k-lo], a[k-lo+1], ..., a[k-hi]
        # Wait, let me be more careful
        # For j in [lo, hi], the contribution to dc[k]/da[j] is:
        # a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
        # k-j is in [k-hi, k-lo] = [max(0,k-n+1), min(k, n-1)] ... hmm
        # Let me just compute it directly for the valid range

        valid1 = (k_minus_j >= 0) & (k_minus_j < n)

def mirror_balance(a, n_iter=1000, seed=42):
    """Iterative mirror balancing to reduce peak convolution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    
    a = a.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(n_iter):
        c = convolve_fft(a)
        k_star = np.argmax(c)
        c_max = c[k_star]
        
        S = np.sum(a)
        J = 2.0 * n * c_max / (S * S)
        if J < best_J:
            best_J = J
            best_a = a.copy()
        
        # Compute contribution of each pair to c[k_star]
        # c[k_star] = sum_j a[j] * a[k_star - j] (for valid j)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        j_range = np.arange(lo, hi + 1)
        pairs = a[j_range] * a[k_star - j_range]
        total = pairs.sum()
        
        if total < 1e-10:
            break
        
        # We want to reduce the large pairs
        # Strategy: reduce a[j] where pairs[j-lo] is above average
        # and increase where below
        avg = total / len(j_range)
        
        # Compute adjustment
        # For j in [lo, hi], the "excess" is pairs[j-lo] - avg
        # We want to reduce a[j] proportional to excess
        excess = pairs - avg
        # But we also need to consider the mirror: reducing a[j] helps
        # but reducing a[k_star-j] also helps
        # Net effect on c[k_star] from reducing a[j] by delta:
        # -delta * a[k_star-j] (from the j term)
        # -delta * a[j-k_star] (from the mirror term, if j-k_star is valid)
        
        # Simple approach: adjust a[j] by -eta * a[k_star-j] for j in [lo,hi]
        eta = 0.01
        adjustment = np.zeros(n)
        adjustment[j_range] -= eta * a[k_star - j_range]
        
        # Also add some to maintain sum: add uniform positive
        adjustment += eta * a[k_star - j_range].sum() / n
        
        a = a + adjustment
        a = np.maximum(a, 0.0)
        S = np.sum(a)
        if S < 0.01:
            a = best_a.copy()
            break
        a = a * (n / S)
    
    return best_a, best_J

def simulated_annealing(a_init, budget_s, seed):
    """Simulated annealing with structure-aware moves."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J(a)
    
    T = 0.1  # Initial temperature
    T_min = 1e-6
    cool_rate = 0.999
    
    start_time = time.monotonic()
    total_iters = 0
    
    while time.monotonic() < deadline:
        total_iters += 1
        elapsed = (time.monotonic() - start_time)
        frac = min(1.0, elapsed / budget_s)
        T = T * (T_min / T) ** frac  # Linear interpolation on log scale
        
        # Generate candidate
        cand = a.copy()
        
        # Move type 1: Perturb a random block
        if rng.random() < 0.5:
            width = int(rng.integers(max(1, n // 500), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width)))
            noise = rng.normal(0.0, 0.1, width)
            cand[start:start+width] *= np.exp(noise)
        # Move type 2: Swap two elements
        elif rng.random() < 0.8:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = (a[i] - a[j]) * 0.1 * rng.random()
                cand[i] -= delta
                cand[j] += delta
        # Move type 3: Targeted peak reduction
        else:
            c = convolve_fft(a)
            k_star = np.argmax(c)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            j_range = np.arange(lo, hi + 1)
            pairs = a[j_range] * a[k_star - j_range]
            # Find the top contributing pair
            top_local = np.argmax(pairs)
            j_top = j_range[top_local]
            # Reduce the larger of the pair
            a_j = a[j_top]
            a_mirror = a[k_star - j_top] if 0 <= k_star - j_top < n else 0
            if a_j > a_mirror:
                cand[j_top] *= 0.95
            else:
                m = k_star - j_top
                if 0 <= m < n:
                    cand[m] *= 0.95
        
        cand = np.maximum(cand, 0.0)
        cS = np.sum(cand)
        if cS < 0.01:
            T = max(T_min, T * 0.9)
            continue
        
        cJ = compute_J(cand)
        
        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
        elif rng.random() < np.exp(-(cJ - best_J) / max(T, 1e-10)):
            a = cand.copy()
        
        T = max(T_min, T * 0.9999)
    
    return best_a, best_J

"""
Multi-phase optimizer for minimizing J(a) = 2n*max(a*a)/sum(a)^2:
1. Exact peak-subgradient descent with adaptive restarts
2. Direct peak-chasing (reducing elements contributing to top peaks)
3. Coordinate descent on sensitive indices
4. Final polish with fine-tuned gradient steps
Uses FFT for all convolution computations.
"""

import numpy as np
import time

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_J(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def exact_subgradient(a, top_k=5):
    """Compute J and subgradient using top-k peaks."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    J = 2.0 * n * c_max / (S * S)
    
    # Get top-k peak positions
    # Use a threshold to include all "near-max" peaks
    threshold = c_max - (c_max - np.min(c)) * 0.01  # top 1% of range
    peak_positions = np.where(c >= threshold)[0]
    if len(peak_positions) > top_k:
        # Keep only top_k
        idx = np.argsort(c[peak_positions])[-top_k:]
        peak_positions = peak_positions[idx]
    
    # Compute weighted subgradient
    weights = c[peak_positions]
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, peak_positions):
        # grad[j] += a[k-j] (if valid) + a[j-k] (if valid)
        # Part 1: a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        lo1 = max(0, int(k) - n + 1)
        hi1 = min(n - 1, int(k))
        if lo1 <= hi1:
            js = np.arange(lo1, hi1 + 1)
            grad[js] += w * a[int(k) - js]
        # Part 2: a[j-k] for j in [max(0,k), min(n-1, k+n-1)]
        lo2 = max(0, int(k))
        hi2 = min(n - 1, int(k) + n - 1)
        if lo2 <= hi2:
            js = np.arange(lo2, hi2 + 1)
            grad[js] += w * a[js - int(k)]
    
    # Full gradient of J = 2n*c_max/S^2
    # dJ/da[j] = 2n * grad_c[j] / S^2 - 4n * c_max / S^3
    # where grad_c[j] = sum_k w_k * (a[k-j] + a[j-k])
    grad = (2.0 * n * grad) / (S * S) - (4.0 * n * c_max) / (S * S * S)
    
    return J, grad

def smooth_subgradient(a, tau=1.0):
    """Compute J and gradient using log-sum-exp smoothing."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    # Cross-correlation for gradient
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad

def peak_chase_step(a, top_peaks=3, strength=0.02):
    """Directly reduce elements contributing to top peaks."""
    n = len(a)
    c = convolve_fft(a)
    
    # Get top peaks
    peak_indices = np.argsort(c)[-top_peaks:]
    
    adjustment = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        
        # Contribution of each j to c[k]
        contrib = a[j_range] * a[k - j_range]
        total = contrib.sum()
        if total < 1e-15:
            continue
        
        # Reduce a[j] proportional to a[k-j] (the mirror)
        # This reduces c[k] by strength * a[j]*a[k-j]
        reduction = strength * a[k - j_range]
        adjustment[j_range] -= reduction
        
        # Also apply mirror reduction if j-k is valid
        jk = j_range - k
        valid = (jk >= 0) & (jk < n)
        if valid.any():
            adjustment[j_range[valid]] -= strength * a[j_range[valid]]
    
    a = a + adjustment
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 0.01:
        return None
    a = a * (n / S)
    return a

def coordinate_refine(a, budget_s, seed):
    """Coordinate descent on sensitive indices."""
    n = len(a)

    deadline = time.monotonic() + budget_s

    best_a = a.copy()

        top_idx = np.argsort(sensitivity)[-n_top:]
        rand_idx = rng.integers(0, n, size=n_top)
        candidates = np.concatenate([top_idx, rand_idx])
        rng.shuffle(candidates)
        
        improved = False
        for j in candidates:
            if time.monotonic() >= deadline:
                    break
            # Try a few multipliers
            for mult in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]:

                cand[j] *= mult
