def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))

def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)

def make_exp_decay(n, tau_frac=0.3):
    """Exponentially decaying (asymmetric)."""
    tau = tau_frac * n
    return np.exp(-np.arange(n) / tau)

def make_exp_rise(n, tau_frac=0.3):
    """Exponentially rising (asymmetric, mirror of decay)."""
    tau = tau_frac * n
    return np.exp(-np.arange(n - 1, -1, -1) / tau)

def make_bimodal(n, center_frac=0.3, width_frac=0.2):
    """Bimodal distribution."""
    c1 = center_frac * n
    c2 = (1 - center_frac) * n
    w = width_frac * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)

def make_asym_cosine(n, alpha=0.7):
    """Asymmetric cosine-like window."""
    x = np.linspace(0, 1, n)
    return 0.5 * (1.0 + np.cos(np.pi * (x ** alpha)))

`= (p-1)/p * M^(1-p) * f_s^(1-p) * M^(p-1) * raw_s[i]`
`= (p-1)/p * f_s^(1-p) * raw_s[i]`

Yes! The `M` factors cancel. So I can safely use the scaled `c` for numerical stability, and the gradient is:
`grad[i] = (p-1)/p * f_s^(1-p) * (convolve(a, c_s^(p-1), 'valid'))[i]`

where `f_s = (sum c_s^p)^(1/p)` and `c_s = c / c.max()`.

This is numerically stable because `c_s` has max 1, so `c_s^(p-1)` won't overflow (it'll be <= 1).

For numerical stability at the low end: if `c_s[k]` is very small, `c_s[k]^(p-1)` will underflow to 0, which is fine (it just means that entry doesn't contribute to the gradient, which is correct behavior).

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-phase FISTA with Lp-smoothed objective, LP-based polishing,
and diverse initializations including asymmetric sequences.
Key insight: asymmetric sequences can achieve lower scores than symmetric ones.
"""

import numpy as np
import time
from scipy.optimize import linprog


def simp_proj(v, s=1.0):
    """Project vector v onto the scaled probability simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def gradient_lp(a, c, p):
    """
    Gradient of (sum(c^p))^(1/p) w.r.t. a, where c = conv(a,a).
    Uses scaled c for numerical stability.
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(n)
    c_s = c / c_max
    c_s = np.maximum(c_s, 0.0)
    S_p = np.sum(c_s ** p)
    if S_p < 1e-30:
        return np.zeros(n)
    f_s = S_p ** (1.0 / p)
    c_pm1 = c_s ** (p - 1)
    raw = np.convolve(a, c_pm1, mode='valid')
    return ((p - 1.0) / p) * (f_s ** (1.0 - p)) * raw


def fista_optimize(a_init, deadline, verbose=False):
    """
    Multi-phase FISTA optimization with Lp-smoothed auto-convolution peak.
    Phase A: p=4-8, larger steps (exploration)
    Phase B: p=16-48, medium steps (refinement)
    Phase C: p=64-256, small steps (polishing)
    """
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)

    total_budget = deadline - time.monotonic()
    if total_budget < 5:
        return best_a, best_obj

    phases = [
        (4, 8, 0.8, 50, 0.25),
        (16, 48, 0.4, 30, 0.35),
        (64, 256, 0.15, 20, 0.40),
    ]

    for phase_idx, (p_start, p_end, lr, eval_int, time_frac) in enumerate(phases):
        if time.monotonic() >= deadline:
            break
        phase_deadline = time.monotonic() + total_budget * time_frac
        phase_deadline = min(phase_deadline, deadline)

        t = 1.0
        a_prev = a.copy()
        a = best_a.copy()
        stall = 0
        it = 0

        while time.monotonic() < phase_deadline:
            it += 1
            progress = min(1.0, it / 1000.0)
            p = p_start + (p_end - p_start) * progress

            c = np.convolve(a, a, mode='full')
            g = gradient_lp(a, c, p)

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            g = g / g_norm

            a_new = simp_proj(a - lr * g, 1.0)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            a_extrap = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
            a = simp_proj(a_extrap, 1.0)
            a_prev = a_new
            t = t_new

            if it % eval_int == 0:
                obj = compute_obj(a)
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall > 30:
                        a = best_a.copy()
                        a_prev = a.copy()
                        t = 1.0
                        lr = max(lr * 0.6, 0.01 / n)
                        stall = 0

            if verbose and it % 10000 == 0:
                obj_now = compute_obj(a)
                print(f"    Ph{phase_idx} it={it} p={p:.0f} obj={obj_now:.6f} best={best_obj:.6f}")

        a = best_a.copy()

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a_init, deadline, verbose=False):
    """LP-based iterative optimization with adaptive steps and perturbations."""
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_obj(a)
    step = 0.08
    stall = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_direction(a, M)
        if g is None or g.sum() < 1e-15:
            if stall > 50:
                break
            stall += 1
            continue
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            sc = compute_obj(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.3, step * 1.04)
        else:
            stall += 1
            step = max(0.003, step * 0.85)
            if stall > 25:
                noise = np.abs(rng.normal(0, 0.12, n))
                a_new = best_a * (1.0 + noise)

                    a_new = a_new / a_new.sum()
                    sc = compute_obj(a_new)
                    if sc < best_score:
                        best_a = a_new
                        best_score = sc
                        a = a_new
                    else:
                        a = a_new
                stall = 0
                step = 0.08

        if stall > 300:
            break

    return best_a, best_score


def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_exp_decay(n, tau_frac):
    """Exponentially decaying (asymmetric)."""
    tau = tau_frac * n
    return np.exp(-np.arange(n) / tau)


def make_exp_rise(n, tau_frac):
    """Exponentially rising (asymmetric)."""
    tau = tau_frac * n
    return np.exp(-np.arange(n - 1, -1, -1) / tau)


def make_bimodal(n, center_frac, width_frac):
    """Bimodal distribution."""
    c1 = center_frac * n
    c2 = (1.0 - center_frac) * n
    w = width_frac * n
    x = np.arange(n)
    return np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)


def make_asym_cosine(n, alpha):
    """Asymmetric cosine-like window."""
    x = np.linspace(0, 1, n)
    return 0.5 * (1.0 + np.cos(np.pi * (x ** alpha)))


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_flat_top(n, width_frac):
    """Flat-top: constant in middle, zero at edges."""
    w = int(width_frac * n)
    a = np.zeros(n)
    start = (n - w) // 2
    a[start:start + w] = 1.0
    return a
