def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 1e-300, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def _greedy(n):
    n = max(int(n), 4)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return _normalize(a)


def _power(n, s):
    n = max(int(n), 4)
    idx = np.arange(n, dtype=np.float64) + 1.0
    return _normalize(np.power(idx, -float(s)))


def _exp_decay(n, rate):
    n = max(int(n), 4)
    idx = np.arange(n, dtype=np.float64)
    return _normalize(np.exp(-float(rate) * idx / n))


def _mix(a, b, lam):
    return _normalize(float(lam) * a + (1.0 - float(lam)) * b)


def _perturb(p, rng, sigma):
    p = np.asarray(p, dtype=np.float64).ravel()
    noise = np.exp(rng.normal(0.0, float(sigma), p.size))
    return _normalize(p * noise)


def _score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    if n <= 512:
        c = np.convolve(a, a)
    else:
        L = _next_pow2(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))

def _adam(p, budget_s, seed=0, alpha_base=8.0, alpha_decay=5.0, lr_scale=1.0):
    """Adam optimizer with smoothed max-convolution objective."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = _normalize(p)
    budget_s = float(budget_s)
    if budget_s < 1.0:
        return p, _score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = 0.06 * lr_scale
    elif n < 256:
        lr = 0.04 * lr_scale
    elif n < 1024:
        lr = 0.025 * lr_scale
    else:
        lr = 0.015 * lr_scale

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = _next_pow2(2 * n - 1 + 64)
    best_p = p.copy()
    best_f = _score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)
    n_steps = 0

    while time.monotonic() < end:
        t += 1
        n_steps += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = alpha_base + (60.0 - alpha_base) * np.exp(-alpha_decay * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Softmax weights
        z = alpha * (c / maxc - 1.0)
        z -= z.max()
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient computation via FFT
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -70.0, 70.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = _score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def _adam_multi_temp(p, budget_s, seed=0):
    """Run Adam with multiple temperature settings, keep best."""
    temps = [(8.0, 5.0), (5.0, 3.0), (15.0, 6.0), (3.0, 2.0)]
    best_p = p.copy()
    best_f = _score_fft(p)
    n_budget = max(1.0, budget_s / len(temps))
    for i, (ab, ad) in enumerate(temps):
        if time.monotonic() - (time.monotonic() - budget_s) > budget_s:
            break
        p_out, f_out = _adam(p, n_budget, seed=seed + i * 7, alpha_base=ab, alpha_decay=ad)
        if f_out < best_f:
            best_f = f_out
            best_p = p_out
        p = _perturb(best_p, np.random.default_rng(seed + i * 13 + 100), 0.05)
    return best_p, best_f


def _coord_polish(p, budget_s, n_coords_per_step=4):
    """
    Coordinate-level polish: for each coordinate (or small group),
    try a few perturbation levels and keep the best.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n == 0:
        return p, np.inf
    p = _normalize(p)
    cur_f = _score_fft(p)
    rng = np.random.default_rng(12345)
    start = time.monotonic()
    end = start + max(0.0, float(budget_s) - 1.0)

    # Precompute FFT setup
    L = _next_pow2(2 * n - 1)

    def fast_score(a):
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        return float(2.0 * n * c.max())

    cur_val = fast_score(p)
    best_val = cur_val
    best_p = p.copy()
    it = 0

    while time.monotonic() < end:
        it += 1
        # Pick random coordinates to perturb
        if n <= 64:
            idxs = list(range(n))
            rng.shuffle(idxs)
        else:
            idxs = list(rng.choice(n, size=min(n, 16), replace=False))

        for idx in idxs:
            if time.monotonic() > end:
                break
            old_val = p[idx]
            # Try a few values
            for scale in (0.5, 0.7, 1.3, 1.6, 0.3, 2.0, 0.8, 1.2):
                p[idx] = old_val * scale
                new_val = fast_score(p)
                if new_val < cur_val * (1.0 - 1e-9):
                    cur_val = new_val
                else:
                    p[idx] = old_val
            # After trying scales, re-normalize p
            p = _normalize(p)
            cur_val = fast_score(p)

        if cur_val < best_val:
            best_val = cur_val
            best_p = p.copy()

    final_f = _score_fft(best_p)
    return best_p, final_f


def _lp_direction(p, budget_s=60.0):
    """
    LP-based direction finding with limited constraints.
    Solve: max sum(g) s.t. (f*g)_k <= M for top constraints, g >= 0.
    """
    if _scipy_opt is None:
        return p, _score_fft(p)

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4:
        return p, _score_fft(p)
    budget_s = float(budget_s)
    if budget_s < 15.0:
        return p, _score_fft(p)

    L = _next_pow2(2 * n - 1)
    S0 = np.sqrt(2.0 * n)
    f = _normalize(p) * S0
    cur_score = _score_fft(f)
    if not np.isfinite(cur_score):
        return p, cur_score

    Pf = np.fft.rfft(f, L)
    conv = np.fft.irfft(Pf * Pf, L)[:2 * n - 1].real
    conv = np.maximum(conv, 0.0)
    M = float(conv.max())

    best_f = f.copy()
    best_score = float(cur_score)
    end = time.monotonic() + max(0.0, budget_s - 5.0)

    # Start with top constraints
    top_k = 30
    order = np.argsort(conv)[::-1]
    rows = sorted(set(int(k) for k in order[:top_k]))
    row_set = set(rows)
    max_rows = min(200, 2 * n - 1)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        r = len(rows_arr)
        A = np.zeros((r, n), dtype=np.float64)
        for ii in range(r):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(r, M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)
        rem = max(1.0, end - time.monotonic() - 2.0)
        tl = max(3.0, min(30.0, rem * 0.3))

        try:
            res = _scipy_opt.linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None),
                                      method='highs', options={'time_limit': tl})
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        Pg = np.fft.rfft(g, L)
        conv_fg = np.fft.irfft(Pf * Pg, L)[:2 * n - 1].real
        conv_fg = np.maximum(conv_fg, 0.0)
        violations = conv_fg - M
        tol = 1e-6 + 1e-6 * M
        if np.any(violations > tol):
            if len(rows) < max_rows:
                bad = np.argsort(violations)[::-1]
                added = 0
                for ix in bad:
                    if added >= 50 or len(rows) >= max_rows:
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            continue

        g = g * (S0 / sg)
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.5, 0.25, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = _score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()

        if accepted is None:
            break
        if accepted_score >= best_score:
            break
        f = accepted
        M = accepted_score * (sn / S0) ** 2 / (2.0 * n) * S0 ** 2 / S0 ** 2
        # Recompute M
        Pf = np.fft.rfft(f, L)
        conv = np.fft.irfft(Pf * Pf, L)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        M = float(conv.max())
        cur_score = accepted_score
        best_f = f.copy()
        best_score = cur_score

    return _normalize(best_f), best_score


def _main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(a, score=None):
        nonlocal best_p, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = _score_fft(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
        n = int(a.size)
        old = best_by_n.get(n)
        if old is None or score < old[0]:
            best_by_n[n] = (float(score), a.copy())

    # Check global height_sequence_1
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                record(arr)
                print("Loaded height_sequence_1: n=%d, score=%.6f" % (arr.size, _score_fft(arr)))
        except Exception:
            pass

    # Quick baselines
    lengths = [128, 192, 256, 320, 384, 448, 512, 640, 768, 1024]
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        for init in [_greedy(n), _power(n, 0.45), _power(n, 0.6), _exp_decay(n, 1.0), _exp_decay(n, 2.0)]:
            record(init)

    print("Baseline best: %.6f (n=%d)" % (best_score, best_p.size if best_p is not None else 0))

    # Phase 1: Adam optimization with diverse starts
    # Focus on the most promising lengths
    focus_lengths = []
    for n in sorted(best_by_n.keys()):
        focus_lengths.append(n)
    if best_p is not None:
        bn = best_p.size
        if bn not in focus_lengths:
            focus_lengths.append(bn)
    focus_lengths = sorted(set(focus_lengths))[:8]

    inits_per_n = 4
    jobs = []
    for n in focus_lengths:
        if n < 64 or n > 2048:
            continue
        for i in range(inits_per_n):
            jobs.append((n, i))

    # Shuffle jobs to interleave lengths
    rng.shuffle(jobs)

    # Time allocation per job
    n_jobs = len(jobs)
    total_budget = max(60.0, deadline - time.monotonic())
    per_job = total_budget / max(1, n_jobs)
    per_job = max(5.0, min(per_job, 60.0))

    done = 0
    for n, i in jobs:
        if time.monotonic() >= deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 15.0:
            break

        # Choose initialization
        if i == 0:
            base = _greedy(n)
        elif i == 1:
            base = _power(n, 0.4 + 0.2 * rng.random())
        elif i == 2:
            if best_p is not None and best_p.size == n:
                base = _perturb(best_p, rng, 0.15)
            else:
                base = _mix(_greedy(n), np.ones(n), 0.3 + 0.4 * rng.random())
        else:
            if best_p is not None:
                base = _perturb(best_p, rng, 0.2 + 0.3 * rng.random())
            else:
                base = _exp_decay(n, 0.5 + 2.0 * rng.random())

        alloc = min(per_job, max(4.0, rem - 10.0))
        p_out, f_out = _adam(base, alloc, seed=int(seed) + n * 100 + i)
        record(p_out, f_out)
        done += 1

        if done % 10 == 0:
            print("Phase1: %d/%d jobs, best=%.6f (n=%d)" % (done, n_jobs, best_score, best_p.size))

    print("After Phase 1: best=%.6f (n=%d)" % (best_score, best_p.size if best_p is not None else 0))

    # Phase 2: Focused optimization on best length
    if best_p is not None:
        rem = deadline - time.monotonic()
        bn = best_p.size
        if rem > 120.0:
            # Multiple perturbation + re-optimize cycles
            n_cycles = max(2, int(rem / 60.0))
            for cyc in range(n_cycles):
                if time.monotonic() >= deadline:
                    break
                sigma = 0.08 + 0.15 * (cyc % 3)
                p_start = _perturb(best_p, rng, sigma)
                alloc = min(50.0, max(8.0, (deadline - time.monotonic()) / max(1, n_cycles - cyc) - 5.0))
                p_out, f_out = _adam(p_start, alloc, seed=int(seed) + 10000 + cyc * 7)
                record(p_out, f_out)
            print("After Phase 2: best=%.6f (n=%d)" % (best_score, best_p.size))

    # Phase 3: Coordinate polish
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30.0:
            alloc = min(120.0, rem - 15.0)
            p_out, f_out = _coord_polish(best_p, alloc)
            record(p_out, f_out)
            print("After Phase 3 (polish): best=%.6f (n=%d)" % (best_score, best_p.size))

    # Phase 4: LP direction finding
    if _scipy_opt is not None and best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 50.0:
            bn = best_p.size
            alloc = min(150.0, rem - 20.0)
            p_out, f_out = _lp_direction(best_p, alloc)
            record(p_out, f_out)
            # Follow up with Adam from LP result
            if f_out < best_score:
                rem2 = deadline - time.monotonic()
                if rem2 > 30.0:
                    p_out2, f_out2 = _adam(p_out, min(60.0, rem2 - 10.0), seed=int(seed) + 99999)
                    record(p_out2, f_out2)
            print("After Phase 4 (LP): best=%.6f (n=%d)" % (best_score, best_p.size))

    # Phase 5: Try a few more lengths with Adam
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 60.0:
            # Try slightly different lengths
            bn = best_p.size
            for n_trial in [bn - 16, bn + 16, bn - 32, bn + 32, bn // 2, bn * 2]:
                if time.monotonic() >= deadline:
                    break
                if n_trial < 64 or n_trial > 2048:
                    continue
                if n_trial in best_by_n:
                    continue
                base = _normalize(np.interp(np.arange(n_trial), np.arange(bn), best_p))
                alloc = min(40.0, max(5.0, (deadline - time.monotonic()) - 10.0))
                p_out, f_out = _adam(base, alloc, seed=int(seed) + n_trial)
                record(p_out, f_out)
            print("After Phase 5: best=%.6f (n=%d)" % (best_score, best_p.size))

    # Final: one more Adam run from best
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 20.0:
            p_out, f_out = _adam(_normalize(best_p), min(60.0, rem - 5.0), seed=int(seed) + 77777)
            record(p_out, f_out)

    if best_p is None:
        best_p = _greedy(384)
        best_score = _score_fft(best_p)

    # Validate top candidates
    candidates = [best_p]
    items = sorted(best_by_n.items(), key=lambda x: x[1][0])
    for n, (sc, p) in items[:5]:
        candidates.append(p)

    for cand in candidates:
        if deadline - time.monotonic() < 10.0:
            break
        seq = _scale_output(cand, 10.0)
        gscore = _try_eval(seq)
        if np.isfinite(gscore) and gscore < best_score:
            best_score = gscore
            best_p = np.asarray(cand, dtype=np.float64).ravel()
        print("Candidate n=%d: fft_score=%.6f, eval_score=%.6f" % (cand.size, _score_fft(cand), gscore))

    final_seq = _scale_output(best_p, 10.0)
    print("FINAL: n=%d, score=%.6f" % (best_p.size, best_score))
    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation."""
    try:
        return _main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main search: %s" % str(e))
        p = _greedy(384)
        return _scale_output(p, 10.0)
