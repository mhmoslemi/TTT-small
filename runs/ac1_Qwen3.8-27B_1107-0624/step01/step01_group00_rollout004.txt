    return c[:2*len(a)-1]

def _smoothed_grad(c, a, L, tau, n):
    """Smoothed gradient of max(c) w.r.t. a."""
    # Get top L indices
    if L >= len(c):
        order = np.arange(len(c))
    else:
        order = np.argpartition(-c, L-1)[:L]
    wc = c[order]
    w = np.exp(wc / max(tau, 1e-12))
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k, wk in zip(order, w):

        idx = np.arange(lo, hi + 1)
        src = k - idx
        grad[idx] += 2.0 * wk * a[src]
    return grad

    mean_g = np.dot(g, a)

def _topL_weights(c, L, tau):
    """Compute softmax weights over top-L entries of c."""
    n_c = len(c)

def _next_pow2(x):
    m = 1
    while m < x:
        m <<= 1
    return m


def _conv_sq(a, m):
    """Full linear convolution a*a via FFT."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * len(a) - 1]


def _topL_idx(c, L):
    """Get indices of top-L entries of c."""
    n_c = len(c)
    L = min(L, n_c)
    if L >= n_c:
        return np.arange(n_c)
    return np.argpartition(-c, L - 1)[:L]


def _compute_grad(a, c, L, tau, n):
    """Compute smoothed gradient of max(c) w.r.t. a using top-L softmax."""
    idx = _topL_idx(c, L)
    vals = c[idx]
    w = np.exp((vals - vals.max()) / max(tau, 1e-12))
    w = w / w.sum()

        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        p = np.arange(lo, hi + 1)
        src = k - p
        grad[p] += 2.0 * wk * a[src]
    return grad

        m_hat = m_buf / (1.0 - b1 ** t_step)
        v_hat = v_buf / (1.0 - b2 ** t_step)
        z = z - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic renormalization of z to prevent drift
        if t_step % 5000 == 0:
            z -= z.mean()
        
        # Update a
        z_shift = z - z.max()
        a_new = np.exp(z_shift)
        s = a_new.sum()
        if s < 1e-10:
            a = a.copy()  # keep old
        else:
            a = a_new / s
        
        # Adaptive schedule
        if t_step % 5000 == 0:
            tau = max(0.005, tau * 0.65)
            L = max(5, int(L * 0.7))
            current_lr = max(1e-7, current_lr * 0.8)
        
        if t_step % 1000 == 0:
            current_lr *= 0.98
        
        # Random restart if stuck
        if patience > max_patience:
            noise_scale = 0.2 * patience / max_patience
            z += rng.normal(0, noise_scale, n)
            z -= z.mean()
            patience = 0
            tau = max(tau, 0.1)
            L = max(L, 30)
            current_lr = max(current_lr, lr0 * 0.3)
        
        # Periodic evaluation
        if time.monotonic() - last_eval > 8.0:
            last_eval = time.monotonic()
            try:
                s = evaluate_sequence([float(x) for x in best_a])
                if s < best_score:
                    best_score = s
            except Exception:
                pass
            if time.monotonic() - last_print > 30.0:
                last_print = time.monotonic()
                print(f"    n={n} step={t_step} mc={mc:.8f} best_mc={best_mc:.8f} score={best_score:.6f} tau={tau:.4f} L={L}")
    
    # Final eval
    try:
        s_final = evaluate_sequence([float(x) for x in best_a])
        if s_final < best_score:
            best_score = s_final
    except Exception:
        pass
    
    return best_a, best_score, t_step


def _make_seeds(n, rng, provided_a=None):
    """Create diverse initial seeds in z-space (log-space)."""
    seeds = []
    
    # Uniform (z=0)
    seeds.append(np.zeros(n))
    
    # Linear ramp
    z = np.linspace(-3, 3, n)
    seeds.append(z)
    
    # Concave (parabolic)
    t = np.arange(n) / max(n - 1, 1)
    z = -6.0 * (t - 0.5) ** 2
    seeds.append(z)
    
    # Sinusoidal
    z = 1.5 * np.sin(2.0 * np.pi * np.arange(n) / n)
    seeds.append(z)
    
    # Two-bump
    z = np.zeros(n)
    z += np.exp(-0.5 * ((np.arange(n) - 0.25 * n) / (0.12 * n)) ** 2)
    z += np.exp(-0.5 * ((np.arange(n) - 0.75 * n) / (0.12 * n)) ** 2)
    z = z - z.mean()
    seeds.append(z)
    
    # Cosine window
    z = np.cos(np.pi * np.arange(n) / n) ** 2
    z = z - z.mean()
    seeds.append(z)
    
    # Random
    seeds.append(rng.normal(0, 1, n))
    seeds.append(rng.normal(0, 1, n))
    
    # High-freq modulation
    z = 0.8 * np.sin(2 * np.pi * 7 * np.arange(n) / n) + 0.4 * np.sin(2 * np.pi * 13 * np.arange(n) / n)
    seeds.append(z)
    
    # Provided sequence
    if provided_a is not None and len(provided_a) == n:
        pa = np.maximum(provided_a, 1e-12)
        z_prov = np.log(pa) - np.log(pa).max()
        seeds.append(z_prov)
    
    return seeds


def _refine(a0, n, m, t_end, seed):
    """Final fine-tuning phase with small learning rate."""
    z0 = np.log(np.maximum(a0, 1e-15))
    z0 = z0 - z0.max()
    return _run_opt(z0, n, m, t_end, tau0=0.02, L0=8, lr0=0.005, seed=seed, use_fft_grad=True)


def propose_candidate(budget_s=1800):
    """Search for the non-negative sequence minimizing 2*n*max(a*a)/(sum a)^2."""
    t_start = time.monotonic()
    t_end = t_start + max(0.0, float(budget_s)) - 5.0
    rng = np.random.default_rng(42)
    
    best_a_global = None
    best_n_global = 0
    best_score_global = float('inf')
    
    # Try to use provided sequence
    provided_a = None
    try:
        b = np.asarray(height_sequence_1, dtype=np.float64)
        b = np.clip(b, 0, 1000.0)
        if b.ndim == 1 and b.size >= 4 and np.all(np.isfinite(b)) and b.sum() > 0.01:
            provided_a = b / b.sum()
            print(f"Using provided sequence, n={b.size}, score=", end="")
            s = evaluate_sequence([float(x) for x in provided_a])
            print(f"{s:.6f}")
            best_score_global = s
            best_a_global = provided_a.copy()
            best_n_global = b.size
    except Exception as e:
        print(f"No provided sequence: {e}")
    
    # Phase 1: Explore different n values (first 50% of time)
    t_phase1_end = t_start + 0.5 * (t_end - t_start)
    
    n_values = [512, 768, 1024, 1536, 2048, 3072]
    # Prioritize n values closer to provided
    if provided_a is not None:
        n_prov = len(provided_a)
        n_values = sorted(n_values, key=lambda x: abs(x - n_prov))
        n_prov = len(provided_a)
        if n_prov not in n_values:
            n_values.insert(0, n_prov)
    
    n_per = (t_phase1_end - time.monotonic()) / len(n_values)
    
    for n_try in n_values:
        if time.monotonic() >= t_phase1_end:
            break
        if n_try > 5000:
            continue
            
        t_n_end = time.monotonic() + n_per
        m = _next_pow2(2 * n_try - 1)
        
        seeds = _make_seeds(n_try, rng, provided_a)
        n_seeds = len(seeds)
        per_seed = (t_n_end - time.monotonic()) / n_seeds
        
        if per_seed < 5.0:
            # Not enough time per seed, just do one quick run
            z0 = seeds[0]
            a_b, sc, _ = _run_opt(z0, n_try, m, t_n_end, tau0=0.15, L0=40, lr0=0.04, seed=n_try)
            if sc < best_score_global:
                best_score_global = sc
                best_a_global = a_b.copy()
                best_n_global = n_try
                print(f"  n={n_try}: score={sc:.6f}")
            continue
        
        best_n_score = float('inf')
        best_n_a = None
        
        for si, z0 in enumerate(seeds):
            seed_end = time.monotonic() + per_seed
            if seed_end <= time.monotonic():
                break
            a_b, sc, steps = _run_opt(z0, n_try, m, seed_end, tau0=0.15, L0=50, lr0=0.04, seed=si * 7 + n_try)
            if sc < best_n_score:
                best_n_score = sc
                best_n_a = a_b.copy()
        
        if best_n_a is not None and best_n_score < best_score_global:
            best_score_global = best_n_score
            best_a_global = best_n_a.copy()
            best_n_global = n_try
            print(f"  n={n_try}: best score={best_n_score:.6f} (global best)")
        else:
            print(f"  n={n_try}: best score={(best_n_score if best_n_a is not None else float('inf')):.6f}")
    
    print(f"\nAfter Phase 1: best_n={best_n_global}, best_score={best_score_global:.6f}, time_left={t_end-time.monotonic():.0f}s")
    
    # Phase 2: Focus on best n with more seeds and longer runs (next 30%)
    t_phase2_end = t_start + 0.8 * (t_end - t_start)
    if best_a_global is not None and time.monotonic() < t_phase2_end:
        n_focus = best_n_global
        m_focus = _next_pow2(2 * n_focus - 1)
        
        # Use the best found as a seed, plus some new random ones
        seeds2 = []
        za = np.log(np.maximum(best_a_global, 1e-15))
        za = za - za.max()
        seeds2.append(za)
        
        # Perturbations of best
        for _ in range(4):
            z_pert = za + rng.normal(0, 0.3, n_focus)
            seeds2.append(z_pert)
        
        # Fresh random
        seeds2.append(rng.normal(0, 1, n_focus))
        seeds2.append(rng.normal(0, 1, n_focus))
        
        # Ramp variants
        for amp in [1.0, 2.0, 3.0]:
            z_ramp = amp * np.linspace(-1, 1, n_focus)
            seeds2.append(z_ramp)
        
        per_seed2 = (t_phase2_end - time.monotonic()) / len(seeds2)
        
        for si, z0 in enumerate(seeds2):
            seed_end = time.monotonic() + per_seed2
            if seed_end <= time.monotonic():
                break
            a_b, sc, steps = _run_opt(z0, n_focus, m_focus, seed_end,
                                       tau0=0.1, L0=30, lr0=0.03, seed=1000 + si)
            if sc < best_score_global:
                best_score_global = sc
                best_a_global = a_b.copy()
                print(f"  Phase2 seed {si}: score={sc:.6f} (NEW GLOBAL BEST)")
    
    print(f"\nAfter Phase 2: best_score={best_score_global:.6f}, time_left={t_end-time.monotonic():.0f}s")
    
    # Phase 3: Fine-tuning (last 20%)
    if best_a_global is not None and time.monotonic() < t_end - 10:
        n_final = best_n_global
        m_final = _next_pow2(2 * n_final - 1)
        
        a_fin, sc_fin, _ = _refine(best_a_global, n_final, m_final, t_end, seed=9999)
        if sc_fin < best_score_global:
            best_score_global = sc_fin
            best_a_global = a_fin.copy()
            print(f"  Phase3 refine: score={sc_fin:.6f} (NEW GLOBAL BEST)")
    
    # Final validation
    if best_a_global is None:
        out = [1.0] * 512
    else:
        out_a = np.maximum(best_a_global, 0.0)
        out_a = np.minimum(out_a, 1000.0)
        if out_a.sum() < 0.01:
            out_a = np.ones(len(out_a))
        # Scale so sum is reasonable (scale invariant, but keep sum ~ n for stability)
        out = [float(x) for x in out_a]
    
    try:
        final_sc = evaluate_sequence(out)
        print(f"\nFINAL: n={len(out)}, score={final_sc:.6f}")
    except Exception as e:
        print(f"FINAL eval error: {e}")
    
    return out
