        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)

def scale_output(p, target_sum=10.0):

def smooth_max_and_grad(f, alpha):
    """
    Compute the smooth max of f*f (log-sum-exp approximation) and its gradient.
    f is normalized to sum to 1.
    Returns (objective, gradient_wrt_f)
    """
    n = f.size
    L = next_power_of_two(2 * n - 1)
    F = np.fft.rfft(f, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    
    c_max = c.max()
    shifted = alpha * (c - c_max)
    exp_sum = np.exp(shifted).sum()
    log_sum_exp = c_max + np.log(exp_sum) / alpha
    obj = 2.0 * n * log_sum_exp
    
    w = np.exp(shifted) / exp_sum
    
    # Gradient: 2 * (f cross-correlate w) at lags 0..n-1
    # = 2 * convolve(f, w_rev)[n-1 : 2n-1][::-1]
    w_rev = w[::-1]
    L2 = next_power_of_two(3 * n - 2)
    Ff = np.fft.rfft(f, L2)
    Fw = np.fft.rfft(w_rev, L2)
    conv = np.fft.irfft(Ff * Fw, L2)[:3 * n - 2]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    
    return obj, grad


def softmax_lobj(x, alpha):
    """Objective function in x-space for scipy L-BFGS."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    f = e / e.sum()
    obj, _ = smooth_max_and_grad(f, alpha)
    return obj


def softmax_lgrad(x, alpha):
    """Gradient in x-space for scipy L-BFGS."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    f = e / e.sum()
    _, grad_f = smooth_max_and_grad(f, alpha)
    return f * (grad_f - np.dot(f, grad_f))


def softmax_objgrad(x, alpha):
    """Combined objective and gradient."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    f = e / e.sum()
    obj, grad_f = smooth_max_and_grad(f, alpha)
    grad_x = f * (grad_f - np.dot(f, grad_f))
    return obj, grad_x
