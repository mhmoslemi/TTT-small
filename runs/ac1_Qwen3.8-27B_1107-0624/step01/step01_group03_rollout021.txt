```python
def subgradient_step(s, eta, n_top=1):
    """One subgradient step to minimize max(conv(s,s)) with sum(s)=1."""
    n = len(s)
    conv = np.convolve(s, s)
    # Find top-K peaks
    top_k = np.sort(conv)[-n_top:][::-1]  # descending
    M = top_k[0]
    # Active set: positions where conv[k] >= M * 0.99 (or top K)
    active = np.where(conv >= top_k[-1])[0]
    
    # Subgradient: for each active k, grad[i] += s[k-i] (if valid)
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += s[j]
    
    # Update
    s_new = s - eta * grad
    s_new = np.maximum(s_new, 0)
    s_new = s_new / s_new.sum()
    return s_new
```

This is O(n * K) per step where K is the number of active peaks. With FFT for the convolution, the total is O(n log n + nK).

For the step size, we can use:
- Fixed eta (e.g., 0.01)
- Polyak step: eta = (f(s) - f_opt_lower_bound) / ||g||^2
- Backtracking line search

```python
def lp_direction(s, M, top_k=1, time_limit=3.0):
    """Solve LP to find direction b that maximizes sum(b) subject to (s*b)[k] <= M."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    # For top-K constraint: we can use a two-level constraint
    # But for simplicity, let's just use M for all (the original approach)
    # Later improvement: use M for top-K and M*1.05 for the rest
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception as e:
        print(f"LP error: {e}")
    return None

def line_search_1d(s, b, best_sc):
    """Find best t in [0,1] that minimizes fast_score((1-t)*s + t*b)."""
    def obj(t):
        cand = (1.0 - t) * s + t * b
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 1e-15:
            return 1e10
        return fast_score(cand / cand.sum())
    
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-5})
    return res.x, res.fun
```
