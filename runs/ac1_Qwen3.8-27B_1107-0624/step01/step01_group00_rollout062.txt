import math
import numpy as np

def _proj_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _conv_full(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    m = 1
    while m < 2*n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def _smoothed_max_grad(a, c, tau):
    """Compute gradient of log-sum-exp(c/tau) w.r.t. a.
    grad[i] = (1/tau) * sum_j w[i+j] * a[j]
    where w = softmax(c/tau)

    n = len(a)
    # Numerically stable softmax
    c_max = c.max()
    log_w = c / tau - c_max / tau
    w = np.exp(log_w)

Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences using
FISTA with log-sum-exp smoothing of the max-convolution objective.
Multi-seed search with adaptive hyperparameters and periodic restarts.
"""

import time
import math
import numpy as np

def proj_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_self(a):
    """Full self-convolution a*a via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    m = 1
    while m < 2*n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def smooth_max_grad(a, c, tau):
    """
    Gradient of log(sum(exp(c/tau))) w.r.t. a.
    grad[i] = (1/tau) * sum_{j=0}^{n-1} w[i+j] * a[j]  (where i+j < 2n-1)
    w = softmax(c/tau)
    Computed via FFT: grad[i] = (1/tau) * (w * a_rev)[n-1+i]
    """
    n = len(a)
    # Stable softmax
    cm = c.max()
    w = np.exp((c - cm) / tau)
    w = w / w.sum()
    
    # Cross-correlation via FFT
    a_rev = a[::-1]
    m = 1
    while m < (2*n-1) + n - 1:
        m <<= 1
    fw = np.fft.rfft(w, m)
    fa = np.fft.rfft(a_rev, m)
    corr = np.fft.irfft(fw * fa, m)
    return (1.0 / tau) * corr[n-1:2*n-1].copy()

def l2_grad(a, c):
    """Gradient of sum(c^2) w.r.t. a. = 2 * (c * a_rev)[n-1:2n-1]"""
    n = len(a)
    a_rev = a[::-1]
    m = 1
    while m < (2*n-1) + n - 1:

    fc = np.fft.rfft(c, m)
    fa = np.fft.rfft(a_rev, m)
    corr = np.fft.irfft(fc * fa, m)
    return 2.0 * corr[n-1:2*n-1].copy()

    lr = lr0

    best_mc = conv_self(a).max()
    best_score = float('inf')
    
    step = 0

    best_off_a = a.copy()
    best_off_score = float('inf')
    
    # Pre-compute m for efficiency
    m_conv = 1
    while m_conv < 2*n - 1:
        m_conv <<= 1
    
    m_grad = 1
    while m_grad < 3*n - 2:
        m_grad <<= 1
    
    while time.monotonic() < time_budget:
        step += 1
        
        # Compute current c and objective
        c = conv_self(a)
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
        
        # Adaptive schedules
        if step % 1000 == 0:

            lr = max(1e-6, lr * 0.92)
            mom = min(0.95, mom + 0.002)
        
        # Nesterov/FISTA step
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_nest * t_nest))
        y = a + ((t_nest - 1.0) / t_new) * (a - a_prev)

        if ys > 1e-12:
            y = y / ys

            y = a.copy()

        # Gradient step + projection
        a_new = y - lr * g
        a_new = proj_simplex(a_new, 1.0)
        
        a_prev = a.copy()
        a = a_new

def conv_self(a, m=None):

    n = len(a)

        while m < 2 * n - 1:

    """Gradient of smoothed max over top-L lags, via FFT cross-correlation.

    tau = tau0

    m_conv = 1
    while m_conv < 2 * n - 1:
        m_conv <<= 1
    m_grad = 1
    while m_grad < 3 * n - 2:
        m_grad <<= 1

    best_a = a.copy()
    best_mc = conv_self(a, m_conv).max()

    step = 0
    last_eval = time.monotonic()

    while time.monotonic() < time_budget:
        step += 1
        c = conv_self(a, m_conv)
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()

        # Adaptive schedules
        if step % 1500 == 0:
            tau = max(1e-6, tau * 0.78)
            L = max(5, int(L * 0.72))
            mom = min(0.96, mom + 0.004)
        if step % 3000 == 0:
            lr = max(1e-5, lr * 0.88)

        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_nest * t_nest))
        beta = (t_nest - 1.0) / t_new
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        y = y / ys if ys > 1e-12 else a.copy()

        # Gradient
        c_y = conv_self(y, m_conv)
        if use_full_softmax:
            g = full_softmax_grad(c_y, y, tau, m_grad, n)
        else:
            g = topL_grad(c_y, y, L, tau, m_grad, n)

        # Step
        a_new = y - lr * g
        a_new = proj_simplex(a_new, 1.0)

        a_prev = a
        a = a_new
        t_nest = t_new

        # Restart with perturbation every 6000 steps
        if step % 6000 == 0:
            pert = rng.normal(0, 0.06, n)
            a_p = best_a * np.exp(pert)
            a_p = np.maximum(a_p, 0)
            if a_p.sum() > 1e-12:
                a = a_p / a_p.sum()
                a = proj_simplex(a, 1.0)
                a_prev = a.copy()
                t_nest = 1.0

    return best_a, best_mc


def make_seeds(n, rng, provided=None):
    """Generate diverse initial seeds."""
    seeds = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    seeds.append(np.full(n, 1.0 / n))

    # Provided
    if provided is not None and len(provided) == n:
        p = np.abs(provided).astype(np.float64)
        ps = p.sum()
        if ps > 1e-12:
            seeds.append(p / ps)
            seeds.append(p[::-1] / p.sum())

    # Sinusoidal shapes
    for fq in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.6]:
            v = 1.0 + amp * np.cos(2 * math.pi * fq * idx / n)
            v = np.maximum(v, 0.01)
            seeds.append(v / v.sum())

    # Gaussian bumps at different positions
    for frac in [0.3, 0.5, 0.7]:
        c = frac * n
        w = 0.3 * n
        g = np.exp(-0.5 * ((idx - c) / w) ** 2)
        seeds.append(g / g.sum())

    # Power-law edge weighting
    d = np.abs(idx - n / 2) / (n / 2)
    for p in [1.5, 3.0]:
        e = 1.0 + 0.5 * d ** p
        seeds.append(e / e.sum())
        e2 = 1.0 + 0.5 * (1 - d) ** p
        seeds.append(e2 / e2.sum())

    # Random
    for _ in range(4):
        r = np.abs(rng.normal(0, 1, n)) + 0.05
        seeds.append(r / r.sum())

    return seeds


def resize_seq(p, n):
    """Resize sequence p to length n via linear interpolation."""
    m = len(p)
    if m == n:
        return p.copy()
    if m == 1:
        return np.full(n, p[0])
    step = (m - 1) / max(1, n - 1)
    idx = np.arange(n) * step
    idx0 = idx.astype(int)
    idx1 = np.minimum(idx0 + 1, m - 1)
    frac = idx - idx0
    return p[idx0] * (1 - frac) + p[idx1] * frac


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(a*a)/(sum(a))^2.
    Multi-seed FISTA with top-L smoothing, parameter racing, and restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    t_end = t_start + max(0, float(budget_s)) - 3.0

    # Get provided sequence
    provided = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.ndim == 1 and p.size >= 4 and np.isfinite(p).all() and p.sum() > 0:
            provided = p
    except Exception:
        pass

    n_default = provided.size if provided is not None else 512

    best_a = None
    best_score = float('inf')

    # Parameter configurations to race
    configs = [
        # (L0, tau0, lr0, mom0, use_full_softmax)
        (60, 0.15, 0.15, 0.88, False),
        (30, 0.08, 0.10, 0.90, False),
        (200, 0.25, 0.12, 0.85, False),
        (50, 0.10, 0.08, 0.92, True),
        (100, 0.12, 0.10, 0.90, False),
        (40, 0.06, 0.06, 0.93, True),
    ]

    # Try the default length first (most time), then one other
    lengths = [n_default]
    if n_default > 300:
        lengths.append(n_default * 2)
    elif n_default < 2000:
        lengths.append(min(2048, n_default * 3))

    for li, n in enumerate(lengths):
        if time.monotonic() > t_end - 15:
            break

        if li == 0:
            time_alloc = (t_end - time.monotonic()) * 0.65
        else:
            time_alloc = (t_end - time.monotonic()) * 0.35

        if time_alloc < 15:
            continue

        print(f"=== n={n}, time={time_alloc:.0f}s ===")

        # Prepare seeds
        if provided is not None and n == n_default:
            seeds = make_seeds(n, rng, provided)
        elif provided is not None and n != n_default:
            resized = resize_seq(provided, n)
            seeds = make_seeds(n, rng, resized)
        else:
            seeds = make_seeds(n, rng, None)

        n_seeds = len(seeds)
        n_cfgs = len(configs)

        # Phase 1: Race - short runs of each (seed, config) combo
        race_time = min(20.0, time_alloc / (n_seeds * n_cfgs + 1) * 1.5)
        race_time = max(5.0, race_time)
        print(f"  Phase 1 (race): {race_time:.1f}s per trial, "
              f"{n_seeds*n_cfgs} trials total")

        results = []  # (score, seed_idx, cfg_idx, best_a)

        t_race_start = time.monotonic()
        t_race_end = t_race_start + min(race_time * 2, time_alloc * 0.25)

        trial_count = 0
        for si in range(n_seeds):
            for ci in range(n_cfgs):
                if time.monotonic() > t_race_end:
                    break
                L0, tau0, lr0, mom0, use_fs = configs[ci]
                a_best, mc_best = run_opt_a0(
                    seeds[si], t_race_end, L0, tau0, lr0, mom0, use_fs, rng
                )
                # Compute internal score
                s_internal = 2.0 * n * mc_best
                results.append((s_internal, si, ci, a_best.copy()))
                trial_count += 1

        if not results:
            # Fallback: just run one config
            a_best, mc_best = run_opt_a0(
                seeds[0], t_end - 5, 60, 0.15, 0.15, 0.90, False, rng
            )
            try:
                sc = float(evaluate_sequence([float(x) for x in a_best]))
            except Exception:
                sc = float('inf')
            if sc < best_score:
                best_score = sc
                best_a = a_best.copy()
            continue

        # Sort by internal score, keep top K
        results.sort(key=lambda x: x[0])
        top_K = min(8, len(results))
        top_results = results[:top_K]

        print(f"  Phase 1 done in {time.monotonic()-t_race_start:.1f}s, "
              f"best internal={top_results[0][0]:.6f}")

        # Phase 2: Continue top-K with longer time
        time_for_phase2 = t_end - time.monotonic() - 5
        time_per_trial = time_for_phase2 / top_K

        for rank, (sc_int, si, ci, a_init) in enumerate(top_results):
            if time.monotonic() > t_end - 5:
                break
            if time_per_trial < 3:
                break

            L0, tau0, lr0, mom0, use_fs = configs[ci]
            # Slightly different params for diversity in phase 2
            tau0_p2 = tau0 * 0.7
            L0_p2 = max(10, int(L0 * 0.8))

            a_best, mc_best = run_opt_a0(
                a_init, t_end - 5, L0_p2, tau0_p2, lr0, mom0, use_fs, rng
            )

            # Official score
            try:
                sc = float(evaluate_sequence([float(x) for x in a_best]))
            except Exception:
                sc = float('inf')

            if sc < best_score:
                best_score = sc
                best_a = a_best.copy()
                print(f"  Phase 2 rank {rank}: OFFICIAL score={sc:.6f} "
                      f"(n={n}, seed={si}, cfg={ci})")

    # Final: also try a very long single run from the absolute best found
    if best_a is not None and time.monotonic() < t_end - 30:
        n_best = len(best_a)
        a_init = best_a.copy()
        L0 = 25
        tau0 = 0.04
        lr0 = 0.05
        mom0 = 0.93
        a_final, _ = run_opt_a0(a_init, t_end - 3, L0, tau0, lr0, mom0, False, rng)
        try:
            sc = float(evaluate_sequence([float(x) for x in a_final]))
            if sc < best_score:
                best_score = sc
                best_a = a_final.copy()
                print(f"  Final polish: OFFICIAL score={sc:.6f}")
        except Exception:
            pass

    # Ensure valid output
    if best_a is None:
        n_out = n_default
        best_a = np.full(n_out, 1.0)

    out = np.clip(best_a.astype(np.float64), 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.full(len(out), 1.0)

    try:
        final_score = float(evaluate_sequence([float(x) for x in out]))
        print(f"\nFINAL SCORE = {final_score:.6f} (n={len(out)})")
    except Exception:
        print(f"\nFINAL SCORE = unknown (n={len(out)})")

    return [float(x) for x in out]
