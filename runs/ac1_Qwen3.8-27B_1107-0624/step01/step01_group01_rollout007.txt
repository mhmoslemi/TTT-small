    k = 1
    while k < n:

"""
Multi-scale optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Start from parent, adaptive soft-max gradient descent in logit space,
multi-length escalation (128->256->512->1024), variance-aware objective,
and simulated annealing escape for local minima.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _soft_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _corr_weights(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def _logit_anneal(p0, deadline, rng, lr=0.08, reg=1e-6, var_weight=0.0,
                  beta_init_frac=0.15, beta_decay=0.72,
                  no_imp_mult=5, name=""):
    """
    Logit-space Adam with adaptive beta and optional variance penalty.
    Returns (best_p, best_score).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = _soft_max_weights(c, beta)
        gp = 2.0 * _corr_weights(w, p)

        if var_weight > 0.0:
            c_center = c[n - 1:2 * n - 1] if c.size >= n else c
            c_mean = float(c_center.mean())
            # Approximate gradient of variance of center convolution
            # d/dp_i var(c_center) ~ 2*(c_center - c_mean) * (contributions from p_i)
            # Simplified: penalize deviation of p from its mean
            gp += var_weight * (p - p.mean())

        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.005, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Adaptive beta decay
        if adam_t % 40 == 0:
            beta = max(1e-12, beta * beta_decay)
            # Check gap between top-2 values for faster sharpening
            if c.size >= 2:
                sorted_idx = np.argsort(c)[::-1]
                if sorted_idx[0] != sorted_idx[1]:
                    gap = c[sorted_idx[0]] - c[sorted_idx[1]]
                    if gap < beta * 0.05:
                        beta = max(1e-12, beta * 0.4)

        # Restart
        no_imp_limit = max(150, no_imp_mult * n)
        if no_imp > no_imp_limit:
            if best_score < 1.52:
                noise_scale = 0.1
            elif best_score < 1.55:
                noise_scale = 0.2
            elif best_score < 1.60:
                noise_scale = 0.35
            else:
                noise_scale = 0.5
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_init_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _simulated_anneal(p0, deadline, rng, n_iter_max=None):
    """
    Patch-based simulated annealing for local escape.
    Returns (best_p, best_score).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    p = p0.copy()
    c = _full_convolve_fft(p)
    maxc = float(c.max())
    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)

    T = 0.05  # initial temperature
    T_min = 0.001
    cool_rate = 0.9995
    patch_frac = 0.08  # patch size as fraction of n
    min_patch = 5
    max_patch = 80

    n_iter = 0
    max_iter = n_iter_max if n_iter_max is not None else 500000

    while time.monotonic() < deadline and n_iter < max_iter:
        n_iter += 1
        # Pick a random patch
        ps = int(np.clip(rng.integers(min_patch, max_patch + 1), min_patch, n // 2))
        start = int(rng.integers(0, n - ps + 1))
        end = start + ps

        # Propose perturbation
        noise_scale = T * 0.5
        p_new = p.copy()
        p_new[start:end] += rng.normal(0, noise_scale, ps)
        p_new = np.maximum(p_new, 1e-12)
        p_new = p_new / p_new.sum()

        # Score
        c_new = _full_convolve_fft(p_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)

        delta = score_new - best_score
        if delta < 0:
            p = p_new
            best_p = p_new.copy()
            best_score = score_new
            c = c_new
            T = max(T_min, T * 0.99)
        else:
            # Accept with probability
            if T > T_min:
                accept_prob = np.exp(-delta / T)
                if rng.random() < accept_prob:
                    p = p_new
                    c = c_new
            T = max(T_min, T * cool_rate)

        if n_iter % 10000 == 0:
            print(f"  SA: iter={n_iter}, T={T:.6f}, best={best_score:.10f}")

    return best_p, best_score


def _multi_start_logit(p0, deadline, rng, lr=0.08, n_starts=3, var_weight=0.0):
    """
    Run logit_anneal from multiple perturbations of p0, return best.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    best_p = p0.copy()
    best_score = _internal_score(p0)

    scale_per_start = 512.0 / n_starts
    for i in range(n_starts):
        if time.monotonic() >= deadline:
            break
        per_end = min(deadline, time.monotonic() + scale_per_start)
        if i == 0:
            p_start = p0
        else:
            z = np.log(np.maximum(p0, 1e-12))
            z -= z.mean()
            noise = 0.3 * (1.0 + i * 0.3)
            z = z + noise * rng.normal(size=p0.size)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_start = _softmax_from_z(z)
        p_best, sc = _logit_anneal(p_start, per_end, rng, lr=lr, var_weight=var_weight)
        if sc < best_score:
            best_score = sc
            best_p = p_best.copy()

    return best_p, best_score

    # Take best candidate, scale up progressively
    phase2_end = min(deadline, t_start + 900.0)
    scale_up_lengths = [256, 512, 1024]

    current_best_p = best_seq.copy() if best_seq is not None else _make_linear_profile(256, 1.5)
    current_best_sc = best_score

    for target_n in scale_up_lengths:
        if time.monotonic() > phase2_end - 60.0:
            break
        # Interpolate current best to target length
        if current_best_p.size != target_n:
            p_interp = _interp_profile(current_best_p, target_n)
        else:
            p_interp = current_best_p.copy()
        if p_interp is None:
            continue

        sc_interp = _internal_score(p_interp)
        print(f"  Escalating to n={target_n}: interp score={sc_interp:.10f}")
        update_best(p_interp, sc_interp, f"interp n={target_n}")

        # Polish at target length with multiple strategies
        per_end_1 = min(phase2_end - 30.0, time.monotonic() + 60.0)
        p_b1, sc_b1 = _logit_anneal(p_interp, per_end_1, rng, lr=0.06, var_weight=0.0,
                                     beta_init_frac=0.10, beta_decay=0.70, no_imp_mult=4)
        update_best(p_b1, sc_b1, f"polish1 n={target_n}")

        if time.monotonic() < phase2_end - 30.0:
            per_end_2 = min(phase2_end - 30.0, time.monotonic() + 45.0)
            p_b2, sc_b2 = _logit_anneal(p_b1, per_end_2, rng, lr=0.04, var_weight=0.01,
                                         beta_init_frac=0.08, beta_decay=0.75, no_imp_mult=3)
            update_best(p_b2, sc_b2, f"polish2 n={target_n}")

        current_best_p = best_seq.copy() if best_seq is not None else p_b1
        current_best_sc = best_score

    print(f"Phase 2: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # --- Phase 3: Aggressive multi-restart at best length ---
    phase3_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        n_best = best_seq.size
        n_starts = 5
        per_start = min(60.0, (phase3_end - time.monotonic()) / max(1, n_starts))

        for i in range(n_starts):
            if time.monotonic() > phase3_end:
                break
            per_end = min(phase3_end, time.monotonic() + per_start)
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            noise = 0.15 * (1.0 + 0.5 * i)
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_start = _softmax_from_z(z)
            p_try, sc_try = _logit_anneal(p_start, per_end, rng, lr=0.05, var_weight=0.0,
                                           beta_init_frac=0.08, beta_decay=0.72, no_imp_mult=3)
            update_best(p_try, sc_try, f"restart{i} n={n_best}")

    print(f"Phase 3: best={best_score:.10f}")

    # --- Phase 4: Simulated annealing on best ---
    phase4_end = min(deadline, t_start + 1680.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        print(f"  Starting SA on best (n={best_seq.size}, score={best_score:.10f})")
        p_sa, sc_sa = _simulated_anneal(best_seq, phase4_end, rng, n_iter_max=200000)
        update_best(p_sa, sc_sa, "SA")
        print(f"  SA done: {sc_sa:.10f}")

    # --- Phase 5: Final multi-start at best length with very small lr ---
    phase5_end = deadline
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        per_final = min(50.0, (phase5_end - time.monotonic()) / 2.0)
        p_fin, sc_fin = _logit_anneal(best_seq, time.monotonic() + per_final,
                                       rng, lr=0.02, var_weight=0.0,
                                       beta_init_frac=0.05, beta_decay=0.80,
                                       no_imp_mult=3)
        update_best(p_fin, sc_fin, "final_polish")

    # --- Final: exact scoring and validation ---
    if best_seq is not None:
        final_seq = best_seq.copy()
        final_n = final_seq.size

        # Exact score
        if final_n <= 2048:
            sc_exact = _true_score(final_seq)
            if np.isfinite(sc_exact):
                print(f"Exact score: {sc_exact:.10f}")

        # Try to call evaluate_sequence for authoritative score
        if final_n <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval) and sc_eval < best_score:
                    best_score = sc_eval
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception:
                pass

        # Ensure valid output
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(256, 1.5)

        print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(256, 1.5)
    print(f"FALLBACK: n={256}")
    return [float(x) for x in fallback]
