The problem: minimize max_k (a*a)(k) subject to a >= 0, sum(a) = 1.

This is a non-convex problem. But we can use the following reformulation:
minimize t
subject to: sum_{i+j=k} a_i * a_j <= t for all k
           a >= 0
           sum(a) = 1

The constraints are quadratic. We can linearize them using the fact that a_i * a_j <= (a_i^2 + a_j^2)/2, but this gives an upper bound, not an exact reformulation.

Another approach: fix the peak location k*. Then the constraint is sum_i a_i * a_{k*-i} <= t. This is a quadratic form: a^T Q_{k*} a <= t where Q_{k*} is a permutation matrix (reversal). The problem becomes:
minimize t
subject to: a^T Q_k a <= t for all k
           a >= 0, sum(a) = 1

This is a QCQP. We can solve it via:
- Semidefinite relaxation (too expensive for n=512)
- Penalty methods
- Alternating minimization (fix t, solve for a; fix a, solve for t)

import numpy as np

import scipy.optimize

# Helper functions at top level

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c

def smooth_max_val(c, beta):
    """Return smooth max value."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    return mval + beta * np.log(s)

def smooth_max_weights(c, beta):
    """Return weights for smooth max."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    return ex / s

def _next_pow2(n):
    k = 1
    while k < n:

    n = a.size

    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _score_grad_z(z, beta):
    """Compute score and gradient in z-space (logit-space). a = softmax(z)."""
    a = _softmax_from_z(z)
    n = a.size
    c = _full_convolve_fft(a)
    w = _smooth_max_weights(c, beta)
    L = _smooth_max_val(c, beta)
    h = _corr_weights(w, a)
    # d(score)/d(a_i) = 4*n*h[i] (with sum(a)=1)
    # d(score)/d(z_j) = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    grad_a = 4.0 * n * h
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    score = 2.0 * n * L
    return score, grad_z


def _adam_logit(z0, deadline, rng, lr=0.12, beta_init=None, reg=1e-7):
    """Adam optimization in logit space for the smooth-max objective."""
    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    n = z.size
    a = _softmax_from_z(z)
    c = _full_convolve_fft(a)
    maxc = float(c.max())
    if beta_init is None:
        beta = max(1e-10, 0.3 * maxc)
    else:
        beta = beta_init

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    score0, _ = _score_grad_z(z, beta)
    best_z = z.copy()
    best_score = float(score0)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        score, gz = _score_grad_z(z, beta)
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Add small regularization to encourage sparsity-like behavior
        a = _softmax_from_z(z)
        gz = gz + reg * a * n

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        step = lr * mh / (np.sqrt(vh) + eps)
        z = z - step

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -80.0, 80.0)
        z -= float(np.mean(z))

        a = _softmax_from_z(z)
        c = _full_convolve_fft(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.03, lr * 0.96)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 80 == 0:
            beta = max(1e-12, beta * 0.72)

        if no_imp > max(400, 8 * n):
            z = best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -80.0, 80.0)
            z -= float(np.mean(z))
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            beta = max(1e-10, 0.3 * float(_full_convolve_fft(_softmax_from_z(z)).max()))
            no_imp = 0

    return best_z, best_score


def _lbgfs_logit(z0, beta, max_iter=100, ftol=1e-12):
    """L-BFGS-B optimization in logit space."""
    def obj_grad(z):
        s, g = _score_grad_z(z, beta)
        return s, g

    n = len(z0)
    res = scipy.optimize.minimize(
        lambda z: obj_grad(z)[0],
        z0,
        jac=lambda z: obj_grad(z)[1],
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': ftol, 'maxfun': max_iter * 2}
    )
    a = _softmax_from_z(res.x)
    c = _full_convolve_fft(a)
    score = float(2.0 * n * c.max())
    return res.x, score


def _peak_shave(p, n_iters=200, rng=None):
    """Iteratively reduce convolution peak by redistributing mass."""
    if rng is None:
        rng = np.random.default_rng(0)
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.maximum(p, 1e-10)
    p = p / p.sum()
    n = p.size
    best_p = p.copy()
    best_score = _internal_score(p)

    for it in range(n_iters):
        c = _full_convolve_fft(p)
        peak_idx = int(np.argmax(c))
        # Find which a_i contribute most to the peak
        # (p*p)(peak_idx) = sum_i p_i * p_{peak_idx - i}
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] = p[j]
        # Reduce top contributors, add to bottom non-contributors
        n_top = max(2, n // 8)
        n_bot = max(2, n // 16)
        top_idx = np.argsort(contrib)[-n_top:]
        bot_idx = np.argsort(contrib)[:n_bot]
        # Only reduce those with significant contribution
        reduce_mask = contrib[top_idx] > np

mean(contrib) * 5
        if not np.any(reduce_mask):
            top_idx = np.argsort(contrib)[-n_top:]
        else:
            top_idx = top_idx[reduce_mask]
        if len(top_idx) == 0:
            continue
        total_reduce = 0.0
        for i in top_idx:
            amt = 0.1 * p[i]
            p[i] -= amt
            total_reduce += amt
        # Redistribute to bottom positions
        n_bot = len(bot_idx)
        if n_bot == 0:
            bot_idx = np.arange(n)
        for j in bot_idx:
            p[j] += total_reduce / n_bot
        p = np.maximum(p, 1e-12)
        p = p / p.sum()
        sc = _internal_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
        else:
            p = best_p.copy()

    return best_p, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quad_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_deterministic_profiles(n):
    profs = []
    for eta in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
        profs.append(_make_linear_profile(n, eta))
    for a1, a2 in ((1.0, -0.5), (1.5, -1.0), (0.5, -0.3), (2.0, -1.5), (1.0, 0.5), (-1.0, -0.5)):
        profs.append(_make_quad_profile(n, a1, a2))
    # Sinc-like
    t = np.linspace(-1.0, 1.0, n)
    a = np.sinc(2.0 * t) ** 2
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    # Gaussian
    t = np.linspace(-1.0, 1.0, n)
    a = np.exp(-2.0 * t * t)
    profs.append(a / a.sum())
    # Cosine window
    t = np.linspace(0.0, np.pi, n)
    a = 0.5 + 0.5 * np.cos(t)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    return profs


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for nonnegative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phases: diverse init -> Adam at n=128 -> interpolate to 512 -> Adam at 512 -> L-BFGS-B -> peak shave.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []  # list of (score, np.array)

    def _update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
        if p is not None and sc is not None and np.isfinite(sc):
            p_arr = np.asarray(p, dtype=np.float64).copy()
            if len(candidates) < 20:
                candidates.append((float(sc), p_arr))
                candidates.sort(key=lambda x: x[0])
            else:
                worst = max(range(len(candidates)), key=lambda i: candidates[i][0])
                if float(sc) < candidates[worst][0]:
                    candidates[worst] = (float(sc), p_arr)
                    candidates.sort(key=lambda x: x[0])

    # Initialize from parent if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pn = _normalize_nonneg(parent)
            if pn is not None:
                sc = _internal_score(pn)
                _update(pn, sc)
                print(f"parent: n={pn.size}, score={sc:.10f}")
    except Exception:
        pass

    # Fallback initializations
    for n_f in (128, 256, 512):
        p = _make_linear_profile(n_f, 2.0)
        sc = _internal_score(p)
        _update(p, sc)

    print(f"initial best={best_score:.10f}")

    # === Phase 1: Optimize at n=128 ===
    n1 = 128
    phase1_end = min(deadline, t_start + 480.0)

    # Gather initial profiles at n=128
    starts_128 = []
    # From parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pn = _normalize_nonneg(parent)
            if pn is not None:
                p128 = _interpolate_profile(pn, n1)
                if p128 is not None:
                    starts_128.append(p128)
    except Exception:
        pass

    # Deterministic profiles
    for p in _make_deterministic_profiles(n1):
        starts_128.append(p)

    # Random smooth profiles
    for _ in range(12):
        t = np.linspace(0.0, 1.0, n1)
        coeffs = rng.uniform(-3, 3, 8)
        a = 1.0
        for k in range(1, 8):
            a += coeffs[k] * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-10)
        starts_128.append(a / a.sum())

    # Optimize each start
    for si, p0 in enumerate(starts_128[:16]):
        if time.monotonic() > phase1_end:
            break
        z0 = np.log(np.maximum(p0, 1e-12))
        z0 -= np.mean(z0)
        per_iter = min(phase1_end, time.monotonic() + 28.0)
        z_best, sc_best = _adam_logit(z0, per_iter, rng, lr=0.12)
        p_best = _softmax_from_z(z_best)
        _update(p_best, sc_best)
        if si % 4 == 0:
            print(f"  phase1 si={si}: score={sc_best:.10f} (best={best_score:.10f})")

    print(f"after phase1 best={best_score:.10f}")

    # === Phase 2: Interpolate to n=512 and optimize ===
    n2 = 512
    phase2_end = min(deadline, t_start + 1200.0)

    # Get top candidates at n=128 and interpolate to 512
    top_128 = [c for c in candidates if c[1].size == n1][:8]
    # Also include the global best if it's at a different length
    if best_seq is not None and best_seq.size != n1:
        p_interp = _interpolate_profile(best_seq, n1)
        if p_interp is not None:
            top_128.append((_internal_score(p_interp), p_interp))
        top_128 = top_128[:8]

    starts_512 = []
    for sc, p in top_128:
        p512 = _interpolate_profile(p, n2)
        if p512 is not None:
            starts_512.append(p512)

    # Also add parent directly at 512 if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pn = _normalize_nonneg(parent)
            if pn is not None and pn.size != n2:
                p512 = _interpolate_profile(pn, n2)
                if p512 is not None:
                    starts_512.append(p512)
            elif pn.size == n2:
                starts_512.append(pn.copy())
    except Exception:
        pass

    # Add a few deterministic profiles at 512
    for p in _make_deterministic_profiles(n2)[:4]:
        starts_512.append(p)

    print(f"phase2: {len(starts_512)} starts at n={n2}")

    for si, p0 in enumerate(starts_512[:10]):
        if time.monotonic() > phase2_end:
            break
        z0 = np.log(np.maximum(p0, 1e-12))
        z0 -= np.mean(z0)
        remaining = phase2_end - time.monotonic()
        per_iter = max(10.0, min(remaining, 120.0))
        z_best, sc_best = _adam_logit(z0, time.monotonic() + per_iter, rng, lr=0.08)
        p_best = _softmax_from_z(z_best)
        _update(p_best, sc_best)
        if si % 2 == 0:
            print(f"  phase2 si={si}: score={sc_best:.10f} (best={best_score:.10f})")

    print(f"after phase2 best={best_score:.10f}")

    # === Phase 3: L-BFGS-B polish ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        p0 = best_seq.copy()
        if p0.size != n2:
            p0 = _interpolate_profile(p0, n2)
        z0 = np.log(np.maximum(p0, 1e-12))
        z0 -= np.mean(z0)
        # Use a moderate beta for L-BFGS-B
        c = _full_convolve_fft(_softmax_from_z(z0))
        beta_lbgfs = 0.1 * float(c.max())
        try:
            z_best, sc_best = _lbgfs_logit(z0, beta_lbgfs, max_iter=80)
            p_best = _softmax_from_z(z_best)
            _update(p_best, sc_best)
            print(f"  lbgfs: score={sc_best:.10f}")
        except Exception as e:
            print(f"  lbgfs failed: {e}")

    # Second L-BFGS-B with smaller beta
    if best_seq is not None and time.monotonic() < deadline - 30:
        p0 = best_seq.copy()
        if p0.size != n2:
            p0 = _interpolate_profile(p0, n2)
        z0 = np.log(np.maximum(p0, 1e-12))
        z0 -= np.mean(z0)
        c = _full_convolve_fft(_softmax_from_z(z0))
        beta_lbgfs2 = 0.03 * float(c.max())
        try:
            z_best, sc_best = _lbgfs_logit(z0, beta_lbgfs2, max_iter=60)
            p_best = _softmax_from_z(z_best)
            _update(p_best, sc_best)
            print(f"  lbgfs2: score={sc_best:.10f}")
        except Exception as e:
            print(f"  lbgfs2 failed: {e}")

    # === Phase 4: Peak shaving ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        p_shaved, sc_shaved = _peak_shave(best_seq, n_iters=300, rng=rng)
        _update(p_shaved, sc_shaved)
        print(f"  peak_shave: score={sc_shaved:.10f}")

    # === Phase 5: Final Adam polish ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        p0 = best_seq.copy()
        if p0.size != n2:
            p0 = _interpolate_profile(p0, n2)
        z0 = np.log(np.maximum(p0, 1e-12))
        z0 -= np.mean(z0)
        per = min(deadline - 3.0, time.monotonic() + 60.0)
        z_best, sc_best = _adam_logit(z0, per, rng, lr=0.04, beta_init=0.01 * float(_full_convolve_fft(_softmax_from_z(z0)).max()))
        p_best = _softmax_from_z(z_best)
        _update(p_best, sc_best)
        print(f"  final_adam: score={sc_best:.10f}")

    # === Final evaluation ===
    if best_seq is not None:
        final_seq = best_seq.copy()
        final_score = best_score
        # Exact score
        if final_seq.size <= 4096:
            c = np.convolve(final_seq, final_seq)
            exact_sc = float(2.0 * final_seq.size * c.max() / (final_seq.sum() ** 2))
            if np.isfinite(exact_sc) and exact_sc < final_score:
                final_score = exact_sc
        # Evaluate using the authoritative function
        if final_seq.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                if np.isfinite(sc_eval) and sc_eval < final_score:
                    final_score = sc_eval
            except Exception:
                pass
        # Also check top candidates
        for sc_c, p_c in candidates[:5]:
            if p_c.size <= 4096:
                c = np.convolve(p_c, p_c)
                exact_sc = float(2.0 * p_c.size * c.max() / (p_c.sum() ** 2))
                if np.isfinite(exact_sc) and exact_sc < final_score:
                    final_score = exact_sc
                    final_seq = p_c.copy()

        final_seq = _normalize_nonneg(final_seq)
        if final_seq is None:
            final_seq = _make_linear_profile(256, 1.5)
            final_score = _internal_score(final_seq)
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        print(f"FINAL: best={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(256, 2.0)
    return [float(x) for x in fallback]
