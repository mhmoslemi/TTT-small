import numpy as np
from scipy.optimize import minimize
import time


def _pow2_at_least(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _make_fft_sizes(N):
    fft_self = _pow2_at_least(2 * N - 1)
    fft_conv = _pow2_at_least(3 * N - 2)
    return fft_self, fft_conv


def _sanitized_z(x, N):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -50.0, 50.0)
    return x


def _probs_from_z(z, N):
    z = _sanitized_z(z, N)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _self_conv(p, N, fft_self):
    f = np.fft.rfft(p, fft_self)
    c = np.fft.irfft(f * f, fft_self).real
    c = c[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _full_conv_padded(a, b, fft_conv):
    fa = np.fft.rfft(a, fft_conv)
    fb = np.fft.rfft(b, fft_conv)
    return np.fft.irfft(fa * fb, fft_conv).real


def _grad_p_from_weights(p, w, N, fft_conv):
    conv = _full_conv_padded(p[::-1], w, fft_conv)
    h = conv[N - 1:2 * N - 1]
    return 2.0 * h

    return 2.0 * float(N) * float(np.max(c))

def _density_to_z(f, N):
    f = np.asarray(f, dtype=np.float64)

    score = _score_p(p, N)
    state.update(z, score)

    noise_scale = 0.08
    fails = 0

    while time.monotonic() < deadline:
        if time.monotonic() >= deadline:
            break

        c = _self_conv(p, N, fft_self)
        cmax = float(np.max(c))

        topk = min(16, 2 * N - 1)
        idx = np.argpartition(c, -topk)[-topk:]
        w_ext = np.zeros(2 * N - 1, dtype=np.float64)
        wv = c[idx] + 1e-15
        w_ext[idx] = wv / float(np.sum(wv))

        gp = _grad_p_from_weights(p, w_ext, N, fft_conv)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        accepted = False
        if gnorm > 1e-15:
            direction = gz / gnorm
            for step in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125):
                zt = z - step * direction
                zt = _sanitized_z(zt, N)
                pt = _probs_from_z(zt, N)
                st = _score_p(pt, N)
                if st < score - 1e-14:
                    z = zt
                    p = pt
                    score = st
                    state.update(z, score)
                    accepted = True
                    break

        if not accepted:
            best_local = score
            best_local_z = z.copy()
            best_local_p = p.copy()
            for _ in range(4):
                zt = z + rng.normal(0.0, noise_scale, N - 1)
                zt = _sanitized_z(zt, N)
                pt = _probs_from_z(zt, N)
                st = _score_p(pt, N)
                if st < best_local - 1e-14:
                    best_local = st
                    best_local_z = zt
                    best_local_p = pt
            if best_local < score - 1e-14:
                z = best_local_z
                p = best_local_p
                score = best_local
                state.update(z, score)
                accepted = True

        if accepted:
            fails = 0
            noise_scale = min(0.2, noise_scale * 1.15)
        else:
            fails += 1
            noise_scale = max(0.01, noise_scale * 0.85)
            if fails > 200:
                break

    return z, score


def _polish_focused(z0, N, fft_self, fft_conv, state, deadline, rng):
    """Focused polishing: target the top few peaks specifically."""
    z = _sanitized_z(z0, N)
    p = _probs_from_z(z, N)
    score = _score_p(p, N)
    state.update(z, score)

    step = 0.3
    stall = 0

    while time.monotonic() < deadline:
        if time.monotonic() >= deadline:
            break

        c = _self_conv(p, N, fft_self)
        cmax = float(np.max(c))

        # Use exact active set (within 0.01% of max)
        threshold = cmax * (1.0 - 1e-4)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        w_ext = np.zeros(2 * N - 1, dtype=np.float64)
        wa = c[active]
        wa = wa / wa.sum()
        w_ext[active] = wa

        gp = _grad_p_from_weights(p, w_ext, N, fft_conv)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        accepted = False
        if gnorm > 1e-16:
            direction = -gz / gnorm
            for s in (step, step * 0.5, step * 0.25, step * 0.125, step * 0.0625):
                zt = z + s * direction
                zt = _sanitized_z(zt, N)
                pt = _probs_from_z(zt, N)
                st = _score_p(pt, N)
                if st < score - 1e-15:
                    z = zt
                    p = pt
                    score = st
                    state.update(z, score)
                    step = min(1.0, step * 1.05)
                    accepted = True
                    break

        if accepted:
            stall = 0
        else:
            stall += 1
            step *= 0.7
            if step < 1e-10:
                break
            # Random kick
            if stall % 30 == 0:
                zt = z + rng.normal(0.0, 0.03, N - 1)
                zt = _sanitized_z(zt, N)
                pt = _probs_from_z(zt, N)
                st = _score_p(pt, N)
                if st < score:
                    z, p, score = zt, pt, st
                    state.update(z, score)
                    accepted = True
                    stall = 0

    return z, score


def _choose_alpha(score):
    if not np.isfinite(score):
        return 1.0e5
    if score > 1.90:
        return 1.0e5
    if score > 1.75:
        return 3.0e5
    if score > 1.60:
        return 8.0e5
    if score > 1.53:
        return 2.0e6
    if score > 1.51:
        return 5.0e6
    return 1.2e7


def _search_single_N(N, seed, budget_s):
    start_time = time.monotonic()
    end_time = start_time + max(1.0, budget_s) - 3.0

    state = _State()
    state.N = N
    fft_self = _pow2_at_least(2 * N - 1)
    fft_conv = _pow2_at_least(3 * N - 2)
    rng = np.random.default_rng(seed)

    # Uniform baseline
    _score_z(np.zeros(N - 1, dtype=np.float64), N, fft_self, state)

    # Generate and sort candidates
    cands = _make_candidates(N, rng)

    # Add inherited solution if available
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, None)
            if float(np.sum(base)) > 0:
                base = base / float(np.sum(base))
                cands.append(_density_to_z(base, N))
    except Exception:
        pass

    # Score all candidates
    scored = []
    for z in cands:
        s = _score_z(z, N, fft_self, state)
        scored.append((s, z))
    scored.sort(key=lambda x: x[0])

    print(f"  N={N}: best initial score={scored[0][0]:.6f} after {len(scored)} candidates")

    # Main optimization loop
    # Reserve time for polishing
    if budget_s > 300:
        main_deadline = start_time + budget_s - 5.0 - max(120.0, 0.12 * budget_s)
    else:
        main_deadline = start_time + max(5.0, budget_s - 10.0)

    # Limit number of L-BFGS runs
    max_runs = min(len(scored), 60) if budget_s > 600 else min(len(scored), 30)

    run_idx = 0
    last_best = state.best_score
    stagnant = 0

    while time.monotonic() < main_deadline:
        rem = main_deadline - time.monotonic()
        if rem < 15.0:
            break

        if run_idx < max_runs:
            z0 = scored[run_idx][1]
        else:
            if state.best_z is not None:
                frac = max(0.0, min(1.0, rem / max(1.0, main_deadline - start_time)))
                noise = 0.03 + 0.3 * frac
                if stagnant > 20:
                    noise *= 2.5
                z0 = state.best_z + rng.normal(0.0, min(2.0, noise), N - 1)
            else:
                z0 = rng.normal(0.0, 1.0, N - 1)

        run_idx += 1
        alpha = _choose_alpha(state.best_score)
        if rem > 600:
            max_fun = 800
        elif rem > 300:
            max_fun = 500
        elif rem > 120:
            max_fun = 300
        else:
            max_fun = 150

        _optimize_z(z0, alpha, N, fft_self, fft_conv, state, max_fun, main_deadline)

        if state.best_score < last_best - 1e-13:
            delta = last_best - state.best_score
            last_best = state.best_score
            stagnant = 0
            elapsed = time.monotonic() - start_time
            print(f"  N={N}: run {run_idx}: best={state.best_score:.8f} delta={delta:.8f} t={elapsed:.0f}s")

    # Polishing phase
    if state.best_z is not None:
        polish_deadline = max(time.monotonic() + 10.0, end_time - 2.0)

        # Focused polishing
        z_best, s_best = _polish_focused(state.best_z, N, fft_self, fft_conv, state, polish_deadline, rng)
        print(f"  N={N}: after focused polish: {state.best_score:.8f}")

        # General subgradient polish
        if time.monotonic() < end_time - 5.0:
            z_best, s_best = _polish_subgradient(state.best_z, N, fft_self, fft_conv, state, end_time - 5.0, rng)
            print(f"  N={N}: after subgrad polish: {state.best_score:.8f}")

        # One more L-BFGS with high alpha
        if minimize is not None and state.best_z is not None and time.monotonic() < end_time - 20.0:
            z0 = state.best_z + rng.normal(0.0, 0.02, N - 1)
            _optimize_z(z0, 1.2e7, N, fft_self, fft_conv, state, 200, end_time - 5.0)
            print(f"  N={N}: after final L-BFGS: {state.best_score:.8f}")

            # Final focused polish
            if time.monotonic() < end_time - 5.0:
                _polish_focused(state.best_z, N, fft_self, fft_conv, state, end_time - 2.0, rng)

    # Convert to sequence
    if state.best_z is None:
        z = np.zeros(N - 1, dtype=np.float64)
    else:
        z = _sanitized_z(state.best_z, N)

    p = _probs_from_z(z, N)
    p = np.clip(p, 0.0, 1000.0)
    seq = [float(v) for v in p]

    try:
        score = float(evaluate_sequence(seq))
    except Exception:
        score = float("inf")

    if not np.isfinite(score) or score > 2.0000001:
        seq = [1.0 / float(N)] * N
        try:
            score = float(evaluate_sequence(seq))
        except Exception:
            score = 2.0

    # Check inherited
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            base_score = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(base_score) and base_score < score:
                seq = [float(v) for v in base]
                score = base_score
    except Exception:
        pass

    print(f"  N={N}: FINAL score={score:.8f}")
    return seq, score


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    start_time = time.monotonic()
    budget_s = float(budget_s)

    # Primary search at N=4096 (most of the budget)
    N1 = 4096
    budget1 = budget_s * 0.75

    seq1, score1 = _search_single_N(N1, seed, budget1)

    # If time remains, try N=8192 with a quick search
    elapsed = time.monotonic() - start_time
    remaining = budget_s - elapsed - 5.0

    best_seq, best_score = seq1, score1

    if remaining > 200.0:
        N2 = 8192
        budget2 = remaining * 0.6
        try:
            seq2, score2 = _search_single_N(N2, seed + 1, budget2)
            if score2 < best_score:
                best_seq, best_score = seq2, score2
        except Exception as e:
            print(f"  N2 search failed: {e}")

        # Try N=2048 if time allows
        elapsed = time.monotonic() - start_time
        remaining = budget_s - elapsed - 5.0
        if remaining > 120.0:
            N3 = 2048
            budget3 = remaining * 0.6
            try:
                seq3, score3 = _search_single_N(N3, seed + 2, budget3)
                if score3 < best_score:
                    best_seq, best_score = seq3, score3
            except Exception as e:
                print(f"  N3 search failed: {e}")

    # Final verification
    try:
        final_score = float(evaluate_sequence(best_seq))
        if not np.isfinite(final_score) or final_score > 2.0:
            best_seq = [1.0 / float(N1)] * N1
            final_score = 2.0
    except Exception:
        best_seq = [1.0 / float(N1)] * N1
        final_score = 2.0

    print(f"GLOBAL FINAL: score={final_score:.8f} len={len(best_seq)}")
    return best_seq
