"""
LP cutting-plane optimizer for minimizing 2*n*max(convolve(f,f))/sum(f)^2.
Primary method: iterative LP direction search with adaptive row selection.
Secondary: Adam softmax-projected gradient for initialization and polishing.
Explores multiple sequence lengths around the best known (384).
"""

import time
import numpy as np
import scipy.optimize as _opt


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def conv_fft(a, b):
    n = a.size
    m = b.size
    L = next_power_of_two(n + m - 1)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:n + m - 1].real


def greedy_start(n):
    n = int(n)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_start(n, s):
    n = int(n)
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def random_perturb(p, rng, sigma):
    p = np.asarray(p, dtype=np.float64).ravel()
    noise = np.exp(rng.normal(0.0, sigma, p.size))
    return normalize_nonneg(p * noise)


def adam_optimize(p, budget_s):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s < 1.0:
        return p, score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 128:
        lr = 0.04
    elif n < 512:
        lr = 0.025
    else:
        lr = 0.015

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(2 * n - 1)
    best_p = p.copy()
    best_f = score_fft(p)

    start = time.monotonic()
    end = start + max(0.1, budget_s - 1.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = 8.0 + 40.0 * np.exp(-5.0 * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f * (1.0 - 1e-10):
            best_f = f_now
            best_p = p.copy()

        z = alpha * (c / maxc - 1.0)
        z = np.clip(z, -50.0, 0.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0.0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -100.0, 100.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

    final_f = score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def build_A(f, rows, n):
    nr = len(rows)
    A = np.zeros((nr, n), dtype=np.float64)
    for ii in range(nr):
        k = int(rows[ii])
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if lo <= hi:
            A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def lp_optimize(p, budget_s, verbose=True):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 2:
        p = normalize_nonneg(np.ones(16, dtype=np.float64))
        n = 16
    budget_s = float(budget_s)
    if budget_s < 30.0:
        return p, score_fft(p)

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_M = score_fft(f)
    if not np.isfinite(cur_M):
        return p, cur_M

    best_f = f.copy()
    best_score = float(cur_M)

    max_rows = min(2 * n - 1, 800)
    end_time = time.monotonic() + max(0.0, budget_s - 10.0)

    rows = _select_rows(f, n, max_rows, frac=0.5)
    row_set = set(int(x) for x in rows)
    it = 0

    while time.monotonic() < end_time:
        it += 1
        rem = end_time - time.monotonic()
        if rem < 8.0:
            break

        A = build_A(f, rows, n)
        b = np.full(len(rows), cur_M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        tl = min(25.0, max(3.0, rem * 0.3))
        res = None
        try:
            res = _opt.linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None),
                               method='highs',
                               options={'time_limit': tl, 'presolve': True})
        except Exception:
            try:
                res = _opt.linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None),
                                   method='highs')
            except Exception:
                break

        if res is None or not getattr(res, 'success', False):
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        conv = conv_fft(f, g)
        conv = np.maximum(conv, 0.0)
        tol = 1e-8 + 1e-8 * cur_M
        violations = conv - cur_M

        if np.max(violations) > tol:
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    if added >= min(200, max_rows - len(rows)):
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                if added > 0:
                    continue
            else:
                if np.max(violations) > 100.0 * tol:
                    break

        g = g * (S0 / sg)

        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.03):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()

        if accepted is None:
            break
        if accepted_score >= best_score * (1.0 - 1e-12):
            if verbose:
                print("  LP n=%d it=%d: no improvement (score=%.8f), stopping" %
                      (n, it, best_score))
            break

        improved = best_score - accepted_score
        f = accepted
        cur_M = score_fft(f)
        if cur_M < best_score:
            best_score = cur_M
            best_f = f.copy()

        if verbose and it % 5 == 0:
            print("  LP n=%d it=%d score=%.8f rows=%d" % (n, it, best_score, len(rows)))

        if it % 10 == 0:
            rows = _select_rows(f, n, max_rows, frac=0.5)
            row_set = set(int(x) for x in rows)

    return normalize_nonneg(best_f), best_score


def _select_rows(f, n, max_rows, frac=0.5):
    c = f * f
    L = next_power_of_two(2 * n - 1)
    Pg = np.fft.rfft(f, L)
    c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    threshold = maxc * (1.0 - frac * 0.3)
    mask = c >= threshold
    if mask.sum() > max_rows:
        idx = np.where(mask)[0]
        order = c[idx].argsort()[::-1][:max_rows]
        idx = idx[order]
    else:
        idx = np.where(mask)[0]
    if len(idx) == 0:
        idx = np.array([n - 1], dtype=int)
    return [int(x) for x in idx]


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, sc=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if sc is None:
            sc = score_fft(a)
        if not np.isfinite(sc):
            return
        if sc < best_score:
            best_p = a.copy()
            best_score = float(sc)
        nn = int(a.size)
        old = best_by_n.get(nn)
        if old is None or sc < old[0]:
            best_by_n[nn] = (float(sc), a.copy())

    # Phase 1: Initialization (40s)
    phase1_end = t0 + 40.0

    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                record(arr)
                print("Loaded height_sequence_1: n=%d score=%.8f" % (arr.size, best_score))
        except Exception:
            pass

    lengths = [360, 372, 384, 396, 408, 420, 432, 444, 456, 468, 480]
    for nn in lengths:
        if time.monotonic() >= phase1_end:
            break
        if nn in best_by_n:
            continue
        pct = time.monotonic() - t0
        alloc = min(3.0, max(0.5, (phase1_end - time.monotonic()) / max(1, len(lengths) - int(pct / 3.0))))
        for kind in ('greedy', 'power'):
            if kind == 'greedy':
                p = greedy_start(nn)
            else:
                p = power_start(nn, 0.45)
            record(p)
        p_out, f_out = adam_optimize(best_by_n.get(nn, (np.inf, greedy_start(nn)))[1], alloc)
        record(p_out, f_out)

    # Quick Adam on best
    if best_p is not None:
        rem = phase1_end - time.monotonic()
        if rem > 5.0:
            p_out, f_out = adam_optimize(best_p, rem)
            record(p_out, f_out)

    print("Phase 1 done. Best: n=%d score=%.8f" % (best_p.size, best_score))

    # Phase 2: LP optimization (majority of budget)
    phase2_end = deadline - 120.0

    # Determine which lengths to LP-optimize
    lp_lengths = sorted(best_by_n.keys(), key=lambda x: best_by_n[x][0])[:3]
    if not lp_lengths:
        lp_lengths = [384]

    for nn in lp_lengths:
        rem = phase2_end - time.monotonic()
        if rem < 60.0:
            break
        base = best_by_n[nn][1]
        alloc = min(500.0, rem * 0.6)
        print("Phase 2: LP on n=%d, alloc=%.0fs" % (nn, alloc))
        p_out, f_out = lp_optimize(base, alloc, verbose=True)
        record(p_out, f_out)

        # After LP, do a short Adam polish
        rem = phase2_end - time.monotonic()
        if rem > 30.0:
            p_pol = random_perturb(p_out, rng, 0.05)
            p_out2, f_out2 = adam_optimize(p_pol, min(30.0, rem - 5.0))
            record(p_out2, f_out2)

    print("Phase 2 done. Best: n=%d score=%.8f" % (best_p.size, best_score))

    # Phase 3: Polish and final exploration (last 120s)
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 20.0:
            p_pol = best_p.copy()
            p_out, f_out = adam_optimize(p_pol, min(60.0, rem - 10.0))
            record(p_out, f_out)

        rem = deadline - time.monotonic()
        if rem > 20.0:
            # Try a random perturbation + LP
            p_pert = random_perturb(best_p, rng, 0.1)
            p_out, f_out = lp_optimize(p_pert, min(40.0, rem - 10.0), verbose=False)
            record(p_out, f_out)

    # Final: try a couple more lengths with short LP if time allows
    rem = deadline - time.monotonic()
    for nn in [384, 420, 480]:
        if rem < 30.0:
            break
        if nn in best_by_n and best_by_n[nn][0] < best_score * 1.01:
            continue
        base = best_by_n.get(nn, (2.0, greedy_start(nn)))[1]
        p_out, f_out = lp_optimize(base, min(25.0, rem - 10.0), verbose=False)
        record(p_out, f_out)
        rem = deadline - time.monotonic()

    if best_p is None:
        best_p = greedy_start(384)
        best_score = score_fft(best_p)

    print("Final best: n=%d score=%.8f" % (best_p.size, best_score))
    return scale_output(best_p, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the objective."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main_search: %s" % str(e))
        import traceback
        traceback.print_exc()
        p = greedy_start(384)
        return scale_output(p, 10.0)
