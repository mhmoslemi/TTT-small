def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    """Full linear convolution via FFT."""
    n = a.size
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    while time.monotonic() < deadline:
        c = fft_convolve(f)
        M = c.max()
        current_score = 2.0 * n * M

        # Solve LP
        g = lp_find_direction(f, M, relaxation=1.0)
        if g is None:
            stagnant += 1
            if stagnant > 5:
                break
            continue

        # Adaptive mixing rate
        t_mix = mix_rate
        f_new = (1 - t_mix) * f + t_mix * g
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        c_new = fft_convolve(f_new)
        sc_new = 2.0 * n * c_new.max()

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_f = f_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 20:
                # Try smaller mix rate
                t_mix = max(0.01, t_mix * 0.5)
                if t_mix < 0.01:
                    break
                stagnant = 0
                continue
            if stagnant > 50:
                break

        f = f_new
        iter_count += 1

    return best_f, best_score, iter_count


def make_initial_profiles(n, rng):
    """Generate diverse initial profiles."""
    profs = []
    t = np.linspace(0, 1, n)

    # Linear profiles
    for eta in (1.0, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0):
        a = 1.0 + eta * t
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())

    # Power-law profiles
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        a = (np.arange(n) + 1.0) ** p
        profs.append(a / a.sum())

    # Exponential profiles
    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = np.exp(alpha * (t - 0.5))
        profs.append(a / a.sum())

    # Cubic profiles
    u = 2.0 * t - 1.0
    for a1, a3 in ((1.5, -1.0), (2.0, -2.0), (1.0, 0.5), (2.5, -1.5), (1.8, -2.5)):
        a = 1.0 + a1 * u + a3 * u ** 3
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())

    # Random smooth profiles via Fourier
    for _ in range(10):
        coeffs = rng.normal(0, 1, 8)
        a = 1.0
        for k, ck in enumerate(coeffs, 1):
            a += ck * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi)) * (0.5 ** k)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())

    # "U-shaped" profiles (high at edges)
    for beta in (1.0, 2.0, 3.0):
        a = 1.0 + beta * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())

    return profs


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    1. Generate diverse initial profiles at multiple lengths.
    2. LP-guided iteration on best profiles.
    3. Logit-space Adam polish with peak-directed gradients.
    4. Cross-length interpolation and re-optimization.
    5. Final exact scoring.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(f, sc):
        nonlocal best_seq, best_score
        if f is not None and np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = np.asarray(f, dtype=np.float64).copy()

    # Use parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            parent = normalize_nonneg(p)
            if parent is not None:
                sc = score_fast(parent)
                update_best(parent, sc)
                print(f"Parent score: {sc:.8f}, n={parent.size}")
    except Exception:
        pass

    # Phase 1: Try various lengths, find good starting points
    lengths = [256, 512, 1024]
    if parent is not None and parent.size in lengths:
        lengths.remove(parent.size)
    lengths.insert(0, parent.size if parent is not None else 512)

    candidates_by_len = {ln: [] for ln in lengths}

    phase1_end = min(deadline, t_start + 120.0)
    for ln in lengths:
        if time.monotonic() > phase1_end:
            break
        profs = make_initial_profiles(ln, rng)
        if parent is not None:
            parent_interp = interpolate_profile(parent, ln)
            if parent_interp is not None:
                profs.insert(0, parent_interp)

        for p0 in profs:
            if time.monotonic() > phase1_end:
                break
            sc = score_fast(p0)
            if np.isfinite(sc):
                update_best(p0, sc)
                candidates_by_len[ln].append((sc, p0))

        candidates_by_len[ln].sort(key=lambda x: x[0])
        print(f"Phase 1 n={ln}: best={candidates_by_len[ln][0][0]:.8f}")

    print(f"After phase 1: best={best_score:.8f}")

    # Phase 2: LP-guided optimization on top candidates
    phase2_end = min(deadline, t_start + 700.0)
    for ln in lengths:
        if time.monotonic() > phase2_end:
            break
        top = candidates_by_len[ln][:4] if candidates_by_len[ln] else []
        for sc0, p0 in top:
            if time.monotonic() > phase2_end:
                break
            per_deadline = min(phase2_end, time.monotonic() + 60.0)
            f_best, sc_best, iters = lp_iterate(p0, per_deadline, rng, mix_rate=0.08)
            update_best(f_best, sc_best)
            if sc_best < sc0:
                candidates_by_len[ln] = [(sc_best, f_best)] + [x for x in candidates_by_len[ln] if x[1].is_contiguous()]
                candidates_by_len[ln].sort(key=lambda x: x[0])
                print(f"Phase 2 n={ln}: LP iter {iters}, score={sc_best:.8f}")

    print(f"After phase 2: best={best_score:.8f}")

    # Phase 3: Logit-space Adam polish
    phase3_end = min(deadline, t_start + 1500.0)
    all_candidates = []
    for ln in lengths:
        all_candidates.extend(candidates_by_len[ln][:6])
    all_candidates.sort(key=lambda x: x[0])

    seen = set()
    for sc0, p0 in all_candidates[:12]:
        if time.monotonic() > phase3_end:
            break
        key = tuple(np.round(p0[:10], 6))
        if key in seen:
            continue
        seen.add(key)

        n0 = p0.size
        per_deadline = min(phase3_end, time.monotonic() + (80.0 if n0 <= 512 else 50.0))
        p_best, sc_best = logit_polish(p0, per_deadline, rng, lr=0.04, alpha_init=30.0, alpha_final=150.0)
        update_best(p_best, sc_best)
        if sc_best < sc0 - 1e-6:
            print(f"Phase 3 n={n0}: polish score={sc_best:.8f}")

    print(f"After phase 3: best={best_score:.8f}")

    # Phase 4: Cross-length optimization
    phase4_end = min(deadline, t_start + 1700.0)
    if best_seq is not None:
        target_lens = [512, 1024]
        if best_seq.size in target_lens:
            target_lens.remove(best_seq.size)
        for tl in target_lens:
            if time.monotonic() > phase4_end:
                break
            p_interp = interpolate_profile(best_seq, tl)
            if p_interp is None:
                continue
            sc_interp = score_fast(p_interp)
            update_best(p_interp, sc_interp)

            # LP on interpolated
            per_deadline = min(phase4_end, time.monotonic() + 50.0)
            f_lp, sc_lp, _ = lp_iterate(p_interp, per_deadline, rng, mix_rate=0.08)
            update_best(f_lp, sc_lp)

            # Polish
            if time.monotonic() < phase4_end - 30:
                per_deadline = min(phase4_end - 10.0, time.monotonic() + 60.0)
                p_pol, sc_pol = logit_polish(f_lp, per_deadline, rng, lr=0.03, alpha_init=40.0, alpha_final=200.0)
                update_best(p_pol, sc_pol)
                print(f"Phase 4 n={tl}: score={best_score:.8f}")

    print(f"After phase 4: best={best_score:.8f}")

    # Phase 5: Final exact scoring and small perturbations
    if best_seq is not None:
        final = best_seq.copy()
        final_score = score_exact(final)
        if final_score < best_score:
            best_score = final_score
        else:
            best_score = final_score

        # Try a few random perturbations
        for _ in range(5):
            if time.monotonic() > deadline - 5:
                break
            n = final.size
            pert = final + 0.01 * rng.normal(size=n)
            pert = normalize_nonneg(pert)
            if pert is not None:
                sc = score_exact(pert)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    final = pert

        # Also try targeted perturbation: reduce the peak element
        c = np.convolve(final, final)
        peak_k = c.argmax()
        # The peak is at position peak_k in the convolution
        # Reduce contribution: find which elements contribute most
        # c[peak_k] = sum_i final[i] * final[peak_k - i]
        # Gradient direction: reduce final[i] where final[peak_k - i] is large
        n = final.size
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                contrib[i] = final[j]
        # Perturb: reduce high-contribution elements
        pert = final.copy()
        pert -= 0.005 * contrib
        pert = normalize_nonneg(pert)
        if pert is not None:
            sc = score_exact(pert)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                final = pert

        print(f"Final: best_score={best_score:.8f}, n={final.size}")
        result = [float(max(0, x)) for x in final]
        return result

    # Fallback
    t = np.linspace(0, 1, 512)
    f = 1.0 + 1.5 * t
    f = f / f.sum()
    return [float(x) for x in f]
