```
def resize_integral(a, N):
    a=normalize_nonneg(a)
    m=a.size
    if m==N: return a
    if m==1: return np.full(N, 1.0/N)
    x_src = np.linspace(0.0,1.0,m+1)
    cdf = np.concatenate(([0.0], np.cumsum(a)))/float(a.sum())
    x_tgt_start = np.arange(N+1, dtype=float)/N
    # interpolate cdf at x_tgt_start: cdf is piecewise linear over x_src
    cdf_tgt = np.interp(x_tgt_start, x_src, cdf)
    mass = np.diff(cdf_tgt)
    mass = np.maximum(mass,0.0)
    return normalize_nonneg(mass)
```

```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

The program uses a portfolio of flatness-oriented initializers, a
multi-scale FFT-based local search, smooth-max / variance gradient steps,
occasional active-set linear programming directions, pool blending, and a
final polishing phase.  The best internally found sequence is returned.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


MAX_POOL = 24
MAX_OPT_N = 4096


def normalize_nonneg(a):
    """Clip to non-negative, remove non-finite values, and normalize sum to 1."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.size, 1.0 / max(1, a.size), dtype=np.float64)
    return a / s


def convolution_square(a):
    """Full convolution a*a, using FFT for moderate/large lengths."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.zeros(1, dtype=np.float64)
    if n <= 128:
        c = np.convolve(a, a, mode="full")
    else:
        m = 2 * n - 1
        size = 1
        while size < m:
            size <<= 1
        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fa * fa, size)[:m]
    c = np.asarray(c, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    c[c < 0.0] = 0.0
    return c


def sequence_score_c(seq):
    """Return (normalized_a, score, convolution)."""
    a = normalize_nonneg(seq)
    c = convolution_square(a)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M):
        return a, float("inf"), c
    return a, float(2.0 * a.size * M), c


def cross_corr_w_a(w, a):
    """
    Given w of length 2n-1 and a of length n, return R_i = sum_j w_{i+j} a_j,
    for i=0..n-1.  This is the gradient contribution for weighted convolution.
    """
    w = np.asarray(w, dtype=np.float64).reshape(-1)
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    m = w.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if m == 0:
        return np.zeros(n, dtype=np.float64)
    if not np.any(w):
        return np.zeros(n, dtype=np.float64)

    if m * n <= 60000:
        c = np.convolve(w, a[::-1], mode="full")
        return np.asarray(c[n - 1:n + n - 1], dtype=np.float64)

    total = m + n - 1
    size = 1
    while size < total:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a[::-1], size)
    c = np.fft.irfft(fw * fa, size)[:total]
    return np.asarray(c[n - 1:n + n - 1], dtype=np.float64)


def make_weights(c, M, temp, mode):
    """Create non-negative lag weights summing to 1."""
    m = c.size
    if m == 0 or M <= 1e-300:
        return None
    p = np.zeros(m, dtype=np.float64)

    if mode == 0 or mode < 0:  # softmax over all lags
        T = max(1e-300, float(temp) * M)
        e = (c - M) / T
        e = np.clip(e, -50.0, 0.0)
        p = np.exp(e)
        s = float(p.sum())
        if not np.isfinite(s) or s <= 0.0:
            p = np.zeros(m)
            p[int(np.argmax(c))] = 1.0
        else:
            p = p / s

    elif mode == 1:  # uniform over top ~10%
        K = int(max(1, min(m, round(0.10 * m))))
        if K >= m:
            p[:] = 1.0 / m
        else:
            rows = np.argpartition(c, -K)[-K:]
            p[rows] = 1.0 / K

    elif mode == 2:  # single largest peak
        p[int(np.argmax(c))] = 1.0

    elif mode == 3:  # softmax over top ~20%
        K = int(max(1, min(m, round(0.20 * m))))
        if K >= m:
            rows = np.arange(m)
        else:
            rows = np.argpartition(c, -K)[-K:]
        T = max(1e-300, float(temp) * M)
        e = (c[rows] - M) / T
        e = np.clip(e, -50.0, 0.0)
        pr = np.exp(e)
        s = float(pr.sum())
        if not np.isfinite(s) or s <= 0.0:
            p[rows] = 1.0 / K
        else:
            p[rows] = pr / s

    return p


def gradient_step(a, c, M, cur, temp, peak_mode, beta):
    """
    Smooth-max + variance gradient step.
    Returns (new_a, new_score, new_c) if an improvement over cur is found.
    """
    n = a.size
    m = c.size
    if n < 2 or m < 3:
        return None

    c_sum = float(c.sum())
    avg = c_sum / m if c_sum > 1e-300 else 1.0 / m
    v = c - avg

    S = cross_corr_w_a(v, a)
    if not np.all(np.isfinite(S)):
        S = np.zeros(n, dtype=np.float64)
    S_c = S - np.mean(S)
    S_scale = float(np.max(np.abs(S_c))) if S.size else 0.0

    R = None
    if peak_mode is not None:
        p = make_weights(c, M, temp, peak_mode)
        if p is not None:
            R = cross_corr_w_a(p, a)
            if not np.all(np.isfinite(R)):
                R = None

    if R is None:
        R_c = np.zeros(n, dtype=np.float64)
        R_scale = 0.0
    else:
        R_c = R - np.mean(R)
        R_scale = float(np.max(np.abs(R_c))) if R.size else 0.0

    if S_scale <= 1e-300 and R_scale <= 1e-300:
        return None

    g = np.zeros(n, dtype=np.float64)
    if R_scale > 1e-300:
        g += R_c / R_scale
    if S_scale > 1e-300 and abs(beta) > 1e-12:
        g += float(beta) * (S_c / S_scale)

    g -= np.mean(g)
    d = -g
    if not np.all(np.isfinite(d)):
        return None

    scale_g = float(np.max(np.abs(d)))
    if scale_g <= 1e-300:
        return None

    best_seq = None
    best_score = cur
    best_c = None

    base_step = max(1e-12, 1.0 / float(n))

    # Additive projected line search.
    for fac in (0.1, 0.3, 1.0, 3.0, 10.0, 30.0, 100.0):
        h = a + (fac * base_step) * d
        cand, sc, cc = sequence_score_c(h)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = cand
            best_c = cc

    # Multiplicative (mirror) step, only when peak weights are available.
    if R is not None and np.all(np.isfinite(R)):
        Rmin = float(np.min(R))
        Rmax = float(np.max(R))
        rrange = Rmax - Rmin
        if rrange > 1e-300:
            base = (R - Rmin) / rrange
            for fac in (0.1, 0.3, 1.0, 3.0, 10.0):
                h = a * np.exp(-fac * base)
                cand, sc, cc = sequence_score_c(h)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_seq = cand
                    best_c = cc

    if best_seq is None:
        return None
    return best_seq, best_score, best_c


def random_perturb(a, rng, sigma):
    """Smooth non-negative multiplicative perturbation."""
    a = normalize_nonneg(a)
    n = a.size
    if n <= 1:
        return a

    B = int(max(4, min(128, n // 8)))
    center = (np.arange(B) + 0.5) / float(B)
    y = (np.arange(n) + 0.5) / float(n)
    block = np.exp(rng.normal(0.0, float(sigma), B))
    mult = np.interp(y, center, block)
    h = a * mult
    return normalize_nonneg(h)


def resize_integral(a, N):
    """Mass-preserving resize by integrating the discrete cumulative function."""
    N = int(max(1, N))
    a = normalize_nonneg(a)
    m = a.size
    if m == N:
        return a
    if m == 1:
        return np.full(N, 1.0 / float(N), dtype=np.float64)

    x_src = np.linspace(0.0, 1.0, m + 1)
    cdf = np.concatenate(([0.0], np.cumsum(a)))
    cdf = cdf / max(1e-300, float(cdf[-1]))

    x_tgt = np.arange(N + 1, dtype=np.float64) / float(N)
    cdf_tgt = np.interp(x_tgt, x_src, cdf)
    mass = np.diff(cdf_tgt)
    mass = np.maximum(mass, 0.0)
    return normalize_nonneg(mass)


def choose_active_rows(c, max_rows, frac):
    """Choose important convolution lags for active-set LP methods."""
    m = c.size
    if m == 0 or max_rows <= 0:
        return np.zeros(0, dtype=np.int64)
    M = float(np.max(c))
    if M <= 1e-300:
        return np.arange(min(max_rows, m), dtype=np.int64)

    thresh = M - float(frac) * M
    rows = np.flatnonzero(c >= thresh)

    if rows.size > int(max_rows):
        if int(max_rows) >= m:
            rows = np.arange(m)
        else:
            rows = np.argpartition(c, -int(max_rows))[-int(max_rows):]
    elif rows.size < 4:
        K = int(min(max(4, int(max_rows)), m))
        if K >= m:
            rows = np.arange(m)
        else:
            rows = np.argpartition(c, -K)[-K:]

    return np.asarray(rows, dtype=np.int64)


def primal_lp_step(a, c, M, cur):
    """
    Find g maximizing sum(g) subject to a*g <= M on active lags and g >= 0.
    Try convex combinations with a.
    """
    if linprog is None:
        return None
    n = a.size
    if n < 2 or n > 640 or M <= 1e-300:
        return None

    rows = choose_active_rows(c, int(min(n, 512)), 0.10)
    if rows.size == 0:
        return None

    K = rows.size
    A = np.zeros((K, n), dtype=np.float64)
    for r, k in enumerate(rows):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            A[r, jlo:jhi] = a[k - np.arange(jlo, jhi)]

    b = np.full(K, float(M), dtype=np.float64)
    cvec = -np.ones(n, dtype=np.float64)

    try:
        res = linprog(cvec, A_ub=A, b_ub=b, bounds=(0.0, None), method="highs")
    except Exception:
        return None

    if res is None or res.status != 0 or res.x is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    S = float(g.sum())
    if S <= 1e-12:
        return None
    g = g / S

    best_seq = None
    best_score = cur
    best_c = None

    for t in (0.05, 0.10, 0.20, 0.35, 0.50, 0.70, 0.90, 1.00):
        h = (1.0 - float(t)) * a + float(t) * g
        cand, sc, cc = sequence_score_c(h)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = cand
            best_c = cc

    if best_seq is None:
        return None
    return best_seq, best_score, best_c


def dual_lp_step(a, c, cur):
    """
    Heuristic dual equalization LP:
    with uniform weights y on active lags, solve
        max z  s.t. (C_y a)_i >= z, sum(a)=1, a>=0.
    Try convex combinations with current a.
    """
    if linprog is None:
        return None
    n = a.size
    if n < 2 or n > 640:
        return None

    rows = choose_active_rows(c, int(min(n, 512)), 0.25)
    if rows.size == 0:
        return None

    m = 2 * n - 1
    y = np.zeros(m, dtype=np.float64)
    y[rows] = 1.0 / float(rows.size)

    C = np.empty((n, n), dtype=np.float64)
    for i in range(n):
        C[i] = y[i:i + n]

    if float(np.min(C.sum(axis=1))) <= 1e-15:
        return None

    cvec = np.zeros(n + 1, dtype=np.float64)
    cvec[-1] = -1.0

    A_ub = -C.copy()
    A_ub[:, -1] = 1.0
    b_ub = np.zeros(n, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(
            cvec,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None

    if res is None or res.status != 0 or res.x is None:
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    S = float(g.sum())
    if S <= 1e-12:
        return None
    g = g / S

    best_seq = None
    best_score = cur
    best_c = None

    for t in (0.05, 0.10, 0.20, 0.35, 0.50, 0.70, 0.90, 1.00):
        h = (1.0 - float(t)) * a + float(t) * g
        cand, sc, cc = sequence_score_c(h)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = cand
            best_c = cc

    if best_seq is None:
        return None
    return best_seq, best_score, best_c


def local_refine(a, time_limit, rng, state, use_lp=True, max_lp=200, polish=False):
    """
    Local search trajectory.  Updates the global state.
    """
    a, cur, c = sequence_score_c(a)
    n = a.size
    if n < 2:
        a = np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64)
        a, cur, c = sequence_score_c(a)
        n = a.size

    local_best = a.copy()
    local_best_score = cur
    update_global(state, local_best, local_best_score)

    if time_limit <= 0.0:
        return local_best, local_best_score

    deadline = time.monotonic() + float(time_limit)
    temp = 0.03 if polish else 0.12
    no_improve = 0
    lp_used = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # If the current sequence has wandered too far, reset to local best.
        if cur > local_best_score * (1.02 + 0.05 * temp) and it > 20:
            a = local_best.copy()
            c = convolution_square(a)
            cur = local_best_score

        # Choose gradient surrogate parameters.
        if polish:
            if it % 4 == 0:
                peak_mode, beta = 1, 1.0
            elif it % 4 == 1:
                peak_mode, beta = 0, 0.5
            elif it % 4 == 2:
                peak_mode, beta = None, 0.7
            else:
                peak_mode, beta = 1, 0.7
        else:
            q = it % 6
            if q == 0:
                peak_mode, beta = 0, 0.7
            elif q == 1:
                peak_mode, beta = 1, 1.0
            elif q == 2:
                peak_mode, beta = None, 1.0
            elif q == 3:
                peak_mode, beta = 3, 2.0
            elif q == 4:
                peak_mode, beta = 2, 0.0
            else:
                peak_mode, beta = 1, 2.0

        res = gradient_step(a, c, float(np.max(c)), cur, temp, peak_mode, beta)

        if res is None and use_lp and n <= 640 and lp_used < int(max_lp):
            if (it % 10 == 0) or (no_improve > 8):
                res = primal_lp_step(a, c, float(np.max(c)), cur)
                if res is None:
                    res = dual_lp_step(a, c, cur)
                if res is not None:
                    lp_used += 1

        if res is not None:
            a, cur, c = res
        else:
            no_improve += 1
            if no_improve > (18 if polish else 35):
                temp = min(0.50, temp * 1.25)
                a, cur, c = sequence_score_c(random_perturb(a, rng, 0.03 + 0.08 * temp))
                if cur > local_best_score * (1.05 + 0.10 * temp):
                    a = local_best.copy()
                    c = convolution_square(a)
                    cur = local_best_score

        if cur < local_best_score - 1e-12:
            local_best_score = cur
            local_best = a.copy()
            update_global(state, local_best, local_best_score)
            no_improve = 0
            if polish:
                temp = max(0.012, temp * 0.96)
            else:
                temp = max(0.025, temp * 0.97)

    # Final update.
    if cur < local_best_score:
        local_best_score = cur
        local_best = a.copy()
    update_global(state, local_best, local_best_score)
    return local_best, local_best_score


def load_provided_sequence():
    """Load the externally supplied construction if available."""
    try:
        raw = height_sequence_1
    except Exception:
        return None
    try:
        a = np.asarray(raw, dtype=np.float64)
    except Exception:
        return None
    if a is None or a.size == 0:
        return None
    a = a.reshape(-1)
    a = np.clip(a, 0.0, None)
    if not np.all(np.isfinite(a)) or float(a.sum()) <= 1e-300:
        return None
    return normalize_nonneg(a)


def add_pool(state, seq, score):
    """Maintain a small pool of good sequences."""
    if not np.isfinite(score):
        return
    pool = state["pool"]
    seq = np.asarray(seq, dtype=np.float64).copy()

    if len(pool) >= MAX_POOL:
        worst_score = max(item["score"] for item in pool)
        if score > worst_score - 1e-9:
            return

    pool.append({"seq": seq, "score": float(score)})
    if len(pool) > MAX_POOL:
        pool.sort(key=pool_sort_key)
        pool.pop()


def pool_sort_key(item):
    return item["score"]


def scored_tuple_key(item):
    return item[0]


def update_global(state, seq, score):
    """Update global best and pool."""
    if not np.isfinite(score):
        return False
    a = normalize_nonneg(seq)
    if a.size == 0:
        return False

    old = state["score"]
    if score < old - 1e-12:
        state["score"] = float(score)
        state["seq"] = a.copy()
        print(f"new best score={state['score']:.12f} n={a.size}", flush=True)
        add_pool(state, a, score)
        return True

    # Add good-but-not-best sequences to the pool for interpolation.
    if np.isfinite(old) and score < old + 0.15:
        add_pool(state, a, score)
    return False


def pool_blend(state, rng, max_evals=30):
    """Try convex combinations of pool sequences at common lengths."""
    if len(state["pool"]) < 2:
        return

    state["pool"].sort(key=pool_sort_key)
    sub = state["pool"][:8]
    evals = 0

    for i in range(len(sub)):
        for j in range(i + 1, len(sub)):
            if evals >= max_evals:
                return
            s1 = sub[i]["seq"]
            s2 = sub[j]["seq"]
            L = int(max(s1.size, s2.size))
            if L > 1024:
                L = 1024
            if L < 256:
                L = max(2, L)

            a1 = resize_integral(s1, L)
            a2 = resize_integral(s2, L)

            for t in (0.25, 0.50, 0.75):
                h = (1.0 - float(t)) * a1 + float(t) * a2
                cand, sc, _ = sequence_score_c(h)
                update_global(state, cand, sc)
            evals += 1


def generate_quick_candidates(N, rng, max_candidates=80):
    """Generate a diverse set of normalized non-negative candidates of length N."""
    N = int(max(2, min(MAX_OPT_N, N)))
    x = (np.arange(N) + 0.5) / float(N)
    out = []

    # Uniform.
    out.append(np.ones(N, dtype=np.float64))

    # Two-block structures.
    for split in (0.30, 0.40, 0.50, 0.60, 0.70):
        L = int(round(split * N))
        L = int(max(1, min(N - 1, L)))
        for w in (0.20, 0.33, 0.50, 0.67, 0.80):
            a = np.zeros(N, dtype=np.float64)
            a[:L] = w / float(L)
            a[L:] = (1.0 - w) / float(N - L)
            out.append(a)

    # Power-law envelopes.
    for alpha in (-0.4, 0.0, 0.5, 1.0, 2.0):
        for beta in (-0.4, 0.0, 0.5, 1.0, 2.0):
            a = np.maximum(x, 1e-6) ** alpha * np.maximum(1.0 - x, 1e-6) ** beta
            if np.all(np.isfinite(a)):
                out.append(a)

    # Half-wave cosine shapes (non-negative after clipping).
    for alpha in (0.10, 0.30, 0.50, 0.70, 0.90):
        for p in (0.5, 1.0, 1.5, 2.0):
            for shift in (0.0, 0.25, 0.50):
                base = 1.0 + alpha * np.cos(np.pi * (x - shift))
                a = np.power(np.maximum(base, 0.0), p)
                out.append(a)

    # Random smooth exponentials in low Fourier modes.
    for _ in range(20):
        M = 8
        theta = rng.normal(0.0, 1.0, size=M)
        phase = rng.uniform(0.0, 2.0 * np.pi, size=M)
        y = np.zeros(N, dtype=np.float64)
        for k in range(M):
            y += theta[k] * np.cos(np.pi * (k + 1) * x + phase[k])
        out.append(np.exp(0.6 * y - float(np.mean(0.6 * y))))

    # Random multi-block constant sequences.
    for _ in range(10):
        K = int(rng.integers(3, 9))
        counts = rng.random(K)
        counts /= max(1e-300, float(counts.sum()))
        counts = np.maximum(1.0, counts * float(N)).astype(np.int64)
        while counts.sum() > N:
            idx = int(mp.argmax(counts)) if False else int(np.argmax(counts))
            if counts[idx] > 1:
                counts[idx] -= 1
            else:
                break
        while counts.sum() < N:
            idx = int(np.argmax(counts))
            counts[idx] += 1

        masses = rng.dirichlet(np.ones(K))
        a = np.zeros(N, dtype=np.float64)
        pos = 0
        for L, p in zip(counts, masses):
            L = int(L)
            if L > 0 and pos + L <= N:
                a[pos:pos + L] = float(p) / float(L)
                pos += L
            elif L > 0:
                a[pos:N] = float(p) / max(1, N - pos)
                break
        out.append(a)

    if len(out) > max_candidates:
        out = out[:max_candidates]
    return out


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for the best sequence within the requested time budget.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    budget = float(max(60.0, budget_s))
    deadline = t0 + max(30.0, budget - 60.0)

    state = {"score": float("inf"), "seq": None, "pool": []}

    # Load user-provided construction.
    provided = load_provided_sequence()
    if provided is not None:
        a, sc, _ = sequence_score_c(provided)
        update_global(state, a, sc)

    # Guaranteed fallbacks.
    update_global(state, np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64), 16.0 / 9.0)
    update_global(state, np.ones(64, dtype=np.float64), None)

    # Quick candidate generation at a reasonable base length.
    if state["seq"] is not None and state["seq"].size > 0:
        base_N = int(state["seq"].size)
    else:
        base_N = 512
    base_N = int(max(128, min(1024, base_N)))

    candidates = generate_quick_candidates(base_N, rng, max_candidates=80)
    for seq in candidates:
        a, sc, _ = sequence_score_c(seq)
        if np.isfinite(sc) and sc < state["score"] + 0.35:
            update_global(state, a, sc)

    if state["seq"] is None:
        state["seq"] = np.ones(base_N, dtype=np.float64) / base_N
        state["score"] = 2.0 * base_N * float(np.max(convolution_square(state["seq"])))

    # Work sequence for phase 1: keep it at a tractable length.
    work_seq = state["seq"]
    if work_seq.size > 1024:
        work_seq = resize_integral(work_seq, 1024)
        a, sc, _ = sequence_score_c(work_seq)
        update_global(state, a, sc)

    # Phase 1: strong refinement at / near current size.
    remaining = deadline - time.monotonic()
    if remaining > 120.0:
        limit = min(520.0, remaining)
        local_refine(work_seq, limit, rng, state, use_lp=True, max_lp=320, polish=False)
        pool_blend(state, rng, max_evals=24)

    # Phase 2: multi-scale refinement at longer lengths.
    for target in (1024, 2048, 4096):
        remaining = deadline - time.monotonic()
        if remaining < 90.0:
            break
        if target > MAX_OPT_N:
            continue

        if state["seq"] is None:
            break

        # If best already has a larger length, still allow downsampling/refinement,
        # but generally start from the current best resized upward.
        start_seq =.resize_integral if False else resize_integral(state["seq"], target)
        a, sc, _ = sequence_score_c(start_seq)
        update_global(state, a, sc)

        if target <= 1024:
            limit = min(260.0, max(0.0, remaining - 160.0))
        elif target <= 2048:
            limit = min(230.0, max(0.0, remaining - 130.0))
        else:
            limit = min(200.0, max(0.0, remaining - 100.0))

        if limit > 45.0:
            local_refine(start_seq, limit, rng, state,
                         use_lp=(target <= 640),
                         max_lp=120 if target <= 640 else 0,
                         polish=False)
            pool_blend(state, rng, max_evals=18)

    # Phase 3: short exploratory refinements from fresh candidates.
    for N in (512, 1024):
        remaining = deadline - time.monotonic()
        if remaining < 80.0:
            break

        cands = generate_quick_candidates(N, rng, max_candidates=36)
        scored = []
        for seq in cands:
            a, sc, _ = sequence_score_c(seq)
            if np.isfinite(sc):
                scored.append((sc, a, N))
                update_global(state, a, sc)

        if not scored:
            continue
        scored.sort(key=scored_tuple_key)

        for sc, a, nn in scored[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 45.0:
                break
            # Do not waste time on obviously poor long candidates.
            if np.isfinite(state["score"]) and sc > state["score"] + 0.45 and N > 512:
                continue
            limit = min(65.0, remaining)
            local_refine(a, limit, rng, state,
                         use_lp=(nn <= 640),
                         max_lp=60 if nn <= 640 else 0,
                         polish=False)
            pool_blend(state, rng, max_evals=12)

    # Final polish using whatever remains.
    if state["seq"] is not None and state["seq"].size > 0:
        remaining = deadline - time.monotonic()
        if remaining > 40.0:
            limit = min(180.0, remaining - 5.0)
            local_refine(state["seq"], limit, rng, state,
                         use_lp=(state["seq"].size <= 640),
                         max_lp=100 if state["seq"].size <= 640 else 0,
                         polish=True)

    if state["seq"] is None:
        return [float(1.0 / 3.0), float(2.0 / 3.0)]

    # Final official validation / tie-break over best and a few pool entries.
    final_seq = state["seq"]
    final_score = state["score"]

    try:
        off = evaluate_sequence([float(x) for x in final_seq])
        if np.isfinite(off):
            final_score = float(off)
        print(f"final internal/official best score={final_score:.12f} n={final_seq.size}", flush=True)
    except Exception:
        pass

    candidates_to_check = [state["seq"]]
    if len(state["pool"]) > 1:
        state["pool"].sort(key=pool_sort_key)
        for item in state["pool"][:5]:
            if item["seq"] is not None and item["seq"].size not in [0]:
                candidates_to_check.append(item["seq"])

    seen = set()
    try:
        for seq in candidates_to_check:
            seq = np.asarray(seq, dtype=np.float64).reshape(-1)
            key = (seq.size, tuple(seq[:8].round(12)), tuple(seq[-8:].round(12)))
            if key in seen:
                continue
            seen.add(key)
            try:
                official = evaluate_sequence([float(x) for x in seq])
            except Exception:
                official = float("inf")
            if np.isfinite(official) and official < final_score:
                final_score = float(official)
                final_seq = seq
                print(f"official refined best score={final_score:.12f} n={final_seq.size}", flush=True)
    except Exception:
        pass

    out = [float(x) for x in np.asarray(final_seq, dtype=np.float64)]
    if not out:
        return [float(1.0 / 3.0), float(2.0 / 3.0)]
    return out
```
