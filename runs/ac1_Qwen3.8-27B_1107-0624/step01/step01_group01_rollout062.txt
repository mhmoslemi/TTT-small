def tuple_first(item):
    return item[0]


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):

    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())

def make_exp_profile(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * t)

    s = float(a.sum())
    return a / s


def make_power_profile(n, alpha, p):

    a = (1.0 + alpha * t) ** p

    s = float(a.sum())
    return a / s

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)

    return normalize_nonneg(np.interp(x1, x0, p0))

def peak_reduction_step(p, num_steps=100, rng=None):
    """Direct peak-reduction: iteratively reduce the max convolution coefficient."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    for step in range(num_steps):
        c = full_convolve_fft(p)
        k_star = int(np.argmax(c))
        max_val = c[k_star]

        # Gradient: reduce a[m] where a[k_star - m] is large
        # dC[k*]/da[m] = 2*a[k*-m] (with bounds)
        contrib = np.zeros(n)
        for m in range(n):
            jm = k_star - m
            if 0 <= jm < n:
                contrib[m] = 2.0 * p[jm]

        # Normalize and apply step
        contrib_mean = float(np.mean(contrib))
        contrib = contrib - contrib_mean
        contrib = np.maximum(contrib, 0.0)
        csum = contrib.sum()
        if csum > 1e-15:
            contrib = contrib / csum

        # Reduce elements contributing to peak, increase others
        lr = 0.05 * (max_val / max_val)  # scale
        p = p - lr * contrib
        p = p + lr * (1.0 / n)
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

    return p
