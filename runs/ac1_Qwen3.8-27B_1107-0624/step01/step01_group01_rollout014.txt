def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the linear programming direction-finding approach with multi-resolution
strategy, combined with Adam gradient polish in logit space.
"""

import numpy as np
import time
from scipy.optimize import linprog


def internal_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[x] = convolve(f,g)[x]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, timeout=10.0):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and convolve(f, g)[x] <= M for all x.
    Returns the optimal g (or None if infeasible).
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    Returns (best_f, best_score).

    n = len(f)

    f = normalize(f)
    if f is None:
        return None, np.inf

    c = np.convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * M
    best_M = M

    no_improve = 0
    t_step = t_init

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        g_raw = lp_direction(f, M)
        if g_raw is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break

        g = normalize(g_raw)
        if g is None:
            break

        # Try full step first
        g_self_score = 2.0 * n * float(np.convolve(g, g).max())

        if g_self_score < best_score * 0.999:
            # Full step is good
            f = g.copy()
            M = float(np.convolve(f, f).max())
            best_f = f.copy()
            best_score = 2.0 * n * M
            best_M = M
            no_improve = 0
        else:
            # Blended step
            f_new = (1.0 - t_step) * f + t_step * g
            f_new = normalize(f_new)
            if f_new is None:
                break
            c_new = np.convolve(f_new, f_new)
            M_new = float(c_new.max())
            score_new = 2.0 * n * M_new

            if score_new < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score_new
                best_M = M_new
                f = f_new.copy()
                M = M_new
                no_improve = 0
                # Decay step size slowly
                t_step = max(t_min, t_step * 0.995)
            else:
                no_improve += 1
                f = best_f.copy()
                M = best_M
                if no_improve % 20 == 0:
                    t_step = max(t_min, t_step * 0.7)
                if no_improve > 200:
                    break

        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={best_M:.6f}, t={t_step:.4f}")

    return best_f, best_score


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def make_linear_profile(n, eta):
    """Create a linear profile 1 + eta*(2t-1), normalized."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + eta * u
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_multi_profile(n, coeffs):
    """Create profile from polynomial coefficients: a(u) = sum coeffs[k] * u^k, u=2t-1."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.ones(n)
    for k, c in enumerate(coeffs, start=1):
        a += c * (u ** k)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..n-1."""
    n = len(p)
    h = np.zeros(n)
    for i in range(n):
        h[i] = np.dot(w[i:i + n], p)
    return h


def adam_polish(p0, deadline, lr=0.04, reg=1e-8, verbose=True):
    """
    Adam gradient descent in logit space to polish a non-negative sequence.
    Minimizes 2*n*max(conv(p,p)) with sum(p)=1.
    Returns (best_p, best_score).
    """
    n = len(p0)
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0

    c = np.convolve(p0, p0)
    max_c = float(c.max())
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * max_c
    no_imp = 0
    base_lr = lr
    beta = max(1e-12, 0.2 * max_c)

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(p0, p0)
        max_c = float(c.max())

        # Soft argmax weights for differentiating the max
        x = (c - max_c) / beta
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s

        # Gradient of max(c) w.r.t. p: grad_p[i] = 2 * sum_x w[x] * p[x-i]
        h = corr_weights(w, p0)
        grad_p = 2.0 * h + reg * p0

        # Logit space gradient
        dot = float(np.dot(p0, grad_p))
        gz = p0 * (grad_p - dot)

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            it = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)

        p = np.exp(z - z.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            z = best_z.copy()
            continue
        p = p / p_sum

        c = np.convolve(p, p)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.6)

        if no_imp > 2000:
            # Restart with perturbation
            z = 0.8 * best_z + 0.3 * np.random.default_rng(it).normal(size=n)
            z = np.clip(z, -50, 50)
            z -= np.mean(z)
            m.fill(0)
            v.fill(0)
            it = 0
            lr = base_lr
            no_imp = 0
            beta = max(1e-10, 0.2 * float(np.convolve(
                np.exp(z - z.max()),
                np.exp(z - z.max())
            ).max()))

        p0 = p

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-resolution LP + Adam search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
      1. Initialize from provided sequence or good analytic profiles.
      2. LP-based iterative refinement at n=128 (fast, many iterations).
      3. Upsample to n=256, continue LP refinement.
      4. Upsample to n=512, continue LP refinement.
      5. Adam gradient polish at n=512 for final improvement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 15.0)
    rng = np.random.default_rng(seed)

    # Get initial sequence
    best_seq = None
    best_score = np.inf

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                parent = parent / parent.sum()
                best_seq = parent
                best_score = internal_score(parent)
                print(f"Parent sequence: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Generate good initial profiles
        n_init = 256
        profiles = [
            make_linear_profile(n_init, 1.5),
            make_linear_profile(n_init, 2.0),
            make_multi_profile(n_init, [1.5, -0.8]),
            make_multi_profile(n_init, [1.2, 0.5, -0.3]),
            make_multi_profile(n_init, [2.0, -1.5, 0.5]),
        ]
        for p in profiles:
            sc = internal_score(p)
            if sc < best_score:
                best_score = sc
                best_seq = p
        print(f"Generated initial: n={len(best_seq)}, score={best_score:.8f}")

    # Phase 1: LP at n=128
    n1 = 128
    deadline_p1 = min(deadline, t_start + 400.0)
    a1 = interpolate_seq(best_seq, n1)
    if a1 is not None:
        print(f"\nPhase 1: LP at n={n1}, initial score={internal_score(a1):.8f}")
        # Time one LP solve
        t0 = time.monotonic()
        M_init = float(np.convolve(a1, a1).max())
        _ = lp_direction(a1, M_init, timeout=5)
        t_lp = time.monotonic() - t0
        print(f"  LP time: {t_lp*1000:.1f}ms")

        max_iters_1 = min(2000, max(100, int((deadline_p1 - time.monotonic()) * 0.7 / max(t_lp + 0.01, 0.01))))
        f_best, sc_best = lp_iterate(a1, max_iters=max_iters_1, t_init=0.4, t_min=0.02,
                                      deadline=deadline_p1, verbose=True)
        if f_best is not None and sc_best < best_score:
            best_score = sc_best
            best_seq = f_best
            print(f"  Phase 1 result: score={best_score:.8f}")

    # Phase 2: LP at n=256
    n2 = 256
    deadline_p2 = min(deadline, t_start + 900.0)
    if time.monotonic() < deadline_p2 - 60:
        a2 = interpolate_seq(best_seq, n2)
        if a2 is not None:
            print(f"\nPhase 2: LP at n={n2}, initial score={internal_score(a2):.8f}")
            t0 = time.monotonic()
            M_init = float(np.convolve(a2, a2).max())
            _ = lp_direction(a2, M_init, timeout=10)
            t_lp = time.monotonic() - t0
            print(f"  LP time: {t_lp*1000:.1f}ms")

            max_iters_2 = min(800, max(50, int((deadline_p2 - time.monotonic()) * 0.7 / max(t_lp + 0.02, 0.02))))
            f_best, sc_best = lp_iterate(a2, max_iters=max_iters_2, t_init=0.35, t_min=0.02,
                                          deadline=deadline_p2, verbose=True)
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 2 result: score={best_score:.8f}")

    # Phase 3: LP at n=512
    n3 = 512
    deadline_p3 = min(deadline, t_start + 1500.0)
    if time.monotonic() < deadline_p3 - 60:
        a3 = interpolate_seq(best_seq, n3)
        if a3 is not None:
            print(f"\nPhase 3: LP at n={n3}, initial score={internal_score(a3):.8f}")
            t0 = time.monotonic()
            M_init = float(np.convolve(a3, a3).max())
            _ = lp_direction(a3, M_init, timeout=30)
            t_lp = time.monotonic() - t0
            print(f"  LP time: {t_lp*1000:.1f}ms")

            max_iters_3 = min(300, max(20, int((deadline_p3 - time.monotonic()) * 0.7 / max(t_lp + 0.05, 0.05))))
            f_best, sc_best = lp_iterate(a3, max_iters=max_iters_3, t_init=0.3, t_min=0.02,
                                          deadline=deadline_p3, verbose=True)
            if f_best is not None and sc_best < best_score:
                best_score = sc_best
                best_seq = f_best
                print(f"  Phase 3 result: score={best_score:.8f}")

    # Phase 4: Adam polish at n=512
    if time.monotonic() < deadline - 20:
        n4 = len(best_seq)
        if n4 != 512:
            best_seq = interpolate_seq(best_seq, 512)
            n4 = 512
        print(f"\nPhase 4: Adam polish at n={n4}, initial score={internal_score(best_seq):.8f}")
        p_best, sc_best = adam_polish(best_seq, deadline - 20, lr=0.03, reg=1e-8, verbose=True)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = p_best
            print(f"  Phase 4 result: score={best_score:.8f}")

    # Final validation and cleanup
    final_score = internal_score(best_seq)
    print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = make_linear_profile(256, 1.5)
        s = best_seq.sum()
    best_seq = best_seq / s

    return [float(x) for x in best_seq]
