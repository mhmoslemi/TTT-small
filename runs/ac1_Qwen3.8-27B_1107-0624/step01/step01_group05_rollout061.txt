Minimize 2*n*max(autocorrelation)/(sum)^2 for non-negative sequences.
Uses FISTA with smooth-max continuation, subgradient polishing with peak
equalization, and multi-start strategy. Works in logit space (soft
parameterization of the probability simplex).

import numpy as np
import time

# ─── Configuration ───────────────────────────────────────────────────────────
N = 4096
FFT_S = 1 << (2 * N - 1).bit_length()       # 8192
FFT_G = 1 << (3 * N - 2).bit_length()        # 16384
CLIP = 45.0

_best_score = float("inf")
_best_p = None


# ─── Core computations ───────────────────────────────────────────────────────

def _self_conv(p):
    """Full linear autocorrelation of p (length N) via FFT."""
    f = np.fft.rfft(p, FFT_S)
    c = np.fft.irfft(f * f, FFT_S)
    c = c[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _true_score(p):
    """Exact objective: 2*N*max(p*p)."""
    c = _self_conv(p)
    return 2.0 * N * float(c.max())


def _z_to_p(z):
    """Logits → probability vector (length N, sums to 1)."""
    z = np.clip(z, -CLIP, CLIP)
    m = float(z.max())
    ez = np.exp(z - m)
    Z = np.exp(-m) + ez.sum()
    p = np.empty(N, dtype=np.float64)
    p[0] = np.exp(-m) / Z
    p[1:] = ez / Z
    return p


def _p_to_z(p):
    """Probability vector → logits (relative to index 0)."""
    p = np.maximum(p, 1e-15)
    return np.clip(np.log(p[1:]) - np.log(p[0]), -CLIP, CLIP)


def _smooth_max_and_grad_p(p, alpha):
    """
    Smooth max surrogate of max_k (p*p)[k] and its gradient w.r.t. p.
    Returns (value, grad_p_of_length_N).

    c = _self_conv(p)
    cmax = float(c.max())
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(e.sum())

        w = np.zeros(2 * N - 1)
        w[int(c.argmax())] = 1.0

    w_flipped = w[::-1]
    conv = np.fft.irfft(np.fft.rfft(p, FFT_G) * np.fft.rfft(w_flipped, FFT_G), FFT_G).real

    return val, grad_p

    """Smooth max objective and gradient w.r.t. logits z (length N-1)."""
    p = _z_to_p(z)
    val, grad_p = _smooth_max_and_grad_p(p, alpha)
    sdot = float(grad_p @ p)
    grad_z = p[1:] * (grad_p[1:] - sdot)
    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)
    return val, grad_z


def _update_best(p, score=None):
    global _best_score, _best_p
    if score is None:
        score = _true_score(p)
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_p = p.copy()


# ─── FISTA optimizer ─────────────────────────────────────────────────────────

def _fista(z0, alpha, max_iter, deadline):
    """
    FISTA (accelerated gradient) on logit space with backtracking.
    Returns final z and best true-score achieved.
    """
    z = np.array(z0, dtype=np.float64).copy()
    t_mom = 1.0
    z_prev = z.copy()
    step = 0.1

    for i in range(max_iter):
        if time.monotonic() >= deadline:
            break

        f_y, grad_y = _obj_grad_z(z, alpha)

        # Momentum
        if i > 0:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            beta = (t_mom - 1.0) / (t_new + 1.0)
            y = z + beta * (z - z_prev)
            t_mom = t_new
            f_y, grad_y = _obj_grad_z(y, alpha)
        else:
            y = z
            t_new = t_mom

        # Backtracking line search
        steplocal = step
        gnorm2 = float(grad_y @ grad_y)
        accepted = False
        for _bt in range(40):
            z_new = y - steplocal * grad_y
            z_new = np.clip(z_new, -CLIP, CLIP)
            f_new, _ = _obj_grad_z(z_new, alpha)
            if f_new <= f_y - 0.01 * steplocal * gnorm2:
                accepted = True
                break
            steplocal *= 0.5

        z_prev = z
        z = z_new
        p = _z_to_p(z)
        ts = _true_score(p)
        _update_best(p, ts)
        step = max(steplocal * 1.4, 1e-8)

    return z


# ─── Subgradient polishing with peak equalization ────────────────────────────

def _subgrad_polish(z0, deadline, rng, max_iters=20000):
    """
    Subgradient method with adaptive step size and peak-aware weighting.
    Uses top-K peaks for gradient direction.
    """
    z = np.array(z0, dtype=np.float64).copy()
    p = _z_to_p(z)
    score = _true_score(p)
    _update_best(p, score)

    # Backtracking step size
    step = 0.02
    fails = 0

    for i in range(max_iters):
        if time.monotonic() >= deadline:
            break

        c = _self_conv(p)
        cmax = float(c.max())

        # Weight top-K peaks with exponential penalty
        topk = min(16, 2 * N - 1)
        idx = np.argpartition(c, -topk)[-topk:]
        cw = c[idx]
        # Soft weighting favoring peaks
        wc = np.exp(500.0 * (cw - cw.max()))
        wc /= wc.sum()
        w = np.zeros(2 * N - 1)
        w[idx] = wc

        # Gradient w.r.t. p
        w_flipped = w[::-1]
        conv = np.fft.irfft(np.fft.rfft(p, FFT_G) * np.fft.rfft(w_flipped, FFT_G), FFT_G).real
        grad_p = 2.0 * conv[N - 1:2 * N - 1][::-1]

        # Gradient w.r.t. z
        sdot = float(grad_p @ p)
        grad_z = p[1:] * (grad_p[1:] - sdot)
        gnrm = float(np.linalg.norm(grad_z))

        if gnrm < 1e-15:
            break

        # Backtracking
        accepted = False
        local_step = step
        for _bt in range(20):
            z_new = z - local_step * grad_z / gnrm
            z_new = np.clip(z_new, -CLIP, CLIP)
            p_new = _z_to_p(z_new)
            s_new = _true_score(p_new)
            if s_new < score - 1e-13:
                z = z_new
                p = p_new
                score = s_new
                _update_best(p, score)
                accepted = True
                step = min(0.1, local_step * 1.3)
                fails = 0
                break
            local_step *= 0.5

        if not accepted:
            fails += 1
            step = max(1e-6, step * 0.7)
            if fails > 50:
                # Try a random perturbation
                z = z + rng.normal(0.0, 0.05, N - 1)
                z = np.clip(z, -CLIP, CLIP)
                p = _z_to_p(z)
                s2 = _true_score(p)
                if s2 < score:
                    score = s2
                    _update_best(p, score)
                fails = 0


# ─── Starting point generators ───────────────────────────────────────────────

def _z_uniform():
    return np.zeros(N - 1)


def _z_exponential(beta):
    x = np.linspace(0, 1, N)
    f = np.exp(-beta * x)
    f = np.maximum(f, 1e-12)
    return _p_to_z(f / f.sum())


def _z_logistic(beta, center=0.5):
    x = np.linspace(0, 1, N)
    f = 1.0 / (1.0 + np.exp(-beta * (x - center)))
    f = np.maximum(f, 1e-12)
    return _p_to_z(f / f.sum())


def _z_bell(center, width):
    x = np.linspace(0, 1, N)
    f = np.exp(-0.5 * ((x - center) / width) ** 2)
    f = np.maximum(f, 1e-12)
    return _p_to_z(f / f.sum())


def _z_power(a):
    x = np.linspace(1e-6, 1, N)
    f = x ** a
    f = np.maximum(f, 1e-12)
    return _p_to_z(f / f.sum())


def _z_talk_ang():
    """Tangerine-shape: high in middle, low at edges."""
    x = np.linspace(0, 1, N)
    f = 1.0 - 2.0 * (x - 0.5) ** 2
    f = np.maximum(f, 1e-8)
    return _p_to_z(f / f.sum())


def _z_w_shaped():
    """W-shape: peaks at 0.25 and 0.75."""
    x = np.linspace(0, 1, N)
    f = np.exp(-40.0 * (x - 0.25) ** 2) + np.exp(-40.0 * (x - 0.75) ** 2) + 0.1
    return _p_to_z(f / f.sum())


def _z_random_poly(rng):
    x = np.linspace(0, 1, N)
    t = x - 0.5
    deg = rng.integers(3, 8)
    coef = rng.normal(0, 1, deg + 1)
    f = np.ones(N)
    val = np.ones(N)
    for k in range(deg, 0, -1):
        val = val * t + coef[k]
    f += val
    f = np.maximum(f, 1e-6)
    return _p_to_z(f / f.sum())


def _z_random_fourier(rng, nfreq=8):
    x = np.linspace(0, 1, N)
    f = np.ones(N)
    for k in range(nfreq):
        freq = rng.integers(1, 12)
        amp = rng.normal(0, 1.5 / (freq + 1))
        ph = rng.uniform(0, 2 * np.pi)
        f += amp * np.sin(2 * np.pi * freq * x + ph)
    f = np.maximum(f, 1e-6)
    return _p_to_z(f / f.sum())


def _make_starts(rng):
    starts = []
    starts.append(_z_uniform())
    for beta in [1, 2, 3, 5, 8, 12]:
        starts.append(_z_exponential(beta))
        starts.append(_z_exponential(-beta))
    for beta in [2, 4, 8, 15]:
        starts.append(_z_logistic(beta, 0.3))
        starts.append(_z_logistic(beta, 0.7))
        starts.append(_z_logistic(-beta, 0.5))
    for c, w in [(0.5, 0.1), (0.5, 0.2), (0.5, 0.3), (0.3, 0.15), (0.7, 0.15)]:
        starts.append(_z_bell(c, w))
    for a in [-1, -0.5, 0.5, 1, 2, 3]:
        starts.append(_z_power(a))
    starts.append(_z_talk_ang())
    starts.append(_z_w_shaped())
    for _ in range(20):
        starts.append(_z_random_poly(rng))
    for _ in range(20):
        starts.append(_z_random_fourier(rng))
    return starts


# ─── Continuation FISTA ──────────────────────────────────────────────────────

def _continuation_fista(z0, deadline, max_total_iter=5000):
    """
    FISTA with alpha continuation: start with small alpha, increase gradually.
    """
    z = np.array(z0, dtype=np.float64).copy()
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000, 20000000]
    iters_per_alpha = max(50, max_total_iter // len(alphas))

    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        deadline_local = min(deadline, time.monotonic() + 0.1 * (deadline - time.monotonic()))
        z = _fista(z, alpha, iters_per_alpha, deadline_local)

    return z


# ─── Main search ─────────────────────────────────────────────────────────────

def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_p

    start_time = time.monotonic()
    end_time = start_time + max(1.0, float(budget_s)) - 3.0

    _best_score = float("inf")
    _best_p = None

    rng = np.random.default_rng(seed)

    print(f"Starting search with N={N}, budget={budget_s:.0f}s")

    # ── Try to use previous best as a starting point ──
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s > 0 and base.size == N:
            p0 = base / s
            score0 = _true_score(p0)
            _update_best(p0, score0)
            print(f"Previous best: score={score0:.6f}")
        elif s > 0:
            # Resample to N
            xs_old = np.linspace(0, 1, base.size)
            xs_new = np.linspace(0, 1, N)
            p0 = np.interp(xs_new, xs_old, base)
            p0 = np.maximum(p0, 0)
            p0 /= p0.sum()
            score0 = _true_score(p0)
            _update_best(p0, score0)
            print(f"Resampled previous best: score={score0:.6f}")
    except Exception:
        pass

    # Uniform baseline
    p_uniform = np.ones(N) / N
    _update_best(p_uniform, _true_score(p_uniform))
    print(f"Uniform baseline: score={_best_score:.6f}")

    # ── Generate starting points ──
    starts = _make_starts(rng)

    # Add the previous best as a start
    if _best_p is not None:
        starts.insert(0, _p_to_z(_best_p))

    # ── Evaluate and sort starts ──
    start_scores = []
    for z in starts:
        p = _z_to_p(z)
        s = _true_score(p)
        _update_best(p, s)
        start_scores.append(s)
    start_scores = np.array(start_scores)
    order = np.argsort(start_scores)
    print(f"Best start score: {start_scores[order[0]]:.6f} (out of {len(starts)} starts)")

    # ── Phase 1: FISTA from top starts ──
    phase1_deadline = start_time + 0.55 * (budget_s - 5.0)
    n_starts_to_try = min(30, len(starts))

    for rank in range(n_starts_to_try):
        if time.monotonic() >= phase1_deadline:
            break
        idx = int(order[rank])
        z0 = starts[idx]
        rem = phase1_deadline - time.monotonic()
        if rem < 10:
            break

        # Adaptive iterations based on remaining time
        max_iter = min(3000, max(200, int(rem * 80)))
        z = _continuation_fista(z0, time.monotonic() + min(rem * 0.9, 120.0), max_iter)

        print(f"  Start {rank}: best={_best_score:.6f} (elapsed={time.monotonic()-start_time:.1f}s)")

    # ── Phase 2: Subgradient polishing on best ──
    if _best_p is not None:
        z_best = _p_to_z(_best_p)
        phase2_deadline = start_time + 0.85 * (budget_s - 5.0)
        _subgrad_polish(z_best, phase2_deadline, rng)
        print(f"After subgradient polish: best={_best_score:.6f} (elapsed={time.monotonic()-start_time:.1f}s)")

    # ── Phase 3: Additional FISTA from perturbed best ──
    if _best_p is not None:
        z_best = _p_to_z(_best_p)
        phase3_deadline = end_time - 10.0
        attempts = 5
        for a in range(attempts):
            if time.monotonic() >= phase3_deadline:
                break
            noise = rng.normal(0, 0.03, N - 1)
            z0 = np.clip(z_best + noise, -CLIP, CLIP)
            rem = phase3_deadline - time.monotonic()
            if rem < 15:
                break
            z = _fista(z0, 5000000, min(2000, int(rem * 60)), time.monotonic() + min(rem * 0.8, 60.0))
            z_best = _p_to_z(_best_p) if _best_p is not None else z0
            print(f"  Phase3 attempt {a}: best={_best_score:.6f}")

    # ── Phase 4: Final aggressive subgradient polish ──
    if _best_p is not None:
        z_best = _p_to_z(_best_p)
        _subgrad_polish(z_best, end_time, rng, max_iters=50000)

    # ── Construct final sequence ──
    if _best_p is not None:
        p = np.maximum(_best_p, 0.0)
        p = np.minimum(p, 1000.0)
        seq = [float(x) for x in p]
    else:
        seq = [1.0 / N] * N

    # Verify with evaluate_sequence
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score) or final_score > 2.0:
        seq = [1.0 / N] * N
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            final_score = 2.0

    # Also check previous best
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.clip(base, 0, 1000).tolist()
        base_score = float(evaluate_sequence(base))
        if np.isfinite(base_score) and base_score < final_score:
            seq = base
            final_score = base_score
    except Exception:
        pass

    print(f"FINAL: score={final_score:.6f}, length={len(seq)}, time={time.monotonic()-start_time:.1f}s")
    return seq
