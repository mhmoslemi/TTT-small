def sinusoidal_perturb(p, rng, num_freqs=5, max_amp=0.3):
    """Add sinusoidal perturbations to a normalized sequence."""

def compute_convolution(a, L=None):
    """Compute the full convolution of a with itself."""
    n = a.size
    if L is None:
        L = next_power_of_two(2*n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2*n-1].real
    return np.maximum(c, 0.0)


def gradient_smoothed_max(p, temperature=1.0, topk=None):
    """
    Compute gradient of smoothed max of self-convolution of p (normalized to sum 1).
    Returns (value, gradient) where value = 2n * smoothed_max and gradient is w.r.t. p.
    """
    n = p.size
    c = compute_convolution(p)
    maxc = float(c.max())
    
    if topk is not None:
        # Top-k approach: assign equal weight to top-k values
        k = min(topk, c.size)
        # Use a soft top-k: weight proportional to max(0, c_k - threshold)
        # threshold is the k-th largest value
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[k-1] if k < c.size else 0.0
        w = np.maximum(c - threshold, 0.0) + 1e-30
        w /= w.sum()
    else:
        # Softmax approach
        z = c / max(temperature, 1e-10)
        z = z - z.max()
        expz = np.exp(z)
        w = expz / expz.sum()
    
    # Value: smoothed max approximation
    value = float(np.dot(w, c))
    
    # Gradient: d/dp_i [sum_k w_k * (p*p)_k] = 2 * sum_k w_k * p_{(k-i mod n)}... 
    # More precisely: (p*p)_k = sum_i p_i * p_{k-i} (with bounds)
    # d(p*p)_k / dp_i = p_{k-i} if 0 <= k-i < n, else 0
    # So: dp [sum_k w_k (p*p)_k] = 2 * sum_k w_k * p_{k-i} for valid k-i
    # = 2 * (w * p)_i where the convolution is appropriate
    
    # Actually: let's think of it as cross-correlation
    # (w * p)_i = sum_k w_k * p_{k-i} (with valid indices)
    # But we need: sum_k w_k * [d(p*p)_k / dp_i] = sum_k w_k * p_{k-i} (where 0<=k-i<n)
    # This is the cross-correlation of w and p, evaluated at lag i
    
    # Using FFT: correlation(w, p) at lag i = sum_k w_k * conj(p_{k-i}) = sum_k w_k * p_{k-i}
    # In FFT terms: ifft(FFT(w) * conj(FFT(p))) gives the cross-correlation
    
    L = next_power_of_two(2*n - 1)
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    # Cross-correlation: sum_k w_k * p_{k-i}
    # This is ifft(W * conj(P)) but we need to be careful with the indexing
    # Actually, the cross-correlation r_i = sum_k w_k * p_{k-i}
    # In terms of convolution: (w * p_reverse)_something...
    # Let me think again.
    # sum_k w_k * p_{k-i} where k ranges over 0..2n-2 and k-i over 0..n-1
    # = sum_j w_{j+i} * p_j (substituting j = k-i)
    # This is the cross-correlation of w and p at lag i
    
    # Using FFT: cross_corr(w, p)[i] = ifft(conj(FFT(w)) * FFT(p))[i]
    # But our w is defined on 0..2n-2 and p on 0..n-1
    # Let's embed both in length L
    corr = np.fft.irfft(np.conj(W) * P, L).real
    # corr[i] = sum_k w_k * p_{(k-i) mod L}
    # We want the non-circular version, which is exactly corr[n-1 : 2n-1] or similar
    # Actually for our setup (w on 0..2n-2, p on 0..n-1, zero-padded to L):
    # corr[i] = sum_{k=0}^{L-1} w_k * p_{(k-i) mod L}
    # Since p is zero outside 0..n-1, this is = sum_{k=i}^{L-1} w_k * p_{k-i} (for i < L)
    # We want: grad_i = 2 * sum_k w_k * p_{k-i} where k-i is in 0..n-1
    # = 2 * sum_{j=0}^{n-1} w_{j+i} * p_j
    # This is exactly corr[i] for our setup (since w is defined on 0..2n-2)
    
    grad = 2.0 * corr[:n]
    
    # The full objective is 2*n*max(c) / sum(p)^2
    # With sum(p)=1: objective = 2*n*max(c)
    # Gradient w.r.t. p: 2*n * grad_above
    # But we need to project onto the simplex (or use the log-barrier trick)
    # For the softmax parameterization, we'll handle this differently
    
    return value, grad


def adam_softmax_optimize(p, budget_s, seed=0, use_topk=False, topk_frac=0.05):
    """
    Adam optimizer with softmax (log) parameterization for the simplex.
    p is normalized to sum 1.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 1.0:
        return p, actual_score_fft(p)
    
    # Initialize log-probabilities
    x = np.log(p + 1e-300)
    x -= x.mean()
    
    # Learning rate schedule
    base_lr = 0.02 if n < 512 else 0.015
    
    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    
    L = next_power_of_two(2*n - 1)
    best_p = p.copy()
    best_f = actual_score_fft(p)
    
    n_topk = max(1, int(n * topk_frac))
    
    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)
    
    # Temperature schedule: start high, decrease
    T0 = max(0.01, float(n) * 0.01)  # initial temperature
    T_min = max(0.001, float(n) * 0.001)
    
    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        
        # Temperature decreases with time
        T = T_min + (T0 - T_min) * np.exp(-3.0 * frac)
        
        # Compute convolution
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2*n-1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        
        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()
        
        # Compute weights
        if use_topk:
            sorted_idx = np.argsort(c)[::-1]
            k = min(n_topk, c.size)
            # Soft top-k based on temperature
            threshold = sorted_idx[k-1] if k < c.size else 0
            c_k = c[sorted_idx[k-1]] if k < c.size else 0.0
            w = np.maximum(c - c_k, 0.0)
            # Add small bump for the top-k to ensure they get weight
            w[sorted_idx[:k]] += T * 0.1
            w = w + 1e-30
            w /= w.sum()

def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def mix_greedy_uniform(n, lam):
    p = lam * greedy_normalized(n) + (1.0 - lam) * np.ones(int(n), dtype=np.float64)
    return normalize_nonneg(p)


def sinusoidal_perturb(p, rng, num_freqs=4, max_amp=0.25):
    a = np.asarray(p, dtype=np.float64).ravel().copy()
    n = a.size
    t = np.arange(n, dtype=np.float64)
    for _ in range(int(num_freqs)):
        freq = rng.uniform(1.0, max(1.5, float(n) // 5))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.0, max_amp)
        a = a + amp * np.sin(2.0 * np.pi * freq * t / n + phase)
    return normalize_nonneg(a)


def compute_self_conv(a):
    n = a.size
    if n <= 512:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def compute_cross_conv(f, g):
    n = f.size
    L = next_power_of_two(2 * n - 1)
    Pf = np.fft.rfft(f, L)
    Pg = np.fft.rfft(g, L)
    c = np.fft.irfft(Pf * Pg, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    c = compute_self_conv(a)
    if c.size == 0:
        return np.inf
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def adam_softmax_optimize(p, budget_s, seed=0, use_topk=False, topk_frac=0.04):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 1.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 128:
        base_lr = 0.04
    elif n < 512:
        base_lr = 0.025
    elif n < 1024:
        base_lr = 0.018
    else:
        base_lr = 0.012

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(2 * n - 1)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    n_topk = max(2, int(n * topk_frac))
    T0 = max(0.5, float(n) * 0.008)
    T_min = max(0.05, float(n) * 0.0005)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)
    stall_count = 0
    patience = 200

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        T = T_min + (T0 - T_min) * np.exp(-2.5 * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > patience and frac > 0.5:
            break

        if use_topk:
            k = min(n_topk, c.size)
            idx_sorted = np.argsort(c)[::-1]
            threshold = c[idx_sorted[k - 1]] if k < c.size else 0.0
            w = np.maximum(c - threshold, 0.0)
            w[idx_sorted[:k]] += T * 0.3
            w = w + 1e-30
            sw = w.sum()
            if sw <= 0 or not np.isfinite(sw):
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(c))] = 1.0
            else:
                w /= sw
        else:
            z = c / max(T, 1e-10)
            z = z - z.max()
            expz = np.exp(z)
            sz = expz.sum()
            if sz <= 0 or not np.isfinite(sz):
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(c))] = 1.0
            else:
                w = expz / sz

        W = np.fft.rfft(w, L)
        P = np.fft.rfft(p, L)
        corr = np.fft.irfft(np.conj(W) * P, L).real
        grad_p = 2.0 * n * corr[:n]

        if not np.all(np.isfinite(grad_p)):
            break

        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        if not np.all(np.isfinite(grad_x)):
            break

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)

        lr = base_lr * (0.7 + 0.6 * np.exp(-2.0 * frac))
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -20.0, 20.0)

        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break

        x = np.clip(x, -50.0, 50.0)
        x -= x.mean()
        e = np.exp(x - x.max())
        se = e.sum()
        if se <= 0 or not np.isfinite(se):
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def lp_cutting_plane(p, budget_s, verbose=False):
    if _scipy_opt is None:
        return p, actual_score_fft(p)

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4:
        return p, actual_score_fft(p)
    budget_s = float(budget_s)
    if budget_s < 15.0:
        return p, actual_score_fft(p)

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    best_f = f.copy()

    c = compute_self_conv(f)
    cur_M = float(c.max())
    best_M = cur_M

    max_len = 2 * n - 1
    if max_len <= 2000:
        rows = list(range(max_len))
    else:
        rows = list(np.linspace(0, max_len - 1, 1200, dtype=int).astype(int))
    row_set = set(rows)

    max_rows = min(max_len, 2000 if n <= 1024 else 3000)

    end = time.monotonic() + max(0.0, budget_s - 3.0)
    it = 0
    no_improve = 0

    while time.monotonic() < end:
        it += 1
        no_improve += 1
        rows_arr = np.asarray(rows, dtype=int)
        nr = rows_arr.size

        if nr * n > 8000000:
            rows_arr = rows_arr[:min(nr, 8000000 // n)]
            nr = rows_arr.size

        A = np.zeros((nr, n), dtype=np.float64)
        for ii in range(nr):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(nr, cur_M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 0.5)
        tl = max(2.0, min(45.0, rem * 0.3))

        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None),
                method='highs',
                options={'time_limit': float(tl), 'presolve': True}
            )
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        full_conv = compute_cross_conv(f, g)
        full_conv = full_conv[:2 * n - 1]
        tol = 1e-8 + 1e-8 * cur_M
        violations = full_conv - cur_M
        max_viol = float(violations.max()) if violations.size > 0 else 0.0

        if max_viol > tol and len(rows) < max_rows:
            order = np.argsort(violations)[::-1]
            added = 0
            for ix in order:
                if len(rows) >= max_rows or added >= 300:
                    break
                ix = int(ix)
                if violations[ix] > tol and ix not in row_set:
                    rows.append(ix)
                    row_set.add(ix)
                    added += 1

        g = g * (S0 / sg)

        accepted = None
        accepted_M = cur_M
        for step in (1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            cn = compute_self_conv(fn)
            Mn = float(cn.max())
            if Mn < accepted_M - 1e-12:
                accepted_M = Mn
                accepted = fn.copy()

        if accepted is None:
            if no_improve > 15:
                break
            continue

        f = accepted
        cur_M = accepted_M
        if cur_M < best_M:
            best_M = cur_M
            best_f = f.copy()
            no_improve = 0
            if verbose:
                print("    LP iter %d: M=%.8f (score=%.6f)" % (it, best_M, 2.0 * n * best_M / (S0 * S0)))

        if no_improve > 30:
            break

    return normalize_nonneg(best_f), float(2.0 * n * best_M / (S0 * S0))


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = actual_score_fft(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_score = float(score)
            best_p = a.copy()
        nn = int(a.size)
        old = best_by_n.get(nn)
        if old is None or score < old[0]:
            best_by_n[nn] = (float(score), a.copy())

    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                record(arr)
                print("Loaded height_sequence_1: n=%d, score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    print("Initial best_score = %.6f" % best_score)

    # Phase 1: Quick multistart (up to 100s)
    phase1_end = t0 + min(100.0, 0.06 * budget_s)

    lengths_to_try = [352, 368, 384, 400, 416, 256, 448, 512, 192]

    for nn in lengths_to_try:
        if time.monotonic() >= phase1_end:
            break
        inits = [
            greedy_normalized(nn),
            power_normalized(nn, 0.45),
            power_normalized(nn, 0.55),
            power_normalized(nn, 0.65),
            mix_greedy_uniform(nn, 0.3),
            mix_greedy_uniform(nn, 0.5),
            sinusoidal_perturb(greedy_normalized(nn), rng, 4, 0.2),
        ]
        for init in inits:
            record(init)

    # Short Adam on top candidates from Phase 1
    adam1_end = t0 + min(350.0, 0.20 * budget_s)

    if best_by_n:
        sorted_items = sorted(best_by_n.items(), key=lambda x: x[1][0])
        top_lengths = [int(k) for k, v in sorted_items[:6]]
    else:
        top_lengths = [384, 256, 512]

    for nn in top_lengths:
        if time.monotonic() >= adam1_end:
            break
        rec = best_by_n.get(nn)
        base = rec[1] if rec is not None else greedy_normalized(nn)

        alloc = min(35.0, max(8.0, (adam1_end - time.monotonic()) / 2.0))
        if alloc < 5.0:
            continue

        p_out, f_out = adam_softmax_optimize(base, alloc, seed=seed + nn, use_topk=False)
        record(p_out, f_out)

        p_out, f_out = adam_softmax_optimize(base, alloc, seed=seed + nn + 77, use_topk=True, topk_frac=0.04)
        record(p_out, f_out)

        # Sinusoidal perturbation variant
        pert = sinusoidal_perturb(base, rng, 3, 0.15)
        p_out, f_out = adam_softmax_optimize(pert, alloc, seed=seed + nn + 154, use_topk=True, topk_frac=0.03)
        record(p_out, f_out)

    print("After Phase 1 (multistart + Adam): best_score = %.6f at n=%d" %
          (best_score, best_p.size if best_p is not None else 0))

    # Phase 2: LP cutting-plane (main optimization)
    if _scipy_opt is not None:
        lp_end = t0 + min(budget_s - 80.0, budget_s * 0.82)

        if best_by_n:
            sorted_items = sorted(best_by_n.items(), key=lambda x: x[1][0])
            lp_lengths = list(dict.fromkeys([int(k) for k, v in sorted_items[:5]] + [384, 256, 512, 400]))
        else:
            lp_lengths = [384, 256, 512]

        seen = set()
        for nn in lp_lengths:
            if time.monotonic() >= lp_end:
                break
            if nn in seen:
                continue
            seen.add(nn)

            rec = best_by_n.get(nn)
            base = rec[1] if rec is not None else greedy_normalized(nn)

            alloc = min(250.0, max(40.0, (lp_end - time.monotonic()) * 0.35))
            p_out, f_out = lp_cutting_plane(base, alloc, verbose=(nn == 384))
            record(p_out, f_out)

            # Adam polish after LP
            if time.monotonic() < lp_end - 30.0:
                alloc2 = min(40.0, lp_end - time.monotonic() - 15.0)
                if alloc2 > 10.0:
                    p2, f2 = adam_softmax_optimize(p_out, alloc2, seed=seed + nn + 500,
                                                   use_topk=True, topk_frac=0.02)
                    record(p2, f2)

                    p2, f2 = adam_softmax_optimize(p_out, alloc2, seed=seed + nn + 600,
                                                   use_topk=False)
                    record(p2, f2)

        print("After Phase 2 (LP): best_score = %.6f at n=%d" %
              (best_score, best_p.size if best_p is not None else 0))
    else:
        # Fallback: extended Adam
        adam_end2 = t0 + min(budget_s - 80.0, budget_s * 0.82)
        attempt = 0
        while time.monotonic() < adam_end2 and attempt < 30:
            attempt += 1
            if best_p is not None:
                nn = best_p.size
                base = best_p
            else:
                nn = 384
                base = greedy_normalized(nn)

            sigma = 0.04 + 0.03 * (attempt % 5)
            noise = np.exp(rng.normal(0.0, sigma, nn))
            perturbed = normalize_nonneg(base * noise)

            alloc = min(60.0, max(10.0, adam_end2 - time.monotonic() - 5.0))
            p_out, f_out = adam_softmax_optimize(perturbed, alloc,
                                                 seed=int(time.monotonic() * 100) % 99999,
                                                 use_topk=True, topk_frac=0.03)
            record(p_out, f_out)

    # Phase 3: Final polishing
    polish_end = deadline
    if best_p is not None and time.monotonic() < polish_end:
        nn = best_p.size

        # Several rounds of decreasing perturbation + Adam
        for round_i in range(8):
            if time.monotonic() >= polish_end - 15.0:
                break
            sigma = 0.025 * np.exp(-0.3 * round_i)
            noise = np.exp(rng.normal(0.0, sigma, nn))
            perturbed = normalize_nonneg(best_p * noise)

            alloc = min(50.0, max(8.0, (polish_end - time.monotonic() - 5.0)))
            p_out, f_out = adam_softmax_optimize(perturbed, alloc,
                                                 seed=seed + 2000 + round_i,
                                                 use_topk=True, topk_frac=0.02)
            record(p_out, f_out)

        # Direct polish on best
        if time.monotonic() < polish_end - 20.0:
            alloc = min(50.0, polish_end - time.monotonic() - 10.0)
            if alloc > 10.0:
                p_out, f_out = adam_softmax_optimize(best_p, alloc, seed=seed + 9999,
                                                     use_topk=True, topk_frac=0.01)
                record(p_out, f_out)

                p_out, f_out = adam_softmax_optimize(best_p, alloc, seed=seed + 8888,
                                                     use_topk=False)
                record(p_out, f_out)

        # If time allows, try LP on the best
        if _scipy_opt is not None and time.monotonic() < polish_end - 40.0:
            nn = best_p.size
            alloc = min(60.0, polish_end - time.monotonic() - 10.0)
            if alloc > 20.0:
                p_out, f_out = lp_cutting_plane(best_p, alloc)
                record(p_out, f_out)

    print("Final: best_score = %.6f at n=%d" %
          (best_score, best_p.size if best_p is not None else 0))

    if best_p is None:
        best_p = greedy_normalized(384)
        print("WARNING: No valid candidate found, using greedy(384)")

    return scale_output(best_p, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation function."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main_search: %s" % str(e))
        import traceback
        traceback.print_exc()
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
