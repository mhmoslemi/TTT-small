def solve_lp_direction(a, M):
    """Solve LP to find best direction g given current a and bound M."""
    n = len(a)
    # A is (2n-1) x n: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # We can build this efficiently
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        A[k, j_start:j_end+1] = a[k - np.arange(j_start, j_end+1)]

    # In linprog: minimize -sum(b) subject to A_ub @ b <= b_ub, b >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve_auto(a):
    """Full linear auto-convolution using FFT."""
    n = a.size
    if n <= 1:
        return a * a
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Clip small negatives from FFT
    mn = c.min()
    if mn < 0:
        c = np.maximum(c, 0.0)
    return c


def score_of(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = fft_convolve_auto(a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1

def fft_conv(a, b=None):
    """Full linear convolution of a and b (or a with itself if b is None)."""
    if b is None:
        b = a
    n = a.size
    m = b.size
    need = n + m - 1

        return np.convolve(a, b)

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:need]

```python
"""
LP-based iterative optimization for step function coefficient sequences.

Minimizes score = 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative
sequences a. Uses linear programming to find optimal update directions
at each iteration, combined with line search for step size selection.
Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_conv_auto(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        c = np.convolve(a, a)
    else:
        N = next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        mn = float(c.min())
        if mn < -1e-12:
            c = np.maximum(c, 0.0)
        elif mn < 0:
            c += (-mn) * 0.5  # gentle correction
    return c


def fft_conv_cross(a, b):
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([])
    need = n + m - 1
    if need <= 32:
        return np.convolve(a, b)
    N = next_pow2(need)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:need]
    mn = float(c.min())
    if mn < -1e-12:
        c = np.maximum(c, 0.0)
    elif mn < 0:
        c += (-mn) * 0.5
    return c


def get_score(a):
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = fft_conv_auto(a)
    return float(2.0 * n * float(c.max()) / (s * s))


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def interp_1d(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return a.copy()
    if n0 == 1:
        return np.full(n_new, float(a[0]))
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, a)


def prof_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    return a / a.sum()


def prof_exp(n, eta=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(eta * t)
    return a / a.sum()
