We want to minimize L = soft_max(c) where c = conv(p, p) and p is on the simplex.

The soft max is: L = m + beta * log(sum exp((c_k - m)/beta)) where m = max(c).

dL/dc_k = exp((c_k - m)/beta) / sum_j exp((c_j - m)/beta) = w_k

So dL/dp_i = sum_k w_k * dc_k/dp_i

Now, c_k = sum_j p_j * p_{k-j} (with bounds ensuring 0 <= j <= n-1 and 0 <= k-j <= n-1)

dc_k/dp_i = p_{k-i} (if 0 <= k-i <= n-1, from the j=i term)
           + p_{k-i} (if 0 <= k-i <= n-1, from the k-j=i term, i.e., j=k-i)
           = 2*p_{k-i} (if 0 <= k-i <= n-1 and i != k-i)
           = p_{k-i} (if 0 <= k-i <= n-1 and i = k-i, i.e., 2i = k)

So dL/dp_i = sum_k w_k * 2*p_{k-i} * 1_{0<=k-i<=n-1} - sum_k w_k * p_{k-i} * 1_{2i=k}
= 2 * sum_k w_k * p_{k-i} * 1_{0<=k-i<=n-1} - w_{2i} * p_i * 1_{2i < 2n-1}

The first term: let j = k-i, then k = i+j, and the sum becomes:
sum_j w_{i+j} * p_j * 1_{0<=j<=n-1} = sum_{j=0}^{n-1} w_{i+j} * p_j

This is exactly the corr_weights function! h[i] = sum_{j=0}^{n-1} w[i+j] * p[j]

The second term is a small diagonal correction: w[2i] * p[i] if 2i < 2n-1.

So dL/dp_i ≈ 2*h[i] - w[2i]*p[i] (for large n, the correction is small)

import numpy as np
import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(e) / e.size
    return e / s

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-30:
        return None
    return a / s

def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def soft_max_vw(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w

def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)

def make_cubic_profile(n, a1, a3):

def make_basis_mixed(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u]
    m = 1
    while len(cols) < K:
        ang = 2.0 * np.pi * m * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
        m += 1
        if len(cols) > K:
            break
    return np.column_stack(cols[:K])

def make_basis_poly(n, K):
    t = np.linspace(0.0, 1.0, n)

    cols = []
    for k in range(1, K + 1):

def interpolate_profile(p0, n_new):

    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))

def add_candidate(candidates, p, score, limit=20):

    else:
        worst_idx = 0
        worst_val = candidates[0][0]
        for i in range(1, len(candidates)):
            if candidates[i][0] > worst_val:
                worst_val = candidates[i][0]
                worst_idx = i
        if float(score) < worst_val:
            candidates[worst_idx] = (float(score), p)
            candidates.sort(key=tuple_first)
