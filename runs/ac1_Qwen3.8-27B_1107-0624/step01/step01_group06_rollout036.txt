    result = np.fft.irfft(fa * fa, size)

    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:

**Alternative: Use the "Importance Sampling" gradient approach.**

Instead of the log-sum-exp smoothing (which approximates the max), directly sample from the peak positions. The gradient of max(c) with respect to a[i] is:
d max(c) / da[i] = sum_k 1[k = argmax(c)] * (a[k-i] + a[i] if k >= n-1 else a[k-i])

Wait, let me be more careful. c[k] = sum_j a[j] * a[k-j]. So:
dc[k]/da[i] = a[k-i] (if 0 <= k-i <= n-1)

And d max(c)/da[i] = dc[k*]/da[i] where k* = argmax(c) (or a weighted average over tied maxima).

So the gradient of J = 2n * max(c) / S^2 with respect to a[i] is:
dJ/da[i] = 2n * (dc[k*]/da[i] * S^2 - max(c) * 2S * da_sum/da[i]) / S^4
= 2n * (a[k*-i] * S^2 - 2S * max(c)) / S^4
= 2n * (a[k*-i] * S - 2 * max(c)) / S^3

(when k* is unique and 0 <= k*-i <= n-1)

This is a very simple gradient! But it's only defined at the maximum point, so it's discontinuous. The log-sum-exp smoothing makes it continuous.

**Key insight**: We can use a "soft argmax" with a very small tau to get a gradient that's close to the true max gradient but still smooth. The previous code does this, but always from scratch.

**My novel approach: "Iterative Peak Chasing"**

Instead of optimizing a smoothed approximation of J, directly work with the peaks:

1. Compute c = conv(a, a)
2. Find all peaks (local maxima above some threshold)
3. For each peak at position k, compute the "sensitivity vector" s_k[i] = a[k-i] (the gradient of c[k] w.r.t. a[i])
4. The "active" sensitivity is the sum of s_k over all peaks weighted by how much they exceed the target
5. Use this as the "gradient" for a descent step

This is different from the LSE approach because:
- It directly targets the actual peaks (not a smooth approximation)
- It can handle multiple peaks simultaneously
- The weighting can be made very aggressive (effectively zero-temperature)

Let me also think about a **second-order** approach. The Hessian of c[k] w.r.t. a is:
d²c[k]/(da[i]da[j]) = 1 if i+j = k (and valid indices), else 0

So the Hessian is very sparse - it's (nearly) a permutation matrix (shifted). This means we could potentially use a Newton-like step, but the constraint a >= 0 makes it complicated.

"""
Peak-directed optimization with multi-scale structure generation and
simulated annealing for minimizing J(a) = 2n*max(conv(a,a))/S^2.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n * max(conv(a,a)) / S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def compute_J_and_peaks(a, num_peaks=50):
    """Compute J and return the positions of the top peaks in the convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, c, []
    max_c = np.max(c)
    J = 2.0 * n * max_c / (S * S)
    # Find top peaks (positions where c is largest)
    peak_positions = np.argsort(c)[::-1][:num_peaks]
    return J, c, peak_positions


def make_structures(n, rng):
    """Generate diverse parametric structures for initialization."""
    structures = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    # Standard windows
    structures['sine'] = np.sin(np.pi * (x + 1) / (n + 1))
    structures['hamming'] = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    structures['hanning'] = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    t = x / (n - 1)
    structures['blackman'] = 0.42 - 0.5 * np.cos(2*np.pi*t) + 0.08 * np.cos(4*np.pi*t)
    structures['flat_top'] = 0.21557895 - 0.41663158*np.cos(2*np.pi*t) + 0.277263158*np.cos(4*np.pi*t) - 0.083578947*np.cos(6*np.pi*t) + 0.006947368*np.cos(8*np.pi*t)
    
    # Gaussian family
    for sf in [0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.45]:
        sigma = sf * n
        structures[f'gauss_{sf}'] = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
    
    # Beta-like (diverging at edges)
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        tt = np.clip((x + 0.5) / n, 1e-8, 1 - 1e-8)
        structures[f'beta_{alpha}'] = tt**(-alpha) * (1-tt)**(-alpha)
    
    # Cosine power
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        base = np.cos(np.pi * (x - mid) / (n + 1))
        structures[f'cospow_{p}'] = np.abs(base) ** (2 * p) + 0.01

    for alpha in [0.5, 1.0, 2.0, 3.0]:
        structures[f'edge_{alpha}'] = 0.3 + 0.7 * (np.abs(x - mid) / mid) ** alpha

    for p in [0.1, 0.3, 0.5, 1.0, 2.0]:
        structures[f'pow_{p}'] = (1.0 - (np.abs(x - mid) / mid) ** p) + 0.01

def lse_gradient_step(a, tau, lr, momentum_buf, momentum_coeff):
    """Single LSE-smoothed gradient step (lightweight, for polishing)."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return a, momentum_buf, 1e15
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted - np.max(shifted))
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    Tmax = c_max + tau * np.log(sum_exp)
    
    # Cross-correlation for gradient
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S**3)
    
    # Momentum update
    momentum_buf = momentum_coeff * momentum_buf - lr * grad
    a_new = a + momentum_buf
    a_new = np.maximum(a_new, 0.0)

def smooth_J_grad(a, tau):
    """Compute smoothed J and its gradient using log-sum-exp."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad

def stochastic_peak_grad(a, num_peaks=20):
    """Compute a stochastic gradient based on the top peaks."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return np.zeros(n)

    peak_idx = np.argsort(c)[::-1][:num_peaks]
    max_c = c[peak_idx[0]]

    # Average value of c is S^2 / len(c)
    avg_c = S * S / len(c)
    weights = np.maximum(c[peak_idx] - avg_c, 0)
    total_w = np.sum(weights)
    if total_w < 1e-10:
        weights = np.ones(num_peaks) / num_peaks
        total_w = 1.0
    else:
        weights = weights / total_w
    
    # Gradient: dJ/da[i] ≈ (2n/S^2) * sum_k w_k * dc[k]/da[i] - (4n*max_c/S^3)
    # dc[k]/da[i] = a[k-i] if valid
    grad = np.zeros(n)
    for w_k, k in zip(weights, peak_idx):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        indices = np.arange(i_lo, i_hi + 1)
        partners = k - indices
        grad[indices] += w_k * a[partners]

    approx_max = np.average(c[peak_idx], weights=weights)
    grad = (2.0 * n * grad) / (S * S) - (2.0 * n * approx_max) / (S * S * S)

"""
Hybrid peak-directed optimizer combining stochastic peak gradients,
block coordinate descent on convolution peaks, structure generation,
and LSE-smoothed Adam polishing to minimize J(a) = 2n*max(conv(a,a))/S^2.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n * max(conv(a,a)) / S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def stochastic_peak_grad(a, num_peaks=20):
    """Compute a stochastic gradient targeting the top peaks in the convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return np.zeros(n)

    # Get top peaks
    peak_idx = np.argsort(c)[::-1][:num_peaks]
    avg_c = (S * S) / len(c)

    # Weight peaks by excess over average
    excess = np.maximum(c[peak_idx] - avg_c, 0)
    total_excess = np.sum(excess)
    if total_excess < 1e-10:
        weights = np.ones(num_peaks) / num_peaks
    else:
        weights = excess / total_excess

    # Compute gradient contribution from each peak
    # c[k] = sum_i a[i] * a[k-i], so dc[k]/da[i] = a[k-i]
    grad = np.zeros(n)
    for w_k, k in zip(weights, peak_idx):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        indices = np.arange(i_lo, i_hi + 1)
        partners = k - indices
        grad[indices] += w_k * a[partners]

    # Full gradient of J = 2n * max_c / S^2
    # dJ/da[i] = 2n * (dc[k*]/da[i] * S^2 - max_c * 2S) / S^4
    #          = 2n * (dc[k*]/da[i] * S - 2*max_c) / S^3
    approx_max = float(np.average(c[peak_idx], weights=weights))
    grad = (2.0 * n * grad * S) / (S * S * S) - (4.0 * n * approx_max) / (S * S * S)

    return grad


def lse_J_and_grad(a, tau):
    """Compute LSE-smoothed objective and gradient."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    shifted -= np.max(shifted)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * (np.max(shifted) + np.log(sum_exp))
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # Cross-correlation for gradient
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad


def generate_structures(n, rng):
    """Generate diverse parametric structures."""
    structures = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    t = x / (n - 1)

    # Standard windows
    structures['sine'] = np.sin(np.pi * (x + 1) / (n + 1))
    structures['hamming'] = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    structures['hanning'] = 0.5 * (1 - np.cos(2 * np.pi * t))
    structures['blackman'] = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    structures['flattop'] = (0.21557895 - 0.41663158 * np.cos(2*np.pi*t)
                             + 0.277263158 * np.cos(4*np.pi*t)
                             - 0.083578947 * np.cos(6*np.pi*t)
                             + 0.006947368 * np.cos(8*np.pi*t))

    # Gaussian family
    for sf in [0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
        sigma = sf * n
        structures[f'gauss_{sf}'] = np.exp(-0.5 * ((x - mid) / sigma) ** 2)

    # Beta-like (edge-diverging)
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:
        tt = np.clip((x + 0.5) / n, 1e-8, 1 - 1e-8)
        structures[f'beta_{alpha}'] = tt**(-alpha) * (1.0 - tt)**(-alpha)

    # Cosine power
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        base = np.cos(np.pi * (x - mid) / (n + 1))
        structures[f'cospow_{p}'] = np.abs(base) ** (2.0 * p) + 0.01

    # Edge-enhanced (U-shaped)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        structures[f'edge_{alpha}'] = 0.2 + 0.8 * (np.abs(x - mid) / mid) ** alpha

    # Power taper
    for p in [0.1, 0.3, 0.5, 1.0, 1.5, 2.0]:
        structures[f'pow_{p}'] = (1.0 - (np.abs(x - mid) / mid) ** p) + 0.01

    # Multi-modal
    for nmod in [2, 3, 4]:
        for spread in [0.15, 0.25, 0.35]:
            components = []
            for m in range(nmod):
                center = (m + 0.5) * n / nmod
                sig = spread * n / nmod
                components.append(np.exp(-0.5 * ((x - center) / sig) ** 2))
            structures[f'multi_{nmod}_{spread}'] = np.sum(components)

    # Random smooth (filtered noise)
    for seed in range(8):
        rng_local = np.random.default_rng(seed + 500)
        noise = rng_local.normal(0, 1, n)
        kernel_size = max(3, n // 50)
        kernel = np.hanning(kernel_size)
        kernel /= np.sum(kernel)
        smooth = np.convolve(noise, kernel, mode='same')
        structures[f'randsm_{seed}'] = np.exp(0.5 * smooth)

    # Hybrid: window * bump
    for wname, warr in [('hanning', structures['hanning']), ('sine', structures['sine'])]:
        for sf in [0.3, 0.5]:
            sigma = sf * n
            bump = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
            structures[f'{wname}_bump_{sf}'] = warr * bump + 0.05

    return structures


def hybrid_optimize(a_init, budget, seed, verbose=True):
    """
    Hybrid optimizer combining:
    - Stochastic peak gradient descent (large moves)
    - Block coordinate descent on peak contributors (targeted moves)
    - Random perturbations (exploration)
    - LSE gradient steps (smooth exploitation)
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J(a)
    c = convolve_fft(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01

    # Annealing state
    temperature = best_J * 0.3
    min_temp = best_J * 0.0005
    cool_rate = 0.99995

    start_time = time.monotonic()
    last_print = start_time
    stall = 0
    mode_cycle = 0

    while time.monotonic() < deadline:
        mode_cycle += 1
        S = np.sum(a)

        # Choose mode based on cycle and stall
        if stall > 100:
            mode = "restart"
        elif mode_cycle % 10 == 0:
            mode = "perturb"
        elif mode_cycle % 3 == 0:
            mode = "block_cd"
        elif mode_cycle % 3 == 1:
            mode = "stoch_grad"
        else:
            mode = "lse_grad"

        if mode == "restart":
            # Reset from best with perturbation
            a = best_a.copy()
            w = int(rng.integers(n // 50, n // 10))
            s = int(rng.integers(0, n - w))
            a[s:s+w] *= np.exp(rng.normal(0, 0.3, w))
            a = np.maximum(a, 0)
            S_a = np.sum(a)
            if S_a > 0:
                a *= (n / S_a)
            c = convolve_fft(a)
            m[:] = 0
            v[:] = 0
            t_adam = 0
            temperature = best_J * 0.1
            stall = 0

        elif mode == "stoch_grad":
            # Stochastic peak gradient step
            n_peaks = int(10 + 20 * (best_J - 1.0))
            n_peaks = min(n_peaks, 50)
            grad = stochastic_peak_grad(a, n_peaks)

            if np.isfinite(grad).all():
                t_adam += 1
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * grad * grad
                m_hat = m / (1 - beta1 ** t_adam)
                v_hat = v / (1 - beta2 ** t_adam)

                a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = np.maximum(a_new, 0)
                S_new = np.sum(a_new)
                if S_new > 0.01:
                    a_new *= (n / S_new)
                    c_new = convolve_fft(a_new)
                    J_new = 2.0 * n * np.max(c_new) / (S_new * S_new)

                    if J_new < best_J or (temperature > min_temp and
                                          rng.random() < np.exp(-(J_new - best_J) / temperature)):
                        a = a_new
                        c = c_new
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            lr = min(lr * 1.01, 0.05)
                        else:
                            stall += 1
                    else:
                        stall += 1
                        lr = max(lr * 0.98, 0.001)

        elif mode == "lse_grad":
            # LSE-smoothed gradient step
            tau = max(0.1, 5.0 * (1.0 - (time.monotonic() - start_time) / max(budget, 1)))
            J, grad = lse_J_and_grad(a, tau)

            if np.isfinite(J) and np.isfinite(grad).all():
                t_adam += 1
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * grad * grad
                m_hat = m / (1 - beta1 ** t_adam)
                v_hat = v / (1 - beta2 ** t_adam)

                a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = np.maximum(a_new, 0)
                S_new = np.sum(a_new)
                if S_new > 0.01:
                    a_new *= (n / S_new)
                    c_new = convolve_fft(a_new)
                    J_new = 2.0 * n * np.max(c_new) / (S_new * S_new)

                    if J_new < best_J:
                        a = a_new
                        c = c_new
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall += 1

        elif mode == "block_cd":
            # Block coordinate descent on peak contributors
            peak_idx = np.argsort(c)[::-1][:20]
            k = peak_idx[0]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            indices = np.arange(i_lo, i_hi + 1)
            partners = k - indices
            contribs = a[indices] * a[partners]

            # Try reducing top contributors in blocks
            n_top = min(5, len(indices))
            top_pos = np.argsort(contribs)[::-1][:n_top]

            for _ in range(3):
                cp = top_pos[int(rng.integers(0, n_top))]
                i = indices[cp]
                partner = partners[cp]
                target = i if a[i] >= a[partner] else partner

                if a[target] < 1e-10:
                    continue

                # Try a few reduction fractions
                for frac in [0.02, 0.05, 0.01]:
                    cand = a.copy()
                    cand[target] *= (1.0 - frac)
                    c_new = convolve_fft(cand)
                    S_new = np.sum(cand)
                    if S_new < 0.01:
                        break
                    J_new = 2.0 * n * np.max(c_new) / (S_new * S_new)

                    if J_new < best_J or (temperature > min_temp and
                                          rng.random() < np.exp(-(J_new - best_J) / temperature)):
                        a = cand
                        c = c_new
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                        break
                else:
                    continue
                break

            # Also try a block-scale move: multiply a random block by (1+delta)
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width)))
            cand = a.copy()
            delta = rng.normal(0, 0.03)
            cand[start:start + width] *= np.exp(delta)
            cand = np.maximum(cand, 0)
            S_c = np.sum(cand)
            if S_c > 0.01:
                c_c = convolve_fft(cand)
                J_c = 2.0 * n * np.max(c_c) / (S_c * S_c)
                if J_c < best_J or (temperature > min_temp and
                                    rng.random() < np.exp(-(J_c - best_J) / temperature)):
                    a = cand
                    c = c_c
                    if J_c < best_J:
                        best_J = J_c
                        best_a = a.copy()
                        stall = 0

        elif mode == "perturb":
            # Larger random perturbation
            width = int(rng.integers(n // 100, n // 15))
            start = int(rng.integers(0, max(1, n - width)))
            cand = a.copy()
            noise = rng.normal(0, 0.05 + 0.1 * temperature / max(best_J, 1e-10), width)
            cand[start:start + width] *= np.exp(noise)
            cand = np.maximum(cand, 0)
            S_c = np.sum(cand)
            if S_c > 0.01:
                c_c = convolve_fft(cand)
                J_c = 2.0 * n * np.max(c_c) / (S_c * S_c)
                if J_c < best_J or (temperature > min_temp and
                                    rng.random() < np.exp(-(J_c - best_J) / temperature)):
                    a = cand
                    c = c_c
                    if J_c < best_J:
                        best_J = J_c
                        best_a = a.copy()
                        stall = 0

        # Cooling
        temperature = max(min_temp, temperature * cool_rate)

        # Adaptive learning rate
        if mode == "stoch_grad" and stall > 30:
            lr = max(0.001, lr * 0.95)

        # Print progress
        if verbose and time.monotonic() - last_print > 30:
            elapsed = time.monotonic() - start_time
            print(f"      hybrid: t={elapsed:.0f}s, J={best_J:.6f}, mode={mode}, stall={stall}, lr={lr:.4f}")
            last_print = time.monotonic()

    return best_a, best_J
