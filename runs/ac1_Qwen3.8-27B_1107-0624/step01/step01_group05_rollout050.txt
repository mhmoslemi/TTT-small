        fw = np.fft.rfft(w, _FFT_X)
        fp = np.fft.rfft(pr, _FFT_X)
        conv = np.fft.irfft(fw * fp, _FFT_X).real
        xcorr = conv[N - 1:2 * N - 1]
        gp = xcorr + p * float(np.sum(w))

        # Project gradient onto tangent space of simplex
        sdot = float(np.dot(gp, p))
        g_dir = gp - sdot  # tangent direction
        gnorm = float(np.linalg.norm(g_dir))

        if gnorm < 1e-15:
            kk = min(64, kk * 2)
            continue

        g_dir /= gnorm
        # Momentum
        v = 0.9 * v - step * g_dir

        accepted = False
        for _ in range(8):
            p_new = p + v
            # Project onto simplex
            p_new = _proj_simplex(p_new)
            s_new = _score_p(p_new)
            if s_new < score - 1e-14:
                p = p_new
                score = s_new
                _update_best(p, score)
                step = min(1.0, step * 1.15)
                accepted = True
                break
            v *= 0.5

        if not accepted:
            fails += 1
            step = max(0.01, step * 0.7)
            mom = 0.0
            v = np.zeros(N)
            if fails > 100:
                # Perturb and restart
                p = _proj_simplex(p + 0.05 * np.random.randn(N))
                score = _score_p(p)
                _update_best(p, score)
                fails = 0
                step = 0.3
                kk = 4
        else:
            fails = 0
            if score < 1.505:
                kk = min(64, kk * 2)


def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_candidates():
    """Generate initial candidate probability vectors."""
    cands = []

    # 1. Previous best
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.clip(base, 0.0, None)
        if base.sum() > 0 and base.size == N:
            cands.append(base / base.sum())
    except Exception:
        pass

    # 2. Uniform
    cands.append(np.ones(N) / N)

    # 3. Structured windows
    x = (np.arange(N, dtype=np.float64) + 0.5) / N

    # Exponential decay
    for beta in (1.0, 2.0, 3.0, 5.0, 8.0):
        f = np.exp(-beta * x)
        f /= f.sum()
        cands.append(f)
        f2 = np.exp(-beta * (1 - x))
        f2 /= f2.sum()
        cands.append(f2)

    # Logistic
    for beta in (2.0, 4.0, 6.0, 10.0):
        f = 1.0 / (1.0 + np.exp(-beta * (x - 0.5)))
        f /= f.sum()
        cands.append(f)

    # Power law
    for a in (0.5, 1.0, 2.0, 3.0):
        f = np.power(x + 1e-6, a)
        f /= f.sum()
        cands.append(f)
        f2 = np.power(1.0 - x + 1e-6, a)
        f2 /= f2.sum()
        cands.append(f2)

    # Cosine window
    f = np.cos(np.pi * (x - 0.5)) ** 2 + 0.01
    f /= f.sum()
    cands.append(f)

    # Gaussian
    for sig in (0.1, 0.15, 0.2, 0.3):
        f = np.exp(-0.5 * ((x - 0.5) / sig) ** 2)
        f /= f.sum()
        cands.append(f)

    # Chebyshev-inspired: |T_n(cos(pi*x))| type
    for order in (10, 20, 50):
        t = 2.0 * x - 1.0
        # Use cos(order * arccos(t)) as Chebyshev polynomial
        f = np.abs(np.cos(order * np.arccos(np.clip(t, -1, 1)))) + 0.01
        f /= f.sum()
        cands.append(f)

    # Triangular (inverted - mass in middle)
    f = 1.0 - np.abs(2.0 * x - 1.0)
    f = np.maximum(f, 0.01)
    f /= f.sum()
    cands.append(f)

    # Random perturbations of best
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.clip(base, 0.0, None)
        if base.sum() > 0 and base.size == N:
            base_n = base / base.sum()
            rng = np.random.default_rng(123)
            for scale in (0.05, 0.1, 0.2, 0.3):
                for _ in range(3):
                    noisy = base_n * (1.0 + scale * rng.standard_normal(N))
                    noisy = np.maximum(noisy, 0.0)
                    if noisy.sum() > 0:
                        cands.append(noisy / noisy.sum())
    except Exception:
        pass

    return cands


def _sort_candidates(cands):
    scored = []
    for i, p in enumerate(cands):
        sc = _score_p(p)
        _update_best(p, sc)
        scored.append((sc, i))
    scored.sort()
    return [cands[i] for _, i in scored]


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_p, _deadline

    start = time.monotonic()
    budget_s = float(budget_s)
    hard_deadline = start + max(1.0, budget_s) - 3.0

    _best_score = float('inf')
    _best_p = None

    print(f"Starting search, budget={budget_s:.1f}s, N={N}")

    # Generate and sort candidates
    cands = _make_candidates()
    cands_sorted = _sort_candidates(cands)
    print(f"Generated {len(cands)} candidates, best initial: {_best_score:.6f}")

    # Main optimization: continuation L-BFGS on best candidates
    # Allocate time: 70% to main optimization, 30% to polishing
    main_budget = 0.65 * budget_s
    polish_budget = 0.35 * budget_s

    main_deadline = start + main_budget

    # How many candidates to optimize
    n_optimize = min(len(cands_sorted), max(5, int(main_budget / 30)))

    # Continuation schedule: geometric sequence of alphas
    # Start smooth, end sharp
    alpha_start = 5e4
    alpha_end = 1.5e7
    n_stages = 5
    alpha_seq_full = np.geomspace(alpha_start, alpha_end, n_stages)

    for idx in range(n_optimize):
        if time.monotonic() > main_deadline:
            break
        rem = main_deadline - time.monotonic()
        if rem < 15:
            break

        p0 = cands_sorted[idx]
        z0 = _z_from_p(p0)

        # Scale maxfun based on remaining time
        max_fun_total = int(min(2000, max(200, rem / 0.008)))
        max_fun_per = max(60, max_fun_total // n_stages)

        # Use subset of alpha stages based on time
        n_use = min(n_stages, max(2, int(rem / 15)))
        alphas = alpha_seq_full[:n_use]
        # If this is a good candidate, go all the way
        if idx < 3:
            alphas = alpha_seq_full

        _lbfgs_run(z0, alphas, max_fun_per, main_deadline)

        sc = _best_score
        print(f"  candidate {idx}: best={sc:.6f}, elapsed={time.monotonic()-start:.1f}s")

    # Extra L-BFGS runs from the best with high alpha
    if _best_p is not None and time.monotonic() < main_deadline - 10:
        z_best = _z_from_p(_best_p)
        for extra_alpha in (2e7, 5e7):
            if time.monotonic() > main_deadline - 5:
                break
            rem = main_deadline - time.monotonic()
            z_start = z_best + np.random.default_rng(99).normal(0, 0.02, N - 1)
            _lbfgs_run(z_start, [extra_alpha], int(min(800, rem / 0.008)), main_deadline)

    print(f"After main optimization: best={_best_score:.6f}, elapsed={time.monotonic()-start:.1f}s")

    # Polishing phase
    if _best_p is not None:
        polish_deadline = start + budget_s - 4.0
        _polish_subgrad(_best_p, polish_deadline)
        print(f"After polishing: best={_best_score:.6f}, elapsed={time.monotonic()-start:.1f}s")

        # One more L-BFGS from polished best
        if time.monotonic() < polish_deadline - 10:
            z_best = _z_from_p(_best_p)
            rem = polish_deadline - time.monotonic() - 5
            if rem > 20:
                _lbfgs_run(z_best, [3e7, 8e7], int(min(500, rem / 0.01)), polish_deadline - 5)
                print(f"Final L-BFGS: best={_best_score:.6f}")

    # Convert best probability to sequence
    if _best_p is not None:
        seq_p = np.maximum(_best_p, 0.0)
        seq_p = seq_p / seq_p.sum()
        seq = [float(v) for v in seq_p]
    else:
        seq = [1.0 / N] * N

    # Verify with evaluate_sequence
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score) or final_score > 2.0:
        seq = [1.0 / N] * N
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            final_score = 2.0

    # Check if inherited sequence is better
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            base_score = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(base_score) and base_score < final_score:
                seq = [float(v) for v in base]
                final_score = base_score
    except Exception:
        pass

    print(f"FINAL: score={final_score:.6f}, length={len(seq)}, elapsed={time.monotonic()-start:.1f}s")
    return seq
