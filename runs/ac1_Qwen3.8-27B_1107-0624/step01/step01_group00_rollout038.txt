Let me also think about using the **dual** of the problem. The dual of:
$$\min_a \max_k c_k \quad \text{s.t.} \quad \sum a_i = 1, a_i \geq 0$$

is (informally):
$$\max_{w \in \Delta_{2n-1}} \min_{a \geq 0, \sum a = 1} \sum_k w_k c_k(a)$$

For a fixed $w$, the inner problem is:
$$\min_a \sum_k w_k \sum_i a_i a_{k-i} = \min_a \sum_{i,j} a_i a_j w_{i+j}$$

This is a quadratic form $a^T W a$ where $W_{ij} = w_{i+j}$ (a Hankel matrix). If $W$ is PSD, this is a convex QP.

The Hankel matrix $W$ with entries $w_{i+j}$ is PSD iff the sequence $w_0, w_1, \ldots$ is the autocorrelation of some sequence (by the Hankel matrix positivity criterion, related to the Carathéodory-Fejér theorem).

So the dual is:
$$\max_{w \in \Delta_{2n-1}, W(w) \succeq 0} \min_{a \geq 0, \sum a = 1} a^T W(w) a$$

where the inner minimum is $\lambda_{\min}^{+}(W(w))$ restricted to the simplex (which is the minimum of a quadratic form over the simplex).

This is complicated but suggests an algorithm:
1. Start with a distribution $w$ over the lags.
2. Compute the Hankel matrix $W(w)$.
3. If $W(w)$ is not PSD, project it to the nearest PSD Hankel matrix (or adjust $w$).
4. Find the minimizer $a^* = \arg\min_{a \geq 0, \sum a = 1} a^T W(a)$.
5. Update $w$ to put more weight on the lags where $c_k(a^*)$ is large.
6. Repeat.

This is essentially the "mirror descent" or "multiplicative weights" algorithm for the minmax problem. It might work well!

def _to_sum1(a):
    s = a.sum()
    if s <= 1e-12:
        return np.full(a.size, 1.0/a.size, dtype=np.float64)
    return a / s

def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.clip(v, 0, None) * (target / max(v.sum(), 1e-12))
    lam = (cssv[rho[-1]] - target) / (rho[-1] + 1.0)
    return np.maximum(v - lam, 0.0)

def _conv_sq(a, m=None):
    if m is None:
        m = 1
        while m < 2*a.size - 1:
            m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*a.size-1], m

def _subgrad_topk(c, a, k=1):
    """Compute subgradient of max(c) using top-k weighted average."""
    n = a.size
    k = min(k, n)
    order = np.argpartition(-c, k-1)[:k]
    # Uniform weights over top-k
    w = np.ones(k) / k
    grad = np.zeros(n, dtype=np.float64)
    for idx in range(k):
        ki = order[idx]
        wk = w[idx]
        # grad_p = 2 * wk * a[ki - p] for p in [max(0, ki-n+1), min(ki, n-1)]
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        p_idx = np.arange(lo, hi + 1)
        src = ki - p_idx
        grad[p_idx] += 2.0 * wk * a[src]
    return grad

def _smoothed_objective(c, tau):
    """Compute smoothed max: tau * log(sum(exp(c/tau)))"""
    c_max = c.max()
    return c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))

def _smoothed_grad(c, a, tau):
    """Gradient of the smoothed max objective."""
    n = a.size
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w = w / w.sum()
    grad = np.zeros(n, dtype=np.float64)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        p_idx = np.arange(lo, hi + 1)
        src = k - p_idx
        grad[p_idx] += 2.0 * wk * a[src]
    return grad

Hmm, the `_smoothed_grad` function loops over all $2n-1$ lags, which for $n = 512$ is 1023 iterations. Each iteration does an $O(n)$ operation. So total is $O(n \cdot 2n) = O(2n^2) \approx 500K$ for $n = 512$. That's fine.

But I can vectorize it using the correlation approach: `grad = 2 * correlate(c, a, mode='valid')`. Let me verify:
`correlate(c, a, mode='valid')` gives $h_p = \sum_j c[p+j] \cdot a[j]$ for $p = 0, \ldots, n-1$ (when $\text{len}(c) = 2n-1$ and $\text{len}(a) = n$).

And I need $grad_p = 2 \sum_k w_k a_{k-p}$. Let $j = k-p$, so $k = j+p$: $grad_p = 2 \sum_j w_{j+p} a_j = 2 \cdot \text{correlate}(w, a, mode='valid')_p$.

Yes! So `grad = 2.0 * scipy.signal.correlate(w, a, mode='valid')`.

This is much faster (uses FFT internally).

```python
"""
Minimize J = 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequence a.

Strategy:
1. Multiple parametric initializations + provided sequence
2. Primary optimizer: adaptive smoothed subgradient with momentum,
   backtracking-style step adjustment, and periodic restarts
3. Secondary optimizer: mirror descent (multiplicative weights) for diversity
4. Final polishing: exact subgradient (top-1) with tiny steps
5. Try multiple sequence lengths if time permits
"""

import time
import math
import numpy as np
from scipy.signal import correlate

# --- Helper functions (top-level, no closures) ---

def _to_sum1(a):
    s = a.sum()
    if s <= 1e-12:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s

def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 1:
        return np.array([target])
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.clip(v, 0.0, None)
    lam = (cssv[rho[-1]] - target) / (rho[-1] + 1.0)
    return np.maximum(v - lam, 0.0)

def _fft_size(n):
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    return m

def _conv_sq(a, m):
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * a.size - 1]

def _grad_topL(c, a, L, tau):
    """Subgradient of the smoothed max (top-L with softmax weights)."""
    n = a.size
    L = min(L, 2 * n - 1)
    order = np.argpartition(-c, L - 1)[:L]
    wc = c[order]
    w_local = np.exp((wc - wc.max()) / max(tau, 1e-12))
    w_local = w_local / w_local.sum()
    # Build full weight vector
    w = np.zeros(2 * n - 1, dtype=np.float64)
    w[order] = w_local
    # Gradient via correlation
    return 2.0 * correlate(w, a, mode='valid')

def _grad_top1(c, a):
    """Exact subgradient: 2 * a shifted to the argmax lag."""
    n = a.size
    k = int(np.argmax(c))
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    p_idx = np.arange(lo, hi + 1)
    src = k - p_idx
    g[p_idx] = 2.0 * a[src]
    return g

def _grad_smooth(c, a, tau):
    """Softmax gradient over all lags."""
    c_max = c.max()
    w = np.exp((c - c_max) / max(tau, 1e-12))
    w = w / w.sum()
    return 2.0 * correlate(w, a, mode='valid')

def _obj(a):
    """Objective value (assuming sum(a)=1): 2*n*max(conv(a))"""
    n = a.size
    m = _fft_size(n)
    c = _conv_sq(a, m)
    return 2.0 * n * c.max()

def _make_seeds(n, base, rng):
    """Generate diverse initial sequences."""
    seeds = []
    # Provided base
    if base is not None and base.size == n:
        b = np.clip(np.abs(base), 0, None)
        if b.sum() > 1e-6:
            seeds.append(_to_sum1(b))
            seeds.append(_to_sum1(b[::-1]))
    # Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))
    # Power laws
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** p
        seeds.append(_to_sum1(a))
    # Concave (bell-shaped)
    for p in [0.5, 1.0, 2.0]:
        a = np.sin(np.linspace(0, math.pi, n)) ** p
        seeds.append(_to_sum1(a))
    # Gaussian
    for sigma in [0.15, 0.25, 0.35]:
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
        seeds.append(_to_sum1(a))
    # Random structured
    for _ in range(4):
        a = np.exp(rng.normal(0, 0.8, n))
        seeds.append(_to_sum1(a))
    # Trimmed (zero out some edges)
    for frac in [0.05, 0.1, 0.2]:
        a = np.full(n, 1.0 / n)
        k = int(frac * n)
        a[:k] = 0
        a[-k:] = 0
        if a.sum() > 0:
            seeds.append(_to_sum1(a))
    return seeds

def _run_smoothed_subgrad(a_init, t_end, rng, verbose=False):
    """Run adaptive smoothed subgradient descent. Returns best sequence."""
    n = a_init.size
    m = _fft_size(n)
    a = _to_sum1(np.abs(a_init).copy().astype(np.float64))
    best_a = a.copy()
    best_obj = _obj(a)
    buf = np.zeros(n, dtype=np.float64)
    
    # Initial hyperparameters
    L = n
    tau = 0.5 / max(_conv_sq(a, m).max(), 1e-6)
    lr = 0.05
    momentum = 0.85
    step = 0
    stall = 0
    best_obj_run = best_obj
    
    while time.monotonic() < t_end - 0.5:
        c = _conv_sq(a, m)
        mc = c.max()
        obj = 2.0 * n * mc
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive schedule
        if step > 0 and step % 2000 == 0:
            L = max(4, int(L * 0.6))
            tau = max(0.02, tau * 0.7)
            momentum = min(0.95, momentum + 0.01)
        
        if step > 0 and step % 500 == 0:
            lr *= 0.95
        
        # Compute gradient
        if L <= 2:
            g = _grad_top1(c, a)
        elif tau < 0.05:
            g = _grad_topL(c, a, L, tau)
        else:
            g = _grad_topL(c, a, L, tau)
        
        # Normalize gradient
        gn = np.linalg.norm(g)

            g = g / gn

            a_new
