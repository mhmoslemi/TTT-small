"""
Search for non-negative step-function sequence minimizing f(seq) = 2*n*max(convolve(seq,seq))/sum(seq)^2.
Uses multi-phase optimization: (1) diverse multi-start, (2) intensive Adam + subgradient local search,
(3) peak-aware coordinate descent with simulated annealing, (4) LP cutting-plane polishing.
Starts from height_sequence_1 when available for continuity.
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _normalize(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(32)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones(a.size) / a.size
    return a / s


def _fft_score(p):
    """Compute 2*n*max(a*a)/sum(a)^2 using FFT. p is assumed normalized to sum=1."""
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    if n <= 256:
        c = np.convolve(a, a)
    else:
        L = _next_pow2(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = c.max()
    return 2.0 * n * maxc / (s * s)


def _conv_direct(a):
    """Full linear convolution, returns length 2n-1 array."""
    n = a.size
    if n <= 512:
        return np.convolve(a, a)
    L = _next_pow2(2 * n - 1)
    P = np.fft.rfft(a, L)
    return np.fft.irfft(P * P, L)[:2 * n - 1].real


def _subgradient(p, alpha=30.0):
    """
    Compute subgradient of max_k (p*p)[k] at point p.
    Uses softmax weighting over near-peak positions for smoothness.
    Returns gradient vector of length n.
    """
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    c = _conv_direct(a)
    c = np.maximum(c, 0.0)
    maxc = c.max()
    if maxc < 1e-30:
        return np.zeros(n)
    # Softmax weights over convolution positions
    z = alpha * (c / maxc - 1.0)
    expz = np.exp(z)
    sz = expz.sum()
    if sz < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = expz / sz
    # Gradient: grad[j] = sum_k w[k] * 2*a[k-j] (valid indices)
    # For efficiency, compute via FFT: grad = 2 * (w_rev * a) truncated
    L = _next_pow2(2 * n - 1)
    Wr = np.fft.rfft(w[::-1], L)
    Pa = np.fft.rfft(a, L)
    Cfull = np.fft.irfft(Wr * Pa, L)
    # Cfull[m] = sum_k w[k] * a[m-k] for appropriate ranges
    # We want grad[j] = sum_k w[k] * a[k-j] for k-j in [0,n-1]
    # = Cfull[n-1+j] ... let me verify indexing
    # w_rev[m] = w[2n-2-m], Pa is rfft of a
    # Cfull[t] = sum_m w_rev[m] * a[t-m] = sum_m w[2n-2-m] * a[t-m]
    # Let k = 2n-2-m, m = 2n-2-k: Cfull[t] = sum_k w[k] * a[t-2n+2+k]
    # We want sum_k w[k] * a[k-j], so set t-2n+2+k = k-j => t = 2n-2-j... that doesn't work directly.
    # Let me just compute it directly for correctness (n is at most ~4096)
    if n <= 2048:
        grad = np.zeros(n)
        for j in range(n):
            # grad[j] = sum_k w[k] * 2 * a[k-j] where k-j in [0, n-1]
            lo = max(0, j)
            hi = min(2 * n - 1, j + n - 1)
            idx = np.arange(lo, hi + 1)
            a_idx = idx - j  # these are in [0, n-1]
            grad[j] = 2.0 * np.dot(w[idx], a[a_idx])
    else:
        # FFT-based
        # We want: grad[j] = 2 * sum_k w[k] * a[k-j] for 0<=k-j<n
        # = 2 * (w * a)[j] where the convolution is cross-correlation
        # cross[j] = sum_k w[k] * a[k-j]
        # In terms of standard convolution: (w * a_rev)[2n-2-j]... 
        # Let me use: cross = correlate(w, a) = convolve(w, a_rev)
        a_rev = a[::-1]
        L2 = _next_pow2(2 * n - 1)
        Wf = np.fft.rfft(w, L2)
        Af = np.fft.rfft(a_rev, L2)
        conv = np.fft.irfft(Wf * Af, L2)[:2 * n - 1].real
        # conv[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)] = sum_k w[k] * a[n-1-m+k]
        # We want sum_k w[k] * a[k-j], so n-1-m+k = k-j => n-1-m = -j => m = n-1+j
        cross = conv[n - 1:2 * n - 1]
        grad = 2.0 * cross
    return grad


def _adam_optimize(p, budget_s, seed=0, lr_scale=1.0):
    """
    Adam optimizer with softmax parameterization for the max-convolution objective.
    p is normalized to sum=1.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n < 2:
        return p, _fft_score(p)
    p = _normalize(p)
    budget_s = float(budget_s)
    if budget_s < 2.0:
        return p, _fft_score(p)

    x = np.log(p + 1e-30)
    x -= x.mean()

    # Adaptive learning rate
    if n < 64:
        lr = 0.04
    elif n < 256:
        lr = 0.025
    elif n < 1024:
        lr = 0.018
    else:
        lr = 0.012
    lr *= lr_scale

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    L = _next_pow2(2 * n - 1)
    best_p = p.copy()
    best_f = _fft_score(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        # Alpha schedule: start high (sharp), decrease (smoother) then increase again at end
        if frac < 0.3:
            alpha = 40.0 * (1.0 - frac / 0.3) + 10.0
        elif frac < 0.7:
            alpha = 10.0 + 20.0 * ((frac - 0.3) / 0.4)
        else:
            alpha = 30.0 + 40.0 * ((frac - 0.7) / 0.3)

        # Compute convolution
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = c.max()
        if maxc < 1e-30:
            break

        f_now = 2.0 * n * maxc  # sum(p)=1
        if f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Weighted gradient via FFT
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = expz.sum()
        if sz < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w = expz / sz

        # Gradient computation via FFT
        Wr = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Wr * Pg, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = np.dot(p, grad_p)
        grad_x = p * (grad_p - gp)

        # Adam update
        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        # Adaptive step: decay learning rate over time
        lr_t = lr * (1.0 - 0.3 * frac)
        step = lr_t * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -70.0, 70.0)
        x -= x.max()
        e = np.exp(x)
        se = e.sum()
        if se < 1e-30:
            break
        p = e / se

    final_f = _fft_score(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def _subgradient_optimize(p, budget_s, step_mode='polyak'):
    """
    Subgradient method on the simplex with projection.
    Minimizes max_k (p*p)[k] subject to p >= 0, sum(p) = 1.
    """
    a = _normalize(p)
    n = a.size
    budget_s = float(budget_s)
    if budget_s < 2.0:
        return a, _fft_score(a)

    best_a = a.copy()
    best_f = _fft_score(a)
    best_val = 2.0 * n * best_f  # since sum=1, score = 2n*max(c)

    # For Polyak step: need a lower bound. Use 0 as lower bound (trivial).
    # Actually use the value at uniform: 2n/n = 2, so max(c) >= 1/n
    lb = 1.0 / n  # lower bound on max(c) when sum=1 (trivial: at least one c[k] >= sum(c)/(2n-1) >= 1/(2n-1))
    # Better: the minimum possible max(c) is at least 1/n by pigeonhole
    # (sum of all c[k] = sum(a)^2 = 1, and there are 2n-1 values)
    # So max(c) >= 1/(2n-1)
    lb = 1.0 / (2.0 * n - 1)

    gap = best_val - lb
    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)
    t = 0
    base_step = 2.0 / np.sqrt(2.0 * n)  # initial step size

    while time.monotonic() < end:
        t += 1
        g = _subgradient(a, alpha=20.0)
        # Projected subgradient step
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        
        if step_mode == 'polyak':
            cur_val = 2.0 * n * _fft_score(a)
            gap = max(cur_val - lb, 1e-10)
            step = (cur_val - lb) / (np.dot(g, g) + 1e-15)
            step = min(step, 0.5)
        else:
            step = base_step / (1.0 + 0.1 * t)

        a_new = a - step * g
        # Project onto simplex: max(x - tau, 0) = y, sum(y) = 1
        a_new = _project_simplex(a_new)

        f_new = _fft_score(a_new)
        if f_new < best_f:
            best_f = f_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_f


def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _frank_wolfe_optimize(p, budget_s):
    """
    Frank-Wolfe (conditional gradient) method for min max_k (p*p)[k] on simplex.
    Uses momentum and adaptive step size.
    """
    a = _normalize(p)
    n = a.size
    budget_s = float(budget_s)
    if budget_s < 3.0:
        return a, _fft_score(a)

    best_a = a.copy()
    best_f = _fft_score(a)
    s = np.zeros(n)  # momentum
    beta = 0.7  # momentum coefficient

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)
    t = 0
    gap = 1.0  # initial gap estimate (max possible is ~1)

    while time.monotonic() < end:
        t += 1
        g = _subgradient(a, alpha=50.0)  # sharp gradient for FW
        s = beta * s + (1.0 - beta) * g

        # LMO: find vertex e_j minimizing <s, e_j> = s[j]
        j_star = np.argmin(s)
        v = np.zeros(n)
        v[j_star] = 1.0

        # Step size: use gamma_t = 2/(t+2) or gap-based
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        gamma = 2.0 / (t + 2.0)
        # Also try: gamma = gap / ||a - v||^2
        diff = a - v
        nn = np.dot(diff, diff)
        if nn > 1e-15:
            gamma2 = np.dot(g, diff) / nn
            gamma = max(gamma, min(gamma2, 1.0))

        a_new = (1.0 - gamma) * a + gamma * v
        f_new = _fft_score(a_new)
        if f_new < best_f:
            best_f = f_new
            best_a = a_new.copy()
            gap = best_f * 2.0 * n - 1.0 / (2.0 * n)
            gap = max(gap, 1e-6)
        a = a_new

    return best_a, best_f


def _peak_shave(p, budget_s, n_perturbations=5000):
    """
    Peak-aware local search: identify peak-contributing coordinates and adjust them.
    Uses simulated annealing acceptance criterion.
    """
    a = _normalize(p).copy()
    n = a.size
    best_a = a.copy()
    best_f = _fft_score(a)
    rng = np.random.default_rng(42)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)
    T = 0.005  # initial temperature
    T_min = 0.0001
    n_iter = 0

    while time.monotonic() < end and n_iter < n_perturbations:
        n_iter += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        T_cur = max(T * (1.0 - frac) + T_min, T_min)

        # Compute convolution and find peaks
        c = _conv_direct(a)
        c = np.maximum(c, 0.0)
        maxc = c.max()
        peak_idx = np.where(c > 0.95 * maxc)[0]

        # Find coordinates that contribute most to peaks
        # Contribution of a[j] to peak at k: a[j] * a[k-j]
        contrib = np.zeros(n)
        for k in peak_idx[:10]:  # top peaks
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    contrib[j] += a[j] * a[kj]

        # Pick a "harmful" coordinate (high contribution) and a "safe" one (low contribution)
        sorted_contrib = np.argsort(contrib)
        harmful_pool = sorted_contrib[::-1][:n // 4]
        safe_pool = sorted_contrib[:n // 4]

        if len(harmful_pool) == 0 or len(safe_pool) == 0:
            break

        j_from = harmful_pool[rng.integers(len(harmful_pool))]
        j_to = safe_pool[rng.integers(len(safe_pool))]

        # Transfer mass
        amt = a[j_from] * rng.uniform(0.01, 0.15)
        if amt > a[j_from] * 0.5:
            amt = a[j_from] * 0.5
        a_new = a.copy()
        a_new[j_from] -= amt
        a_new[j_to] += amt

        f_new = _fft_score(a_new)
        delta = f_new - best_f

        if delta < 0 or np.exp(-delta / T_cur) > rng.random():
            a = a_new
            if f_new < best_f:
                best_f = f_new
                best_a = a_new.copy()
        # else: reject, keep a as is

    return best_a, best_f


def _lp_improve(p, budget_s):
    """
    LP cutting-plane: find direction g that maximizes sum(g) subject to f*g <= M, g >= 0.
    Then blend f and g.
    """
    if _scipy_opt is None:
        return p, _fft_score(p)

    a = _normalize(p)
    n = a.size
    if n < 8 or budget_s < 30.0:
        return a, _fft_score(a)

    S0 = np.sqrt(2.0 * n)
    f = a * S0
    cur_score = _fft_score(f)
    M = cur_score  # max of f*f (since f*f peak / sum(f)^2 * sum(f)^2 = ... )
    # Actually M = max(f*f) directly
    c_ff = _conv_direct(f)
    M = c_ff.max()

    best_f = f.copy()
    best_score = cur_score

    max_rows = min(800, max(128, 4000000 // n))
    rows = list(np.linspace(0, 2 * n - 2, min(max_rows // 2, 2 * n - 1), dtype=int).astype(int))
    row_set = set(rows)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 5.0)
    it = 0

    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(sorted(rows), dtype=int)
        R = rows_arr.size
        A = np.zeros((R, n))
        for ii in range(R):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(R, M)
        cobj = -np.ones(n)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(5.0, min(60.0, rem * 0.3))
        res = None
        try:
            res = _scipy_opt.linprog(cobj, A_ub=A, b_ub=b, bounds=(0, None),
                                     method='highs', options={'time_limit': tl})
        except Exception:
            break

        if res is None or not res.success or res.x is None:
            break

        g = np.clip(res.x, 0, None)
        sg = g.sum()
        if sg < 1e-12:
            break

        # Check violations
        fg = _conv_direct(f) * 0 + _conv_direct(f[:0].size and g)  # placeholder
        fg = np.convolve(f, g) if n <= 512 else None
        if fg is None:
            L2 = _next_pow2(2 * n - 1)
            Pf = np.fft.rfft(f, L2)
            Pg = np.fft.rfft(g, L2)
            fg = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        fg = np.maximum(fg, 0.0)
        violations = fg - M
        tol = 1e-6 * M

        if np.any(violations > tol):
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    if added >= 128 or added >= 0.3 * (max_rows - len(rows)):
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            if added == 0:
                break
            continue

        # Blend
        g = g * (S0 / sg)
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.5, 0.25, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0, None)
            sn = fn.sum()
            if sn < 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = _fft_score(fn)
            if sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()

        if accepted is None or accepted_score >= best_score:
            break

        f = accepted
        best_score = accepted_score
        best_f = f.copy()
        c_ff = _conv_direct(f)
        M = c_ff.max()

        if it % 5 == 0:
            print("  LP iter %d: score=%.6f rows=%d" % (it, best_score, len(rows)))

    return _normalize(best_f), _fft_score(best_f)


def _greedy_norm(n):
    """Coefficients of (1-x)^{-1/2} normalized."""
    n = max(int(n), 8)
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    return _normalize(a)


def _power_norm(n, s):
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return _normalize(a)


def _random_start(n, rng, kind='uniform'):
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'dirichlet_sparse':
        a = rng.dirichlet(0.3 * np.ones(n))
    elif kind == 'dirichlet_conc':
        a = rng.dirichlet(3.0 * np.ones(n))
    elif kind == 'exponential':
        a = np.exp(-np.arange(n, dtype=np.float64) * rng.uniform(0.01, 0.1))
    else:
        a = rng.random(n)
    return _normalize(a)


def _scale_output(p, target_sum=10.0):
    """Scale sequence to have the given sum, clip to [0, 1000]."""
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(32)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(a.size)
        s = a.size
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]

        if not np.isfinite(score) or score >= best_score:
            n = a.size
            old = best_by_n.get(n)
            if old is None or score < old[0]:
                best_by_n[n] = (score, a.copy())
            return
        best_p = a.copy()
        best_score = float(score)
        n = a.size
        best_by_n[n] = (float(score), a.copy())

    # Phase 1: Quick baselines (30s)
    print("Phase 1: Baselines")
    lengths = [64, 96, 128, 160, 192, 224, 256, 320, 384, 448, 512, 640, 768, 1024]
    for n in lengths:
        if time.monotonic() > t0 + 30:
            break
        for kind, fn in [('greedy', lambda: _greedy_norm(n)),
                         ('pow45', lambda: _power_norm(n, 0.45)),
                         ('pow55', lambda: _power_norm(n, 0.55)),
                         ('pow65', lambda: _power_norm(n, 0.65))]:
            record(fn())

    # Try height_sequence_1
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0, 1000.0)
            if arr.size > 0 and arr.sum() > 0.01:
                record(arr)
                print("  Loaded height_sequence_1: n=%d, score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    # Phase 2: Intensive optimization from best starts (1200s)
    print("Phase 2: Intensive optimization")
    opt_budget = min(1200.0, deadline - time.monotonic())

    # Sort candidates by score
    candidates = []
    for n, (sc, p) in best_by_n.items():
        candidates.append((sc, n, p))
    candidates.sort(key=lambda x: x[0])

    # Top candidates to optimize
    top_n = list(dict.fromkeys([c[1] for c in candidates[:8]]))
    # Add some diversity
    for n in [256, 384, 512]:
        if n not in top_n and n <= 1024:
            top_n.append(n)
    top_n = top_n[:6]

    for i, n in enumerate(top_n):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        rec = best_by_n.get(n)
        base = rec[1] if rec else _greedy_norm(n)
        alloc = min(rem * 0.15, 180.0)

        # Adam from base
        p_out, f_out = _adam_optimize(base, alloc, seed=seed + i * 100)
        record(p_out, f_out)
        print("  Adam n=%d: score=%.6f (time=%.0fs)" % (n, best_score, time.monotonic() - t0))

        rem = deadline - time.monotonic()
        if rem < 30:
            break

        # Subgradient from best
        if rem > 60:
            alloc2 = min(rem * 0.1, 120.0)
            p_out2, f_out2 = _subgradient_optimize(best_p, alloc2, step_mode='polyak')
            record(p_out2, f_out2)

    # Phase 3: Peak shaving + simulated annealing (600s)
    print("Phase 3: Peak shaving and annealing")
    rem = deadline - time.monotonic()
    if rem > 100 and best_p is not None:
        alloc = min(rem * 0.4, 500.0)
        p_out, f_out = _peak_shave(best_p, alloc, n_perturbations=20000)
        record(p_out, f_out)
        print("  Peak shave: score=%.6f" % best_score)

    # Phase 4: LP cutting-plane (300s)
    print("Phase 4: LP cutting-plane")
    if _scipy_opt is not None and best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            alloc = min(rem * 0.5, 300.0)
            p_out, f_out = _lp_improve(best_p, alloc)
            record(p_out, f_out)
            print("  LP: score=%.6f" % best_score)

    # Phase 5: More Adam rounds with perturbation (remaining time)
    print("Phase 5: Perturbation + re-optimization")
    rem = deadline - time.monotonic()
    round_num = 0
    while rem > 60 and round_num < 5:
        round_num += 1
        # Perturb best
        sigma = 0.05 + 0.05 * round_num
        noise = np.exp(rng.normal(0, sigma, best_p.size))
        perturbed = _normalize(best_p * noise)
        alloc = min(rem * 0.25, 200.0)
        p_out, f_out = _adam_optimize(perturbed, alloc, seed=seed + 1000 + round_num * 7,
                                      lr_scale=0.8)
        record(p_out, f_out)
        print("  Round %d (sigma=%.2f): score=%.6f" % (round_num, sigma, best_score))
        rem = deadline - time.monotonic()

        # Also try from a different initialization
        if rem > 90:
            n_try = best_p.size
            p_alt = _random_start(n_try, rng, kind='dirichlet_sparse')
            # Blend with best
            p_blend = _normalize(0.7 * best_p + 0.3 * p_alt)
            alloc2 = min(rem * 0.15, 120.0)
            p_out2, f_out2 = _adam_optimize(p_blend, alloc2, seed=seed + 5000 + round_num * 13)
            record(p_out2, f_out2)

    # Final: try a few different lengths from scratch
    rem = deadline - time.monotonic()
    if rem > 45 and best_p is not None:
        best_n = best_p.size
        for n_try in [best_n - 16, best_n + 16, best_n + 32]:
            if rem < 20:
                break
            if n_try < 16:
                continue
            # Initialize from best (pad or truncate)
            if n_try > best_n:
                p_try = np.zeros(n_try)
                p_try[:best_n] = best_p
            elif n_try < best_n:
                p_try = best_p[:n_try]
            else:
                p_try = best_p.copy()
            p_try = _normalize(p_try)
            alloc = min(rem * 0.3, 150.0)
            p_out, f_out = _adam_optimize(p_try, alloc, seed=seed + n_try)
            record(p_out, f_out)
            rem = deadline - time.monotonic()

    # Phase 6: Final polishing
    print("Phase 6: Final polishing")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            alloc = min(rem * 0.7, 200.0)
            p_out, f_out = _adam_optimize(best_p, alloc, seed=seed + 99999, lr_scale=0.5)
            record(p_out, f_out)

    # Validate against global evaluator
    print("Validation")
    if best_p is not None:
        seq = _scale_output(best_p, 10.0)
        gscore = _try_eval(seq)
        print("  Global eval: %.6f (local: %.6f)" % (gscore, best_score))
        if np.isfinite(gscore) and gscore < best_score:
            best_score = gscore

        # Try a few scalings
        for ts in [5.0, 10.0, 20.0, 50.0]:
            seq2 = _scale_output(best_p, ts)
            gscore2 = _try_eval(seq2)
            if np.isfinite(gscore2) and gscore2 < best_score:
                best_score = gscore2
                print("  Better at target_sum=%.0f: %.6f" % (ts, best_score))

    # Also validate top candidates from best_by_n
    items = sorted(best_by_n.items(), key=lambda x: x[1][0])
    for n, (sc, p) in items[:3]:
        if time.monotonic() > deadline - 10:
            break
        if n == best_p.size:
            continue
        seq = _scale_output(p, 10.0)
        gscore = _try_eval(seq)
        if np.isfinite(gscore) and gscore < best_score:
            best_score = gscore
            best_p = p.copy()
            print("  New best from n=%d: %.6f" % (n, best_score))

    if best_p is None:
        best_p = _greedy_norm(256)

    final_seq = _scale_output(best_p, 10.0)
    print("Final: n=%d, local_score=%.6f" % (best_p.size, best_score))
    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation function."""
    try:
        return _main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main search: %s" % str(e))
        import traceback
        traceback.print_exc()
        # Fallback
        p = _greedy_norm(384)
        return _scale_output(p, 10.0)
