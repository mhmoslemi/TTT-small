def _conv_full(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    m = 1
    while m < 2*n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

of non-negative sequences. Uses the approach: at each step, solve an LP to find
the best direction that keeps the cross-convolution bounded, then take a step.

import time

from scipy.optimize import linprog


def _conv_full(a):
    """Full linear auto-convolution a*a via FFT."""
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _build_A(a, n):
    """Build the convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    m = 2 * n - 1
    A = np.zeros((m, n))
    # For efficiency, use the banded structure
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            # A[k, j] = a[k-j] for j in [lo, hi]
            js = np.arange(lo, hi + 1)
            A[k, js] = a[k - js]
    return A


def _lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    m = 2 * n - 1
    A = _build_A(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'disp': False, 'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None
