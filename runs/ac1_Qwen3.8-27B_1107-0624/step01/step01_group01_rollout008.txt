```python
def lp_improve(f, M, solver='highs'):
    """
    Given current sequence f and bound M on convolution,
    find g >= 0 maximizing sum(g) subject to (f*g)[k] <= M for all k.
    """
    from scipy.optimize import linprog
    n = len(f)
    
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i] * g[k-i] where 0 <= i < n and 0 <= k-i < n
    # i.e., max(0, k-n+1) <= i <= min(k, n-1)
    
    # Build constraint matrix A such that A @ g <= M * ones(2n-1)
    # A[k, i] = f[i] if max(0, k-n+1) <= i <= min(k, n-1), else 0
    
    # This is a banded matrix! We can use sparse representation.
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the linear programming "chasing" method: at each step, solve an LP to find
the best g that maximizes sum(g) subject to (f*g) <= M elementwise, then blend.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def full_convolve_fft(a):
    """Full linear convolution of a real 1D array with itself using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    k = 1
    while k < 2 * n - 1:
        k <<= 1
    fa = np.fft.rfft(a, k)
    c = np.fft.irfft(fa * fa, k)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0 and mn > -1e-9:
        c = c - mn
    return c


def internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 efficiently."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def normalize_unit(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def make_linear_profile(n, eta):
    """Monotone linear profile normalized to sum 1."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_exp_profile(n, rate):
    """Exponential profile: a[i] ~ exp(-rate * i / n), normalized to sum 1."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    """Anti-symmetric cubic profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a1, a2):
    """Quadratic profile: 1 + a1*u + a2*u^2."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_power_profile(n, p):
    """Power-law profile: a[i] ~ (i+1)^p, normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    return a / a.sum()


def build_conv_constraint_matrix(f):
    """Build sparse matrix A where A @ g = f * g (full convolution)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_chase_step(f, M, max_iter_inner=50):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for all k.
    Returns optimal g or None if infeasible.
    """
    n = len(f)
    c_obj = -np.ones(n)
    A = build_conv_constraint_matrix(f)
    b = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 10, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_iteration(f0, max_iters=200, t_initial=0.5, t_min=0.05, tol=1e-9,
                 deadline=None, verbose=False):
    """
    Run the LP-based chasing iteration to minimize max(f*f) subject to sum(f)=1, f>=0.
    Returns (best_f, best_score).
    """
    f = normalize_unit(f0)
    n = len(f)
    c = full_convolve_fft(f)
    M = float(c.max())
    score = 2.0 * n * M
    best_f = f.copy()
    best_score = score
    t = t_initial
    no_improve = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        g = lp_chase_step(f, M)
        if g is None:
            # Try with slightly larger M
            M_try = M * 1.001
            g = lp_chase_step(f, M_try)
            if g is None:
                break
        g = normalize_unit(g)
        # Blend
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = normalize_unit(f_new)
        c_new = full_convolve_fft(f_new)
        M_new = float(c_new.max())
        score_new = 2.0 * n * M_new

        if np.isfinite(score_new) and score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        M = M_new

        # Adapt step size
        if no_improve > 20:
            t = max(t_min, t * 0.9)
            no_improve = 0
        if it % 50 == 0:
            t = max(t_min, t * 0.95)

        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: score={score_new:.10f}, M={M:.8f}, t={t:.4f}")

        # Convergence check
        if M - M_new < tol * M:
            break

    return best_f, best_score


def logit_adam_polish(p0, deadline, lr=0.05, reg=1e-8, max_extra_iters=500):
    """
    Polish a probability vector using Adam in logit space with soft-max smoothing.
    Returns (best_p, best_score).
    """
    n = p0.size
    p = np.maximum(p0, 1e-15)
    p = p / p.sum()
    z = np.log(p)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0

    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.1 * maxc)
    best_p = p.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0

    while time.monotonic() < deadline and adam_t < max_extra_iters:
        # Soft max weights
        x = (c - maxc) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s = 1.0
        w = ex / s

        # Gradient: grad_p[i] = 2 * corr(w, p)[i]
        # corr(w, p)[i] = sum_j w[i+j] * p[j] (cross-correlation)
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        grad_p = 2.0 * h[n - 1:2 * n - 1]

        # Project to tangent space of simplex
        grad_p += reg * p
        grad_p -= float(np.dot(p, grad_p)) * p  # Not quite right for logit

        # Actually for logit parameterization: grad_z = p * (grad_p - <p, grad_p>)
        dot = float(np.dot(p, grad_p))
        grad_z = p * (grad_p - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))

        p = np.exp(z)
        p = p / p.sum()
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                lr = max(0.01, lr * 0.95)

        if adam_t % 50 == 0:
            beta = max(1e-12, beta * 0.8)

        if no_imp > 500:
            break

    return best_p, best_score


def make_random_smooth(n, rng, smoothness=3.0):
    """Generate a random smooth profile using Gaussian process-like method."""
    # Random low-frequency Fourier coefficients
    coeffs = []
    for k in range(1, int(smoothness) + 1):
        amp = rng.uniform(0.5, 2.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        coeffs.append((amp, phase, k))
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for amp, phase, k in coeffs:
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Linearly interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_unit(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-15)
    return p1 / p1.sum()

            break
        per_deadline = min(phase2_deadline, time.monotonic() + 40.0)
        f_best, sc_best = lp_iteration(f, max_iters=200,
                                       t_initial=0.3, t_min=0.05,
                                       deadline=per_deadline)
        update(f_best, sc_best)
        if idx % 3 == 0:
            print(f"  Phase 2 refine {idx}: {sc:.8f} -> {sc_best:.8f} | best={best_score:.8f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Try different lengths (256, 1024)
    for n_try in [256, 1024]:
        if time.monotonic() > deadline - 300:
            break
        # Interpolate best solution to new length
        p_interp = interpolate_profile(best_seq, n_try) if best_seq is not None else make_linear_profile(n_try, 1.5)
        sc_interp = internal_score(p_interp)
        print(f"  n={n_try}: interpolated score={sc_interp:.8f}")

        # Also try fresh starts
        starts_try = [
            make_linear_profile(n_try, 1.5),
            make_exp_profile(n_try, 2.0),
            make_cubic_profile(n_try, 1.5, -1.0),
            p_interp,
            make_random_smooth(n_try, rng, 4.0),
        ]
        scored_try = sorted([(internal_score(s), s) for s in starts_try])

        for sc_init, s_init in scored_try[:4]:
            if time.monotonic() > deadline - 300:
                break
            per_deadline = min(deadline - 300, time.monotonic() + 35.0)
            f_best, sc_best = lp_iteration(s_init, max_iters=100,
                                           t_initial=0.5, t_min=0.1,
                                           deadline=per_deadline)
            update(f_best, sc_best)
            print(f"  n={n_try}: init={sc_init:.6f} -> {sc_best:.8f} | best={best_score:.8f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Gradient polishing of the best solution
    if best_seq is not None:
        polish_deadline = min(deadline, time.monotonic() + 300.0)
        p_polished, sc_polished = logit_adam_polish(best_seq, polish_deadline, lr=0.03)
        update(p_polished, sc_polished)
        print(f"After Phase 4 (polish): best={best_score:.10f}")

        # Try polishing with different LR
        if time.monotonic() < deadline - 100:
            p_pol2, sc_pol2 = logit_adam_polish(best_seq, min(deadline, time.monotonic() + 150.0), lr=0.08)
            update(p_pol2, sc_pol2)
            print(f"After Phase 4b (polish2): best={best_score:.10f}")

    # Phase 5: Local perturbation search
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_b = best_seq.size
        p_cur = best_seq.copy()
        sc_cur = best_score
        # Random coordinate perturbations
        for _ in range(200):
            if time.monotonic() > deadline - 30:
                break
            # Perturb a random subset of coordinates
            n_pert = max(3, n_b // 32)
            idx_pert = rng.choice(n_b, size=n_pert, replace=False)
            delta = rng.normal(0, 0.02, n_pert)
            p_trial = p_cur.copy()
            p_trial[idx_pert] += delta
            p_trial = np.maximum(p_trial, 0.0)
            s_sum = p_trial.sum()
            if s_sum > 1e-10:
                p_trial = p_trial / s_sum
                sc_trial = internal_score(p_trial)
                if sc_trial < sc_cur:
                    p_cur = p_trial
                    sc_cur = sc_trial
                    update(p_cur, sc_cur)

        print(f"After Phase 5 (perturb): best={best_score:.10f}")

    # Final: validate and return
    if best_seq is None:
        best_seq = make_linear_profile(512, 1.5)
        best_score = internal_score(best_seq)

    # Final exact scoring
    final_seq = best_seq.copy()
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final > 0:
        final_seq = final_seq / s_final

    sc_final = internal_score(final_seq)
    print(f"FINAL: score={sc_final:.10f}, n={final_seq.size}")

    return [float(x) for x in final_seq]
```
