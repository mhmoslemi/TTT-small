def iterate_lp(f, M, solver='highs'):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(j, j+n):
            A[k, j] = f[j]
    # Or more efficiently:
    # A[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1)
    
    # Objective: minimize -sum(g) (to maximize sum(g))
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    return result
```

def solve_lp_direction(f, n):
    """Solve: min M' s.t. conv(f,g)[k] <= M', sum(g)=1, g>=0"""
    # Variables: [g0, g1, ..., g_{n-1}, M']
    # c = [0, 0, ..., 0, 1] (minimize M')
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M'
    
    # Inequality constraints: conv(f,g)[k] - M' <= 0
    # A_ub has shape (2n-1) x (n+1)
    A_ub = np.zeros((2*n-1, n+1))
    for j in range(n):
        A_ub[j:j+n, j] = f[j]
    A_ub[:, -1] = -1.0  # -M'
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, M' free (or M' >= 0)
    bounds = [(0, None)] * n + [(0, None)]

        return None, None
    
    g = result.x[:n]
    M_prime = result.x[-1]
    return g, M_prime

import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = full_convolve_fft(a)
    m = c.max()
    return 2.0 * n * m / (s * s)

def solve_lp(f):
    """Solve: min M' s.t. conv(f,g)[k] <= M', sum(g)=1, g>=0"""
    n = len(f)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):

    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True})

        return None
    
    g = result.x[:n]
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s < 1e-10:

    return g / s

def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_parabolic_profile(n, a2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + a2 * (t - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_bump_profile(n, center, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-(t - center) ** 2 / (2 * width ** 2))

def make_power_profile(n, p):
    """a[i] = (1 - i/n)^p"""
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p
    a = np.maximum(a, 1e-12)

    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s < 0.01:
            return None
        return p0 / s
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    if s < 0.01:
        return None
    return p1 / s

```python
"""
LP-based iterative search for non-negative sequences minimizing
score = 2*n*max(convolve(a,a))/sum(a)^2.

Algorithm:
1. Start with a diverse set of initial profiles.
2. At each iteration, solve an LP to find the direction g that
   minimizes max(conv(f,g)) subject to g>=0, sum(g)=1.
3. Take a step towards g, accepting only if the score improves.
4. Use adaptive step size and periodic restarts to escape local optima.
5. Scale up to longer sequences via interpolation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a):
    """Full linear convolution a * a, using FFT for large n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Numerical cleanup
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = full_convolve(a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where (A@g)[k] = conv(f,g)[k]."""
    n = f.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f[j]
    return A


def solve_lp_direction(f):
    """
    Solve: min M' s.t. conv(f,g)[k] <= M' for all k, sum(g)=1, g>=0, M'>=0.
    Returns (g, M_prime) or (None, None) if infeasible.
    """
    n = f.size
    # Decision variables: [g_0, g_1, ..., g_{n-1}, M']
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = 1.0  # minimize M'
    
    # A_ub @ x <= b_ub
    # conv(f,g)[k] - M' <= 0 for k = 0, ..., 2n-2
    A_ub = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    for j in range(n):
        A_ub[j:j + n, j] = f[j]
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)
    
    # A_eq @ x = b_eq
    # sum(g) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    
    # Bounds: g_i >= 0, M' >= 0
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10.0})
    except Exception:
        return None, None
    
    if result.status != 0:
        return None, None
    
    g = result.x[:n].copy()
    M_prime = float(result.x[-1])
    
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-15:
        return None, None
    g = g / s
    return g, M_prime


def lp_iterate(f, t):
    """One LP iteration: solve for g, take step of size t towards g."""
    g, M_prime = solve_lp_direction(f)
    if g is None:
        return None
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = float(f_new.sum())
    if s < 0.01:
        return None
    f_new = f_new / s
    return f_new
