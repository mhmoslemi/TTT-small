Search for a non-negative sequence minimizing 2*n*max_k(a*a)[k] / (sum a)^2.

Key improvements over previous approach:
1. Top-k soft-max objective (sharper than full softmax)
2. Outer active-set refinement loop
3. Adaptive k based on the width of the peak cluster
4. Multi-scale optimization (coarse-to-fine)
5. Subgradient polishing on the identified active set

import numpy as np
import time
from scipy.optimize import minimize

N = 4096

FFT_CROSS = 1 << (3 * N - 1).bit_length()  # 32768 for p_rev * w

_best_score = float('inf')
_best_p = None
_deadline = 0.0

def self_conv(p):
    f = np.fft.rfft(p, FFT_SELF)
    return np.fft.irfft(f * f, FFT_SELF).real[:2*N-1]

def cross_conv_p_rev_w(p, w):
    """Compute (p_reversed * w) using FFT, extract elements [N-1 : 2N-1]."""
    pr = p[::-1]
    F = np.fft.rfft(pr, FFT_CROSS)
    W = np.fft.rfft(w, FFT_CROSS)
    c = np.fft.irfft(F * W, FFT_CROSS).real
    return c[N-1:2*N-1]

def probs_from_z(z):
    m = max(0.0, float(np.max(z)))
    ez = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(ez))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = ez / Z
    return p

def score_p(p):
    c = self_conv(p)
    return 2.0 * N * float(np.max(c))

    global _best_score, _best_p
    if np.isfinite(score) and score < _best_score:

        _best_p = p.copy()

def objective_topk(z, topk_idx, alpha):
    """Top-k soft-max objective with gradient."""
    p = probs_from_z(z)
    c = self_conv(p)
    
    hard = 2.0 * N * float(np.max(c))
    update_best(p, hard)
    
    # Extract top-k values
    topk_vals = c[topk_idx]
    max_val = float(np.max(topk_vals))
    u = alpha * (topk_vals - max_val)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        J = max_val
        w = np.zeros(2*N-1)
        w[topk_idx[np.argmax(topk_vals)]] = 1.0
    else:
        J = max_val + np.log(s) / alpha
        w = np.zeros(2*N-1)
        w[topk_idx] = e / s
    
    # Gradient w.r.t. p
    gp = 2.0 * cross_conv_p_rev_w(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return float(J), gz

        # Take top min_k
        active = np.argsort(c)[-min_k:]
    elif len(active) > max_k:
        active = np.argsort(c)[-max_k:]
    return active

        cmax = float(c.max())
        score = 2.0 * N * cmax
        update_best(p, score)

        topk_idx = find_active_set(c, rel_tol=0.02, min_k=3, max_k=25)
        k = len(topk_idx)
        topk_vals = c[topk_idx]
        spread = float(topk_vals.max() - topk_vals.min()) + 1e-12

        alpha = 20.0 / spread
        alpha = max(5e4, min(alpha, 1e7))

        if k <= 20:
            func = obj_topk
            args = (topk_idx, alpha)
        else:
            func = obj_full
            args = (min(alpha, 5e6),)

        rem = deadline - time.monotonic() - 2.0
        if rem < 3.0:
            break
        maxfun = int(min(500, max(80, rem * 80)))

        try:
            res = minimize(func, z, args=args, method='L-BFGS-B', jac=True,
                           bounds=BOUNDS,
                           options={'maxfun': maxfun, 'maxiter': maxfun // 2,
                                    'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 20})
            z_new = np.clip(res.x, -40.0, 40.0)
            z_new = np.where(np.isfinite(z_new), z_new, 0.0)
            p_new = probs_from_z(z_new)
            s_new = score_p(p_new)
            if s_new < score - 1e-13:
                z = z_new
            else:
                break
        except SearchTimeout:
            break
        except Exception:
            break

    p = probs_from_z(z)
    update_best(p, score_p(p))


def polish_subgradient(p0, deadline, rng):
    global _deadline
    _deadline = deadline
    z = p_to_z(p0)
    p = p0.copy()
    score = score_p(p)
    update_best(p, score)

    noise = 0.04
    fails = 0

    while time.monotonic() < deadline - 1.0 and fails < 150:
        c = self_conv(p)
        cmax = float(c.max())
        score = 2.0 * N * cmax
        update_best(p, score)

        k = min(8, 2 * N - 1)
        topk_idx = np.argsort(c)[-k:]
        topk_vals = c[topk_idx]
        topk_max = float(topk_vals.max())

        w = np.zeros(2 * N - 1)
        weights = np.exp(-5e4 * (topk_max - topk_vals) / max(cmax, 1e-15))
        ws = float(weights.sum())
        if ws < 1e-300:
            w[topk_idx[np.argmax(topk_vals)]] = 1.0
        else:
            w[topk_idx] = weights / ws

        gp = grad_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        g = gp - sdot
        gn = float(np.linalg.norm(g))

        if gn < 1e-14:
            break

        accepted = False
        for step in (0.4, 0.2, 0.1, 0.05, 0.02, 0.01):
            dz = -step * g[1:] / gn / np.maximum(p[1:], 1e-10)
            zt = np.clip(z + dz, -40.0, 40.0)
            pt = probs_from_z(zt)
            st = score_p(pt)
            if st < score - 1e-13:
                z, p, score = zt, pt, st
                update_best(p, score)
                accepted = True
                break

        if not accepted:
            best_s = score
            best_z = z
            for _ in range(4):
                zt = np.clip(z + rng.normal(0, noise, N - 1), -40.0, 40.0)
                pt = probs_from_z(zt)
                st = score_p(pt)
                if st < best_s:
                    best_s = st
                    best_z = zt
            if best_s < score - 1e-13:
                z = best_z
                p = probs_from_z(z)
                score = best_s
                update_best(p, score)
                accepted = True

        if accepted:
            fails = 0
            noise = min(0.12, noise * 1.15)
        else:
            fails += 1
            noise = max(0.005, noise * 0.85)


def make_candidates(rng):
    pts = []

    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size > 0:
            b = np.clip(base, 0, None)
            if b.size != N:
                b = np.interp(np.linspace(0, 1, N), np.linspace(0, 1, b.size), b)
            bsum = float(b.sum())
            if bsum > 0:
                b = b / bsum
                pts.append(p_to_z(b))
    except Exception:
        pass

    pts.append(np.zeros(N - 1))

    x = X_GRID

    for beta in [0.5, 1.0, 1.5, 2, 3, 5, 8, 12]:
        f = 2.0 / (1.0 + np.exp(-beta * (x - 0.5)))
        f = np.maximum(f, 1e-10)
        f /= f.sum()
        pts.append(p_to_z(f))
        pts.append(p_to_z(f[::-1]))

    for a in [0.5, 1.0, 1.5, 2, 3, 5]:
        f = np.power(x + 1e-6, a)
        f = np.maximum(f, 1e-10)
        f /= f.sum()
        pts.append(p_to_z(f))
        pts.append(p_to_z(f[::-1]))

    for b in [1, 2, 3, 5, 8]:
        f = np.exp(b * (x - 0.5))
        f = np.maximum(f, 1e-10)
        f /= f.sum()
        pts.append(p_to_z(f))
        pts.append(p_to_z(f[::-1]))

    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.08, 0.15, 0.25, 0.4]:
            f = np.exp(-((x - center) / width) ** 2)
            f = np.maximum(f, 1e-10)
            f /= f.sum()
            pts.append(p_to_z(f))

    for _ in range(25):
        z = np.zeros(N - 1)
        for freq in [1, 2, 3, 4, 6, 8, 12]:
            amp = rng.normal(0, 1.2 / (freq + 1))
            ph = rng.uniform(0, 2 * np.pi)
            z += amp * np.sin(2 * np.pi * freq * x[1:] + ph)
        pts.append(np.clip(z, -40, 40))

    for _ in range(20):
        f = 1.0 + 0.4 * rng.standard_normal(N)
        kernel = np.exp(-np.arange(-30, 31) ** 2 / 100.0)
        kernel /= kernel.sum()
        f = np.convolve(f, kernel, 'same')
        f = np.maximum(f, 1e-6)
        f /= f.sum()
        pts.append(p_to_z(f))

    for _ in range(10):
        f = 1.0 + 0.15 * rng.standard_normal(N)
        f = np.maximum(f, 1e-6)
        f /= f.sum()
        pts.append(p_to_z(f))

    return pts


def sort_candidates(scored):
    arr = np.array([s for s, z in scored])
    order = np.argsort(arr)
    return [scored[int(i)] for i in order]


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_p, _deadline

    t0 = time.monotonic()
    total_budget = max(10.0, float(budget_s))
    deadline = t0 + total_budget - 3.0
    _best_score = float('inf')
    _best_p = None
    _deadline = deadline

    rng = np.random.default_rng(seed)

    candidates = make_candidates(rng)
    scored = []
    for z in candidates:
        p = probs_from_z(z)
        s = score_p(p)
        update_best(p, s)
        scored.append((s, z))

    scored = sort_candidates(scored)
    print(f"Init: {len(scored)} candidates, best={_best_score:.6f}")

    polish_time = max(40.0, 0.10 * (deadline - t0))
    main_deadline = deadline - polish_time

    n_opt = min(len(scored), 12)
    for i in range(n_opt):
        if time.monotonic() >= main_deadline - 5.0:
            break
        rem = main_deadline - time.monotonic()
        if rem < 8.0:
            break
        if scored[i][0] > _best_score * 1.08 and i >= 4:
            continue
        max_time = min(rem, 250.0 / (1.0 + i * 0.3))
        cand_dl = min(time.monotonic() + max_time, main_deadline)
        if cand_dl - time.monotonic() < 5.0:
            break
        optimize_from(scored[i][1], cand_dl, max_outer=5)
        if (i + 1) % 3 == 0:
            print(f"P1 [{i + 1}/{n_opt}]: best={_best_score:.6f} t={time.monotonic() - t0:.0f}s")

    if time.monotonic() < main_deadline - 5.0:
        for attempt in range(40):
            if time.monotonic() >= main_deadline - 5.0:
                break
            if _best_p is None:
                break
            frac = (time.monotonic() - t0) / max(1.0, main_deadline - t0)
            scale = 0.03 + 0.35 * (1.0 - frac)
            if attempt > 25:
                scale *= 1.5
            z0 = p_to_z(_best_p) + rng.normal(0, scale, N - 1)
            z0 = np.clip(z0, -40, 40)
            rem = main_deadline - time.monotonic()
            if rem < 5.0:
                break
            cand_dl = min(time.monotonic() + min(50.0, rem // 3), main_deadline)
            if cand_dl - time.monotonic() < 3.0:
                break
            optimize_from(z0, cand_dl, max_outer=3)
            if attempt % 8 == 0:
                print(f"P2 [{attempt}]: best={_best_score:.6f} t={time.monotonic() - t0:.0f}s")

    if _best_p is not None and time.monotonic() < deadline - 3.0:
        polish_subgradient(_best_p, deadline - 2.0, rng)
        print(f"Polish: best={_best_score:.6f}")

    if _best_p is not None and time.monotonic() < deadline - 2.0:
        z0 = p_to_z(_best_p)
        c = self_conv(_best_p)
        topk = find_active_set(c, rel_tol=0.01, min_k=3, max_k=12)
        try:
            res = minimize(obj_topk, z0, args=(topk, 8e6),
                           method='L-BFGS-B', jac=True, bounds=BOUNDS,
                           options={'maxfun': 300, 'ftol': 1e-15, 'gtol': 1e-11})
            p_f = probs_from_z(np.clip(res.x, -40, 40))
            update_best(p_f, score_p(p_f))
        except Exception:
            pass

    if _best_p is not None and time.monotonic() < deadline - 1.0:
        z0 = p_to_z(_best_p)
        c = self_conv(_best_p)
        topk = find_active_set(c, rel_tol=0.005, min_k=3, max_k=8)
        try:
            res = minimize(obj_topk, z0, args=(topk, 2e7),
                           method='L-BFGS-B', jac=True, bounds=BOUNDS,
                           options={'maxfun': 200, 'ftol': 1e-15, 'gtol': 1e-11})
            p_f = probs_from_z(np.clip(res.x, -40, 40))
            update_best(p_f, score_p(p_f))
        except Exception:
            pass

    if _best_p is None:
        seq = [1.0 / float(N)] * N
    else:
        seq = [float(max(0.0, v)) for v in _best_p]

    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score) or final_score > 2.0:
        seq = [1.0 / float(N)] * N
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            final_score = 2.0

    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size > 0:
            b = np.clip(base, 0, 1000)
            bs = float(evaluate_sequence([float(v) for v in b]))
            if np.isfinite(bs) and bs < final_score:
                seq = [float(v) for v in b]
                final_score = bs
    except Exception:
        pass

    print(f"FINAL: score={final_score:.6f} n={len(seq)}")
    return seq
