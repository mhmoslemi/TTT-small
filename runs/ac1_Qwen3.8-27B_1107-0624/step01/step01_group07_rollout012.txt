Combined optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses iterated LP, subgradient descent, and simulated annealing with 
multiple initializations and adaptive hyperparameters.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M, margin=1.0):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * margin)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_combined(a, deadline, rng, verbose=True):
    """
    Combined LP + subgradient + annealing optimization.
    """
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)
    
    # Phase 1: LP-based optimization
    step = 0.1
    stall = 0
    it = 0
    
    while time.monotonic() < deadline and stall < 100:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        # Try LP with progressively tighter margins
        g = None
        for margin in [1.0, 0.95, 0.9, 0.85]:
            g = lp_solve(a, M, margin)
            if g is not None and g.sum() > 1e-15:
                break
        
        if g is None:
            # Fall back to subgradient
            g = subgradient_direction(a, M)
            if g is None:
                break
        
        g = g / g.sum()
        
        improved = False
        for shrink in [1.0, 0.5, 0.3, 0.15, 0.08, 0.04]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

            step = min(0.35, step * 1.02)

                noise = rng.normal(0, noise_scale, n)
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 0.0, None)

                        a = a_new

    # Phase 2: Subgradient polish
    if time.monotonic() < deadline:
        best_a, best_score = optimize_subgrad(best_a, deadline, rng)
    
    return best_a, best_score


def subgradient_direction(a, M):
    """Compute subgradient of max(conv(a,a)) and return a direction."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    peak_mask = c >= 0.95 * M  # subgradient of max is convex combo of active constraints
    
    # Subgradient w.r.t. a_i: sum over active k of a[k-i] (for valid k-i)
    subgrad = np.zeros(n)
    peak_indices = np.where(peak_mask)[0]
    weights = np.zeros(len(peak_indices))
    # Weight peaks by how close they are to max
    peak_vals = c[peak_indices]
    weights = (peak_vals - 0.9 * M) / (0.1 * M + 1e-15)
    weights = np.maximum(weights, 0.1)
    weights = weights / weights.sum()

        for i in range(n):
            j1 = k - i
            j2 = i - k  # not needed since conv is a*a, symmetric
            if 0 <= j1 < n:
                subgrad[i] += w * a[j1]

    direction = a * np.exp(-0.1 * subgrad / (subgrad.max() + 1e-15))
    direction = np.clip(direction, 0.0, None)
    if direction.sum() < 1e-10:

    return direction

            valid = (js >= 0) & (js < n)

So we want to minimize max(c) = max(IDFT(|A|^2)) subject to A(0) = 1, a = IDFT(A) >= 0.

One approach: 
1. Start with some A (corresponding to our initial a)
2. Compute c = IDFT(|A|^2)
3. Find where c is largest
4. Modify A to reduce those components
5. Ensure a = IDFT(A) >= 0 (project if needed)

The modification in step 4 could be:
- Reduce |A(k)| for k that contribute to the peaks
- This is like "spectral shaping"

But ensuring a >= 0 after modifying A is the tricky part (it's a linear constraint in A).

This is getting complex. Let me stick with the time-domain approach but with the peak-specific LP improvement.

Let me also think about the **optimal n**. The objective is 2n * max(conv) / sum^2. For a given "quality" of flatness (ratio of max to mean of conv), the objective scales as 2n * (max/mean) / (2n-1) ≈ 1 + 2*(max/mean - 1)/(2n-1) for large n. Wait, that's not right.

Let me redo: if conv is "ε-flat" meaning max(conv) = (1+ε) * mean(conv), then:
- mean(conv) = sum(conv)/(2n-1) = sum(a)^2/(2n-1)
- max(conv) = (1+ε) * sum(a)^2/(2n-1)
- Objective = 2n * (1+ε) * sum(a)^2/(2n-1) / sum(a)^2 = 2n(1+ε)/(2n-1) ≈ 1 + ε (for large n)

So the objective ≈ 1 + ε where ε is the "excess" of the peak over the mean. To get objective < 1.503, we need ε < 0.503.

For n=512: 2n/(2n-1) = 1024/1023 ≈ 1.001. So if perfectly flat, objective ≈ 1.001.
The excess is what matters: objective = 2n * max(conv) / sum^2 = 2n/(2n-1) * (max(conv)/mean(conv)) ≈ 1 * (1+ε).

So we need max(conv)/mean(conv) < 1.503 (approximately, for large n).

For the uniform sequence: max(conv)/mean(conv) = (1/n) / (1/(2n-1)) = (2n-1)/n ≈ 2. So objective ≈ 2.

We need to get the ratio from 2 down to about 1.5. That's a 25% reduction in the peak-to-mean ratio.

One more thought: **what about using a "bump function" approach?**

Instead of smooth windows (cosine, Gaussian), what about using multiple bumps? Like:
a = [bump1 + bump2 + ... + bump_m]

where each bump is a narrow Gaussian. The idea is that multiple bumps at different positions can create a more uniform auto-convolution (like how multiple delta functions create a more uniform sum distribution).

For example, if we place m equally-spaced bumps of width w, the auto-convolution will have m^2 peaks (one for each pair of bumps), but if w is small enough, these peaks will be well-separated and the "fill" between them can be relatively uniform.

Actually, this is exactly what happens with a "comb" function. The auto-convolution of a comb is a comb convolved with itself, which is a broader comb. If we have a comb with period T and width w, the auto-convolution has period T and width 2w.

For a flat auto-convolution, we'd want the comb to have the same period as the output... this doesn't quite work.

"""
Step function optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses peak-specific LP directions, subgradient descent, and simulated annealing
with multiple restarts and adaptive hyperparameters. Key improvement: LP that
asymmetrically tightens constraints at peak positions to force peak reduction.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_standard(a, M):
    """Standard LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A_mat = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            if res.x.sum() > 1e-15:
                return res.x
    except Exception:
        pass
    return None


def lp_peak_specific(a, c_full, epsilon, num_peaks=5):
    """LP that specifically targets peak positions with tighter constraints.
    b_ub[k] = c_full[k] * (1-epsilon) for top peaks, c_full[k] for others.
    This forces the direction to reduce the peaks.
    """
    n = len(a)
    A_mat = build_conv_matrix(a)
    c_obj = -np.ones(n)
    
    # Identify top peaks
    b_ub = np.copy(c_full)
    peak_indices = np.argsort(c_full)[-num_peaks:]
    for k in peak_indices:
        b_ub[k] = c_full[k] * (1.0 - epsilon)
    
    # Ensure b_ub is positive where needed
    b_ub = np.maximum(b_ub, 1e-10)
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            if res.x.sum() > 1e-15:
                return res.x
    except Exception:
        pass
    return None


def subgrad_direction(a, lr=0.05, threshold_frac=0.85):
    """Compute subgradient-based multiplicative weight direction."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    if M < 1e-15:
        return a.copy()
    
    threshold = threshold_frac * M
    w = np.zeros(2 * n - 1)
    peak_mask = c > threshold
    w[peak_mask] = (c[peak_mask] - threshold)
    
    w_sum = w.sum()
    if w_sum < 1e-15:
        return a.copy()
    w = w / w_sum
    
    # subgrad[i] = sum_j w[i+j] * a[j] for i in [0, n-1]
    W = np.lib.stride_tricks.sliding_window_view(w, n)
    subgrad = W @ a
    
    max_sg = subgrad.max() + 1e-15
    a_new = a * np.exp(-lr * subgrad / max_sg)
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() < 1e-10:
        return a.copy()
    return a_new / a_new.sum()


def optimize_lp_anneal(a, deadline, rng):
    """LP-based optimization with peak-specific constraints and annealing."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.15
    stall = 0
    it = 0
    use_peak_lp = True
    
    while time.monotonic() < deadline:
        it += 1
        c_full = np.convolve(a, a, mode='full')
        M = c_full.max()
        
        # Try peak-specific LP first (more aggressive at peaks)
        g = None
        if use_peak_lp:
            for eps in [0.05, 0.1, 0.15, 0.2]:
                g = lp_peak_specific(a, c_full, eps, num_peaks=5)
                if g is not None:
                    break
            if g is None:
                g = lp_standard(a, M)
        else:
            g = lp_standard(a, M)
        
        if g is None:
            g = subgrad_direction(a, lr=0.05)
        
        g = g / g.sum()
        
        improved = False
        for shrink in [1.0, 0.6, 0.4, 0.25, 0.12, 0.06]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break
        
        if improved:
            stall = 0
            step = min(0.35, step * 1.02)
            # Alternate between peak-specific and standard LP
            if it % 10 == 0:
                use_peak_lp = not use_peak_lp
        else:
            stall += 1
            step = max(0.003, step * 0.72)
            
            if stall > 6:
                noise_scale = min(0.35, 0.04 + 0.025 * stall)
                noise = rng.normal(0, noise_scale, n)
                a_pert = best_a * (1.0 + noise)
                a_pert = np.clip(a_pert, 0.0, None)
                if a_pert.sum() > 1e-10:
                    a_pert /= a_pert.sum()
                    sc_pert = compute_objective(a_pert)
                    if sc_pert < best_score:
                        best_a = a_pert
                        best_score = sc_pert
                        a = a_pert
                        stall = 0
                        step = 0.15
                    elif rng.random() < 0.03 + 0.012 * stall:
                        a = a_pert
                else:
                    break
            
            if stall > 100:
                break
        
        if it % 100 == 0:
            print(f"        LP iter {it}, score={best_score:.6f}, stall={stall}, peak_lp={use_peak_lp}")
    
    return best_a, best_score


def optimize_subgrad(a, deadline, rng):
    """Pure subgradient optimization with adaptive learning rate."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = compute_objective(a)
    lr = 0.04
    stall = 0
    
    while time.monotonic() < deadline and stall < 150:
        a_new = subgrad_direction(a, lr=lr, threshold_frac=0.8 + 0.05 * np.sin(lr * 50))
        sc = compute_objective(a_new)
        
        if sc < best_score - 1e-12:
            best_a = a_new
            best_score = sc
            a = a_new
            stall = 0
        else:
            stall += 1
            lr = max(0.002, lr * 0.92)
            if stall > 15:
                noise = rng.normal(0, 0.06, n)
                a_pert = best_a * (1.0 + noise)
                a_pert = np.clip(a_pert, 0.0, None)
                if a_pert.sum() > 1e-10:
                    a = a_pert / a_pert.sum()
                    stall = 0
                    lr = 0.04
                else:
                    break
    
    return best_a, best_score


def single_run(a_init, deadline, rng, method='lp'):
    """Run a single optimization from the given initialization."""
    a_init = np.clip(a_init, 0.0, None).astype(np.float64)
    S = a_init.sum()
    if S < 1e-10:
        a_init = np.ones_like(a_init)
        S = a_init.sum()
    a_init = a_init / S
    
    if method == 'lp':
        return optimize_lp_anneal(a_init, deadline, rng)
    elif method == 'subgrad':
        return optimize_subgrad(a_init, deadline, rng)
    else:
        # Combined: LP for a while, then subgrad polish
        mid_deadline = (time.monotonic() + deadline) / 2.0
        a_mid, sc_mid = optimize_lp_anneal(a_init, mid_deadline, rng)
        a_final, sc_final = optimize_subgrad(a_mid, deadline, rng)
        return a_final, sc_final


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_super_gaussian(n, p=6):
    center = (n - 1) / 2.0
    return np.exp(-((np.arange(n) - center) / (0.35 * n)) ** p)


def make_exponential(n, rate=1.5):
    center = (n - 1) / 2.0
    return np.exp(-rate * np.abs(np.arange(n) - center) / n)


def make_multi_bump(n, num_bumps=5, bump_width=0.08):
    """Multiple Gaussian bumps spread across the sequence."""
    a = np.zeros(n)
    centers = np.linspace(0.15 * n, 0.85 * n, num_bumps)
    sigma = bump_width * n
    for c in centers:
        a += np.exp(-0.5 * ((np.arange(n) - c) / sigma) ** 2)
    return a
