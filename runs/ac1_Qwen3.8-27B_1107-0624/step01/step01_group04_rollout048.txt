    from scipy.optimize import linprog as _linprog
except Exception:
    _linprog = None


def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _normalize(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def _greedy(n):
    n = int(max(2, n))
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return _normalize(a)


def _power_seq(n, alpha):
    n = int(max(2, n))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(alpha))
    return _normalize(a)


def _mix_greedy(n, lam):
    n = int(max(2, n))
    g = _greedy(n)
    u = np.ones(n, dtype=np.float64) / n
    return _normalize(lam * g + (1.0 - lam) * u)


def _exp_decay(n, rate):
    n = int(max(2, n))
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-float(rate) * idx / max(1, n - 1))
    return _normalize(a)


def _fft_conv_same(a):
    """Compute full convolution a*a, return only the first len(a) lags (symmetric part)."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    Pa = np.fft.rfft(a, L)
    c = np.fft.irfft(Pa * Pa, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _score(p):
    """Compute 2*n*max(c) / (sum(p))^2 for the full convolution."""
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    if n <= 512:
        c = np.convolve(a, a)
    else:
        c = _fft_conv_same(a)
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def _scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(a.size)
        s = a.size
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    if b.sum() < 0.01:
        b = b * (0.01 / max(b.sum(), 1e-30))
    return [float(x) for x in b]

def _adam_smooth_max(p, budget_s, seed=0, alpha_init=20.0, alpha_end=200.0):
    """
    Adam optimizer minimizing the smoothed max of the full autocorrelation,
    parameterized in log-space to enforce non-negativity.
    Adaptive temperature: starts smooth (low alpha) for exploration,
    ends sharp (high alpha) for exploitation.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 2:
        return p, _score(p)
    p = _normalize(p)
    budget_s = float(max(1.0, budget_s))
    if budget_s < 2.0:
        return p, _score(p)

    x = np.log(p + 1e-30)
    x -= x.mean()

    # Learning rate schedule
    if n <= 100:
        lr0 = 0.04
    elif n <= 300:
        lr0 = 0.025
    elif n <= 600:
        lr0 = 0.018
    else:
        lr0 = 0.012

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t_step = 0

    L = _next_pow2(2 * n - 1)
    best_p = p.copy()
    best_f = _score(p)

    t_start = time.monotonic()
    t_end = t_start + budget_s - 1.0

    n_iter_max = 50000
    while time.monotonic() < t_end and t_step < n_iter_max:
        t_step += 1
        frac = min(1.0, (time.monotonic() - t_start) / max(1e-9, budget_s))

        # Adaptive temperature: start smooth, end sharp
        alpha = alpha_init + (alpha_end - alpha_init) * frac * frac

        # Compute convolution
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = 2.0 * n * maxc  # p sums to 1
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Smoothed max weights
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(np.clip(z, -100, 100))
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient: d/dp [2n * sum(w_k * c_k)] where c_k = sum_i p_i * p_{k-i}
        # d c_k / d p_j = p_{k-j} (for valid indices)
        # So grad_p[j] = 2n * sum_k w_k * p_{k-j} (with appropriate index handling)
        # This is a cross-correlation of w with p
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        # Cfull[j] = sum_k w[k] * p[k-j] for k-j in [0,n-1] and k in [0, 2n-2]
        # We need: grad[j] = sum_{k=0}^{2n-2} w[k] * p[k-j] where k-j in [0, n-1]
        # = sum_{i=0}^{n-1} w[i+j] * p[i]  (substituting k = i+j)
        # This is the cross-correlation of w with p, evaluated at lag j
        # = irfft(FFT(w) * conj(FFT(p))) ... let me just do it directly
        # Actually: Cfull from above: Cfull[m] = sum_k w_flipped[k] * p[k + m - (n-1)]... 
        # Let me just compute it properly
        # grad_p[j] = sum_k w[k] * (dc_k/dp_j) = sum_k w[k] * p_{k-j} (if 0<=k-j<n)
        # = sum_{i=0}^{n-1} w[i+j] * p[i]
        # This is (w * p_bar)[j] where * denotes correlation
        # Using FFT: if W_full is the full-length DFT of w (padded to 2n-1),
        # then grad = ifft(FFT(w_padded) * conj(FFT(p_padded))) shifted appropriately
        
        # Simpler: just compute directly for moderate n, or use the FFT trick
        # Let's use: crosscorr(w, p)[j] = sum_i w[i+j] * p[i]
        # Pad w to length 3n, pad p to length 3n
        L3 = _next_pow2(3 * n)
        Wp = np.fft.rfft(w, L3)
        Pp = np.fft.rfft(p, L3)
        Ccross = np.fft.irfft(Wp * Pp.conj(), L3).real
        # Ccross[j] = sum_i w[i-j] * p[i] (circular) -- not quite what we want
        # Let me just do: grad[j] = sum_i w[i+j]*p[i] = cross-correlation
        # Using the conjugate trick:
        # ifft(conj(FFT(w)) * FFT(p))[j] = sum_i conj(w[i-j]) * p[i] (circular)
        # For real sequences: sum_i w[i-j] * p[i]
        # We want sum_i w[i+j] * p[i] = sum_k w[k] * p[k-j] (sub k=i+j)
        # = (w * p)[j] in the convolution sense (not correlation)
        # conv(w, p)[j] = sum_i w[i] * p[j-i]... no
        # Let me just be explicit. For n up to ~2048, direct computation in the loop is O(n^2) per step which is too slow.
        # 
        # Better approach: note that grad_p = 2 * (w reversed) * p (cross-correlation)
        # Actually, the gradient of max_k c_k w.r.t. p is:
        # d c_k / d p_j = p_{k-j} if k-j in range, else 0
        # So grad_p[j] = 2n * sum_k w_k * p_{k-j} (where k-j in [0,n-1])
        # = 2n * sum_{i=0}^{n-1} w_{i+j} * p_i
        # 
        # Let's compute this via FFT: define w_ext of length 2n-1 (already have it)
        # Then sum_i w[i+j]*p[i] for j=0,...,n-1
        # This is the "cross-correlation" of w and p at lag j
        # crosscorr(w, p)[j] = sum_i w[i+j] * p[i]
        # In FFT: ifft(FFT(w, N) * conj(FFT(p, N))) gives circular cross-correlation
        # For our purposes with zero-padding to at least 2n-1, it should work
        Lx = _next_pow2(3 * n - 2)
        Wx = np.fft.rfft(w, Lx)
        Px = np.fft.rfft(p, Lx)
        X = np.fft.irfft(Wx * Px.conj(), Lx).real
        # X[j] = sum_i w[i-j mod Lx] * p[i] for circular... 
        # Actually for real: ifft(FFT(w)*conj(FFT(p)))[k] = sum_n w[n-k] * p[n] (circular)
        # = sum_n w[(n-k) mod L] * p[n]
        # We want: sum_i w[i+j] * p[i] = sum_i w[i+j] * p[i]
        # Let m = i, then = sum_m w[m+j] * p[m]
        # From the formula: X[k] = sum_m w[m-k] * p[m]
        # Set k = -j: X[-j mod L] = sum_m w[m-(-j)] * p[m] = sum_m w[m+j] * p[m]
        # So X[(Lx-j) % Lx] = sum_m w[m+j] * p[m] (for j in range where no wraparound)
        # For j = 0: X[0] = sum_m w[m] * p[m]
        # For j = 1: X[Lx-1] = sum_m w[m+1] * p[m]
        grad_p = np.zeros(n, dtype=np.float64)
        for j in range(n):
            idx = (Lx - j) % Lx
            if idx < Lx:
                grad_p[j] = X[idx]
        # Actually this loop is slow. Let me just index the X array.
        # X[0], X[Lx-1], X[Lx-2], ..., X[Lx-(n-1)]
        if n <= 2:
            idxs = np.zeros(n, dtype=int)
        else:
            idxs = np.concatenate([np.zeros(1, dtype=int), (Lx - np.arange(1, n)).astype(int)])
        idxs = idxs % Lx
        grad_p = 2.0 * n * X[idxs]
        grad_p = np.maximum(grad_p, 0.0)  # clip to avoid numerical issues
        
        # Normalize gradient to log-parameter space
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad_x
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m_buf / (1.0 - beta1 ** t_step)
        v_hat = v_buf / ((1.0 - beta2 ** t_step) + 1e-30)
        
        # Decaying learning rate
        lr = lr0 * (1.0 + 2.0 * (1.0 - frac)) ** (-0.5)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -20.0, 20.0)
        x = x - step
        x -= x.mean()
        
        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = _score(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def _adam_smooth_max_v2(p, budget_s, seed=0):
    """
    Improved Adam with direct O(n) gradient via circular convolution trick.
    Uses the fact that grad_p = 2n * (w reversed) * p via FFT.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 2:
        return p, _score(p)
    p = _normalize(p)
    budget_s = float(max(1.0, budget_s))
    if budget_s < 2.0:
        return p, _score(p)

    x = np.log(p + 1e-30)
    x -= x.mean()

    lr0 = 0.02 if n <= 400 else 0.015
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t_step = 0

    L = _next_pow2(2 * n - 1)
    best_p = p.copy()
    best_f = _score(p)

    t_start = time.monotonic()
    t_end = t_start + budget_s - 1.0
    n_iter_max = 100000

    while time.monotonic() < t_end and t_step < n_iter_max:
        t_step += 1
        frac = min(1.0, (time.monotonic() - t_start) / max(1e-9, budget_s))
        alpha = 10.0 + 300.0 * frac ** 2

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = 2.0 * n * maxc
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        z = alpha * (c / maxc - 1.0)
        expz = np.exp(np.clip(z, -100, 100))
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient: grad_p[j] = 2n * sum_i w[i+j] * p[i]
        # Use FFT: this is the cross-correlation of w and p
        # corr[j] = sum_i w[i+j]*p[i]
        # FFT method: ifft(FFT(w) * conj(FFT(p))) gives circular correlation
        # With proper zero-padding, linear correlation = circular correlation
        Lc = _next_pow2(3 * n - 2)
        Wc = np.fft.rfft(w, Lc)
        Pc = np.fft.rfft(p, Lc)
        # corr[k] = ifft(FFT(w) * conj(FFT(p)))[k] = sum_n w[n-k]*p[n]
        # We want corr[j] = sum_i w[i+j]*p[i]
        # = sum_i w[i+j]*p[i] -- let m = i+j, i = m-j
        # = sum_m w[m]*p[m-j] = conv(w, p_flipped)[j + n - 1] ... getting confused
        #
        # Let me just use: grad_p = 2n * np.convolve(w, p[::-1], mode='valid')[::-1]
        # No wait. Let me think again.
        #
        # c_k = (p*p)_k = sum_{i=0}^{min(k,n-1)} p[i] * p[k-i] (with k-i in [0,n-1])
        # d c_k / d p_j = p[k-j] if k-j in [0,n-1], else 0
        #
        # Objective: F = sum_k w_k * c_k
        # dF/dp_j = sum_k w_k * p[k-j] (where k-j in [0,n-1])
        # = sum_{i=0}^{n-1} w[i+j] * p[i]  (substituting i = k-j, so k = i+j)
        #
        # So grad_p[j] = 2n * sum_{i=0}^{n-1} w[i+j] * p[i]
        #
        # This is the "cross-correlation" of the sequence w[0], w[1], ..., w[2n-2]
        # with p[0], p[1], ..., p[n-1], evaluated at lags j = 0, 1, ..., n-1.
        #
        # Using FFT: define W_full = FFT of w padded to length >= 3n-2
        #           P_full = FFT of p padded to same length
        # Then ifft(W_full * conj(P_full))[k] = sum_n w[(n-k) mod N] * p[n] (circular)
        # For our case with sufficient padding:
        # ifft(W_full * conj(P_full))[k] = sum_n w[n-k] * p[n] for 0 <= k < 2n-1
        # 
        # We want: sum_i w[i+j] * p[i] = sum_n w[n+j] * p[n]
        # From the formula with k = -j (mod N): sum_n w[n-(-j)] * p[n] = sum_n w[n+j] * p[n]
        # = ifft(W_full * conj(P_full))[(N-j) mod N]
        #
        # So grad_p[j] = 2n * ifft(W_full * conj(P_full))[(Lc - j) % Lc]

        corr_full = np.fft.irfft(Wc * Pc.conj(), Lc).real
        idxs = np.zeros(n, dtype=np.int64)
        if n > 1:
            idxs[1:] = Lc - np.arange(1, n, dtype=np.int64)
        # idxs = [0, Lc-1, Lc-2, ..., Lc-(n-1)]
        grad_p = 2.0 * n * corr_full[idxs % Lc]
        grad_p = np.maximum(grad_p, 0.0)

        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad_x
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m_buf / (1.0 - beta1 ** t_step)
        v_hat = v_buf / (1.0 - beta2 ** t_step)

        lr = lr0 * (1.0 + 3.0 * (1.0 - frac)) ** (-0.7)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -15.0, 15.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -50.0, 50.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = _score(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def _lp_improve(a, budget_s, max_constraints=800):
    """
    LP cutting-plane method to find a direction that increases sum while
    keeping max autocorrelation bounded.
    
    Formulation: given a (sum-normalized), with M = max(a*a),
    find g >= 0 maximizing sum(g) subject to (a*g)_k <= M for all k.
    Then blend: a_new = (1-t)*a + t*g, rescale.
    """
    if _linprog is None:
        return a, _score(a)
    
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n < 4:
        return a, _score(a)
    budget_s = float(max(10.0, budget_s))
    
    a = _normalize(a)
    c = _fft_conv_same(a) if n > 300 else np.convolve(a, a)
    c = np.maximum(c, 0.0)
    M = float(c.max())
    
    if not np.isfinite(M) or M <= 0:
        return a, _score(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Initial constraints: every (2n-1)//max_constraints lags
    n_convs = 2 * n - 1
    n_init = min(max_constraints // 2, n_convs)
    step_size = max(1, n_convs // n_init)
    rows = list(range(0, n_convs, step_size))
    row_set = set(rows)
    
    t_start = time.monotonic()
    t_end = t_start + budget_s - 10.0
    it = 0
    
    # Precompute the convolution matrix rows lazily
    # A[k, j] = a[max(0, k-n+1):min(k, n-1)+1] aligned with j
    # (a*g)_k = sum_j A[k,j] * g[j]
    
    def _build_A(row_list):
        """Build constraint matrix for given rows."""
        nr = len(row_list)
        A = np.zeros((nr, n), dtype=np.float64)
        for ii, k in enumerate(row_list):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                length = hi - lo + 1
                A[ii, lo:hi+1] = a[k - hi : k - lo + 1][::-1]
        return A
    
    while time.monotonic() < t_end:
        it += 1
        rows_arr = list(rows)
        nr = len(rows_arr)
        
        # Limit LP size
        if nr * n > 4000000:
            # Too large, subsample
            idx = np.linspace(0, nr - 1, min(nr, 3000), dtype=int)
            rows_arr = [rows_arr[i] for i in idx]
            nr = len(rows_arr)
        
        A = _build_A(rows_arr)
        b_ub = np.full(nr, M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)
        
        rem = max(1.0, t_end - time.monotonic() - 2.0)
        tl = max(3.0, min(60.0, rem * 0.3))
        
        res = None
        try:
            res = _linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                          method='highs', options={'time_limit': tl, 'presolve': True})
        except Exception:
            try:
                res = _linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None),
                              method='highs', options={'time_limit': tl})
            except Exception:
                break
        
        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break
        
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break
        
        # Check full convolution constraint violations
        if n > 500:
            L = _next_pow2(2 * n - 1)
            Pf = np.fft.rfft(a, L)
            Pg = np.fft.rfft(g, L)
            conv = np.fft.irfft(Pf * Pg, L)[:2 * n - 1].real
        else:
            conv = np.convolve(a, g)
        conv = np.maximum(conv, 0.0)
        
        tol = 1e-8 + 1e-8 * M
        violations = conv - M
        max_viol = float(np.max(violations)) if violations.size > 0 else 0.0
        
        if max_viol > tol:
            # Add most violated constraints
            if len(rows) < max_constraints:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    ix = int(ix)
                    if violations[ix] <= tol:
                        break
                    if ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                        if added >= 128 or len(rows) >= max_constraints:
                            break
                if added > 0:
                    continue
            else:
                # Max constraints reached, use most violated as a penalty
                break
        
        # Blend direction
        best_blend = None
        best_blend_score = np.inf
        for t in (0.3, 0.5, 0.7, 0.85, 1.0, 0.4, 0.6):
            fn = (1.0 - t) * a + t * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn / sn
            sc = _score(fn)
            if np.isfinite(sc) and sc < best_blend_score:
                best_blend_score = sc
                best_blend = fn.copy()
        
        if best_blend is None or not np.isfinite(best_blend_score):
            break
        if best_blend_score >= best_score:
            break
        
        a = best_blend
        best_score = best_blend_score
        best_a = a.copy()
        
        # Update M
        c = _fft_conv_same(a) if n > 300 else np.convolve(a, a)
        c = np.maximum(c, 0.0)
        M = float(c.max())
        
        if it % 10 == 0:
            print("  LP iter %d: score=%.6f M=%.6f rows=%d" % (it, best_score, M, len(rows)))
    
    return best_a, _score(best_a)


def _spectral_shape(p, budget_s):
    """
    Try to shape the DFT magnitude to be flatter, which corresponds to
    flatter autocorrelation. Use iterative reweighting in frequency domain.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4:
        return p, _score(p)
    p = _normalize(p)
    budget_s = float(max(5.0, budget_s))
    
    best_p = p.copy()
    best_f = _score(p)
    
    t_start = time.monotonic()
    t_end = t_start + budget_s - 2.0
    
    # Work in log-domain
    x = np.log(p + 1e-30)
    x -= x.mean()
    
    lr = 0.01
    for it in range(500):
        if time.monotonic() > t_end:
            break
        
        p = _normalize(np.exp(x - x.max()))
        
        # Compute spectrum
        L = _next_pow2(n)
        P = np.fft.rfft(p, L)
        mag = np.abs(P)
        mag2 = mag * mag
        
        # We want mag2 to be as flat as possible (minimize max - min)
        # Gradient approach: push down the peaks
        max_mag = float(mag2.max())
        if max_mag <= 0 or not np.isfinite(max_mag):
            break
        
        # Weights on frequency bins
        w_f = mag2 / max_mag
        # Gradient w.r.t. p: d(mag2[k])/dp = 2*Re(P[k] * conj(FFT(p_shifted)))
        # Simplified: use the phase of P to guide updates
        phase = np.angle(P)
        
        # Compute autocorrelation directly
        c = np.fft.irfft(P * P, L)[:2*n-1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        
        f_now = 2.0 * n * maxc
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()
        
        # Soft gradient
        alpha = 50.0
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(np.clip(z, -50, 50))
        sz = expz.sum()
        if sz <= 0:
            break
        w = expz / sz
        
        # Gradient via cross-correlation (same as before)
        Lc = _next_pow2(3 * n - 2)
        Wc = np.fft.rfft(w, Lc)
        Pc = np.fft.rfft(p, Lc)
        corr_full = np.fft.irfft(Wc * Pc.conj(), Lc).real
        idxs = np.zeros(n, dtype=np.int64)
        if n > 1:
            idxs[1:] = Lc - np.arange(1, n, dtype=np.int64)
        grad_p = 2.0 * n * corr_full[idxs % Lc]
        grad_p = np.maximum(grad_p, 0.0)
        
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)
        
        x = x - lr * grad_x
        x -= x.mean()
        x = np.clip(x, -40, 40)
        
        # Decay learning rate
        lr *= 0.999
    
    final_f = _score(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))
    
    best_p = None
    best_score = np.inf
    best_by_n = {}
    
    def record(p, n=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        sc = _score(a)
        if not np.isfinite(sc):
            return
        if sc < best_score:
            best_p = a.copy()
            best_score = sc
            print("  NEW BEST: n=%d score=%.6f" % (a.size, sc))
        nn = int(a.size)
        if n is not None:
            nn = n
        old = best_by_n.get(nn)
        if old is None or sc < old:
            best_by_n[nn] = sc
    
    # Phase 1: Quick baselines
    print("Phase 1: Baselines...")
    base_lengths = [128, 160, 192, 224, 256, 288, 320, 352, 384, 416, 448, 480, 512, 640, 768]
    for n in base_lengths:
        if time.monotonic() >= deadline - 600:
            break
        record(_greedy(n))
        record(_power_seq(n, 0.4))
        record(_power_seq(n, 0.5))
        record(_power_seq(n, 0.6))
        record(_mix_greedy(n, 0.3))
        record(_mix_greedy(n, 0.7))
        record(_exp_decay(n, 0.5))
        record(_exp_decay(n, 1.0))
        record(_exp_decay(n, 2.0))
    
    # Try the global variable if available
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size >= 8:
                record(arr)
                print("  Baseline from height_sequence_1: n=%d score=%.6f" % (arr.size, _score(arr)))
        except Exception:
            pass
    
    print("  Best after baselines: score=%.6f" % best_score)
    
    # Phase 2: Adam optimization on top candidates
    print("Phase 2: Adam optimization...")
    # Pick top lengths
    items = sorted(best_by_n.items(), key=lambda x: x[1])
    top_lengths = [n for n, sc in items[:8]]
    
    jobs = []
    for n in top_lengths:
        jobs.append((n, 'greedy', 30))
        jobs.append((n, 'perturb', 25))
    
    # Add some diversity
    for n in [256, 320, 384, 448, 512]:
        if n not in top_lengths:
            jobs.append((n, 'greedy', 20))
    
    # Prepare starting points
    for n, kind, alloc in jobs:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        rec = best_by_n.get(int(n))
        if rec is not None:
            base = _normalize(_greedy(n))
            # Use the best found for this length if available
            # (we don't store the actual array, just the score, so rebuild)
            if kind == 'perturb':
                base = _greedy(n)
                noise = np.exp(rng.normal(0, 0.1, n))
                base = _normalize(base * noise)
            else:
                base = _greedy(n)
        else:
            base = _greedy(n)
        
        alloc_use = min(float(alloc), max(5.0, rem * 0.15))
        p_out, f_out = _adam_smooth_max_v2(base, alloc_use, seed=int(seed) + n)
        record(p_out)
    
    print("  Best after Adam: score=%.6f" % best_score)
    
    # Phase 3: LP cutting plane
    print("Phase 3: LP optimization...")
    if _linprog is not None and best_p is not None:
        # Focus on the best length
        best_n = int(best_p.size)
        lp_lengths = [best_n]
        # Add nearby lengths
        for n in [best_n - 32, best_n - 16, best_n + 16, best_n + 32]:
            if n >= 64 and n not in lp_lengths:
                lp_lengths.append(n)
        
        for n in lp_lengths:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            base = _greedy(n)
            # Try to start from a good point
            if n == best_n and best_p is not None:
                base = best_p.copy()
            
            alloc = min(200.0, max(30.0, rem * 0.4))
            p_out, f_out = _lp_improve(base, alloc, max_constraints=600)
            record(p_out)
            
            # Follow up with Adam on the LP result
            rem2 = deadline - time.monotonic()
            if rem2 > 30 and p_out is not None:
                p_out2, f_out2 = _adam_smooth_max_v2(p_out, min(60.0, rem2 * 0.3), seed=int(seed) + n + 7)
                record(p_out2)
    
    print("  Best after LP: score=%.6f" % best_score)
    
    # Phase 4: Additional Adam with different hyperparameters
    print("Phase 4: Refinement...")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            # Fine-tune the best
            p_out, f_out = _adam_smooth_max_v2(best_p, min(120.0, rem * 0.5), seed=int(seed) + 999)
            record(p_out)
        
        rem = deadline - time.monotonic()
        if rem > 40:
            # Perturb and re-optimize
            n = best_p.size
            perturbed = _normalize(best_p * np.exp(rng.normal(0, 0.05, n)))
            p_out, f_out = _adam_smooth_max_v2(perturbed, min(80.0, rem * 0.6), seed=int(seed) + 12345)
            record(p_out)
    
    # Phase 5: Try a few more lengths with aggressive optimization
    print("Phase 5: Exploration...")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Try lengths we haven't optimized much
        for n in [200, 250, 300, 350, 400, 450, 500, 550]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            base = _greedy(n)
            p_out, f_out = _adam_smooth_max_v2(base, min(40.0, rem * 0.4), seed=int(seed) + n * 3)
            record(p_out)
            
            # Quick LP if available
            rem = deadline - time.monotonic()
            if _linprog is not None and rem > 30:
                p_out2, f_out2 = _lp_improve(p_out, min(30.0, rem * 0.5), max_constraints=400)
                record(p_out2)
    
    # Final validation
    print("Final: validating...")
    if best_p is not None:
        seq = _scale_output(best_p, 10.0)
        gscore = _try_eval(seq)
        if np.isfinite(gscore):
            print("  Global eval: %.6f" % gscore)
            if gscore < best_score:
                best_score = gscore
    
    if best_p is None:
        best_p = _greedy(384)
    
    print("Final best score (local): %.6f, length=%d" % (best_score, best_p.size))
    return _scale_output(best_p, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation function."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Exception in main_search: %s" % str(e))
        import traceback
        traceback.print_exc()
        p = _greedy(384)
        return _scale_output(p, 10.0)
