    n = len(v)

def fast_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def normalize(s):
    s = np.maximum(s, 0.0)
    total = s.sum()
    if total < 1e-15:
        return np.ones(len(s)) / len(s)
    return s / total

def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]

    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def rfft_conv(a, b):
    """FFT-based convolution for real sequences."""
    n = len(a) + len(b) - 1

    result = np.fft.irfft(fa * fb, size)

def project_simplex(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0.0)

        hi = min(n-1, k)
        j_range = np.arange(lo, hi+1)
        grad[j_range] += s[k - j_range]

    """LP: max sum(b) s.t. (s*b)[k] <= M, b >= 0."""
    n = len(s)

    for j in range(n):
        A[j:j+n, j] = s
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0)
    except:
        pass
    return None

"""
Optimizes non-negative sequence to minimize 2n*max(conv(s,s))/(sum(s))^2.
Uses subgradient local search with simulated annealing, FFT convolution,
and LP refinement. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time

def rfft_conv(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]

def score_norm(s):
    n = len(s)
    conv = rfft_conv(s, s)
    return 2.0 * n * conv.max()

def norm(s):
    s = np.maximum(s, 0.0)
    t = s.sum()
    if t < 1e-15:
        return np.ones(len(s)) / len(s)
    return s / t

def build_conv_matrix(s):
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A

def lp_dir(s, M, tl=5.0):
    n = len(s)
    A = build_conv_matrix(s)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': tl})
        if res.success:
            return np.maximum(res.x, 0)
    except:
        pass
    return None

def local_search(s, rng, max_iters=50000, deadline=None, T0=0.05, T1=0.001):

    best = s.copy()
    best_sc = score_norm(s)
    cur_sc = best_sc
    no_imp = 0
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        conv = rfft_conv(s, s)
        # Use top-3 peaks for smoother subgradient
        top_k = min(3, 2*n-1)
        peaks = np.argsort(conv)[::-1][:top_k]
        sg = np.zeros(n)
        for k in peaks:
            lo = max(0, k - (n-1))

        # Find high-sg and low-sg coordinates
        if sg.max() - sg.min() < 1e-12:
            # Flat subgradient, random perturbation
