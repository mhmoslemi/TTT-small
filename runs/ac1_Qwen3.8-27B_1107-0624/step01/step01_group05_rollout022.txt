def _z_from_sequence(seq, n):
    """Convert a sequence to logits representation of length n."""
    seq = np.asarray(seq, dtype=np.float64)
    if len(seq) != n:
        # Resample
        old = np.linspace(0, 1, len(seq))
        new = np.linspace(0, 1, n)
        seq = np.interp(new, old, seq)
    seq = np.maximum(seq, 1e-12)
    p = seq / np.sum(seq)
    z = np.log(p[1:]) - np.log(p[0])
    return np.clip(z, -50, 50)

    return g


def _logsumexp_smooth(c, alpha):
    """Return (smooth_max, weights)."""
    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    np.clip(u, -50.0, 50.0, out=u)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
        return cmax, w
    J = cmax + math.log(s) / alpha
    w = e / s
    return J, w


def _optimize_lbfgs(z0, n, alpha, max_fun, deadline):
    """L-BFGS-B on smoothed objective. Returns optimized z."""
    global _best_score, _best_seq

    def obj_and_grad(x):
        _check_time()
        x = np.clip(x, -50.0, 50.0)
        p = _probs_from_z(x, n)
        c = _self_conv(p, n)
        hard = 2.0 * n * float(np.max(c))
        if hard < _best_score:
            _best_score = hard
            _best_seq = np.array(p, copy=True)
        J, w = _logsumexp_smooth(c, alpha)
        gp = _grad_full(p, w, n)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gz = np.where(np.isfinite(gz), gz, 0.0)
        return float(J), gz

    bounds = [(-50.0, 50.0)] * (n - 1)
    try:
        res = minimize(obj_and_grad, np.array(z0, dtype=np.float64, copy=True),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': max(50, max_fun // 3),
                                'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 25})
        return np.array(res.x, dtype=np.float64)
    except _Timeout:
        return np.array(z0, dtype=np.float64)
    except Exception:
        return np.array(z0, dtype=np.float64)


def _mirror_descent_nesterov(z0, n, deadline, topk=16, base_lr=0.08, recompute_every=8):
    """Mirror descent with Nesterov acceleration and cached peak gradients."""
    global _best_score, _best_seq

    z = np.array(z0, dtype=np.float64, copy=True)
    v = np.zeros_like(z)
    z_nest = z.copy()
    p = _probs_from_z(z, n)

    # Initial evaluation
    c = _self_conv(p, n)
    cmax = float(np.max(c))
    score = 2.0 * n * cmax
    best_local = score
    if score < _best_score:
        _best_score = score
        _best_seq = np.array(p, copy=True)

    peak_idx = np.argpartition(c, -topk)[-topk:]
    peak_vals = c[peak_idx]
    peak_w = peak_vals / (float(np.sum(peak_vals)) + 1e-15)

    lr = base_lr
    mu = 0.85
    t = 0
    stagnation = 0

    while time.monotonic() < deadline:
        t += 1
        if t % 50 == 0:
            _check_time()

        # Nesterov lookahead
        beta_t = 1.0 / (1.0 + 0.5 * math.sqrt(1.0 + 16.0 * mu * t))
        z_nest = z + beta_t * (z - z_nest) if t > 1 else z.copy()
        p_nest = _probs_from_z(z_nest, n)

        # Compute gradient at Nesterov point (using cached peaks)
        if t % recompute_every == 1:
            c = _self_conv(p_nest, n)
            cmax = float(np.max(c))
            peak_idx = np.argpartition(c, -topk)[-topk:]
            peak_vals = c[peak_idx] + 1e-15
            peak_w = peak_vals / float(np.sum(peak_vals))

        gp = _grad_fast(p_nest, peak_idx, peak_w, n)
        sdot = float(np.dot(gp, p_nest))
        gz = p_nest[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        if gnorm < 1e-15:
            break

        # Mirror descent step
        v = mu * v - lr * gz / gnorm
        z_candidate = z + v
        z_candidate = np.clip(z_candidate, -50.0, 50.0)
        p_candidate = _probs_from_z(z_candidate, n)

        # Evaluate (less frequently to save time)
        if t % recompute_every == 0 or t < 10:
            c_new = _self_conv(p_candidate, n)
            score_new = 2.0 * n * float(np.max(c_new))
        else:
            # Approximate score using cached peaks
            c_approx = _self_conv(p_candidate, n)  # still need full conv for accuracy
            score_new = 2.0 * n * float(np.max(c_approx))

        if score_new < score + 1e-14:
            z = z_candidate
            p = p_candidate
            score = score_new
            if score < best_local - 1e-12:
                best_local = score
                stagnation = 0
                if score < _best_score:
                    _best_score = score
                    _best_seq = np.array(p, copy=True)
            else:
                stagnation += 1
            # Adaptive learning rate
            lr = min(lr * 1.02, 0.5)
        else:
            lr = max(lr * 0.6, 0.005)
            v *= 0.5
            z = z_candidate * 0.5 + z * 0.5
            z = np.clip(z, -50.0, 50.0)
            p = _probs_from_z(z, n)
            c_tmp = _self_conv(p, n)
            score = 2.0 * n * float(np.max(c_tmp))
            stagnation += 1

        # Noise injection to escape local minima
        if stagnation > 200:
            noise = 0.1 * (1.0 + stagnation / 200.0)
            z = z + np.random.randn(len(z)) * noise
            z = np.clip(z, -50.0, 50.0)
            p = _probs_from_z(z, n)
            c_tmp = _self_conv(p, n)
            score = 2.0 * n * float(np.max(c_tmp))
            stagnation = 100
            v = np.zeros_like(z)

        if stagnation > 800:
            break

    return z


def _make_structured_z(n, variant_id):
    """Create diverse structured initializations."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    t = x - 0.5
    shapes = [
        # 0: logistic
        lambda: np.tanh(4.0 * t) * 2.5,
        # 1: strong logistic
        lambda: np.tanh(8.0 * t) * 3.5,
        # 2: linear
        lambda: 3.0 * t,
        # 3: quadratic (edge-peaked)
        lambda: 4.0 * t * t - 0.6,
        # 4: cosine
        lambda: 2.0 * np.cos(2.0 * np.pi * x) * 0.8,
        # 5: sine
        lambda: 2.5 * np.sin(2.0 * np.pi * x),
        # 6: power
        lambda: (x ** 2 - x) * 3.0,
        # 7: bimodal
        lambda: -3.0 * (1.0 + np.cos(4.0 * np.pi * x)) * 0.6,
        # 8: sigmoid product
        lambda: 4.0 * np.tanh(5.0 * (x - 0.3)) * np.tanh(5.0 * (x - 0.7)),
        # 9: cubic
        lambda: 3.0 * t * (1.0 - 4.0 * t * t * 0.5),
        # 10: mirror
        lambda: np.tanh(6.0 * (0.5 - x)) * 3.0,
        # 11: combined
        lambda: np.tanh(3.0 * t) * 2.0 + 1.5 * np.cos(4.0 * np.pi * x) * 0.5,
    ]
    idx = variant_id % len(shapes)
    f = shapes[idx]()
    f = np.clip(f, -40.0, 40.0)
    p = np.exp(f)
    p = p / (np.sum(p) + 1e-30)
    p = np.maximum(p, 1e-12)
    z = np.log(p[1:]) - np.log(p[0])
    return np.clip(z, -50.0, 50.0)


def _z_from_sequence(seq, n):
    """Resample a sequence to length n and convert to logits."""
    seq = np.asarray(seq, dtype=np.float64)
    if len(seq) != n:
        old = np.linspace(0.0, 1.0, len(seq))
        new = np.linspace(0.0, 1.0, n)
        seq = np.interp(new, old, np.maximum(seq, 0.0))
    seq = np.maximum(seq, 1e-12)
    p = seq / (float(np.sum(seq)) + 1e-30)
    return _z_from_p(p, n)


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_seq, _deadline

    start = time.monotonic()
    _deadline = start + budget_s - 5.0
    _best_score = float('inf')
    _best_seq = None

    rng = np.random.default_rng(seed)

    # Try to load previous best
    prev_seq = None
    try:
        prev = height_sequence_1
        prev = np.asarray(prev, dtype=np.float64)
        if prev.size > 0:
            prev = np.where(np.isfinite(prev), prev, 0.0)
            prev = np.clip(prev, 0.0, None)
            if float(np.sum(prev)) > 0.01:
                prev_seq = prev
                p_prev = prev / float(np.sum(prev))
                s_prev = _score_p(p_prev, len(p_prev))
                _best_score = min(_best_score, s_prev)
                _best_seq = np.array(p_prev, copy=True)
                print(f"Loaded previous best: score={s_prev:.6f}, n={len(prev)}")
    except Exception:
        pass

    # Determine N values to try
    n_list = [4096]
    if budget_s > 600:
        n_list.append(8192)
    if budget_s > 1200:
        n_list.append(16384)

    # If we have a previous sequence with a different length, include it
    if prev_seq is not None and len(prev_seq) not in n_list:
        n_list.insert(0, len(prev_seq))

    print(f"N values to try: {n_list}")
    print(f"Time budget: {budget_s:.0f}s")

    # Allocate time: 60% to N=4096, 30% to next, 10% to last
    time_alloc = {}
    total_weight = float(len(n_list))
    weights = [1.5] + [1.0] * (len(n_list) - 1)
    wsum = sum(weights)
    cum_w = 0.0
    for i, n in enumerate(n_list):
        cum_w += weights[i] / wsum
        time_alloc[n] = start + (budget_s - 20.0) * cum_w

    for n in n_list:
        n_deadline = time_alloc[n] - 5.0
        if time.monotonic() > n_deadline - 15:
            continue

        remaining = n_deadline - time.monotonic()
        print(f"\n=== N={n}: {remaining:.0f}s allocated ===")

        # Phase 1: Structured init + L-BFGS-B (40% of N's time)
        phase1_deadline = start + (n_deadline - start) * 0.4 + start
        # Actually let's do it relative
        phase1_time = remaining * 0.35
        phase2_time = remaining * 0.40
        phase3_time = remaining * 0.25

        p1_deadline = time.monotonic() + phase1_time
        p2_deadline = p1_deadline + phase2_time
        p3_deadline = n_deadline - 3.0

        # Structured initializations
        n_struct = max(6, min(16, int(phase1_time / 40)))
        candidates = []

        for si in range(n_struct):
            z0 = _make_structured_z(n, si * 2 + (n % 7))
            p0 = _probs_from_z(z0, n)
            s0 = _score_p(p0, n)
            if s0 < _best_score:
                _best_score = s0
                _best_seq = np.array(p0, copy=True)
            candidates.append((s0, z0))
            if time.monotonic() > p1_deadline - 30:
                break

        # Add previous best if available
        if prev_seq is not None:
            z_prev = _z_from_sequence(prev_seq, n)
            p_prev_n = _probs_from_z(z_prev, n)
            s_prev_n = _score_p(p_prev_n, n)
            if s_prev_n < _best_score:
                _best_score = s_prev_n
                _best_seq = np.array(p_prev_n, copy=True)
            candidates.append((s_prev_n, z_prev))

        # Sort by score and run L-BFGS-B on top ones
        candidates.sort(key=lambda x: x[0])
        n_lbfgs = min(len(candidates), max(4, int(phase1_time / 50)))

        for ci in range(n_lbfgs):
            if time.monotonic() > p1_deadline - 10:
                break
            _, z0 = candidates[ci]
            # Adaptive alpha based on current best
            if _best_score > 1.7:
                alpha = 2e6
            elif _best_score > 1.52:
                alpha = 5e6
            else:
                alpha = 1e7
            max_fun = max(100, int(min(400, (p1_deadline - time.monotonic()) / 0.5)))
            z_opt = _optimize_lbfgs(z0, n, alpha, max_fun, p1_deadline - 5)
            p_opt = _probs_from_z(z_opt, n)
            s_opt = _score_p(p_opt, n)
            if s_opt < _best_score:
                _best_score = s_opt
                _best_seq = np.array(p_opt, copy=True)
                print(f"  L-BFGS cand {ci}: {s_opt:.6f}")

        print(f"  Phase 1 done: best={_best_score:.6f}")

        # Phase 2: Mirror descent / Nesterov from best
        if _best_seq is not None and time.monotonic() < p2_deadline - 30:
            z_start = _z_from_p(_best_seq, n)
            # Multiple runs with different hyperparameters
            for lr_try in [0.06, 0.12, 0.25]:
                if time.monotonic() > p2_deadline - 20:
                    break
                zres = _mirror_descent_nesterov(
                    z_start + rng.normal(0, 0.05, n - 1), n,
                    p2_deadline - 10, topk=16, base_lr=lr_try, recompute_every=8)
                pres = _probs_from_z(zres, n)
                sres = _score_p(pres, n)
                if sres < _best_score:
                    _best_score = sres
                    _best_seq = np.array(pres, copy=True)
                    print(f"  Mirror lr={lr_try}: {sres:.6f}")
                z_start = zres  # chain from result

        print(f"  Phase 2 done: best={_best_score:.6f}")

        # Phase 3: Random restarts + fine L-BFGS-B
        if time.monotonic() < p3_deadline - 20:
            n_random = max(2, int((p3_deadline - time.monotonic()) / 60))
            for ri in range(n_random):
                if time.monotonic() > p3_deadline - 10:
                    break
                if _best_seq is not None:
                    z_base = _z_from_p(_best_seq, n)
                    noise_scale = 0.8 - 0.5 * (ri / max(1, n_random))
                    z_rand = z_base + rng.normal(0, noise_scale, n - 1)
                else:
                    z_rand = rng.normal(0, 1.5, n - 1)
                z_opt = _optimize_lbfgs(z_rand, n, 5e6, 200, p3_deadline - 5)
                p_opt = _probs_from_z(z_opt, n)
                s_opt = _score_p(p_opt, n)
                if s_opt < _best_score:
                    _best_score = s_opt
                    _best_seq = np.array(p_opt, copy=True)
                    print(f"  Random {ri}: {s_opt:.6f}")

        print(f"  N={n} complete: best={_best_score:.6f}")

    # Final polishing on global best
    if _best_seq is not None and time.monotonic() < _deadline - 10:
        n_final = len(_best_seq)
        z_final = _z_from_p(_best_seq, n_final)
        print(f"\n=== Final polish: n={n_final}, score={_best_score:.6f} ===")

        # High-alpha L-BFGS-B
        for alpha in [8e6, 1.5e7, 2.5e7]:
            if time.monotonic() > _deadline - 8:
                break
            z_pert = z_final + rng.normal(0, 0.02, n_final - 1)
            z_opt = _optimize_lbfgs(z_pert, n_final, alpha, 200, _deadline - 4)
            p_opt = _probs_from_z(z_opt, n_final)
            s_opt = _score_p(p_opt, n_final)
            if s_opt < _best_score:
                _best_score = s_opt
                _best_seq = np.array(p_opt, copy=True)
                z_final = z_opt
                print(f"  Polish alpha={alpha:.0e}: {s_opt:.6f}")

        # Mirror descent fine-tune
        if time.monotonic() < _deadline - 8:
            zres = _mirror_descent_nesterov(z_final, n_final, _deadline - 2,
                                            topk=32, base_lr=0.02, recompute_every=4)
            pres = _probs_from_z(zres, n_final)
            sres = _score_p(pres, n_final)
            if sres < _best_score:
                _best_score = sres
                _best_seq = np.array(pres, copy=True)
                print(f"  Final mirror: {sres:.6f}")

    # Construct final sequence
    if _best_seq is not None:
        seq = np.array(_best_seq, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        s = float(np.sum(seq))
        if s < 0.01:
            seq = seq * (1.0 / (s + 1e-30))
            seq = np.clip(seq, 0.0, 1000.0)
        result = [float(v) for v in seq]
    else:
        result = [1.0 / 4096.0] * 4096

    # Verify with actual evaluator
    try:
        final_score = float(evaluate_sequence(result))
        print(f"\nFinal verified: score={final_score:.6f}, length={len(result)}")
    except Exception as e:
        print(f"Evaluation error: {e}, using fallback")
        result = [1.0 / 4096.0] * 4096
        final_score = 2.0

    return result
