"""
Search for non-negative step-function sequences minimizing
    f = 2*n*max(convolve(seq,seq)) / sum(seq)^2
using a hybrid strategy: FFT-based Adam gradient descent followed by
an LP cutting-plane method with warm-starting and adaptive constraints.

import time
import numpy as np

    import scipy.optimize as _sp_opt
    _HAS_SCIPY = True
except Exception:
    _HAS_SCIPY = False


def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _normalize(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def _score_fft(p):
    """Compute 2*n*max(conv) / sum^2 using FFT."""
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-10:
        return np.inf
    if n <= 512:
        c = np.convolve(a, a, mode='full')
    else:
        L = _next_pow2(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))


def _score_direct(p):
    """Exact score using np.convolve (for verification)."""
    a = np.clip(np.asarray(p, dtype=np.float64).ravel(), 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return float(2.0 * n * max(c) / (s * s))


def _greedy_init(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return _normalize(a)


def _power_init(n, s):
    idx = np.arange(1, n + 1, dtype=np.float64)
    return _normalize(idx ** (-s))


def _mix_init(n, lam):
    return _normalize(lam * _greedy_init(n) + (1.0 - lam) * np.ones(n, dtype=np.float64))


def _dirichlet_init(n, alpha, rng):

def _perturb(p, rng, sigma):

def _scale_output(p, target_sum=10.0):

def _adam_optimize(p0, budget_s, seed=0):
    """Adam with softmax-projected gradient descent on log-space params."""
    p = _normalize(p0.astype(np.float64).copy())
    n = p.size
    if n == 0:
        return p, np.inf
    budget_s = float(budget_s)
    if budget_s < 1.0:
        return p, _score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 128:
        lr = 0.06
    elif n < 512:
        lr = 0.035
    elif n < 2048:
        lr = 0.022
    else:
        lr = 0.015

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t_step = 0

    L = _next_pow2(2 * n - 1)
    best_p = p.copy()
    best_f = _score_fft(p)

    t_start = time.monotonic()
    t_end = t_start + budget_s

    while time.monotonic() < t_end:
        t_step += 1
        elapsed = time.monotonic() - t_start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = 8.0 + 60.0 * np.exp(-5.0 * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = 2.0 * n * maxc  # p sums to 1
        if f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Softmax weights over autocorrelation
        z = alpha * (c / maxc - 1.0)
        z -= z.max()
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz < 1e-300:
            idx = int(np.argmax(c))
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[idx] = 1.0
        else:
            w = expz / sz

        # Gradient: grad_p[j] = 4*n * sum_m w[m] * f[m-j] (valid range)
        # This is a cross-correlation of w with f
        W = np.fft.rfft(w, L)
        grad_raw = np.fft.irfft(W * Pg, L)[:2 * n - 1].real
        # We need grad_p[j] = 2 * sum_m w[m] * p[m-j] for 0 <= m-j < n
        # This equals 2 * (w * p_reversed)[n-1+j] ... let me use the FFT approach
        # Actually: sum_m w[m] * p[m-j] = (w_conj * p)[j] in some sense
        # Let me use: the cross-correlation <w, shift(p, j)>
        # Using FFT: ifft(fft(w) * conj(fft(p))) gives cross-correlation
        # Actually simpler: grad_p[j] = 2*sum_m w[m]*p[m-j]
        # Let's compute via: take fft of w (padded to length >= 2n-1), multiply by fft(p), ifft
        # The result at position (2n-2-j) gives sum_m w[m+j]*p[m]... 
        # Let me just do it directly with the FFT trick:
        # Define h[m] = w[m] for m in [0, 2n-2], padded to L
        # Then sum_m h[m]*p[m-j] for valid m-j is a cross-correlation
        # cross_corr(j) = sum_m h[m] * p[m-j] = ifft(FFT(h) * conj(FFT(p)))[j] ... 
        # No, cross_corr(j) = sum_m h[m] * conj(p[m-j]) = ifft(FFT(h) * conj(FFT(p)))
        # But we want sum_m w[m] * p[m-j], which is the same since p is real.
        # So: grad_p[j] = 2 * cross_corr[j] where cross_corr = ifft(FFT(w_pad) * conj(FFT(p_pad)))
        
        # Actually, let me just reuse the approach from the previous code which worked:
        # Shrink spectrum method:
        Wf = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * Wf, L)
        C = Cfull[n - 1: 2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)
        grad_p = (4.0 * n) * C

        if not np.all(np.isfinite(grad_p)):
            break
        
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad_x
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m_buf / (1.0 - beta1 ** t_step)
        v_hat = v_buf / (1.0 - beta2 ** t_step)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -100.0, 100.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        # Softmax projection
        x_clip = np.clip(x, -100.0, 100.0)
        x_clip -= x_clip.max()
        e = np.exp(x_clip)
        se = float(e.sum())
        if not np.isfinite(se) or se < 1e-300:
            break
        p = e / se

    final_f = _score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def _lp_improve(p0, budget_s, verbose=False):
    """
    LP cutting-plane method: iteratively solve
        max sum(g) s.t. (f*g)[k] <= M, g >= 0
    where f = sqrt(2n) * normalize(p0), M = max(f*f).
    Blend f with solution g to reduce peak autocorrelation.
    """
    if not _HAS_SCIPY:
        return p0, _score_fft(p0)

    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    if n < 4:
        return p0, _score_fft(p0)
    budget_s = float(budget_s)
    if budget_s < 30.0:
        return p0, _score_fft(p0)

    S0 = np.sqrt(2.0 * n)
    f = _normalize(p) * S0
    cur_M = 0.0
    # Compute current max autocorrelation
    if n <= 512:
        c0 = np.convolve(f, f, mode='full')
    else:
        L0 = _next_pow2(2 * n - 1)
        P0 = np.fft.rfft(f, L0)
        c0 = np.fft.irfft(P0 * P0, L0)[:2 * n - 1].real
    c0 = np.maximum(c0, 0.0)
    cur_M = float(c0.max())
    if not np.isfinite(cur_M) or cur_M <= 0:
        return _normalize(f), _score_fft(f)

    best_f = f.copy()
    best_score = float(2.0 * n * cur_M / (S0 * S0))  # = 2n*cur_M/(2n) = cur_M
    # Actually score = 2n * max(conv) / (sum)^2 = 2n*cur_M/(S0^2) = 2n*cur_M/(2n) = cur_M
    # So score = cur_M when sum = sqrt(2n). Nice simplification!

    # Initial constraint set: top indices where c0 is large
    max_rows = min(2 * n - 1, max(200, 3 * n))
    top_idx = np.argsort(c0)[::-1]
    rows = [int(x) for x in top_idx[:min(100, 2 * n - 1)]]
    row_set = set(rows)

    t_start = time.monotonic()
    t_end = t_start + budget_s
    it = 0
    stall_count = 0

    while time.monotonic() < t_end:
        it += 1
        rem = t_end - time.monotonic()
        if rem < 10.0:
            break

        # Build constraint matrix for current rows
        rows_arr = np.asarray(rows, dtype=int)
        nr = len(rows_arr)
        A = np.zeros((nr, n), dtype=np.float64)
        for ii in range(nr):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                # A[ii, j] = f[k - j] for j in [lo, hi]
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b_vec = np.full(nr, cur_M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)

        # Time budget for LP solve
        tl = max(3.0, min(60.0, rem * 0.3))
        res = None
        try:
            res = _sp_opt.linprog(
                c_obj, A_ub=A, b_ub=b_vec,
                bounds=[(0.0, None)] * n,
                method='highs',
                options={'time_limit': tl, 'presolve': True}
            )
        except Exception:
            try:
                res = _sp_opt.linprog(
                    c_obj, A_ub=A, b_ub=b_vec,
                    bounds=[(0.0, None)] * n,
                    method='highs'
                )
            except Exception:
                break

        if res is None or not getattr(res, 'success', False) or res.x is None:
            break

        g = np.clip(np.asarray(res.x, dtype=np.float64).ravel(), 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg < 1e-12:
            break

        # Check full constraint violations
        if n <= 512:
            conv = np.convolve(f, g, mode='full')
        else:
            Lc = _next_pow2(2 * n - 1)
            Pf = np.fft.rfft(f, Lc)
            Pg = np.fft.rfft(g, Lc)
            conv = np.fft.irfft(Pf * Pg, Lc)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        tol = 1e-9 + 1e-9 * cur_M
        violations = conv - cur_M
        max_viol = float(np.max(violations)) if violations.size > 0 else 0.0

        if max_viol > tol:
            # Add violated constraints
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    if added >= max(64, (max_rows - len(rows)) // 2):
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            if max_viol > 100.0 * tol:
                # Too many violations, can't fix with cutting planes alone
                break
            else:
                continue  # Re-solve with new constraints

        # LP solution is feasible for all constraints. Use as direction.
        g_norm = g * (S0 / sg)

        # Line search for step size
        best_step = None
        best_sc = best_score
        for step in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g_norm
            fn = np.clip(fn, 0.0, None)
            sfn = float(fn.sum())
            if sfn < 1e-12:
                continue
            fn = fn * (S0 / sfn)
            # Score
            if n <= 512:
                cn = np.convolve(fn, fn, mode='full')
            else:
                Ln = _next_pow2(2 * n - 1)
                Pn = np.fft.rfft(fn, Ln)
                cn = np.fft.irfft(Pn * Pn, Ln)[:2 * n - 1].real
            cn = np.maximum(cn, 0.0)
            mc = float(cn.max())
            sc = float(2.0 * n * mc / (S0 * S0))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_step = fn.copy()

        if best_step is None:
            break

        improved = best_sc < best_score * (1.0 - 1e-10)
        f = best_step
        cur_M = float(2.0 * n * best_sc / (2.0 * n))  # = best_sc
        # Recompute cur_M properly
        if n <= 512:
            cnew = np.convolve(f, f, mode='full')
        else:
            Lnew = _next_pow2(2 * n - 1)
            Pnew = np.fft.rfft(f, Lnew)
            cnew = np.fft.irfft(Pnew * Pnew, Lnew)[:2 * n - 1].real
        cnew = np.maximum(cnew, 0.0)
        cur_M = float(cnew.max())
        best_score = float(2.0 * n * cur_M / (S0 * S0))
        best_f = f.copy()

        if not improved:
            stall_count += 1
            if stall_count > 10:
                break
        else:
            stall_count = 0
            if verbose and it % 5 == 0:
                print("  LP iter %d: score=%.8f, rows=%d" % (it, best_score, len(rows)))

    final_score = _score_fft(best_f)
    if final_score < best_score:
        best_score = float(final_score)
    return _normalize(best_f), best_score


def _coordinate_polish(p, budget_s, rng):
    """Local coordinate search: try small multiplicative perturbations of individual elements."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    p = _normalize(p)
    cur_score = _score_fft(p)
    t_start = time.monotonic()
    t_end = t_start + budget_s

    factors = [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]

    improved = True
    while improved and time.monotonic() < t_end:
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for j in order:
            if time.monotonic() >= t_end:
                break
            if p[j] < 1e-15:
                continue
            for fac in factors:
                p_new = p.copy()
                p_new[j] *= fac
                p_new = np.clip(p_new, 0.0, None)
                s = float(p_new.sum())
                if s < 1e-12:
                    continue
                p_norm = p_new / s
                sc = _score_fft(p_norm)
                if sc < cur_score * (1.0 - 1e-12):
                    p = p_norm
                    cur_score = sc
                    improved = True
                    break

    return p, cur_score

        v = float(v)
        return v if np.isfinite(v) else np.inf
    except Exception:
        return np.inf


def _main_search(seed, total_budget):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, total_budget - 20.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def _record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = _score_fft(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
        n = a.size
        old = best_by_n.get(n)
        if old is None or score < old[0]:
            best_by_n[n] = (float(score), a.copy())

    # --- Phase 0: Load previous best if available ---
    h1 = globals().get('height_sequence_1', None)
    if h1 is not None:
        try:
            arr = np.asarray(h1, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0 and arr.sum() > 0.01:
                _record(arr)
                print("Loaded height_sequence_1: n=%d, score=%.8f" % (arr.size, best_score))
        except Exception:
            pass

    # --- Phase 1: Baseline initialization ---
    lengths = [128, 192, 256, 320, 384, 448, 512, 640, 768, 1024]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        for kind in ['greedy', 'p45', 'p55', 'mix50']:
            if kind == 'greedy':
                p = _greedy_init(n)
            elif kind == 'p45':
                p = _power_init(n, 0.45)
            elif kind == 'p55':
                p = _power_init(n, 0.55)
            else:
                p = _mix_init(n, 0.5)
            _record(p)

    # Also try some random starts at key lengths
    for n in [256, 384, 512, 768]:
        for alpha in [0.3, 0.5, 1.0, 2.0]:
            if time.monotonic() > deadline:
                break
            _record(_dirichlet_init(n, alpha, rng))

    print("Phase 1 done. Best: score=%.8f, n=%d" % (best_score, len(best_p) if best_p is not None else 0))

    # --- Phase 2: Adam optimization ---
    # Select top candidates by score
    items = []
    for n, (sc, p) in best_by_n.items():
        items.append((sc, n, p))
    items.sort(key=lambda x: x[0])

    # Deduplicate: keep only the best at each length, take top 8
    seen_n = set()
    top_candidates = []
    for sc, n, p in items:
        if n not in seen_n:
            seen_n.add(n)
            top_candidates.append((sc, n, p))
        if len(top_candidates) >= 8:
            break

    # Allocate time for Adam: ~35% of remaining
    adam_budget = min(500.0, max(60.0, 0.35 * (deadline - time.monotonic())))
    per_adam = adam_budget / max(1, len(top_candidates))

    for sc, n, p in top_candidates:
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break
        alloc = min(per_adam, rem - 20.0)
        if alloc < 15.0:
            break
        p_out, f_out = _adam_optimize(p, alloc, seed=seed + n)
        _record(p_out, f_out)

        # Also try a perturbed version
        if rem > 60.0:
            p_pert = _perturb(p_out, rng, 0.08)
            alloc2 = min(per_adam * 0.6, rem - 20.0)
            if alloc2 > 15.0:
                p_out2, f_out2 = _adam_optimize(p_pert, alloc2, seed=seed + n + 7)
                _record(p_out2, f_out2)

    print("Phase 2 done. Best: score=%.8f, n=%d" % (best_score, len(best_p) if best_p is not None else 0))

    # --- Phase 3: LP cutting-plane optimization ---
    if _HAS_SCIPY:
        # Select candidates for LP: best at each distinct length, plus top overall
        lp_items = []
        for n, (sc, p) in best_by_n.items():
            if n <= 2048:  # Cap LP size
                lp_items.append((sc, n, p))
        lp_items.sort(key=lambda x: x[0])

        # Take top 5-6 for LP
        lp_selected = lp_items[:6]

        lp_phase_budget = deadline - time.monotonic() - 30.0
        if lp_phase_budget < 120.0:
            lp_phase_budget = max(60.0, (deadline - time.monotonic()) * 0.7)

        per_lp = lp_phase_budget / max(1, len(lp_selected))

        for idx, (sc, n, p) in enumerate(lp_selected):
            rem = deadline - time.monotonic()
            if rem < 80.0:
                break
            alloc = min(per_lp, rem - 60.0)
            if alloc < 50.0:
                alloc = max(30.0, rem - 30.0)
            if alloc < 25.0:
                break
            print("LP phase: candidate %d/%d, n=%d, current_score=%.8f, alloc=%.0fs" %
                  (idx + 1, len(lp_selected), n, sc, alloc))
            p_out, f_out = _lp_improve(p, alloc, verbose=True)
            _record(p_out, f_out)
            print("  LP result: score=%.8f" % f_out)

    else:
        # Fallback: more Adam
        while deadline - time.monotonic() > 120.0:
            if best_p is not None:
                base = _perturb(best_p, rng, 0.06)
            else:
                base = _greedy_init(384)
            alloc = min(200.0, max(30.0, deadline - time.monotonic() - 30.0))
            p_out, f_out = _adam_optimize(base, alloc, seed=int(time.monotonic() * 1000) % 100000)
            _record(p_out, f_out)

    print("Phase 3 done. Best: score=%.8f, n=%d" % (best_score, len(best_p) if best_p is not None else 0))

    # --- Phase 4: Final polishing ---
    if best_p is not None:
        rem = deadline - time.monotonic()

        if rem > 200.0:
            # Coordinate polish
            p_pol, sc_pol = _coordinate_polish(best_p, min(120.0, rem - 60.0), rng)
            _record(p_pol, sc_pol)
            print("Coord polish: score=%.8f" % sc_pol)

        rem = deadline - time.monotonic()
        if rem > 60.0:
            # Short Adam on best
            p_fin, f_fin = _adam_optimize(best_p, min(80.0, rem - 20.0), seed=seed + 99999)
            _record(p_fin, f_fin)

        rem = deadline - time.monotonic()
        if rem > 40.0 and _HAS_SCIPY and best_p.size <= 1024:
            # One more LP round
            p_lp, f_lp = _lp_improve(best_p, min(60.0, rem - 15.0))
            _record(p_lp, f_lp)

    # --- Phase 5: Validation ---
    if best_p is None:
        best_p = _greedy_init(384)

    final_seq = _scale_output(best_p, 10.0)
    gscore = _try_eval(final_seq)
    print("Validation: evaluate_sequence gives %.8f" % gscore)

    if not np.isfinite(gscore):
        # Fallback: try with different scaling
        final_seq = _scale_output(best_p, 1.0)
        gscore = _try_eval(final_seq)
        print("Retry with sum=1: %.8f" % gscore)

    if not np.isfinite(gscore):
        final_seq = _scale_output(_greedy_init(256), 10.0)
        gscore = _try_eval(final_seq)
        print("Ultimate fallback: %.8f" % gscore)

    # Also verify a second candidate
    if len(best_by_n) > 1:
        other_n = list(best_by_n.keys())[1]
        other_p = best_by_n[other_n][1]
        other_seq = _scale_output(other_p, 10.0)
        other_score = _try_eval(other_seq)
        if np.isfinite(other_score) and other_score < gscore:
            final_seq = other_seq
            gscore = other_score

    total_time = time.monotonic() - t0
    print("Final: score=%.8f, n=%d, total_time=%.1fs" % (gscore, len(final_seq), total_time))

    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation."""
    try:
        return _main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Fatal error in search: %s" % str(e))
        import traceback
        traceback.print_exc()
        # Fallback
        p = _greedy_init(384)
        seq = _scale_output(p, 10.0)
        return seq
