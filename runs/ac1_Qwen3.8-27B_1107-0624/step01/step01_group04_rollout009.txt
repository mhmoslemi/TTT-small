import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except:
    _scipy_opt = None

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(max(a.size, 16), dtype=np.float64)
        s = float(a.sum())
    return a / s

def score_fft(p):
    """Compute the score using FFT for speed."""
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-6:
        return np.inf
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))

def convolve_fft(a, b):
    """Compute a * b (full convolution) using FFT."""
    n = a.size
    m = b.size
    L = next_power_of_two(n + m - 1)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:n + m - 1].real

def greedy_normalized(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)

def lp_direction(f, n, M, max_rows, time_limit):
    """
    Solve the LP: maximize sum(g) subject to (f*g)[k] <= M for k in selected rows, g >= 0.
    Returns the optimal g vector.
    """
    # Build constraint matrix for selected rows
    # For row k: sum_i f[k-i] * g[i] <= M (where valid indices are used)
    # This is: sum_i A[k,i] * g[i] <= M
    # where A[k,i] = f[k-i] if 0 <= k-i <= n-1, else 0
    
    rows = np.arange(0, 2*n-1, max(1, (2*n-1)//max_rows))
    # Add the row with max (f*f)
    max_row = int(np.argmax(convolve_fft(f, f)))
    if max_row not in rows:
        rows = np.append(rows, max_row)
    
    rows = np.sort(np.unique(rows))
    R = rows.size
    
    A = np.zeros((R, n), dtype=np.float64)
    for ii in range(R):
        k = int(rows[ii])
        # (f*g)[k] = sum_i f[k-i] * g[i] for valid i
        # g[i] is valid for i in [0, n-1]
        # f[k-i] is valid when 0 <= k-i <= n-1, i.e., k-(n-1) <= i <= k
        i_lo = max(0, k - (n-1))
        i_hi = min(n-1, k)
        if i_lo <= i_hi:
            A[ii, i_lo:i_hi+1] = f[k - i_hi : k - i_lo + 1][::-1]
    
    b = np.full(R, M, dtype=np.float64)
    cobj = -np.ones(n, dtype=np.float64)
    
    try:
        res = _scipy_opt.linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None),
                                  method='highs', options={'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.clip(res.x, 0.0, None)
    except:
        pass
    return None

def lp_iteration(f, max_rows, time_per_lp):
    """
    One LP-based iteration: find direction g, then do line search.
    """
    n = f.size
    S0 = np.sqrt(2.0 * n)
    f_scaled = f * S0  # sum = S0 = sqrt(2n)
    
    ff = convolve_fft(f_scaled, f_scaled)
    ff = np.maximum(ff, 0.0)
    M = float(ff.max())
    
    if not np.isfinite(M) or M <= 0:
        return f, score_fft(f)
    
    g = lp_direction(f_scaled, n, M, max_rows, time_per_lp)
    if g is None or g.sum() <= 1e-12:
        return f, score_fft(f)
    
    g_scaled = g * (S0 / g.sum())  # sum = S0
    
    # Line search: try multiple t values
    best_t = 0.0
    best_score = score_fft(f)
    
    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
        fn = (1.0 - t) * f_scaled + t * g_scaled
        fn = np.clip(fn, 0.0, None)
        sn = fn.sum()
        if sn <= 1e-12:
            continue
        fn_norm = fn / sn  # normalize to sum = 1
        sc = score_fft(fn_norm)
        if sc < best_score:
            best_score = sc
            best_t = t
    
    if best_t == 0.0:
        return f, best_score
    
    f_new = ((1.0 - best_t) * f_scaled + best_t * g_scaled)
    f_new = normalize_nonneg(f_new)
    return f_new, best_score

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(max(a.size, 16), dtype=np.float64)
        s = float(a.sum())
    return a / s


def score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-6:
        return np.inf
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    maxc = float(np.max(np.maximum(c, 0.0)))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def conv_fft(a, b):

    L = next_power_of_two(n + b.size - 1)
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:n + b.size - 1].real

def gradient_polish(f, budget_s):
    """Simple projected gradient descent for polishing."""
    n = f.size
    p = normalize_nonneg(f).copy()
    L = next_power_of_two(2 * n - 1)
    
    best_p = p.copy()
    best_f = score_fft(p)
    
    lr = 0.1
    end_time = time.monotonic() + budget_s
    
    while time.monotonic() < end_time:
        # Compute gradient of max autocorrelation
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2*n-1].real
        c = np.maximum(c, 0.0)
        k_star = int(np.argmax(c))
        
        # Gradient of c[k_star] w.r.t. p[j] = p[k_star - j] + p[j - k_star] (with bounds)
        grad = np.zeros(n, dtype=np.float64)
        for j in range(n):
            val = 0.0
            idx1 = k_star - j
            if 0 <= idx1 < n:
                val += p[idx1]
            idx2 = j - k_star
            if 0 <= idx2 < n:
                val += p[idx2]
            grad[j] = val
        
        # Projected gradient step
        p_new = p - lr * grad / n
        # Project to simplex (non-negative, sum=1)
        p_new = np.clip(p_new, 0, None)
        s = p_new.sum()
        if s > 1e-12:
            p_new = p_new / s
        else:
            p_new = p
        
        f_val = score_fft(p_new)
        if f_val < best_f:
            best_f = f_val
            best_p = p_new.copy()
        
        p = p_new
        lr *= 0.999  # Slow decay
    
    return best_p, best_f

LP-based optimization for step function score minimization.
Uses high-dimensional LP direction finding with adaptive constraint management
and multi-phase time allocation.

import time
import numpy as np

try:
    import scipy.optimize
    HAS_SCIPY = True
except:
    HAS_SCIPY = False

# === Utility functions ===

def npos2(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def norm_pos(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if not np.isfinite(s) or s < 1e-12:
        a = np.ones(max(a.size, 16))
    else:
        a = a / s
    return a

def sc(a):
    """Score computation via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n < 1:
        return 1e18
    s = a.sum()
    if s < 1e-8:
        return 1e18
    L = npos2(2*n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1].real
    c = np.maximum(c, 0.0)
    mc = c.max()
    if not np.isfinite(mc):
        return 1e18
    return 2.0 * n * mc / (s * s)

    """Full convolution via FFT."""
    L = npos2(a.size + b.size - 1)
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:a.size+b.size-1].real

        hi = min(n - 1, k)
        if lo <= hi:
            A[ii, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    return A

def solve_lp(f, n, M, rows, tlimit):
    if not HAS_SCIPY or len(rows) == 0:
        return None
    A = build_A(f, rows, n)
    b = np.full(len(rows), M)
    c = -np.ones(n)
    try:
        r = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0,None)]*n,
                                    method='highs', options={'time_limit': tlimit})
        if r.success and r.x is not None:
            return np.clip(r.x, 0, None)
    except:
        pass
    return None

def lp_phase(f0, budget):
    """Main LP optimization phase."""
    n = f0.size
    S0 = np.sqrt(2.0 * n)
    f = f0 * S0
    best = f0.copy()
    best_sc = sc(f0)
    
    ff = np.maximum(cconv(f, f), 0.0)
    M = ff.max()
    
    # Initial rows
    kt = 2*n - 1
    nr = min(1200, max(300, 3*n))
    top = np.argsort(ff)[::-1][:nr//2]
    ev = np.linspace(0, kt-1, nr-nr//2, dtype=int)
    rows = np.sort(np.unique(np.concatenate([top, ev])))
    rset = set(rows.tolist())
    
    end = time.monotonic() + budget
    it = 0
    stall = 0
    
    while time.monotonic() < end - 5:
        it += 1
        rem = end - time.monotonic()
        if rem < 8:
            break
        
        g = solve_lp(f, n, M, rows.tolist(), min(20, rem*0.25))
        if g is None or g.sum() < 1e-10:
            break
        
        gs = g * (S0 / g.sum())
        fg = cconv(f, gs)
        gg = np.maximum(cconv(gs, gs), 0.0)
        
        # Vectorized line search
        best_t = 0.0
        cur_sc = 2.0 * n * M / (S0 * S0)  # = 2n*max(ff)/(sum f)^2
        ts = [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]
        
        for t in ts:
            c_new = (1-t)**2 * ff + 2*t*(1-t) * fg + t**2 * gg
            c_new = np.maximum(c_new, 0.0)
            mc = c_new.max()
            s_new = ((1-t)*S0 + t*S0)  # = S0 always since both sum to S0
            # Wait, fn = (1-t)*f + t*gs, sum = (1-t)*S0 + t*S0 = S0
            s_new = S0
            sc_new = 2.0 * n * mc / (s_new * s_new)
            if sc_new < cur_sc:
                cur_sc = sc_new
                best_t = t
        
        if best_t > 0:
            f = (1 - best_t) * f + best_t * gs
            f = np.clip(f, 0, None)
            f = f * (S0 / f.sum())
            ff = np.maximum(cconv(f, f), 0.0)
            M = ff.max()
            sc_now = sc(f / S0)
            stall = 0
            if sc_now < best_sc:
                best_sc = sc_now
                best = (f / S0).copy()
                if it % 20 == 0:
                    print(f"  LP it={it} sc={best_sc:.8f} t={best_t:.3f} M={M:.4f}")
        else:
            stall += 1
            # Add violated constraints
            viol = fg - M
            if stall <= 5:
                mask = viol > 1e-6 * (1 + M)
                if mask.any() and len(rows) < min(2000, 4*n):
                    idx = np.where(mask)[0]
                    order = np.argsort(viol[idx])[::-1]
                    for ix in order[:min(200, min(2000,4*n) - len(rows))]:
                        ix = int(ix)
                        if ix not in rset:
                            rows = np.append(rows, ix)
                            rset.add(ix)
            else:
                break
        
        # Periodic constraint refresh
        if it % 15 == 0 and len(rows) > 200:
            thr = np.percentile(ff, 15)
            keep = [int(x) for x in rows if ff[int(x)] >= thr]
            top_new = np.argsort(ff)[::-1][:max(100, len(rows)//4)]
            for ix in top_new:
                if int(ix) not in set(keep):
                    keep.append(int(ix))
            rows = np.sort(np.array(keep[:min(2000, 4*n)]))
            rset = set(rows.tolist())
    
    return best, best_sc
