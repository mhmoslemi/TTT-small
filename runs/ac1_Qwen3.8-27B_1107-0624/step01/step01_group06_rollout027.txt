import time

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2*n-1]

def compute_J(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S*S)

def smooth_grad(a, tau):
    """Return (smoothed_max, gradient_of_smoothed_max_wrt_a)"""
    n = len(a)
    size = 1
    while size < 3*n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2*n-1]
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    # cross[i] = sum_k p[k] * a[k-i]
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    return Tmax, 2.0 * cross  # dTmax/da_i = 2*cross[i]

    """Compute J and gradient."""
    n = len(a)
    S = np.sum(a)

    return J, grad

def _fft_conv(a):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]


def _J(a):
    """Compute J = 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = _fft_conv(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def _smooth_grad_linear(a, tau):
    """Smoothed J and gradient w.r.t. a (linear parametrization)."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:

def _subgrad_phase(a_init, duration, seed):
    """Sub-gradient method using exact max for final polishing."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + duration
    start = time.monotonic()

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * n / S

    best_a = a.copy()
    best_J = _J(a)
    lr = 1.0
    step = 0
    stagnation = 0

    while time.monotonic() < deadline:
        c = _fft_conv(a)
        max_c = np.max(c)
        S = np.sum(a)
        J = 2.0 * n * max_c / (S * S)

        if J < best_J - 1e-10:
            best_J = J
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation % 100 == 0:
                lr = max(0.05, lr * 0.95)

        step += 1

        # Get sub-gradient from top few max coefficients
        tol = 1e-6 * max_c
        max_idx = np.where(c >= max_c - tol)[0]
        # Limit to at most 10 indices for efficiency
        max_idx = max_idx[:10]

        dmax = np.zeros(n)
        count = len(max_idx)
        for k in max_idx:
            # dc_k/da_i = 2*a[k-i] for valid i
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # a[k-i] for i in [i_min, i_max]
            # = a[k], a[k-1], ..., a[k-i_max]
            # indices into a: k, k-1, ..., k-i_max
            a_idx = np.arange(k, k - (i_max - i_min + 1), -1)
            a_idx = a_idx[(a_idx >= 0) & (a_idx < n)]
            valid_i = np.arange(i_min, i_max + 1)
            valid_i = valid_i[(a_idx >= 0) & (a_idx < n)]
            dmax[valid_i] += 2.0 * a[a_idx] / count

        # Gradient of J
        grad = (2.0 * n * dmax) / (S * S) - (4.0 * n * max_c) / (S**3)

        # Step
        scale = lr / (n * (np.max(np.abs(grad)) + 1e-12))
        a = a - scale * grad * n
        a = np.maximum(a, 0.0)
        S = np.sum(a)
        if S > 0:
            a = a * n / S

        # Occasional perturbation
        if step % 300 == 0:
            cand = best_a.copy()
            w = max(5, n // 80)
            s = int(rng.integers(0, n - w))
            cand[s:s+w] *= np.exp(rng.normal(0, 0.05, w))
            cand = np.maximum(cand, 0)
            Sc = np.sum(cand)
            if Sc > 0:
                cand = cand * n / Sc
            cJ = _J(cand)
            if cJ < best_J:
                best_J = cJ
                best_a = cand.copy()
                a = cand.copy()
                lr = 1.0
                stagnation = 0

    S = np.sum(best_a)
    if S > 0:
        best_a = best_a * n / S
    return best_a, _J(best_a)

        best_global_J = v1

    # Verify with evaluator
    eval1 = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(eval1):
        print(f"  Evaluated: {eval1:.8f}")
        if eval1 < best_global_J:
            best_global_J = eval1

    # === Phase 2: Linear-space Adam (25% of time) ===
    remaining = total_deadline - time.monotonic()
    t2 = remaining * 0.50
    print(f"\n--- Phase 2: Linear-space Adam ({t2:.0f}s) ---")
    a2, J2 = _adam_linear_phase(best_global, t2, seed=seed+100, tau_start=0.5, tau_end=0.01, lr_start=0.008)
    v2 = _J(a2)
    print(f"  Phase 2 J: {v2:.8f} (improvement: {best_global_J - v2:.8f})")
    if v2 < best_global_J:
        best_global = a2.copy()
        best_global_J = v2

    eval2 = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(eval2):
        print(f"  Evaluated: {eval2:.8f}")
        if eval2 < best_global_J:
            best_global_J = eval2

    # === Phase 3: Sub-gradient polish (15% of time) ===
    remaining = total_deadline - time.monotonic()
    t3 = remaining * 0.40
    print(f"\n--- Phase 3: Sub-gradient polish ({t3:.0f}s) ---")
    a3, J3 = _subgrad_phase(best_global, t3, seed=seed+200)
    v3 = _J(a3)
    print(f"  Phase 3 J: {v3:.8f} (improvement: {best_global_J - v3:.8f})")
    if v3 < best_global_J:
        best_global = a3.copy()
        best_global_J = v3

    # === Phase 4: Local search (remaining time minus safety margin) ===
    remaining = total_deadline - time.monotonic() - 10
    if remaining > 20:
        t4 = remaining * 0.7
        print(f"\n--- Phase 4: Local search ({t4:.0f}s) ---")
        a4, J4 = _local_search(best_global, t4, seed=seed+300)
        v4 = _J(a4)
        print(f"  Phase 4 J: {v4:.8f} (improvement: {best_global_J - v4:.8f})")
        if v4 < best_global_J:
            best_global = a4.copy()
            best_global_J = v4

    # === Phase 5: Final fine optimization ===
    remaining = total_deadline - time.monotonic() - 5
    if remaining > 15:
        t5 = remaining
        print(f"\n--- Phase 5: Final refinement ({t5:.0f}s) ---")
        # Quick log-space refinement
        a5_safe = np.maximum(best_global, 1e-10)
        x5 = np.log(a5_safe)
        a5, J5 = _adam_log_phase(x5, t5 * 0.6, seed=seed+400, tau_start=0.1, tau_end=0.005, lr_start=0.005)
        v5 = _J(a5)
        if v5 < best_global_J:
            best_global = a5.copy()
            best_global_J = v5
            print(f"  Phase 5 (log) J: {v5:.8f}")

        # Sub-gradient on the result
        rem = total_deadline - time.monotonic() - 3
        if rem > 10:
            a6, J6 = _subgrad_phase(best_global, rem, seed=seed+500)
            v6 = _J(a6)
            if v6 < best_global_J:
                best_global = a6.copy()
                best_global_J = v6
                print(f"  Phase 5 (subgrad) J: {v6:.8f}")

    # Final verification
    best_global = np.clip(best_global, 0.0, 1000.0)
    final_eval = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_eval):
        print(f"\nFinal evaluated score: {final_eval:.8f}")
    else:
        print(f"\nWARNING: evaluate returned non-finite: {final_eval}")
        # Fallback to parent
        best_global = parent.copy()
        final_eval = float(evaluate_sequence(best_global.tolist()))
        print(f"  Fallback parent score: {final_eval:.8f}")

    return [float(x) for x in best_global]
