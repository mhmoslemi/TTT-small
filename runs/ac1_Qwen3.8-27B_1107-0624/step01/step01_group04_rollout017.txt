"""
Hybrid optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses L-BFGS-B with soft-max temperature annealing and LP cutting-plane refinement.
Focuses on lengths around the previous best (384) with multiple restarts.
"""
import time
import numpy as np
from scipy.optimize import minimize, linprog


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def conv_fft(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    L = next_power_of_two(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def obj_val(a):
    n = len(a)
    if n == 0:
        return np.inf
    c = conv_fft(a)
    return 2.0 * n * float(c.max())


def soft_max_loss_grad(x, alpha):
    """Compute soft-max loss and gradient in x-space where a = softmax(x)."""
    n = len(x)
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    sx = exp_x.sum()
    if not np.isfinite(sx) or sx <= 0:
        return 1e10, np.zeros(n)
    a = exp_x / sx

    c = conv_fft(a)
    maxc = float(c.max())
    if maxc <= 0 or not np.isfinite(maxc):
        return 0.0, np.zeros(n)

    # Soft-max: L = 2n * (1/alpha) * log(sum(exp(alpha * c)))
    z = alpha * c
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    se = exp_z.sum()
    if not np.isfinite(se) or se <= 0:
        return 1e10, np.zeros(n)
    loss = 2.0 * n * (z.max() + np.log(se)) / alpha

    # Weights for soft-max
    w = exp_z / se

    # Cross-correlation: g_m = sum_j a[j] * w[m+j] for m in [0, n-1]
    L = next_power_of_two(3 * n - 2)
    w_padded = np.zeros(L)
    w_padded[:len(w)] = w
    a_padded = np.zeros(L)
    a_padded[:n] = a
    Wf = np.fft.rfft(w_padded)
    Af = np.fft.rfft(a_padded)
    corr = np.fft.irfft(Wf * np.conj(Af), L)
    g = corr[:n]
    g = np.maximum(g, 0.0)

    # Gradient in a-space: G_i = 4*n*g_i
    G = 4.0 * n * g

    # Gradient in x-space: grad_x_j = a_j * (G_j - sum_i G_i * a_i)
    gp = float(np.dot(G, a))
    grad_x = a * (G - gp)

    if not np.all(np.isfinite(grad_x)):
        grad_x = np.zeros(n)

    return float(loss), grad_x


def lbfgs_anneal(a, budget_s, seed=0):
    """Run L-BFGS-B with temperature annealing starting from normalized a."""
    n = len(a)
    if budget_s < 5.0:
        return a, obj_val(a)

    rng = np.random.default_rng(seed)
    x = np.log(a + 1e-300)
    x -= x.mean()

    best_a = a.copy()
    best_score = obj_val(a)

    # Temperature schedule: start smooth, end sharp
    alphas = [3.0, 5.0, 8.0, 15.0, 30.0, 60.0, 120.0, 250.0]
    # Time allocation: spend more time at higher alphas
    total_alpha_time = float(budget_s - 10.0)
    weights = [0.05, 0.08, 0.10, 0.12, 0.15, 0.18, 0.20, 0.12]
    wsum = sum(weights)
    weights = [w / wsum for w in weights]

    start_time = time.monotonic()
    for idx, alpha in enumerate(alphas):
        alloc = weights[idx] * total_alpha_time
        end_t = start_time + alloc + (time.monotonic() - start_time)
        if time.monotonic() >= end_t or time.monotonic() >= start_time + total_alpha_time - 5:
            break

        def fun_and_grad(xv):
            return soft_max_loss_grad(xv, alpha)

        opts = {
            'maxiter': max(100, int(alloc * 30)),
            'maxcor': 20,
            'ftol': 1e-14,
            'gtol': 1e-8,
            'maxls': 40,
            'disp': False,
        }

        try:
            res = minimize(fun_and_grad, x, jac=True, method='L-BFGS-B', options=opts)
            if res.success or res.fun < 1e10:
                x_shifted = res.x - res.x.max()
                exp_x = np.exp(x_shifted)
                sx = exp_x.sum()
                if sx > 0 and np.isfinite(sx):
                    new_a = exp_x / sx
                    sc = obj_val(new_a)
                    if sc < best_score:
                        best_score = sc
                        best_a = new_a.copy()
                x = res.x
        except Exception:
            pass

        # Check time
        if time.monotonic() >= start_time + total_alpha_time - 10:
            break

        # Small perturbation to escape plateaus
        x += rng.normal(0, 0.01, n)
        x -= x.mean()

    # Final evaluation
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    sx = exp_x.sum()
    if sx > 0 and np.isfinite(sx):
        final_a = exp_x / sx
        sc = obj_val(final_a)
        if sc < best_score:
            best_score = sc
            best_a = final_a.copy()

    return best_a, best_score


def lp_cutting_plane(a, budget_s, seed=0):
    """LP cutting-plane refinement."""
    n = len(a)
    if n < 4 or budget_s < 30.0:
        return a, obj_val(a)

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(a) * S0
    cur_score = obj_val(f / S0)  # = 2n*max(f*f)/(2n) = max(f*f) ... wait
    # Actually obj_val expects sum=1. f sums to S0.
    # Conv of f: f*f has max = S0^2 * max((f/S0)*(f/S0)) = 2n * max(normalized_conv)
    # So max(f*f) = 2n * max(normalized_conv)
    # And the score = 2n*max(a*a)/(sum(a))^2 = max(f*f)/(sum(f/S0))^2 * ... 
    # Let me just compute directly.
    
    # Score in terms of f (sum = S0):
    # score = 2*n*max(f*f) / (sum(f))^2 = 2*n*max(f*f) / (2*n) = max(f*f)
    # So the score equals max(f*f) when f sums to sqrt(2n)!
    
    c_f = conv_fft(f)
    M = float(c_f.max())
    cur_score = M  # = 2n*max(normalized_conv)
    
    best_f = f.copy()
    best_score = cur_score

    end_time = time.monotonic() + max(0, budget_s - 5.0)

    # Initial constraint set: evenly spaced
    max_rows = min(2 * n - 1, 1500)
    initial_rows = np.linspace(0, 2 * n - 2, min(max_rows, 200), dtype=int).astype(int)
    rows = list(initial_rows)
    row_set = set(rows)

    it = 0
    while time.monotonic() < end_time:
        it += 1

        # Build constraint matrix
        rows_arr = np.asarray(rows, dtype=int)
        n_rows = len(rows_arr)
        A = np.zeros((n_rows, n), dtype=np.float64)
        for ii in range(n_rows):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(n_rows, M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        # Time limit for LP
        rem = max(1.0, end_time - time.monotonic() - 2.0)
        tl = max(2.0, min(60.0, rem * 0.2))

        res = None
        try:
            res = linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None),
                          method='highs', options={'time_limit': tl, 'presolve': True})
        except Exception:
            try:
                res = linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None),
                              method='highs')
            except Exception:
                break

        if res is None or not getattr(res, 'success', False) or res.x is None:
            break

        g = np.clip(res.x, 0.0, None)
        sg = float(g.sum())
        if sg <= 1e-12 or not np.isfinite(sg):
            break

        # Check full constraint violations via FFT
        L2 = next_power_of_two(2 * n - 1)
        Pf = np.fft.rfft(f, L2)
        Pg = np.fft.rfft(g, L2)
        conv_fg = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        conv_fg = np.maximum(conv_fg, 0.0)

        tol = 1e-6 + 1e-6 * M
        violations = conv_fg - M
        max_viol = float(violations.max())

        if max_viol > tol:
            # Add violated constraints
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    if added >= 200 or len(rows) >= max_rows:
                        break
                    ix = int(ix)
                    if violations[ix] > tol * 1.5 and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                if added == 0 and max_viol > 10.0 * tol:
                    break
                continue
            else:
                if max_viol > 10.0 * tol:
                    break

        # Rescale g to match normalization
        g = g * (S0 / sg)

        # Line search: find best step
        best_step_a = None
        best_step_score = best_score
        for step in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            cn = conv_fft(fn)
            sc = float(cn.max())
            if sc < best_step_score:
                best_step_score = sc
                best_step_a = fn.copy()

        if best_step_a is None or best_step_score >= best_score - 1e-10:
            break

        f = best_step_a
        cur_score = best_step_score
        best_f = f.copy()
        best_score = cur_score
        M = cur_score

        if it % 10 == 0:
            print("  LP: n=%d iter=%d score=%.8f rows=%d" % (n, it, best_score, len(rows)))

        # Convergence check
        if best_score < 1.45:
            break

    return normalize_nonneg(best_f), best_score


def random_perturb(a, rng, sigma):
    noise = np.exp(rng.normal(0.0, sigma, len(a)))
    return normalize_nonneg(a * noise)


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]


def try_evaluate(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
        val = float(val)
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed))

    best_a = None
    best_score = np.inf
    best_by_n = {}

    def record(a, score=None):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = normalize_nonneg(a)
        if score is None:
            score = obj_val(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_a = a.copy()
            best_score = score
        n = a.size
        if n not in best_by_n or score < best_by_n[n][0]:
            best_by_n[n] = (score, a.copy())

    # Stage 0: Get initial candidates
    print("Stage 0: Initial candidates")
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0 and arr.sum() > 1e-6:
                record(arr)
                print("  Loaded height_sequence_1: n=%d, score=%.8f" % (arr.size, best_score))
        except Exception:
            pass

    # Quick baselines at various lengths
    base_lengths = [256, 320, 352, 384, 400, 448, 512, 640, 768]
    for n in base_lengths:
        if time.monotonic() > deadline - 30:
            break
        record(greedy_normalized(n))
        record(power_normalized(n, 0.45))
        record(power_normalized(n, 0.55))
        record(power_normalized(n, 0.40))
        record(power_normalized(n, 0.60))

    print("  Best after baselines: score=%.8f" % best_score)

    # Stage 1: L-BFGS-B optimization at the best length
    print("Stage 1: L-BFGS-B annealing")
    # Priority: best length first, then nearby lengths
    priority_lengths = []
    if best_a is not None:
        bn = len(best_a)
        priority_lengths.append(bn)
    for n in [384, 368, 400, 352, 416, 336, 432, 320, 448, 480, 512, 256]:
        if n not in priority_lengths:
            priority_lengths.append(n)

    for n in priority_lengths[:8]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break

        # Get starting point for this length
        if n in best_by_n:
            start_a = best_by_n[n][1].copy()
        else:
            start_a = greedy_normalized(n)

        alloc = min(200.0, max(20.0, rem * 0.25))
        print("  L-BFGS-B n=%d, budget=%.0fs" % (n, alloc))
        a_out, sc_out = lbfgs_anneal(start_a, alloc, seed=seed + n)
        record(a_out, sc_out)
        print("    score=%.8f" % sc_out)

    print("  Best after L-BFGS-B: score=%.8f" % best_score)

    # Stage 2: LP cutting-plane refinement
    print("Stage 2: LP cutting-plane")
    lp_lengths = []
    scored_items = []
    for n, (sc, a) in best_by_n.items():
        scored_items.append((sc, n))
    scored_items.sort()
    for sc, n in scored_items[:6]:
        if n <= 2048:
            lp_lengths.append(n)
    # Ensure we try the best
    if best_a is not None:
        bn = len(best_a)
        if bn not in lp_lengths and bn <= 2048:
            lp_lengths.insert(0, bn)

    for n in lp_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        start_a = best_by_n.get(n, (np.inf, greedy_normalized(n)))[1]
        alloc = min(300.0, max(30.0, rem * 0.35))
        print("  LP n=%d, budget=%.0fs" % (n, alloc))
        a_out, sc_out = lp_cutting_plane(start_a, alloc, seed=seed + n)
        record(a_out, sc_out)
        print("    score=%.8f" % sc_out)

    print("  Best after LP: score=%.8f" % best_score)

    # Stage 3: Another round of L-BFGS-B on LP results
    print("Stage 3: Post-LP L-BFGS-B polish")
    for n in lp_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        if n in best_by_n:
            start_a = best_by_n[n][1].copy()
        else:
            continue
        alloc = min(150.0, max(15.0, rem * 0.25))
        a_out, sc_out = lbfgs_anneal(start_a, alloc, seed=seed + 1000 + n)
        record(a_out, sc_out)
        print("  n=%d post-polish score=%.8f" % (n, sc_out))

    # Stage 4: Random restarts if time permits
    print("Stage 4: Random restarts")
    n_restarts = 0
    while time.monotonic() < deadline - 60 and n_restarts < 6:
        n_rem = deadline - time.monotonic()
        if n_rem < 60:
            break
        # Pick a length near the best
        if best_a is not None:
            bn = len(best_a)
            n_try = bn + int(rng.choice([-48, -32, -16, 0, 16, 32, 48]))
            n_try = max(64, min(2048, n_try))
        else:
            n_try = 384

        start_a = greedy_normalized(n_try)
        start_a = random_perturb(start_a, rng, 0.3)

        alloc = min(120.0, max(20.0, (deadline - time.monotonic() - 30) * 0.3))
        a_out, sc_out = lbfgs_anneal(start_a, alloc, seed=seed + 5000 + n_restarts)
        record(a_out, sc_out)
        n_restarts += 1
        print("  Restart %d: n=%d score=%.8f" % (n_restarts, n_try, sc_out))

        # If we found something good, do LP on it
        if sc_out < 1.51 and deadline - time.monotonic() > 120:
            a_lp, sc_lp = lp_cutting_plane(a_out, min(200.0, deadline - time.monotonic() - 60), seed=seed + 9000 + n_restarts)
            record(a_lp, sc_lp)
            print("  LP after restart: score=%.8f" % sc_lp)

    # Final: validate with evaluate_sequence
    print("Final: validation")
    if best_a is not None:
        # Try the best and a few alternatives at different scales
        candidates = [best_a]
        items = sorted([(v[0], k) for k, v in best_by_n.items()])
        for sc, n in items[:3]:
            if n != len(best_a):
                candidates.append(best_by_n[n][1])

        best_final_score = best_score
        best_final_a = best_a
        for cand in candidates:
            if deadline - time.monotonic() < 10:
                break
            for scale in [5.0, 10.0, 20.0]:
                seq = scale_output(cand, scale)
                sc = try_evaluate(seq)
                if np.isfinite(sc) and sc < best_final_score:
                    best_final_score = sc
                    best_final_a = cand.copy()
                    print("  Found better via evaluator: %.8f" % sc)

        best_a = best_final_a
        best_score = best_final_score

    if best_a is None:
        best_a = greedy_normalized(384)
        best_score = obj_val(best_a)

    print("FINAL: n=%d, score=%.8f" % (len(best_a), best_score))
    return scale_output(best_a, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the objective."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main_search: %s" % str(e))
        import traceback
        traceback.print_exc()
        # Fallback: use previous best if available
        h = globals().get('height_sequence_1', None)
        if h is not None:
            try:
                arr = np.asarray(h, dtype=np.float64).ravel()
                arr = np.clip(arr, 0.0, 1000.0)
                if arr.size > 0 and arr.sum() > 0.01:
                    return [float(x) for x in arr]
            except Exception:
                pass
        a = greedy_normalized(384)
        return scale_output(a, 10.0)
