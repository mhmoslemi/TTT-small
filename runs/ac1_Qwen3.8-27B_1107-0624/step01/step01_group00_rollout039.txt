```python
import time
import math
import numpy as np

def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    # Sort in descending order
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)

def _conv_sq(a):
    """Compute linear convolution a*a via FFT."""
    n = a.size
    m = 1
    while m < 2*n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def _objective(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_sq(a)
    return 2.0 * n * c.max() / (s * s)

def _smoothed_grad(c, a, L, tau):
    """Gradient of smoothed max(c) using softmax over top-L."""
    n = a.size
    L = min(L, n)
    # Get top L indices
    if L == n:
        order = np.arange(n)
    else:
        order = np.argpartition(-c, L-1)[:L]
    wc = c[order]
    w = np.exp(wc / max(tau, 1e-15))
    w = w / w.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    for k, wk in zip(order, w):
        if wk < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        src = k - idx
        grad[idx] += 2.0 * wk * a[src]
    return grad
```

```python
def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size

        return np.zeros(n)

    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
```

```python
import numpy as np
from scipy.optimize import minimize

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(z), 1.0/len(z))
    return e / s

def _fft_conv_sq(a):
    n = a.size
    m = 1
    while m < 2*n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    cond = sv * np.arange(1, n+1) > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)

def _obj_and_grad(z, tau):
    """Return (objective, gradient) for L-BFGS."""
    a = _softmax(z)
    c = _fft_conv_sq(a)
    n = a.size
    m = c.max()
    # Smoothed max value
    val = m + tau * np.log(np.sum(np.exp((c - m) / tau)))
    obj = 2.0 * n * val
    
    # Gradient
    w = np.exp((c - m) / tau)
    w = w / w.sum()
    C_w = np.dot(w, c)
    R = np.correlate(w, a, mode='valid')[:n]
    grad = 4.0 * n * a * ((1.0 - a) * R - C_w)
    
    return obj, grad

            method='L-BFGS-B',
            options={'maxiter': max_iter_per_level, 'ftol': 1e-12, 'gtol': 1e-10, 'maxls': 50}
        )
        
        if result.fun < best_f:
            best_x = result.x.copy()
            best_f = result.fun
        
        z = best_x
        a = _softmax(z)
        score = _eval_score(a)
        print(f"  Level {level}, tau={tau:.6f}, score={score:.8f}, lbfgs_fun={result.fun:.8f}")
        
        tau *= 0.4
        
        if time.monotonic() > t_end - 60:

    return z, _softmax(z)

    """Coordinate descent polish."""
    n = len(a)
    a = a.copy()
```

```python
"""
Minimize obj = 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with softmax parameterization
and continuation method, combined with targeted coordinate descent polish.

Key ideas:
- Parameterize a = softmax(z) to handle simplex constraint smoothly
- Use log-sum-exp smoothing of max with decreasing temperature (continuation)
- Derive clean gradient: g_i = 4n*a_i*[(1-a_i)*R_i - C_w] where R_i = sum_k w_k*a[k-i]
- Polish with targeted coordinate descent focused on peak-reducing transfers
- Multiple seeds and strategies to avoid local minima
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# Global variables for L-BFGS function/gradient (to avoid closures/lambdas)
_current_tau = 1.0
_current_L = None


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(z.size, 1.0 / z.size)
    return e / s


def _fft_conv_sq(a):
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    cond = sv * np.arange(1, n + 1) > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _obj_grad_impl(z, tau, L):
    """Compute smoothed objective and gradient in z-space."""
    a = _softmax(z)
    c = _fft_conv_sq(a)
    n = a.size
    total_len = 2 * n - 1

    if L is not None and L < total_len:
        top_idx = np.argpartition(-c, L - 1)[:L]
    else:
        top_idx = np.arange(total_len)

    c_top = c[top_idx]
    m = c_top.max()
    w_top = np.exp((c_top - m) / tau)
    w_top = w_top / w_top.sum()

    val = m + tau * np.log(np.sum(np.exp((c_top - m) / tau)))
    obj = 2.0 * n * val

    w = np.zeros(total_len)
    w[top_idx] = w_top

    C_w = np.dot(w, c)
    R = np.correlate(w, a, mode='valid')[:n]
    grad = 4.0 * n * a * ((1.0 - a) * R - C_w)

    return obj, grad


def _func_z(z):
    return _obj_grad_impl(z, _current_tau, _current_L)[0]


def _grad_z(z):
    return _obj_grad_impl(z, _current_tau, _current_L)[1]


def _eval_score(a):
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float('inf')


def _quick_obj(a):
    """Fast objective (no security checks), assuming sum(a)=1, a>=0."""
    n = a.size
    c = _fft_conv_sq(a)
    return 2.0 * n * c.max()


def _lbfgs_one_level(z, tau, L, max_iter):
    """Run one L-BFGS level and return (z, best_fun)."""
    global _current_tau, _current_L
    _current_tau = tau
    _current_L = L

    try:
        result = minimize(
            _func_z, z, jac=_grad_z, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
        )
        if result.fun < _func_z(z):
            return result.x.copy(), result.fun
    except Exception:
        pass
    return z.copy(), _func_z(z)


def _lbfgs_continuation(z0, t_end, rng):
    """Full L-BFGS continuation from z0 until time budget."""
    z = z0.copy()
    n = z.size
    best_z = z.copy()
    best_score = _eval_score(_softmax(z))

    tau = 0.8
    L = None  # use all lags initially

    level = 0
    while time.monotonic() < t_end - 5:
        level += 1
        # Run this level
        z_new, fun = _lbfgs_one_level(z, tau, L, max_iter=300)
        a = _softmax(z_new)
        a = _proj_simplex(a, 1.0)
        score = _eval_score(a)

        if score < best_score:
            best_score = score
            best_z = z_new.copy()

        if level <= 5:
            print(f"    L{level}: tau={tau:.4f}, score={score:.8f}, fun={fun:.8f}")

        # Decrease tau and optionally restrict to top-L
        tau *= 0.45
        if level >= 4:
            L = max(20, int((2 * n - 1) * 0.3))
        if level >= 7:
            L = max(10, int((2 * n - 1) * 0.15))
        if level >= 10:
            L = max(5, int((2 * n - 1) * 0.08))

        z = z_new

        # Periodic jitter to escape local minima
        if level % 4 == 0:
            z_pert = z + rng.normal(0, 0.1 * tau, n)
            _current_tau = tau * 2
            _current_L = None
            z_pert_new, _ = _lbfgs_one_level(z_pert, tau * 2, None, max_iter=100)
            a_pert = _softmax(z_pert_new)
            score_pert = _eval_score(a_pert)
            if score_pert < best_score:
                best_score = score_pert
                best_z = z_pert_new.copy()
                z = z_pert_new

    return best_z, _softmax(best_z), best_score


def _coord_descent(a, t_end, rng):
    """Targeted coordinate descent: reduce peak convolution by transferring mass."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_obj = _quick_obj(a)
    no_improve = 0

    while time.monotonic() < t_end:
        c = _fft_conv_sq(a)
        # Find top peak lags
        n_peaks = min(5, 2 * n - 1)
        peak_lags = np.argpartition(-c, n_peaks - 1)[:n_peaks]
        peak_lags = peak_lags[np.argsort(-c[peak_lags])]

        improved = False

        for k in peak_lags:
            if time.monotonic() > t_end:
                break
            # Coordinates contributing to c[k]: pairs (j, k-j) for valid j
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            contributors = list(range(lo, hi + 1))

            # Try reducing each contributor and transferring to a random non-contributor
            for j in contributors[:20]:  # limit work per lag
                if a[j] < 1e-8:
                    continue
                partner = k - j
                if partner == j:
                    continue
                # Find a random target that doesn't contribute to c[k]
                for _ in range(3):
                    target = rng.integers(n)
                    if target == j or target == partner:
                        continue
                    if (0 <= k - target < n):
                        continue  # this would also contribute to c[k]

                    for frac in [0.1, 0.3, 0.5]:
                        delta = frac * a[j]
                        if delta < 1e-10:
                            break
                        a[j] -= delta
                        a[target] += delta
                        obj = _quick_obj(a)
                        if obj < best_obj - 1e-12:
                            best_obj = obj
                            improved = True
                        else:
                            a[j] += delta
                            a[target] -= delta

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return a, best_obj


def _gradient_descent_fallback(a, t_end, rng):
    """Simple gradient descent with top-L smoothing as fallback."""
    n = a.size
    a = _proj_simplex(a.copy(), 1.0)
    best_a = a.copy()
    best_mc = _fft_conv_sq(a).max()

    momentum = 0.9
    buf = np.zeros(n)
    lr = 0.05
    tau = 0.3
    L = 50
    step = 0

    while time.monotonic() < t_end:
        c = _fft_conv_sq(a)
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()

        if step % 2000 == 0:
            tau = max(0.01, tau * 0.8)
            L = max(5, int(L * 0.75))
            lr = max(1e-6, lr * 0.9)
            momentum = min(0.98, momentum + 0.001)

        # Compute gradient via top-L softmax
        Lc = min(L, 2 * n - 1)
        top_idx = np.argpartition(-c, Lc - 1)[:Lc]
        c_top = c[top_idx]
        m = c_top.max()
        w_top = np.exp((c_top - m) / tau)
        w_top = w_top / w_top.sum()
        w = np.zeros(2 * n - 1)
        w[top_idx] = w_top

        C_w = np.dot(w, c)
        R = np.correlate(w, a, mode='valid')[:n]
        # Gradient in a-space (not z-space): g_a[i] = 2*sum_k w_k * a[k-i] * 2 ... 
        # Actually for the a-space: d(max_c)/da_i = 2*sum_k w_k * a[k-i]
        # (by our formula dc_k/da_i = 2*a[k-i])
        grad_a = 2.0 * R  # this is sum_k w_k * 2*a[k-i] = 2*R_i
        # But we need to project to the tangent space of the simplex
        # The projected gradient: g - mean(g) (for the sum constraint)
        # And clip for the non-negativity
        g = grad_a - grad_a.mean()
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn
            buf = momentum * buf - lr * g
            a = _proj_simplex(a + lr * buf, 1.0)

        step += 1

    return best_a
```
