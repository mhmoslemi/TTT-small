```python
def find_best_direction(a, M):
    """Find g >= 0 maximizing sum(g) subject to (a*g)_k <= M for all k."""
    n = len(a)
    # Convolution a*g has length 2n-1
    # (a*g)_k = sum_i a_i * g_{k-i} for valid i
    # This is a linear function of g for fixed a
    # Constraint: (A @ g)_k <= M for each k, where A is the convolution matrix
    # Objective: maximize sum(g) = - minimize -sum(g)
    
    # Build the constraint matrix
    # For k = 0, ..., 2n-2:
    # (a*g)_k = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a_i * g_{k-i}
    # Let j = k-i, so i = k-j, j from max(0,k-n+1) to min(k,n-1)
    # (a*g)_k = sum_j a_{k-j} * g_j
    
    # So the constraint matrix R has R[k, j] = a_{k-j} if 0 <= k-j < n, else 0
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    
    R = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) s.t. R @ g <= M, g >= 0
    # = Minimize -sum(g) s.t. R @ g <= M, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=R, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using LP-guided direction search.

The main algorithm finds the optimal direction g via LP (maximize sum(g) 
subject to g>=0 and (a*g)_k <= M for all k), then moves towards g.
This is combined with line search, multiple seeds, and local refinement.

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _conv(a):
    """Linear convolution of a with itself."""
    return np.convolve(a, a)


def _obj(a):
    """Objective value."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)


def _build_conv_matrix(a):
    """Build sparse matrix R where (R @ g)_k = (a * g)_k for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i
            if a[i] != 0.0:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    R = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return R


def _solve_direction_lp(a, M):
    """Solve max sum(g) s.t. g >= 0, (a*g)_k <= M for all k."""
    n = len(a)
    R = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=R, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _lp_step(a, t_max=0.1, max_iter=50):
    """One LP-guided step. Returns new a and whether it improved."""
    n = len(a)
    s = a.sum()
    if s <= 0:
        return a, False
    
    a_norm = a / s  # normalize to sum 1
    c = np.convolve(a_norm, a_norm)
    M = c.max()
    
    # Solve LP to find best direction
    g = _solve_direction_lp(a_norm, M)
    if g is None or g.sum() <= 0:
        return a, False
    
    # Rescale g to have sum = 1 (same as a_norm)
    g = g / g.sum()
    
    # Try to find best t via line search
    best_a = a_norm.copy()
    best_M = M
    for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        a_try = (1 - t) * a_norm + t * g
        a_try = np.maximum(a_try, 0.0)
        a_try = a_try / a_try.sum()
        c_try = np.convolve(a_try, a_try)
        M_try = c_try.max()
        if M_try < best_M:
            best_M = M_try
            best_a = a_try.copy()
    
    # Scale back to original sum
    best_a = best_a * s
    return best_a, best_M < M


def _smoothed_grad_step(a, L, tau, lr):
    """Smoothed subgradient step (fallback)."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s <= 0:
        return a
    
    # Top-L smoothing
    L_eff = min(L, n)
    order = np.argpartition(-c, L_eff - 1)[:L_eff]
    wc = c[order]
    w = np.exp(wc / max(tau, 1e-12))
    w = w / w.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    for k, wk in zip(order, w):
        if wk <= 0:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i
            grad[i] += 2.0 * wk * a[j]
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 0:
        a_new = a_new / a_new.sum() * s
    else:
        a_new = a.copy()
    return a_new


def _proj_simplex(v, target=1.0):
    """Project onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)

def _run_smoothed_descent(a, t_end, L=100, tau0=0.2, lr0=0.1, verbose=True):

    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()

    best_obj = 2.0 * n * best_mc  # since sum=1
    
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    
    momentum = 0.9
    tau = tau0
    lr = lr0
    buf = np.zeros(n)
    step = 0
    
    if verbose:
        print(f"  Start: obj={best_obj:.8f}, n={n}")
    
    while time.monotonic() < t_end - 1.0:
        fa = np.fft.rfft(a, m)
        c = np.fft.irfft(fa * fa, m)[:2*n-1]
        mc = c.max()
        
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
            best_obj = 2.0 * n * mc
        
        step += 1
        
        # Adaptive parameters
        if step % 1000 == 0:
            tau = max(0.02, tau * 0.85)
            lr = max(1e-5, lr * 0.75)
            Lc = max(10, L - step // 200)
        else:
            Lc = L
        
        # Compute smoothed gradient
        L_eff = min(Lc, 2*n-1)
        order = np.argpartition(-c, L_eff - 1)[:L_eff]
        wc = c[order]
        w = np.exp(wc / max(tau, 1e-12))
        w = w / w.sum()
        
        grad = np.zeros(n, dtype=np.float64)
        for k, wk in zip(order, w):
            if wk <= 0:

            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * wk * a[k - i]
        
        gn = np.linalg.norm(grad)
        if gn > 0:
            grad = grad / gn

        a_new = a + lr * buf
        a_new = _proj_simplex(a_new, 1.0)
        a = a_new
        
        if step % 500 == 0 and verbose:

def _generate_seeds(n):

    seeds.append(tri / tri.sum())
    
    # Inverted triangular (peak at edges)
    inv_tri = np.maximum(1.0, (n - 1) - tri)
    seeds.append(inv_tri / inv_tri.sum())
    
    # Sine wave
    sine = np.sin(np.pi * (x + 0.5) / n) ** 2 + 0.1
    seeds.append(sine / sine.sum())
    
    # Cosine tapered
    cos_t = 0.5 + 0.5 * np.cos(2 * np.pi * x / (2*n))

    # Chirp-like (absolute value of complex chirp, made real)

    # This is all 1s, so let's do something else
```

```python
"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using LP-guided direction search
combined with smoothed subgradient descent.

The main algorithm:
1. Solve an LP to find the optimal direction g (maximize sum(g) subject to
   g>=0 and (a*g)_k <= M for all k), which identifies the best way to
   redistribute mass to flatten the auto-convolution.
2. Move towards g with adaptive line search.
3. Between LP solves, use smoothed subgradient steps for faster local descent.
4. Multiple structured seeds and periodic restarts to escape local minima.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _conv(a):
    """Linear convolution of a with itself."""
    return np.convolve(a, a)


def _obj_val(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)


def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    idx = np.arange(1, n + 1)
    cond = sv * idx > (cssv - target)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, target / n)
    rho = rho_arr[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _build_conv_matrix(a):
    """Build sparse matrix R where (R @ g)_k = (a * g)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i
            if a[i] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    if len(vals) == 0:
        return csr_matrix((2 * n - 1, n))
    R = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return R


def _solve_lp_direction(a_norm, M, timeout=3.0):
    """Solve max sum(g) s.t. g >= 0, (a_norm*g)_k <= M for all k."""
    n = len(a_norm)
    R = _build_conv_matrix(a_norm)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=R, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _smoothed_grad(c, a, L, tau):
    """Compute smoothed subgradient using top-L lags."""
    n = a.size
    c_len = len(c)
    L_eff = min(L, c_len)
    order = np.argpartition(-c, L_eff - 1)[:L_eff]
    wc = c[order]
    w = np.exp(wc / max(tau, 1e-12))
    w_sum = w.sum()
    if w_sum <= 0:
        return np.zeros(n)
    w = w / w_sum

    grad = np.zeros(n, dtype=np.float64)
    for k, wk in zip(order, w):
        if wk <= 0:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i
            grad[i] += 2.0 * wk * a[j]
    return grad


def _run_lp_descent(a, t_end, verbose=False):
    """LP-guided descent with periodic smoothed subgradient steps."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s <= 0:
        return np.ones(n) / n, 2.0 * n * (1.0 / n)

    a = a / s  # normalize to sum 1
    c = np.convolve(a, a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    best_obj = 2.0 * n * M

    if verbose:
        print(f"    LP descent start: obj={best_obj:.8f}")

    step = 0
    stall_count = 0
    lp_dir = None
    lp_step_count = 0
    momentum = 0.85
    buf = np.zeros(n)
    tau = 0.15
    L = min(60, 2 * n - 1)
    lr = 0.08

    while time.monotonic() < t_end - 1.5:
        step += 1

        # Solve LP every 30 steps or if direction is stale
        need_lp = (lp_dir is None) or (step % 30 == 0) or (stall_count > 20)
        if need_lp:
            g = _solve_lp_direction(a, M, timeout=2.0)
            if g is not None and g.sum() > 1e-10:
                lp_dir = g / g.sum()
                lp_step_count = 0
            else:
                # Fallback: use smoothed gradient direction
                grad = _smoothed_grad(c, a, L, tau)
                gn = np.linalg.norm(grad)
                if gn > 0:
                    lp_dir = np.maximum(a - grad / gn, 0)
                    ls = lp_dir.sum()
                    lp_dir = lp_dir / ls if ls > 0 else np.ones(n) / n

        lp_step_count += 1

        # Take step towards lp_dir with adaptive size
        if lp_step_count <= 5:
            t = 0.03 + 0.02 * lp_step_count
        else:
            t = max(0.01, 0.12 * math.exp(-0.05 * lp_step_count))

        a_new = (1.0 - t) * a + t * lp_dir
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 0:
            a_new = a_new / sn

        c_new = np.convolve(a_new, a_new)
        M_new = c_new.max()

        if M_new < M:
            a = a_new
            M = M_new
            c = c_new
            stall_count = 0
            obj_new = 2.0 * n * M
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
        else:
            # Step didn't help; use smoothed subgradient instead
            grad = _smoothed_grad(c, a, L, tau)
            gn = np.linalg.norm(grad)
            if gn > 0:
                grad = grad / gn
            buf = momentum * buf - lr * grad
            a_try = a + lr * buf
            a_try = _proj_simplex(a_try, 1.0)
            c_try = np.convolve(a_try, a_try)
            M_try = c_try.max()
            if M_try < M:
                a = a_try
                M = M_try
                c = c_try
                obj_try = 2.0 * n * M
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Adaptive parameter updates
        if step % 200 == 0:
            tau = max(0.03, tau * 0.85)
            lr = max(0.005, lr * 0.8)
            L = max(10, int(L * 0.85))
            momentum = min(0.93, momentum + 0.002)

        # If stalled, try a larger perturbation
        if stall_count > 60:
            rng = np.random.default_rng(step)
            a_pert = a * np.exp(rng.normal(0, 0.1, n))
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 0:
                a_pert = a_pert / sp
                c_pert = np.convolve(a_pert, a_pert)
                if c_pert.max() < M:
                    a = a_pert
                    M = c_pert.max()
                    c = c_pert
                    obj_pert = 2.0 * n * M
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a.copy()
            stall_count = 0
            lp_dir = None

        if verbose and step % 100 == 0:
            print(f"      step {step}: obj={2*n*M:.8f}, best={best_obj:.8f}, stall={stall_count}")

    return best_a, best_obj


def _run_fista_descent(a, t_end, L=80, tau0=0.2, lr0=0.1, verbose=False):
    """FISTA-style smoothed subgradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s <= 0:
        return np.ones(n) / n, 2.0 * n * (1.0 / n)
    a = a / s

    m = 1
    while m < 2 * n - 1:
        m <<= 1

    best_a = a.copy()
    best_M = np.convolve(a, a).max()
    best_obj = 2.0 * n * best_M

    if verbose:
        print(f"    FISTA start: obj={best_obj:.8f}")

    momentum_t = 1.0
    tau = tau0
    lr = lr0
    buf = np.zeros(n)
    a_ys = a.copy()
    step = 0

    while time.monotonic() < t_end - 1.5:
        step += 1
        momentum_t = (1.0 + math.sqrt(1.0 + 4.0 * momentum_t * momentum_t)) / 2.0
        mu = (momentum_t - 1.0) / (momentum_t + 2.0)

        fa = np.fft.rfft(a_ys, m)
        c = np.fft.irfft(fa * fa, m)[:2 * n - 1]

        L_eff = min(L, 2 * n - 1)
        grad = _smoothed_grad(c, a_ys, L_eff, tau)
        gn = np.linalg.norm(grad)
        if gn > 0:
            grad = grad / gn

        buf = mu * buf - lr * grad
        a_new = a_ys + (lr / momentum_t) * buf
        a_new = _proj_simplex(a_new, 1.0)

        c_new = np.convolve(a_new, a_new)
        M_new = c_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            best_obj = 2.0 * n * M_new

        a = a_new
        a_ys = a + mu * (a - a_ys)

        if step % 1500 == 0:
            tau = max(0.02, tau * 0.8)
            lr = max(0.005, lr * 0.85)
            L = max(15, int(L * 0.9))

        if verbose and step % 500 == 0:
            print(f"      FISTA step {step}: obj={2*n*M_new:.8f}, best={best_obj:.8f}")

    return best_a, best_obj


def _generate_seeds(n, base_seq=None):
    """Generate diverse structured seeds."""
    seeds = []
    x = np.arange(n, dtype=np.float64)

    # Base sequence (normalized)
    if base_seq is not None:
        b = np.maximum(base_seq, 0.0)
        if b.sum() > 0:
            seeds.append(b / b.sum())
        # Reversed
        seeds.append(b[::-1].copy() / b.sum())

    # Uniform
    seeds.append(np.ones(n) / n)

    # Quadratic bump (peak at center)
    quad = 1.0 - ((x - n / 2.0) / (n / 2.0)) ** 2
    quad = np.maximum(quad, 0.01)
    seeds.append(quad / quad.sum())

    # Sine window
    sine_w = 0.5 + 0.5 * np.sin(np.pi * (x + 0.5) / n)
    sine_w = np.maximum(sine_w, 0.01)
    seeds.append(sine_w / sine_w.sum())

    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / (2.0 * n))
    cos_w = np.maximum(cos_w, 0.01)
    seeds.append(cos_w / cos_w.sum())

    # Power-law ramp
    for p in [0.5, 2.0, 4.0]:
        ramp = (x + 1.0) ** p
        seeds.append(ramp / ramp.sum())
        ramp_r = (n - x + 1.0) ** p
        seeds.append(ramp_r / ramp_r.sum())

    # Two-sided peak
    peak = np.minimum(x + 1, n - x) ** 0.5
    seeds.append(peak / peak.sum())

    # Multiplicative sinusoidal modulation
    for freq in [3, 5, 7]:
        mod = 1.0 + 0.4 * np.sin(2.0 * np.pi * freq * x / n)
        mod = np.maximum(mod, 0.05)
        seeds.append(mod / mod.sum())

    # Smoothed random
    rng = np.random.default_rng(123)
    for _ in range(4):
        r = rng.uniform(0.2, 1.0, n)
        # Gaussian smoothing
        sigma = max(1.0, n / 40.0)
        kernel_size = int(6 * sigma) | 1
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / sigma ** 2)
        kernel /= kernel.sum()
        smoothed = np.convolve(r, kernel, mode='same')
        smoothed = np.maximum(smoothed, 0.01)
        seeds.append(smoothed / smoothed.sum())

    # Exponential decay pairs
    for center_frac in [0.25, 0.5, 0.75]:
        center = center_frac * n
        bell = np.exp(-((x - center) ** 2) / (2.0 * (n / 8.0) ** 2))
        bell = np.maximum(bell, 0.001)
        seeds.append(bell / bell.sum())

    return seeds


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    Uses LP-guided direction search combined with smoothed subgradient descent.
    """
    t_start = time.monotonic()
    t_end = t_start + max(0.0, float(budget_s)) - 4.0

    # Get base sequence
    base = None
    try:
        b = np.asarray(height_sequence_1, dtype=np.float64)
        b = np.clip(b, 0.0, 1000.0)
        if b.ndim == 1 and b.size >= 4 and np.isfinite(b).all() and b.sum() > 0.01:
            base = b
    except Exception:
        pass

    if base is not None:
        n = base.size
    else:
        n = 512
        base = np.ones(n)

    print(f"Sequence length n={n}, budget={budget_s}s")

    # Generate and evaluate seeds
    seeds = _generate_seeds(n, base)
    print(f"Generated {len(seeds)} seeds")

    seed_scores = []
    for i, s in enumerate(seeds):
        s = np.maximum(s, 0.0)
        ss = s.sum()
        if ss <= 0:
            continue
        s = s / ss
        obj = 2.0 * n * np.convolve(s, s).max()
        seed_scores.append((obj, i))
    seed_scores.sort()

    best_a = seeds[seed_scores[0][1]].copy()
    best_score = seed_scores[0][0]
    print(f"Best seed score: {best_score:.8f}")

    # Verify with evaluator
    try:
        ev = float(evaluate_sequence([float(x) for x in best_a]))
        if ev < best_score:
            best_score = ev
    except Exception:
        pass

    # Phase 1: LP descent on top 3 seeds
    top_k = min(3, len(seed_scores))
    phase1_time = (t_end - time.monotonic()) * 0.55
    time_per_seed = phase1_time / top_k
    print(f"\nPhase 1: LP descent, {top_k} seeds, ~{time_per_seed:.0f}s each")

    for rank in range(top_k):
        if time.monotonic() > t_end - 20:
            break
        obj_i, seed_i = seed_scores[rank]
        seed_a = seeds[seed_i].copy()
        t_seed_end = min(time.monotonic() + time_per_seed - 2, t_end - 15)

        print(f"  Seed {rank} (obj={obj_i:.8f}):")
        a_res, obj_res = _run_lp_descent(seed_a, t_seed_end, verbose=(rank == 0))

        if obj_res < best_score:
            best_score = obj_res
            best_a = a_res.copy()
            print(f"    New best: {best_score:.8f}")

        # Carry forward
        seeds[seed_i] = a_res.copy()

    # Phase 2: FISTA refinement on best solution
    if time.monotonic() < t_end - 30:
        phase2_time = (t_end - time.monotonic()) * 0.35
        print(f"\nPhase 2: FISTA refinement, ~{phase2_time:.0f}s")

        for L_try in [50, 100]:
            for tau_try in [0.1, 0.25]:
                if time.monotonic() > t_end - 20:
                    break
                t_sub = min(time.monotonic() + phase2_time / 4.0, t_end - 15)
                a_res, obj_res = _run_fista_descent(best_a, t_sub, L=L_try, tau0=tau_try, lr0=0.08, verbose=False)
                if obj_res < best_score:
                    best_score = obj_res
                    best_a = a_res.copy()
                    print(f"    FISTA (L={L_try}, tau={tau_try}): {best_score:.8f}")

    # Phase 3: LP refinement on the best solution
    if time.monotonic() < t_end - 15:
        print(f"\nPhase 3: LP refinement, time_left={t_end - time.monotonic():.1f}s")
        t_lp2 = t_end - 5
        a_res, obj_res = _run_lp_descent(best_a, t_lp2, verbose=False)
        if obj_res < best_score:
            best_score = obj_res
            best_a = a_res.copy()
            print(f"    LP refinement: {best_score:.8f}")

    # Phase 4: Fine local search with small perturbations
    if time.monotonic() < t_end - 5:
        print(f"\nPhase 4: Local perturbation search")
        rng = np.random.default_rng(seed + 1)
        t_pert_end = t_end - 3
        while time.monotonic() < t_pert_end:
            # Random small perturbation
            for amp in [0.01, 0.03, 0.05]:
                a_pert = best_a * np.exp(rng.normal(0, amp, n))
                a_pert = np.maximum(a_pert, 0)
                sp = a_pert.sum()
                if sp > 0:
                    a_pert = a_pert / sp * best_a.sum()
                    obj_pert = _obj_val(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_a = a_pert.copy()
                        print(f"    Perturb (amp={amp}): {best_score:.8f}")
            # Targeted: reduce peak lags
            c = np.convolve(best_a, best_a)
            peak_idx = np.argmax(c)
            # Reduce elements that contribute most to the peak
            lo = max(0, peak_idx - n + 1)
            hi = min(peak_idx, n - 1)
            contrib = np.zeros(n)
            for i in range(lo, hi + 1):
                j = peak_idx - i
                contrib[i] = best_a[i] * best_a[j]
            top_contrib = np.argsort(-contrib)[:n // 4]
            a_pert2 = best_a.copy()
            a_pert2[top_contrib] *= 0.95
            a_pert2 = np.maximum(a_pert2, 0)
            sp2 = a_pert2.sum()
            if sp2 > 0:
                a_pert2 = a_pert2 / sp2 * best_a.sum()
                obj_pert2 = _obj_val(a_pert2)
                if obj_pert2 < best_score:
                    best_score = obj_pert2
                    best_a = a_pert2.copy()
                    print(f"    Peak reduce: {best_score:.8f}")

    # Final verification
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_a]))
        print(f"\nFinal verified score: {final_score:.8f}")
        if final_score < best_score:
            best_score = final_score
    except Exception as e:
        print(f"Evaluation error: {e}")

    # Ensure valid output
    out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.full(n, 1.0)

    print(f"Returning sequence of length {len(out)}, best obj={best_score:.8f}")
    return [float(x) for x in out]
```
