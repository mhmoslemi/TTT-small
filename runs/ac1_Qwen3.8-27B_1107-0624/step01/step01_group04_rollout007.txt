def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 1e-300, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def cosine_taper_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    x = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * x / 2.0) ** 2
    a = np.maximum(a, 1e-10)
    return normalize_nonneg(a)


def gaussian_taper_normalized(n, sigma_frac=0.4):
    n = int(n)
    if n <= 0:
        n = 16
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    s = float(sigma_frac) * n
    a = np.exp(-0.5 * ((x - center) / s) ** 2)
    a = np.maximum(a, 1e-10)
    return normalize_nonneg(a)


def perturb_greedy(n, rng, sigma):

def mix_greedy_uniform(n, lam):

def actual_score_fft(p):

def softmax_gradient_optimize(p, budget_s, seed=0):
    """
    Softmax-smoothed gradient descent with Nesterov momentum in log-domain.
    p is a probability vector (sums to 1).
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 1.0:
        f = actual_score_fft(p)
        return p, f

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 128:
        lr = 0.06
    elif n < 512:
        lr = 0.04
    elif n < 2048:
        lr = 0.025
    else:
        lr = 0.015

    beta = 0.9  # Nesterov momentum
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 3.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Adaptive temperature: start smooth, end sharp
        alpha = 6.0 + 54.0 * np.exp(-3.0 * frac)
        # Adaptive LR: decay over time
        lr_t = lr * (1.0 + 0.5 * np.exp(-5.0 * frac))

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Softmax weights over convolution values
        z = alpha * (c / maxc - 1.0)
        z = np.clip(z, -50.0, 50.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0.0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient: convolution of p with reversed w
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        # Nesterov-style momentum
        m = beta * m + (1.0 - beta) * grad_x
        v = m  # simplified: use m as velocity

        step = lr_t * v / (np.sqrt(np.mean(v * v)) + eps)
        step = np.clip(step, -20.0, 20.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -100.0, 100.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

        # Periodic annealing restart (small perturbation)
        if t % 500 == 0 and frac > 0.3:
            small_noise = rng_local.normal(0.0, 0.01, n)
            x = np.log(p + 1e-300) + small_noise
            x -= x.mean()
            x = np.clip(x, -100.0, 100.0)
            x -= x.max()
            e = np.exp(x)
            p = e / e.sum()

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f

def peak_deflation(p, budget_s, rng):
    """
    Target the peak of the autocorrelation directly by deflating.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s < 2.0:
        return p, actual_score_fft(p)

    best_p = p.copy()
    best_f = actual_score_fft(p)
    end = time.monotonic() + max(0.0, budget_s - 2.0)

    it = 0
    while time.monotonic() < end:
        it += 1
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(p, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        k_star = int(np.argmax(c))

        # Subgradient direction: 2 * a flipped/shifted
        subgrad = np.zeros(n, dtype=np.float64)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                subgrad[i] = 2.0 * p[j]

        # Small step in negative subgradient direction
        sg_norm = float(np.linalg.norm(subgrad))
        if sg_norm < 1e-12:
            break
        # Use adaptive step
        step_size = 0.01 * (1.0 + 0.5 * np.sin(it * 0.1))
        p_new = p - step_size * subgrad / sg_norm
        p_new = np.clip(p_new, 0.0, None)
        sn = float(p_new.sum())
        if sn < 1e-12:
            break
        p_new = p_new / sn

        f_new = actual_score_fft(p_new)
        if np.isfinite(f_new) and f_new < best_f:
            best_f = f_new
            best_p = p_new.copy()
            p = p_new
        else:
            # Reduce step
            step_size *= 0.5
            p_new = p - step_size * subgrad / sg_norm
            p_new = np.clip(p_new, 0.0, None)
            sn = float(p_new.sum())
            if sn > 1e-12:
                p_new = p_new / sn
                f_new2 = actual_score_fft(p_new)
                if np.isfinite(f_new2) and f_new2 < best_f:
                    best_f = f_new2
                    best_p = p_new.copy()
                    p = p_new
                else:
                    break

        # Occasionally apply random perturbation to escape
        if it % 200 == 0:
            noise = rng.normal(0.0, 0.02, n)
            p = np.clip(p + noise, 0.0, None)
            sn = float(p.sum())
            if sn > 1e-12:
                p = p / sn

    return best_p, best_f

    if remaining > 60.0 and best_p is not None:
        # Deflation on best
        alloc_def = min(200.0, remaining * 0.3)
        p_out, f_out = peak_deflation(best_p, alloc_def, rng)
        record(p_out, f_out)

        # Restart from perturbed best
        remaining = deadline - time.monotonic()
        if remaining > 60.0:
            for attempt in range(3):
                if time.monotonic() >= deadline - 30.0:
                    break
                sigma = 0.05 + 0.05 * attempt
                p0 = random_perturb(best_p, rng, sigma)
                alloc = min(150.0, max(20.0, (deadline - time.monotonic() - 20.0) * 0.5))
                p_out, f_out = softmax_gradient_optimize(p0, alloc, seed=int(seed) + 1000 + attempt)
                record(p_out, f_out)

    # Phase 5: Final Adam polish on absolute best
    print("Phase 5: Final polish")
    remaining = deadline - time.monotonic()
    if remaining > 40.0 and best_p is not None:
        alloc = min(200.0, remaining - 10.0)
        p_out, f_out = softmax_gradient_optimize(best_p, alloc, seed=int(seed) + 9999)
        record(p_out, f_out)

    # Additional: try a few more lengths near the best
    if best_p is not None:
        best_n = int(best_p.size)
        for delta in [-128, -64, 64, 128, 256]:
            n_try = best_n + delta
            if n_try < 64 or n_try > 8192:
                continue
            if time.monotonic() >= deadline - 30.0:
                break
            p_try = greedy_normalized(n_try)
            # Interpolate with best
            if n_try != best_n:
                # Resample best to new length
                x_old = np.linspace(0, 1, best_n)
                x_new = np.linspace(0, 1, n_try)
                p_resampled = np.interp(x_new, x_old, best_p)
                p_mix = 0.5 * normalize_nonneg(p_resampled) + 0.5 * p_try
                p_try = normalize_nonneg(p_mix)
            record(p_try)
            # Quick gradient step
            if time.monotonic() < deadline - 20.0:
                p_out, f_out = softmax_gradient_optimize(p_try, min(60.0, deadline - time.monotonic() - 10.0), seed=int(seed) + n_try)
                record(p_out, f_out)

    if best_p is None:
        best_p = greedy_normalized(256)
        best_score = actual_score_fft(best_p)

    print("Final internal best: %.6f at n=%d" % (best_score, best_p.size))

    # Phase 6: Validation
    print("Phase 6: Validation")
    candidates = [best_p]
    items = []
    for n, (sc, p) in best_by_n.items():
        items.append((sc, n, p))
    items.sort(key=lambda x: x[0])
    for sc, n, p in items[:5]:
        if p is not None and n != best_p.size:
            candidates.append(p)

    for cand in candidates:
        if time.monotonic() >= deadline - 5.0:
            break
        try:
            seq = scale_output(cand, 10.0)
            gscore = try_global_eval(seq)
            print("  Global eval n=%d -> %.6f" % (cand.size, gscore))
            if np.isfinite(gscore) and gscore < best_score:
                best_score = gscore
                best_p = np.asarray(cand, dtype=np.float64).ravel()
        except Exception:
            pass

    try:
        final_seq = scale_output(best_p, 10.0)
    except Exception:
        final_seq = scale_output(greedy_normalized(256), 10.0)

    print("DONE. Final score: %.6f, length: %d" % (best_score, len(final_seq)))
    return final_seq
