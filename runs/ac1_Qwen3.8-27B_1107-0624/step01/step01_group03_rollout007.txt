def fast_score(s):

    lse = max_c + np.log(np.sum(np.exp(shifted)))
    val = 2.0 * n * lse

    # Gradient: sum_k w_k * 2 * s_{k-i}
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)

    # grad[i] = 2 * sum_k w[k] * s[k-i] for valid k-i
    # This is a cross-correlation: grad = 2 * (w * s) where * is cross-corr
    # Using convolution: cross_corr(w, s)[i] = sum_k w[k] * s[k-i]
    # = conv(w_flipped, s)[i + (2n-1 - n)] ... let's just use the direct formula
    # Actually: sum_k w[k] * s[k-i] for 0<=k-i<n
    # = sum_{j=0}^{n-1} w[i+j] * s[j]  (where j = k-i, so k = i+j)
    # This is a cross-correlation of w and s
    # cross_corr(w, s)[i] = sum_j w[i+j] * s[j]
    # We can compute this via: if we pad w to length 2n-1, then
    # cross_corr = conv(w_padded, s_flipped) shifted...
    # Easier: use the fact that it's equivalent to
    # For each i: sum over valid k of w[k]*s[k-i]
    # Let's use FFT:
    # cross_corr(w, s)[i] = sum_j w[i+j] s[j]
    # = (w * reverse(s))[i + n - 1] where * is linear conv
    # Actually let me just compute it directly for clarity (n is at most 2048)
    grad = np.zeros(n)
    # For efficiency, use the convolution structure
    # grad[i] = 2 * sum_{k=max(i,0)}^{min(i+2n-2, 2n-2)} w[k] * s[k-i] / (k-i < n)
    # = 2 * sum_{j=0}^{n-1} w[i+j] * s[j]  (where i+j < 2n-1)
    # This is the cross-correlation of w and s at lag i
    # We can compute all at once using FFT
    N_fft = 2 * (2 * n - 1) - 1
    W = np.fft.rfft(w, N_fft)
    S = np.fft.rfft(s, N_fft)
    # cross_corr(w, s)[i] = sum_j w[i+j] s[j]
    # This equals the inverse FFT of W * conj(S) shifted
    # Actually: if we define g[i] = sum_j w[i+j]*s[j], then
    # This is the "anti-convolution": g = ifft(fft(w_padded) * conj(fft(s_padded)))
    # Let me use a different approach: g[i] = sum_j w[i+j]*s[j]
    # = conv(w, reverse(s))[i + n - 1 + ... ]
    # Ugh, let me just do it the simple way for n <= 2048
    # Actually for n=2048, the sum has at most 2048 terms per i, and there are 2048 i's
    # So it's 2048*2048 = 4M operations, which is fast in numpy
    # Use broadcasting:
    # w has length 2n-1, we need w[i+j] for j=0..n-1, i=0..n-1
    # indices = i + j where i in [0,n), j in [0,n)
    # w[i+j] is valid when i+j < 2n-1
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]  # (n, n) array of i+j
    valid = idx < (2 * n - 1)
    w_expanded = np.where(valid, w[np.minimum(idx, 2*n-2)], 0.0)  # (n, n)
    s_row = s[None, :]  # (1, n)
    grad = 2.0 * np.sum(w_expanded * s_row, axis=1)

    return val, grad


def smooth_optimize(s, n, deadline=None, max_iter=300):
    """Optimize using L-BFGS on log-sum-exp surrogate."""
    s = s / s.sum()
    try:
        res = minimize(smooth_objective_and_grad, s, args=(n,),
                       jac=True, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': max_iter, 'gtol': 1e-7,
                                'ftol': 1e-12, 'maxls': 50})
        if res.x is not None:
            s_opt = np.maximum(res.x, 0.0)
            if s_opt.sum() > 1e-14:
                s_opt = s_opt / s_opt.sum()
                sc = fast_score(s_opt)
                if sc < fast_score(s):
                    return s_opt, sc
    except Exception as e:
        pass
    return s, fast_score(s)


def multiplicative_updates(s, n, max_iters=200, eta=None, deadline=None):
    """Mirror descent with softmax subgradient."""
    s = s / s.sum()
    best = s.copy()
    best_sc = fast_score(s)
    no_imp = 0

    if eta is None:
        eta = 1.0 / n

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = np.convolve(s, s)
        max_c = np.max(conv)
        # Softmax weights (temperature decreasing)
        temp = max(0.01, 50.0 / (it + 10))
        shifted = conv / temp - np.max(conv) / temp
        exp_c = np.exp(shifted)
        w = exp_c / np.sum(exp_c)

        # Gradient: grad[i] = 2 * sum_k w[k] * s[k-i]
        # Use the same trick as smooth_objective
        idx = np.arange(n)[:, None] + np.arange(n)[None, :]
        valid = idx < (2 * n - 1)
        w_expanded = np.where(valid, w[np.minimum(idx, 2 * n - 2)], 0.0)
        grad = 2.0 * np.sum(w_expanded * s[None, :], axis=1)

        # Multiplicative update (mirror descent with entropy)
        log_s = np.log(s + 1e-30) - eta * grad
        s_new = np.exp(log_s - np.max(log_s))
        s_new = s_new / s_new.sum()
        s_new = np.maximum(s_new, 1e-30)
        s_new = s_new / s_new.sum()

        sc = fast_score(s_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best = s_new.copy()
            no_imp = 0
            # Decrease eta slowly
            eta *= 0.999
        else:
            no_imp += 1
            if no_imp > 30:
                break
            # Try smaller step
            eta *= 0.9

    return best, best_sc


def generate_candidates(n, rng, hseq=None):
    """Generate diverse initial candidates for given length n."""
    candidates = []

    # From hseq (resized)
    if hseq is not None:
        x_old = np.linspace(0, 1, hseq.size)
        x_new = np.linspace(0, 1, n)
        h_r = np.interp(x_new, x_old, hseq)
        h_r = np.maximum(h_r, 0)
        if h_r.sum() > 0.01:
            h_r = h_r / h_r.sum()
            candidates.append(h_r)

    # Cosine modulation: 1 + a*cos(2*pi*f*t)
    for f in range(1, 8):
        for a in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            t = np.arange(n) / n
            s = 1.0 + a * np.cos(2 * np.pi * f * t)
            s = np.maximum(s, 0.01)
            s = s / s.sum()
            candidates.append(s)

    # Two-harmonic
    for f1 in [1, 2, 3]:
        for f2 in [2, 3, 4, 5]:
            for a1 in [0.3, 0.5]:
                for a2 in [0.2, 0.4]:
                    t = np.arange(n) / n
                    s = 1.0 + a1 * np.cos(2*np.pi*f1*t) + a2 * np.cos(2*np.pi*f2*t)
                    s = np.maximum(s, 0.01)
                    s = s / s.sum()
                    candidates.append(s)

    # Uniform
    candidates.append(np.ones(n) / n)

    # Exponential decay
    for a in [0.5, 1.0, 2.0, 3.0, 5.0]:
        t = np.arange(n) / n
        s = np.exp(-a * t)
        s = s / s.sum()
        candidates.append(s)
        s_rev = np.exp(-a * (1 - t))
        s_rev = s_rev / s_rev.sum()
        candidates.append(s_rev)

    # Sine (half wave)
    t = np.arange(n) / n
    s = np.sin(np.pi * t)
    s = s / s.sum()
    candidates.append(s)

    # Triangle
    s = 1.0 - 2.0 * np.abs(t - 0.5)
    s = s / s.sum()
    candidates.append(s)

    # Random
    for _ in range(8):
        s = rng.random(n) + 0.05
        s = s / s.sum()
        candidates.append(s)

    # Random smooth (spline-like)
    for _ in range(4):
        # Random control points, interpolate
        n_ctrl = rng.integers(3, 10)
        ctrl_x = np.sort(rng.random(n_ctrl))
        ctrl_y = rng.random(n_ctrl) + 0.1
        s = np.interp(t, ctrl_x, ctrl_y)
        s = s / s.sum()
        candidates.append(s)

    # Sort by score
    candidates.sort(key=lambda x: fast_score(x))
    return candidates


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid optimization combining LP-guided search, smooth L-BFGS optimization,
    and multiplicative updates. Tries multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0

    global_best_score = float('inf')
    global_best_seq = None

    # Try to load height_sequence_1
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}, score={fast_score(h):.6f}")
    except Exception:
        pass

    # Sequence lengths to try (larger n gives more degrees of freedom)
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining_time={remaining:.0f}s ===")

        # Generate candidates
        candidates = generate_candidates(n, rng, hseq)
        print(f"  {len(candidates)} candidates generated, best initial: {fast_score(candidates[0]):.6f}")

        # Process top candidates
        max_cands = min(10, len(candidates))
        per_cand_time = max(10.0, (remaining - 60) / max_cands)

        for ci in range(max_cands):
            if time.monotonic() > deadline - 30:
                break

            s0 = candidates[ci].copy()
            sc0 = fast_score(s0)
            cand_deadline = time.monotonic() + per_cand_time

            if sc0 > global_best_score + 0.1 and ci > 2:
                continue  # Skip if too far from best

            # Phase 1: Smooth optimization (L-BFGS)
            t0 = time.monotonic()
            if n <= 2048 and time.monotonic() < cand_deadline - 5:
                s1, sc1 = smooth_optimize(s0, n, deadline=cand_deadline - 5, max_iter=200)
                if sc1 < global_best_score:
                    global_best_score = sc1
                    global_best_seq = s1.copy()
                    print(f"  [{ci}] Smooth: {sc1:.6f} **NEW BEST**")
                else:
                    print(f"  [{ci}] Smooth: {sc1:.6f} (init {sc0:.6f})")
                s0 = s1
            elif time.monotonic() > cand_deadline - 5:
                continue

            # Phase 2: LP-guided search (original LP)
            if time.monotonic() < deadline - 30:
                t0 = time.monotonic()
                s2, sc2 = lp_iterate(s0, rng, max_iters=100, deadline=min(deadline - 20, time.monotonic() + per_cand_time * 0.6), use_new_lp=False)
                if sc2 < global_best_score:
                    global_best_score = sc2
                    global_best_seq = s2.copy()
                    print(f"  [{ci}] LP-orig: {sc2:.6f} ({time.monotonic()-t0:.1f}s) **NEW BEST**")
                else:
                    print(f"  [{ci}] LP-orig: {sc2:.6f} ({time.monotonic()-t0:.1f}s)")
                s0 = s2

            # Phase 3: LP-guided search (direct LP minimizing M')
            if time.monotonic() < deadline - 30 and n <= 1024:
                t0 = time.monotonic()
                s3, sc3 = lp_iterate(s0, rng, max_iters=80, deadline=min(deadline - 20, time.monotonic() + per_cand_time * 0.4), use_new_lp=True)
                if sc3 < global_best_score:
                    global_best_score = sc3
                    global_best_seq = s3.copy()
                    print(f"  [{ci}] LP-direct: {sc3:.6f} ({time.monotonic()-t0:.1f}s) **NEW BEST**")
                else:
                    print(f"  [{ci}] LP-direct: {sc3:.6f} ({time.monotonic()-t0:.1f}s)")
                s0 = s3

            # Phase 4: Multiplicative updates
            if time.monotonic() < deadline - 30:
                t0 = time.monotonic()
                s4, sc4 = multiplicative_updates(s0, n, max_iters=200, deadline=min(deadline - 20, time.monotonic() + 30))
                if sc4 < global_best_score:
                    global_best_score = sc4
                    global_best_seq = s4.copy()
                    print(f"  [{ci}] Mult: {sc4:.6f} ({time.monotonic()-t0:.1f}s) **NEW BEST**")
                else:
                    print(f"  [{ci}] Mult: {sc4:.6f} ({time.monotonic()-t0:.1f}s)")
                s0 = s4

            # Phase 5: Final LP polish
            if time.monotonic() < deadline - 30:
                t0 = time.monotonic()
                s5, sc5 = lp_iterate(s0, rng, max_iters=50, deadline=min(deadline - 20, time.monotonic() + 30), use_new_lp=(n <= 1024))
                if sc5 < global_best_score:
                    global_best_score = sc5
                    global_best_seq = s5.copy()
                    print(f"  [{ci}] LP-polish: {sc5:.6f} ({time.monotonic()-t0:.1f}s) **NEW BEST**")

            print(f"  [{ci}] After all phases: best so far = {global_best_score:.6f}")

        # Additional random restarts if time permits
        if time.monotonic() < deadline - 60:
            n_restarts = min(5, int((deadline - time.monotonic() - 60) / 30))
            for r in range(n_restarts):
                if time.monotonic() > deadline - 40:
                    break
                s_r = rng.random(n) + 0.05
                s_r = s_r / s_r.sum()
                # Quick smooth opt + LP
                if n <= 1024:
                    s_r, sc_r = smooth_optimize(s_r, n, deadline=deadline - 40, max_iter=100)
                s_r, sc_r = lp_iterate(s_r, rng, max_iters=50, deadline=deadline - 40, use_new_lp=(n <= 1024))
                if sc_r < global_best_score:
                    global_best_score = sc_r
                    global_best_seq = s_r.copy()
                    print(f"  Restart {r}: {sc_r:.6f} **NEW BEST**")

        # Print best for this n
        if global_best_seq is not None and global_best_seq.size == n:
            print(f"  Best for n={n}: {global_best_score:.6f}")

    # Try to refine the global best with more iterations
    if global_best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Final refinement of best (score={global_best_score:.6f}, n={len(global_best_seq)}) ===")
        n_best = len(global_best_seq)
        s_final = global_best_seq.copy()

        # More LP iterations
        s_final, sc = lp_iterate(s_final, rng, max_iters=200, deadline=deadline - 30, use_new_lp=(n_best <= 1024))
        if sc < global_best_score:
            global_best_score = sc
            global_best_seq = s_final.copy()
            print(f"  After final LP: {sc:.6f} **NEW BEST**")

        # Smooth optimization
        if n_best <= 2048 and time.monotonic() < deadline - 30:
            s_final, sc = smooth_optimize(s_final, n_best, deadline=deadline - 30, max_iter=300)
            if sc < global_best_score:
                global_best_score = sc
                global_best_seq = s_final.copy()
                print(f"  After final smooth: {sc:.6f} **NEW BEST**")

        # One more LP
        s_final, sc = lp_iterate(s_final, rng, max_iters=100, deadline=deadline - 30, use_new_lp=(n_best <= 1024))
        if sc < global_best_score:
            global_best_score = sc
            global_best_seq = s_final.copy()
            print(f"  After final LP2: {sc:.6f} **NEW BEST**")

    # Final fallback
    if global_best_seq is None:
        global_best_seq = np.ones(256) / 256.0
        global_best_score = fast_score(global_best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_sc = evaluate_sequence(result)
    print(f"\n=== FINAL: score={final_sc:.6f}, length={len(result)} ===")
    return result
