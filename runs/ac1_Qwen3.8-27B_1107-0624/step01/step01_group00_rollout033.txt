def _next_pow2(x):
    """Next power of 2 >= x."""
    return 1 << (x - 1).bit_length() if x > 1 else 1


def _proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if s <= 0:
        return np.zeros(n)
    # Scale to sum=1 case
    v = np.asarray(v, dtype=np.float64)
    # Algorithm from dcp / Giftis-O'Donnell
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not np.any(cond):
        return np.zeros(n)
    rho_max = np.max(rho[cond])
    theta = (cssv[rho_max - 1] - s) / rho_max
    return np.maximum(v - theta, 0.0)

    """Normalize a to sum=1, non-negative."""
    a = np.maximum(a, 0.0)

    if s < 1e-15:

    return a / s


def _fft_convolve(a):
    """Compute full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _fft_gradient(w, a):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] using FFT.
    w has length 2n-1, a has length n. Returns grad of length n."""
    n = len(a)
    # Build A_full of length 2n-1 (zero-padded a)
    A_full = np.zeros(2 * n - 1)
    A_full[:n] = a
    A_rev = A_full[::-1]
    # Convolution of w (length 2n-1) and A_rev (length 2n-1)
    # Result length: 4n-3
    m = _next_pow2(4 * n - 3)
    fw = np.fft.rfft(w, m)
    fa = np.fft.rfft(A_rev, m)
    conv = np.fft.irfft(fw * fa, m)
    # Extract indices n-1 to 2n-2
    return 2.0 * conv[n - 1:2 * n - 1]

    """Compute the gradient of (1/beta)*log(sum_k exp(beta*c_k)) where c = conv(a,a).

    c = _fft_convolve(a)
    # Stable log-sum-exp
    mc = c.max()
    exp_bc = np.exp(beta * (c - mc))
    lse = mc + np.log(exp_bc.sum()) / beta
    w = exp_bc / exp_bc.sum()
    grad = _fft_gradient(w, a)
    return grad, c, lse

    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _fft_grad(w, a):
    n = len(a)
    A_full = np.zeros(2 * n - 1, dtype=np.float64)
    A_full[:n] = a
    A_rev = A_full[::-1]
    m = _next_pow2(4 * n - 3)
    fw = np.fft.rfft(w, m)
    fa = np.fft.rfft(A_rev, m)
    conv = np.fft.irfft(fw * fa, m)
    return 2.0 * conv[n - 1:2 * n - 1]


def _lse_grad(a, beta):
    c = _fft_conv(a)
    mc = c.max()
    shifted = beta * (c - mc)
    max_shifted = shifted.max()
    exp_s = np.exp(shifted - max_shifted)
    w = exp_s / exp_s.sum()
    grad = _fft_grad(w, a)
    return grad, c


def _raw_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e18
    c = _fft_conv(a)
    return 2.0 * n * c.max() / (s * s)


def _fista_run(a0, t_end, beta0, lr0, tag):
    n = len(a0)
    a = _normalize(a0.astype(np.float64).copy())
    x_prev = a.copy()
    t_k = 1.0
    beta = beta0
    lr = lr0
    best_a = a.copy()
    best_raw = _raw_obj(a)
    best_score = 1e18
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_a]))
    except Exception:
        pass
    step = 0
    last_eval = time.monotonic()
    stall = 0
    while time.monotonic() < t_end - 1.0:
        t_prev = t_k
        t_k = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k))
        mu = (t_prev - 1.0) / t_k
        y = a + mu * (a - x_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = a.copy()
        grad, c = _lse_grad(y, beta)
        a_new = y - lr * grad
        a_new = _proj_simplex(a_new, 1.0)
        x_prev = a.copy()
        a = a_new
        step += 1
        if step % 300 == 0:
            beta = min(beta * 1.8, 5000.0)
            lr = max(0.005, lr * 0.88)
            raw = _raw_obj(a)
            if raw < best_raw - 1e-8:
                best_raw = raw
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    lr = max(0.003, lr * 0.7)
                    stall = 0
        if time.monotonic() - last_eval > 2.5 and step % 50 == 0:
            last_eval = time.monotonic()
            try:
                sc = float(evaluate_sequence([float(x) for x in a]))
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"    {tag} step={step} score={sc:.6f} beta={beta:.1f} lr={lr:.4f}")
            except Exception:
                pass
    try:
        sc = float(evaluate_sequence([float(x) for x in a]))
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    except Exception:
        pass
    return best_a, best_score


def _coord_polish(a, t_end, tag):
    n = len(a)
    a = _normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_raw = _raw_obj(a)
    best_score = 1e18
    try:
        best_score = float(evaluate_sequence([float(x) for x in best_a]))
    except Exception:
        pass
    c = _fft_conv(a)
    rng = np.random.default_rng(12345)
    iter_count = 0
    while time.monotonic() < t_end - 1.0:
        j = int(rng.integers(0, n))
        a0 = a[j]
        if a0 < 1e-12:
            a[j] = 1.0 / n
        else:
            a[j] = 0.0
        a = _normalize(a)
        c = _fft_conv(a)
        raw = c.max() / (a.sum() ** 2)
        new_raw = 2.0 * n * raw
        if new_raw < best_raw - 1e-10:
            best_raw = new_raw
            best_a = a.copy()
        else:
            a = best_a.copy()
        iter_count += 1
        if iter_count % 500 == 0:
            try:
                sc = float(evaluate_sequence([float(x) for x in a]))
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            except Exception:
                pass
    return best_a, best_score


def _seed_bell(n, center, width):
    x = np.arange(n, dtype=np.float64) / n
    return np.exp(-0.5 * ((x - center) / width) ** 2)


def _make_seeds(n0, a_init):
    seeds = []
    if a_init is not None:
        seeds.append(("init", _normalize(a_init)))
        seeds.append(("init_rev", _normalize(a_init[::-1])))
    n_seeds_len = [n0]
    for nl in [2048, 3072, 4096, 5120, 6144, 8192, 10240]:
        if nl != n0 and 500 <= nl <= 12000:
            n_seeds_len.append(nl)
    shapes = [
        ("bell_c", lambda nl: _seed_bell(nl, 0.5, 0.35)),
        ("bell_n", lambda nl: _seed_bell(nl, 0.35, 0.25)),
        ("arch", lambda nl: 1.0 - 0.7 * (2.0 * np.arange(nl) / nl - 1.0) ** 2),
        ("flat", lambda nl: np.ones(nl)),
        ("sin2", lambda nl: np.sin(np.pi * np.arange(nl) / nl) ** 3),
    ]
    for nl in n_seeds_len:
        if nl == n0 and a_init is not None:
            continue
        for name, fn in shapes:
            try:
                s = _normalize(fn(nl))
                if s.sum() > 0:
                    seeds.append((f"{name}_{nl}", s))
            except Exception:
                pass
    return seeds

    print(f"FINAL: n={len(out)} score={final_sc:.6f} time={time.monotonic() - t_start:.1f}s")
    return [float(x) for x in out]
