import time
from scipy.optimize import minimize

N = 4096

FFT_SELF = 2 * N  # for self-convolution (need 2N-1, round up to power of 2)
# Actually for self-conv of length N, we need FFT size >= 2N-1
FFT_SELF = 1 << (2*N - 1).bit_length()
FFT_CROSS = 1 << (2*N - 1).bit_length()  # same size for cross-conv

best_score = float('inf')
best_seq = None
deadline = 0.0

class Timeout(Exception):
    pass

def probs_from_z(z):
    """Convert logits to probabilities via softmax."""
    z = np.clip(z, -50, 50)
    z = np.where(np.isfinite(z), z, 0.0)
    m = np.max(z)
    expz = np.exp(z - m)
    total = np.sum(expz)
    p = expz / total
    return p

def self_conv(p):
    """Compute full self-convolution of p (length 2N-1)."""
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f * f, FFT_SELF).real
    return c[:2*N-1]

def score(p):
    """Compute the objective value."""
    c = self_conv(p)
    return 2.0 * N * np.max(c)

def grad_conv(p, w):
    """Compute gradient of sum_k w_k * (p*p)_k with respect to p.
    Result has length N (the middle part of the cross-conv)."""
    # (p*p)_k = sum_i p_i * p_{k-i}
    # d/dp_j of (p*p)_k = p_{k-j} + p_{k-j} = 2*p_{k-j} (for j in range, with bounds)
    # Actually: d/dp_j sum_k w_k (p*p)_k = 2 * sum_k w_k * p_{k-j}
    # This is 2 * (w * p_reversed)_j in the right range
    w_rev = w[::-1]
    cross = np.fft.rfft(p, FFT_CROSS) * np.fft.rfft(w_rev, FFT_CROSS)
    result = np.fft.irfft(cross, FFT_CROSS).real
    # The cross-conv of p (length N) and w_rev (length 2N-1) has length 3N-2
    # We want positions N-1 to 2N-2 (0-indexed) which correspond to j=0 to N-1
    return 2.0 * result[N-1:2*N-1]

        w = e / s

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using iterative subgradient and
smooth optimization (L-BFGS-B with soft-max surrogate) on probability
vectors parameterized by logits. Combines diverse initialization,
adaptive smoothing, top-K peak reduction, and mirror-descent-style
updates to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

N = 4096
FFT_SELF = 1 << (2 * N - 1).bit_length()
FFT_CROSS = 1 << (3 * N - 2).bit_length()
BOUNDS = [(-50.0, 50.0)] * (N - 1)

_global_best_score = float("inf")
_global_best_p = None
_global_deadline = 0.0


class _Timeout(Exception):
    pass


def _san(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    return np.clip(z, -50.0, 50.0)


def _probs(z):
    z = _san(z)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    Z = 1.0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = 1.0 / Z
    p[1:] = expz / Z
    return p


def _self_conv(p):
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f * f, FFT_SELF).real[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(p):
    c = _self_conv(p)
    return 2.0 * N * float(np.max(c))


def _update_best(p, s):
    global _global_best_score, _global_best_p
    if np.isfinite(s) and s < _global_best_score:
        _global_best_score = float(s)
        _global_best_p = p.copy()


def _grad_p(p, w):
    """Gradient of sum_k w_k*(p*p)_k w.r.t. p. Returns length-N array."""
    flip_p = p[::-1]
    fw = np.fft.rfft(w, FFT_CROSS)
    fp = np.fft.rfft(flip_p, FFT_CROSS)
    c = np.fft.irfft(fw * fp, FFT_CROSS).real
    h = c[N - 1:2 * N - 1]
    return 2.0 * h


def _obj_grad(x, alpha):
    global _global_deadline
    if time.monotonic() >= _global_deadline:
        raise _Timeout
    z = _san(x)
    p = _probs(z)
    c = _self_conv(p)
    hard = 2.0 * N * float(np.max(c))
    _update_best(p, hard)

    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(np.sum(e))
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
        J = cmax
    else:
        J = cmax + np.log(s) / alpha
        w = e / s

    gp = _grad_p(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e8, 1e8)
    return float(J), gz


def _lbfgs(z0, alpha, max_iter, deadline):
    global _global_deadline
    _global_deadline = float(deadline)
    try:
        res = scipy_minimize(
            _obj_grad, _san(z0), args=(float(alpha),),
            method="L-BFGS-B", jac=True, bounds=BOUNDS,
            options={"maxiter": max_iter, "ftol": 1e-15, "gtol": 1e-11,
                     "maxls": 40, "disp": False},
        )
        return _san(res.x)
    except _Timeout:
        return _san(z0)
    except Exception:
        return _san(z0)


def _subgrad_step(z, top_k, step, deadline):
    z = _san(z)
    p = _probs(z)
    c = _self_conv(p)
    cur = 2.0 * N * float(np.max(c))
    _update_best(p, cur)

    if top_k <= 1:
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    elif top_k >= len(c):
        w = c + 1e-15
        w = w / float(np.sum(w))
    else:
        idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros_like(c)
        wv = c[idx] + 1e-15
        w[idx] = wv
        w = w / float(np.sum(w))

    gp = _grad_p(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gnorm = float(np.linalg.norm(gz))
    if gnorm < 1e-16:
        return z, cur

    direction = -gz / gnorm
    best_z, best_s = z, cur
    for s in (step, step * 0.5, step * 0.25, step * 0.125, step * 0.0625):
        zt = z + s * direction
        zt = _san(zt)
        pt = _probs(zt)
        st = _score(pt)
        if st < best_s:
            best_s = st
            best_z = zt
            _update_best(pt, st)
    return best_z, best_s
