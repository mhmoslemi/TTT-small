def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - max(0, -mn)
    return c

def soft_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10 * (mval + 1)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:

        return mval, w
    x = (c - mval) / beta

    # where (w*q)[m] = sum_k w[k]*q[m-k]

    dot = float(np.dot(p, grad))
    return p * (grad - dot)

            gz = simplex_gradient(p, grad_p)
            gtheta = basis.T @ gz
            
            if not np.all(np.isfinite(gtheta)):
                theta = best_th.copy()
                m[:] = 0; v[:] = 0; t_adam = 0
                continue
            
            t_adam += 1
            m = b1 * m + (1-b1) * gtheta

            vh = v / (1 - b2**t_adam)

            c = full_convolve_fft(p)

            else:

    # Phase 2: Full logit-space polish at n0
    phase2_end = min(deadline, t_start + 540)

        per = min(phase2_end, time.monotonic() + 25.0)

        update(p_best, sc)

    # Phase 3: Multi-resolution extension

            print(f"Exact score: {sc_exact:.10f} (n={n})")

                best_score = sc_eval

        except:

    # Final normalization and return
    if best_p is not None:
        best_p = np.maximum(best_p, 0.0)
        s = best_p.sum()
        if s > 0:
            best_p = best_p / s
        print(f"FINAL: n={best_p.size}, score={best_score:.10f}")

    # Fallback
