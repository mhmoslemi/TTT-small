Hybrid optimization combining soft-subgradient descent with multiplicative updates
and LP-based refinement to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses temperature annealing, Polyak step sizes, and multiple diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

        M = c.max()

        g = lp_solve_direction(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.7, 0.4, 0.2, 0.1]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-13:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.25, step * 1.03)
        else:
            stall += 1
            step = max(0.004, step * 0.78)
            if stall > 80:
                noise = rng.normal(0, 0.04, n)
                a_new = best_a + noise
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    a = a_new
                    stall = 0
        if stall > 250:
            break
        if verbose and it % 30 == 0:
            print(f"      LP it={it}, score={best_score:.7f}, step={step:.4f}")

    return best_a, best_score


def optimize_soft_subgrad(a, deadline, rng, verbose=True):
    """
    Soft-subgradient descent with multiplicative updates on the simplex.
    Uses temperature annealing for the soft-max over convolution peaks.
    """
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)

    # Temperature schedule: start high (smooth) and anneal down
    T = 0.2
    T_min = 0.003
    T_decay = 0.9997
    base_lr = 0.015
    lr_decay = 0.9995

    it = 0
    stall = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        if it % 5000 == 0:
            current_score = 2.0 * n * M  # sum(a)=1
            if current_score > best_score + 0.01:
                a = best_a.copy()
                c = np.convolve(a, a, mode='full')
                M = c.max()

        # Soft-max weights over convolution values
        # Scale so that the peak gets the most weight but others contribute
        T_eff = max(T, 1e-6)
        # Shift by max to avoid overflow; use negative so peaks get highest weight
        c_neg = M - c  # 0 at peak, positive elsewhere
        w = np.exp(-c_neg / T_eff)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient: grad[i] = 2 * sum_m w[m] * a[m-i] (with a[k]=0 outside [0,n-1])
        # This is 2 * cross-correlation of w and a, indexed appropriately
        grad = 2.0 * np.correlate(w, a, mode='full')[n - 1:2 * n - 1]
        grad = np.maximum(grad, 0.0)  # ensure non-negative

        # Center the gradient (tangent to simplex)
        grad_mean = grad.mean()
        grad_c = grad - grad_mean

        # Adaptive learning rate
        # Use Polyak-like: lr ~ gap / ||grad||^2
        current_score = 2.0 * n * M
        gap = max(current_score - best_score, 0.0)
        grad_norm_sq = (grad_c ** 2).sum() + 1e-20
        lr_polyak = 2.0 * gap / grad_norm_sq if grad_norm_sq > 0 else base_lr
        lr = min(base_lr, max(0.0005, lr_polyak))

        # Multiplicative update (mirror descent with entropy)
        # a_new[i] = a[i] * exp(-lr * grad_c[i])
        # Clamp the exponent to avoid underflow
        exponent = -lr * grad_c
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        new_score = compute_objective(a_new)
        if new_score < best_score - 1e-13:
            best_a = a_new.copy()
            best_score = new_score
            stall = 0
        else:
            stall += 1
            a = a_new  # continue from new point for exploration

        T = max(T_min, T * T_decay)
        base_lr = max(0.003, base_lr * lr_decay)

        if verbose and it % 3000 == 0:
            print(f"      subgrad it={it}, best={best_score:.7f}, curr={new_score:.7f}, T={T:.4f}, lr={lr:.5f}")

        if stall > 800:
            # Reset and perturb
            a = best_a * (1.0 + rng.uniform(0, 0.03, n))
            a = np.clip(a, 0, None)
            if a.sum() > 0:
                a /= a.sum()
            T = 0.15
            base_lr = 0.015
            stall = 0

    return best_a, best_score


def optimize_combined(a, deadline, rng):
    """
    Combined optimization: alternate between soft-subgradient and LP.
    """
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)

    # Phase 1: Soft-subgradient (longer)
    t1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.55
    a, score = optimize_soft_subgrad(a, t1_deadline, rng, verbose=True)
    if score < best_score:
        best_a, best_score = a.copy(), score

    # Phase 2: LP refinement (shorter)
    if time.monotonic() < deadline - 5:
        a, score = optimize_lp(best_a, deadline, rng, verbose=True)
        if score < best_score:
            best_a, best_score = a.copy(), score

    return best_a, best_score


def objective_from_evaluator(seq):
    """Use the external evaluator to verify a sequence."""
    try:
        s = evaluate_sequence([float(x) for x in seq])
        return s
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization function. Uses soft-subgradient + LP hybrid with
    multiple diverse initializations and temperature annealing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p
    except Exception:
        pass

    # Build list of (name, initial_array) pairs
    initializations = []

    if parent is not None:
        initializations.append(('parent', parent.copy()))
        # Slightly perturbed parent
        for k in range(3):
            sigma = 0.03 * (k + 1)
            pert = parent + rng.normal(0, sigma, len(parent))
            pert = np.clip(pert, 0, None)
            if pert.sum() > 0:
                initializations.append((f'parent_s{sigma:.2f}', pert))

    # Generate diverse initializations at n=512
    n_main = len(parent) if parent is not None else 512

    def make_cosine(n):
        return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))

    def make_gaussian(n, sigma_frac=0.22):
        center = (n - 1) / 2.0
        sigma = sigma_frac * n
        return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)

    def make_gaussian_wide(n):
        center = (n - 1) / 2.0
        sigma = 0.35 * n
        return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)

    def make_parabolic(n):
        c = (n - 1) / 2.0
        return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))

    def make_bimodal(n):
        # Two bumps
        c1 = 0.3 * (n - 1)
        c2 = 0.7 * (n - 1)
        s = 0.12 * n
        return np.exp(-0.5 * ((np.arange(n) - c1) / s) ** 2) + \
               np.exp(-0.5 * ((np.arange(n) - c2) / s) ** 2)

    def make_random_smooth(n):
        raw = np.abs(rng.normal(0, 1, n))
        # Smooth it
        kernel = np.ones(5) / 5
        return np.convolve(raw, kernel, mode='same')

    inits_512 = [
        ('cos512', make_cosine(n_main)),
        ('gauss512', make_gaussian(n_main)),
        ('gaussW512', make_gaussian_wide(n_main)),
        ('par512', make_parabolic(n_main)),
        ('bimodal512', make_bimodal(n_main)),
        ('randS512', make_random_smooth(n_main)),
    ]
    initializations.extend(inits_512)

    # Try other lengths
    for n_alt in [448, 640, 768]:
        a_alt = make_cosine(n_alt)
        initializations.append((f'cos_{n_alt}', a_alt))
        a_alt2 = make_gaussian(n_alt)
        initializations.append((f'gauss_{n_alt}', a_alt2))

    # If parent is available, try sub-sampling
    if parent is not None and len(parent) > 384:
        a_sub = parent[:384].copy()
        initializations.append(('parent_384', a_sub))

    print(f"Total initializations: {len(initializations)}, budget={budget_s}s")

    # Time allocation: weight parent-based configs higher
    weights = []
    for name, _ in initializations:
        if 'parent_s0' in name or name == 'parent':
            weights.append(3.0)
        elif 'parent' in name:
            weights.append(2.0)
        elif '512' in name:
            weights.append(1.5)
        else:
            weights.append(1.0)

    total_weight = sum(weights)
    available = budget_s - 40.0
    cum_time = 0.0
    time_allocs = []
    for w in weights:
        cum_time += available * w / total_weight
        time_allocs.append(cum_time)

    print(f"Time allocated (cumulative): {[f'{t:.0f}' for t in time_allocs]}")

    for idx, (name, a_init) in enumerate(initializations):
        if time.monotonic() > deadline:
            break

        n = len(a_init)
        a_init = np.clip(a_init, 0, None).astype(np.float64)
        if a_init.sum() < 1e-10:
            continue
        a_init = a_init / a_init.sum()

        config_deadline = min(start_time + time_allocs[idx], deadline)
        remaining = config_deadline - time.monotonic()
        if remaining < 30:
            continue

        # Quick score of initial
        init_score = compute_objective(a_init)
        print(f"[{idx + 1}/{len(initializations)}] {name} (n={n}, init_score={init_score:.6f}, time={remaining:.0f}s)")

        # If initialization already beats best, update
        if init_score < best_score:
            best_score = init_score
            best_seq = a_init.copy()
            print(f"  => init already best: {best_score:.7f}")

        # Run combined optimization
        a_opt, score = optimize_combined(a_init, config_deadline, rng)

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: {best_score:.7f} (n={n}, {name})")
        else:
            print(f"  => score={score:.7f} (best: {best_score:.7f})")

    # Final extended LP polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nFinal LP polish (time left: {deadline - time.monotonic():.0f}s)...")
        polish_deadline = deadline
        final_seq, final_score = optimize_lp(best_seq.copy(), polish_deadline, rng, verbose=True)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"  After polish: {best_score:.7f}")
        else:
            # Try one more round of soft-subgrad then LP
            if time.monotonic() < deadline - 10:
                print("  Trying subgrad + LP again...")
                s1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.5
                a_s, sc_s = optimize_soft_subgrad(best_seq.copy(), s1_deadline, rng, verbose=True)
                if sc_s < best_score:
                    best_score = sc_s
                    best_seq = a_s.copy()
                    print(f"  After subgrad: {best_score:.7f}")
                if time.monotonic() < deadline - 5:
                    a_l, sc_l = optimize_lp(best_seq.copy(), deadline, rng, verbose=True)
                    if sc_l < best_score:
                        best_score = sc_l
                        best_seq = a_l.copy()
                        print(f"  After final LP: {best_score:.7f}")

    # Verify with evaluator
    if best_seq is not None:
        eval_score = objective_from_evaluator(best_seq)
        print(f"\nEvaluator score: {eval_score:.7f}")
        if np.isfinite(eval_score) and eval_score < best_score:
            best_score = eval_score

    if best_seq is None:
        best_seq = np.ones(512)
        print("WARNING: No valid sequence found, returning uniform.")

    elapsed = time.monotonic() - start_time
    print(f"\nDONE in {elapsed:.1f}s. Best score={best_score:.7f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
