def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def fft_convolve(a, b):
    """Compute full convolution using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1
    while size < n_out:
        size <<= 1
    fa = fft(a, size)
    fb = fft(b, size)
    return ifft(fa * fb).real[:n_out]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def smoothed_grad_and_obj(p, M):
    """Compute smoothed objective and its gradient.
    Objective: (1/M) * log(sum_k exp(M * r[k])) where r = conv(p,p)
    Gradient: sum_k w[k] * 2 * shifted_p
    """
    r = fft_convolve(p, p)
    r_max = r.max()
    shifted = M * (r - r_max)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    obj = r_max + np.log(exp_shifted.sum()) / M

    # Gradient: g[a] = 2 * sum_j p[j] * w[j+a] for j=0..n-1
    # This is 2 * np.correlate(w, p, 'full')[:n]
    cross = np.correlate(weights, p, mode='full')
    g = 2.0 * cross[:len(p)]

    return obj, g

def lp_direction(p, M_target):
    """Solve LP: max sum(g) s.t. conv(p, g) <= M_target, g >= 0.
    Returns g or None."""
    n = len(p)
    # Build constraint matrix: (2n-1) x n
    # A[i, j] = p[j] if i-j >= 0 and i-j < n, else 0
    # conv(p, g)[k] = sum_j p[j] * g[k-j] ... 
    # Actually conv(p, g)[k] = sum_j p[j] * g[k-j] for valid indices
    # = sum_j g[j] * p[k-j] for valid indices
    # So if we write conv(p,g) = A @ g, then A[k, j] = p[k-j] if 0<=k-j<n, else 0

    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = p[k - j_max:k - j_min + 1][::-1]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

        return compute_objective(p)

    best_p = None
    best_score = np.inf
    n_modes_list = [2, 3, 4, 6]

    for n_modes in n_modes_list:
        if time.monotonic() > deadline:
            break

        x0 = rng.uniform(-0.3, 0.3, 2 * n_modes)
        try:
            res = minimize(objective, x0, method='powell',
                          options={'maxiter': 200, 'xtol': 1e-6, 'ftol': 1e-8})
            if res.fun < best_score:
                best_score = res.fun
                # Reconstruct p
                i = np.arange(n, dtype=np.float64)
                p = np.ones(n)
                for m in range(1, n_modes + 1):
                    am = res.x[2 * (m - 1)]
                    bm = res.x[2 * (m - 1) + 1]
                    p += am * np.cos(2 * np.pi * m * i / n) + bm * np.sin(2 * np.pi * m * i / n)
                p = np.maximum(p, 0)
                if p.sum() > 1e-10:
                    p /= p.sum()
                    best_p = p.copy()
        except Exception:
            pass

        # Also try a few random starts
        for attempt in range(3):
            if time.monotonic() > deadline:
                break
            x0 = rng.uniform(-0.5, 0.5, 2 * n_modes)
            try:
                res = minimize(objective, x0, method='nelder-mead',
                              options={'maxiter': 500, 'xatol': 1e-5, 'fatol': 1e-7})
                if res.fun < best_score:
                    best_score = res.fun
                    i = np.arange(n, dtype=np.float64)
                    p = np.ones(n)
                    for m in range(1, n_modes + 1):
                        am = res.x[2 * (m - 1)]
                        bm = res.x[2 * (m - 1) + 1]
                        p += am * np.cos(2 * np.pi * m * i / n) + bm * np.sin(2 * np.pi * m * i / n)
                    p = np.maximum(p, 0)
                    if p.sum() > 1e-10:
                        p /= p.sum()
                        best_p = p.copy()
            except Exception:
                pass

    if best_p is None:
        best_p = np.ones(n) / n
        best_score = 2.0

    return best_p, best_score


def make_cosine(n):
    """Raised cosine window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic window."""
    c = (n - 1) / 2.0
    return np.maximum(1.0 - np.abs(np.arange(n) - c) / (c + 1.0), 0.01)


def make_sine(n):
    """Sine (Blackman-like) window."""
    i = np.arange(n)
    c = (n - 1) / 2.0
    return np.maximum(np.sin(np.pi * (i - c) / n + np.pi / 2), 0.01)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Combines parametric Fourier search, smoothed gradient descent,
    and LP-based optimization with multiple initializations and lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all() and p.sum() > 0.01:
            parent = p / p.sum()
            parent_score = compute_objective(parent)
            print(f"Parent: n={parent.size}, score={parent_score:.6f}")
    except Exception:
        pass

    # Phase 1: Parametric search (fast, ~60s)
    print("Phase 1: Parametric Fourier search...")
    param_candidates = []
    param_deadline = start_time + 80.0

    n_values = [256, 384, 512, 768, 1024]
    if parent is not None:
        n_values = [parent.size] + [n for n in n_values if n != parent.size]

    for n in n_values[:5]:
        if time.monotonic() > param_deadline:
            break
        print(f"  Parametric search n={n}...")
        p_param, sc_param = parametric_search(n, param_deadline, rng)
        print(f"  => n={n}: score={sc_param:.6f}")
        param_candidates.append((p_param, sc_param, n))
        if sc_param < best_score:
            best_score = sc_param
            best_seq = p_param.copy()

    # Phase 2: Smoothing gradient descent (~600s)
    print("Phase 2: Smoothed gradient descent...")
    grad_deadline = start_time + 650.0

    # Build list of starting points
    start_points = []
    if parent is not None:
        start_points.append(('parent', parent))
    for p_param, sc_param, n in param_candidates:
        if sc_param < best_score + 0.1:
            start_points.append((f'param_n{n}', p_param))

    # Add standard initializations
    for n in [256, 512, 1024]:
        for name, func in [('cos', make_cosine), ('gauss', make_gaussian),
                           ('parab', make_parabolic), ('sine', make_sine)]:
            p_init = func(n).astype(np.float64)
            p_init = p_init / p_init.sum()
            start_points.append((f'{name}_n{n}', p_init))

    # Limit to top candidates by initial score
    scored_starts = []
    for name, p in start_points:
        sc = compute_objective(p)
        scored_starts.append((sc, name, p))
    scored_starts.sort(key=lambda x: x[0])
    top_starts = scored_starts[:12]

    for sc_init, name, p_init in top_starts:
        if time.monotonic() > grad_deadline:
            break
        print(f"  Gradient: {name}, init_score={sc_init:.6f}")
        p_opt, sc_opt = optimized_gradient_descent(p_init, grad_deadline, rng=rng)
        print(f"  => {name}: score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = p_opt.copy()
            print(f"  ** NEW BEST: {best_score:.6f}")

    # Phase 3: LP optimization (~700s)
    if best_seq is not None:
        print("Phase 3: LP optimization...")
        lp_deadline = start_time + 1400.0
        n_best = len(best_seq)

        # LP on the best sequence
        print(f"  LP on best seq n={n_best}...")
        p_lp, sc_lp = lp_optimize(best_seq, lp_deadline, rng=rng)
        print(f"  => LP score={sc_lp:.6f}")
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = p_lp.copy()

        # LP on a couple other good candidates
        for sc_init, name, p_init in top_starts[1:4]:
            if time.monotonic() > lp_deadline:
                break
            if len(p_init) != n_best:
                continue
            if time.monotonic() > lp_deadline - 100:
                break
            print(f"  LP on {name}...")
            p_lp2, sc_lp2 = lp_optimize(p_init, min(lp_deadline, time.monotonic() + 200), rng=rng)
            if sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = p_lp2.copy()
                print(f"  ** NEW BEST: {best_score:.6f}")

    # Phase 4: Final exploration with remaining time
    if time.monotonic() < deadline - 30:
        print("Phase 4: Final exploration...")
        remaining = deadline - time.monotonic()

        # Try different n values with gradient + LP
        for n in [384, 640, 896]:
            if time.monotonic() > deadline - 30:
                break
            p_init = make_cosine(n).astype(np.float64) / (make_cosine(n).sum())
            sub_deadline = time.monotonic() + min(remaining * 0.2, 120.0)
            p_opt, sc_opt = optimized_gradient_descent(p_init, sub_deadline, rng=rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = p_opt.copy()
                print(f"  ** NEW BEST n={n}: {best_score:.6f}")
            elif sc_opt < best_score + 0.05 and time.monotonic() < deadline - 60:
                # Run LP on this candidate
                sub_deadline2 = time.monotonic() + min(remaining * 0.1, 60.0)
                p_lp, sc_lp = lp_optimize(p_opt, sub_deadline2, rng=rng)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = p_lp.copy()
                    print(f"  ** NEW BEST (LP) n={n}: {best_score:.6f}")

        # Try perturbation of best sequence
        if best_seq is not None and time.monotonic() < deadline - 30:
            n_best = len(best_seq)
            for attempt in range(5):
                if time.monotonic() > deadline - 30:
                    break
                noise = np.abs(rng.normal(0, 0.02, n_best))
                p_pert = best_seq * (1.0 + noise)
                p_pert = np.clip(p_pert, 0, None)
                if p_pert.sum() < 1e-10:
                    continue
                p_pert /= p_pert.sum()
                sub_dl = time.monotonic() + 30
                p_opt, sc_opt = optimized_gradient_descent(p_pert, sub_dl, rng=rng)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_seq = p_opt.copy()

    # Final: ensure valid output
    if best_seq is None:
        best_seq = make_cosine(256)
        best_score = compute_objective(best_seq)

    # Scale to ensure sum >= 0.01
    if best_seq.sum() < 0.01:
        best_seq = best_seq * 0.01 / best_seq.sum()

    # Final evaluation
    final_score = compute_objective(best_seq)
    elapsed = time.monotonic() - start_time
    print(f"\nDONE in {elapsed:.1f}s. Final score={final_score:.6f}, n={len(best_seq)}")

    # Verify with evaluate_sequence if available
    try:
        eval_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"eval
uate_sequence confirms: {eval_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
