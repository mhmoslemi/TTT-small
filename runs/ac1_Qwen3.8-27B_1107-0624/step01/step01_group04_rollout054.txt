"""
Minimize 2*n*max(autocorrelation)/sum^2 for non-negative sequences using:
1. LP-based cutting-plane block coordinate descent (large steps)
2. L-BFGS-B with softmax parameterization and log-sum-exp smoothing (fine-tuning)
3. Multi-start with diverse initializations and length exploration
"""
import time
import numpy as np
from scipy.optimize import minimize, linprog


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    L = next_power_of_two(n)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n]


def actual_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = fft_convolve(a, a)
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    return float(2.0 * n * maxc / (s * s))


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(16)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-12:
        return np.ones(a.size)
    return a / s


def greedy_normalized(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def exp_decay_normalized(n, s):
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-float(s) * idx / n)
    return normalize_nonneg(a)


def gaussian_normalized(n, s):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-float(s) * (idx - center) ** 2 / n)
    return normalize_nonneg(a)


def two_sided_normalized(n, s):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-float(s) * np.abs(idx - center) / n)
    return normalize_nonneg(a)


def mix_normalize(a, b, lam):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = max(len(a), len(b))
    if len(a) < n:
        a = np.concatenate([a, np.zeros(n - len(a))])
    if len(b) < n:
        b = np.concatenate([b, np.zeros(n - len(b))])
    return normalize_nonneg(float(lam) * a + (1.0 - lam) * b)


def scale_output(a, target_sum=10.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(16, dtype=np.float64)
        s = 16.0
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def smooth_obj_grad(x, beta):
    n = x.size
    a = softmax(x)
    c = fft_convolve(a, a)
    c = np.maximum(c, 0.0)
    bc = beta * c
    bc_max = bc.max()
    log_sum = bc_max + np.log(np.exp(bc - bc_max).sum())
    s = log_sum / beta
    w = np.exp(bc - bc_max)
    w = w / w.sum()
    f = 2.0 * n * s
    wc = float(np.dot(w, c))
    # d_l = sum_k w[k] * a[k-l]
    # Using FFT: d[l] = cross-correlation
    L = next_power_of_two(4 * n)
    v = np.zeros(2 * n - 1, dtype=np.float64)
    v[:n] = a
    rev_v = v[::-1]
    W = np.fft.rfft(w, L)
    Rv = np.fft.rfft(rev_v, L)
    conv_full = np.fft.irfft(W * Rv, L)
    d = conv_full[2 * n - 2:3 * n - 2]
    d = np.clip(d, 0.0, None)
    grad = 4.0 * n * float(beta) * a * (d - wc)
    return f, grad


def lbfgsb_phase(a_init, n, betas, budget_s):
    """Run multi-stage L-BFGS-B with annealed beta."""
    a = normalize_nonneg(a_init)
    x0 = np.log(a + 1e-30)
    x0 -= x0.mean()
    best_a = a.copy()
    best_f = actual_score(a)
    end_time = time.monotonic() + budget_s

    for beta in betas:
        if time.monotonic() >= end_time:
            break
        alpha = float(beta)

        def fun_g(x):
            f, g = smooth_obj_grad(x, alpha)
            if not np.all(np.isfinite(g)):
                g = np.zeros_like(x)
            return f, g

        remaining = max(5.0, end_time - time.monotonic())
        max_iter = max(50, int(remaining * 50))
        try:
            res = minimize(
                fun_g, x0, method='L-BFGS-B',
                bounds=[(-50.0, 50.0)] * n,
                options={'maxiter': max_iter, 'ftol': 1e-14, 'maxls': 30}
            )
            if res.x is not None:
                x_new = res.x
                if np.all(np.isfinite(x_new)):
                    a_new = softmax(x_new)
                    f_new = actual_score(a_new)
                    if f_new < best_f:
                        best_f = f_new
                        best_a = a_new.copy()
                    x0 = x_new
        except Exception:
            pass
    return best_a, best_f


def lp_improve(a_init, budget_s, n_target=None):
    """LP cutting-plane improvement."""
    a = normalize_nonneg(a_init)
    n = a.size
    if n < 2:
        return a, actual_score(a)

    S0 = np.sqrt(2.0 * n)
    f = a * S0
    cur_M = float(fft_convolve(f, f).max())
    cur_score = 2.0 * n * cur_M / (S0 * S0)
    best_f = f.copy()
    best_score = cur_score
    end_time = time.monotonic() + budget_s

    # Initial constraint rows: all lags where conv > 0.85 * max
    conv_full = fft_convolve(f, f)
    threshold = 0.85 * cur_M
    rows = list(np.where(conv_full > threshold)[0].astype(int))
    if len(rows) < 50:
        rows = list(range(0, 2 * n - 1, max(1, (2 * n - 1) // 200)))
    rows = sorted(set(rows))
    max_rows = min(2000, max(300, n))
    row_set = set(rows)

    it = 0
    while time.monotonic() < end_time and len(rows) >= 10:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        r = int(rows_arr.size)
        A = np.zeros((r, n), dtype=np.float64)
        for ii in range(r):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(r, cur_M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(2.0, end_time - time.monotonic() - 2.0)
        tl = max(5.0, min(120.0, rem * 0.3))
        try:
            res = linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None),
                          method='highs', options={'time_limit': float(tl)})
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg <= 1e-12:
            break

        # Check violations
        conv_fg = fft_convolve(f, g)
        conv_fg = np.maximum(conv_fg, 0.0)
        viol = conv_fg - cur_M
        max_viol = float(viol.max())
        tol = 1e-6 + 1e-6 * cur_M

        if max_viol > tol:
            if len(rows) < max_rows:
                order = np.argsort(viol)[-200:][::-1]
                added = 0
                for ix in order:
                    ix = int(ix)
                    if viol[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                    if added >= 100:
                        break
                if added == 0 and max_viol > 10 * tol:
                    break
                continue

        # Try step sizes
        g_scaled = g * (S0 / sg)
        accepted = None
        accepted_score = cur_score
        for step in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g_scaled
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            mc = float(fft_convolve(fn, fn).max())
            sc = 2.0 * n * mc / (S0 * S0)
            if sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()

        if accepted is None or accepted_score >= best_score:
            break

        f = accepted
        cur_M = float(fft_convolve(f, f).max())
        cur_score = 2.0 * n * cur_M / (S0 * S0)
        best_f = f.copy()
        best_score = cur_score

        if it % 10 == 0:
            print("  LP n=%d it=%d score=%.6f rows=%d" % (n, it, best_score, len(rows)))

    return normalize_nonneg(best_f), best_score


def prepare_starts(n, rng):
    starts = []
    starts.append(greedy_normalized(n))
    starts.append(power_normalized(n, 0.45))
    starts.append(power_normalized(n, 0.55))
    starts.append(power_normalized(n, 0.35))
    starts.append(exp_decay_normalized(n, 1.0))
    starts.append(exp_decay_normalized(n, 2.0))
    starts.append(gaussian_normalized(n, 1.0))
    starts.append(gaussian_normalized(n, 2.0))
    starts.append(two_sided_normalized(n, 1.5))
    starts.append(two_sided_normalized(n, 3.0))
    starts.append(mix_normalize(greedy_normalized(n), np.ones(n, dtype=np.float64), 0.5))
    starts.append(mix_normalize(greedy_normalized(n), np.ones(n, dtype=np.float64), 0.3))
    # Perturbed greedy
    p = greedy_normalized(n)
    noise = np.exp(rng.normal(0, 0.1, n))
    starts.append(normalize_nonneg(p * noise))
    noise = np.exp(rng.normal(0, 0.3, n))
    starts.append(normalize_nonneg(p * noise))
    return starts


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = actual_score(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
        n = int(a.size)
        old = best_by_n.get(n)
        if old is None or score < old[0]:
            best_by_n[n] = (float(score), a.copy())

    # Phase 1: Quick initialization
    print("Phase 1: Initialization")
    phase1_end = t0 + 30.0
    lengths = [96, 128, 160, 192, 256, 320, 384, 448, 512, 640, 768, 896, 1024]

    for n in lengths:
        if time.monotonic() >= phase1_end:
            break
        for start in prepare_starts(n, rng):
            sc = actual_score(start)
            if sc < best_score:
                record(start, sc)

    # Use height_sequence_1 if available
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                rec = []
                sc = actual_score(arr)
                record(arr, sc)
                print("  height_sequence_1: n=%d score=%.6f" % (arr.size, sc))
        except Exception:
            pass

    print("  After init: best_score=%.6f (n=%d)" % (best_score, len(best_p) if best_p is not None else 0))

    # Phase 2: LP-based improvement
    print("Phase 2: LP improvement")
    phase2_end = t0 + 0.62 * float(budget_s)

    # Sort candidates by score
    candidates = []
    for n, (sc, p) in best_by_n.items():
        candidates.append((sc, n, p))
    candidates.sort(key=lambda x: x[0])
    # Also add greedy at various lengths not yet tried
    for n in [256, 320, 384, 448, 512, 640, 768]:
        if n not in best_by_n:
            g = greedy_normalized(n)
            sc = actual_score(g)
            candidates.append((sc, n, g))
            record(g, sc)
    candidates.sort(key=lambda x: x[0])

    # LP on top candidates
    lp_done = set()
    for sc, n, p in candidates[:8]:
        if time.monotonic() >= phase2_end:
            break
        if n in lp_done:
            continue
        lp_done.add(n)
        rem = max(20.0, phase2_end - time.monotonic() - 5.0)
        alloc = min(rem * 0.4, max(30.0, 120.0))
        print("  LP n=%d (score=%.6f) alloc=%.0fs" % (n, sc, alloc))
        p_out, f_out = lp_improve(p, alloc)
        record(p_out, f_out)

    # Additional LP rounds on best
    if best_p is not None and time.monotonic() < phase2_end:
        rem = phase2_end - time.monotonic()
        if rem > 30:
            print("  LP round 2 on best (n=%d)" % len(best_p))
            p_out, f_out = lp_improve(best_p, min(rem - 5, 200.0))
            record(p_out, f_out)

        # Try LP from perturbed best
        if time.monotonic() < phase2_end - 30:
            rem = phase2_end - time.monotonic()
            if rem > 30:
                p_pert = normalize_nonneg(best_p * np.exp(rng.normal(0, 0.05, len(best_p))))
                p_out, f_out = lp_improve(p_pert, min(rem - 5, 150.0))
                record(p_out, f_out)

    print("  After LP: best_score=%.6f (n=%d)" % (best_score, len(best_p) if best_p is not None else 0))

    # Phase 3: L-BFGS-B polishing
    print("Phase 3: L-BFGS-B polishing")
    phase3_end = t0 + 0.92 * float(budget_s)

    # Collect top candidates for polishing
    polish_candidates = []
    if best_p is not None:
        polish_candidates.append(best_p.copy())
    for n, (sc, p) in sorted(best_by_n.items(), key=lambda x: x[1][0])[:3]:
        if len(p) != len(polish_candidates[0]) if polish_candidates else True:
            polish_candidates.append(p.copy())
        if len(polish_candidates) >= 4:
            break

    for i, cand in enumerate(polish_candidates[:3]):
        if time.monotonic() >= phase3_end:
            break
        rem = max(10.0, phase3_end - time.monotonic())
        ncand = len(cand)
        # Scale down for very long sequences to keep L-BFGS-B fast
        if ncand > 1024:
            alloc = min(120.0, rem * 0.5)
        elif ncand > 512:
            alloc = min(180.0, rem * 0.5)
        else:
            alloc = min(250.0, rem * 0.5)
        print("  L-BFGS-B on cand %d (n=%d, score=%.6f) alloc=%.0fs" % (i, ncand, actual_score(cand), alloc))
        betas = [5.0, 15.0, 40.0, 80.0]
        p_out, f_out = lbfgsb_phase(cand, ncand, betas, alloc)
        record(p_out, f_out)

    # Phase 4: Final polishing with more aggressive beta
    if best_p is not None and time.monotonic() < phase3_end:
        rem = phase3_end - time.monotonic()
        if rem > 20:
            p_pert = normalize_nonneg(best_p * np.exp(rng.normal(0, 0.03, len(best_p))))
            rec2 = []
            sc2 = actual_score(p_pert)
            record(p_pert, sc2)
            if sc2 < best_score + 0.005:
                print("  Final L-BFGS-B polish alloc=%.0fs" % min(rem - 5, 100.0))
                p_out, f_out = lbfgsb_phase(p_pert, len(p_pert), [30.0, 60.0, 100.0], min(rem - 5, 100.0))
                record(p_out, f_out)

    print("  After polishing: best_score=%.6f (n=%d)" % (best_score, len(best_p) if best_p is not None else 0))

    # Phase 5: Validate with evaluate_sequence
    print("Phase 5: Validation")
    final_candidates = []
    if best_p is not None:
        final_candidates.append(best_p)
    for n, (sc, p) in sorted(best_by_n.items(), key=lambda x: x[1][0])[:3]:
        if p is not None and len(p) != len(best_p):
            final_candidates.append(p)

    best_eval_score = np.inf
    best_eval_seq = None
    for cand in final_candidates:
        if time.monotonic() >= deadline:
            break
        try:
            seq = scale_output(cand, 10.0)
            ev = evaluate_sequence(seq)
            if np.isfinite(ev) and ev < best_eval_score:
                best_eval_score = float(ev)
                best_eval_seq = seq
                print("  Eval: n=%d score=%.6f" % (len(cand), ev))
        except Exception:
            pass

    if best_eval_seq is not None:
        return best_eval_seq
    return scale_output(best_p if best_p is not None else greedy_normalized(256), 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    except Exception as e:
