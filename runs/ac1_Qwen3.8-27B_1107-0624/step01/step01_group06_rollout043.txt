"""
Optimizes step function coefficients to minimize J(a) = 2n*max(a*a)/(sum a)^2
using FFT-accelerated log-sum-exp smoothed gradients with Adam optimizer,
multiple parametric initializations (power ramps, exponentials), and
local search refinement. Key insight: non-symmetric (ramp-like) sequences
can achieve J < 2 by reducing the peak autocorrelation below 1/n.
"""

import numpy as np
import time


def convolve_fft(a, fft_size=None):
    """Fast self-convolution using FFT."""
    n = len(a)
    if fft_size is None:
        size = 1
        while size < 2 * n - 1:
            size <<= 1
    else:
        size = fft_size
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/(sum a)^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_obj_grad(a, tau, fft_size=None):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    if fft_size is None:
        size = 1
        while size < 2 * n - 1:
            size <<= 1
    else:
        size = fft_size
    
    # Ensure size is large enough for cross-correlation too
    # Cross-correlation needs size >= 2n-1 as well (same as convolution)
    # But to avoid circular convolution issues, use size >= 3n-2
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip to prevent overflow
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    # Cross-correlation: cross[i] = sum_k p[k] * a[k-i]
    # = sum_j a[j] * p[i+j] (cross-correlation of a and p)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    # Gradient: dJ/da[i] = 4n*cross[i]/S^2 - 4n*Tmax/S^3
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad


def make_power_ramp(n, alpha):
    """Create a power ramp: p[i] proportional to (i+1)^alpha."""
    i = np.arange(1, n + 1)
    p = i ** alpha
    p = p / np.sum(p)
    return p * n  # Scale so sum = n (convenient for computation)


def make_exp_ramp(n, rate):
    """Create exponential ramp: p[i] proportional to rate^i."""
    i = np.arange(n)
    p = rate ** i
    p = p / np.sum(p)
    return p * n


def make_symmetric_ramp(n, alpha):
    """Create a symmetric bump: p[i] proportional to (1 + alpha * (i - mid)/mid)^2, clipped to positive."""
    mid = (n - 1) / 2.0
    x = (np.arange(n) - mid) / mid  # in [-1, 1]
    p = (1.0 + alpha * x) ** 2
    p = np.maximum(p, 0.01)
    p = p / np.sum(p)
    return p * n


def run_adam(a_init, budget_s, seed, tau_start=5.0, tau_end=0.01, lr=0.05):
    """Adam gradient descent with adaptive tau. Returns (best_array, best_J)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy().astype(np.float64)
    # Normalize so sum = n (doesn't change J, but keeps numbers in a good range)
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    start_time = time.monotonic()
    last_check = 0.0
    stall_count = 0
    stall_best = best_J
    no_improve_iters = 0
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        # Adaptive learning rate: decrease slowly
        lr_eff = lr / (1.0 + 0.1 * t / 10000.0)

            t = 0
            tau = max(1.0, tau)
            continue

        beta1, beta2, eps = 0.8, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

                    else:

                        m[:] = 0
                        v[:] = 0
                        t = 0
                    no_improve_iters = 0
    
    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)

    n = len(a_init)

            best_a = cand.copy()
            a = cand.copy()
            step = max(0.01, step * 0.9)

            step = min(0.5, step * 1.05)
