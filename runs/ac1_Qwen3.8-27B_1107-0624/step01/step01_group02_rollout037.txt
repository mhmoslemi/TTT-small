"""
Search for non-negative step-function heights minimizing the official score.

The main ingredients are:
  * fast normalized proxy scoring using self-convolution,
  * multiplicative mirror-descent on the simplex with adaptive smooth-max peaks,
  * L-BFGS smoothing of the maximum convolution coefficient,
  * LP-guided Frank-Wolfe style directions from active convolution peaks,
  * a multi-start portfolio around the provided sequence and structured seeds.
"""

import time
import numpy as np
import operator

try:
    from scipy.optimize import minimize
    from scipy.optimize import linprog
except Exception:
    minimize = None
    linprog = None

_PROBLEM = {"n": 0, "tau": 1.0}
_CACHE = {"valid": False, "n": 0, "tau": 0.0, "a": None, "c": None, "p": None}
_CBEST = {"active": False, "best_score": float("inf"), "best_y": None, "best_a": None}


def normalize_nonneg(a, target=1.0):
    """Clip to non-negative and rescale to target sum."""
    if a is None:
        return np.ones(1, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    return a * (float(target) / s)


def softmax_y(y):
    """Stable softmax of a log-vector."""
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return np.ones(1, dtype=np.float64)
    m = float(np.max(y))
    z = np.exp(y - m)
    s = float(z.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(y.size, 1.0 / y.size, dtype=np.float64)
    return z / s


def self_conv(a):
    """Linear self-convolution a * a."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)

    if n <= 1200:
        c = np.convolve(a, a)
        c = np.asarray(c, dtype=np.float64)
        c[c < 0.0] = 0.0
        return c

    need = 2 * n - 1
    m = 1
    while m < need:
        m <<= 1
    fa = np.zeros(m, dtype=np.float64)
    fa[:n] = a
    F = np.fft.rfft(fa)
    c = np.fft.irfft(F * F, m)[:need]
    c = np.asarray(c, dtype=np.float64)
    c[~np.isfinite(c)] = 0.0
    c[c < 0.0] = 0.0
    return c


def shift_corr(p, a):
    """
    Return r[i] = sum_j a[j] * p[i+j] for i=0..len(a)-1.
    This is used for gradients of weighted sums of convolution coefficients.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    m = int(p.size)
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)

    if n <= 1200:
        out = np.convolve(a, p[::-1])
        idx = (m - 1) - np.arange(n, dtype=np.int64)
        out = np.asarray(out, dtype=np.float64)[idx]
    else:
        need = m + n - 1
        size = 1
        while size < need:
            size <<= 1
        fa = np.zeros(size, dtype=np.float64)
        fa[:n] = a
        pb = np.zeros(size, dtype=np.float64)
        pb[:m] = p[::-1]
        out = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(pb, size), size)
        out = np.asarray(out, dtype=np.float64)
        idx = (m - 1) - np.arange(n, dtype=np.int64)
        out = out[idx]

    out[~np.isfinite(out)] = 0.0
    return out


def score_norm(a):
    """Scale-invariant proxy score equal to the official formula for valid inputs."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return float("inf")
    c = self_conv(a)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def official_score(seq):
    """Call the official evaluator safely."""
    try:
        if isinstance(seq, np.ndarray):
            seq = seq.tolist()
        val = evaluate_sequence(list(seq))
    except Exception:
        return float("inf")
    if not np.isfinite(val):
        return float("inf")
    return float(val)


def update_global(state, seq, off=None):
    """Update global best and a small pool of diverse best candidates."""
    if state is None:
        return
    seq = normalize_nonneg(np.asarray(seq, dtype=np.float64))
    if seq.size == 0:
        return
    if off is None:
        off = official_score(seq)
    if not np.isfinite(off):
        return

    if off < state.get("best_off", float("inf")) - 1e-12:
        state["best_off"] = float(off)
        state["best_seq"] = seq.copy()
        print(f"update_best official={off:.12f} n={seq.size}", flush=True)

    pool = state.setdefault("pool", [])
    dup = False
    for entry in pool:
        oseq = entry["seq"]
        if oseq.size == seq.size:
            if float(np.max(np.abs(oseq - seq))) < 1e-7:
                dup = True
                if off + 1e-12 < entry["off"]:
                    entry["seq"] = seq.copy()
                    entry["off"] = float(off)
                break
    if not dup:
        pool.append({"seq": seq.copy(), "off": float(off)})
        pool.sort(key=operator.itemgetter("off"))
        while len(pool) > 12:
            pool.pop()


def consider_candidate(state, local_best_seq, local_best_off, candidate, cur_off, cur_proxy):
    """Evaluate a stage candidate officially/proxy-wise and decide whether to continue."""
    candidate = normalize_nonneg(np.asarray(candidate, dtype=np.float64))
    new_off = official_score(candidate)
    update_global(state, candidate, new_off)
    new_proxy = score_norm(candidate)

    if np.isfinite(new_off) and new_off < local_best_off - 1e-12:
        local_best_seq = candidate.copy()
        local_best_off = float(new_off)

    use = False
    if np.isfinite(new_off):
        if np.isfinite(cur_off):
            use = (new_off < cur_off - 1e-12) or (new_proxy < cur_proxy - 1e-12)
        else:
            use = True
    elif np.isfinite(cur_off) is False:
        use = new_proxy < cur_proxy - 1e-12

    return use, local_best_seq, local_best_off, float(new_off), float(new_proxy)

def downsample_to(a, n):
    """Downsample by summing equal-size bins."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(a.size)
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if m <= 1:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if m == n:
        return normalize_nonneg(a)
    out = np.zeros(n, dtype=np.float64)
    for i in range(n):
        lo = (i * m) // n
        hi = ((i + 1) * m) // n
        if hi > lo:
            out[i] = float(a[lo:hi].sum())
    return normalize_nonneg(out)


def upsample_to(a, n):
    """Upsample by linear interpolation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(a.size)
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if m <= 1:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if m == n:
        return normalize_nonneg(a)
    x = np.linspace(0.0, float(m - 1), n)
    i0 = np.floor(x).astype(np.int64)
    i1 = np.minimum(i0 + 1, m - 1)
    frac = x - i0
    out = a[i0] * (1.0 - frac) + a[i1] * frac
    out = np.clip(out, 0.0, None)
    return normalize_nonneg(out)


def resize_to_length(a, n):
    """Resize a sequence to length n while preserving non-negativity and mass."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if a.size == n:
        return normalize_nonneg(a)
    if a.size < n:
        return upsample_to(a, n)
    return downsample_to(a, n)


def random_perturb(a, rng, scale=0.08):
    """Smooth/logarithmic nonnegative perturbation."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = int(a.size)
    if n < 2:
        return normalize_nonneg(a)

    a *= np.exp(rng.normal(0.0, float(scale), size=n))

    f1 = int(rng.integers(1, 6))
    ph1 = float(rng.uniform(0.0, 2.0 * np.pi))
    amp1 = float(rng.uniform(0.3, 1.2)) * float(scale)
    x = np.linspace(0.0, 1.0, n)
    a *= np.exp(amp1 * np.sin(2.0 * np.pi * f1 * x + ph1))

    wmin = max(1, n // 200)
    wmax = max(wmin + 1, n // 8)
    if n > wmin and wmax > wmin:
        w = int(rng.integers(wmin, min(wmax, n) + 1))
        if w < n:
            s = int(rng.integers(0, n - w + 1))
            block_scale = float(rng.normal(0.0, max(0.0, float(scale) * 1.5)))
            a[s:s + w] *= np.exp(block_scale)

    a = np.clip(a, 0.0, None)
    return normalize_nonneg(a)


def make_two_block(n, rng):
    """Random dense two-block seed."""
    n = int(max(2, n))
    Llo = max(2, n // 10)
    Lhi = min(n - 2, n - n // 10)
    if Lhi <= Llo:
        L = n // 2
    else:
        L = int(rng.integers(Llo, Lhi + 1))
    r = float(rng.uniform(1.0 / 4.0, 2.0 / 3.0))
    a = np.zeros(n, dtype=np.float64)
    a[:L] = r / L
    if n - L > 0:
        a[L:] = (1.0 - r) / (n - L)
    return normalize_nonneg(a)


def make_blocks(n, rng, kmax=24):
    """Piecewise-constant Dirichlet-block seed."""
    n = int(max(2, n))
    K = int(rng.integers(2, min(kmax, n) + 1))
    lengths = rng.dirichlet(np.ones(K, dtype=np.float64)) * float(n)
    lengths = np.maximum(1, lengths.astype(np.int64))
    while lengths.sum() < n:
        lengths[int(rng.integers(K))] += 1
    while lengths.sum() > n:
        i = int(np.argmax(lengths))
        if lengths[i] > 1:
            lengths[i] -= 1
        else:
            break
    masses = rng.dirichlet(0.25 * np.ones(K, dtype=np.float64))
    a = np.zeros(n, dtype=np.float64)
    pos = 0
    for L, p in zip(lengths, masses):
        L = int(L)
        if L > 0 and pos + L <= n:
            a[pos:pos + L] = float(p) / L
            pos += L
        elif L > 0:
            a[pos:n] = float(p) / max(1, n - pos)
            break
    return normalize_nonneg(a)


def make_smooth_random(n, rng, kind=0):
    """Smooth positive seeds."""
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    if kind == 0:
        base = np.zeros(n, dtype=np.float64)
        for _ in range(int(rng.integers(2, 5))):
            f = int(rng.integers(1, 8))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            amp = float(rng.uniform(0.4, 1.4))
            base += amp * np.sin(2.0 * np.pi * f * x + ph)
        base += rng.normal(0.0, 0.45, size=n)
        a = np.exp(0.85 * base)
    elif kind == 1:
        alpha = float(rng.uniform(0.2, 3.0))
        beta = float(rng.uniform(0.2, 3.0))
        a = x ** alpha * (1.0 - x + 1e-12) ** beta
    else:
        f1 = int(rng.integers(1, 8))
        f2 = int(rng.integers(1, 8))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        p2 = float(rng.uniform(0.0, 2.0 * np.pi))
        a = 1.0 + 0.9 * np.sin(2.0 * np.pi * f1 * x + p1) * np.sin(2.0 * np.pi * f2 * x + p2)
    return normalize_nonneg(np.maximum(a, 1e-300))


def make_binary_sparse(n, rng):
    """Random sparse equal-mass seed."""
    n = int(max(2, n))
    target = int(np.sqrt(n) * float(rng.uniform(0.7, 1.4)))
    target = int(np.clip(target, 1, n))
    if target == n:
        return np.full(n, 1.0 / n, dtype=np.float64)
    perm = rng.permutation(n)
    a = np.zeros(n, dtype=np.float64)
    a[perm[:target]] = 1.0 / float(target)
    return a


def make_quadratic_mod(n, rng):
    """Seed from quadratic residues modulo n, embedded linearly."""
    n = int(max(2, n))
    a_coeff = int(rng.integers(1, max(2, n)))
    b_coeff = int(rng.integers(0, max(1, n)))
    idx = np.mod((np.arange(n, dtype=np.int64) * a_coeff + b_coeff) ** 2, n)
    counts = np.bincount(idx, minlength=n).astype(np.float64)
    return normalize_nonneg(counts)


def softmax_from_coeffs(c, cap, tau):
    """p = softmax((c-cap)/tau), where cap is usually max(c)."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.zeros(0, dtype=np.float64)
    if (not np.isfinite(tau)) or tau <= 0.0:
        tau = 1e-12
    z = (c - float(cap)) / float(tau)
    z = z - float(np.max(z))
    z = np.where(z < -70.0, -70.0, z)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(c.size, 1.0 / c.size, dtype=np.float64)
    return ez / s


def refine_sequence_with_mirror(a, time_limit, rng):
    """
    Multiplicative mirror-descent on log-simplex coordinates.

    It minimizes a softmax-smoothed maximum convolution coefficient and uses
    adaptive temperature, line search, and periodic perturbations.
    """
    a = normalize_nonneg(np.asarray(a, dtype=np.float64))
    n = int(a.size)
    if n < 2 or time_limit <= 0.0:
        return a, score_norm(a)

    c0 = self_conv(a)
    M0 = float(np.max(c0)) if c0.size else 1e-12
    min_tau = max(1e-18, 1e-9 * max(M0, 1e-18))
    tau = max(min_tau, 0.20 * M0)

    y = np.log(a + 1e-300)
    y = np.asarray(y, dtype=np.float64)
    y -= float(np.mean(y))

    best_a = a.copy()
    best_sc = score_norm(a)
    start = time.monotonic()
    no_improve = 0
    iter_no = 0

    while time.monotonic() - start < max(0.0, time_limit):
        a = normalize_nonneg(softmax_y(y))
        c = self_conv(a)
        M = float(np.max(c)) if c.size else 0.0
        sc = 2.0 * n * M

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 45:
            a = random_perturb(best_a, rng, 0.11)
            y = np.log(a + 1e-300)
            y = np.asarray(y, dtype=np.float64)
            y -= float(np.mean(y))
            c = self_conv(a)
            M = float(np.max(c)) if c.size else 1e-12
            tau = max(min_tau, 0.25 * M)
            no_improve = 0
            iter_no += 1
            continue

        if iter_no % 12 == 0 and tau > min_tau:
            tau = max(min_tau, tau * 0.85)

        p = softmax_from_coeffs(c, M, tau)
        h = 2.0 * shift_corr(p, a)
        base = float(np.dot(a, h))
        if not np.isfinite(base) or base <= 1e-300:
            base = 1e-12

        delta = base - h
        if delta.size:
            scale = float(np.percentile(np.abs(delta), 95))
        else:
            scale = 0.0
        if not np.isfinite(scale) or scale <= 1e-300:
            tau = max(min_tau, tau * 0.5)
            iter_no += 1
            continue

        delta = delta / scale
        cur_sc = sc
        best_eps = 0.0
        for eps in (0.03, 0.08, 0.20, 0.55, 1.4, 3.0):
            y2 = y + float(eps) * delta
            y2 = np.clip(y2, -70.0, 70.0)
            a2 = normalize_nonneg(softmax_y(y2))
            sc2 = score_norm(a2)
            if sc2 < cur_sc - 1e-14:
                cur_sc = sc2
                best_eps = float(eps)

        if best_eps != 0.0:
            y += best_eps * delta
            y = np.clip(y, -70.0, 70.0)
            y -= float(np.mean(y))
        else:
            tau = max(min_tau, tau * 0.5)

        iter_no += 1

    return best_a, best_sc


def _smooth_func(y):
    """Smooth max objective in softmax coordinates."""
    n = int(_PROBLEM.get("n", 0))
    tau = float(_PROBLEM.get("tau", 1.0))
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size != n:
        y = y[:n] if y.size > n else np.pad(y, (0, max(0, n - y.size)))
    a = softmax_y(y)
    c = self_conv(a)
    if c.size == 0:
        return float("inf")

    max_c = float(np.max(c))
    if not np.isfinite(tau) or tau <= 0.0:
        tau = max(1e-18, 0.1 * max_c if max_c > 0 else 1e-18)

    z = (c - max_c) / tau
    z = z - float(np.max(z))
    z = np.where(z < -70.0, -70.0, z)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 1e-300:
        p = np.full(c.size, 1.0 / c.size, dtype=np.float64)
        val = max_c
    else:
        p = ez / s
        val = max_c + float(tau) * float(np.log(s))

    _CACHE["valid"] = True
    _CACHE["n"] = n
    _CACHE["tau"] = float(tau)
    _CACHE["a"] = a
    _CACHE["c"] = c
    _CACHE["p"] = p

    if _CBEST.get("active", False):
        sc = 2.0 * a.size * max_c
        if np.isfinite(sc) and sc < _CBEST.get("best_score", float("inf")) - 1e-14:
            _CBEST["best_score"] = float(sc)
            _CBEST["best_y"] = y.copy()
            _CBEST["best_a"] = a.copy()

    return float(val)


def _smooth_jac(y):
    """Gradient of the smooth max objective in softmax coordinates."""
    n = int(_PROBLEM.get("n", 0))
    tau = float(_PROBLEM.get("tau", 1.0))
    if (
        not _CACHE.get("valid", False)
        or int(_CACHE.get("n", -1)) != n
        or abs(float(_CACHE.get("tau", -1.0)) - tau) > 1e-300
    ):
        _smooth_func(y)

    a = _CACHE.get("a")
    p = _CACHE.get("p")
    if a is None or p is None:
        _smooth_func(y)
        a = _CACHE.get("a")
        p = _CACHE.get("p")

    h = 2.0 * shift_corr(p, a)
    base = float(np.dot(a, h))
    grad = a * (h - base)
    grad[~np.isfinite(grad)] = 0.0
    return grad


def refine_sequence_with_lbfgs(a, time_limit, rng):
    """Polish using L-BFGS on a softened maximum convolution coefficient."""
    a = normalize_nonneg(np.asarray(a, dtype=np.float64))
    n = int(a.size)
    if n < 2 or time_limit <= 0.0 or minimize is None:
        return a, score_norm(a)

    y = np.log(a + 1e-300)
    y = np.asarray(y, dtype=np.float64)
    y -= float(np.mean(y))

    best_a = a.copy()
    best_sc = score_norm(a)
    start = time.monotonic()

    c0 = self_conv(a)
    M0 = float(np.max(c0)) if c0.size else 1e-12
    taus = (
        0.30 * M0,
        0.10 * M0,
        0.03 * M0,
        0.008 * M0,
        0.002 * M0,
        0.0005 * M0,
    )

    maxfun_stage = int(np.clip(300000.0 / max(1, n), 80, 1200))
    maxiter_stage = int(max(30, maxfun_stage // 2))

    for r, tau in enumerate(taus):
        if time.monotonic() - start >= max(0.0, time_limit):
            break
        _PROBLEM["n"] = n
        _PROBLEM["tau"] = float(max(1e-18, tau))
        _CACHE["valid"] = False
        _cache_n = n

        _CBEST["active"] = True
        _CBEST["best_score"] = float(best_sc)
        _CBEST["best_y"] = y.copy()
        _CBEST["best_a"] = best_a.copy()

        opts = {
            "maxiter": maxiter_stage,
            "maxfun": maxfun_stage,
            "gtol": 1e-12,
            "ftol": 1e-12,
            "maxls": 40,
            "disp": False,
        }
        try:
            res = minimize(
                _smooth_func,
                y,
                jac=_smooth_jac,
                method="L-BFGS-B",
                bounds=[(-65.0, 65.0)] * n,
                options=opts,
            )
        except Exception:
            res = None

        cand_y = _CBEST.get("best_y")
        if cand_y is None and res is not None and res.x is not None:
            cand_y = res.x
        if cand_y is None:
            cand_y = y

        cand = normalize_nonneg(softmax_y(cand_y))
        sc = score_norm(cand)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = cand.copy()
            y = np.log(best_a + 1e-300)
            y = np.asarray(y, dtype=np.float64)
            y -= float(np.mean(y))

    return best_a, best_sc


def build_lp_matrix(a, rows):
    """Rows of the linear operator g -> a*g for selected convolution lags."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    rows = np.asarray(rows, dtype=np.int64).ravel()
    A = np.zeros((rows.size, n), dtype=np.float64)
    for r, k in enumerate(rows):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            js = np.arange(lo, hi, dtype=np.int64)
            A[r, lo:hi] = a[k - js]
    return A


def solve_mass_lp(A, M, n):
    """Solve maximize sum(g) subject to A g <= M, g >= 0."""
    if linprog is None:
        return None
    A = np.asarray(A, dtype=np.float64)
    if A.size == 0 or A.shape[0] == 0 or A.shape[1] != n:
        return None
    M = float(M)
    if not np.isfinite(M) or M <= 0.0:
        return None

    keep = np.any(A != 0.0, axis=1)
    if not np.all(keep):
        A = A[keep]
    if A.shape[0] == 0:
        return None

    A = A / M
    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.ones(A.shape[0], dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=bounds)
        except Exception:
            return None

    if res is not None and getattr(res, "status", -1) == 0 and getattr(res, "x", None) is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        if float(g.sum()) > 1e-12:
            return g
    return None


def refine_sequence_with_lp(a, time_limit, rng):
    """LP-guided active-peak Frank-Wolfe refinement."""
    a = normalize_nonneg(np.asarray(a, dtype=np.float64))
    n = int(a.size)
    if n < 2 or time_limit <= 0.0 or linprog is None or n > 2600:
        return a, score_norm(a)

    best_a = a.copy()
    best_sc = score_norm(a)
    start = time.monotonic()
    lp_no = 0

    while time.monotonic() - start < max(0.0, time_limit):
        c = self_conv(a)
        M = float(np.max(c)) if c.size else 0.0
        sc = 2.0 * n * M
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        if n <= 700:
            cap = min(2 * n - 1, 900)
        elif n <= 1500:
            cap = 700
        else:
            cap = 500

        if c.size:
            order = np.argsort(c)[::-1][:cap]
            close = np.flatnonzero(c >= M - max(1e-12, 1e-9 * M))
            if close.size <= cap and close.size > 0:
                rows = close
            else:
                if close.size > 0:
                    rows = np.unique(np.concatenate((order, close)))
                    if rows.size > cap:
                        rows = rows[:cap]
                else:
                    rows = order
        else:
            rows = np.empty(0, dtype=np.int64)

        A = build_lp_matrix(a, rows)
        factor = (1.00, 1.02, 1.05, 1.10)[lp_no % 4]
        g = solve_mass_lp(A, M * float(factor), n)

        if g is None:
            a = random_perturb(a, rng, 0.05)
            lp_no += 1
            continue

        Sg = float(g.sum())
        if Sg <= 1.0 + 1e-7:
            a = random_perturb(a, rng, 0.05)
            lp_no += 1
            continue

        g = g / Sg
        cur_sc = score_norm(a)
        best_cand = a.copy()
        best_csc = cur_sc

        for t in (0.005, 0.02, 0.05, 0.10, 0.20, 0.40, 0.70, 1.00):
            cand = (1.0 - float(t)) * a + float(t) * g
            cand = normalize_nonneg(cand)
            sc2 = score_norm(cand)
            if sc2 < best_csc - 1e-12:
                best_csc = sc2
                best_cand = cand.copy()

        if best_csc < cur_sc - 1e-12:
            a = best_cand
        else:
            a = random_perturb(a, rng, 0.05)

        lp_no += 1
        if lp_no > 300:
            break

    c = self_conv(best_a)
    final_sc = 2.0 * int(best_a.size) * float(np.max(c)) if c.size else score_norm(best_a)
    if final_sc < best_sc:
        best_sc = final_sc
    return best_a, best_sc


def optimize_sequence(a, time_limit, rng, state):
    """Run a portfolio of local optimizers on one sequence for a fixed budget."""
    a = normalize_nonneg(np.asarray(a, dtype=np.float64))
    if a.size < 8:
        a = resize_to_length(a, 256)
    if a.size > 4200:
        a = resize_to_length(a, 2048)

    total = float(max(0.0, time_limit))
    if total < 2.0:
        if state is not None:
            update_global(state, a, official_score(a))
        return a, official_score(a)

    start = time.monotonic()
    seq = a
    cur_off = official_score(seq)
    update_global(state, seq, cur_off)
    cur_proxy = score_norm(seq)

    local_best_seq = seq.copy()
    local_best_off = float(cur_off) if np.isfinite(cur_off) else float("inf")

    elapsed = time.monotonic() - start
    remaining = max(0.0, total - elapsed - 2.0)
    t = min(remaining, 0.42 * total)
    if t > 2.0:
        cand, _ = refine_sequence_with_mirror(seq, t, rng)
        use, local_best_seq, local_best_off, off2, pr2 = consider_candidate(
            state, local_best_seq, local_best_off, cand, cur_off, cur_proxy
        )
        if use:
            seq = cand
            cur_off = off2
            cur_proxy = pr2

    if minimize is not None:
        elapsed = time.monotonic() - start
        remaining = max(0.0, total - elapsed - 2.0)
        t = min(remaining, 0.22 * total)
        if t > 4.0:
            cand, _ = refine_sequence_with_lbfgs(seq, t, rng)
            use, local_best_seq, local_best_off, off2, pr2 = consider_candidate(
                state, local_best_seq, local_best_off, cand, cur_off, cur_proxy
            )
            if use:
                seq = cand
                cur_off = off2
                cur_proxy = pr2

    elapsed = time.monotonic() - start
    remaining = max(0.0, total - elapsed - 2.0)
    t = min(remaining, 0.15 * total)
    if t > 3.0:
        cand, _ = refine_sequence_with_mirror(seq, t, rng)
        use, local_best_seq, local_best_off, off2, pr2 = consider_candidate(
            state, local_best_seq, local_best_off, cand, cur_off, cur_proxy
        )
        if use:
            seq = cand
            cur_off = off2
            cur_proxy = pr2

    if linprog is not None and seq.size <= 2600:
        elapsed = time.monotonic() - start
        remaining = max(0.0, total - elapsed - 2.0)
        t = min(remaining, 0.12 * total)
        if t > 4.0:
            cand, _ = refine_sequence_with_lp(seq, t, rng)
            use, local_best_seq, local_best_off, off2, pr2 = consider_candidate(
                state, local_best_seq, local_best_off, cand, cur_off, cur_proxy
            )
            if use:
                seq = cand
                cur_off = off2
                cur_proxy = pr2

    elapsed = time.monotonic() - start
    remaining = max(0.0, total - elapsed - 1.0)
    t = min(remaining, 0.09 * total)
    if t > 2.0:
        cand, _ = refine_sequence_with_mirror(seq, t, rng)
        use, local_best_seq, local_best_off, off2, pr2 = consider_candidate(
            state, local_best_seq, local_best_off, cand, cur_off, cur_proxy
        )
        if use:
            seq = cand
            cur_off = off2
            cur_proxy = pr2

    final_off = official_score(seq)
    update_global(state, seq, final_off)
    if np.isfinite(final_off) and final_off < local_best_off - 1e-12:
        local_best_seq = seq.copy()
        local_best_off = float(final_off)

    return local_best_seq, local_best_off


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search entry point.

    Builds a small portfolio of sequences, refines the best one aggressively,
    then refines diversions / longer lengths, and finally returns the best
    officially evaluated sequence found within the time budget.
    """
    t0 = time.monotonic()
    budget = float(min(max(60.0, float(budget_s)), 1800.0))
    hard_deadline = t0 + max(30.0, budget - 45.0)
    rng = np.random.default_rng(seed)

    state = {"best_off": float("inf"), "best_seq": None, "pool": []}

    # Initial seed pool.
    seed_seq = load_provided()
    cands = []
    if seed_seq is not None:
        cands.append(seed_seq)
        for N in (256, 512, 1024):
            if seed_seq.size != N:
                cands.append(resize_to_length(seed_seq, N))

    for N in (256, 512, 1024):
        cands.append(make_two_block(N, rng))
        cands.append(make_blocks(N, rng))
        cands.append(make_smooth_random(N, rng, 0))
    cands.append(make_smooth_random(512, rng, 1))
    cands.append(make_smooth_random(512, rng, 2))
    cands.append(make_binary_sparse(512, rng))
    cands.append(make_quadratic_mod(512, rng))
    cands.append(np.ones(512, dtype=np.float64) / 512.0)

    fallback_seq = None
    fallback_proxy = float("inf")

    for seq in cands:
        seq = normalize_nonng(np.asarray(seq, dtype=np.float64))
        if seq.size < 2 or seq.size > 4200:
            continue
        off = official_score(seq)
        if np.isfinite(off):
            update_global(state, seq, off)
        else:
            pr = score_norm(seq)
            if np.isfinite(pr) and pr < fallback_proxy:
                fallback_proxy = pr
                fallback_seq = seq.copy()

    if state["best_seq"] is None and fallback_seq is not None:
        state["best_off"] = float(fallback_proxy)
        state["best_seq"] = fallback_seq.copy()

    if state["best_seq"] is None:
        state["best_off"] = score_norm(make_blocks(512, rng))
        state["best_seq"] = make_blocks(512, rng)

    base_budget = max(60.0, budget - 55.0)

    # Primary refinement of the current best.
    now = time.monotonic()
    rem = max(0.0, hard_deadline - now - 5.0)
    t = min(0.42 * base_budget, rem)
    if t > 8.0 and state["best_seq"] is not None:
        optimize_sequence(state["best_seq"], t, rng, state)

    # Perturb the best to escape a basin.
    if state["best_seq"] is not None:
        now = time.monotonic()
        rem = max(0.0, hard_deadline - now - 5.0)
        t = min(0.12 * base_budget, rem)
        if t > 6.0:
            pert = random_perturb(state["best_seq"], rng, 0.25)
            optimize_sequence(pert, t, rng, state)

    # If the best is not length 512, optimize a length-512 version.
    if state["best_seq"] is not None and state["best_seq"].size != 512:
        now = time.monotonic()
        rem = max(0.0, hard_deadline - now - 5.0)
        t = min(0.11 * base_budget, rem)
        if t > 6.0:
            seq512 = resize_to_length(state["best_seq"], 512)
            optimize_sequence(seq512, t, rng, state)

    # Blend the best two pool entries if possible.
    pool = state.get("pool", [])
    if len(pool) >= 2:
        s1 = pool[0]["seq"]
        s2 = pool[1]["seq"]
        if s1.size != s2.size:
            s2 = resize_to_length(s2, s1.size)
        blend = normalize_nonneg(0.5 * s1 + 0.5 * s2)
        now = time.monotonic()
        rem = max(0.0, hard_deadline - now - 5.0)
        t = min(0.09 * base_budget, rem)
        if t > 6.0:
            optimize_sequence(blend, t, rng, state)

    # Upsample to 1024 if the best is shorter.
    if state["best_seq"] is not None and state["best_seq"].size < 1024:
        now = time.monotonic()
        rem = max(0.0, hard_deadline - now - 5.0)
        t = min(0.14 * base_budget, rem)
        if t > 8.0:
            seq1024 = resize_to_length(state["best_seq"], 1024)
            optimize_sequence(seq1024, t, rng, state)

    # Upsample to 2048 if the best is shorter.
    if state["best_seq"] is not None and state["best_seq"].size < 2048:
        now = time.monotonic()
        rem = max(0.0, hard_deadline - now - 5.0)
        t = min(0.07 * base_budget, rem)
        if t > 8.0:
            seq2048 = resize_to_length(state["best_seq"], 2048)
            optimize_sequence(seq2048, t, rng, state)

    # Final polish on the current global best.
    now = time.monotonic()
    rem = max(0.0, hard_deadline - now - 4.0)
    t = min(0.06 * base_budget, rem)
    if t > 4.0 and state["best_seq"] is not None:
        optimize_sequence(state["best_seq"], t, rng, state)

    if state["best_seq"] is None:
        final_seq = make_blocks(512, rng)
    else:
        final_seq = state["best_seq"]

    # Final safety net / fallback selection.
    fallbacks = [
        final_seq,
        make_two_block(final_seq.size, rng),
        make_blocks(final_seq.size, rng),
        make_smooth_random(final_seq.size, rng, 0),
        make_binary_sparse(final_seq.size, rng),
    ]

    best_final = None
    best_final_key = float("inf")
    for seq in fallbacks:
        seq = normalize_nonng(np.asarray(seq, dtype=np.float64))
        off = official_score(seq)
        if np.isfinite(off):
            key = off
        else:
            key = score_norm(seq)
        if np.isfinite(key) and key < best_final_key:
            best_final_key = float(key)
            best_final = seq
        if np.isfinite(off):
            update_global(state, seq, off)

    if best_final is None:
        best_final = normalize_nonng(np.ones(512, dtype=np.float64))

    final_official = official_score(best_final)
    print(f"final_best official={final_official:.12f} n={best_final.size}", flush=True)

    return [float(x) for x in best_final]
