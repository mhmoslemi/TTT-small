import time

def fft_convolve(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]

    """Compute J = 2n * max(a*a) / S^2."""
    n = len(a)

    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (S * S)

def smooth_max_grad(a, tau, K=10):

    Uses top-K peaks with softmax weighting for better gradient concentration.

    c_sorted_idx = np.argsort(c)[::-1][:K]
    c_top = c[c_sorted_idx]
    c_max = c_top[0]

    p_full = np.exp(c_shifted)
    p_full /= np.sum(p_full)

    rest[c_sorted_idx] = 0

"""
Step function coefficient optimization using multi-phase adaptive smoothing,
peak-tracking gradients with top-K concentration, diverse initializations,
and coordinate-wise polish. Combines FFT-accelerated Adam with L-BFGS
refinement for the final stages.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fft_convolve(a):
    """Self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n * max(a*a) / S^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e15
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (S * S)


def smoothed_objective_and_grad(a, tau, K=5):
    """
    Compute smoothed J and gradient using log-sum-exp with top-K sharpening.
    The top-K peaks get extra weight to focus the gradient where it matters.
    """
    n = len(a)
    c = fft_convolve(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_c = np.exp(shifted)
    sum_exp = np.sum(exp_c)
    Tmax = c_max + tau * np.log(sum_exp)
    J = 2.0 * n * Tmax / (S * S)

    # Base probabilities
    p = exp_c / sum_exp

    # Top-K sharpening: re-weight to emphasize top peaks
    if K > 0 and K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
        c_top_max = c_top[0]
        p_top = np.exp((c_top - c_top_max) / (tau * 0.5))  # sharper within top-K
        p_top /= np.sum(p_top)

        # Blend: increase weight on top-K
        alpha = min(0.9, 0.5 + 0.5 * K / len(c) * 100)  # more concentration with smaller K
        p_new = p.copy()
        p_new[top_idx] = p_top * alpha + p[top_idx] * (1 - alpha)
        p = p_new / np.sum(p_new)

    # Cross-correlation for gradient
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = 4.0 * n * cross / (S * S) - 4.0 * n * Tmax / (S ** 3)
    return J, grad


def exact_max_grad(a):
    """
    Compute subgradient of the true max objective.
    Uses uniform weighting over all indices achieving (approximately) the max.
    """
    n = len(a)
    c = fft_convolve(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    # Find indices within 1% of max (to handle near-ties)
    threshold = c_max * 0.999
    active = c >= threshold
    if not np.any(active):
        active = c >= c_max

    p = np.zeros_like(c)
    p[active] = 1.0 / np.sum(active)

    Tmax = c_max
    J = 2.0 * n * Tmax / (S * S)

    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = 4.0 * n * cross / (S * S) - 4.0 * n * Tmax / (S ** 3)
    return J, grad


def run_adam(a_init, budget_s, seed, tau_start=20.0, tau_end=0.05,
             K_start=20, K_end=3, lr_start=0.008, lr_decay=0.5):
    """
    Adam gradient descent with adaptive tau and K.
    Normalizes to sum=1 at each step (since objective is scale-invariant).
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    start_time = time.monotonic()
    stall_count = 0
    stall_threshold = best_J
    last_improve_t = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1e-6))

        # Adaptive tau: exponential decay with slight slow-down at end
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        # Adaptive K: start broad, narrow down
        K = max(1, int(K_start * (K_end / K_start) ** frac))
        # Learning rate decay
        lr = lr_start / (1.0 + lr_decay * t / 3000.0)

        J, grad = smoothed_objective_and_grad(a, tau, K)

        if not np.isfinite(J) or not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m[:] = 0
            v[:] = 0
            t = 0
            tau = max(tau, 1.0)
            continue

        t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mhat = m / (1 - b1 ** t)
        vhat = v / (1 - b2 ** t)

        a = a - lr * mhat / (np.sqrt(vhat) + eps)
        a = np.maximum(a, 0.0)

        # Renormalize to sum=1
        S_new = np.sum(a)
        if S_new > 1e-10:
            a = a / S_new
        else:
            a = best_a.copy()
            continue

        # Periodic evaluation
        if t % 200 == 0 or (t - last_improve_t > 2000 and t % 100 == 0):
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall_count = 0
                stall_threshold = local_J
                last_improve_t = t
            else:
                if local_J < stall_threshold:
                    stall_threshold = local_J
                stall_count += 1

                if stall_count > 30:
                    # Try structured perturbation
                    improved = False
                    for attempt in range(5):
                        cand = best_a.copy()
                        # Option 1: block perturbation
                        if attempt < 2:
                            w = max(1, n // (50 + 20 * attempt))
                            s = int(rng.integers(0, n - w))
                            cand[s:s + w] *= np.exp(rng.normal(0, 0.3 + 0.2 * attempt, w))
                        # Option 2: global sinusoidal perturbation
                        elif attempt == 2:
                            freq = rng.uniform(0.5, 5.0)
                            phase = rng.uniform(0, 2 * np.pi)
                            pert = 1.0 + 0.1 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                            cand *= pert
                        # Option 3: edge boost
                        elif attempt == 3:
                            edge = max(1, n // 20)
                            cand[:edge] *= np.exp(rng.normal(0, 0.2, edge))
                            cand[-edge:] *= np.exp(rng.normal(0, 0.2, edge))
                        # Option 4: random Gaussian
                        else:
                            cand += rng.normal(0, 0.01 * np.mean(cand), n)

                        cand = np.maximum(cand, 0)
                        cS = np.sum(cand)
                        if cS < 1e-10:
                            continue
                        cand /= cS
                        cJ = compute_J(cand)
                        if cJ < stall_threshold:
                            a = cand
                            m[:] = 0
                            v[:] = 0
                            t = 0
                            stall_count = 0
                            improved = True
                            break

                    if not improved:
                        a = best_a.copy()
                        m[:] = 0
                        v[:] = 0
                        t = 0
                        stall_count = 0

    return best_a, best_J


def run_lbfgs(a_init, budget_s, tau=0.1, K=3):
    """
    L-BFGS-B refinement using smoothed gradient.
    Works well when the landscape is nearly convex (small tau).
    """
    n = len(a_init)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a / S

    best_a = a.copy()
    best_J = compute_J(a)

    def obj_and_grad(x):
        x_pos = np.maximum(x, 0)
        Sx = np.sum(x_pos)
        if Sx < 1e-10:
            return 1e15, np.zeros(n)
        # Don't normalize inside (let L-BFGS handle it)
        J, g = smoothed_objective_and_grad(x_pos, tau, K)
        # Project gradient: only positive direction matters
        g = np.where(x_pos < 1e-10, np.maximum(g, 0), g)
        return J, g

    # Run L-BFGS with time limit
    class TimeLimiter:
        def __init__(self, deadline):
            self.deadline = deadline

        def __call__(self, pk):
            if time.monotonic() > self.deadline:
                return True

    limiter = TimeLimiter(deadline)

    try:
        result = scipy_minimize(
            obj_and_grad, a, method='L-BFGS-B',
            jac=True, options={'maxiter': 5000, 'maxfun': 5000,
                               'ftol': 1e-12, 'gtol': 1e-8, 'iprint': -1},
            callback=limiter
        )
        if result.x is not None:
            res_a = np.maximum(result.x, 0)
            res_S = np.sum(res_a)
            if res_S > 1e-10:
                res_a /= res_S
            res_J = compute_J(res_a)
            if res_J < best_J:
                best_J = res_J
                best_a = res_a.copy()
    except Exception:
        pass

    return best_a, best_J


def run_coord_descent(a_init, budget_s, seed, n_passes=5):
    """
    Coordinate descent: for each index, try multiplicative changes.
    Focuses on indices that contribute most to the active peaks.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a / S

    best_a = a.copy()
    best_J = compute_J(a)

    # Compute sensitivity: which indices affect the max peaks most?
    c = fft_convolve(a)
    c_max = np.max(c)
    active_idx = np.where(c >= c_max * 0.999)[0]

    # For each active peak at position m, the contributing indices are i where
    # a[i] * a[m-i] is significant. Build sensitivity map.
    sensitivity = np.zeros(n)
    for m in active_idx:
        for i in range(n):
            j = m - i
            if 0 <= j < n:
                sensitivity[i] += a[i] * a[j]

    # Sort indices by sensitivity (high first)
    order = np.argsort(sensitivity)[::-1]

    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]

    improved = True
    pass_count = 0
    while improved and time.monotonic() < deadline and pass_count < n_passes * 10:
        improved = False
        pass_count += 1

        # Process in order of sensitivity (high first)
        for i in order[min(n, 500)]:  # Only top 500 most sensitive
            if time.monotonic() > deadline:
                break
            for f in factors:
                cand = a.copy()
                cand[i] *= f
                cand = np.maximum(cand, 0)
                cS = np.sum(cand)
                if cS < 1e-10:
                    continue
                cand /= cS
                cJ = compute_J(cand)
                if cJ < best_J - 1e-10:
                    best_J = cJ
                    best_a = cand.copy()
                    a = cand.copy()
                    improved = True
                    break

            # Occasionally update sensitivity
            if int(i / max(1, n // 20)) != 0 and pass_count % 5 == 0:
                c = fft_convolve(a)
                c_max = np.max(c)
                active_idx = np.where(c >= c_max * 0.999)[0]
                sensitivity = np.zeros(n)
                for m in active_idx:
                    for i2 in range(min(n, 200)):
                        j2 = m - i2
                        if 0 <= j2 < n:
                            sensitivity[i2] += a[i2] * a[j2]
                        j2b = m - (n - 1 - i2)
                        if 0 <= j2b < n:
                            sensitivity[n - 1 - i2] += a[n - 1 - i2] * a[j2b]
                order = np.argsort(sensitivity)[::-1]

    return best_a, best_J


def run_simulated_annealing(a_init, budget_s, seed, T_start=0.01, T_end=0.0001):
    """
    Simulated annealing: accept worse solutions with decreasing probability.
    Uses block multiplicative perturbations.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a / S

    best_a = a.copy()
    best_J = compute_J(a)

    start_time = time.monotonic()
    total_iters = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1e-6))
        T = T_start * (T_end / T_start) ** frac

        # Generate candidate
        cand = a.copy()
        # Block perturbation
        w = int(rng.integers(max(1, n // 200), max(2, n // 10)))
        s = int(rng.integers(0, n - w))
        cand[s:s + w] *= np.exp(rng.normal(0, 0.1, w))
        # Single element perturbation
        if rng.random() < 0.3:
            idx = int(rng.integers(0, n))
            cand[idx] *= np.exp(rng.normal(0, 0.3))

        cand = np.maximum(cand, 0)
        cS = np.sum(cand)
        if cS < 1e-10:
            continue
        cand /= cS

        cJ = compute_J(cand)
        total_iters += 1

        delta = cJ - compute_J(a)  # approximate (could cache)
        # Actually, let's compare to current, not best
        current_J = compute_J(a)
        delta = cJ - current_J

        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = cand.copy()

    return best_a, best_J


def make_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []

    # Parent (normalized)
    p = parent.copy()
    p = np.maximum(p, 0)
    S = np.sum(p)
    if S > 0:
        p /= S
    inits.append(("parent", p))

    # Uniform
    inits.append(("uniform", np.ones(n) / n))

    # Gaussian (various widths)
    x = np.arange(n) - (n - 1) / 2.0
    for sigma_frac in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-x * x / (2.0 * (sigma_frac * n) ** 2))
        g /= np.sum(g)
        inits.append((f"gauss_{sigma_frac}", g))

    # Cosine windows
    for freq in [1, 2, 3]:
        c = 1.0 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n)
        c = np.maximum(c, 0.001)
        c /= np.sum(c)
        inits.append((f"cos_{freq}", c))

    # Blackman-like window
    w = 0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / n) + 0.08 * np.cos(4 * np.pi * np.arange(n) / n)
    w = np.maximum(w, 0.001)
    w /= np.sum(w)
    inits.append(("blackman", w))

    # Edge-concentrated
    edge = np.ones(n)
    edge[:n // 4] *= 2.0
    edge[-(n // 4):] *= 2.0
    edge /= np.sum(edge)
    inits.append(("edge2x", edge))

    # Center dip
    dip = np.ones(n)
    mid = n // 2
    dip[mid - n // 5:mid + n // 5] *= 0.3
    dip /= np.sum(dip)
    inits.append(("center_dip", dip))

    # Random (log-normal)
    for sigma in [0.3, 0.7, 1.0]:
        r = np.exp(rng.normal(0, sigma, n))
        r /= np.sum(r)
        inits.append((f"rand_{sigma}", r))

    # Multi-peak
    npeaks = 3
    peaks = rng.uniform(0.1, 0.9, npeaks) * n
    mp = np.zeros(n)
    for pidx in peaks:
        mp += np.exp(-(np.arange(n) - pidx) ** 2 / (2 * (n / 20) ** 2))
    mp = np.maximum(mp, 1e-6)
    mp /= np.sum(mp)
    inits.append(("multipeak", mp))

    # Parent with slight noise
    noisy = p * np.exp(rng.normal(0, 0.1, n))
    noisy = np.maximum(noisy, 0)
    noisy /= np.sum(noisy)
    inits.append(("parent_noisy", noisy))

    # Parent with edge modification
    edge_mod = p.copy()
    edge_mod[:n // 10] *= 1.5
    edge_mod[-(n // 10):] *= 1.5
    edge_mod[n // 4:3 * n // 4] *= 0.7
    edge_mod = np.maximum(edge_mod, 0)
    edge_mod /= np.sum(edge_mod)
    inits.append(("parent_edge", edge_mod))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Score diverse initializations, select top candidates
    Phase 2: Adam gradient descent with adaptive smoothing
    Phase 3: L-BFGS-B refinement
    Phase 4: Coordinate descent polish
    Phase 5: Final simulated annealing
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"n = {n}")

    # Initial score
    best_global = parent.copy()
    S_parent = np.sum(best_global)
    if S_parent < 1e-10:
        best_global = np.ones(n)
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score (parent): {best_global_score:.6f}")

    # Generate and score initializations
    inits = make_initializations(n, parent, rng)
    scored_inits = []
    for name, arr in inits:
        J = compute_J(arr)
        scored_inits.append((name, J, arr))

    scored_inits.sort(key=lambda x: x[1])
    print("\nInitializations ranked:")
    for name, J, _ in scored_inits[:8]:
        print(f"  {name:20s}: J = {J:.6f}")

    # Phase 1: Quick Adam on top 5 initializations (15% of time)
    phase1_time = (total_deadline - time.monotonic()) * 0.15
    per_init = phase1_time / 5.0

    phase1_results = []
    for i in range(min(5, len(scored_inits))):
        if time.monotonic() > total_deadline - 30:
            break
        name, init_J, init_arr = scored_inits[i]
        alloc = min(per_init, total_deadline - time.monotonic() - 20)
        if alloc < 5:
            break
        print(f"\nPhase 1: Adam on '{name}' ({alloc:.1f}s)")
        res_a, res_J = run_adam(
            init_arr, alloc, seed=seed + i * 1000,
            tau_start=15.0, tau_end=0.5, K_start=30, K_end=5,
            lr_start=0.01
        )
        val = float(evaluate_sequence(res_a.tolist()))
        print(f"  Result: local={res_J:.6f}, verified={val:.6f}")
        phase1_results.append((res_a, min(res_J, val)))
        if val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")

    # Phase 2: Long Adam on best candidates (45% of remaining time)
    remaining = total_deadline - time.monotonic()
    phase2_time = remaining * 0.45

    # Pick top candidates from phase 1 + parent
    candidates = [(a, j) for a, j in phase1_results]
    candidates.append((best_global, best_global_score))

    # Deduplicate and sort
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:3]

    phase2_results = []
    for i, (cand_a, cand_J) in enumerate(top_candidates):
        if time.monotonic() > total_deadline - 30:
            break
        alloc = phase2_time / (len(top_candidates) - i)
        alloc = min(alloc, total_deadline - time.monotonic() - 20)
        if alloc < 10:
            break
        print(f"\nPhase 2: Long Adam on candidate {i} ({alloc:.1f}s)")
        res_a, res_J = run_adam(
            cand_a, alloc, seed=seed + 50000 + i * 777,
            tau_start=5.0, tau_end=0.05, K_start=15, K_end=2,
            lr_start=0.005, lr_decay=0.8
        )
        val = float(evaluate_sequence(res_a.tolist()))
        print(f"  Result: local={res_J:.6f}, verified={val:.6f}")
        phase2_results.append((res_a, min(res_J, val)))
        if val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")

    # Phase 3: L-BFGS refinement (15% of remaining time)
    remaining = total_deadline - time.monotonic()
    phase3_time = remaining * 0.15

    all_results = phase1_results + phase2_results
    if best_global_score < min(j for _, j in all_results):
        all_results.append((best_global, best_global_score))
    all_results.sort(key=lambda x: x[1])

    for i in range(min(2, len(all_results))):
        if time.monotonic() > total_deadline - 20:
            break
        alloc = phase3_time / 2
        alloc = min(alloc, total_deadline - time.monotonic() - 15)
        if alloc < 5:
            break
        cand_a, cand_J = all_results[i]
        print(f"\nPhase 3: L-BFGS on candidate {i} ({cand_J:.6f}) ({alloc:.1f}s)")
        res_a, res_J = run_lbfgs(cand_a, alloc, tau=0.05, K=2)
        val = float(evaluate_sequence(res_a.tolist()))
        print(f"  Result: local={res_J:.6f}, verified={val:.6f}")
        if val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")

    # Phase 4: Coordinate descent polish (15% of remaining time)
    remaining = total_deadline - time.monotonic()
    phase4_time = remaining * 0.15
    if phase4_time > 10:
        alloc = min(phase4_time, total_deadline - time.monotonic() - 10)
        print(f"\nPhase 4: Coordinate descent ({alloc:.1f}s)")
        res_a, res_J = run_coord_descent(best_global, alloc, seed=seed + 99999)
        val = float(evaluate_sequence(res_a.tolist()))
        print(f"  Result: local={res_J:.6f}, verified={val:.6f}")
        if val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")

    # Phase 5: Final short Adam + SA on global best (remaining time)
    remaining = total_deadline - time.monotonic() - 5
    if remaining > 10:
        # Short final Adam with very small tau
        final_alloc = remaining * 0.6
        print(f"\nPhase 5a: Final Adam ({final_alloc:.1f}s)")
        res_a, res_J = run_adam(
            best_global, final_alloc, seed=seed + 123456,
            tau_start=1.0, tau_end=0.01, K_start=5, K_end=1,
            lr_start=0.002, lr_decay=1.0
        )
        val = float(evaluate_sequence(res_a.tolist()))
        print(f"  Result: local={res_J:.6f}, verified={val:.6f}")
        if val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")

        # Simulated annealing on best
        remaining = total_deadline - time.monotonic() - 5
        if remaining > 5:
            print(f"\nPhase 5b: Simulated annealing ({remaining:.1f}s)")
            res_a, res_J = run_simulated_annealing(
                best_global, remaining, seed=seed + 789,
                T_start=0.005, T_end=0.0001
            )
            val = float(evaluate_sequence(res_a.tolist()))
            print(f"  Result: local={res_J:.6f}, verified={val:.6f}")
            if val < best_global_score:
                best_global = res_a.copy()
                best_global_score = val
                print(f"  *** NEW BEST: {val:.6f} ***")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_global_score:.6f}")
    print(f"{'='*50}")

    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
