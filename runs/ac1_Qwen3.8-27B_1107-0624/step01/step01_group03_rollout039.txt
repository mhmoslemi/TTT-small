For coordinate descent, the key insight is: given that we want to minimize max_k (s*s)[k] / (sum(s))^2 * 2n, and we're changing one coordinate at a time, the optimal value of s[i] (given all other coordinates) can be found by a 1D minimax problem.

Specifically, if we fix all coordinates except s[i] = x, then:
- (s*s)[k] = f_k(x) = 2*x*s[k-i] + x^2*[k=2i] + c_k for k in [i, i+n-1]
  where c_k is the contribution from all other terms.
  For k not in [i, i+n-1], (s*s)[k] = c_k (unaffected).

- sum(s) = S + x where S = sum of all other coordinates.

We want to minimize: 2n * max_k f_k(x) / (S+x)^2

def _lp_direction_min_peak(s, time_limit=3.0):
    """LP: min t s.t. (s*b)[k] <= t, sum(b) = sum(s), b >= 0."""
    n = len(s)
    target_sum = s.sum()
    
    # Variables: b[0],...,b[n-1], t
    # Minimize t
    # Constraints: (s*b)[k] - t <= 0 for k=0,...,2n-2
    #              sum(b) = target_sum
    #              b[i] >= 0
    
    # Build A_ub for inequality constraints: (s*b)[k] - t <= 0
    A_conv = np.zeros((2*n-1, n))
    for j in range(n):
        A_conv[j:j+n, j] = s
    
    # Full A_ub: [A_conv, -ones] * [b; t] <= 0
    A_ub = np.hstack([A_conv, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(b) = target_sum
    A_eq = np.hstack([np.ones(1, n), [[0.0]]])
    b_eq = np.array([target_sum])
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success and res.x is not None:
            b = np.maximum(res.x[:n], 0.0)
            if b.sum() > 1e-10:
                return b / b.sum()
    except Exception:
        pass
    return None

                best_sc = sc
                best_s = s.copy()
    if best_s is None:
        best_s = np.ones(n) / n
    return best_s


def _iaa_refine(s, iters=50, deadline=None):
    """Iterative Amplitude Adjustment to flatten the power spectrum."""
    n = len(s)
    best_s = s.copy()
    best_score = _fast_score(s)

    for it in range(iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        S = np.fft.fft(s)
        mag = np.abs(S)
        target_mag = np.full(n, mag.mean())
        S_new = target_mag * np.exp(1j * np.angle(S))
        s_new = np.real(np.fft.ifft(S_new))
        s_new = np.maximum(s_new, 0)
        if s_new.sum() < 1e-10:
            break
        s_new = s_new / s_new.sum()

        sc = _fast_score(s_new)
        if sc < best_score:
            best_score = sc
            best_s = s_new.copy()
        s = s_new

    return best_s, best_score


def _coord_descent(s, max_sweeps=100, deadline=None, rng=None):
    """Coordinate descent with incremental convolution update."""
    n = len(s)
    if rng is None:
        rng = np.random.default_rng()

    conv = np.convolve(s, s)
    total = s.sum()
    best_s = s.copy()
    best_score = 2.0 * n * conv.max() / (total * total) if total > 1e-15 else float('inf')
    no_imp_sweeps = 0

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        order = rng.permutation(n)

        for i in order:
            s_i = s[i]
            if s_i < 1e-12:
                candidates = [0.05 * total / n, 0.1 * total / n, 0.2 * total / n]
            else:
                candidates = [0.0, 0.2 * s_i, 0.5 * s_i, 0.8 * s_i, 1.2 * s_i, 1.5 * s_i, 2.0 * s_i]

            best_val = s_i
            best_obj = 2.0 * n * conv.max() / (total * total)

            for x in candidates:
                delta = x - s_i
                if abs(delta) < 1e-18:
                    continue

                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)

                idx_lo = max(0, k_start - i)
                idx_hi = min(n - 1, k_end - i)

                if idx_hi >= idx_lo:
                    new_affected = conv[k_start:k_end + 1].copy()
                    new_affected += 2.0 * delta * s[idx_lo:idx_hi + 1]
                    if k_start <= 2 * i <= k_end:
                        new_affected[2 * i - k_start] += delta * delta
                    aff_max = float(new_affected.max())
                else:
                    aff_max = 0.0

                unaff_max = 0.0
                if k_start > 0:
                    unaff_max = max(unaff_max, float(conv[:k_start].max()))
                if k_end + 1 < 2 * n - 1:
                    unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

                new_max = max(aff_max, unaff_max)
                new_total = total + delta
                if new_total < 1e-12:
                    continue
                obj = 2.0 * n * new_max / (new_total * new_total)

                if obj < best_obj:
                    best_obj = obj
                    best_val = x

            if best_val != s_i:
                delta = best_val - s_i
                k_start = max(0, i)
                k_end = min(2 * n - 2, i + n - 1)
                idx_lo = max(0, k_start - i)
                idx_hi = min(n - 1, k_end - i)
                if idx_hi >= idx_lo:
                    conv[k_start:k_end + 1] += 2.0 * delta * s[idx_lo:idx_hi + 1]
                if k_start <= 2 * i <= k_end:
                    conv[2 * i] += delta * delta
                s[i] = best_val
                total += delta

        if total < 1e-15:
            break
        cur_score = 2.0 * n * conv.max() / (total * total)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_s = s.copy()
            no_imp_sweeps = 0
        else:
            no_imp_sweeps += 1
            if no_imp_sweeps > 15:
                break

    return best_s, best_score


def _build_conv_matrix(s):
    """Build (2n-1) x n matrix A where A[k,j] = s[k-j] if valid."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def _lp_min_peak_direction(s, time_limit=3.0):
    """LP: min t s.t. (s*b)[k] <= t for all k, sum(b)=sum(s), b>=0. Returns normalized b."""
    n = len(s)
    target_sum = s.sum()
    A_conv = _build_conv_matrix(s)

    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([target_sum])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x[:n], 0.0)
            if b.sum() > 1e-10:
                return b / b.sum()
    except Exception:
        pass
    return None


def _lp_max_sum_direction(s, M, time_limit=3.0):
    """LP: max sum(b) s.t. (s*b)[k] <= M for all k, b>=0. Returns normalized b."""
    n = len(s)
    A_conv = _build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_conv, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-10:
                return b / b.sum()
    except Exception:
        pass
    return None


def _lp_local_search(s, max_iters=100, deadline=None, rng=None):
    """LP-guided local search alternating between min-peak and max-sum LPs."""
    n = len(s)
    if rng is None:
        rng = np.random.default_rng()

    best_s = s.copy()
    best_score = _fast_score(s)
    no_imp = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Alternate LP formulations
        if it % 2 == 0:
            b = _lp_min_peak_direction(s, time_limit=2.5)
        else:
            M = np.max(np.convolve(s, s))
            b = _lp_max_sum_direction(s, M, time_limit=2.5)

        if b is None:
            break

        best_t = 0.0
        best_ts = best_score
        for t in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]:
            cand = (1.0 - t) * s + t * b
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-10:
                continue
            cand = cand / cand.sum()
            ts = _fast_score(cand)
            if ts < best_ts - 1e-12:
                best_ts = ts
                best_t = t

        if best_t == 0.0:
            no_imp += 1
            if no_imp > 4:
                # Perturb to escape
                scale = 0.05 + 0.05 * no_imp
                s = s * (1.0 + scale * rng.standard_normal(n))
                s = np.maximum(s, 0.0)
                if s.sum() < 1e-10:
                    break
                s = s / s.sum()
                if no_imp > 12:
                    break
            continue

        s = (1.0 - best_t) * s + best_t * b
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        no_imp = 0

        cur_score = _fast_score(s)
        if cur_score < best_score:
            best_score = cur_score
            best_s = s.copy()

    return best_s, best_score


def _parameterized_inits(n, rng):
    """Generate diverse parameterized initial sequences."""
    results = []

    # Uniform
    s = np.ones(n) / n
    results.append(s.copy())

    # Linear ramps
    for beta in [-0.9, -0.5, 0.0, 0.5, 0.9]:
        x = np.linspace(-1, 1, n)
        s = 1.0 + beta * x
        s = np.maximum(s, 0.01)
        s = s / s.sum()
        results.append(s.copy())

    # Exponential
    for alpha in [-4.0, -2.0, -1.0, 1.0, 2.0, 4.0]:
        s = np.exp(alpha * np.arange(n) / (n - 1))
        s = s / s.sum()
        results.append(s.copy())

    # Quadratic (Gaussian-like)
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.3, 0.5]:
            x = np.arange(n) / (n - 1)
            s = np.exp(-((x - center) ** 2) / (2 * width ** 2))
            s = s / s.sum()
            results.append(s.copy())

    # Two-level
    for frac in [0.3, 0.5, 0.7]:
        k = int(frac * n)
        for ratio in [0.5, 2.0]:
            s = np.ones(n)
            s[:k] = ratio
            s = s / s.sum()
            results.append(s.copy())

    # Spectral inits
    for _ in range(5):
        s = _spectral_init(n, rng, n_tries=10)
        results.append(s)

    # Random
    for _ in range(3):
        s = rng.random(n) + 0.05
        s = s / s.sum()
        results.append(s.copy())

    return results


def _hybrid_local_search(s, deadline, rng):
    """Alternating IAA + coordinate descent + LP local search."""
    best_s = s.copy()
    best_score = _fast_score(s)

    phase = 0
    while time.monotonic() < deadline - 3:
        if phase % 3 == 0:
            # IAA phase
            s, sc = _iaa_refine(s, iters=15, deadline=deadline - 3)
            if sc < best_score:
                best_score = sc
                best_s = s.copy()
        elif phase % 3 == 1:
            # Coordinate descent
            s, sc = _coord_descent(s, max_sweeps=40, deadline=deadline - 3, rng=rng)
            if sc < best_score:
                best_score = sc
                best_s = s.copy()
        else:
            # LP phase
            if time.monotonic() < deadline - 20:
                s, sc = _lp_local_search(s, max_iters=15, deadline=deadline - 3, rng=rng)
                if sc < best_score:
                    best_score = sc
                    best_s = s.copy()

        phase += 1

    return best_s, best_score


def propose_candidate():
    """
    Hybrid local search to minimize 2n*max(conv(s,s))/(sum(s))^2.
    Combines spectral initialization, IAA refinement, coordinate descent,
    and LP-guided direction finding with multiple sequence lengths.
    """
    rng = np.random.default_rng(42)
    deadline = time.monotonic() + 1760.0

    global_best_score = float('inf')
    global_best_seq = None

    # Try to load height_sequence_1
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}, sum={h.sum():.4f}")
            sc_h = _fast_score(h / h.sum())
            print(f"  Rescaled score: {sc_h:.6f}")
    except Exception:
        pass

    # Sequence lengths to try with time allocation
    lengths = [(256, 500), (512, 500), (1024, 500), (2048, 300)]

    for n, alloc_time in lengths:
        if time.monotonic() > deadline - 50:
            break

        remaining = deadline - time.monotonic()
        time_budget = min(alloc_time, remaining - 30)
        if time_budget < 40:
            break

        t_start = time.monotonic()
        print(f"\n=== n={n}, time_budget={time_budget:.0f}s ===")

        # Generate initializations
        inits = _parameterized_inits(n, rng)

        # Add height_sequence_1
        if hseq is not None:
            x_old = np.linspace(0, 1, hseq.size)
            x_new = np.linspace(0, 1, n)
            h_r = np.interp(x_new, x_old, hseq)
            h_r = np.maximum(h_r, 0.0)
            if h_r.sum() > 0.01:
                h_r = h_r / h_r.sum()
                inits.append(h_r)

        # Score and sort
        scored = []
        for s in inits:
            sc = _fast_score(s)
            scored.append((sc, s))
        scored.sort(key=lambda x: x[0])

        print(f"  {len(scored)} inits, best={scored[0][0]:.6f}, "
              f"5th best={scored[min(4, len(scored)-1)][0]:.6f}")

        # Run hybrid local search on top candidates
        n_cands = min(5, len(scored))
        time_per_cand = time_budget / (n_cands + 1)

        for idx in range(n_cands):
            if time.monotonic() > deadline - 40:
                break

            sc0, s0 = scored[idx]
            cand_deadline = min(t_start + (idx + 1) * time_per_cand, deadline - 40)
            time_left = cand_deadline - time.monotonic()
            if time_left < 5:
                continue

            print(f"  Cand {idx}: score={sc0:.6f}, time={time_left:.0f}s")

            best_s, best_sc = _hybrid_local_search(s0, cand_deadline, rng)

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** New global best: {best_sc:.6f} (n={n}, cand={idx})")

        # Random restarts with IAA + coord descent
        restart_time = deadline - 60
        while time.monotonic() < min(t_start + time_budget, restart_time):
            s0 = rng.random(n) + 0.1
            s0 = s0 / s0.sum()
            s0, sc0 = _iaa_refine(s0, iters=15, deadline=restart_time)
            if sc0 >= global_best_score * 0.95:
                s0, sc0 = _coord_descent(s0, max_sweeps=30, deadline=restart_time, rng=rng)
            if sc0 < global_best_score:
                best_s, best_sc = _lp_local_search(s0, max_iters=20,
                                                    deadline=min(t_start + time_budget, restart_time),
                                                    rng=rng)
                if best_sc < global_best_score:
                    global_best_score = best_sc
                    global_best_seq = best_s.copy()
                    print(f"  ** Restart: new best {best_sc:.6f}")

    # Final: spend remaining time on the best sequence with LP
    if global_best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final LP polish, time={deadline - time.monotonic():.0f}s ===")
        best_s, best_sc = _lp_local_search(global_best_seq, max_iters=50,
                                           deadline=deadline - 5, rng=rng)
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()
            print(f"  ** Final polish: {best_sc:.6f}")

    if global_best_seq is None:
        global_best_seq = np.ones(256) / 256.0

    # Verify with actual evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_score = float(evaluate_sequence(result))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(global_best_seq)}")

    return result
