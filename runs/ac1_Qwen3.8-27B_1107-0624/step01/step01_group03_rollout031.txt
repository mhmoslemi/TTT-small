```python
def compute_pressure(w, s):
    """Compute pressure[i] = sum_k w[k]*s[k-i] for valid k-i using FFT."""
    n = len(s)

    U = np.fft.rfft(s, N)
    corr = np.fft.irfft(V * np.conj(U), N)

    n = len(s)
```

```python
def greedy_binary_init(n, rng):
    """Greedy construction of a binary sequence with low peak convolution."""
    # We want to choose w positions from {0, ..., n-1} to minimize max pairwise sum count
    w = n // 2  # target weight
    
    # Greedy: add positions one by one, choosing the one that minimizes the increase in max
    positions = []
    sums_count = np.zeros(2*n-1)  # counts of pairwise sums
    max_sum = 0
    
    for _ in range(w):
        best_pos = -1
        best_max = float('inf')
        
        for pos in range(n):
            if pos in positions:
                continue
            # If we add pos, the new sums are pos + p for p in positions ∪ {pos}
            new_sums = sums_count.copy()
            for p in positions:
                new_sums[pos + p] += 1
            new_sums[2*pos] += 1  # pos + pos
            
            cur_max = np.max(new_sums)
            if cur_max < best_max:
                best_max = cur_max
                best_pos = pos
        
        positions.append(best_pos)
        # Update sums_count
        for p in positions[:-1]:
            sums_count[best_pos + p] += 1
        sums_count[2*best_pos] += 1
    
    s = np.zeros(n)
    for pos in positions:
        s[pos] = 1.0
    s = s / s.sum()
    return s
```

For each candidate position `pos`, I need to compute the max of (sums_count + contributions from pos). The contributions from pos are: for each existing position p, add 1 to sums_count[pos+p], and add 1 to sums_count[2*pos].

So the new max is max(sums_count + delta) where delta is a sparse vector with at most w+1 non-zero entries.

I can compute this as: max(max(sums_count[non-affected]), max(sums_count[affected] + delta[affected])).

The affected indices are {pos + p : p in positions} ∪ {2*pos}. That's at most w+1 indices.

So for each candidate, the computation is O(w) to check the affected indices, plus O(1) for the global max of unaffected indices (if I maintain it).

Total: O(n * w) per position added, O(w * n * w) = O(n * w^2) total. For n=256, w=128: about 4M operations. Fine.

Let me implement this more efficiently:

```python

    w = n // 2
    positions = []
    sums_count = np.zeros(2*n-1, dtype=np.int32)
    global_max = 0
    
    for step in range(w):
        best_pos = -1
        best_max = float('inf')
        
        for pos in range(n):
            if pos in positions:
                continue
            
            # Compute new max if we add pos
            # Affected indices: {pos + p for p in positions} ∪ {2*pos}
            affected = set()
            for p in positions:
                affected.add(pos + p)
            affected.add(2 * pos)
            
            # New values at affected indices
            new_max = 0
            for idx in affected:
                if idx < 2*n-1:
                    candidate = sums_count[idx] + 1
                    if candidate > new_max:
                        new_max = candidate

            # But we need to be careful: the old global_max might be at an affected index
            # For simplicity, just compute the overall max
            # Actually, the new max is max(global_max (if not at affected), new_max)
            # But global_max might decrease if it was at an affected index... no, we're only adding.
            # So new overall max = max(global_max, new_max)
            # Wait, global_max is the max of the OLD sums_count. After adding, the affected indices increase by 1.
            # So new max = max(max of unaffected, max of affected + 1)
            # max of unaffected <= global_max (since we removed some from the "unaffected" set)
            # Actually no: max of unaffected = global_max if the argmax is not in affected, else might be less.
            # For simplicity and correctness:
            overall_new_max = max(global_max, new_max)

            # Let's just be conservative and use this.
            
            if overall_new_max < best_max:
                best_max = overall_new_max
                best_pos = pos
        
        # Add best_pos
        positions.append(best_pos)
        for p in positions[:-1]:
            idx = best_pos + p
            if idx < 2*n-1:
                sums_count[idx] += 1
                if sums_count[idx] > global_max:
                    global_max = sums_count[idx]
        idx = 2 * best_pos
        if idx < 2*n-1:
            sums_count[idx] += 1
            if sums_count[idx] > global_max:
                global_max = sums_count[idx]
    
    s = np.zeros(n)
    for pos in positions:
        s[pos] = 1.0
    s = s / s.sum()
    return s
```
