"""
Step function coefficient optimization minimizing J(a) = 2n*max(a*a)/(sum(a))^2.
Uses two-phase FFT-accelerated gradient descent (Adam) with log-sum-exp smoothing,
L-BFGS-B refinement, coordinate descent polish, and multi-start strategy.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def convolve_full(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/(sum(a))^2."""
    n = len(a)
    c = convolve_full(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_J_grad(a, tau):
    """Compute log-sum-exp smoothed objective and gradient via FFT."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    shifted = np.clip(shifted, -60, 60)
    e = np.exp(shifted)
    se = np.sum(e)
    if se < 1e-300:
        return 1e15, np.zeros(n)

    Tmax = c_max + tau * np.log(se)
    p = e / se
    J = 2.0 * n * Tmax / (S * S)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad


def make_initializations(n, rng):
    """Generate diverse initial sequences."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    inits['ones'] = np.ones(n)

    for sf in [0.08, 0.15, 0.25, 0.4]:
        sigma = sf * n
        inits[f'gauss_{sf}'] = np.exp(-((x - mid) ** 2) / (2 * sigma * sigma))

    for freq in [1, 2, 3, 5, 8]:
        inits[f'cos_{freq}'] = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x / n)

    for sigma in [0.2, 0.4, 0.6, 1.0]:
        inits[f'rand_{sigma}'] = np.exp(rng.normal(0, sigma, n))

    inits['taper'] = 1.0 - 0.4 * np.abs(x - mid) / max(mid, 1.0)
    inits['edge'] = 0.3 + 0.7 * np.abs(x - mid) / max(mid, 1.0)
    inits['quadratic'] = 1.0 + 0.3 * (1.0 - (x - mid) / max(mid, 1.0)) ** 2

    # Spectral initializations: flat-magnitude FFT
    for s in [7, 13, 42]:
        r = np.random.default_rng(s)
        phase = r.uniform(0, 2 * np.pi, n)
        mag = 1.0 + 0.05 * r.standard_normal(n)
        mag = np.maximum(mag, 0.1)
        spec = mag * np.exp(1j * phase)
        a_spec = np.real(np.fft.ifft(spec))

        inits[f'bumps_{nb}'] = np.maximum(bumps, 0.01)

    return inits

    t = 0

    best_J = compute_J(a)

            continue

        t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)

        # Periodic true objective check
        if t % check_interval == 0:
            local_J = compute_J(a)
            if local_J < best_J - 1e-10:
                best_J = local_J
                best_a = a.copy()
                last_improve = time.monotonic()
                if shake_count > 0:
                    shake_count = max(0, shake_count - 1)
            elif local_J > best_J + 0.02 and (time.monotonic() - last_improve) > 25.0:
                # Reset with perturbation
                a = best_a.copy()
                n_perturb = max(10, n // 20)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a[indices] *= np.exp(rng.normal(0, 0.3, n_perturb))
                a = np.maximum(a, 0.0)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0

    # Final check
    local_J = compute_J(a)
    if local_J < best_J:
        best_J = local_J
        best_a = a.copy()

    return best_a, best_J


def run_lbfgs(a_init, budget_s, tau=0.1):
    """L-BFGS-B refinement on the smoothed objective."""
    n = len(a_init)
    deadline = time.monotonic() + budget_s
    iterations = [0]

    def obj_grad(x):
        iterations[0] += 1
        if time.monotonic() > deadline:
            a = np.maximum(x, 0.0)
            J, _ = smooth_J_grad(a, tau)
            return J, np.zeros(n)
        a = np.maximum(x, 0.0)
        S = np.sum(a)
        if S < 0.01:
            return 1e15, np.zeros(n)
        J, grad = smooth_J_grad(a, tau)
        # Zero gradient where a is 0 (boundary)
        grad[x < 1e-10] = 0.0
        return J, grad

    best_a = a_init.copy()
    best_J = compute_J(a_init)

    try:
        result = scipy_minimize(
            obj_grad, a_init.copy(),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': 5000, 'maxfun': 8000, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        a_res = np.maximum(result.x, 0.0)
        J_res = compute_J(a_res)
        if J_res < best_J:
            best_a = a_res
            best_J = J_res
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    # Try with smaller tau for final polish
    if time.monotonic() < deadline - 5.0:
        try:
            result2 = scipy_minimize(
                obj_grad, best_a.copy(),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 3000, 'maxfun': 5000, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            a_res2 = np.maximum(result2.x, 0.0)
            J_res2 = compute_J(a_res2)
            if J_res2 < best_J:
                best_a = a_res2
                best_J = J_res2
        except Exception:
            pass

    return best_a, best_J


def coordinate_polish(a_init, budget_s, seed):
    """Targeted coordinate descent: decrease high-contribution coordinates at peaks."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    best_a = a_init.copy()
    best_J = compute_J(a_init)
    c = convolve_full(best_a)
    S = np.sum(best_a)

    # Find peak indices
    max_c = np.max(c)
    peak_thresh = 0.92 * max_c
    peak_indices = np.where(c >= peak_thresh)[0]

    # Compute contribution of each coordinate to peaks
    contrib = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            contrib[j] += best_a[j] * best_a[k - j]

    order = np.argsort(-contrib)

    # Try decreasing coordinates in order of contribution
    attempts = 0
    while time.monotonic() < deadline and attempts < 20000:
        improved = False
        for idx in order[:300]:
            if time.monotonic() >= deadline:
                break
            if best_a[idx] < 1e-8:
                continue

            # Try several decrease fractions
            best_frac = 0
            best_val = best_J
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
                d = best_a[idx] * frac
                if d < 1e-10:
                    break
                cand = best_a.copy()
                cand[idx] -= d
                cJ = compute_J(cand)
                if cJ < best_val:
                    best_val = cJ
                    best_frac = frac
            if best_frac > 0:
                best_a[idx] *= (1.0 - best_frac)
                best_J = best_val
                improved = True
                # Recompute contribution occasionally
                if attempts % 50 == 0:
                    c = convolve_full(best_a)
                    max_c = np.max(c)
                    peak_thresh = 0.92 * max_c
                    peak_indices = np.where(c >= peak_thresh)[0]
                    contrib = np.zeros(n)
                    for k in peak_indices:
                        lo = max(0, k - n + 1)
                        hi = min(k, n - 1)
                        for j in range(lo, hi + 1):
                            contrib[j] += best_a[j] * best_a[k - j]
                    order = np.argsort(-contrib)

        attempts += 1
        if not improved and attempts > 100:
            # Try random perturbations
            n_pert = rng.integers(5, 20)
            indices = rng.choice(n, size=n_pert, replace=False)
            best_a[indices] *= np.exp(rng.normal(0, 0.1, n_pert))
            best_a = np.maximum(best_a, 0)
            new_J = compute_J(best_a)
            if new_J < best_J:
                best_J = new_J
            else:
                break

    return best_a, best_J
