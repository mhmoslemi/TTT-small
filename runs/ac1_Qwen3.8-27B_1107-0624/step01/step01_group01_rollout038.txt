"""
LP-based iterative refinement with subgradient and logit-Adam polishing
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovations over previous approach:
1. Linear programming to find guaranteed descent directions (from SOTA description)

3. Multi-length exploration with careful time management
4. Adam in logit-space for final polish

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a):
    """Full linear convolution via FFT."""
    n = a.size
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < 0:
        c = np.maximum(c, 0.0)
    return c


def _score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-10:
        return np.inf
    c = _convolve_fft(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _score_exact(a):
    """Score using direct convolution (for final verification)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _softmax(z):
    """Numerically stable softmax."""
    z = np.clip(z, -100.0, 100.0)
    z -= float(z.max())
    e = np.exp(z)
    p = e + 1e-30
    return p / float(p.sum())


def _normalize(a):
    """Clip to [0,1000] and normalize to sum=1."""
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-10:
        return None
    return a / s


def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    return a / a.sum()


def _make_exp(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _make_poly(n, coeffs):
    """Polynomial profile: sum coeffs[k] * t^k, clamped to non-negative."""
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for k, c in enumerate(coeffs):
        a += c * (t ** k)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_edge(n, sigma, center=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _lp_solve(f, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns g or None if infeasible/unbounded.
    """
    n = f.size
    # Build A: A[k,i] = f[k-i] if 0 <= k-i < n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        end = min(i + n, 2 * n - 1)
        A[i:end, i] = f[:end - i]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5.0, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.maximum(g, 0.0)
            return g
    except Exception:
        pass
    return None


def _lp_refine(f, deadline, max_iters=5000, t_base=0.15, rng=None):
    """
    LP-based iterative refinement of sequence f.
    Returns (best_f, best_score).
    """
    f = np.asarray(f, dtype=np.float64).copy()
    n = f.size
    c = _convolve_fft(f)
    M = float(c.max())
    s = float(f.sum())
    best_f = f.copy()
    best_score = float(2.0 * n * M / (s * s))
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = _lp_solve(f, M)
        if g is None:
            break

        gs = float(g.sum())
        if gs < 1e-10:
            break
        g = g / gs  # normalize to sum 1

        # Try a few step sizes
        best_t = t_base
        best_h_score = np.inf
        for t_try in [t_base * 0.5, t_base, t_base * 1.5, t_base * 2.0]:
            h = (1.0 - t_try) * f + t_try * g
            h = _normalize(h)
            if h is None:
                continue
            sc = _score(h)
            if np.isfinite(sc) and sc < best_h_score:
                best_h_score = sc
                best_t = t_try

        if best_h_score == np.inf:
            break

        h = (1.0 - best_t) * f + best_t * g
        h = _normalize(h)
        if h is None:
            break

        new_score = best_h_score
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_f = h.copy()
            f = h
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                # Try a smaller step or break
                if no_imp > 200:
                    break
                # Perturb
                if rng is not None:
                    h = f + 0.01 * rng.normal(size=n)
                    h = _normalize(h)
                    if h is not None:
                        f = h

        # Update M
        c = _convolve_fft(f)
        M = float(c.max())

    return best_f, best_score


def _subgradient_step(f, c, eta=0.5, beta=0.1):
    """
    One subgradient/multiplicative update step.
    f: current sequence (sum=1)
    c: convolution of f with f
    Returns updated sequence.
    """
    n = f.size
    mval = float(c.max())
    if beta <= 1e-14:
        # Hard max: uniform weights on argmax set
        mask = c >= mval - 1e-12
        count = max(1, int(mask.sum()))
        w = mask.astype(np.float64) / count
    else:
        x = (c - mval) / beta
        ex = np.exp(np.clip(x, -50, 50))
        w = ex / float(ex.sum())

    # Compute gradient: grad[i] = sum_k w[k] * 2*f[k-i] (where valid)
    # This is 2 * (w * f_reversed) convolved appropriately
    # grad[i] = 2 * sum_k w[k] * f[k-i] where 0 <= k-i < n
    # = 2 * (w convolve f_flipped)[i + n - 1] ... let me think differently.
    #
    # grad[i] = sum_k w[k] * 2 * f[k-i] (if 0 <= k-i < n)
    # Let j = k-i, so k = i+j, j ranges 0..n-1:
    # grad[i] = 2 * sum_j w[i+j] * f[j]
    #
    # This is a cross-correlation: grad[i] = 2 * sum_j w[i+j] * f[j]
    # = 2 * (w_cross_correlate f)[i]
    #
    # Using FFT: if we define h[i] = sum_j w[i+j] * f[j], this is like
    # the "anti-convolution" or cross-correlation.
    # h = ifft(fft(w_full) * conj(fft(f_full))) ... let me just do it directly
    # using the same structure as the LP matrix.

    # More efficient: use the fact that this is a "sliding dot product"
    # h[i] = dot(w[i:i+n], f)
    # We can compute this via cross-correlation using FFT.

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    ff = np.fft.rfft(f, N)
    # cross-correlation: h[i] = sum_j w[i+j] * f[j]
    # = (w * flip(f))[i] where * is convolution and flip(f) = f[::-1]
    # Let's verify: conv(w, f_rev)[i] = sum_j w[j] * f_rev[i-j] = sum_j w[j] * f[n-1-(i-j)]
    # Hmm, this is getting complicated. Let me just use the direct formula.
    #
    # Actually: h[i] = sum_j w[i+j] * f[j] for j=0..n-1 (where i+j < 2n-1)
    # This is the cross-correlation of w and f at lag i.
    # = ifft(fft(w) * conj(fft(f))) at the appropriate indices.

    # cross = ifft(fft(w) * conj(fft(f)))
    # cross[k] = sum_j w[j] * conj(f[j-k]) ... this is getting confusing.
    #
    # Let me just compute it using the convolution structure:
    # h[i] = sum_j w[i+j] * f[j]
    # Define v = w padded to length 2n-1. Then h[i] = sum_j v[i+j] * f[j].
    # This is the i-th element of the "anti-convolution" or "cross-correlation".
    #
    # cross-correlation cross[i] = sum_j f[j] * conj(v[j+i])
    # If v is real: cross[i] = sum_j f[j] * v[j+i] = sum_j f[j] * w[j+i] = h[i] ✓ (when w is defined for j+i < 2n-1)
    #
    # cross = ifft(fft(f) * conj(fft(v)))
    # But we need v to be the right length.

    # Simpler approach: just build the matrix for the current w and multiply.
    # For n=512, this is O(n^2) = 262K ops, fast enough.
    grad = np.zeros(n)
    for i in range(n):
        # grad[i] = 2 * sum_j w[i+j] * f[j] for j where i+j < 2n-1
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            grad[i] = 2.0 * float(np.dot(w[i:i + j_max], f[:j_max]))

    # Multiplicative update in log-space
    log_f = np.log(np.maximum(f, 1e-30))
    log_f = log_f - eta * grad
    log_f = log_f - float(log_f.mean())  # keep sum approximately 1
    f_new = _softmax(log_f)
    return f_new


def _logit_adam(f0, deadline, lr=0.1, reg=1e-8, max_no_imp=200, rng=None):
    """
    Adam optimization in logit space.
    f0: initial sequence (sum=1, non-negative)
    Returns (best_f, best_score).
    """
    f0 = np.asarray(f0, dtype=np.float64).ravel()
    n = f0.size
    f0 = np.maximum(f0, 1e-15)
    f0 = f0 / f0.sum()

    z = np.log(f0)
    z -= float(z.mean())

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    f = f0.copy()
    c = _convolve_fft(f)
    maxc = float(c.max())
    beta = max(1e-12, 0.3 * maxc)

    best_f = f0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        # Soft-max gradient
        mval = float(c.max())
        if beta <= 1e-14:
            mask = c >= mval - 1e-12
            count = max(1, int(mask.sum()))
            w = mask.astype(np.float64) / count
        else:
            x = (c - mval) / beta
            ex = np.exp(np.clip(x, -50, 50))
            s_ex = float(ex.sum())
            if s_ex < 1e-30:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ex / s_ex

        # Gradient: vp[i] = 2 * sum_k w[k] * f[k-i] (same as subgradient)
        vp = np.zeros(n)
        for i in range(n):
            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:
                vp[i] = 2.0 * float(np.dot(w[i:i + j_max], f[:j_max]))

        # Add small regularization
        vp += reg * f
        # Project gradient onto tangent space of simplex
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= float(gz.mean())

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(z.mean())
        f = _softmax(z)
        c = _convolve_fft(f)
        maxc = float(c.max())
        sc = float(2.0 * n * maxc)

        if np.isfinite(sc):
            if sc < best_score - 1e-12:
                best_score = sc
                best_f = f.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.02, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t = 0
            continue

        if t % 50 == 0:
            beta = max(1e-14, beta * 0.7)

        if no_imp > max_no_imp:
            if rng is not None:
                z = 0.8 * best_z + 0.3 * rng.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(z.mean())
            else:
                break
            f = _softmax(z)
            c = _convolve_fft(f)
            maxc = float(c.max())
            beta = max(1e-10, 0.3 * maxc)
            m.fill(0)
            v.fill(0)
            t = 0
            lr = 0.1
            no_imp = 0

    return best_f, best_score


def _interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    return _normalize(p1)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based refinement, subgradient updates, and logit-Adam polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []  # list of (score, array)

    def _update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
        # Add to candidate list (maintain top 20)
        if p is not None and np.isfinite(sc):
            p = np.asarray(p, dtype=np.float64).copy()
            if len(candidates) < 20:
                candidates.append((float(sc), p))
                candidates.sort(key=lambda x: x[0])
            else:
                if float(sc) < candidates[-1][0]:
                    candidates[-1] = (float(sc), p)
                    candidates.sort(key=lambda x: x[0])

    # Try to use previous solution as a starting point
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pp = _normalize(parent)
            if pp is not None:
                sc = _score(pp)
                _update(pp, sc)
                print(f"Parent score: {sc:.8f} (n={pp.size})")
    except Exception:
        pass

    # Phase 1: Generate diverse initial candidates
    print("Phase 1: Initial candidates")
    t_phase1_end = t_start + 30.0

    init_profiles = []
    for n in [64, 128, 256, 512]:
        for eta in [0.8, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0]:
            init_profiles.append(_make_linear(n, eta))
        for rate in [1.0, 1.5, 2.0, 3.0, 4.0]:
            init_profiles.append(_make_exp(n, rate))
        for c1, c2 in [(1.0, -0.5), (2.0, -1.0), (1.5, -1.5), (3.0, -2.0), (0.5, 1.0)]:
            init_profiles.append(_make_poly(n, [1.0, c1, c2]))
        for sigma in [0.15, 0.2, 0.3]:
            for center in [0.3, 0.4, 0.5]:
                init_profiles.append(_make_gaussian_edge(n, sigma, center))

    for p in init_profiles:
        if p is not None:
            sc = _score(p)
            _update(p, sc)

    # Random smooth profiles
    for n in [128, 256, 512]:
        for _ in range(10):
            t = np.linspace(0, 1, n)
            a = 1.0 + rng.uniform(0, 3) * t + rng.uniform(-1, 1) * t**2 + rng.uniform(-0.5, 0.5) * t**3
            a = np.maximum(a, 1e-12)
            p = a / a.sum()
            sc = _score(p)
            _update(p, sc)

    print(f"Phase 1 best: {best_score:.8f} (n={best_seq.size})")

    # Phase 2: LP-based refinement of top candidates
    print("Phase 2: LP refinement")
    t_phase2_end = min(deadline, t_start + 1000.0)
    top_n = min(8, len(candidates))

    for idx in range(top_n):
        if time.monotonic() > t_phase2_end:
            break
        sc_orig, p_orig = candidates[idx]
        # Work at a manageable length
        n_work = min(512, max(128, p_orig.size))
        p_work = _interpolate_profile(p_orig, n_work)
        if p_work is None:
            continue
        sc_work = _score(p_work)

        # LP refinement
        per_deadline = min(t_phase2_end, time.monotonic() + 80.0)
        p_lp, sc_lp = _lp_refine(p_work, per_deadline, max_iters=3000, t_base=0.12, rng=rng)
        _update(p_lp, sc_lp)

        if best_score < 1.52:
            print(f"  LP candidate {idx}: {sc_lp:.8f} (n={p_lp.size})")

    print(f"Phase 2 best: {best_score:.8f}")

    # Phase 3: Subgradient refinement
    print("Phase 3: Subgradient refinement")
    t_phase3_end = min(deadline, t_start + 1300.0)
    top_n = min(6, len(candidates))

    for idx in range(top_n):
        if time.monotonic() > t_phase3_end:
            break
        _, p_orig = candidates[idx]
        n_work = min(512, max(128, p_orig.size))
        p_work = _interpolate_profile(p_orig, n_work)
        if p_work is None:
            continue

        # Subgradient iterations
        f = p_work.copy()
        c = _convolve_fft(f)
        M = float(c.max())
        beta = 0.2 * M
        best_sub = f.copy()
        best_sub_score = float(2.0 * f.size * M)
        no_imp = 0
        eta = 0.3

        sub_deadline = min(t_phase3_end, time.monotonic() + 60.0)
        for it in range(2000):
            if time.monotonic() > sub_deadline:
                break

            f_new = _subgradient_step(f, c, eta=eta, beta=beta)
            sc = _score(f_new)
            if np.isfinite(sc) and sc < best_sub_score - 1e-12:
                best_sub_score = sc
                best_sub = f_new.copy()
                f = f_new
                c = _convolve_fft(f)
                M = float(c.max())
                no_imp = 0
                beta = max(1e-14, beta * 0.95)
            else:
                no_imp += 1
                if no_imp > 100:
                    break
                beta = max(1e-14, beta * 0.9)
                eta = max(0.05, eta * 0.98)

        _update(best_sub, best_sub_score)
        if best_sub_score < 1.52:
            print(f"  Subgrad candidate {idx}: {best_sub_score:.8f}")

    print(f"Phase 3 best: {best_score:.8f}")

    # Phase 4: Logit-Adam polish
    print("Phase 4: Logit-Adam polish")
    t_phase4_end = min(deadline, t_start + 1650.0)
    top_n = min(8, len(candidates))

    for idx in range(top_n):
        if time.monotonic() > t_phase4_end:
            break
        _, p_orig = candidates[idx]
        n_work = min(512, max(128, p_orig.size))
        p_work = _interpolate_profile(p_orig, n_work)
        if p_work is None:
            continue

        per_deadline = min(t_phase4_end, time.monotonic() + 30.0)
        p_adam, sc_adam = _logit_adam(p_work, per_deadline, lr=0.08, rng=rng)
        _update(p_adam, sc_adam)

        if best_score < 1.52:
            print(f"  Adam candidate {idx}: {sc_adam:.8f} (n={p_adam.size})")

    print(f"Phase 4 best: {best_score:.8f}")

    # Phase 5: Length extension and final polish
    print("Phase 5: Length extension")
    t_phase5_end = deadline - 30.0

    # Try extending best solution to different lengths
    if best_seq is not None:
        for n_ext in [256, 512, 1024]:
            if time.monotonic() > t_phase5_end:
                break
            p_ext = _interpolate_profile(best_seq, n_ext)
            if p_ext is None:
                continue
            sc_ext = _score(p_ext)
            _update(p_ext, sc_ext)

            # Quick LP polish at new length
            if n_ext <= 512:
                per_deadline = min(t_phase5_end, time.monotonic() + 40.0)
                p_lp, sc_lp = _lp_refine(p_ext, per_deadline, max_iters=2000, t_base=0.1, rng=rng)
                _update(p_lp, sc_lp)

    # Phase 6: Final Adam polish on the absolute best
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        n_final = min(512, best_seq.size)
        p_final = _interpolate_profile(best_seq, n_final) if best_seq.size != n_final else best_seq.copy()
        if p_final is not None:
            per_deadline = deadline - 10.0
            p_polished, sc_polished = _logit_adam(p_final, per_deadline, lr=0.05, reg=1e-9, max_no_imp=500, rng=rng)
            _update(p_polished, sc_polished)

    # Final exact evaluation
    print("Final evaluation")
    final_best_score = np.inf
    final_best_seq = best_seq

    # Check top candidates with exact scoring
    for sc_c, p_c in candidates[:10]:
        if p_c.size <= 2048:
            sc_exact = _score_exact(p_c)
            if np.isfinite(sc_exact) and sc_exact < final_best_score:
                final_best_score = sc_exact
                final_best_seq = p_c

    # Also check our tracked best
    if best_seq is not None and best_seq.size <= 2048:
        sc_exact = _score_exact(best_seq)
        if np.isfinite(sc_exact) and sc_exact < final_best_score:
            final_best_score = sc_exact
            final_best_seq = best_seq

    # Call the authoritative evaluator
    if final_best_seq is not None and final_best_seq.size <= 5000:
        try:
            sc_eval = float(evaluate_sequence(final_best_seq.tolist()))
            if np.isfinite(sc_eval):
                final_best_score = sc_eval
        except Exception:
            pass

    # Prepare final output
    if final_best_seq is not None:
        final_seq = _normalize(final_best_seq)
        if final_seq is None:
            final_seq = _make_linear(256, 1.5)

        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s

        print(f"Final: score={final_best_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]
    else:
        fallback = _make_linear(256, 1.5)
        return [float(x) for x in fallback]
