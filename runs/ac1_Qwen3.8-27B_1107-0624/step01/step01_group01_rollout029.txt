```python
import numpy as np
import time
from scipy.optimize import minimize

# Helper functions (top-level, no closures)

def fft_convolve(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    if n <= 64:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0)  # numerical safety

def corr_with_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i = 0..n-1."""
    n = len(p)
    N = 1
    while N < 2*n - 1:
        N *= 2
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n-1:2*n-1]

def softmax_stable(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def score_and_grad_z(z, beta):
    """Compute soft-max score and gradient w.r.t. logit z."""
    n = len(z)
    p = softmax_stable(z)
    c = fft_convolve(p)
    cmax = c.max()
    x = (c - cmax) / beta
    ex = np.exp(x)
    s = ex.sum()
    score = cmax + beta * np.log(s)
    w = ex / s
    # Gradient of score w.r.t. p
    gp = 2.0 * corr_with_weights(w, p)
    # Chain rule for softmax: d(score)/d(z_i) = p_i * (gp_i - sum(p*gp))
    dot = np.dot(p, gp)
    gz = p * (gp - dot)
    gz -= gz.mean()  # remove gauge
    return score, gz

def hard_score(p):
    """Exact score for a normalized non-negative sequence."""
    n = len(p)
    c = fft_convolve(p)
    return 2.0 * n * c.max()

    betas = [cmax0 * f for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]]
    
    time_per_stage = (deadline - time.monotonic()) / n_stages

        if time.monotonic() > deadline - 30:

            fun=lambda z: score_and_grad_z(z, beta)[0],
            x0=best_z,

            method='L-BFGS-B',

        )

        z_new -= z_new.mean()
        p_new = softmax_stable(z_new)
```

```python
Multi-stage optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS-B in logit space with beta annealing, multiple diverse starts,

from scipy.optimize import minimize

# === Helper functions ===

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < 0:
        c = np.maximum(c, 0.0)
    return c

def _corr_w(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1: 2 * n - 1]

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _score_grad(z, beta):
    n = len(z)
    p = _softmax(z)
    c = _fft_convolve(p)
    cmax = c.max()
    x = (c - cmax) / beta
    ex = np.exp(x)
    s = ex.sum()
    score = cmax + beta * np.log(s)
    w = ex / s
    gp = 2.0 * _corr_w(w, p)
    dot = np.dot(p, gp)
    gz = p * (gp - dot)
    gz -= gz.mean()
    return score, gz

def _obj(z, beta):
    return _score_grad(z, beta)[0]

def _jac(z, beta):
    return _score_grad(z, beta)[1]

def _hard_score(p):
    n = len(p)
    c = _fft_convolve(p)
    return 2.0 * n * c.max()

def _linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _cosine_profile(n, A, k):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + A * np.cos(2.0 * np.pi * k * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _tapered_uniform(n, taper_frac):
    t = np.linspace(0.0, 1.0, n)
    # Raised cosine taper
    d = taper_frac
    w = np.ones(n)
    for i in range(n):
        if t[i] < d:
            w[i] = 0.5 * (1 + np.cos(np.pi * t[i] / d))
        elif t[i] > 1 - d:
            w[i] = 0.5 * (1 + np.cos(np.pi * (1 - t[i]) / d))
    w = np.maximum(w, 1e-10)
    return w / w.sum()

def _poly_profile(n, coeffs):
    t = np.linspace(-1.0, 1.0, n)
    a = np.ones(n)
    for k, c in enumerate(coeffs, 1):
        a += c * t ** k
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _random_smooth(n, rng, n_modes=15):
    t = np.arange(n, dtype=np.float64)
    z = np.zeros(n)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.5, 2.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * k * t / n + phase)
    return _softmax(z - z.mean())

def _lbfgs_optimize(z0, betas, deadline, maxiter=3000):
    best_z = z0.copy()
    best_p = _softmax(z0)
    best_sc = _hard_score(best_p)
    
    for beta in betas:
        if time.monotonic() > deadline - 60:
            break
        try:
            res = minimize(
                _obj, best_z, jac=_jac, args=(beta,),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'maxfun': maxiter * 2,
                         'ftol': 1e-15, 'gtol': 1e-12}
            )
            z_new = res.x
            z_new -= z_new.mean()
            p_new = _softmax(z_new)
            sc = _hard_score(p_new)
            if sc < best_sc:
                best_sc = sc
                best_z = z_new.copy()
                best_p = p_new.copy()
        except Exception:
            pass
    
    return best_p, best_sc

            m.fill(0); v.fill(0); t = 0

        mh = m / (1 - b1**t)

        z = np.clip(z, -50, 50)

    return best_p, best_sc

def _soft_max_info(c, beta):
```
