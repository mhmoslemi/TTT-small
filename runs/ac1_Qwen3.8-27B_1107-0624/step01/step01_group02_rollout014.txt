"""
Step function optimization minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses LP-based direction finding (min t s.t. f_cross_g <= t, sum(g)=1, g>=0),
spectral flat-top initialization, adaptive blending, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def normalize_nonneg(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    np.clip(a, 0.0, None, out=a)
    s = a.sum()
    if s < 1e-300:
        return np.full(len(a), 1.0 / len(a))
    a /= s
    return a


def get_score(a):
    """Score = 2*n*max(conv) for a sum-1 non-negative sequence."""
    c = np.convolve(a, a)
    return 2.0 * len(a) * c.max()


def build_A(f):
    """Build m x n matrix: A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            A[k, jlo:jhi] = f[k - np.arange(jlo, jhi)]
    return A


def solve_lp_direction(f):
    """Solve: min t s.t. A_f @ g <= t, sum(g)=1, g>=0. Returns g or None."""
    n = len(f)
    m = 2 * n - 1
    A = build_A(f)

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if res.status == 0 and res.x is not None:
        g = np.clip(res.x[:n], 0.0, None)
        s = g.sum()
        if s > 1e-12:
            return g / s
    return None


def lp_refine(f, deadline, rng, max_iters=50000):
    """Iterate LP direction finding with adaptive blending until deadline."""
    f = normalize_nonneg(f)
    n = len(f)
    best = f.copy()
    best_sc = get_score(f)
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline and iter_count < max_iters:
        iter_count += 1
        g = solve_lp_direction(f)
        if g is None:
            break

        # Convergence check
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 5:
                # Perturb to escape
                scale = 0.05 + 0.05 * min(no_improve / 20.0, 1.0)
                f = normalize_nonneg(f * np.exp(rng.normal(0, scale, n)))
                no_improve = 0
            continue

        cur_sc = get_score(f)
        improved = False

        # Adaptive blend: larger when far from convergence, smaller near it
        dist = np.mean(np.abs(g - f))
        ts = [0.5, 0.3, 0.1, 0.7, 0.05, 0.8, 0.2, 0.4, 0.15, 0.6]
        if dist < 0.05:
            ts = [0.1, 0.05, 0.02, 0.01, 0.03, 0.08, 0.04]

        for t in ts:
            h = normalize_nonneg((1.0 - t) * f + t * g)
            sc = get_score(h)
            if sc < best_sc - 1e-12:
                f = h
                best = h.copy()
                best_sc = sc
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                scale = 0.03 + 0.03 * min(no_improve / 30.0, 1.0)
                f = normalize_nonneg(f * np.exp(rng.normal(0, scale, n)))
                no_improve = 0

    return best, best_sc


def spectral_flat_init(n, rng):
    """Generate initialization via flat-spectrum retrospective with random phases."""
    L = 1
    while L < 2 * n - 1:
        L *= 2

    phase = np.exp(1j * rng.uniform(0, 2 * np.pi, L))
    F = phase.copy()
    F[0] = 1.0  # DC = 1 for sum=1

    a = np.fft.ifft(F).real[:n]
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-10:
        # Retry with different phase
        phase = np.exp(1j * rng.uniform(0, 2 * np.pi, L))
        F = phase.copy()
        F[0] = 1.0
        a = np.fft.ifft(F).real[:n]
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-10:
            return np.full(n, 1.0 / n)
    return a / s


def spectral_refine(f, deadline, max_iters=5000):
    """Iteratively flatten the convolution via spectral projection."""
    f = normalize_nonneg(f)
    n = len(f)
    L = 1
    while L < 2 * n - 1:
        L *= 2

    best = f.copy()
    best_sc = get_score(f)

    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break

        f_pad = np.zeros(L)
        f_pad[:n] = f
        F = np.fft.fft(f_pad)

        # Set flat magnitude (keep phase)
        mag = np.abs(F)
        mag[mag < 1e-15] = 1e-15
        F_flat = F / mag
        F_flat[0] = 1.0  # Keep DC

        a_new = np.fft.ifft(F_flat).real[:n]
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        sc = get_score(a_new)
        if sc < best_sc - 1e-12:
            best = a_new.copy()
            best_sc = sc
        f = a_new

    return best, best_sc


def make_two_block(n, p):
    """Two-block sequence with mass p in first half."""
    L = n // 2
    a = np.zeros(n)
    a[:L] = p / L
    a[L:] = (1.0 - p) / (n - L)
    return a


def make_three_block(n, weights):
    """Three-block sequence."""
    L1 = n // 3
    L2 = n // 3
    L3 = n - L1 - L2
    a = np.zeros(n)
    a[:L1] = weights[0] / L1
    a[L1:L1+L2] = weights[1] / L2
    a[L1+L2:] = weights[2] / L3
    return a


def make_cosine_nets(n, rng):
    """Cosine-based smooth initialization."""
    a = np.zeros(n)
    x = np.linspace(0, 1, n)
    for _ in range(rng.integers(1, 4)):
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.clip(a + 0.5, 0.0, None)
    return normalize_nonneg(a)


def upsample(a, N):
    n = len(a)
    if n == N:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, N)
    return normalize_nonneg(np.interp(x_new, x_old, a))


def downsample(a, N):
    n = len(a)
    if n == N:
        return a.copy()
    out = np.zeros(N)
    for i in range(N):
        lo = int(round(i * n / N))
        hi = int(round((i + 1) * n / N))
        hi = max(hi, lo + 1)
        out[i] = a[lo:min(hi, n)].sum()
    return normalize_nonneg(out)


def resize(a, N):
    if len(a) == N:
        return a.copy()
    if len(a) < N:
        return upsample(a, N)
    return downsample(a, N)
