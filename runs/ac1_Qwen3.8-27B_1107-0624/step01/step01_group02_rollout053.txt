    """Project onto probability simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s

def compute_conv(a):
    return np.convolve(a, a)

def score_normalized(a):
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * float(c.max())

def soft_weights(c, eps):
    """Compute softmax weights over the convolution values."""
    M = c.max()
    e = (c - M) / eps
    e = np.clip(e, -500, 0)
    w = np.exp(e)
    s = w.sum()
    if s < 1e-300:

        w[int(np.argmax(c))] = 1.0
    else:

    return w, M

    grad[j] = 2 * sum_i a[i] * w[i+j] for j = 0..n-1.

"""
Optimize step function heights to minimize 2n*max(conv(a,a))/(sum a)^2.
Uses smooth-max gradient descent with mirror (exponential) updates,
periodic LP-based direction kicks, and multi-scale refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def normalize_simplex(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def score_normalized(a):
    """Score = 2n * max(conv(a,a)) for normalized (sum=1) non-negative a."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * float(c.max())


def soft_weights(c, eps):
    """Compute softmax weights over convolution values (for smooth max)."""
    M = float(c.max())
    e = (c - M) / eps
    e = np.clip(e, -500.0, 0.0)
    w = np.exp(e)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w, M


def gradient_smoothmax(a, w):
    """
    Gradient of smooth-max of conv(a,a) w.r.t. a.
    grad[j] = 2 * sum_i a[i] * w[i+j] for j=0..n-1.
    Uses FFT for efficiency.
    """
    n = len(a)
    m = len(w)  # 2n-1
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = a
    pb = np.zeros(size, dtype=np.float64)
    pb[:m] = w[::-1]
    out = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(pb, size), size)
    idx = (m - 1) - np.arange(n, dtype=np.int64)
    return 2.0 * out[idx]


def solve_lp_direction(f, M, max_rows=500):
    """Solve LP: max sum(g) s.t. conv-active-constraints(f,g) <= M, g>=0."""
    n = len(f)
    if n > 4096 or M <= 1e-300 or linprog is None:
        return None
    c = np.convolve(f, f)
    total_rows = 2 * n - 1
    if max_rows >= total_rows:
        rows = np.arange(total_rows, dtype=np.int64)
    else:
        k = max(1, min(max_rows, total_rows))
        rows = np.argpartition(c, -k)[-k:].astype(np.int64)
    A = np.zeros((len(rows), n), dtype=np.float64)
    for r, k in enumerate(rows):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            A[r, jlo:jhi] = f[k - np.arange(jlo, jhi)]
    A = A / M
    try:
        res = linprog(
            -np.ones(n), A_ub=A, b_ub=np.ones(len(rows)),
            bounds=[(0.0, None)] * n, method="highs"
        )
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1.0 + 1e-4:
                return g
    except Exception:
        pass
    return None


def gradient_optimize(a, time_limit, rng, state, n_lp_interval=30):

def resize_sequence(a, target_n):
    """Resize sequence to target length using interpolation or averaging."""
    a = np.asarray(a, dtype=np.float64)
    m = len(a)
    if m == target_n:
        return normalize_simplex(a)
    if m < target_n:
        x_old = np.linspace(0.0, 1.0, m)
        x_new = np.linspace(0.0, 1.0, target_n)
        a_new = np.interp(x_new, x_old, a)
    else:
        # Downsample by binning
        a_new = np.zeros(target_n, dtype=np.float64)
        for i in range(target_n):
            lo = int(i * m / target_n)
            hi = int((i + 1) * m / target_n)
            if hi > lo:
                a_new[i] = a[lo:hi].sum()
    return normalize_simplex(a_new)

        if time.monotonic() > hard_deadline - 80:
            break
        t_remaining = hard_deadline - time.monotonic()
        t_alloc = min(60.0, 0.06 * t_remaining)
        if t_alloc < 15:
            break
        init = make_initial_v2(best_n, kind, rng, state["seq"])
        init_s = score_normalized(init)
        if init_s > 2.5 * state["score"]:
            continue  # Skip obviously bad starts
        gradient_optimize(init, t_alloc, rng, state)
        print(f"Phase 3: kind={kind}, best={state['score']:.10f}", flush=True)

    # Phase 4: Fine polish on the best
    if time.monotonic() < hard_deadline - 20:
        t_remaining = hard_deadline - time.monotonic()
        t_alloc = min(200.0, 0.4 * t_remaining)
        print(f"Phase 4: Fine polish, t={t_alloc:.0f}s", flush=True)
        fine_polish(state["seq"], t_alloc, rng, state)
        print(f"  After Phase 4: best={state['score']:.10f}, n={len(state['seq'])}", flush=True)

    # Phase 5: Final micro-polish
    if time.monotonic() < hard_deadline - 10:
        t_remaining = hard_deadline - time.monotonic()
        fine_polish(state["seq"], t_remaining, rng, state)

    # Return best
    if state["seq"] is None:
        return [1.0 / 3.0, 2.0 / 3.0]

    result = [float(x) for x in state["seq"]]
    try:
        official = evaluate_sequence(result)
        print(f"FINAL: score={official:.10f}, n={len(result)}", flush=True)
    except Exception:
        pass

    elapsed = time.monotonic() - total_start
    print(f"Total time: {elapsed:.1f}s", flush=True)
    return result
