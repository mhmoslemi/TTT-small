import numpy as np
from scipy.optimize import linprog
import time

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def subgradient_step(s, alpha=0.01, active_frac=0.85):
    """Take a subgradient step to reduce max(conv(s,s))."""
    n = len(s)
    conv = np.convolve(s, s)
    M = np.max(conv)
    
    # Active set: indices where conv is close to max
    active_mask = conv > active_frac * M
    active_indices = np.where(active_mask)[0]
    
    if len(active_indices) == 0:
        return s
    
    # Compute subgradient
    # d(conv[k])/d(s[i]) = 2*s[k-i] if 0 <= k-i < n, else 0
    g = np.zeros(n)
    for k in active_indices:
        # For each k, the contribution to g[i] is 2*s[k-i] (if valid)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * s[j]
    
    g /= len(active_indices)
    
    # Step
    s_new = s - alpha * g
    
    # Project onto simplex: s >= 0, sum(s) = 1
    s_new = np.maximum(s_new, 0.0)
    s_sum = np.sum(s_new)
    if s_sum < 1e-15:
        return s
    s_new = s_new / s_sum
    
    return s_new

def coordinate_descent_step(s, rng):

    best_s = s.copy()
    best_score = fast_score(s)

    # Keep s[i] - delta >= 0, so delta <= s[i]
    max_delta = s[i]

        sc = fast_score(s_try)
        if sc < best_score:
            best_score = sc
            best_s = s_try
    
    return best_s

def block_descent(s, rng, block_size=10):
    """Optimize a random block of coordinates."""
    n = len(s)
    start = rng.integers(0, n - block_size + 1)
    idx = np.arange(start, start + block_size)
    
    # Optimize the block using Nelder-Mead or similar

        sc = fast_score(s_try)

def lp_direction(s, M, time_limit=3.0):
    """Solve LP to find a good direction."""
    n = len(s)
    # Build constraint matrix: (s*b)[k] <= M for all k
    # (s*b)[k] = sum_j s[j] * b[k-j]
    # This is a linear function of b.
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # b[j] appears in (s*b)[k] for k = j, j+1, ..., j+n-1 (with s[k-j])
        for k in range(j, j+n):
            A[k, j] = s[k-j]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:

                      options={'time_limit': time_limit})

            b = np.maximum(res.x, 0.0)
            if np.sum(b) > 1e-14:

    except Exception:
        pass
    return None
