import numpy as np
import time

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-phase gradient optimization in logit space with Nesterov momentum,
adaptive soft-max annealing, direct top-k subgradients, and diverse restarts.
Improves upon prior approaches with: (1) exact top-k subgradient computation,
(2) Nesterov accelerated momentum, (3) Polyak-inspired adaptive step sizes,
(4) aggressive beta annealing, (5) structured diverse initializations.
"""


def _fft_conv(a):
    """Full linear self-convolution via FFT."""
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = np.maximum(c, 0.0)
    return c


def _compute_score(a):
    """Compute 2*n*max(conv(a,a)) / sum(a)^2."""
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = _fft_conv(a)
    mb = float(c.max())
    return 2.0 * n * mb / (s * s)


def _score_norm(a):
    """Score for sum-normalized a: 2*n*max(conv)."""
    n = a.size
    c = _fft_conv(a)
    return 2.0 * n * float(c.max())


def _softmax(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-30:
        return np.ones_like(z) / z.size
    return e / s


def _normalize(a):
    """Clip to [0,1000] and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s


def _interp_profile(p, n_new):
    """Linear interpolation of profile to new length, then normalize."""
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == n_new:
        return _normalize(p)
    x0 = np.linspace(0.0, 1.0, p.size)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p))


def _soft_weights(c, beta):
    """Compute soft-max weights (sum to 1) over conviction values c."""
    m = c.max()
    x = np.clip((c - m) / beta, -60.0, 60.0)
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        mask = (c >= m - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            w[np.argmax(c)] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    return ex / s


def _grad_soft(a, w):
    """
    Compute 2 * cross_corr(w, a) using FFT.
    cross_corr(w, a)[i] = sum_k w[k+i] * a[k]
    """
    n = a.size
    N = 1
    while N < w.size + n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fw * np.conjugate(fa), N)[:n]
    return 2.0 * corr


def _grad_topk(a, c, k=8, power=3.0):
    """
    Compute approximate subgradient using top-k lags with power-weighted averaging.
    O(k*n) direct computation.
    """
    n = a.size
    if k >= c.size:
        top_idx = np.arange(c.size)
    else:
        top_idx = np.argsort(c)[-k:]
    vals = c[top_idx]
    vmax = vals.max()
    vmin = vals.min()
    if vmax - vmin > 1e-15:
        w = (vals - vmin + 1e-12) ** power
    else:
        w = np.ones_like(vals)
    w = w / w.sum()

    grad = np.zeros(n, dtype=np.float64)
    for lag, wi in zip(top_idx, w):
        lag = int(lag)
        j_lo = max(0, lag - n + 1)
        j_hi = min(n - 1, lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += 2.0 * wi * a[j_lo:j_hi + 1][::-1]
    return grad


def _logit_optimize(z0, n, deadline, rng, base_lr=0.12, reg=0.0, use_topk=False, topk_k=8):
    """
    Optimize in logit space: a = softmax(z), minimize 2n*max(conv(a,a)).
    Uses Nesterov momentum + adaptive soft-max (or top-k) gradient.
    Returns (best_z, best_a, best_score).
    """
    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    z = z - z.mean()
    z = np.clip(z, -50.0, 50.0)

    a = _softmax(z)
    c = _fft_conv(a)
    maxc = float(c.max())
    current_score = 2.0 * n * maxc

    best_score = current_score
    best_z = z.copy()
    best_a = a.copy()

    # Nesterov momentum
    m = np.zeros(n, dtype=np.float64)

    # Beta for soft-max
    beta = max(1e-6, 0.3 * maxc)
    beta_min = 1e-8

    lr = base_lr
    no_imp = 0
    t = 0
    gamma = 0.85  # momentum coefficient

    while time.monotonic() < deadline:
        t += 1

        # Compute gradient
        if use_topk:
            grad_a = _grad_topk(a, c, k=topk_k)
        else:
            w = _soft_weights(c, beta)
            grad_a = _grad_soft(a, w)

        # Gradient in logit space
        dot = float(np.dot(a, grad_a))
        if reg > 0:
            grad_a = grad_a + reg * a
        grad_z = a * (grad_a - float(np.dot(a, grad_a)))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0.0)
            continue

        # Nesterov step
        m = gamma * m + grad_z
        z_new = z - lr * m
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -50.0, 50.0)

        a_new = _softmax(z_new)
        c_new = _fft_conv(a_new)
        maxc_new = float(c_new.max())
        score_new = 2.0 * n * maxc_new

        if np.isfinite(score_new) and score_new < best_score - 1e-13:
            best_score = score_new
            best_z = z_new.copy()
            best_a = a_new.copy()
            no_imp = 0
            z = z_new
            a = a_new
            c = c_new
        else:
            no_imp += 1
            m *= 0.7

        # Anneal beta (for soft-max mode)
        if not use_topk and t % 40 == 0:
            beta = max(beta_min, beta * 0.65)

        # Decay learning rate when stuck
        if no_imp > 200:
            lr = max(0.005, lr * 0.94)

        # Adaptive restart
        if no_imp > max(400, 3 * n // 4):
            z = best_z + 0.4 * rng.standard_normal(n)
            z = z - z.mean()
            z = np.clip(z, -50.0, 50.0)
            a = _softmax(z)
            c = _fft_conv(a)
            beta = max(1e-5, 0.3 * float(c.max()))
            m = np.zeros(n, dtype=np.float64)
            lr = base_lr
            no_imp = 0

    return best_z, best_a, best_score


def _logit_optimize_polyak(z0, n, deadline, rng, base_lr=0.08, reg=0.0):
    """
    Logit-space optimization with Polyak-inspired adaptive step size.
    Uses exact top-k subgradients for more accurate descent direction.
    """
    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    z = z - z.mean()
    z = np.clip(z, -50.0, 50.0)

    a = _softmax(z)
    c = _fft_conv(a)
    current_score = 2.0 * n * float(c.max())

    best_score = current_score
    best_z = z.copy()
    best_a = a.copy()

    # Polyak: estimate optimal value as 2n * mean(conv) = 2n/(2n-1)
    target = 2.0 * n / (2 * n - 1)  # theoretical lower bound for flat conv

    m = np.zeros(n, dtype=np.float64)
    gamma = 0.82
    lr = base_lr
    no_imp = 0
    t = 0
    k_top = 6

    while time.monotonic() < deadline:
        t += 1

        # Top-k gradient
        grad_a = _grad_topk(a, c, k=k_top, power=2.0)

        # Logit gradient
        dot_val = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot_val)
        if reg > 0:
            grad_z += reg * a

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0.0)
            continue

        # Adaptive step size (Polyak-like)
        gap = current_score - target
        if gap > 1e-10:
            gnorm = float(np.dot(grad_z, grad_z))
            if gnorm > 1e-20:
                lr_polyak = min(0.5, 2.0 * gap / (gnorm + 1e-15))
                lr = 0.6 * lr + 0.4 * lr_polyak
            lr = max(0.001, min(lr, 0.5))

        # Nesterov
        m = gamma * m + grad_z
        z_new = z - lr * m
        z_new = z_new - z_new.mean()
        z_new = np.clip(z_new, -50.0, 50.0)

        a_new = _softmax(z_new)
        c_new = _fft_conv(a_new)
        score_new = 2.0 * n * float(c_new.max())

        if np.isfinite(score_new) and score_new < best_score - 1e-13:
            best_score = score_new
            best_z = z_new.copy()
            best_a = a_new.copy()
            current_score = score_new
            no_imp = 0
            z = z_new
            a = a_new
            c = c_new
        else:
            no_imp += 1
            m *= 0.6
            lr = max(0.001, lr * 0.97)

        # Increase top-k as we converge
        if t % 200 == 0:
            k_top = min(20, 4 + t // 100)

        # Restart if stuck
        if no_imp > max(300, n // 2):
            z = best_z + 0.35 * rng.standard_normal(n)
            z = z - z.mean()
            z = np.clip(z, -50.0, 50.0)
            a = _softmax(z)
            c = _fft_conv(a)
            current_score = 2.0 * n * float(c.max())
            m = np.zeros(n, dtype=np.float64)
            lr = base_lr
            no_imp = 0
            k_top = 6

    return best_z, best_a, best_score


def _make_structured_profiles(n, count=20, rng=None):
    """Generate diverse structured starting profiles of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profs = []

    # Linear ramps
    for eta in [0.8, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())

    # Cosine windows
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        a = np.cos(alpha * np.pi * (t - 0.5)) ** 2 + 1e-10
        profs.append(a / a.sum())
        a = np.exp(-alpha * (u) ** 2)
        profs.append(a / a.sum())

    # Power laws
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (t + 1e-6) ** p
        profs.append(a / a.sum())
        a = ((1 - t) + 1e-6) ** p
        profs.append(a / a.sum())

    # Triangle / piecewise linear
    for peak in [0.3, 0.5, 0.7]:
        a = np.maximum(0, 1 - np.abs(t - peak) / min(peak, 1 - peak + 1e-6))
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())

    # Random smooth
    for _ in range(min(count, 10)):
        ncoef = np.random.randint(3, 8)
        coefs = rng.standard_normal(ncoef) * 0.8
        a = 1.0
        for k, ck in enumerate(coefs, 1):
            a += ck * np.cos(2.0 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())

    return profs
