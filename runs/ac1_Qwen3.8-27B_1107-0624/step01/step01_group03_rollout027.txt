    best_s = s.copy()
    best_val = soft_max_obj(s, tau)
    
    v = np.zeros_like(s)  # momentum
    y = s.copy()  # Nesterov lookahead
    
    for t in range(n_iters):
        # Nesterov step
        beta = (1 + sqrt(1 + 4*t*t)) / 2
        y = s + (beta - 1) / beta * (s - s_prev) if t > 0 else s
        
        # Compute gradient at y
        c = np.convolve(y, y)
        w = softmax(c / tau)  # normalized weights
        g = 2.0 * np.correlate(y, w, 'full')[n-1 : 2*n-1]
        
        # Update
        v = momentum * v + g
        s_new = y - step_size * v
        s_new = project_simplex(s_new)
        
        # Check improvement
        val = soft_max_obj(s_new, tau)
        if val < best_val:
            best_val = val
            best_s = s_new.copy()
        
        s_prev = s
        s = s_new
    
    return best_s, best_val

import numpy as np
from scipy.optimize import linprog
import time
import math


"""
Soft-max continuation + Adam optimizer + LP refinement.
Minimizes 2n*max(conv(s,s))/(sum(s))^2 via smoothed objective with continuation,
then refines with LP-guided local search. Uses FFT for fast convolutions.
"""


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    need = len(a) + len(b) - 1
    L = 1
    while L < need:
        L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:need]


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = fft_convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def softmax_gradient(s, tau):
    """Compute gradient of tau * log(sum(exp(c_k/tau))) where c = s*s."""
    n = len(s)
    conv = fft_convolve(s, s)
    max_c = np.max(conv)
    exp_c = np.exp((conv - max_c) / tau)
    w = exp_c / exp_c.sum()
    # g_i = 2 * sum_k w[k] * s[k-i]  (cross-correlation)
    g = 2.0 * np.correlate(s, w, 'full')[n - 1:2 * n - 1]
    return g


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    k = idx[-1]
    theta = cssv[k] / (k + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def smooth_sequence(s, width):
    """Apply Gaussian smoothing and re-normalize."""
    n = len(s)
    x = np.arange(n)
    center = n // 2
    kernel = np.exp(-(x - center) ** 2 / (2.0 * width ** 2))
    kernel = kernel / kernel.sum()
    s_smooth = fft_convolve(s, kernel)
    s_smooth = s_smooth[:n]
    s_smooth = np.maximum(s_smooth, 0.0)
    total = s_smooth.sum()
    if total < 1e-15:
        return s
    return s_smooth / total


def adam_optimize(s0, tau_schedule, steps_per_tau, lr, deadline, smooth_every=0):
    """Adam optimization with continuation on tau. Returns best sequence and score."""
    n = len(s0)
    s = s0.copy()
    s = np.maximum(s, 0.0)
    s = s / s.sum()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_s = s.copy()
    best_score = fast_score(s)
    beta1, beta2, eps = 0.8, 0.999, 1e-8

    for phase, tau in enumerate(tau_schedule):
        stagnation = 0
        for step in range(steps_per_tau):
            if deadline is not None and time.monotonic() > deadline:
                break

            t += 1
            g = softmax_gradient(s, tau)

            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * g * g
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            s_new = s - lr * m_hat / (np.sqrt(v_hat) + eps)
            s_new = project_simplex(s_new)

            # Periodic smoothing to escape local minima
            if smooth_every > 0 and step > 0 and step % smooth_every == 0:
                s_new = smooth_sequence(s_new, n * 0.05)

            score = fast_score(s_new)
            if score < best_score - 1e-12:
                best_score = score
                best_s = s_new.copy()
                stagnation = 0
            else:
                stagnation += 1

            s = s_new

        print(f"    tau={tau:.5f}, current={fast_score(s):.6f}, best={best_score:.6f}")

    return best_s, best_score


def build_conv_matrix(s):
    """Build (2n-1) x n matrix for convolution constraints."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def lp_refine(s, max_iters=50, deadline=None):
    """LP-guided local search refinement."""
    n = len(s)
    s = s.copy()
    s = np.maximum(s, 0.0)
    s = s / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)
    no_imp = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = fft_convolve(s, s)
        M = np.max(conv)

        A = build_conv_matrix(s)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': 3.0, 'presolve': True})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0.0)
            else:
                break
        except Exception:
            break

        b_sum = b.sum()
        if b_sum < 1e-14:
            break
        b = b / b_sum

        best_t = 0.0
        best_ts = best_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            cand = (1.0 - t) * s + t * b
            cand = np.maximum(cand, 0.0)
            cs = fast_score(cand)
            if cs < best_ts - 1e-12:
                best_ts = cs
                best_t = t

        if best_t == 0.0:
            no_imp += 1
            if no_imp > 6:
                break
            # Perturb
            rng = np.random.default_rng(it)
            s = s + 0.01 * rng.standard_normal(n)
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            continue

        s = (1.0 - best_t) * s + best_t * b
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        no_imp = 0

        if best_ts < best_score:

            best_s = s.copy()

        global_best_seq = np.ones(256) / 256.0
        global_best_score = fast_score(global_best_seq)

    # Verify with actual evaluator
    final_verified = float(evaluate_sequence(global_best_seq.tolist()))
    print(f"\nFinal verified score: {final_verified:.6f}, length={len(global_best_seq)}")

    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
