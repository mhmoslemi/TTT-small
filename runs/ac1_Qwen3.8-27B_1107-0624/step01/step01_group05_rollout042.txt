to reduce the peak convolution, then polishes with L-BFGS-B on a smooth surrogate.
Tries multiple sequence lengths (512, 1024, 2048, 4096) and returns the best.

import numpy as np

# ---------- Core computation ----------
def _fft_size(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _full_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = _fft_size(n)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def _self_conv(p):
    """Autocorrelation-like: full convolution of p with itself."""
    return _full_conv(p, p)

def _score_p(p):
    """Compute 2*n*max_conv/sum²."""
    n = len(p)
    s = float(np.sum(p))
    if s < 0.01:
        return float('inf')
    c = _self_conv(p)
    return float(2.0 * n * np.max(c) / (s * s))

def _max_conv(p):
    return float(np.max(_self_conv(p)))

def _lp_step(p, verbose=False):
    """
    One LP iteration: find q >= 0 maximizing sum(q)
    subject to (p*q)[k] <= M for all k.
    Returns (q_normalized, improvement_factor).
    """
    n = len(p)
    M = _max_conv(p)
    if M <= 0:
        return p.copy(), 1.0
    
    # Build constraint matrix A (2n-1) x n, A[k,i] = p[k-i] if 0 <= k-i < n
    # Column i has p[0..n-1] at rows i..i+n-1
    A = sp.lil_matrix((2*n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = p
    A = A.tocsr()
    
    c = -np.ones(n)  # minimize -sum(q)
    b = np.full(2*n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': 60})

        if verbose:
            print(f"  LP failed: {e}")
        return p.copy(), 1.0
    
    if not res.success or res.x is None:
        return p.copy(), 1.0
    
    q = np.maximum(res.x, 0.0)
    sum_q = float(np.sum(q))
    if sum_q <= 1.0 + 1e-12:
        return p.copy(), sum_q
    
    q_norm = q / sum_q
    return q_norm, sum_q

# ---------- Gradient-based polishing ----------
def _grad_from_weights(p, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. p."""
    # conv[k] = sum_j p[j]*p[k-j]
    # d/dp[i] sum_k w[k]*conv[k] = 2 * sum_j p[j] * w[i+j]
    # (factor of 2 from the two appearances of p[i] in the product)
    n = len(p)
    # We need h[i] = sum_j p[j] * w[i+j] for i=0..n-1
    # This is the correlation of p (reversed) with w, offset appropriately
    p_rev = p[::-1]
    conv_pw = _full_conv(p_rev, w)
    # conv_pw[m] = sum_j p_rev[j] * w[m-j] = sum_j p[n-1-j] * w[m-j]
    # Let j' = n-1-j: = sum_j' p[j'] * w[m - n + 1 + j']
    # So conv_pw[n-1+i] = sum_j' p[j'] * w[i + j']
    h = conv_pw[n-1 : 2*n-1]
    return 2.0 * h

"""
Minimize 2*n*max_conv/sum^2 via LP-based refinement, KKT fixed-point,
and gradient polishing. Tries multiple n values starting from known good sequences.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog, minimize
from scipy.sparse.linalg import spsolve

def _fft_size(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _full_conv(a, b):
    n = len(a) + len(b) - 1
    sz = _fft_size(n)
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:n]


def _self_conv(p):
    return _full_conv(p, p)


def _score_p(p):
    n = len(p)
    s = float(np.sum(p))
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * np.max(_self_conv(p)) / (s * s))


def _max_conv(p):
    return float(np.max(_self_conv(p)))

    global _best_score, _best_seq

        _best_score = float(score)
        _best_seq = np.array(seq, copy=True)


def _probs_from_z(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64), -50.0, 50.0)
    m = float(np.max(z)) if z.size else 0.0
    ez = np.exp(z - m)
    e0 = np.exp(-m)
    Z = e0 + float(np.sum(ez))
    p = np.empty(n, dtype=np.float64)
    p[0] = e0 / Z
    p[1:] = ez / Z
    return p


def _density_to_z(f, n):
    f = np.maximum(f, 1e-15)
    lf = np.log(f)
    return np.clip(lf[1:] - lf[0], -50.0, 50.0)


def _grad_w(p, w):
    """d/dp[sum_k w[k]*conv[k]] = 2 * sum_j p[j]*w[i+j]."""
    n = len(p)
    cr = _full_conv(p[::-1], w)
    return 2.0 * cr[n - 1:2 * n - 1]


# ---------- LP phase ----------

def _lp_step(p, tlim=20.0):
    n = len(p)
    M = _max_conv(p)
    if M <= 0:
        return p.copy(), 1.0
    A = sp.lil_matrix((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = p
    A = A.tocsr()
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                    bounds=[(0.0, None)] * n, method='highs',
                    options={'time_limit': tlim})
    except Exception:
        return p.copy(), 1.0
    if not r.success or r.x is None:
        return p.copy(), 1.0
    q = np.maximum(r.x, 0.0)
    sq = float(np.sum(q))
    if sq <= 1.0 + 1e-12:
        return p.copy(), sq
    return q / sq, sq


def _lp_phase(p, budget, tag="LP"):

def _resized(seq_target, n):
    st = np.asarray(seq_target, dtype=np.float64)
    st = np.where(np.isfinite(st), st, 0.0)
    st = np.clip(st, 0.0, None)
    if st.sum() <= 0:
        return np.full(n, 1.0 / n)
    if st.size == n:
        pn = st / st.sum()
    else:
        old_x = np.linspace(0, 1, st.size)
        new_x = np.linspace(0, 1, n)
        pn = np.interp(new_x, old_x, st)
        pn = np.maximum(pn, 0.0)
        if pn.sum() <= 0:
            return np.full(n, 1.0 / n)
        pn /= pn.sum()
    return pn
