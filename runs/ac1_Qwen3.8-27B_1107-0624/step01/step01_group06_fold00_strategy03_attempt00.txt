def convolve_fft(a):
    n = len(a)
    if n == 0:

    # power-of-two to speed up FFT -> # linear convolution is full mode

    if n_val == 0:
        return 0.0
    convolve_result = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    max_area = np.max(convolve_result)
    return float(2 * n_val * max_area / (S**2))

def smooth_J_and_grad(a, tau):
    """Computes smoothed J and gradient for Adam-long protocol"""
    n_val = len(a)
    if n_val == 0 or not(isinstance(a, (list, np.ndarray))):
        return 1e15, np.zeros(n_val)
    # FFT smoother for lower bound risks: compute convolution for self-sum feature
    size = 1
    while size < 3*n_val - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    cross_correlation = np.fft.irfft(fa * np.conj(fa), size)  # Effects earlier than self
    # Now normalize the convolution input
    c_norm = cross_correlation - np.log(np.exp(cross_correlation - max(cross_correlation)) / tau)
    S_val_normalized = np.max(c_norm) + tau * np.log(np.array(len(c_norm)))
    J_val_smoothed = 2 * n_val * S_val_normalized / (np.sum(a))**2
    
    # Gradient calculation: we model via projection
    exp_sm = np.exp(c_norm / tau)
    grad_val = (4.0 * n_val * compute_gradient(c_norm, a, tau)) / (np.sum(a))**2
    return J_val_smoothed, grad_val

def make_taper(n):
    """Taper with background resonance; uses np.exp for smooth bump"""
    max_index = (n-1) / 2.0
    if max_index < 0.5:
        # Ensure at least a baseline width
        sigma = 10
    else:
        sigma = n // 3
    arr = np.exp(-(((np.arange(n)) - max_index)/sigma)**2)
    # Boost the tails minimally to counteract FFT round-off losses
    arr *= 1/np.sqrt(2 * np.pi * sigma**2)
    return arr
