"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 for a >= 0.
Uses Adam-based smoothed-max descent with adaptive annealing,
multi-length search, and final polish.
"""

import time
import math
import numpy as np


def _to_sum1(a):
    s = a.sum()
    if s <= 1e-12:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _conv_full(a):
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.zeros(n, dtype=np.float64)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_grad(c, a, L, tau):
    n = a.size
    nc = len(c)
    L = min(L, nc)
    if L < nc:
        top_idx = np.argpartition(-c, L - 1)[:L]
    else:
        top_idx = np.arange(nc)
    vals = c[top_idx]
    vmax = vals.max()
    exp_vals = np.exp((vals - vmax) / tau)
    wsum = exp_vals.sum()
    if wsum < 1e-30:
        weights = np.ones(L) / L
    else:
        weights = exp_vals / wsum
    grad = np.zeros(n, dtype=np.float64)
    for k, w in zip(top_idx, weights):
        if w < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            src = k - np.arange(lo, hi + 1)
            grad[lo:hi + 1] += 2.0 * w * a[src]
    return grad


def _opt_run(a_init, t_end, L0=300, tau0=0.05, lr0=0.03):
    """Adam-based smoothed-max descent."""
    n = a_init.size
    a = _to_sum1(np.abs(a_init).astype(np.float64))
    m_arr = np.zeros(n, dtype=np.float64)
    v_arr = np.zeros(n, dtype=np.float64)
    adam_t = 0
    L = L0
    tau = tau0
    lr = lr0
    best_a = a.copy()
    best_raw = 2.0 * n * _conv_full(a).max()
    best_eval = np.inf
    best_eval_a = a.copy()
    step = 0
    last_ev = 0.0
    no_improve = 0
    while time.monotonic() < t_end - 0.5:
        c = _conv_full(a)
        raw = 2.0 * n * c.max()
        if raw < best_raw:
            best_raw = raw
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        step += 1
        if step % 300 == 0:
            L = max(3, int(L * 0.88))
            tau = max(0.002, tau * 0.82)
        if step % 100 == 0:
            lr = max(0.0005, lr * 0.97)
        if no_improve > 2000:
            # Reset Adam state and perturb
            m_arr[:] = 0
            v_arr[:] = 0
            adam_t = 0
            a = _to_sum1(np.abs(a + 0.01 * np.random.default_rng(step).normal(0, 1, n)))
            no_improve = 0
            lr = min(lr * 2.0, 0.05)
        grad = _smoothed_grad(c, a, L, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            grad = grad / gnorm
        adam_t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m_arr = b1 * m_arr + (1 - b1) * grad
        v_arr = b2 * v_arr + (1 - b2) * (grad ** 2)
        m_hat = m_arr / (1 - b1 ** adam_t)
        v_hat = v_arr / (1 - b2 ** adam_t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _proj_simplex(np.maximum(a_new, 0), 1.0)
        if time.monotonic() - last_ev > 5.0:

            try:

                if np.isfinite(s) and s < best_eval:

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 for a >= 0, |a| = n.
Uses Adam-based smoothed-max descent with aggressive annealing of the
top-L/temperature parameters, multiple restarts, and a final sharp polish.
Explores multiple sequence lengths for better attack surface.
"""

import time
import math
import numpy as np


def _to_sum1(a):
    s = a.sum()
    if s <= 1e-12:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _conv_full(a):
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.zeros(n, dtype=np.float64)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_grad(c, a, L, tau):
    """Gradient of softmax-smoothed max over top-L lags of c."""
    n = a.size
    nc = len(c)
    L = min(L, nc)
    if L < nc:
        top_idx = np.argpartition(-c, L - 1)[:L]
    else:
        top_idx = np.arange(nc)
    vals = c[top_idx]
    vmax = vals.max()
    exp_vals = np.exp((vals - vmax) / max(tau, 1e-8))
    wsum = exp_vals.sum()
    if wsum < 1e-30:
        weights = np.full(L, 1.0 / L)
    else:
        weights = exp_vals / wsum
    grad = np.zeros(n, dtype=np.float64)
    for k, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            src = k - np.arange(lo, hi + 1)
            grad[lo:hi + 1] += 2.0 * w * a[src]
    return grad


def _opt_run(a_init, t_end, L0, tau0, lr0, seed=0):
    """Adam-based smoothed-max descent with adaptive annealing."""
    n = a_init.size
    a = _to_sum1(np.abs(a_init).astype(np.float64))
    m_arr = np.zeros(n, dtype=np.float64)
    v_arr = np.zeros(n, dtype=np.float64)
    adam_t = 0
    L = L0
    tau = tau0
    lr = lr0
    best_a = a.copy()
    best_raw = 2.0 * n * _conv_full(a).max()
    best_eval = np.inf
    best_eval_a = a.copy()
    step = 0
    last_ev = 0.0
    no_improve = 0
    rng = np.random.default_rng(seed)
    b1, b2, eps = 0.9, 0.999, 1e-8
    while time.monotonic() < t_end - 0.5:
        c = _conv_full(a)
        raw = 2.0 * n * c.max()
        if raw < best_raw * (1 - 1e-7):
            best_raw = raw
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        step += 1
        # Anneal schedule
        if step % 200 == 0:
            L = max(3, int(L * 0.87))
            tau = max(0.001, tau * 0.80)
        if step % 50 == 0:
            lr = max(0.0003, lr * 0.985)
        # Reset if stuck

            m_arr[:] = 0.0
            v_arr[:] = 0.0
            adam_t = 0
            a = _to_sum1(np.abs(a + 0.005 * rng.normal(0, 1, n)))
            no_improve = 0
            lr = min(lr * 3.0, 0.02)
        # Compute gradient
        grad = _smoothed_grad(c, a, L, tau)
        # Adam update
        adam_t += 1
        m_arr = b1 * m_arr + (1 - b1) * grad
        v_arr = b2 * v_arr + (1 - b2) * (grad ** 2)
        m_hat = m_arr / (1 - b1 ** adam_t)
        v_hat = v_arr / (1 - b2 ** adam_t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _proj_simplex(np.maximum(a_new, 0.0), 1.0)
        # Periodic official evaluation

            try:
