def smooth_max_grad(a, temperature):
    """Compute gradient of log-sum-exp of auto-convolution."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    # log-sum-exp for numerical stability
    c_max = c.max()
    exp_c = np.exp((c - c_max) / temperature)
    weights = exp_c / exp_c.sum()
    # Gradient with respect to a[i] is sum_k weights[k] * a[k-i]
    grad = np.zeros(n)
    for i in range(n):
        js = np.arange(len(c)) - i
        valid = (js >= 0) & (js < n)
        if valid.any():
            grad[i] = np.dot(weights[valid], a[js[valid]])
    return grad, c.max()

    for t in ts:
        val = ((1.0 - t) ** 2 * c_aa + 2.0 * t * (1.0 - t) * c_ag + t ** 2 * c_gg).max()
        if val < best_val:
            best_val = val
            best_t = t

    # Golden section refinement around best grid point
    if best_t > 0.005 and best_t < 1.0:
        lo = max(0.0, best_t - 0.015)
        hi = min(1.0, best_t + 0.015)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(40):
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)
            f1 = ((1.0 - m1) ** 2 * c_aa + 2.0 * m1 * (1.0 - m1) * c_ag + m1 ** 2 * c_gg).max()
            f2 = ((1.0 - m2) ** 2 * c_aa + 2.0 * m2 * (1.0 - m2) * c_ag + m2 ** 2 * c_gg).max()
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        t_star = (lo + hi) / 2.0
        f_star = ((1.0 - t_star) ** 2 * c_aa + 2.0 * t_star * (1.0 - t_star) * c_ag +
                  t_star ** 2 * c_gg).max()
        if f_star < best_val:
            best_t = t_star
            best_val = f_star

    return best_t


def optimize_lp(a, deadline, rng):
    """LP-based optimization with exact line search."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    stall = 0
    it = 0
    last_print = 0

    while time.monotonic() < deadline:
        it += 1
        c_aa = np.convolve(a, a, mode='full')
        M = c_aa.max()

        if M < 1e-15:
            break

        # Try LP with current peak
        g = lp_solve(a, M)
        if g is None or g.sum() < 1e-15:
            # Try slightly relaxed
            g = lp_solve(a, M * 1.001)
        if g is None or g.sum() < 1e-15:
            break

        g = np.clip(g, 0.0, None)
        if g.sum() < 1e-15:
            break
        g = g / g.sum()

        # Precompute for line search
        c_ag = np.convolve(a, g, mode='full')
        c_gg = np.convolve(g, g, mode='full')

        t = line_search(a, g, c_aa, c_ag, c_gg)

        improved = False
        if t > 0.001:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() >= 1e-10:
                a_new = a_new / a_new.sum()
                sc = compute_objective(a_new)
                if sc < best_score - 1e-12:
                    best_a = a_new
                    best_score = sc
                    a = a_new
                    stall = 0
                    improved = True

        if not improved:
            stall += 1

        # Restart strategy
        if stall > 0 and stall % 8 == 0:
            strength = min(0.3, 0.03 * stall)
            noise = np.abs(rng.normal(0, strength, n))
            a_new = best_a * (1.0 + noise)
            a_new = np.clip(a_new, 1e-15, None)
            if a_new.sum() > 1e-10:
                a_new /= a_new.sum()
                sc = compute_objective(a_new)
                if sc < best_score:
                    best_a = a_new
                    best_score = sc
                    a = a_new
                    stall = 0
                elif rng.random() < 0.25:
                    a = a_new
                    stall = 0
                else:
                    a = best_a.copy()

        if stall > 60:
            # Big restart
            noise = np.abs(rng.normal(0, 0.15, n))
            a = best_a * (1.0 + noise)
            a = np.clip(a, 1e-15, None)
            a = a / a.sum()
            stall = 0

        if stall > 150:
            break

        if it - last_print >= 200:
            print(f"      LP iter {it}, score={best_score:.6f}, stall={stall}, t={t:.4f}")
            last_print = it

    return best_a, best_score


def smooth_gradient_step(a, temperature, lr):
    """One step of smooth gradient descent using log-sum-exp approximation."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    c_max = c.max()

    # Smooth max weights
    exp_c = np.exp((c - c_max) / temperature)
    w = exp_c / exp_c.sum()

    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        max_j = min(n, 2 * n - 1 - i)
        if max_j > 0:
            grad[i] = 2.0 * np.dot(w[i:i + max_j], a[:max_j])

    # Normalize gradient
    grad_mean = grad.mean() + 1e-15
    grad_norm = grad / grad_mean

    # Multiplicative update
    a_new = a * np.exp(-lr * grad_norm)
    a_new = np.clip(a_new, 1e-15, None)
    a_new = a_new / a_new.sum()

    return a_new, c.max(), grad


def optimize_gradient(a, deadline, rng):
    """Smooth gradient descent with adaptive parameters."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    lr = 0.08
    temperature = 0.05
    stall = 0
    it = 0
    last_print = 0

    while time.monotonic() < deadline:
        it += 1
        a_new, c_max, grad = smooth_gradient_step(a, temperature, lr)
        sc = compute_objective(a_new)

        if sc < best_score - 1e-12:
            best_a = a_new
            best_score = sc
            a = a_new
            stall = 0
            temperature = max(0.002, temperature * 0.998)
        else:
            stall += 1
            lr = max(0.002, lr * 0.92)

        if stall > 25:
            strength = min(0.2, 0.02 * stall)
            noise = np.abs(rng.normal(0, strength, n))
            a = best_a * (1.0 + noise)
            a = np.clip(a, 1e-15, None)
            a /= a.sum()
            stall = 0

        if stall > 120:
            break

        if it - last_print >= 500:
            print(f"      Grad iter {it}, score={best_score:.6f}, lr={lr:.4f}, T={temperature:.5f}")
            last_print = it

    return best_a, best_score


def init_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0.0, np.pi, n)))


def init_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def init_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def init_random(n, rng):
    return np.abs(rng.normal(0, 1, n))


def init_sinc(n, width=0.4):
    """Approximate sinc window."""
    x = np.linspace(-1, 1, n)
    return np.maximum(0.0, np.sinc(width * x))


def init_plateau(n, frac=0.5):
    """Plateau (top-hat) window."""
    x = np.linspace(-1, 1, n)
    result = np.ones(n)
    result[np.abs(x) > frac] = 0.0
    return result + 0.1 * np.abs(np.random.default_rng(0).normal(0, 1, n))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-start LP + gradient optimization to minimize the auto-convolution peak ratio.
    Uses exact line search for LP directions and smooth gradient for large n.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 8.0

    best_seq = None
    best_score = np.inf

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all() and p.sum() > 0:
            parent = p.copy()
    except Exception:
        pass

    print(f"Starting optimization. Budget={budget_s}s, deadline in {budget_s - 8:.0f}s")
    if parent is not None:
        print(f"Parent sequence available: n={len(parent)}, score={compute_objective(parent):.6f}")

    # Define search configurations: (n, init_func, method)
    configs = []

    # Parent-based (highest priority)
    if parent is not None:
        pn = len(parent)
        configs.append((pn, parent.copy(), 'lp'))
        if pn > 512:
            configs.append((512, parent[:512].copy(), 'lp'))
        if pn <= 512:
            configs.append((1024, np.concatenate([parent, parent[::-1]] * (1024 // pn + 1))[:1024], 'lp'))

    # Standard inits at n=512 (sweet spot for LP speed)
    for init_f in [init_cosine, init_gaussian, init_parabolic, init_sinc, init_plateau]:
        configs.append((512, init_f(512), 'lp'))

    # Random inits at n=512
    for _ in range(3):
        configs.append((512, init_random(512, rng), 'lp'))

    # n=1024
    for init_f in [init_cosine, init_gaussian]:
        configs.append((1024, init_f(1024), 'lp'))

    # n=256 (fast, good for exploration)
    configs.append((256, init_cosine(256), 'lp'))
    configs.append((256, init_gaussian(256), 'lp'))

    # n=2048 (use gradient method, slower LP)
    configs.append((2048, init_cosine(2048), 'gradient'))
    configs.append((2048, init_gaussian(2048, 0.2), 'gradient'))

    # Time allocation: weight by promisingness
    # n=512 LP configs get the most time
    total_weight = 0.0
    weights = []
    for i, (n, init_a, method) in enumerate(configs):
        if method == 'lp':
            if n == 512:
                w = 1.5
            elif n == 1024:
                w = 1.0
            elif n == 256:
                w = 0.8
            else:
                w = 1.0
        else:
            w = 0.7
        weights.append(w)
        total_weight += w

    available_time = budget_s - 40.0  # reserve 40s for final polish
    time_allocs = []
    cum = 0.0
    for w in weights:
        cum += available_time * w / total_weight
        time_allocs.append(cum)

    n_configs = len(configs)
    print(f"Total configs: {n_configs}")

    for idx, (n, init_a, method) in enumerate(configs):
        if time.monotonic() > deadline:
            break

        config_deadline = min(start_time + time_allocs[idx], deadline)
        if config_deadline <= time.monotonic():
            continue

        a = np.clip(init_a, 1e-10, None).astype(np.float64)
        if a.sum() < 1e-10:
            a = np.ones(n)
        a = a / a.sum()

        elapsed = time.monotonic() - start_time
        remaining = deadline - time.monotonic()
        config_time = config_deadline - time.monotonic()
        print(f"[{idx + 1}/{n_configs}] n={n}, method={method}, alloc={config_time:.0f}s, remaining={remaining:.0f}s")

        try:
            if method == 'lp' and n <= 1536:
                a_opt, score = optimize_lp(a, config_deadline, rng)
            else:
                a_opt, score = optimize_gradient(a, config_deadline, rng)
        except Exception as e:
            print(f"      Error: {e}")
            a_opt, score = a, compute_objective(a)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: score={best_score:.6f} (n={n}, config={idx + 1})")
        else:
            print(f"  => score={score:.6f} (best: {best_score:.6f})")

    # Final polish on best sequence
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nFinal polish on best (score={best_score:.6f}, n={len(best_seq)})...")
        n_best = len(best_seq)
        if n_best <= 1536:
            best_seq, best_score = optimize_lp(best_seq, deadline, rng)
        else:
            best_seq, best_score = optimize_gradient(best_seq, deadline, rng)
        print(f"  After polish: score={best_score:.6f}")

    if best_seq is None:
        best_seq = init_cosine(512)
        best_score = compute_objective(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Ensure valid for evaluator
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (1.0 / best_seq.sum())

    elapsed_total = time.monotonic() - start_time
    print(f"\nDONE in {elapsed_total:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
