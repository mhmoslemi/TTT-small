def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _conv_sq(a, m=None):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    if m is None:
        m = 1
        while m < 2 * n - 1:
            m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]

def _smoothed_obj_grad(c, a, L, tau):
    """Compute smoothed objective value and gradient using top-L softmax."""
    n = a.size
    L = min(L, 2 * n - 1)
    # Get top L indices
    if L >= 2 * n - 1:

    else:
        order = np.argpartition(-c, L - 1)[:L]
    wc = c[order]
    # Softmax with temperature
    wc_shifted = wc - wc.max()
    w = np.exp(wc_shifted / max(tau, 1e-10))

    grad = np.zeros(n, dtype=np.float64)

        k = int(k)

        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            src = k - idx
            grad[idx] += 2.0 * wk * a[src]

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using multi-start Adam optimization
with temperature-annealed smoothed max, on the non-negative simplex.
"""

import time
import math
import numpy as np


def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_sq(a, m):
    n = a.size
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _get_L(m, n):
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    return m


def _smoothed_max_and_grad(c, a, L, tau):
    """Returns (smoothed_max_value, gradient) using top-L softmax smoothing."""
    n = a.size
    length = 2 * n - 1
    L = min(L, length)
    
    if L >= length:
        idx = np.arange(length)
    else:
        idx = np.argpartition(-c, L - 1)[:L]
    
    vals = c[idx]
    vmax = vals.max()
    w = np.exp((vals - vmax) / max(tau, 1e-10))
    w_sum = w.sum()
    w = w / w_sum
    
    smooth_val = float(np.dot(w, vals))
    
    # grad[j] = 2 * sum_{k in idx} w[k] * a[k-j] for 0 <= k-j < n
    grad = np.zeros(n, dtype=np.float64)
    for k, wk in zip(idx.astype(int), w):
        if wk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            # grad[lo:hi+1] += 2*wk*a[k-hi : k-lo+1][::-1]
            # a[k - j] for j in lo..hi means a[k-hi], a[k-hi+1], ..., a[k-lo]
            # which is a[k-lo : k-hi+1][::-1]... 
            # Let me just be explicit:
            # j ranges from lo to hi
            # a[k-j] ranges from a[k-lo] down to a[k-hi]
            # So grad[lo:hi+1] += 2*wk * a[k-hi : k-lo+1][::-1]
            grad[lo:hi + 1] += 2.0 * wk * a[k - hi:k - lo + 1][::-1]
    
    return smooth_val, grad


def _adam_step(a, grad, m_t, v_t, t, lr, beta1, beta2, eps, m_fft):
    """One Adam step with projection to simplex."""
    m_t = beta1 * m_t + (1 - beta1) * grad
    v_t = beta2 * v_t + (1 - beta2) * grad ** 2
    m_hat = m_t / (1 - beta1 ** t)
    v_hat = v_t / (1 - beta2 ** t)
    a_new = a + lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = _proj_simplex(a_new, 1.0)
    return a_new, m_t, v_t

def _smoothed_max_grad(c, a, tau, topL):
    """Compute softmax-smoothed max value and gradient w.r.t. a.
    Uses top-L lags for efficiency. Returns (value, gradient)."""
    n = a.size
    length = 2 * n - 1

    if L >= length:

def _adam_update(a, grad, m_buf, v_buf, t, lr, b1, b2, eps):
    """One Adam step with simplex projection."""
    m_buf = b1 * m_buf + (1.0 - b1) * grad
    v_buf = b2 * v_buf + (1.0 - b2) * grad * grad
    m_hat = m_buf / (1.0 - b1 ** t)
    v_hat = v_buf / (1.0 - b2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a + step
    a_new = _proj_simplex(a_new, 1.0)
    return a_new, m_buf, v_buf

    times = []
    for i in range(n_seeds):
        w = 3.0 if i == 0 else 1.0
        times.append(w / total_weight * remaining * 0.5)  # Use 50% for phase 1

    for i in range(n_seeds):
        if time.monotonic() >= t_end - 10.0:
            break
        tower = time.monotonic() + times[i]
        a_best, mc = _optimize(ordered_seeds[i], tower, rng, m)
        if mc < best_mc:
            best_mc = mc
            best_a = a_best.copy()
            try:
                sc = float(evaluate_sequence([float(x) for x in a_best]))
                if sc < best_official:
                    best_official = sc
                    print(f"Phase1 seed{i}: NEW BEST score={sc:.6f}")
            except Exception:
                pass
        ordered_seeds[i] = a_best

    print(f"After Phase1: best_mc={best_mc:.6f}, best_official={best_official:.6f}, "
          f"time_left={t_end - time.monotonic():.1f}s")

    # Phase 2: Long optimization from best + perturbations
    if time.monotonic() < t_end - 30.0:
        # Create perturbations of the current best
        perturb_seeds = [best_a.copy()]
        for _ in range(4):
            p = best_a * np.exp(rng.normal(0, 0.04, n))
            p = _proj_simplex(p, 1.0)
            perturb_seeds.append(p)

        # Also mix best with other top seeds
        for i in range(1, min(3, len(ordered_seeds))):
            mix = 0.6 * best_a + 0.4 * ordered_seeds[i]
            mix = _proj_simplex(mix, 1.0)
            perturb_seeds.append(mix)

        remaining2 = t_end - time.monotonic() - 5.0
        time_per_seed = remaining2 * 0.4 / len(perturb_seeds)

        for i, ps in enumerate(perturb_seeds):
            if time.monotonic() >= t_end - 10.0:
                break
            tower = time.monotonic() + time_per_seed
            a_best, mc = _optimize(ps, tower, rng, m)
            if mc < best_mc:
                best_mc = mc
                best_a = a_best.copy()
                try:
                    sc = float(evaluate_sequence([float(x) for x in a_best]))
                    if sc < best_official:
                        best_official = sc
                        print(f"Phase2 pert{i}: NEW BEST score={sc:.6f}")
                except Exception:
                    pass

    print(f"After Phase2: best_mc={best_mc:.6f}, best_official={best_official:.6f}, "
          f"time_left={t_end - time.monotonic():.1f}s")

    # Phase 3: Final intensive optimization from absolute best
    if time.monotonic() < t_end - 10.0:
        tower = t_end - 2.0
        a_best, mc = _optimize(best_a, tower, rng, m)
        if mc < best_mc:
            best_mc = mc
            best_a = a_best.copy()
        try:
            sc = float(evaluate_sequence([float(x) for x in a_best]))
            if sc < best_official:
                best_official = sc
                best_a = a_best.copy()
                print(f"Phase3: NEW BEST score={sc:.6f}")
        except Exception:
            pass

    # Final validation and return
    out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.full(n, 1.0)

    try:
        final_score = float(evaluate_sequence([float(x) for x in out]))
        print(f"FINAL: score={final_score:.6f}")
        if final_score < best_official:
            best_official = final_score
    except Exception as e:
        print(f"Final eval error: {e}")

    return [float(x) for x in out]
