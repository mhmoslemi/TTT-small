"""
Search for non-negative step-function sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using a combination of softmax-projected gradient descent and LP-based direction finding
with aggressive line search. The LP phase is the main improvement: for n<=512 the full
convolution matrix is small enough to solve the LP with all constraints at each iteration.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(a.size) / max(a.size, 1)
    return a / s


def _score_from_conv(c, n):
    return 2.0 * float(n) * float(np.max(c))


def _score(a):
    """Score assuming sum(a)=1."""
    n = a.size
    if n == 0:
        return np.inf
    c = np.convolve(a, a)
    return _score_from_conv(c, n)


def _greedy_init(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return _normalize(a)


def _power_init(n, s):
    idx = np.arange(n, dtype=np.float64) + 1.0
    return _normalize(np.power(idx, -s))


def _uniform_init(n):
    return np.full(n, 1.0 / n)


def _perturb(a, rng, sigma):
    noise = np.exp(rng.normal(0.0, sigma, a.size))
    return _normalize(a * noise)


def _mix(a, b, t):
    return _normalize((1.0 - t) * a + t * b)


def _build_conv_matrix(f, n):
    """A[k, i] = f[k-i] if 0<=k-i<n, else 0. Shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _adam_phase(a, budget_s, alpha_init=30.0, alpha_end=3.0):
    """Softmax-projected gradient descent on log-sum-exp smoothed max-conv."""
    n = a.size
    a = _normalize(a)
    best_a = a.copy()
    best_f = _score(a)

    if budget_s < 1.0:
        return best_a, best_f

    L = 2 * n  # FFT length (enough for full conv of length 2n-1)
    lr = 0.04 if n <= 256 else (0.025 if n <= 512 else 0.015)
    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    x = np.log(a + 1e-300)
    x -= x.mean()

    start = time.monotonic()
    end = start + budget_s

    while time.monotonic() < end:
        t += 1
        frac = min(1.0, (time.monotonic() - start) / max(1e-9, budget_s))
        alpha = alpha_init + (alpha_end - alpha_init) * frac

        Pf = np.fft.rfft(a, L)
        c = np.fft.irfft(Pf * Pf, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = c.max()
        if maxc <= 0 or not np.isfinite(maxc):
            break

        f_now = _score_from_conv(c, n)
        if f_now < best_f:
            best_f = f_now
            best_a = a.copy()

        z = alpha * (c / maxc - 1.0)
        z -= z.max()
        expz = np.exp(z)
        sz = expz.sum()
        if sz < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient: d/d(a[i]) of sum_k w[k]*(a*a)[k] = 2 * (w_rev * a)[i]
        Wr = w[::-1]
        Pw = np.fft.rfft(Wr, L)
        C = np.fft.irfft(Pf * Pw, L)
        grad = 2.0 * C[n - 1:2 * n - 1][::-1].real
        grad = np.maximum(grad, 0.0)

        gp = float(np.dot(a, grad))
        gx = a * (grad - gp)

        m = mu * m + (1 - mu) * gx
        v = nu * v + (1 - nu) * gx * gx
        mhat = m / (1 - mu ** t)
        vhat = v / (1 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()
        x = np.clip(x, -60.0, 60.0)
        x -= x.max()
        e = np.exp(x)
        se = e.sum()
        if not np.isfinite(se) or se < 1e-300:
            break
        a = e / se

    final_f = _score(a)
    if final_f < best_f:
        best_f = final_f
        best_a = a.copy()
    return best_a, best_f


def _lp_refine(a, budget_s, max_iters=50000):
    """
    LP-based refinement. a is normalized to sum=1.
    At each iteration:
      M = max(a*a)
      Solve: max sum(g) s.t. A*g <= M, g >= 0
      Line search: a_new = (1-t)*a + t*g for best t
    """
    n = a.size
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(a)
    cur = a.copy()
    cur_score = best_score

    A = _build_conv_matrix(cur, n)
    ones_b = np.ones(2 * n - 1, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)

    start = time.monotonic()
    end = start + budget_s
    it = 0
    stall = 0

    while time.monotonic() < end and it < max_iters:
        it += 1

        # Recompute M and matrix (matrix changes with cur)
        c = np.convolve(cur, cur)
        M = float(c.max())
        if not np.isfinite(M) or M <= 0:
            break

        # Rebuild A (needed because cur changed)
        A = _build_conv_matrix(cur, n)
        b_vec = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve LP
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=(0, None),
                          method='highs', options={'time_limit': 5.0, 'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        sg = g.sum()
        if sg < 1e-15:
            break

        g = g / sg  # normalize to sum 1

        # Line search
        best_t = 0.0
        best_sc = cur_score
        for tt in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            cand = (1.0 - tt) * cur + tt * g
            cand = np.maximum(cand, 0.0)
            sc = _score(cand)
            if sc < best_sc:
                best_sc = sc
                best_t = tt

        if best_t == 0.0 or best_sc >= cur_score - 1e-12:
            stall += 1
            if stall > 50:
                break
            # Try a small random perturbation to escape
            rng_esc = np.random.default_rng(it * 7919)
            cur = _perturb(cur, rng_esc, 0.02)
            cur_score = _score(cur)
            A = _build_conv_matrix(cur, n)
            continue
        else:
            stall = 0
            cur = (1.0 - best_t) * cur + best_t * g
            cur = np.maximum(cur, 0.0)
            cur_score = best_sc
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur.copy()

        if it % 2000 == 0:
            print(f"  LP n={n} it={it} score={best_score:.7f} t={best_t:.3f}")

    return best_a, best_score


def _graduate_lp(a, budget_s):
    """
    Graduated LP: start with a subset of constraints (top violators),
    then add more as we go. Useful for larger n.
    """
    n = a.size
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(a)
    cur = a.copy()

    # Start with all constraints (for n <= 512 this is fine)
    if n <= 600:
        return _lp_refine(cur, budget_s)

    # For larger n, use cutting planes
    c = np.convolve(cur, cur)
    M = c.max()
    # Pick top constraints
    top_k = min(2 * n - 1, max(500, 2 * n // 2))
    idx = np.argsort(c)[::-1][:top_k]
    idx = np.sort(idx)

    start = time.monotonic()
    end = start + budget_s
    it = 0
    stall = 0

    while time.monotonic() < end and it < 30000:
        it += 1
        c = np.convolve(cur, cur)
        M = float(c.max())
        if not np.isfinite(M) or M <= 0:
            break

        # Build sub-matrix
        A = np.zeros((len(idx), n))
        for ii, k in enumerate(idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                A[ii, i] = cur[k - i]
        b_vec = np.full(len(idx), M)
        c_obj = -np.ones(n)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=(0, None),
                          method='highs', options={'time_limit': 5.0, 'presolve': True})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.maximum(res.x, 0.0)
        sg = g.sum()
        if sg < 1e-15:
            break
        g = g / sg

        # Check full feasibility
        cg = np.convolve(cur, g)
        viol = cg - M
        if viol.max() > 1e-7 * M:
            # Add violated constraints
            new_idx = np.where(viol > 1e-7 * M)[0]
            idx = np.sort(np.unique(np.concatenate([idx, new_idx])))
            if len(idx) > 2 * n - 1:
                idx = np.argsort(np.abs(cg) + np.abs(c))[::-1][:2 * n - 1]
                idx = np.sort(idx)
            continue

        # Line search
        best_t = 0.0
        best_sc = _score(cur)
        for tt in [1.0, 0.5, 0.25, 0.1, 0.05]:
            cand = np.maximum((1 - tt) * cur + tt * g, 0.0)
            sc = _score(cand)
            if sc < best_sc:
                best_sc = sc
                best_t = tt

        if best_t == 0 or best_sc >= best_sc - 1e-12:
            stall += 1
            if stall > 30:
                break
            continue

        cur = np.maximum((1 - best_t) * cur + best_t * g, 0.0)
        stall = 0
        if _score(cur) < best_score:
            best_score = _score(cur)
            best_a = cur.copy()

    return best_a, best_score

        if n == primary_n:
            continue
        trials.append((n, 'greedy', 25.0, 100.0))
        trials.append((n, 'power45', 20.0, 80.0))

    # Add perturbation restarts of current best
    if best_a is not None:
        for sigma in [0.05, 0.1, 0.2]:
            trials.append((best_n, f'pert{sigma}', 20.0, 80.0))

    print(f"Total trials planned: {len(trials)}, primary_n={primary_n}")

    # Execute trials
    for n, init_type, g_time, l_time in trials:
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break

        # Build initial point
        if init_type == 'greedy':
            a = _greedy_init(n)
        elif init_type == 'power45':
            a = _power_init(n, 0.45)
        elif init_type == 'power55':
            a = _power_init(n, 0.55)
        elif init_type == 'uniform':
            a = _uniform_init(n)
        elif init_type.startswith('pert'):
            sigma = float(init_type[4:])
            if best_a is not None and best_a.size == n:
                a = _perturb(best_a, rng, sigma)
            else:
                a = _perturb(_greedy_init(n), rng, sigma)
        elif init_type == 'best':
            if best_a is not None and best_a.size == n:
                a = best_a.copy()
            else:
                a = _greedy_init(n)
        else:
            a = _greedy_init(n)

        # Phase 1: Gradient
        alloc_g = min(g_time, rem * 0.3)
        if alloc_g > 2.0:
            a, sc_g = _adam_phase(a, alloc_g)
            _record(a, n, sc_g)

        rem = deadline - time.monotonic()
        if rem < 20.0:
            break

        # Phase 2: LP refinement
        alloc_l = min(l_time, rem * 0.5)
        if alloc_l > 5.0:
            a, sc_l = _lp_refine(a, alloc_l)
            _record(a, n, sc_l)

        rem = deadline - time.monotonic()
        if rem < 15.0:
            break

        # Phase 3: Brief gradient polish after LP
        if rem > 30.0:
            a, sc_p = _adam_phase(a, min(20.0, rem * 0.2))
            _record(a, n, sc_p)

    # Final: if time remains, do more LP on the best
    rem = deadline - time.monotonic()
    if rem > 60.0 and best_a is not None:
        print(f"  Final LP polish: rem={rem:.1f}s")
        a, sc = _lp_refine(best_a, min(rem - 20.0, 300.0))
        _record(a, best_a.size, sc)

    # Validate with actual evaluator
    rem = deadline - time.monotonic()
    if rem > 10.0 and best_a is not None:
        candidates = [best_a]
        # Add a few perturbations
        for sigma in [0.01, 0.03]:
            candidates.append(_perturb(best_a, rng, sigma))

        for cand in candidates:
            if rem < 5.0:
                break
            seq = _scale_output(cand, 10.0)
            real_sc = _try_eval(seq)
            if np.isfinite(real_sc) and real_sc < best_score:
                best_score = real_sc
                best_a = _normalize(cand)
                print(f"  Validated improvement: {best_score:.7f}")

    # Ensure we always return something valid
    if best_a is None:
        best_a = _greedy_init(256)

    result = _scale_output(best_a, 10.0)
    print(f"FINAL: n={best_a.size} local_score={best_score:.7f} time={time.monotonic()-t0:.1f}s")
    return result
