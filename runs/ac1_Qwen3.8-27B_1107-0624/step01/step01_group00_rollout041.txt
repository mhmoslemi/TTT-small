"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using exponentiated gradient
and coordinate descent.
"""

import time
import math
import numpy as np


def _conv_sq(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]


def _objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    S = a.sum()
    if S < 1e-12:
        return float('inf')
    c = _conv_sq(a)
    return 2.0 * len(a) * c.max() / (S * S)


def _proj_simplex(v, target=1.0):
    """Project onto {x >= 0, sum x = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _eg_step(a, c, T, eta):
    """One exponentiated gradient step."""
    n = len(a)
    # Compute softmax weights on c
    c_shifted = c - c.max()
    w = np.exp(c_shifted / T)
    w = w / w.sum()
    
    # Gradient: g[p] = 2 * sum_k w[k] * a[k-p] for valid (k,p)
    # This is 2 * convolve(w, a_reversed) truncated to length n
    w_rev = w[::-1]
    conv_wa = np.convolve(w_rev, a, mode='full')[:n]
    g = 2.0 * conv_wa
    
    # Multiplicative update
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 1e-30)
    S = a_new.sum()
    if S > 0:
        a_new = a_new / S
    else:
        a_new = np.full(n, 1.0/n)
    return a_new


def _cd_pass(a, n_steps=1):
    """One pass of coordinate descent."""
    n = len(a)
    a = a.copy()
    for _ in range(n_steps):
        for p in range(n):
            # Save current value
            old = a[p]
            a[p] = 0.0
            
            # Compute c0 = convolve(a, a) with a[p] = 0
            c0 = _conv_sq(a)
            
            # Now c[k] = c0[k] + 2*x*a[k-p] (if valid, k≠2p) or c0[k] + x^2 (if k=2p)
            # We want to minimize max_k c[k] over x in [0, x_max]
            
            # Background: max of c0[k] for k not affected by x
            # Affected positions: k where 0 <= k-p < n, i.e., k in [p, p+n-1]
            # For these: c[k] = c0[k] + 2*x*a[k-p] (or +x^2 if k=2p)
            # For others: c[k] = c0[k]
            
            lo = max(0, p)  # k-p >= 0 => k >= p
            hi = min(2*n-2, p+n-1)  # k-p < n => k < p+n
            
            # Background max (positions not affected)
            if lo > 0 or hi < 2*n-2:
                bg = np.max(np.concatenate([c0[:lo], c0[hi+1:]])) if (lo > 0) or (hi+1 < 2*n-1) else 0.0
            else:
                bg = 0.0
            
            # For affected positions, c[k](x) = c0[k] + 2*x*a[k-p] (+ x^2 if k=2p)
            # Let's define: for k in [lo, hi], coeff[k] = 2*a[k-p] (or a[k-p] if k=2p, since x^2 = x*x... 
            # Actually: if k = 2p, c[k] = c0[k] + x^2. Otherwise c[k] = c0[k] + 2*x*a[k-p].
            
            # The function to minimize is:
            # f(x) = max(max_{k in [lo,hi], k≠2p} (c0[k] + 2*x*a[k-p]),
            #             c0[2p] + x^2 (if 2p in [lo,hi]),
            #             bg)
            
            # This is convex in x. Use ternary search.
            
            def f(x):
                if x < 0:
                    x = 0.0
                vals = c0[lo:hi+1] + 2.0 * x * a[lo-p:hi+1-p]
                if 2*p >= lo and 2*p <= hi:
                    idx = 2*p - lo
                    vals[idx] = c0[2*p] + x*x
                return max(vals.max(), bg)
            
            # Ternary search on [0, x_upper]
            # Upper bound: current value or something reasonable
            x_upper = max(old * 3, 1.0 / n * 10)

            a[p] = old
            affected_coeffs = 2.0 * a[lo-p:hi+1-p].copy()
            if 2*p >= lo and 2*p <= hi:
                idx_2p = 2*p - lo
                c0_2p = c0[2*p]
            else:
                idx_2p = -1
                c0_2p = 0.0
            a[p] = 0.0
            
            c0_affected = c0[lo:hi+1].copy()
            
            def f_inner(x):
                vals = c0_affected + x * affected_coeffs
                if idx_2p >= 0:
                    vals[idx_2p] = c0_2p + x*x
                return max(vals.max(), bg)
            
            # Golden section search
            gr = (math.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
            x0 = 0.0
            x1 = x_upper
            x2 = x0 + gr * (x1 - x0)
            x3 = x1 - gr * (x1 - x0)
            f2 = f_inner(x2)
            f3 = f_inner(x3)
            
            for _ in range(60):
                if f2 < f3:
                    x1 = x3
                    x3 = x2
                    f3 = f2
                    x2 = x0 + gr * (x1 - x0)
                    f2 = f_inner(x2)
                else:
                    x0 = x2
                    x2 = x3
                    f2 = f3
                    x3 = x1 - gr * (x1 - x0)
                    f3 = f_inner(x3)
            
            x_opt = (x0 + x1) / 2
            a[p] = x_opt
            
            # Renormalize to keep sum = 1 (or just keep going, since obj is scale-invariant)
    
    # Normalize
    S = a.sum()
    if S > 0:
        a = a / S
    return a

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using coordinate descent with
golden section search, combined with exponentiated gradient initialization
and diverse restarts to escape local minima.
"""

import time
import math
import numpy as np


def _conv_sq_fft(a):
    """Compute linear convolution a*a via FFT, length 2n-1."""
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _eval_local(a):
    """Fast local objective evaluation."""
    S = a.sum()
    if S < 1e-12:
        return float('inf')
    c = _conv_sq_fft(a)
    return 2.0 * a.size * c.max() / (S * S)


def _norm1(a):
    """Normalize to sum 1."""
    S = a.sum()
    if S > 1e-15:
        return a / S
    return np.full(a.size, 1.0 / a.size)


def _cd_pass(a, c_full, n_golden=40):
    """One coordinate descent pass, modifying a and c_full in place."""
    n = a.size
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    bound = 2 * n - 2

    for p in range(n):
        old = a[p]
        lo = p
        hi = min(bound, p + n - 1)

        # c0 = convolution with a[p] removed
        c0_aff = c_full[lo:hi + 1].copy()
        c0_aff -= 2.0 * old * a[lo - p:hi + 1 - p]
        if 2 * p <= bound:
            c0_aff[2 * p - lo] += old * old

        # Background max (unaffected positions)
        bg = 0.0
        if lo > 0:
            bg = np.max(c_full[:lo])
        if hi < bound:
            m2 = np.max(c_full[hi + 1:])
            if m2 > bg:
                bg = m2

        aff_a = a[lo - p:hi + 1 - p].copy()
        has_2p = (2 * p <= bound)
        idx_2p = 2 * p - lo if has_2p else -1
        c0_2p = c0_aff[idx_2p] if has_2p else 0.0

        # Golden section search for optimal x = a[p]
        x_upper = max(10.0 * old + 1e-8, 5.0 / n)

        def f_of_x(x):
            vals = c0_aff + 2.0 * x * aff_a
            if has_2p:
                vals[idx_2p] = c0_2p + x * x
            m = vals.max()
            return m if m > bg else bg

        # Expand upper bound if needed
        f0 = f_of_x(0.0)
        if f_of_x(x_upper) < f0 * 0.99:
            x_upper *= 8.0

        x0 = 0.0
        x1 = x_upper
        x2 = x0 + gr * (x1 - x0)
        x3 = x1 - gr * (x1 - x0)
        f2 = f_of_x(x2)
        f3 = f_of_x(x3)

        for _ in range(n_golden):
            if f2 < f3:
                x1 = x3
                x3 = x2
                f3 = f2
                x2 = x0 + gr * (x1 - x0)
                f2 = f_of_x(x2)
            else:
                x0 = x2
                x2 = x3
                f2 = f3
                x3 = x1 - gr * (x1 - x0)
                f3 = f_of_x(x3)

        x_opt = max(0.0, (x0 + x1) * 0.5)
        a[p] = x_opt

        # Update c_full
        c_full[lo:hi + 1] = c0_aff + 2.0 * x_opt * a[lo - p:hi + 1 - p]
        if has_2p:
            c_full[2 * p] = c0_2p + x_opt * x_opt

    return a, c_full


def _cd_optimize(a, max_passes=200, t_limit=None):
    """Run coordinate descent for up to max_passes or until t_limit."""
    n = a.size
    a = _norm1(a.copy())
    c_full = _conv_sq_fft(a)
    best_local = _eval_local(a)
    best_a = a.copy()

    for pidx in range(max_passes):
        if t_limit is not None and time.monotonic() > t_limit:
            break
        a, c_full = _cd_pass(a, c_full)
        a = _norm1(a)
        c_full = _conv_sq_fft(a)
        obj = _eval_local(a)
        if obj < best_local:
            best_local = obj
            best_a = a.copy()
        # Early stop if converged
        if pidx > 10 and 1.0 - obj / best_local < 1e-7:
            break

    return best_a, best_local


def _eg_init(a, n_steps=300, T0=0.5, T_min=0.05, eta0=0.5):
    """Exponentiated gradient initialization."""
    n = a.size
    a = _norm1(a.copy().astype(np.float64))
    c = _conv_sq_fft(a)
    T = T0
    eta = eta0

    for _ in range(n_steps):
        # Softmax weights
        cs = c - c.max()
        w = np.exp(cs / T)
        ws = w.sum()
        if ws > 0:
            w = w / ws
        else:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))

        # Gradient: g[p] = 2 * sum_k w[k] * a[k-p]
        w_rev = w[::-1]
        conv_wa = np.convolve(w_rev, a, mode='full')[:n]
        g = 2.0 * conv_wa

        # Multiplicative update
        a *= np.exp(-eta * g)
        a = np.maximum(a, 1e-15)
        a = _norm1(a)

        # Update params
        T = max(T_min, T * 0.97)
        eta *= 0.995

        c = _conv_sq_fft(a)

    return a


def _make_diverse_seeds(n, rng):
    """Create diverse initial seeds of length n."""

    # Cosine squared

    # Exponential decay from center

    x = np.arange(n)
    seeds.append(np.exp(-((x - center) ** 2) / (2 * (n / 6.0) ** 2)) + 0.01)
    # Ramp up
    seeds.append(np.linspace(0.01, 1.0, n))
    # Ramp down
    seeds.append(np.linspace(1.0, 0.01, n))
    # Two bumps
    peak1 = n * 0.25
    peak2 = n * 0.75
    s = np.exp(-((x - peak1) ** 2) / (2 * (n / 10.0) ** 2)) + np.exp(-((x - peak2) ** 2) / (2 * (n / 10.0) ** 2))
    seeds.append(s + 0.01)
    # Random
    seeds.append(rng.random(n) + 0.01)
    # Periodic (low freq)
    seeds.append(0.5 + 0.4 * np.sin(2 * math.pi * np.arange(n) / max(1, n // 4)) + 0.01)
    # Triangular
    t = 2.0 * np.abs(np.arange(n) / (n - 1.0) - 0.5)
    seeds.append((1.0 - t) + 0.01)
    return seeds
