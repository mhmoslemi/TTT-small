```python
def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
"""
Step function optimizer using smooth-max gradient descent with FFT,
LP-based direction finding, and adaptive perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2 * n - 1]


def score_of(a):
    """Score = 2*n*max(conv(a,a)) for sum-normalized non-neg a."""
    c = fft_convolve(a)
    return 2.0 * len(a) * float(c.max())


def gradient_smooth_max(a, w):
    """
    Gradient of soft_max(conv(a,a)) w.r.t. a.
    w: softmax weights over convolution (length 2n-1).
    Returns gradient of length n.
    g[j] = 2 * sum_m a[m] * w[m+j]
    """
    n = len(a)
    rows = np.arange(n)[:, None]
    cols = np.arange(n)[None, :]
    T = w[rows + cols]
    g = 2.0 * (T @ a)
    return g


def solve_lp_direction(f, M, max_rows):
    """Solve LP: max sum(g) s.t. A*g <= M, g >= 0."""
    n = len(f)
    c = fft_convolve(f)
    if M <= 1e-300:
        return None
    
    # Select top max_rows peaks
    k = min(max_rows, len(c))
    if k < 2:
        return None
    top_idx = np.argpartition(c, -k)[-k:]
    
    # Build constraint matrix
    A = np.zeros((k, n))
    for r, lag in enumerate(top_idx):
        lag = int(lag)
        for j in range(n):
            idx = lag - j
            if 0 <= idx < n:
                A[r, j] = f[idx]
    
    # Solve LP
    cvec = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(cvec, A_ub=A / M, b_ub=np.ones(k),
                      bounds=bounds, method='highs',
                      options={'time_limit': 30})
    except Exception:
        return None
    
    if res is not None and res.status == 0 and res.x is not None:
        g = np.clip(res.x, 0, None)
        if g.sum() > 1.0 + 1e-6:
            return g
    return None


def random_perturb(a, rng, scale=0.1):
    """Random multiplicative perturbation, maintaining non-negativity."""
    n = len(a)
    h = a * np.exp(rng.normal(0, scale, n))
    return project_simplex(h)


def smooth_block_perturb(a, rng, scale=0.05):
    """Smooth block perturbation: multiply a random block by a random factor."""
    n = len(a)
    h = a.copy()
    block_size = max(1, n // 32)
    start = rng.integers(0, n - block_size)
    factor = np.exp(rng.normal(0, scale * 2))
    h[start:start + block_size] *= factor
    return project_simplex(h)


def optimize_sequence(a, time_limit, rng):
    """
    Main optimization loop using smooth-max gradient + LP + perturbations.
    a: initial sequence (will be normalized).
    Returns (best_sequence, best_score).
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_score = score_of(a)
    
    deadline = time.monotonic() + time_limit
    temp = 0.05 / n  # temperature for smooth max
    it = 0
    no_improve = 0
    alpha_base = 0.1
    
    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(a)
        M = float(c.max())
        cur_score = 2.0 * n * M
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        if no_improve > 30:
            temp = min(0.5 / n, temp * 1.3)
        
        # Compute softmax weights
        e = (c - M) / temp
        e = np.clip(e, -50.0, 0.0)
        w = np.exp(e)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum
        
        moved = False
        
        # Method 1: Smooth-max gradient step
        g = gradient_smooth_max(a, w)
        g_norm = np.linalg.norm(g)
        
        if g_norm > 1e-15:
            # Backtracking line search
            alpha = alpha_base / g_norm
            for _ in range(40):
                new_a = project_simplex(a - alpha * g)
                new_score = score_of(new_a)
                if new_score < cur_score - 1e-14:
                    a = new_a
                    moved = True
                    break
                alpha *= 0.7
        
        # If no improvement from gradient, try LP
        if not moved and (it % 15 == 0 or no_improve > 20):
            if n <= 1024:
                max_rows = min(2 * n - 1, max(100, n // 2))
                g_lp = solve_lp_direction(a, M, max_rows)
                if g_lp is not None:
                    S = g_lp.sum()
                    best_t = 0
                    best_h = a
                    for t in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
                        h = project_simplex((1 - t) * a + t * g_lp / S)
                        sc = score_of(h)
                        if sc < best_score or sc < cur_score:
                            if best_t == 0 or sc < score_of(best_h):
                                best_t = t
                                best_h = h
                    if best_t > 0:
                        a = best_h
                        moved = True
        
        # If still stuck, random perturbation
        if not moved:
            if no_improve > 50:
                a = random_perturb(a, rng, scale=min(0.3, 0.1 + 0.01 * no_improve / 10))
            elif no_improve > 20:
                a = smooth_block_perturb(a, rng, scale=0.08)
            else:
                a = random_perturb(a, rng, scale=0.03)
        
        # Cool temperature
        if no_improve == 0:
            temp = max(0.001 / n, temp * 0.97)
    
    return best_a, best_score


def make_two_block(n, ratio):
    """Create a two-block sequence with given mass ratio."""
    a = np.zeros(n)
    L = int(n * ratio)
    R = n - L
    if L > 0:
        a[:L] = ratio / L
    if R > 0:
        a[L:] = (1 - ratio) / R
    return a


def make_ramp(n, alpha):
    """Create a power-law ramp sequence."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** alpha
    return project_simplex(a)


def make_sine(n, freq, amp, phase):
    """Create a 1 + amp*sin(...) sequence."""
    x = np.linspace(0, 1, n)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return project_simplex(a)


def make_gaussian(n, mu, sigma):
    """Create a Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return project_simplex(a)
```
