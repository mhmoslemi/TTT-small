Wait, but we also have the sum constraint. If we change a_i, we need to adjust another coordinate or normalize. This makes it more complex.

Alternatively, we can use the softmax parameterization and do coordinate descent on x_i. That might work but is probably slower than full gradient.

Let me stick with the Adam approach but make it more robust.

**Key improvements I'll implement:**

1. **Aggressive temperature annealing**: Start with alpha=100 (very smooth), anneal to alpha=5 (nearly exact max). Use exponential annealing.

2. **Multi-restart within a single optimization run**: Every ~100 steps, if no improvement for 50 steps, add a random kick to x.

3. **Diverse initializations**: Include cosine, linear, sqrt, and "bump" sequences.

4. **Length sweep**: Try n = 256, 320, 360, 384, 400, 432, 448, 480, 512, 576, 640.

5. **Use the global starting point** if available.

6. **Final polishing**: Very fine-tune the best sequence with low learning rate.

"""
Minimizes f(seq) = 2*n*max(autocorr(seq)) / (sum(seq))^2 using:
- Multi-start Adam optimization with softmax parameterization
- Adaptive smoothed-max objective with aggressive annealing
- Periodic perturbation to escape local minima
- Diverse initialization (greedy, power, cosine, linear, bump)
- Fine length sweep around known-best range

import time
import numpy as np


def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full(a.size, target / max(a.size, 1))
    return a * (target / s)


def score(a):
    """Fast score computation via FFT. Assumes a >= 0."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    b = a / s
    L = next_pow2(2 * n - 1)
    B = np.fft.rfft(b, L)
    c = np.fft.irfft(B * B, L)[:2 * n - 1].real
    mx = c.max()
    if not np.isfinite(mx) or mx < 0:
        return np.inf
    return 2.0 * n * mx


def greedy(n):
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    return normalize(a)


def power(n, alpha):
    idx = np.arange(1, n + 1, dtype=np.float64)
    return normalize(idx ** (-alpha))


def cosine_half(n):
    t = np.linspace(0, np.pi/2, n)
    return normalize(np.cos(t) ** 2)


def linear_dec(n):
    return normalize(np.arange(n, 0, -1, dtype=np.float64) / n)


def bump(n, w=0.3):
    """Gaussian bump centered in the middle."""
    t = np.linspace(0, 1, n)
    return normalize(np.exp(-(t - 0.5) ** 2 / (2 * w ** 2)))

    h = globals().get("height_sequence_1", None)

def adam_optimize(a0, budget_s, lr_base=0.03, alpha0=50.0, alpha_min=3.0,
                  kick_interval=80, kick_sigma=0.15, stagnation_limit=40):
    """
    Adam optimizer with softmax parameterization for min 2n*max(autocorr).
    Input: a0 (non-negative, will be normalized internally)
    Returns: (best_a, best_score)
    """
    a = normalize(a0)
    n = a.size
    if n < 2:
        a = np.ones(2) / 2
        n = 2
    L = next_pow2(2 * n - 1)
    
    # Softerm parameterization
    x = np.log(a + 1e-300)
    x -= x.mean()
    
    best_a = a.copy()
    best_f = score(a)
    
    # Adam state
    mu1, mu2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_t = 0
    
    start = time.monotonic()
    end = start + budget_s
    
    while time.monotonic() < end:
        # Adaptive temperature
        elapsed_frac = min(1.0, (time.monotonic() - start) / max(budget_s, 1e-9))
        alpha = alpha_min + (alpha0 - alpha_min) * np.exp(-3.0 * elapsed_frac)
        
        # Adaptive learning rate: decay over time
        lr = lr_base * (0.3 + 0.7 * np.exp(-2.0 * elapsed_frac))
        
        # Compute autocorrelation
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = c.max()
        
        if not np.isfinite(maxc) or maxc <= 0:
            break
        
        f_now = 2.0 * n * maxc
        if f_now < best_f:
            best_f = f_now
            best_a = a.copy()
            best_t = t
        
        # Soft-max weights
        z = alpha * (c / maxc - 1.0)
        # Clip z to avoid overflow
        z = np.clip(z, -50.0, 0.0)
        ez = np.exp(z)
        sz = ez.sum()
        if sz < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w = ez / sz
        
        # Gradient: convolve w (reversed) with a
        W = np.fft.rfft(w[::-1], L)
        grad_raw = np.fft.irfft(P * W, L)[n - 1:2 * n - 1][::-1].real
        grad_raw = np.maximum(grad_raw, 0.0)
        
        # This is the gradient of max_c w.r.t. a (up to constant)
        # d/d(a_i) [2n * max_c] = 4n * sum_j w_j * a_{|i+j-(n-1)|} ... 
        # Actually, the gradient of c_k w.r.t. a_i is a_{k-i} (for valid indices).
        # So grad of sum_k w_k * c_k w.r.t. a_i = sum_k w_k * a_{k-i} = (w * a)(i)
        # where the convolution is defined appropriately.
        # The full convolution (w*a)(i) for i=0..n-1:
        # (w*a)(i) = sum_j w_j * a_{i-j} (with appropriate bounds)
        # This is what we computed as grad_raw.
        # The score is 2n * max_c, and d(score)/d(a_i) = 4n * grad_raw_i (when sum a = 1 constraint is handled by softmax).
        # Wait, let me be more careful.
        # score = 2n * max_k (a*a)(k)
        # (a*a)(k) = sum_i a_i * a_{k-i}
        # d(a*a)(k)/da_i = a_{k-i} (for k-i in [0, n-1])
        # So d(max_k (a*a)(k))/da_i = sum_k w_k * a_{k-i} (for the active set)
        # = (w * a)_i where * denotes the appropriate cross-correlation
        # Which is exactly what we computed.
        
        # With softmax: a = softmax(x), so da_i/dx_j = a_i*(delta_ij - a_j)
        # grad_x_i = sum_j a_j*(delta_ij - a_j) * 4n * grad_raw_j = 4n * a_i * (grad_raw_i - a_i * sum(grad_raw * a))
        # Wait: grad_x_i = sum_j (da_i/dx_j) * dL/da_j
        # da_i/dx_j = a_i * delta_ij - a_i * a_j
        # grad_x_i = sum_j (a_i*delta_ij - a_i*a_j) * 4n*grad_raw_j
        # = 4n * a_i * grad_raw_i - 4n * a_i * sum_j a_j * grad_raw_j
        # = 4n * a_i * (grad_raw_i - a . grad_raw)
        
        grad_p = 4.0 * n * grad_raw
        gp = np.dot(a, grad_p)
        grad_x = a * (grad_p - gp)
        
        # Adam update
        t += 1
        m = mu1 * m + (1 - mu1) * grad_x
        v = mu2 * v + (1 - mu2) * (grad_x * grad_x)
        mhat = m / (1 - mu1 ** t)
        vhat = v / (1 - mu2 ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -40.0, 40.0)
        
        x = x - step
        
        # Stagnation check: kick
        if t - best_t > stagnation_limit and t % kick_interval == 0:
            x += np.random.standard_normal(n) * kick_sigma
            print(f"  Kick at t={t}, best_f={best_f:.6f}")
        
        # Re-derive a from x
        x = np.clip(x, -80.0, 80.0)
        x -= x.max()
        e = np.exp(x)
        se = e.sum()
        if not np.isfinite(se) or se < 1e-300:
            break
        a = e / se
    
    final_f = score(a)
    if final_f < best_f:
        best_f = final_f
        best_a = a.copy()
    
    return best_a, best_f

"""
Minimizes f(seq) = 2*n*max(autocorr(seq)) / (sum(seq))^2 using:
- Multi-start Adam optimization with softmax parameterization
- Adaptive smoothed-max objective with aggressive temperature annealing
- Periodic random kicks to escape local minima
- Diverse initialization (greedy, power-law, cosine, linear, bump, mixes)
- Fine length sweep around known-best range (256-512)
- Resampling to adjacent lengths for cross-pollination
"""

import time
import numpy as np


def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full(max(a.size, 1), target / max(a.size, 1))
    return a * (target / s)


def score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    b = a / s
    L = next_pow2(2 * n - 1)
    B = np.fft.rfft(b, L)
    c = np.fft.irfft(B * B, L)[:2 * n - 1].real
    mx = c.max()
    if not np.isfinite(mx) or mx < 0:
        return np.inf
    return 2.0 * n * mx


def init_greedy(n):
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize(a)


def init_power(n, alpha):
    idx = np.arange(1, n + 1, dtype=np.float64)
    return normalize(idx ** (-alpha))


def init_cosine(n):
    t = np.linspace(0.0, np.pi / 2.0, n)
    return normalize(np.cos(t) ** 2)


def init_linear(n):
    return normalize(np.arange(n, 0, -1, dtype=np.float64) / float(n))


def init_bump(n, w=0.3):
    t = np.linspace(0.0, 1.0, n)
    return normalize(np.exp(-(t - 0.5) ** 2 / (2.0 * w * w)))


def mix_seqs(a, b, lam):
    return normalize(lam * a + (1.0 - lam) * b)


def resample_seq(a, new_n):
    old_n = a.size
    if new_n == old_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, a)

def adam_run(a0, budget_s, lr_base=0.03, alpha0=50.0, alpha_min=3.0,
             kick_interval=80, kick_sigma=0.15, stagnation_limit=40, seed=0):
    """
    Adam optimizer with softmax parameterization.
    Minimizes 2n*max(autocorr(a)) subject to a>=0, sum(a)=1.
    Returns (best_a_normalized, best_score).
    """
    a = normalize(a0)
    n = a.size
    if n < 2:
        a = np.ones(2) / 2.0
        n = 2
    L = next_pow2(2 * n - 1)
    
    x = np.log(a + 1e-300)
    x -= x.mean()
    
    best_a = a.copy()
    best_f = score_fft(a)
    
    rng_kick = np.random.default_rng(seed)
    
    mu1 = 0.85
    mu2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_t = 0
    
    start_time = time.monotonic()
    end_time = start_time + budget_s
    
    while time.monotonic() < end_time:
        elapsed_frac = min(1.0, (time.monotonic() - start_time) / max(budget_s, 1e-9))
        alpha = alpha_min + (alpha0 - alpha_min) * np.exp(-3.0 * elapsed_frac)
        lr = lr_base * (0.3 + 0.7 * np.exp(-2.0 * elapsed_frac))
        
        # Compute autocorrelation
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = c.max()
        
        if not np.isfinite(maxc) or maxc <= 0:
            break
        
        f_now = 2.0 * n * maxc
        if f_now < best_f:
            best_f = f_now
            best_a = a.copy()
            best_t = t
        
        # Softmax weights over convolution
        z = alpha * (c / maxc - 1.0)
        z = np.clip(z, -50.0, 0.0)
        ez = np.exp(z)
        sz = ez.sum()
        if sz < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w = ez / sz
        
        # Gradient: cross-correlation of w with a
        W = np.fft.rfft(w[::-1], L)
        grad_raw = np.fft.irfft(P * W, L)[n - 1:2 * n - 1][::-1].real
        grad_raw = np.maximum(grad_raw, 0.0)
        
        # Transform to softmax gradient
        grad_p = 4.0 * n * grad_raw
        gp = np.dot(a, grad_p)
        grad_x = a * (grad_p - gp)
        
        # Adam update
        t += 1
        m = mu1 * m + (1.0 - mu1) * grad_x
        v = mu2 * v + (1.0 - mu2) * (grad_x * grad_x)
        mhat = m / (1.0 - mu1 ** t)
        vhat = v / (1.0 - mu2 ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -40.0, 40.0)
        x = x - step
        
        # Stagnation kick
        if t - best_t > stagnation_limit and t % kick_interval == 0:
            x += rng_kick.standard_normal(n) * kick_sigma
            print("    Kick at t=%d, best_f=%.6f" % (t, best_f))
        
        # Recover a from softmax
        x = np.clip(x, -80.0, 80.0)
        x -= x.max()
        e = np.exp(x)
        se = e.sum()
        if not np.isfinite(se) or se < 1e-300:
            break
        a = e / se
    
    final_f = score_fft(a)
    if final_f < best_f:
        best_f = final_f
        best_a = a.copy()
    
    return best_a, best_f

    # We want lr_k = c / k for subgradient method
    # But that converges slowly. Let's use a heuristic.
    
    lr = lr_init

    start_time = time.monotonic()

        c = np.maximum(c, 0.0)

            best_a = a.copy()
            patience = 0

            lr = max(0.01, lr * 0.99)
        else:
            patience += 1
            if patience > max_patience:
                # Restart with a kick
                a = normalize(a + rng.standard_normal(n) * 0.1)
                lr = lr_init * 0.5
                patience = 0
                continue
        
        # Find argmax
        k_star = np.argmax(c)
        
        # Subgradient: g_j = 2 * a[k_star - j] (for valid indices)
        # k_star ranges from 0 to 2n-2
        # j ranges from 0 to n-1
        # k_star - j must be in [0, n-1]
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        g = np.zeros(n)
        if lo <= hi:
            # g[j] = 2 * a[k_star - j] for j in [lo, hi]
            # When j = lo: k_star - j = k_star - lo = k_star - max(0, k_star-n+1)
            # When j = hi: k_star - j = k_star - hi = k_star - min(n-1, k_star)
            # So the indices of a are from k_star-hi to k_star-lo
            a_idx = k_star - hi : k_star - lo + 1  # this gives k_star-hi, ..., k_star-lo
            g[lo:hi+1] = 2.0 * a[a_idx][::-1]
        
        # Step
        a_new = a - lr * g
        
        # Project onto simplex {a >= 0, sum a = 1}
        a_new = proj_simplex(a_new)
        
        a = a_new
    
    return best_a, best_f


def proj_simplex(b):
    """Project b onto the standard simplex {x >= 0, sum x = 1}."""
    n = b.size
    if n == 0:
        return b
    u = np.sort(b)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(b - theta, 0)

Minimizes f(seq) = 2*n*max(autocorr(seq)) / (sum(seq))^2 using:
- Multi-start Adam optimization with softmax parameterization
- Top-K peak-focusing for better gradient signal
- Adaptive temperature annealing and learning rate scheduling

import time
import numpy as np


def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full(max(a.size, 1), target / max(a.size, 1))
    return a * (target / s)


def score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    b = a / s
    L = next_pow2(2 * n - 1)
    B = np.fft.rfft(b, L)
    c = np.fft.irfft(B * B, L)[:2 * n - 1].real
    mx = c.max()
    if not np.isfinite(mx) or mx < 0:
        return np.inf
    return 2.0 * n * mx


def init_greedy(n):
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize(a)


def init_power(n, alpha):
    idx = np.arange(1, n + 1, dtype=np.float64)
    return normalize(idx ** (-alpha))


def init_cosine(n):

def proj_simplex(b):
    n = b.size
    if n == 0:
        return b
    u = np.sort(b)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)

    rho = idx[-1]

    return np.maximum(b - theta, 0.0)

                w = ez / sz
        else:
            z = alpha * (c / maxc - 1.0)
            z = np.clip(z, -50.0, 0.0)
            ez = np.exp(z)
            sz = ez.sum()
            if sz < 1e-300:
                w = np.zeros(2 * n - 1)
                w[np.argmax(c)] = 1.0
            else:
                w = ez / sz

        W = np.fft.rfft(w[::-1], L)
        grad_raw = np.fft.irfft(P * W, L)[n - 1:2 * n - 1][::-1].real
        grad_raw = np.maximum(grad_raw, 0.0)

        grad_p = 4.0 * n * grad_raw
        gp = np.dot(a, grad_p)
        grad_x = a * (grad_p - gp)

        t += 1
        m = mu1 * m + (1.0 - mu1) * grad_x
        v = mu2 * v + (1.0 - mu2) * (grad_x * grad_x)
        mhat = m / (1.0 - mu1 ** t)
        vhat = v / (1.0 - mu2 ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps_adam)
        step = np.clip(step, -40.0, 40.0)
        x = x - step

        # Stagnation kick
        if t - best_t > stagnation_limit and t % kick_interval == 0:
            kick_size = kick_sigma * min(1.0, (t - best_t) / (2.0 * stagnation_limit))
            x += rng_kick.standard_normal(n) * kick_size

        x = np.clip(x, -80.0, 80.0)
        x -= x.max()
        e = np.exp(x)
        se = e.sum()
        if not np.isfinite(se) or se < 1e-300:
            break
        a = e / se

    final_f = score_fft(a)
    if final_f < best_f:
        best_f = final_f
        best_a = a.copy()

    return best_a, best_f


def subgrad_run(a0, budget_s, lr_init=0.05, top_k=3, seed=0):
    """
    Subgradient method with simplex projection.
    Uses top-K peaks for a smoother descent direction.
    """
    a = normalize(a0)
    n = a.size
    if n < 2:
        a = np.ones(2) / 2.0
        n = 2
    L = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_f = score_fft(a)
    rng = np.random.default_rng(seed)

    lr = lr_init
    t = 0
    stagnation = 0

    start_time = time.monotonic()
    end_time = start_time + budget_s

    while time.monotonic() < end_time:
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = c.max()
        f_now = 2.0 * n * maxc

        if f_now < best_f:
            best_f = f_now
            best_a = a.copy()
            stagnation = 0
            lr = max(0.005, lr * 0.995)
        else:
            stagnation += 1
            if stagnation > 300:
                a = normalize(a + rng.standard_normal(n) * 0.15)
                lr = lr_init * 0.5
                stagnation = 0
                continue

        # Compute gradient using top-K peaks
        if top_k > 0 and top_k < (2 * n - 1):
            top_idx = np.argpartition(c, -top_k)[-top_k]
            top_vals = c[top_idx]
            weights = np.exp(-np.arange(top_k) / 2.0)
            weights /= weights.sum()
            g = np.zeros(n)
            for wi, ki in zip(weights, top_idx):
                ki = int(ki)
                lo = max(0, ki - n + 1)
                hi = min(n - 1, ki)
                if lo <= hi:
                    g[lo:hi + 1] += wi * 2.0 * a[ki - hi:ki - lo + 1][::-1]
        else:
            k_star = int(np.argmax(c))
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            g = np.zeros(n)
            if lo <= hi:
                g[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        a_new = a - lr * g
        a = proj_simplex(a_new)
        t += 1

    final_f = score_fft(a)
    if final_f < best_f:
        best_f = final_f
        best_a = a.copy()

    return best_a, best_f

def scale_output(a, target_sum=10.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(16)
        s = 16.0
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = b.sum()
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(v) for v in b]
