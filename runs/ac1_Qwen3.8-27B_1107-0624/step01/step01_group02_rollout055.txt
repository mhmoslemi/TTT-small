"""
Optimizes non-negative step function heights to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses FFT-based smooth gradient (log-sum-exp surrogate), projected gradient descent
with temperature annealing, L-BFGS polish via softmax parameterization, and
progressive length upsampling. Multi-start with diverse structural initializations.
"""

import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.optimize import minimize


def _fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _cross_corr_valid(f, p):
    """
    Compute g_i = sum_{j=0}^{n-1} f[j] * p[i+j] for i=0..n-1.
    p has length 2n-1.
    """
    n = len(f)
    m = len(p)
    rp = p[::-1]
    full = _fft_convolve(f, rp)
    return full[m - 1:m - 1 + n][::-1].copy()


def _score_norm(f):
    """Score for sum-1 nonnegative sequence: 2n * max(convolve(f,f))."""
    c = _fft_convolve(f, f)
    return 2.0 * len(f) * float(np.max(c))


def _softmax_obj_grad(x, tau, n):
    """
    Objective and gradient w.r.t. softmax(x) for the smooth score.
    f = softmax(x), score = 2n * tau * log(sum exp(c/tau)) where c = conv(f,f).
    """
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    f = e / np.sum(e)
    c = _fft_convolve(f, f)
    c_max = np.max(c)
    e2 = np.exp((c - c_max) / tau)
    s = np.sum(e2)
    L = c_max + tau * np.log(s)
    val = 2.0 * n * L
    p = e2 / s
    g = _cross_corr_valid(f, p)
    fg = np.dot(f, g)
    # dS/dx_i = 4n * f_i * (g_i - f·g)
    grad = 4.0 * n * f * (g - fg)
    return val, grad


def _project_vsimplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _anneal_gradient_step(f, tau, alpha, rng):
    """
    One projected gradient step with backtracking.
    Returns (new_f, score, grad, c, p).
    """
    n = len(f)
    c = _fft_convolve(f, f)
    c_max = np.max(c)
    e = np.exp((c - c_max) / tau)
    p = e / np.sum(e)
    g = _cross_corr_valid(f, p)
    grad = 4.0 * n * g
    cur_score = _score_norm(f)

    # Backtracking line search
    step = alpha
    for _ in range(20):
        h = f - step * grad
        h = np.maximum(h, 0.0)
        s = np.sum(h)
        if s < 1e-15:
            step *= 0.5
            continue
        h = h / s
        new_score = _score_norm(h)
        if new_score < cur_score - 1e-15:
            return h, new_score, grad, c, p
        step *= 0.5

    return f, cur_score, grad, c, p


def _optimize_gradient(f, time_limit, tau_start, tau_end, alpha_init, seed):
    """
    Run projected gradient descent with tau annealing.
    """
    rng = np.random.default_rng(seed)
    n = len(f)
    f = _project_vsimplex(np.maximum(f, 0.0)).copy()
    deadline = time.monotonic() + time_limit

    tau = tau_start
    cur_score = _score_norm(f)
    best_f = f.copy()
    best_score = cur_score
    alpha = alpha_init
    it = 0
    stagnation = 0

    while time.monotonic() < deadline:
        it += 1
        # Anneal tau
        frac = min(1.0, it / max(1, int(time_limit * 200)))
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, 1e-9 * cur_score)

        f, new_score, grad, c, p = _anneal_gradient_step(f, tau, alpha, rng)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_f = f.copy()
            stagnation = 0
            alpha = min(alpha * 1.1, 0.5)
        else:
            stagnation += 1
            if stagnation > 50:
                # Random kick
                kick = rng.normal(0, 0.01, n)
                f = _project_vsimplex(f + kick)
                alpha = alpha * 0.7
                stagnation = 0

        # Periodically recompute true score (the smooth approx may drift)
        if it % 100 == 0:
            true_score = _score_norm(f)
            if true_score < best_score:
                best_score = true_score
                best_f = f.copy()

    return best_f, best_score


def _optimize_lbfgs(f, time_limit, tau, seed):
    """
    L-BFGS optimization using softmax parameterization.
    """
    n = len(f)
    f = _project_vsimplex(np.maximum(f, 0.0)).copy()

    # Initialize x from f: x_i = log(f_i) (with small epsilon for zeros)
    eps = 1e-20
    x0 = np.log(np.maximum(f, eps))

    def obj_grad(x):
        return _softmax_obj_grad(x, tau, n)

    deadline = time.monotonic() + time_limit

    # Run L-BFGS with time limit via maxiter
    max_iter = max(50, int(time_limit * 50))
    res = minimize(
        lambda x: obj_grad(x)[0],
        x0,
        jac=lambda x: obj_grad(x)[1],
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                 'maxls': 50, 'maxfun': max_iter * 5}
    )

    if res.x is not None:
        x_opt = res.x
        x_shifted = x_opt - np.max(x_opt)
        e = np.exp(x_shifted)
        f_new = e / np.sum(e)
    else:
        f_new = f

    score = _score_norm(f_new)
    return f_new, score


def _make_gauss_init(n, rng, bimodal=True):

def _make_block_init(n, rng, nblocks):
    """Random block initialization."""
    boundaries = np.sort(rng.choice(np.arange(1, n), nblocks - 1, replace=False))
    boundaries = np.concatenate([[0], boundaries, [n]])
    masses = rng.dirichlet(np.ones(nblocks))
    f = np.zeros(n)
    for i in range(nblocks):
        lo, hi = boundaries[i], boundaries[i + 1]
        if hi > lo:
            f[lo:hi] = masses[i] / (hi - lo)
    return _project_vsimplex(f)


def _make_smooth_random(n, rng):
    """Smooth random initialization via random Fourier features."""
    x = np.linspace(0, 1, n)
    nfreq = rng.integers(2, 8)
    f = np.ones(n)

        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-0.5, 0.5)
        f += amp * np.cos(2 * np.pi * freq * x + phase)
    f = np.maximum(f, 0.1)
    return _project_vsimplex(f)


def _upsample_f(f, new_n):
    """Upsample via interpolation and renormalize."""
    n = len(f)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f_interp = np.interp(x_new, x_old, f)
    return _project_vsimplex(f_interp)

    """Try a sequence, return score or inf."""
    try:
        f = np.asarray(f, dtype=np.float64)
        f = np.maximum(f, 0.0)
        s = np.sum(f)
        if s < 1e-10:
            return np.inf
        f = f / s
        return _score_norm(f)
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for low-score non-negative step function heights.
    
    Strategy:
    1. Multi-start with diverse structural initializations at n=512
    2. FFT-based projected gradient with temperature annealing
    3. L-BFGS polish with softmax parameterization
    4. Progressive upscaling to n=1024, 2048 with refinement
    """
    master_rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    hard_deadline = t_start + budget_s - 30.0  # 30s buffer

    state_score = float('inf')
    state_seq = None

    def _update_best(f, score):
        nonlocal state_score, state_seq
        if np.isfinite(score) and score < state_score:
            state_score = float(score)
            state_seq = _project_vsimplex(f).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [T+{elapsed:.0f}s] New best: score={state_score:.10f}, n={len(state_seq)}", flush=True)

    # Load provided sequence if available
    provided = None
    try:
        raw = height_sequence_1
        a = np.asarray(raw, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.size > 0 and np.sum(a) > 1e-10:
            provided = _project_vsimplex(a)
            ps = _score_norm(provided)
            _update_best(provided, ps)
            print(f"Provided sequence: n={provided.size}, score={ps:.10f}", flush=True)
    except Exception:
        pass

    # Base length
    N = 512

    # Generate diverse initializations
    inits = []
    labels = []

    # From provided (resized to N)
    if provided is not None and provided.size != N:
        inits.append(_upsample_f(provided, N))
        labels.append("provided_up")
    elif provided is not None:
        inits.append(provided.copy())
        labels.append("provided")

    # Cosine modulations (bimodal-like)
    for freq in [1, 2, 3, 4]:
        for amp in [0.7, 0.9, 0.95]:
            inits.append(_make_cosc_init(N, freq, amp))
            labels.append(f"cos_f{freq}_a{amp}")

    # Block initializations
    for nb in [2, 3, 4, 5]:
        for _ in range(2):
            inits.append(_make_block_init(N, master_rng, nb))
            labels.append(f"block_{nb}")

    # Gaussian / bimodal
    for _ in range(3):
        inits.append(_make_gauss_init(N, master_rng, bimodal=True))
        labels.append("bimodal_gauss")
    for _ in range(2):
        inits.append(_make_gauss_init(N, master_rng, bimodal=False))
        labels.append("unimodal_gauss")

    # Smooth random
    for _ in range(3):
        inits.append(_make_smooth_random(N, master_rng))
        labels.append("smooth_rand")

    # Uniform
    inits.append(np.ones(N) / N)
    labels.append("uniform")

    print(f"Total initializations: {len(inits)}", flush=True)

    # Phase 1: Quick screening - 3 seconds per init
    t_phase1 = 4.0
    phase1_scores = []
    for i, (f_init, lab) in enumerate(zip(inits, labels)):
        if time.monotonic() > hard_deadline:
            break
        f_best, sc = _optimize_gradient(f_init, t_phase1, tau_start=0.05, tau_end=0.01,
                                        alpha_init=0.1, seed=seed + i)
        phase1_scores.append((sc, i, lab))
        _update_best(f_best, sc)
        if i % 5 == 0:
            print(f"  Phase1 [{i}/{len(inits)}] best_so_far={state_score:.8f} (label={lab}, sc={sc:.8f})", flush=True)

    phase1_scores.sort()
    print(f"Phase 1 done. Best: {phase1_scores[0][0]:.10f} ({phase1_scores[0][2]})", flush=True)

    # Phase 2: Extended gradient optimization on top 8
    t_phase2_each = 15.0
    top_k = min(8, len(phase1_scores))
    for rank in range(top_k):
        if time.monotonic() > hard_deadline:
            break
        sc1, idx1, lab1 = phase1_scores[rank]
        f_init = inits[idx1]
        # Re-optimize with more time and lower tau
        f_best, sc = _optimize_gradient(f_init, t_phase2_each, tau_start=0.02, tau_end=0.002,
                                        alpha_init=0.05, seed=seed + 100 + rank)
        _update_best(f_best, sc)
        print(f"  Phase2 [{rank}/{top_k}] score={sc:.10f} (label={lab1})", flush=True)

    # Phase 3: L-BFGS polish on top candidates
    t_phase3_each = 30.0
    for rank in range(min(4, top_k)):
        if time.monotonic() > hard_deadline:
            break
        # Use the current best as starting point
        f_start = state_seq.copy()
        f_best, sc = _optimize_lbfgs(f_start, t_phase3_each, tau=0.005, seed=seed + 200 + rank)
        _update_best(f_best, sc)
        print(f"  Phase3 [{rank}] L-BFGS score={sc:.10f}", flush=True)

    # Phase 4: Second round of gradient + L-BFGS from the best
    if time.monotonic() < hard_deadline - 60:
        t_avail = min(120.0, hard_deadline - time.monotonic() - 30)
        if t_avail > 30:
            f_best, sc = _optimize_gradient(state_seq, t_avail * 0.5,
                                            tau_start=0.005, tau_end=0.0005,
                                            alpha_init=0.02, seed=seed + 300)
            _update_best(f_best, sc)

            if time.monotonic() < hard_deadline - 20:
                t_lb = min(60.0, hard_deadline - time.monotonic() - 10)
                if t_lb > 10:
                    f_best, sc = _optimize_lbfgs(state_seq, t_lb, tau=0.001, seed=seed + 400)
                    _update_best(f_best, sc)
                    print(f"  Phase4 final L-BFGS: score={sc:.10f}", flush=True)

    # Phase 5: Try upscaling to larger n
    for target_n in [1024, 2048]:
        if time.monotonic() > hard_deadline - 120:
            break
        f_up = _upsample_f(state_seq, target_n)
        sc_up = _score_norm(f_up)
        print(f"  Phase5: Upscaled to n={target_n}, score={sc_up:.10f}", flush=True)

        t_opt = min(180.0, hard_deadline - time.monotonic() - 60)
        if t_opt > 30:
            f_best, sc = _optimize_gradient(f_up, t_opt * 0.6,
                                            tau_start=0.01, tau_end=0.001,
                                            alpha_init=0.03, seed=seed + 500 + target_n)
            _update_best(f_best, sc)

            if time.monotonic() < hard_deadline - 30:
                t_lb = min(60.0, hard_deadline - time.monotonic() - 10)
                if t_lb > 10:
                    f_best, sc = _optimize_lbfgs(state_seq, t_lb, tau=0.001, seed=seed + 600 + target_n)
                    _update_best(f_best, sc)

    # Phase 6: Final intensive polish on the best found
    if time.monotonic() < hard_deadline - 20:
        t_final = hard_deadline - time.monotonic() - 10
        if t_final > 20:
            # Very low temperature gradient
            f_best, sc = _optimize_gradient(state_seq, t_final * 0.6,
                                            tau_start=0.001, tau_end=0.0001,
                                            alpha_init=0.01, seed=seed + 700)
            _update_best(f_best, sc)

            if time.monotonic() < hard_deadline - 15:
                t_lb = min(45.0, hard_deadline - time.monotonic() - 5)
                if t_lb > 8:
                    f_best, sc = _optimize_lbfgs(state_seq, t_lb, tau=0.0005, seed=seed + 800)
                    _update_best(f_best, sc)
                    print(f"  Phase6 final: score={sc:.10f}", flush=True)

    # Final result
    if state_seq is None:
        # Fallback: 2-element optimal
        result = [1.0 / 3.0, 2.0 / 3.0]
    else:
        result = [float(x) for x in state_seq]

    # Verify with official evaluator
    try:
        official = evaluate_sequence(result)
        print(f"OFFICIAL SCORE: {official:.12f} (n={len(result)})", flush=True)
    except Exception as ex:
        print(f"Verification failed: {ex}", flush=True)

    elapsed = time.monotonic() - t_start
    print(f"Total time: {elapsed:.1f}s. Final best: {state_score:.10f}", flush=True)

    return result
