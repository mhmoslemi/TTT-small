    # a_i = r^i
    S = (r**n - 1) / (r - 1) if r != 1 else n
    # b_k for k = 0..2n-2
    # For 0 <= k <= n-1: b_k = (k+1)*r^k
    # For n <= k <= 2n-2: b_k = (2n-1-k)*r^k
    # Max of first part: n * r^(n-1) (at k=n-1)
    # Max of second part: find k that maximizes (2n-1-k)*r^k
    # Continuous: k* = 2n-1-1/ln(r), value = (1/ln(r))*r^(2n-1-1/ln(r))
    # But k must be in [n, 2n-2]
    
    if r <= 1:
        # Uniform-ish, max at center
        max_b = n * r**(n-1) if r > 1 else n  # for r=1: n
        # Actually for r=1: b_k = k+1 for k<n, 2n-1-k for k>=n. Max = n at k=n-1.
    
    ln_r = math.log(r)
    k_star = 2*n - 1 - 1/ln_r
    if k_star < n:
        max_b2 = (n-1) * r**n  # at k=n
    elif k_star > 2*n-2:
        max_b2 = 1.0 * r**(2*n-2)  # at k=2n-2
    else:

    max_b = max(n * r**(n-1), max_b2)
    return 2*n*max_b/S**2
