def lp_weierstrass(a, budget_s, max_iters=200):

    minimizes peak autocorrelation while increasing sum.

    end = time.monotonic() + budget_s
    rows = list(range(0, 2*n-1))  # Start with all constraints (small n only)
    if n > 200:
        # For larger n, start with a subset

        rows = sorted(set(int(x) for x in rows))
    
    for it in range(max_iters):
        if time.monotonic() > end - 2:
            break
        if len(rows) > 2*n-1:
            break
            
        # Build constraint matrix
        # (f*g)[k] = sum_j f[j] * g[k-j] <= M for k in rows
        # In standard form: A @ g <= b where A has shape (len(rows), n)
        A = np.zeros((len(rows), n), dtype=np.float64)
        for i, k in enumerate(rows):
            # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[j] * g[k-j]
            # = sum_{j} f[j] * g[k-j]
            # For a fixed k, the coefficient of g[m] is f[k-m] (if 0<=k-m<n)
            # So A[i, m] = f[k-m] if 0 <= k-m < n, else 0
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for j in range(lo, hi + 1):
                A[i, j] = a[k - j]
        
        b = np.full(len(rows), cur_M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)  # maximize sum(g) = minimize -sum(g)
        
        # Solve LP
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, 
                         method='highs', options={'time_limit': 30, 'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        g = np.clip(res.x, 0, None)
        sg = float(g.sum())
        if sg < 1e-10:
            break
        
        # Check all constraints (not just active ones)
        if n <= 512:
            conv_fg = np.convolve(a, g)
        else:
            L = next_pow2(2*n-1)
            P = np.fft.rfft(a, L)
            Pg = np.fft.rfft(g, L)
            conv_fg = np.fft.irfft(P*Pg, L)[:2*n-1].real
        
        violations = conv_fg - cur_M
        max_viol = float(violations.max()) if violations.size > 0 else 0
        
        if max_viol > 1e-6 * cur_M:
            # Add violated constraints
            violated_idx = np.where(violations > 1e-6 * cur_M)[0]
            rows_set = set(rows)
            added = 0
            for idx in violated_idx:
                if idx not in rows_set and added < 200:
                    rows.append(int(idx))
                    rows_set.add(int(idx))
                    added += 1
            rows.sort()
            if len(rows) > 2*n - 1:
                rows = list(range(2*n-1))
            continue
        
        # Step: f_new = (1-t)*a + t*g/scale
        # We want to keep the same normalization (sum=1)
        # The new peak should be <= cur_M
        best_step = None
        best_new_M = cur_M
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_new = (1-t)*a + t*g/sg  # g/sg is normalized to sum 1
            if np.any(a_new < -1e-10):
                continue
            a_new = np.clip(a_new, 0, None)
            s_new = float(a_new.sum())
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            new_M = float(np.max(np.convolve(a_new, a_new))) if n <= 512 else float(np.max(np.fft.irfft(np.fft.rfft(a_new, next_pow2(2*n-1))**2, next_pow2(2*n-1))[:2*n-1].real))

        best_M = cur_M
        
        if it % 10 == 0:
            print(f"  LP iter {it}: M={cur_M:.8f}, score={2*n*cur_M:.6f}")
    
    return best_a, 2*n*best_M

def pnrm_gradient_descent(a, budget_s, p=20, lr=None):
    """Gradient descent on p-norm smoothed objective."""
    a = normalize(a)
    n = a.size
    if budget_s < 3:
        return a, score(a)
    
    if lr is None:
        lr = 0.1 / np.sqrt(n)
    
    best_a = a.copy()
    best_val = score(a)
    
    # Work on log-parametrization for non-negativity
    x = np.log(a + 1e-300)
    x -= x.mean()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    end = time.monotonic() + budget_s
    L = next_pow2(2*n - 1)
    
    while time.monotonic() < end - 1:
        t += 1
        # Adaptive p: start large, decrease
        p_cur = int(p * max(0.1, 1.0 - 0.5 * (t / max(1, t//5))))
        p_cur = max(4, min(100, p_cur))
        
        # Compute gradient
        c = np.fft.irfft(np.fft.rfft(a, L)**2, L)[:2*n-1].real
        c = np.maximum(c, 1e-30)
        cp = c ** p_cur
        sum_cp = cp.sum()
        val = sum_cp ** (1.0/p_cur)
        
        if np.isfinite(2*n*val) and 2*n*val < best_val:
            best_val = 2*n*val
            best_a = a.copy()
        
        # Gradient of (sum c^p)^(1/p)
        dc = (1.0/p_cur) * sum_cp ** (1.0/p_cur - 1) * c ** (p_cur - 1)
        
        # d/da_j sum_k dc[k] * (a[k-j] + a[j-k])
        Pdc = np.fft.rfft(dc, L)
        Pa = np.fft.rfft(a, L)
        # cross_corr(dc, a)[j] = sum_k dc[k+j]*a[k] = ifft(fft(dc)*conj(fft(a)))[j]
        cc1 = np.fft.irfft(Pdc * np.conj(Pa), L)[:n]
        cc2 = np.fft.irfft(Pa * np.conj(Pdc), L)[:n]
        grad_a = cc1 + cc2  # This is the gradient of sum_k c[k]^p w.r.t. a
        
        # Project to tangent space of simplex: subtract mean weighted by a
        grad_x = a * (grad_a - float(np.dot(a, grad_a)))
        
        # Clip gradient
        gnorm = float(np.linalg.norm(grad_x))
        if gnorm > 1e6:
            grad_x = grad_x / gnorm * 1e6
        
        # Adam update
        m = beta1*m + (1-beta1)*grad_x
        v = beta2*v + (1-beta2)*grad_x**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Adaptive learning rate
        frac = (t) / max(1, int(budget_s * 100))
        lr_cur = lr * (1.0 + 2.0 * np.exp(-5.0 * frac))
        
        x = x - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        x -= x.mean()
        
        if not np.all(np.isfinite(x)):
            break
        
        x = np.clip(x, -50, 50)
        e = np.exp(x - x.max())
        se = float(e.sum())
        if se < 1e-300:
            break
        a = e / se
    
    final_val = score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    
    return best_a, best_val

def scale_output(a, target_sum=10.0):

Also, I need to be more careful about the gradient computation. Let me re-derive it.

We have J(a) = (sum_k c_k^p)^(1/p) where c = conv(a,a) (linear convolution).

dJ/da_j = (1/p) * (sum_k c_k^p)^(1/p - 1) * p * sum_k c_k^(p-1) * dc_k/da_j
= (sum_k c_k^p)^(1/p - 1) * sum_k (p*c_k^(p-1)) * (a[k-j] + a[j-k])

where we use the convention a[m] = 0 for m < 0 or m >= n.

Let w_k = p * c_k^(p-1). Then:

dJ/da_j = (sum c^p)^(1/p-1) * [sum_k w_k * a[k-j] + sum_k w_k * a[j-k]]

First term: sum_k w_k * a[k-j]
Let m = k-j: sum_m w_{m+j} * a[m] for m = max(0, -(j)) to min(n-1, 2n-2-j)
= sum_m w[m+j] * a[m] for valid m
This is the cross-correlation of w and a, evaluated at lag j.
cross_corr(w, a)[j] = sum_m w[m+j] * a[m]

Using FFT: cross_corr(w, a) = ifft(fft(w) * conj(fft(a))) (circular)
For linear: pad to length >= 2n-1.

Second term: sum_k w_k * a[j-k]
Let m = j-k: sum_m w[j-m] * a[m] for valid m
= sum_m w[j-m] * a[m]
This is the linear convolution (w * a)[j] = sum_m w[m] * a[j-m] ... wait, that's not quite right.
sum_m w[j-m] * a[m] = (w * a)[j] where * is linear convolution. Yes!

So:
First term = cross_corr(w, a)[j] = sum_m w[m+j] * a[m]
Second term = (w * a)[j] = sum_m w[m] * a[j-m] = sum_m w[j-m] * a[m]

These are related: cross_corr(w,a)[j] = (w_rev * a)[j] where w_rev is the reversal of w.

Actually, let me just compute both via FFT:
- cross_corr(w, a)[j] for j=0,...,n-1: 
  Pad w to length L >= 2n-1 (zero-padded at the end)
  Pad a to length L
  result = ifft(fft(w) * conj(fft(a)))[:n]

- conv(w, a)[j] for j=0,...,n-1:
  Pad w to length L
  Pad a to length L

"""
Search for non-negative step-function sequence minimizing
f(seq) = 2*n*max(convolve(seq,seq)) / sum(seq)^2

smoothed gradient descent for multi-scale optimization.

import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.ones(16, dtype=np.float64) / 16.0
    return a / s


def greedy_coeffs(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def conv_full(a):
    n = a.size
    if n <= 600:
        return np.convolve(a, a)
    L = next_pow2(2 * n - 1)
    P = np.fft.rfft(a, L)
    return np.fft.irfft(P * P, L)[:2 * n - 1].real


def score_normalized(a):
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    an = a / s
    c = conv_full(an)
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max())


def build_lp_matrix(f, rows, n):
    """Build constraint matrix for LP: (f*g)[k] <= M for k in rows."""
    A = np.zeros((len(rows), n), dtype=np.float64)
    for i, k in enumerate(rows):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[i, lo:hi + 1] = f[lo:hi + 1][::-1] * f[k - lo:k - lo + 1]
    # Wait, that's wrong. (f*g)[k] = sum_j f[j]*g[k-j]
    # Coefficient of g[m] in (f*g)[k] is f[k-m] if 0<=k-m<n
    # So A[i, m] = f[k-m] if 0 <= k-m < n
    A = np.zeros((len(rows), n), dtype=np.float64)
    for i, k in enumerate(rows):
        # m ranges over 0..n-1, coefficient is f[k-m] if 0<=k-m<n
        m_lo = max(0, k - (n - 1))
        m_hi = min(k, n - 1)
        if m_lo <= m_hi:
            A[i, m_lo:m_hi + 1] = f[k - m_hi:k - m_lo + 1]
    return A


def lp_weierstrass(a, budget_s):
    """Weierstrass LP method: minimize peak autocorrelation iteratively."""
    if linprog is None:
        return a, score_normalized(a)

    a = normalize(a)
    n = a.size
    c_aa = conv_full(a)
    c_aa = np.maximum(c_aa, 0.0)
    cur_M = float(c_aa.max())
    best_a = a.copy()
    best_M = cur_M

    if budget_s < 10:
        return a, score_normalized(a)

    end = time.monotonic() + budget_s

    # Initial active set: top lags
    n_lags = 2 * n - 1
    if n <= 256:
        rows = list(range(n_lags))
    else:
        # Start with subset: lags where c_aa is in top 60%
        threshold = np.percentile(c_aa, 40)
        rows = [int(i) for i in range(n_lags) if c_aa[i] >= threshold]
        if len(rows) < n:
            rows = list(range(0, n_lags, max(1, n_lags // (2 * n))))

    max_rows = min(n_lags, 8 * n)

    for it in range(500):
        if time.monotonic() > end - 3:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= cur_M for k in rows, g >= 0
        A = build_lp_matrix(a, rows, n)
        b = np.full(len(rows), cur_M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)

        t_lp = min(60.0, max(5.0, (end - time.monotonic()) * 0.15))
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'time_limit': t_lp, 'presolve': True})
        except Exception:
            break

        if res is None or not res.success or res.x is None:
            break

        g = np.clip(res.x, 0.0, None)
        sg = float(g.sum())
        if sg < 1e-10:
            break

        # Check full constraint satisfaction
        c_fg = conv_full(g) * 0  # placeholder
        # Actually compute f*g
        if n <= 600:
            c_fg = np.convolve(a, g)
        else:
            L = next_pow2(2 * n - 1)
            Pa = np.fft.rfft(a, L)
            Pg = np.fft.rfft(g, L)
            c_fg = np.fft.irfft(Pa * Pg, L)[:2 * n - 1].real

        violations = c_fg - cur_M
        max_viol = float(np.max(violations)) if violations.size > 0 else 0.0

        if max_viol > 1e-7 * max(1.0, cur_M):
            # Add violated constraints
            rows_set = set(int(r) for r in rows)
            violated = np.where(violations > 1e-7 * max(1.0, cur_M))[0]
            # Sort by violation magnitude
            order = np.argsort(violations)[::-1]
            added = 0
            for idx in order:
                idx = int(idx)
                if len(rows) >= max_rows:
                    break
                if added >= 512:
                    break
                if idx not in rows_set:
                    rows.append(idx)
                    rows_set.add(idx)
                    added += 1
            rows.sort()
            if added > 0:
                continue
            else:
                break

        # g is feasible. Take a step.
        g_norm = g / sg  # normalize to sum 1
        best_new_M = cur_M
        best_new_a = None
        for t in [1.0, 0.9, 0.75, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.clip(a_new, 0.0, None)
            s_new = float(a_new.sum())
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = conv_full(a_new)
            c_new = np.maximum(c_new, 0.0)
            m_new = float(c_new.max())
            if m_new < best_new_M - 1e-12:
                best_new_M = m_new
                best_new_a = a_new

        if best_new_a is None:
            break

        a = best_new_a
        cur_M = best_new_M
        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()

        if it % 20 == 0:
            print(f"    LP iter={it}, M={cur_M:.8f}, score={2*n*cur_M:.6f}, rows={len(rows)}")

        # Check convergence
        if cur_M < best_M * (1 - 1e-10):
            pass  # always updating best
        if it > 50 and cur_M > best_M * (1 - 1e-12):
            break

    return best_a, 2.0 * n * best_M


def pnrm_gradient_step(a, p=20):
    """Compute gradient of (sum c_k^p)^(1/p) w.r.t. a, where c=conv(a,a)."""
    n = a.size
    c = conv_full(a)
    c = np.maximum(c, 1e-30)
    cp = c ** p
    sum_cp = float(cp.sum())
    if sum_cp < 1e-300:
        return np.zeros(n), 1.0
    val = sum_cp ** (1.0 / p)
    w = p * c ** (p - 1)

    # grad_a[j] = sum_k w[k] * a[k-j] + sum_k w[k] * a[j-k]  (with a[m]=0 out of range)
    # First term: cross-correlation of w and a at lag j
    # Second term: convolution of w and a at index j
    L = next_pow2(2 * n - 1)
    Pw = np.fft.rfft(w, L)
    Pa = np.fft.rfft(a, L)
    # cross_corr(w, a)[j] = sum_m w[m+j]*a[m] = ifft(fft(w)*conj(fft(a)))[j]
    cc = np.fft.irfft(Pw * np.conj(Pa), L)[:n]
    # conv(w, a)[j] = sum_m w[m]*a[j-m] = ifft(fft(w)*fft(a))[j]
    cv = np.fft.irfft(Pw * Pa, L)[:n]
    grad_a = cc + cv
    # Scale by derivative of p-norm
    scale = (1.0 / p) * sum_cp ** (1.0 / p - 1)
    grad_a *= scale
    return grad_a, val


def pnrm_gd(a, budget_s, p=20, lr=None):
    """Adam gradient descent on p-norm smoothed peak autocorrelation."""
    a = normalize(a)
    n = a.size
    if budget_s < 3:
        return a, score_normalized(a)

    if lr is None:
        lr = 0.5 / np.sqrt(float(n))

    x = np.log(a + 1e-300)
    x -= x.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    best_s = score_normalized(a)

    L = next_pow2(2 * n - 1)
    end = time.monotonic() + budget_s
    steps_per_sec_target = 50.0

    while time.monotonic() < end - 1:
        t += 1
        frac = min(1.0, t / max(1, int(budget_s * steps_per_sec_target)))
        p_cur = max(4, int(p * (1.5 - 1.0 * frac)))
        p_cur = min(80, p_cur)

        # Compute p-norm objective and gradient
        c = np.fft.irfft(np.fft.rfft(a, L) ** 2, L)[:2 * n - 1].real
        c = np.maximum(c, 1e-30)
        cp = c ** p_cur
        sum_cp = float(cp.sum())
        if sum_cp < 1e-300:
            break
        obj_val = sum_cp ** (1.0 / p_cur)
        curr_score = 2.0 * n * obj_val

        if np.isfinite(curr_score) and curr_score < best_s:
            best_s = curr_score
            best_a = a.copy()

        # Gradient
        w = p_cur * c ** (p_cur - 1)
        Pw = np.fft.rfft(w, L)
        Pa = np.fft.rfft(a, L)
        cc = np.fft.irfft(Pw * np.conj(Pa), L)[:n]
        cv = np.fft.irfft(Pw * Pa, L)[:n]
        grad_a = (cc + cv) / (p_cur * sum_cp ** (1.0 - 1.0 / p_cur))

        # Project to simplex tangent space
        mean_grad = float(np.dot(a, grad_a))
        grad_x = a * (grad_a - mean_grad)

        if not np.all(np.isfinite(grad_x)):
            break

        # Adam
        m = b1 * m + (1 - b1) * grad_x
        v = b2 * v + (1 - b2) * grad_x ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        lr_cur = lr * (1.0 + 3.0 * np.exp(-5.0 * frac))
        step = lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -20.0, 20.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            x = x - x * 0.5  # dampen
            if not np.all(np.isfinite(x)):
                break
        x = np.clip(x, -60.0, 60.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if se < 1e-300 or not np.isfinite(se):
            break
        a = e / se

    final_s = score_normalized(a)
    if final_s < best_s:
        best_s = final_s
        best_a = a.copy()
    return best_a, best_s


def scale_output(a, target_sum=10.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:

    a = np.clip(a, 0.0, 1000.0)

    return [float(x) for x in b]

        else:
            base = normalize(greedy_coeffs(n))
        # Perturb
        sigma = 0.05 + 0.1 * (trial % 4)
        noisy = base * np.exp(rng.normal(0, sigma, n))
        a0 = normalize(noisy)
        # Quick optimize
        gd_alloc = min(80.0, rem * 0.2)
        a_out, s_out = pnrm_gd(a0, gd_alloc, p=20)
        record(a_out, s_out)
        # If good, LP
        if s_out < 1.52 and rem - time.monotonic() > 40:
            lp_alloc = min(80.0, rem * 0.2)
            a_out2, s_out2 = lp_weierstrass(a_out, lp_alloc)
            record(a_out2, s_out2)

    print(f"  After multi-start: best={best_score:.6f}")

    # === Phase 6: Final intensive polish ===
    print("Phase 6: Final polish")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 120:
            # Extended LP
            alloc = min(rem - 20, 300.0)
            print(f"  Final LP, budget={alloc:.0f}s")
            a_out, s_out = lp_weierstrass(best_a.copy(), alloc)
            record(a_out, s_out)
        rem = deadline - time.monotonic()
        if rem > 60:
            # Extended GD with high p
            alloc = min(rem - 10, 150.0)
            print(f"  Final GD p=50, budget={alloc:.0f}s")
            a_out, s_out = pnrm_gd(best_a.copy(), alloc, p=50)
            record(a_out, s_out)
        rem = deadline - time.monotonic()
        if rem > 30:
            # Another LP pass
            alloc = min(rem - 5, 80.0)
            a_out, s_out = lp_weierstrass(best_a.copy(), alloc)
            record(a_out, s_out)

    # === Validation ===
    if best_a is None:
        best_a = normalize(greedy_coeffs(256))

    print(f"\nBest found: n={best_a.size}, internal score={best_score:.6f}")

    # Try different scalings
    best_seq = scale_output(best_a, 10.0)
    best_val = np.inf
    for ts in [1.0, 5.0, 10.0, 20.0, 50.0, 100.0]:
        seq = scale_output(best_a, ts)
        try:
            val = evaluate_sequence(seq)
            if np.isfinite(val) and val < best_val:
                best_val = val
                best_seq = seq
        except Exception:
            pass

    if np.isfinite(best_val):
        print(f"Validated score: {best_val:.6f}")
    else:
        print("Validation failed, using internal score")

    return best_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
