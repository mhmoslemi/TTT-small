    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def make_chirp_pos(n, freq=1.0):
    i = np.arange(n, dtype=np.float64)
    return 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * i * i / n)

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return 1.0 + np.exp(-x*x / (2.0*sigma*sigma))  # offset to keep positive

def make_linear_taper(n):
    x = np.linspace(0.5, 1.0, n)
    return x * (2.0 - x)  # symmetric quartic-ish bump
