import time
import numpy as np

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s

def greedy_normalized(n):
    n = int(max(n, 16))
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return normalize_nonneg(a)

def power_normalized(n, s):
    n = int(max(n, 16))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)

def actual_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    if n <= 1024:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2*n-1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P*P, L)[:2*n-1].real
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s*s))

def score_and_conv(a):
    """Returns (score, convolution) for a normalized sequence."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-12:
        return np.inf, np.zeros(2*n-1)
    an = a / s  # normalize to sum 1
    if n <= 1024:
        c = np.convolve(an, an)

        L = next_power_of_two(2*n-1)
        P = np.fft.rfft(an, L)
        c = np.fft.irfft(P*P, L)[:2*n-1].real
    c = np.maximum(c, 0.0)

    score = float(2.0 * n * maxc)  # since sum=1

"""
Search for a non-negative step-function sequence minimizing
    f(seq) = 2*n*max(convolve(seq, seq)) / sum(seq)^2
using a novel power-mean annealing approach combined with subgradient
max-minimization and coordinate descent polishing.

Key innovation: Instead of directly optimizing the (non-smooth) max objective,
we use a power-mean objective (sum c^p)^(1/p) with increasing p, whose gradient
can be computed in O(n log n) via FFT. This "annealing" from smooth (p=2) to
sharp (p=large) effectively flattens the convolution.
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(max(n, 16))
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(max(n, 16))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def conv_fft(a, b):
    """Linear convolution via FFT."""
    na, nb = len(a), len(b)
    L = next_power_of_two(na + nb - 1)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:na + nb - 1].real


def compute_conv(a):
    """Compute linear auto-convolution of a."""
    n = len(a)
    if n <= 512:
        return np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        return np.fft.irfft(P * P, L)[:2 * n - 1].real


def actual_score(a):
    """Compute the evaluation score for a sequence (normalized or not)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    c = compute_conv(a)
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def power_mean_obj_and_grad(a, p):
    """
    Compute power mean objective and its gradient.
    Objective: (sum c_k^p)^(1/p) where c = conv(a,a), a sums to 1.
    Returns (objective_value, gradient_array).
    """
    n = len(a)
    c = compute_conv(a)
    c = np.maximum(c, 1e-300)  # avoid 0^0 issues
    c_sum_p = float(np.sum(c ** p))
    if c_sum_p <= 0 or not np.isfinite(c_sum_p):
        return np.inf, np.zeros(n)
    obj = c_sum_p ** (1.0 / p)
    # Gradient: 2n * (sum c^p)^(1/p - 1) * h[i]
    # where h[i] = sum_j c[i+j]^(p-1) * a[j]
    # h = conv(c^(p-1), a[::-1])[n-1:2n-1]
    c_pw = c ** (p - 1.0)
    a_rev = a[::-1]
    if n <= 512:
        full_conv = np.convolve(c_pw, a_rev)
    else:
        full_conv = conv_fft(c_pw, a_rev)
    h = full_conv[n - 1:2 * n - 1]
    scale = 2.0 * n * c_sum_p ** (1.0 / p - 1.0)
    grad = scale * h
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return obj, grad


def power_mean_optimize(a, budget_s, p_schedule, base_lr, minlr):
    """
    Optimize using power-mean objective with annealing schedule.
    a: sequence summing to 1, non-negative.
    p_schedule: list of p values to anneal through.
    Returns best sequence found.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = normalize_nonneg(a)
    n = len(a)
    best_a = a.copy()
    best_score = actual_score(a)
    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)

    lr = base_lr
    stall_count = 0
    prev_obj = None

    for p_val in p_schedule:
        if time.monotonic() >= end:
            break
        p_current = float(p_val)
        it_in_p = 0
        max_it_in_p = int(max(50, 3000.0 / max(1.0, n / 64.0)))

        while it_in_p < max_it_in_p and time.monotonic() < end:
            it_in_p += 1
            obj, grad = power_mean_obj_and_grad(a, p_current)
            if not np.isfinite(obj):
                break

            # Check actual score
            sc = actual_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

            if prev_obj is not None and (prev_obj - obj) < 1e-15 * max(1.0, prev_obj):
                # Converged at this p level
                lr *= 0.5
                if lr < minlr:
                    break
            prev_obj = obj

            # Gradient step with projection
            a_new = a - lr * grad
            a_new = np.clip(a_new, 0.0, None)
            s_new = float(a_new.sum())
            if s_new <= 1e-12:
                a_new = a  # revert
                lr *= 0.5
                continue
            a_new = a_new / s_new

            # Line search: check if score improved
            sc_new = actual_score(a_new)
            if sc_new <= sc + 1e-12:
                a = a_new
            else:
                # Try half step
                a_half = a - 0.5 * lr * grad
                a_half = np.clip(a_half, 0.0, None)
                sh = float(a_half.sum())
                if sh > 1e-12:
                    a_half = a_half / sh
                    sc_half = actual_score(a_half)
                    if sc_half <= sc + 1e-12:
                        a = a_half
                        lr *= 0.8
                    else:
                        lr *= 0.5
                else:
                    lr *= 0.5

            if lr < minlr:
                break

        # Reset lr for next p level (slightly smaller)
        lr = max(minlr, base_lr * 0.7 ** (len(p_schedule)))

    return best_a, best_score


def adam_max_optimize(a, budget_s, seed=0):
    """
    Adam optimizer for the max-convolution objective using softmax gradient.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = normalize_nonneg(a)
    n = len(a)
    if n == 0:
        return a, np.inf

    if n < 64:
        lr = 0.04
    elif n < 256:
        lr = 0.025
    elif n < 1024:
        lr = 0.015
    else:
        lr = 0.01

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(2 * n - 1)
    best_a = a.copy()
    best_score = actual_score(a)
    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)
    stall = 0

    while time.monotonic() < end:
        t += 1
        frac = min(1.0, (time.monotonic() - start) / max(1e-9, budget_s))
        alpha = 8.0 + 60.0 * np.exp(-5.0 * frac)

        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
        c = np.maximum(c, 1e-300)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        sc = float(2.0 * n * maxc)  # a sums to 1
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Softmax weights on convolution peaks
        z = alpha * (c / maxc - 1.0)
        z = np.clip(z, -100.0, 0.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient: 4n * conv(w_reversed, a) at appropriate indices
        W = np.fft.rfft(w[::-1], L)
        C = np.fft.irfft(P * W, L)
        grad_a = (4.0 * n) * C[n - 1:2 * n - 1][::-1].real
        grad_a = np.maximum(grad_a, 0.0)

        # Subtract mean for normalization invariance
        gp = float(np.dot(a, grad_a))
        grad_x = grad_a - gp

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)

        # Work in log-space for non-negativity
        log_a = np.log(a + 1e-300)
        log_a = log_a - log_a.mean() - step
        log_a = np.clip(log_a, -80.0, 80.0)
        log_a = log_a - log_a.max()
        ea = np.exp(log_a)
        se = float(ea.sum())
        if not np.isfinite(se) or se <= 0:
            break
        a = ea / se

        if stall > 200:
            break

    final_sc = actual_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score


def coordinate_descent_polish(a, budget_s, n_trials=200):
    """
    Coordinate descent: try small perturbations on individual coordinates.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = normalize_nonneg(a)
    n = len(a)
    best_a = a.copy()
    best_score = actual_score(a)
    start = time.monotonic()
    end = start + max(0.0, budget_s)

    delta = 0.02
    no_improve = 0
    max_no_improve = 50

    idx = np.arange(n)
    np.random.shuffle(idx)
    pos = 0

    while time.monotonic() < end and no_improve < max_no_improve:
        i = idx[pos % n]
        pos += 1

        for sign in (+1, -1):
            a_trial = a.copy()
            a_trial[i] = max(0.0, a_trial[i] + sign * delta)
            s = float(a_trial.sum())
            if s <= 1e-12:
                continue
            a_trial = a_trial / s
            sc = actual_score(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_a = a.copy()
                best_score = sc
                no_improve = 0
                break
        else:
            no_improve += 1

        # Adaptively reduce delta
        if no_improve > max_no_improve // 2:
            delta *= 0.7

    return best_a, best_score


def symmetrize(a):
    """Average with reversed version."""
    return normalize_nonneg(0.5 * (a + a[::-1]))


def lp_improve(p, budget_s):
    """
    Cutting-plane LP approach for refinement.
    """
    if _scipy_opt is None:
        return p, actual_score(p)

    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n < 2 or n > 8000:
        return p, actual_score(p)
    budget_s = float(budget_s)
    if budget_s < 20.0:
        return p, actual_score(p)

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score(f)
    best_f = f.copy()
    best_score = cur_score

    end = time.monotonic() + max(0.0, budget_s - 5.0)

    max_rows = min(800, max(100, 6000000 // max(1, n)))
    r0 = min(max_rows, max(60, max_rows // 2))
    rows = list(np.unique(np.linspace(0, 2 * n - 2, r0, dtype=int)).astype(int))
    row_set = set(rows)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        A = np.zeros((int(rows_arr.size), n), dtype=np.float64)
        for ii in range(int(rows_arr.size)):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(int(rows_arr.size), cur_score, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(5.0, min(60.0, rem * 0.3))
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None), method="highs",
                options={"time_limit": float(tl)}
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False):
            break

        g = np.clip(np.asarray(res.x, dtype=np.float64).ravel(), 0.0, None)
        sg = float(g.sum())
        if sg <= 1e-12:
            break

        conv_check = compute_conv(f * g / sg)
        conv_check = np.maximum(conv_check, 0.0)
        violations = conv_check - cur_score * (sg / sg)
        tol = 1e-7 + 1e-7 * cur_score

        if np.any(violations > tol) and len(rows) < max_rows:
            order = np.argsort(violations)[::-1]
            added = 0
            for ix in order:
                if added >= 128:
                    break
                ix = int(ix)
                if violations[ix] > tol and ix not in row_set:
                    rows.append(ix)
                    row_set.add(ix)
                    added += 1
            if added == 0 and float(np.max(violations)) > 10.0 * tol:
                break
            continue

        g = g * (S0 / sg)
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.5, 0.25, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()

        if accepted is None or accepted_score >= best_score:
            break

        f = accepted
        cur_score = accepted_score
        best_f = f.copy()
        best_score = cur_score

        if it % 3 == 0:
            print("  LP iter %d: score=%.6f rows=%d" % (it, best_score, len(rows)))

    return normalize_nonneg(best_f), best_score

        if cd_budget > 15:
            p_cd, sc_cd = coordinate_descent_polish(best_p, cd_budget)
            record(p_cd, sc_cd)
            # Also try symmetrized version
            p_sym = symmetrize(best_p)
            p_cd2, sc_cd2 = coordinate_descent_polish(p_sym, cd_budget * 0.5)
            record(p_cd2, sc_cd2)

    # Phase 4: LP cutting plane (if available and time permits)
    if _scipy_opt is not None:
        print("\nPhase 4: LP cutting plane")
        lp_total = min(400.0, max(30.0, 0.25 * (deadline - time.monotonic())))
        lp_lengths = sorted(best_by_n.keys(), key=lambda x: best_by_n[x][0])[:3]
        for n in lp_lengths:
            if time.monotonic() >= deadline:
                break
            if n > 5000:
                continue
            rec = best_by_n.get(int(n))
            base = rec[1] if rec else greedy_normalized(n)
            alloc = min(lp_total, max(30.0, (deadline - time.monotonic()) * 0.3))
            if alloc < 25:
                break
            p_lp, sc_lp = lp_improve(base, alloc)
            record(p_lp, sc_lp)

    # Phase 5: Final refinement loop
    print("\nPhase 5: Final refinement")
    if best_p is not None:
        while time.monotonic() < deadline - 30:
            # Short Adam run from best
            alloc = min(120.0, max(20.0, (deadline - time.monotonic()) * 0.4))
            base = normalize_nonneg(best_p * np.exp(rng.normal(0, 0.03, len(best_p))))
            p_out, sc = adam_max_optimize(base, alloc, seed=int(rng.integers(1e6)))
            record(p_out, sc)

            # Quick polish
            if best_p is not None and time.monotonic() < deadline - 20:
                p_pol, sc_pol = coordinate_descent_polish(best_p, min(30, deadline - time.monotonic() - 15))
                record(p_pol, sc_pol)

            if best_score < 1.503:
                break

    # Symmetrize final best and check
    if best_p is not None:
        p_sym = symmetrize(best_p)
        sc_sym = actual_score(p_sym)
        if sc_sym < best_score:
            best_p = p_sym
            best_score = sc_sym
            print("Symmetrization improved: %.6f" % best_score)

    # Validate with global evaluator
    print("\nFinal validation")
    if best_p is not None:
        seq = scale_output(best_p, 10.0)
        gval = try_eval(seq)
        print("  Global eval: %.6f" % gval)
        if np.isfinite(gval) and gval < best_score:
            best_score = gval

    # Also check a few other candidates
    items = sorted(best_by_n.items(), key=lambda x: x[1][0])[:3]
    for n, (sc, p) in items:
        if time.monotonic() >= deadline - 5:
            break
        seq = scale_output(p, 10.0)
        gv = try_eval(seq)
        print("  n=%d: local=%.6f global=%.6f" % (n, sc, gv))

    print("\nBest score: %.6f" % best_score)

    if best_p is None:
        best_p = greedy_normalized(384)

    return scale_output(best_p, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
