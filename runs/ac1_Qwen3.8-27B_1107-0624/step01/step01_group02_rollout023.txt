import numpy as np
import time
from scipy.optimize import linprog

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def score_of(a):
    """Score for normalized (sum=1) sequence."""
    if len(a) < 1:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * len(a) * c.max()

def compute_subgrad(a, k_star):
    """Compute subgradient of max(conv) at k_star."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k_star - (n-1))
    j_end = min(n-1, k_star)
    if j_end >= j_start:
        length = j_end - j_start + 1
        idx = k_star - j_end
        grad[j_start:j_end+1] = a[idx:idx+length][::-1]
    return grad  # factor of 2 absorbed into step size

def multiplicative_step(a, grad, step):
    """Multiplicative (mirror descent) step."""
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - step * grad
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    return a_new / a_new.sum()

def lp_direction(f, M, max_rows):
    """Solve LP for a good direction."""
    n = len(f)
    c = np.convolve(f, f)
    # Select active rows
    if max_rows >= len(c):
        rows = np.arange(len(c))
    else:
        rows = np.argpartition(c, -max_rows)[-max_rows:]
    
    # Build constraint matrix
    A = np.zeros((len(rows), n))
    for i, k in enumerate(rows):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        if j_hi > j_lo:
            A[i, j_lo:j_hi] = f[k - (j_lo + np.arange(j_hi - j_lo))][::-1]
    
    # Normalize by M
    A = A / M
    cvec = -np.ones(n)
    bvec = np.ones(len(rows))
    
    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=(0, None), method='highs',
                      options={'time_limit': 30})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 1.0 + 1e-6:
                return normalize(g)
    except:
        pass
    return None

def optimize_at_length(a, time_budget, rng):
    """Run optimization at a fixed length."""
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_score = score_of(a)
    deadline = time.monotonic() + time_budget
    
    step = 0.5  # initial step size
    momentum_buf = np.zeros(n)
    momentum = 0.9
    
    it = 0
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = c.max()
        cur_score = 2.0 * n * M
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        k_star = int(np.argmax(c))
        grad = compute_subgrad(a, k_star)
        
        # Momentum
        momentum_buf = momentum * momentum_buf + (1 - momentum) * grad
        
        # Try several step sizes
        best_next = a
        best_next_score = cur_score
        for s in [step * 0.5, step, step * 2.0]:
            cand = multiplicative_step(a, momentum_buf, s)
            sc = score_of(cand)
            if sc < best_next_score:
                best_next_score = sc
                best_next = cand
        
        if best_next_score < cur_score:
            a = best_next
            step = min(step * 1.1, 2.0)
        else:
            step = max(step * 0.7, 0.01)
        
        it += 1
        
        # Periodic LP injection
        if n <= 4096 and it % 50 == 0 and time.monotonic() < deadline - 10:
            max_rows = min(2*n-1, max(100, n))
            g = lp_direction(a, M, max_rows)
            if g is not None:
                # Blend with current
                for t in [0.3, 0.5, 0.7]:
                    blend = normalize((1-t)*a + t*g)
                    sc = score_of(blend)
                    if sc < cur_score:
                        a = blend
                        break
        
        # Periodic perturbation
        if it % 200 == 0:
            a = normalize(a + rng.normal(0, 0.01, n))
    
    return best_a, best_score

    else:

        a[n//2:] = (1/3) / (n - n//2)
        best_seq = normalize(a)
        best_score = score_of(best_seq)
    
    # Multi-scale optimization
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    
    for target_n in lengths:
        if time.monotonic() > deadline - 30:
            break
        budget = min(150, deadline - time.monotonic() - 10)
        if budget < 10:

        # Resize best to target_n
        if best_seq.size != target_n:
            # Interpolate
            x_old = np.linspace(0, 1, best_seq.size)

Algorithm: Multi-scale multiplicative mirror descent with LP injection and scipy smooth optimization.
- Normalizes to sum=1 (score = 2*n*max(conv(a,a)))
- Uses mirror descent (multiplicative updates) on the simplex for scalability
- FFT-based convolution and smooth subgradient computation
- Periodic LP direction injection for large jumps
- scipy L-BFGS-B on smooth-max approximation for moderate n
- Multi-scale refinement: start small, progressively increase n

import numpy as np
import time

try:
    from scipy.optimize import linprog
    from scipy.optimize import minimize as sp_minimize
except Exception:
    linprog = None
    sp_minimize = None


def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def score_of(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size < 1:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * a.size * c.max())


def fft_convolve(a, b):
    n = a.size + b.size - 1
    size = 1
    while size < n:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]


def compute_smooth_grad(a, weights):
    """grad[j] = sum_k weights[k] * a[k-j] for valid indices, j=0..n-1."""
    n = a.size
    size = 1
    while size < 3 * n:
        size <<= 1
    a_flip = np.zeros(size, dtype=np.float64)
    a_flip[:n] = a[::-1]
    w_pad = np.zeros(size, dtype=np.float64)
    w_pad[:len(weights)] = weights
    conv = np.fft.irfft(np.fft.rfft(w_pad) * np.fft.rfft(a_flip), size)
    return conv[n - 1:n - 1 + n]


def multiplicative_update(a, grad, eta):
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * grad
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(a, M, max_rows, time_limit=15.0):
    if linprog is None:
        return None
    n = a.size
    if n > 4096 or M < 1e-300:
        return None
    c = np.convolve(a, a)
    if max_rows >= c.size:
        rows = np.arange(c.size)
    else:
        rows = np.argpartition(c, -max_rows)[-max_rows:]
    A = np.zeros((rows.size, n), dtype=np.float64)
    for i, k in enumerate(rows):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        if j_hi > j_lo:
            idx = j_lo + np.arange(j_hi - j_lo)
            A[i, j_lo:j_hi] = a[k - idx]
    A = A / M
    cvec = -np.ones(n)
    bvec = np.ones(rows.size)
    t0 = time.monotonic()
    try:
        opts = {'presolve': True}
        if time_limit:
            opts['time_limit'] = time_limit
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=(0.0, None),
                      method='highs', options=opts)
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1.0 + 1e-6:
                return normalize(g)
    except Exception:
        pass
    return None


def resize_sequence(a, target_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == target_n:
        return normalize(a)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, target_n)
    out = np.interp(x_new, x_old, a)
    return normalize(np.maximum(out, 0.0))


def optimize_mirror_descent(a, budget, rng):
    """Core multiplicative mirror descent optimizer."""
    n = a.size
    a = normalize(a)
    best_a = a.copy()
    best_score = score_of(a)
    deadline = time.monotonic() + budget

    eta = 1.0
    momentum = 0.7
    buf = np.zeros(n)
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        c = fft_convolve(a, a)
        M = float(c.max())
        cur = 2.0 * n * M

        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 100:
            a = normalize(a + rng.normal(0, 0.02, n))
            eta = max(eta * 0.5, 0.05)
            stall = 0
            continue

        k_star = int(np.argmax(c))
        # Exact subgradient (shifted a)
        grad = np.zeros(n)
        j_start = max(0, k_star - (n - 1))
        j_end = min(n - 1, k_star)
        if j_end >= j_start:
            length = j_end - j_start + 1
            idx = k_star - j_end
            grad[j_start:j_end + 1] = a[idx:idx + length][::-1]

        # Smooth grad using top few lags
        if n <= 4096:
            topk = min(16, c.size)
            top_idx = np.argpartition(c, -topk)[-topk:]
            top_c = c[top_idx]
            temp = max(0.01, M * 0.1)
            w_local = np.exp((top_c - top_c.max()) / temp)
            w_local /= w_local.sum()
            # Map to full weights
            w_full = np.zeros(c.size)
            w_full[top_idx] = w_local
            smooth_g = compute_smooth_grad(a, w_full)
            combined = 0.5 * grad + 0.5 * smooth_g
        else:
            combined = grad

        buf = momentum * buf + (1.0 - momentum) * combined
        max_g = float(np.max(np.abs(buf)))
        if max_g < 1e-300:
            break

        eta_scaled = eta / max_g
        best_next = a
        best_next_score = cur
        for s in [eta_scaled * 0.3, eta_scaled * 0.7, eta_scaled * 1.5, eta_scaled * 3.0]:
            cand = multiplicative_update(a, buf, s)
            sc = score_of(cand)
            if sc < best_next_score:
                best_next_score = sc
                best_next = cand

        if best_next_score < cur - 1e-15:
            a = best_next
            eta = min(eta * 1.05, 10.0)
        else:
            eta = max(eta * 0.75, 0.01)

        # LP injection
        if it % 80 == 40 and n <= 4096 and time.monotonic() < deadline - 20:
            max_rows = min(2 * n - 1, max(200, n))
            g = lp_direction(a, M, max_rows, time_limit=10.0)
            if g is not None:
                for t in [0.2, 0.4, 0.6, 0.8]:
                    blend = normalize((1.0 - t) * a + t * g)
                    sc = score_of(blend)
                    if sc < cur:
                        a = blend
                        break

        it += 1

    return best_a, best_score

        t0 = time.monotonic()

        # Mirror descent
        a_md, sc_md = optimize_mirror_descent(a, alloc * 0.6, rng)
        print(f"  Mirror descent: {sc_md:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)

        # Scipy smooth (only for moderate n)
        if target_n <= 4096 and time.monotonic() < hard_deadline - 20:
            a_sp, sc_sp = optimize_scipy_smooth(a_md, alloc * 0.35, rng)
            print(f"  Scipy smooth:   {sc_sp:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)
            if sc_sp < sc_md:
                a_md, sc_md = a_sp, sc_sp

        if sc_md < best_score:
            best_score = sc_md
            best_seq = a_md.copy()
            print(f"  ** New best: {best_score:.8f} at n={target_n} **", flush=True)

    # Final polish at the best length
    if time.monotonic() < hard_deadline - 10:
        remaining = hard_deadline - time.monotonic() - 5
        print(f"\n--- Final polish: n={best_seq.size}, budget={remaining:.0f}s ---", flush=True)
        a_pol, sc_pol = optimize_mirror_descent(best_seq, remaining, rng)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
            print(f"  Polish result: {best_score:.8f}", flush=True)

    # Verify with official evaluator
    result = [float(x) for x in best_seq]
    try:
        official = evaluate_sequence(result)
        print(f"\nFinal official score: {official:.10f}, n={len(result)}", flush=True)
    except Exception:
        pass

    return result
