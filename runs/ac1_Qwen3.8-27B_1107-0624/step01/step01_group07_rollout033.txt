```python
import time
from scipy.optimize import minimize, linprog

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def softmax_obj_and_grad(x, peak_tau=0.05):
    """Objective and gradient for softmax parameterization."""
    # Numerically stable softmax
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    n = len(a)
    
    conv = np.convolve(a, a)
    
    # Softmax weights over peaks
    conv_max = conv.max()
    conv_min = conv.min()
    # Use a window around the max
    threshold = conv_max - peak_tau
    weights = np.where(conv >= threshold, np.exp((conv - conv_max) / peak_tau), 0.0)
    if weights.sum() == 0:
        # Only the max
        weights = np.zeros(len(conv))
        weights[conv.argmax()] = 1.0
    else:
        weights /= weights.sum()
    
    obj = 2.0 * n * conv.max()
    
    # Subgradient: for each peak k with weight w_k, d(conv[k])/da[i] = a[k-i]
    # Smoothed: grad_a[i] = 2n * sum_k w_k * a[k-i]
    grad_a = np.zeros(n)
    for k, w in enumerate(weights):
        if w == 0:
            continue
        # conv[k] = sum_{i: 0<=k-i<n} a[i] * a[k-i]
        # d(conv[k])/da[i] = a[k-i] for i in [max(0,k-n+1), min(k, n-1)]
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            j = k - np.arange(i_start, i_end + 1)  # j = k - i
            grad_a[i_start:i_end+1] += w * a[j]
    
    grad_a *= 2.0 * n
    
    # Chain rule for softmax: dx/da
    # d(f)/dx[j] = grad_a[j] * a[j] - a[j] * sum_i grad_a[i] * a[i]
    A = np.dot(grad_a, a)
    grad_x = grad_a * a - a * A
    
    return obj, grad_x
```

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS with softmax parameterization (smooth subgradient),
multiplicative weight updates, LP refinement, and coordinate descent.
Multiple diverse initializations across several sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def compute_objective_fft(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    size = 2 * n
    fa = np.fft.fft(a, size)
    c = np.fft.ifft(fa * fa).real[:2*n-1]
    return 2.0 * n * c.max() / (S * S)

def obj_grad(x, top_k=3, tau=0.1):
    a = softmax(x)
    n = len(a)
    
    # Use FFT for convolution when n is large
    if n > 512:
        size = 2 * n
        fa = np.fft.fft(a, size)
        conv = np.fft.ifft(fa * fa).real[:2*n-1]
    else:
        conv = np.convolve(a, a)
    
    obj = 2.0 * n * conv.max()
    
    # Smoothed subgradient using top-k peaks
    if top_k >= len(conv):
        top_indices = np.arange(len(conv))
        peak_vals = conv
    else:
        top_indices = np.argsort(conv)[::-1][:top_k]
        peak_vals = conv[top_indices]
    
    # Softmax weighting over peaks
    pv = peak_vals - peak_vals.max()
    weights = np.exp(pv)
    weights /= weights.sum()
    
    grad_a = np.zeros(n)
    for idx, k in enumerate(top_indices):
        w = weights[idx]
        if w < 1e-10:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start > i_end:
            continue
        j = k - np.arange(i_start, i_end + 1)
        grad_a[i_start:i_end+1] += w * a[j]
    
    grad_a *= 2.0 * n
    
    # Softmax chain rule
    A = np.dot(grad_a, a)
    grad_x = grad_a * a - a * A
    
    return obj, grad_x

def obj_grad_simple(x):
    """Simpler version using just the argmax peak."""
    a = softmax(x)
    n = len(a)
    
    if n > 512:
        size = 2 * n
        fa = np.fft.fft(a, size)
        conv = np.fft.ifft(fa * fa).real[:2*n-1]
    else:
        conv = np.convolve(a, a)
    
    k = conv.argmax()
    obj = 2.0 * n * conv[k]
    
    grad_a = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(k, n - 1)
    j = k - np.arange(i_start, i_end + 1)
    grad_a[i_start:i_end+1] = a[j]
    grad_a *= 2.0 * n
    
    A = np.dot(grad_a, a)
    grad_x = grad_a * a - a * A
    
    return obj, grad_x

def multiplicative_update(a, lr=0.1, n_iter=200):
    """Multiplicative weight method to flatten the convolution."""
    n = len(a)
    a = a.copy()
    a /= a.sum()
    best = a.copy()
    best_score = compute_objective(a)
    
    for it in range(n_iter):
        conv = np.convolve(a, a)
        M = conv.max()
        
        # Weighted by proximity to peak
        diff = M - conv
        threshold = 0.1 * M
        mask = diff < threshold
        if not mask.any():
            mask[conv.argmax()] = True
        
        # Compute contribution of each a[i] to masked convolution positions
        grad_a = np.zeros(n)
        peak_indices = np.where(mask)[0]
        for k in peak_indices:
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            j = k - np.arange(i_start, i_end + 1)
            grad_a[i_start:i_end+1] += a[j]
        
        # Multiplicative update
        avg = grad_a.mean() + 1e-15
        a *= np.exp(-lr * grad_a / avg)
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        
        score = compute_objective(a)
        if score < best_score:
            best_score = score
            best = a.copy()
    
    return best, best_score

def lp_refine(a, deadline):
    """LP-based refinement."""
    n = len(a)
    a = a.copy()
    a /= a.sum()
    best = a.copy()
    best_score = compute_objective(a)
    
    # Pre-build the convolution matrix (sparse would be better but dense is OK for n<=2048)
    # A @ g = convolve(a, g), shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    it = 0
    while time.monotonic() < deadline - 5:
        it += 1
        conv = np.convolve(a, a)
        M = conv.max()
        
        # Solve LP: max sum(g) s.t. A @ g <= M, g >= 0
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': 30})
            if res.status == 0:
                g = res.x
                if g.sum() > 1e-10:
                    g /= g.sum()
                    # Try different step sizes
                    best_local = a.copy()
                    best_local_score = best_score
                    for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
                        a_new = (1 - t) * best + t * g
                        a_new = np.maximum(a_new, 0)
                        if a_new.sum() < 1e-10:
                            continue
                        a_new /= a_new.sum()
                        sc = compute_objective(a_new)
                        if sc < best_score:
                            best_score = sc
                            best = a_new.copy()
                            a = a_new
                            break
        except Exception:
            pass
        
        if time.monotonic() > deadline - 10:
            break
    
    return best, best_score

def coordinate_descent(a, max_iter=500):
    """Coordinate descent: optimize pairs of coordinates."""
    n = len(a)
    a = a.copy()
    a /= a.sum()
    best = a.copy()
    best_score = compute_objective(a)
    
    for it in range(max_iter):
        improved = False
        # Randomly select a few coordinate pairs to optimize
        rng = np.random.default_rng(it)
        for _ in range(10):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Optimize a[i] and a[j] (keep sum constant)
            # Actually, let's just vary a[i] (with a[j] fixed, sum changes but objective is scale-invariant)
            # Binary search for optimal a[i]
            a_old = a[i]
            best_i = a_old
            best_sc = best_score
            
            for _ in range(20):
                lo, hi = 0.0, a_old * 3 + 0.1
                if a_old < 1e-6:
                    lo, hi = 0.0, hi
                for _ in range(15):
                    mid = (lo + hi) / 2
```

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS with softmax parameterization and smoothed subgradients,
multiplicative weight updates, simulated annealing restarts,
LP refinement, and coordinate descent polishing.
Multiple diverse initializations across several sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def safe_softmax(x):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def compute_objective_fast(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if n > 1024:
        size = 2 * n
        fa = np.fft.fft(a, size)
        c = np.fft.ifft(fa * fa).real[:2 * n - 1]
    else:
        c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def lbfogs_objective(x, top_k=3, tau=0.1):
    a = safe_softmax(x)
    n = len(a)
    if n > 1024:
        size = 2 * n
        fa = np.fft.fft(a, size)
        conv = np.fft.ifft(fa * fa).real[:2 * n - 1]
    else:
        conv = np.convolve(a, a)
    obj = 2.0 * n * conv.max()
    # Smoothed subgradient using top-k peaks with softmax weighting
    top_indices = np.argsort(conv)[::-1][:top_k]
    peak_vals = conv[top_indices]
    pv = peak_vals - peak_vals.max()
    weights = np.exp(pv)
    weights /= weights.sum()
    grad_a = np.zeros(n)
    for idx, k in enumerate(top_indices):
        w = weights[idx]
        if w < 1e-10:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start > i_end:
            continue
        j = k - np.arange(i_start, i_end + 1)
        grad_a[i_start:i_end + 1] += w * a[j]
    grad_a *= 2.0 * n
    A = np.dot(grad_a, a)
    grad_x = grad_a * a - a * A
    return obj, grad_x


def lbfogs_objective_simple(x):
    a = safe_softmax(x)
    n = len(a)
    if n > 1024:
        size = 2 * n
        fa = np.fft.fft(a, size)
        conv = np.fft.ifft(fa * fa).real[:2 * n - 1]
    else:
        conv = np.convolve(a, a)
    k = int(conv.argmax())
    obj = 2.0 * n * conv[k]
    grad_a = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(k, n - 1)
    j = k - np.arange(i_start, i_end + 1)
    grad_a[i_start:i_end + 1] = a[j]
    grad_a *= 2.0 * n
    A = np.dot(grad_a, a)
    grad_x = grad_a * a - a * A
    return obj, grad_x


def run_lbfogs(x0, maxiter=500, maxfun=2000, gtol=1e-5, top_k=3, tau=0.1):
    n = len(x0)
    if top_k == 1:
        fun = lbfogs_objective_simple
    else:
        fun = lambda x: lbfogs_objective(x, top_k=top_k, tau=tau)
    try:
        result = minimize(
            fun, x0, method='L-BFGS-B',
            options={'maxiter': maxiter, 'maxfun': maxfun, 'gtol': gtol}
        )
        return safe_softmax(result.x), result
    except Exception:
        return safe_softmax(x0), None


def multiplicative_update(a, lr=0.1, n_iter=300):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    best = a.copy()
    best_score = compute_objective(a)
    for it in range(n_iter):
        conv = np.convolve(a, a)
        M = conv.max()
        if M < 1e-15:
            break
        threshold = M - 0.1 * M
        mask = conv >= threshold
        if not mask.any():
            mask[conv.argmax()] = True
        peak_indices = np.where(mask)[0]
        grad_a = np.zeros(n)
        for k in peak_indices:
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            j = k - np.arange(i_start, i_end + 1)
            grad_a[i_start:i_end + 1] += a[j]
        avg = grad_a.mean() + 1e-15
        a = a * np.exp(-lr * grad_a / avg)
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        score = compute_objective(a)
        if score < best_score:
            best_score = score
            best = a.copy()
    return best, best_score


def simulated_annealing(a_best, best_score, deadline, seed=42):
    n = len(a_best)
    rng = np.random.default_rng(seed)
    a = a_best.copy()
    current_score = best_score
    T = 0.001
    T_min = 1e-8
    cooling = 0.999
    it = 0
    while time.monotonic() < deadline:
        it += 1
        t_mix = 0.05 + 0.15 * T / 0.001
        if T < T_min:
            t_mix = 0.02
        a_new = (1 - t_mix) * a + t_mix * rng.random(n)
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            continue
        a_new /= a_new.sum()
        new_score = compute_objective(a_new)
        delta = new_score - current_score
        if delta < 0:
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                a_best = a_new.copy()
        else:
            if rng.random() < np.exp(-delta / max(T, 1e-10)):
                a = a_new
                current_score = new_score
        T *= cooling
        if it % 200 == 0:
            print(f"    SA iter {it}, T={T:.6f}, current={current_score:.6f}, best={best_score:.6f}")
    return a_best, best_score


def lp_refine(a, deadline):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    best = a.copy()
    best_score = compute_objective(a)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_mat[j:j + n, j] = a
    it = 0
    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(a, a)
        M = conv.max()
        A_mat[:, :] = 0
        for j in range(n):
            A_mat[j:j + n, j] = a
        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20})
            if res.status == 0:
                g = res.x
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g /= g_sum
                    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                        a_new = (1 - t) * best + t * g
                        a_new = np.maximum(a_new, 0)
                        if a_new.sum() < 1e-10:
                            continue
                        a_new /= a_new.sum()
                        sc = compute_objective(a_new)
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best = a_new.copy()
                            a = a_new
                            break
        except Exception:
            pass
        if it % 5 == 0:
            print(f"    LP iter {it}, score={best_score:.6f}")
        if time.monotonic() > deadline:
            break
    return best, best_score


def coordinate_descent(a, deadline):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    best = a.copy()
    best_score = compute_objective(a)
    rng = np.random.default_rng(12345)
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        improved = False
        n_tries = min(50, max(5, n // 20))
        for _ in range(n_tries):
            i = int(rng.integers(0, n))
            a_old = a[i]
            candidates = [0.0, a_old * 0.2, a_old * 0.5, a_old * 0.7,
                          a_old, a_old * 1.3, a_old * 1.7, a_old * 2.5,
                          a_old + 0.005, a_old - 0.005,
                          0.01, 0.05, 0.1, 0.5]
            candidates = [max(0.0, c) for c in candidates]
            best_c = a_old
            best_sc = best_score
            for c in candidates:
                a[i] = c
                if a.sum() < 1e-10:
                    a[i] = a_old
                    continue
                a_norm = a / a.sum()
                sc = compute_objective(a_norm)
                if sc < best_sc:
                    best_sc = sc
                    best_c = c
            a[i] = best_c
            a = a / a.sum()
            sc = compute_objective(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best = a.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    return best, best_score


def get_initializations(n, seed, parent):
    inits = []
    rng = np.random.default_rng(seed)
    # Random
    for _ in range(3):
        inits.append(rng.normal(0, 1, n))
    # Cosine
    x = 0.5 * (1 + np.cos(np.linspace(0, np.pi, n)))
    inits.append(np.log(np.maximum(x, 1e-10)) - 2)
    # Gaussian
    c = n / 2
    x = -((np.arange(n) - c) / (n / 4)) ** 2
    inits.append(x - x.max())
    # V-shaped (high at edges)
    x = -np.abs(np.arange(n) - c) * 0.05
    inits.append(x - x.max())
    # Two-bump
    x = np.exp(-((np.arange(n) - n / 4) / (n / 8)) ** 2) + np.exp(-((np.arange(n) - 3 * n / 4) / (n / 8)) ** 2)
    lx = np.log(np.maximum(x, 1e-10))
    inits.append(lx - lx.max())
    # Ramp (asymmetric)
    x = np.linspace(0.1, 1.0, n)
    lx = np.log(x)
    inits.append(lx - lx.max())
    # Anti-ramp
    x = np.linspace(1.0, 0.1, n)
    lx = np.log(x)
    inits.append(lx - lx.max())
    # Sparse random (mostly zeros with a few bumps)
    x = np.zeros(n)
    for _ in range(n // 10):
        pos = int(rng.integers(0, n))
        width = max(1, n // 50)
        x[max(0, pos - width):min(n, pos + width)] += rng.random()
    x = np.maximum(x, 1e-10)
    inits.append(np.log(x) - np.log(x).max())
    # Uniform (should give score ~2)
    inits.append(np.zeros(n))
    # Parent
    if parent is not None:
        p_r = np.interp(np.linspace(0, len(parent) - 1, n), np.arange(len(parent)), parent)
        p_r = np.maximum(p_r, 1e-10)
        lx = np.log(p_r)
        inits.append(lx - lx.max())
    return inits
