import time
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_step(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a, deadline, rng):
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.1
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        g = lp_step(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new.copy()
                best_score = sc
                a = a_new
                improved = True
                break
        if improved:
            stall = 0
            step = min(0.4, step * 1.05)
        else:
            stall += 1
            step = max(0.01, step * 0.8)
            if stall > 30:
                noise = np.abs(rng.normal(0, 0.1, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a = a_new / a_new.sum()
                    stall = 0
            if stall > 150:
                break
    return best_a, best_score


def compute_grad_from_peaks(a, c, peak_mask):
    """Compute gradient contribution from peak positions."""
    n = len(a)
    w = peak_mask.astype(np.float64)
    a_flipped = a[::-1]
    conv_res = np.convolve(w, a_flipped, mode='full')
    return conv_res[n-1:2*n-1].copy()
