Optimized step function coefficient search using multi-phase strategy:
1) FFT-accelerated gradient descent with adaptive log-sum-exp smoothing
2) Peak-tracking coordinate refinement
3) Multi-restart with diverse initializations
4) Final polishing with tight tau

import numpy as np

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_J(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def smooth_objective_and_grad(a, tau):
    """Compute smoothed J and gradient using log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tsmooth = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tsmooth / (S * S)
    
    # Gradient: dJ/da[i] = 2n * (sum_k p[k] a[k-i]) / S^2 - 2n * Tsmooth * 2 / S^3

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tsmooth) / (S * S * S)

"""
Multi-phase optimizer for step function coefficient minimization:
J(a) = 2n * max(conv(a,a)) / (sum(a))^2
Uses FFT-accelerated gradient descent with adaptive log-sum-exp smoothing,
backtracking line search, peak-tracking coordinate refinement, and diverse
restarts with carefully tuned hyperparameters.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast full self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_obj_grad(a, tau):
    """
    Compute smoothed objective and gradient via log-sum-exp.
    T_smooth = tau * log(sum exp(c[k]/tau)) approximates max(c) as tau->0.
    Returns (J_smooth, gradient).
    """
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tsmooth = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tsmooth / (S * S)

    # dJ/da[j] = 4n * cross[j] / S^2 - 4n * Tsmooth / S^3
    # where cross[j] = sum_k p[k] * a[k-j]
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tsmooth) / (S ** 3)
    return J, grad

    S = np.sum(a)
    if S > 0:

def run_gd(a_init, budget, seed, tau_start=10.0, tau_end=0.005, lr_base=0.01,
           momentum=True, anneal_prob=0.0):
    """
    Adam gradient descent with adaptive tau, optional Nesterov momentum,
    backtracking line search, and random annealing restarts.
    Normalizes to sum = sqrt(2n) internally.
    Returns (best_a, best_J).
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget
    t0 = time.monotonic()

    S_target = np.sqrt(2.0 * n)
    a = normalize_sum(a_init.copy(), S_target)

    best_a = a.copy()
    best_J = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    anneals = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(budget, 1.0))
        # Geometric tau decay
        tau = tau_start * (tau_end / tau_start) ** frac

        J, grad = smooth_obj_grad(a, tau)
        if not np.isfinite(J) or not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m, v = np.zeros(n), np.zeros(n)
            t = 0
            continue

        t += 1
        lr = lr_base / (1.0 + 0.0005 * t)
        b1, b2, eps = 0.9, 0.999, 1e-8

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        direction = -m_hat / (np.sqrt(v_hat) + eps)

        # Backtracking line search (check every 10 steps to save time)
        if t % 10 == 0:
            best_step = lr
            best_trial_J = J
            for trial_lr in [lr * 2, lr * 1.5, lr, lr * 0.5, lr * 0.25]:
                trial = a + trial_lr * direction
                trial = np.maximum(trial, 0.0)
                trial = normalize_sum(trial, S_target)
                trial_J = compute_J(trial)
                if trial_J < best_trial_J:
                    best_trial_J = trial_J
                    best_step = trial_lr
            a = np.maximum(a + best_step * direction, 0.0)
            a = normalize_sum(a, S_target)
        else:
            a = np.maximum(a + lr * direction, 0.0)
            a = normalize_sum(a, S_target)

def run_nesterov_gd(a_init, budget, seed, tau_start=5.0, tau_end=0.005, lr_base=0.005):
    """Nesterov accelerated gradient descent with adaptive tau."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget
    t0 = time.monotonic()

    S_target = np.sqrt(2.0 * n)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(budget, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_base / (1.0 + 0.0003 * t)

            a = best_a.copy()
            y = a.copy()
            t = 0
            continue

        t += 1
        new_a = a - lr * grad
        new_y = new_a + ((t - 1) / beta) * (new_a - a)

        a = np.maximum(new_a, 0.0)
        a = normalize_sum(a, S_target)
        y = np.maximum(new_y, 0.0)

        if t % 200 == 0:
            local_J = compute_J(a)

def run_peak_refinement(a_init, budget, seed):
    """
    Coordinate refinement targeting peak-contributing indices.
    Computes sensitivity (gradient of max-convolution w.r.t. each a[j])
    and performs directed moves.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget

    S_target = np.sqrt(2.0 * n)
    a = normalize_sum(a_init.copy(), S_target)
    best_a = a.copy()
    best_J = compute_J(a)

    iteration = 0

    while time.monotonic() < deadline:
        c = convolve_fft(a)
        M = np.max(c)
        S = np.sum(a)
        J = 2.0 * n * M / (S * S)

        if J < best_J:
            best_J = J
            best_a = a.copy()

        # Find hot positions
        threshold = M * (0.90 + 0.05 * min(1.0, iteration / 500.0))
        hot = np.where(c >= threshold)[0]

        if len(hot) == 0:
            hot = np.array([int(np.argmax(c))])

        # Compute sensitivity: sens[j] = sum_{k in hot} a[k-j] (valid range)
        sens = np.zeros(n)
        for k in hot:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            sens[lo:hi + 1] += a[k - hi:k - lo + 1]

        s_max = np.max(sens)
        if s_max > 0:
            sens = sens / s_max

        # Direction: decrease high-sensitivity, increase low-sensitivity
        direction = sens - np.mean(sens)

        # Try multiple step sizes
        found = False
        for step_scale in [0.1, 0.05, 0.02, 0.01, 0.005]:
            cand = a - step_scale * direction * S
            cand = np.maximum(cand, 0.0)
            cS = np.sum(cand)
            if cS < 0.01:
                continue
            cand = normalize_sum(cand, S_target)
            cand_J = compute_J(cand)
            if cand_J < best_J:
                best_J = cand_J
                best_a = cand.copy()
                a = cand.copy()
                found = True
                break

        if not found:
            # Try random single-coordinate moves
            for _ in range(20):
                idx = int(rng.integers(0, n))
                for factor in [0.7, 0.85, 1.15, 1.3]:
                    cand = a.copy()
                    cand[idx] *= factor
                    cand = np.maximum(cand, 0.0)
                    cS = np.sum(cand)
                    if cS < 0.01:
                        continue
                    cand = normalize_sum(cand, S_target)
                    cand_J = compute_J(cand)
                    if cand_J < best_J:
                        best_J = cand_J
                        best_a = cand.copy()
                        a = cand.copy()
                        break
                if time.monotonic() > deadline:
                    break

        iteration += 1

    return best_a, best_J

            best_J = cand_J
            best_a = cand.copy()
            a = cand.copy()

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Diverse gradient descent runs (Adam + Nesterov) with varied init and hyperparams
    Phase 2: Peak-tracking coordinate refinement
    Phase 3: Block perturbation + final polishing GD with tight tau
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    # Get parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.finite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(4096, dtype=np.float64)

    n = parent.size
    S_target = np.sqrt(2.0 * n)
    print(f"n = {n}, S_target = {S_target:.2f}")

    parent_norm = normalize_sum(parent, S_target)
    best_global = parent_norm.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    best_global_J = compute_J(best_global)
    best_global_J = min(best_global_J, best_global_score)
    print(f"Initial: local={compute_J(parent_norm):.6f}, verified={best_global_score:.6f}, tracked={best_global_J:.6f}")

    # Build diverse initializations
    inits = [
        ("parent", parent_norm.copy()),
        ("ones", np.ones(n)),
        ("gauss_tight", make_gaussian(n, 0.5, 0.12)),
        ("gauss_mid", make_gaussian(n, 0.5, 0.25)),
        ("gauss_wide", make_gaussian(n, 0.5, 0.40)),
        ("cos1", make_cosine_mod(n, 1.0, 0.25)),
        ("cos2", make_cosine_mod(n, 2.0, 0.15)),
        ("tapered", make_tapered(n, 0.08)),
        ("rand_smooth", make_smooth_random(n, rng, 5, 0.15)),
        ("rand_smooth2", make_smooth_random(n, np.random.default_rng(seed + 1), 8, 0.1)),
    ]

    for i, (name, arr) in enumerate(inits):
        inits[i] = (name, normalize_sum(arr, S_target))

    # Score and sort
    scored = []
    for name, arr in inits:
        J = compute_J(arr)
        scored.append((J, name, arr))
    scored.sort(key=lambda x: x[0])
    for J, name, _ in scored:
        print(f"  init '{name}': J = {J:.6f}")

    # Time allocation
    remaining = total_deadline - time.monotonic()
    p1_frac = 0.50
    p2_frac = 0.25
    p3_frac = 0.25
    p1_time = remaining * p1_frac
    p2_time = remaining * p2_frac
    p3_time = remaining * p3_frac

    # ===== Phase 1: Gradient Descent (diverse) =====
    print(f"\n{'='*50}")
    print(f"Phase 1: Gradient Descent ({p1_time:.0f}s)")
    print(f"{'='*50}")

    # Strategy: top inits get Adam GD, next get Nesterov, all get different tau schedules
    gd_configs = [
        # (init_idx, method, tau_start, tau_end, lr, budget_frac)
        (0, 'adam', 10.0, 0.005, 0.01, 0.25),
        (1, 'adam', 5.0, 0.005, 0.008, 0.20),
        (0, 'nesterov', 5.0, 0.003, 0.005, 0.20),
        (2, 'adam', 20.0, 0.01, 0.015, 0.12),
        (3, 'nesterov', 3.0, 0.003, 0.004, 0.10),
        (4, 'adam', 8.0, 0.01, 0.01, 0.13),
    ]

    for init_idx, method, tau_s, tau_e, lr, frac in gd_configs:
        if time.monotonic() >= total_deadline - 30:
            break
        budget = min(p1_time * frac, total_deadline - time.monotonic() - 15)
        if budget < 5:
            break

        J_init, name, arr = scored[init_idx] if init_idx < len(scored) else (1e10, "unk", np.ones(n))
        print(f"  {method} on '{name}' (tau: {tau_s}->{tau_e}, lr={lr}, budget={budget:.0f}s)")

        if method == 'adam':
            res_a, res_J = run_gd(arr, budget, seed=seed + init_idx * 111,
                                   tau_start=tau_s, tau_end=tau_e, lr_base=lr)
        else:
            res_a, res_J = run_nesterov_gd(arr, budget, seed=seed + init_idx * 222,
                                            tau_start=tau_s, tau_end=tau_e, lr_base=lr)

        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_J:
            best_global_J = val
            best_global = res_a.copy()
            print(f"    *** NEW BEST: {val:.6f} ***")
        else:
            print(f"    (local={res_J:.6f}, verified={val:.6f}, best={best_global_J:.6f})")

    # ===== Phase 2: Peak refinement =====
    print(f"\n{'='*50}")
    print(f"Phase 2: Peak Refinement ({p2_time:.0f}s)")
    print(f"{'='*50}")

    refine_budget = p2_time * 0.6
    if time.monotonic() < total_deadline - 30 and refine_budget > 10:
        print(f"  Refining best global (budget={refine_budget:.0f}s)")
        res_a, res_J = run_peak_refinement(best_global, refine_budget, seed=seed + 5000)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_J:
            best_global_J = val
            best_global = res_a.copy()
            print(f"    *** NEW BEST: {val:.6f} ***")
        else:
            print(f"    (local={res_J:.6f}, verified={val:.6f}, best={best_global_J:.6f})")

    # Also refine a good init that wasn't the starting point
    if time.monotonic() < total_deadline - 30:
        remaining_p2 = total_deadline - time.monotonic() - 20
        if remaining_p2 > 10:
            # Pick the best init that differs from best_global's source
            for J_i, name_i, arr_i in scored:
                if name_i != "parent" and J_i < 2.0:
                    budget = min(remaining_p2 * 0.5, 30.0)
                    if budget > 5:
                        print(f"  Refining '{name_i}' (budget={budget:.0f}s)")
                        res_a, res_J = run_peak_refinement(arr_i, budget, seed=seed + 6000)
                        val = float(evaluate_sequence(res_a.tolist()))
                        if np.isfinite(val) and val < best_global_J:
                            best_global_J = val
                            best_global = res_a.copy()
                            print(f"    *** NEW BEST: {val:.6f} ***")
                        break

    # ===== Phase 3: Block perturbation + final polish =====
    print(f"\n{'='*50}")
    print(f"Phase 3: Polishing ({p3_time:.0f}s)")
    print(f"{'='*50}")

    # Block perturbation
    bp_budget = p3_time * 0.3
    if time.monotonic() < total_deadline - 20 and bp_budget > 5:
        print(f"  Block perturbation (budget={bp_budget:.0f}s)")
        res_a, res_J = run_block_perturbation(best_global, bp_budget, seed=seed + 8000)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_J:
            best_global_J = val
            best_global = res_a.copy()
            print(f"    *** NEW BEST: {val:.6f} ***")
        else:
            print(f"    (local={res_J:.6f}, verified={val:.6f}, best={best_global_J:.6f})")

    # Final GD polish with tight tau
    remaining = total_deadline - time.monotonic() - 5
    if remaining > 10:
        # Multiple short polish runs with different tau
        polish_configs = [
            (1.0, 0.003, 0.004, 0.4),
            (0.5, 0.002, 0.003, 0.3),
            (2.0, 0.005, 0.005, 0.3),
        ]
        for tau_s, tau_e, lr, frac in polish_configs:
            if time.monotonic() >= total_deadline - 5:
                break
            budget = min(remaining * frac, total_deadline - time.monotonic() - 5)
            if budget < 5:
                break
            print(f"  Polish GD (tau: {tau_s}->{tau_e}, lr={lr}, budget={budget:.0f}s)")
            res_a, res_J = run_gd(best_global, budget, seed=seed + 9000 + int(tau_s * 100),
                                   tau_start=tau_s, tau_end=tau_e, lr_base=lr)
            val = float(evaluate_sequence(res_a.tolist()))
            if np.isfinite(val) and val < best_global_J:
                best_global_J = val
                best_global = res_a.copy()
                print(f"    *** NEW BEST: {val:.6f} ***")
            else:
                print(f"    (local={res_J:.6f}, verified={val:.6f}, best={best_global_J:.6f})")

    # One final peak refinement on the global best
    remaining = total_deadline - time.monotonic() - 3
    if remaining > 5:
        print(f"  Final peak refinement (budget={remaining:.0f}s)")
        res_a, res_J = run_peak_refinement(best_global, remaining, seed=seed + 12345)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_J:
            best_global_J = val
            best_global = res_a.copy()
            print(f"    *** NEW BEST: {val:.6f} ***")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_J = min(best_global_J, final_val)

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_global_J:.6f}")
    print(f"{'='*50}")

    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
