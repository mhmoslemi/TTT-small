            # So A[k, j] = s[k-j] for k-j in [0, n-1]

            hi = min(2*n-2, j+n-1)
            A[lo:hi+1, j] = s[j: j + (hi-lo+1)]

        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 10, 'presolve': True})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0)
                b_sum = b.sum()
                if b_sum < 1e-14:
                    break
                b = b / b_sum
                
                # Line search
                best_t = 0
                best_ts = best_sc
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    cand = (1-t)*s + t*b
                    cand = np.maximum(cand, 0)
                    cand_sum = cand.sum()
                    if cand_sum < 1e-10:
                        continue
                    cand = cand / cand_sum
                    ts = fast_score(cand)
                    if ts < best_ts:
                        best_ts = ts
                        best_t = t
                
                if best_t == 0:
                    break
                
                s = (1-best_t)*s + best_t*b
                s = np.maximum(s, 0)
                s = s / s.sum()
                
                sc = fast_score(s)
                if sc < best_sc:
                    best_sc = sc
                    best = s.copy()
        except Exception:
            break
    
    return best, best_sc

def spectral_flattening(s, n_iter=5000, deadline=None, verbose=False):
