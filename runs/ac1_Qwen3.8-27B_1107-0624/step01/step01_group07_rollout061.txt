def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with asymmetric initializations,
line search, and subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a, deadline):
    """LP-based optimization with line search."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        # Try multiple M values for diversity
        best_new_a = None
        best_new_score = best_score

        for M_factor in [1.0, 0.95, 0.9, 0.8]:
            g = lp_direction(a, M * M_factor)
            if g is None or g.sum() < 1e-15:
                continue
            g = g / g.sum()

            # Line search
            for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                a_new = (1 - t) * a + t * g
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() < 1e-10:
                    continue
                a_new = a_new / a_new.sum()
                sc = compute_objective(a_new)
                if sc < best_new_score:
                    best_new_score = sc
                    best_new_a = a_new

        if best_new_a is not None:
            a = best_new_a
            if best_new_score < best_score:
                best_a = a.copy()
                best_score = best_new_score
        else:
            # Stuck - try perturbation
            if it % 10 == 0:
                noise = np.abs(np.random.default_rng(it).normal(0, 0.1, n))
                a_pert = a * (1 + noise)
                a_pert = np.clip(a_pert, 0, None)
                if a_pert.sum() > 1e-10:
                    a_pert /= a_pert.sum()
                    sc = compute_objective(a_pert)
                    if sc < best_score:
                        best_a = a_pert.copy()
                        best_score = sc
                    a = a_pert  # Accept for exploration
            if it % 50 == 0:
                break

        if it % 200 == 0:
            print(f"      LP iter {it}, score={best_score:.6f}")

    return best_a, best_score


def optimize_subgradient(a, deadline):
    """Subgradient method with simplex projection."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    lr = 0.01
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        max_val = c.max()
        # Find all peak positions (within 1% of max)
        peak_mask = c >= 0.99 * max_val
        peak_indices = np.where(peak_mask)[0]

        # Subgradient: average of gradients at peak positions
        sg = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                val = 0.0
                if 0 <= k - i < n:
                    val += a[k - i]
                if 0 <= k - i < n and (k - i) != i:
                    val += a[k - i]
                sg[i] += val
        sg /= len(peak_indices)

        # Subgradient step
        a_new = a - lr * sg
        a_new = project_simplex(a_new, s=1.0)

        sc = compute_objective(a_new)
        if sc < best_score:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.05, lr * 1.05)
        else:
            stall += 1
            lr = max(0.001, lr * 0.9)
            if stall > 30:
                break

        if it % 100 == 0:
            print(f"      SG iter {it}, score={best_score:.6f}, lr={lr:.5f}")

    return best_a, best_score


def make_exp_decay(n, rate=1.0):
    return np.exp(-rate * np.arange(n) / n)


def make_hyperbolic(n, offset=1.0):
    return 1.0 / (np.arange(n) + offset)


def make_power_law(n, alpha=1.0):
    return (1.0 - np.arange(n) / n) ** alpha


def make_asym_gaussian(n, center_frac=0.3, sigma_frac=0.2):
    center = center_frac * n
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
