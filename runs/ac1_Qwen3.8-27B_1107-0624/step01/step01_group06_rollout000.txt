    """Compute smoothed J and its gradient for x (non-negative vector)."""
    n = len(x)
    a = np.maximum(x, 0)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c = convolve_fft(a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp  # softmax weights
    
    J = 2.0 * n * Tmax / (S * S)
    
    # Gradient: dJ/da[i] = 4n*cross[i]/S^2 - 4n*Tmax/S^3
    # where cross[i] = sum_k p[k] * a[|k-i|] (cross-correlation)
    size = 1

    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    grad -= np.mean(grad)

import time

def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)

    c = convolve_fft(a)

    c = np.fft.irfft(fa * fa, csize)[:2 * n - 1]

    # Log-sum-exp
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * (max_shifted + np.log(sum_exp))
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # Cross-correlation for gradient
    xsize = 1
    while xsize < 3 * n:
        xsize <<= 1
    fa_x = np.fft.rfft(a, xsize)
    fp = np.fft.rfft(p, xsize)
    cross = np.fft.irfft(np.conj(fa_x) * fp, xsize)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    # Project out mean (scale invariance)
    grad = grad - np.mean(grad)

    # Clip gradient for stability
    gmax = np.max(np.abs(grad))
    if gmax > 1e6:
        grad = grad / gmax * 1e6

    return J, grad


def spectral_init(n, n_iter=50, flatten_pow=0.5):
    """Initialize via iterative spectral shaping with non-negativity clipping."""
    # Start with flat spectrum target
    size = 1
    while size < n:
        size <<= 1
    A = np.zeros(size // 2 + 1, dtype=complex)
    A[0] = float(n)  # DC component
    # Add some initial structure
    A[1:] = np.exp(-np.arange(1, size // 2 + 1) * 0.1) * np.sqrt(n / size)

    a = np.zeros(n)
    a[:] = 1.0  # Start with ones

    for it in range(n_iter):
        # Current spectrum
        fa = np.fft.rfft(a, size)
        power = np.abs(fa) ** 2

        # Target: flatten the spectrum (geometric mean approach)
        # Blend current with flat
        power_target = np.ones(size // 2 + 1) * np.mean(power)
        power_new = power ** (1 - flatten_pow) * power_target ** flatten_pow

        # Reconstruct with current phase
        phase = np.angle(fa)
        fa_new = np.sqrt(power_new) * (fa / np.maximum(np.abs(fa), 1e-10))
        # Fix DC
        fa_new[0] = float(n)
        a_new = np.fft.irfft(fa_new, size)[:n]

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)

        # Renormalize
        S = np.sum(a_new)
        if S > 0.01:
            a_new = a_new * n / S
        else:
            break

        a = a_new

    return a


def cosine_window(n, order=1):
    """Raised cosine window."""
    x = np.arange(n) / (n - 1)
    return 0.5 * (1 - np.cos(2 * np.pi * order * x))


def symmetric_taper(n, power=2.0):
    """Symmetric taper with adjustable sharpness."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    t = 1.0 - np.abs(x - mid) / mid
    return t ** power


def run_lbfgs(a_init, budget_s, tau_schedule, max_iter=500, verbose=True):
    """Run L-BFGS-B with a schedule of tau values (decreasing)."""
    n = len(a_init)
    rng = np.random.default_rng(12345)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * n / S

    best_a = a.copy()
    best_J = compute_J(a)

    if verbose:
        print(f"    L-BFGS start: J={best_J:.6f}")

    for tau in tau_schedule:
        if time.monotonic() > deadline - 5.0:
            break

        # Wrap objective and gradient for scipy
        def fun(x):
            J, _ = smooth_obj_and_grad(x, tau)
            return J

        def jac(x):
            _, g = smooth_obj_and_grad(x, tau)
            return g

        # Run L-BFGS-B
        options = {
            'maxiter': max_iter,
            'ftol': 1e-12,
            'gtol': 1e-8,
            'maxls': 50,
            'disp': False,
        }

        a_work = a.copy()
        try:
            result = minimize(fun, a_work, jac=jac, method='L-BFGS-B',
                              bounds=[(0.0, None)] * n, options=options)
            a_work = np.maximum(result.x, 0.0)
            S = np.sum(a_work)
            if S > 0.01:
                a_work = a_work * n / S
        except Exception as e:
            if verbose:
                print(f"    L-BFGS tau={tau} failed: {e}")
            continue

        # Check improvement
        local_J = compute_J(a_work)
        if local_J < best_J:
            best_J = local_J
            best_a = a_work.copy()
            a = a_work.copy()
            if verbose:
                print(f"    L-BFGS tau={tau}: J={best_J:.6f} (improved)")
        else:
            # Try with a small perturbation
            a_pert = a.copy()
            w = max(1, n // 20)
            s = int(rng.integers(0, n - w))
            a_pert[s:s+w] *= np.exp(rng.normal(0, 0.1, w))
            a_pert = np.maximum(a_pert, 0)
            S = np.sum(a_pert)
            a_pert = a_pert * n / S
            pJ = compute_J(a_pert)
            if pJ < best_J:
                best_J = pJ
                best_a = a_pert.copy()
                a = a_pert.copy()
                if verbose:
                    print(f"    L-BFGS tau={tau}: perturbed J={best_J:.6f}")
            else:
                a = a_work.copy()
                if verbose:
                    print(f"    L-BFGS tau={tau}: no improvement (J={local_J:.6f})")

    # Final exact evaluation
    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def active_set_refine(a_init, budget_s, n_active=50, verbose=True):
    """Refine by focusing on indices contributing to peak convolution values."""
    n = len(a_init)
    rng = np.random.default_rng(99999)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * n / S
    best_a = a.copy()
    best_J = compute_J(a)

    if verbose:
        print(f"    Active set start: J={best_J:.6f}")

    iteration = 0
    while time.monotonic() < deadline - 2.0:
        iteration += 1
        c = convolve_fft(a)
        c_max = np.max(c)

        # Find active indices (near peak)
        threshold = c_max - 0.01 * c_max
        active_idx = np.where(c >= threshold)[0]
        if len(active_idx) > n_active:
            # Take the worst n_active
            order = np.argsort(-c[active_idx])
            active_idx = active_idx[order[:n_active]]

        # For each active index k, c[k] = sum_j a[j]*a[k-j]
        # Gradient contribution to a[i] from c[k] is a[k-i] (if valid)
        # Build sub-gradient from active set
        grad_sub = np.zeros(n)
        for k in active_idx:
            # dc[k]/da[i] = a[k-i] if 0<=k-i<n, plus a[i] if 0<=k-i<n
            # More precisely: dc[k]/da[i] = a[k-i] (from the j=i term) + a[i] (from the k-j=i term)
            # But we need to be careful about bounds
            i_range = np.arange(n)
            k_minus_i = k - i_range
            # a[k-i] where 0<=k-i<n
            mask1 = (k_minus_i >= 0) & (k_minus_i < n)
            grad_sub[mask1] += a[k_minus_i[mask1]]
            # a[i] where 0<=k-i<n (i.e., k-n+1 <= i <= k)
            mask2 = (k_minus_i >= 0) & (k_minus_i < n)
            grad_sub[mask2] += a[i_range[mask2]]

        # Normalize gradient
        gnorm = np.linalg.norm(grad_sub)
        if gnorm < 1e-10:
            break
        grad_sub = grad_sub / gnorm

        # Take a step in the negative gradient direction (reduces peak)
        # Use a small step size with backtracking
        step = 0.01 / np.sqrt(n)
        for attempt in range(20):
            cand = a - step * grad_sub
            cand = np.maximum(cand, 0.0)
            Sc = np.sum(cand)
            if Sc < 0.01:
                break
            cand = cand * n / Sc
            cand_J = compute_J(cand)
            if cand_J < best_J:
                best_J = cand_J
                best_a = cand.copy()
                a = cand.copy()
                break
            step *= 0.7

        # Also try random perturbations on active-contributing elements
        if iteration % 10 == 0:
            # Identify elements that contribute to peaks
            contributing = set()
            for k in active_idx[:10]:
                j_min = max(0, k - n + 1)
                j_max = min(k, n - 1)
                for j in range(j_min, j_max + 1):
                    if a[j] > 0.01 * np.max(a):
                        contributing.add(j)
            contributing = list(contributing)
            if contributing:
                cand = a.copy()
                for idx in rng.choice(contributing, size=min(50, len(contributing)), replace=False):
                    cand[idx] *= np.exp(rng.normal(0, 0.05))
                cand = np.maximum(cand, 0)
                Sc = np.sum(cand)
                cand = cand * n / Sc
                cJ = compute_J(cand)
                if cJ < best_J:
                    best_J = cJ
                    best_a = cand.copy()
                    a = cand.copy()

    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def fine_local_search(a_init, budget_s, verbose=True):
    """Very fine-grained local search with small random perturbations."""
    n = len(a_init)
    rng = np.random.default_rng(54321)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * n / S
    best_a = a.copy()
    best_J = compute_J(a)
    scale = 0.01  # perturbation scale

    if verbose:
        print(f"    Fine search start: J={best_J:.6f}")

    while time.monotonic() < deadline - 1.0:
        # Random block perturbation
        block_size = int(rng.integers(max(1, n // 500), max(2, n // 20)))
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + block_size)

        cand = a.copy()
        cand[start:end] *= np.exp(rng.normal(0, scale, end - start))
        cand = np.maximum(cand, 0)
        Sc = np.sum(cand)
        if Sc < 0.01:
            continue
        cand = cand * n / Sc
        cJ = compute_J(cand)

        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            scale = max(0.001, scale * 0.97)
        else:
            scale = min(0.1, scale * 1.01)

    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer:
    1. Diverse initialization including spectral shaping
    2. L-BFGS with decreasing tau (coarse to fine)
    3. Active-set peak reduction
    4. Fine local search
    5. Final L-BFGS polish
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"n = {n}")

    # Normalize parent
    parent_norm = parent.copy()
    S = np.sum(parent_norm)
    if S > 0:
        parent_norm = parent_norm * n / S

    # Initialize best
    best_a = parent_norm.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Initial score: {best_score:.6f}")

    # --- Phase 1: Generate diverse initializations (5% of time) ---
    t1_deadline = time.monotonic() + 0.05 * (total_deadline - time.monotonic())

    inits = []
    inits.append(("parent", parent_norm.copy()))
    inits.append(("ones", np.ones(n)))
    inits.append(("spectral", spectral_init(n, 50, 0.5)))
    inits.append(("spectral2", spectral_init(n, 80, 0.7)))
    inits.append(("cosine", cosine_window(n, 1)))
    inits.append(("taper2", symmetric_taper(n, 2.0)))
    inits.append(("taper4", symmetric_taper(n, 4.0)))
    inits.append(("ratio", parent_norm * (1.0 + 0.3 * np.cos(2 * np.pi * np.arange(n) / n))))

    # Score and normalize initializations
    scored_inits = []
    for name, arr in inits:
        arr = np.maximum(arr, 0.0)
        S = np.sum(arr)
        if S > 0.01:
            arr = arr * n / S
        J = compute_J(arr)
        scored_inits.append((name, J, arr.copy()))
        if time.monotonic() > t1_deadline:
            break

    scored_inits.sort(key=lambda x: x[1])
    print("Initializations:")
    for name, J, _ in scored_inits:
        print(f"  {name}: J = {J:.6f}")

    # Update best if any init is better
    for name, J, arr in scored_inits:
        if J < best_score:
            val = float(evaluate_sequence(arr.tolist()))
            if np.isfinite(val) and val < best_score:
                best_score = val
                best_a = arr.copy()
                print(f"  *** New best from init '{name}': {val:.6f}")

    # --- Phase 2: L-BFGS optimization on top initializations (55% of time) ---
    remaining = total_deadline - time.monotonic()
    lbfgs_budget = remaining * 0.55
    n_top = min(4, len(scored_inits))

    # Time allocation: more time for better initializations
    weights = []
    for i in range(n_top):
        if i == 0:
            weights.append(0.40)
        elif i == 1:
            weights.append(0.25)
        elif i == 2:
            weights.append(0.20)
        else:
            weights.append(0.15)
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    tau_schedules = [
        [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1],
        [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
        [10.0, 3.0, 1.0, 0.3, 0.1, 0.05, 0.02],
        [5.0, 2.0, 0.5, 0.2, 0.1, 0.05],
    ]

    for i in range(n_top):
        if time.monotonic() > total_deadline - 30:
            break
        name, init_J, init_arr = scored_inits[i]
        budget_i = min(lbfgs_budget * weights[i], total_deadline - time.monotonic() - 20)
        if budget_i < 3.0:
            break
        tau_sched = tau_schedules[i % len(tau_schedules)]

        print(f"\nPhase 2: L-BFGS on '{name}' (budget={budget_i:.1f}s, init_J={init_J:.4f})")
        res_a, res_J = run_lbfgs(init_arr, budget_i, tau_sched, max_iter=300, verbose=True)

        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_score:
            best_score = val
            best_a = res_a.copy()
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  (local={res_J:.6f}, verified={val:.6f}, best={best_score:.6f})")

    # --- Phase 3: Active-set refinement (20% of time) ---
    remaining = total_deadline - time.monotonic()
    active_budget = min(remaining * 0.20, total_deadline - time.monotonic() - 15)
    if active_budget > 5.0:
        print(f"\nPhase 3: Active set refinement (budget={active_budget:.1f}s)")
        res_a, res_J = active_set_refine(best_a, active_budget, n_active=30, verbose=True)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_score:
            best_score = val
            best_a = res_a.copy()
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  (local={res_J:.6f}, verified={val:.6f}, best={best_score:.6f})")

    # --- Phase 4: Fine local search (15% of time) ---
    remaining = total_deadline - time.monotonic()
    fine_budget = min(remaining * 0.25, total_deadline - time.monotonic() - 10)
    if fine_budget > 5.0:
        print(f"\nPhase 4: Fine local search (budget={fine_budget:.1f}s)")
        res_a, res_J = fine_local_search(best_a, fine_budget, verbose=True)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_score:
            best_score = val
            best_a = res_a.copy()
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  (local={res_J:.6f}, verified={val:.6f}, best={best_score:.6f})")

    # --- Phase 5: Final L-BFGS polish (remaining time) ---
    remaining = total_deadline - time.monotonic() - 3.0
    if remaining > 5.0:
        print(f"\nPhase 5: Final L-BFGS polish (budget={remaining:.1f}s)")
        tau_final = [1.0, 0.3, 0.1, 0.05, 0.02, 0.01]
        res_a, res_J = run_lbfgs(best_a, remaining, tau_final, max_iter=200, verbose=True)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_score:
            best_score = val
            best_a = res_a.copy()
            print(f"  *** NEW BEST: {val:.6f} ***")

    # Final verification
    final_val = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_val):
        best_score = min(best_score, final_val)

    print(f"\nFinal score: {best_score:.6f}")
    return [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
