```python
import numpy as np
import time

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def softmax(x, T=1.0):
    x = T * x
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

def gradient_step(s, T, lr):
    """One gradient step of the smoothed objective."""
    n = len(s)
    c = np.convolve(s, s)
    p = softmax(c, T)
    grad = 2.0 * T * np.correlate(p, s, mode='valid')
    return s - lr * grad, p, grad

def adam_update(m, v, t, grad, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    return m, v, m_hat / (np.sqrt(v_hat) + eps) * lr

def optimize_sequence(s, T_start=1.0, T_end=500.0, n_steps=5000, lr=0.01, deadline=None):
    """Optimize using Adam with T annealing."""
    n = len(s)
    s = s.copy()
    s = np.maximum(s, 0)
    s = s / s.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_s = s.copy()
    best_score = fast_score(s)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal T
        frac = step / max(n_steps - 1, 1)
        T = T_start * (T_end / T_start) ** frac  # exponential annealing
        
        c = np.convolve(s, s)
        p = softmax(c, T)
        grad = 2.0 * T * np.correlate(p, s, mode='valid')
        
        t += 1
        m, v, update = adam_update(m, v, t, grad, lr=lr)
        
        s_new = s - update
        s_new = np.maximum(s_new, 0)
        s_sum = s_new.sum()
        if s_sum < 1e-15:
            s = best_s.copy()
            continue
        s_new = s_new / s_sum
        s = s_new
        
        score = fast_score(s)
        if score < best_score:
            best_score = score
            best_s = s.copy()
    
    return best_s, best_score
```

```python
import numpy as np
import time
from scipy.optimize import linprog

"""
Hybrid optimization for minimizing 2n*max(conv(s,s))/(sum(s))^2.
Uses logsumexp-smoothed gradient descent with backtracking line search,
multiple initializations, T-annealing, and LP-based polishing.
"""


def fast_score_normalized(s):
    """Compute 2n*max(conv(s,s)) assuming sum(s)=1."""
    n = len(s)
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv)


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def softmax_safe(x, T):
    """Compute softmax(T*x) safely."""
    tx = T * x
    tx = tx - np.max(tx)
    e = np.exp(tx)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def compute_smoothed_grad(s, T):
    """Compute gradient of logsumexp(T*conv(s,s)) w.r.t. s (sum(s) fixed at 1)."""
    c = np.convolve(s, s)
    p = softmax_safe(c, T)
    grad = 2.0 * T * np.correlate(p, s, mode='valid')
    return grad


def optimize_gradient(s0, T_start=3.0, T_end=2000.0, n_steps=20000,

    s = np.maximum(s0, 0.0).astype(np.float64)

    if s_sum < 1e-15:

def build_conv_matrix(s):
    """Build (2n-1) x n matrix for convolution constraints."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def solve_lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. (s*b)[k] <= M, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_polish(s0, max_iters=50, deadline=None):
    """LP-guided polishing steps."""
    n = len(s0)
    s = s0.copy().astype(np.float64)
    s = np.maximum(s, 0.0)
    s_sum = s.sum()
    if s_sum < 1e-15:
        return s0, fast_score_normalized(s0)

    best_sc = fast_score_normalized(s)
    no_imp = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = np.convolve(s, s)
        M = np.max(conv)

        b = solve_lp_direction(s, M, time_limit=3.0)
        if b is None or np.sum(b) < 1e-14:
            break

        b = b / b.sum()

        # Line search
        best_t = 0.0
        best_ts = best_sc
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            cand = (1.0 - t) * s + t * b
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            ts = fast_score_normalized(cand)
            if ts < best_ts:
                best_ts = ts
                best_t = t

            no_imp += 1

            s = s + 0.01 * rng_local.standard_normal(n)

            ss = s.sum()
            if ss > 1e-15:
                s = s / ss

        if ss > 1e-15:
            s = s / ss
        cur_sc = fast_score_normalized(s)
        no_imp = 0

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_s = s.copy()

    return best_s, best_sc

    final_score = float(evaluate_sequence(result))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(result)}")

    return result
```
