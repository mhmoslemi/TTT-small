```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using full-constraint LP iteration
with adaptive step sizes, diverse initializations, and Adam polishing.
"""

import time
import numpy as np
import scipy.optimize as scipy_opt

def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def score_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    L = next_pow2(2*n-1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P*P, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    mc = c.max()
    if not np.isfinite(mc):
        return np.inf
    return 2.0*n*mc/(s*s)

def greedy(n):
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k-1)/(2*k)
    a /= a.sum()
    return a

def power_decay(n, alpha):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = idx**(-alpha)
    a /= a.sum()
    return a

def exp_decay(n, beta):
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-beta * idx / n)
    a /= a.sum()
    return a

def build_A(f):
    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = f.size
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # A[k,j] = f[k-j] for j in [max(0,k-n+1), min(k,n-1)]
        j_lo = max(0, k-n+1)
        j_hi = min(k, n-1)
        if j_lo <= j_hi:
            # f[k-j] for j from j_lo to j_hi
            # k-j goes from k-j_lo down to k-j_hi
            # Since j_lo >= k-n+1, k-j_lo <= n-1
            # Since j_hi <= k, k-j_hi >= 0
            A[k, j_lo:j_hi+1] = f[k-j_hi:k-j_lo+1][::-1]
    return A

def lp_direction(f):
    """Solve LP to find direction g maximizing sum(g) s.t. f*g <= max(f*f), g>=0."""
    n = f.size
    S = np.sqrt(2.0 * n)
    f_norm = f * S  # now sum(f_norm) = S, and score(f_norm) = max(f_norm * f_norm)
    
    c_conv = np.convolve(f_norm, f_norm, mode='full')
    c_conv = np.maximum(c_conv, 0.0)
    M = c_conv.max()
    
    if M <= 0 or not np.isfinite(M):
        return None
    
    A = build_A(f_norm)
    b = np.full(2*n-1, M)
    c_obj = -np.ones(n)
    
    res = scipy_opt.linprog(c_obj, A_ub=A, b_ub=b, bounds=(0, None), method='highs',
                            options={'presolve': True, 'time_limit': 10})
    if not res.success or res.x is None:
        return None
    
    g = np.maximum(res.x, 0.0)
    sg = g.sum()
    if sg < 1e-12:
        return None
    
    # Rescale g to have sum = S
    g = g * (S / sg)
    return g

def lp_iterate(f, max_iter=200, time_budget=60.0):
    """Run LP-based iteration to improve f."""
    n = f.size
    f = f.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    
    best_f = f.copy()
    best_score = score_fft(f)
    
    start = time.monotonic()
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        g = lp_direction(f)
        if g is None:
            break
        
        # Line search over step sizes
        best_t = 0.0
        best_sc = best_score
        for t in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:

            fn = np.maximum(fn, 0.0)
            sf = fn.sum()

            if sc < best_sc:
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using full-constraint LP iteration
with adaptive step sizes, diverse initializations, multi-lag weighted LP variants,
and Adam polishing. Focuses on sequence lengths 200-600 where the best solutions
are expected.

import time
import numpy as np
import scipy.optimize as scipy_opt


def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def score_fft(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    L = next_pow2(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))


def greedy_init(n):
    n = int(n)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    a /= a.sum()
    return a


def power_decay(n, alpha):
    n = int(n)
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, -float(alpha))
    a /= a.sum()
    return a


def exp_decay(n, beta):
    n = int(n)
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-float(beta) * idx / float(n))
    a /= a.sum()
    return a


def build_A(f):
    n = f.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_direction(f, top_k=None, relax=1.0):
    n = f.size
    S = np.sqrt(2.0 * n)
    f_n = f * S
    c_conv = np.convolve(f_n, f_n, mode='full')
    c_conv = np.maximum(c_conv, 0.0)
    M = float(c_conv.max())
    if M <= 0 or not np.isfinite(M):
        return None

    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(c_conv)[::-1][:top_k]
        idx_sorted = np.sort(idx)
        A_full = build_A(f_n)
        A = A_full[idx_sorted, :]
        b = np.full(top_k, M * relax)
    else:
        A = build_A(f_n)
        b = np.full(2 * n - 1, M * relax)

    c_obj = -np.ones(n)
    try:
        res = scipy_opt.linprog(c_obj, A_ub=A, b_ub=b, bounds=(0, None),
                                method='highs', options={'time_limit': 5})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.maximum(res.x, 0.0)
    sg = float(g.sum())
    if sg < 1e-12:
        return None
    g = g * (S / sg)
    return g

            break

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        sf2 = float(f.sum())
        if sf2 < 1e-12:
            break
        f = f / sf2

        if best_sc < best_score:
            best_score = best_sc
            best_f = f.copy()

        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP n={n} iter={it+1} score={best_score:.9f} t={best_t:.3f} elapsed={elapsed:.1f}s")

    return best_f, best_score


def adam_polish(f, time_budget=60.0, lr=None):
    n = f.size
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()

    if lr is None:
        if n < 200:
            lr = 0.03
        elif n < 500:
            lr = 0.02
        else:
            lr = 0.012

    x = np.log(f)
    x -= x.mean()
    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    L = next_pow2(3 * n)
    best_f = f.copy()
    best_score = score_fft(f)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, time_budget))
        alpha = 8.0 + 50.0 * np.exp(-3.5 * frac)

        P = np.fft.rfft(f, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        mc = float(c.max())
        if mc <= 0 or not np.isfinite(mc):
            break

        z = alpha * (c / mc - 1.0)
        z = z - z.max()
        w = np.exp(z)
        sw = float(w.sum())
        if sw < 1e-15:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / sw

        w_rev = w[::-1]
        W = np.fft.rfft(w_rev, L)
        Cfull = np.fft.irfft(P * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_f = 4.0 * n * C
        gp = float(np.dot(f, grad_f))
        grad_x = f * (grad_f - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -15.0, 15.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if se < 1e-15:
            break
        f = e / se

        sc = score_fft(f)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f.copy()

    return best_f, best_score


def coordinate_descent(f, time_budget=30.0):
    n = f.size
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    sf = float(f.sum())
    if sf < 1e-12:
        return f, np.inf
    f = f / sf

    best_f = f.copy()
    best_score = score_fft(f)
    start = time.monotonic()
    L = next_pow2(2 * n - 1)

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        P = np.fft.rfft(f, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        k_star = int(np.argmax(c))
        c_max = c[k_star]

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = f[i]
            f[i] = 0.0
            for delta in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.0, 1.3, 1.7, 2.0, 2.5, 3.0]:
                f[i] = orig * delta
                sc = score_fft(f)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
            f[i] = best_f[i]

        if it % 100 == 0:
            print(f"    CD iter={it} score={best_score:.9f}")

    return best_f, best_score


def scale_output(p, target=10.0):
    a = np.asarray(p, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        a = np.ones(16, dtype=np.float64)
        s = 16.0
    b = a * (target / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def try_eval(seq):
    try:
        val = evaluate_sequence(list(seq))
        val = float(val)
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf

INIT_FNS = [init_greedy, init_power45, init_power55, init_power65, init_exp1, init_exp2, init_uniform, init_triangular, init_invtriangular]


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 15.0
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = float('inf')
    best_by_n = {}

    def do_record(p, sc=None):
        nonlocal best_p, best_score
        if p is None or p.size == 0:
            return
        p = np.asarray(p, dtype=np.float64)
        p = np.maximum(p, 0.0)
        if sc is None:
            sc = score_fft(p)
        if not np.isfinite(sc):
            return
        if sc < best_score:
            best_p = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: n={p.size}, score={sc:.9f} ***")
        n = int(p.size)
        old = best_by_n.get(n)
        if old is None or sc < old[0]:
            best_by_n[n] = (float(sc), p.copy())

    # Phase 0: Evaluate all initializations across lengths
    lengths = [160, 192, 224, 256, 288, 320, 352, 384, 416, 448, 480, 512, 576, 640]
    print("Phase 0: Baseline evaluations")
    for n in lengths:
        for fn in INIT_FNS:
            try:
                p = fn(n)
                do_record(p)
            except Exception:
                pass

    # Check height_sequence_1
    try:
        h = globals()['height_sequence_1']
        h = np.asarray(h, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.size > 0:
            do_record(h)
            hs = score_fft(h)
            print(f"  height_sequence_1: n={h.size}, score={hs:.9f}")
    except Exception:
        pass

    # Phase 1: LP iteration on best candidates per length
    print("Phase 1: LP iteration")
    for n in sorted(best_by_n.keys()):
        if time.monotonic() >= deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break

        rec = best_by_n.get(n)
        if rec is None:
            continue
        base = rec[1].copy()
        base_sc = rec[0]

        if n <= 320:
            alloc = min(90.0, rem * 0.35)
        elif n <= 512:
            alloc = min(70.0, rem * 0.3)
        else:
            alloc = min(45.0, rem * 0.2)

        if alloc < 10.0:
            continue

        print(f"  LP n={n} start_score={base_sc:.9f} alloc={alloc:.0f}s")
        p_out, sc_out = lp_iterate(base, max_iter=500, time_budget=alloc, top_k=None)
        do_record(p_out, sc_out)

        # Try top-k variant (only constrain top 30 lags)
        if time.monotonic() < deadline - 20.0:
            alloc2 = min(30.0, (deadline - time.monotonic()) * 0.15)
            if alloc2 > 8.0:
                p_out2, sc_out2 = lp_iterate(base, max_iter=300, time_budget=alloc2, top_k=30)
                do_record(p_out2, sc_out2)

    # Phase 2: LP iteration with perturbed starts
    print("Phase 2: LP with perturbed starts")
    targeted_lengths = [n for n in sorted(best_by_n.keys()) if 200 <= n <= 600]
    for n in targeted_lengths:
        if time.monotonic() >= deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 30.0:
            break

        rec = best_by_n.get(n)
        if rec is None:
            continue
        base = rec[1].copy()

        for trial in range(3):
            if time.monotonic() >= deadline:
                break
            alloc = min(40.0, (deadline - time.monotonic()) * 0.2)
            if alloc < 10.0:
                break
            sigma = 0.1 + 0.05 * trial
            noise = np.exp(rng.normal(0.0, sigma, n))
            p_pert = base * noise
            p_pert = np.maximum(p_pert, 0.0)
            sp = float(p_pert.sum())
            if sp < 1e-12:
                continue
            p_pert = p_pert / sp
            p_out, sc_out = lp_iterate(p_pert, max_iter=300, time_budget=alloc)
            do_record(p_out, sc_out)

    # Phase 3: Adam polishing
    print("Phase 3: Adam polishing")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 40.0:
            alloc = min(rem * 0.5, 120.0)
            print(f"  Adam on best (n={best_p.size}), alloc={alloc:.0f}s")
            p_out, sc_out = adam_polish(best_p, time_budget=alloc)
            do_record(p_out, sc_out)

        rem = deadline - time.monotonic()
        if rem > 30.0:
            p_sc = best_p * np.exp(rng.normal(0.0, 0.06, best_p.size))
            p_sc = np.maximum(p_sc, 0.0)
            p_sc = p_sc / p_sc.sum()
            alloc2 = min(rem * 0.5, 60.0)
            p_out2, sc_out2 = adam_polish(p_sc, time_budget=alloc2)
            do_record(p_out2, sc_out2)

    # Phase 4: Coordinate descent on best
    if best_p is not None and time.monotonic() < deadline - 20.0:
        rem = deadline - time.monotonic()
        if rem > 30.0 and best_p.size <= 400:
            alloc = min(rem * 0.6, 80.0)
            print(f"  CD on best (n={best_p.size}), alloc={alloc:.0f}s")
            p_out, sc_out = coordinate_descent(best_p, time_budget=alloc)
            do_record(p_out, sc_out)

    # Phase 5: Final Adam round
    if best_p is not None and time.monotonic() < deadline - 10.0:
        rem = deadline - time.monotonic()
        if rem > 15.0:
            alloc = min(rem - 5.0, 60.0)
            p_out, sc_out = adam_polish(best_p, time_budget=alloc, lr=0.005)
            do_record(p_out, sc_out)

    # Final validation
    if best_p is not None:
        seq = scale_output(best_p, 10.0)
        gv = try_eval(seq)
        print(f"Final: fft_score={best_score:.9f}, actual_eval={gv:.9f}")
        if np.isfinite(gv) and gv < best_score:
            best_score = gv
    else:
        best_p = greedy_init(256)

    return scale_output(best_p, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print(f"Fatal error: {e}")
        import traceback
        traceback.print_exc()
        p = greedy_init(256)
        return scale_output(p, 10.0)
```
