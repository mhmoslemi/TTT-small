def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones_like(a) / a.size
    return a / s

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:len(a)] = a
    fb = np.zeros(size, dtype=np.float64)
    fb[:len(b)] = b
    result = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(fb, size), size)
    return result[:n]

def fft_cross_corr(w, f):
    """Compute g[i] = sum_j f[j] * w[i+j] for i=0..len(f)-1.
    w has length 2*len(f)-1.
    """
    n = len(f)
    # g[i] = conv[f * w_rev][2n-2-i]
    w_rev = w[::-1]
    size = 1

    m = len(a)

"""
Optimizes the L-infinity norm of the auto-convolution of a non-negative
step function using multiplicative weight (mirror descent) iterations with
adaptive temperature, FFT-based acceleration, and multi-length search.
"""

import numpy as np
import time

def normalize_nonneg(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / max(1, a.size)
    return a / s

def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1
    while size < n_out:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:len(a)] = a
    fb = np.zeros(size, dtype=np.float64)
    fb[:len(b)] = b
    return np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(fb, size), size)[:n_out]

def fft_cross_corr(w, f):
    """
    Compute g[i] = sum_{j=0}^{n-1} f[j] * w[i+j] for i = 0, ..., n-1.
    w has length 2n-1, f has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(f)
    w_rev = w[::-1]  # length 2n-1
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = f
    wa = np.zeros(size, dtype=np.float64)
    wa[:2*n-1] = w_rev
    conv = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(wa, size), size)
    # g[i] = conv[2n-2-i] for i = 0, ..., n-1
    g = conv[n-1:2*n-1].copy()
    # Wait, let me re-derive.
    # conv[p] = sum_k f[k] * w_rev[p-k] = sum_k f[k] * w[2n-2-(p-k)] = sum_k f[k] * w[2n-2-p+k]
    # We want g[i] = sum_j f[j] * w[i+j]
    # Setting j = k: g[i] = sum_k f[k] * w[i+k]
    # conv[2n-2-i] = sum_k f[k] * w[2n-2-(2n-2-i)+k] = sum_k f[k] * w[i+k] = g[i] ✓

```python
"""
Optimizes the L-infinity norm of the auto-convolution of a non-negative step
function using multiplicative weight (mirror descent) iterations with adaptive
temperature, FFT-based acceleration, multi-length search, and power-softmax
weighting for aggressive peak flattening.
"""

import numpy as np
import time


def normalize_nonneg(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / max(1, a.size)
    return a / s


def fft_convolve_auto(f):
    """Compute linear auto-convolution f*f via FFT. Returns array of length 2n-1."""
    n = len(f)
    n_out = 2 * n - 1
    size = 1
    while size < n_out:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = f
    spec = np.fft.rfft(fa, size)
    c = np.fft.irfft(spec * spec, size)[:n_out]
    return np.maximum(c, 0.0)


def fft_cross_corr(w, f):
    """
    Compute g[i] = sum_{j=0}^{n-1} f[j] * w[i+j] for i=0..n-1.
    w has length 2n-1, f has length n. Uses FFT.
    """
    n = len(f)
    n_out = 2 * n - 1
    w_rev = w[::-1].copy()
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = f
    wa = np.zeros(size, dtype=np.float64)
    wa[:n_out] = w_rev
    conv = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(wa, size), size)
    g = conv[n - 1:2 * n - 1][::-1].copy()
    return np.maximum(g, 0.0)


def resize_seq(a, N):
    """Resize sequence to length N preserving sum via interpolation/averaging."""
    a = np.asarray(a, dtype=np.float64)
    m = len(a)
    if m == N:
        return normalize_nonneg(a)
    if m < 2:
        return np.full(N, 1.0 / N)
    if m < N:
        x_old = np.linspace(0.0, m - 1, m)
        x_new = np.linspace(0.0, N - 1, N)
        a_new = np.interp(x_new, x_old, a)
        return normalize_nonneg(np.maximum(a_new, 0.0))
    else:
        # Downsample by averaging
        bins = np.linspace(0, m, N + 1).astype(int)
        a_new = np.array([a[bins[i]:bins[i+1]].sum() for i in range(N)])
        return normalize_nonneg(a_new)
