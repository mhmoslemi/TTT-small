2. FFT-accelerated gradient descent with adaptive log-sum-exp smoothing
3. Peak-aware local search
4. LP-based refinement (if time permits)

import numpy as np
import time
from scipy.optimize import linprog

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_J(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def smooth_objective_and_grad(a, tau):
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    # J = 2n * Tmax / S^2
    J = 2.0 * n * Tmax / (S * S)
    
    # Gradient: dT/da[i] = sum_j a[j] * p[i+j]
    # This is cross-correlation of a and p
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    # dJ/da[i] = 2n * (cross[i]/S - 2*Tmax/S^3)  ... wait
    # J = 2n * Tmax / S^2
    # dJ/da[i] = 2n * (dTmax/da[i] * S^2 - Tmax * 2S * dS/da[i]) / S^4
    #          = 2n * (dTmax/da[i] / S - 2*Tmax/S^3)
    # dS/da[i] = 1
    grad = 2.0 * n * (cross / S - 2.0 * Tmax / (S * S * S))
    
    return J, grad

def spectral_init(n, rng, n_iters=200):
    """Initialize using iterative spectral flattening."""
    a = np.ones(n)
    # Start with some structure
    a = a * (1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(0.5, 3.0) * np.arange(n) / n))
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    
    for it in range(n_iters):
        # Compute circular convolution (using FFT)
        fa = np.fft.rfft(a, n)
        power = fa.real**2 + fa.imag**2
        # Want power to be flat: target = mean(power)
        target = np.mean(power)
        # Update magnitude: new_mag = sqrt(target)
        new_mag = np.sqrt(target)
        # Keep phase, update magnitude
        new_fa = new_mag * (fa / (np.abs(fa) + 1e-15))
        # Inverse FFT
        a_new = np.fft.irfft(new_fa, n)
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        # Preserve sum
        a_new = a_new / np.sum(a_new)
        a = a_new
    
    return a

def run_gradient_descent(a_init, budget_s, seed, tau_start=10.0, tau_min=0.01):
    """Adam gradient descent with adaptive tau and restarts."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    # Normalize to sum = 1
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    start_time = time.monotonic()
    last_check = 0.0
    stall = 0
    stall_best = best_J
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        # Adaptive tau: start large, decrease, but adapt based on effective support
        tau = tau_start * (tau_min / tau_start) ** (frac ** 0.5)
        
        J, grad = smooth_objective_and_grad(a, tau)
        
        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        t += 1
        lr = 0.01 * tau / (1.0 + 0.1 * t / 500.0)
        
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        
        # Normalize
        S = np.sum(a)
        if S > 0:
            a = a / S
        
        # Check every 5000 iterations
        if t % 5000 == 0:
            local_J = compute_J(a)
            if local_J < best_J - 1e-8:
                best_J = local_J
                best_a = a.copy()
                stall = 0
                stall_best = local_J
            else:
                stall += 1
                if stall > 10:
                    # Restart with perturbation
                    a = best_a.copy()
                    w = max(1, n // 50)
                    s = int(rng.integers(0, n - w))
                    a[s:s+w] *= np.exp(rng.normal(0, 0.3, w))
                    a = np.maximum(a, 0.0)
                    a = a / np.sum(a)
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    stall = 0
                    tau = max(tau * 5, 1.0)
        
        # Periodic verification
        now = time.monotonic()
        if now - last_check > 10.0:
            last_check = now
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
    
    return best_a, best_J

def lp_refine(a, max_iters=5):
    """LP-based refinement following the state-of-art approach."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a_norm = a / S  # sum = 1
    
    M = np.max(convolve_fft(a_norm))
    
    for it in range(max_iters):
        # Build the convolution matrix A (2n-1 by n)
        # A[k, i] = a_norm[k-i] if 0 <= k-i <= n-1 else 0
        # This is a banded Toeplitz-like matrix
        # We'll build it implicitly
        
        # Actually for linprog, we need explicit matrix or use sparse
        from scipy.sparse import csr_matrix
        from scipy.sparse import diags
        
        # A is (2n-1) x n, A[k,i] = a_norm[k-i] if 0 <= k-i <= n-1
        # This is a "anti-Toeplitz" or "Hankel" type matrix
        # Row k has nonzeros at columns i where 0 <= k-i <= n-1, i.e., max(0,k-n+1) <= i <= min(n-1,k)
        
        # For large n, building this explicitly is expensive (12241 x 6121 ~ 75M entries)
        # But it's sparse (each row has at most n nonzeros, total ~ n^2)
        # Let's use sparse
        
        # Build sparse matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a_norm[k - i])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        
        # LP: max 1^T b s.t. A b <= M, b >= 0
        # linprog minimizes, so: min -1^T b s.t. A b <= M, 0 <= b
        c_lp = -np.ones(n)
        b_lp = np.full(2 * n - 1, M)
        
        res = linprog(c_lp, A_ub=A, b_ub=b_lp, bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0 or not np.isfinite(res.fun):
            break
        
        b_opt = res.x
        sum_b = np.sum(b_opt)
        if sum_b < 1e-10:
            break
        
        # Rescale to match normalization
        b_rescaled = b_opt / sum_b
        
        # Update
        t_mix = 0.3
        a_new = (1 - t_mix) * a_norm + t_mix * b_rescaled
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / np.sum(a_new)
        
        new_M = np.max(convolve_fft(a_new))
        if new_M < M - 1e-10:
            a_norm = a_new
            M = new_M
            print(f"    LP iter {it}: M = {M:.6f}")
        else:
            # Try smaller mixing
            t_mix = 0.1
            a_new = (1 - t_mix) * a_norm + t_mix * b_rescaled
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / np.sum(a_new)
            new_M = np.max(convolve_fft(a_new))
            if new_M < M - 1e-10:
                a_norm = a_new
                M = new_M
                print(f"    LP iter {it} (small mix): M = {M:.6f}")
            else:
                break
    
    return a_norm

def peak_reduction(a, n_steps=100):
    """Targeted reduction of peak convolution values."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    a = a / S
    
    for step in range(n_steps):
        c = convolve_fft(a)
        k_star = np.argmax(c)
        M = c[k_star]
        
        # Find the (i, k_star-i) pairs with largest product
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        products = np.array([a[i] * a[k_star - i] for i in range(i_min, i_max + 1)])
        
        # Reduce the largest contributors
        top_k = min(5, len(products))
        top_indices = np.argsort(products)[-top_k:]
        
        improved = False
        for idx in top_indices:
            i = i_min + idx
            j = k_star - i
            
            # Try reducing a[i]
            for factor in [0.9, 0.8, 0.7]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = a_trial / np.sum(a_trial)
                c_trial = convolve_fft(a_trial)
                if np.max(c_trial) < M:
                    a = a_trial
                    M = np.max(c_trial)
                    improved = True
                    break
            if improved:
                break
            
            # Try reducing a[j] (if j != i)
            if j != i:
                for factor in [0.9, 0.8, 0.7]:
                    a_trial = a.copy()
                    a_trial[j] *= factor
                    a_trial = a_trial / np.sum(a_trial)
                    c_trial = convolve_fft(a_trial)
                    if np.max(c_trial) < M:
                        a = a_trial
                        M = np.max(c_trial)
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return a

"""
Optimized step function coefficient search combining:
1. Multiple diverse initializations (spectral, structured, parent)
2. L-BFGS on smoothed (log-sum-exp) objective with progressive tau reduction
3. Peak-aware local search for final refinement
4. Scale-invariant formulation with tiny regularization for well-conditioning
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _fft_size(n_min):
    size = 1
    while size < n_min:
        size <<= 1
    return size


def convolve_fft(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = _fft_size(2 * n - 1)
    fa = rfft(a, size)
    result = irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smoothed_J_and_grad(x, tau, scale_target=None, scale_weight=0.0):
    """
    Compute smoothed J and its gradient.
    J_smooth = 2n * logsumexp(conv/tau) / S^2 + scale_weight*(S - scale_target)^2
    """
    n = len(x)
    a = np.maximum(x, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return 1e15, np.zeros_like(x)

    size = _fft_size(3 * n - 2)
    fa = rfft(a, size)
    c = irfft(fa * fa, size)[:2 * n - 1]

    c_max = c.max()
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # Gradient of 2n*Tmax/S^2 wrt a
    fp = rfft(p, size)
    cross = irfft(np.conj(fa) * fp, size)[:n]
    # dTmax/da[i] = cross[i]
    # dJ/da[i] = 2n*(cross[i]*S^2 - 2*S*Tmax*1) / S^4 = 2n*(cross[i]/S - 2*Tmax/S^3)
    grad = 2.0 * n * (cross / S - 2.0 * Tmax / (S * S * S))

    if scale_weight > 0 and scale_target is not None:
        J += scale_weight * (S - scale_target) ** 2
        grad += 2.0 * scale_weight * (S - scale_target)

    return J, grad


def spectral_init(n, rng, n_iters=300):
    """Iterative spectral flattening for initialization."""
    a = np.ones(n)
    freq = rng.uniform(1, 5)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n) / n)
    a = np.maximum(a, 0.05)
    a = a / a.sum()

    for it in range(n_iters):
        fa = rfft(a, n)
        mag = np.abs(fa)
        power = mag ** 2
        target_power = power.sum() / n
        new_mag = np.sqrt(target_power)
        # Preserve phase
        phase = fa / (mag + 1e-15)
        new_fa = new_mag * phase
        a = irfft(new_fa, n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
            break
        a = a / s
    return a


def structured_init(n, rng, style):
    """Various structured initializations."""
    x = np.arange(n)
    mid = n / 2.0
    if style == 'gauss':
        a = np.exp(-((x - mid) ** 2) / (2 * (0.2 * n) ** 2))
    elif style == 'cos1':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x / n)
    elif style == 'cos2':
        a = 1.0 + 0.25 * np.cos(4 * np.pi * x / n)
    elif style == 'cos3':
        a = 1.0 + 0.2 * np.cos(6 * np.pi * x / n)
    elif style == 'mod':
        a = (1.0 + 0.35 * np.cos(2 * np.pi * 3 * x / n)) * np.exp(-((x - mid) ** 2) / (2 * (0.35 * n) ** 2))
    elif style == 'wedge':
        a = 1.0 - 0.4 * np.abs(x - mid) / mid
    elif style == 'rand_smooth':
        a = np.exp(rng.normal(0, 0.4, n))
        # Smooth it
        kernel = np.exp(-np.arange(-10, 11) ** 2 / 20)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def run_lbfgs(a_init, budget_s, tau_start, tau_end, seed):
    """Run L-BFGS with progressive tau reduction."""
    n = len(a_init)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()

    # Scale to have mean ~1 for numerical stability
    a = a * (n / a.sum())

    best_a = a.copy()
    best_J = compute_J(a)
    scale_target = n  # prefer sum(a) = n, i.e., mean = 1
    scale_weight = 1e-8  # very tiny to break scale invariance

    # Progressive tau schedule: run L-BFGS in phases
    n_phases = 5
    tau_schedule = np.geomspace(tau_start, tau_end, n_phases)

    for phase in range(n_phases):
        if time.monotonic() >= deadline:
            break
        tau = tau_schedule[phase]
        phase_deadline = deadline

        def obj_grad(x):
            J, g = smoothed_J_and_grad(x, tau, scale_target, scale_weight)
            if not np.isfinite(J):
                return 1e15, np.zeros_like(x)
            return J, np.maximum(g, -1e6)  # clip for stability

        res = minimize(
            obj_grad, a, jac=True, method='L-BFGS-B',
            bounds=[(0.0, 1000.0)] * n,
            options={
                'maxiter': 500,
                'maxfun': 2000,
                'ftol': 1e-12,
                'gtol': 1e-8,
                'maxls': 50,
                'disp': False
            }
        )

        if res.x is not None and np.isfinite(res.fun):
            a_new = np.maximum(res.x, 0.0)
            if a_new.sum() > 0.01:
                a = a_new
                local_J = compute_J(a)
                if local_J < best_J:
                    best_J = local_J
                    best_a = a.copy()

            # Verify periodically
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"    L-BFGS phase {phase} (tau={tau:.3f}): verified J={val:.6f}")

    # Final check
    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def peak_reduction(a, time_budget):
    """Targeted peak reduction via coordinate search."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    if S < 0.01:
        return a
    a = a * (n / S)  # normalize mean to 1

    deadline = time.monotonic() + time_budget
    best_J = compute_J(a)
    best_a = a.copy()
    stall = 0

    while time.monotonic() < deadline:
        c = convolve_fft(a)
        M = c.max()
        k_star = int(np.argmax(c))

        # Find top contributing pairs for the peak lag
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_max < i_min:
            break
        indices = np.arange(i_min, i_max + 1)
        products = a[indices] * a[k_star - indices]
        top_k = min(10, len(products))
        top_idx = indices[np.argsort(products)[-top_k:]]

        improved = False
        for i in top_idx:
            j = k_star - i
            if j < 0 or j >= n:
                continue
            # Try reducing a[i]
            for factor in [0.95, 0.9, 0.85, 0.8]:
                a_trial = a.copy()
                a_trial[i] *= factor
                c_trial = convolve_fft(a_trial)
                if c_trial.max() < M * (1 - 1e-10):
                    a = a_trial
                    M = c_trial.max()
                    improved = True
                    break
            if improved:
                break
            # Try reducing a[j] (if different)
            if j != i:
                for factor in [0.95, 0.9, 0.85, 0.8]:
                    a_trial = a.copy()
                    a_trial[j] *= factor
                    c_trial = convolve_fft(a_trial)
                    if c_trial.max() < M * (1 - 1e-10):
                        a = a_trial
                        M = c_trial.max()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 20:
            break

    return best_a, best_J


def coordinated_hill_climbing(a_init, time_budget, seed):
    """Adaptive block perturbation local search."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + time_budget
    a = a_init.copy()
    S = a.sum()
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J(a)
    step_scale = 0.03
    stall = 0

    while time.monotonic() < deadline:
        # Random block perturbation
        width = int(rng.integers(max(2, n // 300), max(3, n // 10)))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)

        cand = a.copy()
        noise = rng.normal(0.0, step_scale, end - start)
        cand[start:end] *= np.exp(noise)
        cand = np.maximum(cand, 0.0)

        # Sometimes also nudge a specific peak contributor
        if rng.random() < 0.3:
            c = convolve_fft(cand)
            k_s = int(np.argmax(c))
            i_min = max(0, k_s - n + 1)
            i_max = min(n - 1, k_s)
            if i_max >= i_min:
                idx = int(rng.integers(i_min, i_max + 1))
                cand[idx] *= np.exp(rng.normal(-0.1, 0.1))
                cand = np.maximum(cand, 0.0)

        cS = cand.sum()
        if cS < 0.01:
            stall += 1
            continue

        cJ = compute_J(cand)
        if cJ < best_J - 1e-12:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            step_scale = max(0.002, step_scale * 0.97)
            stall = 0
        else:
            step_scale = min(0.15, step_scale * 1.03)
            stall += 1

        if stall > 500:
            # Larger random restart perturbation
            w = max(1, n // 20)
            s = int(rng.integers(0, n - w))
            a[s:s + w] *= np.exp(rng.normal(0, 0.5, w))
            a = np.maximum(a, 0.0)
            a = a * (n / a.sum())
            stall = 0
            step_scale = 0.05

    return best_a, best_J
