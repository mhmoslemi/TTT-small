    Maximize sum(g) subject to g >= 0 and (p*g)[k] <= M for all k.
    Returns the optimal g (or None if infeasible).
    """

    # This is a (2N-1) x N matrix

    # For each k (row), non-zero entries are at j where 0 <= k-j < N

    cols = []

        j_max = min(k, N - 1)

            cols.append(j)

```python
def solve_lp_direction(p, M, max_time=5.0):
    """Solve the LP: max sum(g) s.t. g >= 0, (p*g)[k] <= M for all k."""
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    import time
    
    start = time.monotonic()
    n = len(p)
    
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = p[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(p[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                  method='highs', options={'time_limit': max_time})
    
    if res.status == 0 and time.monotonic() - start < max_time:
        return res.x
    return None
```

                zt = z + step * direction
                zt = np.clip(zt, -30.0, 30.0)
                pt = probs_from_z(zt)
                st = score_p(pt)
                if st < score - 1e-13:
                    z = zt
                    p = pt
                    score = st
                    update_best(z, score)
                    accepted = True
                    break
            if not accepted:
                for step in (step_scale * 2.0, step_scale * 1.5):
                    zt = z + step * direction
                    zt = np.clip(zt, -30.0, 30.0)
                    pt = probs_from_z(zt)
                    st = score_p(pt)
                    if st < score - 1e-13:
                        z = zt
                        p = pt
                        score = st
                        update_best(z, score)
                        accepted = True
                        break

        if not accepted:
            best_local = score
            best_local_z = z.copy()
            for _ in range(5):
                zt = z + rng.normal(0.0, 0.08, N - 1)
                zt = np.clip(zt, -30.0, 30.0)
                pt = probs_from_z(zt)
                st = score_p(pt)
                if st < best_local - 1e-13:
                    best_local = st
                    best_local_z = zt.copy()
            if best_local < score - 1e-13:
                z = best_local_z
                p = probs_from_z(z)
                score = best_local
                update_best(z, score)
                accepted = True

        if accepted:
            fails = 0
            step_scale = min(0.5, step_scale * 1.15)
        else:
            fails += 1
            step_scale = max(0.02, step_scale * 0.85)
            if fails > 200:
                break

    return z


def make_initializations(rng):
    cands = []
    x = (np.arange(N) + 0.5) / float(N)

    # Uniform
    cands.append(np.zeros(N - 1))

    # Symmetric shapes
    for a in (0.5, 1.0, 1.5, 2.0, 3.0):
        f = np.cos(np.pi * (x - 0.5) * a) ** 2 + 1e-6
        p = f / f.sum()
        cands.append(z_from_p(p))

    # Cosine-based
    for freq in (1, 2, 3, 4):
        for amp in (0.3, 0.5, 0.7):
            f = 1.0 + amp * np.cos(2 * np.pi * freq * (x - 0.5))
            f = np.maximum(f, 1e-10)
            p = f / f.sum()
            cands.append(z_from_p(p))

    # Exponential
    for beta in (0.5, 1.0, 2.0, 3.0, 5.0):
        f = np.exp(-beta * np.abs(x - 0.5) * 2)
        p = f / f.sum()
        cands.append(z_from_p(p))

    # Logistic
    for beta in (2.0, 4.0, 6.0, 10.0):
        f = 1.0 / (1.0 + np.exp(-beta * (x - 0.5)))
        p = f / f.sum()
        cands.append(z_from_p(p))

    # Power
    for a in (0.5, 1.0, 2.0, 3.0, 5.0):
        f = np.power(np.minimum(x, 1 - x) + 1e-10, a)
        p = f / f.sum()
        cands.append(z_from_p(p))

    # Polynomial (symmetric)
    for deg in (2, 4, 6):
        t = x - 0.5
        for a in (-1.0, -0.5, 0.5, 1.0):
            f = 1.0 + a * (2 * t) ** deg
            f = np.maximum(f, 1e-10)
            p = f / f.sum()
            cands.append(z_from_p(p))

    # Random smooth
    for _ in range(30):
        t = x - 0.5
        f = np.ones(N)
        for k in range(1, 8):
            amp = rng.normal(0, 1.0 / (k + 0.5))
            ph = rng.uniform(0, 2 * np.pi)
            f += amp * np.cos(2 * np.pi * k * t + ph)
        f = np.maximum(f, 1e-10)
        p = f / f.sum()
        cands.append(z_from_p(p))

    return cands


def try_base_sequence():
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size == 0:
            return None
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.maximum(base, 0.0)
        s = float(np.sum(base))
        if s < 0.01:
            return None
        if base.size != N:
            old = np.linspace(0, 1, base.size)
            new = np.linspace(0, 1, N)
            base = np.interp(new, old, base)
            s = float(np.sum(base))
        base = base / s
        sc = score_p(base)
        update_best(base, sc)
        return z_from_p(base)
    except Exception:
        return None


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_p, _best_z, _deadline

    t_start = time.monotonic()
    budget = float(budget_s)
    t_end = t_start + budget - 3.0

    _best_score = float('inf')
    _best_p = None
    _best_z = None
    _deadline = t_end

    rng = np.random.default_rng(seed)

    # Try base sequence
    base_z = try_base_sequence()
    if base_z is not None:
        print(f"Base sequence score: {_best_score:.8f}")

    # Generate and score initializations
    cands = make_initializations(rng)
    if base_z is not None:
        cands.insert(0, base_z)

    scores = np.empty(len(cands))
    for i, z in enumerate(cands):
        p = probs_from_z(z)
        scores[i] = score_p(p)
        update_best(p, scores[i])

    order = np.argsort(scores)
    print(f"Best init score: {scores[order[0]]:.8f} (from {len(cands)} candidates)")
    print(f"Top 5 scores: {scores[order[:5]]}")

    # Phase 1: L-BFGS-B on top candidates
    n_l_bfgs = min(12, len(cands))
    main_deadline = t_start + budget - max(120.0, 0.15 * budget)

    for i in range(n_l_bfgs):
        if time.monotonic() > main_deadline - 10.0:
            break
        z0 = cands[order[i]]
        rem = main_deadline - time.monotonic()
        if rem < 15.0:
            break

        alpha = 5.0e6 if _best_score < 1.55 else 2.0e6
        if _best_score < 1.52:
            alpha = 1.0e7
        max_fun = int(min(600, max(100, rem / 0.008)))
        run_l_bfgs(z0, alpha, max_fun, main_deadline)
        if i % 3 == 0:
            print(f"L-BFGS run {i}: best={_best_score:.8f} elapsed={time.monotonic()-t_start:.0f}s")

    print(f"After L-BFGS: best={_best_score:.8f}")

    # Phase 2: LP-based Frank-Wolfe
    if _best_p is not None and time.monotonic() < t_end - 80.0:
        fw_deadline = min(t_end - 60.0, time.monotonic() + 250.0)
        print(f"Starting FW polish at score={_best_score:.8f}")
        p_fw = fw_polish(_best_p.copy(), fw_deadline, max_iters=40)
        if _best_p is not None:
            _best_z = z_from_p(_best_p)

    # Phase 3: Subgradient polishing
    if _best_z is not None and time.monotonic() < t_end - 15.0:
        sg_deadline = t_end - 5.0
        print(f"Starting subgradient polish at score={_best_score:.8f}")
        _best_z = subgradient_polish(_best_z, sg_deadline, rng)

    # Phase 4: One more L-BFGS pass if time remains
    if _best_z is not None and time.monotonic() < t_end - 30.0:
        rem = t_end - 5.0 - time.monotonic()
        if rem > 30.0:
            z0 = _best_z + rng.normal(0, 0.02, N - 1)
            max_fun = int(min(400, max(80, rem / 0.01)))
            run_l_bfgs(z0, 1.0e7, max_fun, t_end - 5.0)

    # Final subgradient
    if _best_z is not None and time.monotonic() < t_end - 5.0:
        _best_z = subgradient_polish(_best_z, t_end - 2.0, rng)

    # Build final sequence
    if _best_z is None:
        z = np.zeros(N - 1)
    else:
        z = np.clip(_best_z, -30.0, 30.0)

    p = probs_from_z(z)
    p = np.clip(p, 0.0, 1000.0)
    seq = [float(v) for v in p]

    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')

    if not np.isfinite(final_score) or final_score > 2.00001:
        seq = [1.0 / float(N)] * N
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            final_score = 2.0

    # Safety: check base
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            bs = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(bs) and bs < final_score:
                seq = [float(v) for v in base]
                final_score = bs
    except Exception:
        pass

    print(f"FINAL score={final_score:.8f} length={len(seq)} time={time.monotonic()-t_start:.1f}s")
    return seq
