"""
Step function coefficient optimization using multi-method local search.
Combines: (1) smooth-max subgradient with adaptive step size,
(2) LP direction finding, (3) structured mutations,
(4) multi-resolution refinement, and (5) population diversity.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    HAS_LINPROG = True
except Exception:
    HAS_LINPROG = False


def normalize_nonneg(a, target_sum=1.0):
    """Normalize to non-negative with given sum."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-300:
        return np.full(a.size, target_sum / max(1, a.size))
    return a * (target_sum / s)


def proj_simplex(v, target_sum=1.0):
    """Exact projection onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = v.size
    if n == 0:
        return v
    # If already feasible
    if v.min() >= -1e-15 and abs(v.sum() - target_sum) < 1e-10:
        return np.clip(v, 0, None)
    
    # Standard algorithm (parrot et al.)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.zeros(n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute 2*n*max(convolve(a,a)) for sum-1 non-negative a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max())


def get_subgradient(a, c, top_k=32, power=6):
    """Compute weighted subgradient of max(c) at a."""
    n = a.size
    k = min(top_k, c.size)
    idx = np.argpartition(c, -k)[-k:].astype(np.int64)
    w = c[idx] ** power
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    
    grad = np.zeros(n)
    for ki, wi in zip(idx, w):
        ki = int(ki)
        jlo = max(0, ki - n + 1)
        jhi = min(n, ki + 1)
        if jhi > jlo:
            jj = jlo + np.arange(jhi - jlo)
            grad[jj] += wi * a[ki - jj]
    return 2.0 * grad


def subgradient_step(a, c, eta, top_k=32, power=6):
    """Take one subgradient step with exact simplex projection."""
    grad = get_subgradient(a, c, top_k, power)
    v = a - eta * grad
    return proj_simplex(v)


def solve_lp_direction(a, c, n_constraints=1023):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= c[k], g >= 0."""
    if not HAS_LINPROG:
        return None
    n = a.size
    if n > 2048:
        return None
    
    M = c.max()
    if M < 1e-300:
        return None
    
    # Build constraint matrix: (a*g)[k] = sum_i a[i]*g[k-i] <= c[k]
    # For each k, the constraint is: g * A_k <= c[k]
    # where (A_k)_i = a[k-i] if 0<=k-i<n else 0
    nrows = min(n_constraints, c.size)
    # Use all constraints for smaller n, top constraints for larger n
    if nrows >= c.size:
        rows = np.arange(c.size)
    else:
        rows = np.argpartition(c, -nrows)[-nrows:]
    
    A = np.zeros((rows.size, n))
    b = np.zeros(rows.size)
    for r, k in enumerate(rows):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            A[r, jlo:jhi] = a[k - np.arange(jhi - jlo)]
        b[r] = c[int(k)]
    
    # Scale constraints
    A = A / M
    b = b / M
    
    cvec = -np.ones(n)
    try:
        res = linprog(cvec, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    except Exception:
        return None
    
    if res.status == 0 and res.x is not None:
        g = np.clip(res.x, 0, None)
        if g.sum() > 1.0 + 1e-6:
            return g
    return None


def mutate_sinusoid(a, rng, amplitude=0.1):
    """Add a small sinusoidal perturbation."""
    n = a.size
    freq = rng.integers(1, max(2, n // 16))
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    pert = amplitude * np.sin(2 * np.pi * freq * x / n + phase)
    return proj_simplex(a * (1 + pert))


def mutate_ramp(a, rng, amplitude=0.08):
    """Add a small ramp perturbation."""
    n = a.size
    slope = rng.uniform(-amplitude, amplitude)
    intercept = rng.uniform(-amplitude, amplitude)
    x = np.linspace(0, 1, n)
    pert = slope * x + intercept
    return proj_simplex(a * (1 + pert))


def mutate_block(a, rng, amplitude=0.15):
    """Perturb a random block."""
    n = a.size
    width = max(2, int(rng.integers(n // 16, n // 4)))
    start = int(rng.integers(0, max(1, n - width)))
    scale = np.exp(rng.normal(0, amplitude, width))
    a_new = a.copy()
    a_new[start:start + width] *= scale
    return proj_simplex(a_new)


def mutate_random(a, rng, scale=0.1):
    """Random multiplicative noise."""
    n = a.size
    noise = np.exp(rng.normal(0, scale, n))
    return proj_simplex(a * noise)


def crossover(a1, a2, rng, alpha=None):
    """Convex combination crossover."""
    if alpha is None:
        alpha = rng.uniform(0.2, 0.8)

def resize_sequence(a, n_target):
    """Resize sequence to n_target length while preserving sum."""
    a = np.asarray(a, dtype=np.float64)
    n_src = a.size
    if n_src == n_target:
        return proj_simplex(a)
    
    if n_target > n_src:
        # Upsample by interpolation
        x_src = np.linspace(0, 1, n_src)
        x_dst = np.linspace(0, 1, n_target)
        a_up = np.interp(x_dst, x_src, a)
    else:
        # Downsample by block averaging
        a_down = np.zeros(n_target)
        for i in range(n_target):
            lo = int(i * n_src / n_target)
            hi = int((i + 1) * n_src / n_target)
            if hi > lo:
                a_down[i] = a[lo:hi].sum()
    
    return proj_simplex(a_down)
