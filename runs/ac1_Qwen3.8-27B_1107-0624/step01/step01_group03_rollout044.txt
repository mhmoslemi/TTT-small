import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def build_sparse_conv_matrix(s):
    """Build sparse matrix for (s*b)[k] = sum_j s[k-j]*b[j]."""
    n = len(s)
    rows, cols, data = [], [], []
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * n)
        data.extend(list(s))
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def solve_lp_direction(s, M, K=50, time_limit=10.0):
    """Solve LP: max sum(b) s.t. (s*b)[k] <= M for k in top-K lags, b >= 0."""
    n = len(s)
    conv = np.convolve(s, s)
    # Get top K lag indices
    top_k = np.argsort(conv)[-K:]
    
    A = build_sparse_conv_matrix(s)
    # Only use top K constraints
    A_sub = A[top_k]
    b_ub = np.full(len(top_k), M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lp_guided_search(s, rng, max_iters=200, deadline=None, K=50):
    best = s.copy()
    best_sc = fast_score(s)
    no_imp = 0
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        M = conv.max()
        
        b = solve_lp_direction(s, M, K=K, time_limit=10.0)
        if b is None or b.sum() < 1e-14:
            break
        
        b = b / b.sum()
        
        # Line search
        best_t = 0.0
        best_ts = best_sc
        t_vals = np.linspace(0.01, 1.0, 20)
        for t in t_vals:
            cand = (1-t)*s + t*b
            cand = np.maximum(cand, 0)
            ts = fast_score(cand)
            if ts < best_ts - 1e-14:
                best_ts = ts
                best_t = t
        
        if best_t == 0:
            no_imp += 1
            if no_imp > 10:
                break
            # Perturb
            s = s + 0.01 * rng.standard_normal(n)
            s = np.maximum(s, 0)
            s = s / s.sum()
            continue
        
        s = (1-best_t)*s + best_t*b
        s = np.maximum(s, 0)
        s = s / s.sum()
        no_imp = 0
        
        sc = fast_score(s)
        if sc < best_sc:
            best_sc = sc
            best = s.copy()
    
    return best, best_sc
