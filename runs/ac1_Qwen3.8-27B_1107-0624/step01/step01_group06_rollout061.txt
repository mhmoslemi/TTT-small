"""
Optimized step function coefficient search using multi-strategy gradient descent
with FFT-accelerated log-sum-exp smoothing, diverse initializations including
frequency-domain design, and coordinate descent refinement.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_objective_and_grad(a, tau):
    """Compute smoothed J and gradient."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(np.clip(shifted, -50, 50))
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    # Gradient of Tmax with respect to a[i]:
    # dTmax/da[i] = sum_k p[k] * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] (for valid k-i)
    # So dTmax/da[i] = sum_k p[k] * a[k-i] = (p * a_flipped)[i] ... 
    # Actually, this is the cross-correlation of p and a
    
    # (p * a)[i] in convolution sense = sum_k p[k] * a[i-k]
    # But we want sum_k p[k] * a[k-i] which is the cross-correlation
    
    # Using FFT: cross_corr(p, a)[i] = sum_k p[k+i] * a[k] ... 
    # Let me think more carefully.
    
    # c[k] = sum_j a[j] * a[k-j]
    # dc[k]/da[i] = a[k-i] + a[i-k] if i != k-i, else a[k-i]
    # Wait, for the full convolution: c[k] = sum_j a[j]*a[k-j] where j ranges so that both j and k-j are in [0,n-1]
    # dc[k]/da[i] = a[k-i] if 0 <= k-i < n, plus a[i-k] if 0 <= i-k < n (but only if i != k-i, otherwise just a[k-i])
    # Actually for the full convolution, each pair (j, k-j) appears once, so:
    # dc[k]/da[i] = a[k-i] if 0 <= k-i <= n-1 (this accounts for the case where i is the first index)
    # But we also need to account for when i is the second index: a[i-k] if 0 <= i-k <= n-1
    # Wait, in the convolution c[k] = sum_j a[j]*a[k-j], if we differentiate w.r.t. a[i]:
    # The term a[i]*a[k-i] appears when j=i, giving a[k-i]
    # The term a[j]*a[i] appears when k-j=i, i.e., j=k-i, giving a[k-i]
    # But if i = k-i (i.e., k=2i), we only get a[i] (not 2*a[i])
    # So dc[k]/da[i] = a[k-i] + [i != k-i] * a[k-i] ... no that's wrong
    # Actually: c[k] = sum_j a[j]*a[k-j]. When j=i, we get a[i]*a[k-i]. When j=k-i, we get a[k-i]*a[i]. These are the same term if and only if i = k-i.
    # In a convolution, each pair is counted once. So dc[k]/da[i] = a[k-i] if 0<=k-i<n, and that's it (the convolution sums over j, and the term involving a[i] is when j=i, giving a[k-i], OR when k-j=i, i.e., j=k-i, giving a[k-i]).
    # Wait, I'm overcomplicating this. Let's just use:
    # dc[k]/da[i] = a[(k-i) mod something] ... 
    # 
    # Actually the simplest way: c = convolve(a, a), so dc[k]/da[i] is the (i-th) element of the "response" which is just a shifted version of a.
    # Specifically, dc/da[i] = convolve(delta_i, a) + convolve(a, delta_i) = 2*a shifted, except at the diagonal.
    # 
    # For the purpose of the gradient, we can compute:
    # dTmax/da[i] = sum_k p[k] * dc[k]/da[i]
    # = sum_k p[k] * (a[k-i] if 0<=k-i<n else 0) + sum_k p[k] * (a[i-k] if 0<=i-k<n else 0) [but avoid double counting when k-i = i-k]
    # 
    # Hmm, this is getting complicated. Let me use a different approach.
    # 
    # Note that c[k] = sum_j a[j]*a[k-j]. This is a quadratic form: c = D(a) where D is the "convolution matrix" applied to a.
    # Actually, c[k] = a^T C_k a where C_k is a permuted identity... 
    # 
    # The simplest approach: use the fact that for the full convolution,
    # sum_k p[k]*c[k] = sum_k p[k] * sum_j a[j]*a[k-j] = sum_{j,l} a[j]*a[l]*p[j+l]
    # This is a^T B a where B[j,l] = p[j+l] (a Hankel matrix)
    # 
    # d/da[i] (a^T B a) = 2*(Ba)[i] (if B is symmetric, which it is since B[j,l] = p[j+l] = p[l+j] = B[l,j])
    # So dTmax/da[i] = 2*(B_p a)[i]
    
    # B_p a [i] = sum_l p[i+l] * a[l]
    # This is a convolution! (B_p a) = convolve(a, p_reversed) ... 
    # Actually: (B_p a)[i] = sum_l a[l] * p[i+l]
    # Let's define q[m] = p[m] for m in [0, 2n-2], 0 otherwise
    # Then (B_p a)[i] = sum_l a[l] * q[i+l]
    # This is the cross-correlation of a and q: (a ⋆ q)[i] = sum_l a[l] * q[i+l]
    
    # Using FFT: cross_corr(a, q)[i] = sum_l a[l] * q[i+l]
    # = IFFT(FFT(a) * conj(FFT(q)))... let me verify
    # 
    # Actually, let's define it differently. Let r[m] = q[m] for m in range, 0 otherwise.
    # (a ⋆ r)[i] = sum_l a[l] * r[l+i]
    # In terms of convolution: if we define a_flipped[l] = a[n-1-l], then
    # (a_flipped * r)[n-1+i] = sum_l a_flipped[l] * r[n-1+i-l] = sum_l a[n-1-l] * r[n-1+i-l]
    # Let m = n-1-l: = sum_m a[m] * r[i+m] = (a ⋆ r)[i] ✓
    
    # So (B_p a)[i] = (a_flipped * r)[n-1+i]
    # where r = p (padded to appropriate length)
    
    # Alternatively, we can compute this directly with FFT:
    # Let A = FFT(a, size), P = FFT(p_padded, size)
    # Then IFFT(A * conj(P)) gives the cross-correlation... 
    
    # Actually, the simplest: just do the FFT multiplication
    # (B_p a)[i] = sum_{l=0}^{n-1} a[l] * p[i+l] (where p is defined on [0, 2n-2])
    # = (a * p_padded)[i] in the "cross-correlation" sense
    
    # Let's use: if we pad p to length size (FFT size), and compute:
    # result = IFFT(FFT(a_padded) * FFT(p_padded_shifted))
    # Hmm, this is getting confusing. Let me just use a direct formula.
    
    # Actually, I realize the cleanest way:
    # sum_k p[k]*c[k] = sum_k p[k]*sum_j a[j]*a[k-j]
    # Let's change variables: let m = j, n' = k-j, so k = m+n'
    # = sum_{m,n'} a[m]*a[n']*p[m+n']
    # = a^T B a where B[m,n'] = p[m+n']
    
    # Gradient: 2*B*a (since B is symmetric)
    # (Ba)[m] = sum_{n'} p[m+n'] * a[n']
    
    # This is: for each m, sum over n' of p[m+n']*a[n']
    # = (a convolved with p_inverted)[m] where p_inverted[x] = p[-x] ... 
    
    # Let me just compute it with a second FFT:
    # Define q of length size: q[k] = p[k] for k in [0, 2n-2], 0 otherwise
    # Then (Ba)[m] = sum_{n'=0}^{n-1} a[n'] * p[m+n']
    # This is the cross-correlation of a (length n) and p (length 2n-1) evaluated at lag m
    
    # Using FFT: cross_corr(a, p)[m] = sum_n a[n] * p[m+n]
    # = IFFT(FFT(a_padded * size) * conj(FFT(p_padded * size))) evaluated at appropriate indices
    # 
    # Actually, standard cross-correlation: (a ⋆ p)[m] = sum_n a[n] * conj(p[n-m])
    # But we want sum_n a[n] * p[m+n] which is different.
    # 
    # Let me define: r[k] = p[k] for k in [0, 2n-2]
    # We want: t[m] = sum_{n=0}^{n-1} a[n] * r[m+n] for m = 0, ..., n-1
    # 
    # Note: t[m] = (a * r_rev)[m + (2n-2)] where r_rev[k] = r[2n-2-k]
    # Because (a * r_rev)[m + 2n-2] = sum_n a[n] * r_rev[m+2n-2-n] = sum_n a[n] * r[n-2+m+2n-2-2n+2] ... 
    # This is getting too confusing. Let me just use a practical approach.
    
    # PRACTICAL APPROACH: Use the identity that the gradient of sum_k p[k]*c[k] 
    # where c = convolve(a,a) can be computed as:
    # grad[i] = 2 * convolve(p, a)[i] ... let me verify with a small example
    
    # For n=3, a=[a0,a1,a2], p=[p0,p1,p2,p3,p4]
    # c[0]=a0^2, c[1]=2*a0*a1, c[2]=2*a0*a2+a1^2, c[3]=2*a1*a2, c[4]=a2^2
    # sum p[k]*c[k] = p0*a0^2 + 2*p1*a0*a1 + (2*p2*a0*a2+p2*a1^2) + 2*p3*a1*a2 + p4*a2^2
    # d/da0 = 2*p0*a0 + 2*p1*a1 + 2*p2*a2
    # d/da1 = 2*p1*a0 + 2*p2*a1 + 2*p3*a2
    # d/da2 = 2*p2*a0 + 2*p3*a1 + 2*p4*a2
    # 
    # So grad[i] = 2 * sum_j p[i+j] * a[j]
    # 
    # This is: grad = 2 * (Hankel multiplication)
    # Where the Hankel matrix H[i,j] = p[i+j]
    # 
    # To compute this with FFT: H*a is the cross-correlation of a with p (in a specific sense)
    # Specifically: (Ha)[i] = sum_j a[j]*p[i+j]
    # 
    # This equals: if we define the "reversed p" as p_rev[k] = p[2n-2-k], then
    # (Ha)[i] = sum_j a[j] * p[i+j]
    # Let's use the convolution: (a * p_rev)[i + (n-1)] = sum_j a[j] * p_rev[i+(n-1)-j]
    # = sum_j a[j] * p[2n-2-i-(n-1)+j] = sum_j a[j] * p[n-1-i+j]
    # Hmm, that gives p[n-1-i+j] not p[i+j].
    # 
    # Let me try differently. Define q[k] = p[k] (length 2n-1, padded to FFT size)
    # We want t[i] = sum_{j=0}^{n-1} a[j] * q[i+j]
    # 
    # This is the same as: t = cross_correlation of a and q, where
    # (a ⋆ q)[i] = sum_j a[j] * q[j+i] (this is the "sliding dot product")
    # 
    # In terms of standard convolution: (a ⋆ q)[i] = (a_rev * q)[(n-1) + i] where a_rev[j] = a[n-1-j]
    # Because (a_rev * q)[(n-1)+i] = sum_j a_rev[j] * q[(n-1)+i-j] = sum_j a[n-1-j] * q[(n-1)+i-j]
    # Let m = n-1-j: = sum_m a[m] * q[i+m] ✓
    
    # Great! So:
    # Ha = IFFT(FFT(a_rev, size) * FFT(q, size))[(n-1):2n-1]
    # where q is p padded to length size
    
    # Actually, let me just use the property that:
    # cross_correlation(a, q)[i] = IFFT(conj(FFT(a)) * FFT(q)) (with appropriate indexing)
    # or equivalently = IFFT(FFT(a_rev) * FFT(q)) with a shift
    
    # For simplicity, let me just compute it directly using the FFT approach:
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, size)
    # p is already length 2n-1, pad to size
    p_padded = np.zeros(size)
    p_padded[:2*n-1] = p
    fp = np.fft.rfft(p_padded)
    # cross-correlation result
    cc = np.fft.irfft(fa_rev * fp, size)
    # t[i] = cc[n-1+i] for i = 0, ..., n-1
    Ha = cc[n-1:n-1+n]
    
    dTmax_da = 2.0 * Ha
    
    # Now J = 2n * Tmax / S^2
    # dJ/da[i] = 2n/S^2 * dTmax/da[i] - 4n*Tmax/S^3
    grad = (2.0 * n / (S * S)) * dTmax_da - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-x * x / (2.0 * sigma * sigma))


def make_raised_cosine(n, beta=0.5):
    """Raised cosine window."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    t = (x - mid) / (mid + 0.5)
    return 0.5 * (1 + np.cos(np.pi * t / beta)) ** 2


def make_blackman(n):
    """Blackman window."""
    x = np.arange(n)
    N = n - 1
    return 0.42 - 0.5 * np.cos(2 * np.pi * x / N) + 0.08 * np.cos(4 * np.pi * x / N)


def make_freq_design(n, rng, num_iters=50):
    """Frequency-domain design: find non-negative a with nearly flat |FFT(a)|."""
    a = np.ones(n)
    for _ in range(num_iters):
        fa = np.fft.rfft(a)
        # Target: constant magnitude spectrum
        target_mag = np.ones(len(fa))
        # Update in frequency domain
        new_fa = target_mag * np.exp(1j * np.angle(fa))
        a = np.fft.irfft(new_fa, n)
        a = np.maximum(a, 0)
        # Renormalize
        if np.sum(a) > 0:
            a = a / np.sum(a) * n
    return a


def make_waterfill_init(n, num_iters=100):
    """Iterative water-filling: start with ones, iteratively flatten convolution."""
    a = np.ones(n)
    for _ in range(num_iters):
        c = convolve_fft(a)
        c_max = np.max(c)
        c_min = np.max(0, np.min(c))
        # Reduce elements that contribute to the peak
        # Gradient of max(c) w.r.t. a
        peak_indices = np.where(c > 0.9 * c_max)[0]
        # For each peak index k, the elements a[i] with a[k-i] also large contribute
        # Simple heuristic: reduce a proportionally to its "contribution" to peaks
        contrib = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib[i] += a[j]
        # Scale down high-contribution elements
        if np.max(contrib) > 0:
            scale = 1.0 - 0.1 * contrib / np.max(contrib)
            a = a * np.maximum(0.5, scale)
        a = np.maximum(a, 1e-6)
    return a


def run_optimization(a_init, budget_s, seed, tau_start=10.0, tau_end=0.01, lr=0.01):
    """Run Adam gradient descent with adaptive tau and line search."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    start_time = time.monotonic()
    stall_count = 0
    stall_threshold = best_J
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        current_lr = lr / (1.0 + 0.1 * t / 500.0)
        
        J, grad = smooth_objective_and_grad(a, tau)
        
        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        
        # Check improvement
        if t % 5000 == 0:
            local_J = compute_J(a_new)
            if local_J < best_J:
                best_J = local_J
                best_a = a_new.copy()
                stall_count = 0
                stall_threshold = local_J
            else:
                stall_count += 1
                if stall_count > 3:
                    # Perturbation
                    cand = best_a.copy()
                    w = max(1, n // 100)
                    s = int(rng.integers(0, n - w))
                    cand[s:s+w] *= np.exp(rng.normal(0, 0.3, w))
                    cand = np.maximum(cand, 0)
                    cJ = compute_J(cand)
                    if cJ < best_J:
                        a = cand.copy()
                        best_J = cJ
                        best_a = cand.copy()
                    else:
                        a = best_a.copy()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    stall_count = 0
            
            a = a_new
        
        a = a_new
    
    return best_a, best_J


def run_coordinate_descent(a_init, budget_s, seed):
    """Coordinate descent on sensitive indices."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    best_a = a.copy()
    best_J = compute_J(a)
    
    while time.monotonic() < deadline:
        c = convolve_fft(a)
        c_max = np.max(c)
        S = np.sum(a)
        
        # Find sensitive indices: those that contribute to peak convolution values
        peak_threshold = 0.95 * c_max
        peak_indices = np.where(c >= peak_threshold)[0]
        
        # Compute sensitivity for each a[i]
        sensitivity = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    sensitivity[i] += a[j]
        
        # Sort by sensitivity (most sensitive first)
        order = np.argsort(-sensitivity)
        
        improved = False
        for idx in order[:min(50, n)]:
            if time.monotonic() >= deadline:
                break
            # Try scaling a[idx] by different factors
            best_scale = 1.0
            best_local_J = compute_J(a)
            for scale in [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.5]:
                a_old = a[idx]
                a[idx] = a_old * scale
                if a[idx] < 0:
                    a[idx] = 0
                local_J = compute_J(a)
                if local_J < best_local_J:
                    best_local_J = local_J
                    best_scale = scale
                a[idx] = a_old
            
            a[idx] *= best_scale
            a[idx] = max(0, a[idx])
            
            if best_scale != 1.0:
                improved = True
        
        current_J = compute_J(a)
        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
        
        if not improved:
            # Try random perturbations
            for _ in range(10):
                idx = int(rng.integers(0, n))
                a_old = a[idx]
                a[idx] = a_old * np.exp(rng.normal(0, 0.2))
                a[idx] = max(0, a[idx])
                if compute_J(a) < best_J:
                    best_J = compute_J(a)
                    best_a = a.copy()
                else:
                    a[idx] = a_old
    
    return best_a, best_J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimizer combining:
    1. Diverse initializations (Gaussian, raised cosine, Blackman, freq-design, waterfill)
    2. FFT-accelerated gradient descent with log-sum-exp smoothing
    3. Coordinate descent refinement on sensitive indices
    4. Adaptive time allocation based on initial quality
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)
    
    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)
    
    n = parent.size
    print(f"n = {n}")
    
    # Initial score
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score: {best_global_score:.6f}")
    
    # Build diverse initializations
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("ones", np.ones(n)))
    inits.append(("gauss_narrow", make_gaussian(n, 0.15)))
    inits.append(("gauss_mid", make_gaussian(n, 0.25)))
    inits.append(("gauss_wide", make_gaussian(n, 0.35)))
    inits.append(("raised_cos", make_raised_cosine(n, 0.5)))
    inits.append(("blackman", make_blackman(n)))
    inits.append(("freq_design", make_freq_design(n, rng, 30)))
    inits.append(("waterfill", make_waterfill_init(n, 50)))
    inits.append(("rand1", np.exp(rng.normal(0, 0.4, n))))
    inits.append(("rand2", np.exp(rng.normal(0, 0.8, n))))
    
    # Score and sort
    scored_inits = []
    for name, arr in inits:
        arr = np.maximum(arr, 0)
        S = np.sum(arr)
        if S < 0.01:
            arr = np.ones(n)
            S = n
        J = compute_J(arr)
        scored_inits.append((name, J, arr))
        print(f"  init '{name}': J = {J:.6f}")
    
    scored_inits.sort(key=lambda x: x[1])
    
    # Time allocation
    remaining = total_deadline - time.monotonic()
    # Reserve 30s for final refinement
    search_budget = remaining - 30.0
    
    # Run optimization on top initializations
    # Top 4 get gradient descent, then coordinate descent on best
    results = [(name, J, arr.copy()) for name, J, arr in scored_inits[:8]]
    
    # Allocate time: top gets 40%, second 20%, third 15%, rest split
    if search_budget > 0:
        budgets = []
        for i in range(min(6, len(results))):
            if i == 0:
                budgets.append(search_budget * 0.35)
            elif i == 1:
                budgets.append(search_budget * 0.20)
            elif i == 2:

But for a monotone sequence, a[i] < a[n-1-i] for i < n/2, so the products a[i]*a[n-1-i] are all "small * large" which is moderate.

What if we use a sequence that's 0 for the first half and 1 for the second half? Then a[i]*a[n-1-i] = 0 for all i (since one is always 0 when the other is 1). So c[n-1] = 0!

But then c[k] for k in the "overlap" region would be large. Specifically, for the step function a[i] = 0 for i < n/2, 1 for i >= n/2:
c[k] = number of i such that i >= n/2 and k-i >= n/2 = number of i in [n/2, k-n/2] (if k >= n)
For k = n: c[n] = n/2 (number of valid pairs)
S = n/2
J = 2n*(n/2)/(n/2)^2 = 4. Very bad!

So the step function is terrible. The ramp is much better because it spreads the values more evenly.

What about a sequence that's smoothly increasing but with a "kink"? Like a[i] = i/n for i < n/2, and a[i] = 1 for i >= n/2?

This would have c[n-1] = sum_{i=0}^{n/2-1} (i/n)*1 + sum_{i=n/2}^{n-1} 1*1
= (1/n) * sum_{i=0}^{n/2-1} i + n/2
= (1/n) * (n/2-1)(n/2)/2 + n/2
≈ (n/4)/2n + n/2 = n/8 + n/2 = 5n/8

S = sum_{i=0}^{n/2-1} i/n + n/2 ≈ n/4 + n/2 = 3n/4
J = 2n*(5n/8)/(3n/4)^2 = 2n*5n/8/(9n^2/16) = 10n^2/8 * 16/(9n^2) = 20/9 ≈ 2.22

Worse than the ramp! So the pure ramp is better than the ramp+step.
