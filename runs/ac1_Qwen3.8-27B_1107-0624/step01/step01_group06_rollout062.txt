"""
Optimized step function search using subgradient descent on the convolution peak,
combined with FFT-accelerated gradient descent (log-sum-exp smoothing) and local
coordinate search. Key improvement: linear ramp initialization which provides
a strong starting point (J ≈ 4/3) far below the previous 1.5158.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)


def smooth_J_and_grad(a, tau):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad


def subgradient_step(a, lr):
    """One subgradient step on max(conv(a,a)). Returns new a and the peak location."""
    n = len(a)
    c = convolve_fft(a)
    k_star = int(np.argmax(c))
    S = np.sum(a)
    
    # Subgradient of b[k*] w.r.t. a[i] is a[k*-i] (with bounds)
    # db[k*]/da[i] = a[k*-i] if 0 <= k*-i <= n-1, else 0
    s = np.zeros(n)
    # For each i, the contribution to b[k*] from a[i] is a[k*-i]
    # i ranges from max(0, k*-(n-1)) to min(k*, n-1)
    i_min = max(0, k_star - (n - 1))
    i_max = min(k_star, n - 1)
    for i in range(i_min, i_max + 1):
        s[i] = a[k_star - i]
    
    # Subgradient of J = 2n*max(b)/S^2:
    # dJ/da[i] = 2n/S^2 * s[i] - 4n*max(b)/S^3 * 1
    max_b = c[k_star]
    grad = (2.0 * n / (S * S)) * s - (4.0 * n * max_b / (S * S * S)) * np.ones(n)
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    
    return a_new, k_star, max_b


def run_subgradient(a_init, budget_s, seed, lr_start=0.1, lr_decay=0.999):
    """Subgradient descent on max convolution peak."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    
    best_a = a.copy()
    best_J = compute_J_fft(a)
    lr = lr_start
    iteration = 0
    
    while time.monotonic() < deadline:
        iteration += 1
        lr = lr_start * (lr_decay ** (iteration % 10000))
        
        a_new, k_star, max_b = subgradient_step(a, lr)
        
        # Renormalize to maintain sum = n
        S_new = np.sum(a_new)
        if S_new > 0.01:
            a_new = a_new * (n / S_new)
        
        J_new = compute_J_fft(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Accept with probability (simulated annealing)
            if rng.random() < 0.1:
                a = a_new.copy()
            # Don't update best
        
        if iteration % 10000 == 0:
            print(f"    subgrad iter={iteration} J={best_J:.6f} max_b={max_b:.4e}")
    
    # Final verification
    final_J = compute_J_fft(best_a)
    return best_a, min(best_J, final_J)


def run_gradient_descent(a_init, budget_s, seed, tau_start=10.0, tau_end=0.1, lr_start=0.01):
    """Adam gradient descent with adaptive tau."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J_fft(a)
    start_time = time.monotonic()
    stall_count = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start / (1.0 + 0.3 * t / 5000.0)

        J, grad = smooth_J_and_grad(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            tau = max(1.0, tau)
            continue

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        
        # Renormalize
        S_curr = np.sum(a)
        if S_curr > 0.01:
            a = a * (n / S_curr)

        if t % 5000 == 0:
            local_J = compute_J_fft(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 10:
                    a = best_a.copy()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    stall_count = 0

    final_local = compute_J_fft(best_a)
    return best_a, min(best_J, final_local)


def run_local_search(a_init, budget_s, seed):
    """Coordinate-wise local search for final refinement."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J_fft(a)
    
    # Precompute convolution
    c = convolve_fft(a)
    k_star = int(np.argmax(c))
    
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # Pick a random index
        i = int(rng.integers(0, n))
        
        # Try different multipliers
        best_mult = 1.0
        best_trial_J = np.inf
        
        for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
            cand = a.copy()
            cand[i] *= mult
            cand = np.maximum(cand, 0)
            
            # Quick evaluation using the precomputed convolution
            # b[k] changes only at positions where i or k-i is involved
            # b_new[k] = b[k] + (cand[i] - a[i]) * (sum of relevant a's)
            # For efficiency, just compute J directly if n is manageable
            S_trial = np.sum(cand)
            if S_trial < 0.01:
                continue
            c_trial = convolve_fft(cand)
            J_trial = 2.0 * n * np.max(c_trial) / (S_trial * S_trial)
            
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_mult = mult
        
        if best_mult != 1.0 and best_trial_J < best_J:
            a[i] *= best_mult
            a = np.maximum(a, 0)
            # Renormalize
            S_curr = np.sum(a)
            a = a * (n / S_curr)
            best_J = compute_J_fft(a)
            best_a = a.copy()
            c = convolve_fft(a)
            k_star = int(np.argmax(c))
        
        # Occasionally do a block perturbation
        if iteration % 100 == 0:
            width = int(rng.integers(max(1, n // 100), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width)))
            cand = a.copy()
            noise = rng.normal(0, 0.1, width)
            cand[start:start+width] *= np.exp(noise)
            cand = np.maximum(cand, 0)
            S_c = np.sum(cand)
            if S_c > 0.01:
                cand = cand * (n / S_c)
                cJ = compute_J_fft(cand)
                if cJ < best_J:
                    a = cand
                    best_a = a.copy()
                    best_J = cJ
                    c = convolve_fft(a)
    
    return best_a, best_J


def make_linear_ramp(n):
    """Linear ramp: a[i] = (i+1)/n. Theoretical J ≈ 4/3."""
    return (np.arange(n) + 1) / n


def make_quadratic_ramp(n):
    """Quadratic ramp: a[i] = ((i+1)/n)^2."""
    x = (np.arange(n) + 1) / n
    return x * x


def make_cubic_ramp(n):
    """Cubic ramp."""
    x = (np.arange(n) + 1) / n
    return x ** 3


def make_sqrt_ramp(n):
    """Square root ramp: a[i] = sqrt((i+1)/n)."""
    return np.sqrt((np.arange(n) + 1) / n)


def make_cosine_ramp(n):
    """Cosine ramp: high at start, low at end."""
    x = np.arange(n) / (n - 1)
    return np.cos(np.pi * x / 2) ** 2 + 0.01


def make_sine_window(n):
    """Sine window."""
    x = np.arange(n)
    return np.sin(np.pi * (x + 0.5) / n)


def make_power_ramp(n, alpha):
    """Power ramp: a[i] = ((i+1)/n)^alpha."""
    x = (np.arange(n) + 1) / n
    return x ** alpha


def run_spectral_flattening(a_init, budget_s, seed, gamma=0.5):
    """Spectral flattening: iteratively flatten the power spectrum of a.
    
    The idea: in the frequency domain, B = |A|^2. We want B to be as flat as

    """

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()

"""
Optimized step function search combining linear ramp initialization (J≈4/3),
subgradient descent on convolution peaks, FFT-accelerated log-sum-exp gradient
descent, spectral flattening, and coordinate-wise local search.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)


def smooth_J_and_grad(a, tau):
    """Compute smoothed objective and gradient using log-sum-exp.
    
    J = 2n * Tmax / S^2 where Tmax = tau*log(sum_k exp(c[k]/tau))
    """

    while size < 3 * n - 2:
        size <<= 1

    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

        return 1e15, np.zeros(n)

    Tmax = c_max + tau * np.log(sum_exp)

    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    # dTmax/da[i] = cross[i] (cross-correlation of p with a)
    # dJ/da[i] = 2n/S^2 * cross[i] - 4n*Tmax/S^3
    grad = (2.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad


def get_subgradient(a):
    """Compute subgradient of J(a) = 2n*max(b)/S^2 at the argmax peak.
    
    Returns (grad, k_star, max_b) where grad is the subgradient w.r.t. a.
    db[k*]/da[i] = 2*a[k*-i] if 0 <= k*-i <= n-1, else 0.
    """
    n = len(a)
    c = convolve_fft(a)
    k_star = int(np.argmax(c))
    max_b = c[k_star]
    S = np.sum(a)

    s = np.zeros(n)
    i_min = max(0, k_star - (n - 1))
    i_max = min(k_star, n - 1)
    # s[i] = 2*a[k_star - i] for i in [i_min, i_max]
    indices = np.arange(i_min, i_max + 1)
    s[indices] = 2.0 * a[k_star - indices]

    grad = (2.0 * n / (S * S)) * s - (4.0 * n * max_b / (S ** 3)) * np.ones(n)
    return grad, k_star, max_b


def run_subgradient(a_init, budget_s, seed, lr_start=0.02, lr_end=0.001):
    """Subgradient descent with Polyak averaging on max convolution peak."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0.01:
        a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J_fft(a)
    avg_a = a.copy()
    iteration = 0
    start_time = time.monotonic()

    while time.monotonic() < deadline:
        iteration += 1
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        lr = lr_start * (lr_end / lr_start) ** frac

        grad, k_star, max_b = get_subgradient(a)

        a = a - lr * grad
        a = np.maximum(a, 0.0)

        S_curr = np.sum(a)
        if S_curr > 0.01:
            a = a * (n / S_curr)

        # Polyak averaging
        avg_a = avg_a + (a - avg_a) / (iteration + 1)

        # Check objective periodically
        if iteration % 2000 == 0:
            J_check = compute_J_fft(a)
            if J_check < best_J:
                best_J = J_check
                best_a = a.copy()

            J_avg = compute_J_fft(avg_a)
            if J_avg < best_J:
                best_J = J_avg
                best_a = avg_a.copy()

            if iteration % 10000 == 0:
                print(f"    subgrad iter={iteration} J={best_J:.6f} lr={lr:.5f}")

    # Final check on both a and avg_a
    J_a = compute_J_fft(a)
    J_avg = compute_J_fft(avg_a)
    if J_a < best_J:
        best_J = J_a
        best_a = a.copy()
    if J_avg < best_J:
        best_J = J_avg
        best_a = avg_a.copy()

    return best_a, best_J


def run_gradient_descent(a_init, budget_s, seed, tau_start=10.0, tau_end=0.05, lr_start=0.01):
    """Adam gradient descent with adaptive tau using log-sum-exp smoothing."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0.01:
        a = a * (n / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J_fft(a)
    start_time = time.monotonic()
    stall_count = 0
    stall_threshold = best_J

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start / (1.0 + 0.5 * t / 5000.0)

        J, grad = smooth_J_and_grad(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            tau = max(1.0, tau)
            continue

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)

        S_curr = np.sum(a)
        if S_curr > 0.01:
            a = a * (n / S_curr)

        if t % 3000 == 0:
            local_J = compute_J_fft(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall_count = 0
                stall_threshold = local_J
            else:
                stall_count += 1
                if stall_count > 8:
                    # Perturbation to escape local minimum
                    cand = a.copy()
                    w = int(max(1, n // 100))
                    s = int(rng.integers(0, max(1, n - w)))
                    cand[s:s + w] *= np.exp(rng.normal(0, 0.3, w))
                    cand = np.maximum(cand, 0)
                    cS = np.sum(cand)
                    if cS > 0.01:
                        cand = cand * (n / cS)
                        cJ = compute_J_fft(cand)
                        if cJ < stall_threshold:
                            a = cand
                            m = np.zeros(n)
                            v = np.zeros(n)
                            t = 0
                            stall_count = 0
                        else:
                            a = best_a.copy()
                            m = np.zeros(n)
                            v = np.zeros(n)
                            t = 0
                            stall_count = 0

    final_local = compute_J_fft(best_a)
    return best_a, min(best_J, final_local)


def run_spectral_flattening(a_init, budget_s, seed, mix_rate=0.05):
    """Spectral flattening: iteratively flatten power spectrum to reduce convolution peak."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0.01:
        a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J_fft(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        fa = np.fft.rfft(a, size)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Target: geometric mean of current magnitude and uniform
        uniform_mag = np.full_like(mag, S / np.sqrt(size))
        # Use geometric mean for smoother transition
        new_mag = np.sqrt((1 - mix_rate) * mag**2 + mix_rate * uniform_mag**2)

        fa_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(fa_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)

        S_new = np.sum(a_new)
        if S_new > 0.01:
            a_new = a_new * (n / S_new)

        J_new = compute_J_fft(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new.copy()

        # Adaptive mix rate
        if iteration % 5000 == 0:
            mix_rate = max(0.01, mix_rate * 0.95)
            if iteration % 20000 == 0:
                print(f"    spectral iter={iteration} J={best_J:.6f} mix={mix_rate:.4f}")

    return best_a, best_J


def run_local_search(a_init, budget_s, seed):
    """Coordinate-wise and block local search for final refinement."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0.01:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J_fft(a)

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        if iteration % 10 == 0:
            # Block perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 25)))
            start = int(rng.integers(0, max(1, n - width)))
            cand = a.copy()
            noise = rng.normal(0, 0.05, width)
            cand[start:start + width] *= np.exp(noise)
            cand = np.maximum(cand, 0)
            cS = np.sum(cand)
            if cS > 0.01:
                cand = cand * (n / cS)
                cJ = compute_J_fft(cand)
                if cJ < best_J:
                    best_J = cJ
                    best_a = cand.copy()
                    a = cand.copy()
        else:
            # Single coordinate update
            i = int(rng.integers(0, n))
            best_mult = 1.0
            best_trial = np.inf

            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                cand = a.copy()
                cand[i] *= mult
                cand = np.maximum(cand, 0)
                cS = np.sum(cand)
                if cS < 0.01:
                    continue
                cJ = compute_J_fft(cand)
                if cJ < best_trial:
                    best_trial = cJ
                    best_mult = mult

            if best_mult != 1.0 and best_trial < best_J:
                a[i] *= best_mult
                a = np.maximum(a, 0)
                S_curr = np.sum(a)
                if S_curr > 0.01:
                    a = a * (n / S_curr)
                best_J = compute_J_fft(a)
                best_a = a.copy()

    return best_a, best_J


def make_power_ramp(n, alpha):
    """Power ramp: a[i] = ((i+1)/n)^alpha."""
    x = (np.arange(n) + 1.0) / n
    return x ** alpha


def make_cosine_ramp(n):
    """Cosine ramp: smooth transition from high to low."""
    x = np.arange(n) / max(n - 1, 1)
    return 0.01 + 0.99 * np.cos(np.pi * x / 2.0) ** 2


def make_sine_window(n):
    """Sine window: zero at both ends (approximately)."""
    x = np.arange(n)
    return np.sin(np.pi * (x + 0.5) / n)


def make_asymmetric(n, skew=0.3):
    """Asymmetric bump: skewed Gaussian."""
    x = np.arange(n) / n
    center = 0.5 + skew
    width = 0.3
    return np.exp(-((x - center) ** 2) / (2 * width ** 2)) + 0.001
