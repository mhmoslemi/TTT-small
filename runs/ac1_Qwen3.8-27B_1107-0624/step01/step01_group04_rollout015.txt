            if len(rows) < max_rows:
                viol_idx = np.where(violations > tol)[0]
                order = viol_idx[np.argsort(violations[viol_idx])[::-1]]
                added = 0
                for ix in order[:min(150, max_rows - len(rows))]:
                    ix = int(ix)
                    if ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                if added > 0:
                    continue
            else:
                if max_viol > 1000.0 * tol:
                    break

        # Rescale g to have sum S0
        g_rescaled = g * (S0 / sg)

        # Try blends
        best_blend_f = None
        best_blend_M = np.inf
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            fn = (1.0 - t) * f + t * g_rescaled
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn < 1e-10:
                continue
            fn = fn * (S0 / sn)
            if n <= 2048:
                cn = np.convolve(fn, fn)
            else:
                cn = convolve_fft(fn)
            cn = np.maximum(cn, 0.0)
            M_new = float(np.max(cn))
            if M_new < best_blend_M:
                best_blend_M = M_new
                best_blend_f = fn.copy()

        if best_blend_f is None or best_blend_M >= best_M * (1.0 - 1e-12):
            break

        f = best_blend_f
        M = best_blend_M
        best_f = f.copy()
        best_M = M
        score = M  # since sum(f) = S0 = sqrt(2n), score = 2n*M/(2n) = M

        if verbose and it % 10 == 0:
            print("  LP n=%d iter=%d M=%.8f rows=%d time=%d/%d" % (
                n, it, best_M, len(rows), int(time.monotonic() + budget_s - end_time), int(budget_s)))

        if score < 1.50:
            if verbose:
                print("  Achieved score < 1.50!")
            break

    return best_f / S0, best_M


def subgradient_optimize(p_in, budget_s):
    """
    Smoothed subgradient descent on the simplex to minimize max convolve(p,p).
    p_in: sum=1 vector. Returns improved p and score.
    """
    p = np.asarray(p_in, dtype=np.float64).ravel()
    n = p.size
    p = normalize_nonneg(p)

    c = np.convolve(p, p) if n <= 2048 else convolve_fft(p)
    c = np.maximum(c, 0.0)
    current_max = float(np.max(c))
    best_p = p.copy()
    best_max = current_max

    end_time = time.monotonic() + budget_s
    L = next_power_of_two(2 * n - 1)

    # Learning rate schedule
    lr0 = 0.02 if n > 256 else 0.04
    decay = 0.9995

    t = 0
    while time.monotonic() < end_time:
        t += 1
        rem = end_time - time.monotonic()
        if rem < 1.0:
            break

        # Compute smoothed weights
        alpha = 10.0 + 50.0 * np.exp(-3.0 * (1.0 - rem / max(budget_s, 1.0)))
        z = alpha * (c - current_max)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Compute subgradient: grad[j] = 2 * sum_k w[k] * p[k-j] for valid k-j
        # This is the cross-correlation of w (len 2n-1) and p (len n) at lags 0..n-1
        if n <= 2048:
            # Direct computation via correlate
            # r[j] = sum_i w[j+i] * p[i] for i=0..n-1
            # This is np.correlate(w, p, mode='valid') but w is longer than p
            # Actually: r[j] = sum_{i=0}^{n-1} w[j+i] * p[i]
            # = inner product of w[j:j+n] with p
            # Use FFT for efficiency
            L2 = next_power_of_two(3 * n - 2)
            Wf = np.fft.rfft(w, L2)
            Pf = np.fft.rfft(p, L2)
            # We want r[j] = sum_i w[j+i]*p[i]
            # This equals the cross-correlation: (w ⋆ p)[j] = sum_i w[i+j]*p[i]
            # In FFT: ifft(conj(FFT(p)) * FFT(w)) gives the cross-correlation
            # But we need to be careful about the convention
            # Let's use: r = ifft(FFT(w, L2) * conj(FFT(p, L2)))[:n]
            # Verify: this gives sum_k w[k]*p[k-j] ... no
            # Actually the standard result is:
            # ifft(FFT(a)*conj(FFT(b)))[n] = sum_k a[k+n]*conj(b[k]) (circular cross-correlation)
            # With zero-padding, this gives linear cross-correlation
            # (a ⋆ b)[n] = sum_k a[k+n]*conj(b[k])
            # We want r[j] = sum_i w[j+i]*p[i] = sum_k w[k]*p[k-j] (substituting k=j+i)
            # This is (w ⋆ p)[j] with the convention (a⋆b)[n] = sum_k a[k+n]*conj(b[k])
            # So we need ifft(FFT(w)*conj(FFT(p)))[j] for j=0..n-1
            r = np.fft.irfft(Wf * np.conj(Pf), L2)[:n].real
        else:
            r = np.fft.irfft(Wf * np.conj(Pf), L2)[:n].real

        grad = 2.0 * np.maximum(r, 0.0)

        # Normalize gradient for step size
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Update on simplex: p_new = p - lr * grad, then project
        lr = lr0 * (decay ** t)
        p_raw = p - lr * grad / gnorm

        # Project onto simplex {x >= 0, sum(x) = 1}
        # Using the standard algorithm
        v = p_raw.copy()
        rho = 1e-12
        for _ in range(50):
            v_sorted = np.sort(v)[::-1]
            css = np.cumsum(v_sorted)
            arange = np.arange(1, n + 1, dtype=np.float64)
            cond = v_sorted - (css - 1.0) / arange > rho
            if not np.any(cond):
                break
            idx = np.where(cond)[0][-1]
            t_val = (css[idx] - 1.0) / (idx + 1.0)
            p_new = np.maximum(p_raw - t_val, 0.0)
            break
        else:
            p_new = np.maximum(p_raw - np.mean(p_raw), 0.0)
            sn = float(p_new.sum())
            p_new = p_new / sn if sn > 0 else np.ones(n) / n

        sn = float(p_new.sum())
        if sn < 1e-10 or not np.isfinite(sn):
            p_new = np.ones(n, dtype=np.float64) / n
        else:
            p_new = p_new / sn

        # Ensure non-negativity and renormalize
        p_new = np.clip(p_new, 0.0, None)
        sn = float(p_new.sum())
        if sn < 1e-10:
            p_new = np.ones(n, dtype=np.float64) / n
        else:
            p_new = p_new / sn

        # Evaluate
        if n <= 2048:
            c = np.convolve(p_new, p_new)
        else:
            c = convolve_fft(p_new)
        c = np.maximum(c, 0.0)
        current_max = float(np.max(c))

        if current_max < best_max:
            best_max = current_max
            best_p = p_new.copy()

        p = p_new

    return best_p, 2.0 * n * best_max


def refine_with_local_lp(p_in, budget_s, n_rows_lp=300):
    """
    Local LP refinement: fix most elements, optimize a small subset via LP.
    This allows more precise optimization.
    """
    p = np.asarray(p_in, dtype=np.float64).ravel()
    n = p.size
    p = normalize_nonneg(p)
    S0 = np.sqrt(2.0 * n)
    f = p * S0

    if n <= 2048:
        c = np.convolve(f, f)
    else:
        c = convolve_fft(f)
    c = np.maximum(c, 0.0)
    M = float(np.max(c))

    best_f = f.copy()
    best_M = M
    end_time = time.monotonic() + budget_s

    # Try optimizing blocks
    block_size = max(4, n // 8)
    n_blocks = n // block_size

    for bi in range(n_blocks):
        if time.monotonic() >= end_time:
            break
        lo = bi * block_size
        hi = lo + block_size

        # Fix all elements except [lo, hi)
        # Amount in fixed part
        f_fixed = f.copy()
        f_fixed[lo:hi] = 0.0
        sum_fixed = float(f_fixed.sum())

        if sum_fixed < 1e-10:
            continue

        # Constraints: conv(f, g)[k] <= M for all k, where g has support in [lo, hi)
        # conv(f, g)[k] = conv(f_fixed, g)[k] + conv(f_block, g)[k]
        # Since g only has support in [lo, hi), this is still linear in g.

        # For simplicity, use the full constraint with the current f
        # But only optimize g in the block
        total_k = 2 * n - 1
        rows = list(np.linspace(0, total_k - 1, min(n_rows_lp, total_k), dtype=int))
        rows = sorted(set(rows + [int(np.argmax(c))]))
        rows = [r for r in rows if 0 <= r <= total_k - 1]

        # Build A for the block variables
        bs = hi - lo
        A = np.zeros((len(rows), bs))
        for ii, k in enumerate(rows):
            # conv(f, g_block)[k] = sum_j f_block[j] * f[k-(lo+j)]
            # = sum_j f[j+lo] * f[k-lo-j]... 
            # Actually: conv(f, g)[k] = sum_i f[i]*g[k-i]
            # g only nonzero at positions lo..hi-1, so k-i in [lo, hi-1], i in [k-hi+1, k-lo]
            # conv(f, g)[k] = sum_{i=k-hi+1}^{k-lo} f[i] * g[k-i]
            # Let j = k-i (index in g, which is 0..bs-1 corresponding to position lo..hi-1)
            # Then i = k-j, and j ranges from k-(k-lo) to k-(k-hi+1) = lo to hi-1... 
            # Actually j = k-i, i in [max(0,k-hi+1), min(n-1,k-lo)]
            # j in [k-min(n-1,k-lo), k-max(0,k-hi+1)]
            # And the local index in g is j (since g[j] corresponds to position lo+j... no)
            # Let me just use: g_local[j] = f[lo+j] for j=0..bs-1 (the variables)
            # conv(f, g)[k] = sum_i f[i] * g[k-i] where g is zero outside [lo,hi)
            # = sum_{j=0}^{bs-1} f[k-(lo+j)] * g_local[j]  (where k-lo-j in [0,n-1])
            for j in range(bs):
                i = k - (lo + j)
                if 0 <= i < n:
                    A[ii, j] = f[i]

        b = np.full(len(rows), M)
        c_obj = -np.ones(bs)

        tl = max(0.5, min(10.0, (end_time - time.monotonic()) / max(1, n_blocks - bi)))
        try:
            res = scipy_opt.linprog(c_obj, A_ub=A, b_ub=b,
                                     bounds=[(0.0, None)] * bs,
                                     method='highs',
                                     options={'time_limit': float(tl)})
        except Exception:
            continue

        if not res.success or res.x is None:
            continue

        g_block = np.clip(res.x, 0.0, None)
        sg = float(g_block.sum())
        if sg < 1e-12:
            continue

        # Total sum of new f
        new_f = f_fixed + np.zeros(n)
        new_f[lo:hi] = g_block
        sn = float(new_f.sum())
        if sn < 1e-10:
            continue
        new_f = new_f * (S0 / sn)

        # Evaluate
        if n <= 2048:
            cn = np.convolve(new_f, new_f)
        else:
            cn = convolve_fft(new_f)
        cn = np.maximum(cn, 0.0)
        M_new = float(np.max(cn))

        if M_new < best_M:
            best_M = M_new
            best_f = new_f.copy()
            c = cn

    return best_f / S0, 2.0 * n * best_M / (S0 * S0)


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(16, dtype=np.float64)
        s = 16.0
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def try_evaluate(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
    except Exception:
        return np.inf
    try:
        val = float(val)
    except Exception:
        return np.inf
    return val if np.isfinite(val) else np.inf


def prepare_start(n, kind, rng):
    n = int(n)
    if kind == "greedy":
        return greedy_normalized(n)
    if kind == "power45":
        return power_normalized(n, 0.45)
    if kind == "power55":
        return power_normalized(n, 0.55)
    if kind == "mix25":
        return mix_greedy_uniform(n, 0.25)
    if kind == "mix50":
        return mix_greedy_uniform(n, 0.50)
    if kind == "mix75":
        return mix_greedy_uniform(n, 0.75)
    if kind == "pert05":
        p = greedy_normalized(n)
        noise = np.exp(rng.normal(0.0, 0.05, n))
        return normalize_nonneg(p * noise)
    if kind == "pert15":
        p = greedy_normalized(n)
        noise = np.exp(rng.normal(0.0, 0.15, n))
        return normalize_nonneg(p * noise)
    if kind == "pert30":
        p = greedy_normalized(n)
        noise = np.exp(rng.normal(0.0, 0.30, n))
        return normalize_nonneg(p * noise)
    if kind == "dirichlet":
        try:
            return rng.dirichlet(np.ones(n))
        except Exception:
            return np.ones(n) / n
    return greedy_normalized(n)


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = score_norm(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
            print("NEW BEST: n=%d score=%.8f" % (a.size, best_score))
        n = int(a.size)
        old = best_by_n.get(n)
        if old is None or score < old[0]:
            best_by_n[n] = (float(score), a.copy())

    # Load previous best if available
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                score_h = score_norm(arr)
                if np.isfinite(score_h):
                    record(arr, score_h)
                    print("Loaded height_sequence_1: n=%d score=%.8f" % (arr.size, score_h))
        except Exception:
            pass

    # Quick baselines
    base_lengths = [128, 192, 256, 384, 512, 768, 1024]
    for n in base_lengths:
        if time.monotonic() >= deadline:
            break
        for kind in ["greedy", "power45", "mix50"]:
            p = prepare_start(n, kind, rng)
            record(p)

    # Phase 1: Subgradient optimization on several lengths
    print("Phase 1: Subgradient optimization")
    sg_lengths = [384, 512, 768]
    for n in sg_lengths:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        rec = best_by_n.get(n)
        if rec is not None:
            base = rec[1]
        else:
            base = greedy_normalized(n)
        alloc = min(60.0, max(15.0, 0.08 * rem))
        print("  Subgradient n=%d alloc=%.1fs" % (n, alloc))
        p_out, f_out = subgradient_optimize(base, alloc)
        record(p_out, f_out)

    # Phase 2: LP cutting-plane (main workhorse)
    print("Phase 2: LP cutting-plane")
    lp_lengths = sorted(best_by_n.keys(), key=lambda x: best_by_n[x][0])[:5]
    # Ensure we include some key lengths
    for plen in [384, 512, 768, 1024]:
        if plen not in lp_lengths:
            lp_lengths.append(plen)
    lp_lengths = sorted(set(lp_lengths))

    for n in lp_lengths:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        rec = best_by_n.get(int(n))
        if rec is not None:
            base = rec[1]
        else:
            base = greedy_normalized(n)
        alloc = min(400.0, max(20.0, 0.15 * rem))
        print("  LP n=%d alloc=%.1fs (score=%.8f)" % (n, alloc, score_norm(base)))
        p_out, f_out = lp_improve(base, alloc, verbose=True)
        record(p_out, f_out)

    # Phase 3: More LP on the best lengths
    print("Phase 3: Extended LP on best")
    lp_lengths2 = sorted(best_by_n.keys(), key=lambda x: best_by_n[x][0])[:3]
    for n in lp_lengths2:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        rec = best_by_n.get(int(n))
        if rec is None:
            continue
        base = rec[1]
        alloc = min(500.0, max(30.0, 0.2 * rem))
        print("  LP2 n=%d alloc=%.1fs (score=%.8f)" % (n, alloc, rec[0]))
        p_out, f_out = lp_improve(base, alloc, verbose=True)
        record(p_out, f_out)

    # Phase 4: Subgradient polishing on best
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            n_best = int(best_p.size)
            alloc = min(120.0, rem - 10.0)
            print("Phase 4: Subgradient polish n=%d alloc=%.1fs" % (n_best, alloc))
            p_out, f_out = subgradient_optimize(best_p, alloc)
            record(p_out, f_out)

    # Phase 5: Local LP block refinement
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 50:
            alloc = min(200.0, rem - 15.0)
            print("Phase 5: Local LP block refinement n=%d alloc=%.1fs" % (best_p.size, alloc))
            p_out, f_out = refine_with_local_lp(best_p, alloc)
            record(p_out, f_out)

    # If still not good enough, try more subgradient with restarts
    if best_score > 1.505 and deadline - time.monotonic() > 60:
        print("Phase 6: Random restarts + subgradient")
        n_try = int(best_p.size) if best_p is not None else 384
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            base = normalize_nonneg(best_p * np.exp(rng.normal(0, 0.15, n_try)))
            alloc = min(80.0, rem - 10.0)
            p_out, f_out = subgradient_optimize(base, alloc)
            record(p_out, f_out)
            if best_score < 1.505:
                break
            # Also try LP if available
            if rem > 40:
                p_out2, f_out2 = lp_improve(p_out, min(60.0, rem - 20.0), verbose=False)
                record(p_out2, f_out2)

    # Final: try a few more LP rounds on the absolute best
    if best_p is not None and deadline - time.monotonic() > 30:
        rem = deadline - time.monotonic()
        alloc = min(rem - 10.0, 180.0)
        print("Final LP polish: n=%d alloc=%.1fs" % (best_p.size, alloc))
        p_out, f_out = lp_improve(best_p, alloc, verbose=True)
        record(p_out, f_out)

    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = score_norm(best_p)

    print("Final best: n=%d score=%.8f" % (best_p.size, best_score))

    # Validate with the actual evaluator
    candidates = []
    if best_p is not None:
        candidates.append(best_p)
    for n, rec in sorted(best_by_n.items(), key=lambda x: x[1][0])[:3]:
        candidates.append(rec[1])

    best_eval_score = np.inf
    best_eval_seq = None
    for cand in candidates:
        if deadline - time.monotonic() < 5.0:
            break
        seq = scale_output(cand, 10.0)
        ev = try_evaluate(seq)
        if np.isfinite(ev) and ev < best_eval_score:
            best_eval_score = ev
            best_eval_seq = seq
        if np.isfinite(ev):
            print("  Eval: n=%d score=%.8f" % (len(cand), ev))

    if best_eval_seq is not None:
        print("Best evaluated score: %.8f" % best_eval_score)
        return best_eval_seq

    return scale_output(best_p, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the score."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main_search: %s" % str(e))
        # Fallback
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
