For the full convolution (2N-1 = 8191 elements), an FFT of size 8192 is sufficient.

Let me now write the complete solution:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: Multi-stage softmax L-BFGS-B with adaptive active-set subgradient
polishing, LP-based escape, and multi-resolution warm-starting.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize, linprog
    from scipy.sparse import csc_matrix
except Exception:
    minimize = None
    linprog = None
    csc_matrix = None

N = 4096
X = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
FFT_SELF = 1 << (2 * N - 2).bit_length()  # 8192
FFT_GRAD = 1 << (3 * N - 3).bit_length()  # 16384
BOUNDS = [(-40.0, 40.0)] * (N - 1)

_best_score = float('inf')
_best_z = None
_deadline = 0.0


class Timeout(Exception):
    pass


def _pw2(m):
    r = 1
    while r < m:
        r <<= 1
    return r


def _clean_z(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    return np.clip(x, -40.0, 40.0)


def _z2p(z):
    z = _clean_z(z)
    m = float(np.max(z)) if z.size else 0.0
    e = np.exp(z - m)
    e0 = np.exp(-m)
    Z = e0 + float(e.sum())
    p = np.empty(N, dtype=np.float64)
    p[0] = e0 / Z
    p[1:] = e / Z
    return p


def _p2z(p):
    p = np.maximum(p, 1e-15)
    logp = np.log(p)
    z = logp[1:] - logp[0]
    return np.clip(z, -40.0, 40.0)


def _conv_self(p):
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f * f, FFT_SELF).real
    c = c[:2*N-1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(p):
    c = _conv_self(p)
    return 2.0 * float(N) * float(c.max())


def _update_best(z, s):
    global _best_score, _best_z
    if np.isfinite(s) and s < _best_score:
        _best_score = float(s)
        _best_z = np.array(z, copy=True)


def _grad_from_weights(p, w):
    """Compute grad_i of sum_k w[k]*conv(k) = 2*cross_corr(p, w)[i]."""
    fp = np.fft.rfft(p, FFT_GRAD)
    fw = np.fft.rfft(w, FFT_GRAD)
    cross = np.fft.irfft(np.conj(fp) * fw, FFT_GRAD).real[:N]
    return 2.0 * cross


def _objective_grad(x, alpha):
    if time.monotonic() >= _deadline:
        raise Timeout
    x = _clean_z(x)
    p = _z2p(x)
    c = _conv_self(p)
    hard = 2.0 * N * float(c.max())
    _update_best(x, hard)
    
    cmax = float(c.max())
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(e.sum())
    if s <= 1e-300:
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + math.log(s) / alpha
        w = e / s
    
    gp = _grad_from_weights(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e4, 1e4)
    return float(J), gz


def _lbfgs(z0, alpha, max_fun, dl):
    global _deadline
    if minimize is None:
        return
    _deadline = float(dl)
    try:
        minimize(_objective_grad, _clean_z(z0), args=(alpha,),
                 method='L-BFGS-B', jac=True, bounds=BOUNDS,
                 options={'maxfun': max_fun, 'maxiter': max_fun//2,
                          'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 25,
                          'disp': False})
    except Timeout:
        pass
    except Exception:
        pass


def _subgradient_polish(z0, dl, rng, m_start=5, m_end=40):
    """Aggressive subgradient polishing with adaptive active set size."""
    z = _clean_z(z0)
    p = _z2p(z)
    score = _score(p)
    _update_best(z, score)
    
    noise = 0.08
    fails = 0
    m = m_start
    
    while time.monotonic() < dl:
        c = _conv_self(p)
        cmax = float(c.max())
        score = 2.0 * N * cmax
        
        # Adaptive top-m
        m_eff = min(m, len(c) - 1)
        idx = np.argpartition(c, -m_eff)[-m_eff:]
        w = np.zeros(len(c))
        cw = c[idx]
        # Use temperature-based weighting
        scale = 50.0
        wv = np.exp(scale * (cw - cw.max()))
        w[idx] = wv / wv.sum()
        
        gp = _grad_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))
        
        accepted = False
        if gnorm > 1e-15:
            direction = -gz / gnorm
            for step in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
                zt = z + step * direction
                zt = _clean_z(zt)
                pt = _z2p(zt)
                st = _score(pt)
                if st < score - 1e-13:
                    z, p, score = zt, pt, st
                    _update_best(z, score)
                    accepted = True
                    break
        
        if not accepted:
            # Try random perturbations
            best_s = score
            best_z = z
            best_p = p
            ntrials = 5
            for _ in range(ntrials):
                zt = z + rng.normal(0, noise, N-1)
                zt = _clean_z(zt)
                pt = _z2p(zt)
                st = _score(pt)
                if st < best_s:
                    best_s, best_z, best_p = st, zt, pt
            if best_s < score - 1e-13:
                z, p, score = best_z, best_p, best_s
                _update_best(z, score)
                accepted = True
        
        if accepted:
            fails = 0
            noise = min(0.3, noise * 1.3)
            m = min(m_end, m + 2)
        else:
            fails += 1
            noise = max(0.02, noise * 0.75)
            m = max(m_start, m - 1)
            if fails > 80:
                break


def _lp_escape(z0, dl, rng):
    """Use LP to find a direction that reduces the max convolution."""
    if linprog is None or csc_matrix is None:
        return
    z = _clean_z(z0)
    p = _z2p(z)
    c = _conv_self(p)
    cmax = float(c.max())
    
    # Solve: max sum(g) s.t. (p*g)[k] <= cmax for all k, g >= 0
    # Decision variables: g[0], ..., g[N-1] (N variables)
    # Constraints: for each k in 0..2N-2, sum_j p[j]*g[k-j] <= cmax
    # This is a large LP. Let's use a subset of constraints.
    
    # Actually, for large N this LP is too big. Let's use a simplified version.
    # Just find the top few peaks and use those as constraints.
    topk = np.argpartition(c, -10)[-10:]
    
    # For a subset of constraints, we can solve a smaller QP or LP.
    # But the constraint (p*g)[k] <= cmax is linear in g (since p is fixed).
    # So this IS a valid LP.
    
    # c_matrix: (2N-1) x N, where c_matrix[k, j] = p[j] if 0 <= k-j < N else 0
    # But we only use 10 constraints, so it's 10 x N.
    
    # Objective: maximize sum(g) => minimize -sum(g)
    # c_obj = -1 for all N variables
    # A_ub[k, j] = p[j] if 0 <= k-j < N else 0 (for k in topk)
    # b_ub = cmax for each constraint
    # Bounds: g >= 0
    
    # This LP has N=4096 variables and 10 constraints. Should be fast.
    
    c_obj = -np.ones(N)
    A_rows = []
    for k in topk:
        row = np.zeros(N)
        # (p*g)[k] = sum_j p[j]*g[k-j] where 0<=k-j<N => j >= k-N+1 and j <= k and j >= 0
        # So g index = k-j, which ranges from max(0, k-N+1) to min(k, N-1)
        # For a given g index m = k-j, j = k-m, and we need 0 <= j < N => 0 <= k-m < N => k-N < m <= k
        # Also 0 <= m < N.
        lo = max(0, k - N + 1)
        hi = min(k, N - 1)
        for m in range(lo, hi + 1):
            j = k - m
            row[m] = p[j]
        A_rows.append(row)
    A_ub = np.array(A_rows)
    b_ub = np.full(len(topk), cmax)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * N, method='highs')
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        gsum = float(g.sum())
        if gsum > 1e-10:
            g = g / gsum
            # Interpolate
            t = 0.3
            p_new = (1-t)*p + t*g
            p_new = np.maximum(p_new, 0)
            p_new = p_new / p_new.sum()
            z_new = _p2z(p_new)
            s_new = _score(p_new)
            _update_best(z_new, s_new)
            # Continue from here
            if s_new < _score(p):
                # Do a few more L-BFGS steps from z_new
                _lbfgs(z_new, 3e6, 200, dl)
