"""
Multi-phase optimizer for step function coefficients minimizing
J(a) = 2n * max(convolve(a,a)) / (sum(a))^2.
Uses mirror descent with log-sum-exp smoothing, LP-based direction finding,
and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def cross_convolve_fft(a, g):
    n = len(a)
    m = len(g)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    result = np.fft.irfft(fa * fg, size)
    return result[:n + m - 1]

def compute_J_fft(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)

def smooth_objective_and_grad(a, tau):
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # Gradient: dJ/da[i] = 4n * cross[i] / S^2 - 4n * Tmax / S^3
    # where cross[i] = sum_k p[k] * a[k-i] (cross-correlation)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad

def mirror_descent(a_init, budget_s, seed, tau_start=10.0, tau_end=0.1):
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (np.sqrt(2 * n) / S)

    best_a = a.copy()
    best_J = compute_J_fft(a)
    eps = 0.01  # multiplicative step size
    t = 0
    start_time = time.monotonic()
    last_verify = 0.0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac

        J, grad = smooth_objective_and_grad(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            tau = max(1.0, tau)
            continue

        # Multiplicative update (mirror descent with L1 mirror)
        # a[i] <- a[i] * exp(-eps * grad[i])
        # But we need to handle the normalization
        a = a * np.exp(-eps * grad / (np.max(np.abs(grad)) + 1e-10))
        a = np.maximum(a, 1e-10)

        # Renormalize
        S = np.sum(a)
        if S > 0:

                best_J = val
                best_a = a.copy()

    final_J = compute_J_fft(best_a)
    return best_a, min(best_J, final_J)

"""
Multi-phase optimization of step function coefficients to minimize
J(a) = 2n * max(convolve(a,a)) / (sum(a))^2.
Uses mirror descent with adaptive log-sum-exp smoothing, periodic deflation,
diverse initializations, and coordinate descent polishing.
"""

import numpy as np
import time

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def smooth_grad(a, tau):
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_s / sum_exp
    J = 2.0 * n * Tmax / (S * S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = 4.0 * n * cross / (S * S) - 4.0 * n * Tmax / (S ** 3)
    return J, grad

def mirror_descent_phase(a_init, budget_s, seed, tau_start=5.0, tau_end=0.05):
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy().astype(np.float64)
    S = np.sum(a)
    target_S = np.sqrt(2.0 * n)
    if S > 0:
        a *= target_S / S
    
    best_a = a.copy()
    best_J = compute_J(a)
    t = 0
    start_t = time.monotonic()
    last_check = 0.0
    eps = 2.0 / np.sqrt(n)  # step size
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_t
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        J_smooth, grad = smooth_grad(a, tau)
        if not np.isfinite(J_smooth) or not np.all(np.isfinite(grad)):
            a = best_a.copy()
            continue
        
        # Mirror descent step
        # Normalize gradient for stable updates
        grad_scale = np.max(np.abs(grad))
        if grad_scale < 1e-15:
            break
        a = a * np.exp(-eps * grad / grad_scale)
        a = np.maximum(a, 1e-15)
        
        # Renormalize
        S = np.sum(a)
        if S > 0:
            a *= target_S / S
        
        t += 1
        
        # Periodic check
        if t % 50000 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
        
        # Verify with evaluate_sequence
        now = time.monotonic()
        if now - last_check > 3.0:
            last_check = now
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
        
        # Adaptive eps: decrease if stuck
        if t % 100000 == 0 and t > 200000:
            eps *= 0.9
    
    return best_a, best_J

def deflation_step(a, n_top=5):
    """Identify top peaks of b and reduce contributing a[i]."""
    n = len(a)
    b = convolve_fft(a)
    # Find top n_top peaks
    top_k = np.argsort(b)[::-1][:n_top]
    # For each peak, find contributing a[i]
    # b[k] = sum_j a[j]*a[k-j]
    # Contribution of a[i] to b[k] is a[k-i] (if valid)
    reduce = np.zeros(n)
    b_max = b[top_k[0]]
    for k in top_k:
        weight = b[k] / b_max
        # a[i] contributes a[k-i] to b[k]
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                reduce[i] += weight * a[ki]
    # Apply deflation
    a_new = a / (1.0 + 0.1 * reduce / (np.max(reduce) + 1e-15))
    a_new = np.maximum(a_new, 1e-15)
    S = np.sum(a_new)
    if S > 0:
        a_new *= np.sqrt(2.0 * n) / S
    return a_new

def coordinate_descent_polish(a, budget_s, seed):
    """Polish with coordinate descent on sensitive coordinates."""
    n = len(a)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    best_a = a.copy()
    best_J = compute_J(a)
    target_S = np.sqrt(2.0 * n)
    
    iterations = 0
    while time.monotonic() < deadline:
        # Pick a random coordinate to update
        i = int(rng.integers(0, n))
        
        # Try several values for a[i]
        current_val = a[i]
        best_local = a[i]
        best_local_J = best_J
        
        # We need to evaluate how changing a[i] affects J
        # b[k] changes by: delta * (a[k-i] + a[i-k]) + delta^2 (if k=2i)
        # For efficiency, compute the "effect vector"
        effect = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            ki = k - i
            if 0 <= ki < n:
                effect[k] += a[ki]
            ik = i - k
            if 0 <= ik < n:
                effect[k] += a[ik]
        # The a[i]^2 term: if 2i is in [0, 2n-2], effect[2i] needs +1 for the quadratic
        if 0 <= 2 * i <= 2 * n - 2:
            effect[2 * i] += 0  # the quadratic term is delta^2, handled separately
            
        # Actually, let me just do a few trial values
        b_base = convolve_fft(a)  # current b
        for delta_frac in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 0.0, -0.5, -0.7]:
            trial = max(0.0, current_val * delta_frac)
            delta = trial - current_val
            # New b: b[k] + delta * effect[k] + delta^2 * (1 if k=2i else 0)
            b_trial = b_base + delta * effect
            if 0 <= 2 * i <= 2 * n - 2:
                b_trial[2 * i] += delta * delta
            # New S
            S_trial = np.sum(a) + delta
            if S_trial < 0.01:
                continue
            J_trial = 2.0 * n * np.max(np.maximum(b_trial, 0)) / (S_trial * S_trial)
            if J_trial < best_local_J:
                best_local_J = J_trial
                best_local = trial
        
        if best_local != current_val:
            a[i] = best_local
            # Renormalize
            S = np.sum(a)
            if S > 0:
                a *= target_S / S
            if best_local_J < best_J:
                best_J = best_local_J
                best_a = a.copy()
        
        iterations += 1
    
    return best_a, best_J
