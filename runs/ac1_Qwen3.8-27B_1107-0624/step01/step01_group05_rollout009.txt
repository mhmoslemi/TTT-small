Uses multi-scale log-sum-exp smoothed optimization with L-BFGS-B,
adaptive alpha scheduling, top-K peak targeting, multiplicative update
polishing, and multi-length exploration. Improves upon previous best

import numpy as np
import time
from scipy.optimize import minimize

# --- FFT size helpers ---

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


# --- Core computation for a given sequence length n ---

def _make_computer(n):
    fft_self = _next_pow2(2 * n - 1)
    fft_grad = _next_pow2(3 * n - 1)

    def self_conv(p):
        f = np.fft.rfft(p, fft_self)
        c = np.fft.irfft(f * f, fft_self).real[:2 * n - 1]
        np.maximum(c, 0.0, out=c)
        return c

    def grad_p(p, w):
        conv = np.fft.irfft(
            np.fft.rfft(p[::-1], fft_grad) * np.fft.rfft(w, fft_grad),
            fft_grad
        ).real
        h = conv[n - 1:2 * n - 1]
        return 2.0 * h

    def score(p):
        c = self_conv(p)
        return 2.0 * float(n) * float(np.max(c))

    return self_conv, grad_p, score


# --- Parameterization: logits -> probability vector ---

def _probs_from_z(z, n):
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -50.0, 50.0)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(n, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _score_z(z, n, computer):

# --- Smoothed objective and gradient ---

def _objective_and_grad(x, n, computer, alpha, state, deadline):
    if time.monotonic() >= deadline:
        raise SearchTimeout

    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -50.0, 50.0)
    p = _probs_from_z(x, n)
    c = computer[0](p)

    cmax = float(np.max(c))
    hard = 2.0 * float(n) * cmax
    _update_best(hard, x, n, state)

    u = alpha * (c - cmax)
    u = np.clip(u, -700.0, 700.0)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s

    gp = computer[1](p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    return float(J), gz

                  f"n={n} elapsed={elapsed:.0f}s")

    print(f"Phase 2 done: best={state['best_score']:.6f} (n={state['best_n']})")

    # Phase 3: Polishing
    print(f"Phase 3: Polishing")
    if state['best_z'] is not None and time.monotonic() < polish_deadline - 20:
        n_best = state['best_n']
        # Round n_best to a power of 2 if close, for FFT efficiency
        if n_best not in computers:
            computers[n_best] = _make_computer(n_best)

        # Mirror descent polishing
        _mirror_descent_polish(state['best_z'], n_best, computers[n_best],
                               state, polish_deadline - 30, rng)

        # Final short L-BFGS pass with very high alpha
        if time.monotonic() < polish_deadline - 15:
            z_final = state['best_z'] + rng.normal(0.0, 0.03, n_best - 1)
            z_final = np.clip(z_final, -50.0, 50.0)
            _multiscale_optimize(z_final, n_best, computers[n_best], state,
                                 polish_deadline - 15, [5e6, 2e7, 1e8], 150)

    print(f"Phase 3 done: best={state['best_score']:.6f} (n={state['best_n']})")

    # Convert best z to sequence
    if state['best_z'] is not None:
        n_best = state['best_n']
        p = _probs_from_z(state['best_z'], n_best)
    else:
        n_best = 4096
        p = np.ones(n_best, dtype=np.float64) / n_best

    seq = [float(v) for v in p]

    # Validate with evaluate_sequence
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float('inf')

    # Fallback to uniform if something went wrong
    if not np.isfinite(final_score) or final_score > 2.0:
        seq = [1.0 / 4096.0] * 4096
        try:
            final_score = float(evaluate_sequence(seq))
        except Exception:
            final_score = 2.0

    # Also check if a different length gives better
    if prev_seq is not None:
        try:
            ps = [float(v) for v in prev_seq]
            ps_score = float(evaluate_sequence(ps))
            if np.isfinite(ps_score) and ps_score < final_score:
                seq = ps
                final_score = ps_score
        except Exception:
            pass

    # Also try uniform at different lengths
    for n_try in [2048, 4096, 8192]:
        try:
            u_seq = [1.0 / n_try] * n_try
            u_score = float(evaluate_sequence(u_seq))
            if np.isfinite(u_score) and u_score < final_score:
                seq = u_seq
                final_score = u_score
        except Exception:
            pass

    print(f"Final: score={final_score:.6f}, length={len(seq)}")
    return seq
