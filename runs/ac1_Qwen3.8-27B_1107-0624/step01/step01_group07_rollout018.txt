def kkt_solve(a, P):
    """Solve the KKT linear system. P is a boolean array of peak positions."""
    n = len(a)
    # Build C where C[j,i] = p[j+i]
    C = np.zeros((n, n))
    for j in range(n):
        for i in range(n):
            k = j + i
            if k < 2*n - 1 and P[k]:
                C[j, i] = 1
    
    # Solve C * a = lambda * 1, sum(a) = 1
    # [C | -1; 1^T | 0] [a; lambda] = [0; 1]
    A = np.zeros((n+1, n+1))
    b = np.zeros(n+1)
    A[:n, :n] = C
    A[:n, n] = -1
    A[n, :n] = 1
    b[n] = 1
    
    try:
        x = np.linalg.solve(A, b)
        a_new = x[:n]
        lam = x[n]
        if np.all(a_new >= -1e-10):
            a_new = np.clip(a_new, 0, None)
            a_new /= a_new.sum()
            return a_new
    except:
        pass
    return None

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()

    return 2.0 * n * c.max() / (S * S)


def kkt_solve(a, peak_threshold=0.95):
    """
    Solve the KKT linear system for the given peak set.
    Returns a new sequence or None if no valid solution.
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    c_max = c.max()
    if c_max < 1e-15:
        return None

def lp_direction(a, M):
    """Find LP direction: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix: A[g] = conv(a, g)
    # conv(a, g)[k] = sum_i a[i] * g[k-i]
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a

    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except:
        pass
    return None

"""
KKT-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Karush-Kuhn-Tucker conditions to directly solve for stationary points of the
peak auto-convolution, combined with LP direction finding and multiplicative
gradient descent for refinement. Exploits the structure that at a KKT point,
the cross-correlation of a with the peak indicator must be constant on the support.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def compute_cmax(a):
    """Compute max of conv(a,a) assuming sum(a)=1."""
    c = np.convolve(a, a, mode='full')
    return c.max(), c


def kkt_solve(a, peak_threshold=0.95):
    """
    Solve the KKT linear system for the given peak set.
    At a KKT point: sum_{k in P} a[k-j] = lambda for all j in support(a).
    Returns a new sequence (normalized to sum=1) or None if no valid solution.
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    c_max = c.max()
    if c_max < 1e-15:
        return None

    P = (c >= peak_threshold * c_max).astype(float)
    if P.sum() < 2:
        return None

    # Build C matrix: C[j,i] = P[j+i]
    idx = np.add.outer(np.arange(n), np.arange(n))
    C = P[idx]

    # Solve [C, -1; 1^T, 0] [a; lambda] = [0; 1]
    A_mat = np.zeros((n + 1, n + 1))
    A_mat[:n, :n] = C
    A_mat[:n, n] = -1.0
    A_mat[n, :n] = 1.0
    b_vec = np.zeros(n + 1)
    b_vec[n] = 1.0

    try:
        x = np.linalg.solve(A_mat, b_vec)
    except np.linalg.LinAlgError:
        try:
            x, _, _, _ = np.linalg.lstsq(A_mat, b_vec, rcond=None)
        except Exception:
            return None

    a_new = x[:n]

    # Check non-negativity
    if np.any(a_new < -1e-6):
        return None

    a_new = np.clip(a_new, 0.0, None)
    S = a_new.sum()
    if S < 1e-10:
        return None
    a_new /= S
    return a_new


def kkt_iteration(a, max_iters=300, thresholds=None):
    """
    Iterate the KKT solve, updating the peak set each time.
    Tries multiple thresholds to explore different peak configurations.
    Returns the best sequence found (normalized to sum=1).
    """
    if thresholds is None:
        thresholds = [0.995, 0.99, 0.98, 0.97, 0.95, 0.93, 0.90, 0.85, 0.80, 0.70]

    best_a = a.copy()
    best_cmax, _ = compute_cmax(a)

    for thresh in thresholds:
        a_curr = best_a.copy()
        prev_cmax = best_cmax
        for it in range(max_iters):
            a_new = kkt_solve(a_curr, peak_threshold=thresh)
            if a_new is None:
                break

            c_new_max, _ = compute_cmax(a_new)

            if c_new_max < best_cmax - 1e-10:
                best_a = a_new.copy()
                best_cmax = c_new_max
                a_curr = a_new
            else:
                # Check convergence: if cmax didn't improve, try a small perturbation
                if it > 5:
                    break
                else:
                    a_curr = 0.9 * a_curr + 0.1 * a_new
                    a_curr = np.clip(a_curr, 0, None)
                    if a_curr.sum() > 0:
                        a_curr /= a_curr.sum()
                    else:
                        break

    return best_a


def lp_direction(a, M):
    """Find LP direction: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    # Build constraint matrix: conv(a, g)[k] = sum_i a[i] * g[k-i]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a, deadline, rng, max_iters=300):
    """LP-based optimization with adaptive step size and perturbation."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    stall = 0
    step = 0.3

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_direction(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        # Backtracking line search
        improved = False
        for shrink in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            t = step * shrink
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.5, step * 1.05)
        else:
            stall += 1
            step = max(0.01, step * 0.7)
            if stall > 0 and stall % 25 == 0:
                # Random perturbation
                noise = np.abs(rng.normal(0, 0.06, n))
                a_pert = a * (1.0 + noise)
                a_pert = np.clip(a_pert, 0, None)
                if a_pert.sum() > 1e-10:
                    a_pert /= a_pert.sum()
                    sc = compute_score(a_pert)
                    if sc < best_score:
                        a = a_pert
                        best_a = a_pert
                        best_score = sc
                        stall = 0
                    else:
                        a = best_a
            if stall > 120:
                break

    return best_a, best_score


def gradient_refine(a, deadline, rng, max_iters=10000):
    """Multiplicative gradient descent targeting peak convolution positions."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.08
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        # Use a band of near-peak positions for smoother gradient
        peak_thresh = 0.85 * M
        peaks = c >= peak_thresh
        peak_idx = np.where(peaks)[0]

        if len(peak_idx) == 0:
            break

        # Compute subgradient: s_j = sum_{k in peaks} 2*a[k-j]
        grad = np.zeros(n)
        for k in peak_idx:
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            js = np.arange(j_min, j_max + 1)
            vals = k - js
            grad[js] += 2.0 * a[vals]

        # Normalize gradient for stable updates
        grad_max = grad.max()
        if grad_max < 1e-15:
            break

        # Multiplicative update
        a_new = a * np.exp(-lr * grad / grad_max)
        a_new = np.clip(a_new, 1e-15, None)
        S = a_new.sum()
        if S < 1e-10:
            break
        a_new /= S
        sc = compute_score(a_new)

        if sc < best_score - 1e-12:
            best_a = a_new
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.15, lr * 1.01)
        else:
            stall += 1
            lr = max(0.002, lr * 0.85)
            if stall > 80:
                # Perturbation to escape
                noise = np.abs(rng.normal(0, 0.04, n))
                a_pert = a * (1.0 + noise)
                a_pert = np.clip(a_pert, 1e-15, None)
                a_pert /= a_pert.sum()
                sc_p = compute_score(a_pert)
                if sc_p < best_score:
                    a = a_pert
                    best_a = a_pert
                    best_score = sc_p
                    stall = 0
                else:
                    a = best_a
                    if rng.random() < 0.2:
                        a = a_pert
                    elif stall > 200:
                        break
            if stall > 300:
                break

    return best_a, best_score


def kkt_lp_hybrid(a, deadline, rng):

    a = np.clip(a, 0.0, None).astype(np.float64)

                    if a_new.sum() > 1e-10:
                        a_new /= a_new.sum()
                        sc = compute_score(a_new)
                        if sc < best_score - 1e-12:
                            best_a = a_new
                            best_score = sc
                            a = a_new
                            improved = True
                            break

        if improved:
            stall = 0
        else:
            stall += 1
            # Perturbation
            noise = np.abs(rng.normal(0, 0.05, n))
            a_pert = a * (1.0 + noise)
            a_pert = np.clip(a_pert, 0, None)

                a_pert /= a_pert.sum()
                sc = compute_score(a_pert)
                if sc < best_score:
                    a = a_pert
                    best_a = a_pert
                    best_score = sc
                    stall = 0
                else:

def make_cosine(n):
    """Raised cosine (Hann) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_flat_taper(n, taper_frac=0.3):
    """Flat center with tapered edges."""
    a = np.ones(n)
    t = int(n * taper_frac / 2)
    if t > 0:
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(t) / t))
        a[:t] = taper
        a[-t:] = taper[::-1]
    return a


def make_quadratic(n):
    """Quadratic taper."""
    c = (n - 1) / 2.0
    x = (np.arange(n) - c) / (c + 1.0)
    return np.maximum(0.0, 1.0 - x * x)
