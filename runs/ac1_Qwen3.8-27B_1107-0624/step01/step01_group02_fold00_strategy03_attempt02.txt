import numpy as np

def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients with appropriate security checks.
    Returns np.inf if the input is invalid or if any checks fail."""
    # Security checks
    if not isinstance(sequence, list):
        return np.inf
    if len(sequence) == 0:
        return np.inf
    
    sequence = [float(x) for x in sequence]
    n = len(sequence)
    sequence = np.asarray(sequence, dtype=np.float64)
    if not np.isfinite(sequence).all():
        return np.inf
    
    # Non-integer types are rejected
    for x in sequence:
        if not isinstance(x, (bool, int, float)):
            return np.inf
    
    # Booleans are rejected (subclass of int)
    if np.isin(sequence, [0, 1]).any():
        return np.inf
    
    # Reject infinite and NaN values
    if not np.isfinite(sequence).all():
        return np.inf
    
    sequence = np.clip(sequence, 0.0, 1000.0)
    
    # Check if sequence contains any negative numbers (though clip should ensure non-negativity, double-check)
    if any(x < 0 for x in sequence):
        return np.inf
    
    # Check sum after clipping
    s = sequence.sum()
    if s < 1e-12:
        return np.inf
    if n > 2048 and not np.isfinite(s):
        return np.inf
    
    # Normalize the sequence for consistency
    sequence = sequence / s * 1.0
    
    # Full convolution
    conv = np.convolve(sequence, sequence, mode='full')
    max_conv = conv.max()
    n_val = float(n)
    min_denom = 1.0
    score = float(2 * n_val * max_conv / (min_denom * (n_val**2)))  # Keeping the denominator minimum
    
    return score
