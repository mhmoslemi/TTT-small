def compute_objective_and_gradient(x, t, n):

    x: parameter vector (length n)
    t: smoothing parameter
    n: sequence length
    Returns: (objective_value, gradient)
    """
    # Softmax parameterization
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    s = exp_x / exp_x.sum()

    q = np.convolve(s, s)  # length 2n-1

    exp_tq = np.exp(tq - max_tq)
    logsumexp = max_tq + np.log(np.sum(exp_tq))

Uses L-BFGS-B with softmax parameterization and continuation on smoothing parameter.
Combines with LP-guided local search for final refinement.

def softmax(x):
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / e.sum()

def compute_obj_grad_softmax(x, t, n):
    """Compute smoothed objective and gradient using softmax parameterization."""
    s = softmax(x)
    
    q = np.convolve(s, s)  # length 2n-1
    tq = t * q
    max_tq = np.max(tq)
    exp_tq = np.exp(tq - max_tq)
    sum_exp = np.sum(exp_tq)
    logsumexp = max_tq + np.log(sum_exp)
    p = exp_tq / sum_exp
    
    # Objective: 2n * (1/t) * logsumexp  (since sum(s)=1 by softmax)
    obj = 2.0 * n * logsumexp / t
    
    # grad_s[i] = sum_k p[k] * dq[k]/ds[i]
    first = np.correlate(p, s, mode='full')[:n]
    second = np.convolve(s, p, mode='full')[:n]
    grad_s = first + second
    
    # Chain rule through softmax:
    # d obj / d x[j] = sum_i (2n * grad_s[i]) * s[i] * (delta(i,j) - s[j])
    # = 2n * (grad_s[j]*s[j] - s[j]*dot(grad_s, s))
    grad_x = 2.0 * n * (grad_s * s - s * np.dot(grad_s, s))
    
    return obj, grad_x

def compute_obj_grad_direct(s, t):

    if total < 1e-15:
        return np.inf, np.zeros(n)
    
    q = np.convolve(s, s)
    tq = t * q
    max_tq = np.max(tq)
    exp_tq = np.exp(tq - max_tq)
    sum_exp = np.sum(exp_tq)
    logsumexp = max_tq + np.log(sum_exp)
    p = exp_tq / sum_exp
    
    # obj = 2n/t * logsumexp / total^2
    obj = 2.0 * n * logsumexp / (t * total * total)
    
    first = np.correlate(p, s, mode='full')[:n]
    second = np.convolve(s, p, mode='full')[:n]
    grad_s = first + second  # = sum_k p[k] * dq[k]/ds[i] (without the factor of t)
    
    # d obj / d s[i] = 2n * [grad_s[i]/total^2 - 2*logsumexp/(t*total^3)]
    grad = 2.0 * n * (grad_s / (total * total) - 2.0 * logsumexp / (t * total * total * total))
    
    return obj, grad

    """Compute the true (non-smoothed) score."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    q = np.convolve(s, s)
    return 2.0 * n * np.max(q) / (total * total)

def smooth_optimize(x_init, n, t_values, deadline, max_iter=300):
    """Run continuation method with L-BFGS-B."""
    x = x_init.copy()
    for t in t_values:
        if time.monotonic() > deadline:
            break
        
        def fun(x):
            return compute_obj_grad_softmax(x, t, n)
        
        result = minimize(fun, x, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'gtol': 1e-10, 'maxfun': 500})
        if result.x is not None:
            x = result.x
    
    s = softmax(x)
    return s, true_score(s)

def build_conv_matrix(s):
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A

def lp_refine(s, deadline, max_iters=100):
    """LP-guided local search for fine-tuning."""
    n = len(s)
    best = s.copy()
    best_sc = true_score(s)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        q = np.convolve(s, s)
        M = np.max(q)
        
        A = build_conv_matrix(s)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 3.0})
            if not res.success or res.x is None:
                break
            b = np.maximum(res.x, 0.0)
        except Exception:
            break
        
        if np.sum(b) < 1e-14:
            break
        
        b = b / b.sum()
        
        # Line search
        best_t = 0.0
        best_ts = best_sc
        for t in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = np.maximum((1.0 - t) * s + t * b, 0.0)
            if cand.sum() < 0.01:
                continue
            ts = true_score(cand)
            if ts < best_ts - 1e-12:
                best_ts = ts
                best_t = t
        
        if best_t == 0.0:
            break
        
        s = np.maximum((1.0 - best_t) * s + best_t * b, 0.0)
        s = s / s.sum()
        
        if best_ts < best_sc:
            best_sc = best_ts
            best = s.copy()
    
    return best, best_sc

    """Smoothed objective and gradient via softmax parameterization."""
    s = softmax(x)
    q = np.convolve(s, s)
    tq = t * q
    max_tq = np.max(tq)
    exp_tq = np.exp(tq - max_tq)
    sum_exp = np.sum(exp_tq)
    logsumexp = max_tq + np.log(sum_exp)
    p = exp_tq / sum_exp

    # grad_s[i] = sum_k p[k] * dq[k]/ds[i]
    first = np.correlate(p, s, mode='full')[:n]
    second = np.convolve(s, p, mode='full')[:n]
    grad_s = first + second
    
    # Gradient through softmax
    grad_x = 2.0 * n * (grad_s * s - s * np.dot(grad_s, s))
