Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based direction finding with cutting planes (HiGHS solver)

    maxc = float(np.max(c))

def mix_greedy_uniform(n, lam):

def lp_cutting_plane(p_init, budget_s, verbose=True):
    p = np.asarray(p_init, dtype=np.float64).ravel()
    n = p.size
    if n < 16:
        n = 16
        p = greedy_normalized(n)

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    best_f = f.copy()
    best_score = actual_score(f)

    max_rows = min(2 * n - 1, 1600)
    r0 = min(max_rows, max(80, max_rows // 2))
    rows = [int(k) for k in np.linspace(0, 2 * n - 2, r0, dtype=int)]
    row_set = set(rows)

    end_time = time.monotonic() + budget_s
    it = 0
    stall_count = 0
    patience = 25

    while time.monotonic() < end_time:
        it += 1
        rem = end_time - time.monotonic()
        if rem < 5.0:
            break

        conv_ff = convolve_fft(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0:
            break

        tl = max(2.0, min(40.0, rem * 0.30))
        g = lp_solve_direction(f, M, rows, n, tl)
        if g is None:
            break

        sg = float(g.sum())
        if sg <= 1e-12:
            break

        conv_fg = convolve_fft(f, g)
        violations = conv_fg - M
        max_viol = float(np.max(violations))

        if max_viol > 1e-9 * M:
            if len(rows) < max_rows:
                tol = 1e-9 * M
                order = np.argsort(violations)[::-1]
                target_add = max(1, (max_rows - len(rows)) // 3)
                added = 0
                for idx in order[:min(len(order), 512)]:
                    idx = int(idx)
                    if violations[idx] > tol and idx not in row_set:
                        rows.append(idx)
                        row_set.add(idx)
                        added += 1
                        if added >= target_add:
                            break
                continue
            else:
                break

        g = g * (S0 / sg)
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.8, 0.65, 0.5, 0.4, 0.3, 0.2, 0.12, 0.06):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score(fn)
            if sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()

        if accepted is None:
            break
        if accepted_score >= best_score * (1.0 - 1e-10) + 1e-12:
            stall_count += 1
            if stall_count >= patience:
                break
            continue

        f = accepted
        cur_score = actual_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            stall_count = 0

        if verbose and it % 3 == 0:
            print("  LP n=%d it=%d score=%.8f rows=%d" % (n, it, best_score, len(rows)))

    return normalize_nonneg(best_f), best_score


def adam_polish(p_init, budget_s, omega_start=35.0, omega_end=6.0, lr_scale=1.0, verbose=True):
    p = np.asarray(p_init, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf

    p = normalize_nonneg(p)
    end_time = time.monotonic() + budget_s

    if n <= 128:
        lr = 0.04 * lr_scale
    elif n <= 384:
        lr = 0.025 * lr_scale
    elif n <= 1024:
        lr = 0.015 * lr_scale
    else:
        lr = 0.010 * lr_scale

    x = np.log(p + 1e-300)
    x -= x.mean()
    x = np.clip(x, -60.0, 60.0)
    x -= x.max()

    mu, nu, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score(p)

    while time.monotonic() < end_time:
        t += 1
        elapsed = (time.monotonic() + budget_s - end_time) / max(1e-9, budget_s)
        frac = min(1.0, max(0.0, elapsed))
        omega = omega_start + (omega_end - omega_start) * frac

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_val = 2.0 * n * maxc
        if f_val < best_f:
            best_f = f_val
            best_p = p.copy()

        z = omega * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        Wf = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * Wf, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = 4.0 * n * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m_buf = mu * m_buf + (1.0 - mu) * grad_x
        v_buf = nu * v_buf + (1.0 - nu) * (grad_x * grad_x)
        m_hat = m_buf / (1.0 - mu ** t)
        v_hat = v_buf / (1.0 - nu ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = actual_score(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    if verbose:
        print("  Adam n=%d final_score=%.8f" % (n, best_f))
    return best_p, best_f

                                   omega_start=50.0, omega_end=8.0, lr_scale=0.7, verbose=True)
        record(p_out, f_out)

    # Phase 5: Short LP on absolute best
    rem = deadline - time.monotonic()
    if rem > 40 and best_p is not None:
        bn = best_p.size
        p_out, f_out = lp_cutting_plane(best_p, min(rem - 5, 180.0), verbose=True)
        record(p_out, f_out)

    # Final Adam if time
    rem = deadline - time.monotonic()
    if rem > 20 and best_p is not None:
        p_out, f_out = adam_polish(best_p, min(rem - 3, 60.0),
                                   omega_start=60.0, omega_end=10.0, lr_scale=0.5, verbose=False)
        record(p_out, f_out)

    # Validate with evaluate_sequence
    if best_p is not None:
        seq = scale_output(best_p, 10.0)
        gval = try_eval(seq)
        print("  Global eval: %.8f (local: %.8f)" % (gval, best_score))
        if np.isfinite(gval) and gval < best_score:
            best_score = gval

    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = actual_score(best_p)

    print("FINAL: n=%d score=%.8f" % (best_p.size, best_score))
    return scale_output(best_p, 10.0)
