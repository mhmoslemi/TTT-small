   **Approach: Alternating Linearization (ALA)**

   1. Fix a "reference" sequence a_ref.
   2. Linearize c[k] around a_ref: c[k] ≈ c_ref[k] + sum_i (a_ref[i]*a[k-i] + a_ref[k-i]*a[i] - a_ref[i]*a_ref[k-i])

      min t
      s.t. sum_i a_ref[i]*a[k-i] + sum_i a_ref[k-i]*a[i] - c_ref[k] <= t for all k
           sum(a) = 1, a >= 0
   4. Update a_ref = a, go to step 2.

   This is a "linearized" version that should converge to a local optimum. The LP can be solved efficiently with scipy or cvxpy.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative a via:
Phase 1: Projected subgradient (top-K exact) with heavy-ball momentum + Nesterov.
Phase 2: Active-set ALA (Alternating Linearization) solving small LPs per iteration.
Phase 3: Final polishing and verification.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not cond.any():
        return np.zeros(n, dtype=np.float64)
    rho_max = np.nonzero(cond)[0][-1] + 1
    theta = (cssv[rho_max - 1] - target) / rho_max
    return np.maximum(v - theta, 0.0)


def _to_sum1(a):
    s = a.sum()
    if s <= 1e-15:
        return np.full(a.size, 1.0 / a.size, dtype=np.float64)
    return a / s


def _conv_sq_fast(a, m):
    """Linear convolution a*a via FFT. m is next pow2 >= 2n-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * a.size - 1]


def _next_pow2(x):
    m = 1
    while m < x:
        m <<= 1
    return m


def _topk_subgradient(a, c, K):
    """Compute average subgradient of top-K lags of c."""
    n = a.size
    K = min(K, 2 * n - 1)
    # Find top K indices
    if K == 2 * n - 1:
        order = np.arange(2 * n - 1)
    else:
        order = np.argpartition(-c, K - 1)[:K]
    # Weight by value (proportional subgradient)
    wc = c[order]
    w = wc / wc.sum()
    grad = np.zeros(n, dtype=np.float64)
    for k, wk in zip(order, w):
        if wk <= 0:
            continue
        # grad[i] += 2 * wk * a[k-i] for valid i
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * wk * a[k - i]
    return grad


def _vec_topk_subgradient(a, c, K):
    """Vectorized top-K subgradient computation."""
    n = a.size
    K = min(K, 2 * n - 1)
    if K == 2 * n - 1:
        order = np.arange(2 * n - 1)
    else:
        order = np.argpartition(-c, K - 1)[:K]
    wc = c[order]
    w = wc / wc.sum()
    grad = np.zeros(n, dtype=np.float64)
    for idx in range(len(order)):
        k = order[idx]
        wk = 2.0 * w[idx]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        src = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] += wk * a[src]
    return grad


def _subgradient_phase(a, t_end, K=5, momentum=0.85, lr0=0.08, n_steps_max=500):
    """Phase 1: Projected subgradient with momentum."""
    n = a.size
    a = _to_sum1(np.abs(a.astype(np.float64)))
    m = _next_pow2(2 * n - 1)
    a = _to_sum1(a)
    buf = np.zeros(n, dtype=np.float64)
    best_a = a.copy()
    best_mc = _conv_sq_fast(a, m).max()
    lr = lr0
    step = 0
    stall_count = 0

    while step < n_steps_max and time.monotonic() < t_end - 30.0:
        c = _conv_sq_fast(a, m)
        mc = c.max()
        if mc < best_mc - 1e-9:
            best_mc = mc
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 100:
                lr *= 0.5
                stall_count = 0
                if lr < 1e-8:
                    break

        g = _vec_topk_subgradient(a, c, K)
        gn = np.linalg.norm(g)
        if gn > 1e-12:
            g = g / gn
        buf = momentum * buf - lr * g
        a_new = a + lr * buf
        a_new = _proj_simplex(a_new, 1.0)

        # Adaptive step size
        if step % 50 == 0:
            c_new = _conv_sq_fast(a_new, m).max()
            if c_new > mc * 1.01:
                lr *= 0.7
            elif step % 200 == 0:
                lr = max(1e-7, lr * 0.95)

        a = a_new
        step += 1

    return best_a, best_mc


def _build_ala_lp(a_ref, c_ref, active_set):
    """Build the ALA LP for the given active set of lags.
    Returns linprog args or None on failure.
    """
    n = a_ref.size
    K = len(active_set)
    if K == 0:
        return None

    # Variables: x = [a_0, ..., a_{n-1}, t]  (n+1 vars)
    # Constraints: for each k in active_set:
    #   2 * sum_i a_ref[i]*a[k-i] - t <= c_ref[k]
    # Equality: sum(a) = 1
    # Bounds: a[i] >= 0, t >= -inf (or 0)

    # Build sparse A_ub (K x (n+1))
    rows = []
    cols = []
    vals = []
    b_ub = np.zeros(K)

    for idx, k in enumerate(active_set):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            j = k - i  # column index for a[j]
            rows.append(idx)
            cols.append(j)
            vals.append(2.0 * a_ref[i])
        # t variable (column n)
        rows.append(idx)
        cols.append(n)
        vals.append(-1.0)
        b_ub[idx] = c_ref[k]

    if len(rows) == 0:
        return None

    A_ub = sparse.coo_matrix((vals, (rows, cols)), shape=(K, n + 1)).tocsr()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    bounds = [(0, None)] * n + [(-np.inf, None)]

    return A_ub, b_ub, A_eq, b_eq, c_obj, bounds


def _ala_phase(a, t_end, max_iter=40, K_active=80, damping=1.0):
    """Phase 2: ALA with active set LP."""
    n = a.size
    a = _to_sum1(np.abs(a.astype(np.float64)))
    m = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_mc = _conv_sq_fast(a, m).m_max() if hasattr(_conv_sq_fast(a, m), 'm_max') else _conv_sq_fast(a, m).max()
    best_mc = _conv_sq_fast(a, m).max()

    for it in range(max_iter):
        if time.monotonic() > t_end - 30.0:
            break

        c = _conv_sq_fast(a, m)
        mc = c.max()

        if mc < best_mc - 1e-10:
            best_mc = mc
            best_a = a.copy()

        # Select active set: top K_active lags
        K_act = min(K_active, 2 * n - 1)
        if K_act < 2 * n - 1:
            active = np.argpartition(-c, K_act - 1)[:K_act]
        else:
            active = np.arange(2 * n - 1)

        # Build and solve LP
        lp_args = _build_ala_lp(a, c, active)
        if lp_args is None:
            break
        A_ub, b_ub, A_eq, b_eq, c_obj, bounds = lp_args

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'time_limit': 20})
        except Exception as e:
            print(f"  ALA LP error at iter {it}: {e}")
            break

        if not res.success:
            # Try with fewer constraints
            if K_act > 20:
                K_act = max(20, K_act // 2)
                active = np.argpartition(-c, K_act - 1)[:K_act]
                lp_args = _build_ala_lp(a, c, active)
                if lp_args is None:
                    break
                A_ub, b_ub, A_eq, b_eq, c_obj, bounds = lp_args
                try:
                    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                  bounds=bounds, method='highs', options={'time_limit': 15})
                except Exception:
                    break
            if not res.success:
                break

        a_lp = res.x[:n]
        a_lp = np.maximum(a_lp, 0.0)
        a_lp = _to_sum1(a_lp)

        # Damped update
        a_new = (1.0 - damping) * a + damping * a_lp
        a_new = _proj_simplex(a_new, 1.0)

        c_new = _conv_sq_fast(a_new, m).max()

        if c_new < mc:
            a = a_new
        else:
            # Try with less damping
            if damping > 0.3:
                damping *= 0.7
            else:
                break
        # Slowly increase damping back
        damping = min(1.0, damping * 1.02)

    return best_a, best_mc


def _get_seeds(base, n):
    """Generate diverse seed sequences."""
    seeds = []
    seeds.append(base.copy())
    seeds.append(base[::-1].copy())

    # Linear ramp down
    ramp = np.linspace(1.0, 0.01, n)
    seeds.append(ramp)

    # Linear ramp up
    seeds.append(ramp[::-1].copy())

    # Quadratic ramp down
    quad = np.linspace(1.0, 0.01, n) ** 2
    seeds.append(quad)

    # Perturbation of base
    rng = np.random.default_rng(123)
    pert = base * np.exp(rng.normal(0, 0.1, n))
    pert = np.maximum(pert, 0)
    if pert.sum() > 0:
        seeds.append(pert)

    # Another perturbation
    pert2 = base * np.exp(rng.normal(0, 0.2, n))
    pert2 = np.maximum(pert2, 0)
    if pert2.sum() > 0:
        seeds.append(pert2)

    return [_to_sum1(np.abs(s)) for s in seeds]
