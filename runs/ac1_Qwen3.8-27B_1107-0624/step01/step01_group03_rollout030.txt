    """Upsample a sequence to a new length using interpolation."""
    old_n = len(s)
    if old_n == new_n:
        return s.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    s_new = np.interp(x_new, x_old, s)
    s_new = np.maximum(s_new, 0)
    if s_new.sum() > 1e-15:
        s_new = s_new / s_new.sum()
    return s_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid mirror descent + LP search to find low-peak autocorrelation sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    
    global_best_score = float('inf')
    global_best_seq = np.ones(256) / 256.0
    
    # Try to load height_sequence_1
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h / h.sum()
            print(f"Loaded height_sequence_1, length={h.size}, score={compute_score(hseq):.6f}")
    except Exception:
        pass
    
    # Sequence lengths to try (sorted by priority)
    # Focus on lengths that are likely to give good results
    lengths_to_try = [256, 128, 512, 192, 384, 1024, 320, 640]
    
    # Time allocation: more time for first (promising) lengths
    total_time = deadline - time.monotonic()
    # Allocate time proportionally to index weight
    weights = [30, 15, 20, 10, 10, 8, 5, 2]
    total_w = sum(weights)
    
    for idx, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 20:
            break
        time_alloc = total_time * weights[idx] / total_w
        n_deadline = time.monotonic() + time_alloc
        if n_deadline > deadline - 10:
            n_deadline = deadline - 10
        
        print(f"\n=== n={n}, time_alloc={time_alloc:.0f}s ===")
        
        # Gather initial candidates
        candidates = []
        
        # Parameterized families
        param_results = try_parameterized(n, rng)
        for sc, s in param_results[:10]:
            candidates.append((sc, s))
        
        # From height_sequence_1 (upsampled/downsampled)
        if hseq is not None:
            h_resized = upsample(hseq, n)
            candidates.append((compute_score(h_resized), h_resized))
        
        # Sort and take top
        candidates.sort(key=lambda x: x[0])
        
        # Optimize top candidates
        max_cands = min(4, len(candidates))
        cand_deadline = n_deadline - 10
        cand_time = max(5, (cand_deadline - time.monotonic()) / max_cands)
        
        for c_idx in range(max_cands):
            if time.monotonic() > cand_deadline:
                break
            sc0, s0 = candidates[c_idx]
            print(f"  Candidate {c_idx}: initial score={sc0:.6f}")
            
            # Main optimization
            best_s, best_sc = optimize_sequence(
                s0, n, rng, 
                time.monotonic() + cand_time,
                use_lp=True,
                max_iters=3000
            )
            
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** New global best: {global_best_score:.6f} (n={n}, cand={c_idx})")
            
            # LP polishing if time remains
            if time.monotonic() < cand_deadline - 5 and best_sc < global_best_score + 0.01:
                best_s2, best_sc2 = optimize_with_lp_only(
                    best_s, n, time.monotonic() + max(5, cand_deadline - time.monotonic() - 3)
                )
                if best_sc2 < global_best_score:
                    global_best_score = best_sc2
                    global_best_seq = best_s2.copy()
                    print(f"  ** After LP polish: {global_best_score:.6f}")
        
        # Additional random restarts if time permits
        restarts = 0
        while time.monotonic() < n_deadline - 10 and restarts < 3:
            r = rng.random(n) + 0.05
            r = r / r.sum()
            best_s, best_sc = optimize_sequence(
                r, n, rng, 
                time.monotonic() + max(5, (n_deadline - 10 - time.monotonic()) / (3 - restarts)),
                use_lp=True,
                max_iters=1000
            )
            restarts += 1
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Restart {restarts}: {global_best_score:.6f}")
    
    # Final polish on the best sequence with pure LP
    if time.monotonic() < deadline - 5:
        print(f"\n=== Final LP polish, n={len(global_best_seq)} ===")
        final_s, final_sc = optimize_with_lp_only(
            global_best_seq, len(global_best_seq), 
            time.monotonic() + max(5, deadline - time.monotonic() - 2),
            max_iters=100
        )
        if final_sc < global_best_score:
            global_best_score = final_sc
            global_best_seq = final_s.copy()
            print(f"  Final after polish: {global_best_score:.6f}")
    
    # Verify with actual evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_verified = evaluate_sequence(result)
    print(f"\nFinal verified score: {final_verified:.6f}, length={len(result)}")
    
    # Use the verified score's sequence
    if np.isinf(final_verified):
        # Fallback to uniform
        result = [1.0] * 256
    
    return result
