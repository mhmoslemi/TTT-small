```python
def subgradient_step(a, top_k=10, lr=0.01):
    n = len(a)
    c = np.convolve(a, a)
    # Find top k peaks
    peak_indices = np.argsort(c)[-top_k:]
    
    # Compute gradient: for each peak at lag p, grad[i] += a[p-i] (if valid)
    grad = np.zeros(n)
    for p in peak_indices:
        for i in range(n):
            j = p - i
            if 0 <= j < n:
                grad[i] += a[j]
    
    # Normalize gradient
    g_norm = np.linalg.norm(grad)
    if g_norm < 1e-15:
        return a
    
    # Move opposite to gradient, maintain sum=1
    a_new = a - lr * grad / g_norm
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-10:
        return a
    a_new /= a_new.sum()
    return a_new
```

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected subgradient for shape finding, LP-based refinement for polishing,
and peak equalization to drive all prominent convolution lags toward a common value.
Multiple initializations and sequence lengths are explored within the time budget.

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ d = convolve(a, d)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve_direction(a, M, n_peaks=5):
    """
    Solve LP: max sum(d) s.t. conv(a, d) <= M - conv(a,a), d >= -a, sum(d)=0
    This finds a direction that reduces the peak while maintaining feasibility.
    Returns d or None.
    """
    n = len(a)
    c_current = np.convolve(a, a)
    A = build_conv_matrix(a)
    # conv(a, d) <= M - c_current (elementwise)
    rhs = np.full(2 * n - 1, M) - c_current
    # Only use constraints that are near-active (top peaks)
    # Use all constraints for correctness but the LP will handle it
    # Maximize sum(d) (so minimize -sum(d))
    c_obj = np.zeros(n + 1)  # d variables + slack for sum constraint handled by equality
    # Actually let's just do: max sum(d) s.t. A@d <= rhs, -a <= d, sum(d) = 0
    # Convert to standard form for linprog
    c_obj = -np.ones(n)
    A_ub = np.vstack([A, -np.eye(n)])
    b_ub = np.concatenate([rhs, a])  # d <= a isn't needed, we need d >= -a
    # d >= -a means -d <= a, so row of -I with bound a
    A_eq = np.array([[1.0] * n])
    b_eq = np.array([0.0])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(-a[i], None) for i in range(n)], method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_solve_target(a, M):
    """
    Solve LP: max sum(g) s.t. conv(a, g) <= M, g >= 0.
    Returns g or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def subgradient_optimize(a, deadline, top_k=15, lr_init=0.03):
    """Projected subgradient method with adaptive step size."""
    n = len(a)
    a_work = np.clip(a, 1e-15, None).astype(np.float64)
    a_work /= a_work.sum()

    best_a = a_work.copy()
    best_score = compute_objective(a_work)
    lr = lr_init
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a_work, a_work)
        c_max = c.max()
        # Find peaks above 90% of max
        threshold = c_max * 0.90
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) > top_k:
            order = np.argsort(c[peak_indices])[::-1][:top_k]
            peak_indices = peak_indices[order]

        # Compute subgradient: grad[k] = sum_{p in peaks} 2*a[p-k] for valid p-k
        grad = np.zeros(n)
        for p in peak_indices:
            k_min = max(0, p - n + 1)
            k_max = min(p, n - 1)
            j = p - np.arange(k_min, k_max + 1)
            grad[k_min:k_max + 1] += 2.0 * a_work[j]

        # Normalize step
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        a_new = a_work - lr * grad / g_norm
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            a_new = best_a
        else:
            a_new /= a_new.sum()

        sc = compute_objective(a_new)
        if sc < best_score - 1e-10:
            best_a = a_new.copy()
            best_score = sc
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.9

        a_work = a_new

        if it % 5000 == 0:
            print(f"      subgrad iter={it}, best={best_score:.6f}, lr={lr:.5f}, n_peaks={len(peak_indices)}")

            break

    return best_a, best_score


def lp_optimize(a, deadline, rng):

def peak_equalize(a, deadline, rng, n_rounds=50):
    """
    Iteratively equalize the top peaks of the convolution.
    At each step, identify the highest peak and the lowest non-zero peak,
    then adjust the sequence to reduce the gap.
    """
    n = len(a)
    a_work = np.clip(a, 1e-15, None).astype(np.float64)
    a_work /= a_work.sum()

    best_a = a_work.copy()
    best_score = compute_objective(a_work)
    it = 0

    while time.monotonic() < deadline and it < n_rounds * 10:
        it += 1
        c = np.convolve(a_work, a_work)
        c_max = c.max()
        c_min_nonzero = c.min()

        # Find the top peaks
        top_peaks = np.argsort(c)[::-1][:5]

        # Compute gradient for top peaks
        grad = np.zeros(n)
        for p in top_peaks:
            k_min = max(0, p - n + 1)
            k_max = min(p, n - 1)
            j = p - np.arange(k_min, k_max + 1)
            grad[k_min:k_max + 1] += a_work[j]

        # Also add a small "spreading" force: increase values at positions
        # that contribute to low peaks
        # Find lags where c is below average
        c_avg = c.mean()
        low_peaks = np.where(c < c_avg * 0.5)[0]
        if len(low_peaks) > 0:
            # Positions that contribute to low peaks should be increased
            spread_grad = np.zeros(n)
            for p in low_peaks[:5]:
                k_min = max(0, p - n + 1)
                k_max = min(p, n - 1)
                j = p - np.arange(k_min, k_max + 1)
                spread_grad[k_min:k_max + 1] += a_work[j]
            # Reduce spread_grad contribution (we want to increase these positions)
            grad -= 0.3 * spread_grad

        # Move opposite to gradient
        lr = 0.01
        a_new = a_work - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            break
        a_new /= a_new.sum()

        sc = compute_objective(a_new)
        if sc < best_score:
            best_a = a_new.copy()
            best_score = sc
            a_work = a_new
        else:
            lr *= 0.9
            if lr < 1e-6:
                break

    return best_a, best_score


def make_init_exponential(n, alpha):
    """Exponential decay: a[k] = exp(-alpha * k / n)."""
    k = np.arange(n)
    return np.exp(-alpha * k / n)


def make_init_bilateral(n, alpha, asym=0.0):
    """Bilateral exponential: high on both sides with asymmetry."""
    k = np.arange(n)
    center = n / 2.0
    left = np.exp(-alpha * np.abs(k - center + asym * n / 2) / n)
    return left


def make_init_power(n, beta):
    """Power law: a[k] = (1 + k/n)^(-beta)."""
    k = np.arange(n)
    return (1.0 + k / n) ** (-beta)


def make_init_trapezoid(n, frac=0.3):
    """Trapezoidal window (asymmetric option)."""
    k = np.arange(n)
    rise = int(n * frac)
    fall = n - rise
    a = np.zeros(n)
    a[:rise] = np.linspace(0.5, 1.0, rise)
    a[rise:] = np.linspace(1.0, 0.1, fall)
    return a


def make_init_smooth(n, gamma=2.0, offset=0.0):
    """Smooth asymmetric: a[k] = 1/(1+exp(gamma*(k/n - 0.5 + offset)))."""
    k = np.arange(n)
    return 1.0 / (1.0 + np.exp(gamma * (k / n - 0.5 + offset)))


def make_init_cosine_mod(n, phi=0.0):
    """Modulated cosine: a[k] = 0.5 + 0.5*cos(2*pi*k/n + phi)."""
    k = np.arange(n)

    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using hybrid optimization
    (subgradient + peak equalization + LP) with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0  # buffer for return

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p
    except Exception:
        pass

    # Define initializations: (n, init_func, init_args)
    inits = []

    # Parent-based
    if parent is not None:
        pn = parent.size
        inits.append((pn, lambda n, p=parent: p.copy(), 'parent'))
        # Truncated parent
        inits.append((min(pn, 512), lambda n, p=parent: p[:n].copy(), 'parent_trunc'))

    # Theory-motivated initializations
    for n in [256, 512]:
        # Exponential decay (various rates)
        for alpha in [2.0, 4.0, 6.0, 8.0]:
            inits.append((n, lambda n, a=alpha: make_init_exponential(n, a), f'exp_{alpha}'))
        # Power law
        for beta in [1.0, 2.0, 3.0]:
            inits.append((n, lambda n, b=beta: make_init_power(n, b), f'pow_{beta}'))
        # Smooth asymmetric
        for gamma in [2.0, 4.0, 6.0]:
            for offset in [-0.2, -0.1, 0.1, 0.2]:
                inits.append((n, lambda n, g=gamma, o=offset: make_init_smooth(n, g, o),
                              f'smooth_{gamma}_{offset}'))
        # Trapezoid
        for frac in [0.2, 0.3, 0.5]:
            inits.append((n, lambda n, f=frac: make_init_trapezoid(n, f), f'trap_{frac}'))
        # Cosine modulated
        for phi in [0.0, np.pi/4, np.pi/2, -np.pi/4]:
            inits.append((n, lambda n, ph=phi: make_init_cosine_mod(n, ph), f'cos_{ph:.2f}'))

    # Larger n for more degrees of freedom
    for n in [1024]:
        for alpha in [3.0, 5.0]:
            inits.append((n, lambda n, a=alpha: make_init_exponential(n, a), f'exp_large_{alpha}'))
        for beta in [1.5, 2.5]:
            inits.append((n, lambda n, b=beta: make_init_power(n, b), f'pow_large_{beta}'))
        for offset in [-0.15, 0, 0.15]:
            inits.append((n, lambda n, o=offset: make_init_smooth(n, 4.0, o),
                          f'smooth_large_{offset}'))

    # Random initializations
    for n in [256, 512]:
        def make_rand(n, rng_local=rng):
            return np.abs(rng_local.normal(0, 1, n)) + 0.1
        inits.append((n, make_rand, f'random_{n}'))

    print(f"Total {len(inits)} initializations to try, budget={budget_s}s")
    print(f"Parent available: {parent is not None}")

    # Time allocation: parent gets more time, then theory-motivated, then random
    # Weight: parent=3, theory=1, random=0.5
    weights = []
    for n, init_f, name in inits:
        if 'parent' in name:
            weights.append(3.0)
        elif 'large' in name:
            weights.append(1.5)
        elif 'random' in name:
            weights.append(0.5)
        else:
            weights.append(1.0)

    total_w = sum(weights)
    available = deadline - time.monotonic()

    # Sort by weight (heaviest first) to ensure best candidates get time
    # But keep parent at the front
    idxs = list(range(len(inits)))
    # Parent first, then others by weight
    idxs.sort(key=lambda i: (-weights[i], i))
    # Force parent to front
    parent_idxs = [i for i in range(len(inits)) if 'parent' in inits[i][2]]
    other_idxs = [i for i in idxs if i not in parent_idxs]
    idxs = parent_idxs + other_idxs

    # Allocate time
    time_per_unit = available / total_w
    cumulative = 0.0
    time_allocs = {}
    for i in idxs:
        cumulative += time_per_unit * weights[i]
        time_allocs[i] = min(start_time + cumulative, deadline)

    # Limit per-init time to avoid spending too long on one
    max_per_init = min(120.0, available / max(3, len(inits)) * 3)

    for idx in idxs:
        n, init_f, name = inits[idx]
        if time.monotonic() > deadline:
            break

        init_deadline = min(time_allocs[idx], time.monotonic() + max_per_init, deadline)
        if init_deadline <= time.monotonic() + 5:
            continue

        try:
            a_init = init_f(n)
        except Exception:
            a_init = np.ones(n)

        a_init = np.clip(a_init, 1e-10, None).astype(np.float64)
        if a_init.sum() < 1e-10:
            a_init = np.ones(n)

        elapsed = time.monotonic() - start_time
        remaining = deadline - time.monotonic()
        print(f"[{idx}] n={n}, init={name}, remaining~{remaining:.0f}s", flush=True)

        # For small n, use full hybrid; for large n, use LP only
        if n <= 512:
            a_opt, score = optimize_hybrid(a_init, init_deadline, rng)
        else:
            # For large n, skip subgradient (too slow) and go straight to LP
            # But first do a few subgradient steps
            t_sub = min(time.monotonic() + 30, init_deadline - 30)
            if t_sub > time.monotonic():
                a_init, s_sub = subgradient_optimize(a_init, t_sub, top_k=10, lr_init=0.03)
                if s_sub < compute_objective(a_init):
                    pass  # already updated
            a_opt, score = lp_optimize(a_init, init_deadline, rng)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: score={best_score:.6f} (n={n}, init={name})", flush=True)
        else:
            print(f"  => score={score:.6f} (best={best_score:.6f})", flush=True)

    # Final polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\nFinal polish on best (score={best_score:.6f}, n={len(best_seq)})...", flush=True)
        remaining_time = deadline - time.monotonic()

        # Phase 1: More LP iterations
        polish_deadline_1 = time.monotonic() + 0.6 * remaining_time
        best_seq, s1 = lp_optimize(best_seq, polish_deadline_1, rng)
        print(f"  After LP polish: {s1:.6f}", flush=True)
        if s1 < best_score:
            best_score = s1

        # Phase 2: Subgradient fine-tuning
        if time.monotonic() < deadline - 30:
            polish_deadline_2 = deadline
            best_seq, s2 = subgradient_optimize(best_seq, polish_deadline_2,
                                               top_k=20, lr_init=0.01)
            print(f"  After subgrad polish: {s2:.6f}", flush=True)
            if s2 < best_score:
                best_score = s2

        # Phase 3: One more LP pass
        if time.monotonic() < deadline - 10:
            best_seq, s3 = lp_optimize(best_seq, deadline, rng)
            print(f"  After final LP: {s3:.6f}", flush=True)
            if s3 < best_score:
                best_score = s3

    # Also try a few random restarts if time permits
    if time.monotonic() < deadline - 120:
        print(f"\nRandom restarts (remaining={deadline - time.monotonic():.0f}s)", flush=True)
        for _ in range(5):
            if time.monotonic() > deadline - 30:
                break
            n_rand = rng.choice([256, 384, 512])
            a_rand = np.abs(rng.normal(0, 1, n_rand)) + 0.05
            a_rand /= a_rand.sum()
            # Quick optimization
            t_dead = min(time.monotonic() + 60, deadline - 10)
            a_opt, sc = lp_optimize(a_rand, t_dead, rng)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Random restart improved: {best_score:.6f}", flush=True)

    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0

    # Ensure valid for evaluator
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    elapsed_total = time.monotonic() - start_time
    print(f"\nDONE in {elapsed_total:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
```
