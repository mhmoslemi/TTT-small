def _coord_descent(p, deadline, rng):

    while time.monotonic() < deadline:
        improved = False
        for _ in range(10):  # 10 passes
            if time.monotonic() > deadline:
                break
            for j in range(N_local):
                if time.monotonic() > deadline:
                    break
                # Try moving mass from j to j+1 or j-1
                for delta in [0.001, 0.01, 0.05]:
                    if p[j] < delta:
                        continue
                    p_new = p.copy()
                    p_new[j] -= delta
                    target = (j + 1) if j < N_local - 1 else (j - 1)
                    if target < 0:
                        target = j + 1
                    p_new[target] += delta
                    if p_new[j] < 0:
                        continue
                    s = _score_p(p_new)
                    if s < score - 1e-12:
                        p = p_new
                        score = s
                        _best_score = score
                        _best_seq = p.copy()
                        improved = True
                        break
        if not improved:
            break
    return p

Search for a non-negative sequence minimizing 2*n*max(a*a)/(sum a)^2.
Uses logit parameterization, L-BFGS-B with adaptive top-K log-sum-exp smoothing,
directed subgradient polishing with line search, and multi-start strategy.

import numpy as np
import time
from scipy.optimize import minimize

N = 4096
_FFT_SELF = 8192
_FFT_GRAD = 16384
_BOUNDS = [(-50.0, 50.0)] * (N - 1)

_best_score = float('inf')
_best_seq = None
_deadline = 0.0


class _Timeout(Exception):
    pass


def _to_p(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)
    z = np.clip(z, -50.0, 50.0)
    m = float(z.max()) if z.size else 0.0
    e = np.exp(z - m)
    Z = 1.0 + float(e.sum())
    p = np.empty(N, dtype=np.float64)
    p[0] = 1.0 / Z
    p[1:] = e / Z
    return p


def _to_z(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 1e-30)
    z = np.log(p[1:]) - np.log(p[0])
    return np.clip(z, -50.0, 50.0)


def _conv(p):
    f = np.fft.rfft(p, _FFT_SELF)
    c = np.fft.irfft(f * f, _FFT_SELF)[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_p(p):
    return 2.0 * N * float(_conv(p).max())


def _grad_p(p, w):
    w_flip = w[::-1]
    fp = np.fft.rfft(p, _FFT_GRAD)
    fw = np.fft.rfft(w_flip, _FFT_GRAD)
    conv = np.fft.irfft(fp * fw, _FFT_GRAD)
    idx = 2 * N - 2 - np.arange(N)

        topk = c[topk_idx]

        topk = c.copy()

    m = float(topk.max())

    e = np.exp(shifted)
    s = float(e.sum())

        J = m
        w = np.zeros(2 * N - 1)
        w[topk_idx[int(np.argmax(topk))]] = 1.0
    else:
        J = m + float(np.log(s)) / alpha
        w = np.zeros(2 * N - 1)
        w[topk_idx] = e / s

    gp = _grad_p(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.nan_to_num(gz, nan=0.0, posinf=0.0, neginf=0.0)
    gz = np.clip(gz, -1e8, 1e8)
    return float(J), gz


def _optimize(z0, alpha, K, maxfun, deadline):
    global _deadline
    _deadline = float(deadline)
    z0 = np.nan_to_num(z0, nan=0.0, posinf=40.0, neginf=-40.0)
    z0 = np.clip(z0, -50.0, 50.0)
    try:
        minimize(_obj_grad, z0, args=(float(alpha), int(K)),
                 method='L-BFGS-B', jac=True, bounds=_BOUNDS,
                 options={'maxfun': int(maxfun), 'maxiter': int(maxfun),
                          'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50,
                          'disp': False})
    except _Timeout:
        pass
    except Exception:
        pass


def _subgradient_polish(z0, deadline, rng):
    global _best_score, _best_seq
    z = _to_z(_to_p(z0))
    p = _to_p(z)
    score = _score_p(p)
    _update_best(p, score)

    K = 20
    noise_scale = 0.15
    fails = 0
    max_fails = 300

    while time.monotonic() < deadline:
        c = _conv(p)
        topk_idx = np.argpartition(c, -K)[-K:]
        topk = c[topk_idx]
        w = np.zeros(2 * N - 1)
        w[topk_idx] = topk / float(topk.sum())

        gp = _grad_p(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gn = float(np.linalg.norm(gz))

        accepted = False
        if gn > 1e-15:
            d = -gz / gn
            for step in [0.8, 0.4, 0.2, 0.1, 0.05, 0.025]:
                zt = z + step * d
                zt = np.clip(zt, -50.0, 50.0)
                pt = _to_p(zt)
                st = _score_p(pt)
                if st < score - 1e-13:
                    z, p, score = zt, pt, st
                    _update_best(p, score)
                    accepted = True
                    break

        if not accepted:
            best_local = score
            best_local_z = z.copy()
            best_local_p = p.copy()
            n_trials = 8
            for _ in range(n_trials):
                zt = z + rng.normal(0.0, noise_scale, N - 1)
                zt = np.clip(zt, -50.0, 50.0)
                pt = _to_p(zt)
                st = _score_p(pt)
                if st < best_local - 1e-13:
                    best_local = st
                    best_local_z = zt.copy()
                    best_local_p = pt.copy()
            if best_local < score - 1e-13:
                z, p, score = best_local_z, best_local_p, best_local
                _update_best(p, score)
                accepted = True

        if accepted:
            fails = 0
            noise_scale = min(0.4, noise_scale * 1.2)
        else:
            fails += 1
            noise_scale = max(0.005, noise_scale * 0.75)
            if fails > max_fails:
                break


def _make_gauss(beta, center=0.5):
    x = (np.arange(N) + 0.5) / N
    f = np.exp(-beta * (x - center) ** 2)
    f = np.maximum(f, 1e-15)
    return _to_z(f / f.sum())


def _make_cosh(beta, center=0.5):
    x = (np.arange(N) + 0.5) / N
    f = 1.0 / np.cosh(beta * (x - center))
    f = np.maximum(f, 1e-15)
    return _to_z(f / f.sum())


def _make_asym(a, b):
    x = (np.arange(N) + 0.5) / N
    f = np.exp(-a * x) * np.exp(-b * (1 - x))
    f = np.maximum(f, 1e-15)
    return _to_z(f / f.sum())


def _make_mix(rng):
    x = (np.arange(N) + 0.5) / N
    c1 = 0.1 + 0.8 * rng.random()
    c2 = 1.0 - c1
    w1 = rng.random()
    w2 = 1.0 - w1
    f = w1 * np.exp(-3.0 / w1 * (x - c1) ** 2) + w2 * np.exp(-3.0 / w2 * (x - c2) ** 2)
    f = np.maximum(f, 1e-15)
    return _to_z(f / f.sum())


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_seq, _deadline

    start = time.monotonic()
    bud = float(budget_s)
    end = start + bud - 4.0

    _best_score = float('inf')
    _best_seq = None
    rng = np.random.default_rng(seed)

    z_start = None
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.maximum(base, 0.0)
        if base.size == N and base.sum() > 1e-10:
            p0 = base / base.sum()
            z_start = _to_z(p0)
            s0 = _score_p(p0)
            _update_best(p0, s0)
            print(f"Loaded height_sequence_1: score={s0:.6f}")
        elif base.size > 0 and base.sum() > 1e-10:
            old_x = np.linspace(0, 1, base.size)
            new_x = np.linspace(0, 1, N)
            b_interp = np.interp(new_x, old_x, base)
            b_interp = np.maximum(b_interp, 0.0)
            p0 = b_interp / b_interp.sum()
            z_start = _to_z(p0)
            s0 = _score_p(p0)
            _update_best(p0, s0)
            print(f"Interpolated height_sequence_1: score={s0:.6f}")
    except Exception as e:
        print(f"No height_sequence_1: {e}")

    if z_start is None:
        z_start = _make_gauss(10.0)
        p0 = _to_p(z_start)
        _update_best(p0, _score_p(p0))
        print(f"Using Gaussian start: score={_best_score:.6f}")

    # Phase 1: Coarse-to-fine L-BFGS-B optimization
    # Schedule: (alpha, K, weight)
    schedule = [
        (5e4, 200, 1),
        (1e5, 100, 1),
        (3e5, 50, 2),
        (1e6, 30, 2),
        (3e6, 20, 3),
        (1e7, 15, 3),
        (3e7, 10, 4),
        (1e8, 8, 4),
        (3e8, 6, 5),
        (1e9, 5, 5),
    ]

    phase1_frac = 0.55
    phase1_end = start + phase1_frac * bud

    # Build expanded schedule based on time
    total_weight = sum(s[2] for s in schedule)
    available_runs = max(10, int((phase1_frac * bud) / 8.0))

    run_queue = []
    for alpha, K, w in schedule:
        n_runs = max(1, int(available_runs * w / total_weight))
        for _ in range(n_runs):
            run_queue.append((alpha, K))

    rng.shuffle(run_queue)

    last_improve = _best_score
    run_count = 0

    for alpha, K in run_queue:
        if time.monotonic() > phase1_end - 15:
            break
        rem = phase1_end - time.monotonic()
        if rem < 15:
            break

        # Choose starting point
        r = rng.random()
        if r < 0.6 and _best_seq is not None:
            # Perturb best
            noise = 0.03 + 0.1 * (run_count / max(1, len(run_queue)))
            z0 = _to_z(_best_seq) + rng.normal(0, noise, N - 1)
        elif r < 0.8:
            # Different Gaussian width
            z0 = _make_gauss(3.0 + 15.0 * rng.random())
        elif r < 0.9:
            z0 = _make_cosh(5.0 + 15.0 * rng.random())
        elif r < 0.95:
            z0 = _make_asym(1.0 + 8.0 * rng.random(), 1.0 + 8.0 * rng.random())
        else:
            z0 = _make_mix(rng)

        maxfun = int(min(5000, max(300, rem / 0.06)))
        _optimize(z0, alpha, K, maxfun, phase1_end)
        run_count += 1

        if _best_score < last_improve - 1e-7:
            elapsed = time.monotonic() - start
            print(f"  Run {run_count} (alpha={alpha:.0e}, K={K}): "
                  f"score={_best_score:.6f} (delta={last_improve - _best_score:.2e}) "
                  f"t={elapsed:.0f}s")
            last_improve = _best_score

    print(f"Phase 1 done: best={_best_score:.6f} t={time.monotonic()-start:.0f}s")

    # Phase 2: Subgradient polishing
    phase2_end = start + 0.82 * bud
    if _best_seq is not None and time.monotonic() < phase2_end:
        z_best = _to_z(_best_seq)
        _subgradient_polish(z_best, phase2_end, rng)
        print(f"Phase 2 done: best={_best_score:.6f} t={time.monotonic()-start:.0f}s")

    # Phase 3: Fine L-BFGS-B with very high alpha
    phase3_end = start + 0.93 * bud
    if _best_seq is not None and time.monotonic() < phase3_end - 30:
        z_best = _to_z(_best_seq)
        for alpha, K in [(5e8, 5), (2e9, 4), (1e10, 3)]:
            if time.monotonic() > phase3_end - 20:
                break
            z0 = z_best + rng.normal(0, 0.02, N - 1)
            rem = phase3_end - time.monotonic()
            maxfun = int(min(3000, max(200, rem / 0.05)))
            _optimize(z0, alpha, K, maxfun, phase3_end)
            z_best = _to_z(_best_seq) if _best_seq is not None else z0
        print(f"Phase 3 done: best={_best_score:.6f} t={time.monotonic()-start:.0f}s")

    # Phase 4: Final subgradient + L-BFGS
    if _best_seq is not None and time.monotonic() < end - 20:
        z_best = _to_z(_best_seq)
        _subgradient_polish(z_best, min(end - 10, time.monotonic() + 0.05 * (end - time.monotonic())), rng)
        if time.monotonic() < end - 10:
            z_final = _to_z(_best_seq)
            _optimize(z_final, 1e9, 4, 500, end - 3)

    # Final: extract best sequence
    if _best_seq is not None:
        seq = [float(v) for v in _best_seq]
    else:
        seq = [1.0 / N] * N

    try:
        fs = evaluate_sequence(seq)
        print(f"evaluate_sequence: {fs:.6f}")
    except Exception:
        fs = float('inf')

    if not np.isfinite(fs) or fs > 2.0001:
        seq = [1.0 / N] * N
        try:
            fs = evaluate_sequence(seq)
        except Exception:
            fs = 2.0

    # Safety: compare with height_sequence_1
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.maximum(base, 0.0)
        if base.size > 0:
            bs = evaluate_sequence([float(v) for v in base])
            if np.isfinite(bs) and bs < fs:
                seq = [float(v) for v in base]
                fs = bs
                print(f"Using height_sequence_1: {fs:.6f}")
    except Exception:
        pass

    print(f"DONE: score={fs:.6f} n={len(seq)}")
    return seq
