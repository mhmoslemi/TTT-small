        frac = min(1.0, elapsed / max(1e-6, budget_s))

        # Alpha schedule: start high, decay to low
        alpha = 3.0 + 30.0 * np.exp(-6.0 * frac)

        # Nesterov: y = x + momentum * (x - x_prev)
        # Simplified: just use x directly (Adam already has momentum)
        p_curr = _softmax(x)
        if not np.all(np.isfinite(p_curr)) or p_curr.sum() < 1e-12:
            break

        grad_p, _, _ = gradient_smoothed(p_curr, alpha, L)
        if not np.all(np.isfinite(grad_p)):
            break

        gp = float(np.dot(p_curr, grad_p))
        grad_x = p_curr * (grad_p - gp)

        # Adam update
        m_buf = mu1 * m_buf + (1.0 - mu1) * grad_x
        v_buf = mu2 * v_buf + (1.0 - mu2) * (grad_x * grad_x)
        m_hat = m_buf / (1.0 - mu1**t)
        v_hat = v_buf / (1.0 - mu2**t)

        # Cosine annealing with warm restarts
        T = 3000
        cycle = t % T
        lr = lr0 * 0.5 * (1.0 + np.cos(np.pi * cycle / T))
        lr = max(lr, 0.001)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -50.0, 50.0)
        x -= x.mean()

        p_curr = _softmax(x)

        f_curr = score_fft(p_curr)
        if np.isfinite(f_curr) and f_curr < best_f:
            best_f = f_curr
            best_p = p_curr.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                # Restart with perturbation
                x = np.log(best_p + 1e-30)
                x -= x.mean()
                x += 0.1 * np.random.default_rng(t).normal(0, 1, n)
                x -= x.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                stall_count = 0

    final_f = score_fft(best_p)
    return best_p, final_f

def _softmax(x):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-30:
        return np.ones(x.size) / x.size
    return e / s

def lp_direction(f_cur, M, max_rows=500, time_limit=30.0):
    """Solve the LP: max sum(g) s.t. f*g <= M, g >= 0. Returns g or None."""
    n = f_cur.size
    if n < 2:
        return None

    # Build constraint matrix for a subset of rows
    if max_rows >= 2*n - 1:
        rows = np.arange(2*n - 1)
    else:
        # Stratified sampling: include endpoints and uniform middle
        rows = np.unique(np.linspace(0, 2*n-2, max_rows).astype(int))

    A = np.zeros((len(rows), n), dtype=np.float64)
    for ii, k in enumerate(rows):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            # A[ii, m] = f[k - m] for m in [lo, hi]
            idx = k - np.arange(lo, hi+1)
            A[ii, lo:hi+1] = f_cur[idx]

    c_obj = -np.ones(n)
    b_ub = np.full(len(rows), M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None),
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res is not None and res.success and res.x is not None:
            g = np.clip(res.x, 0.0, None)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None

def lp_polish(p0, budget_s, seed=0):
    """LP cutting-plane Frank-Wolfe polishing."""
    p = npng(p0)
    n = p.size
    if n < 2 or budget_s < 10.0:
        return p, score_fft(p)

    best_p = p.copy()
    best_f = score_fft(p)
    L = next_power_of_two(2*n - 1)

    # Determine max rows based on n
    if n > 2000:
        max_rows = 600
    elif n > 1000:
        max_rows = 800
    else:
        max_rows = 2*n - 1  # use all constraints for small n

    # Adaptive time limit per LP solve
    base_tl = min(60.0, max(5.0, budget_s / 30.0))

    start = time.monotonic()
    end = start + budget_s

    it = 0
    while time.monotonic() < end - 3.0:
        it += 1
        c = full_convolution_fft(p, L)
        M = float(c.max())
        if M <= 0:
            break

        # LP direction
        rem = end - time.monotonic()
        tl = min(base_tl, max(2.0, rem * 0.4))
        g = lp_direction(p, M, max_rows=max_rows, time_limit=tl)
        if g is None:
            break

        sg = float(g.sum())
        if sg <= 1e-12:
            break

        # Normalize g to sum = 1
        g = g / sg

        # Verify full constraints (if we didn't use all rows)
        if max_rows < 2*n - 1:
            L2 = next_power_of_two(2*n - 1)
            Pg = np.fft.rfft(g, L2)
            Pf = np.fft.rfft(p, L2)
            conv_check = np.fft.irfft(Pf * Pg, L2)[:2*n-1].real
            conv_check = np.maximum(conv_check, 0.0)
            viol = float(conv_check.max()) - M
            if viol > 0.01 * M:
                # Direction violates constraints; skip
                continue

        # Line search
        best_step_val = None
        best_score_so_far = best_f
        for step in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02):
            fn = (1.0 - step) * p + step * g
            fn = np.clip(fn, 0.0, None)
            sn = fn.sum()
            if sn < 1e-12:
                continue
            fn = fn / sn
            sc = score_fft(fn)
            if np.isfinite(sc) and sc < best_score_so_far:
                best_score_so_far = sc
                best_step_val = fn.copy()

        if best_step_val is None or not np.isfinite(best_score_so_far):
            # Try a small gradient step instead
            grad_p, _, _ = gradient_smoothed(p, 10.0, L)
            gp = float(np.dot(p, grad_p))
            grad_x = p * (grad_p - gp)
            # Simple step
            for step in (0.01, 0.05, 0.1, 0.2):
                fn = p - step * grad_x
                fn = np.clip(fn, 0.0, None)
                sn = fn.sum()
                if sn < 1e-12:
                    continue
                fn = fn / sn
                sc = score_fft(fn)
                if np.isfinite(sc) and sc < best_f:
                    best_step_val = fn.copy()
                    best_score_so_far = sc
                    break
            if best_step_val is None:
                break

        p = best_step_val
        if best_score_so_far < best_f:
            best_f = best_score_so_far
            best_p = p.copy()
            if it % 3 == 0:
                print("  LP iter %d: score=%.6f" % (it, best_f))
        else:
            break

    return best_p, best_f

def local_refine(p0, budget_s, seed=0):
    """Use scipy.optimize.minimize for local refinement."""
    p = npng(p0)
    n = p.size
    if n < 2 or budget_s < 5.0:
        return p, score_fft(p)

    best_p = p.copy()
    best_f = score_fft(p)

    def objective(x):
        f = _softmax(x)
        return score_fft(f)

    x0 = np.log(p + 1e-30)
    x0 -= x0.mean()

    try:
        res = minimize(objective, x0, method='L-BFGS-B',
                       options={'maxiter': 200, 'maxfun': 5000,
                                'ftol': 1e-10, 'gtol': 1e-8})
        if res.success or res.fun < best_f:
            f_new = _softmax(res.x)
            f_new = npng(f_new)
            sc = score_fft(f_new)
            if np.isfinite(sc) and sc < best_f:
                best_f = sc
                best_p = f_new.copy()
    except Exception:
        pass

    return best_p, best_f

def coordinate_descent(p0, budget_s, seed=0):
    """Coordinate-wise optimization: optimize one element at a time."""
    p = npng(p0).copy()
    n = p.size
    if n < 2 or budget_s < 3.0:
        return p, score_fft(p)

    best_f = score_fft(p)
    L = next_power_of_two(2*n - 1)

    start = time.monotonic()
    end = start + budget_s
    rng = np.random.default_rng(seed)

    it = 0
    while time.monotonic() < end - 1.0:
        it += 1
        # Pick a random coordinate (or cycle through)
        if it % 10 < 7:
            idx = rng.integers(n)
        else:
            idx = (it * 7) % n  # deterministic cycle

        # Optimize p[idx] with others fixed
        # Score = 2n * max(c) / S^2 where S = sum(p)
        # c_k depends on p[idx] quadratically
        # Simply try a few values
        p_save = p[idx]
        s_rest = float(p.sum() - p[idx])

        # The contribution to c from element idx:
        # c_k gets contributions: p[idx]*p[k-idx] (j=idx) and p[k-idx]*p[idx] (j=k-idx)
        # For the peak, we want to minimize max(c)
        # Try: reduce p[idx] if it's at the peak, increase if not

        # Simple approach: try scaling p[idx] by factors
        best_val = p[idx]
        best_sc = best_f
        for factor in (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5):
            p[idx] = p_save * factor
            if p[idx] < 0:
                p[idx] = 0
            sc = score_fft(p)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_val = p[idx]
        p[idx] = best_val
        p = npng(p)

        sc = score_fft(p)
        if sc < best_f - 1e-10:
            best_f = sc

        if it % 500 == 0:
            print("    CD iter %d: %.6f" % (it, best_f))

    return p, best_f

def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size)
        s = float(a.size)
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / max(bs, 1e-30))
    return [float(x) for x in b]

def try_eval(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
        val = float(val)
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf

def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        if p.size == 0:
            return
        if score is None:
            score = score_fft(p)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = p.copy()
            best_score = float(score)
            print("  NEW BEST: n=%d score=%.6f (t=%.0fs)" % (p.size, score, time.monotonic()-t0))
        n = p.size
        old = best_by_n.get(n)
        if old is None or score < old:
            best_by_n[n] = float(score)

    # Load existing best if available
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                sc = score_fft(arr)
                record(arr, sc)
                print("Loaded height_sequence_1: n=%d score=%.6f" % (arr.size, sc))
        except Exception:
            pass

    # Phase 1: Quick baselines
    print("Phase 1: Baselines")
    base_lengths = [128, 192, 256, 300, 350, 384, 420, 450, 512, 600, 768, 1024, 1536, 2048]
    for n in base_lengths:
        if time.monotonic() > deadline:
            break
        for init_fn in [greedy_seq, lambda nn: power_seq(nn, 0.4),
                        lambda nn: power_seq(nn, 0.5), lambda nn: power_seq(nn, 0.6),
                        lambda nn: u_shape_seq(nn, 4.0), lambda nn: u_shape_seq(nn, 8.0)]:
            p = init_fn(n)
            record(p)

    # Sort by best score per length
    sorted_by_n = sorted(best_by_n.items(), key=lambda x: x[1])
    print("Top lengths by baseline score:", [(n, "%.4f" % s) for n, s in sorted_by_n[:8]])

    # Phase 2: Gradient optimization
    print("Phase 2: Gradient optimization")
    remaining = deadline - time.monotonic()

    # Build job list: (n, init_type)
    jobs = []
    # Focus on promising lengths
    top_lengths = [n for n, _ in sorted_by_n[:12]]
    # Ensure we have diversity
    all_lengths = [128, 192, 256, 300, 350, 384, 420, 450, 512, 600, 768, 1024]
    for n in all_lengths:
        if n not in top_lengths:
            top_lengths.append(n)
    top_lengths = top_lengths[:16]

    # For each length, try a few initializations
    inits_per_length = 3
    for n in top_lengths:
        for i in range(inits_per_length):
            jobs.append((n, i))

    # Allocate time proportionally (more time for better-performing lengths)
    total_alloc = min(remaining - 200.0, 600.0)
    total_weight = 0.0
    job_weights = []
    for n, i in jobs:
        base_sc = best_by_n.get(n, 2.0)
        w = max(0.5, 2.5 - base_sc)  # better scores get more weight
        job_weights.append(w)
        total_weight += w

    for idx, (n, init_i) in enumerate(jobs):
        if time.monotonic() > deadline - 200.0:
            break
        alloc = total_alloc * job_weights[idx] / total_weight
        alloc = max(8.0, min(alloc, 120.0))

        # Choose initialization
        if init_i == 0:
            p = greedy_seq(n)
        elif init_i == 1:
            p = perturb_seq(greedy_seq(n), rng, 0.15)
        elif init_i == 2:
            p = u_shape_seq(n, 5.0 + rng.random() * 5.0)
        else:
            p = mixed_seq(n, rng, 0.3 + 0.4*rng.random())

        # Also try from best_by_n if available
        if n in best_by_n:
            # Already recorded, use the best as base for gradient
            p_base = p  # keep as is for diversity

        p_out, f_out = adam_simplex(p, alloc, seed=seed + n*31 + init_i*7)
        record(p_out, f_out)

    # Phase 3: LP polishing
    print("Phase 3: LP polishing")
    remaining = deadline - time.monotonic()
    if remaining > 100.0:
        # Pick top candidates for LP polishing
        sorted_candidates = sorted(best_by_n.items(), key=lambda x: x[1])
        lp_n = [n for n, _ in sorted_candidates[:6]]

        for n in lp_n:
            if time.monotonic() > deadline - 80.0:
                break
            # Find the best sequence of this length
            # We need to re-derive it; for simplicity, re-run brief gradient
            # Actually, let's just polish from a greedy start at this length
            # with the best known score as reference
            p_init = greedy_seq(n)
            # Quick gradient to get close
            quick_time = min(20.0, max(5.0, (deadline - time.monotonic()) * 0.1))
            p_init, _ = adam_simplex(p_init, quick_time, seed=seed + n*17)

            alloc = min(remaining * 0.3, 200.0)
            alloc = max(alloc, 30.0)
            p_out, f_out = lp_polish(p_init, alloc, seed=seed + n*41)
            record(p_out, f_out)

    # Phase 4: Local refinement
    print("Phase 4: Local refinement")
    remaining = deadline - time.monotonic()
    if remaining > 30.0 and best_p is not None:
        # Try scipy local refinement on the best
        ref_time = min(remaining * 0.4, 120.0)
        p_out, f_out = local_refine(best_p, ref_time, seed=seed)
        record(p_out, f_out)

        # Try coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 20.0:
            cd_time = min(remaining * 0.5, 100.0)
            if cd_time > 10.0:
                p_out2, f_out2 = coordinate_descent(best_p, cd_time, seed=seed+1)
                record(p_out2, f_out2)

        # Try another gradient pass from perturbed best
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            p_pert = perturb_seq(best_p, rng, 0.05)
            p_out3, f_out3 = adam_simplex(p_pert, min(remaining * 0.6, 100.0), seed=seed+999)
            record(p_out3, f_out3)

    # Phase 5: Validation
    print("Phase 5: Validation")
    if best_p is None:
        best_p = greedy_seq(384)
        best_score = score_fft(best_p)

    # Try a few scalings to see if evaluation differs
    candidates = [best_p]
    # Also try the best by length for a few other lengths
    sorted_final = sorted(best_by_n.items(), key=lambda x: x[1])
    for n, sc in sorted_final[:4]:
        # We don't have the actual sequence saved, so skip
        pass

    best_seq = scale_output(best_p, 10.0)
    print("Final best: n=%d internal_score=%.6f" % (best_p.size, best_score))

    # Validate with actual evaluator
    try:
        actual = try_eval(best_seq)
        print("Validated score: %.6f" % actual)
    except Exception:
        pass

    return best_seq

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation function."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main_search: %s" % str(e))
        import traceback
        traceback.print_exc()
        p = greedy_seq(384)
        return scale_output(p, 10.0)
