"""
Multi-strategy optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FFT-accelerated log-sum-exp smooth L-BFGS optimization, (2) exact
coordinate descent "peaking" heuristic, (3) multiple restarts with diverse
initializations. Combines gradient-based and combinatorial approaches.

import numpy as np
import time
from scipy.optimize import minimize


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_obj_grad(a, tau):
    """
    Log-sum-exp smoothed objective and gradient.
    J_smooth = 2n * Ttau / S^2
    where Ttau = tau * log(sum exp(c[k]/tau))
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 1e-10:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Ttau = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Ttau / (S * S)

    # Gradient: dJ/da[i] = (4n/S^2) * cross[i] - (4n*Ttau/S^3)
    # cross[i] = sum_k p[k] * a[k-i] (cross-correlation of p and a)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Ttau) / (S ** 3)

    return J, grad


def local_obj_grad_wrapper(args):
    """Wrapper for scipy minimize."""
    a, tau = args
    return smooth_obj_grad(a, tau)


def run_lbfgs(a_init, tau, budget_s, maxiter=5000):
    """Run L-BFGS-B on smoothed objective. Returns (best_a, best_J)."""
    n = len(a_init)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)  # normalize to sum = n

    best_a = a.copy()
    best_J = compute_J(a)

    def func(g):
        return smooth_obj_grad(g, tau)[0]

    def grad(g):
        return smooth_obj_grad(g, tau)[1]

    # Run L-BFGS-B
    try:
        result = minimize(
            func, a, jac=grad, method='L-BFGS-B',
            bounds=[(0, 1000)] * n,
            options={'maxiter': maxiter, 'maxfun': maxiter * 5,
                     'ftol': 1e-12, 'gtol': 1e-8, 'iprint': 0}
        )
        if result.success or np.isfinite(result.fun):
            cand = np.maximum(result.x, 0)
            cand_J = compute_J(cand)
            if cand_J < best_J:
                best_J = cand_J
                best_a = cand.copy()
    except Exception:
        pass

    # Check time
    if time.monotonic() > deadline - 1.0:
        return best_a, best_J

    return best_a, best_J


def run_peaking(a_init, budget_s, n_passes_limit=200):
    """
    Coordinate descent 'peaking' algorithm.
    For each coordinate j, set a[j] to the maximum value that keeps
    all convolution values <= current max M. This redistributes weight
    to maximize sum for a given peak level.
    """
    n = len(a_init)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n)
        S = n
    a = a / S * n  # normalize to sum = n

    c = convolve_fft(a)
    best_a = a.copy()
    best_J = compute_J(a)
    n_passes = 0

    while time.monotonic() < deadline and n_passes < n_passes_limit:
        n_passes += 1
        M = np.max(c)
        if M < 1e-10:
            break

        # For each coordinate j, compute the ceiling
        # Affected k range: [j, j+n-1]
        for j in range(n):
            # c_base[k] for affected k:
            # For k in [j, j+n-1]: a[k-j] is the "partner"
            partner = a  # a[k-j] for k = j..j+n-1 is a[0]..a[n-1]

            # conv[k] for k in [j, j+n-1]:
            # c[k] = c_base[k] + 2*a[j]*a[k-j] (if k != 2j)
            # c[2j] = c_base[2j] + a[j]^2 (if 2j in [0, 2n-2])

            # Ceiling from non-diagonal constraints:
            # a[j] <= (M - c_base[k]) / (2*a[k-j]) = (M - c[k] + 2*a[j]*a[k-j]) / (2*a[k-j])
            # = (M - c[k]) / (2*a[k-j]) + a[j]
            # So: a[j]_new <= (M - c[k]) / (2*a[k-j]) + a[j]
            # We want: a[j]_new = min over k of [ (M - c[k]) / (2*a[k-j]) + a[j] ]
            # = a[j] + min over k of [ (M - c[k]) / (2*a[k-j]) ]

            # Vectorized: for k = j..j+n-1, partner = a[0..n-1]
            k_range = np.arange(j, min(j + n, 2 * n - 1))
            c_affected = c[k_range]
            partner_vals = a[:min(n, 2 * n - 1 - j)]

            # Exclude diagonal (k = 2j) from non-diagonal computation
            if 2 * j < len(c):
                diag_idx_in_affected = 2 * j - j  # position of k=2j in k_range
                mask = np.ones(len(k_range), dtype=bool)
                if diag_idx_in_affected < len(mask):
                    mask[diag_idx_in_affected] = False
            else:
                mask = np.ones(len(k_range), dtype=bool)

            valid = mask & (partner_vals > 1e-15) & (k_range < len(c))
            if np.any(valid):
                ratios = (M - c_affected[valid]) / (2.0 * partner_vals[valid])
                non_diag_ceil = a[j] + np.min(ratios)
            else:
                non_diag_ceil = a[j]

            # Diagonal constraint: a[j]^2 + c_base[2j] <= M
            # c_base[2j] = c[2j] - a[j]^2 (if 2j in range)
            diag_ceil = 1e10
            if 2 * j < len(c):
                c_base_diag = c[2 * j] - a[j] * a[j]
                if M > c_base_diag:
                    diag_ceil = np.sqrt(M - c_base_diag)
                else:
                    diag_ceil = 0.0

            # Also ensure non-negativity
            new_val = max(0.0, min(non_diag_ceil, diag_ceil))

            # Update
            delta = new_val - a[j]
            if abs(delta) > 1e-15:
                a[j] = new_val
                # Update convolution incrementally
                # conv[k] changes by:
                # For k in [j, j+n-1]: delta * a[k-j] (from i=j term)
                #                        + delta * a[k-j] (from i=k-j term, if k != 2j)
                # For k = 2j: delta * a[j] (only once, it's a[j]^2)
                # Actually: old contribution was 2*a[j]*a[k-j] (or a[j]^2)
                # New contribution is 2*new_val*a[k-j] (or new_val^2)
                # Change: 2*delta*a[k-j] (or 2*delta*a[j] for diagonal, i.e., delta*(old+new))

                k_start = j
                k_end = min(j + n, 2 * n - 1)
                if 2 * j < 2 * n - 1:
                    # Diagonal at k = 2j
                    c[2 * j] += delta * (a[j] + a[j] - delta)  # new^2 - old^2 = delta*(old+new)
                    # Actually: new_val^2 - a[j_old]^2 where a[j] is already new_val
                    # Let me redo: old_val was a[j_before], new_val is a[j]
                    # a[2j] change = new_val^2 - old_val^2
                    # We already set a[j] = new_val, so old_val = new_val - delta
                    old_val = new_val - delta
                    c[2 * j] = c[2 * j] - old_val * old_val + new_val * new_val
                else:
                    # No diagonal in range, just linear terms
                    partner_slice = a[:min(n, 2 * n - 1 - j)]
                    c[k_start:k_end] += 2.0 * delta * partner_slice

                # Re-do the diagonal correction properly
                if 2 * j < 2 * n - 1:
                    old_val = new_val - delta
                    c[2 * j] = c[2 * j] - (2.0 * delta * a[2 * j - j]) + (new_val * new_val - old_val * old_val - 2.0 * delta * a[2 * j - j])
                    # This is getting confusing. Let me just recompute the affected range.

        # Periodic full recompute to correct drift
        if n_passes % 5 == 0:
            c = convolve_fft(a)
            current_J = compute_J(a)
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()

    # Final
    final_J = compute_J(a)
    if final_J < best_J:
        best_J = final_J
        best_a = a.copy()

    return best_a, best_J


def run_peaking_v2(a_init, budget_s):
    """
    Improved peaking: recompute conv every pass for accuracy.
    For each j, compute ceiling and update.
    """
    n = len(a_init)
    deadline = time.monotonic() + budget_s
    a = a_init.copy().astype(np.float64)
    S = np.sum(a)

    while time.monotonic() < deadline - 2.0:
        pass_count += 1
        c = convolve_fft(a)
        M = np.max(c)
        S = np.sum(a)

        # Compute all ceilings vectorized in blocks
        # For each j: ceiling involves c[j:j+n] and a[0:n-1] (shifted)
        # This is O(n^2) total but can be done in chunks

        improved = False
        for j in range(n):
            k_lo = j
            k_hi = min(j + n, 2 * n - 1)
            length = k_hi - k_lo
            if length <= 0:
                continue

            partner = a[:length]  # a[k-j] for k = j..j+n-1
            c_aff = c[k_lo:k_hi]

            # Non-diagonal ceiling:
            # (M - c_aff) / (2 * partner) for positions where partner > 0 and k != 2j
            valid = partner > 1e-12
            diag_pos = 2 * j - k_lo  # position of k=2j within [k_lo, k_hi)
            if 0 <= diag_pos < length:
                valid[diag_pos] = False

            if np.any(valid):
                ratios = (M - c_aff[valid]) / (2.0 * partner[valid])
                non_diag_limit = np.min(ratios)
            else:
                non_diag_limit = 1e10

            # Diagonal ceiling:
            diag_limit = 1e10
            if 0 <= diag_pos < length:
                # c_base[2j] = c[2j] - a[j]^2
                c_base_diag = c[2 * j] - a[j] * a[j]
                if M > c_base_diag:
                    diag_limit = np.sqrt(M - c_base_diag)
                else:
                    diag_limit = 0.0

            new_val = max(0.0, min(non_diag_limit, diag_limit))

            if new_val != a[j]:
                a[j] = new_val
                improved = True

        # Check improvement
        c = convolve_fft(a)
        S = np.sum(a)
        if S > 0.01:
            current_J = 2.0 * n * np.max(c) / (S * S)
            if current_J < best_J - 1e-9:
                best_J = current_J
                best_a = a.copy()
                print(f"    peaking pass {pass_count}: J = {current_J:.6f}")

        if not improved:
            break

    return best_a, best_J


def make_random_init(n, rng, mode='lognormal'):
    """Generate random positive initialization."""
    if mode == 'lognormal':
        return np.exp(rng.normal(0, 0.8, n))
    elif mode == 'uniform':
        return rng.uniform(0.1, 1.0, n)
    elif mode == 'bimodal':
        x = rng.uniform(0, 1, n)
        return np.where(x < 0.3, 0.05, rng.uniform(0.5, 2.0, n))
    elif mode == 'phase':
        # Multi-frequency phase pattern
        freqs = [1, 3, 7, 13]
        result = np.zeros(n)
        for f in freqs:
            result += np.abs(np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 2 * np.pi)))
        return result + 0.1
    else:
        return np.exp(rng.normal(0, 0.5, n))


def interpolate_to_length(arr, target_n):
    """Interpolate array to target length."""
    from scipy.interpolate import interp1d
    n = len(arr)
    if n == target_n:
        return arr.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    f = interp1d(x_old, arr, kind='cubic', fill_value='extrapolate')
    return np.maximum(f(x_new), 0)

            res_a, res_J = run_lbfgs(best_global, 0.2, remaining2, maxiter=1000)
            val = float(evaluate_sequence(np.clip(res_a, 0, 1000).tolist()))
            if np.isfinite(val) and val < best_global_score:
                best_global = np.clip(res_a, 0, 1000)
                best_global_score = val
                print(f"  -> {best_global_score:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(np.clip(best_global, 0, 1000).tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\n=== FINAL: score = {best_global_score:.6f}, n = {len(best_global)} ===")
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
