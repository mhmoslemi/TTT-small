def lp_direction(f, S):
    """
    Given f (current sequence) and S (target sum), solve:
    minimize t
    subject to (f*g)_k <= t for all k
               sum(g) = S
               g >= 0
    
    Returns (g, t) where g is the optimal direction and t is the optimal peak.
    """
    from scipy.optimize import linprog
    import numpy as np
    
    n = len(f)
    # Variables: g_0, ..., g_{n-1}, t
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Constraints: (f*g)_k - t <= 0 for k = 0, ..., 2n-2
    # (f*g)_k = sum_i f_i * g_{k-i}
    # In matrix form: A_ub * [g, t] <= b_ub
    
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = f[j]  # f[j] * g[i] contributes to (f*g)_k
        A_ub[k, n] = -1.0  # -t
    
    # Equality constraint: sum(g) = S
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(-np.inf, None)]  # g_i >= 0, t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if not res.success:
        return None, np.inf
    
    g = res.x[:n]
    t = res.x[n]
    return g, t

def adam_optimize(p, budget_s, seed=0):
    """
    Adam optimizer for the smoothed max-convolution objective.
    Similar to previous code but with some improvements.
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    
    if budget_s <= 0.5:
        return p, score_fft(p)
    
    x = np.log(p + 1e-300)
    x -= x.mean()
    
    if n < 64:
        lr = 0.05
    elif n < 256:
        lr = 0.03
    else:
        lr = 0.015
    
    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    L = next_power_of_two(3*n - 2)
    best_p = p.copy()
    best_f = score_fft(p)
    
    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)
    
    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = 5.0 + 40.0 * np.exp(-4.0 * frac)
        
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2*n-1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break
        
        f_now = float(2.0 * n * maxc)
        if f_now < best_f:
            best_f = f_now
            best_p = p.copy()
        
        # Softmax weights for the max
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2*n-1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz
        
        # Gradient: d/max(a*a) w.r.t. a_i
        # Using the weighted average: grad_i = sum_k w_k * d(a*a)_k/d(a_i)
        # d(a*a)_k/d(a_i) = 2*a_{k-i} (if valid)
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n-1:2*n-1][::-1].real
        C = np.maximum(C, 0.0)
        
        grad_p = (4.0 * n) * C  # derivative of 2n*max(a*a) w.r.t. a_i (for sum=1 normalization)
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)
        
        m = mu * m + (1 - mu) * grad_x
        v = nu * v + (1 - nu) * (grad_x * grad_x)
        mhat = m / (1 - mu**t)
        vhat = v / (1 - nu**t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -50.0, 50.0)
        x = x - step
        x -= x.mean()
        
        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -80.0, 80.0)
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se
    
    final_f = score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f
