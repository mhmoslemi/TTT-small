```python
def _proj_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - target)
    rho_val = np.nonzero(cond)[0][-1] + 1
    theta = (cssv[rho_val - 1] - target) / rho_val
    return np.maximum(v - theta, 0.0)

def _lp_refine(a, M, max_time=300):
    """LP-based refinement: find g that maximizes sum(g) subject to a*g <= M, g>=0.
    Then move towards g."""
    n = a.size
    t_start = time.monotonic()
    
    # Build constraint matrix A where (A@g)[k] = (a*g)[k]
    # A is (2n-1) x n, A[k,i] = a[k-i] if 0 <= k-i < n else 0
    # This is a convolution matrix - build as sparse
    from scipy import sparse
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for i in range(max(0, k-n+1), min(k, n-1) + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k-i])
    
    A_mat = sparse.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Solve LP: maximize sum(g) subject to A_mat @ g <= M, g >= 0
    g_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ g_var <= M]
    objective = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.PDLP, verbose=False)
    except:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            return a.copy()
    
    if g_var.value is None:
        return a.copy()
    
    g = np.asarray(g_var.value).flatten()
    g_sum = g.sum()
    if g_sum <= 1e-12:
        return a.copy()
    
    # Rescale g to have sum 1
    g = g / g_sum
    return g
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start FFT-based gradient descent with log-sum-exp smoothing
2. LP-based refinement (reduced constraints) for escaping local minima
3. Simulated annealing restarts
4. Final coordinate descent fine-tuning
"""

import time
import numpy as np
import cvxpy as cp
from scipy import sparse


def _next_pow2(n):
    return 1 << (n - 1).bit_length()


def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.zeros(n)
    rho_val = idx[-1] + 1
    theta = (cssv[rho_val - 1] - target) / rho_val
    return np.maximum(v - theta, 0.0)


def _smoothed_max_grad(a, tau, N):
    """Compute max of self-convolution and gradient of log-sum-exp smoothing."""
    n = a.size
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N).real[:2 * n - 1]
    c_max = c.max()
    # Softmax weights for numerical stability
    shifted = (c - c_max) / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    # Gradient: (2/tau) * cross_correlation(w, a)
    # cross_correlation(w, a)[i] = sum_k w[k] * a[k-i]
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(W * np.conj(A), N).real[:n]
    grad = (2.0 / tau) * R
    return c_max, grad, w


def _gradient_descent(a, t_end, tau_init, lr_init, momentum_init, N, patience=2000):
    """Run smoothed gradient descent with adaptive schedule and restart."""
    n = a.size
    a = _proj_simplex(np.abs(a), 1.0).copy()
    momentum = momentum_init
    tau = tau_init
    lr = lr_init
    buf = np.zeros(n)
    best_a = a.copy()
    best_mc = np.inf
    step = 0
    stagnant = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() < t_end:
        c_max, grad, w = _smoothed_max_grad(a, tau, N)

        if c_max < best_mc:
            best_mc = c_max
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        step += 1

        # Adaptive schedule
        if step % 300 == 0:
            tau = max(0.005, tau * 0.82)
            lr = max(1e-6, lr * 0.88)
            momentum = min(0.99, momentum + 0.003)

        # Restart if stagnant
        if stagnant > patience:
            # Perturb best solution and restart
            a = best_a * np.exp(rng.normal(0, max(0.05, tau), n))
            a = _proj_simplex(np.abs(a), 1.0)
            buf = np.zeros(n)
            tau = max(0.05, tau * 3)
            lr = max(lr, 0.01)
            stagnant = 0
            patience = max(500, patience * 0.7)
            continue

        # Gradient step
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-14:
            grad = grad / gnorm
        buf = momentum * buf - lr * grad
        a_new = a + lr * buf
        a_new = _proj_simplex(a_new, 1.0)
        a = a_new

    return best_a, best_mc


def _lp_refine_reduced(a, n, top_k=80, time_limit=60):
    """LP with reduced constraints to find improving direction."""
    t0 = time.monotonic()
    if time.monotonic() - t0 > time_limit:
        return None

    c = np.convolve(a, a, mode='full')
    M = c.max()

    # Use top-k + some random constraints for diversity
    top_idx = np.argsort(c)[::-1][:top_k]
    # Add a few random middle constraints
    mid_pool = np.setdiff1d(np.arange(2 * n - 1), top_idx)
    n_rand = min(20, len(mid_pool))
    rand_idx = np.random.choice(mid_pool, n_rand, replace=False)
    all_idx = np.concatenate([top_idx, rand_idx])

    # Build reduced constraint matrix
    K = len(all_idx)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(all_idx):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1)
            rows.extend(np.full(len(i_range), idx, dtype=np.int32))
            cols.extend(i_range.astype(np.int32))
            vals.extend(a[k - i_range])

    if len(rows) == 0:
        return None

    A_mat = sparse.csc_matrix((vals, (rows, cols)), shape=(K, n))

    g_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ g_var <= M]
    objective = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(objective, constraints)

    t1 = time.monotonic()
    try:

    g_sum = g.sum()
    if g_sum <= 1e-12:

def _coordinate_descent(a, t_end, n, N):
    """Simple coordinate descent / block refinement."""
    a = _proj_simplex(np.abs(a), 1.0).copy()
    best_a = a.copy()
    best_mc = np.convolve(a, a, mode='full').max()
    rng = np.random.default_rng(99)
    block_size = max(3, n // 200)

    while time.monotonic() < t_end:
        # Pick random block
        start = rng.integers(0, n - block_size)
        end = start + block_size
        block = a[start:end].copy()

        # Try to optimize this block: perturb and check
        for _ in range(20):
            perturbation = rng.normal(0, 0.02, block_size)
            new_block = np.maximum(block + perturbation, 0)
            if new_block.sum() < 1e-10:
                continue
            candidate = a.copy()
            candidate[start:end] = new_block / new_block.sum() * block.sum()
            candidate = _proj_simplex(candidate, 1.0)
            mc = np.convolve(candidate, candidate, mode='full').max()
            if mc < best_mc:
                best_mc = mc
                best_a = candidate.copy()
                block = candidate[start:end].copy()

        a = best_a.copy()

    return best_a, best_mc

                best_a = a_try.copy()

        if lp_count % 3 == 0:
            print(f"  LP iter {lp_count}: score = {best_score:.8f} (t_left={t_end - time.monotonic():.0f}s)")

    print(f"After Phase 2: best score = {best_score:.8f} (LP iters: {lp_count})")

    # Phase 3: Fine gradient descent + coordinate descent (last 20%)
    phase3_end = t_end
    # Fine gradient descent first 60% of phase 3
    fine_end = t_start + (budget_s - 10) * 0.92
    a_fine, mc_fine = _gradient_descent(best_a, fine_end, tau_init=0.01, lr_init=0.005,
                                        momentum_init=0.95, N, patience=3000)
    score_fine = _get_score(a_fine, n)
    if score_fine < best_score:
        best_score = score_fine
        best_a = a_fine.copy()

    # Coordinate descent for the remainder
    a_cd, mc_cd = _coordinate_descent(best_a, phase3_end, n, N)
    score_cd = _get_score(a_cd, n)
    if score_cd < best_score:
        best_score = score_cd
        best_a = a_cd.copy()

    print(f"After Phase 3: best score = {best_score:.8f}")

    # Final: verify with evaluate_sequence and return
    out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.full(n, 1.0)
    out = out / out.sum()

    final_score = _get_score(out, n)
    print(f"Final computed score: {final_score:.8f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in out]))
        print(f"Final evaluate_sequence score: {eval_score:.8f}")
    except Exception as e:
        print(f"Warning: evaluate_sequence failed: {e}")

    return [float(x) for x in out]
```
