    """Solve the active-set linear system for the KKT condition."""
    import cvxpy as cp
    
    a = cp.Variable(n)
    # Build constraints: sum_k w_k (a[k-i] - a[k]) == 0 for i = 1,...,n-1
    constraints = [cp.sum(a) == 1, a >= 0]

    prob.solve(solver=cp.SCS)

def _fft_conv_sq(a, m=None):

    n = len(a)

def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum x = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # = sum_j W[j+i] a[j]
    # This is the cross-correlation of W and a, which equals convolve(W, flip(a))[n-1+i]
    fa_flip = np.fft.rfft(a[::-1], m)
    prod = fW * fa_flip
    conv_result = np.fft.irfft(prod, m)
    # conv_result[p] = sum_j W[j] * a_flip[p-j] = sum_j W[j] * a[n-1-(p-j)]
    # = sum_j W[j] * a[n-1-p+j]
    # We want sum_j W[j] * a[j+i]... hmm this is getting confusing.
    # Let me just do it directly for each top lag (L is not too large)
    g = np.zeros(n)
    for idx, wk in zip(top_idx, w):
        if wk < 1e-15:
            continue
        k = idx
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1)
        a_vals = a[k - i_range]
        g[i_range] += 2.0 * wk * a_vals
    return g


def _adam_update(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. state = (m, v, t)."""
    m, v, t = state
    t += 1
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * grads * grads
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return params, (m, v, t)


def _run_subgradient(a, t_end, L_init, tau_init, lr_init, verbose=False):
    """Improved subgradient descent with adaptive parameters."""
    n = len(a)
    a = _normalize(np.maximum(a, 0.0))
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    
    momentum_buf = np.zeros(n)
    momentum = 0.85
    L = L_init
    tau = tau_init
    lr = lr_init
    best_a = a.copy()
    best_obj = _objective(a)
    step = 0
    stall_count = 0
    
    while time.monotonic() < t_end - 1.0:
        c = _fft_conv_sq(a, m)
        obj = 2.0 * n * c.max()  # S=1 so S^2=1
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        step += 1
        # Adaptive schedule
        if step % 500 == 0:
            tau = max(0.02, tau * 0.85)
            L = max(10, int(L * 0.8))
            lr = max(1e-6, lr * 0.85)
            momentum = min(0.95, momentum + 0.005)
        
        g = _smoothed_grad_fast(a, c, L, tau)
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn
        momentum_buf = momentum * momentum_buf - lr * g
        a_new = a + lr * momentum_buf
        a = _proj_simplex(a_new, 1.0)
        
        if step % 200 == 0 and verbose:
            print(f"  step {step}: obj={_objective(a):.6f}, best={best_obj:.6f} ({2*n*best_obj:.4f}), tau={tau:.4f}, L={L}")
    
    return best_a, 2.0 * n * best_obj


def _run_adam_logdomain(a0, t_end, lr=0.01, verbose=False):
    """Adam optimization in log-domain (softmax parameterization)."""
    n = len(a0)
    a = np.maximum(a0, 1e-10)
    theta = np.log(a / a.sum())  # log-domain params
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_count = 0
    
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    
    best_a = a.copy()
    best_obj = _objective(a)
    step = 0
    
    while time.monotonic() < t_end - 1.0:
        # Softmax to get a from theta
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        a = exp_theta / exp_theta.sum()
        
        c = _fft_conv_sq(a, m)
        obj = 2.0 * n * c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        step += 1
        if step % 200 == 0 and verbose:
            print(f"  adam step {step}: obj={_objective(a):.6f}, best={2*n*best_obj:.4f}")
        
        # Compute gradient w.r.t. theta
        L_tau = 50
        tau = 0.1
        ga = _smoothed_grad_fast(a, c, L_tau, tau)
        # Chain rule: d(a_j)/d(theta_i) = a_j*(delta_ji - a_i)
        # grad_theta_i = sum_j ga_j * a_j * (delta_ji - a_i)
        # = ga_i * a_i - a_i * sum_j ga_j * a_j
        ga_dot_a = np.dot(ga, a)
        grad_theta = a * (ga - a * ga_dot_a)
        
        # Adam step
        t_count += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m_buf = beta1 * m_buf + (1 - beta1) * grad_theta
        v_buf = beta2 * v_buf + (1 - beta2) * grad_theta ** 2
        m_hat = m_buf / (1 - beta1 ** t_count)
        v_hat = v_buf / (1 - beta2 ** t_count)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Adaptive lr
        if step % 500 == 0:
            lr = max(0.001, lr * 0.9)
    
    return best_a, 2.0 * n * best_obj


def _run_active_set(a0, t_end, verbose=False):
    """Active-set KKT method.
    At optimum: sum_k w_k * a[k-i] = lambda for all i (where a_i > 0).
    We iteratively build the active set and solve the resulting linear system.
    """
    n = len(a0)
    a = _normalize(np.maximum(a0, 1e-10))
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    
    best_a = a.copy()
    best_obj = _objective(a)
    
    # Start with center lag
    K_active = [n - 1]
    max_iter = 200
    
    for outer in range(max_iter):
        if time.monotonic() > t_end - 2.0:
            break
        
        c = _fft_conv_sq(a, m)
        t_star = c.max()
        
        # Identify lags near the max (within 5% of max)
        threshold = t_star * 0.95
        near_max = np.where(c >= threshold)[0]
        
        # Build active set from near-max lags
        # Use softmax weighting based on c[k]
        cv = c[near_max]
        w = np.exp(cv / 0.05)
        w = w / w.sum()
        K_active = list(near_max)
        weights = w
        
        # Solve: sum_k w_k * a[k-i] = lambda for all i, sum a = 1, a >= 0
        # This is: M @ a = lambda * ones, sum(a) = 1, a >= 0
        # where M[i,j] = sum_k w_k * [j == k-i] = sum_k w_k * [i+j == k]
        # M[i,j] = w_k where k = i+j (if i+j is in K_active)
        
        # Build M as sparse (implicit)
        # M[i,j] = weights[idx] if (i+j) is in K_active at position idx, else 0
        # For efficiency, create a lookup: for each possible sum s = i+j (0 to 2n-2),
        # M[i,j] = W[s] where W[s] = w_k if s == K_active[k]
        W = np.zeros(2 * n - 1)
        for k_idx, k_val in enumerate(K_active):
            W[k_val] = weights[k_idx]
        
        # M[i,j] = W[i+j]
        # We want M @ a = lambda * 1
        # And sum(a) = 1, a >= 0
        
        # This is equivalent to: find a >= 0, sum(a)=1 such that (W * reverse(a))[center] is constant
        # Let's solve it as a QP using the constraint that all (Ma)_i are equal.
        # Alternatively: minimize sum((Ma)_i - (Ma)_0)^2 subject to sum(a)=1, a>=0
        
        # For speed, use a simpler approach: solve the linear system M@a = lambda*1, 1^T a = 1
        # with a >= 0 handled by clamping
        
        # Build augmented system:
        # [M  -1] [a]   [0]
        # [1^T 0] [lam] = [1]
        # This gives (n+1) x (n+1) system
        
        # Actually, let's use a different approach: 
        # Minimize ||Ma - 1^T(Ma)/n * 1||^2 subject to sum(a)=1, a>=0
        # This is a QP. Let's use scipy or just project.
        
        # Simple approach: solve M@a = lambda*1, 1^T a = 1 (ignoring a>=0)
        # Then project onto {a>=0, sum(a)=1}
        
        # Build M (n x n): M[i,j] = W[i+j]
        # This is a Hankel-like matrix. We can build it efficiently.
        indices_i = np.arange(n)
        indices_j = np.arange(n)
        I, J = np.meshgrid(indices_i, indices_j, indexing='ij')
        S = I + J  # shape (n, n), S[i,j] = i+j
        M_mat = W[S]  # M[i,j] = W[i+j]
        
        # Solve [M | -1; 1^T | 0] [a; lambda] = [0; 1]
        # i.e., M@a - lambda*1 = 0, 1^T a = 1
        # => M@a = lambda*1
        # Take first equation: (M@a)[0] = lambda => lambda = (M@a)[0]
        # So: M@a = (M@a)[0] * 1
        # (M - M[0,:]) @ a = 0 (subtract first row from all rows)
        # Plus: 1^T a = 1
        
        # System: (n-1) + 1 = n equations in n unknowns
        A_sys = np.vstack([M_mat - M_mat[0:1, :], np.ones((1, n))])
        b_sys = np.zeros(n)
        b_sys[-1] = 1.0
        
        try:
            a_sol = np.linalg.solve(A_sys, b_sys)
        except np.linalg.LinAlgError:
            a_sol = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
        
        # Project onto simplex
        a_new = _proj_simplex(a_sol, 1.0)
        
        # Check if this is better
        obj_new = _objective(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose and outer % 20 == 0:
                c_new = _fft_conv_sq(a)
                print(f"  active_set iter {outer}: J={2*n*best_obj:.6f}, max_c={c_new.max():.6f}")
        else:
            # Try a blend
            for t_blend in [0.3, 0.5, 0.7]:
                a_blend = (1 - t_blend) * a + t_blend * a_new
                a_blend = _proj_simplex(a_blend, 1.0)
                obj_blend = _objective(a_blend)
                if obj_blend < best_obj:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    break
        
        # Check convergence
        if outer > 10 and time.monotonic() > t_end - 60.0:
            break
    
    return best_a, 2.0 * n * best_obj


def _generate_seeds(n, base):
    """Generate diverse structured seeds."""
    seeds = []
    
    # Base and its reverse
    seeds.append(base.copy())
    seeds.append(base[::-1].copy())
    
    # Uniform
    seeds.append(np.ones(n) / n)
    
    # Triangle (symmetric)
    t = np.abs(np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)
    tri = 1.0 - t
    seeds.append(tri / tri.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    rc = np.maximum(rc, 1e-6)
    seeds.append(rc / rc.sum())
    
    # Gaussian (centered)
    x = np.arange(n) / n - 0.5
    gauss = np.exp(-4 * x * x)
    seeds.append(gauss / gauss.sum())
    
    # Wide Gaussian
    gauss2 = np.exp(-2 * x * x)
    seeds.append(gauss2 / gauss2.sum())
    
    # Power law
    pl = (1.0 - t) ** 2 + 1e-6
    seeds.append(pl / pl.sum())
    
    # Sine bump
    sinb = np.sin(np.pi * np.arange(n) / n) ** 2
    sinb = np.maximum(sinb, 1e-6)
    seeds.append(sinb / sinb.sum())
    
    # Two-lobe
    x1 = (np.arange(n) / n - 0.3)
    x2 = (np.arange(n) / n - 0.7)
    two_lobe = np.exp(-8 * x1**2) + np.exp(-8 * x2**2)
    seeds.append(two_lobe / two_lobe.sum())
    
    # Random smooth
    rng = np.random.default_rng(123)
    freq = rng.uniform(0.5, 3.0, 5)
    phase = rng.uniform(0, 2 * np.pi, 5)
    amp = rng.uniform(0.5, 1.0, 5)
    random_smooth = np.ones(n)
    for f, p, a_val in zip(freq, phase, amp):
        random_smooth += a_val * np.cos(2 * np.pi * f * np.arange(n) / n + p)
    random_smooth = np.maximum(random_smooth, 1e-6)
    seeds.append(random_smooth / random_smooth.sum())
    
    # Perturbed base
    pert = base * np.exp(rng.normal(0, 0.2, n))
    pert = np.maximum(pert, 0)
    if pert.sum() > 0:
        seeds.append(pert / pert.sum())
    
    return seeds


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for the non-negative sequence minimizing the objective."""
    t_start = time.monotonic()
    t_end = t_start + max(0.0, float(budget_s)) - 3.0
    
    # Get base sequence
    n = 512
    base = np.ones(n) / n
    try:
        b = np.asarray(height_sequence_1, dtype=np.float64)
        b = np.clip(b, 0.0, 1000.0)
        if b.ndim == 1 and b.size >= 4 and np.isfinite(b).all() and b.sum() > 0.01:
            base = b / b.sum()
            n = base.size
    except Exception:
        pass
    
    print(f"Starting with n={n}")
    print(f"Initial objective: {2*n*_fft_conv_sq(base).max() / (base.sum()**2):.6f}")
    
    # Evaluate with official evaluator
    try:
        init_score = float(evaluate_sequence([float(x) for x in base]))
    except Exception:
        init_score = float('inf')
    print(f"Initial evaluate_sequence: {init_score:.6f}")
    
    best_a = base.copy()
    best_score = init_score
    
    def _check_and_update(a):
        nonlocal best_a, best_score
        try:
        # Scale to sum = 1 for evaluation (scale invariant, but evaluator clips)
            s = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            s = float('inf')
        if s < best_score:
            best_score = s
            best_a = a.copy()
            print(f"  NEW BEST: {s:.6f} at t={time.monotonic()-t_start:.1f}s")
        return s
    
    _check_and_update(base)
    
    # Generate seeds
    seeds = _generate_seeds(n, base)
    print(f"Generated {len(seeds)} seeds")
    
    # Time allocation strategy:
    # - 40%: Subgradient (largest improvement potential)
    # - 30%: Active-set KKT
    # - 20%: Adam log-domain
    # - 10%: Final polish and evaluation
    
    total_time = t_end - time.monotonic()
    t_sg = time.monotonic() + total_time * 0.35
    t_as = t_sg + total_time * 0.25
    t_adam = t_as + total_time * 0.20
    t_polish = t_adam + total_time * 0.10
    
    # ---- Phase 1: Subgradient with multiple seeds ----
    print(f"\n=== Phase 1: Subgradient ({t_sg - time.monotonic():.0f}s) ===")
    n_seeds_sg = min(4, len(seeds))
    sg_time_each = (t_sg - time.monotonic()) / n_seeds_sg
    for i in range(n_seeds_sg):
        if time.monotonic() > t_sg - 5:
            break
        seed_a = seeds[i].copy()
        t_seed_end = min(t_sg - 1, time.monotonic() + max(10, sg_time_each))
        print(f"  Seed {i}: running subgradient...")
        a_res, obj_res = _run_subgradient(
            seed_a, t_seed_end,
            L_init=min(200, 2 * n - 1),
            tau_init=0.15,
            lr_init=0.15,
            verbose=False
        )
        print(f"  Seed {i} done: J={obj_res:.6f}")
        _check_and_update(a_res)
        # Carry forward
        seeds[i] = a_res.copy()
    
    # Additional subgradient rounds with best result
    if time.monotonic() < t_sg - 30:
        print("  Extra subgradient on best...")
        a_res, obj_res = _run_subgradient(
            best_a.copy(), t_sg - 5,
            L_init=min(300, 2 * n - 1),
            tau_init=0.08,
            lr_init=0.05,
            verbose=True
        )
        print(f"  Extra subgradient done: J={obj_res:.6f}")
        _check_and_update(a_res)
    
    # ---- Phase 2: Active-set KKT ----
    print(f"\n=== Phase 2: Active-set KKT ({t_as - time.monotonic():.0f}s) ===")
    # Run active-set from best and a few other seeds
    as_seeds = [best_a.copy(), seeds[0].copy(), seeds[3].copy() if len(seeds) > 3 else seeds[0].copy()]
    n_as = min(3, len(as_seeds))
    as_time_each = (t_as - time.monotonic()) / max(1, n_as)
    for i in range(n_as):
        if time.monotonic() > t_as - 5:
            break
        print(f"  Active-set from seed {i}...")
        a_res, obj_res = _run_active_set(
            as_seeds[i],
            min(t_as - 5, time.monotonic() + max(20, as_time_each)),
            verbose=True
        )
        print(f"  Active-set {i} done: J={obj_res:.6f}")
        _check_and_update(a_res)
    
    # ---- Phase 3: Adam in log-domain ----
    print(f"\n=== Phase 3: Adam log-domain ({t_adam - time.monotonic():.0f}s) ===")
    adam_seeds = [best_a.copy()]
    if len(seeds) > 1:
        adam_seeds.append(seeds[0].copy())
    n_adam = min(2, len(adam_seeds))
    adam_time_each = (t_adam - time.monotonic()) / max(1, n_adam)
    for i in range(n_adam):
        if time.monotonic() > t_adam - 5:
            break
        print(f"  Adam from seed {i}...")
        a_res, obj_res = _run_adam_logdomain(
            adam_seeds[i],
            min(t_adam - 5, time.monotonic() + max(20, adam_time_each)),
            lr=0.02,
            verbose=True
        )
        print(f"  Adam {i} done: J={obj_res:.6f}")
        _check_and_update(a_res)
    
    # ---- Phase 4: Final polish ----
    print(f"\n=== Phase 4: Final polish ({t_polish - time.monotonic():.0f}s) ===")
    # Run active-set on the best found so far
    if time.monotonic() < t_polish - 10:
        a_res, obj_res = _run_active_set(best_a.copy(), t_polish - 5, verbose=False)
        _check_and_update(a_res)
    
    # Fine subgradient polish
    if time.monotonic() < t_end - 10:
        a_res, obj_res = _run_subgradient(
            best_a.copy(), t_end - 5,
            L_init=min(500, 2 * n - 1),
            tau_init=0.03,
            lr_init=0.02,
            verbose=True
        )
        _check_and_update(a_res)
    
    # Final evaluation
    _check_and_update(best_a)
    
    # Ensure valid output
    out = np.clip(best_a, 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.ones(n)
    
    try:
        final_score = float(evaluate_sequence([float(x) for x in out]))
        print(f"\nFINAL score: {final_score:.6f}")
    except Exception:
        print("Final evaluation failed")
    
    return [float(x) for x in out]
