For the problem of minimizing max_k (a*a)[k] subject to a >= 0, sum(a) = 1, the optimal solution is known to have a specific property: the top peaks of the convolution should be approximately equal. This is because if one peak is strictly higher than all others, we can reduce it by adjusting a, and the maximum won't increase (since the other peaks are lower).

So the strategy is: **make the top K peaks equal**. This is a system of K-1 equations (equalizing K peaks) plus the optimization of the common value.

For K=2 (make the top 2 peaks equal), we have 1 constraint + 1 objective. For K=3, we have 2 constraints + 1 objective. Etc.

- This encourages the top peaks to be equal

def _self_conv(p, fft_len=None):
    n = len(p)
    if fft_len is None:
        fft_len = 1 << (2*n - 1).bit_length()
    f = np.fft.rfft(p, fft_len)
    c = np.fft.irfft(f * f, fft_len).real[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

def _grad_p_from_weights(p, w, fft_len):
    """Compute gradient of sum_k w[k]*conv(p,p)[k] w.r.t. p."""
    n = len(p)
    p_rev = p[::-1]
    fa = np.fft.rfft(p_rev, fft_len)
    fb = np.fft.rfft(w, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len).real
    h = conv[n-1:2*n-1]
    return 2.0 * h

def _check():
    if time.monotonic() >= _deadline:

def _probs_from_z(z, n):
    z = np.clip(np.where(np.isfinite(z), z, 0.0), -50.0, 50.0)
    m = float(np.max(z))
    expz = np.exp(z - m)
    Z = float(np.sum(expz))
    return expz / Z


def _probs_to_z(p, n):
    p = np.maximum(p, 1e-30)
    logp = np.log(p)
    z = logp - float(logp[0])
    return np.clip(z, -50.0, 50.0)


def _conv_fft_len(n):
    return 1 << (2 * n - 1).bit_length()


def _grad_fft_len(n):
    return 1 << (3 * n - 2).bit_length()


def _self_conv(p):
    n = len(p)
    fl = _conv_fft_len(n)
    f = np.fft.rfft(p, fl)
    c = np.fft.irfft(f * f, fl).real[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(p, n):
    c = _self_conv(p)
    return 2.0 * n * float(np.max(c))

def _grad_from_w(p, w):
    n = len(p)
    fl = _grad_fft_len(n)
    p_rev = p[::-1]
    fa = np.fft.rfft(p_rev, fl)
    fb = np.fft.rfft(w, fl)
    conv = np.fft.irfft(fa * fb, fl).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h
