def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:

def compute_J(a):

import numpy as np

def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e15
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_obj_grad(a, tau):
    """Compute smoothed objective and gradient via log-sum-exp."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 1e-10:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp < 1e-300:
        return 1e15, np.zeros(n)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    # Phase 2: Coordinate descent on the best
    cd_budget = total_deadline - time.monotonic() - 40.0
    if cd_budget > 30:
        print(f"\n[{cd_budget:.0f}s left] Coordinate descent phase")
        res_a, res_J = coordinate_descent(best_a, cd_budget * 0.6, seed + 12345, top_k=100)
        val = float(evaluate_sequence(res_a.tolist()))
        if val < best_score:
            best_score = val
            best_a = res_a.copy()
            print(f"  CD improved: {val:.6f}")
        else:
            print(f"  CD: J={res_J:.6f}, verified={val:.6f} (best={best_score:.6f})")

    # Phase 3: Final GD polish with very small tau
    remaining = total_deadline - time.monotonic() - 5.0
    if remaining > 20:
        print(f"\n[{remaining:.0f}s left] Final polish GD (tau=0.01->0.003)")
        res_a, res_J = run_gd(best_a, remaining, seed + 99999,
                              tau_start=0.02, tau_end=0.003,
                              lr_start=0.015, lr_decay=0.2)
        val = float(evaluate_sequence(res_a.tolist()))
        if val < best_score:
            best_score = val
            best_a = res_a.copy()
            print(f"  Final polish: {val:.6f}")
        else:
            print(f"  Final polish: J={res_J:.6f}, verified={val:.6f} (best={best_score:.6f})")

    # Final verification
    final_val = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_val):
        best_score = min(best_score, final_val)

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"{'='*50}")

    return [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
