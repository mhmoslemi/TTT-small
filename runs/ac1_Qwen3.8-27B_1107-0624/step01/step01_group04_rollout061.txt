Let me also think about the mathematical structure more. The objective is:
f(a) = 2n * max_k (a*a)_k / (sum a)^2

This is a ratio of a piecewise-linear function (max of linear functions) to a quadratic. It's not convex, which makes optimization hard.

However, for a fixed "active set" (the k that achieves the max), the objective becomes:
f(a) = 2n * (a*a)_{k*} / (sum a)^2

This is a ratio of a quadratic to a quadratic, which is a fractional quadratic. This can be optimized using eigenvalue methods or by fixing the denominator.

Actually, if we fix S = sum a = 1, then we want to minimize max_k (a*a)_k subject to sum a = 1, a >= 0. This is a quadratic programming problem (for a fixed active set) or a non-convex problem (optimizing over the active set).

For a fixed active set K, we want to minimize (a*a)_{k*} subject to (a*a)_k <= (a*a)_{k*} for all k in K, sum a = 1, a >= 0. This is a quadratic constraint problem.

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1

def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:

        s = float(a.sum())

def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def cosine_taper_power(n, alpha):
    """a_i = cos^2(i*pi/(2n)) / (i+1)^alpha, normalized."""
    n = int(n)
    idx = np.arange(n, dtype=np.float64)
    taper = np.cos(idx * np.pi / (2.0 * n)) ** 2
    base = np.power(idx + 1.0, -float(alpha))
    a = taper * base
    return normalize_nonneg(a)


def mix_sequences(a, b, lam):

def actual_score_fft(p):

        return np.inf

    if not np.isfinite(maxc):

def adam_optimize_v2(p, budget_s, seed=0, base_lr=None, max_temp=80.0):
    """
    Improved Adam optimizer with:
    - Better learning rate schedule
    - Adaptive temperature for smoothed max
    - Periodic restarts within the call
    - Better gradient computation
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 0.5:
        f = actual_score_fft(p)
        return p, f

    x = np.log(p + 1e-300)
    x -= x.mean()

    if base_lr is None:
        if n < 64:
            lr = 0.06
        elif n < 256:
            lr = 0.04
        elif n < 1024:
            lr = 0.025
        else:
            lr = 0.018

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        
        # Adaptive temperature: start high, decrease over time
        alpha = 3.0 + 60.0 * np.exp(-3.0 * frac)
        
        # Adaptive learning rate: decay over time
        lr_t = lr * (1.0 + 2.0 * np.exp(-5.0 * frac))

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)  # p sums to one
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Smoothed max via softmax
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0.0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient of smoothed max w.r.t. p
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (2.0 * n) * C  # gradient of n * smoothed_max w.r.t. p (p sums to 1)
        # Actually, d/dp_i [2n * max_k (a*a)_k] = 2n * sum_k w_k * d(a*a)_k/dp_i
        # d(a*a)_k/dp_i = sum_j [j+k-i >= 0 and j+k-i < n] p_{j+k-i} ... 
        # This is the convolution of w with p, indexed appropriately.
        # The gradient w.r.t. p (in the unconstrained sense) is 2n * (w * p)
        # where (w*p) is the appropriate cross-correlation.
        
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr_t * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -100.0, 100.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -100.0, 100.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def lbfgs_smoothed_optimize(p, budget_s, maxiter=None):
    """
    L-BFGS-B optimization of the smoothed max objective.
    """
    if _scipy_opt is None:
        return p, actual_score_fft(p)
    
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    
    alpha = 20.0  # fixed temperature for L-BFGS
    
    def objective_and_grad(x):
        p_temp = np.exp(x - x.max())
        s = float(p_temp.sum())
        p_temp = p_temp / s
        
        L = next_power_of_two(3 * n - 2)
        Pg = np.fft.rfft(p_temp, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if maxc <= 0 or not np.isfinite(maxc):
            return 2.0 * n * 0.0, np.zeros(n)
        
        f_val = float(2.0 * n * maxc)
        
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if sz <= 0 or not np.isfinite(sz):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz
        
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)
        
        grad_p = (2.0 * n) * C
        gp = float(np.dot(p_temp, grad_p))
        grad_x = p_temp * (grad_p - gp)
        
        return f_val, grad_x
    
    x0 = np.log(p + 1e-300)
    x0 -= x0.mean()
    
    end = time.monotonic() + max(0.0, budget_s - 5.0)
    best_p = p.copy()
    best_f = actual_score_fft(p)
    
    # Run L-BFGS in chunks
    while time.monotonic() < end:
        chunk_iter = 100 if n < 512 else 50
        try:
            res = _scipy_opt.minimize(
                lambda x: objective_and_grad(x)[0],
                x0,
                jac=lambda x: objective_and_grad(x)[1],
                method='L-BFGS-B',
                options={'maxiter': chunk_iter, 'gtol': 1e-6, 'maxfun': int(chunk_iter * 1.5)}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_new = np.exp(res.x - res.x.max())
                p_new = p_new / p_new.sum()
                f_new = actual_score_fft(p_new)
                if f_new < best_f:
                    best_f = f_new
                    best_p = p_new
                x0 = res.x - res.x.mean()
            else:
                break
        except Exception:
            break
    
    return best_p, best_f

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def cosine_taper_power(n, alpha):
    """a_i = cos^2(i*pi/(2n)) / (i+1)^alpha, normalized."""
    n = int(n)
    idx = np.arange(n, dtype=np.float64)
    taper = np.cos(idx * np.pi / (2.0 * n)) ** 2
    base = np.power(idx + 1.0, -float(alpha))
    a = taper * base
    a = np.maximum(a, 0.0)
    return normalize_nonneg(a)


def sine_taper_power(n, alpha):
    """a_i = sin((i+1)*pi/(n+1)) / (i+1)^alpha, normalized."""
    n = int(n)
    idx = np.arange(n, dtype=np.float64)
    taper = np.sin((idx + 1.0) * np.pi / (float(n) + 1.0))
    base = np.power(idx + 1.0, -float(alpha))
    a = taper * base
    a = np.maximum(a, 0.0)
    return normalize_nonneg(a)

    b = np.asarray(b, dtype=np.float64).ravel()

def actual_score_fft(p):

    a = np.clip(a, 0.0, 1000.0)

    maxc = float(np.max(c))
    if not np.isfinite(maxc):

    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):

        a = np.ones(16, dtype=np.float64)

    bs = float(b.sum())

    return [float(x) for x in b]

        w[int(np.argmax(c))] = 1.0
        return w, maxc
    z = alpha * (c / maxc - 1.0)
    z = z - z.max()
    expz = np.exp(z)
    sz = float(expz.sum())
    if not np.isfinite(sz) or sz <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = expz / sz
    return w, maxc


def compute_gradient(p, L, alpha):
    """Compute gradient of 2n*softmax_smoothed_max w.r.t. p (log-param space)."""
    n = p.size
    Pg = np.fft.rfft(p, L)
    c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    if maxc <= 0 or not np.isfinite(maxc):
        return np.zeros(n), 0.0, c, maxc

    w, _ = compute_soft_max_weights(c, alpha)

    W = np.fft.rfft(w[::-1], L)
    Cfull = np.fft.irfft(Pg * W, L)
    C = Cfull[n - 1:2 * n - 1][::-1].real
    C = np.maximum(C, 0.0)

    # Gradient w.r.t. p (unconstrained): 4n * C
    grad_p = (4.0 * n) * C
    gp = float(np.dot(p, grad_p))
    grad_x = p * (grad_p - gp)

    f_val = float(2.0 * n * maxc)
    return grad_x, f_val, c, maxc


def adam_optimize_v2(p, budget_s, seed=0):
    """
    Adam optimizer with corrected temperature annealing:
    start smooth (low alpha) → end sharp (high alpha).
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 0.5:
        f = actual_score_fft(p)
        return p, f

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = 0.08
    elif n < 256:
        lr = 0.05
    elif n < 1024:
        lr = 0.03
    else:
        lr = 0.02

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Temperature: start smooth (alpha=3) → end sharp (alpha=200)
        alpha = 3.0 + 197.0 * frac ** 1.5

        # Learning rate decay
        lr_t = lr * (0.3 + 0.7 * (1.0 - frac))

        grad_x, f_now, c, maxc = compute_gradient(p, L, alpha)

        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        if not np.all(np.isfinite(grad_x)):
            break

        if np.max(np.abs(grad_x)) < 1e-15:
            break

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr_t * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -200.0, 200.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -120.0, 120.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def adam_optimize_v3(p, budget_s, seed=0):
    """
    Adam with dual-temperature: alternates between smooth and sharp phases
    to escape shallow local minima while still exploiting.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 0.5:
        f = actual_score_fft(p)
        return p, f

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 256:
        lr = 0.05
    elif n < 1024:
        lr = 0.03
    else:
        lr = 0.02

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Oscillating temperature: cycles between smooth and sharp
        phase = np.sin(2.0 * np.pi * frac * 3.0)  # 3 cycles
        alpha = 10.0 + 40.0 * frac + 15.0 * phase

        lr_t = lr * (0.4 + 0.6 * (1.0 - frac))

        grad_x, f_now, c, maxc = compute_gradient(p, L, alpha)

        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        if not np.all(np.isfinite(grad_x)):
            break

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr_t * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -150.0, 150.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -120.0, 120.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def lbfgs_smoothed_optimize(p, budget_s, alpha=30.0):
    """
    L-BFGS-B optimization of the smoothed max objective with fixed temperature.
    """
    if _scipy_opt is None:
        return p, actual_score_fft(p)

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)

    L = next_power_of_two(3 * n - 2)

    def objective_and_grad(x):
        p_temp = np.exp(x - x.max())
        s = float(p_temp.sum())
        if not np.isfinite(s) or s <= 0:
            return 0.0, np.zeros(n)
        p_temp = p_temp / s

        Pg = np.fft.rfft(p_temp, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if maxc <= 0 or not np.isfinite(maxc):
            return 0.0, np.zeros(n)

        f_val = float(2.0 * n * maxc)

        z = alpha * (c / maxc - 1.0)
        z = z - z.max()
        expz = np.exp(z)
        sz = float(expz.sum())
        if sz <= 0 or not np.isfinite(sz):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        gp = float(np.dot(p_temp, grad_p))
        grad_x = p_temp * (grad_p - gp)

        return f_val, grad_x

    x0 = np.log(p + 1e-300)
    x0 -= x0.mean()

    end = time.monotonic() + max(0.0, budget_s - 3.0)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    while time.monotonic() < end:
        chunk_iter = 80 if n < 512 else 50
        try:
            res = _scipy_opt.minimize(
                lambda x: objective_and_grad(x)[0],
                x0,
                jac=lambda x: objective_and_grad(x)[1],
                method='L-BFGS-B',
                options={'maxiter': chunk_iter, 'gtol': 1e-8,
                         'maxfun': int(chunk_iter * 2)}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_new = np.exp(res.x - res.x.max())
                p_new = p_new / p_new.sum()
                f_new = actual_score_fft(p_new)
                if f_new < best_f:
                    best_f = f_new
                    best_p = p_new.copy()
                x0 = res.x - res.x.mean()
            else:
                break
        except Exception:
            break

    final_f = actual_score_fft(best_p)
    if final_f < best_f:
        best_f = final_f
    return best_p, best_f


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 6.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    # Load previous best if available
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size and arr.sum() > 0.01:
                best_p, best_score = record_candidate(arr, best_p, best_score, best_by_n)
                print("Loaded height_sequence_1: n=%d, score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    # Quick baselines
    base_lengths = [192, 256, 320, 384, 448, 512, 640]
    for n in base_lengths:
        if time.monotonic() >= deadline:
            break
        p = greedy_normalized(n)
        best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)
        p = cosine_taper_power(n, 0.5)
        best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)
        p = cosine_taper_power(n, 0.45)
        best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)
        p = sine_taper_power(n, 0.5)
        best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)

    print("After baselines: best_score=%.6f, n=%d" % (best_score, best_p.size if best_p is not None else 0))

    best_n = best_p.size if best_p is not None else 384

    # Phase 1: Adam v2 (smooth→sharp) from diverse starts
    starts = []
    if best_p is not None:
        starts.append(("best", best_p.copy()))
        starts.append(("best_p05", gaussian_perturb(best_p, rng, 0.05)))
        starts.append(("best_p10", gaussian_perturb(best_p, rng, 0.10)))
        starts.append(("best_p15", gaussian_perturb(best_p, rng, 0.15)))
    starts.append(("greedy", greedy_normalized(best_n)))
    starts.append(("cos045", cosine_taper_power(best_n, 0.45)))
    starts.append(("cos050", cosine_taper_power(best_n, 0.50)))
    starts.append(("sin050", sine_taper_power(best_n, 0.50)))
    starts.append(("pow050", power_normalized(best_n, 0.50)))

    # Try adjacent lengths
    for n_adj in [best_n - 32, best_n + 32]:
        if 64 <= n_adj <= 4096 and n_adj != best_n:
            starts.append(("greedy_%d" % n_adj, greedy_normalized(n_adj)))
            starts.append(("cos_%d" % n_adj, cosine_taper_power(n_adj, 0.50)))

    n_jobs = len(starts)
    remaining = deadline - time.monotonic()
    time_per_job = (remaining - 120.0) / max(1, n_jobs)
    time_per_job = max(15.0, time_per_job)

    for i, (name, p_start) in enumerate(starts):
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break
        alloc = min(time_per_job, max(10.0, rem - 5.0))
        p_out, f_out = adam_optimize_v2(p_start, alloc, seed=int(seed) + i * 137)
        best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
        print("Adam v2 [%s, n=%d]: %.6f (best=%.6f)" % (name, p_out.size, f_out, best_score))

    # Phase 2: Adam v3 (oscillating temperature) from best
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 60.0:
            p0 = gaussian_perturb(best_p, rng, 0.03)
            alloc = min(rem - 30.0, 200.0)
            p_out, f_out = adam_optimize_v3(p0, alloc, seed=int(seed) + 555)
            best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
            print("Adam v3: %.6f (best=%.6f)" % (f_out, best_score))

    # Phase 3: L-BFGS polishing
    if _scipy_opt is not None:
        items = []
        for n, rec in best_by_n.items():
            items.append((rec[0], int(n), rec[1]))
        items.sort(key=lambda x: x[0])

        for sc, n, p in items[:4]:
            rem = deadline - time.monotonic()
            if rem < 35.0:
                break
            # Try multiple temperatures
            for temp in [20.0, 50.0, 100.0]:
                rem = deadline - time.monotonic()
                if rem < 20.0:
                    break
                alloc = min(rem - 10.0, 60.0)
                p_out, f_out = lbfgs_smoothed_optimize(p, alloc, alpha=temp)
                best_p, best_score = record_candidate(p_out, best_p, best_by_n, f_out, f_out)
                print("L-BFGS (n=%d, alpha=%.0f): %.6f (best=%.6f)" % (n, temp, f_out, best_score))

    # Phase 4: Final Adam fine-tuning from best
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 40.0:
            p0 = best_p.copy()
            alloc = min(rem - 15.0, 150.0)
            p_out, f_out = adam_optimize_v2(p0, alloc, seed=int(seed) + 9999)
            best_p, best_score = record_candidate(p_out, best_p, best_by_n, f_out)
            print("Final Adam: %.6f (best=%.6f)" % (f_out, best_score))

        # One more L-BFGS pass
        rem = deadline - time.monotonic()
        if rem > 30.0 and _scipy_opt is not None:
            alloc = min(rem - 10.0, 80.0)
            p_out, f_out = lbfgs_smoothed_optimize(best_p, alloc, alpha=80.0)
            best_p, best_score = record_candidate(p_out, best_p, best_by_n, f_out)
            print("Final L-BFGS: %.6f (best=%.6f)" % (f_out, best_score))

    # Phase 5: Try a slightly different length
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 50.0:
            bn = best_p.size
            for n_try in [bn + 16, bn - 16, bn + 64, bn - 64]:
                if rem < 25.0 or n_try < 64 or n_try > 4096:
                    continue
                if n_try < bn:
                    p_try = normalize_nonneg(best_p[:n_try])
                else:
                    p_try = np.zeros(n_try)
                    p_try[:bn] = best_p
                    # Interpolate the taper
                    if n_try > bn:
                        tail = best_p[-1] * np.exp(-np.arange(n_try - bn) * 0.05)
                        p_try[bn:] = tail
                    p_try = normalize_nonneg(p_try)
                alloc = min(rem - 10.0, 60.0)
                p_out, f_out = adam_optimize_v2(p_try, alloc, seed=int(seed) + int(n_try))
                best_p, best_score = record_candidate(p_out, best_p, best_by_n, f_out)
                rem = deadline - time.monotonic()
                print("Length n=%d: %.6f (best=%.6f)" % (n_try, f_out, best_score))

    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = actual_score_fft(best_p)

    # Final validation with global evaluator
    candidates = [best_p]
    items = []
    for n, rec in best_by_n.items():
        items.append((rec[0], int(n), rec[1]))
    items.sort(key=lambda x: x[0])
    for sc, n, p in items[:3]:
        candidates.append(p)

    for cand in candidates:
        if deadline - time.monotonic() < 10.0:
            break
        try:
            seq = scale_output(cand, 10.0)
            gscore = try_global_eval(seq)
            if np.isfinite(gscore) and gscore < best_score:
                best_score = gscore
                best_p = np.asarray(cand, dtype=np.float64).ravel()
                print("Global eval improved to %.6f" % gscore)
        except Exception:
            pass

    # Try different scaling
    for ts in [5.0, 10.0, 20.0, 50.0]:
        if deadline - time.monotonic() < 5.0:
            break
        try:
            seq = scale_output(best_p, ts)
            gscore = try_global_eval(seq)
            if np.isfinite(gscore) and gscore < best_score:
                best_score = gscore
                print("Scaling ts=%.1f: %.6f" % (ts, gscore))
        except Exception:
            pass

    try:
        final_seq = scale_output(best_p, 10.0)
    except Exception:
        final_seq = scale_output(greedy_normalized(384), 10.0)

    print("FINAL: n=%d, score=%.6f" % (len(final_seq), best_score))
    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation function."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main_search: %s" % str(e))
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
