def lp_step(f, M, solver='auto'):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k."""
    n = len(f)
    # Build constraint matrix A (2n-1 x n) where (A @ b)[k] = (f*b)[k]
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = f[k-j]
    
    # This is O(n²) which is fine for n=4096 (16M entries)
    # But building it this way is slow in Python. Use Toeplitz structure.
    
    # Actually, A is a Toeplitz matrix. A[k,j] = f[k-j] (when valid).
    # The first column of A is [f[0], f[1], ..., f[n-1], 0, ..., 0] (length 2n-1)
    # The (k,j) entry is f[k-j] if 0<=k-j<n
    # Row k of A has nonzeros at columns j = max(0,k-n+1) to min(n-1, k)
    # and the values are f[k-j] for j in that range = f[k-min(n-1,k)] to f[k-max(0,k-n+1)]
    # = f[max(0, k-(n-1))] to f[min(n-1, k)]
    # So row k is: [0...0, f[max(0,k-n+1)], ..., f[min(n-1,k)], 0...0]
    #              ^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^
    #              k-max(0,k-n+1) zeros       length: min(n,k+1)-max(0,k-n+1)    rest zeros
    #
    # This is a lower triangular Toeplitz + upper triangular Toeplitz (bidiagonal band)
    # Actually it's a full band matrix with bandwidth n-1.
    
    # For scipy linprog, we need A_ub @ b <= b_ub
    # A_ub is (2n-1) x n, b_ub = M * ones(2n-1)
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=M*np.ones(2*n-1), bounds=bounds, method='highs')
    return -result.x  # negate back

The key insight is that if f is the current solution with max(f*f) = M, then f itself is feasible (since (f*f)[k] <= M). So the LP optimum is at least sum(f) = 1. If the LP gives sum(b) > 1, then b is "better" in the sense that it has more total mass while staying within the same peak constraint.

After finding b, we rescale: g = b / sum(b) (so sum(g) = 1). Then the new peak max(g*g) could be less than M (since the "excess mass" in b was distributed in a way that avoids the peaks).

Actually, the rescaling is the key. If b has sum > 1 and (f*b)[k] <= M, then g = b/sum(b) has (f*g)[k] <= M/sum(b) < M. This means the cross-convolution f*g has peak < M. But we need g*g to have peak < M too.

The update f_new = (1-t)f + t*g gives:
f_new * f_new = (1-t)² f*f + 2t(1-t) f*g + t² g*g
max <= (1-t)² M + 2t(1-t) * M/sum(b) + t² * max(g*g)

If max(g*g) <= M (which might not hold), then:
max(f_new*f_new) <= M[(1-t)² + 2t(1-t)/sum(b) + t²]
= M[1 - 2t + t² + 2t(1-t)/sum(b) + t²]
= M[1 + 2t² - 2t + 2t(1-t)/sum(b)]
= M[1 - 2t(1-t)(1 - 1/sum(b))]

Since sum(b) > 1, we have 1 - 1/sum(b) > 0, so the max decreases! This is the key theoretical guarantee.

The rate of decrease is proportional to t(1-t)(1 - 1/sum(b)). For small t, this is ≈ t * (1 - 1/sum(b)). So we want t to be as large as possible (while maintaining feasibility of g*g).

        cands.append(f3 / np.sum(f3))

    # Power law
    for a in (0.5, 1.0, 1.5, 2.0, 3.0):
        f = np.power(x + 1e-8, a)
        cands.append(f / np.sum(f))
        f2 = np.power(1 - x + 1e-8, a)
        cands.append(f2 / np.sum(f2))

    # Double hump
    for w in (0.2, 0.3, 0.4):
        f = np.exp(-5 / (2 * w**2) * (x - 0.25)**2) + np.exp(-5 / (2 * w**2) * (x - 0.75)**2)
        cands.append(f / np.sum(f))

    # Sinusoidal modulations
    for freq in (1, 2, 3, 4, 6, 8):
        for amp in (0.3, 0.5, 0.7):
            f = 1.0 + amp * np.sin(2 * np.pi * freq * x)
            f = np.maximum(f, 1e-8)
            cands.append(f / np.sum(f))

    # Random smooth
    for _ in range(20):
        f = np.ones(n)
        for fq in range(1, 8):
            amp = rng.normal(0, 1.0 / (fq + 0.5))
            ph = rng.uniform(0, 2 * np.pi)
            f += amp * np.sin(2 * np.pi * fq * x + ph)
        f = np.maximum(f, 1e-8)
        cands.append(f / np.sum(f))

    return cands

def _logit_obj(x, alpha, n):
    """Objective and gradient for logit parameterization."""
    global _deadline
    _chk()
    x = np.clip(x, -40.0, 40.0)
    m = max(0.0, float(np.max(x)))
    expz = np.exp(x - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(n, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z

    c = _conv_full(p)
    hard = 2.0 * n * float(np.max(c))
    _upd(p)

    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(u)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / se

    # Gradient
    N = _pow2(3 * n - 1)
    p_rev = p[::-1]
    conv = np.fft.irfft(
        np.fft.rfft(p_rev, N) * np.fft.rfft(w, N), N
    ).real
    gp = 2.0 * conv[n - 1: 2 * n - 1]
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e4, 1e4)

    J = cmax + np.log(se) / alpha if se > 0 else cmax
    return float(J), gz

def _logit_optimize(z0, alpha, max_iter, n):
    """Run L-BFGS-B on logit parameterization."""
    global _deadline
    if not HAS_MINIMIZE:
        return z0
    _chk()
    try:
        rem = _deadline - time.monotonic()
        res = minimize(
            _logit_obj, z0, args=(alpha, n),
            method='L-BFGS-B', jac=True,
            bounds=[(-40.0, 40.0)] * (n - 1),
            options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                     'ftol': 1e-12, 'gtol': 1e-8, 'disp': False}
        )
        if res.x is not None:
            return np.clip(res.x, -40.0, 40.0)
    except Exception:
        pass
    return z0

def _p_to_z(p, n):
    p = np.maximum(p, 1e-15)
    logp = np.log(p)
    z = logp[1:] - logp[0]
    return np.clip(z, -40.0, 40.0)

def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_p, _deadline

    start = time.monotonic()
    _deadline = start + float(budget_s) - 15.0
    _best_score = float('inf')
    _best_p = None

    n = 4096
    rng = np.random.default_rng(seed)

    # Phase 1: Generate and score initial candidates
    print(f"Phase 1: Generating candidates for n={n}")
    cands = _make_structured(n, rng)

    # Add previous best if available
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, None)
            if base.sum() > 1e-10:
                if base.size != n:
                    old_x = np.linspace(0, 1, base.size)
                    new_x = np.linspace(0, 1, n)
                    base = np.interp(new_x, old_x, base)
                base = np.maximum(base, 0)
                base /= base.sum()
                cands.insert(0, base)
    except Exception:
        pass

    # Score all candidates
    scored = []
    for i, c in enumerate(cands):
        _chk()
        s = _score_p(c)
        scored.append((s, i, c))
    scored.sort(key=lambda x: x[0])
    print(f"  {len(cands)} candidates generated, best initialized: {_best_score:.6f}")

    # Phase 2: Optimize top candidates with LP-based iterations
    print("Phase 2: LP-based optimization")
    top_k = min(12, len(scored))
    alloc_time = 1400.0
    per_cand = alloc_time / top_k

    for ci in range(top_k):
        _chk()
        rem = _deadline - time.monotonic()
        if rem < 30:
            break
        s_init, _, p_init = scored[ci]
        cand_time = min(per_cand, rem - 30)
        if cand_time < 15:
            break

        print(f"  Candidate {ci}: init score={s_init:.6f}, time_budget={cand_time:.0f}s")
        _deadline_save = _deadline
        _deadline = time.monotonic() + cand_time

        try:
            # Try full LP first
            p_lp, sc_lp = _lp_iterate(p_init, max_iters=50, use_reduced=False,
                                       single_iter_time=min(20.0, cand_time / 3))
        except SearchTimeout:
            p_lp, sc_lp = p_init, s_init
        except Exception:
            p_lp, sc_lp = p_init, s_init

        _deadline = max(_deadline_save, time.monotonic() + 1.0)

        # Also try reduced LP
        if time.monotonic() < _deadline - 20:
            try:
                p_lr, sc_lr = _lp_iterate(p_init, max_iters=80, use_reduced=True,
                                           single_iter_time=min(10.0, 5.0))
                if sc_lr < _best_score:
                    p_lp, sc_lp = p_lr, sc_lr
            except SearchTimeout:
                pass
            except Exception:
                pass

        # Logit-based L-BFGS refinement
        if time.monotonic() < _deadline - 20:
            try:
                z = _p_to_z(p_lp, n)
                alpha = 3e6 if _best_score < 1.55 else 1e6
                max_it = max(50, int(( _deadline - time.monotonic() - 10) / 0.05))
                z_new = _logit_optimize(z, alpha, min(max_it, 300), n)
                m = max(0.0, float(np.max(z_new)))
                expz = np.exp(z_new - m)
                exp0 = np.exp(-m)
                Z = exp0 + float(np.sum(expz))
                p_new = np.empty(n)
                p_new[0] = exp0 / Z
                p_new[1:] = expz / Z
                _upd(p_new)
            except Exception:
                pass

        print(f"  Candidate {ci}: best after optimization = {_best_score:.6f}")

    _deadline = start + float(budget_s) - 15.0

    # Phase 3: Subgradient polishing on the best
    print(f"Phase 3: Subgradient polishing, current best = {_best_score:.6f}")
    if _best_p is not None and time.monotonic() < _deadline - 20:
        polish_budget = min(300.0, _deadline - time.monotonic() - 20)
        try:
            p_fin, sc_fin = _subgrad_polish(_best_p, polish_budget, rng)
        except SearchTimeout:
            pass
        except Exception:
            pass
        print(f"  After polishing: {_best_score:.6f}")

    # Phase 4: Final L-BFGS pass
    if _best_p is not None and time.monotonic() < _deadline - 15:
        try:
            z = _p_to_z(_best_p, n)
            alpha = 8e6
            max_it = max(100, int((_deadline - time.monotonic() - 5) / 0.05))
            z_new = _logit_optimize(z + rng.normal(0, 0.02, n - 1), alpha, min(max_it, 200), n)
            m = max(0.0, float(np.max(z_new)))
            expz = np.exp(z_new - m)
            exp0 = np.exp(-m)
            Z = exp0 + float(np.sum(expz))
            p_new = np.empty(n)
            p_new[0] = exp0 / Z
            p_new[1:] = expz / Z
            _upd(p_new)
        except Exception:
            pass

    print(f"Final best score: {_best_score:.6f}")

    # Construct output
    if _best_p is not None:
        p = np.array(_best_p, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        seq = [float(v) for v in p]
    else:
        seq = [1.0 / n] * n

    # Validate with evaluate_sequence
    try:
        sc = float(evaluate_sequence(seq))
        if not np.isfinite(sc) or sc > 2.0:
            seq = [1.0 / n] * n
            sc = float(evaluate_sequence(seq))
        print(f"Verified score: {sc:.6f}")
    except Exception:
        seq = [1.0 / n] * n
        print("Fallback to uniform")

    # Safety: check against previous best
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            bs = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(bs) and bs < float(evaluate_sequence(seq)):
                seq = [float(v) for v in base]
                print(f"Using inherited best: {bs:.6f}")
    except Exception:
        pass

    return seq
