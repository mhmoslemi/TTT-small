    count = 0
    for k in range(m):
        js = max(0, k - n + 1)
        je = min(k, n - 1) + 1
        length = je - js
        if length > 0:
            rows[count:count+length] = k
            cols[count:count+length] = np.arange(js, je)
            vals[count:count+length] = f[k - np.arange(js, je)]
            count += length
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(m, n))


def _lp_solve(A, M, n, solver=None):
    """Solve max sum(g) s.t. g >= 0, A@g <= M."""
    if not HAVE_CP:
        return None
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    try:
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        kwargs = {}
        if solver:
            prob.solve(solver=solver)
        else:
            prob.solve()
        if g.value is not None:
            return np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    except Exception:
        pass
    return None


def _subgrad_gradient(p, n, c, topk=32):
    """Compute subgradient of max(p*p) at p."""
    cmax = np.max(c)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    w = np.zeros(2*n-1, dtype=np.float64)
    wv = c[idx]
    sw = float(np.sum(wv))
    if sw <= 0:
        w[idx] = 1.0 / len(idx)
    else:
        w[idx] = wv / sw
    
    # SG_i = 2 * sum_j p_j * w_{i+j}
    # = 2 * (flip(p) * w)[i + n - 1]
    flip_p = p[::-1]
    sz = _fft_size(3 * n - 1)
    FP = np.fft.rfft(flip_p, sz)
    W = np.fft.rfft(w, sz)
    conv = np.fft.irfft(FP * W, sz)
    SG = 2.0 * conv[n-1:2*n-1]
    return SG


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Simple projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _lbfgs_smooth(p, n, alpha, max_iter=200, deadline=None):
    """L-BFGS-B on the smooth surrogate with log-sum-exp."""
    # Work in log-space to ensure non-negativity
    p = np.maximum(p, 1e-15)
    log_p = np.log(p)
    
    sz_conv = _fft_size(2 * n - 1)
    sz_grad = _fft_size(3 * n - 1)
    
    def obj_and_grad(log_p):
        if deadline is not None and time.monotonic() >= deadline:
            raise SearchTimeout
        p_local = np.exp(log_p - np.max(log_p))
        p_local = p_local / np.sum(p_local)
        c = _self_conv(p_local, n)
        cmax = np.max(c)
        u = alpha * (c - cmax)
        e = np.exp(u)
        s = np.sum(e)
        if s <= 0 or not np.isfinite(s):
            J = cmax
            w = np.zeros(2*n-1)
            w[np.argmax(c)] = 1.0
        else:
            J = cmax + np.log(s) / alpha
            w = e / s
        
        # Gradient wrt p: SG_i = 2 * sum_j p_j * w_{i+j}
        flip_p = p_local[::-1]
        FP = np.fft.rfft(flip_p, sz_grad)
        W = np.fft.rfft(w, sz_grad)
        conv = np.fft.irfft(FP * W, sz_grad)
        grad_p = 2.0 * conv[n-1:2*n-1]
        
        # Gradient wrt log_p: grad_logp_i = p_i * (grad_p_i - sum(p * grad_p))
        sdot = np.dot(grad_p, p_local)
        grad_logp = p_local * (grad_p - sdot)
        
        # Zero out the first component (normalization constraint)
        # Actually, since we re-normalize, we can just zero the gradient of the "mean"
        grad_logp -= np.mean(grad_logp)  # This makes it orthogonal to the "flat" direction
        # Hmm, this isn't quite right. Let me think...
        # The constraint is sum(exp(log_p)) = const (after normalization)
        # Actually, we normalize p_local after computing exp, so the parameterization
        # is: p_i = exp(log_p_i) / sum(exp(log_p_j))
        # The gradient of the objective wrt log_p is:
        # dp_i/dlog_p_j = p_i * (delta_ij - p_j)
        # dJ/dlog_p_j = sum_i (dJ/dp_i) * p_i * (delta_ij - p_j)
        # = p_j * (grad_p_j - sum_i grad_p_i * p_i)
        # = p_j * (grad_p_j - sdot)
        # This is what we have. Good.
        
        grad_logp = np.where(np.isfinite(grad_logp), grad_logp, 0.0)
        return float(J), grad_logp
    
    x0 = log_p.copy()
    bounds = [(-50, 50)] * n
    
    try:
        result = minimize(
            obj_and_grad, x0, method='L-BFGS-B', jac=True,
            bounds=bounds,
            options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                     'ftol': 1e-14, 'gtol': 1e-10, 'disp': False}
        )
        if result.x is not None:
            p_new = np.exp(result.x - np.max(result.x))
            p_new = p_new / np.sum(p_new)
            return p_new
    except SearchTimeout:
        pass
    except Exception:
        pass
    return p  # Return original if optimization fails
