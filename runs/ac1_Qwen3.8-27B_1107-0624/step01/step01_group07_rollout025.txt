"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast multiplicative gradient updates with top-K peak tracking,
LP-based refinement, and coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def make_hann(n):
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))


def make_gaussian(n, sigma=0.2):
    center = (n - 1) / 2.0
    return np.exp(-0.5 * ((np.arange(n) - center) / (sigma * n)) ** 2)


def make_flattop(n):
    # Flat-top window (Blackman-Harris like)
    x = np.arange(n)
    c = (n - 1) / 2.0
    t = (x - c) / (c + 0.5)
    return 0.35875 - 0.48829 * np.cos(2*np.pi*t/(n)) + 0.14128 * np.cos(4*np.pi*t/(n)) - 0.01168 * np.cos(6*np.pi*t/(n))


def fast_gradient(a, deadline, rng, top_k=5, verbose=False):
    """Fast multiplicative update with top-K peaks."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a.copy(), np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = compute_objective(a)
    eta = 0.15
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        # Top-K peaks
        if top_k >= len(c):
            peak_idx = np.arange(len(c))
        else:
            peak_idx = np.argpartition(c, -top_k)[-top_k:]

        grad = np.zeros(n)
        for k in peak_idx:
            js = np.arange(n)
            idx = k - js
            valid = (idx >= 0) & (idx < n)
            grad[valid] += a[idx[valid]]
        grad /= len(peak_idx)
        
        # Multiplicative update
        mean_g = grad.mean() + 1e-15
        a_new = a * np.exp(-eta * grad / mean_g)
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new /= S_new
        
        sc = compute_objective(a_new)
        if sc < best_score - 1e-13:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            eta = min(0.4, eta * 1.02)
        else:

            eta = max(0.002, eta * 0.92)

                noise = np.abs(rng.normal(0, 0.05, n)) * (1.0 + a)
                a_new = best_a * (1.0 + noise * 0.1)
                a_new = np.clip(a_new, 1e-15, None)
                a_new /= a_new.sum()
                a = a_new
                stall = 0

                break
        
        if verbose and it % 500 == 0:
            print(f"      it={it}, score={best_score:.6f}, eta={eta:.4f}")
    
    return best_a, best_score


def lp_step(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # Build sparse-ish constraint matrix
    # A[k,j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a, deadline, rng, verbose=False):

    n = len(a)

        g = g / g.sum()

            a_new /= S_new

            if stall % 30 == 0 and stall > 30:

            if stall > 100:
                break
        
        if verbose and it % 100 == 0:
            print(f"      lp it={it}, score={best_score:.6f}, step={step:.4f}")

"""
Hybrid optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast multiplicative gradient with top-K peaks, LP refinement,
and coordinate descent polish. Tries multiple lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def make_hann(n):
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))


def make_gaussian(n, sigma_frac=0.2):
    center = (n - 1) / 2.0
    return np.exp(-0.5 * ((np.arange(n) - center) / (sigma_frac * n)) ** 2)


def make_flattop(n):
    """Blackman-Harris flat-top window."""
    x = np.arange(n)
    c = (n - 1) / 2.0
    t = (x - c) / (c + 0.5)
    w = (0.35875 - 0.48829 * np.cos(2 * np.pi * t / n)
         + 0.14128 * np.cos(4 * np.pi * t / n)
         - 0.01168 * np.cos(6 * np.pi * t / n))
    return np.clip(w, 0.01, None)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.clip(1.0 - np.abs(np.arange(n) - c) / (c + 1.0), 0.01, None)


def fast_gradient(a, deadline, rng, top_k=5, initial_eta=0.15):
    """Fast multiplicative update with top-K peak tracking."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a.copy(), np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)
    eta = initial_eta
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')

        # Top-K peaks

            peak_idx = np.arange(len(c))
        else:

            grad[valid] += a[idx[valid]]
        grad /= len(peak_idx)

        # Multiplicative update
        mean_g = grad.mean() + 1e-15
        a_new = a * np.exp(-eta * grad / mean_g)

                a = a_new

    return best_a, best_score


def lp_solve_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return np.clip(res.x, 0.0, None)
    except Exception:
        pass
    return None


def lp_optimize(a, deadline, rng):
    """LP-based iterative optimization with adaptive step."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a.copy(), np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.15
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_solve_direction(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g_sum = g.sum()
        g = g / g_sum

        # Check if g is close to a (convergence)
        cos_sim = np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-15)
        if cos_sim > 0.9999:
            break

        improved = False
        for shrink in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0.0, None)
            S_new = a_new.sum()
            if S_new < 1e-10:
                continue
            a_new /= S_new
            sc = compute_objective(a_new)
            if sc < best_score - 1e-13:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.5, step * 1.04)
        else:
            stall += 1
            step = max(0.005, step * 0.8)
            if stall % 25 == 0 and stall > 25:
                noise = np.abs(rng.normal(0, 0.06, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-15, None)
                a_new /= a_new.sum()
                sc = compute_objective(a_new)
                if sc < best_score:
                    best_a = a_new
                    best_score = sc
                    a = a_new
                    stall = 0
                elif rng.random() < 0.25:
                    a = a_new
            if stall > 120:
                break

    return best_a, best_score


def coord_descent_polish(a, deadline, rng):
    """Coordinate descent: optimize each coordinate with scaling."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a /= a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    sweep = 0
    stall = 0

    while time.monotonic() < deadline:
        sweep += 1
        improved_this_sweep = 0
        a_curr = best_a.copy()

        for j in range(0, n, 1):
            if time.monotonic() > deadline:
                break
            aj = a_curr[j]
            if aj < 1e-10:
                continue

            best_val = aj
            best_sc = best_score

            for scale in [0.5, 0.7, 0.85, 1.1, 1.25, 1.5]:
                a_trial = a_curr.copy()
                a_trial[j] = aj * scale
                a_trial = np.clip(a_trial, 1e-15, None)
                S_t = a_trial.sum()
                if S_t < 1e-10:
                    continue
                a_trial /= S_t
                sc = compute_objective(a_trial)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_val = aj * scale

            if best_val != aj:
                a_curr[j] = best_val
                a_curr = np.clip(a_curr, 1e-15, None)
                a_curr /= a_curr.sum()
                improved_this_sweep += 1

        # End of sweep
        if improved_this_sweep > 0:
            sc = compute_objective(a_curr)
            if sc < best_score - 1e-13:
                best_a = a_curr.copy()
                best_score = sc
            stall = 0
        else:
            stall += 1
            if stall > 3:
                # Random perturbation
                a_curr = best_a.copy()
                for _ in range(5):
                    j = rng.integers(n)
                    a_curr[j] *= 0.5 + rng.random()
                a_curr = np.clip(a_curr, 1e-15, None)
                a_curr /= a_curr.sum()
                sc = compute_objective(a_curr)
                if sc < best_score:
                    best_a = a_curr.copy()
                    best_score = sc
                stall = 0
            if stall > 15:
                break

    return best_a, best_score


def soft_max_gradient(a, deadline, rng, beta=5.0):
    """Gradient method using soft-max for smoother objective."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a.copy(), np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)
    eta = 0.1
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        S2 = S * S

        # Soft-max weights
        c_scaled = beta * c / (S2 + 1e-30)
        c_shifted = c_scaled - c_scaled.max()
        w = np.exp(c_shifted)
        w /= w.sum()

        # Weighted gradient
        grad = np.zeros(n)
        # Only need to check top few weights for efficiency
        top_idx = np.argsort(w)[-min(20, len(w)):]
        for k in top_idx:
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] += w[k] * a[idx[valid]]

        mean_g = grad.mean() + 1e-15
        a_new = a * np.exp(-eta * grad / mean_g)
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new /= S_new

        sc = compute_objective(a_new)
        if sc < best_score - 1e-13:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0

                a_new = best_a * (1.0 + noise)

                a_new /= a_new.sum()
                a = a_new
                stall = 0

        if score < best_score:
            best_seq, best_score = a_opt, score
            print(f"    => NEW BEST: {score:.6f}")

    # 3b: Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        sub_dl = min(time.monotonic() + remaining * 0.4, deadline)
        print(f"  Coord descent, time={sub_dl - time.monotonic():.0f}s")
        a_opt, score = coord_descent_polish(best_seq, sub_dl, rng)
        if score < best_score:
            best_seq, best_score = a_opt, score
            print(f"    => NEW BEST: {score:.6f}")

    # 3c: One more LP pass if time allows
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sub_dl = min(time.monotonic() + remaining * 0.7, deadline)
        print(f"  Final LP, time={sub_dl - time.monotonic():.0f}s")
        a_opt, score = lp_optimize(best_seq, sub_dl, rng)
        if score < best_score:
            best_seq, best_score = a_opt, score
            print(f"    => NEW BEST: {score:.6f}")

    # 3d: Final fine gradient
    remaining = deadline - time.monotonic()
    if remaining > 10:
        sub_dl = min(deadline, time.monotonic() + remaining - 3)
        a_opt, score = fast_gradient(best_seq, sub_dl, rng, top_k=8, initial_eta=0.03)
        if score < best_score:
            best_seq, best_score = a_opt, score
            print(f"    => NEW BEST (final): {score:.6f}")

    # Verify with evaluate_sequence
    best_seq_list = [float(x) for x in best_seq]
    eval_score = evaluate_sequence(best_seq_list)
    print(f"\nFinal: internal_score={best_score:.6f}, eval_score={eval_score:.6f}, n={len(best_seq)}")

    if np.isinf(eval_score) or eval_score > best_score * 1.01:
        # Fallback to known good
        print("Warning: eval mismatch, using internal best")
        best_seq_list = [float(max(0, x)) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")
    return best_seq_list
