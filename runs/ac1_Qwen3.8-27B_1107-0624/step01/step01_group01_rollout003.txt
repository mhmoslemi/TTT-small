"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over nonnegative sequences.
Uses L-BFGS with annealed smoothing, diverse initializations, and direct simplex optimization.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0 and mn > -1e-9:
        c = c - mn
    return np.maximum(c, 0.0)


def _logsumexp_beta(c, beta):
    """Smoothed max: returns (value, weights) where weights sum to 1."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _corr_weights(w, p):
    """h[i] = sum_j w[j] * p[i+j] via FFT."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_stable(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -80.0, 80.0)
    z = z - float(z.max())
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = _fft_convolve(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _obj_and_grad(z, p, beta, n):
    """Objective (smoothed) and gradient w.r.t. logit z. p = softmax(z)."""
    c = _fft_convolve(p)
    val, w = _logsumexp_beta(c, beta)
    gp = 2.0 * _corr_weights(w, p)
    gz = p * (gp - float(np.dot(p, gp)))
    gz -= np.mean(gz)
    return val, gz


def _anneal_lbfgs(p0, deadline, rng, beta_init=None, max_iter=2000):
    """L-BFGS optimization in logit space with annealing."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n = p0.size
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= np.mean(z)

    c = _fft_convolve(p0)
    maxc = float(c.max())
    if beta_init is None:
        beta_init = maxc

    best_p = p0.copy()
    best_score = float(2.0 * n * maxc)

    current_beta = beta_init
    T_total = max_iter
    it_count = [0]

    def objective(zv):
        it_count[0] += 1
        p = _softmax_stable(zv)
        c = _fft_convolve(p)
        # Adaptive beta: cosine annealing
        frac = min(1.0, it_count[0] / T_total)
        b = maxc * (0.001 + 0.999 * (1.0 + np.cos(np.pi * frac)) / 2.0)
        val, w = _logsumexp_beta(c, b)
        return val

    def jac(zv):
        p = _softmax_stable(zv)
        c = _fft_convolve(p)
        frac = min(1.0, it_count[0] / T_total)
        b = maxc * (0.001 + 0.999 * (1.0 + np.cos(np.pi * frac)) / 2.0)
        _, w = _logsumexp_beta(c, b)
        gp = 2.0 * _corr_weights(w, p)
        gz = p * (gp - float(np.dot(p, gp)))
        gz -= np.mean(gz)
        return gz

    # Run L-BFGS with time limit
    t_start = time.monotonic()
    per_iter_time = 0.005  # estimate
    max_steps = min(max_iter, int((deadline - t_start) / per_iter_time))
    max_steps = max(100, max_steps)

    try:
        res = scipy_minimize(
            objective, z, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_steps, 'maxfun': max_steps * 2,
                     'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            p_new = _softmax_stable(res.x)
            sc = _score_fast(p_new)
            if np.isfinite(sc) and sc < best_score:
                best_p = p_new
                best_score = sc
    except Exception:
        pass

    return best_p, best_score


def _direct_simplex_opt(p0, deadline, rng, lr_init=0.05, iters=5000):
    """Direct multiplicative update optimization on the simplex."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    p = np.maximum(p, 1e-15)
    p = p / p.sum()

    c = _fft_convolve(p)
    maxc = float(c.max())
    beta = maxc * 0.5
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    lr = lr_init
    no_imp = 0

    t_start = time.monotonic()
    for it in range(iters):
        if time.monotonic() > deadline:
            break
        val, w = _logsumexp_beta(c, beta)
        gp = 2.0 * _corr_weights(w, p)
        # Multiplicative update
        g_mean = float(np.mean(gp))
        p = p * np.exp(-lr * (gp - g_mean))
        p = np.maximum(p, 1e-15)
        s = float(p.sum())
        if s > 0:
            p = p / s
        c = _fft_convolve(p)
        maxc = float(c.max())
        sc = float(2.0 * n * maxc)

        if sc < best_score - 1e-12:
            best_score = sc
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr *= 0.95
                beta *= 0.8

        if no_imp > 500:
            # Restart with perturbation
            p = best_p * np.exp(0.3 * rng.normal(size=n))
            p = np.maximum(p, 1e-15)
            p = p / p.sum()
            c = _fft_convolve(p)
            maxc = float(c.max())
            beta = maxc * 0.5
            lr = lr_init
            no_imp = 0

    return best_p, best_score


def _make_profiles(n, rng):
    """Generate diverse initial profiles."""
    profs = []
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    # Monotone linear
    for eta in (1.0, 1.25, 1.5, 1.75, 2.0, 2.5):
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())

    # Cubic (antisymmetric)
    for a1, a3 in ((1.5, -1.0), (1.8, -2.0), (1.6, -1.2), (2.0, -2.5), (1.2, -0.5), (2.5, -3.0)):
        a = 1.0 + a1 * u + a3 * u ** 3
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())

    # RHS-heavy (asymmetric peak)
    a = np.exp(2.5 * t)
    profs.append(a / a.sum())
    a = np.exp(3.5 * t)
    profs.append(a / a.sum())
    a = np.exp(-2.5 * t)
    profs.append(a / a.sum())

    # Raised cosine
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())

    # Parabolic
    a = 1.0 + 2.0 * t - t * t
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())

    # Power law
    for p_val in (0.5, 1.5, 2.0, 3.0):
        a = (t + 0.01) ** p_val
        profs.append(a / a.sum())
        a = ((1 - t) + 0.01) ** p_val
        profs.append(a / a.sum())

    # Random smooth via Fourier
    for _ in range(10):
        coeffs = rng.normal(0, 1, 6)
        a = 1.0
        for k in range(1, 6):
            a += coeffs[k-1] * 0.3 / k * np.cos(2 * np.pi * k * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())

    # Sigmoid-like
    for center in (0.3, 0.4, 0.5):
        for width in (0.1, 0.15, 0.2):
            a = 1.0 / (1.0 + np.exp(-10 * (t - center) / width))
            a = np.maximum(a, 1e-12)
            profs.append(a / a.sum())

    return profs


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phase 1: Diverse initializations at n=512.
    Phase 2: L-BFGS annealing on top profiles.
    Phase 3: Direct simplex optimization.
    Phase 4: Fine-tuning and exact evaluation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_p = None
    best_score = np.inf
    all_candidates = []  # list of (score, profile)

    def add_candidate(p, sc):
        nonlocal best_p, best_score
        if p is not None and sc is not None and np.isfinite(sc):
            if sc < best_score:
                best_p = p.copy()
                best_score = float(sc)
            all_candidates.append((float(sc), p.copy()))
            all_candidates.sort(key=lambda x: x[0])
            if len(all_candidates) > 30:
                all_candidates = all_candidates[:30]

    # --- Get parent if available ---
    parent_p = None
    try:
        par = np.asarray(height_sequence_1, dtype=np.float64)
        if par.ndim == 1 and par.size > 0:
            pn = _normalize_nonneg(par)
            if pn is not None:
                parent_p = pn
    except Exception:
        pass

    # --- Determine search length ---
    # We'll primarily work at n=512 since that's what was used before
    n_search = 512

    # --- Phase 1: Generate and score initial profiles ---
    print(f"Phase 1: Generating initial profiles at n={n_search}")
    profiles = _make_profiles(n_search, rng)

    if parent_p is not None:
        if parent_p.size != n_search:
            p_par = _interpolate_profile(parent_p, n_search)
        else:
            p_par = parent_p.copy()
        if p_par is not None:
            profiles.insert(0, p_par)
            # Also keep parent at native length
            if parent_p.size <= 2048:
                sc_par = _score_fast(parent_p)
                add_candidate(parent_p, sc_par)
                print(f"  Parent score: {sc_par:.10f} (n={parent_p.size})")

    # Score all profiles
    for i, p in enumerate(profiles):
        sc = _score_fast(p)
        add_candidate(p, sc)
        if (i + 1) % 10 == 0:
            print(f"  Scored {i+1}/{len(profiles)} profiles, best so far: {best_score:.10f}")

    print(f"Phase 1 done. Best initial: {best_score:.10f}")

    # --- Phase 2: L-BFGS annealing on top profiles ---
    print(f"Phase 2: L-BFGS optimization")
    phase2_deadline = min(deadline, t_start + 1080.0)
    top_k = min(8, len(all_candidates))

    for i in range(top_k):
        if time.monotonic() > phase2_deadline:
            break
        sc_i, p_i = all_candidates[i]
        per = min(phase2_deadline, time.monotonic() + 120.0)
        try:
            p_best, sc_best = _anneal_lbfgs(p_i, per, rng, max_iter=800)
            add_candidate(p_best, sc_best)
            print(f"  L-BFGS on candidate {i}: score {sc_i:.8f} -> {sc_best:.8f}")
        except Exception as e:
            print(f"  L-BFGS error on candidate {i}: {e}")

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # --- Phase 3: Direct simplex (multiplicative update) ---
    print(f"Phase 3: Direct simplex optimization")
    phase3_deadline = min(deadline, t_start + 1500.0)
    top_k3 = min(6, len(all_candidates))

    for i in range(top_k3):
        if time.monotonic() > phase3_deadline:
            break
        sc_i, p_i = all_candidates[i]
        per = min(phase3_deadline, time.monotonic() + 80.0)
        try:
            p_best, sc_best = _direct_simplex_opt(p_i, per, rng, lr_init=0.03, iters=8000)
            add_candidate(p_best, sc_best)
            print(f"  Simplex on candidate {i}: score {sc_i:.8f} -> {sc_best:.8f}")
        except Exception as e:
            print(f"  Simplex error on candidate {i}: {e}")

    # Also try with different learning rates
    for lr_try in (0.01, 0.08):
        if time.monotonic() > phase3_deadline:
            break
        p_i = best_p.copy()
        per = min(phase3_deadline, time.monotonic() + 60.0)
        try:
            p_best, sc_best = _direct_simplex_opt(p_i, per, rng, lr_init=lr_try, iters=6000)
            add_candidate(p_best, sc_best)
            print(f"  Simplex lr={lr_try}: score -> {sc_best:.8f}")
        except Exception as e:
            print(f"  Simplex lr={lr_try} error: {e}")

    print(f"Phase 3 done. Best: {best_score:.10f}")

    # --- Phase 4: Final L-BFGS polish with tighter annealing ---
    print(f"Phase 4: Final polish")
    phase4_deadline = min(deadline, t_start + budget_s - 60.0)

    for i in range(min(5, len(all_candidates))):
        if time.monotonic() > phase4_deadline:
            break
        sc_i, p_i = all_candidates[i]
        per = min(phase4_deadline, time.monotonic() + 60.0)
        try:
            p_best, sc_best = _anneal_lbfgs(p_i, per, rng, beta_init=None, max_iter=1500)
            add_candidate(p_best, sc_best)
            print(f"  Polish on candidate {i}: {sc_i:.8f} -> {sc_best:.8f}")
        except Exception:
            pass

    # --- Phase 5: Try a few more random restarts if time allows ---
    if time.monotonic() < deadline - 120 and best_score > 1.503:
        print(f"Phase 5: Additional random restarts")
        for _ in range(5):
            if time.monotonic() > deadline - 120:
                break
            # Perturb best solution
            noise = rng.normal(0, 0.2, n_search)
            p_try = best_p * np.exp(noise)
            p_try = p_try / p_try.sum()
            sc_try = _score_fast(p_try)
            add_candidate(p_try, sc_try)

            per = min(deadline - 120, time.monotonic() + 40.0)
            p_best, sc_best = _anneal_lbfgs(p_try, per, rng, max_iter=500)
            add_candidate(p_best, sc_best)
            print(f"  Restart: {sc_best:.8f}")

    # --- Final evaluation ---
    print(f"Phase 6: Final exact evaluation")
    final_p = best_p.copy()
    final_score = best_score

    # Exact score
    if final_p is not None and final_p.size <= 4096:
        sc_exact = _score_exact(final_p)
        if np.isfinite(sc_exact):
            print(f"  Exact score: {sc_exact:.10f}")
            final_score = min(final_score, sc_exact)

    # Also evaluate top candidates exactly
    for sc_c, p_c in all_candidates[:10]:
        if p_c.size <= 4096:
            sc_c_ex = _score_exact(p_c)
            if np.isfinite(sc_c_ex) and sc_c_ex < final_score:
                final_score = sc_c_ex
                final_p = p_c.copy()
                print(f"  Found better exact: {sc_c_ex:.10f}")

    # Call evaluate_sequence for authoritative check
    if final_p is not None and final_p.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_p]))
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < final_score:
                    final_score = sc_eval
        except Exception:
            pass

    # Normalize and clip final
    final_p = _normalize_nonneg(final_p)
    if final_p is None:
        # Fallback
        t = np.linspace(0, 1, n_search)
        final_p = 1.0 + 1.5 * (t - 0.5)
        final_p = np.maximum(final_p, 1e-6)
        final_p = final_p / final_p.sum()

    final_p = np.clip(final_p, 0.0, 1000.0)
    s = float(final_p.sum())
    if s > 0:
        final_p = final_p / s

    print(f"FINAL: score={final_score:.10f}, n={final_p.size}")
    return [float(x) for x in final_p]
