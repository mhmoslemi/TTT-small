def build_A(f):
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # For row k, column i has value f[k-i] if 0 <= k-i < n
    # k-i is in [0, n-1] iff max(0, k-n+1) <= i <= min(n-1, k)
    # For row k, the values are f[k-min(n-1,k) : k-max(0,k-n+1)+1] in reverse
    # Let me just do it with advanced indexing
    k_idx = np.arange(rows)
    i_idx = np.arange(n)
    # Create meshgrid
    K, I = np.meshgrid(k_idx, i_idx, indexing='ij')
    diff = K - I
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((rows, n), dtype=np.float64)
    A[mask] = f[diff[mask]]
    return A

    f(seq) = 2*n*max(convolve(seq, seq)) / sum(seq)^2

  1. L-BFGS-B with log-sum-exp smoothed max-convolution objective

  2. Full-constraint LP-based refinement (no cutting planes, for n <= ~1500)

  4. Adaptive time management across phases

import numpy as np
from scipy.optimize import minimize, linprog
import time


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def convolve_fft(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    L = next_power_of_two(2 * n - 1)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def fast_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    c = convolve_fft(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * np.max(c) / (s * s))


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / max(1, len(a))
    return a / s


def greedy_normalized(n):
    a = np.empty(int(n), dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_sum1(a)

    idx = np.arange(int(n), dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_sum1(a)

def smooth_obj_and_grad(x, alpha):
    """Compute S and dS/dx where a = softmax(x), S = (1/alpha)*log(sum(exp(alpha*c)), c = a*a."""
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    se = float(e.sum())
    if se < 1e-300:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = e / se

    c = convolve_fft(a)

    ac = alpha * c
    ac_max = float(ac.max())
    ac_shifted = ac - ac_max
    exp_sum = float(np.sum(np.exp(ac_shifted)))
    if exp_sum < 1e-300:
        S = ac_max / alpha
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        S = (ac_max + np.log(exp_sum)) / alpha
        w = np.exp(ac_shifted) / exp_sum

    # Gradient of S w.r.t. a: grad_a_i = sum_l w_{i+l} * a_l (cross-correlation)
    L2 = next_power_of_two(3 * n)
    Wc = np.fft.fft(w, L2)
    Ac = np.fft.fft(a, L2)
    corr = np.fft.ifft(np.conj(Wc) * Ac).real
    grad_a = corr[:n]

    # Gradient w.r.t. x: grad_x = a * (grad_a - a^T @ grad_a)
    ga = float(np.dot(a, grad_a))
    grad_x = a * (grad_a - ga)

    return float(S), grad_x


def lbfgsb_run(a_init, alpha, time_limit, maxiter):
    """Run L-BFGS-B on the smoothed objective with softmax parameterization."""
    a0 = normalize_sum1(a_init)
    n = a0.size
    x0 = np.log(a0 + 1e-30)
    x0 -= x0.mean()

    t0 = time.monotonic()
    call_count = [0]

    def obj_func(x):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return 1e10
        val, _ = smooth_obj_and_grad(x, alpha)
        return val if np.isfinite(val) else 1e10

    def grad_func(x):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(x)
        _, g = smooth_obj_and_grad(x, alpha)
        g -= g.mean()
        if not np.all(np.isfinite(g)):
            return np.zeros_like(x)
        return g

    try:
        res = minimize(obj_func, x0, jac=grad_func, method='L-BFGS-B',
                       options={'maxiter': int(maxiter), 'max_fun': 20000,
                                'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 30})
    except Exception:
        return a0

    x = np.asarray(res.x, dtype=np.float64).ravel()
    x = x - x.max()
    e = np.exp(x)
    se = float(e.sum())
    if se < 1e-300:
        return a0
    return e / se


def build_lp_matrix(f):
    """Build the full constraint matrix A where A[k,i] = f[k-i] if 0<=k-i<n."""
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    k_idx = np.arange(rows)
    i_idx = np.arange(n)
    K, I = np.meshgrid(k_idx, i_idx, indexing='ij')
    diff = K - I
    mask = (diff >= 0) & (diff < n)
    A[mask] = f[diff[mask]]
    return A


def lp_refine_full(a_init, time_limit, max_iters=300):
    """Full-constraint LP-based refinement without cutting planes."""
    a0 = normalize_sum1(a_init)
    n = a0.size
    if n < 2 or n > 2500:
        return a0, fast_score(a0)

    S0 = np.sqrt(2.0 * n)
    f = a0 * S0
    M = float(np.max(convolve_fft(f)))
    cur_score = float(2.0 * n * M / (S0 * S0))

    best_f = f.copy()
    best_score = cur_score

    t0 = time.monotonic()
    end = t0 + max(1.0, float(time_limit) - 5.0)
    it = 0

    while time.monotonic() < end and it < int(max_iters):
        it += 1
        rem = end - time.monotonic()
        if rem < 5.0:
            break

        # Build constraint matrix
        A = build_lp_matrix(f)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        tl = max(1.0, min(60.0, rem * 0.25))
        try:
            res = linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None),
                          method='highs', options={'time_limit': float(tl), 'presolve': True})
        except Exception:
            break

        if res is None or not res.success or res.x is None:
            break

        g = np.clip(np.asarray(res.x, dtype=np.float64).ravel(), 0.0, None)
        sg = float(g.sum())
        if sg < 1e-15:
            break

        g_scaled = g * (S0 / sg)

        # Best improvement test: if sum(g) <= sum(f) + eps, no improvement possible
        if sg <= S0 * (1.0 + 1e-10):
            break

        # Line search with golden section + direct evaluation
        best_t = 0.0
        best_sc = best_score
        best_fn = None

        # Try a range of step sizes
        for t in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01):
            fn = (1.0 - t) * f + t * g_scaled
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn < 1e-15:
                continue
            fn_norm = fn * (S0 / sn)
            M_new = float(np.max(convolve_fft(fn_norm)))
            sc = float(2.0 * n * M_new / (S0 * S0))
            if sc < best_sc:
                best_sc = sc
                best_t = t
                best_fn = fn_norm.copy()

        if best_fn is None or best_t <= 0.0:
            break

        f = best_fn
        M = float(np.max(convolve_fft(f)))
        best_score = best_sc
        best_f = f.copy()

        if it % 5 == 0:
            print("    LP iter %d: score=%.8f, t=%.3f, it_time=%.1fs" % (it, best_score, best_t, time.monotonic() - t0))

        # Convergence check
        if best_score < 1.5000:
            break

    return normalize_sum1(best_f), best_score


def adam_polish(a_init, time_limit, lr_base=0.02):
    """Adam optimizer with smoothed max objective (backup for large n)."""
    a = normalize_sum1(a_init)
    n = a.size
    x = np.log(a + 1e-30)
    x -= x.mean()

    L = next_power_of_two(3 * n - 2)
    mu, nu, eps = 0.9, 0.999, 1e-8
    m_arr = np.zeros(n, dtype=np.float64)
    v_arr = np.zeros(n, dtype=np.float64)
    t = 0

    best_p = a.copy()
    best_f = fast_score(a)

    t0 = time.monotonic()
    end = t0 + max(0.0, float(time_limit) - 2.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(1e-9, float(time_limit)))
        alpha = 10000.0 + 80000.0 * np.exp(-3.0 * frac)
        lr = float(lr_base) * (1.0 + 2.0 * (1.0 - frac))

        pag = np.fft.rfft(a, L)
        c = np.fft.irfft(pag * pag, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if f_now < best_f:
            best_f = f_now
            best_p = a.copy()

        z = alpha * (c / maxc - 1.0)
        z -= z.max()
        exp_z = np.exp(z)
        sz = float(exp_z.sum())
        if sz < 1e-300:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exp_z / sz

        # Gradient via FFT cross-correlation
        Wc = np.fft.fft(w, L)
        Ac = np.fft.fft(a, L)
        corr = np.fft.ifft(np.conj(Wc) * Ac).real[:n]
        grad_a = np.maximum(corr, 0.0)

        ga = float(np.dot(a, grad_a))
        grad_x = a * (grad_a - ga)

        m_arr = mu * m_arr + (1.0 - mu) * grad_x
        v_arr = nu * v_arr + (1.0 - nu) * (grad_x * grad_x)
        m_hat = m_arr / (1.0 - mu ** t)
        v_hat = v_arr / (1.0 - nu ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -70.0, 70.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if se < 1e-300:
            break
        a = e / se

    final_f = fast_score(a)
    if final_f < best_f:
        best_f = final_f
        best_p = a.copy()
    return best_p, best_f


def scale_output(a, target_sum=10.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return [1.0 / 16.0] * 16
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]

def record_candidate(p, state):
    """Record a candidate. state is a dict with keys: best_p, best_score, best_by_n."""
    if p is None:
        return
    a = np.asarray(p, dtype=np.float64).ravel()
    s = float(a.sum())
    if s < 1e-10 or a.size == 0:
        return
    a = a / s
    sc = fast_score(a)
    if not np.isfinite(sc):
        return
    if sc < state['best_score']:
        state['best_p'] = a.copy()
        state['best_score'] = float(sc)
        print("  NEW BEST: score=%.8f, n=%d" % (sc, a.size))
    n = a.size
    if n not in state['best_by_n'] or sc < state['best_by_n'][n][0]:
        state['best_by_n'][n] = (float(sc), a.copy())
