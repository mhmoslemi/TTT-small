def convolve(a, b=None):
    """Compute auto-convolution efficiently."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)

        return np.convolve(a, b)
    # FFT-based

    while size < n + m - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n+m-1]

def softmax(x, beta=1.0):

    s = np.sum(e)
    if s < 1e-300:
        out = np.zeros_like(x)
        out[np.argmax(x)] = 1.0
        return out
    return e / s
