def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size

def simplex_project(v, target=1.0):
    """Euclidean projection of v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 1:
        return np.array([target])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv - target) / rho
    rho_max = int(np.max(np.where(cond)[0])) + 1 if np.any(cond) else 0
    if rho_max == 0:
        theta = 0.0
    else:
        theta = (cssv[rho_max - 1] - target) / rho_max

def resize(a, N):
    """Resize sequence to length N via interpolation."""
    a = normalize_nonneg(a)
    m = a.size
    if m == N:
        return a.copy()
    x_new = np.linspace(0, 1.0, N)
    x_old = np.linspace(0, 1.0, m)
    new_a = np.interp(x_new, x_old, a)
    return normalize_nonneg(new_a)

def coord_descent(a, time_limit, rng, state):
    """Coordinate descent: optimize one or a few elements at a time."""
    a = normalize_nonneg(a.copy())
    n = a.size
    deadline = time.monotonic() + time_limit
    cur = score_norm(a)
    best = a.copy()
    best_sc = cur
    
    while time.monotonic() < deadline:
        # Pick random coordinates to adjust
        n_coords = min(n, int(rng.integers(5, 50)))
        indices = rng.choice(n, size=n_coords, replace=False)
        
        improved = False
        for idx in indices:
            # Try to optimize a[idx] by golden section search
            # The score as a function of a[idx] (with rest fixed, then renormalize)
            a_save = a.copy()
            a_save_sum_rest = a_save.sum() - a_save[idx]
            
            # We vary a[idx] and renormalize
            # Actually, it's easier to just try a few values
            base = a[idx]
            for factor in [0.5, 0.8, 1.2, 1.5, 2.0, 0.3]:
                a_trial = a.copy()
                a_trial[idx] = base * factor
                a_trial = normalize_nonneg(a_trial)
                s = score_norm(a_trial)

        if cur < best_sc - 1e-12:

    final_sc = score_norm(a)
