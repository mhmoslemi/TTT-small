def soft_max_val(c, temperature):
    """Compute soft max value of array c."""
    c = np.asarray(c, dtype=np.float64)
    m = float(c.max())
    if temperature < 1e-12:
        return m
    x = (c - m) / temperature
    # Clip for numerical stability
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        return m
    return m + temperature * float(np.log(s))


def soft_max_weights(c, temperature):

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # grad_a[j] = 2 * conv_w_af[n-1+j] for j=0..n-1
        grad_a = 2.0 * conv_w_af[n - 1:2 * n - 1]
        
        # Now convert to z-gradient: a = softmax(z), so da/dz = a*(I - a*a^T) (Jacobian of softmax)
        # grad_z = a * (grad_a - sum(a * grad_a))
        # (This is the standard softmax gradient trick)
        grad_a_dot_a = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - grad_a_dot_a)
        
        # Also add small regularization to prevent z from drifting
        grad_z += 1e-8 * (z - z.mean())
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m_state.fill(0)
            v_state.fill(0)
            t_adam = 0
            continue
        
        # Adam update
        t_adam += 1
        m_state = b1 * m_state + (1 - b1) * grad_z
        v_state = b2 * v_state + (1 - b2) * grad_z ** 2
        m_hat = m_state / (1 - b1 ** t_adam)
        v_hat = v_state / (1 - b2 ** t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        # Update temperature
        if t_adam % 50 == 0:
            temperature = max(1e-8, temperature * 0.85)
        
        # Check improvement
        if exact_score < best_score - 1e-12:
            best_score = exact_score
            best_a = a.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr = max(0.01, lr * 0.95)
        
        if no_improve > 300:
            # Restart with perturbation
            z = best_z + 0.3 * rng.normal(size=n)
            z = z - z.mean()
            z = np.clip(z, -50.0, 50.0)
            m_state.fill(0)
            v_state.fill(0)
            t_adam = 0
            lr = base_lr
            temperature = max(1e-6, 0.1 * float(fft_convolve_self(best_a).max()))
            no_improve = 0
    
    return best_z, best_a, best_score
