    # Actually, let's use the top_k lags
    idx = np.argsort(C)[::-1][:top_k]
    
    # Build LP
    # Variables: d[0..n-1], t  (total n+1 variables)
    # Objective: minimize t -> c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: (d*a)[k] - t <= 0 for k in idx
    # (d*a)[k] = sum_i d[i] * a[k-i] for valid i
    n_cons = len(idx)
    A_ub = np.zeros((n_cons, n + 1))
    b_ub = np.zeros(n_cons)
    for r, k in enumerate(idx):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[r, i] = a[j]
        A_ub[r, -1] = -1.0  # coefficient of t
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -a[i]/alpha <= d[i] <= alpha, t unbounded below
    bounds = [(-a[i] / alpha, alpha) for i in range(n)]
    bounds.append((None, None))  # t is free
    
    try:

                     bounds=bounds, method='highs')

            d = res.x[:n]
            t_opt = res.x[-1]
            if t_opt < -1e-10:
                return d, t_opt
            else:
                return None, None
        else:
            return None, None
    except Exception:
        return None, None


def take_step(a, d, C, cmax, n):
    """
    Take a step in direction d, finding the best step size via line search.
    Returns (a_new, score_new, t_used).
    """
    # Max step from non-negativity
    t_max = 1.0
    neg_mask = d < -1e-15
    if np.any(neg_mask):
        ratios = a[neg_mask] / (-d[neg_mask])
        t_max = min(t_max, float(np.min(ratios)))
    
    if t_max <= 0:
        return a, compute_score(a), 0.0
    
    # Try different step fractions
    best_a = a
    best_score = compute_score(a)
    best_t = 0.0
    
    for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07, 0.03]:
        t = t_max * frac
        if t <= 1e-15:
            continue
        a_new = a + t * d
        if a_new.min() < -1e-12:
            continue
        a_new = np.maximum(a_new, 0.0)
        # Normalize to maintain sum = 1
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        C_new = full_convolve(a_new)
        sc = float(2.0 * n * C_new.max() / 1.0)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            best_t = t
    
    return best_a, best_score, best_t


def active_set_search(a, deadline, rng, verbose=True):
    """
    Main active set search loop.
    a is initialized to sum to 1.
    """
    n = a.size
    a = a.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    C = full_convolve(a)
    cmax = C.max()
    
    no_improve = 0
    top_k = 10
    alpha = 5.0
    iter_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        # Find LP direction
        d, t_opt = lp_descent_direction(a, C, top_k=top_k, alpha=alpha)
        
        if d is None:
            # No improvement found - try relaxing
            no_improve += 1
            if no_improve == 5:
                top_k = max(3, top_k // 2)
            elif no_improve == 10:
                alpha = min(20.0, alpha * 1.5)
            elif no_improve == 20:
                # Perturb and restart
                a = a + 0.01 * rng.normal(size=n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                C = full_convolve(a)
                top_k = 10
                alpha = 5.0
                no_improve = 0
            continue
        
        # Take a step
        a_new, sc_new, t_used = take_step(a, d, C, cmax, n)
        
        if sc_new < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = sc_new
            a = a_new
            C = full_convolve(a)
            cmax = C.max()
            no_improve = 0
            if verbose and iter_count % 100 == 0:
                print(f"  AS iter={iter_count} score={best_score:.8f} t={t_used:.4f}")
        else:
            no_improve += 1
            if no_improve > 30:
                # Reset
                a = best_a + 0.005 * rng.normal(size=n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                C = full_convolve(a)
                cmax = C.max()
                no_improve = 0
                top_k = 10
                alpha = 5.0
    
    return best_a, best_score


def softmax_adam(a0, deadline, rng, lr=0.1, top_k_smooth=50):
    """
    Soft-max smoothed Adam optimization in logit space.
    a0 sums to 1.
    """
    n = a0.size
    z = np.log(np.maximum(a0, 1e-30))
    z -= z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    
    p = a0.copy()
    C = full_convolve(p)
    cmax = C.max()
    beta = 0.2 * cmax
    
    best_p = p.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * cmax)
    no_imp = 0
    
    while time.monotonic() < deadline:
        # Soft max weights
        x = (C - cmax) / beta
        ex = np.exp(x)
        s = ex.sum()
        w = ex / s
        L = cmax + beta * np.log(s)
        
        # Gradient of L w.r.t. p: 2 * corr(w, p)
        # corr(w, p)[i] = sum_j w[i+j] * p[j] (adjusted for bounds)
        # Use FFT
        N = next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        gp = 2.0 * h[n - 1:2 * n - 1]
        
        # Project to tangent space of simplex
        dot = np.dot(p, gp)
        gz = p * (gp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -30, 30)
        z -= z.mean()
        
        # Softmax
        e = np.exp(z - z.max())
        p = e / e.sum()
        
        C = full_convolve(p)
        cmax = C.max()
        score = float(2.0 * n * cmax)
        
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.02, lr * 0.97)
        
        if t_adam % 80 == 0:
            beta = max(1e-6, beta * 0.8)
        
        if no_imp > max(300, 3 * n):
            z = 0.8 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -30, 30)
            z -= z.mean()
            e = np.exp(z - z.max())
            p = e / e.sum()
            C = full_convolve(p)
            cmax = C.max()
            beta = 0.2 * cmax
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = 0.1
            no_imp = 0
    
    return best_p, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64)
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0)
    s = p1.sum()
    if s < 1e-10:
        return None
    return p1 / s

"""
Active Set LP + Soft-max Adam hybrid search for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovations over previous approach:
1. LP-based descent directions using active set of peak convolution lags
2. Precomputed correlation matrix for fast LP construction
3. Multi-scale search (128 -> 512 -> 1024)
4. Combined LP + Adam with adaptive switching
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def build_corr_matrix(a):
    """Build matrix M where M[k,i] = a[k-i] if 0<=k-i<n else 0. (2n-1) x n."""
    n = a.size
    M = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        length = i_end - i_start + 1
        M[k, i_start:i_end + 1] = a[k - i_end:k - i_start + 1][::-1]
    return M


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_quartic_profile(n, a1, a3, a4):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3 + a4 * u ** 4
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_profiles(n):
    profs = []
    for eta in [1.0, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0, 3.5, 4.0]:
        profs.append(make_linear_profile(n, eta))
    for a1, a3 in [(1.5, -1.0), (2.0, -2.0), (2.5, -3.0), (1.8, -1.5),
                   (2.0, 0.0), (3.0, -4.0), (2.5, -1.0), (2.0, -0.5)]:
        profs.append(make_cubic_profile(n, a1, a3))
    for a1, a3, a4 in [(2.0, -2.0, 1.0), (2.5, -3.0, 2.0), (1.5, -1.0, 0.5),
                       (2.0, 0.0, 1.5), (3.0, -4.0, 3.0)]:
        profs.append(make_quartic_profile(n, a1, a3, a4))
    for eta in [1.5, 2.0, 3.0]:
        t = np.linspace(0.0, 1.0, n)
        a = np.power(np.maximum(1.0 + eta * (t - 0.5), 0.01), 3.0)
        profs.append(a / a.sum())
    for p in [2.0, 3.0, 5.0, 8.0]:
        t = np.linspace(0.0, 1.0, n)
        a = np.power(t, p) + np.power(1.0 - t, p)
        a = np.maximum(a, 1e-10)
        profs.append(a / a.sum())
    # Exponential ramp
    for lam in [2.0, 3.0, 5.0]:
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(lam * t)
        profs.append(a / a.sum())
    return profs


def lp_descent_direction(a, M, lags, alpha=5.0):
    """
    Find descent direction via LP.
    Minimize t s.t. M[k,:]@d - t <= 0 for k in lags, sum(d)=0, bounds.
    Returns (d, t_opt) or (None, None).
    """
    n = a.size
    A_ub_rows = M[lags, :].copy()  # shape (len(lags), n)
    n_lags = len(lags)
    
    # Full constraint matrix: [A_ub_rows | -1] (n_lags x (n+1))
    A_ub = np.zeros((n_lags, n + 1), dtype=np.float64)
    A_ub[:, :n] = A_ub_rows
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_lags)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    lb = -a / alpha
    ub = np.full(n, alpha)
    bounds = list(zip(lb, ub))
    bounds.append((None, None))  # t free
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            d = res.x[:n]
            t_opt = res.x[n]
            if t_opt < -1e-12:
                return d, float(t_opt)
            return None, None
        return None, None
    except Exception:
        return None, None


def take_step(a, d, n):
    """Line search along direction d. Returns (a_new, score, t)."""
    t_max = 1.0
    neg = d < -1e-14
    if np.any(neg):
        t_max = min(t_max, float(np.min(a[neg] / (-d[neg]))))
    t_max = min(t_max, 0.5)
    if t_max <= 1e-14:
        return a, compute_score(a), 0.0
    
    best_a = a
    best_sc = compute_score(a)
    best_t = 0.0
    
    for frac in [1.0, 0.8, 0.6, 0.4, 0.25, 0.12, 0.06]:
        t = t_max * frac
        if t < 1e-14:
            continue
        a_new = a + t * d
        if a_new.min() < -1e-10:
            continue
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = compute_score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()
            best_t = t
    
    return best_a, best_sc, best_t

def softmax_adam(a0, deadline, rng, lr=0.1):

    n = a0.size

    t_adam = 0

    cmax = C.max()
    beta = 0.15 * cmax
    
    best_p = p.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * cmax)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        x = (C - cmax) / beta
        ex = np.exp(np.clip(x, -50, 50))
        s = ex.sum()
        if s < 1e-30:
            ex = np.zeros_like(C)
            ex[int(np.argmax(C))] = 1.0
            s = 1.0
        w = ex / s
        L = cmax + beta * np.log(s)

        fp = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        gp = 2.0 * h[n - 1:2 * n - 1]
        
        # Project to tangent of simplex
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()
        
        e = np.exp(z - z.max())
        p = e / e.sum()

            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.01, lr * 0.97)
        
        if t_adam % 60 == 0:
            beta = max(1e-8, beta * 0.85)

            z -= z.mean()
            e = np.exp(z - z.max())
            p = e / e.sum()
            C = full_convolve(p)
            cmax = C.max()
            beta = 0.15 * cmax
            m.fill(0)
            v.fill(0)

    return best_p, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64)
    n0 = p0.size
    if n0 == n_new:
        p = p0.copy()
    else:
        x0 = np.linspace(0.0, 1.0, n0)
        x1 = np.linspace(0.0, 1.0, n_new)
        p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return None
    return p / s
