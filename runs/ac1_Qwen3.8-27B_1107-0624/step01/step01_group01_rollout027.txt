"""
Optimization of non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses theoretical initialization (binomial-coefficient shape), progressive length
scaling, and aggressive logit-space Adam with soft-max annealing.
Key insight: the one-sided ideal shape C(2k,k)/4^k gives flat autocorrelation
for the first n lags, yielding score ~ pi/2. We improve beyond this by
allowing the optimizer to find shapes with larger sum while maintaining
low peak autocorrelation.

import numpy as np
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_max_weights(c, beta):
    """Return (soft_max_value, weights) for smooth max approximation."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 (cross-correlation)."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def internal_score(a):

    n = a.size
    if n == 0:

    s = float(a.sum())

    if not np.isfinite(mval):

    return float(2.0 * n * mval / (s * s))

def binomial_shape(n):
    """One-sided ideal: C(2k,k)/4^k for k=0..n-1, normalized to sum 1."""
    k = np.arange(n, dtype=np.float64)
    # Use log for numerical stability
    # log(C(2k,k)/4^k) = log(C(2k,k)) - k*log(4)
    # log(C(2k,k)) = sum_{j=1}^{k} log(2k-j+1) - log(j)
    # For large k, use Stirling or direct computation
    # a[k] ~ 1/sqrt(pi*k) for large k
    log_a = np.zeros(n)
    if n > 1:
        # Compute log(C(2k,k)) - k*log(4) using cumulative sums
        # log(C(2k,k)) = log((2k)!) - 2*log(k!)
        # For numerical stability, use the fact that a[0]=1 and
        # a[k]/a[k-1] = (2k-1)/(2k)
        ratios = (2.0 * np.arange(1, n) - 1.0) / (2.0 * np.arange(1, n))
        log_ratios = np.log(ratios)
        log_a = np.concatenate([[0.0], np.cumsum(log_ratios)])
    a = np.exp(log_a)
    s = float(a.sum())
    return a / s


def tophat_shape(n, taper_frac=0.3):
    """Top-hat with 1/sqrt(k) tapers at edges."""
    t = np.arange(n, dtype=np.float64)
    K = max(1, int(taper_frac * n))
    a = np.ones(n)
    # Left taper
    left = np.minimum(t, K * np.float64(1.0))
    a[:K] = np.power(1.0 - left[:K] / (K + 1), 0.5)
    # Right taper
    right = np.minimum(n - 1 - t, K * np.float64(1.0))
    a[-K:] = np.power(1.0 - right[-K:] / (K + 1), 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def smooth_taper_shape(n, alpha=0.5):
    """Symmetric smooth taper: a[k] = (sin(pi*(k+0.5)/n))^alpha."""
    t = np.arange(n, dtype=np.float64)
    a = np.power(np.sin(np.pi * (t + 0.5) / n), alpha)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def linear_shape(n, eta=1.5):
    """Monotone linear: 1 + eta*(t - 0.5)."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def cos_taper(n, p=1.0):
    """Cosine taper: a[k] = (cos(pi*(t-0.5)))^p for t in [0,1]."""
    t = np.linspace(0.0, 1.0, n)
    a = np.power(np.maximum(np.cos(np.pi * (t - 0.5)), 0.0), p)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def all_initializations(n, rng):
    """Generate a diverse set of initial profiles."""
    profs = []
    profs.append(binomial_shape(n))
    profs.append(binomial_shape(n)[::-1])
    # Mix of binomial and uniform
    mix = 0.5 * binomial_shape(n) + 0.5 * np.ones(n) / n
    profs.append(mix / mix.sum())
    mix2 = 0.7 * binomial_shape(n) + 0.3 * np.ones(n) / n
    profs.append(mix2 / mix2.sum())
    # Top-hat variants
    for tf in (0.2, 0.3, 0.5):
        profs.append(tophat_shape(n, tf))
    # Smooth tapers
    for alpha in (0.3, 0.5, 0.7, 1.0, 1.5):
        profs.append(smooth_taper_shape(n, alpha))
    # Cosine tapers
    for p in (0.5, 1.0, 2.0, 3.0):
        profs.append(cos_taper(n, p))
    # Linear shapes
    for eta in (0.5, 1.0, 1.5, 2.0):
        profs.append(linear_shape(n, eta))
    # Random smooth perturbations of binomial
    base = binomial_shape(n)
    for _ in range(4):
        noise = 0.1 * rng.normal(size=n)
        z = np.log(base) + noise
        profs.append(softmax_stable(z))
    # Product of left and right binomial
    left = binomial_shape(n)
    right = binomial_shape(n)[::-1]
    prod = left * right
    profs.append(prod / prod.sum())
    # Geometric decay (exponential taper)
    for rate in (0.01, 0.02, 0.05):
        a = np.exp(-rate * np.arange(n))
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
        profs.append(a[::-1] / a.sum())
    return profs


def logit_adam_optimize(p0, deadline, rng, lr=0.06, reg=1e-8, beta_scale=0.3):
    """
    Adam optimization in logit space to minimize 2*n*max(conv(p,p)).
    p0: initial profile (sum 1, non-negative)
    Returns (best_p, best_score) where best_score = 2*n*max(conv).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_scale * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    start_time = time.monotonic()

    while time.monotonic() < deadline:
        L, w = soft_max_weights(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_stable(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.01, lr * 0.96)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 50 == 0:
            beta = max(1e-12, beta * 0.7)

        # Restart condition
        if no_imp > max(400, 8 * n):
            # Perturb in z-space
            z = 0.85 * best_z + 0.25 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_stable(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, beta_scale * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def logit_adam_optimize_v2(p0, deadline, rng, lr=0.05):
    """
    Enhanced Adam with variance-based adaptive beta and periodic
    'flattening' steps that push the autocorrelation toward uniformity.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    betamax = maxc
    beta = max(1e-12, 0.25 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    flatten_count = 0

    while time.monotonic() < deadline:
        L, w = soft_max_weights(c, beta)

        # Gradient of soft-max w.r.t. p
        gp = 2.0 * corr_weights(w, p)

        # Add flattening regularization: push c toward its mean
        # This encourages a more uniform autocorrelation
        c_mean = float(np.mean(c))
        c_var_weights = (c - c_mean) / (float(np.std(c)) + 1e-12)
        # Gradient of variance w.r.t. p: 2 * c_var_weights convolved with p
        gp_flat = 2.0 * corr_weights(c_var_weights, p)
        # Combined: minimize max + small flattening term
        gp_total = gp + 0.05 * gp_flat

        vp = gp_total + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_stable(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 400 == 0:
                    lr = max(0.01, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        # Adaptive beta: reduce when close to convergence
        if adam_t % 40 == 0:
            beta = max(1e-12, beta * 0.72)

        # Periodic flattening: explicitly nudge p toward more uniform
        if no_imp > 200 and no_imp % 50 == 0:
            uniform_z = np.zeros(n)
            z = 0.97 * z + 0.03 * uniform_z
            z -= float(np.mean(z))
            p = softmax_stable(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            score = float(2.0 * n * maxc)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            flatten_count += 1

        # Full restart
        if no_imp > max(500, 10 * n):
            mix = 0.8 * best_z + 0.2 * rng.normal(size=n)
            z = np.clip(mix, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_stable(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.25 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 1e-12)
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-12)
    return p1 / p1.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
      1. Generate diverse initializations (binomial, top-hat, cosine, etc.)
      2. Optimize at n=128 with multiple starts (quick phase)
      3. Refine best at n=256
      4. Refine best at n=512 with enhanced optimizer
      5. If time permits, try n=1024
      6. Return best sequence found
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def local_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Try to use parent sequence if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            parent = np.maximum(parent, 0.0)
            s = float(parent.sum())
            if s > 1e-6:
                parent_norm = parent / s
                sc_parent = internal_score(parent_norm)
                if np.isfinite(sc_parent):
                    local_update(parent_norm, sc_parent)
                    print(f"Parent score: {sc_parent:.10f} (n={parent_norm.size})")
    except Exception:
        pass

    if best_score < np.inf:
        print(f"Initial best: {best_score:.10f}")

    # Phase 1: Quick optimization at n=128
    n1 = 128
    p1_end = min(deadline, t_start + 240.0)
    profs_128 = all_initializations(n1, rng)
    print(f"Phase 1: {len(profs_128)} initializations at n={n1}")

    candidates_128 = []
    for i, p0 in enumerate(profs_128):
        if time.monotonic() > p1_end:
            break
        per = min(p1_end, time.monotonic() + 8.0)
        p_best, sc = logit_adam_optimize(p0, per, rng, lr=0.07)
        sc_full = float(2.0 * n1 * sc)
        local_update(p_best, sc_full)
        candidates_128.append((sc_full, p_best))
        if (i + 1) % 5 == 0:
            print(f"  Phase 1: {i+1}/{len(profs_128)} done, best={best_score:.10f}")

    # Sort candidates
    candidates_128.sort(key=lambda x: x[0])
    print(f"Phase 1 complete: best={best_score:.10f}")

    # Phase 2: Refine top candidates at n=256
    n2 = 256
    p2_end = min(deadline, t_start + 600.0)
    print(f"Phase 2: refining at n={n2}")

    candidates_256 = []
    for i in range(min(8, len(candidates_128))):
        if time.monotonic() > p2_end:
            break
        p0 = interpolate_profile(candidates_128[i][1], n2)
        per = min(p2_end, time.monotonic() + 30.0)
        p_best, sc = logit_adam_optimize_v2(p0, per, rng, lr=0.05)
        sc_full = float(2.0 * n2 * sc)
        local_update(p_best, sc_full)
        candidates_256.append((sc_full, p_best))
        print(f"  Phase 2: candidate {i} -> score={sc_full:.10f}, best={best_score:.10f}")

    # Also try fresh initializations at n=256
    fresh_256 = all_initializations(n2, rng)
    for i, p0 in enumerate(fresh_256[:6]):
        if time.monotonic() > p2_end:
            break
        per = min(p2_end, time.monotonic() + 25.0)
        p_best, sc = logit_adam_optimize_v2(p0, per, rng, lr=0.05)
        sc_full = float(2.0 * n2 * sc)
        local_update(p_best, sc_full)
        candidates_256.append((sc_full, p_best))

    candidates_256.sort(key=lambda x: x[0])
    print(f"Phase 2 complete: best={best_score:.10f}")

    # Phase 3: Refine at n=512
    n3 = 512
    p3_end = min(deadline, t_start + 1400.0)
    print(f"Phase 3: refining at n={n3}")

    candidates_512 = []
    for i in range(min(6, len(candidates_256))):
        if time.monotonic() > p3_end:
            break
        p0 = interpolate_profile(candidates_256[i][1], n3)
        per = min(p3_end, time.monotonic() + 50.0)
        p_best, sc = logit_adam_optimize_v2(p0, per, rng, lr=0.04)
        sc_full = float(2.0 * n3 * sc)
        local_update(p_best, sc_full)
        candidates_512.append((sc_full, p_best))
        print(f"  Phase 3: interp candidate {i} -> score={sc_full:.10f}, best={best_score:.10f}")

    # Fresh starts at n=512
    fresh_512 = all_initializations(n3, rng)
    for i, p0 in enumerate(fresh_512[:8]):
        if time.monotonic() > p3_end:
            break
        per = min(p3_end, time.monotonic() + 40.0)
        p_best, sc = logit_adam_optimize_v2(p0, per, rng, lr=0.04)
        sc_full = float(2.0 * n3 * sc)
        local_update(p_best, sc_full)
        candidates_512.append((sc_full, p_best))
        if (i + 1) % 4 == 0:
            print(f"  Phase 3: fresh {i+1}/{min(8, len(fresh_512))}, best={best_score:.10f}")

    candidates_512.sort(key=lambda x: x[0])
    print(f"Phase 3 complete: best={best_score:.10f}")

    # Phase 4: Final polish of the absolute best at n=512
    if time.monotonic() < deadline - 60.0 and best_seq is not None:
        if best_seq.size != n3:
            best_seq = interpolate_profile(best_seq, n3)
        print(f"Phase 4: final polish at n={n3}")
        p0 = best_seq.copy()
        per = deadline - 60.0
        p_best, sc = logit_adam_optimize_v2(p0, per, rng, lr=0.03)
        sc_full = float(2.0 * n3 * sc)
        local_update(p_best, sc_full)
        print(f"Phase 4 complete: best={best_score:.10f}")

    # Phase 5: Try n=1024 if time allows and score is already good
    if time.monotonic() < deadline - 120.0 and best_score < 1.52:
        n4 = 1024
        print(f"Phase 5: trying n={n4}")
        p0 = interpolate_profile(best_seq, n4) if best_seq is not None else binomial_shape(n4)
        per = min(deadline - 60.0, time.monotonic() + 100.0)
        p_best, sc = logit_adam_optimize_v2(p0, per, rng, lr=0.03)
        sc_full = float(2.0 * n4 * sc)
        if np.isfinite(sc_full) and sc_full < best_score:
            local_update(p_best, sc_full)
        print(f"Phase 5: n={n4} score={sc_full:.10f}, best={best_score:.10f}")

    # Final verification with true score
    if best_seq is not None:
        final_seq = best_seq
        final_score = best_score

        # Re-verify with exact convolution
        if final_seq.size <= 2048:
            sc_exact = true_score(final_seq)
            if np.isfinite(sc_exact) and sc_exact < final_score:
                final_score = sc_exact

            # Check candidates too
            for sc_c, p_c in candidates_512:
                if p_c.size <= 2048:
                    sc_c_exact = true_score(p_c)
                    if np.isfinite(sc_c_exact) and sc_c_exact < final_score:
                        final_score = sc_c_exact
                        final_seq = p_c.copy()

        # Final validation call
        if final_seq.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                if np.isfinite(sc_eval) and sc_eval < final_score:
                    final_score = sc_eval
            except Exception:
                pass

        # Ensure valid output
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 1e-6:
            final_seq = final_seq / s
        else:
            final_seq = binomial_shape(512)
            final_score = internal_score(final_seq)

        print(f"FINAL: score={final_score:.10f} n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = binomial_shape(512)
    print(f"FALLBACK: score={internal_score(fallback):.10f}")
    return [float(x) for x in fallback]
