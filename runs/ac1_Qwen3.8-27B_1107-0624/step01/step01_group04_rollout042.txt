    A = np.zeros((n_rows, n), dtype=np.float64)
    for i in range(n_rows):
        k = rows[i]
        # (f*g)[k] = sum_j f[k-j] * g[j]
        # For constraint: sum_j A[i,j] * g[j] <= b[i]
        # A[i,j] = f[k-j] if 0 <= k-j < n, else 0
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        for j in range(lo, hi+1):
            A[i, j] = f[k-j]
    return A

    n = len(p)

    best_p = p.copy()
    best_f = score_fft(p)
    
    start = time.monotonic()
    end = start + max(0, budget_s - 5)
    
    lr = 0.1
    it = 0
    
    while time.monotonic() < end:
        it += 1
        P = np.fft.rfft(p, L)
        c = np.fft.irfft(P*P, L)[:2*n-1].real
        c = np.maximum(c, 0)
        M = c.max()
        M_target = M * (1 - eps)
        
        # Excess
        e = np.maximum(c - M_target, 0)
        e.e... just c - M_target clipped to be non-negative
        
        # Gradient of sum_k e[k] * (c[k] - M_target) w.r.t. p
        # = sum_k (c[k] - M_target) * dc[k]/dp[j] (for c[k] > M_target)
        # dc[k]/dp[j] = p[k-j] (if valid)
        # So grad[j] = sum_{k: c[k]>M_t} (c[k]-M_t) * p[k-j]
        
        e_weighted = np.maximum(c - M_target, 0)
        # grad[j] = sum_k e_weighted[k] * p[k-j]
        # = cross-correlation of e_weighted and p
        
        E = np.fft.rfft(e_weighted, L)
        # cross_corr: sum_k e[k] * p[k-j] = ifft(conj(fft(e)) * fft(p))
        G = np.fft.irfft(np.conj(E) * np.fft.rfft(p, L), L)[:n].real
        
        # Update: p_new = p - lr * G, projected to non-negative simplex
        p_new = p - lr * G
        p_new = np.maximum(p_new, 0)
        s = p_new.sum()
        if s < 1e-12:
            break
        p_new = p_new / s
        
        f_now = score_fft(p_new)
        if f_now < best_f:
            best_f = f_now
            best_p = p_new.copy()
        
        p = p_new
        
        # Adaptive learning rate
        if it % 50 == 0 and f_now > best_f:
            lr *= 0.9
        
        if it % 100 == 0:
            print(f"    spectral it={it} score={best_f:.6f} lr={lr:.4f}")
    
    return best_p, best_f

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using:
1. Multistart Adam on log-simplex with smoothed max objective
2. Cutting-plane LP for precise local optimization
3. Spectral excess-gradient descent as complementary method
4. Multiple sequence lengths with warm-start from height_sequence_1

import time
import numpy as np

try:
    import scipy.optimize as scipy_opt
except Exception:
    scipy_opt = None


def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nn(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(a.size, dtype=np.float64)
    return a / s


def score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    an = a / s
    L = next_pow2(2 * n - 1)
    P = np.fft.rfft(an, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return 2.0 * n * mc


def greedy_init(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nn(a)


def power_init(n, alpha):
    idx = np.arange(1, n + 1, dtype=np.float64)
    return normalize_nn(idx ** (-float(alpha)))


def mixed_init(n, lam):
    g = greedy_init(n)
    u = np.ones(n, dtype=np.float64)
    return normalize_nn(lam * g + (1.0 - lam) * u)


def f Exp_init(n, rate):
    idx = np.arange(n, dtype=np.float64)
    return normalize_nn(np.exp(-float(rate) * idx))


def build_constraint_matrix(f, rows, n):
    n_rows = len(rows)
    A = np.zeros((n_rows, n), dtype=np.float64)
    for i in range(n_rows):
        k = int(rows[i])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[i, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def lp_optimize(p0, budget_s, max_rows=None, verbose=True):
    if scipy_opt is None:
        return p0, score_fft(p0)

    p = normalize_nn(p0)
    n = int(p.size)
    if n < 4:
        n = 16
        p = np.ones(n, dtype=np.float64)
    budget_s = float(budget_s)
    if budget_s < 20.0:
        return p, score_fft(p)

    if max_rows is None:
        if n > 3000:
            max_rows = 700
        elif n > 1000:
            max_rows = 1000
        elif n > 400:
            max_rows = 1300
        else:
            max_rows = 1600

    max_el = 8000000
    while max_rows * n > max_el:
        max_rows = max(80, int(max_rows * 0.8))
    max_rows = max(80, min(max_rows, max_el // max(1, n)))

    S0 = np.sqrt(2.0 * n)
    f = p * S0
    cur_score = score_fft(f)
    best_f = f.copy()
    best_score = float(cur_score)

    end = time.monotonic() + max(0.0, budget_s - 8.0)

    n_init = min(max_rows, max(60, max_rows // 3))
    rows = list(np.unique(np.linspace(0, 2 * n - 2, n_init, dtype=int)).astype(int))
    row_set = set(int(r) for r in rows)

    it = 0
    stagnation = 0

    while time.monotonic() < end:
        it += 1
        rem = end - time.monotonic()
        if rem < 3.0:
            break

        A = build_constraint_matrix(f, rows, n)
        b = np.full(len(rows), cur_score, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)

        tl = max(2.0, min(45.0, rem * 0.3))
        res = None
        try:
            res = scipy_opt.linprog(
                c_obj, A_ub=A, b_ub=b, bounds=(0.0, None),
                method='highs', options={'time_limit': float(tl)}
            )
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if sg < 1e-12:
            break

        # Check full constraint satisfaction
        L2 = next_pow2(2 * n - 1)
        Pf = np.fft.rfft(f, L2)
        Pg = np.fft.rfft(g, L2)
        conv = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        tol = 1e-6 * max(1.0, cur_score)
        violations = conv - cur_score
        max_viol = float(violations.max())

        if max_viol > tol:
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                n_add = min(300, max_rows - len(rows))
                added = 0
                for ix in order:
                    if added >= n_add:
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(int(ix))
                        added += 1
                if added == 0:
                    if max_viol > 50.0 * tol:
                        break
                else:
                    continue
            else:
                if max_viol > 20.0 * tol:
                    break

        # Scale g
        g = g * (S0 / sg)

        # Line search with golden-ratio-like steps
        best_step_f = None
        best_step_score = np.inf
        steps = (1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02)
        for step_val in steps:
            fn = (1.0 - step_val) * f + step_val * g
            fn = np.maximum(fn, 0.0)
            sn = float(fn.sum())
            if sn < 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = score_fft(fn)
            if np.isfinite(sc) and sc < best_step_score:
                best_step_score = sc
                best_step_f = fn.copy()

        if best_step_f is None or best_step_score >= best_score:
            stagnation += 1
            if stagnation > 8:
                break
            # Try very small steps
            found = False
            for step_val in (0.01, 0.005, 0.002):
                fn = (1.0 - step_val) * f + step_val * g
                fn = np.maximum(fn, 0.0)
                sn = float(fn.sum())
                if sn < 1e-12:
                    continue
                fn = fn * (S0 / sn)
                sc = score_fft(fn)
                if np.isfinite(sc) and sc < best_score:
                    best_step_f = fn.copy()
                    best_step_score = sc
                    found = True
                    break
            if not found:
                if stagnation > 15:
                    break
                continue

        f = best_step_f
        cur_score = best_step_score
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            stagnation = 0
            if verbose and it % 10 == 0:
                print("  LP n=%d it=%d score=%.6f rows=%d" % (n, it, best_score, len(rows)))

    final_sc = score_fft(best_f)
    if final_sc < best_score:
        best_score = final_sc
    return normalize_nn(best_f), best_score


def adam_log_simplex(p0, budget_s, alpha_init=40.0, base_lr=None, verbose=False):
    p = normalize_nn(p0)
    n = int(p.size)
    if n == 0:
        return p, np.inf

    if base_lr is None:
        if n < 200:
            base_lr = 0.05
        elif n < 800:
            base_lr = 0.025
        else:
            base_lr = 0.012

    x = np.log(p + 1e-300)
    x -= x.mean()

    L = next_pow2(3 * n - 2)
    best_p = p.copy()
    best_f = score_fft(p)

    mu_a, nu_a, eps_a = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = alpha_init * np.exp(-3.5 * frac) + 4.0

        P = np.fft.rfft(p, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if maxc <= 0 or not np.isfinite(maxc):
            break

        f_now = 2.0 * n * maxc
        if f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Softmax weights on convolution
        z = alpha * (c / maxc - 1.0)
        z = np.clip(z, -60.0, 60.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz < 1e-300:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient of sum_k w_k * c_k w.r.t. p
        # c_k = sum_i p_i * p_{k-i}
        # d/dp_j = sum_k w_k * p_{k-j} (with 0-padding)
        # = cross-correlation of w and p at lag j
        L2 = next_pow2(2 * n - 1)
        Wf = np.fft.rfft(w, L2)
        Pf2 = np.fft.rfft(p, L2)
        G = np.fft.irfft(np.conj(Wf) * Pf2, L2)[:n].real
        grad_p = 2.0 * G

        if not np.all(np.isfinite(grad_p)):
            break

        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu_a * m + (1.0 - mu_a) * grad_x
        v = nu_a * v + (1.0 - nu_a) * (grad_x * grad_x)
        mhat = m / (1.0 - mu_a ** t)
        vhat = v / (1.0 - nu_a ** t)
        step = base_lr * mhat / (np.sqrt(vhat) + eps_a)
        step = np.clip(step, -40.0, 40.0)
        x = x - step
        x -= x.mean()
        x = np.clip(x, -80.0, 80.0)
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se < 1e-300:
            break
        p = e / se

    final_f = score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def spectral_excess_descent(p0, budget_s, eps=0.008, lr_init=0.15, verbose=False):
    p = normalize_nn(p0).copy()
    n = int(p.size)
    if n == 0:
        return p, np.inf
    budget_s = float(budget_s)
    if budget_s < 10.0:
        return p, score_fft(p)

    L = next_pow2(2 * n - 1)
    best_p = p.copy()
    best_f = score_fft(p)
    lr = lr_init

    start = time.monotonic()
    end = start + max(0.0, budget_s - 3.0)
    it = 0

    while time.monotonic() < end:
        it += 1
        P = np.fft.rfft(p, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        Mc = float(c.max())
        Mt = Mc * (1.0 - eps)

        e_w = np.maximum(c - Mt, 0.0)
        if e_w.max() < 1e-12:
            break

        # grad[j] = sum_k e_w[k] * p[k-j]
        L2 = next_pow2(2 * n - 1)
        Ef = np.fft.rfft(e_w, L2)
        Pf2 = np.fft.rfft(p, L2)
        G = np.fft.irfft(np.conj(Ef) * Pf2, L2)[:n].real

        p_new = p - lr * G
        p_new = np.maximum(p_new, 0.0)
        s = float(p_new.sum())
        if s < 1e-12:
            break
        p_new = p_new / s

        f_now = score_fft(p_new)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p_new.copy()
            p = p_new
        else:
            # Backtrack
            lr *= 0.85
            if lr < 1e-5:
                break
            p_new2 = p - lr * G
            p_new2 = np.maximum(p_new2, 0.0)
            s2 = float(p_new2.sum())
            if s2 < 1e-12:
                break
            p_new2 = p_new2 / s2
            f2 = score_fft(p_new2)
            if np.isfinite(f2) and f2 < best_f:
                best_f = f2
                best_p = p_new2.copy()
                p = p_new2
            else:
                lr = max(lr * 0.8, 1e-6)

        if verbose and it % 200 == 0:
            print("    spectral it=%d score=%.6f lr=%.5f" % (it, best_f, lr))

    return best_p, best_f


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        a = np.ones(16, dtype=np.float64)
        s = 16.0
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]


def record_best(p, cur_best_p, cur_best_score, best_by_n):
    if p is None:
        return cur_best_p, cur_best_score
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        return cur_best_p, cur_best_score
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return cur_best_p, cur_best_score
    a = a / s
    sc = score_fft(a)
    if not np.isfinite(sc):
        return cur_best_p, cur_best_score
    n = int(a.size)
    if sc < cur_best_score:
        cur_best_p = a.copy()
        cur_best_score = float(sc)
        print("  NEW BEST: %.6f (n=%d)" % (sc, n))
    if n not in best_by_n or sc < best_by_n[n][0]:
        best_by_n[n] = (float(sc), a.copy())
    return cur_best_p, cur_best_score

        best_p, best_score = record_best(p_out, best_p, best_score, best_by_n)

    rem = deadline - time.monotonic()
    if rem > 40 and best_p is not None:
        alloc = min(120.0, rem * 0.5)
        p_out, _ = adam_log_simplex(best_p, alloc, alpha_init=20.0)
        best_p, best_score = record_best(p_out, best_p, best_score, best_by_n)

    # One more LP if time permits
    rem = deadline - time.monotonic()
    if rem > 80 and best_p is not None and scipy_opt is not None:
        alloc = min(200.0, rem * 0.6)
        print("  Final LP, rem=%.0fs" % rem)
        p_out, f_out = lp_optimize(best_p, alloc)
        best_p, best_score = record_best(p_out, best_p, best_score, best_by_n)

    # Spectral on top candidates
    rem = deadline - time.monotonic()
    if rem > 40 and len(best_by_n) > 1:
        for n, (sc, p) in sorted(best_by_n.items(), key=lambda x: x[1][0])[:3]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            alloc = min(80.0, max(10.0, rem * 0.3))
            p_out, f_out = spectral_excess_descent(p, alloc, eps=0.004, lr_init=0.08)
            best_p, best_score = record_best(p_out, best_p, best_score, best_by_n)

    print("=== Final: best=%.6f (n=%d) ===" % (best_score, len(best_p) if best_p is not None else 0))

    if best_p is None:
        best_p = greedy_init(384)

    final_seq = scale_output(best_p, 10.0)

    # Validate
    try:
        val = evaluate_sequence(final_seq)
        print("Validated: %.6f" % val)
        if not np.isfinite(val):
            # Try different scaling
            final_seq = scale_output(best_p, 5.0)
            val = evaluate_sequence(final_seq)
            print("Rescaled: %.6f" % val)
    except Exception:
        pass

    elapsed = time.monotonic() - t0
    print("Total time: %.1fs" % elapsed)
    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    except Exception as e:
