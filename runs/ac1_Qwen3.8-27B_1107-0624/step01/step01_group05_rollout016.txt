def _build_conv_matrix(p):
    """Build sparse matrix A (2N-1 x N) where A @ g = p * g (linear conv)."""
    n = N
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = n - 1 if k >= n else k
        length = j_max - j_min + 1
        rows[idx:idx+length] = k
        cols[idx:idx+length] = np.arange(j_min, j_max + 1)
        vals[idx:idx+length] = p[k - np.arange(j_min, j_max + 1)]
        idx += length
    
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _lp_refine(p, deadline, max_iters=100):

    _check()
    p = p.copy()
    c = _self_conv(p)
    M = float(np.max(c))
    best_p = p.copy()
    best_score = _score_p(p)
    _update_best(p, best_score)
    
    A = _build_conv_matrix(p)
    
    for it in range(max_iters):
        _check()
        if time.monotonic() >= deadline:
            break
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(N)
        b_ub = np.full(2*N-1, M)
        
        try:
            result = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                           bounds=[(0, 1e10)]*N, method='highs')
        except Exception:
            break
        
        if not result.success:
            # Try relaxed M
            for relax in (1.01, 1.05, 1.1, 1.5):
                b_ub2 = np.full(2*N-1, M * relax)
                try:
                    result = linprog(c_obj, A_ub=A, b_ub=b_ub2,
                                   bounds=[(0, 1e10)]*N, method='highs')
                except Exception:
                    continue
                if result.success:
                    break
            if not result.success:
                break
        
        g = result.x
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break
        
        g = g / g_sum  # normalize to probability vector
        
        # Try mixing
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            p_new = (1-t) * p + t * g
            p_new = np.maximum(p_new, 0)
            s_new = float(np.sum(p_new))
            if s_new < 1e-10:
                continue
            p_new = p_new / s_new
            sc = _score_p(p_new)
            if sc < best_score - 1e-12:
                p = p_new
                best_score = sc
                best_p = p.copy()
                _update_best(p, sc)
                improved = True
                break
        
        if not improved:
            # Rebuild matrix with new p
            M = float(np.max(_self_conv(p)))
            A = _build_conv_matrix(p)
            # Check convergence
            if M <= best_score / (2.0 * N) * (1 + 1e-9):
                # Very close to optimum
                # Try smaller steps
                for t in (0.02, 0.01, 0.005):
                    p_new = (1-t) * p + t * g
                    p_new = np.maximum(p_new, 0)
                    p_new = p_new / float(np.sum(p_new))
                    sc = _score_p(p_new)
                    if sc < best_score - 1e-12:
                        p = p_new
                        best_score = sc
                        best_p = p.copy()
                        _update_best(p, sc)
                        break
                else:
                    break
        
        if it % 10 == 0:
            print(f"  LP iter {it}: score={best_score:.6f}, M={M:.8f}")
    
    return best_p, best_score

def _subgradient_polish(p0, deadline, rng):
    """Active-peak subgradient method."""
    p = p0.copy()
    sc = _score_p(p)
    _update_best(p, sc)
    
    noise = 0.05
    fails = 0
    
    while time.monotonic() < deadline:
        _check()
        c = _self_conv(p)
        cmax = float(np.max(c))
        
        # Find top peaks
        topk = min(16, 2*N-1)
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros(2*N-1)
        wv = c[idx] + 1e-15
        w[idx] = wv / float(np.sum(wv))
        
        # Compute subgradient
        p_rev = p[::-1]
        conv = np.fft.irfft(np.fft.rfft(p_rev, 2*_FFT_SIZE) * 
                           np.fft.rfft(w, 2*_FFT_SIZE), 2*_FFT_SIZE).real
        gp = 2.0 * conv[N-1:2*N-1]
        
        # Project gradient to tangent space of simplex: g - mean(g*p)/sum(p^2) * p
        # Actually, for the probability simplex, the projected gradient is:
        # g_proj = g - (g . p) * p / sum(p) ... no
        # The tangent space of the simplex {p: p>=0, sum(p)=1} at p is {v: sum(v)=0}
        # So we project: g_proj = g - mean(g) * 1 ... no
        # Project onto {v: sum(v) = 0}: g_proj = g - (sum(g)/N) * 1
        # But we also need to respect p >= 0 constraint.
        # For now, just use the "centered" gradient:
        sdot = float(np.dot(gp, p))
        g_centered = gp - sdot / float(np.dot(p, p)) * p
        # Hmm, this isn't quite right either. Let me just use the direct p-update:
        # p_new = p - eta * g, then clip and renormalize.
        
        gnorm = float(np.linalg.norm(gp))
        if gnorm < 1e-14:
            break
        
        direction = -gp / gnorm  # negative gradient direction
        
        accepted = False
        for eta in (0.02, 0.01, 0.005, 0.002, 0.001):
            p_new = p + eta * direction
            p_new = np.maximum(p_new, 0)
            s = float(np.sum(p_new))
            if s < 1e-10:
                continue
            p_new = p_new / s
            sc_new = _score_p(p_new)
            if sc_new < sc - 1e-13:
                p = p_new
                sc = sc_new
                _update_best(p, sc)
                accepted = True
                break
        
        if not accepted:
            # Try random perturbation
            best_local = sc
            best_local_p = p
            for _ in range(5):
                p_try = p + rng.normal(0, noise, N)
                p_try = np.maximum(p_try, 0)
                s = float(np.sum(p_try))
                if s < 1e-10:
                    continue
                p_try = p_try / s
                sc_try = _score_p(p_try)
                if sc_try < best_local:
                    best_local = sc_try
                    best_local_p = p_try
            if best_local < sc - 1e-13:
                p = best_local_p
                sc = best_local
                _update_best(p, sc)
                accepted = True
        
        if accepted:
            fails = 0
            noise = min(0.1, noise * 1.3)
        else:
            fails += 1
            noise = max(0.005, noise * 0.8)
            if fails > 80:
                break
    
    return p, sc

# --- Initializations ---

def _uniform_z():
    return np.zeros(N-1)

def _logistic_z(beta):
    x = (np.arange(N) + 0.5) / N
    f = 2.0 / (1.0 + np.exp(-beta * (x - 0.5)))
    f = np.maximum(f, 1e-12)
    return _z_from_probs(f / f.sum())

def _symmetric_triangle_z():
    x = (np.arange(N) + 0.5) / N
    f = 1.0 - np.abs(2.0 * x - 1.0)
    f = np.maximum(f, 1e-12)
    return _z_from_probs(f / f.sum())

def _raised_cosine_z():
    x = (np.arange(N) + 0.5) / N
    f = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    f = np.maximum(f, 1e-12)
    return _z_from_probs(f / f.sum())

def _bump_z(sigma):
    x = (np.arange(N) + 0.5) / N
    f = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return _z_from_probs(f / f.sum())

def _two_bump_z(sigma, sep):
    x = (np.arange(N) + 0.5) / N
    f = np.exp(-0.5 * ((x - 0.3) / sigma) ** 2) + np.exp(-0.5 * ((x - 0.7) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return _z_from_probs(f / f.sum())

def _random_poly_z(rng):
    x = (np.arange(N) + 0.5) / N
    t = x - 0.5
    deg = rng.integers(3, 8)
    coef = rng.normal(0, 1.5, size=deg+1)
    f = np.ones(N)
    for k in range(deg, 0, -1):
        f = f * t + coef[k]
    f += coef[0]
    f = np.maximum(f, 1e-8)
    return _z_from_probs(f / f.sum())

def _random_fourier_z(rng):
    x = (np.arange(N) + 0.5) / N
    f = np.ones(N)
    for freq in (1, 2, 3, 4, 5, 6, 8):
        amp = rng.normal(0, 0.8 / (0.5 + freq * 0.3))
        ph = rng.uniform(0, 2*np.pi)
        f += amp * np.sin(2*np.pi*freq*x + ph)
        f += amp * np.cos(2*np.pi*freq*x + ph)
    f = np.maximum(f, 1e-8)
    return _z_from_probs(f / f.sum())
