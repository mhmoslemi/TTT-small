def lp_direction(a, M, time_limit=10, max_iter=20):
    """
    Compute approximately the LP direction:
    max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Uses a few iterations of the Frank-Wolfe / multiplicative weights method.
    Returns g (not normalized).
    """
    n = a.size
    # Use multiplicative weights on the constraints
    # Dual: min M * sum(y) s.t. (a_rev * y)_i >= 1 for all i, y >= 0
    # Primal-dual gap approach
    
    # Initialize
    y = np.ones(2*n-1) / (2*n-1)  # uniform
    g = np.ones(n) / n
    
    L = 1
    while L < 2*n: L <<= 1
    
    Fa = np.fft.rfft(a, L)
    
    for it in range(max_iter):
        if time_limit > 0 and it > 0:
            pass  # time checking done externally
        
        # Primal step: given y, find g that maximizes sum(g) - <y, A*g> (relaxed)
        # Actually, use the dual to update:
        # The dual constraint for coordinate i: sum_k y_k * a_{k-i} >= 1
        # Compute the "tightness" for each i:
        Wy = np.fft.rfft(y[::-1], L)
        atight = np.fft.irfft(Fa * Wy, L)[n-1:2*n-1][::-1]
        atight = np.maximum(atight, 1e-300)
        
        # Update g: proportional to 1/atight (KKT-inspired)
        g = np.maximum(1.0/atight - 1.0, 0)
        sg = g.sum()
        if sg < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / sg
        
        # Dual step: given g, update y proportional to constraint violation
        Fg = np.fft.rfft(g, L)
        convag = np.fft.irfft(Fa * Fg, L)[:2*n-1]
        convag = np.maximum(convag, 0)
        
        # y_k proportional to how tight constraint k is
        eta = 0.5
        y = y * np.exp(eta * convag / (M + 1e-300))
        sy = y.sum()
        y = y / sy
    
    # Final g
    sg = g.sum()
    if sg > 1e-15:
        g = g * (np.sqrt(2*n) / sg)
    return g

"""
Search for a non-negative step-function sequence minimizing
    f(seq) = 2*n*max(convolve(seq, seq)) / sum(seq)^2
using a hybrid approach:
  1. FFT-based Adam gradient descent with adaptive smoothed-maximum objective.
  2. Periodic LP/cutting-plane corrections via scipy to escape local minima.
  3. Multiple diverse initializations (greedy, power-law, flat-spectrum, Dirichlet).
  4. Adaptive restart strategy when progress stalls.
  5. Subgradient polishing at the active constraints.
Focus on length range [256, 768] where the previous best (n=384) was found.
"""

import time
import numpy as np

try:
    import scipy.optimize as scipy_opt
except Exception:
    scipy_opt = None


def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def norm1(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(16)
    a = np.clip(a, 0.0, 1e10)
    s = float(a.sum())
    if s < 1e-15:
        return np.full(a.shape, 1.0 / a.size)
    return a / s


def conv_full(a, L=None):
    n = a.size
    if L is None:
        L = next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def score_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n < 1:
        return np.inf
    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = conv_full(a)
    return float(2.0 * n * c.max() / (s * s))


def conv_cross(a, b, L=None):
    n = a.size
    if L is None:
        L = next_pow2(2 * n - 1)
    Fa = np.fft.rfft(a, L)
    Fb = np.fft.rfft(b, L)
    c = np.fft.irfft(Fa * Fb, L)[:2 * n - 1]
    return np.maximum(c.real, 0.0)


def init_greedy(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return norm1(a)


def init_power(n, s=0.45):
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return norm1(a)


def init_flat_spectrum(n, rng):
    L = next_pow2(2 * n)
    nfreq = L // 2 + 1
    phases = rng.uniform(0.0, 2.0 * np.pi, nfreq)
    F = np.zeros(L, dtype=np.complex128)
    for k in range(nfreq):
        F[k] = np.exp(1j * phases[k])
    for k in range(1, L // 2):
        F[L - k] = np.conj(F[k])
    F[0] = abs(F[0])
    if L % 2 == 0:
        F[L // 2] = abs(F[L // 2])
    a_full = np.fft.ifft(F).real[:n]
    a = np.clip(a_full, 0.0, None)
    if a.sum() < 1e-10:
        return init_greedy(n)
    return norm1(a)


def init_dirichlet(n, alpha, rng):
    a = rng.dirichlet(float(alpha) * np.ones(n))
    return norm1(a)


def init_mixed(n, lam, rng):
    g = init_greedy(n)
    u = rng.dirichlet(np.ones(n))
    return norm1(lam * g + (1.0 - lam) * u)


def init_perturbed(base, rng, sigma):
    noise = np.exp(rng.normal(0.0, float(sigma), base.size))
    return norm1(base * noise)


def lp_direction_scipy(a, M, time_limit=5.0, max_rows=800):
    """
    Solve approximately: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Uses scipy linprog with a subset of constraints (cutting plane).
    Returns g (not rescaled) or None on failure.
    """
    if scipy_opt is None:
        return None
    n = a.size
    if n < 2:
        return np.ones(n)
    if n > 4096:
        return None

    # Start with a subset of constraints: every (2n-1)/max_rows indices
    total_k = 2 * n - 1
    step = max(1, total_k // max_rows)
    rows = list(range(0, total_k, step))
    if rows[-1] != total_k - 1:
        rows.append(total_k - 1)
    rows = sorted(set(rows))
    row_set = set(rows)

    A_list = []
    for k in rows:
        row = np.zeros(n, dtype=np.float64)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            row[lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
        A_list.append(row)
    A = np.array(A_list)
    b = np.full(len(rows), M, dtype=np.float64)
    cobj = -np.ones(n, dtype=np.float64)

    L = next_pow2(2 * n - 1)

    forouter in range(6):
        tl = max(1.0, time_limit / (6 - forouter))
        try:
            res = scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": float(tl)}
            )
        except Exception:
            try:
                res = scipy_opt.linprog(
                    cobj, A_ub=A, b_ub=b,
                    bounds=(0.0, None),
                    method="highs"
                )
            except Exception:
                return None

        if res is None or not getattr(res, "success", False):
            return None
        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        if g.sum() < 1e-15:
            return None

        # Check violations
        Fg = np.fft.rfft(g, L)
        Fa = np.fft.rfft(a, L)
        conv = np.fft.irfft(Fa * Fg, L)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        tol = 1e-6 + 1e-6 * M
        viol = conv - M
        max_viol = float(viol.max())

        if max_viol < tol:
            return g

        # Add violated constraints
        if len(rows) < max_rows * 2:
            order = np.argsort(viol)[::-1]
            added = 0
            for ix in order:
                ix = int(ix)
                if viol[ix] < tol:
                    break
                if ix not in row_set and added < 200:
                    row = np.zeros(n, dtype=np.float64)
                    lo = max(0, ix - n + 1)
                    hi = min(n - 1, ix)
                    if lo <= hi:
                        row[lo:hi + 1] = a[ix - hi:ix - lo + 1][::-1]
                    A = np.vstack([A, row])
                    b = np.append(b, M)
                    rows.append(ix)
                    row_set.add(ix)
                    added += 1
        if max_viol > 10.0 * tol:
            return g

    return g


def subgradient_step(a, L, step_size=0.005):
    """
    One subgradient step: move in the direction that reduces the max constraint.
    The subgradient of max(c_k) at the active index k* is 2 * a shifted to lag k*.
    """
    n = a.size
    c = conv_full(a, L)
    maxc = c.max()
    if maxc < 1e-15:
        return a
    active = np.where(c >= 0.999 * maxc)[0]
    # Average subgradient over active constraints
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    grad *= 2.0 / len(active)
    # Normalize and step
    s = a.sum()
    a_new = a - step_size * grad * s
    a_new = np.clip(a_new, 0.0, None)
    sn = a_new.sum()
    if sn < 1e-15:
        return a
    return a_new / sn


def optimize_hybrid(a0, time_limit, seed=0):
    """
    Main optimizer: Adam with smoothed max + periodic LP + subgradient polishing.
    a0 is normalized to sum 1.
    Returns (best_a, best_score).
    """
    n = a0.size
    a = norm1(a0)
    L = next_pow2(2 * n - 1)

    if n <= 128:
        lr = 0.04
    elif n <= 384:
        lr = 0.03
    elif n <= 1024:
        lr = 0.02
    else:
        lr = 0.012

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    x = np.log(a + 1e-300)
    x -= x.mean()

    c = conv_full(a, L)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    start = time.monotonic()
    last_lp_time = 0.0
    stall_count = 0
    stall_threshold = 300

    while time.monotonic() - start < time_limit:
        t += 1
        elapsed = (time.monotonic() - start) / max(1e-9, time_limit)
        frac = min(1.0, elapsed)

        # Compute convolution
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)

        maxc = float(c.max())
        if maxc < 1e-15:
            break
        cur_score = 2.0 * n * maxc

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive alpha
        meanc = c.sum() / (2 * n - 1)
        flatness = (maxc - meanc) / (maxc + 1e-300)
        base_alpha = 12.0 * np.exp(-6.0 * frac) + 1.5
        alpha = base_alpha * (0.4 + 0.6 * flatness)
        alpha = max(1.0, min(80.0, alpha))

        # Softmax weights
        z = alpha * (c / maxc - 1.0)
        ez = np.exp(z)
        sz = float(ez.sum())
        if not np.isfinite(sz) or sz < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ez / sz

        # Gradient via cross-correlation
        W_rev = np.fft.rfft(w[::-1], L)
        Cconv = np.fft.irfft(F * W_rev, L)
        C = Cconv[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_a = 4.0 * n * C
        gp = float(np.dot(a, grad_a))
        grad_x = a * (grad_a - gp)

        # Adam update
        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -20.0, 20.0)
        x = x - step
        x -= x.mean()

        # Periodic LP correction (every ~40 seconds or at specific intervals)
        lp_interval_frames = max(50, n // 4)
        if (t % lp_interval_frames == 0 and scipy_opt is not None and
                time.monotonic() - last_lp_time > max(10.0, time_limit * 0.03) and
                time.monotonic() - start < time_limit * 0.85):
            last_lp_time = time.monotonic()
            M_cur = maxc
            g_dir = lp_direction_scipy(a, M_cur, time_limit=8.0)
            if g_dir is not None and g_dir.sum() > 1e-10:
                g_dir = norm1(g_dir)
                # Blend: take a step towards LP direction
                blend = 0.15
                a_new = (1.0 - blend) * a + blend * g_dir
                a_new = np.clip(a_new, 0.0, None)
                sn = a_new.sum()
                if sn > 1e-10:
                    a_new = a_new / sn
                    test_score = 2.0 * n * conv_full(a_new, L).max()
                    if test_score < cur_score * 1.01:  # accept if not too bad
                        a = a_new
                        x = np.log(a + 1e-300)
                        x -= x.mean()

        # Periodic subgradient step (last 20% of budget)
        if frac > 0.80 and t % 5 == 0:
            a_new = subgradient_step(a, L, step_size=0.003)
            test_score = 2.0 * n * conv_full(a_new, L).max()
            if test_score < cur_score:
                a = a_new
                x = np.log(a + 1e-300)
                x -= x.mean()

        # Stall detection: perturb and restart
        if stall_count > stall_threshold:
            stall_count = 0
            x += 0.05 * np.random.randn(n)
            x -= x.mean()
            x = np.clip(x, -70.0, 70.0)
            x -= x.max()
            ea = np.exp(x)
            sa = ea.sum()
            if sa < 1e-300:
                break
            a = ea / sa

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -70.0, 70.0)
        x -= x.max()
        ea = np.exp(x)
        sa = float(ea.sum())
        if not np.isfinite(sa) or sa < 1e-300:
            break
        a = ea / sa

    # Final evaluation
    c_final = conv_full(a, L)
    fin_score = float(2.0 * n * c_final.max())
    if fin_score < best_score:
        best_score = fin_score
        best_a = a.copy()

    # Subgradient polishing
    for _ in range(50):
        a_pg = subgradient_step(best_a, L, step_size=0.002)
        sc_pg = 2.0 * n * conv_full(a_pg, L).max()
        if sc_pg < best_score:
            best_score = sc_pg
            best_a = a_pg
        else:
            break

    return best_a, best_score

                    # Resample best_p to new length
                    old_n = best_p.size
                    if nl > old_n:
                        # Extend by zero-padding or interpolation
                        new_idx = np.linspace(0, old_n - 1, nl)
                        base = np.interp(new_idx, np.arange(old_n), best_p)
                    else:
                        new_idx = np.linspace(0, old_n - 1, nl)
                        base = np.interp(new_idx, np.arange(old_n), best_p)
                    base = norm1(base)

                print("  Refining n=%d alloc=%.1fs" % (nl, alloc))
                p_out, f_out = optimize_hybrid(base, alloc, seed=seed + 1000 + nl)
                record(p_out, f_out)

    # Phase 4: More diverse restarts on the best length
    if best_p is not None:
        rem = deadline - time.monotonic()
        best_n = best_p.size
        if rem > 60.0:
            n_restarts = max(2, int(rem / 80.0))
            n_restarts = min(n_restarts, 6)
            per_restart = rem / (n_restarts + 1)
            for r in range(n_restarts):
                rem = deadline - time.monotonic()
                if rem < 25.0:
                    break
                # Different perturbation strategies
                if r == 0:
                    base = init_flat_spectrum(best_n, rng)
                elif r == 1:
                    base = init_perturbed(best_p, rng, 0.25)
                elif r == 2:
                    base = init_mixed(best_n, 0.5, rng)
                else:
                    base = init_perturbed(best_p, rng, 0.35)
                t_alloc = min(per_restart, rem - 10.0)
                if t_alloc < 12.0:
                    break
                print("  Restart %d on n=%d alloc=%.1fs" % (r, best_n, t_alloc))
                p_out, f_out = optimize_hybrid(base, t_alloc, seed=seed + 5000 + r * 111)
                record(p_out, f_out)

    # Phase 5: Final subgradient polishing
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30.0:
            print("Phase 5: Final polishing, remaining=%.1fs" % rem)
            L = next_pow2(2 * best_p.size - 1)
            a_polish = best_p.copy()
            for step_sz in [0.005, 0.003, 0.001]:
                for _ in range(100):
                    if time.monotonic() >= deadline - 5.0:
                        break
                    a_new = subgradient_step(a_polish, L, step_size=step_sz)
                    sc = 2.0 * best_p.size * conv_full(a_new, L).max()
                    if sc < best_score:
                        best_score = sc
                        best_p = a_new
                        best_a = a_new.copy()
                    else:
                        break

    # Fallback
    if best_p is None:
        best_p = init_greedy(384)
        best_score = score_fft(best_p)

    print("Final best: n=%d score=%.6f" % (best_p.size, best_score))

    # Verify with evaluate_sequence if possible
    final_seq = scale_output(best_p, 10.0)
    eval_score = try_eval(final_seq)
    if np.isfinite(eval_score) and eval_score < best_score:
        best_score = eval_score
        print("Verified score: %.6f" % best_score)
    elif eval_score != best_score:
        # The scale_output might change the score slightly; use the direct score
        print("Note: eval gave %.6f, FFT gave %.6f" % (eval_score, best_score))

    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
