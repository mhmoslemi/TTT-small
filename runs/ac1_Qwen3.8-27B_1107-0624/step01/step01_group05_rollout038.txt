import time

def _fft_convolve_full(a, b):
    """Full linear convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]

    """Autocorrelation (self-convolution) of probability vector p."""
    return _fft_convolve_full(p, p)

    """Project v onto probability simplex."""

    """Compute g[i] = 2 * sum_k w_k * p[k-i] (out-of-range = 0)."""

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 via projected subgradient on the probability simplex.
Uses FFT-based convolution, adaptive top-k active set, multi-start with diverse initializations,
and tries multiple sequence lengths (4096, 8192, 16384) to find the global best.
Key improvement over previous: works directly on the simplex (not logits), uses adaptive
step sizes with backtracking, and leverages longer sequences for better asymptotic performance.
"""

import numpy as np
import time


def _fft_conv_full(a, b):
    """Full linear convolution of arrays a and b using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]


def _self_conv(p):
    """Autocorrelation (self-convolution) of probability vector p via FFT."""
    n = len(p)
    size = 1 << (2 * n - 2).bit_length()
    fp = np.fft.rfft(p, size)
    c = np.fft.irfft(fp * fp, size)
    c = c[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_val(n, c):
    """Compute 2*n*max(c) where c is the self-convolution."""
    return 2.0 * float(n) * float(np.max(c))


def _project_simplex(v):
    """Project vector v onto the probability simplex {p >= 0, sum(p) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    # Fix any numerical issues
    s = np.sum(out)
    if s <= 0 or not np.isfinite(s):
        return np.ones(n) / n
    return out / s


def _compute_subgrad(p, active_idx, weights, n):
    """
    Compute subgradient g[i] = 2 * sum_k w_k * p[k-i] (out-of-range = 0).
    active_idx: array of active convolution positions
    weights: array of same length as active_idx
    """
    g = np.zeros(n)
    for k, w in zip(active_idx, weights):
        k = int(k)
        if w == 0.0:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * w * p[lo:hi + 1][::-1]
    return g


def _adaptive_active_set(c, max_k=30, threshold_frac=0.05):
    """
    Select active positions: all positions where c[k] > max(c) * (1 - threshold_frac),
    capped at max_k highest positions.
    Returns (indices, weights) with weights proportional to c values.
    """
    cmax = np.max(c)
    if cmax <= 0:
        return np.array([0]), np.array([1.0])
    # Find positions above threshold
    thresh = cmax * (1.0 - threshold_frac)
    mask = c >= thresh
    idx = np.nonzero(mask)[0]
    if len(idx) > max_k:
        # Take top max_k
        top_vals = c[idx]
        order = np.argsort(top_vals)[::-1][:max_k]
        idx = idx[order]
    if len(idx) == 0:
        idx = np.argpartition(c, -5)[-5:]
    weights = c[idx]
    wsum = np.sum(weights)
    if wsum <= 0 or not np.isfinite(wsum):
        weights = np.ones(len(idx)) / len(idx)
    else:
        weights = weights / wsum
    return idx, weights


def _make_uniform(n):
    return np.ones(n) / n


def _make_logistic(n, beta):
    x = np.linspace(0, 1, n)
    f = 1.0 / (1.0 + np.exp(-beta * (x - 0.5)))
    f = np.maximum(f, 1e-12)
    return f / np.sum(f)


def _make_power(n, a, reverse=False):
    x = np.linspace(0, 1, n)
    if reverse:
        f = (1.0 - x + 1e-6) ** a
    else:
        f = (x + 1e-6) ** a
    f = np.maximum(f, 1e-12)
    return f / np.sum(f)


def _make_gaussian_perturbed(n, rng, sigma=0.3):
    x = np.linspace(0, 1, n)
    f = 1.0 + sigma * rng.standard_normal(n)
    f = np.maximum(f, 1e-8)
    return f / np.sum(f)


def _make_poly(n, rng, deg=4):
    x = np.linspace(0, 1, n)
    t = x - 0.5
    coef = rng.normal(0, 1, deg + 1)
    f = np.ones(n)
    for k in range(deg, 0, -1):
        f = f + coef[k] * t ** k
    f = np.maximum(f, 1e-8)
    return f / np.sum(f)


def _make_sinusoidal(n, rng, n_freq=3):
    x = np.linspace(0, 1, n)
    f = np.ones(n)
    for _ in range(n_freq):
        freq = rng.integers(1, 8)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        f += amp * np.sin(2 * np.pi * freq * x + phase)
    f = np.maximum(f, 1e-8)
    return f / np.sum(f)


def _generate_initials(n, rng, count=20):
    """Generate diverse initial probability vectors."""
    inits = [_make_uniform(n)]
    for beta in [1.0, 2.0, 4.0, 6.0, 10.0]:
        inits.append(_make_logistic(n, beta))
        inits.append(_make_logistic(n, -beta))
    for a in [0.5, 1.0, 2.0, 3.0]:
        inits.append(_make_power(n, a))
        inits.append(_make_power(n, a, reverse=True))
    for _ in range(max(0, count - len(inits))):
        method = rng.integers(0, 3)
        if method == 0:
            inits.append(_make_gaussian_perturbed(n, rng, 0.3))
        elif method == 1:
            inits.append(_make_poly(n, rng, 4))
        else:
            inits.append(_make_sinusoidal(n, rng, 3))
    return inits[:count]


def _optimize_single(p, n, time_limit, rng, verbose=False):
    """
    Optimize a single probability vector using projected subgradient with adaptive top-k.
    Returns (best_p, best_score).
    """
    deadline = time.monotonic() + time_limit
    c = _self_conv(p)
    score = _score_val(n, c)
    best_p = p.copy()
    best_score = score

    # Parameters
    momentum = 0.9
    v = np.zeros(n)
    base_step = 0.5 / np.sqrt(n)
    avg_iter = p.copy()
    avg_count = 0

    iter_count = 0
    stagnant = 0
    last_improve = score

    while time.monotonic() < deadline:
        if time.monotonic() >= deadline:
            break

        iter_count += 1

        # Update active set periodically
        if iter_count % 10 == 1:
            k_top = min(5 + iter_count // 200, 50)
            active_idx, active_w = _adaptive_active_set(c, max_k=k_top)
        else:
            k_top = min(5 + iter_count // 200, 50)
            active_idx, active_w = _adaptive_active_set(c, max_k=k_top)

        # Compute subgradient
        g = _compute_subgrad(p, active_idx, active_w, n)
        gnorm = np.linalg.norm(g)

        if gnorm < 1e-15:
            break

        # Step size with adaptive schedule
        step = base_step / np.sqrt(iter_count + 1)
        step = min(step, 1.0 / (gnorm + 1e-10))

        # Momentum update
        v = momentum * v + (1.0 - momentum) * g
        vnorm = np.linalg.norm(v)
        if vnorm > 1e-10:
            v = v / vnorm * min(vnorm, gnorm)

        # Projected step
        p_new = p - step * v
        p_new = _project_simplex(p_new)

        # Evaluate
        c_new = _self_conv(p_new)
        score_new = _score_val(n, c_new)

        # Backtracking if needed
        if score_new > score + 1e-12:
            for _ in range(5):
                step *= 0.5
                p_new = p - step * v
                p_new = _project_simplex(p_new)
                c_new = _self_conv(p_new)
                score_new = _score_val(n, c_new)
                if score_new <= score + 1e-12:
                    break

        if score_new < score - 1e-14:
            p = p_new
            c = c_new
            score = score_new
            avg_iter += p
            avg_count += 1
            stagnant = 0
        else:
            stagnant += 1
            # If stuck, perturb
            if stagnant > 50:
                p = _project_simplex(p + rng.normal(0, 0.02, n))
                c = _self_conv(p)
                score = _score_val(n, c)
                v = np.zeros(n)
                stagnant = 0
            if stagnant > 200:
                break

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            if verbose and (iter_count % 500 == 0):
                elapsed = time.monotonic() - (deadline - time_limit)
                print(f"  iter {iter_count}: score={best_score:.6f} elapsed={elapsed:.1f}s")

        if score < last_improve - 1e-8:
            last_improve = score

    # Also check the average iterate
    if avg_count > 50:
        p_avg = avg_iter / avg_count
        p_avg = _project_simplex(p_avg)
        c_avg = _self_conv(p_avg)
        score_avg = _score_val(n, c_avg)
        if score_avg < best_score:
            best_score = score_avg
            best_p = p_avg.copy()

    return best_p, best_score

    """
    deadline = time.monotonic() + time_limit

    best_p = p.copy()

    cmax = np.max(c)

    active = np.nonzero(c >= cmax - tol)[0]

        active = np.array([int(np.argmax(c))])

    for it in range(10000):

            if len(active) > 100:
                top_vals = c[active]
                order = np.argsort(top_vals)[::-1][:100]

            w = c[active]
            wsum = np.sum(w)
            if wsum > 0:

            else:
                w = np.ones(len(active)) / len(active)
        else:

        g = _compute_subgrad(p, active, w, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Very small adaptive step
        step = 0.1 / (gnorm * np.sqrt(it + 1) + 1.0)

        p_new = _project_simplex(p - step * g)
        c_new = _self_conv(p_new)
        score_new = _score_val(n, c_new)

        if score_new < score - 1e-14:
            p = p_new

            score = score_new
            if score < best_score:
                best_score = score
                best_p = p.copy()
        else:

            step *= 0.1
            p_new = _project_simplex(p - step * g)

def _resample(p, new_n):
    """Resample a probability vector to a new length using linear interpolation."""
    old_n = len(p)
    if old_n == new_n:
        return p.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    p_new = np.interp(new_x, old_x, p)
    p_new = np.maximum(p_new, 0)
    s = np.sum(p_new)
    if s <= 0:
        return np.ones(new_n) / new_n
    return p_new / s
