def _fft_conv(a, b, n_fft=None):

    fa = np.fft.rfft(a, n_fft)
    fb = np.fft.rfft(b, n_fft)
    return np.fft.irfft(fa * fb, n_fft).real[:n]

def _self_conv(p):
    """Autocorrelation (full convolution with self)."""
    n = len(p)

    f = np.fft.rfft(p, n_fft)
    c = np.fft.irfft(f * f, n_fft).real[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _density_to_logits(f, n):

    f = np.asarray(f, dtype=np.float64)

    f = np.maximum(f, 1e-15)
    p = f / float(np.sum(f))
    logp = np.log(p)
    x = logp[1:] - logp[0]
    return np.clip(x, -50.0, 50.0)

def _logits_to_density(x, n):

def _fourier_repr(n, K):
    """Generate a random Fourier-based density with K harmonics."""
    # f(i/n) = 1 + sum_{k=1}^{K} a_k * cos(2*pi*k*i/n + phi_k)
    # Choose coefficients to ensure f > 0
    a = np.random.normal(0, 1.0/(K+1), K)
    phi = np.random.uniform(0, 2*np.pi, K)
    x = np.arange(n, dtype=np.float64) / n
    f = np.ones(n)
    for k in range(K):
        f += a[k] * np.cos(2*np.pi*(k+1)*x + phi[k])
    f = np.maximum(f, 1e-10)
    return f

def _spline_density(n, rng):
    """Generate a random smooth density using a random spline."""
    # Use a few control points and interpolate
    n_ctrl = rng.integers(4, 12)
    ctrl_x = np.sort(rng.uniform(0, 1, n_ctrl))

    x = np.arange(n, dtype=np.float64) / n
    f = np.interp(x, ctrl_x, ctrl_y, left=ctrl_y[0], right=ctrl_y[-1])

"""
Search for a non-negative sequence minimizing 2*n*max(a*a)/(sum a)^2.

Multi-stage approach:
1. Generate diverse starting points (parametric, random smooth, base resampled).
2. L-BFGS-B optimization with adaptive softmax temperature on full logit space.
3. Active-peak subgradient polishing for final refinement.
Tries multiple sequence lengths to find the globally best n and shape.
"""

import numpy as np
import time
from scipy.optimize import minimize

_best_score = float('inf')
_best_p = None
_deadline = 0.0


class _Timeout(Exception):
    pass


def _make_fft(n_out):
    n = 1
    while n < n_out:
        n <<= 1
    return n


def _self_conv(p):
    n = len(p)
    n_fft = _make_fft(2 * n - 1)
    f = np.fft.rfft(p, n_fft)
    c = np.fft.irfft(f * f, n_fft).real[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(p):
    n = len(p)
    c = _self_conv(p)
    return 2.0 * n * float(np.max(c))


def _update_best(p, score):
    global _best_score, _best_p
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_p = p.copy()


def _logits_to_p(x, n):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -50.0, 50.0)
    m = float(np.max(x)) if x.size else 0.0
    if m < 0:
        m = 0.0
    e = np.exp(x - m)
    e0 = np.exp(-m)
    Z = e0 + float(np.sum(e))
    p = np.empty(n, dtype=np.float64)
    p[0] = e0 / Z
    p[1:] = e / Z
    return p


def _p_to_logits(p, n):
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 1e-15)
    lp = np.log(p)
    x = lp[1:] - lp[0]
    return np.clip(x, -50.0, 50.0)


def _obj_grad(x, alpha, n):
    if time.monotonic() >= _deadline:
        raise _Timeout
    p = _logits_to_p(x, n)
    c = _self_conv(p)
    cmax = float(np.max(c))
    s = 2.0 * n * cmax
    _update_best(p, s)

    u = alpha * (c - cmax)
    e = np.exp(u)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(se) / alpha
        w = e / se

    # g[i] = sum_j p[j] * w[i+j]  (cross-correlation)
    n_fft_g = _make_fft(n + (2 * n - 1) - 1)
    fr = np.fft.rfft(p[::-1], n_fft_g)
    fw = np.fft.rfft(w, n_fft_g)
    conv = np.fft.irfft(fr * fw, n_fft_g).real
    g = conv[n - 1:n - 1 + n]

    sdot = float(np.dot(g, p))
    gx = p[1:] * (g[1:] - sdot)
    gx = np.where(np.isfinite(gx), gx, 0.0)
    gx = np.clip(gx, -1e8, 1e8)
    return float(J), gx


def _lbfgs(x0, n, alpha, max_fun, dl):
    global _deadline
    _deadline = float(dl)
    try:
        minimize(_obj_grad, x0, args=(alpha, n),
                 method='L-BFGS-B', jac=True,
                 bounds=[(-50.0, 50.0)] * (n - 1),
                 options={'maxfun': max_fun, 'maxiter': max_fun // 2,
                          'ftol': 1e-14, 'gtol': 1e-10,
                          'maxls': 40, 'disp': False})
    except _Timeout:
        pass
    except Exception:
        pass


def _polish(x0, n, dl, rng):
    x = np.clip(x0, -50.0, 50.0)
    p = _logits_to_p(x, n)
    score = _score(p)
    _update_best(p, score)

    nscale = 0.1
    fails = 0
    topk = 6

    while time.monotonic() < dl:
        c = _self_conv(p)
        cmax = float(np.max(c))
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros(len(c))
        wv = c[idx] + 1e-15
        w[idx] = wv / float(np.sum(wv))

        n_fft_g = _make_fft(n + (2 * n - 1) - 1)
        fr = np.fft.rfft(p[::-1], n_fft_g)
        fw = np.fft.rfft(w, n_fft_g)
        conv = np.fft.irfft(fr * fw, n_fft_g).real
        g = conv[n - 1:n - 1 + n]

        sdot = float(np.dot(g, p))
        gx = p[1:] * (g[1:] - sdot)
        gn = float(np.linalg.norm(gx))

        ok = False
        if gn > 1e-14:
            d = gx / gn
            for st in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
                xt = np.clip(x - st * d, -50, 50)
                pt = _logits_to_p(xt, n)
                st2 = _score(pt)
                if st2 < score - 1e-13:
                    x, p, score = xt, pt, st2
                    _update_best(p, score)
                    ok = True
                    break

        if not ok:
            bl_score, bl_x = score, x
            for _ in range(5):
                xt = np.clip(x + rng.normal(0, nscale, n - 1), -50, 50)
                pt = _logits_to_p(xt, n)
                st2 = _score(pt)
                if st2 < bl_score - 1e-13:
                    bl_score, bl_x = st2, xt
            if bl_score < score - 1e-13:
                x = bl_x
                p = _logits_to_p(x, n)
                score = bl_score
                _update_best(p, score)
                ok = True

        if ok:
            fails = 0
            nscale = min(0.3, nscale * 1.25)
        else:
            fails += 1
            nscale = max(0.01, nscale * 0.8)
            if fails > 150:
                break


def _parametric(ns, rng):
    pts = []
    x = np.arange(ns, dtype=np.float64) / ns

    for a in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 4.0):
        pts.append((x + 1e-6) ** a)
        pts.append((1.0 - x + 1e-6) ** a)

    for b in (1.0, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0):
        pts.append(np.exp(-b * x))
        pts.append(np.exp(-b * (1.0 - x)))

    for sig in (0.08, 0.12, 0.15, 0.2, 0.25, 0.3):
        for mu in (0.3, 0.4, 0.5, 0.6, 0.7):
            pts.append(np.exp(-0.5 * ((x - mu) / sig) ** 2))

    for b in (2.0, 4.0, 6.0, 10.0, 15.0, 25.0):
        pts.append(1.0 / (1.0 + np.exp(-b * (x - 0.5))))
        pts.append(1.0 / (1.0 + np.exp(b * (x - 0.5))))

    # Generalized sigmoid with power
    for b in (4.0, 8.0, 12.0):
        for pw in (0.5, 1.0, 1.5, 2.0):
            base = 1.0 / (1.0 + np.exp(-b * (x - 0.5)))
            pts.append(np.maximum(base, 1e-10) ** pw)
            pts.append(np.maximum(1.0 - base, 1e-10) ** pw)

    # Bimodal
    for gap in (0.2, 0.3, 0.4, 0.5):
        for sig in (0.05, 0.08, 0.12):
            f = (np.exp(-0.5 * ((x - 0.5 + gap / 2) / sig) ** 2) +
                 np.exp(-0.5 * ((x - 0.5 - gap / 2) / sig) ** 2))
            pts.append(f)

    # Cubic/quartic polynomials (positive)
    for _ in range(8):
        c = rng.normal(0, 1.0, 5)
        t = x - 0.5
        f = np.ones(ns) + c[0] * t + c[1] * t**2 + c[2] * t**3 + c[3] * t**4
        f = np.maximum(f, 1e-10)
        pts.append(f)

    # Random Fourier
    for _ in range(10):
        K = int(rng.integers(3, 10))
        a = rng.normal(0, 0.5 / np.sqrt(K), K)
        phi = rng.uniform(0, 2 * np.pi, K)
        f = np.ones(ns)
        for k in range(K):
            f += a[k] * np.cos(2 * np.pi * (k + 1) * x + phi[k])
        f = np.maximum(f, 1e-10)
        pts.append(f)

    return pts


def _resample_to(f_src, n):
    if len(f_src) == n:
        return f_src.copy()
    old = np.linspace(0, 1, len(f_src))
    new = np.linspace(0, 1, n)
    return np.interp(new, old, f_src)
