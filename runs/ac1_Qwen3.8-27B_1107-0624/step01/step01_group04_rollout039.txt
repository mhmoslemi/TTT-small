def _npo2(m):
    """Next power of two >= m."""
    L = 1
    while L < m:
        L <<= 1
    return L


def _normalize(u):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(max(a.size, 1), dtype=np.float64)
        s = float(a.sum())
    return a / s


def _greedy(n):
    """Coefficients of (1-x)^(-1/2), normalized."""
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return _normalize(a)


def _power(n, s):
    """Power-law decay sequence."""
    n = int(n)
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return _normalize(a)


def _convolve_fft(a, n=None):
    """Compute full linear convolution of a with itself using FFT."""
    if n is None:
        n = a.size
    L = _npo2(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _score(a):
    """Compute the objective value (scale-invariant)."""
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-12:
        return np.inf
    c = _convolve_fft(a, n)
    maxc = float(c.max())
    return float(2.0 * n * maxc / (s * s))


def _softmax_adam(p0, budget_s, seed=0, alpha_max=60.0, alpha_min=3.0, lr=None):

    if n < 2:

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    stall_count = 0

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(budget_s, 1e-9))

        # Adaptive temperature: high initially for smoothness, low later for sharpness
        alpha = alpha_min + (alpha_max - alpha_min) * np.exp(-2.5 * frac)

        # Compute convolution
        Pg = np.fft.rfft(p, L)

        c = np.maximum(c, 0.0)

        if maxc <= 0:

        f_now = 2.0 * n * maxc
        if f_now < best_f - 1e-10:

            best_p = p.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Softmax weights over convolution values
        z = alpha * (c / maxc - 1.0)
        z = np.clip(z, -60.0, 0.0)

            w[int(np.argmax(c))] = 1.0

        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)

        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

def _peak_shave(p0, budget_s=60.0, step_size=0.01):
    """
    Coordinate-descent peak shaving: iteratively reduce the max convolution value.
    """
    n = p0.size
    a = _normalize(p0).copy()
    best_a = a.copy()
    best_c = _convolve_fft(a, n).copy()
    best_max = float(best_c.max())

    start = time.monotonic()
    end = start + max(1.0, budget_s - 2.0)

    L = _npo2(2 * n - 1)

    while time.monotonic() < end:
        # Recompute convolution
        Pa = np.fft.rfft(a, L)
        c = np.fft.irfft(Pa * Pa, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        k_star = int(c.argmax())
        M = float(c[k_star])

        if M < best_max - 1e-10:
            best_max = M
            best_c = c.copy()
            best_a = a.copy()

        # Find indices contributing to c[k_star]
        # c[k_star] = sum_{i: 0<=i<n, 0<=k_star-i<n} a[i] * a[k_star-i]
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        if i_lo > i_hi:
            break

        # Contribution of each pair
        pairs = a[i_lo:i_hi + 1] * a[k_star - i_hi:k_star - i_lo + 1][::-1]
        total_pair = float(pairs.sum())
        if total_pair <= 0:
            break

        # Find the index with the largest contribution
        max_idx_rel = int(np.argmax(pairs))
        i_main = i_lo + max_idx_rel
        i_pair = k_star - i_main

        # Try to reduce a[i_main] (or a[i_pair] if it contributes more elsewhere)
        # and redistribute to an index that doesn't contribute to c[k_star]
        # Find an index that doesn't appear in the convolution at position k_star
        # (i.e., a[j] where k_star-j is out of range, meaning j < k_star-n+1 or j > k_star)
        free_indices = []
        for j in range(n):
            if j < i_lo or j > i_hi:
                free_indices.append(j)

        if not free_indices:
            # All indices contribute; try to reduce the largest a[i] in the active set
            # and increase the smallest
            active = list(range(i_lo, i_hi + 1))
            src = active[int(np.argmax(a[i_lo:i_hi + 1]))]
            dst = active[int(np.argmin(a[i_lo:i_hi + 1]))]
        else:
            # Pick the free index with the smallest value
            src = i_main
            dst = int(np.argmin([a[j] for j in free_indices]))
            dst = free_indices[int(np.argmin([a[j] for j in free_indices]))]

        delta = a[src] * step_size
        if delta <= 0:
            break

        a_trial = a.copy()
        a_trial[src] -= delta
        a_trial[dst] += delta

        if a_trial[src] < 0:
            a_trial[src] = 0
            a_trial[dst] += a[src]
            continue

        # Check if the max decreased
        Pa2 = np.fft.rfft(a_trial, L)
        c2 = np.fft.irfft(Pa2 * Pa2, L)[:2 * n - 1].real
        c2 = np.maximum(c2, 0.0)
        max2 = float(c2.max())

        if max2 < M - 1e-12:
            a = a_trial.copy()
        else:
            # Try smaller step
            step_size = max(0.001, step_size * 0.5)

    return _normalize(best_a), 2.0 * n * best_max


def _rescale_output(p, target_sum=10.0):
    """Scale sequence to have the given sum, clip to valid range."""
    a = np.asarray(p, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(16, dtype=np.float64)
        s = float(a.sum())
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    if float(b.sum()) < 0.01:
        b = b * (0.01 / max(float(b.sum()), 1e-30))
    return [float(x) for x in b]


def _validate(seq):
    """Try to evaluate through the global evaluate_sequence."""
    try:
        val = float(evaluate_sequence(list(seq)))
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def _mix(a, b, t):
    """Convex combination of two same-length sequences."""
    return _normalize(t * np.asarray(a) + (1 - t) * np.asarray(b))


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = _score(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
            print("  NEW BEST: n=%d score=%.6f" % (a.size, score))
        n = int(a.size)
        old = best_by_n.get(n)
        if old is None or score < old[0]:
            best_by_n[n] = (float(score), a.copy())

    # Load provided sequence if available
    provided = None
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0 and float(arr.sum()) > 0.01:
                provided = _normalize(arr)
                record(provided)
                print("Loaded provided sequence: n=%d score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    # Phase 1: Quick baselines at various lengths
    base_lengths = [128, 192, 256, 384, 512, 768, 1024, 1536, 2048]
    for n in base_lengths:
        if time.monotonic() >= deadline:
            break
        for init_func in [_greedy, lambda: _power(n, 0.4), lambda: _power(n, 0.5),
                          lambda: _power(n, 0.6), lambda: _power(n, 0.8)]:
            if time.monotonic() >= deadline:
                break
            try:
                p = init_func()
                record(p)
            except Exception:
                pass

    print("Phase 1 done. Best: n=%d score=%.6f" % (best_p.size, best_score))

    # Phase 2: Softmax-Adam optimization from multiple starting points
    # Prioritize the best known length and nearby lengths
    target_lengths = set()
    if provided is not None:
        target_lengths.add(int(provided.size))
    for n in best_by_n:
        target_lengths.add(n)
    target_lengths.update([256, 384, 512, 768, 1024, 1536, 2048, 3072])

    # Sort by score (best first), limiting to top lengths
    length_scores = [(best_by_n[n][0], n) for n in best_by_n if n <= 4096]
    length_scores.sort()
    priority_lengths = [n for _, n in length_scores[:6]]
    # Ensure we have a good spread
    for n in [256, 512, 1024, 2048]:
        if n not in priority_lengths and len(priority_lengths) < 8:
            priority_lengths.append(n)

    # Generate starting points
    starts_info = []
    for n in priority_lengths:
        if n > 4096 or n < 64:
            continue
        # Greedy
        starts_info.append((n, _greedy(n)))
        # Power variants
        for ps in [0.35, 0.45, 0.55, 0.7]:
            starts_info.append((n, _power(n, ps)))
        # Perturbed greedy
        for sig in [0.1, 0.3, 0.5]:
            base = _greedy(n)
            noise = np.exp(rng.normal(0, sig, n))
            starts_info.append((n, _normalize(base * noise)))
        # Dirichlet
        starts_info.append((n, _normalize(rng.dirichlet(np.ones(n) * 0.5))))

        # If we have a good sequence at this length, perturb it
        if n in best_by_n and best_by_n[n][0] < 1.55:
            bp = best_by_n[n][1]
            for sig in [0.05, 0.1, 0.2]:
                noise = np.exp(rng.normal(0, sig, n))
                starts_info.append((n, _normalize(bp * noise)))

    # Sort starts by initial score
    scored_starts = []
    for n, p in starts_info:
        sc = _score(p)
        if np.isfinite(sc):
            scored_starts.append((sc, n, p))
    scored_starts.sort()

    # Optimize top starts
    max_starts = 20
    per_start_budget = max(8.0, (deadline - time.monotonic() - 60.0) / max(1, min(max_starts, len(scored_starts))))

    count = 0
    for sc, n, p in scored_starts:
        if time.monotonic() >= deadline - 90.0:
            break
        if count >= max_starts:
            break
        if sc >= best_score * 1.1:  # Skip if too far from best
            count += 1
            continue

        alloc = min(per_start_budget, deadline - time.monotonic() - 30.0)
        if alloc < 5.0:
            break

        # Select alpha range based on how good the start is
        if sc < best_score * 1.01:
            alpha_max, alpha_min = 30.0, 2.5
        elif sc < best_score * 1.05:
            alpha_max, alpha_min = 50.0, 3.0
        else:
            alpha_max, alpha_min = 80.0, 4.0

        p_out, f_out = _softmax_adam(p, alloc, seed=int(seed) + count,
                                      alpha_max=alpha_max, alpha_min=alpha_min)
        record(p_out, f_out)
        count += 1

    print("Phase 2 done. Best: n=%d score=%.6f" % (best_p.size, best_score))

    # Phase 3: LP improvement on the best sequence
    if HAS_SCIPY and best_p is not None:
        lp_n = int(best_p.size)
        if 64 <= lp_n <= 4096:
            lp_budget = min(300.0, max(60.0, deadline - time.monotonic() - 60.0))
            if lp_budget > 40:
                print("Running LP improvement for n=%d..." % lp_n)
                p_out, f_out = _lp_improve(best_p, budget_s=lp_budget)
                record(p_out, f_out)

        # Also try LP at a nearby length
        for alt_n in [int(lp_n * 1.5), int(lp_n * 0.75), int(lp_n * 2)]:
            if time.monotonic() >= deadline - 90.0:
                break
            if not (64 <= alt_n <= 4096):
                continue
            base = _greedy(alt_n)
            if alt_n in best_by_n:
                base = best_by_n[alt_n][1]
            lp_budget = min(180.0, max(40.0, deadline - time.monotonic() - 60.0))
            if lp_budget < 35:
                break
            p_out, f_out = _lp_improve(base, budget_s=lp_budget)
            record(p_out, f_out)

    print("Phase 3 done. Best: n=%d score=%.6f" % (best_p.size, best_score))

    # Phase 4: Final polishing with softmax-Adam + peak shaving
    if best_p is not None:
        rem = deadline - time.monotonic()

        if rem > 60:
            # More Adam on the best
            alloc = min(rem - 20.0, 200.0)
            p_out, f_out = _softmax_adam(best_p, alloc, seed=int(seed) + 9999,
                                          alpha_max=20.0, alpha_min=2.0)
            record(p_out, f_out)

        rem = deadline - time.monotonic()
        if rem > 30:
            # Peak shaving
            alloc = min(rem - 10.0, 80.0)
            if alloc > 10:
                p_out, f_out = _peak_shave(best_p, budget_s=alloc)
                record(p_out, f_out)

        # Try one more Adam pass from a slightly perturbed best
        rem = deadline - time.monotonic()
        if rem > 40:
            noise = np.exp(rng.normal(0, 0.05, best_p.size))
            p_pert = _normalize(best_p * noise)
            alloc = min(rem - 10.0, 100.0)
            p_out, f_out = _softmax_adam(p_pert, alloc, seed=int(seed) + 7777,
                                          alpha_max=15.0, alpha_min=2.0)
            record(p_out, f_out)

    # Phase 5: Final validation
    if best_p is None:
        best_p = _greedy(256)
        best_score = _score(best_p)

    # Validate with the actual evaluator
    final_seq = _rescale_output(best_p, 10.0)
    gscore = _validate(final_seq)
    if np.isfinite(gscore) and gscore < best_score:
        best_score = gscore
        print("Global eval improved! score=%.6f" % best_score)

    # Also validate a few alternatives
    for n, (sc, p) in sorted(best_by_n.items(), key=lambda x: x[1][0])[:3]:
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break
        seq = _rescale_output(p, 10.0)
        gv = _validate(seq)
        if np.isfinite(gv) and gv < best_score:
            best_p = p.copy()
            best_score = gv
            print("  Alt n=%d: global score=%.6f (better!)" % (n, gv))

    print("FINAL: n=%d score=%.6f" % (best_p.size, best_score))
    return _rescale_output(best_p, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the objective."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main search: %s" % str(e))
        import traceback
        traceback.print_exc()
        # Fallback
        try:
            h = globals().get("height_sequence_1", None)
            if h is not None:
                arr = np.asarray(h, dtype=np.float64).ravel()
                arr = np.clip(arr, 0.0, 1000.0)
                if arr.size > 0 and float(arr.sum()) > 0.01:
                    return _rescale_output(arr, 10.0)
        except Exception:
            pass
        return _rescale_output(_greedy(256), 10.0)
