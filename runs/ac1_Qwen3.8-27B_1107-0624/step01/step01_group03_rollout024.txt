import numpy as np
from scipy.optimize import linprog
import time

"""
Gradient-based optimization with LP polishing for minimizing
2*n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
Uses log-sum-exp smoothed gradient for the main optimization,
then LP-guided local search for final polishing.
"""

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not np.any(cond):
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def compute_gradient(s, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau * conv(s,s)[k])).
    Returns gradient with respect to s.
    """
    n = len(s)
    c = np.convolve(s, s)  # length 2n-1
    
    # Softmax weights
    tc = tau * c
    tc_shifted = tc - np.max(tc)
    exp_tc = np.exp(tc_shifted)
    p = exp_tc / np.sum(exp_tc)
    
    # grad[i] = 2 * sum_j p[i+j] * s[j] = 2 * corr(p, s)[i]
    # Using numpy: corr(p, s)[i] = np.correlate(p, s, 'full')[i + n - 1]
    corr = np.correlate(p, s, mode='full')
    grad = 2.0 * corr[n-1 : 2*n-1]
    
    return grad

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def smoothed_score(s, tau):
    """Compute (1/tau)*log(sum_k exp(tau * conv(s,s)[k])) * 2n / (sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    c = np.convolve(s, s)
    tc = tau * c
    logsumexp = np.max(tc) + np.log(np.sum(np.exp(tc - np.max(tc))))
    return 2.0 * n * (logsumexp / tau) / (total * total)

def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where row k, col j = s[k-j] if valid."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A

def solve_lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. (s*b)[k] <= M, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog
import time

"""
Hybrid smoothed-gradient + LP local search to minimize 2*n*max(conv(s,s))/(sum(s))^2
over non-negative sequences. Uses log-sum-exp smoothing for the max in the objective,
computes the gradient via cross-correlation, and refines with LP-guided direction
finding. Multi-start with diverse parameterized families and multiple sequence lengths.
"""


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using the standard algorithm."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u * ind > cssv
    if not np.any(cond):
        return np.ones(n) / n
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def compute_gradient(s, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv(s,s)[k])) w.r.t. s.
    Uses the identity: grad[i] = 2 * sum_j p[i+j] * s[j] = 2 * corr(p,s)[i]
    where p = softmax(tau * conv(s,s)).
    """
    n = len(s)
    c = np.convolve(s, s)  # length 2n-1
    tc = tau * c
    tc_shifted = tc - np.max(tc)
    exp_tc = np.exp(tc_shifted)
    p = exp_tc / np.sum(exp_tc)
    # grad[i] = 2 * corr(p, s)[i] where corr(p,s)[i] = sum_j p[i+j]*s[j]
    # numpy: np.correlate(p, s, 'full')[i + n - 1] = sum_j p[i+j]*s[j]
    corr = np.correlate(p, s, mode='full')
    grad = 2.0 * corr[n - 1: 2 * n - 1]
    return grad


def gradient_descent(s, max_time=120.0, tau_start=40.0, tau_end=2.0,

    Gradient descent on smoothed objective with simplex projection.
    Uses backtracking line search and adaptive tau.
    """

    s = project_simplex(s)

    no_improve = 0
    lr_reset = lr_start

    while time.monotonic() < start_time + max_time:
        progress = min(1.0, (time.monotonic() - start_time) / max_time)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.5)

        g = compute_gradient(s, tau)
        # Project gradient onto tangent space of simplex
        g = g - np.mean(g)

        improved = False
        for _ in range(30):
            s_new = s - lr * g
            s_new = project_simplex(s_new)
            sc = fast_score(s_new)

                s = s_new
                best_s = s.copy()
                best_score = sc
                no_improve = 0
                improved = True
                break
            elif sc < fast_score(s) - 1e-14:
                s = s_new
                no_improve += 1
                improved = True

            no_improve += 1

def build_conv_matrix(s):
    """Build (2n-1) x n matrix where row k, col j = s[k-j] if 0<=k-j<n else 0."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def solve_lp_direction(s, M, time_limit=2.0):
    """Solve LP: maximize sum(b) s.t. conv(s,b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    best_sc = fast_score(s)
    no_imp = 0
    start_time = time.monotonic()

        if time.monotonic() > start_time + max_time:
            break

        M = np.max(conv)

        b = solve_lp_direction(s, M, time_limit=1.5)
        if b is None or np.sum(b) < 1e-14:
            break

        b = b / b.sum()

        best_t = 0.0
        best_ts = best_sc

def upsample(s, new_n):
    """Upsample sequence s to length new_n using linear interpolation."""
    old_n = len(s)
    if old_n == new_n:
        return s.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    s_new = np.interp(x_new, x_old, s)
    s_new = np.maximum(s_new, 0.0)
    if s_new.sum() > 0:
        s_new = s_new / s_new.sum()
    else:
        s_new = np.ones(new_n) / new_n
    return s_new

                lr_start=3.0, rng=rng
            )
            print(f"-> GD: {best_sc:.6f}", end=" ")

            if best_sc < best_score_n:
                best_score_n = best_sc
                best_for_n = best_s.copy()

            # LP polishing
            if time.monotonic() < deadline - 40:
                lp_time = min(45, (deadline - time.monotonic()) * 0.25)
                polished_s, polished_sc = lp_polish(best_s.copy(), max_time=lp_time, rng=rng)
                print(f"-> LP: {polished_sc:.6f}")
                if polished_sc < best_score_n:
                    best_score_n = polished_sc
                    best_for_n = polished_s.copy()
            else:
                print()

            if best_score_n < global_best_score:
                global_best_score = best_score_n
                global_best_seq = best_for_n.copy()
                print(f"  ** GLOBAL BEST: {global_best_score:.6f} (n={n})")

        print(f"  Best for n={n}: {best_score_n:.6f}")

        # Multi-scale: upsample best solution to larger n and continue
        # (handled implicitly by trying larger n with hseq-like init from best)

    # Additional random restarts at the best length if time permits
    if time.monotonic() < deadline - 30:
        # Find the length that gave the best result
        best_n = len(global_best_seq)
        print(f"\n=== Random restarts at n={best_n} ===")
        restarts = 0
        while time.monotonic() < deadline - 20 and restarts < 10:
            s_r = rng.random(best_n) + 0.01
            s_r = s_r / s_r.sum()
            # Quick GD
            gd_time = min(30, (deadline - time.monotonic() - 20) * 0.5)
            if gd_time < 5:
                break
            best_s, best_sc = gradient_descent(s_r, max_time=gd_time,
                                                tau_start=30.0, tau_end=3.0,
                                                lr_start=2.0, rng=rng)
            # Quick LP
            if best_sc < global_best_score and time.monotonic() < deadline - 20:
                polished_s, polished_sc = lp_polish(best_s, max_time=20, rng=rng)
                if polished_sc < global_best_score:
                    global_best_score = polished_sc
                    global_best_seq = polished_s.copy()
                    print(f"  Restart {restarts}: NEW BEST {global_best_score:.6f}")
            restarts += 1

    # Try multi-scale refinement: take global best, upsample, and polish
    if time.monotonic() < deadline - 30 and len(global_best_seq) < 2048:
        for target_n in [1024, 2048]:
            if time.monotonic() > deadline - 30:
                break
            if len(global_best_seq) >= target_n:
                continue
            print(f"\n=== Multi-scale: upsampling to n={target_n} ===")
            s_up = upsample(global_best_seq, target_n)
            # Add some noise to avoid getting stuck
            s_up = s_up + 0.005 * rng.standard_normal(target_n)
            s_up = project_simplex(s_up)

            gd_time = min(60, (deadline - time.monotonic() - 30) * 0.5)
            if gd_time < 5:
                break
            best_s, best_sc = gradient_descent(s_up, max_time=gd_time,
                                                tau_start=30.0, tau_end=3.0,
                                                lr_start=2.0, rng=rng)
            print(f"  GD at n={target_n}: {best_sc:.6f}")

            if best_sc < global_best_score and time.monotonic() < deadline - 20:
                ls, lsc = lp_polish(best_s, max_time=30, rng=rng)
                if lsc < global_best_score:
                    global_best_score = lsc
                    global_best_seq = ls.copy()
                    print(f"  ** Multi-scale NEW BEST: {global_best_score:.6f}")

    # Final verification
    final_score = evaluate_sequence(global_best_seq.tolist())
    print(f"\nFinal verified score: {final_score:.6f}, length={len(global_best_seq)}")

    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
