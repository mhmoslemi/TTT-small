```python
                    parent2 = current_pop[np.random.randint(0, batch_size)]
                    # Select reproduction style
                    if np.random.random() < 0.1 or allows_structured:  # Specialized reproduction
                        child = specialized_crossover(parent1, parent2)
                    else:
                        child = crossover(parent1, parent2)
                    # Optional mutation
                    if np.random.random() < 0.7:  # 70% mutation probability
                        child = mutation(child)
                    new_pop.append(child)
            current_pop = np.array(new_pop)
        
        # Evaluate only the new cousins
        fitness_scores = np.apply_along_axis(evaluate_individual, 0, current_pop)
        total_evals += batch_size
        
        # Check if we reached the solution
        if np.min(fitness_scores) < 1.5027:
            print(f"Discovered solution at generation {generation} with score {best_score:.6f}")
            break
            
        # Check time boundaries
        if time.time() - start_time > random_search_budget * n_default:
            break
    
    # If no solution, use the fallback
    if best_score == float("inf"):
        best_score = fallback_alpha
        best_z = np.zeros(n_default-1)  # This will be converted in the final return
    
    # Final normalization to ensure valid output
    final_sequence = [np.exp(x) for x in best_z]
    final_sum = np.sum(final_sequence)
    # Adjust scaling if necessary (but inf is found so likely not)
    if final_sum < 0.01:
        final_sequence = final_sequence * (0.01 / final_sum)
    # Clip and prepare return
    final_sequence = np.clip(final_sequence, 0.0, 1000.0) 
    return final_sequence.tolist()
```
