def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    f = max(n, 1)
    # Round up to power of 2 for FFT efficiency
    while (1 << f) < n:
        f += 1
    F = np.fft.rfft(a, 1 << f)
    G = np.fft.rfft(b, 1 << f)
    return np.fft.irfft(F * G, 1 << f)[:n]

**Key question: What structure of a minimizes max(conv(a,a))?**

Intuitively, we want the sequence to be "spread out" but in a way that avoids constructive addition at any single lag. This is like a "Barker sequence" or "costas array" problem, but for continuous non-negative values.

For the continuous case, the optimal solution is related to the "Water-filling" principle or the solution of a variational problem.

**Variational approach:**
Minimize max_k conv[k] subject to sum(a) = 1, a >= 0.

By KKT conditions, at the optimum:
- If a[i] > 0, the gradient of the objective with respect to a[i] should be constant (for the active constraint).
- The active constraint is the max (or the set of lag values that achieve the max).

If the max is achieved at multiple lags k_1, k_2, ..., then:
For all i with a[i] > 0: sum_m a[k_m - i] + sum_m a[i - k_m] = lambda (constant)
(where we only include terms where the index is in [0, n-1])

This is a system of equations that characterizes the optimum.

**Practical approach: Iterative water-filling**
1. Start with a_0 (e.g., uniform or cosine window)
2. Compute conv(a, a), find the peak lags K
3. For each i, compute g[i] = sum_{k in K} (a[k-i] + a[i-k] if valid) / |K|
4. Update: a[i] <- a[i] * (1 - eta * g[i] / mean(g)) [if a[i] > 0]
5. Renormalize
6. Repeat

This is essentially the subgradient method I mentioned before. The issue is convergence speed.

"""
Hybrid optimization combining LP-based direction finding, 
least-squares convolution shaping, and subgradient methods
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except:
        pass
    return None

def ls_direction(a, c, target):
    """Solve least-squares: min ||conv(a, delta) - (target - c)||^2 + lambda*||delta||^2."""
    n = len(a)
    A = build_conv_matrix(a)
    rhs = target - c
    # Solve (A^T A + lambda I) delta = A^T rhs
    lam = 1e-6
    AT = A.T
    # Use normal equations
    try:
        delta = np.linalg.solve(AT @ A + lam * np.eye(n), AT @ rhs)
        return delta
    except:
        try:
            delta, _, _, _ = np.linalg.lstsq(A + lam * np.eye(A.shape[1]), rhs, rcond=None)
            return delta
        except:
            return None

def subgradient_step(a, c, peak_frac=0.85, eta=0.1):
    """Subgradient step based on peak contributions."""
    n = len(a)
    M = c.max()
    threshold = peak_frac * M
    peak_mask = c > threshold
    peak_indices = np.where(peak_mask)[0]
    
    contrib = np.zeros(n)
    for i in range(n):
        for k in peak_indices:
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[j]
    
    contrib /= len(peak_indices) if len(peak_indices) > 0 else 1
    avg_c = contrib.mean() + 1e-15
    update = np.exp(-eta * contrib / avg_c)
    return a * update

        return a, np.inf

    best_score = compute_objective(a)

    stall = 0
    it = 0

        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        # Choose method based on iteration
        if it % 3 == 0:
            # LP step
            g = lp_direction(a, M)
            if g is None or g.sum() < 1e-15:
                g = None
            if g is not None:
                g = g / g.sum()
                direction = g
            else:
                direction = None
        elif it % 3 == 1:

            target = c.mean()
            delta = ls_direction(a, c, target)
            if delta is not None:
                # Normalize delta
                delta = delta / (np.max(np.abs(delta)) + 1e-15)
                direction = None
                ls_delta = delta
            else:
                direction = None
                ls_delta = None
        else:

            if a_sub.sum() > 1e-10:
                a_sub = a_sub / a_sub.sum()
                direction = a_sub
            else:
                direction = None
            ls_delta = None

"""
Hybrid optimization combining LP-based direction finding, least-squares
convolution shaping, and subgradient methods to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive step sizes, backtracking line search, and structured perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def ls_direction(a, c, target, lam=1e-4):
    """Least-squares direction to flatten convolution."""
    n = len(a)
    if n > 3000:
        return None
    A = build_conv_matrix(a)
    rhs = target - c
    try:
        AT = A.T
        ATA = AT @ A + lam * np.eye(n)
        ATrhs = AT @ rhs
        delta = np.linalg.solve(ATA, ATrhs)
        if np.any(np.isnan(delta)):
            return None
        return delta
    except Exception:
        return None


def subgradient_direction(a, c, peak_frac=0.9, eta=0.15):
    """Multiplicative subgradient step."""
    n = len(a)
    M = c.max()
    threshold = peak_frac * M
    peak_mask = c > threshold
    peak_indices = np.where(peak_mask)[0]
    if len(peak_indices) == 0:
        return a.copy()
    contrib = np.zeros(n)
    for i in range(n):
        js = peak_indices - i
        valid = (js >= 0) & (js < n)
        if valid.any():
            contrib[i] = a[js[valid]].sum()
    avg_c = contrib.mean() + 1e-15
    update = np.exp(-eta * contrib / avg_c)
    return a * update


def kkt_check(a, c):
    """Check how close to KKT conditions we are. Returns variance of peak contribution on support."""
    n = len(a)
    M = c.max()
    peak_mask = c >= 0.99 * M
    peak_indices = np.where(peak_mask)[0]
    if len(peak_indices) == 0:
        return 0.0
    g = np.zeros(n)
    for i in range(n):
        total = 0.0
        for k in peak_indices:
            j1 = k - i
            j2 = i - k + (2 * n - 2) - (n - 1)  # not quite, let me redo
            # conv[k] = sum_j a[j]*a[k-j], contribution of a[i] is a[k-i] if valid
            j = k - i
            if 0 <= j < n:
                total += a[j]
        g[i] = total
    support = a > 1e-8
    if support.sum() < 2:
        return 0.0
    gv = g[support]
    return gv.std() / (gv.mean() + 1e-15)


def backtracking_line_search(best_a, direction, best_score, max_iter=10):
    """Find step size via backtracking."""
    s = 0.2
    for _ in range(max_iter):
        a_new = best_a + s * direction
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            s *= 0.5
            continue
        a_new = a_new / a_new.sum()
        sc = compute_objective(a_new)
        if sc < best_score - 1e-10:
            return a_new, sc
        s *= 0.5
    return None, best_score

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.28):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.01, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_optimal_shape(n):
    """Initialize based on approximate KKT solution."""
    t = np.linspace(0, 1, n)
    return 1.0 + 0.2 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t)


def make_bell-shaped(n):
    """A bell-shaped window with tunable width."""
    center = (n - 1) / 2.0
    x = (np.arange(n) - center) / (n * 0.35)
    return np.exp(-0.5 * x ** 2)
