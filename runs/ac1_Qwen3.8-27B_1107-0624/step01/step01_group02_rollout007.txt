        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        for j in range(jlo, jhi):
            A[r, j] = f[k - j]
    
    if not np.any(A):
        return None
    
    A = A / M
    c_vec = -np.ones(n, dtype=np.float64)
    b_vec = np.ones(rows.size, dtype=np.float64)
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=(0.0, None), method='highs')
    except Exception:
        return None
    
    if res is not None and res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        s = float(g.sum())
        if s > 1e-12:
            return g / s  # normalize to sum 1
    return None

def random_perturb(f, rng, scale=0.1):
    """Random perturbation for diversity."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    h = np.maximum(f, 0.0).copy()
    if n > 1:
        # Multiplicative noise
        h = h * np.exp(rng.normal(0.0, scale, size=n))
        # Block perturbation
        width = max(2, n // 10)
        start = int(rng.integers(0, max(1, n - width)))
        h[start:start+width] *= np.exp(rng.normal(0.0, scale * 2.0))
    return normalize_nonneg(h)

def make_block_sequence(N, K, rng):
    """Create a K-block sequence with random masses."""
    # Use Dirichlet for masses, but bias toward asymmetric
    if K == 2:
        # Optimal 2-block: ratio 2:1
        p1 = rng.uniform(0.55, 0.85)
        p2 = 1.0 - p1
        masses = np.array([p1, p2])
    else:
        # Asymmetric masses
        alphas = rng.uniform(0.3, 2.0, size=K)
        # Bias: first block gets more mass
        alphas[0] *= 1.5
        masses = alphas / alphas.sum()
    
    lengths = np.zeros(K, dtype=np.int64)
    base_len = N // K
    remainder = N % K
    for i in range(K):
        lengths[i] = base_len + (1 if i < remainder else 0)
    
    a = np.zeros(N, dtype=np.float64)
    pos = 0
    for i in range(K):
        L = int(lengths[i])
        if L > 0 and pos + L <= N:
            a[pos:pos+L] = masses[i] / L
            pos += L
    
    return normalize_nonneg(a)

def make_gaussian_sequence(N, rng):
    """Create a Gaussian-bump sequence."""
    mu = rng.uniform(0.2, 0.8)
    sigma = rng.uniform(0.1, 0.35)
    x = np.linspace(0, 1, N)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return normalize_nonneg(a)

def make_power_sequence(N, rng):
    """Create a power-law sequence."""
    alpha = rng.uniform(-2.0, 3.0)
    x = np.linspace(0.01, 1.0, N)
    a = x ** alpha
    return normalize_nonneg(a)

def optimize_local(f, time_limit, rng, state):
    """
    Local optimization: alternate between LP, subgradient, and peak cancellation.
    """
    f = normalize_nonneg(f)
    n = f.size
    deadline = time.monotonic() + max(0.5, float(time_limit))

    it = 0
    no_improve = 0
    phase = 0  # 0: LP, 1: subgrad, 2: perturb

        c = np.convolve(f, f)
        M = float(c.max())
        cur_score = float(2.0 * n * M)

            best_score = cur_score
            best = f.copy()
            if best_score < state['score'] - 1e-12:
                state['score'] = best_score
                state['seq'] = best.copy()
                print(f"  improve: score={best_score:.10f} n={n} it={it}", flush=True)
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Reset with perturbation
            f = random_perturb(f, rng, 0.12)
            no_improve = 0
            continue
        
        improved = False
        
        # Phase 0: LP direction
        if n <= 2048:
            if n <= 512:
                max_rows = 2*n - 1  # Use all constraints
            elif n <= 1024:
                max_rows = 800
            else:
                max_rows = 500
            
            g = lp_direction(f, c, M, max_rows)
            if g is not None:
                # Line search for blend
                best_blend = None
                best_blend_sc = cur_score
                Sg_check = float(g.sum())  # g is already normalized to sum 1
                for t in (0.1, 0.2, 0.3, 0.5, 0.7, 0.9):
                    h = (1.0 - t) * f + t * g
                    h = normalize_nonneg(h)
                    sc = score_norm(h)
                    if sc < best_blend_sc:
                        best_blend_sc = sc
                        best_blend = h
                
                if best_blend is not None and best_blend_sc < cur_score - 1e-12:
                    f = best_blend
                    improved = True
        
        # Phase 1: Subgradient step
        if not improved and time.monotonic() < deadline:
            alphas = (0.01, 0.03, 0.08, 0.15, 0.3) if n <= 512 else (0.005, 0.01, 0.03, 0.08, 0.15)
            h, sc = subgradient_step(f, c, top_k=min(20, 2*n-1), alpha_list=alphas)
            if h is not None and sc < cur_score - 1e-12:
                f = h
                improved = True
        
        # Phase 2: Peak cancellation (SOC-like)
        if not improved and time.monotonic() < deadline:
            # Find the peak lag and reduce contributing elements
            peak_idx = int(np.argmax(c))
            # Subgradient at the peak: grad[j] = 2*f[peak_idx - j] (valid indices)
            jlo = max(0, peak_idx - n + 1)
            jhi = min(n, peak_idx + 1)
            grad = np.zeros(n, dtype=np.float64)
            grad[jlo:jhi] = 2.0 * f[peak_idx - np.arange(jlo, jhi)]
            grad -= grad.mean()
            
            for alpha in (0.02, 0.05, 0.1, 0.2):
                h = f - alpha * grad
                h = normalize_nonneg(h)
                sc = score_norm(h)
                if sc < cur_score - 1e-12:
                    f = h
                    improved = True
                    break
        
        if not improved:
            f = random_perturb(f, rng, 0.03)
    
    # Final check
    c = np.convolve(f, f)
    final_sc = score_norm(f)
    if final_sc < best_score:
        best_score = final_sc
        best = f.copy()
        if best_score < state['score']:
            state['score'] = best_score
            state['seq'] = best.copy()
    
    return best, best_score
