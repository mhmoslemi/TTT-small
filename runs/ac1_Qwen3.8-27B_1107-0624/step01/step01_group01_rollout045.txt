def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c

    c = np.asarray(c, dtype=np.float64)
    m = c.size
    if m == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    idx = np.argsort(c)[::-1]
    if m >= 2:
        gap = c[idx[0]] - c[idx[1]]
    else:
        gap = mval
    if gap < 1e-14:
        beta = max(1e-12, 0.1 * max(mval, 1e-10))
    else:
        beta = max(1e-12, 0.5 * gap)
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j], for i=0..len(p)-1."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / e.size
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = _full_convolve_fft(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _score_exact(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _raised_cosine_profile(n, alpha, beta):
    t = np.linspace(-1.0, 1.0, n)
    a = np.power(np.maximum(0.0, (1.0 + t) / 2.0), alpha) * np.power(np.maximum(0.0, (1.0 - t) / 2.0), beta)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _deterministic_profiles(n):
    profs = []
    for eta in (0.8, 1.0, 1.2, 1.5, 1.8, 2.0, 2.5):
        profs.append(_linear_profile(n, eta))
    for alpha, beta in ((0.5, 0.5), (0.8, 0.8), (1.0, 1.0), (0.5, 1.0), (1.0, 0.5),
                        (0.3, 0.3), (1.5, 1.5), (0.7, 1.3), (1.3, 0.7), (2.0, 2.0)):
        profs.append(_raised_cosine_profile(n, alpha, beta))
    t = np.linspace(0, 1, n)
    a = 1.0 + 1.5 * t - 0.5 * t * t
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    a = 1.0 + 0.8 * np.sin(np.pi * t)
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    return profs


def _basis_mixed(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u]
    m = 1
    while len(cols) < K:
        ang = 2.0 * np.pi * m * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
        m += 1
        if len(cols) > K:
            break
    return np.column_stack(cols[:K])


def _basis_poly(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = []
    for k in range(1, K + 1):
        cols.append(u ** k)
    return np.column_stack(cols)


def _basis_window(n, K):
    """Basis of window-like functions."""
    t = np.linspace(-1.0, 1.0, n)
    abs_t = np.abs(t)
    cols = [abs_t]
    m = 1
    while len(cols) < K:
        ang = np.pi * m * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang) * np.sign(t))
        m += 1
        if len(cols) > K:
            break
    return np.column_stack(cols[:K])


def _profile_from_basis(theta, basis):
    z = basis @ np.asarray(theta, dtype=np.float64)
    return _softmax_z(z)


def _adam_param_space(theta0, basis, deadline, rng, lr=0.3, reg=1e-7):
    """Adam in parameter space with adaptive soft-max."""
    basis = np.asarray(basis, dtype=np.float64)
    theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
    dim = theta.size
    n = basis.shape[0]

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(dim)
    v = np.zeros(dim)
    t_adam = 0

    p = _profile_from_basis(theta, basis)
    c = _full_convolve_fft(p)
    best_theta = theta.copy()
    best_p = p.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    while time.monotonic() < deadline:
        L, w = _soft_max_adaptive(c)
        gp = 2.0 * _corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gtheta = basis.T @ gz

        if not np.all(np.isfinite(gtheta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gtheta
        v = b2 * v + (1.0 - b2) * (gtheta * gtheta)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        theta = theta - lr * mh / (np.sqrt(vh) + eps)
        theta = np.clip(theta, -8.0, 8.0)

        p = _profile_from_basis(theta, basis)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_theta = theta.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.05, lr * 0.96)
        if t_adam % 200 == 0 and no_imp > max(200, 4 * dim):
            theta = best_theta + 0.5 * rng.normal(size=dim)
            theta = np.clip(theta, -8.0, 8.0)
            p = _profile_from_basis(theta, basis)
            c = _full_convolve_fft(p)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = 0.3
            no_imp = 0

    return best_theta, best_p, best_score


def _adam_logit_space(p0, deadline, rng, lr=0.1, reg=1e-7):
    """Full logit-space Adam with adaptive soft-max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve_fft(p)
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    while time.monotonic() < deadline:
        L, w = _soft_max_adaptive(c)
        gp = 2.0 * _corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))

        p = _softmax_z(z)
        c = _full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 400 == 0:
                    lr = max(0.02, lr * 0.97)
        if no_imp > max(600, 5 * n):
            z = 0.8 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_z(z)
            c = _full_convolve_fft(p)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = 0.1
            no_imp = 0

    return best_p, best_score


def _peak_shave(p, deadline, rng, n_iters_max=500):
    """
    Targeted peak reduction: find the lag of max convolution, 
    then adjust elements that contribute most to that lag.
    """
    n = p.size
    p = p.copy()
    best_p = p.copy()
    best_score = float(2.0 * n * _full_convolve_fft(p).max())

    for it in range(n_iters_max):
        if time.monotonic() > deadline:
            break
        c = _full_convolve_fft(p)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]

        # Find which elements contribute to this peak
        # c[peak_idx] = sum_i p[i] * p[peak_idx - i] for valid i
        i_min = max(0, peak_idx - (n - 1))
        i_max = min(n - 1, peak_idx)
        idx = np.arange(i_min, i_max + 1)
        partners = peak_idx - idx
        contrib = p[idx] * p[partners]

        # Gradient-like direction: reduce the peak-contributing elements
        # For each i in idx, d(c[peak_idx])/d(p[i]) = p[partner] + (if i != partner) p[i]
        # Actually c[m] = sum_j p[j]*p[m-j], so d/dp[i] = p[m-i] (if i != m-i) or 2*p[i] (if i == m-i)
        grad = np.zeros(n)
        for i in idx:
            j = peak_idx - i
            if i == j:
                grad[i] += 2.0 * p[i]
            else:
                grad[i] += p[j]

        # Also add gradient from other near-peak lags to avoid creating new peaks
        sorted_idx = np.argsort(c)[::-1]
        for si in sorted_idx[:5]:
            if si == peak_idx:
                continue
            if c[si] > 0.95 * peak_val:
                ii_min = max(0, si - (n - 1))
                ii_max = min(n - 1, si)
                for i in range(ii_min, ii_max + 1):
                    j = si - i
                    if i == j:
                        grad[i] += 0.5 * p[i]
                    else:
                        grad[i] += 0.5 * p[j]

        # Project gradient onto simplex tangent (maintain sum=1)
        grad -= np.mean(grad)
        # Only move in non-negative direction
        step = 0.01
        for _ in range(10):
            p_new = p - step * grad
            p_new = np.maximum(p_new, 1e-15)
            p_new = p_new / p_new.sum()
            c_new = _full_convolve_fft(p_new)
            sc_new = float(2.0 * n * c_new.max())
            if sc_new < best_score:
                p = p_new
                best_score = sc_new
                best_p = p.copy()
                break
            step *= 0.5
        else:
            # Try a random perturbation
            if it % 50 == 0:
                direction = rng.normal(size=n)
                direction -= np.mean(direction)
                for trial in range(20):
                    p_new = p + 0.02 * direction * (1.0 / (1.0 + trial * 0.1))
                    p_new = np.maximum(p_new, 1e-15)
                    p_new = p_new / p_new.sum()
                    sc_new = float(2.0 * n * _full_convolve_fft(p_new).max())
                    if sc_new < best_score:
                        p = p_new
                        best_score = sc_new
                        best_p = p.copy()
                        break

    return best_p, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining basis optimization, full logit-space Adam,
    and peak shaving. Tries multiple lengths and returns the best sequence.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent if available
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0:
            parent = _normalize_nonneg(parent_arr)
    except Exception:
        parent = None

    # Phase 1: Evaluate deterministic profiles at multiple lengths
    lengths = [128, 256, 384, 512, 768, 1024]
    for n in lengths:
        for p0 in _deterministic_profiles(n):
            sc = _score_fast(p0)
            _update(p0, sc)
    if parent is not None:
        sc = _score_fast(parent)
        _update(parent, sc)
        # Also try interpolated parent at different lengths
        for n in [256, 512, 1024]:
            if parent.size != n:
                pi = _interpolate_profile(parent, n)
                if pi is not None:
                    sc = _score_fast(pi)
                    _update(pi, sc)

    print(f"Phase 1 (deterministic): best={best_score:.10f}")

    # Phase 2: Low-dimensional basis optimization
    # Use multiple bases and many random starts
    basis_configs = [
        (256, _basis_mixed(256, 15)),
        (256, _basis_poly(256, 14)),
        (256, _basis_window(256, 15)),
        (512, _basis_mixed(512, 15)),
        (512, _basis_poly(512, 14)),
    ]

    phase2_end = min(deadline, t_start + 480.0)
    per_alloc = 480.0 / (len(basis_configs) * 25)

    for n, basis in basis_configs:
        dim = basis.shape[1]
        n_starts = max(5, int(per_alloc * 25 / 1.0))
        for s in range(n_starts):
            if time.monotonic() > phase2_end:
                break
            if s == 0:
                th0 = np.zeros(dim)
                th0[0] = 1.5
            elif s == 1:
                th0 = np.zeros(dim)
                th0[0] = 2.0
                if dim > 3:
                    th0[2] = -1.5
            elif s == 2:
                th0 = np.zeros(dim)
                th0[0] = 1.0
                if dim > 5:
                    th0[4] = -1.0
            else:
                th0 = rng.uniform(-4.0, 4.0, dim)

            per = min(phase2_end, time.monotonic() + per_alloc)
            if per - time.monotonic() < 0.5:
                break
            _, p_best, sc = _adam_param_space(th0, basis, per, rng, lr=0.3)
            _update(p_best, float(2.0 * n * sc / (2.0 * n)))
            # sc is already 2*n*maxc, so just use it
            _update(p_best, sc)

    print(f"Phase 2 (basis opt): best={best_score:.10f}")

    # Phase 3: Full logit-space optimization on top candidates
    # Collect best candidates by score
    all_cands = []
    if parent is not None:
        all_cands.append(parent)
    for n in [256, 512]:
        all_cands.extend(_deterministic_profiles(n)[:6])
    if best_seq is not None:
        all_cands.append(best_seq)

    # Deduplicate by length
    seen_lengths = {}
    for c in all_cands:
        key = c.size
        if key not in seen_lengths:
            seen_lengths[key] = c
    all_cands = list(seen_lengths.values())

    phase3_end = min(deadline, t_start + 1080.0)
    n_polish = min(len(all_cands), 12)
    per_polish = (phase3_end - time.monotonic()) / max(1, n_polish) - 1.0

    for i, p0 in enumerate(all_cands[:n_polish]):
        if time.monotonic() > phase3_end:
            break
        per = max(1.0, min(phase3_end - 1.0, time.monotonic() + per_polish))
        p_best, sc = _adam_logit_space(p0, per, rng, lr=0.08)
        _update(p_best, sc)

    print(f"Phase 3 (logit polish): best={best_score:.10f}")

    # Phase 4: Peak shaving on the best
    if best_seq is not None:
        phase4_end = min(deadline, t_start + 1500.0)
        # Peak shave at the current best length
        p_shaved, sc_shaved = _peak_shave(best_seq, phase4_end, rng, n_iters_max=1000)
        _update(p_shaved, sc_shaved)
        # Also try peak shaving on a few other good candidates
        for n in [256, 512]:
            if time.monotonic() > phase4_end:
                break
            for p0 in _deterministic_profiles(n)[:3]:
                if time.monotonic() > phase4_end:
                    break
                p_sh, sc_sh = _peak_shave(p0, min(phase4_end, time.monotonic() + 30.0), rng, n_iters_max=300)
                _update(p_sh, sc_sh)

    print(f"Phase 4 (peak shave): best={best_score:.10f}")

    # Phase 5: Final logit polish on the best found so far (if time remains)
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        per_final = deadline - 30.0 - time.monotonic()
        if per_final > 5.0:
            p_best, sc = _adam_logit_space(best_seq, t_start + time.monotonic() - t_start + per_final, rng, lr=0.05)
            _update(p_best, sc)
            # Another peak shave
            if time.monotonic() < deadline - 20.0:
                p_sh, sc_sh = _peak_shave(best_seq, deadline - 20.0, rng, n_iters_max=500)
                _update(p_sh, sc_sh)

    print(f"Phase 5 (final polish): best={best_score:.10f}")

    # Phase 6: Exact evaluation and return
    if best_seq is None:
        best_seq = _linear_profile(256, 1.5)
        best_score = _score_fast(best_seq)

    # Exact score
    if best_seq.size <= 4096:
        sc_exact = _score_exact(best_seq)
        if np.isfinite(sc_exact):
            best_score = sc_exact

    # Try a few nearby variations
    if time.monotonic() < deadline:
        for n_try in [best_seq.size // 2, best_seq.size, best_seq.size * 2]:
            if n_try <= 0 or n_try > 4096:
                continue
            if n_try != best_seq.size:
                p_try = _interpolate_profile(best_seq, n_try)
                if p_try is not None:
                    sc_try = _score_exact(p_try) if n_try <= 2048 else _score_fast(p_try)
                    if np.isfinite(sc_try) and sc_try < best_score:
                        best_seq = p_try
                        best_score = sc_try

    # Final validation with evaluate_sequence
    if best_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in best_seq]))
            if np.isfinite(sc_eval):
                best_score = sc_eval
        except Exception:
            pass

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s > 1e-10:
        best_seq = best_seq / s
    else:
        best_seq = _linear_profile(256, 1.5)

    print(f"FINAL: score={best_score:.10f}, n={best_seq.size}")
    return [float(x) for x in best_seq]
