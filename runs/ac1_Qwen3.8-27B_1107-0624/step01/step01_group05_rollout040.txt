    return max(0.0, TIME_LIMIT - 2.0 - (time.monotonic() - START_TIME))

def _self_conv_fft(p):
    n = len(p)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    f = np.fft.rfft(p, size)
    c = np.fft.irfft(f * f, size).real[:2*n-1]
    np.maximum(c, 0.0, out=c)
    return c

    return 2.0 * n * float(np.max(c)) / (sp * sp)

    return gz - np.mean(gz)
