def fast_conv(s):
    """Compute full convolution using FFT."""
    n = len(s)
    size = 2*n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = np.fft.rfft(s, fft_size)
    conv = np.fft.irfft(F * F, fft_size)[:size]
    return conv

            break
        s_new = s_new / s_new_total
        
        new_score = score(s_new)
        
        if new_score < best_score:

            best_s = s_new.copy()

                s = np.maximum(s, 0)
                s = s / s.sum()
                no_improve = 0
                alpha = 0.01
