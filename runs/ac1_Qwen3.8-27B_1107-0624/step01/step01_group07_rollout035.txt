def compute_objective(a):

    return 2.0 * n * c.max() / (S * S)

Key improvements: Full-step exploitation when LP direction has low self-convolution,

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def subgradient_step(a, lr):
    """Perform a subgradient/multiplicative update step."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    threshold = 0.85 * M
    peak_mask = c >= threshold
    peak_indices = np.where(peak_mask)[0]
    
    contrib = np.zeros(n)
    for i in range(n):
        js = peak_indices - i
        valid = (js >= 0) & (js < n)
        if valid.any():
            contrib[i] = a[js[valid]].sum()
    
    avg_c = contrib.mean() + 1e-15
    update = np.exp(-lr * contrib / avg_c)
    a_new = a * update
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-10:
        return None
    a_new /= a_new.sum()
    return a_new

"""
Improved hybrid LP-Subgradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: full-step exploitation when LP direction has low self-convolution,
dual-guided subgradient steps, adaptive strategy switching, and better time budgeting.

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns (g, dual) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            dual = None
            try:
                if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                    dual = res.ineqlin.marginals.copy()
            except Exception:
                pass
            return res.x, dual
    except Exception:
        pass
    return None, None


def subgradient_step(a, lr, peak_weights=None):
    """Multiplicative subgradient update. peak_weights optionally biases which peaks matter."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    if M < 1e-15:
        return a.copy()
    if peak_weights is not None:
        threshold = 0.7 * M
        peak_mask = c >= threshold
        peak_indices = np.where(peak_mask)[0]
        # Weight peaks by dual values
        weighted_sum = np.zeros(n)
        for i in range(n):
            js = peak_indices - i
            valid = (js >= 0) & (js < n)
            if valid.any():
                w = peak_weights[js[valid]]
                weighted_sum[i] = (a[js[valid]] * w).sum()
    else:
        threshold = 0.85 * M
        peak_mask = c >= threshold
        peak_indices = np.where(peak_mask)[0]
        weighted_sum = np.zeros(n)
        for i in range(n):
            js = peak_indices - i
            valid = (js >= 0) & (js < n)
            if valid.any():
                weighted_sum[i] = a[js[valid]].sum()
    avg = weighted_sum.mean() + 1e-15
    update = np.exp(-lr * weighted_sum / avg)
    a_new = a * update
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-10:
        return None
    a_new /= a_new.sum()
    return a_new

def make_cosine(n):
    """Hann (raised cosine) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Triangular window."""
    c = (n - 1) / 2.0
    return np.maximum(0.01, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_super_gaussian(n, p=6):
    """Super-Gaussian window."""
    center = (n - 1) / 2.0
    sigma = 0.3 * n
    vals = np.exp(-((np.arange(n) - center) / sigma) ** p)
    return np.maximum(vals, 1e-10)


def make_bimodal(n, width_frac=0.3):
    """Bimodal: two Gaussian bumps near the edges."""
    center1 = 0.2 * n
    center2 = 0.8 * n
    sigma = width_frac * n
    v1 = np.exp(-0.5 * ((np.arange(n) - center1) / sigma) ** 2)
    v2 = np.exp(-0.5 * ((np.arange(n) - center2) / sigma) ** 2)
    return np.maximum(v1 + v2, 1e-10)


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for best non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses hybrid LP + subgradient optimization with full-step exploitation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_objective(parent)
        print(f"Parent: n={parent.size}, score={parent_score:.6f}")
        if parent_score < best_score:
            best_score = parent_score
            best_seq = parent.copy()

    # Phase 1: Heavy optimization on parent sequence
    if parent is not None and (deadline - time.monotonic()) > 60:
        alloc1 = (deadline - time.monotonic()) * 0.45
        d1 = start_time + alloc1
        n_p = parent.size
        print(f"Phase 1: Optimize parent n={n_p}, budget={alloc1:.0f}s")
        a_opt, score = optimize_hybrid(parent.copy(), d1, rng, verbose=True)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 1 improved: {score:.6f}")
        else:
            print(f"  Phase 1: {score:.6f} (best: {best_score:.6f})")

    # Phase 2: Try diverse initializations
    remaining = deadline - time.monotonic()
    configs = []

    # Add parent variants
    if parent is not None:
        if parent.size > 512:
            configs.append((512, 'parent_trunc'))
        if parent.size > 256:
            configs.append((256, 'parent_trunc'))

    # Standard initializations at various sizes
    init_types = ['cosine', 'gaussian', 'parabolic', 'super6', 'bimodal']
    for n in [256, 384, 512, 768, 1024, 1536, 2048]:
        for it_type in init_types:
            configs.append((n, it_type))

    # Deduplicate
    seen = set()
    unique_configs = []
    for cfg in configs:
        if cfg not in seen:
            seen.add(cfg)
            unique_configs.append(cfg)
    configs = unique_configs

    n_configs = len(configs)
    # Allocate time proportional to promise (favor n=384-1024)
    weights = []
    for n, it_type in configs:
        if it_type == 'parent_trunc':
            weights.append(2.0)
        elif 384 <= n <= 1024:
            weights.append(1.2)
        elif n == 256 or n == 1536:
            weights.append(0.8)
        else:
            weights.append(0.5)

    total_w = sum(weights)
    time_per = remaining * 0.5 / total_w

    print(f"Phase 2: {n_configs} configs, total budget={remaining * 0.5:.0f}s")
    for idx, (n, it_type) in enumerate(configs):
        if time.monotonic() > deadline - 60:
            break
        cfg_time = time_per * weights[idx]
        cfg_deadline = min(time.monotonic() + cfg_time, deadline - 60)
        if cfg_deadline <= time.monotonic():
            continue

        # Initialize
        if it_type == 'parent_trunc':
            if parent is not None and parent.size >= n:
                a = parent[:n].copy()
            else:
                a = make_cosine(n)
        elif it_type == 'cosine':
            a = make_cosine(n)
        elif it_type == 'gaussian':
            a = make_gaussian(n)
        elif it_type == 'parabolic':
            a = make_parabolic(n)
        elif it_type == 'super6':
            a = make_super_gaussian(n, p=6)
        elif it_type == 'bimodal':
            a = make_bimodal(n)
        else:
            a = make_cosine(n)

        a = np.clip(a, 1e-10, None)
        if a.sum() < 1e-10:
            a = make_cosine(n)

        a_opt, score = optimize_hybrid(a, cfg_deadline, rng)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  [{idx+1}] NEW BEST: {score:.6f} (n={n}, {it_type})")
        elif (idx + 1) % 5 == 0:
            print(f"  [{idx+1}/{n_configs}] n={n}, {it_type}: {score:.6f}")

    # Phase 3: Final polish on global best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"Phase 3: Final polish, budget={remaining:.0f}s, n={len(best_seq)}")
        a_opt, score = optimize_hybrid(best_seq.copy(), deadline, rng, verbose=True)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
        print(f"  After polish: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = make_cosine(256)
        best_score = compute_objective(best_seq)

    # Ensure validity for evaluator
    best_seq = np.clip(best_seq, 0, None)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / max(best_seq.sum(), 1e-30))

    # Final verification
    try:
        final_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Verified via evaluate_sequence: {final_score:.6f}")
        if not np.isfinite(final_score):
            print("Warning: inf score, using fallback")
            best_seq = make_cosine(256)
    except NameError:
        print("evaluate_sequence not available for verification")
    except Exception as e:
        print(f"Verification failed: {e}")

    elapsed = time.monotonic() - start_time
    print(f"DONE in {elapsed:.1f}s. Final: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
