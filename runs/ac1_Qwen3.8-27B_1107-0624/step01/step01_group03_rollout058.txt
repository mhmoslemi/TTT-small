                    best_s = s.copy()
                no_imp = 0
                improved = True
                break

        if not improved:
            no_imp += 1
            if no_imp > 80:
                break
            # Perturb to escape local minimum
            if no_imp % 15 == 0:
                scale = 0.002 * (1 + no_imp / 50.0)
                s = s + scale * rng.standard_normal(n)
                s = np.maximum(s, 0.0)
                s = s / s.sum()

    return best_s, best_score


def lp_refine(s, max_iters=30, deadline=None):
    """LP-based refinement: find direction b maximizing sum(b) s.t. (s*b)[k]<=M."""
    n = len(s)
    s = s.copy()
    best_score = fast_score(s)
    best_s = s.copy()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = compute_conv_fft(s)
        M = np.max(conv)

        # Build convolution matrix A where A[k,j] = s[k-j] if 0<=k-j<n
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = s

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 8, 'presolve': True})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0.0)
                b_sum = b.sum()
                if b_sum < 1e-12:
                    break
                b = b / b_sum
                # Line search
                for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                    s_new = (1.0 - t) * s + t * b
                    s_new = np.maximum(s_new, 0.0)
                    ss = s_new.sum()
                    if ss < 1e-10:
                        continue
                    s_new = s_new / ss
                    sc = fast_score(s_new)
                    if sc < best_score - 1e-13:
                        s = s_new
                        best_score = sc
                        best_s = s.copy()
                        break
        except Exception:
            break

    return best_s, best_score


def make_starts(n, rng):
    """Generate diverse starting points for sequence of length n."""
    starts = []
    x = np.linspace(0.0, 1.0, n)

    # Uniform
    starts.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    starts.append(rc / rc.sum())

    # Gaussian centered
    g = np.exp(-8.0 * (x - 0.5) ** 2)
    starts.append(g / g.sum())

    # Two bumps
    b2 = np.exp(-30.0 * (x - 0.3) ** 2) + np.exp(-30.0 * (x - 0.7) ** 2)
    starts.append(b2 / b2.sum())

    # Linear ramp up
    ramp = np.linspace(0.05, 1.0, n)
    starts.append(ramp / ramp.sum())

    # Linear ramp down
    ramp2 = np.linspace(1.0, 0.05, n)
    starts.append(ramp2 / ramp2.sum())

    # Parabolic (high in middle)
    par = 1.0 - 4.0 * (x - 0.5) ** 2
    par = np.maximum(par, 0.01)
    starts.append(par / par.sum())

    # Inverted parabola (high at edges)
    ipar = 4.0 * (x - 0.5) ** 2 + 0.05
    starts.append(ipar / ipar.sum())

    # Sinc-like
    sc = np.sinc((x * 10.0) - 5.0)
    sc = np.abs(sc) + 0.01
    starts.append(sc / sc.sum())

    # Random starts
    for _ in range(5):
        r = rng.random(n) + 0.05
        starts.append(r / r.sum())

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(s,s))/(sum(s))^2.
    Uses FFT-based Gauss-Newton iterations with LP refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 40.0

    global_best_score = float('inf')
    global_best_seq = None

    # Load height_sequence_1 if available
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}")
    except Exception:
        pass

    # Sequence lengths to try (larger n allows lower objective)
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        if time.monotonic() > deadline - 90:
            break
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")

        # Build starting points
        starts = make_starts(n, rng)

        # Add height_sequence_1 resized
        if hseq is not None:
            h_r = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, hseq.size), hseq)
            h_r = np.maximum(h_r, 0.0)
            if h_r.sum() > 0.01:
                starts.append(h_r / h_r.sum())

        # Sort starts by initial score
        scored_starts = []
        for s0 in starts:
            sc = fast_score(s0)
            scored_starts.append((sc, s0))
        scored_starts.sort(key=lambda x: x[0])

        print(f"  Best initial score: {scored_starts[0][0]:.6f}")
        print(f"  Starts available: {len(scored_starts)}")

        # Phase 1: GN optimization on top starts
        n_candidates = min(6, len(scored_starts))
        iters_per_start = max(200, int(remaining / n_candidates / 0.05))

        for sc0, s0 in scored_starts[:n_candidates]:
            if time.monotonic() > deadline - 60:
                break

            # Try different top-K values
            for top_k in [1, 3, 5]:
                if time.monotonic() > deadline - 40:
                    break
                best_s, best_sc = gn_optimize(s0, max_iters=iters_per_start,
                                              deadline=deadline - 30,
                                              top_k=top_k)
                if best_sc < global_best_score:
                    global_best_score = best_sc
                    global_best_seq = best_s.copy()
                    print(f"  GN top_k={top_k}: new best {best_sc:.6f} (n={n})")

        # Phase 2: Random restarts with GN
        restarts_done = 0
        while time.monotonic() < deadline - 90 and restarts_done < 4:
            r = rng.random(n) + 0.05
            r = r / r.sum()
            best_s, best_sc = gn_optimize(r, max_iters=300,
                                          deadline=deadline - 60,
                                          top_k=3)
            restarts_done += 1
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  Restart {restarts_done}: new best {best_sc:.6f} (n={n})")

    print(f"\nAfter GN phase: best score = {global_best_score:.6f}")

    # Phase 3: LP refinement on the best sequence
    if global_best_seq is not None and time.monotonic() < deadline - 30:
        print("\n=== LP Refinement ===")
        n_best = len(global_best_seq)
        best_s, best_sc = lp_refine(global_best_seq, max_iters=20,
                                    deadline=deadline - 20)
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()
            print(f"  LP refined: {best_sc:.6f}")

    # Phase 4: Try a few more GN iterations on the best
    if global_best_seq is not None and time.monotonic() < deadline - 15:
        print("\n=== Final GN polish ===")
        for top_k in [1, 3]:
            if time.monotonic() > deadline - 10:
                break
            best_s, best_sc = gn_optimize(global_best_seq, max_iters=500,
                                          deadline=deadline - 10,
                                          top_k=top_k)
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  Final GN top_k={top_k}: {best_sc:.6f}")

    # Final verification
    print(f"\nFinal internal score: {global_best_score:.6f}")
    final_score = float(evaluate_sequence(global_best_seq.tolist()))
    print(f"Final verified score: {final_score:.6f}, length={len(global_best_seq)}")

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
