from scipy.signal import fftconvolve

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = fftconvolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def compute_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    return fftconvolve(a, a)

    """Objective in logit space: 2n * max(conv(softmax(z), softmax(z)))"""
    z = z - z.max()  # for numerical stability
    a = np.exp(z)
    s = a.sum()
    if s < 1e-30:

    a = a / s
    c = fftconvolve(a, a)
    return 2.0 * n * c.max()

def interpolate(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, float).ravel()
    n0 = len(a)
    if n0 == n_new:
        return a / a.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, a)
    s = result.sum()
    return result / s if s > 0 else result

In signal processing, the problem of finding a sequence with low peak-to-average power ratio (PAPR) in its autocorrelation is well-studied. The "Golay complementary sequences" and "Zadoff-Chu sequences" are examples, but those are for complex sequences.

For real, non-negative sequences, the problem is less studied. But the key insight from the signal processing literature is that the optimal sequence tends to be "smooth" (low-frequency) rather than "spiky."

This suggests that the optimal profile should be a smooth curve, and the question is what shape.

Let me think about this from a different angle. If we parameterize a[i] = f(i/(n-1)) for some smooth function f on [0,1], then:
- The convolution peak is approximately n * (f * f)(1/2) / (integral f)^2 * n (roughly)
- Where (f*f)(x) is the convolution of f with itself on [0,1]

Actually, for large n, the discrete convolution approximates the continuous convolution scaled by n:
conv[k] ≈ n * (f * f)(k/n) where (f*f)(x) = ∫ f(t) f(x-t) dt (with appropriate limits)

The peak of the continuous convolution for a non-negative function f on [0,1] is at x=1 (full overlap), where (f*f)(1) = ∫_0^1 f(t) f(1-t) dt.

If f is symmetric (f(t) = f(1-t)), then (f*f)(1) = ∫_0^1 f(t)^2 dt = ||f||_2^2.

But the peak might not be at x=1. For a function concentrated at the edges, the peak might be at x=0 or x=2 (the edges of the convolution).

Wait, for the linear convolution of [0,1] with itself, the support is [0,2]. The value at x is:
(f*f)(x) = ∫ max(0,x-1)^{min(1,x)} f(t) f(x-t) dt

At x=1: ∫_0^1 f(t) f(1-t) dt
At x=0: f(0)^2 (degenerate)
At x=0.5: ∫_0^0.5 f(t) f(0.5-t) dt

So the maximum is likely at x=1 (the center) for most reasonable profiles.

If the profile is symmetric, the problem reduces to minimizing ∫_0^1 f(t)^2 dt subject to ∫_0^1 f(t) dt = 1 and f >= 0. By Cauchy-Schwarz, the minimum is 1 (achieved by f = 1 uniformly). But that gives score = 2 (the uniform case).

For non-symmetric profiles, the center value is ∫_0^1 f(t) f(1-t) dt, which can be less than ∫_0^1 f(t)^2 dt. The product f(t)*f(1-t) is maximized when f(t) = f(1-t) (by AM-GM or rearrangement), so for non-symmetric f, the center convolution is actually smaller!

But the issue is that other convolution values might become larger. For instance, if f is concentrated at t=0, then (f*f)(0) = f(0)^2 could be large.

So the trade-off is:
- Asymmetric profiles reduce the center peak
- But they might increase the edge peaks
- The optimal balances these

This suggests that the optimal profile is slightly asymmetric, with more mass at one end. This is consistent with what the previous optimizer found (linear or ramp profiles).

"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines exact subgradient multiplicative updates, L-BFGS-B with annealed
soft-max, polynomial parameterization with Nelder-Mead, multi-scale
interpolation, and coordinate-wise polishing.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import fftconvolve


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = fftconvolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def compute_conv(a):
    """Full linear convolution of a with itself."""
    return fftconvolve(a, a)


def subgradient_at_peak(a, k):
    """
    Compute exact subgradient of conv[k] wrt a.
    grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    k = int(k)
    grad = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min <= j_max:
        idx_start = k - j_min
        idx_end = k - j_max
        grad[j_min:j_max + 1] = 2.0 * a[idx_end:idx_start + 1][::-1]
    return grad


def soft_max_obj_and_grad(z, n, beta):
    """
    Soft-max smoothed objective and gradient in logit space.
    Objective: 2n * soft_max(conv(softmax(z), softmax(z)), beta)
    """
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-30:
        return 1e10, np.zeros_like(z)
    a = a / s

    c = fftconvolve(a, a)
    mval = float(c.max())

    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    esum = float(ex.sum())
    if esum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        softmax_val = mval
    else:
        w = ex / esum
        softmax_val = mval + beta * float(np.log(esum))

    obj = 2.0 * n * softmax_val

    # h[j] = sum_k w[k] * a[k-j] for valid k-j
    h = np.correlate(w, a, mode='full')[:n]
    grad_a = 2.0 * h

    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot) * 2.0 * n

    return obj, grad_z


def subgrad_optimize(a0, deadline, rng, eta0=0.05, eta_decay=0.9995,
                     top_k=1, noise_scale=0.0, restart_every=300):
    """
    Subgradient descent with multiplicative updates.
    Uses exact subgradients at the top-k peaks.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    eta = eta0
    no_improve = 0

    while time.monotonic() < deadline:
        c = fftconvolve(a, a)
        top_k_actual = min(top_k, len(c))
        peak_indices = np.argpartition(c, -top_k_actual)[-top_k_actual:]

        # Average subgradient over top-k peaks
        grad = np.zeros(n)
        for k in peak_indices:
            grad += subgradient_at_peak(a, k)
        grad /= top_k_actual

        # Multiplicative update
        z = np.log(a + 1e-30)
        grad_centered = grad - grad.mean()
        z = z - eta * grad_centered

        if noise_scale > 0:
            z += noise_scale * rng.normal(size=n)

        z = z - z.max()
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            # Accept with probability based on improvement
            if sc < best_score + 0.002 and rng.random() < 0.3:
                a = a_new

        eta *= eta_decay
        if eta < 1e-6:
            eta = 1e-6

        if no_improve > restart_every:
            z = np.log(best_a + 1e-30)
            z += 0.03 * rng.normal(size=n)
            z = z - z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            eta = eta0
            no_improve = 0

    return best_a, best_score


def lbfgs_with_beta(a0, deadline, beta, max_iter=300):
    """
    L-BFGS-B optimization in logit space with given beta.
    """
    n = len(a0)
    z0 = np.log(a0 + 1e-30)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def fun(z):
        if time.monotonic() > deadline:
            return 1e10, np.zeros(n)
        return soft_max_obj_and_grad(z, n, beta)

    try:
        result = minimize(fun, z0, method='L-BFGS-B', jac=True,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                                   'gtol': 1e-10, 'ftol': 1e-14})
        z_final = result.x
        z_final = z_final - z_final.max()
        a_final = np.exp(z_final)
        a_final = a_final / a_final.sum()
        return a_final, compute_score(a_final)
    except Exception:
        return a0, compute_score(a0)


def annealed_lbfgs(a0, deadline, beta_list=None):
    """
    L-BFGS with annealed beta (from smooth to sharp).
    """
    if beta_list is None:
        beta_list = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        max_iter = min(300, max(50, int(remaining / 0.5)))
        a_new, sc = lbfgs_with_beta(a, deadline, beta, max_iter)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        elif sc < best_score + 0.001:
            a = a_new

    return best_a, best_score


def polynomial_profile(n, coeffs):
    """
    Generate a profile from polynomial coefficients.
    a[i] = max(eps, sum_j coeffs[j] * (i/(n-1))^j)
    """
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * coeffs[0]
    for j in range(1, len(coeffs)):
        a = a + coeffs[j] * t**j
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def poly_nelder_mead(n, deadline, rng, degree=5):
    """
    Optimize a polynomial profile using Nelder-Mead.
    """
    t0 = time.monotonic()

    def obj(params):
        if time.monotonic() > deadline:
            return 1e10
        a = polynomial_profile(n, params)
        return compute_score(a)

    # Initial guess: linear profile
    x0 = np.zeros(degree + 1)
    x0[0] = 1.0
    x0[1] = rng.uniform(-2.0, 2.0)
    x0[2] = rng.uniform(-1.0, 1.0)
    for j in range(3, degree + 1):
        x0[j] = rng.uniform(-0.5, 0.5)

    try:
        result = minimize(obj, x0, method='Nelder-Mead',
                          options={'maxiter': 2000, 'xatol': 1e-8,
                                   'fatol': 1e-10, 'adaptive': True})
        a_best = polynomial_profile(n, result.x)
        return a_best, compute_score(a_best)
    except Exception:
        a_best = polynomial_profile(n, x0)
        return a_best, compute_score(a_best)


def coordinate_polish(a, deadline, max_rounds=50):
    """
    Coordinate-wise optimization.
    """
    a = a.copy()
    n = len(a)
    best_score = compute_score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-18:
                continue

            # Try multiplicative perturbations
            candidates = [old_val * f for f in [0.85, 0.92, 0.97, 1.03, 1.08, 1.15]]
            best_cand = old_val
            best_sc = best_score

            for cv in candidates:
                a[i] = cv
                sc = compute_score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_cand = cv

            a[i] = best_cand
            if best_cand != old_val:
                improved = True
                best_score = best_sc

        if not improved:
            break

    return a, best_score


def diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]:
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cosine
    for k in [1, 2, 3, 4, 5]:
        for amp in [0.2, 0.4, 0.6]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Gaussian bumps
    for width in [0.2, 0.35, 0.5]:
        for offset in [0.25, 0.4, 0.5, 0.6]:
            a = np.exp(-((t - offset) / width) ** 2)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Power law (edge-concentrated)
    for p in [1.0, 2.0, 3.0, 4.0, 6.0]:
        for asym in [0.3, 0.5, 0.7]:
            a = t**(p * asym) + (1 - t)**(p * (1 - asym)) + 0.05
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Random smooth
    for _ in range(10):
        coeffs = rng.uniform(-0.3, 0.3, 6)
        a = np.ones(n)
        for i, c_val in enumerate(coeffs, 1):
            a = a + c_val * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def interpolate_to(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = len(a)
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 0 else a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    result = np.interp(x1, x0, a)
    s = result.sum()
    return result / s if s > 0 else result


def multi_topk_subgrad(a0, deadline, rng, eta0=0.03):
    """
    Subgradient method that cycles through different top_k values.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    top_k_schedule = [1, 1, 2, 3, 5, 1, 2, 4, 1, 3]
    idx = 0
    eta = eta0

    while time.monotonic() < deadline:
        top_k = top_k_schedule[idx % len(top_k_schedule)]
        idx += 1

        c = fftconvolve(a, a)
        top_k_actual = min(top_k, len(c))
        peak_indices = np.argpartition(c, -top_k_actual)[-top_k_actual:]

        grad = np.zeros(n)
        for k in peak_indices:
            grad += subgradient_at_peak(a, k)
        grad /= top_k_actual

        z = np.log(a + 1e-30)
        grad_centered = grad - grad.mean()
        z = z - eta * grad_centered
        z = z - z.max()
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            eta = max(0.001, eta * 1.001)
        else:
            eta = max(0.001, eta * 0.999)
            if rng.random() < 0.1:
                a = a_new

    return best_a, best_score


def peak_shaving(a, deadline, rng):
    """
    Targeted peak shaving: identify the main peak and try to reduce it
    by adjusting the sequence in a structured way.
    """
    a = a.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    for iteration in range(10000):
        if time.monotonic() > deadline:
            break

        c = fftconvolve(a, a)
        peak_k = int(np.argmax(c))
        peak_val = c[peak_k]

        # The peak is at position peak_k.
        # conv[peak_k] = sum_j a[j] * a[peak_k - j]
        # To reduce this, we want to reduce the product terms.
        # Strategy: find the pair (j, peak_k-j) with the largest product and reduce it.

        # Build the list of contributing pairs
        valid_j = []
        for j in range(n):
            kj = peak_k - j
            if 0 <= kj < n:
                valid_j.append(j)

        if not valid_j:
            break

        valid_j = np.array(valid_j)
        products = a[valid_j] * a[peak_k - valid_j]
        max_idx = int(np.argmax(products))
        j_star = valid_j[max_idx]
        k_star = peak_k - j_star

        # Try reducing the larger of a[j_star] and a[k_star]
        if a[j_star] >= a[k_star]:
            target = j_star
        else:
            target = k_star

        # Try a few reduction factors
        best_factor = 1.0
        best_sc = best_score
        for factor in [0.98, 0.95, 0.92, 0.88, 0.85]:
            a[target] = a[target] * factor
            sc = compute_score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_factor = factor

        if best_factor < 1.0:
            a[target] = a[target] * best_factor
            best_score = best_sc
            best_a = a.copy()
        else:
            # No improvement from shaving this peak; try small random perturbation
            z = np.log(a + 1e-30)
            z[int(target)] += 0.01 * rng.normal()
            z = z - z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search combining subgradient, L-BFGS, polynomial
    parameterization, and coordinate polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting hybrid search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Multi-n exploration =====
    print("\n=== Phase 1: Multi-n exploration ===")
    n_candidates = [512, 768, 1024, 1536, 2048, 2560, 3072, 3584]
    phase1_deadline = min(deadline, t_start + 250.0)

    for n in n_candidates:
        if time.monotonic() > phase1_deadline - 10:
            break

        if init_seq is not None:
            p0 = interpolate_to(init_seq, n)
        else:
            p0 = np.ones(n) / n

        sc0 = compute_score(p0)
        update(p0, sc0)

        # Quick subgradient
        dl = min(phase1_deadline, time.monotonic() + 25)
        r, sc_r = subgrad_optimize(p0, dl, rng, eta0=0.04, top_k=1)
        update(r, sc_r)

        # Quick L-BFGS
        if time.monotonic() < phase1_deadline - 5:
            r2, sc2 = lbfgs_with_beta(r, time.monotonic() + 15, beta=0.01, max_iter=100)
            update(r2, sc2)

        print(f"  n={n}: best_local={min(sc0, sc_r, sc2):.8f}, global={best_score:.8f}")

    # ===== Phase 2: Focused optimization =====
    print(f"\n=== Phase 2: Focused optimization ===")
    n_focus = len(best_seq) if best_seq is not None else 2048
    phase2_deadline = min(deadline, t_start + 1200.0)
    print(f"  Focus n={n_focus}")

    # Strategy 1: Annealed L-BFGS (long)
    if time.monotonic() < phase2_deadline - 120:
        print("  [1] Annealed L-BFGS (long)...")
        dl = min(phase2_deadline, time.monotonic() + 300)
        r1, sc1 = annealed_lbfgs(best_seq, dl,
                                  beta_list=[0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002])
        update(r1, sc1)
        print(f"      Score: {sc1:.10f}")

    # Strategy 2: Multi-topK subgradient
    if time.monotonic() < phase2_deadline - 100:
        print("  [2] Multi-topK subgradient...")
        dl = min(phase2_deadline, time.monotonic() + 200)
        r2, sc2 = multi_topk_subgrad(best_seq, dl, rng, eta0=0.02)
        update(r2, sc2)
        print(f"      Score: {sc2:.10f}")

    # Strategy 3: L-BFGS with very small beta
    if time.monotonic() < phase2_deadline - 80:
        print("  [3] L-BFGS (beta=0.0003)...")
        dl = min(phase2_deadline, time.monotonic() + 150)
        r3, sc3 = lbfgs_with_beta(best_seq, dl, beta=0.0003, max_iter=500)
        update(r3, sc3)
        print(f"      Score: {sc3:.10f}")

    # Strategy 4: Peak shaving
    if time.monotonic() < phase2_deadline - 50:
        print("  [4] Peak shaving...")
        dl = min(phase2_deadline, time.monotonic() + 120)
        r4, sc4 = peak_shaving(best_seq, dl, rng)
        update(r4, sc4)
        print(f"      Score: {sc4:.10f}")

    # Strategy 5: L-BFGS again (from best so far)
    if time.monotonic() < phase2_deadline - 30:
        print("  [5] L-BFGS refinement (beta=0.001)...")
        dl = min(phase2_deadline, time.monotonic() + 100)
        r5, sc5 = annealed_lbfgs(best_seq, dl,
                                  beta_list=[0.005, 0.003, 0.001, 0.0005])
        update(r5, sc5)
        print(f"      Score: {sc5:.10f}")

    # ===== Phase 3: Polynomial exploration =====
    print(f"\n=== Phase 3: Polynomial exploration ===")
    phase3_deadline = min(deadline, t_start + 1500.0)

    # Try polynomial profiles at the focus n
    if time.monotonic() < phase3_deadline - 30:
        for degree in [3, 4, 5, 6]:
            if time.monotonic() > phase3_deadline - 10:
                break
            print(f"  Nelder-Mead poly degree={degree}...")
            dl = min(phase3_deadline, time.monotonic() + 30)
            rp, scp = poly_nelder_mead(n_focus, dl, rng, degree=degree)
            update(rp, scp)
            print(f"      Score: {scp:.10f}")
            if scp < best_score:
                # Refine the best polynomial with L-BFGS on the full vector
                if time.monotonic() < phase3_deadline - 10:
                    dr = min(phase3_deadline, time.monotonic() + 40)
                    rr, scr = annealed_lbfgs(rp, dr, beta_list=[0.01, 0.005, 0.002, 0.001])
                    update(rr, scr)
                    print(f"      After L-BFGS: {scr:.10f}")

    # Try at other n values
    for n_try in [1536, 1792, 1920, 2048, 2304, 2560]:
        if n_try == n_focus or time.monotonic() > phase3_deadline - 15:
            continue
        if best_seq is not None:
            p_try = interpolate_to(best_seq, n_try)
        else:
            p_try = np.ones(n_try) / n_try
        sc_try = compute_score(p_try)
        update(p_try, sc_try)
        if sc_try < best_score + 0.005:
            dl_try = min(phase3_deadline, time.monotonic() + 20)
            rt, sct = subgrad_optimize(p_try, dl_try, rng, eta0=0.03, top_k=1)
            update(rt, sct)
            print(f"  n={n_try}: {sct:.8f}")

    # ===== Phase 4: Final polish =====
    print(f"\n=== Phase 4: Final polish ===")
    phase4_deadline = deadline

    # Coordinate polish
    if time.monotonic() < phase4_deadline - 30:
        print("  Coordinate polish...")
        dl_c = min(phase4_deadline, time.monotonic() + 180)
        rc, sc_c = coordinate_polish(best_seq, dl_c, max_rounds=30)
        update(rc, sc_c)
        print(f"    Score: {sc_c:.10f}")

    # Final L-BFGS with tiny beta
    if time.monotonic() < phase4_deadline - 20:
        print("  Final L-BFGS (beta=0.0001)...")
        dl_f = min(phase4_deadline, time.monotonic() + 80)
        rf, sc_f = lbfgs_with_beta(best_seq, dl_f, beta=0.0001, max_iter=400)
        update(rf, sc_f)
        print(f"    Score: {sc_f:.10f}")

    # Another round of peak shaving
    if time.monotonic() < phase4_deadline - 10:
        print("  Final peak shaving...")
        dl_ps = min(phase4_deadline, time.monotonic() + 60)
        rps, sc_ps = peak_shaving(best_seq, dl_ps, rng)
        update(rps, sc_ps)
        print(f"    Score: {sc_ps:.10f}")

    # Tiny subgradient with small eta
    if time.monotonic() < phase4_deadline - 5:
        print("  Tiny subgradient...")
        dl_ts = min(phase4_deadline, time.monotonic() + 40)
        rts, sc_ts = subgrad_optimize(best_seq, dl_ts, rng, eta0=0.005, top_k=1,
                                       eta_decay=0.9999, restart_every=500)
        update(rts, sc_ts)
        print(f"    Score: {sc_ts:.10f}")

    # Final coordinate polish
    if time.monotonic() < phase4_deadline - 2:
        rcp, sc_cp = coordinate_polish(best_seq, phase4_deadline, max_rounds=10)
        update(rcp, sc_cp)

    print(f"\n  FINAL best score: {best_score:.10f}")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.maximum(best_seq, 0.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        final_score = compute_score(final_seq)
        print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

        try:
            eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(eval_score):
                print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"Evaluation check: {e}")

        return [float(x) for x in final_seq]
    else:
        fallback = np.ones(512) / 512
        return [float(x) for x in fallback]
