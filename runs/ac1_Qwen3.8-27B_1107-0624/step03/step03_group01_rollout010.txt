```python
"""
Multi-phase optimizer combining LP-based peak reduction with L-BFGS continuation
and Adam polishing. The LP approach finds the optimal "peeling direction" by
solving max sum(g) s.t. (f*g)[k] <= M, g >= 0, which provides a globally
informed step direction unlike local gradient methods.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _make_gaussian_profile(n, width_frac):
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    width = width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_raised_cosine(n, power):
    t = np.linspace(0.0, 1.0, n)
    a = np.maximum(np.cos(np.pi * (t - 0.5)), 0.0) ** power
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, alpha, beta):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.exp(-alpha * u**2) + beta * np.exp(-2.0 * alpha * u**4)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _solve_peak_lp(f, M):
    n = len(f)
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 15})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def _lp_refine(f_init, deadline, max_iters=40):
    f = f_init.copy()
    n = len(f)
    best_f = f.copy()
    best_score = _score_norm_fn(f)
    print(f"  LP start: score={best_score:.10f}")

    no_improve_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(f, f)
        M = float(c.max())
        t_start_lp = time.monotonic()

        g = _solve_peak_lp(f, M)
        if g is None:
            print(f"  LP iter {it}: solver failed, stopping")
            break

        lp_time = time.monotonic() - t_start_lp

        # Try blending
        improved = False
        for t_blend in [0.5, 0.3, 0.1, 0.05, 0.02]:
            f_new = (1.0 - t_blend) * f + t_blend * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            sc = _score_norm_fn(f_new)
            if sc < best_score:
                best_score = sc
                best_f = f_new.copy()
                f = f_new.copy()
                improved = True
                break

        if improved:
            no_improve_count = 0
            if it % 5 == 0:
                print(f"  LP iter {it}: score={best_score:.10f} (lp_time={lp_time:.1f}s)")
        else:
            no_improve_count += 1
            if no_improve_count >= 8:
                print(f"  LP: no improvement for {no_improve_count} iters, stopping")
                break

    print(f"  LP done: best score={best_score:.10f}")
    return best_f, best_score


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_continuation(a_init, deadline, rng, max_wall_time=60.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break

        level_start = time.monotonic()
        level_deadline = level_start + per_level

        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 20, 'maxfun': 1500,
                         'maxiter': 300}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)) and a_new.min() > -1e-10:
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, rng, lr=0.03, beta_frac=0.005, n_max_iter=50000):
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[idx] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 500 == 0:
            beta = max(1e-12, beta * 0.8)

        if t % 5000 == 0 and best_score < 1.51:
            lr = max(0.002, lr * 0.9)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def _topk_gradient_step(f, k_top=20, step_size=0.01):
    """Direct subgradient step targeting the top-k correlation peaks."""
    n = len(f)
    c = _full_convolve(f, f)
    max_c = float(c.max())
    # Use top-k elements as "active" set
    threshold_idx = np.argsort(c)[-k_top:]
    w = np.zeros_like(c)
    w[threshold_idx] = 1.0 / k_top

    a_rev = f[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

    # Gradient in log-space
    grad_z = 4.0 * n * f * (h - W)

    # Step in log-space
    z = np.log(np.maximum(f, 1e-15))
    z = z - z.mean()
    z = z - step_size * grad_z
    z = np.clip(z, -50, 50)
    z = z - z.mean()
    a_new = _softmax_fn(z)
    return a_new

                    no_improve = 0

        if no_improve > 500:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP peak reduction -> L-BFGS continuation -> Adam polish.
    Uses LP to find globally-informed step directions, combined with
    continuation-based gradient methods for fine optimization.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Phase 1: Initialization (0-30s) ---
    phase1_end = min(deadline, t_start + 30.0)
    print("=== Phase 1: Initialization ===")

    candidates = []

    # Try parent sequence
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        candidates.append((sc_parent, parent.copy(), len(parent), "parent"))
        update_best(parent, sc_parent, "parent")
        print(f"  Parent: n={len(parent)}, score={sc_parent:.10f}")

    # Analytical profiles
    n_default = 2048
    if parent is not None:
        n_default = len(parent)

    profiles = []
    for width in [0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
        profiles.append(_make_gaussian_profile(n_default, width))
    for power in [1.0, 1.5, 2.0, 3.0, 4.0]:
        profiles.append(_make_raised_cosine(n_default, power))
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(_make_linear_profile(n_default, slope))
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        for beta_p in [0.0, 0.1, 0.3, 0.5]:
            profiles.append(_make_bell_profile(n_default, alpha, beta_p))

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase1_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_default, f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    # Add random profiles
    for _ in range(10):
        if time.monotonic() > phase1_end:
            break
        z = rng.normal(size=n_default)
        z = z - z.mean()
        p = _softmax_fn(z)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_default, "rand"))
        update_best(p, sc, "rand")

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 best: {best_score:.10f}")

    # --- Phase 2: LP-based refinement (30-350s) ---
    phase2_end = min(deadline, t_start + 350.0)
    print("=== Phase 2: LP-based refinement ===")

    if time.monotonic() < phase2_end and best_seq is not None:
        # Use the best candidate for LP
        lp_start = best_seq.copy()
        lp_deadline = min(phase2_end - 5, time.monotonic() + 200.0)
        lp_result, lp_score = _lp_refine(lp_start, lp_deadline, max_iters=35)
        update_best(lp_result, lp_score, "lp_refine")

        # Also try LP on second-best candidate if different
        if len(candidates) > 1:
            sc2, p2, n2, label2 = candidates[1]
            if p2 is not best_seq or not np.allclose(p2, best_seq):
                if time.monotonic() < phase2_end - 30:
                    lp2_deadline = min(phase2_end - 10, time.monotonic() + 100.0)
                    lp2_result, lp2_score = _lp_refine(p2.copy(), lp2_deadline, max_iters=20)
                    update_best(lp2_result, lp2_score, f"lp_{label2}")

    print(f"  Phase 2 best: {best_score:.10f}")

    # --- Phase 3: Top-k gradient refinement (350-500s) ---
    phase3_end = min(deadline, t_start + 500.0)
    print("=== Phase 3: Top-k gradient refinement ===")

    if time.monotonic() < phase3_end and best_seq is not None:
        tk_deadline = min(phase3_end - 5, time.monotonic() + 120.0)
        tk_result, tk_score = _topk_refine(best_seq.copy(), tk_deadline, max_iters=3000)
        update_best(tk_result, tk_score, "topk_refine")

    print(f"  Phase 3 best: {best_score:.10f}")

    # --- Phase 4: L-BFGS continuation (500-1100s) ---
    phase4_end = min(deadline, t_start + 1100.0)
    print("=== Phase 4: L-BFGS continuation ===")

    if time.monotonic() < phase4_end and best_seq is not None:
        # Main L-BFGS run
        lbfgs_deadline = min(phase4_end - 60, time.monotonic() + 300.0)
        lbfgs_result, lbfgs_score = _run_lbfgs_continuation(
            best_seq.copy(), lbfgs_deadline, rng, max_wall_time=280.0)
        update_best(lbfgs_result, lbfgs_score, "lbfgs_main")

        # Multi-restart with noise
        if time.monotonic() < phase4_end - 30:
            n_best = len(best_seq)
            for i in range(4):
                if time.monotonic() > phase4_end - 20:
                    break
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                noise_scale = 0.05 * (i + 1)
                z = z + noise_scale * rng.normal(size=n_best)
                z = np.clip(z, -40, 40)
                z = z - z.mean()
                p_start = _softmax_fn(z)

                per_dl = min(phase4_end - 15, time.monotonic() + 80.0)
                p_best, sc_best = _run_lbfgs_continuation(p_start, per_dl, rng, max_wall_time=70.0)
                update_best(p_best, sc_best, f"restart{i}")

    print(f"  Phase 4 best: {best_score:.10f}")

    # --- Phase 5: Adam fine-tuning (1100-1450s) ---
    phase5_end = min(deadline, t_start + 1450.0)
    print("=== Phase 5: Adam fine-tuning ===")

    if time.monotonic() < phase5_end and best_seq is not None:
        adam_deadline = min(phase5_end - 10, time.monotonic() + 250.0)
        adam_result, adam_score = _adam_polish(
            best_seq.copy(), adam_deadline, rng, lr=0.02, beta_frac=0.003, n_max_iter=50000)
        update_best(adam_result, adam_score, "adam_polish")

        # Second round with different params
        if time.monotonic() < phase5_end - 20:
            adam2_deadline = min(phase5_end - 5, time.monotonic() + 100.0)
            adam2_result, adam2_score = _adam_polish(
                best_seq.copy(), adam2_deadline, rng, lr=0.01, beta_frac=0.001, n_max_iter=30000)
            update_best(adam2_result, adam2_score, "adam_polish2")

    print(f"  Phase 5 best: {best_score:.10f}")

    # --- Phase 6: Multi-length exploration (1450-1700s) ---
    phase6_end = min(deadline, t_start + 1700.0)
    print("=== Phase 6: Multi-length exploration ===")

    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        target_lengths = [1024, 2048, 4096, 8192]
        target_lengths = [tl for tl in target_lengths if tl != n_curr and tl <= 8192]

        for tgt_n in target_lengths:
            if time.monotonic() > phase6_end - 30:
                break
            p_interp = _interp_profile(best_seq, tgt_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tgt_n}")

            # Quick L-BFGS on interpolated
            if time.monotonic() < phase6_end - 60:
                per_dl = min(phase6_end - 40, time.monotonic() + 80.0)
                p_best, sc_best = _run_lbfgs_continuation(
                    p_interp, per_dl, rng, max_wall_time=60.0)
                update_best(p_best, sc_best, f"lbfgs_n{tgt_n}")

            # Quick LP on interpolated
            if time.monotonic() < phase6_end - 40:
                lp_dl = min(phase6_end - 20, time.monotonic() + 60.0)
                p_lp, sc_lp = _lp_refine(p_interp, lp_dl, max_iters=10)
                update_best(p_lp, sc_lp, f"lp_n{tgt_n}")

    print(f"  Phase 6 best: {best_score:.10f}")

    # --- Phase 7: Final polish (1700-1800s) ---
    phase7_end = deadline
    print("=== Phase 7: Final polish ===")

    if time.monotonic() < phase7_end and best_seq is not None:
        # Final L-BFGS
        if time.monotonic() < phase7_end - 20:
            per_dl = min(phase7_end - 15, time.monotonic() + 50.0)
            p_best, sc_best = _run_lbfgs_continuation(
                best_seq.copy(), per_dl, rng, max_wall_time=40.0)
            update_best(p_best, sc_best, "final_lbfgs")

        # Final Adam
        if time.monotonic() < phase7_end - 5:
            per_dl2 = min(phase7_end - 3, time.monotonic() + 30.0)
            p_best2, sc_best2 = _adam_polish(
                best_seq.copy(), per_dl2, rng, lr=0.005, beta_frac=0.001, n_max_iter=10000)
            update_best(p_best2, sc_best2, "final_adam")

    # --- Final validation ---
    print(f"=== FINAL: score={best_score:.10f} ===")

    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_gaussian_profile(2048, 0.25)

        sc_final = _score_fn(final_seq)
        print(f"  Final score: {sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluate_sequence
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"  Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"  evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_gaussian_profile(2048, 0.25)
    print(f"  FALLBACK: score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
```
