"""
Spectral-flattening optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses frequency-domain insights (PAPR reduction analogy), adaptive subgradient
with Nesterov momentum, LP refinement, and diverse spectral initializations.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M, top_k=100):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, max_evals=30):
    """Golden section search for optimal blend."""
    n = len(a)
    best_score = _score_norm(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c * (g - a), 0))
    fd = _score_norm(np.maximum(a + d * (g - a), 0))
    for _ in range(max_evals):
        if fc < fd:
            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c * (g - a), 0))
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d * (g - a), 0))
    return (lo + hi) / 2.0

def _lse_optimize(a, deadline, max_iter=100000, T_start=0.03, T_end=1e-6, lr=0.08):

    lr_cur = lr

            break

            w[int(np.argmax(c))] = 1.0

        else:
            w = ex / s

        fw = np.fft.rfft(w, N_fft)

            z = np.log(np.maximum(a, 1e-14))

        state['v'] = v
        
        c_new = _fft_convolve(a_new)

            best_a = a_new.copy()
            no_imp = 0

def _spectral_init(n, rng):
    """Initialize from flat spectrum with random phase, projected to non-negative."""
    N = _next_pow2(2 * n - 1)
    # Create a flat-ish magnitude spectrum
    k = np.arange(N // 2 + 1)
    # Slightly shape: boost low frequencies a bit (to get smoother signals)
    mag = 1.0 / (1.0 + 0.1 * (k / (N // 2)) ** 2)
    mag[0] = 0  # Remove DC component to allow zero-mean
    phase = rng.uniform(0, 2 * np.pi, size=n // 2 + 1)
    phase[0] = 0
    
    spectrum = mag * np.exp(1j * phase)
    # Make it Hermitian
    if N % 2 == 0:
        full_spec = np.zeros(N, dtype=complex)
        full_spec[:n // 2 + 1] = spectrum
        full_spec[n // 2 + 1:] = np.conj(spectrum[n // 2 - 1:0:-1])
    else:
        full_spec = np.zeros(N, dtype=complex)
        full_spec[:n // 2 + 1] = spectrum
        full_spec[n // 2 + 1:] = np.conj(spectrum[n // 2:0:-1])
    
    signal = np.fft.ifft(full_spec).real[:n]
    signal = np.maximum(signal, 0)
    s = signal.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return signal / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_with_edges(n, edge_weight=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_periodic(n, periods=4):
    """Create a periodic-ish profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.3 * np.cos(2.0 * np.pi * periods * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, center_sep=0.4):
    """Bimodal profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.3) / 0.1) ** 2) + np.exp(-0.5 * ((t - 0.7) / 0.1) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _coordinate_descent(a, deadline, max_iter=50000):
    """Coordinate descent: optimize one element at a time."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    
    # Precompute: for each coordinate j, the change in score when we change a_j
    # c_k = sum_j a_j * a_{j+k}
    # If we change a_j by delta, c_k changes by:
    #   delta * a_{k-j} (from the j-th term) + a_j * delta_contribution
    # Actually: c_k = sum_i a_i * a_{i+k}
    # dc_k/da_j = a_{k-j} + a_{k+j} (where indices are bounded)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Pick a random coordinate
        j = np.random.default_rng(it).integers(n)
        
        # Binary search for optimal a[j] given others fixed
        a_old = a[j]
        a_floor = 0.0
        a_ceil = 1.0  # since sum=1, max any element can be is 1
        
        # The score as a function of a[j] = t:
        # sum(a) = s - a_old + t, so we need to re-normalize
        # Actually, let's work with the unnormalized problem
        # Score = 2n * max_k c_k / sum(a)^2
        # With sum fixed = 1 (normalized), if we change a[j]:
        # new_sum = 1 - a_old + t (before renormalization)
        # After normalization: a'[j] = t / new_sum, a'[i] = a[i] / new_sum
        # This is complex. Let's just try a few values.
        
        # Try: current value, slightly up, slightly down
        candidates = [a_old, max(0, a_old * 1.1), max(0, a_old * 0.9), 
                      max(0, a_old + 0.01/n), max(0, a_old - 0.01/n)]
        
        best_local = a_old
        best_local_score = best_score
        
        for t in candidates:
            a[j] = t
            s_new = a.sum()
            if s_new < 1e-10:
                a[j] = a_old
                continue
            a_norm = a / s_new
            c = _fft_convolve(a_norm)
            sc = 2.0 * n * c.max()
            if sc < best_local_score:
                best_local_score = sc
                best_local = t
        
        a[j] = best_local
        s_new = a.sum()
        if s_new > 1e-10:
            a = a / s_new
        
        c = _fft_convolve(a)
        sc = 2.0 * n * c.max()
        
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 1000:
                break
    
    return best_a

Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines spectral initialization (PAPR-reduction analogy), adaptive subgradient
with Nesterov momentum, LSE gradient with Adam, LP refinement with line search,
and diverse candidate generation. Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _lp_spread(a, M, top_k=120, window=4):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-window, window + 1):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, max_evals=35):
    n = len(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c * (g - a), 0))
    fd = _score_norm(np.maximum(a + d * (g - a), 0))
    for _ in range(max_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c * (g - a), 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d * (g - a), 0))
    t = (lo + hi) / 2.0
    return t


def _subgrad_optimize(a, deadline, max_iter=80000, lr=0.04, momentum=0.92, top_k=1):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    N_fft = _next_pow2(2 * n - 1)
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())

        if top_k <= 1:
            max_idx = np.where(c >= m_val - 1e-10)[0]
            if len(max_idx) == 0:
                max_idx = np.array([np.argmax(c)])
            w = np.zeros_like(c)
            w[max_idx] = 1.0 / len(max_idx)
        else:
            sorted_idx = np.argsort(c)[::-1][:top_k]
            vals = c[sorted_idx]
            if vals.sum() < 1e-30:
                w = np.zeros_like(c)
                w[sorted_idx[0]] = 1.0
            else:
                weights = vals / vals.sum()
                w = np.zeros_like(c)
                for i, idx in enumerate(sorted_idx):
                    w[idx] = weights[i]

        fw_full = np.fft.fft(w, N_fft)
        fa_full = np.fft.fft(a, N_fft)
        g1 = np.fft.ifft(fw_full * fa_full).real[:n]
        g2 = np.fft.ifft(np.conj(fw_full) * fa_full).real[:n]
        grad = g1 + g2

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr_cur * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr_cur = max(0.001, lr_cur * 0.95)
            if no_imp > max(4000, 6 * n):
                rng_r = np.random.default_rng(777 + it % 10000)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-14)
                a = a / a.sum()
                v = np.zeros(n)
                lr_cur = base_lr
                no_imp = 0

        a = a_new

    return best_a


def _lse_optimize(a, deadline, max_iter=80000, T_start=0.04, T_end=1e-6, lr=0.08):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    N_fft = _next_pow2(2 * n - 1)

    if max_iter > 1:
        T_ratio = (T_end / T_start) ** (1.0 / (max_iter - 1))
    else:
        T_ratio = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        x = np.clip((c - m_val) / T, -50, 50)
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            s = 1.0
        else:
            w = ex / s

        fa = np.fft.rfft(a, N_fft)
        fw = np.fft.rfft(w, N_fft)
        fa_rev = np.fft.rfft(a[::-1], N_fft)
        h_full = np.fft.irfft(fw * fa_rev, N_fft)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-14))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue

        t_count = state['t'] + 1
        b1, b2 = 0.9, 0.999
        m = b1 * state['m'] + (1.0 - b1) * gz
        v_arr = b2 * state['v'] + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_count)
        vh = v_arr / (1.0 - b2 ** t_count)

        z = z - lr_cur * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a_new = e / e.sum()

        state['t'] = t_count
        state['m'] = m
        state['v'] = v_arr

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 600 == 0:
                lr_cur = max(0.01, lr_cur * 0.9)
            if no_imp > max(5000, 8 * n):
                rng_r = np.random.default_rng(99999 + it % 20000)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-14)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr_cur = base_lr
                no_imp = 0

        a = a_new
        T = max(T * T_ratio, T_end)

    return best_a


def _lp_refine(a, deadline, max_iter=5000, inflate=1.0004, blend=0.15, use_ls=True, top_k=120):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp, top_k=top_k)

        if g is None:
            inflate_cur = min(1.015, inflate_cur * 1.4)
            no_imp += 5
            if no_imp > 300:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 400:
                break
            continue

        g = g / sum_g

        if use_ls:
            t_opt = _line_search(a, g, max_evals=30)
            if t_opt < 0.005:
                t_opt = blend_cur * 0.5
        else:
            t_opt = blend_cur

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend_cur = max(0.02, blend_cur * 0.88)
                inflate_cur = max(1.0001, inflate_cur * 0.93)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _spectral_init(n, rng):
    N = _next_pow2(2 * n - 1)
    half = N // 2 + 1
    k = np.arange(half)
    freq_scale = k / max(half - 1, 1)
    mag = 1.0 / (1.0 + 0.3 * freq_scale ** 4)
    mag[0] = 0.0
    phase = rng.uniform(0, 2 * np.pi, size=half)
    phase[0] = 0.0
    spectrum = mag * np.exp(1j * phase)
    full_spec = np.zeros(N, dtype=complex)
    full_spec[:half] = spectrum
    if N % 2 == 0:
        full_spec[half:] = np.conj(spectrum[half - 1:0:-1])
    else:
        full_spec[half:] = np.conj(spectrum[half - 1:0:-1])
    signal = np.fft.ifft(full_spec).real[:n]
    signal = np.maximum(signal, 0)
    s = signal.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return signal / s


def _make_gaussian(n, sigma_frac=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, w):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + w * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_periodic(n, periods):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.25 * np.cos(2.0 * np.pi * periods * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.3) / 0.12) ** 2) + np.exp(-0.5 * ((t - 0.7) / 0.12) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_smooth_bump(n, width=0.6):
    t = np.linspace(0.0, 1.0, n)
    x = (t - 0.5) / (width * 0.5)
    mask = np.abs(x) <= 1.0
    a = np.zeros(n)
    a[mask] = 0.5 * (1.0 + np.cos(np.pi * x[mask]))
    a = np.maximum(a, 1e-12)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _iterative_refine(a, deadline, max_rounds=5):
    """Alternate between subgradient and LP for faster convergence."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 30:
            break

        sg_time = min(40.0, (deadline - time.monotonic() - 30) * 0.4)
        if sg_time < 10:
            break
        a = _subgrad_optimize(a, time.monotonic() + sg_time,
                              max_iter=30000, lr=0.03, momentum=0.92, top_k=3)
        sc = _score_norm(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

        lp_time = min(50.0, (deadline - time.monotonic() - 30) * 0.5)
        if lp_time < 10:
            break
        a = _lp_refine(a, time.monotonic() + lp_time,
                       max_iter=2000, inflate=1.0003, blend=0.12,
                       use_ls=True, top_k=100)
        sc = _score_norm(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: spectral init, LSE gradient, subgradient with
    momentum, LP with line search, iterative refinement. Explores n=1024,1536,2048.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: n={parent.size}, score={sc:.10f}")
    except Exception:
        pass

    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [1024, 1536, 2048]:
        if ln not in lengths:
            lengths.append(ln)
    if len(lengths) > 3:
        lengths = lengths[:3]

    print(f"Lengths to try: {lengths}")
    print(f"Budget: {budget_s}s, deadline at t={deadline - t_start:.1f}s")

    # Phase 1: Diverse initialization + LSE
    print("\n=== Phase 1: Initialization + LSE ===")
    results = {}

    for n_t in lengths:
        if time.monotonic() > deadline - 300:
            break

        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            for _ in range(2):
                pp = _interpolate_profile(parent, n_t) * (1.0 + 0.05 * rng.normal(size=n_t))
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    cands.append(pp / pp.sum())

        for _ in range(4):
            cands.append(_spectral_init(n_t, rng))
        cands.append(_make_gaussian(n_t, 0.3))
        cands.append(_make_gaussian(n_t, 0.5))
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_flat_edges(n_t, 0.5))
        cands.append(_make_flat_edges(n_t, 0.8))
        cands.append(_make_periodic(n_t, 4))
        cands.append(_make_periodic(n_t, 6))
        cands.append(_make_bimodal(n_t))
        cands.append(_make_smooth_bump(n_t, 0.5))
        cands.append(_make_smooth_bump(n_t, 0.7))

        best_here = None
        best_here_score = np.inf
        per_cand_time = max(8.0, min(40.0, 240.0 / len(cands)))

        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 300:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            lse_end = time.monotonic() + per_cand_time
            a_lse = _lse_optimize(a0, lse_end, max_iter=25000,
                                  T_start=0.05, T_end=1e-5, lr=0.08)
            sc_lse = _score(a_lse)

            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse

            if (ci + 1) % 4 == 0:
                print(f"  n={n_t}, {ci+1}/{len(cands)}: best_here={best_here_score:.10f}")

        if best_here is not None:
            results[n_t] = best_here
            print(f"  n={n_t} Phase1 best: {best_here_score:.10f}")

    # Phase 2: Subgradient refinement
    print("\n=== Phase 2: Subgradient ===")
    for n_t in lengths:
        if n_t not in results:
            continue
        if time.monotonic() > deadline - 180:
            break

        a_in = results[n_t]
        sc_in = _score(a_in)
        sg_end = time.monotonic() + min(80.0, deadline - time.monotonic() - 180)
        a_sg = _subgrad_optimize(a_in, sg_end, max_iter=50000,
                                 lr=0.03, momentum=0.93, top_k=3)
        sc_sg = _score(a_sg)
        if np.isfinite(sc_sg) and sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
        if sc_sg < sc_in:
            results[n_t] = a_sg
        print(f"  n={n_t}: {sc_in:.10f} -> {sc_sg:.10f}")

    # Phase 3: LP refinement
    print("\n=== Phase 3: LP ===")
    for n_t in lengths:
        if n_t not in results:
            continue
        if time.monotonic() > deadline - 100:
            break

        a_in = results[n_t]
        sc_in = _score(a_in)
        lp_end = time.monotonic() + min(120.0, deadline - time.monotonic() - 100)
        a_lp = _lp_refine(a_in, lp_end, max_iter=4000,
                          inflate=1.0003, blend=0.12, use_ls=True, top_k=120)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        if sc_lp < sc_in:
            results[n_t] = a_lp
        print(f"  n={n_t}: {sc_in:.10f} -> {sc_lp:.10f}")

    # Phase 4: Iterative (subgrad + LP) on best
    print("\n=== Phase 4: Iterative refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_iter = _iterative_refine(best_seq, time.monotonic() + min(120.0, deadline - time.monotonic() - 40),
                                   max_rounds=4)
        sc_iter = _score(a_iter)
        if np.isfinite(sc_iter) and sc_iter < best_score:
            best_score = sc_iter
            best_seq = a_iter.copy()
        print(f"  Iterative: {sc_iter:.10f}")

    # Phase 5: Final LP polish
    if best_seq is not None and time.monotonic() < deadline - 25:
        lp_end = time.monotonic() + min(50.0, deadline - time.monotonic() - 20)
        a_fin = _lp_refine(best_seq, lp_end, max_iter=2000,
                           inflate=1.0002, blend=0.08, use_ls=True, top_k=150)
        sc_fin = _score(a_fin)
        if np.isfinite(sc_fin) and sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
        print(f"  Final LP: {sc_fin:.10f}")

    # Phase 6: Final LSE micro-polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        lse_end = time.monotonic() + min(30.0, deadline - time.monotonic() - 10)
        a_last = _lse_optimize(best_seq, lse_end, max_iter=20000,
                               T_start=0.01, T_end=1e-8, lr=0.03)
        sc_last = _score(a_last)
        if np.isfinite(sc_last) and sc_last < best_score:
            best_score = sc_last
            best_seq = a_last.copy()
        print(f"  Final LSE: {sc_last:.10f}")

    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.asarray(best_seq, dtype=np.float64)
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
