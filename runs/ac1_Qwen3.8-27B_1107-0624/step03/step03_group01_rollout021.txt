    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Many more beta levels for finer continuation
    beta_fracs = [0.15, 0.10, 0.07, 0.05, 0.035, 0.025, 0.018,
                  0.013, 0.009, 0.006, 0.004, 0.0025, 0.0015, 0.0008, 0.0004]
    betas = [b * max_c0 for b in beta_fracs]

    per_level = max_wall_time / len(betas)
    current_z = z.copy()

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        elapsed_in_level = 0.0

        try:
            start_t = time.monotonic()
            result = minimize(
                _lbfgs_obj, current_z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_grad,
                options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 5000,
                         'iprint': -1}
            )
            elapsed_in_level = time.monotonic() - start_t
            if result.x is not None and np.all(np.isfinite(result.x)):
                current_z = result.x.copy()
                a_new = _softmax_fn(current_z)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
        except Exception:
            pass

        # After each level, re-center z to prevent drift
        current_z = current_z - current_z.mean()
        a_curr = _softmax_fn(current_z)
        sc_curr = _score_norm_fn(a_curr)
        if np.isfinite(sc_curr) and sc_curr < best_score:
            best_score = sc_curr
            best_a = a_curr.copy()

    return best_a, best_score


def _fischer_iter(a_init, deadline, n_max_iter=5000):
    """Fischer-like subgradient projection method."""
    n = len(a_init)
    a = a_init.copy()
    S = float(a.sum())
    if S < 1e-15:
        return a_init, _score_norm_fn(a_init)
    a = a / S

    best_a = a.copy()
    best_score = _score_norm_fn(a)

    # Adaptive step size
    step = 0.5
    patience = 0
    max_patience = 200

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())
        k_star = int(np.argmax(c))

        # Subgradient: d(max(c))/da_j = 2*a[k_star - j] for valid j
        sg = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            sg[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Take step and project
        a_new = a - step * sg / sg_norm
        a_new = _project_simplex(a_new, 1.0)

        sc_new = _score_norm_fn(a_new)
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            a = a_new  # still move, but track patience
            patience += 1
            if patience >= max_patience:
                step *= 0.7
                patience = 0
                if step < 1e-6:
                    break

        # Slowly increase step if improving
        if patience == 0:
            step = min(step * 1.01, 2.0)

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.05, beta_frac=0.01, n_max_iter=50000):
    """Adam optimizer in log-space with adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-18))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = max(1e-12, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        G = 4.0 * n * h
        Ga = float(np.dot(G, a))
        grad = a * (G - Ga)

        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptively reduce beta
        if t % 100 == 0:
            c_now = _full_convolve(a, a)
            max_c_now = float(c_now.max())
            beta = max(1e-12, beta * 0.8)
            if best_score < 1.51:
                lr = max(0.003, lr * 0.97)

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * (t - 0.5))
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_psp_profile(n, f1, f2, amp):
    """Polynomial / sinusoidal profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * f1 * t) + 0.5 * amp * np.cos(2.0 * np.pi * f2 * t + 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _nesterov_optimize(a_init, deadline, lr=0.1, momentum=0.9, n_max_iter=30000):
    """Nesterov-accelerated gradient descent in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    v = np.zeros(n)

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    beta = max(1e-12, 0.01 * c.max())

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        G = 4.0 * n * h
        Ga = float(np.dot(G, a))
        grad = a * (G - Ga)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        # Nesterov look-ahead
        z_look = z + momentum * v
        a_look = _softmax_fn(z_look)
        c_look = _full_convolve(a_look, a_look)
        cm_look = float(c_look.max())
        c_look_shifted = (c_look - cm_look) / beta
        e_look = np.exp(c_look_shifted)
        s_look = float(e_look.sum())
        if s_look < 1e-300 or not np.isfinite(s_look):
            w_look = np.zeros(len(c_look))
            w_look[int(np.argmax(c_look))] = 1.0
        else:
            w_look = e_look / s_look
        a_look_rev = a_look[::-1]
        w_conv_look = _full_convolve(w_look, a_look_rev)
        h_look = w_conv_look[n - 1:2 * n - 1]
        G_look = 4.0 * n * h_look
        Ga_look = float(np.dot(G_look, a_look))
        grad_look = a_look * (G_look - Ga_look)
        if not np.all(np.isfinite(grad_look)):
            grad_look = grad

        v = momentum * v - lr * grad_look
        z = z + v
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.85)
            lr = max(0.01, lr * 0.98)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS continuation + Fischer subgradient + Adam polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
        _update(parent, _score_norm_fn(parent), "parent")
    else:
        n = 2048
        parent = _make_linear_profile(n, 2.0)
        print(f"No parent, using linear profile n={n}")
        _update(parent, _score_norm_fn(parent), "init")

    # --- Phase 0: Evaluate diverse initial candidates (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []
    n_best = len(best_seq) if best_seq is not None else 2048

    profiles_to_try = [
        _make_linear_profile(n_best, 1.0),
        _make_linear_profile(n_best, 1.5),
        _make_linear_profile(n_best, 2.0),
        _make_linear_profile(n_best, 2.5),
        _make_linear_profile(n_best, 3.0),
        _make_exp_profile(n_best, 2.0),
        _make_exp_profile(n_best, 3.0),
        _make_exp_profile(n_best, 4.0),
        _make_quadratic_profile(n_best, 1.5, 0.3),
        _make_quadratic_profile(n_best, 2.0, -0.2),
        _make_psp_profile(n_best, 1.0, 2.0, 0.3),
        _make_psp_profile(n_best, 2.0, 5.0, 0.2),
    ]

    for i, p in enumerate(profiles_to_try):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"prof{i}"))
        _update(p, sc, f"prof{i}")

    # Also try random perturbations of parent
    for i in range(3):
        if time.monotonic() > phase0_end:
            break
        z = np.log(np.maximum(parent, 1e-15))
        z = z - z.mean()
        z = z + 0.3 * rng.normal(size=n_best)
        z = z - z.mean()
        p = _softmax_fn(z)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"rand{i}"))
        _update(p, sc, f"rand{i}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: L-BFGS continuation on top candidates (700s) ---
    phase1_end = min(deadline, t_start + 730.0)
    count = 0
    n_candidates = min(4, len(candidates))
    per_cand = (phase1_end - time.monotonic()) / max(1, n_candidates)

    for ci in range(n_candidates):
        if time.monotonic() > phase1_end:
            break
        if per_cand < 30:
            break
        sc_c, p_c, label_c = candidates[ci]
        per_dl = min(phase1_end - 5, time.monotonic() + per_cand)
        p_best, sc_best = _run_lbfgs_continuation(p_c, per_dl, max_wall_time=per_cand)
        _update(p_best, sc_best, f"lbfgs_{label_c}")
        count += 1
        rem = phase1_end - time.monotonic()
        n_remaining = n_candidates - count
        if n_remaining > 0:
            per_cand = rem / n_remaining

    print(f"Phase 1 done: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # --- Phase 2: Fischer subgradient method (250s) ---
    phase2_end = min(deadline, t_start + 980.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        per_dl = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()))
        p_best, sc_best = _fischer_iter(best_seq, per_dl, n_max_iter=8000)
        _update(p_best, sc_best, "fischer")

        # Try Fischer from a perturbed start
        if time.monotonic() < phase2_end - 30:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean() + 0.1 * rng.normal(size=len(best_seq))
            z = z - z.mean()
            p_start = _softmax_fn(z)
            per_dl2 = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
            p_best2, sc_best2 = _fischer_iter(p_start, per_dl2, n_max_iter=5000)
            _update(p_best2, sc_best2, "fischer_perturbed")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam with adaptive beta (250s) ---
    phase3_end = min(deadline, t_start + 1230.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 5, time.monotonic() + (phase3_end - time.monotonic()))
        p_best, sc_best = _adam_optimize(best_seq, per_dl, lr=0.04, beta_frac=0.008, n_max_iter=40000)
        _update(p_best, sc_best, "adam")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Nesterov accelerated (150s) ---
    phase4_end = min(deadline, t_start + 1380.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = min(phase4_end - 5, time.monotonic() + (phase4_end - time.monotonic()))
        p_best, sc_best = _nesterov_optimize(best_seq, per_dl, lr=0.05, momentum=0.85, n_max_iter=20000)
        _update(p_best, sc_best, "nesterov")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart L-BFGS on best (120s) ---
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_restarts = 3
        per_restart = (phase5_end - time.monotonic()) / max(1, n_restarts)
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            if per_restart < 20:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.05 + 0.03 * i
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            per_dl = min(phase5_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _run_lbfgs_continuation(p_start, per_dl, max_wall_time=per_restart)
            _update(p_best, sc_best, f"restart{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different length if beneficial (100s) ---
    phase6_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        # Try interpolating to a different length
        for target_n in [n_curr * 2, n_curr // 2] if n_curr > 256 else [n_curr * 2]:
            if time.monotonic() > phase6_end - 30:
                break
            if target_n < 128 or target_n > 8192:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            _update(p_interp, sc_interp, f"interp_n{target_n}")
            if sc_interp < best_score:
                per_dl = min(phase6_end - 10, time.monotonic() + 60.0)
                p_best, sc_best = _run_lbfgs_continuation(p_interp, per_dl, max_wall_time=55.0)
                _update(p_best, sc_best, f"lbfgs_n{target_n}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final polish (remaining time) ---
    phase7_end = deadline
    if time.monotonic() < phase7_end and best_seq is not None:
        rem = phase7_end - time.monotonic()
        if rem > 30:
            per_dl = min(phase7_end - 3, time.monotonic() + rem * 0.5)
            p_best, sc_best = _run_lbfgs_continuation(best_seq, per_dl, max_wall_time=rem * 0.5)
            _update(p_best, sc_best, "final_lbfgs")

        if time.monotonic() < phase7_end - 10:
            rem2 = phase7_end - time.monotonic()
            per_dl2 = min(phase7_end - 3, time.monotonic() + rem2 * 0.7)
            p_best2, sc_best2 = _adam_optimize(best_seq, per_dl2, lr=0.01, beta_frac=0.003, n_max_iter=10000)
            _update(p_best2, sc_best2, "final_adam")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(2048, 2.0)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence check: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(2048, 2.0)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
