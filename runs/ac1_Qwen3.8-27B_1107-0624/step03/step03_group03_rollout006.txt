1. Add a "flatness" regularization to the LSE objective to provide gradient signal even when the max doesn't change
2. Use cosine annealing with warm restarts for better exploration
3. Improve the LP spreader with better constraint selection
4. Add a "water-filling" equalization step

"""
Improved multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements: flatness-regularized LSE, cosine annealing with restarts,
improved LP spreading, water-filling equalization, and multi-length search.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score for sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix_full(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _make_conv_matrix_reduced(a, M, top_k=80):
    """Reduced constraint matrix using near-active constraints."""
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        for j in range(jlo, jhi):
            A[row, j] = a[j]
    return A, b


def _lp_spread(a, M, use_reduced=True, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    if use_reduced and n > 32:
        A, b = _make_conv_matrix_reduced(a, M, top_k)
    else:
        A = _make_conv_matrix_full(a)
        b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_spread_v2(a, M, top_k=60):
    """Alternative LP: minimize max of conv(g,g) subject to conv(a,g) being spread."""
    n = len(a)
    # We want g that maximizes sum(g) while keeping conv(a,g) <= M
    # Also add a quadratic penalty approximation via additional constraints
    if n <= 32:
        A = _make_conv_matrix_full(a)
    else:
        A, _ = _make_conv_matrix_reduced(a, M, top_k)
    b = np.full(A.shape[0], M)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _line_search(a, g, max_iter=25):
    """Golden section search for optimal blend."""
    best_score = _score_norm(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c * (g - a), 0))
    fd = _score_norm(np.maximum(a + d * (g - a), 0))
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c * (g - a), 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d * (g - a), 0))
    best_t = (lo + hi) / 2.0
    return best_t


def _lp_refine(a, deadline, max_iter=4000, inflate=1.001, blend=0.25):
    """LP-based spreading with line search and adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        g = _lp_spread(a, M * inflate_cur)

        if g is None:
            inflate_cur = min(1.02, inflate_cur * 1.3)
            no_imp += 3
            if no_imp > 200:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate_cur = min(1.015, inflate_cur * 1.1)
            if no_imp > 300:
                break
            continue

        g = g / sum_g
        t_opt = _line_search(a, g)
        if t_opt < 0.005:
            t_opt = blend_cur * 0.3

        a_new = np.maximum((1.0 - t_opt) * a + t_opt * g, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc = _score_norm(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.02, blend_cur * 0.85)
                inflate_cur = min(1.005, inflate_cur * 0.9)
            if no_imp > max(150, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _lse_step(a, z, state, T, lr, flat_weight):
    """One Adam step with flatness-regularized LSE objective."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    mean_c = float(c.mean())

    # LSE approximation of max
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    # Flatness component: weights proportional to (c - mean)^2
    if flat_weight > 0:
        x2 = (c - mean_c) / (T * 0.5)
        x2 = np.clip(x2, -40, 40)
        ex2 = np.exp(x2 * 0.5)  # softer
        ex2 = ex2 / ex2.sum()
        w = w + flat_weight * ex2
        w = w / w.sum()

    # Gradient: 2 * conv(w, reverse(a)) centered
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    # Project gradient to tangent space (sum-preserving)
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    # Adam
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v

    return a_new, z, state, m_val


def _lse_optimize(a, deadline, max_iter=100000, T_start=0.08, T_end=1e-5,
                  lr=0.15, flat_weight=0.3, n_restarts=5):
    """LSE gradient with cosine annealing, flatness penalty, and warm restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for restart in range(n_restarts + 1):
        if time.monotonic() > deadline:
            break

        # Set up restart
        if restart > 0:
            rng_r = np.random.default_rng(12345 + restart * 7919)
            noise_scale = 0.05 + 0.03 * restart
            a = best_a * (1.0 + noise_scale * rng_r.normal(size=n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

        z = np.log(a)
        z -= float(np.mean(z))
        state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

        # Cosine annealing schedule for this restart
        T_cur = T_start / (1.0 + 0.5 * restart)
        T_min = max(T_end, T_start * 0.01 / (restart + 1))
        lr_cur = lr / (1.0 + 0.3 * restart)
        no_imp = 0
        iter_this_restart = 0

        for it in range(max_iter):
            if time.monotonic() > deadline:
                break

            # Cosine schedule
            progress = it / float(max_iter)
            T = T_min + 0.5 * (T_cur - T_min) * (1.0 + np.cos(np.pi * progress))
            lr_s = lr_cur * (0.5 + 0.5 * np.cos(np.pi * progress))

            a, z, state, m_val = _lse_step(a, z, state, T, lr_s, flat_weight)
            iter_this_restart += 1

            if not np.all(np.isfinite(a)):
                a = best_a.copy()
                z = np.log(np.maximum(a, 1e-12))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                continue

            c = _fft_convolve(a)
            score = float(2.0 * n * c.max())

            if score < best_score - 1e-13:
                best_score = score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > max(3000, 8 * n):
                    break

        print(f"    LSE restart {restart}: best={best_score:.10f}, iters={iter_this_restart}")

    return best_a


def _waterfill_equalize(a, n_steps=50, step_size=0.1):
    """Direct water-filling: try to equalize convolution values."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    target = 1.0 / (2 * n - 1)

    for step in range(n_steps):
        c = _fft_convolve(a)
        # Find lags above target
        excess = c - target
        # For each position i, compute its "contribution weight" to excess
        # Position i contributes to c[k] for k in [i, i+n-1] with weight a[i]*a[k-i]
        # Approximate: weight of position i ~ a[i] * sum of excess where i is involved
        N = _next_pow2(2 * n - 1)
        # conv(excess_clipped, a_rev) gives the gradient of the excess
        pos_excess = np.maximum(excess, 0)
        neg_excess = np.maximum(-excess, 0)

        fe = np.fft.rfft(pos_excess, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_pos = np.fft.irfft(fe * fa_rev, N)[n - 1:2 * n - 1]

        fn = np.fft.rfft(neg_excess, N)
        h_neg = np.fft.irfft(fn * fa_rev, N)[n - 1:2 * n - 1]

        # Push down where positive excess, push up where negative
        update = step_size * (h_neg - h_pos)
        # Scale by a to stay in probability simplex (multiplicative update)
        a_new = a * (1.0 + update / (a + 1e-15))
        a_new = np.maximum(a_new, 1e-12)
        s = a_new.sum()
        a_new = a_new / s

        sc = _score_norm(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.7

    return best_a


def _subgradient_polish(a, deadline, max_iter=60000, lr=0.02, momentum=0.85):
    """Polish using subgradient at the actual max with momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        # Use soft max for smoother gradient
        x = (c - m_val) / 0.01
        x = np.clip(x, -50, 50)
        w = np.exp(x)
        w = w / w.sum()

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        sc = _score_norm(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > max(4000, 6 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, w=0.5):
    """Flat in middle, elevated at edges."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + w * (np.cos(2.0 * np.pi * t) + 1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_power_profile(n, p):
    """a(t) = t^p * (1-t)^p normalized - concentrates at edges for large p."""
    t = np.linspace(0.0, 1.0, n)
    a = (t + 1e-6) ** p * (1.0 - t + 1e-6) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, center_sep=0.3):
    """Two peaks."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-50.0 * (t - 0.5 + center_sep) ** 2) + np.exp(-50.0 * (t - 0.5 - center_sep) ** 2)
    a = a + 1e-6
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining flatness-regularized LSE gradient,
    LP spreading, water-filling equalization, and subgradient polishing.
    Explores multiple sequence lengths and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 45.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Lengths to try - focus on 1024 and nearby
    lengths = [1024]
    if parent is not None:
        pn = parent.size
        if pn != 1024:
            lengths.append(pn)
    # Add a couple more for diversity
    for ln in [768, 1536]:
        if ln not in lengths:
            lengths.append(ln)

    def get_candidates(n_t):
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            # Perturbed
            p_int = _interpolate_profile(parent, n_t)
            p_pert = p_int * (1.0 + 0.1 * rng.normal(size=n_t))
            p_pert = np.maximum(p_pert, 0)
            if p_pert.sum() > 1e-10:
                cands.append(p_pert / p_pert.sum())
        # Structured alternatives
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_flat_edges(n_t, 0.4))
        cands.append(_make_power_profile(n_t, 2.0))
        cands.append(_make_bimodal(n_t, 0.25))
        return cands

    # Phase 1: LSE optimization on diverse candidates
    print("\n=== Phase 1: LSE gradient optimization ===")
    best_per_length = {}

    # Allocate time: 40% of budget for Phase 1
    phase1_deadline = min(deadline, t_start + 0.40 * budget_s)

    for n_t in lengths:
        if time.monotonic() > phase1_deadline:
            break

        cands = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > phase1_deadline:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # LSE optimization
            lse_deadline = min(phase1_deadline, time.monotonic() + 120.0)
            a_lse = _lse_optimize(a0, lse_deadline, max_iter=80000,
                                  T_start=0.08, T_end=1e-5, lr=0.15,
                                  flat_weight=0.3, n_restarts=3)
            sc_lse = _score(a_lse)

            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse

            print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> LSE={sc_lse:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best for n={n_t}: {best_here_score:.10f}")

    # Phase 2: LP refinement
    print("\n=== Phase 2: LP refinement ===")
    phase2_deadline = min(deadline, t_start + 0.65 * budget_s)

    for n_t in sorted(best_per_length.keys(), key=lambda x: _score(best_per_length[x])):
        if time.monotonic() > phase2_deadline:
            break

        a_in = best_per_length[n_t]
        sc_in = _score(a_in)
        print(f"  LP start n={n_t}: score={sc_in:.10f}")

        lp_deadline = min(phase2_deadline, time.monotonic() + 150.0)
        a_lp = _lp_refine(a_in, lp_deadline, max_iter=3000,
                          inflate=1.001, blend=0.2)
        sc_lp = _score(a_lp)

        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP result n={n_t}: score={sc_lp:.10f}")

    # Phase 3: Water-filling equalization
    print("\n=== Phase 3: Water-filling equalization ===")
    phase3_deadline = min(deadline, t_start + 0.78 * budget_s)

    for n_t in sorted(best_per_length.keys(), key=lambda x: _score(best_per_length[x])):
        if time.monotonic() > phase3_deadline:
            break

        a_in = best_per_length[n_t]
        a_wf = _waterfill_equalize(a_in, n_steps=80, step_size=0.08)
        sc_wf = _score(a_wf)

        if np.isfinite(sc_wf) and sc_wf < best_score:
            best_score = sc_wf
            best_seq = a_wf.copy()
        best_per_length[n_t] = a_wf
        print(f"  Waterfill n={n_t}: score={sc_wf:.10f}")

    # Phase 4: Subgradient polish on best
    print("\n=== Phase 4: Subgradient polish ===")
    if best_seq is not None:
        phase4_deadline = min(deadline, t_start + 0.90 * budget_s)
        sp_end = min(phase4_deadline, time.monotonic() + 120.0)
        a_sp = _subgradient_polish(best_seq, sp_end, max_iter=80000, lr=0.015)
        sc_sp = _score(a_sp)
        if np.isfinite(sc_sp) and sc_sp < best_score:
            best_score = sc_sp
            best_seq = a_sp.copy()
        print(f"  Subgradient: score={sc_sp:.10f}")

    # Phase 5: Final combined polish (LSE + LP alternating)
    print("\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = best_seq.size
        final_deadline = deadline

        # LSE pass
        lse_end = min(final_deadline, time.monotonic() + 60.0)
        a_f1 = _lse_optimize(best_seq, lse_end, max_iter=50000,
                             T_start=0.03, T_end=1e-6, lr=0.08,
                             flat_weight=0.5, n_restarts=2)
        sc_f1 = _score(a_f1)
        if sc_f1 < best_score:
            best_score = sc_f1
            best_seq = a_f1.copy()
        print(f"  Final LSE: {sc_f1:.10f}")

        # LP pass
        lp_end = min(final_deadline, time.monotonic() + 60.0)
        a_f2 = _lp_refine(best_seq, lp_end, max_iter=2000,
                          inflate=1.0005, blend=0.1)
        sc_f2 = _score(a_f2)
        if sc_f2 < best_score:
            best_score = sc_f2
            best_seq = a_f2.copy()
        print(f"  Final LP: {sc_f2:.10f}")

        # Water-fill
        a_f3 = _waterfill_equalize(best_seq, n_steps=50, step_size=0.05)
        sc_f3 = _score(a_f3)
        if sc_f3 < best_score:
            best_score = sc_f3
            best_seq = a_f3.copy()
        print(f"  Final Waterfill: {sc_f3:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
