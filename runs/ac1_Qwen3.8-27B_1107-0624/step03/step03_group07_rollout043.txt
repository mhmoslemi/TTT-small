    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _conv_gradient_at_peak(a, peak_idx):
    """Compute d(conv[peak])/da for a given peak position."""
    n = len(a)
    grad = np.zeros(n)
    # d(conv[k])/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0
    lo = max(0, peak_idx - (n - 1))
    hi = min(n - 1, peak_idx)
    for j in range(lo, hi + 1):
        idx = peak_idx - j
        if 0 <= idx < n:
            grad[j] += 2.0 * a[idx]
    return grad


def _soft_max_weights(c, beta):
    """Compute soft-max weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _compute_h_fft(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0..n-1 (correlation-like)."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def linear_ramp(n, direction=1):
    """Create a linear ramp sequence."""
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def power_ramp(n, p=1.0):
    """Create a power-law ramp."""
    a = np.arange(1, n + 1, dtype=np.float64) ** p
    return a / a.sum()


def cosine_ramp(n, k=1, amp=0.3):
    """Ramp with cosine modulation."""
    t = np.linspace(0, 1, n)
    a = t + amp * np.cos(2 * np.pi * k * t)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def smooth_asymmetric(n, alpha=1.0):
    """Smooth asymmetric profile."""
    t = np.linspace(0, 1, n)
    a = t**alpha + 0.05
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gradient_optimize(a0, deadline, rng, lr=0.01, beta_factor=0.05,
                      beta_decay=0.95, beta_min=1e-9, mu=0.9):
    """
    Gradient descent on logit-space with soft-max approximation of max.
    Uses FFT for efficient gradient computation.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = _full_convolve(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-10:
        beta = 1e-5

    # Adam
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _soft_max_weights(c, beta)
        h = _compute_h_fft(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1**t_adam)
        vh = v / (1.0 - b2**t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_a.copy()
            z = np.log(np.maximum(z, 1e-15)) - np.log(np.maximum(best_a, 1e-15)).mean()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s

        c = _full_convolve(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-15:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 500:
            current_lr = max(0.002, current_lr * 0.95)

        if no_improve > 2000:
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean() + 0.1 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = _full_convolve(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.05 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def single_peak_optimize(a0, deadline, rng, lr=0.005):
    """
    Direct gradient descent targeting the single peak position.
    More aggressive than soft-max when peak is clearly identified.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    best_a = a0.copy()
    c = _full_convolve(a0)
    best_score = float(2.0 * n * c.max())
    no_improve = 0
    current_lr = lr
    base_lr = lr

    # Momentum
    vel = np.zeros(n)
    b1 = 0.9

    while time.monotonic() < deadline:
        c = _full_convolve(a0)
        peak_idx = int(np.argmax(c))
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score - 1e-15:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = _conv_gradient_at_peak(a0, peak_idx)
        # Project to tangent space of simplex
        dot = float(np.dot(a0, grad))
        grad_t = grad - dot  # projection

        vel = b1 * vel + (1.0 - b1) * grad_t
        a0 = a0 - current_lr * vel
        a0 = np.maximum(a0, 1e-15)
        s = a0.sum()
        a0 = a0 / s

        if no_improve > 1000:
            current_lr = max(0.001, current_lr * 0.98)

        if no_improve > 3000:
            a0 = best_a + 0.001 * rng.normal(size=n)
            a0 = np.maximum(a0, 1e-15)
            a0 = a0 / a0.sum()
            vel.fill(0.0)
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search exploiting the key insight that monotonic ramps
    have much lower scores than uniform sequences (~1.33 vs ~2.0).
    Combines linear/power ramps with gradient-based refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Check for pre-existing sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc_init = compute_score(init_seq)
                print(f"Existing sequence: n={init_seq.size}, score={sc_init:.10f}")
                update(init_seq, sc_init)
    except Exception:
        pass

    n = 2048
    print(f"\n=== Starting optimization at n={n} ===")

    # Generate diverse initial profiles
    profiles = [
        ("linear_ramp_up", linear_ramp(n, 1)),
        ("linear_ramp_down", linear_ramp(n, -1)),
        ("power_0.5", power_ramp(n, 0.5)),
        ("power_1.5", power_ramp(n, 1.5)),
        ("smooth_asym_1", smooth_asymmetric(n, 1.0)),
        ("smooth_asym_2", smooth_asymmetric(n, 2.0)),
        ("cosine_ramp_1", cosine_ramp(n, 1, 0.2)),
        ("cosine_ramp_2", cosine_ramp(n, 2, 0.15)),
    ]

    if init_seq is not None:
        from scipy.ndimage import zoom
        try:
            if init_seq.size != n:
                x0 = np.linspace(0, 1, init_seq.size)
                x1 = np.linspace(0, 1, n)
                init_interp = np.interp(x1, x0, init_seq)
                init_interp = np.maximum(init_interp, 1e-10)
                init_interp = init_interp / init_interp.sum()
            else:
                init_interp = init_seq.copy()
            profiles.insert(0, ("existing", init_interp))
        except Exception:
            pass

    # Phase 1: Quick evaluation and short optimization of each profile
    phase1_deadline = min(deadline, t_start + 120.0)
    print(f"\n--- Phase 1: Profile evaluation (until t={120}s) ---")

    candidates = []
    for name, p in profiles:
        if time.monotonic() > phase1_deadline - 2:
            break
        sc = compute_score(p)
        candidates.append((sc, name, p))
        print(f"  {name}: score={sc:.10f}")
        update(p, sc)

    # Sort by score and optimize the best few
    candidates.sort(key=lambda x: x[0])
    print(f"\n  Best initial: {candidates[0][1]} with score {candidates[0][0]:.10f}")

    # Phase 2: Gradient optimization of top profiles
    phase2_deadline = min(deadline, t_start + 600.0)
    print(f"\n--- Phase 2: Gradient optimization (until t={600}s) ---")

    for sc_init, name, p in candidates[:4]:
        if time.monotonic() > phase2_deadline - 30:
            break
        alloc = min(phase2_deadline, time.monotonic() + 120.0)
        print(f"  Optimizing {name} (init score={sc_init:.8f})...")

        # Try different learning rates
        for lr in [0.02, 0.01, 0.005]:
            if time.monotonic() > phase2_deadline - 10:
                break
            dl = min(phase2_deadline, time.monotonic() + 80.0)
            p_try = p + 0.0001 * rng.normal(size=n)
            p_try = np.maximum(p_try, 1e-12)
            p_try = p_try / p_try.sum()

            result, sc_r = gradient_optimize(p_try, dl, rng, lr=lr,
                                             beta_factor=0.03, beta_decay=0.96)
            sc_final = compute_score(result)
            update(result, sc_final)
            if sc_final < sc_init:
                print(f"    lr={lr}: improved to {sc_final:.10f}")
            break  # Just try one lr per profile in phase 2

    # Phase 3: Extended optimization of the best
    phase3_deadline = min(deadline, t_start + 1200.0)
    print(f"\n--- Phase 3: Extended optimization (until t={1200}s) ---")

    if best_seq is not None:
        print(f"  Current best: {best_score:.10f}")

        # Long soft-max optimization
        dl3a = min(phase3_deadline, time.monotonic() + 400.0)
        p3 = best_seq.copy()
        p3 = p3 + 0.0002 * rng.normal(size=n)
        p3 = np.maximum(p3, 1e-12)
        p3 = p3 / p3.sum()

        result3, sc3 = gradient_optimize(p3, dl3a, rng, lr=0.008,
                                         beta_factor=0.02, beta_decay=0.97)
        sc3_final = compute_score(result3)
        update(result3, sc3_final)
        print(f"  After soft-max (lr=0.008): {sc3_final:.10f}")

        # Single-peak optimization
        if time.monotonic() < phase3_deadline - 60:
            dl3b = min(phase3_deadline, time.monotonic() + 200.0)
            result3b, sc3b = single_peak_optimize(best_seq, dl3b, rng, lr=0.003)
            sc3b_final = compute_score(result3b)
            update(result3b, sc3b_final)
            print(f"  After single-peak: {sc3b_final:.10f}")

    # Phase 4: Try different n and final polish
    phase4_deadline = deadline
    print(f"\n--- Phase 4: Final polish and size exploration ---")

    # Try larger n
    for n_try in [4096]:
        if time.monotonic() > phase4_deadline - 60:
            break
        if best_score < 1.34:  # Only if we're already doing well
            p_large = linear_ramp(n_try, 1)
            sc_large = compute_score(p_large)
            print(f"  n={n_try} linear ramp: {sc_large:.10f}")
            update(p_large, sc_large)

            # Interpolate best to this size and optimize
            if best_seq is not None:
                x0 = np.linspace(0, 1, best_seq.size)
                x1 = np.linspace(0, 1, n_try)
                p_interp = np.interp(x1, x0, best_seq)
                p_interp = np.maximum(p_interp, 1e-12)
                p_interp = p_interp / p_interp.sum()
                sc_interp = compute_score(p_interp)
                print(f"  n={n_try} interpolated: {sc_interp:.10f}")
                update(p_interp, sc_interp)

                dl = min(phase4_deadline, time.monotonic() + 60.0)
                result, sc_r = gradient_optimize(p_interp, dl, rng, lr=0.005,
                                                 beta_factor=0.02, beta_decay=0.97)
                sc_r = compute_score(result)
                update(result, sc_r)
                print(f"  n={n_try} after optimization: {sc_r:.10f}")

    # Try smaller n
    for n_small in [1024, 512]:
        if time.monotonic() > phase4_deadline - 30:
            break
        p_small = linear_ramp(n_small, 1)
        sc_small = compute_score(p_small)
        update(p_small, sc_small)
        print(f"  n={n_small} linear ramp: {sc_small:.10f}")

    # Final polish at best size
    if best_seq is not None and time.monotonic() < phase4_deadline - 20:
        n_final = best_seq.size
        p_final = best_seq + 0.00005 * rng.normal(size=n_final)
        p_final = np.maximum(p_final, 1e-12)
        p_final = p_final / p_final.sum()
        dl = min(phase4_deadline, time.monotonic() + 30.0)
        result, sc_r = gradient_optimize(p_final, dl, rng, lr=0.003,
                                         beta_factor=0.01, beta_decay=0.98)
        sc_r = compute_score(result)
        update(result, sc_r)
        print(f"  Final polish: {sc_r:.10f}")

    print(f"\n=== FINAL: n={best_seq.size if best_seq is not None else 0}, "
          f"score={best_score:.10f} ===")

    # Validate and return
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = linear_ramp(2048, 1)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    final_score = compute_score(final_seq)
    print(f"Final validated score: {final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
        if np.isfinite(eval_score) and eval_score < best_score:
            pass
        if np.isfinite(eval_score) and eval_score < final_score:
            final_score = eval_score
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
