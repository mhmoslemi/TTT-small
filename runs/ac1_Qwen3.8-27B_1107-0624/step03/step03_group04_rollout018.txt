    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(a)
    w = ex / s

    # grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * (w * a_rev)[n-1+i]
    N = _next_pow2(3 * n)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h[n - 1:2 * n - 1]

    val = mval + beta * float(np.log(s))
    return val, grad


def _interpolate_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        if s > 1e-30:
            return p.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    s = p_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p_new / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_edge_profile(n, gamma=2.0):
    """Profile boosted at edges."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + gamma * (t ** 2 + (1 - t) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_spectrum(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _alternating_lp_optimize(a, deadline, max_iter=5000, momentum=0.3):
    """
    Alternating LP optimization with momentum.
    Minimizes max conv(a,a) by alternating best-response LPs.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    c = _fft_convolve(a)
    best_max_conv = float(c.max())

    v = np.zeros(n)  # momentum velocity
    no_imp = 0
    alpha = 0.5  # mixing parameter

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Solve LP: find b that minimizes max_k conv(a, b)[k]
        b_star, t_star = _lp_best_response(a)
        if b_star is None:
            break

        # Update with momentum
        v = momentum * v + (1.0 - momentum) * (b_star - a)
        a_new = a + alpha * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = b_star.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        c_new = _fft_convolve(a_new)
        max_conv_new = float(c_new.max())
        score_new = float(2.0 * n * max_conv_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_max_conv = max_conv_new
            best_a = a_new.copy()
            no_imp = 0
            # Slowly decrease alpha as we converge
            if it > 50:
                alpha = max(0.1, alpha * 0.999)
        else:
            no_imp += 1
            if no_imp > 100:
                # Try smaller step
                alpha = max(0.05, alpha * 0.8)
                if no_imp > 500:
                    # Revert and reduce momentum
                    a = best_a.copy()
                    v = np.zeros(n)
                    alpha = max(0.1, alpha * 0.5)
                    momentum = momentum * 0.8
                    no_imp = 0
                    if alpha < 0.05:
                        break

        a = a_new

    return best_a


def _gradient_optimize(a, deadline, max_iter=30000, lr=0.03):
    """Gradient descent in log-space with smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.05 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= np.mean(z)

    # Adam
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    base_lr = lr
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        _, grad = _smooth_max_and_grad(p, beta)
        # Project gradient to simplex tangent space
        dot = float(np.dot(p, grad))
        gz = p * (grad - dot)
        gz -= np.mean(gz)

        if not np.all(np.isfinite(gz)):
            beta *= 2
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v_arr = b2 * v_arr + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v_arr / (1 - b2 ** t_adam)

        z = z - base_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.95)
            if no_imp > max(5000, 10 * n):
                break

        # Adaptive beta
        if it % 200 == 0 and it > 200:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                gap = top5[0] - top5[-1]
                if gap < beta * 0.3:
                    beta = max(1e-12, beta * 0.7)
                elif gap > beta * 5:
                    beta = min(0.5 * float(c.max()), beta * 1.3)

    return best_a


def _coord_descent(a, deadline, n_passes=10, delta_init=0.02):
    """
    Coordinate-wise optimization: for each coordinate, try small multiplicative
    perturbations and keep the best.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_score = _score(a)
    best_a = a.copy()
    delta = delta_init

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order for diversity
        order = np.random.default_rng(p * 31 + 7).permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                a[i] = 1e-8
                a = a / a.sum()
                continue
            # Try a few multipliers
            best_local = old_val
            best_local_score = best_score
            for mult in [1 + delta, 1 - delta, 1 + 2*delta, 1 - 2*delta, 1 + 0.5*delta, 1 - 0.5*delta]:
                if mult <= 0:
                    continue
                a[i] = old_val * mult
                sc = _score(a)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = old_val * mult
            a[i] = best_local
            if best_local != old_val:
                improved = True
                a = a / a.sum()
                if _score(a) < best_score:
                    best_score = _score(a)
                    best_a = a.copy()

        if not improved:
            delta *= 0.7
            if delta < 0.001:
                break

    return best_a


def _multi_start(a, deadline, n_starts=5, rng=None):
    """Perturb and re-optimize to escape local minima."""
    if rng is None:
        rng = np.random.default_rng(999)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        scale = 0.03 * (i + 1)
        a_pert = a * (1.0 + scale * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        if sc > best_score * 1.1:
            continue

        # Quick gradient optimization
        sub_dl = min(deadline, time.monotonic() + 30.0)
        a_ref = _gradient_optimize(a_pert, sub_dl, max_iter=3000, lr=0.02)
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Diverse initialization
    2. Coarse-to-fine alternating LP
    3. Gradient refinement
    4. Coordinate descent polish
    5. Multi-start escape
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Target lengths
    target_lengths = [768]
    if parent is not None:
        target_lengths.insert(0, parent.size)
    target_lengths = list(dict.fromkeys(target_lengths))  # dedupe, preserve order

    # Phase 1: Initialization
    print(f"Phase 1: Initialization (time left: {deadline - time.monotonic():.0f}s)", flush=True)
    init_pool = []
    for n_t in target_lengths:
        if parent is not None:
            init_pool.append(('parent', n_t, _interpolate_profile(parent, n_t)))
        init_pool.append(('linear1.5', n_t, _make_linear_profile(n_t, 1.5)))
        init_pool.append(('linear2.0', n_t, _make_linear_profile(n_t, 2.0)))
        init_pool.append(('bell', n_t, _make_bell_profile(n_t, 0.35)))
        init_pool.append(('edge', n_t, _make_edge_profile(n_t, 2.0)))
        init_pool.append(('flat_spec', n_t, _make_flat_spectrum(n_t, rng)))

    # Score all inits
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print("  Top 5 inits:", flush=True)
    for sc, name, n_t, _ in init_scores[:5]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Alternating LP on top candidates
    top_inits = init_scores[:3]
    for sc0, name, n_t, a0 in top_inits:
        if time.monotonic() > deadline - 200:
            break
        if sc0 > best_score * 1.05 and name != 'parent':
            continue

        # LP optimization - give generous time
        lp_budget = 400.0
        lp_dl = min(deadline - 150, t_start + 120 + lp_budget)
        # Actually, be more dynamic
        remaining = deadline - time.monotonic()
        lp_dl = min(deadline - 100, time.monotonic() + min(lp_budget, remaining * 0.4))

        print(f"Phase 2: Alt-LP on {name} (n={n_t}), start={sc0:.8f}, budget={lp_dl - time.monotonic():.0f}s", flush=True)
        a_lp = _alternating_lp_optimize(a0, lp_dl, max_iter=10000, momentum=0.3)
        sc_lp = _score(a_lp)
        print(f"  Alt-LP result: {sc_lp:.10f}", flush=True)

        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()

        # Gradient refinement
        if time.monotonic() < deadline - 60:
            g_dl = min(deadline - 50, time.monotonic() + 120.0)
            a_g = _gradient_optimize(a_lp, g_dl, max_iter=20000, lr=0.03)
            sc_g = _score(a_g)
            print(f"  Gradient result: {sc_g:.10f}", flush=True)
            if sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            a_current = a_g
        else:
            a_current = a_lp

        # Second LP pass with lower momentum for fine-tuning
        if time.monotonic() < deadline - 40:
            lp2_dl = min(deadline - 30, time.monotonic() + 80.0)
            a_lp2 = _alternating_lp_optimize(a_current, lp2_dl, max_iter=5000, momentum=0.1)
            sc_lp2 = _score(a_lp2)
            print(f"  Alt-LP2 result: {sc_lp2:.10f}", flush=True)
            if sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = a_lp2.copy()

    # Phase 3: Coordinate descent on best
    if best_seq is not None and time.monotonic() < deadline - 20:
        cd_dl = min(deadline - 10, time.monotonic() + 120.0)
        print(f"Phase 3: Coord descent, budget={cd_dl - time.monotonic():.0f}s", flush=True)
        a_cd = _coord_descent(best_seq, cd_dl, n_passes=15, delta_init=0.015)
        sc_cd = _score(a_cd)
        print(f"  Coord descent: {sc_cd:.10f}", flush=True)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()

    # Phase 4: Fine gradient polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        gp_dl = min(deadline - 5, time.monotonic() + 60.0)
        if gp_dl - time.monotonic() > 5:
            print(f"Phase 4: Fine gradient, budget={gp_dl - time.monotonic():.0f}s", flush=True)
            a_fine = _gradient_optimize(best_seq, gp_dl, max_iter=20000, lr=0.01)
            sc_fine = _score(a_fine)
            print(f"  Fine gradient: {sc_fine:.10f}", flush=True)
            if sc_fine < best_score:
                best_score = sc_fine
                best_seq = a_fine.copy()

    # Phase 5: Multi-start escape
    if best_seq is not None and time.monotonic() < deadline - 10:
        ms_dl = deadline - 5
        if ms_dl - time.monotonic() > 20:
            print(f"Phase 5: Multi-start, budget={ms_dl - time.monotonic():.0f}s", flush=True)
            a_ms = _multi_start(best_seq, ms_dl, n_starts=4, rng=rng)
            sc_ms = _score(a_ms)
            print(f"  Multi-start: {sc_ms:.10f}", flush=True)
            if sc_ms < best_score:
                best_score = sc_ms
                best_seq = a_ms.copy()

    # Final: one more coord descent pass if time
    if best_seq is not None and time.monotonic() < deadline - 3:
        fd = deadline - 2
        if fd - time.monotonic() > 10:
            a_final = _coord_descent(best_seq, fd, n_passes=5, delta_init=0.005)
            sc_final = _score(a_final)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final coord: {sc_final:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(512, 1.5)

    # Validate with authoritative evaluator
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
