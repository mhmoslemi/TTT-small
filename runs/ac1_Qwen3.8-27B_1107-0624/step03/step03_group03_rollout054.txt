"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines mirror descent, spectral error reduction, and LP refinement.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    if n <= 1:
        return 2
    k = 1
    while k < n:
        k <<= 1
    return k

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0)  # clip tiny negative values from FFT
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _mirror_descent_step(a, T, eta):
    """One mirror descent step with soft subgradient."""
    n = len(a)
    c = _fft_convolve(a)
    # Softmax weights
    c_shifted = c - c.max()
    w = np.exp(c_shifted / T)
    s = w.sum()
    if s < 1e-30 or not np.isfinite(s):
        # Fallback: hard subgradient at argmax
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = w / s
    # Gradient: g_j = sum_k w_k * a_{k-j}
    # This is a cross-correlation: g = correlate(w, a) at appropriate lags
    # Specifically: g_j = sum_k w_k * a[k-j] for valid a[k-j]
    # = sum_k w_k * a[max(0, min(n-1, k-j))] ... no, a[k-j] is 0 if out of range
    # So g_j = sum_{k: 0<=k-j<n} w_k * a[k-j] = sum_{m: 0<=m<n} w_{m+j} * a[m]
    # = (w shifted) correlated with a
    # Using FFT: g = ifft(fft(w_full) * conj(fft(a_full))) ... 
    # Actually, let's think of it as: g_j = sum_m a_m * w_{m+j}
    # This is the cross-correlation of a and w at lag j.
    # In numpy: np.correlate(w, a, mode='valid') gives c_j = sum_m a_m * w_{m+j} for j=0,...,len(w)-len(a)
    # But we need it for j = 0, ..., n-1 and w has length 2n-1.
    # g_j = sum_{m=0}^{n-1} a_m * w_{m+j}
    # This is the "valid" part of the cross-correlation of w (length 2n-1) and a (length n).
    
    # Use FFT for efficiency:
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # Cross-correlation: ifft(conj(fa) * fw) gives sum_m a_m * w_{m+j} ... 
    # Actually: ifft(fa * conj(fw)) gives the convolution a * flip(w)
    # ifft(conj(fa) * fw) gives the cross-correlation
    # Let me be careful:
    # conv(a, b)_k = sum_j a_j * b_{k-j} => DFT: A * B
    # corr(a, b)_k = sum_j a_{j+k} * b_j => DFT: conj(A) * B
    # We want: g_j = sum_m a_m * w_{m+j} = corr(w, a)_j (cross-corr of w and a at lag j)
    # corr(w, a)_j = sum_m w_{m+j} * a_m
    # DFT of corr(w,a) = conj(W) * A
    # So g = ifft(conj(FW) * FA)
    
    g_full = np.fft.irfft(np.conj(fw) * fa, N)
    g = g_full[:n]  # first n elements
    
    # Mirror descent update: a_j <- a_j * exp(-eta * g_j) / Z
    # For numerical stability, shift g
    g_shifted = g - g.max()
    logits = np.log(np.maximum(a, 1e-300)) + eta * g_shifted
    # Wait, mirror descent: a_new_j = a_j * exp(-eta * g_j) / Z
    # = exp(log(a_j) - eta * g_j) / Z
    logits = np.log(np.maximum(a, 1e-300)) - eta * g
    logits -= logits.max()  # for numerical stability
    exp_logits = np.exp(logits)
    a_new = exp_logits / exp_logits.sum()
    
    return a_new, c

def _mirror_descent(a, deadline, max_iter=500000, T_init=0.01, T_min=0.0001,

    """Mirror descent optimization with adaptive temperature and step size."""

    T = T_init

        else:

def _spectral_error_reduction(a, deadline, max_iter=100000, alpha=0.5):
    """Spectral error reduction for convolution flattening."""
    a = _normalize(a.copy())
    n = len(a)
    N = _next_pow2(2 * n - 1)
    target = 1.0 / (2 * n - 1)  # flat target
    
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Forward: compute convolution
        fa = np.fft.rfft(a, N)
        fa2 = fa * fa
        c = np.fft.irfft(fa2, N)[:2*n-1]
        
        # Project toward flat
        c_proj = (1 - alpha) * c + alpha * target
        c_proj = np.maximum(c_proj, 0)
        
        # Inverse: recover a from c_proj
        fc = np.fft.rfft(c_proj, N)
        # A_new = sqrt(fc) with phase from fa
        # |A_new| = sqrt(|fc|) ... but fc might be complex
        # Better: A_new = fa * sqrt(|fc| / |fa|^2) = fa / |fa| * sqrt(|fc|)
        abs_fa = np.abs(fa)
        abs_fc = np.abs(fc)
        # Guard against zero
        mask = abs_fa > 1e-30
        fa_new = np.zeros_like(fa)
        fa_new[mask] = fa[mask] / abs_fa[mask] * np.sqrt(abs_fc[mask])
        fa_new[~mask] = np.sqrt(abs_fc[~mask])  # for zero freq bins
        
        # Inverse FFT
        a_new = np.fft.irfft(fa_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        a_new = _normalize(a_new)
        
        score = _score_norm(a_new)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 5000 == 0:
                alpha = max(0.01, alpha * 0.9)
            if no_imp > max(10000, 10 * n):
                a = best_a.copy()
                alpha = 0.5
                no_imp = 0
        
        a = a_new
    
    return best_a, best_score

def _lp_refine(a, deadline, max_iter=5000, inflate=1.0003, top_k=30):
    """LP-based refinement with reduced constraint set."""
    a = _normalize(a.copy())
    n = len(a)

    no_imp = 0
    blend = 0.2

        c = _fft_convolve(a)
        M = float(c.max())

        # Find active constraints (near-max lags)
        idx = np.argsort(c)[::-1][:top_k]
        active = set()
        for i in idx:
            for off in range(-2, 3):
                j = i + off
                if 0 <= j < 2*n-1:
                    active.add(j)
        active = sorted(active)

        A_ub = np.zeros((len(active), n))
        b_ub = np.full(len(active), M_lp)
        for row, k in enumerate(active):
            for j in range(max(0, k-n+1), min(n, k+1)):
                A_ub[row, j] = a[j]
        
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
