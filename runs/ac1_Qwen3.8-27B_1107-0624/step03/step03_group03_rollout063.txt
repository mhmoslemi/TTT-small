Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines log-sum-exp gradient descent with adaptive annealing,
LP-based peak shaving, and multi-start random perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    """Score with sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return float(2.0 * n * _fft_convolve(a_norm).max())

Optimizes step function to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp gradient, LP peak shaving, and multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_norm(a):
    """Score assuming sum(a)=1."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def _project_simplex(v):
    """Project v onto the probability simplex (sum=1, all>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def _lse_gradient(a, T, top_k=None):
    """Compute log-sum-exp gradient of max(conv) w.r.t. a, in logit space."""
    n = a.size
    c = _fft_convolve(a)
    
    if top_k is not None and top_k < len(c):
        # Use only top K lags
        idx = np.argsort(c)[::-1][:top_k]
        c_top = c[idx]
        m = c_top.max()
        x = (c_top - m) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        w_top = ex / ex.sum()
        # Build full w vector
        w = np.zeros_like(c)
        w[idx] = w_top
    else:
        m = c.max()
        x = (c - m) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        w = ex / ex.sum()
    
    # Gradient: 2 * cross_conv(w, a) centered
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    
    # Logit-space gradient
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    
    return gz, float(c.max())

                    active.add(j)

        A = np.zeros((n_active, n))
        b = np.full(n_active, M_lp)

            j_end = min(n, k + 1)

                A[row, j] = a[j]

                no_imp = 0
                inflate_cur = inflate
                blend_cur = blend

        dd = lo + gr * (hi - lo)

                fd = fc

                cc = dd
                fc = fd
                dd = lo + gr * (hi - lo)

def _interpolate_profile(p0, n_new):

"""
Optimizes step function to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive top-K log-sum-exp gradient, dynamic LP peak shaving,
multi-start exploration, and subgradient polishing with Nesterov momentum.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_norm(a):
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def _project_simplex(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(ind[cond]))
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def _adaptive_top_k(c, frac=0.01):
    m = c.max()
    thresh = m * (1.0 - frac)
    k = int(np.sum(c >= thresh))
    return max(k, 5)


def _lse_gradient_topk(a, T, top_k):
    n = a.size
    c = _fft_convolve(a)
    m_val = float(c.max())
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_top = c[idx]
        m_top = c_top.max()
        x = (c_top - m_top) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros_like(c)
            w[idx] = ex / s
    else:
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    return gz, m_val


def _lse_optimize(a, deadline, T_start=0.04, T_end=1e-6, lr=0.07,
                  max_iter=200000, top_k_init=30, top_k_min=5,
                  reheat_interval=6000, reheat_T=0.025):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z -= np.mean(z)
    state_t = 0
    state_m = np.zeros(n)
    state_v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    best_a = a.copy()
    best_score = _score_norm(a)
    ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    T = T_start
    no_imp = 0
    base_lr = lr
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        top_k = max(top_k_min, int(top_k_init * (T / T_start) ** 0.3))
        gz, m_val = _lse_gradient_topk(a, T, top_k)
        if not np.all(np.isfinite(gz)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-15))
            z -= np.mean(z)
            state_t = 0
            state_m = np.zeros(n)
            state_v = np.zeros(n)
            continue
        state_t += 1
        state_m = b1 * state_m + (1.0 - b1) * gz
        state_v = b2 * state_v + (1.0 - b2) * (gz * gz)
        mh = state_m / (1.0 - b1 ** state_t)
        vh = state_v / (1.0 - b2 ** state_t)
        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -40, 40)
        z -= np.mean(z)
        e = np.exp(z)
        a_new = e / e.sum()
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        a = a_new
        T = max(T * ratio, T_end)
        if no_imp > reheat_interval:
            T = reheat_T
            lr = base_lr
            a = best_a * (1.0 + 0.025 * rng.normal(size=n))
            a = _normalize(a)
            z = np.log(a)
            z -= np.mean(z)
            state_t = 0
            state_m = np.zeros(n)
            state_v = np.zeros(n)
            no_imp = 0
        if no_imp > 3000 and no_imp % 2000 == 0:
            lr = max(0.01, lr * 0.92)
    return best_a


def _lp_refine(a, deadline, max_iter=1500, inflate=1.0004, blend=0.15, top_k=120):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        n_c = min(top_k, 2 * n - 1)
        idx = np.argsort(c)[::-1][:n_c]
        active = set()
        for i in idx:
            for off in range(-3, 4):
                j = int(i) + off
                if 0 <= j < 2 * n - 1:
                    active.add(j)
        active = sorted(active)
        n_active = len(active)
        A = np.zeros((n_active, n))
        b = np.full(n_active, M_lp)
        for row, k in enumerate(active):
            k = int(k)
            j_s = max(0, k - n + 1)
            j_e = min(n, k + 1)
            A[row, j_s:j_e] = a[j_s:j_e]
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        if not res.success:
            inflate_cur = min(1.01, inflate_cur * 1.3)
            no_imp += 3
            if no_imp > 250:
                a = best_a.copy()
                no_imp = 0
                inflate_cur = inflate
            continue
        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 150:
                a = best_a.copy()
                no_imp = 0
                inflate_cur = inflate
                blend_cur = blend
            continue
        g = g / sum_g
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        cc = hi - gr * (hi - lo)
        dd = lo + gr * (hi - lo)
        blend_c = a + cc * (g - a)
        blend_c = np.maximum(blend_c, 0)
        if blend_c.sum() > 1e-10:
            blend_c = blend_c / blend_c.sum()
            fc = _score_norm(blend_c)
        else:
            fc = np.inf
        blend_d = a + dd * (g - a)
        blend_d = np.maximum(blend_d, 0)
        if blend_d.sum() > 1e-10:
            blend_d = blend_d / blend_d.sum()
            fd = _score_norm(blend_d)
        else:
            fd = np.inf
        for _ in range(14):
            if fc < fd:
                hi = dd
                dd = cc
                fd = fc
                cc = hi - gr * (hi - lo)
                blend_c = a + cc * (g - a)
                blend_c = np.maximum(blend_c, 0)
                if blend_c.sum() > 1e-10:
                    blend_c = blend_c / blend_c.sum()
                    fc = _score_norm(blend_c)
                else:
                    fc = np.inf
            else:
                lo = cc
                cc = dd
                fc = fd
                dd = lo + gr * (hi - lo)
                blend_d = a + dd * (g - a)
                blend_d = np.maximum(blend_d, 0)
                if blend_d.sum() > 1e-10:
                    blend_d = blend_d / blend_d.sum()
                    fd = _score_norm(blend_d)
                else:
                    fd = np.inf
        t_opt = (lo + hi) / 2.0
        if t_opt < 0.005:
            t_opt = blend_cur * 0.5
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = _normalize(a_new)
        score_new = _score_norm(a_new)
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 25 == 0:
                blend_cur = max(0.02, blend_cur * 0.9)
            if no_imp > 250:
                a = best_a.copy()
                no_imp = 0
                inflate_cur = inflate
                blend_cur = blend
        a = a_new
    return best_a


def _nesterov_polish(a, deadline, max_iter=40000, lr=0.008, momentum=0.85):
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    y = a.copy()
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c_y = _fft_convolve(y)
        m_val = float(c_y.max())
        max_idx = np.where(c_y >= m_val - 1e-13)[0]
        w = np.zeros_like(c_y)
        w[max_idx] = 1.0 / len(max_idx)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fy_rev = np.fft.rfft(y[::-1], N)
        h_full = np.fft.irfft(fw * fy_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        dot = float(np.dot(y, grad))
        grad_tan = grad - dot * y
        v = momentum * v - lr * grad_tan
        y_new = _project_simplex(y + v)
        score = _score_norm(y_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = y_new.copy()
            no_imp = 0
            a = y_new
        else:
            no_imp += 1
        y = y_new
        if no_imp % 1000 == 0:
            lr = max(0.001, lr * 0.9)
        if no_imp > 8000:
            a = best_a.copy()
            y = a.copy()
            v = np.zeros(n)
            lr = base_lr
            no_imp = 0
    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    return _normalize(p)


def _make_profiles(n, rng):
    profiles = []
    t = np.linspace(0, 1, n)
    profiles.append(_normalize(1.0 + 0.2 * t))
    profiles.append(_normalize(1.0 + 0.3 * np.sin(np.pi * t)))
    profiles.append(_normalize(np.exp(-2.0 * t)))
    profiles.append(_normalize(np.exp(-2.0 * (1.0 - t))))
    profiles.append(_normalize(1.0 + 0.15 * np.cos(2.0 * np.pi * t)))
    profiles.append(_normalize((1.0 - t) ** 0.8))
    return profiles
