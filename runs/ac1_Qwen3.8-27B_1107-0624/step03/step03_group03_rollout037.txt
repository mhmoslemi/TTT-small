Combines adaptive top-K gradient, spectral initialization, LP refinement,

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

    a = np.asarray(a, dtype=np.float64).ravel()

    c = _fft_convolve(a)

        hi = min(n, k + 1)
        for j in range(lo, hi):

    # Tangent space gradient
    dot = float(np.dot(a, grad))
    grad_t = grad - dot * a
    a_new = a - lr * grad_t
    a_new = np.maximum(a_new, 0.0)

    # In log space: z -= lr * grad

def _spectral_init(n, seed=0):
    """Generate a sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Random power spectrum (flat except DC)
    n_fft = _next_pow2(2 * n - 1)
    power = rng.uniform(0.5, 1.5, n_fft // 2 + 1)
    power[0] = 0  # Don't set DC from power spectrum
    # Random phases
    phases = rng.uniform(0, 2*np.pi, n_fft // 2 + 1)
    # Build spectrum
    spectrum = np.sqrt(power) * np.exp(1j * phases)
    # Make it Hermitian
    full = np.zeros(n_fft, dtype=complex)
    full[:n_fft//2+1] = spectrum
    if n_fft % 2 == 0:
        for k in range(1, n_fft//2):
            full[n_fft - k] = np.conj(full[k])
    else:
        for k in range(1, n_fft//2 + 1):
            full[n_fft - k] = np.conj(full[k])
    
    # Inverse FFT
    a_full = np.fft.irfft(full, n_fft).real
    a = a_full[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_gaussian(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma * 0.5))**2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_bimodal(n, center_sep=0.3):
    """Two Gaussian humps."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5 + center_sep) / 0.15)**2) + \
        np.exp(-0.5 * ((t - 0.5 - center_sep) / 0.15)**2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64)
    n_old = a.size
    if n_old == n_new:
        s = a.sum()
        return a / s if s > 1e-15 else np.ones(n_new) / n_new
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    return a_new / s if s > 1e-15 else np.ones(n_new) / n_new

"""

2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovations over previous approach:
1. Dual fixed-point iteration exploiting the KKT structure: alternates between

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _make_conv_matrix_rows(a, indices):
    n = len(a)
    m = len(indices)
    A = np.zeros((m, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[i, j] = a[k - j]
    return A


def _solve_trievelis_top(lam, n):
    """Solve upper-triangular Toeplitz system H@a = 1 where H[j,k]=lam[k-j] for k>=j."""
    a = np.zeros(n)
    if abs(lam[0]) < 1e-30:
        return a
    a[n - 1] = 1.0 / lam[0]
    for j in range(n - 2, -1, -1):
        val = 1.0
        seg_len = n - 1 - j
        if seg_len > 0:
            val -= np.dot(lam[1:1 + seg_len], a[j + 1:n])
        a[j] = val / lam[0]
    return a


def _dual_fixed_point(a_init, deadline, tau_start=0.1, tau_end=0.001, damping=0.5):
    """
    Dual fixed-point iteration:
    - Given dual variables lam (lag weights), solve triangular system for a.
    - Given a, update lam = softened peak of conv(a,a).
    Alternates with annealing of temperature and damping.
    """
    a = np.maximum(a_init, 1e-12).copy()
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / tau_start
    x = np.clip(x, -80, 0)
    lam = np.exp(x)
    lam = lam / lam.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    total_iters = max(100, min(5000, int((deadline - time.monotonic()) * 50)))
    if total_iters > 1:
        tau_ratio = (tau_end / tau_start) ** (1.0 / total_iters)
    else:
        tau_ratio = 1.0

    tau = tau_start
    no_imp = 0

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break

        # Solve H(a) = 1 where H is upper triangular Toeplitz from lam
        a_sol = _solve_trievelis_top(lam[:n], n)

        # Clip negatives
        a_sol = np.maximum(a_sol, 0.0)
        s = a_sol.sum()
        if s < 1e-15:
            s = 1.0
        a_sol = a_sol / s

        # Damped update
        a_new = damping * a + (1.0 - damping) * a_sol
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a_sol
        else:
            a_new = a_new / s

        sc = _score_norm(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        # Update dual: lam = softmax(conv(a,a) / tau)
        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / tau
        x = np.clip(x, -80, 0)
        lam_new = np.exp(x)
        s_lam = lam_new.sum()
        if s_lam < 1e-30:
            lam_new = np.ones(len(c))
            s_lam = len(c)
        lam_new = lam_new / s_lam

        # Damped dual update
        lam = 0.5 * lam + 0.5 * lam_new

        tau *= tau_ratio
        tau = max(tau, tau_end)

        if no_imp > max(500, 5 * n):
            # Reset with more aggressive temperature
            tau = max(tau * 2.0, 0.01)
            damping = 0.3
            c = _fft_convolve(a)
            m_val = float(c.max())
            x = (c - m_val) / tau
            x = np.clip(x, -80, 0)
            lam = np.exp(x)
            lam = lam / lam.sum()
            no_imp = 0

    return best_a


def _topk_gradient_optimize(a, deadline, lr=0.05, K=None, method='additive',
                            tau_adaptive=True):
    """Optimize using top-K gradient with Adam and adaptive temperature."""
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()
    n = len(a)

    if K is None:
        K = max(5, min(20, n // 50))

    best_a = a.copy()
    best_score = _score_norm(a)

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999

    no_imp = 0
    base_lr = lr
    current_lr = lr

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        m_val = float(c.max())

        if tau_adaptive:
            sorted_c = np.sort(c)[::-1]
            k_eff = min(K, len(sorted_c) - 1)
            tau = max(1e-10, (m_val - sorted_c[k_eff] + 1e-15) * 0.5)
        else:
            tau = 0.01

        x = (c - m_val) / tau
        x = np.clip(x, -100, 0)
        ex = np.exp(x)
        w = ex / ex.sum()

        # Gradient via FFT
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        grad = 2.0 * h_full[n - 1:2 * n - 1]

        # Tangent space
        dot = float(np.dot(a, grad))
        grad_t = grad - dot * a

        # Adam
        t += 1
        m_state = b1 * m_state + (1.0 - b1) * grad_t
        v_state = b2 * v_state + (1.0 - b2) * (grad_t * grad_t)
        m_hat = m_state / (1.0 - b1 ** t)
        v_hat = v_state / (1.0 - b2 ** t)
        step = m_hat / (np.sqrt(v_hat) + 1e-8)

        if method == 'additive':
            a_new = a - current_lr * step
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a
            else:
                a_new = a_new / s
        else:
            log_a = np.log(np.maximum(a, 1e-300))
            log_a -= current_lr * step
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a
            else:
                a_new = a_new / s

        if not np.all(np.isfinite(a_new)):
            a_new = a
            no_imp += 1
        else:
            sc = _score_norm(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                no_imp = 0
            else:
                # Try half step
                if method == 'additive':
                    a_half = a - current_lr * 0.5 * step
                    a_half = np.maximum(a_half, 0.0)
                    s = a_half.sum()
                    if s > 1e-15:
                        a_half = a_half / s
                        sc_half = _score_norm(a_half)
                        if sc_half < best_score - 1e-12:
                            best_score = sc_half
                            best_a = a_half.copy()
                            a = a_half
                            no_imp = 0
                            continue
                no_imp += 1

        if no_imp % 2000 == 0 and no_imp > 0:
            current_lr = max(0.003, current_lr * 0.85)

        if no_imp > max(5000, 5 * n):
            rng_r = np.random.default_rng(t)
            a = best_a * (1.0 + 0.04 * rng_r.normal(size=n))
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t = 0
            current_lr = base_lr
            no_imp = 0

    return best_a


def _lp_refine(a, deadline, max_iter=3000, inflate=1.001, top_k=40):
    """LP-based refinement with reduced constraints and line search."""
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        threshold = M * 0.995

        active_idx = np.where(c >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])
        if len(active_idx) > top_k:
            order = np.argsort(c[active_idx])[::-1][:top_k]
            active_idx = active_idx[order]

        A = _make_conv_matrix_rows(a, active_idx)
        b = np.full(len(active_idx), M * inflate_cur)
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')

        if not res.success:
            inflate_cur = min(1.01, inflate_cur * 1.2)
            no_imp += 3
            if no_imp > 200:
                break
            continue

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 15 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.08)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        # Line search with golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = 1.0
        cc = hi - gr * (hi - lo)
        dd = lo + gr * (hi - lo)
        fc = _score_norm(_normalize(np.maximum(a + cc * (g - a), 0)))
        fd = _score_norm(_normalize(np.maximum(a + dd * (g - a), 0)))
        for _ in range(15):
            if fc < fd:
                hi = dd
                dd = cc
                fd = fc
                cc = hi - gr * (hi - lo)
                fc = _score_norm(_normalize(np.maximum(a + cc * (g - a), 0)))
            else:
                lo = cc
                cc = dd
                fc = fd
                dd = lo + gr * (hi - lo)
                fd = _score_norm(_normalize(np.maximum(a + dd * (g - a), 0)))
        t_opt = (lo + hi) / 2.0

        a_new = _normalize(np.maximum((1.0 - t_opt) * a + t_opt * g, 0))
        sc = _score_norm(a_new)

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            a = a_new
            no_imp += 1
            if no_imp % 30 == 0:
                inflate_cur = min(1.005, inflate_cur * 0.92)
            if no_imp > max(150, n):
                a = best_a.copy()
                inflate_cur = inflate
                no_imp = 0

    return best_a


def _spectral_init(n, seed=0):
    """Generate sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    n_fft = _next_pow2(2 * n - 1)
    power = rng.uniform(0.3, 1.5, n_fft // 2 + 1)
    power[0] = 0.0
    phases = rng.uniform(0.0, 2.0 * np.pi, n_fft // 2 + 1)
    spectrum = np.sqrt(power) * np.exp(1j * phases)

    full = np.zeros(n_fft, dtype=complex)
    full[:n_fft // 2 + 1] = spectrum
    for k in range(1, min(n_fft // 2, n_fft - 1)):
        full[n_fft - k] = np.conj(full[k])

    a_full = np.fft.irfft(full, n_fft).real
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, sep=0.25):

    a = (np.exp(-0.5 * ((t - 0.5 + sep) / 0.12) ** 2) +
         np.exp(-0.5 * ((t - 0.5 - sep) / 0.12) ** 2))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_triangle(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64)
    n_old = a.size
    if n_old == n_new:
        s = a.sum()
        return a / s if s > 1e-15 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    return a_new / s if s > 1e-15 else np.ones(n_new) / n_new


def _get_candidates(n_t, parent, rng, seed_offset=0):
    """Generate diverse initial candidates for a given length."""
    cands = []
    if parent is not None:
        cands.append(_interpolate_to_length(parent, n_t))
        p_pert = _interpolate_to_length(parent, n_t)
        p_pert = p_pert * (1.0 + 0.05 * rng.normal(size=n_t))
        cands.append(_normalize(p_pert))
        p_pert2 = _interpolate_to_length(parent, n_t)
        p_pert2 = p_pert2 * (1.0 + 0.12 * rng.normal(size=n_t))
        cands.append(_normalize(p_pert2))

    cands.append(_make_raised_cosine(n_t))
    cands.append(_make_gaussian(n_t, 0.35))
    cands.append(_make_gaussian(n_t, 0.25))
    cands.append(_make_bimodal(n_t, 0.25))
    cands.append(_make_bimodal(n_t, 0.15))
    cands.append(_make_triangle(n_t))

    for s in range(2):
        cands.append(_spectral_init(n_t, seed=seed_offset + s * 7 + 1))

    return cands


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining dual fixed-point, adaptive top-K
    gradient, and LP refinement with diverse length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Lengths to explore
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 1024, 2048]:
        if ln not in lengths:
            lengths.append(ln)
    # Sort by priority (parent length first, then medium)
    lengths.sort(key=lambda x: (x != (parent.size if parent is not None else -1), abs(x - 1024)))

    print(f"Lengths to try: {lengths}")

    best_per_length = {}

    # ===== Phase 1: Dual fixed-point exploration =====
    print("\n=== Phase 1: Dual fixed-point exploration ===")
    for n_t in lengths:
        if time.monotonic() > deadline - 400:
            break

        cands = _get_candidates(n_t, parent, rng, seed_offset=n_t)

        # Score and sort
        scored = []
        for a0 in cands:
            sc0 = _score_norm(a0)
            scored.append((sc0, a0))
        scored.sort(key=lambda x: x[0])

        # Run dual fixed-point on top 2 candidates
        for ci in range(min(2, len(scored))):
            sc0, a0 = scored[ci]
            if time.monotonic() > deadline - 400:
                break

            rem = deadline - time.monotonic() - 400
            fp_budget = min(rem / max(1, min(2, len(scoreed)) - ci), 120.0)
            if fp_budget < 10:
                break
            fp_end = time.monotonic() + fp_budget

            a_fp = _dual_fixed_point(a0, fp_end, tau_start=0.08, tau_end=0.002, damping=0.4)
            sc_fp = _score_norm(a_fp)

            if np.isfinite(sc_fp) and sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()

            if n_t not in best_per_length or sc_fp < _score_norm(best_per_length[n_t]):
                best_per_length[n_t] = a_fp

            print(f"  n={n_t}, cand {ci} (init={sc0:.6f}): dual_fp={sc_fp:.10f}")

    # ===== Phase 2: Top-K gradient optimization =====
    print("\n=== Phase 2: Top-K gradient optimization ===")
    for n_t in lengths:
        if time.monotonic() > deadline - 350:
            break

        if n_t not in best_per_length:
            continue

        a_start = best_per_length[n_t]
        rem = deadline - time.monotonic() - 350
        grad_budget = min(rem / max(1, len(lengths)), 150.0)
        if grad_budget < 15:
            continue
        grad_end = time.monotonic() + grad_budget

        # Additive gradient
        a_grad = _topk_gradient_optimize(a_start, grad_end, lr=0.04, K=10,
                                         method='additive', tau_adaptive=True)
        sc_grad = _score_norm(a_grad)

        if np.isfinite(sc_grad) and sc_grad < best_score:
            best_score = sc_grad
            best_seq = a_grad.copy()

        if n_t not in best_per_length or sc_grad < _score_norm(best_per_length[n_t]):
            best_per_length[n_t] = a_grad

        print(f"  n={n_t}: gradient={sc_grad:.10f}")

        # Also try multiplicative on a different candidate
        if time.monotonic() < deadline - 350:
            cands = _get_candidates(n_t, parent, rng, seed_offset=n_t + 100)
            a_alt = cands[min(3, len(cands) - 1)]
            alt_end = time.monotonic() + min(40.0, deadline - time.monotonic() - 350)
            if alt_end > time.monotonic() + 10:
                a_alt_opt = _topk_gradient_optimize(a_alt, alt_end, lr=0.03, K=15,
                                                    method='multiplicative', tau_adaptive=True)
                sc_alt = _score_norm(a_alt_opt)
                if np.isfinite(sc_alt) and sc_alt < best_score:
                    best_score = sc_alt
                    best_seq = a_alt_opt.copy()
                if n_t not in best_per_length or sc_alt < _score_norm(best_per_length[n_t]):
                    best_per_length[n_t] = a_alt_opt
                print(f"  n={n_t}: mult_gradient={sc_alt:.10f}")

    # ===== Phase 3: LP refinement =====
    print("\n=== Phase 3: LP refinement ===")
    for n_t in sorted(best_per_length.keys(), key=lambda x: _score_norm(best_per_length[x])):
        if time.monotonic() > deadline - 200:
            break

        a_in = best_per_length[n_t]
        sc_in = _score_norm(a_in)

        rem = deadline - time.monotonic() - 200
        lp_budget = min(rem * 0.5, 180.0)
        if lp_budget < 15:
            break
        lp_end = time.monotonic() + lp_budget

        a_lp = _lp_refine(a_in, lp_end, max_iter=2000, inflate=1.0008, top_k=35)
        sc_lp = _score_norm(a_lp)

        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp

        print(f"  n={n_t}: pre={sc_in:.10f} -> LP={sc_lp:.10f}")

    # ===== Phase 4: Second dual fixed-point on best =====
    print("\n=== Phase 4: Second dual pass on best ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = best_seq.size
        fp_end = time.monotonic() + min(80.0, deadline - time.monotonic() - 100)
        if fp_end > time.monotonic() + 10:
            a_fp2 = _dual_fixed_point(best_seq, fp_end, tau_start=0.03,
                                      tau_end=0.0005, damping=0.3)
            sc_fp2 = _score_norm(a_fp2)
            if np.isfinite(sc_fp2) and sc_fp2 < best_score:
                best_score = sc_fp2
                best_seq = a_fp2.copy()
            print(f"  Dual pass 2: score={sc_fp2:.10f}")

    # ===== Phase 5: Fine gradient polish =====
    print("\n=== Phase 5: Fine gradient polish ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        pol_end = time.monotonic() + min(60.0, deadline - time.monotonic() - 40)
        if pol_end > time.monotonic() + 10:
            a_pol = _topk_gradient_optimize(best_seq, pol_end, lr=0.015, K=5,
                                            method='additive', tau_adaptive=True)
            sc_pol = _score_norm(a_pol)
            if np.isfinite(sc_pol) and sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
            print(f"  Polish: score={sc_pol:.10f}")

    # ===== Phase 6: Final LP micro-refinement =====
    if best_seq is not None and time.monotonic() < deadline - 15:
        lp_end = time.monotonic() + min(30.0, deadline - time.monotonic() - 15)
        if lp_end > time.monotonic() + 5:
            a_fin = _lp_refine(best_seq, lp_end, max_iter=500, inflate=1.0003, top_k=20)
            sc_fin = _score_norm(a_fin)
            if np.isfinite(sc_fin) and sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
            print(f"  Final LP: score={sc_fin:.10f}")

    # Ensure valid output
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0

    # Validate with authoritative evaluator
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
