def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)

    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_solve(a, M_constraint, n):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M_constraint, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M_constraint)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=50000, init_inflate=1.0005, init_blend=0.2):
    """Aggressive LP refinement with adaptive hyperparameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    inflate = init_inflate
    blend = init_blend
    no_imp = 0
    imp_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _lp_solve(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 5 == 0:
                inflate = min(1.02, inflate * 1.03)
            if no_imp > 300:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        sc_new = float(2.0 * n * M_new)

        if sc_new < best_score - 1e-13:
            best_score = sc_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            imp_count += 1
            inflate = max(1.0003, inflate * 0.98)
        else:
            no_imp += 1
            if no_imp % 10 == 0:
                blend = max(0.05, blend * 0.92)
                inflate = min(1.015, inflate * 1.02)
            if no_imp > max(100, 2 * n):
                a = best_a.copy()
                blend = init_blend
                inflate = init_inflate
                no_imp = 0

        a = a_new

    return best_a


def _subgrad_polish(a, deadline, max_iter=100000):
    """Subgradient descent on max(conv(a,a)) with simplex projection."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _simplex_project(a)

    best_a = a.copy()
    best_score = _score(a)

    lr = 0.05
    base_lr = lr
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))

        # Compute subgradient at peak
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        grad = np.zeros(n)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            grad[js] = a[k_star - js] + a[js]
            mid = k_star // 2
            if 2 * mid == k_star and j_lo <= mid <= j_hi:
                grad[mid] = 2.0 * a[mid]

        v = momentum * v - lr * grad
        a_new = _simplex_project(a + v)
        a = a_new

        sc = _score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > 5000:
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _smooth_max_gradient_polish(a, deadline, max_iter=50000):
    """Gradient descent using smooth max approximation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _simplex_project(a)

    best_a = a.copy()
    best_score = _score(a)
    beta = 0.02 * float(_fft_convolve(a).max())
    lr = 0.03
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # Gradient: grad_j = 2 * sum_k w_k * a_{k-j} (valid range)
        # Using FFT: cross-correlation of w and a
        N = _next_pow2(2 * n)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a, N)
        # IDFT(conj(fw) * fa)[j] = sum_k w[k] * a[k-j] (linear cross-corr)
        corr = np.fft.irfft(np.conj(fw) * fa, N)
        grad = 2.0 * corr[:n]

        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            beta *= 2.0
            continue

        v = momentum * v - lr * grad
        a_new = _simplex_project(a + v)
        a = a_new

        sc = _score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.95)
            if no_imp > 5000:
                a = best_a.copy()
                v = np.zeros(n)
                beta = 0.02 * float(_fft_convolve(a).max())
                lr = base_lr
                no_imp = 0

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.85)

    return best_a


def _peak_shave(a, deadline, max_iter=20000):
    """Targeted peak reduction: move mass from peak-contributing positions."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))

        # Find the pair (i, j) with i+j = k_star that maximizes a[i]*a[j]
        i_lo = max(0, k_star - n + 1)
        i_hi = min(k_star, n - 1)
        if i_lo > i_hi:
            break

        js = np.arange(i_lo, i_hi + 1)
        products = a[js] * a[k_star - js]
        idx_best = js[int(np.argmax(products))]

        # Find target position: one that contributes least to any high lag
        # Simply pick a random position with small value
        candidates = np.where(a < 0.3 * a.max())[0]
        if len(candidates) == 0:
            candidates = np.where(a < a.mean())[0]
        if len(candidates) == 0:
            break

        target = candidates[rng.integers(0, len(candidates))]
        if target == idx_best:
            continue

        # Move some mass
        frac = 0.1
        delta = frac * a[idx_best]
        a_new = a.copy()
        a_new[idx_best] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break

    return best_a


def _adam_logit_polish(a, deadline, max_iter=50000, lr=0.03):
    """Adam optimization in logit space with smooth max objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_score = _score(a)
    beta = 0.03 * float(_fft_convolve(a).max())
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # Gradient in probability space
        N = _next_pow2(2 * n)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p, N)
        corr = np.fft.irfft(np.conj(fw) * fp, N)
        vp = 2.0 * corr[:n] + 1e-8 * p

        # Transform to logit gradient
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = np.log(np.maximum(best_a, 1e-12))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_a, 1e-12))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        sc = _score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.8)

        if no_imp > 5000:
            rng_r = np.random.default_rng(it * 7919 + n)
            z = 0.6 * np.log(np.maximum(best_a, 1e-12)) + 0.4 * rng_r.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = 0.03 * float(_fft_convolve(a).max())
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP refinement (primary), subgradient polish,
    smooth-max gradient, peak shaving, and Adam logit optimization.
    Tries parent length and nearby lengths with diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is None:
        parent = _make_linear_profile(768, 1.5)
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, using linear init: score={best_score:.10f}")

    n_parent = parent.size

    # Primary length: parent length
    # Secondary lengths to try
    lengths_to_try = [n_parent]
    for ln in [512, 640, 1024, 1280]:
        if ln != n_parent and ln not in lengths_to_try:
            lengths_to_try.append(ln)

    print(f"Lengths to try: {lengths_to_try}")
    print(f"Time budget: {budget_s}s")

    # PHASE 1: Primary LP refinement on parent length
    # Allocate ~50% of time to this
    phase1_budget = (deadline - t_start) * 0.50
    phase1_deadline = t_start + phase1_budget

    print(f"\n=== Phase 1: LP refinement (n={n_parent}), budget={phase1_budget:.0f}s ===")

    a_work = parent.copy()
    a_work = a_work / a_work.sum()

    # LP round 1: generous
    a1 = _lp_refine(a_work, phase1_deadline - phase1_budget * 0.3,
                    max_iter=30000, init_inflate=1.0003, init_blend=0.15)
    sc1 = _score(a1)
    print(f"  LP round 1: {sc1:.10f}")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()

    # Subgradient polish
    rem1 = phase1_deadline - time.monotonic()
    if rem1 > 20:
        a2 = _subgrad_polish(a1, time.monotonic() + min(rem1 * 0.5, 120), max_iter=50000)
        sc2 = _score(a2)
        print(f"  Subgrad: {sc2:.10f}")
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            a1 = a2.copy()

    # LP round 2: tighter
    rem1 = phase1_deadline - time.monotonic()
    if rem1 > 20:
        a3 = _lp_refine(a1, time.monotonic() + rem1 * 0.7,
                        max_iter=20000, init_inflate=1.0002, init_blend=0.1)
        sc3 = _score(a3)
        print(f"  LP round 2: {sc3:.10f}")
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()

    # PHASE 2: Try other lengths
    phase2_budget = (deadline - t_start) * 0.25
    phase2_deadline = t_start + min(deadline, phase1_deadline + phase2_budget)

    print(f"\n=== Phase 2: Other lengths, deadline in {phase2_deadline - time.monotonic():.0f}s ===")

    for n_t in lengths_to_try[1:]:
        if time.monotonic() > phase2_deadline - 30:
            break

        a_init = _interpolate_profile(parent, n_t)
        sc_init = _score(a_init)
        print(f"  n={n_t}: init score={sc_init:.10f}")

        if sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()

        # Quick LP + polish
        sub_budget = min(120.0, phase2_deadline - time.monotonic() - 20)
        if sub_budget < 15:
            continue

        a_lp = _lp_refine(a_init, time.monotonic() + sub_budget * 0.6,
                          max_iter=10000, init_inflate=1.0005, init_blend=0.2)
        sc_lp = _score(a_lp)
        print(f"  n={n_t}: LP score={sc_lp:.10f}")

        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()

        # Quick subgradient
        rem = phase2_deadline - time.monotonic() - 10
        if rem > 10:
            a_sg = _subgrad_polish(a_lp, time.monotonic() + min(rem * 0.5, 60), max_iter=30000)
            sc_sg = _score(a_sg)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"  n={n_t}: subgrad score={sc_sg:.10f}")

    # PHASE 3: Intensive polish on the best
    print(f"\n=== Phase 3: Intensive polish ===")
    print(f"  Current best: {best_score:.10f}, n={best_seq.size}")

    remaining = deadline - time.monotonic() - 10
    if remaining > 30:
        # Smooth max gradient
        b1 = min(remaining * 0.3, 150.0)
        if b1 > 15:
            a_sm = _smooth_max_gradient_polish(best_seq, time.monotonic() + b1, max_iter=30000)
            sc_sm = _score(a_sm)
            print(f"  SmoothMax: {sc_sm:.10f}")
            if sc_sm < best_score:
                best_score = sc_sm
                best_seq = a_sm.copy()

        # LP final
        b2 = min(remaining * 0.3, 120.0)
        if b2 > 15:
            a_lp2 = _lp_refine(best_seq, time.monotonic() + b2,
                               max_iter=20000, init_inflate=1.0002, init_blend=0.08)
            sc_lp2 = _score(a_lp2)
            print(f"  LP final: {sc_lp2:.10f}")
            if sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = a_lp2.copy()

        # Adam logit
        b3 = min(remaining * 0.2, 100.0)
        if b3 > 15:
            a_ad = _adam_logit_polish(best_seq, time.monotonic() + b3, max_iter=30000, lr=0.02)
            sc_ad = _score(a_ad)
            print(f"  Adam: {sc_ad:.10f}")
            if sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()

        # Peak shave
        b4 = min(remaining * 0.1, 60.0)
        if b4 > 10:
            a_ps = _peak_shave(best_seq, time.monotonic() + b4, max_iter=15000)
            sc_ps = _score(a_ps)
            print(f"  PeakShave: {sc_ps:.10f}")
            if sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()

        # One more LP with very tight params
        rem_final = deadline - time.monotonic() - 5
        if rem_final > 20:
            a_final = _lp_refine(best_seq, time.monotonic() + rem_final,
                                 max_iter=20000, init_inflate=1.0001, init_blend=0.05)
            sc_final = _score(a_final)
            print(f"  Final LP: {sc_final:.10f}")
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()

    # PHASE 4: Try diverse initializations if time remains
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Diverse initializations ===")
        n_best = best_seq.size
        inits = []
        inits.append(_make_linear_profile(n_best, 2.0))
        inits.append(_make_linear_profile(n_best, 0.8))
        inits.append(_make_u_profile(n_best, 3.0))
        inits.append(_make_quadratic_profile(n_best, 1.0, 0.5, 0.5))
        # Perturbed best
        a_pert = best_seq * (1.0 + 0.1 * rng.normal(size=n_best))
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        inits.append(a_pert)

        for a0 in inits:
            if time.monotonic() > deadline - 30:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
                print(f"  Better init: {sc0:.10f}")
                continue

            # Quick LP
            sub_rem = deadline - time.monotonic() - 20
            if sub_rem < 15:
                break
            a_q = _lp_refine(a0, time.monotonic() + min(sub_rem * 0.5, 60),
                             max_iter=5000, init_inflate=1.001, init_blend=0.2)
            sc_q = _score(a_q)
            if sc_q < best_score:
                best_score = sc_q
                best_seq = a_q.copy()
                print(f"  Diverse LP: {sc_q:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(768, 1.5)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = final_seq / 0.001
        s = float(final_seq.sum())
    # Normalize to ensure good behavior
    if s > 100:
        final_seq = final_seq / s
        final_seq = final_seq * s  # keep same scale, just ensure sum > 0.01

    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
