    # For peak at index k: g[i] = 2*a[k-i] if 0 <= k-i < n
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]
    return g

def subgradient_at_peaks(a, peak_indices):
    """Compute averaged subgradient at multiple peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in peak_indices:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * a[j]
    g /= len(peak_indices)
    return g

def find_active_peaks(c, max_c, tol=1e-6):
    """Find indices where c is within tol of the maximum."""
    threshold = max_c - max_c * tol
    return np.where(c >= threshold)[0]

def solve_lp_direction(f, M, n, max_constraints=2048, tol=1e-8):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and conv(f, g)[k] <= M for all k.
    Uses constraint generation for efficiency.
    """
    # Start with a subset of constraints
    # The most important constraints are near the peaks of conv(f, f)
    c_ff = full_convolve_fft(f)
    
    # Start with top K constraints
    K = min(100, 2 * n - 1)
    peak_idx = np.argsort(c_ff)[-K:]
    
    active_constraints = set(peak_idx.tolist())
    
    for iteration in range(50):
        # Build constraint matrix for active constraints
        m_constraints = len(active_constraints)
        A_ub = np.zeros((m_constraints, n))
        b_ub = np.full(m_constraints, M)
        
        for row_idx, k in enumerate(active_constraints):
            for j in range(n):
                jj = k - j
                if 0 <= jj < n:
                    A_ub[row_idx, j] = f[jj]
        
        # Solve LP
        c_obj = -np.ones(n)  # maximize sum = minimize -sum
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, 
                      method='highs', options={'presolve': True})
        
        if not res.success:
            return None
        
        g = res.x
        
        # Check if all constraints are satisfied
        c_fg = full_convolve_fft(f)  # Wait, this is conv(f,f), not conv(f,g)
        # I need conv(f, g)
        c_fg = np.convolve(f, g)
        
        violated = np.where(c_fg > M + tol)[0]
        
        if len(violated) == 0:
            return g
        
        # Add violated constraints
        for v in violated:
            active_constraints.add(int(v))
        
        if len(active_constraints) > max_constraints:
            # Keep only the most violated
            violations = c_fg[active_constraints] - M
            sorted_idx = np.argsort(violations)[-max_constraints:]
            active_constraints = set([list(active_constraints)[i] for i in sorted_idx])
    
    return None

def lp_refine(f, deadline, rng, max_lp_iters=50, step_candidates=None):
    """
    Refine f using LP-based direction finding.
    """
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    f = np.maximum(f, 0.0)
    s = f.sum()
    f = f / s
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    if step_candidates is None:
        step_candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
    
    for it in range(max_lp_iters):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve_fft(f)
        M = float(c.max())
        
        # Solve LP
        t0 = time.monotonic()
        g = solve_lp_direction(f, M, n)
        t_lp = time.monotonic() - t0
        
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        # Rescale g to have same sum as f
        g_rescaled = g * (s / g_sum)
        
        # Try different step sizes
        improved = False
        for t in step_candidates:
            f_new = (1 - t) * f + t * g_rescaled
            f_new = np.maximum(f_new, 0.0)
            f_new_sum = f_new.sum()
            if f_new_sum < 0.01:
                continue
            score_new = compute_score(f_new)
            if score_new < best_score:
                best_f = f_new.copy()
                best_score = score_new
                f = f_new
                improved = True
                break
        
        s = f.sum()
        
        if not improved:
            # Try smaller steps
            for t in [0.01, 0.005, 0.001]:
                f_new = (1 - t) * f + t * g_rescaled
                f_new = np.maximum(f_new, 0.0)
                f_new_sum = f_new.sum()
                if f_new_sum < 0.01:
                    continue
                score_new = compute_score(f_new)
                if score_new < best_score:
                    best_f = f_new.copy()
                    best_score = score_new
                    f = f_new
                    improved = True
                    break
        
        if not improved:
            break
        
        print(f"    LP iter {it}: score={best_score:.10f}, t_lp={t_lp:.2f}s")
    
    return best_f, best_score

def subgradient_refine(f, deadline, rng, lr=0.001, momentum=0.9, 
                       peak_tol=1e-4, restart_period=500):
    """
    Subgradient descent with exact peak tracking and momentum.
    """
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    
    z = np.log(f)
    z = z - z.mean()
    
    best_f = f.copy()
    best_score = compute_score(f)
    best_z = z.copy()
    
    vel = np.zeros(n)
    no_improve = 0
    current_lr = lr
    base_lr = lr
    
    iters = 0
    while time.monotonic() < deadline:
        c = full_convolve_fft(f)
        max_c = float(c.max())
        
        # Find active peaks
        peaks = find_active_peaks(c, max_c, tol=peak_tol)
        
        # Compute subgradient
        g = np.zeros(n)
        for k in peaks:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += 2.0 * f[j]
        g /= len(peaks)
        
        # Project to tangent space of simplex (in logit space)
        dot = float(np.dot(f, g))
        dz = f * (g - dot)
        
        # Momentum
        vel = momentum * vel + (1 - momentum) * dz
        z = z - current_lr * vel
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        f_new = np.exp(z - z.max())
        f_sum = f_new.sum()
        if f_sum < 1e-15:
            z = best_z.copy()
            f = best_f.copy()
            vel = np.zeros(n)
            continue
        f_new = f_new / f_sum
        
        score = compute_score(f_new)
        
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            best_z = z.copy()
            f = f_new
            no_improve = 0
        else:
            # Roll back
            z = z + current_lr * vel  # undo
            z = z - z.mean()
            no_improve += 1
        
        iters += 1
        
        # Adapt learning rate
        if no_improve % 100 == 0:
            current_lr = max(1e-5, current_lr * 0.98)
        
        # Restart
        if no_improve > restart_period:
            z = best_z + 0.1 * current_lr * 100 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            vel = np.zeros(n)
            current_lr = base_lr
            no_improve = 0
    
    return best_f, best_score

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

def softmax_gradient_refine(a0, deadline, rng, lr=0.03, beta_factor=0.05,
                            beta_decay=0.93, beta_min=1e-8, restart_period=1000):
    """
    Logit-space optimization with soft-max (similar to previous but with
    different hyperparameters and adaptive peak tracking).
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-10:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        # Soft max weights
        mval = max_c
        x = (c - mval) / beta
        x = np.clip(x, -60.0, 0.0)
        ex = np.exp(x)
        s_ex = float(np.sum(ex))
        if s_ex < 1e-300:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        # Gradient
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # Adam
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 30 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 200:
            current_lr = max(0.003, current_lr * 0.97)

        if no_improve > restart_period:
            z = best_z + 0.15 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.06 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score

def coordinate_descent_refine(f, deadline, rng, n_iterations_per_round=100):
    """
    Coordinate descent: try to improve one coordinate at a time.
    For each coordinate i, find the optimal value of f[i] (keeping others fixed)
    that minimizes the max of the convolution.
    """
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    f = np.maximum(f, 1e-12)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    rng_perm = rng.permutation(n)
    
    while time.monotonic() < deadline:
        improved = False
        for idx in range(len(rng_perm)):
            i = rng_perm[idx]
            
            # Try a few values for f[i]
            # Current contribution to convolution from coordinate i:
            # conv(a,a)[k] includes terms a[i]*a[k-i] and a[k-i]*a[i]
            # So changing a[i] by delta changes conv by delta * (a[k-i] + a[i-k]) for each k
            
            # Binary search for optimal f[i]
            # Actually, let's just try a few multiplicative changes
            for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
                f_trial = f.copy()
                f_trial[i] = f[i] * factor
                f_trial = np.maximum(f_trial, 1e-15)
                # Renormalize
                f_trial = f_trial / f_trial.sum()
                
                score_trial = compute_score(f_trial)
                if score_trial < best_score - 1e-12:
                    f = f_trial
                    best_f = f.copy()
                    best_score = score_trial
                    improved = True
                    break
        
        if not improved:
            # Shuffle and try again
            rng_perm = rng.permutation(n)
    
    return best_f, best_score

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP-guided direction finding, soft-max gradient descent in logit space,
exact subgradient methods, and coordinate descent for final polishing.
Uses adaptive multi-phase time allocation and constraint generation for LP.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def solve_lp_constrained(f, M, n, max_constraints=1536, max_iters=30):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and conv(f, g)[k] <= M for all k.
    Uses constraint generation for efficiency.
    Returns g or None if infeasible.
    """
    c_ff = full_convolve_fft(f)

    K_init = min(200, 2 * n - 1)
    top_idx = np.argsort(c_ff)[-K_init:]
    active_set = list(top_idx)

    g = None
    for iteration in range(max_iters):
        m_c = len(active_set)
        A_ub = np.zeros((m_c, n))
        for row_i, k in enumerate(active_set):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                A_ub[row_i, j] = f[k - j]

        c_obj = -np.ones(n)
        b_ub = np.full(m_c, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-9,
                                                   'primal_feasibility_tolerance': 1e-9})
        except Exception:
            return None

        if not res.success:
            return None

        g = res.x

        c_fg = np.convolve(f, g)
        violations = c_fg - M
        violated = np.where(violations > 1e-8)[0]

        if len(violated) == 0:
            return g

        new_violated = [int(v) for v in violated if int(v) not in active_set]
        for v in new_violated:
            active_set.append(v)

        if len(active_set) > max_constraints:
            active_arr = np.array(active_set)
            viol_vals = violations[active_arr]
            keep_idx = np.argsort(viol_vals)[-max_constraints:]
            active_set = [int(active_arr[i]) for i in keep_idx]

    return g

def softmax_adam_refine(a0, deadline, rng, lr=0.03, beta_factor=0.06,

    """

    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()

    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-10:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        mval = max_c
        x = (c - mval) / beta
        x = np.clip(x, -60.0, 0.0)
        ex = np.exp(x)
        s_ex = float(np.sum(ex))
        if s_ex < 1e-300:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 30 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 200:
            current_lr = max(0.003, current_lr * 0.97)

        if no_improve > restart_period:
            z = best_z + 0.12 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.05 * max_c)
            m.fill(0.0)
            v.fill(0.0)

            no_improve = 0

def exact_subgradient_refine(a0, deadline, rng, lr=0.0003, momentum=0.93,
                             peak_tol=5e-4, patience=600):
    """
    Exact subgradient descent in logit space with momentum.
    Tracks all near-maximum peaks and averages their subgradients.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    best_a = a0.copy()
    best_z = z.copy()
    best_score = compute_score(a0)

    vel = np.zeros(n)
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())

        threshold = max_c * (1.0 - peak_tol)
        peaks = np.where(c >= threshold)[0]

        g = np.zeros(n)
        for k in peaks:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += 2.0 * a0[j]
        g /= float(len(peaks))

        dot = float(np.dot(a0, g))
        dz = a0 * (g - dot)

        vel = momentum * vel + (1.0 - momentum) * dz
        z = z - current_lr * vel
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a_new = np.exp(z - z.max())
        s = a_new.sum()
        if s < 1e-15:
            z = z + current_lr * vel
            z = z - z.mean()
            continue
        a_new = a_new / s

        score = compute_score(a_new)
        if np.isfinite(score) and score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            best_z = z.copy()
            a0 = a_new
            no_improve = 0
        else:
            z = z + current_lr * vel
            z = z - z.mean()
            no_improve += 1

        if no_improve % 200 == 0:
            current_lr = max(1e-6, current_lr * 0.95)

        if no_improve > patience:
            z = best_z + 0.05 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            vel = np.zeros(n)

def coordinate_descent_polish(f, deadline, rng, n_factors=5):

        do_update(init_seq, sc_init)

    n_work = init_seq.size
    if n_work > 2048:
        n_work = 2048
        p_work = interpolate_profile(init_seq, n_work)
    elif n_work < 512:
        n_work = 512
        p_work = interpolate_profile(init_seq, n_work)
    else:
        p_work = init_seq.copy()

    # ===== Phase 1: Quick softmax-gradient (200s) =====
    print(f"\n--- Phase 1: Softmax gradient (n={n_work}) ---")
    p1_deadline = min(deadline, t_start + 200.0)

    r1, s1 = softmax_adam_refine(p_work, p1_deadline, rng,
                                  lr=0.04, beta_factor=0.08, beta_decay=0.92)
    s1f = compute_score(r1)
    do_update(r1, s1f)
    print(f"  Phase 1 best: {s1f:.10f}")

    # Also try a few diverse starts quickly
    if time.monotonic() < p1_deadline - 10:
        starts = diverse_starts(n_work, rng)[:4]
        for idx, st in enumerate(starts):
            if time.monotonic() > p1_deadline - 5:
                break
            dl = min(p1_deadline, time.monotonic() + 30.0)
            r, s = softmax_adam_refine(st, dl, rng, lr=0.05, beta_factor=0.1)
            sf = compute_score(r)
            do_update(r, sf)
            if sf < best_score + 0.02:
                print(f"  Alt start {idx}: {sf:.8f}")

    # ===== Phase 2: LP-guided refinement (400s) =====
    print(f"\n--- Phase 2: LP refinement ---")
    p2_deadline = min(deadline, t_start + 650.0)

    p2_base = best_seq.copy() if best_seq is not None else r1.copy()
    sc_p2_init = compute_score(p2_base)
    print(f"  Starting LP from score={sc_p2_init:.10f}")

    max_lp = 25 if n_work <= 1024 else 15
    max_con = 1200 if n_work <= 1024 else 800
    r2, s2 = lp_direction_refine(p2_base, p2_deadline,
                                  max_lp_iters=max_lp, max_constraints=max_con)
    s2f = compute_score(r2)
    do_update(r2, s2f)
    print(f"  Phase 2 best: {s2f:.10f}")

    # ===== Phase 3: Exact subgradient polishing (350s) =====
    print(f"\n--- Phase 3: Subgradient polish ---")
    p3_deadline = min(deadline, t_start + 1100.0)

    p3_base = best_seq.copy() if best_seq is not None else r2.copy()
    r3, s3 = exact_subgradient_refine(p3_base, p3_deadline, rng,
                                       lr=0.0004, momentum=0.94,
                                       peak_tol=3e-4, patience=500)
    s3f = compute_score(r3)
    do_update(r3, s3f)
    print(f"  Phase 3 best: {s3f:.10f}")

    # ===== Phase 4: Coordinate descent (200s) =====
    print(f"\n--- Phase 4: Coordinate descent ---")
    p4_deadline = min(deadline, t_start + 1500.0)

    p4_base = best_seq.copy() if best_seq is not None else r3.copy()
    r4, s4 = coordinate_descent_polish(p4_base, p4_deadline, rng)
    s4f = compute_score(r4)
    do_update(r4, s4f)
    print(f"  Phase 4 best: {s4f:.10f}")

    # ===== Phase 5: Final softmax with very small steps (remaining) =====
    print(f"\n--- Phase 5: Final refinement ---")

    if best_seq is not None:
        p5 = best_seq.copy()
    else:
        p5 = r4.copy()

    # Round 1: moderate
    if time.monotonic() < deadline - 60:
        dl5a = min(deadline, time.monotonic() + 120.0)
        p5a = p5 + 5e-5 * rng.normal(size=p5.size)
        p5a = np.maximum(p5a, 1e-12)
        p5a = p5a / p5a.sum()
        r5a, s5a = softmax_adam_refine(p5a, dl5a, rng, lr=0.01,
                                        beta_factor=0.02, beta_decay=0.96)
        s5af = compute_score(r5a)
        do_update(r5a, s5af)
        print(f"  Final round 1: {s5af:.10f}")

    # Round 2: very small steps
    if time.monotonic() < deadline - 40:
        p5b = best_seq.copy() if best_seq is not None else p5.copy()
        dl5b = min(deadline, time.monotonic() + 80.0)
        r5b, s5b = exact_subgradient_refine(p5b, dl5b, rng,
                                             lr=0.0001, momentum=0.96,
                                             peak_tol=1e-4, patience=400)
        s5bf = compute_score(r5b)
        do_update(r5b, s5bf)
        print(f"  Final round 2: {s5bf:.10f}")

    # Try different n
    if best_seq is not None:
        for n_try in [1024, 2048, 4096]:
            if time.monotonic() > deadline - 30:
                break
            if n_try == best_seq.size:
                continue
            p_try = interpolate_profile(best_seq, n_try)
            sc_try = compute_score(p_try)
            do_update(p_try, sc_try)
            if sc_try < best_score + 0.005 and time.monotonic() < deadline - 20:
                dl_try = min(deadline, time.monotonic() + 25.0)
                r_try, _ = softmax_adam_refine(p_try, dl_try, rng,
                                                lr=0.02, beta_factor=0.04)
                sc_try2 = compute_score(r_try)
                do_update(r_try, sc_try2)
                print(f"  n={n_try}: {sc_try2:.10f}")

    print(f"\n  FINAL best score: {best_score:.10f}")

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(1024, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s_final = float(final_seq.sum())
    if s_final > 0.01:
        final_seq = final_seq / s_final

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Eval check: {e}")

    return [float(x) for x in final_seq]
