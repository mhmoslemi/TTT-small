"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: FFT-based smooth-max gradient descent (Adam) in logit space with adaptive
temperature, diverse profile initialization at multiple scales (n=1024..8192),
LP-based best-response refinement, multi-restart with noise injection, and
multi-scale escalation. Targets score < 1.5027.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a, mode='full')
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _score_and_conv(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max()), c


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    max_x = float(x.max())
    ex = np.exp(x - max_x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s) + max_x)
    w = ex / s
    return val, w


def _cross_corr_fft(a, w):
    """Compute corr[j] = sum_m a[m]*w[j+m] for j=0..len(a)-1 via FFT."""
    n = len(a)
    L = len(w)
    N = _next_pow2(n + L - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(np.conj(fa) * fw, N)
    return corr[:n]


def _grad_smooth_max(w, a):
    """Gradient of smooth_max(conv(a)) w.r.t. a."""
    return 2.0 * _cross_corr_fft(a, w)


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _make_uniform(n):
    return np.ones(n) / n


def _make_exponential(n, lam):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_parabolic(n):
    t = np.linspace(0.0, 1.0, n)
    a = t * (1.0 - t) + 1e-6
    return a / a.sum()


def _make_sine(n, freq=1, amp=0.7):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine(n, freq=1, amp=0.7):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_two_bump(n, c1, c2, sigma):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / sigma) ** 2) + np.exp(-0.5 * ((t - c2) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_alt(n, ratio=2.0):
    a = np.ones(n)
    a[::2] = ratio
    return _normalize_nonneg(a)


def _make_random_smooth(n, rng, n_basis=16):
    coeffs = rng.normal(size=n_basis)
    t = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for k in range(1, n_basis):
        a += coeffs[k] * (k + 1) ** (-1.0) * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    if n0 == 1:
        return np.full(n_new, 1.0 / n_new)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.95,
                   no_imp_limit=500, noise_scale=None, max_iter=None):
    """Adam smooth-max optimization in logit space."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n = p0.size
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    it = 0

    while time.monotonic() < deadline:
        if max_iter is not None and it >= max_iter:
            break
        it += 1

        _, w = _smooth_max_weights(c, beta)
        g = _grad_smooth_max(w, p)

        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.93)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(5e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(5e-5, beta * 0.5)

        limit = max(50, no_imp_limit)
        if no_imp > limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.12
                elif best_score < 1.53:
                    ns = 0.2
                else:
                    ns = 0.4
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=15):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.1, t * 0.85)
    return current_best, current_score

    cands.append(('cos2', _make_cosine(n, 2, 0.7)))
    cands.append(('cos3', _make_cosine(n, 3, 0.5)))
    cands.append(('twobump', _make_two_bump(n, 0.3, 0.7, 0.15)))
    cands.append(('twobump2', _make_two_bump(n, 0.25, 0.75, 0.2)))
    cands.append(('alt2', _make_alt(n, 2.0)))
    cands.append(('alt3', _make_alt(n, 3.0)))
    for i in range(4):
        cands.append((f'rand{i}', _make_random_smooth(n, rng)))
    if parent is not None and include_parent:
        pi = _interp_profile(parent, n)
        if pi is not None:
            cands.append(('parent', pi))
    return cands


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s, seed={seed}")

    # === Phase 1: Generate and evaluate initial candidates ===
    phase1_end = t_start + 90.0
    all_cands = []

    for n in [512, 1024, 2048]:
        cands = _generate_candidates(n, rng, include_parent=True, parent=parent)
        for label, p in cands:
            sc = _score_norm(p)
            all_cands.append((sc, p, n, label))

    if parent is not None and 0 < pn <= 8192:
        sc = _score_norm(parent)
        all_cands.append((sc, parent.copy(), pn, 'parent_orig'))

    all_cands.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(all_cands)} candidates, best={all_cands[0][0]:.10f} n={all_cands[0][2]}")
    _update(all_cands[0][1], all_cands[0][0], 'init')

    # === Phase 2: Polish top candidates with Adam ===
    phase2_end = t_start + 700.0
    top_k = min(8, len(all_cands))
    polished = []
    for i in range(top_k):
        if time.monotonic() > phase2_end - 30:
            break
        sc, p, n, label = all_cands[i]
        per_budget = min(90.0, phase2_end - time.monotonic() - 10)
        if per_budget < 15:
            break
        p_b, sc_b = _adam_optimize(p, time.monotonic() + per_budget, rng,
                                   lr=0.06, beta_frac=0.08, beta_decay=0.96,
                                   no_imp_limit=400)
        _update(p_b, sc_b, f'polish_{label}_n{n}')
        polished.append((sc_b, p_b, n, label + '_pol'))

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: LP refinement on best small-n candidate ===
    phase3_end = t_start + 1000.0
    if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase3_end - 20:
        n_lp = best_seq.size
        print(f"  LP refine n={n_lp}, score={best_score:.10f}")
        lp_deadline = min(phase3_end, time.monotonic() + 180.0)
        p_lp, sc_lp = _lp_refine(best_seq, lp_deadline, t=0.35, max_solves=20)
        _update(p_lp, sc_lp, 'LP_refine')

        # Adam polish after LP
        if time.monotonic() < phase3_end - 15:
            per = min(90.0, phase3_end - time.monotonic() - 10)
            if per > 15:
                p_ap, sc_ap = _adam_optimize(p_lp, time.monotonic() + per, rng,
                                             lr=0.04, beta_frac=0.06, beta_decay=0.96,
                                             no_imp_limit=300)
                _update(p_ap, sc_ap, 'post_LP')

    print(f"Phase 3 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-scale escalation ===
    phase4_end = t_start + 1400.0
    if time.monotonic() < phase4_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048, 4096]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096, 8192]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            per = min(120.0, phase4_end - time.monotonic() - 30)
            if per > 20:
                p_e, sc_e = _adam_optimize(p_int, time.monotonic() + per, rng,
                                           lr=0.05, beta_frac=0.07, beta_decay=0.96,
                                           no_imp_limit=400)
                _update(p_e, sc_e, f'escalate_n{tn}')

            # Try LP at new scale if manageable
            if best_seq is not None and best_seq.size <= 2048:
                if time.monotonic() < phase4_end - 30:
                    lp_d = min(phase4_end - time.monotonic() - 10, 120.0)
                    p_lp2, sc_lp2 = _lp_refine(best_seq, time.monotonic() + lp_d,
                                               t=0.3, max_solves=10)
                    _update(p_lp2, sc_lp2, f'LP_n{best_seq.size}')

    print(f"Phase 4 done: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-restart at best length ===
    phase5_end = deadline - 40.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(30.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 5:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 15:
                break
            noise = 0.08 + 0.08 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_optimize(p_s, time.monotonic() + per, rng,
                                       lr=0.05, beta_frac=0.07, beta_decay=0.95,
                                       no_imp_limit=max(100, int(0.2 * n_best)))
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

    # === Phase 6: Final polish ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 15:
            p_f, sc_f = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                       lr=0.02, beta_frac=0.05, beta_decay=0.97,
                                       no_imp_limit=150)
            _update(p_f, sc_f, 'final_polish')

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_uniform(1024)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL CONFIRMED: {eval_sc:.10f}")
            else:
                print(f"EVAL FAILED: {eval_sc}")
        except Exception as e:
            print(f"EVAL ERROR: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_parabolic(1024)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
