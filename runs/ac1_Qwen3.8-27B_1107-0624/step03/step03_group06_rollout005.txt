"""
Multi-scale LP + Adam optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements: adaptive inflate with aggressive tightening, golden-section
line search, Adam gradient polish, multi-length exploration, and random
perturbation kicks to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-30:
        return a
    return a / s


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)


def _make_profile(n, profile_type='ramp', alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    if profile_type == 'ramp':
        a = t ** alpha + 1e-8
    elif profile_type == 'ramp_r':
        a = (1.0 - t) ** alpha + 1e-8
    elif profile_type == 'lin':
        a = 1.0 + alpha * t
    elif profile_type == 'quad':
        a = 1.0 + alpha * t + alpha * t * t
    elif profile_type == 'cos':
        a = 0.5 + 0.5 * np.cos(np.pi * t) + 1e-8
    elif profile_type == 'bell':
        a = np.exp(-alpha * (t - 0.5) ** 2) + 1e-8
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _build_conv_matrix(f, n):
    """Build the matrix A such that (A @ g)[k] = (f*g)[k] for k=0..2n-2."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A


def _solve_lp(f, M_lp, n):
    """Solve max sum(g) s.t. f*g <= M_lp, g >= 0."""
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M_lp)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _golden_section_search(a, g, sc_a, iters=20):
    """Golden section search for best blend t in (0,1) of a and g."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0  # 0.618...
    lo = 0.0
    hi = 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    sc_c = _score(np.maximum((1 - c) * a + c * g, 0))
    sc_d = _score(np.maximum((1 - d) * a + d * g, 0))
    
    for _ in range(iters):
        if sc_c < sc_d:
            hi = d
            d = c
            sc_d = sc_c
            c = hi - phi * (hi - lo)
            sc_c = _score(np.maximum((1 - c) * a + c * g, 0))
        else:
            lo = c
            c = d
            sc_c = sc_d
            d = lo + phi * (hi - lo)
            sc_d = _score(np.maximum((1 - d) * a + d * g, 0))
    
    best_t = (lo + hi) / 2.0
    best_sc = min(sc_c, sc_d, sc_a)
    return best_t, best_sc

def _adam_polish(a0, deadline, max_iter=50000):
    """Adam optimizer on smooth max of convolution."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    beta = 0.02 * float(c.max())
    base_beta = beta
    lr = 0.01
    base_lr = lr
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)
    v = np.zeros(n)
    no_imp = 0
    t = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        grad, mval = _log_sum_exp_grad(a, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3)
            m = np.zeros(n)
            v = np.zeros(n)
            continue
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        a = _normalize(a_new)
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                beta = base_beta
                lr = base_lr
                no_imp = 0
        
        if it % 200 == 199:
            beta = max(1e-8, beta * 0.85)
    
    return best_a


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _momentum_polish(a0, deadline, max_iter=50000):
    """Momentum-based optimization with smooth max."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    beta = 0.03 * float(c.max())
    lr = 0.02
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, mval = _log_sum_exp_grad(a, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3)
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.93)

                v = np.zeros(n)

                lr = base_lr
                no_imp = 0
        
        if it % 300 == 299:
            beta = max(1e-8, beta * 0.8)
    
    return best_a
