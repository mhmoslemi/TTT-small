    """Newton method for smooth max of autocorrelation on the simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    c = _fft_convolve(a)
    c_max = c.max()
    beta = beta_init if beta_init is not None else 0.01 * c_max
    
    N = _next_pow2(4 * n)  # FFT size
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        mval = c.max()

        x = np.clip(x, -100, 100)

        if s < 1e-30:

        w = w / s

        W[:2*n-1] = w
        fa = np.zeros(N, dtype=complex)
        fa[:n] = a

        # correlate: (w correlate a)[i] = sum_j w[j] * a[j-i]
        # In FFT: ifft(fft(w) * conj(fft(a)))  [need to verify convention]

        corr = np.fft.ifft(fw * np.conj(fa_f)).real
        grad = 2.0 * corr[:n]

        grad_proj = grad - grad.mean()

        # H * v = 2 * sum_k w_k * v[k-i] (same structure as grad but with v)
        
        def hess_vec(v):

            fv[:n] = v
            fv_c = np.fft.fft(fv)
            hv = np.fft.ifft(fw * np.conj(fv_c)).real
            return 2.0 * hv[:n]

        p = r.copy()
        rs_old = r @ r
        delta = np.zeros(n)
        
        for cg_it in range(min(100, n)):
            Hp = hess_vec(p)

            Hp_proj = Hp - Hp.mean()
            pHp = p @ Hp_proj

                break
            p_new = r_new + (rs_new / rs_old) * p
            p = p_new
            r = r_new
            rs_old = rs_new
        
        # Ensure delta is in tangent space
        delta = delta - delta.mean()

        for ls in range(20):

def _next_pow2(n):
    k = 1
    while k < n:

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        mn = float(c.min())
        if mn < -1e-10:
            c = c - mn
    return c


def _score(a):

    s = float(a.sum())

def _make_conv_matrix(a):

    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):

    return A


def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weights(c, beta):
    """Compute smooth max weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w

def _compute_gradient(a, w):
    """Compute grad[i] = 2 * sum_k w[k] * a[k-i] (valid indices only).
    w has length 2n-1. Returns array of length n.
    """
    n = len(a)
    grad = np.zeros(n)
    # Only iterate over nonzero w entries for efficiency
    nonzero = np.nonzero(w > 1e-15)[0]
    for k in nonzero:
        w_k = w[k]
        if w_k == 0:
            continue
        # grad[i] += 2 * w_k * a[k-i] for i such that 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * w_k * a[k - np.arange(lo, hi+1)]
    return grad


def _compute_hessian_vec(a, v, w):
    """Compute H*v where H is the Hessian of smooth_max(conv(a,a)).
    (H*v)[i] = 2 * sum_k w[k] * v[k-i]
    w has length 2n-1. v has length n. Returns array of length n.
    """
    n = len(v)
    Hv = np.zeros(n)
    nonzero = np.nonzero(w > 1e-15)[0]
    for k in nonzero:
        w_k = w[k]
        if w_k == 0:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        Hv[lo:hi+1] += 2.0 * w_k * v[k - np.arange(lo, hi+1)]
    return Hv
