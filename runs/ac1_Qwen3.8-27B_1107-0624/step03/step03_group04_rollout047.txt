def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _compute_grad(a, w, n):
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_raised_cosine(n):
    t = np.linspace(0.0, np.pi, n)
    a = np.sin(t)
    a = a * a
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_gaussian(n, width=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_symmetric_mix(n, rng, alpha=0.5):
    """Mix of raised cosine and flat-spectrum."""
    rc = _make_raised_cosine(n)
    fs = _make_flat_spectrum_init(n, rng)
    a = alpha * rc + (1 - alpha) * fs
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _smooth_max_adam(a, deadline, max_iter=60000, lr=0.04, beta_decay=0.997,
                     nesterov=True):
    """Smooth-max Adam with Nesterov momentum and aggressive beta annealing."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.08 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()

    no_imp = 0
    base_lr = lr
    prev_grad = None
    prev_z = z.copy()
    bb_scale = 1.0

    # Nesterov momentum
    z_nest = z.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute gradient at current (or Nesterov-lookahead) point
        if nesterov and it > 0:
            p = a
            # Use current a for gradient (simpler and more stable)
        else:
            p = a

        c = _fft_convolve(p)
        _, w = _smooth_max(c, beta)
        grad_p = _compute_grad(p, w, n)
        vp = grad_p + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            beta = max(1e-6, 0.08 * float(_fft_convolve(a).max()))
            prev_grad = None
            prev_z = z.copy()
            continue

        # BB step size
        if prev_grad is not None and np.all(np.isfinite(prev_grad)) and it > 10:
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20:
                bb_candidate = dot_ss / max(abs(dot_sg), 1e-20)
                bb_candidate = np.clip(bb_candidate, 0.2, 5.0)
                bb_scale = 0.6 * bb_scale + 0.4 * bb_candidate
        prev_z = z.copy()
        prev_grad = gz.copy()

        cur_lr = base_lr * bb_scale

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -55.0, 55.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.93)

        # Aggressive beta annealing
        if it % 50 == 0:
            beta *= beta_decay
            beta = max(1e-10, beta)

        # Adaptive beta based on peak concentration
        if it % 200 == 0 and it > 100:
            top5 = np.sort(c)[::-1][:5]
            if len(top5) > 1:
                peak_gap = top5[0] - top5[-1]
                if peak_gap < beta * 0.3:
                    beta = max(1e-12, beta * 0.5)

        # Restart if stuck
        if no_imp > max(4000, 6 * n):
            rng_r = np.random.default_rng(it * 104729 + 7)
            noise = 0.2 * rng_r.standard_normal(n)
            z_restart = best_z + noise
            z_restart = np.clip(z_restart, -55.0, 55.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            bb_scale = 1.0
            prev_grad = None
            prev_z = z.copy()
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            no_imp = 0

        z = z_new

    return best_a


def _coordinate_descent(a, deadline, max_iter=20000):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        improved = False
        # Cycle through a random subset of coordinates
        rng_cd = np.random.default_rng(it * 31 + 13)
        n_try = min(n, max(32, n // 4))
        indices = rng_cd.choice(n, size=n_try, replace=False)

        for j in indices:
            if time.monotonic() > deadline:
                break
            # Try different values for a[j] via golden section or line search
            old_val = a[j]
            # The objective as a function of a[j] is quadratic (piecewise)
            # Try a few multipliers
            best_local = old_val
            best_local_score = best_score

            # Scale sum: if we change a[j], sum changes
            # Work with normalized: a_new = a with a[j] changed, then renormalize
            # Actually let's work in the space where sum = 1
            # c_k depends on a[j] quadratically
            # Try: a[j] * factor for factor in [0.5, 0.6, 0.7, 0.8, 1.2, 1.3, 1.5]
            for factor in [0.7, 0.8, 0.9, 1.1, 1.15, 1.25, 1.4]:
                a_test = a.copy()
                a_test[j] = old_val * factor
                a_test = np.maximum(a_test, 0.0)
                s_test = a_test.sum()
                if s_test < 1e-10:
                    continue
                a_test = a_test / s_test
                c_test = _fft_convolve(a_test)
                sc = float(2.0 * n * c_test.max())
                if sc < best_local_score - 1e-14:
                    best_local_score = sc
                    best_local = a_test[j]

            if best_local != old_val:
                # Do a finer search around best_local
                a[j] = best_local
                a = np.maximum(a, 0.0)
                s2 = a.sum()
                if s2 < 1e-10:
                    a[j] = old_val
                    continue
                a = a / s2
                c_now = _fft_convolve(a)
                sc = float(2.0 * n * c_now.max())
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a.copy()
                    no_imp = 0
                    improved = True

        if not improved:
            no_imp += 1
            if no_imp > 50:
                break

    return best_a


def _peak_target_subgrad(a, deadline, max_iter=30000, n_top=10):
    """Direct subgradient using exact top-peak structure."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    z = np.log(a)
    z -= np.mean(z)
    lr = 0.02
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        score = float(2.0 * n * c_max)

        # Find top peaks
        thresh = c_max * 0.999
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Build subgradient from active peaks
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if k < n:
                # c[k] = sum_{i=k-n+1}^{k} a[i]*a[k-i], valid i: max(0,k-n+1)..min(n-1,k)
                for i in range(lo, hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        grad[i] += a[j]
                        grad[j] += a[i]
            else:
                # k >= n: c[k] = sum_{i=k-n+1}^{n-1} a[i]*a[k-i]
                for i in range(lo, hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        grad[i] += a[j]
                        grad[j] += a[i]

        grad /= len(active)
        grad = 2.0 * grad + 1e-8 * a
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        # Polyak-like step size
        z = z - lr * gz
        z = np.clip(z, -55.0, 55.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
            lr = min(0.05, lr * 1.01)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.001, lr * 0.92)

        if no_imp > max(2000, 4 * n):
            break

    return best_a


def _lp_refine(a, deadline, max_iter=10000, inflate=1.0005, blend=0.2):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.005, inflate_cur * 1.05)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.002, inflate_cur * 0.97)
            if no_imp > max(100, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _fine_polish(a, deadline, max_iter=20000):
    """Very fine gradient polish with tiny steps."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-10, 0.005 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= np.mean(z)
    lr = 0.005
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        _, w = _smooth_max(c, beta)
        vp = _compute_grad(a, w, n) + 1e-9 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -55.0, 55.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.0005, lr * 0.93)

        if it % 200 == 0:
            beta = max(1e-14, beta * 0.7)

        if no_imp > max(2000, 4 * n):
            break

    return best_a


def _multi_target_shave(a, deadline, max_iter=5000, n_peaks=8, strength=0.03):
    """Targeted multi-peak shaving."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]
        med = np.median(c)
        weights = np.maximum(top_vals - med, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = weights / weights.sum()

        pressure = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * a[j]

        a_new = a * np.exp(-str_cur * pressure / (pressure.max() + 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.15, str_cur * 1.005)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                str_cur = max(0.005, str_cur * 0.92)
            if no_imp > 500:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining multi-scale smooth-max Adam, coordinate
    descent, direct subgradient, peak shaving, and LP refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    if parent is not None:
        n_parent = parent.size
    else:
        n_parent = 768

    lengths = sorted(set([n_parent, 512, 768, 1024]))
    # Limit to reasonable lengths
    lengths = [ln for ln in lengths if 128 <= ln <= 2048]

    print(f"Lengths to try: {lengths}", flush=True)
    print(f"Deadline in {deadline - time.monotonic():.0f}s", flush=True)

    # Phase 1: Generate diverse initializations
    init_pool = []

    for n_t in lengths:
        if parent is not None:
            init_pool.append(('parent', n_t, _interpolate_profile(parent, n_t)))
        init_pool.append(('raised_cos', n_t, _make_raised_cosine(n_t)))
        init_pool.append(('gauss', n_t, _make_gaussian(n_t, 0.25)))
        init_pool.append(('gauss_wide', n_t, _make_gaussian(n_t, 0.35)))
        init_pool.append(('linear_1', n_t, _make_linear_profile(n_t, 1.0)))
        init_pool.append(('linear_2', n_t, _make_linear_profile(n_t, 2.0)))
        init_pool.append(('flat_spec', n_t, _make_flat_spectrum_init(n_t, rng)))
        init_pool.append(('sym_mix', n_t, _make_symmetric_mix(n_t, rng, 0.6)))

    # Quick evaluation
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print("Phase 1: Top 8 initializations:", flush=True)
    for sc, name, n_t, _ in init_scores[:8]:
        print(f"  {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Optimize top candidates
    k_top = min(6, len(init_scores))
    time_per_candidate = (deadline - t_start - 300) / max(1, k_top)

    for idx in range(k_top):
        if time.monotonic() > deadline - 250:
            break
        sc0, name, n_t, a0 = init_scores[idx]
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        # Skip if obviously worse and not the parent length
        if sc0 > best_score * 1.1 and name != 'parent' and n_t != n_parent:
            print(f"  Skipping {name} (n={n_t}): {sc0:.6f} > best", flush=True)
            continue

        # Allocate time
        t_alloc = min(time_per_candidate, remaining - 100)
        t_alloc = max(t_alloc, 100)

        print(f"\nPhase 2: Optimizing {name} (n={n_t}), start={sc0:.8f}, budget={t_alloc:.0f}s", flush=True)

        # Step A: Smooth-max Adam (main workhorse)
        adam_time = t_alloc * 0.45
        opt_deadline_a = time.monotonic() + adam_time
        a_opt = _smooth_max_adam(a0, opt_deadline_a, max_iter=60000, lr=0.04,
                                 beta_decay=0.996, nesterov=True)
        sc_opt = _score(a_opt)
        print(f"  Smooth-max Adam: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # Step B: Peak targeting subgradient
        if time.monotonic() < deadline - 80:
            ps_time = min(t_alloc * 0.15, 120.0)
            a_ps = _peak_target_subgrad(a_opt, time.monotonic() + ps_time,
                                        max_iter=30000, n_top=10)
            sc_ps = _score(a_ps)
            print(f"  Peak target subgrad: {sc_ps:.10f}", flush=True)
            if sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
            a_opt = a_ps if sc_ps < sc_opt else a_opt

        # Step C: Coordinate descent
        if time.monotonic() < deadline - 60:
            cd_time = min(t_alloc * 0.1, 60.0)
            a_cd = _coordinate_descent(a_opt, time.monotonic() + cd_time,
                                       max_iter=10000)
            sc_cd = _score(a_cd)
            print(f"  Coordinate descent: {sc_cd:.10f}", flush=True)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            a_opt = a_cd if sc_cd < sc_opt else a_opt

        # Step D: Multi-peak shave
        if time.monotonic() < deadline - 40:
            ms_time = min(t_alloc * 0.1, 45.0)
            a_ms = _multi_target_shave(a_opt, time.monotonic() + ms_time,
                                        max_iter=5000, n_peaks=8, strength=0.02)
            sc_ms = _score(a_ms)
            print(f"  Multi-peak shave: {sc_ms:.10f}", flush=True)
            if sc_ms < best_score:
                best_score = sc_ms
                best_seq = a_ms.copy()
            a_opt = a_ms if sc_ms < sc_opt else a_opt

        # Step E: Fine polish
        if time.monotonic() < deadline - 20:
            fp_time = min(t_alloc * 0.1, 30.0)
            a_fp = _fine_polish(a_opt, time.monotonic() + fp_time, max_iter=15000)
            sc_fp = _score(a_fp)
            print(f"  Fine polish: {sc_fp:.10f}", flush=True)
            if sc_fp < best_score:
                best_score = sc_fp
                best_seq = a_fp.copy()
            a_opt = a_fp if sc_fp < sc_opt else a_opt

        # Step F: LP refinement (only for smaller n)
        if n_t <= 1024 and time.monotonic() < deadline - 15:
            lp_time = min(t_alloc * 0.2, 80.0)
            a_lp = _lp_refine(a_opt, time.monotonic() + lp_time,
                              max_iter=5000, inflate=1.0004, blend=0.15)
            sc_lp = _score(a_lp)
            print(f"  LP refine: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()

    print(f"\nAfter Phase 2: best_score={best_score:.10f}", flush=True)

    # Phase 3: Global fine-tuning on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5

        # Additional smooth-max with very low initial beta
        if remaining > 60:
            t3 = min(remaining * 0.4, 300.0)
            a3 = _smooth_max_adam(best_seq, time.monotonic() + t3,
                                   max_iter=40000, lr=0.015, beta_decay=0.995)
            sc3 = _score(a3)
            print(f"Phase 3a: Refine smooth-max: {sc3:.10f}", flush=True)
            if sc3 < best_score:
                best_score = sc3
                best_seq = a3.copy()

        # Peak target subgrad on best
        if time.monotonic() < deadline - 10:
            t4 = min(remaining * 0.2, 150.0)
            a4 = _peak_target_subgrad(best_seq, time.monotonic() + t4,
                                       max_iter=20000, n_top=15)
            sc4 = _score(a4)
            print(f"Phase 3b: Peak target: {sc4:.10f}", flush=True)
            if sc4 < best_score:
                best_score = sc4
                best_seq = a4.copy()

        # Fine polish
        if time.monotonic() < deadline - 5:
            t5 = min(remaining * 0.2, 120.0)
            a5 = _fine_polish(best_seq, time.monotonic() + t5, max_iter=15000)
            sc5 = _score(a5)
            print(f"Phase 3c: Fine polish: {sc5:.10f}", flush=True)
            if sc5 < best_score:
                best_score = sc5
                best_seq = a5.copy()

        # Final LP if n is manageable
        if best_seq.size <= 1024 and time.monotonic() < deadline - 5:
            t6 = min(remaining * 0.15, 60.0)
            if t6 > 10:
                a6 = _lp_refine(best_seq, time.monotonic() + t6,
                                 max_iter=3000, inflate=1.0003, blend=0.1)
                sc6 = _score(a6)
                print(f"Phase 3d: Final LP: {sc6:.10f}", flush=True)
                if sc6 < best_score:
                    best_score = sc6
                    best_seq = a6.copy()

    # Phase 4: Final evaluation
    if best_seq is None:
        best_seq = _make_raised_cosine(768)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_raised_cosine(768)

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
