"""

import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b=None):
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([])
    if n <= 200:
        return np.convolve(a, b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

    a = np.asarray(a, dtype=np.float64).ravel()

    return float(2.0 * n * c.max())

def _solve_lp(f, M, n):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

    b = np.full(2 * n - 1, M)

    ts = np.linspace(0.02, 1.0, n_grid)

        sc = _normalized_score(a_new)

                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            consec_no_imp = 0
            # Slowly decrease inflation
            if it % 200 == 199:
                inflate = max(inflate_min, inflate * 0.97)
        else:
            no_imp += 1
            consec_no_imp += 1
            if consec_no_imp % 30 == 0:
                inflate = max(0.005, inflate * 0.9)
            if consec_no_imp > 300:
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                consec_no_imp = 0
                # If we're back at best and still can't improve, try perturbation
                if best_score < 2.0 * n * 0.7:
                    # Add small random perturbation
                    rng = np.random.default_rng(hash((n, best_score)) % (2**32))
                    pert = rng.random(n) * 0.01
                    a = a + pert
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

    return best_a


def _smooth_max_subgrad(a, beta):
    """Compute subgradient of smooth approximation of max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # Subgradient: d/d f[j] of sum_k w_k * conv(a,a)[k]
    # = 2 * sum_k w_k * a[k-j] (for valid indices)
    # This is 2 * (a_flipped * w) at appropriate offset
    a_flipped = a[::-1]
    N_pad = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a_flipped, N_pad)
    fw = np.fft.rfft(w, N_pad)
    corr = np.fft.irfft(fa * fw, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _accelerated_subgradient(a0, deadline, max_iter=50000):
    """Nesterov-accelerated projected subgradient method."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-7, 0.02 * float(c.max()))
    lr = 0.1
    base_lr = lr
    momentum = 0.9
    v = np.zeros(n)
    y = a.copy()  # Nesterov look-ahead
    t_k = 1.0
    no_imp = 0
    phase = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        if it > 0 and it % 500 == 0:
            # Update Nesterov parameter
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
            momentum = (t_k - 1.0) / t_k
            beta = max(1e-8, beta * 0.85)

        # Compute subgradient at y
        grad, maxc = _smooth_max_subgrad(y, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-7, beta * 3.0)
            continue

        # Gradient step
        v = momentum * v - lr * grad / (maxc + 1e-10)
        a_new = _project_simplex(a + v)

        # Nesterov update for y
        if it > 0:
            y = a_new + ((t_k - 1.0) / t_k) * (a_new - a)
        else:
            y = a_new
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.01, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                y = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.02 * float(_fft_convolve(best_a).max()))
                lr = base_lr
                no_imp = 0
                t_k = 1.0
                if phase >= 2:
                    break
                phase += 1

    return best_a


def _all_rounds_optimize(a0, deadline, total_budget=600.0):
    """Alternate between LP and gradient for best results."""
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    s = a.sum()
    if s > 1e-10:
        a = a / s
        best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    # Round 1: LP
    lp_budget = min(total_budget * 0.4, deadline - time.monotonic() - 60)
    if lp_budget > 10:
        lp_end = time.monotonic() + lp_budget
        a_lp = _lp_refine(a, lp_end, max_iter=30000, inflate_start=1.003)
        sc_lp = _normalized_score(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
            a = a_lp
        print(f"    LP: score={sc_lp:.10f}")

    # Round 2: Gradient
    if time.monotonic() < deadline - 30:
        gp_budget = min(total_budget * 0.35, deadline - time.monotonic() - 30)
        if gp_budget > 5:
            gp_end = time.monotonic() + gp_budget
            a_gp = _accelerated_subgradient(a, gp_end, max_iter=30000)
            sc_gp = _normalized_score(a_gp)
            if sc_gp < best_score:
                best_score = sc_gp
                best_a = a_gp.copy()
            print(f"    Grad: score={sc_gp:.10f}")

    # Round 3: LP again (finer)
    if time.monotonic() < deadline - 20:
        lp_budget2 = min(total_budget * 0.2, deadline - time.monotonic() - 20)
        if lp_budget2 > 5:
            lp_end2 = time.monotonic() + lp_budget2
            a_lp2 = _lp_refine(best_a, lp_end2, max_iter=20000, inflate_start=1.001, inflate_min=1.0001)
            sc_lp2 = _normalized_score(a_lp2)
            if sc_lp2 < best_score:
                best_score = sc_lp2
                best_a = a_lp2.copy()
            print(f"    LP2: score={sc_lp2:.10f}")

    # Round 4: Final gradient
    if time.monotonic() < deadline - 10:
        gp_budget2 = min(30.0, deadline - time.monotonic() - 10)
        if gp_budget2 > 3:
            gp_end2 = time.monotonic() + gp_budget2
            a_g2 = _accelerated_subgradient(best_a, gp_end2, max_iter=10000)
            sc_g2 = _normalized_score(a_g2)
            if sc_g2 < best_score:
                best_score = sc_g2
                best_a = a_g2.copy()
            print(f"    Grad2: score={sc_g2:.10f}")

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initializations -> LP refine -> gradient polish.
    Explores U-shaped, sine, and multi-modal profiles at multiple lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate diverse initializations at multiple lengths
    lengths = [384, 512, 768, 1024, 1536]
    if parent is not None:
        lengths = [parent.size, 1024, 1536, 2048]

    configs = []
    for n_t in lengths:
        if parent is not None:
            configs.append(('parent_interp', n_t, _interpolate(parent, n_t)))
        configs.append(('u_shape_2', n_t, _make_u_shape(n_t, 2.0, 0.01)))
        configs.append(('u_shape_3', n_t, _make_u_shape(n_t, 3.0, 0.01)))
        configs.append(('u_shape_4', n_t, _make_u_shape(n_t, 4.0, 0.005)))
        configs.append(('power_u_2', n_t, _make_power_u(n_t, 2.0, 0.001)))
        configs.append(('power_u_4', n_t, _make_power_u(n_t, 4.0, 0.001)))
        configs.append(('sine_1', n_t, _make_sine(n_t, 1.0)))
        configs.append(('sine_2', n_t, _make_sine(n_t, 2.0)))
        configs.append(('cos_u_1', n_t, _make_cos_u(n_t, 1.0)))
        configs.append(('w_shape_2', n_t, _make_w_shape(n_t, 2.0, 0.6)))
        configs.append(('gauss_edge', n_t, _make_gaussian_edges(n_t, 0.15)))
        configs.append(('ramp_1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp_2', n_t, _make_ramp(n_t, 2.0)))

    # Score all initializations
    scored = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        if np.isfinite(sc):
            scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    print(f"Total configs: {len(scored)}")
    print(f"Top 10 initializations:")
    for i, (sc, name, n_t, _) in enumerate(scored[:10]):
        print(f"  {i+1}. {name} n={n_t} score={sc:.8f}")

    # Track the best per length to avoid redundant work
    best_per_length = {}
    for sc, name, n_t, a0 in scored:
        if n_t not in best_per_length or sc < best_per_length[n_t][0]:
            best_per_length[n_t] = (sc, name, a0)

    # Phase 1: Optimize top candidates
    # Sort by score, take top candidates with diversity
    seen_lengths = set()
    selected = []
    for sc, name, n_t, a0 in scored:
        if len(selected) >= 8:
            break
        if n_t in seen_lengths:
            continue
        seen_lengths.add(n_t)
        selected.append((sc, name, n_t, a0))

    # Also make sure we include the absolute best
    if scored and scored[0][2] not in seen_lengths:
        selected.insert(0, scored[0])

    print(f"\nSelected {len(selected)} candidates for optimization:")
    for sc, name, n_t, _ in selected:
        print(f"  {name} n={n_t} score={sc:.8f}")

    # Allocate time for each candidate
    n_cands = len(selected)
    total_opt_time = min(budget_s * 0.7, deadline - time.monotonic() - 200)
    time_per_cand = total_opt_time / max(1, n_cands)

    for idx, (sc0, name, n_t, a0) in enumerate(selected):
        remaining = deadline - time.monotonic() - 120
        if remaining < 30:
            break

        if sc0 >= best_score + 0.05 and idx > 0:
            # Only optimize if potentially competitive
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Allocate time: more for better initializations
        if sc0 < best_score + 0.02:
            cand_time = min(time_per_cand * 1.5, remaining * 0.4)
        else:
            cand_time = min(time_per_cand, remaining * 0.25)

        cand_time = max(cand_time, 15.0)
        if cand_time > remaining * 0.5:
            cand_time = remaining * 0.5

        print(f"\nOptimizing {name} n={n_t} (budget={cand_time:.0f}s):")
        t_cand_start = time.monotonic()
        cand_deadline = t_cand_start + cand_time

        a_opt = _all_rounds_optimize(a0, cand_deadline, total_budget=cand_time)
        sc_opt = _score(a_opt)
        print(f"  Result: {name} n={n_t} -> score={sc_opt:.10f}")

        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  *** NEW BEST: {sc_opt:.10f} ***")

    # Phase 2: Fine-tune the best with longer optimization
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 60:
            print(f"\nFine-tuning best (n={best_seq.size}, score={best_score:.10f}), budget={remaining:.0f}s:")
            fine_deadline = time.monotonic() + remaining
            a_fine = _all_rounds_optimize(best_seq, fine_deadline, total_budget=remaining)
            sc_fine = _score(a_fine)
            print(f"  Fine-tune result: {sc_fine:.10f}")
            if np.isfinite(sc_fine) and sc_fine < best_score:
                best_score = sc_fine
                best_seq = a_fine.copy()
                print(f"  *** NEW BEST: {sc_fine:.10f} ***")

    # Phase 3: Try interpolating best to slightly different lengths and re-optimizing
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        for n_try in [n_best + 32, n_best + 64, n_best - 32]:
            if n_try < 64 or n_try > 4096:
                continue
            if time.monotonic() > deadline - 20:
                break
            a_interp = _interpolate(best_seq, n_try)
            remaining2 = deadline - time.monotonic() - 15
            if remaining2 < 10:
                break
            a_try = _lp_refine(a_interp, time.monotonic() + min(20, remaining2 * 0.5),
                              max_iter=5000, inflate_start=1.002)
            sc_try = _score(a_try)
            if np.isfinite(sc_try) and sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()
                print(f"  Interp n={n_try}: {sc_try:.10f} (new best)")

    # Fallback
    if best_seq is None:
        best_seq = _make_u_shape(512, 2.0, 0.01)
        best_score = _score(best_seq)

    # Final validation using evaluate_sequence
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_u_shape(256, 2.0, 0.01)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL SCORE: {best_score:.10f}")
    return [float(x) for x in final_seq]
