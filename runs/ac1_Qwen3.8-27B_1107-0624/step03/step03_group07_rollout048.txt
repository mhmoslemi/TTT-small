def lbfqs_polish(a0, max_iters=200, beta=0.001):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z0 = np.log(a0)
    z0 = z0 - z0.mean()

    best_a = a0.copy()
    c0 = full_convolve_fft(a0)
    best_score = float(2.0 * n * c0.max())

    def objective(z_vec):
        z_vec = np.asarray(z_vec, dtype=np.float64).ravel()
        if not np.all(np.isfinite(z_vec)):
            return 1e10
        z_vec = np.clip(z_vec, -50.0, 50.0)
        z_vec = z_vec - z_vec.mean()
        a_tmp = np.exp(z_vec - z_vec.max())
        s_tmp = a_tmp.sum()
        if s_tmp < 1e-15:
            return 1e10
        a_tmp = a_tmp / s_tmp
        c_tmp = full_convolve_fft(a_tmp)
        val, _ = soft_max_weights(c_tmp, beta)
        return 2.0 * n * val

    def gradient(z_vec):
        z_vec = np.asarray(z_vec, dtype=np.float64).ravel()
        z_vec = np.clip(z_vec, -50.0, 50.0)
        z_vec = z_vec - z_vec.mean()
        a_tmp = np.exp(z_vec - z_vec.max())
        s_tmp = a_tmp.sum()
        if s_tmp < 1e-15:
            return np.zeros_like(z_vec)
        a_tmp = a_tmp / s_tmp
        c_tmp = full_convolve_fft(a_tmp)
        _, w_tmp = soft_max_weights(c_tmp, beta)
        h_tmp = compute_h(w_tmp, a_tmp)
        grad_a = 2.0 * h_tmp
        dot_val = float(np.dot(a_tmp, grad_a))
        dz = a_tmp * (grad_a - dot_val)
        return 2.0 * n * dz

    try:
        result = minimize(
            objective, z0, jac=gradient, method='L-BFGS-B',
            options={'maxiter': max_iters, 'ftol': 1e-12, 'gtol': 1e-8,
                     'maxfun': max_iters * 3}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            z_opt = result.x
            z_opt = np.clip(z_opt, -50.0, 50.0)
            z_opt = z_opt - z_opt.mean()
            a_opt = np.exp(z_opt - z_opt.max())
            s_opt = a_opt.sum()
            if s_opt > 1e-15:
                a_opt = a_opt / s_opt
                c_opt = full_convolve_fft(a_opt)
                sc = float(2.0 * n * c_opt.max())
                if sc < best_score:
                    best_a = a_opt
                    best_score = sc
    except Exception:
        pass

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Try to get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting hybrid search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Quick multi-n exploration =====
    n_candidates = [512, 768, 1024, 1536, 2048, 2560]
    phase1_deadline = min(deadline, t_start + 200.0)
    print(f"\n--- Phase 1: Multi-n exploration ---")

    phase1_results = {}
    for n_try in n_candidates:
        if time.monotonic() > phase1_deadline - 5:
            break

        if init_seq is not None:
            p_base = interpolate_profile(init_seq, n_try)
        else:
            p_base = make_linear_profile(n_try, 1.5)

        sc_base = compute_score(p_base)
        update(p_base, sc_base)

        dl = min(phase1_deadline, time.monotonic() + 25.0)
        result, sc_r = logit_adam_optimize(p_base, dl, rng, lr=0.05,
                                            beta_factor=0.1, beta_decay=0.88,
                                            use_adaptive_beta=True)
        sc_final = compute_score(result)
        update(result, sc_final)
        phase1_results[n_try] = sc_final
        print(f"  n={n_try}: {sc_final:.10f}")

    # Sort by score to pick best n values
    sorted_ns = sorted(phase1_results.keys(), key=lambda k: phase1_results[k])
    top_ns = sorted_ns[:3]
    print(f"  Top n values: {[f'{n}:{phase1_results[n]:.6f}' for n in top_ns]}")

    # ===== Phase 2: Detailed optimization at top n values =====
    phase2_deadline = min(deadline, t_start + 1300.0)
    print(f"\n--- Phase 2: Detailed optimization ---")

    for rank, n_opt in enumerate(top_ns):
        if time.monotonic() > phase2_deadline - 50:
            break

        if best_seq is not None and best_seq.size == n_opt:
            p_start = best_seq.copy()
        elif init_seq is not None:
            p_start = interpolate_profile(init_seq, n_opt)
        else:
            p_start = make_linear_profile(n_opt, 1.5)

        print(f"\n  Optimizing n={n_opt} (rank {rank+1})...")

        # Adam optimization (longer)
        time_per_n = max(60.0, (phase2_deadline - time.monotonic()) / max(1, len(top_ns) - rank))
        dl_adam = min(phase2_deadline - 30, time.monotonic() + time_per_n * 0.6)
        result, sc_r = logit_adam_optimize(p_start, dl_adam, rng, lr=0.03,
                                            beta_factor=0.06, beta_decay=0.92,
                                            beta_min=1e-12, restart_period=1000,
                                            use_adaptive_beta=True)
        sc_final = compute_score(result)
        update(result, sc_final)
        print(f"    After Adam: {sc_final:.10f}")

        # Try a couple of diverse starting points if time allows
        if time.monotonic() < phase2_deadline - 60:
            profiles = diversified_profiles(n_opt, rng)
            for idx in range(min(3, len(profiles))):
                if time.monotonic() > phase2_deadline - 30:
                    break
                p_alt = profiles[idx]
                sc_alt_init = compute_score(p_alt)
                if sc_alt_init < best_score + 0.05:
                    dl_alt = min(phase2_deadline - 30, time.monotonic() + 30.0)
                    r_alt, sc_alt = logit_adam_optimize(p_alt, dl_alt, rng,
                                                         lr=0.04, beta_factor=0.08,
                                                         use_adaptive_beta=True)
                    sc_alt_f = compute_score(r_alt)
                    update(r_alt, sc_alt_f)
                    if sc_alt_f < best_score + 0.01:
                        print(f"    Alt profile {idx}: {sc_alt_f:.10f}")
                    if sc_alt_f < sc_final:
                        result = r_alt
                        sc_final = sc_alt_f

        # Peak targeting
        if time.monotonic() < phase2_deadline - 20:
            dl_pt = min(phase2_deadline - 10, time.monotonic() + time_per_n * 0.25)
            r_pt, sc_pt = peak_targeting_refine(result, dl_pt, rng,
                                                 n_peaks=5, lr=0.015, adaptive=True)
            sc_pt_f = compute_score(r_pt)
            update(r_pt, sc_pt_f)
            print(f"    After peak targeting: {sc_pt_f:.10f}")
            if sc_pt_f < sc_final:
                result = r_pt
                sc_final = sc_pt_f

        # L-BFGS polish
        if time.monotonic() < phase2_deadline - 10:
            r_lb, sc_lb = lbfqs_polish(result, max_iters=100, beta=0.0005)
            sc_lb_f = compute_score(r_lb)
            update(r_lb, sc_lb_f)
            print(f"    After L-BFGS: {sc_lb_f:.10f}")
            if sc_lb_f < sc_final:
                result = r_lb
                sc_final = sc_lb_f

    print(f"  Phase 2 best: {best_score:.10f}")

    # ===== Phase 3: Final intensive polishing =====
    phase3_deadline = deadline
    print(f"\n--- Phase 3: Final polishing ---")

    if best_seq is not None:
        n_polish = best_seq.size

        # Multiple rounds with decreasing learning rates
        polish_configs = [
            (0.012, 0.015, "adam", 80),
            (0.008, 0.010, "adam", 60),
            (0.005, 0.005, "lbfqs", 50),
            (0.003, 0.003, "peak", 40),
        ]

        for round_idx, (lr_val, beta_f, method, time_s) in enumerate(polish_configs):
            if time.monotonic() > phase3_deadline - 10:
                break

            p0 = best_seq.copy()
            p0 = p0 + 0.0003 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()

            dl = min(phase3_deadline - 5, time.monotonic() + float(time_s))

            if method == "adam":
                r, sc_r = logit_adam_optimize(p0, dl, rng, lr=lr_val,
                                               beta_factor=beta_f, beta_decay=0.94,
                                               use_adaptive_beta=True)
            elif method == "lbfqs":
                r, sc_r = lbfqs_polish(p0, max_iters=150, beta=max(0.0001, beta_f * 0.01))
            else:
                r, sc_r = peak_targeting_refine(p0, dl, rng, n_peaks=3,
                                                 lr=lr_val, adaptive=True)

            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  Polish {round_idx} ({method}, lr={lr_val}): {sc_f:.10f}")

        # Final L-BFGS with very small beta
        if time.monotonic() < phase3_deadline - 10:
            r_final, sc_final = lbfqs_polish(best_seq, max_iters=200, beta=0.0001)
            sc_final_f = compute_score(r_final)
            update(r_final, sc_final_f)
            print(f"  Final L-BFGS (beta=0.0001): {sc_final_f:.10f}")

    print(f"\n  Final best score: {best_score:.10f}")

    # ===== Validate and return =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(1024, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
