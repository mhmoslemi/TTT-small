def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

"""
Refined multi-stage optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses hybrid exact-subgradient / soft-max optimization with momentum,
adaptive step sizes, structured restarts, and multi-scale exploration.
Exploits convolution symmetry and uses simplex projection for robustness.

import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 256:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_grad_from_weights(a, w):
    """Compute 2 * sum_i a[i] * w[j+i] for j=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    N = next_pow2(n + len(w) - 1)
    fa_rev = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa_rev * fw, N)
    return 2.0 * conv[n - 1:2 * n - 1]


def softmax_weights(c, beta):
    """Compute soft-max weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        mask = c >= mval - 1e-10
        cnt = int(np.count_nonzero(mask))
        w = np.zeros_like(c)
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def project_simplex(v):
    """Project v onto probability simplex {z >= 0, sum(z)=1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize_sum1(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_bump(n, width=0.5, offset=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - offset) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Interpolate profile p0 to length n_new and normalize."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        return normalize_sum1(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    return normalize_sum1(result)


def subgradient_descent_momentum(a0, deadline, rng, momentum=0.9,
                                  base_lr=0.002, lr_decay=0.999,
                                  backtrack_c=0.5, max_backtracks=20,
                                  min_lr=1e-8):
    """
    Exact subgradient descent with momentum and backtracking line search
    on the simplex. Minimize f(a) = n * max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    a = normalize_sum1(a0.copy())
    n = a.size
    v = np.zeros(n)  # momentum
    lr = base_lr

    c = full_convolve_fft(a)
    f_val = float(n * c.max())
    best_a = a.copy()
    best_f = f_val
    no_improve = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        max_c = float(c.max())

        # Get subgradient: uniform over argmax
        mask = c >= max_c - 1e-10 * max_c
        cnt = int(np.count_nonzero(mask))
        w = np.zeros_like(c)
        w[mask] = 1.0 / cnt

        # Compute gradient
        grad = compute_grad_from_weights(a, w)

        # Project gradient onto tangent of simplex (remove mean component)
        grad = grad - np.mean(grad)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        direction = -v

        # Backtracking line search
        best_alpha = lr
        best_new_f = np.inf
        stp = lr
        for bt in range(max_backtracks):
            trial = project_simplex(a + stp * direction)
            trial_c = full_convolve_fft(trial)
            trial_f = float(n * trial_c.max())
            if trial_f < f_val - 1e-15:
                best_alpha = stp
                best_new_f = trial_f
                break
            stp *= backtrack_c
        else:
            # Accept anyway with smallest step
            best_alpha = stp
            best_new_f = trial_f

        a = project_simplex(a + best_alpha * direction)
        f_val = best_new_f

        if f_val < best_f - 1e-14:
            best_f = f_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay learning rate
        lr = max(min_lr, lr * lr_decay)

        # Restart if stuck
        if no_improve > 500:
            # Structured restart: perturb in a smooth direction
            freq = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.linspace(0, 1, n)
            perturbation = np.cos(2 * np.pi * freq * t + phase)
            a = a + 0.05 * perturbation
            a = normalize_sum1(a)
            v = np.zeros(n)
            lr = base_lr
            no_improve = 0
            c = full_convolve_fft(a)
            f_val = float(n * c.max())

    return best_a, best_f


def logit_adam_optimize(a0, deadline, rng, lr=0.03, beta_factor=0.05,
                        beta_decay=0.94, beta_min=1e-10,
                        restart_period=2000, exact_threshold=500):
    """
    Logit-space Adam optimization with adaptive soft-max beta.
    Switches to exact subgradient when beta is very small.
    """
    a = normalize_sum1(a0.copy())
    n = a.size

    c = full_convolve_fft(a)
    max_c = float(c.max())

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr

    while time.monotonic() < deadline:
        # Compute weights (soft-max or exact)
        if beta < exact_threshold * 1e-10:
            # Use exact subgradient
            mask = c >= max_c - 1e-10
            cnt = int(np.count_nonzero(mask))
            w = np.zeros_like(c)
            w[mask] = 1.0 / cnt
        else:
            _, w = softmax_weights(c, beta)

        grad_a = compute_grad_from_weights(a, w)
        dot = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot)

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        # Stabilize
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            a = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr_reset = True
            continue

        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-15:
            z = best_z.copy()
            a = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a = a / s

        # Evaluate
        c = full_convolve_fft(a)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt beta
        if t_adam % 30 == 0:
            beta = max(beta_min, beta * beta_decay)

        # Adapt learning rate
        if no_improve > 200:
            current_lr = max(0.003, current_lr * 0.98)

        # Random restart
        if no_improve > restart_period:
            noise_scale = 0.12
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = full_convolve_fft(a)
            max_c = float(c.max())
            beta = max(beta_min, max(beta_factor * max_c, 0.02))
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = lr
            no_improve = 0

    return best_a, best_score

def coordinate_perturbation_refine(a0, deadline, rng, n_trials=50):
    """
    Refine by trying random coordinate perturbations.
    For each trial, perturb a small number of coordinates and accept if improved.
    """
    a = normalize_sum1(a0.copy())
    n = a.size
    best_a = a.copy()
    best_score = float(2.0 * n * full_convolve_fft(a).max())
    improvements = 0

    while time.monotonic() < deadline:
        improved = False
        for _ in range(n_trials):
            # Choose a random subset of coordinates to perturb
            n_perturb = rng.integers(1, min(10, n // 10) + 1)
            indices = rng.choice(n, size=n_perturb, replace=False)

            # Random perturbation in logit space
            z = np.log(np.maximum(a, 1e-15))
            dz = np.zeros(n)
            for idx in indices:
                dz[idx] = rng.normal(0, 0.01)
            # Compensate to maintain sum (approximately)
            dz -= np.sum(dz * a)
            z_new = z + dz
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()

            c_new = full_convolve_fft(a_new)
            score_new = float(2.0 * n * c_new.max())

            if score_new < best_score - 1e-15:
                a = a_new
                best_a = a.copy()
                best_score = score_new
                improved = True
                improvements += 1

        if not improved and improvements > 0:
            # Decay perturbation scale by reducing n_trials effect
            break

    return best_a, best_score

        dl = min(phase3_deadline, time.monotonic() + 25.0)
        r, _ = logit_adam_optimize(p, dl, rng, lr=0.03,
                                    beta_factor=0.06, beta_decay=0.93)
        sc_r = compute_score(r)
        try_update(r, sc_r)
        if sc_r < best_score + 0.02:
            print(f"  Restart {idx}: {sc_r:.10f}")

    print(f"  Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Try different n values =====
    phase4_deadline = min(deadline, t_start + 1000.0)
    print(f"\n=== Phase 4: Multi-scale refinement (deadline at {phase4_deadline - t_start:.0f}s) ===")

    n4_candidates = sorted(set([best_seq.size,
                                 max(512, best_seq.size // 2),
                                 best_seq.size * 2,
                                 768, 1536]))
    n4_candidates = [n for n in n4_candidates if 128 <= n <= 4096]

    for n4 in n4_candidates:
        if time.monotonic() > phase4_deadline - 20:
            break
        p4 = interpolate_profile(best_seq, n4)
        sc4 = compute_score(p4)
        try_update(p4, sc4)

        if sc4 < best_score + 0.03:
            dl = min(phase4_deadline, time.monotonic() + 60.0)
            r4, _ = logit_adam_optimize(p4, dl, rng, lr=0.02,
                                         beta_factor=0.03, beta_decay=0.95)
            sc4_f = compute_score(r4)
            try_update(r4, sc4_f)
            print(f"  n={n4}: {sc4_f:.10f}")

    print(f"  Phase 4 best: {best_score:.10f}")

    # ===== Phase 5: Coordinate perturbation refinement =====
    phase5_deadline = min(deadline, t_start + 1300.0)
    print(f"\n=== Phase 5: Coordinate perturbation (deadline at {phase5_deadline - t_start:.0f}s) ===")

    if time.monotonic() < phase5_deadline - 10:
        dl_cp = min(phase5_deadline, time.monotonic() + 150.0)
        r_cp, sc_cp = coordinate_perturbation_refine(best_seq, dl_cp, rng, n_trials=100)
        sc_cp_f = compute_score(r_cp)
        try_update(r_cp, sc_cp_f)
        print(f"  After coordinate perturbation: {sc_cp_f:.10f}")

    # ===== Phase 6: Final intensive polish =====
    print(f"\n=== Phase 6: Final polish ===")

    if best_seq is not None and time.monotonic() < deadline - 10:
        n_polish = best_seq.size
        # Multiple rounds with decreasing learning rate
        for round_idx, (lr_val, beta_f, n_peaks) in enumerate([
            (0.012, 0.02, 2),
            (0.008, 0.015, 3),
            (0.005, 0.01, 2),
            (0.003, 0.008, 3),
        ]):
            if time.monotonic() > deadline - 10:
                break
            dl = min(deadline, time.monotonic() + 60.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.0003 * rng.normal(size=n_polish)
            p0 = normalize_sum1(p0)

            if round_idx % 2 == 0:
                r, sc_r = logit_adam_optimize(p0, dl, rng, lr=lr_val,
                                               beta_factor=beta_f, beta_decay=0.96)
            else:
                r, sc_r = peak_targeting_refine(p0, dl, rng, n_peaks=n_peaks, lr=lr_val)
            sc_f = compute_score(r)
            try_update(r, sc_f)
            print(f"  Polish {round_idx} (lr={lr_val}): {sc_f:.10f}")

        # Final subgradient polish
        if time.monotonic() < deadline - 20:
            dl_sg2 = min(deadline, time.monotonic() + 40.0)
            r_sg2, sc_sg2 = subgradient_descent_momentum(best_seq, dl_sg2, rng,
                                                          momentum=0.8, base_lr=0.0005,
                                                          lr_decay=0.9998, max_backtracks=10)
            sc_sg2_f = compute_score(r_sg2)
            try_update(r_sg2, sc_sg2_f)
            print(f"  Final subgradient: {sc_sg2_f:.10f}")

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
