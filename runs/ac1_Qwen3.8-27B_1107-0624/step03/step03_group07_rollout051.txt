import time

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

"""
Hybrid LP + gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based directional refinement combined with logit-space Adam optimization.

import numpy as np
import cvxpy as cp
import time

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def cross_convolve_fft(f, g):
    """Compute full convolution of f and g using FFT."""
    n = max(len(f), len(g))
    N = next_pow2(n + len(g) - 1)
    F = np.fft.rfft(f, N)
    G = np.fft.rfft(g, N)
    c = np.fft.irfft(F * G, N)
    return c[:len(f) + len(g) - 1]

def compute_h(w, a):
    """h[i] = sum_{m} a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_uniform_profile(n):
    return np.ones(n) / n

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

    best_z = z.copy()

def lp_refine(f, M, timeout=30.0):
    """Solve LP: max sum(g) s.t. C_f @ g <= M, g >= 0.
    Returns normalized g or None."""
    n = len(f)
    
    # Build convolution matrix C: C[k,j] = f[k-j] if 0 <= k-j < n
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = f
    
    g = cp.Variable(n)
    constraints = [g >= 0, C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    # Try solvers in order of preference
    solvers = [cp.ECOS, cp.PDLP, cp.SCS]
    if hasattr(cp, 'CLARABEL'):
        solvers.insert(0, cp.CLARABEL)
    
    t0 = time.monotonic()
    for solver in solvers:
        if time.monotonic() - t0 > timeout:
            return None
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                g_val = np.maximum(g.value, 0)
                s = g_val.sum()
                if s > 1e-15:
                    return g_val / s
                break
        except Exception:
            continue
    
    return None

def lp_iteration(f, timeout=30.0):
    """One LP-based refinement iteration.
    Returns (new_f, new_score) or (None, None)."""
    n = len(f)
    c = full_convolve_fft(f)
    M = float(c.max())
    s = float(f.sum())
    # Normalize M relative to sum
    M_norm = M / (s * s)  # M / sum^2
    
    g = lp_refine(f / s, M_norm, timeout=timeout)
    if g is None:
        return None, None
    
    return g, compute_score(g)

def lp_refine_loop(f0, deadline, rng, initial_t=0.3, t_decay=0.995, min_t=0.01):
    """Iterative LP-based refinement with decreasing step size."""
    f = f0.copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    
    n = f.size
    c = full_convolve_fft(f)
    best_score = float(2.0 * n * c.max())

    no_improve = 0

            t *= 0.9

        # Blend
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)

            no_improve = 0
            f = f_new.copy()
        else:
            no_improve += 1

    n = len(v)

"""
Hybrid LP + subgradient + logit-Adam optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovation: LP-based directional refinement that directly enforces convolution constraints,
combined with subgradient methods and logit-space Adam for fast initial progress.

import numpy as np
import cvxpy as cp
import time


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_h(w, a):
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def make_uniform_profile(n):
    return np.ones(n) / n


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = int(np.argmax(u * np.arange(1, n + 1) - (cssv - 1) > 0))
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def logit_adam_optimize(a0, deadline, rng, lr=0.04, beta_factor=0.08,
                        beta_decay=0.92, beta_min=1e-10, restart_period=1500):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 40 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 300:
            current_lr = max(0.005, current_lr * 0.97)

        if no_improve > restart_period:
            noise_scale = 0.15
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.08 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def subgradient_optimize(a0, deadline, rng, lr=0.005, lr_decay=0.999,
                         num_peaks=5, smooth_factor=0.1):
    """Subgradient method with peak averaging and simplex projection."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-12)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    best_a = a0.copy()
    best_score = float(2.0 * n * max_c)
    current_lr = lr
    no_improve = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())

        # Get weights on top peaks (smoothed)
        if num_peaks > 1 and smooth_factor > 0:
            _, w = soft_max_weights(c, smooth_factor * max_c)
        else:
            w = np.zeros_like(c)
            top_idx = np.argsort(c)[-num_peaks:]
            w[top_idx] = 1.0 / num_peaks

        h = compute_h(w, a0)
        grad = 2.0 * h

        a0 = a0 - current_lr * grad
        a0 = project_simplex(a0)

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        current_lr = max(1e-6, current_lr * lr_decay)

        if no_improve > 500:
            break

    return best_a, best_score


def solve_lp(f, M, timeout=20.0):
    """Solve LP: max sum(g) s.t. C_f @ g <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    if n > 4096:
        return None

    # Build convolution matrix C: C[k,j] = f[k-j] if 0 <= k-j < n
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f

    g_var = cp.Variable(n)
    constraints = [g_var >= 0, C @ g_var <= M]
    objective = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(objective, constraints)

    t0 = time.monotonic()
    solvers = []
    if hasattr(cp, 'CLARABEL'):
        solvers.append(cp.CLARABEL)
    solvers.extend([cp.ECOS, cp.PDLP])

    for solver in solvers:
        if time.monotonic() - t0 > timeout:
            return None
        try:
            prob.solve(solver=solver)
            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0)
                s = float(g_val.sum())
                if s > 1e-15:
                    return g_val / s
                break
        except Exception:
            continue

    return None


def lp_refine_loop(f0, deadline, initial_t=0.4, t_decay=0.997, min_t=0.005,
                   lp_timeout=15.0, verbose=True):
    """Iterative LP-based refinement with decreasing step size."""
    f = f0.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return f0, np.inf
    f = f / s

    n = f.size
    c = full_convolve_fft(f)
    best_score = float(2.0 * n * c.max())
    best_f = f.copy()

    t = initial_t
    no_improve = 0
    n_iters = 0
    stuck_count = 0

    while time.monotonic() < deadline:
        if t < min_t:
            break

        c = full_convolve_fft(f)
        M = float(c.max())

        g = solve_lp(f, M, timeout=lp_timeout)

        if g is None:
            stuck_count += 1
            if stuck_count > 10:
                break
            t *= 0.9
            continue

        # Check if g is meaningfully different from f
        diff = float(np.sum(np.abs(g - f)))
        if diff < n * 1e-8:
            stuck_count += 1
            if stuck_count > 5:
                t *= 0.95
            if stuck_count > 20:
                break
            continue

        # Blend
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            t *= 0.9
            continue
        f_new = f_new / s_new

        c_new = full_convolve_fft(f_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new.copy()
            no_improve = 0
            stuck_count = 0
        else:
            no_improve += 1
            if no_improve > 30:
                t *= 0.95
                no_improve = 0

        t = max(min_t, t * t_decay)
        n_iters += 1

        if verbose and n_iters % 100 == 0:
            print("    LP iter {}: score={:.10f}, t={:.5f}".format(n_iters, best_score, t))

    return best_f, best_score
