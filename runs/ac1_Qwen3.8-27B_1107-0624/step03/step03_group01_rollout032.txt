LP-based subgradient method with smooth L-BFGS polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Score assuming sum(a)=1, so score = 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_lp_matrix(a, n):
    """Build the LP constraint matrix for (a*b)_k <= M constraints.
    Returns matrix of shape (2n-1, n) where row k, col j = a[k-j] if valid."""
    # For constraint k: sum_j a[k-j] * b[j] <= M, where k-j must be in [0, n-1]
    # So j must satisfy: 0 <= j <= n-1 and 0 <= k-j <= n-1
    # i.e., max(0, k-(n-1)) <= j <= min(k, n-1)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def _solve_lp(a, M, n):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0. Returns b or None."""
    # c = -1 (minimize -sum(b))
    c = -np.ones(n)
    # A_ub: shape (2n-1, n)
    A_ub = _build_lp_matrix(a, n)
    b_ub = np.full(2*n-1, M)
    # bounds: b >= 0
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(0, -res.x)
        else:
            return None
    except Exception:
        return None


def _lp_subgradient_step(a, n, t=0.1):
    """Perform one LP-based subgradient step.
    a should be normalized so sum(a)=1.
    Returns new a."""
    c = _full_convolve(a, a)
    M = float(c.max())
    
    b = _solve_lp(a, M, n)
    if b is None or b.sum() <= 0:
        return a.copy()
    
    # Rescale b to sum to 1
    b = b / b.sum()
    
    # Update
    a_new = (1 - t) * a + t * b
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new

def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _make_u_shaped(n, curvature):
    """Create a U-shaped profile: high at edges, low in middle."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + curvature * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, center_sep, width):
    """Create a bimodal profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.25 + center_sep/2) / width) ** 2) + \
        np.exp(-((t - 0.75 - center_sep/2) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asymmetric(n, bias):
    """Create an asymmetric profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + bias * np.sin(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_asymmetric(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

"""
LP-based subgradient method with smooth L-BFGS and Adam polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key insight: For non-negative sequences, the self-convolution peak can be below
the uniform baseline by using asymmetric/U-shaped profiles. The LP-based
subgradient method directly optimizes the max-convolution constraint, while
smooth L-BFGS with log-sum-exp provides gradient-based polishing.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_lp_matrix(a, n):
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        js = np.arange(j_min, j_max + 1)
        A[k, js] = a[k - js]
    return A


def _solve_lp(a, M, n):
    c_obj = -np.ones(n)
    A_ub = _build_lp_matrix(a, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return np.maximum(0.0, -res.x)
        return None
    except Exception:
        return None


def _lp_subgradient_step(a, n, step_size):
    c = _full_convolve(a, a)
    M = float(c.max())
    b = _solve_lp(a, M, n)
    if b is None or b.sum() <= 1e-15:
        return a.copy()
    b = b / b.sum()
    a_new = (1.0 - step_size) * a + step_size * b
    a_new = np.maximum(0.0, a_new)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _interp_profile(p0, n_new):

    print(f"Phase 4 done: best={best_score:.10f}")

    # Phase 5: Multi-restart with perturbation
    phase5_end = min(deadline, t_start + 1680.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = min(50.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15.0:
                break
            noise = 0.04 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=n_best)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            dl = time.monotonic() + per_restart
            if i % 2 == 0:
                p_best, sc_best = _run_lbfgs_continuation(p_start, dl, max_wall_time=per_restart)
            else:
                p_best, sc_best = _run_lp_subgradient(p_start, dl, max_iters=1500)
            _update(p_best, sc_best, f"restart{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # Phase 6: Try longer sequence via interpolation
    phase6_end = min(deadline, t_start + 1740.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr <= 2048:
            target_n = min(4096, n_curr * 2)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            _update(p_interp, sc_interp, f"interp_n{target_n}")
            if time.monotonic() < phase6_end - 30.0:
                wl = min(60.0, phase6_end - time.monotonic() - 10.0)
                if wl > 15.0:
                    dl = time.monotonic() + wl
                    p_best, sc_best = _run_lbfgs_continuation(p_interp, dl, max_wall_time=wl)
                    _update(p_best, sc_best, f"lbfgs_n{target_n}")

    # Phase 7: Final polish
    phase7_end = deadline
    if time.monotonic() < phase7_end and best_seq is not None:
        remaining = phase7_end - time.monotonic() - 5.0
        if remaining > 30.0:
            half = remaining / 2.0
            dl1 = time.monotonic() + half
            p_best, sc_best = _run_lbfgs_continuation(best_seq, dl1, max_wall_time=half)
            _update(p_best, sc_best, "final_lbfgs")
            remaining2 = phase7_end - time.monotonic() - 3.0
            if remaining2 > 15.0:
                dl2 = time.monotonic() + remaining2
                p_best2, sc_best2 = _adam_polish(best_seq, dl2, lr=0.01, beta_frac=0.002, n_max_iter=15000)
                _update(p_best2, sc_best2, "final_adam")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_u_shaped(1024, 3.0)
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")
        return [float(x) for x in final_seq]

    fallback = _make_u_shaped(1024, 3.0)
    print(f"FALLBACK: n=1024, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
