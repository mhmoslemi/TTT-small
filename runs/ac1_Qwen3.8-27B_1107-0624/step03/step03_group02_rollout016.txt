"""
Optimized step function coefficient search using multi-scale Adam smooth-max,

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of autocorrelation."""
    n = len(a)
    N = _next_pow2(3 * n)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    # Compute gradient via FFT
    w_pad = np.zeros(N)
    w_pad[:len(c)] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h, c


def _flatten_seq(a, target_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == target_n:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, target_n)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(target_n) / target_n
    return p / s


def _adam_smooth_max(a, deadline, max_iter=30000, lr=0.03, beta_init=0.05,
                     beta_decay=0.995, restart_every=3000, verbose=False):
    """Adam in logit-space with adaptive beta and structured restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_init * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    rng = np.random.default_rng(42 + n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        mval = float(c_curr.max())
        x = (c_curr - mval) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c_curr)
            w[int(np.argmax(c_curr))] = 1.0
        else:
            w = ex / s_ex

        # Gradient
        N = _next_pow2(3 * n)
        w_pad = np.zeros(N)
        w_pad[:len(c_curr)] = w
        a_rev = a[::-1].copy()
        a_pad = np.zeros(N)
        a_pad[:n] = a_rev
        fw = np.fft.rfft(w_pad, N)
        fa = np.fft.rfft(a_pad, N)
        conv_result = np.fft.irfft(fw * fa, N)
        h = conv_result[n - 1:2 * n - 1]
        grad_a = 2.0 * h

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            a = best_a.copy()
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.0005, lr_cur * 0.92)

        z = z_new
        a = a_new

        # Adaptive beta: decrease slowly, reset on restart
        if it % 100 == 0:
            beta = max(1e-12, beta * beta_decay)

        # Structured restart
        if no_imp > restart_every:
            strategy = rng.integers(0, 3)
            if strategy == 0:
                # Random noise
                z = best_z + 0.5 * rng.normal(size=n)
            elif strategy == 1:
                # Mirror perturbation
                z = best_z.copy()
                mirror = z[::-1].copy()
                z = 0.7 * z + 0.3 * mirror
            else:
                # Shift perturbation
                shift = rng.integers(1, max(2, n // 32))
                z = best_z.copy()
                z = np.roll(z, shift)
                z = 0.8 * z + 0.2 * best_z
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(0.02 * float(_fft_convolve(a).max()), 1e-10)
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr_cur = base_lr
            no_imp = 0

        if verbose and it % 5000 == 0:
            print(f"        Adam n={n} it={it}: best={best_score:.10f}, beta={beta:.2e}, lr={lr_cur:.6f}")

    return best_a, best_score


def _peak_shave(a, deadline, max_iter=500, decay=0.15, verbose=False):
    """Directly shave peaks by reducing contributors and redistributing mass."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        max_val = float(c.max())
        max_idx = int(np.argmax(c))

        # Find contributors to the peak
        contrib = np.zeros(n)
        j1_start = max(0, max_idx - n + 1)
        j1_end = min(max_idx, n - 1)
        if j1_start <= j1_end:
            idx = np.arange(j1_start, j1_end + 1)
            contrib[idx] += a[max_idx - idx]
        j2_start = max(max_idx, 0)
        j2_end = min(max_idx + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            contrib[idx] += a[idx - max_idx]

        # Reduce high contributors, increase low ones
        contrib_max = contrib.max()
        if contrib_max < 1e-15:
            break

        # Scale factor
        scale = decay / (1.0 + it * 0.1)
        a_new = a * (1.0 - scale * contrib / contrib_max)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break

        a = a_new

    return best_a


def _lp_spread(a, M):
    """LP: maximize sum(g) subject to conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    # Build constraint matrix A where A[k, j] = a[|k-j|] effectively
    # conv(a, g)[k] = sum_j a[j] * g[k-j]
    # For fixed a, this is linear in g
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[j] * g[k-j] contributes to position k
        # For position k, g index is k-j, valid when 0 <= k-j < n
        # So A[k, k-j] += a[j] for valid k-j
        valid_k_start = max(0, j)  # k-j >= 0 => k >= j
        valid_k_end = min(2 * n - 2, j + n - 1)  # k-j < n => k < j+n
        if valid_k_start <= valid_k_end:
            g_indices = np.arange(valid_k_start - j, valid_k_end - j + 1)
            A[valid_k_start:valid_k_end + 1, g_indices] += a[j]

    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _basis_optimize(a_init, deadline, max_iter=2000, n_basis=20, lr=0.05):
    """Optimize in a basis of Chebyshev polynomials for faster convergence."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    x = np.linspace(-1, 1, n)

    # Initialize basis coefficients by projecting
    T_k = np.zeros((n_basis, n))
    T_k[0] = 1.0
    if n_basis > 1:
        T_k[1] = x
    for k in range(2, n_basis):
        T_k[k] = 2 * x * T_k[k - 1] - T_k[k - 2]

    # Least squares fit
    coeffs, _, _, _ = np.linalg.lstsq(T_k, a, rcond=None)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    m_adam = np.zeros(n_basis)
    v_adam = np.zeros(n_basis)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute a from basis
        a_curr = np.maximum(T_k @ coeffs, 1e-12)
        a_curr = a_curr / a_curr.sum()

        # Compute gradient in coefficient space
        c = _fft_convolve(a_curr)
        mval = float(c.max())
        max_idx = int(np.argmax(c))

        # Subgradient of max autocorrelation w.r.t. a
        sg_a = np.zeros(n)
        j1s = max(0, max_idx - n + 1)
        j1e = min(max_idx, n - 1)
        if j1s <= j1e:
            idx = np.arange(j1s, j1e + 1)
            sg_a[idx] += a_curr[max_idx - idx]
        j2s = max(max_idx, 0)
        j2e = min(max_idx + n - 1, n - 1)
        if j2s <= j2e:
            idx = np.arange(j2s, j2e + 1)
            sg_a[idx] += a_curr[idx - max_idx]

        # Gradient w.r.t. coefficients (through the max(., 1e-12) approximation)
        # d a_curr / d coeff_k = T_k[k] (where a_curr > 0)
        active = a_curr > 1e-8
        grad_coeff = np.array([np.dot(T_k[k], sg_a[active]) / max(np.sum(active), 1)
                               for k in range(n_basis)])

        # Adam step
        t_adam += 1
        m_adam = b1 * m_adam + (1 - b1) * grad_coeff
        v_adam = b2 * v_adam + (1 - b2) * grad_coeff ** 2
        mh = m_adam / (1 - b1 ** t_adam)
        vh = v_adam / (1 - b2 ** t_adam)

        coeffs = coeffs - lr * mh / (np.sqrt(vh) + eps)

        a_new = np.maximum(T_k @ coeffs, 1e-12)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score_new = float(2.0 * n * _fft_convolve(a_new).max())
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()

    return best_a, best_score


def _multi_subgrad(a, deadline, max_iter=5000, top_k=5, verbose=False):
    """Subgradient targeting top-k peaks with decaying step size."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    base_lr = 0.3 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        max_val = float(c[top_idx[0]])

        # Weighted subgradient (heavier weight on higher peaks)
        peak_vals = c[top_idx]
        weights = np.exp((peak_vals - max_val) / max(0.01 * max_val, 1e-10))
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_idx):
            idx = int(idx)
            j1s = max(0, idx - n + 1)
            j1e = min(idx, n - 1)
            if j1s <= j1e:
                ii = np.arange(j1s, j1e + 1)
                sg[ii] += weights[i] * a[idx - ii]
            j2s = max(idx, 0)
            j2e = min(idx + n - 1, n - 1)
            if j2s <= j2e:
                ii = np.arange(j2s, j2e + 1)
                sg[ii] += weights[i] * a[ii - idx]

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score_new = float(2.0 * n * _fft_convolve(a_new).max())
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 5 * n):
                break

        a = a_new

    return best_a, best_score


def _spectral_shaping(a, deadline, max_iter=1000, lr=0.1):
    """Shape the spectrum to be flatter, reducing autocorrelation peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0

    # Initialize complex spectrum
    A = np.fft.rfft(a, N)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_curr = np.real(np.fft.irfft(A, N))[:n]
        a_curr = np.maximum(a_curr, 0)
        s_curr = a_curr.sum()
        if s_curr < 1e-10:
            break
        a_curr = a_curr / s_curr

        c = _fft_convolve(a_curr)
        score = float(2.0 * n * c.max())
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                break

        # Gradient: we want to reduce peaks in c = IFFT(|A|²)
        # In frequency domain, flatten |A|
        mag = np.abs(A)
        # Target: make |A[1:]| as equal as possible (keep A[0] fixed for sum constraint)
        if len(mag) > 1:
            target_mag = np.ones_like(mag)
            target_mag[0] = mag[0]
            # Gradient of (|A| - target)² w.r.t. A
            grad_A = A / mag * (mag - target_mag)
            grad_A = np.real(grad_A)  # take real part for the real-signal constraint
            # Actually for real signals, we work with rfft
            # The gradient w.r.t. the rfft coefficients needs care
            # Simplified: just nudge the magnitudes
            A_mag = mag.copy()
            A_phase = np.angle(A)
            A_mag_new = A_mag - lr * 0.01 * (A_mag[1:] - A_mag[1:].mean())
            A_mag_new[0] = A_mag[0]  # preserve DC
            A_mag_new = np.maximum(A_mag_new, 1e-10)
            A_new = A_mag_new * np.exp(1j * A_phase)
            # Enforce Hermitian symmetry is automatic with rfft/irfft
            A = A_new

    return best_a


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []

    # Linear taper
    a = 1.0 + 1.5 * (t - 0.5)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Stronger linear
    a = 1.0 + 2.5 * (t - 0.5)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Cosine
    a = 1.0 + 1.0 * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # DCOS (raised cosine window)
    a = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.5)))
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Gaussian
    a = np.exp(-2.0 * (t - 0.5) ** 2)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Wide Gaussian
    a = np.exp(-0.5 * (t - 0.5) ** 2)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Cubic
    a = 1.0 + 0.5 * u - 0.3 * u ** 3
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Quadratic (parabola)
    a = 1.0 - 0.5 * u ** 2
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Power law taper
    a = 1.0 - 0.3 * np.abs(u) ** 0.5
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Symmetric two-hump
    a = 1.0 + 0.4 * np.cos(2 * np.pi * u)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    return profiles


def _full_optimize(a0, deadline, n, label=""):
    """Run full optimization pipeline on a0 of length n."""
    sc0 = _score(a0)
    best_a = a0.copy()
    best_sc = sc0
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.10f}")

    # Phase 1: Adam (primary)
    if time.monotonic() < deadline - 30:
        d1 = min(deadline - 30, time.monotonic() + 180)
        a1, s1 = _adam_smooth_max(a0, d1, max_iter=20000, lr=0.03, beta_init=0.05)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam1: {best_sc:.10f}")

    # Phase 2: Peak shaving
    if time.monotonic() < deadline - 20:
        d2 = min(deadline - 20, time.monotonic() + 40)
        a2 = _peak_shave(best_a, d2, max_iter=300, decay=0.1)
        s2 = _score(a2)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} Shave: {best_sc:.10f}")

    # Phase 3: Multi-subgradient
    if time.monotonic() < deadline - 15:
        d3 = min(deadline - 15, time.monotonic() + 60)
        a3, s3 = _multi_subgrad(best_a, d3, max_iter=5000, top_k=5)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} MultiSG: {best_sc:.10f}")

    # Phase 4: LP refinement (only for n <= 1500)
    if n <= 1500 and time.monotonic() < deadline - 10:
        d4 = min(deadline - 10, time.monotonic() + 80)
        a4 = _lp_refine(best_a, d4, max_iter=1500, inflate=1.0005, blend=0.25)
        s4 = _score(a4)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} LP: {best_sc:.10f}")

    # Phase 5: Fine Adam
    if time.monotonic() < deadline - 5:
        d5 = deadline - 5
        a5, s5 = _adam_smooth_max(best_a, d5, max_iter=15000, lr=0.01, beta_init=0.02, beta_decay=0.997)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} Adam2: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining Adam smooth-max with adaptive beta,
    peak shaving, multi-peak subgradient, LP refinement, and spectral shaping.
    Explores multiple sequence lengths (512, 1024, 2048) with structured restarts
    and basis-function parameterization.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Prepare interpolated parents
    if parent is not None:
        p512 = _flatten_seq(parent, 512)
        p1024 = _flatten_seq(parent, 1024)
        p2048 = _flatten_seq(parent, 2048)
    else:
        p512 = _make_profiles(512)[0]
        p1024 = _make_profiles(1024)[0]
        p2048 = _make_profiles(2048)[0]

    # === Phase A: Quick baseline at n=512 (200s) ===
    print("=== Phase A: n=512 baseline ===")
    dA = min(deadline - 1300, t_start + 200)
    aA, sA = _full_optimize(p512, dA, 512, "A")
    if sA < best_score:
        best_score = sA
        best_seq = aA.copy()
    print(f"Phase A done: best={best_score:.10f}")

    # === Phase B: Main optimization at n=1024 (500s) ===
    print("=== Phase B: n=1024 main ===")
    dB = min(deadline - 700, t_start + 700)
    aB, sB = _full_optimize(p1024, dB, 1024, "B")
    if sB < best_score:
        best_score = sB
        best_seq = aB.copy()
    print(f"Phase B done: best={best_score:.10f}")

    # === Phase C: Try n=2048 (300s) ===
    if time.monotonic() < deadline - 400:
        print("=== Phase C: n=2048 ===")
        dC = min(deadline - 400, time.monotonic() + 300)
        aC, sC = _full_optimize(p2048, dC, 2048, "C")
        if sC < best_score:
            best_score = sC
            best_seq = aC.copy()
        print(f"Phase C done: best={best_score:.10f}")

    # === Phase D: Diverse initializations at n=1024 (200s) ===
    if time.monotonic() < deadline - 200:
        print("=== Phase D: Diverse inits @ n=1024 ===")
        n = 1024
        profiles = _make_profiles(n)
        # Add perturbed parent
        p_pert = p1024 * (1.0 + 0.2 * rng.normal(size=n))
        p_pert = np.maximum(p_pert, 0)
        profiles.append(p_pert / p_pert.sum())
        # Add shifted/mirrored variants
        profiles.append(p1024[::-1] / p1024.sum())
        mid = len(p1024) // 2
        p_mir = np.concatenate([p1024[:mid][::-1], p1024[mid:]])
        profiles.append(p_mir / p_mir.sum())

        for i, a0 in enumerate(profiles):
            if time.monotonic() > deadline - 200:
                break
            per_budget = min(30.0, max(15.0, (deadline - 200 - time.monotonic()) / max(1, len(profiles) - i)))
            dDi = min(deadline - 200, time.monotonic() + per_budget)
            # Quick: Adam + shave
            a_q, s_q = _adam_smooth_max(a0, dDi, max_iter=5000, lr=0.025, beta_init=0.04)
            if time.monotonic() < dDi + 10:
                a_q2 = _peak_shave(a_q, time.monotonic() + 5, max_iter=100, decay=0.08)
                s_q2 = _score(a_q2)
                if s_q2 < s_q:
                    a_q, s_q = a_q2, s_q2
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
                print(f"  D-{i}: IMPROVED to {s_q:.10f}")
            else:
                print(f"  D-{i}: {s_q:.10f}")

    # === Phase E: Final polish (200s) ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"=== Phase E: Final polish (current best={best_score:.10f}) ===")
        n_best = best_seq.size

        # E1: Fine Adam
        dE1 = min(deadline - 60, time.monotonic() + 80)
        aE1, sE1 = _adam_smooth_max(best_seq, dE1, max_iter=10000, lr=0.008,
                                     beta_init=0.01, beta_decay=0.998)
        if sE1 < best_score:
            best_score = sE1
            best_seq = aE1.copy()
        print(f"  E1 Adam: {best_score:.10f}")

        # E2: Peak shave
        if time.monotonic() < deadline - 40:
            dE2 = min(deadline - 40, time.monotonic() + 20)
            aE2 = _peak_shave(best_seq, dE2, max_iter=200, decay=0.05)
            sE2 = _score(aE2)
            if sE2 < best_score:
                best_score = sE2
                best_seq = aE2.copy()
            print(f"  E2 Shave: {best_score:.10f}")

        # E3: LP polish
        if n_best <= 1500 and time.monotonic() < deadline - 20:
            dE3 = min(deadline - 20, time.monotonic() + 40)
            aE3 = _lp_refine(best_seq, dE3, max_iter=1000, inflate=1.0003, blend=0.15)
            sE3 = _score(aE3)
            if sE3 < best_score:
                best_score = sE3
                best_seq = aE3.copy()
            print(f"  E3 LP: {best_score:.10f}")

        # E4: Final ultra-fine Adam
        if time.monotonic() < deadline - 5:
            dE4 = deadline - 5
            aE4, sE4 = _adam_smooth_max(best_seq, dE4, max_iter=8000, lr=0.003,
                                         beta_init=0.005, beta_decay=0.999)
            if sE4 < best_score:
                best_score = sE4
                best_seq = aE4.copy()
            print(f"  E4 Final Adam: {best_score:.10f}")

        # E5: One more subgradient pass
        if time.monotonic() < deadline - 3:
            dE5 = deadline - 3
            aE5, sE5 = _multi_subgrad(best_seq, dE5, max_iter=3000, top_k=3)
            if sE5 < best_score:
                best_score = sE5
                best_seq = aE5.copy()
            print(f"  E5 Subgrad: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_profiles(512)[0]
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_profiles(256)[0]

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                # Invalid, try fixing
                final_seq = np.maximum(final_seq, 0)
                final_seq[~np.isfinite(final_seq)] = 0
                s2 = final_seq.sum()
                if s2 < 0.01:
                    final_seq = _make_profiles(256)[0]
                else:
                    final_seq = final_seq / s2
        except Exception as e:
            print(f"  Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
