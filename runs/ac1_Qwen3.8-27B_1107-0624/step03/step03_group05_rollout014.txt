"""
Optimized step function search using:
1. Constrained LP (top-K peaks only) for fast spreading
2. Aggressive full-step updates with backtracking line search
3. Exact subgradient method at peak positions
4. Multi-length search with diverse initializations
5. Frequency-domain smoothing for regularization

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _make_conv_matrix_subset(a, constraint_indices):
    """Create constraint matrix only for specified convolution positions."""
    n = len(a)
    m = len(constraint_indices)
    A = np.zeros((m, n))
    for i, k in enumerate(constraint_indices):
        # conv(a,g)[k] = sum_j g[j] * a[k-j] for valid j
        # For each j, the coefficient of g[j] in conv(a,g)[k] is a[k-j] if 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[i, j] = a[k - j]

import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a, constraint_indices):
    """Create constraint matrix rows for specified convolution output positions."""
    n = len(a)
    m = len(constraint_indices)
    A = np.zeros((m, n))
    for i, k in enumerate(constraint_indices):
        # conv(a,g)[k] = sum_j g[j] * a[k-j] where 0<=k-j<n
        # So coefficient of g[j] is a[k-j] if 0 <= k-j < n
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[i, j] = a[k - j]
    return A


def _lp_spread(a, M, top_k=100):
    """LP: maximize sum(g) s.t. conv(a,g)[k] <= M for top-k positions k."""
    n = len(a)
    c = _fft_convolve(a)
    # Get top-k positions
    idx = np.argsort(c)[::-1][:top_k]
    
    A = _make_conv_matrix(a, idx)
    b = np.full(top_k, M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _gradient_smooth_max(a, beta):

    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)

    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c

    c = _fft_convolve(a)
    top_k = np.argsort(c)[::-1][:num_peaks]

    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:

    a_new = a_new / s

def _adam_polish(a, deadline, max_iter=20000, lr=0.04):
    """Adam-based polish in log-parameterization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    
    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.03 * float(c.max()))
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p  # small regularization
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.96)
        
        if it % 150 == 0:
            beta = max(1e-10, beta * 0.8)
        
        if no_imp > max(2000, 3 * n):
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.7 * best_z + 0.3 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.03 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_a, best_score


def _block_swap_optimize(a, deadline, max_iter=5000):
    """Random block swaps to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        block_size = rng.integers(max(2, n // 32), max(3, n // 8))
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) < block_size:
            continue
        
        a_new = a.copy()
        tmp = a_new[i:i + block_size].copy()
        a_new[i:i + block_size] = a_new[j:j + block_size]
        a_new[j:j + block_size] = tmp
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        
        sc_new = _score(a_new)
        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 1500:
                break
    
    return best_a, best_score


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_mixture(n, centers, sigmas, weights):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c, s, w in zip(centers, sigmas, weights):
        a += w * np.exp(-0.5 * ((t - c) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    """Run the full optimization pipeline for a given initial sequence."""
    a = a_init.copy()

    a = a / a.sum()
    n = len(a)

    best_score = _score(a)

    p1_budget = total_budget * 0.35
    p1_end = time.monotonic() + p1_budget
    a1, s1 = _lp_refine_fast(a, p1_end, top_k_start=30, top_k_end=100, max_iter=5000)
    elapsed = time.monotonic() - t0
    if s1 < best_score:

        best_a = a1.copy()

        best_score = s2
        best_a = a2.copy()
    print(f"    Phase2 LP-slow: {s2:.10f} ({elapsed:.1f}s)")
    
    # Phase 3: Subgradient polish
    p3_budget = total_budget * 0.15
    p3_end = time.monotonic() + p3_budget
    a3, s3 = _subgrad_polish(best_a, p3_end, max_iter=10000)

    print(f"    Phase3 SubGrad: {s3:.10f} ({elapsed:.1f}s)")
    
    # Phase 4: Smooth gradient polish
    p4_budget = total_budget * 0.1
    p4_end = time.monotonic() + p4_budget
    a4, s4 = _smooth_grad_polish(best_a, p4_end, max_iter=10000)
    elapsed = time.monotonic() - t0
    if s4 < best_score:
        best_score = s4
        best_a = a4.copy()
    print(f"    Phase4 SmoothGrad: {s4:.10f} ({elapsed:.1f}s)")

    p5_budget = total_budget * 0.1
    p5_end = time.monotonic() + p5_budget
    a5, s5 = _adam_polish(best_a, p5_end, max_iter=10000, lr=0.03)

    if s5 < best_score:
        best_score = s5
        best_a = a5.copy()

    p6_budget = total_budget * 0.05
    p6_end = time.monotonic() + p6_budget
    a6, s6 = _block_swap_optimize(best_a, p6_end, max_iter=3000)

    if s6 < best_score:
        best_score = s6
        best_a = a6.copy()
    print(f"    Phase6 BlockSwap: {s6:.10f} ({elapsed:.1f}s)")
    
    return best_a, best_score
