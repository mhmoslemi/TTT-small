            if no_imp == 10000:
                # Restart from best with noise
                z = best_z + 0.1 * np.random.default_rng(it * 31 + phase * 7).standard_normal(n)
                z -= np.mean(z)
                z_prev = z.copy()
                lr = lr_init * (0.8 ** phase)
                no_imp = 0
    
    return best_a

def _peak_equalize_vec(a, deadline, n_iter=5000, strength=0.05, n_peaks=3):
    """Vectorized peak equalization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    str_cur = strength
    no_imp = 0
    
    # Precompute the "pressure matrix" structure
    # pressure[i] = sum_k w_k * a_{k-i} for top k's
    # We'll compute this vectorized using FFT
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Build weight vector for top peaks
        w = np.zeros(2 * n - 1)
        for k in top_idx:
            w[k] = c[k] / c_max
        
        # Pressure = 2 * cross-correlation of w and a (same as LSE gradient with these weights)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        pressure = h_full[n - 1:2 * n - 1]
        
        if pressure.max() < 1e-15:
            break
        
        # Exponential tilt
        a_new = a * np.exp(-str_cur * pressure / (pressure.max() + 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = _score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.2, str_cur * 1.002)
        else:
            no_imp += 1
            str_cur = max(0.01, str_cur * 0.95)
            if no_imp > 500:
                break
        
        a = a_new
    
    return best_a

def _lp_spread(a, M, time_limit=15):
    """LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return res.x
    return None

def _lp_refine(a, deadline, max_iter=2000, inflate=1.0003, blend=0.15):
    """LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max()) * inflate_cur
        
        g = _lp_spread(a, M, time_limit=10)
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 50:
                break
            inflate_cur = min(1.003, inflate_cur * 1.01)
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 10 == 0:
                blend_cur = max(0.05, blend_cur * 0.95)
                inflate_cur = min(1.001, inflate_cur * 0.98)
            if no_imp > 100:
                a = best_a.copy()
                no_imp = 0
        
        a = a_new
    
    return best_a

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _make_bell(n, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
