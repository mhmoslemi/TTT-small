For fixed `a` of length n, the constraint `conv(a, g)[k] <= M` becomes:
`sum_{j: 0<=j<n, 0<=k-j<n} a[j] * g[k-j] <= M`

This is linear in `g`. The matrix is:

"""
Optimized step-function coefficient search minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive LP-based spreading with adaptive M, full-step updates,
simulated annealing escapes, multi-length search, and gradient polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h, float(c.max()), c


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, width=0.2):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.0) / (width * 0.5)) ** 2) + \
        np.exp(-0.5 * ((t - 1.0) / (width * 0.5)) ** 2) + 0.1
    a = a / a.sum()
    return a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_step(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _lp_optimize_aggressive(a, deadline, max_iter=50000, sa_temperature=0.0):
    """
    Aggressive LP-based optimization with adaptive M, full-step updates,
    and optional simulated annealing.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = 1.001
    blend = 0.4
    temperature = sa_temperature
    base_temperature = sa_temperature
    rng = np.random.default_rng(999)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        g = _lp_step(a, M_lp)

        if g is None:
            inflate *= 1.02
            if inflate > 1.1:
                break
            continue

        sum_g = float(g.sum())
        sum_a = float(a.sum())

        if sum_g <= sum_a * (1.0 + 1e-6):
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(1.05, inflate * 1.03)
            if no_imp > 500:
                # Perturbation: random block swap
                bs = max(2, n // 16)
                i = rng.integers(0, n - bs)
                j = rng.integers(0, n - bs)
                if abs(i - j) >= bs:
                    a_new = a.copy()
                    a_new[i:i+bs], a_new[j:j+bs] = a_new[j:j+bs].copy(), a_new[i:i+bs].copy()
                    a_new = np.maximum(a_new, 0.0)
                    sn = a_new.sum()
                    if sn > 1e-10:
                        a = a_new / sn
                        no_imp = 0
                        inflate = 1.001
                        blend = 0.4
                        continue
                break
            continue

        # We have an improvement. Try full step first.
        g_norm = g / sum_g
        c_g = _fft_convolve(g_norm)
        score_g = float(2.0 * n * c_g.max())

        if score_g < best_score - 1e-12:
            # Full step accepted
            best_score = score_g
            best_a = g_norm.copy()
            a = g_norm.copy()
            no_imp = 0
            inflate = max(1.0005, inflate * 0.95)
            blend = min(0.8, blend * 1.1)
        else:
            # Try blend
            t_blend = blend
            a_blend = (1.0 - t_blend) * a + t_blend * g_norm
            a_blend = np.maximum(a_blend, 0.0)
            s_blend = a_blend.sum()
            if s_blend > 1e-10:
                a_blend = a_blend / s_blend
            score_blend = _score(a_blend)

            if score_blend < best_score - 1e-12:
                best_score = score_blend
                best_a = a_blend.copy()
                a = a_blend.copy()
                no_imp = 0
                inflate = max(1.0005, inflate * 0.95)
            else:
                # Simulated annealing: accept with probability exp(-delta/T)
                delta = score_blend - best_score
                if temperature > 1e-10 and delta > 0:
                    accept_prob = np.exp(-delta / (temperature * best_score))
                    if rng.random() < accept_prob:
                        a = a_blend.copy()
                        no_imp = 0
                    else:
                        no_imp += 1
                else:
                    no_imp += 1

                if no_imp % 50 == 0:
                    blend = max(0.05, blend * 0.9)
                    inflate = min(1.02, inflate * 1.01)
                # Decay temperature
                temperature = max(0.0, base_temperature * (1.0 - it / max(1, max_iter)) ** 2)

    return best_a


def _gradient_adam_polish(a, deadline, max_iter=30000, lr=0.03):
    """Adam-based gradient polish on logit parameterization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c_arr = _fft_convolve(p)
        w = _smooth_max_weight(c_arr, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v_arr / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c_arr = _fft_convolve(a)
        score = float(2.0 * n * c_arr.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(3000, 4 * n):
            rng_r = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.8 * best_z + 0.2 * rng_r.normal(size=n) * 0.1
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c_arr = _fft_convolve(a)
            beta = max(1e-8, 0.05 * float(c_arr.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _momentum_gradient_polish(a, deadline, max_iter=30000, lr=0.02, momentum=0.9):
    """Momentum-based gradient polish on simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.05 * float(c.max()))
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0.0)
        s_nest = a_nest.sum()
        if s_nest > 1e-10:
            a_nest = a_nest / s_nest
        else:
            a_nest = a

        grad, maxc, c_arr = _gradient_smooth_max(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c_arr = _fft_convolve(a)
        score = float(2.0 * n * c_arr.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 150 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a


def _block_perturb_and_refine(a, deadline, rng, max_swaps=500):
    """Apply random block swaps and keep the best."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_swaps):
        if time.monotonic() > deadline:
            break

        bs = rng.integers(max(2, n // 32), max(3, n // 8))
        i = rng.integers(0, n - bs)
        j = rng.integers(0, n - bs)
        if abs(i - j) < bs:
            continue

        a_new = a.copy()
        a_new[i:i+bs], a_new[j:j+bs] = a_new[j:j+bs].copy(), a_new[i:i+bs].copy()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 1000:
                break

    return best_a, best_score


def _peak_repair(a, deadline, max_iter=5000):
    """Target the top peaks of the convolution for direct reduction."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(42)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        max_c = float(c.max())
        # Find top peaks
        top_k = np.argsort(c)[::-1][:3]
        peak_lag = int(top_k[0])

        # For peak at lag k, c[k] = sum_j a[j]*a[k-j]
        # Find the j that contributes most
        j_valid = np.arange(max(0, peak_lag - n + 1), min(peak_lag + 1, n))
        if len(j_valid) == 0:
            break
        contributions = np.array([a[j] * a[peak_lag - j] for j in j_valid])
        j_peak = j_valid[int(np.argmax(contributions))]
        j_partner = peak_lag - j_peak

        # Try moving mass from j_peak to a random position
        j_target = rng.integers(0, n)
        for frac in [0.05, 0.1, 0.2]:
            delta = frac * a[j_peak]
            a_try = a.copy()
            a_try[j_peak] -= delta
            a_try[j_target] += delta
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_score


def _run_pipeline_for_length(a_init, n, deadline):
    """Run full optimization pipeline for one length."""
    a = np.asarray(a_init, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_u_profile(n, 2.0)
    a = a / s
    a = a[:n] if a.size >= n else np.pad(a, (0, n - a.size))
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    print(f"    Initial: {best_score:.10f}")

    # Phase 1: Aggressive LP (most important)
    rem = deadline - time.monotonic()
    if rem > 30:
        lp_deadline = time.monotonic() + min(rem * 0.5, 300.0)
        a_lp = _lp_optimize_aggressive(a, lp_deadline, max_iter=30000, sa_temperature=0.001)
        sc = _score(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
        print(f"    LP1: {sc:.10f}")

    # Phase 2: Gradient polish
    rem = deadline - time.monotonic()
    if rem > 20:
        gp_deadline = time.monotonic() + min(rem * 0.2, 120.0)
        a_gp = _gradient_adam_polish(best_a, gp_deadline, max_iter=20000, lr=0.025)
        sc = _score(a_gp)
        if sc < best_score:
            best_score = sc
            best_a = a_gp.copy()
        print(f"    Adam: {sc:.10f}")

    # Phase 3: More LP (with restart)
    rem = deadline - time.monotonic()
    if rem > 20:
        lp2_deadline = time.monotonic() + min(rem * 0.25, 150.0)
        # Perturb slightly before second LP
        rng = np.random.default_rng(777)
        a_pert = best_a * (1.0 + 0.05 * rng.normal(size=n))
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()
        a_lp2 = _lp_optimize_aggressive(a_pert, lp2_deadline, max_iter=20000, sa_temperature=0.0005)
        sc = _score(a_lp2)
        if sc < best_score:
            best_score = sc
            best_a = a_lp2.copy()
        # Also try LP on best directly
        a_lp2b = _lp_optimize_aggressive(best_a, lp2_deadline, max_iter=10000, sa_temperature=0.0)
        sc = _score(a_lp2b)
        if sc < best_score:
            best_score = sc
            best_a = a_lp2b.copy()
        print(f"    LP2: {sc:.10f}")

    # Phase 4: Block perturb
    rem = deadline - time.monotonic()
    if rem > 15:
        bs_deadline = time.monotonic() + min(rem * 0.15, 60.0)
        rng = np.random.default_rng(12345)
        a_bs, sc_bs = _block_perturb_and_refine(best_a, bs_deadline, rng, max_swaps=2000)
        if sc_bs < best_score:
            best_score = sc_bs
            best_a = a_bs.copy()
        print(f"    BlockSwap: {sc_bs:.10f}")

    # Phase 5: Peak repair
    rem = deadline - time.monotonic()
    if rem > 10:
        pr_deadline = time.monotonic() + min(rem * 0.1, 40.0)
        a_pr, sc_pr = _peak_repair(best_a, pr_deadline, max_iter=3000)
        if sc_pr < best_score:
            best_score = sc_pr
            best_a = a_pr.copy()
        print(f"    PeakRepair: {sc_pr:.10f}")

    # Phase 6: Final momentum gradient
    rem = deadline - time.monotonic()
    if rem > 10:
        mg_deadline = time.monotonic() + min(rem * 0.15, 60.0)
        a_mg = _momentum_gradient_polish(best_a, mg_deadline, max_iter=15000, lr=0.015)
        sc = _score(a_mg)
        if sc < best_score:
            best_score = sc
            best_a = a_mg.copy()
        print(f"    Momentum: {sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: aggressive LP with SA escapes, Adam gradient,
    block perturbations, peak repair, and multi-length search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 768, 1024, 1536]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))
    # Sort: try the parent length first, then others
    lengths.sort(key=lambda x: (0 if (parent is not None and x == parent.size) else 1, x))

    print(f"Lengths to try: {lengths}")
    print(f"Deadline in {deadline - time.monotonic():.0f}s")

    # Allocate budget: more time for more promising lengths
    # Parent length gets 40%, others share the rest
    num_lengths = len(lengths)
    if num_lengths == 1:
        budgets = [deadline - time.monotonic() - 30.0]
    else:
        # First length (parent) gets 45%, rest share 55%
        parent_budget = (deadline - time.monotonic() - 30.0) * 0.45
        other_budget = (deadline - time.monotonic() - 30.0) * 0.55
        others_per = other_budget / (num_lengths - 1)
        budgets = [parent_budget] + [others_per] * (num_lengths - 1)

    for i, n_t in enumerate(lengths):
        if time.monotonic() > deadline - 60:
            break

        elapsed = time.monotonic() - t_start
        remaining = deadline - time.monotonic() - 30.0
        if remaining < 60:
            break

        # Scale budget based on remaining time
        alloc = min(budgets[i], remaining - 30.0)
        if alloc < 30:
            continue

        print(f"\n=== Length {n_t} (budget {alloc:.0f}s) ===")

        # Create initialization
        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
        else:
            a_init = _make_u_profile(n_t, 2.0)

        sc_init = _score(a_init)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()

        # Run pipeline
        phase_deadline = time.monotonic() + alloc
        a_best_n, sc_best_n = _run_pipeline_for_length(a_init, n_t, phase_deadline)
        if sc_best_n < best_score:
            best_score = sc_best_n
            best_seq = a_best_n.copy()
            print(f"  ** New best at n={n_t}: {sc_best_n:.10f}")

    # Phase 2: Try diverse initializations at the best length
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = best_seq.size
        print(f"\n=== Phase 2: Diverse inits at n={n_best} ===")

        inits = []
        # U-shaped with various alphas
        for alpha in [1.0, 1.5, 2.5, 3.5, 5.0]:
            inits.append(_make_u_profile(n_best, alpha))
        # Bimodal
        inits.append(_make_bimodal_profile(n_best, 0.3))
        inits.append(_make_bimodal_profile(n_best, 0.15))
        # Linear
        for eta in [1.0, 2.0, 3.0]:
            inits.append(_make_linear_profile(n_best, eta))
        # Perturbed best
        for mag in [0.1, 0.2, 0.3]:
            a_pert = best_seq * (1.0 + mag * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            inits.append(a_pert)
        # Random
        for _ in range(3):
            a_rand = np.abs(rng.normal(1.0, 0.8, n_best)) + 0.01
            a_rand = a_rand / a_rand.sum()
            inits.append(a_rand)

        # Limit based on time
        rem = deadline - time.monotonic() - 50.0
        max_inits = max(1, int(rem / 40.0))
        inits = inits[:min(len(inits), max_inits)]

        for a0 in inits:
            if time.monotonic() > deadline - 50:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            rem = deadline - time.monotonic() - 40.0
            if rem < 20:
                break
            sub_budget = min(50.0, rem * 0.4)
            a_res, sc_res = _run_pipeline_for_length(a0, n_best, time.monotonic() + sub_budget)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best: {sc_res:.10f} (from diverse init)")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        final_rem = deadline - time.monotonic() - 25.0
        print(f"\n=== Phase 3: Final polish (budget {final_rem:.0f}s) ===")

        # 3a: LP with very tight inflate
        b = min(final_rem * 0.3, final_rem)
        if b > 10:
            a_lp = _lp_optimize_aggressive(best_seq, time.monotonic() + b,
                                           max_iter=30000, sa_temperature=0.0)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  Final LP: {sc:.10f}")

        # 3b: Adam
        b = min(final_rem * 0.2, time.monotonic() + final_rem - time.monotonic())
        b = min(final_rem * 0.2, max(0, deadline - time.monotonic() - 20.0))
        if b > 10:
            a_ad = _gradient_adam_polish(best_seq, time.monotonic() + b,
                                         max_iter=20000, lr=0.015)
            sc = _score(a_ad)
            if sc < best_score:
                best_score = sc
                best_seq = a_ad.copy()
            print(f"  Final Adam: {sc:.10f}")

        # 3c: Momentum
        b = min(final_rem * 0.15, max(0, deadline - time.monotonic() - 20.0))
        if b > 10:
            a_mg = _momentum_gradient_polish(best_seq, time.monotonic() + b,
                                             max_iter=15000, lr=0.01)
            sc = _score(a_mg)
            if sc < best_score:
                best_score = sc
                best_seq = a_mg.copy()
            print(f"  Final Momentum: {sc:.10f}")

        # 3d: Peak repair
        b = min(final_rem * 0.1, max(0, deadline - time.monotonic() - 20.0))
        if b > 5:
            a_pr, sc_pr = _peak_repair(best_seq, time.monotonic() + b, max_iter=2000)
            if sc_pr < best_score:
                best_score = sc_pr
                best_seq = a_pr.copy()
            print(f"  Final PeakRepair: {sc_pr:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_u_profile(768, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_u_profile(768, 2.0)

    # Evaluate with authoritative function
    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
