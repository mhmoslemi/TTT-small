# Helper functions at top level

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    c = _fft_convolve(a)

def _lse_step(a, z, state, T, lr):

    c = _fft_convolve(a)

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999

    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)

6. We can do this by transferring mass from high-product pairs to low-product pairs

"""
Multi-method optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LSE gradient with cyclic annealing, aggressive LP spreading,
targeted coordinate descent on peak-affecting coordinates, and
multi-length exploration. Starts from parent sequence for warm start.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-10)
    s = p.sum()
    return p / s


def _lse_gradient(a, T):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    return gz if np.all(np.isfinite(gz)) else None


def _lse_optimize(a, deadline, max_iter=100000, T_start=0.08, T_end=1e-5, lr=0.08,
                  cyclic=False, cycle_len=5000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    state_t = 0
    state_m = np.zeros(n)
    state_v = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    b1, b2 = 0.9, 0.999
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        if cyclic:
            phase = (it % cycle_len) / cycle_len
            T = T_start * (T_end / T_start) ** (phase * 0.5)
            T = max(T, T_end)
        else:
            T = T_start * T_ratio ** it
            T = max(T, T_end)

        gz = _lse_gradient(a, T)
        if gz is None:
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state_t = 0
            state_m = np.zeros(n)
            state_v = np.zeros(n)
            continue

        state_t += 1
        state_m = b1 * state_m + (1.0 - b1) * gz
        state_v = b2 * state_v + (1.0 - b2) * (gz * gz)
        mh = state_m / (1.0 - b1 ** state_t)
        vh = state_v / (1.0 - b2 ** state_t)
        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a_new = e / e.sum()

        if not np.all(np.isfinite(a_new)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state_t = 0
            state_m = np.zeros(n)
            state_v = np.zeros(n)
            continue

        c = _fft_convolve(a_new)
        sc = float(2.0 * n * c.max())

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 5 * n):
                rr = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.04 * rr.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state_t = 0
                state_m = np.zeros(n)
                state_v = np.zeros(n)
                no_imp = 0

    return best_a


def _lp_direction(a, M, n_top=150):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:min(n_top, len(c_aa))]:
        for off in range(-3, 4):
            j = i + off
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate_start=1.001, inflate_end=1.0001,
               blend_start=0.3, blend_end=0.05):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    inflate = inflate_start
    blend = blend_start
    total_iters = max(10, min(max_iter, int((deadline - time.monotonic()) * 5)))
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        g = _lp_direction(a, M_lp)
        if g is None:
            inflate = min(1.01, inflate * 1.3)
            no_imp += 3
            if no_imp > 200:
                break
            continue
        sg = float(g.sum())
        if sg <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(1.01, inflate * 1.1)
            if no_imp > 300:
                break
            continue
        g = g / sg
        t_frac = float(it) / total_iters
        blend_cur = blend_start + (blend_end - blend_start) * t_frac
        inflate_cur = inflate_start + (inflate_end - inflate_start) * t_frac

        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()
        c_new = _fft_convolve(a_new)
        sc_new = float(2.0 * n * c_new.max())

        if sc_new < best_score - 1e-13:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 10 == 0:
                blend = max(blend_end, blend * 0.85)
                inflate = min(1.008, inflate * 0.95)
            if no_imp > 200:
                a = best_a.copy()
                no_imp = 0
                inflate = inflate_start
                blend = blend_start
        a = a_new
    return best_a


def _peak_target_descent(a, deadline, max_iter=30000, lr=0.01):
    """Directly target reduction of max convolution entries."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    no_imp = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        threshold = m_val - 0.02 * m_val
        hot = np.where(c >= threshold)[0]
        if len(hot) == 0:
            hot = [int(np.argmax(c))]

        grad = np.zeros(n)
        for k in hot:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += a[partner]
        grad *= 2.0
        dot = float(np.dot(a, grad))
        grad_t = grad - dot * a
        lr_cur = lr * (1.0 - 0.5 * float(it) / max_iter)
        v = 0.7 * v - lr_cur * grad_t
        a_new = a + v
        a_new = np.maximum(a_new, 1e-12)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        sc = float(2.0 * n * c_new.max())
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.001, lr * 0.85)
            if no_imp > 5000:
                a = best_a.copy()
                v = np.zeros(n)
                no_imp = 0
                lr = 0.01
    return best_a


def _coord_descent(a, deadline, n_iter=80000, seed=77):
    """Randomized coordinate pair descent targeting peak convolution."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(seed)
    c = _fft_convolve(a)
    max_k = int(np.argmax(c))
    no_imp = 0

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # Focus on coordinates that affect the peak lag
        if rng.random() < 0.6:
            lo = max(0, max_k - n + 1)
            hi = min(n, max_k + 1)
            i = int(rng.integers(lo, hi))
            j = int(rng.integers(lo, hi))
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))

        if i == j:
            continue

        ai = a[i]
        aj = a[j]
        if ai < 1e-12 and aj < 1e-12:
            continue

        # Try transferring mass from larger to smaller
        trial_done = False
        for frac in [0.3, 0.5, 0.7]:
            if ai >= aj:
                delta = frac * (ai - aj)
                na, nb = ai - delta, aj + delta
            else:
                delta = frac * (aj - ai)
                na, nb = ai + delta, aj - delta
            if na < 0 or nb < 0:
                continue
            a_trial = a.copy()
            a_trial[i] = na
            a_trial[j] = nb
            s = a_trial.sum()
            a_trial = a_trial / s
            sc = _score_norm(a_trial)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                c = _fft_convolve(a)
                max_k = int(np.argmax(c))
                no_imp = 0
                trial_done = True
                break
        if not trial_done:
            no_imp += 1
            if no_imp > 10000:
                break
    return best_a


def _symmetric_optimize(a, deadline, max_iter=5000, lr=0.05):
    """Optimize assuming symmetry a[i] = a[n-1-i]."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    # Enforce symmetry
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 1e-12)
    a_sym = a_sym / a_sym.sum()

    best_a = a_sym.copy()
    best_score = _score_norm(a_sym)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a_sym)
        m_val = float(c.max())
        x = (c - m_val) / 0.03
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        w = ex / ex.sum()
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_sym[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        # Symmetrize gradient
        grad = 0.5 * (grad + grad[::-1])
        dot = float(np.dot(a_sym, grad))
        grad = grad - dot * a_sym
        a_new = a_sym - lr * grad / max(np.linalg.norm(grad), 1e-10)
        a_new = 0.5 * (a_new + a_new[::-1])
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()
        sc = _score_norm(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a_sym = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break
    return best_a

        sc_fin = _score(a_fin)
        print(f"  Final LSE: {sc_fin:.10f}")
        if np.isfinite(sc_fin) and sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()

    # Final LP micro-polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        end_flp = min(time.monotonic() + 30, deadline - 15)
        a_flp = _lp_refine(best_seq, end_flp, max_iter=500,
                           inflate_start=1.0005, inflate_end=1.0001,
                           blend_start=0.1, blend_end=0.02)
        sc_flp = _score(a_flp)
        print(f"  Final LP: {sc_flp:.10f}")
        if np.isfinite(sc_flp) and sc_flp < best_score:
            best_score = sc_flp
            best_seq = a_flp.copy()

    # Validate
    if best_seq is None:
        best_seq = np.ones(256) / 256.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
