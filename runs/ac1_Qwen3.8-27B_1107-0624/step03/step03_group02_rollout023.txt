def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _fista_smooth_max(a0, deadline, lr=None, beta0=0.1, beta_min=0.001, max_iter=30000):
    """
    FISTA (Nesterov accelerated) method in logit space for smooth max of autocorrelation.
    Returns best_a found and its score.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return a, np.inf
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    if lr is None:
        lr = 0.02

    beta = beta0
    t_fista = 1.0
    z_y = z.copy()  # Nesterov iterate
    
    no_imp = 0
    lr_cur = lr
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        alpha = (1.0 - t_fista) / t_new
        z_y = z + alpha * (z_y - z)
        t_fista = t_new

        # Convert z_y to a_y
        z_y_clipped = np.clip(z_y, -60.0, 60.0)
        z_y_clipped -= float(np.mean(z_y_clipped))
        e_y = np.exp(z_y_clipped)
        a_y = e_y / e_y.sum()

        # Compute smooth max weights
        c_y = _fft_convolve(a_y)
        max_c = float(c_y.max())
        x = (c_y - max_c) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = ex / s_ex

        # Compute gradient in a-space
        grad_a = _grad_smooth_max(a_y, w, beta)

        # Convert to logit-space gradient
        dot = float(np.dot(a_y, grad_a))
        grad_z = a_y * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_a.copy()
            z = np.log(np.maximum(z, 1e-14))
            z -= float(np.mean(z))
            z_y = z.copy()
            t_fista = 1.0
            continue

        # Clamp gradient
        gnorm = float(np.linalg.norm(grad_z))
        if gnorm > 100.0:
            grad_z = grad_z * (100.0 / gnorm)

        # Update
        z_new = z_y - lr_cur * grad_z
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        # Check score
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(0.0005, lr_cur * 0.95)
            if no_imp > max(3000, 4 * n):
                # Restart
                rng_r = np.random.default_rng(42 + it)
                z = np.log(np.maximum(best_a, 1e-14))
                z -= float(np.mean(z))
                z += 0.15 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                z_y = z.copy()
                t_fista = 1.0
                lr_cur = base_lr
                no_imp = 0
                beta = max(beta_min, beta * 0.8)
                break

        z = z_new

        # Anneal beta
        if it % 100 == 0:
            beta = max(beta_min, beta * 0.85)

    return best_a, best_score


def _adam_smooth_max(a0, deadline, lr=0.03, beta_factor=0.05, max_iter=30000):

    v_arr = np.zeros(n)
    adam_t = 0

    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    base_lr = lr

            v_arr = np.zeros(n)

        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)

                z = np.clip(z, -60.0, 60.0)

                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)

                adam_t = 0
                lr_cur = base_lr

        a = a_new

def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _subgradient_at_max(a, k):
    """Compute dc[k]/da vectorized for full convolution index k."""
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _multi_subgradient(a0, deadline, max_iter=10000, top_k=5, base_lr=None):
    """Multi-peak subgradient method."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    if base_lr is None:
        base_lr = 0.15 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])

        # Exponential weighting for top peaks
        weights = np.array([np.exp(-0.5 * i) for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _lp_find_direction(f, M):
    """
    Find g >= 0 maximizing sum(g) subject to (f*g)[k] <= M for all k.
    Returns the optimal g (unnormalized) or None.
    """
    n = len(f)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] for valid indices
    # For each k in [0, 2n-2]: sum_j f[j]*g[k-j] <= M
    # g has n components, indexed 0..n-1
    
    # For constraint k: g appears with coefficients f[j] where k-j is in [0,n-1]
    # i.e., j in [k-n+1, k] intersected with [0, n-1]
    # So g[k-j] appears with coefficient f[j], meaning g[i] appears with coefficient f[k-i]
    
    # Let's build the matrix A where A[k, i] = coefficient of g[i] in constraint k
    # (f*g)[k] = sum_i f[k-i] * g[i] where 0<=k-i<n and 0<=i<n
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # g[i] contributes to (f*g)[k] for k = i+j where j in [0, n-1]
        # coefficient is f[j] = f[k-i]
        for j in range(n):
            k = i + j
            A[k, i] = f[j]
    
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_solution': False})
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=2000, inflate=1.002, blend=0.3):
    """LP-based refinement: find directions that spread mass while keeping peak bounded."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return a, np.inf
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        M = float(c_curr.max())
        M_lp = M * inflate_cur

        g = _lp_find_direction(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.04)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.97)
            if no_imp > max(150, int(1.5 * n)):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a, best_score


def _peak_chase(a0, deadline, max_iter=5000):
    """Directly chase and reduce the current maximum peak."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    lr = 0.5 / n

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_idx = int(np.argmax(c_curr))
        max_val = float(c_curr[max_idx])

        sg = _subgradient_at_max(a, max_idx)
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
        else:
            # Reduce step size
            lr *= 0.9

        a = a_new

    return best_a, best_score


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_shape_profile(n, gamma=0.5):
    """U-shaped profile: higher at edges, lower in middle."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + gamma * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _pipeline_fista_lp(a0, deadline, label=""):
    """FISTA + LP hybrid pipeline."""
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"    {label} init: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: FISTA
    if time.monotonic() < deadline - 30:
        d1 = min(deadline - 30, time.monotonic() + 180)
        a1, s1 = _fista_smooth_max(a0, d1, lr=0.02, beta0=0.08, beta_min=0.001)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"    {label} FISTA: {best_sc:.10f}")

    # Phase 2: LP
    if time.monotonic() < deadline - 20:
        d2 = min(deadline - 20, time.monotonic() + 80)
        a2, s2 = _lp_refine(best_a, d2, max_iter=500, inflate=1.001, blend=0.25)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"    {label} LP: {best_sc:.10f}")

    # Phase 3: Adam
    if time.monotonic() < deadline - 15:
        d3 = min(deadline - 15, time.monotonic() + 120)
        a3, s3 = _adam_smooth_max(best_a, d3, lr=0.015, beta_factor=0.03)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"    {label} Adam: {best_sc:.10f}")

    # Phase 4: Peak chase
    if time.monotonic() < deadline - 10:
        d4 = min(deadline - 10, time.monotonic() + 60)
        a4, s4 = _peak_chase(best_a, d4, max_iter=3000)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"    {label} PeakChase: {best_sc:.10f}")

    # Phase 5: Multi-subgradient
    if time.monotonic() < deadline - 5:
        d5 = min(deadline - 5, time.monotonic() + 40)
        a5, s5 = _multi_subgradient(best_a, d5, max_iter=3000, top_k=5)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"    {label} MultiSG: {best_sc:.10f}")

    return best_a, best_sc


def _pipeline_deep(a0, deadline, label=""):
    """Deeper pipeline with multiple LP passes and adaptive FISTA."""
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"    {label} init: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: FISTA (aggressive)
    if time.monotonic() < deadline - 60:
        d1 = min(deadline - 60, time.monotonic() + 250)
        a1, s1 = _fista_smooth_max(a0, d1, lr=0.025, beta0=0.1, beta_min=0.0005)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"    {label} FISTA1: {best_sc:.10f}")

    # Phase 2: LP
    if time.monotonic() < deadline - 50:
        d2 = min(deadline - 50, time.monotonic() + 100)
        a2, s2 = _lp_refine(best_a, d2, max_iter=800, inflate=1.001, blend=0.2)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"    {label} LP1: {best_sc:.10f}")

    # Phase 3: FISTA (fine)
    if time.monotonic() < deadline - 40:
        d3 = min(deadline - 40, time.monotonic() + 150)
        a3, s3 = _fista_smooth_max(best_a, d3, lr=0.01, beta0=0.02, beta_min=0.0001)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"    {label} FISTA2: {best_sc:.10f}")

    # Phase 4: LP again
    if time.monotonic() < deadline - 30:
        d4 = min(deadline - 30, time.monotonic() + 80)
        a4, s4 = _lp_refine(best_a, d4, max_iter=500, inflate=1.0005, blend=0.15)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"    {label} LP2: {best_sc:.10f}")

    # Phase 5: Peak chase + multi-SG
    if time.monotonic() < deadline - 10:
        d5 = min(deadline - 10, time.monotonic() + 50)
        a5, s5 = _peak_chase(best_a, d5, max_iter=3000)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        if time.monotonic() < deadline - 5:
            d6 = min(deadline - 5, time.monotonic() + 30)
            a6, s6 = _multi_subgradient(best_a, d6, max_iter=2000, top_k=7)
            if s6 < best_sc:
                best_a, best_sc = a6, s2
        print(f"    {label} Final: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: FISTA smooth-max + LP spreading + peak chasing.
    Explores multiple initializations and sequence lengths with adaptive time allocation.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Prepare initializations
    if parent is not None:
        p256 = _interpolate_profile(parent, 256)
        p512 = _interpolate_profile(parent, 512)
        p768 = _interpolate_profile(parent, 768)
        p1024 = _interpolate_profile(parent, 1024)
    else:
        p256 = _make_linear_profile(256, 1.5)
        p512 = _make_linear_profile(512, 1.5)
        p768 = _make_linear_profile(768, 1.5)
        p1024 = _make_linear_profile(1024, 1.5)

    # Also prepare U-shape variants
    u512 = _make_u_shape_profile(512, 0.8)
    u1024 = _make_u_shape_profile(1024, 0.8)

    print(f"Starting optimization with budget {budget_s}s, deadline at t={deadline - t_start:.1f}s")

    # === Phase A: Best effort on parent@1024 (or n=1024 linear) ===
    if parent is not None:
        print("=== Phase A: Parent @ n=1024 (deep pipeline) ===")
        d_a = min(deadline - 600, t_start + 550)
        a_best, s_best = _pipeline_deep(p1024, d_a, "A")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
    else:
        print("=== Phase A: Linear @ n=1024 ===")
        d_a = min(deadline - 600, t_start + 400)
        a_best, s_best = _pipeline_fista_lp(p1024, d_a, "A")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()

    # === Phase B: Try n=512 (sometimes shorter is better) ===
    if time.monotonic() < deadline - 400:
        print("=== Phase B: Best @ n=512 ===")
        if parent is not None:
            init_b = p512
        else:
            init_b = _make_linear_profile(512, 1.5)
        d_b = min(deadline - 400, time.monotonic() + 350)
        a_b, s_b = _pipeline_fista_lp(init_b, d_b, "B")
        if s_b < best_score:
            best_score = s_b
            best_seq = a_b.copy()

    # === Phase C: U-shape at n=1024 ===
    if time.monotonic() < deadline - 250:
        print("=== Phase C: U-shape @ n=1024 ===")
        d_c = min(deadline - 250, time.monotonic() + 200)
        a_c, s_c = _pipeline_fista_lp(u1024, d_c, "C")
        if s_c < best_score:
            best_score = s_c
            best_seq = a_c.copy()

    # === Phase D: U-shape at n=512 ===
    if time.monotonic() < deadline - 150:
        print("=== Phase D: U-shape @ n=512 ===")
        d_d = min(deadline - 150, time.monotonic() + 120)
        a_d, s_d = _pipeline_fista_lp(u512, d_d, "D")
        if s_d < best_score:
            best_score = s_d
            best_seq = a_d.copy()

    # === Phase E: Diverse short initializations at n=512 ===
    if time.monotonic() < deadline - 100:
        print("=== Phase E: Diverse inits @ n=512 ===")
        n = 512
        diverse = [
            _make_cubic_profile(n, 2.0, -2.5),
            _make_cubic_profile(n, 1.0, -0.8),
            _make_cosine_profile(n, 0.7),
            _make_cosine_profile(n, 1.5),
            _make_bell_profile(n, 0.4),
            _make_u_shape_profile(n, 1.2),
            _make_linear_profile(n, 2.5),
        ]
        # Add perturbed parent
        if parent is not None:
            p_pert = p512 * (1.0 + 0.1 * rng.normal(size=n))
            p_pert = np.maximum(p_pert, 0.0)
            diverse.append(p_pert / p_pert.sum())

        for i, a0 in enumerate(diverse):
            if time.monotonic() > deadline - 100:
                break
            per_time = min(25.0, (deadline - 100 - time.monotonic()) / max(1, len(diverse) - i))
            if per_time < 8:
                break
            d_ei = min(deadline - 100, time.monotonic() + per_time)
            a_ei, s_ei = _adam_smooth_max(a0, d_ei, lr=0.02, beta_factor=0.04)
            if s_ei < best_score:
                best_score = s_ei
                best_seq = a_ei.copy()
                print(f"    E-{i} improved: {best_score:.10f}")

    # === Phase F: Final polish on best ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"=== Phase F: Final polish (best={best_score:.10f}) ===")

        # Ensure we're working with the right length
        bn = best_seq.size

        # FISTA fine
        if time.monotonic() < deadline - 15:
            d_f1 = min(deadline - 15, time.monotonic() + 80)
            a_f1, s_f1 = _fista_smooth_max(best_seq, d_f1, lr=0.008, beta0=0.01, beta_min=0.0001)
            if s_f1 < best_score:
                best_score = s_f1
                best_seq = a_f1.copy()
            print(f"    FISTA final: {best_score:.10f}")

        # LP polish
        if time.monotonic() < deadline - 10:
            d_f2 = min(deadline - 10, time.monotonic() + 50)
            a_f2, s_f2 = _lp_refine(best_seq, d_f2, max_iter=500, inflate=1.0005, blend=0.1)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"    LP final: {best_score:.10f}")

        # Peak chase
        if time.monotonic() < deadline - 5:
            d_f3 = min(deadline - 5, time.monotonic() + 30)
            a_f3, s_f3 = _peak_chase(best_seq, d_f3, max_iter=2000)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"    PeakChase final: {best_score:.10f}")

        # Multi-SG final
        if time.monotonic() < deadline - 3:
            d_f4 = min(deadline - 3, time.monotonic() + 20)
            a_f4, s_f4 = _multi_subgradient(best_seq, d_f4, max_iter=2000, top_k=7)
            if s_f4 < best_score:
                best_score = s_f4
                best_seq = a_f4.copy()
            print(f"    MultiSG final: {best_score:.10f}")

    # === Phase G: Try extending best to n=1024 and re-optimizing ===
    if best_seq is not None and best_seq.size != 1024 and time.monotonic() < deadline - 30:
        print("=== Phase G: Extend to n=1024 ===")
        best_1024 = _interpolate_profile(best_seq, 1024)
        d_g = min(deadline - 10, time.monotonic() + 40)
        a_g, s_g = _adam_smooth_max(best_1024, d_g, lr=0.01, beta_factor=0.02)
        if s_g < best_score:
            best_score = s_g
            best_seq = a_g.copy()
            print(f"    Extended: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
                elif sc_eval > best_score * 1.01:
                    # Something is off, use our score
                    pass
        except Exception as e:
            print(f"  Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
