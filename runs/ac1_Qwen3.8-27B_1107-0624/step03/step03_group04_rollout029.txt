    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _coord_descent(a, deadline, max_sweeps=20000, tol=1e-12):
    """
    Coordinate descent exploiting 1D convex structure.
    For each coordinate i, minimize max_k conv(a,a)[k] over a[i] >= 0
    with all other coordinates fixed.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / a.sum()
    n = len(a)
    
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    
    no_sweep_imp = 0
    total_imp = 0.0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        sweep_imp = 0.0
        c = _fft_convolve(a)  # Recompute at start of sweep
        
        for i in range(n):
            if time.monotonic() > deadline + 5:  # Small tolerance
                break
            
            x_old = a[i]
            
            # Compute A_k and B_k for all k
            # conv(a,a)[k] = sum_{i'} a[i'] * a[k-i']
            # Term involving a[i] = x:
            #   if 0 <= k-i < n and k-i != i: contributes 2 * x * a[k-i] to lag k
            #   if k == 2*i (and 0 <= 2*i <= 2*n-2): contributes x^2 to lag k
            #   (note: when k-i == i, i.e., k=2i, the single term is x^2, not 2*x*x)
            
            # A[k] = coefficient of x in lag k
            # B[k] = coefficient of x^2 in lag k
            # C[k] = constant part of lag k (when x=x_old, C[k] = c[k] - A[k]*x_old - B[k]*x_old^2)
            
            A = np.zeros(2 * n - 1)
            B = np.zeros(2 * n - 1)
            
            # For lag k, a[i] participates if:
            # i' = i: need 0 <= k-i < n, contributes a[k-i] * x (if k-i != i) or x^2 (if k-i == i)
            # i' = k-i: need 0 <= k-i < n and k-i != i, contributes a[k-i] * x
            
            # k ranges from i to i + (n-1) for the first case (i' = i)
            # k ranges from i to i + (n-1) for the second case (i' = k-i, meaning k = i + j for j in [0,n-1], j != i)
            
            # More carefully:
            # For i' = i: k - i' = k - i must be in [0, n-1], so k in [i, i+n-1]
            #   If k - i != i (i.e., k != 2i): contributes x * a[k-i]
            #   If k - i == i (i.e., k == 2i): contributes x * x = x^2
            # For i' = k - i (must be != i, and in [0, n-1]): 
            #   k - i in [0, n-1] and k - i != i, so k in [i, i+n-1] and k != 2i
            #   Contributes a[k-i] * x
            
            # So for k in [i, min(i+n-1, 2n-2)]:
            #   If k == 2i: B[k] = 1, A[k] = 0
            #   Else: A[k] = 2 * a[k-i], B[k] = 0
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            for k in range(k_start, k_end + 1):
                if k == 2 * i:
                    B[k] = 1.0
                else:
                    A[k] = 2.0 * a[k - i]
            
            # C[k] = c[k] - A[k] * x_old - B[k] * x_old^2
            C = c - A * x_old - B * x_old * x_old
            
            # Now minimize f(x) = max_k (C[k] + A[k]*x + B[k]*x^2) for x >= 0
            # This is a 1D convex optimization (max of convex functions)
            
            # Find a reasonable upper bound for x
            # f(x) >= C[k] + A[k]*x + B[k]*x^2 for all k
            # For large x, the max will be large. Let's find an upper bound.
            # f(x_old) = c_max (approximately, since at x_old, f = max(c) which is the current max)
            # Actually f(x_old) = max_k c[k] = c_max
            c_max = float(c.max())
            f_at_old = c_max
            
            # Upper bound: we want f(x_upper) > f_at_old significantly
            # The quadratic terms give x^2 growth, so x_upper = c_max + 1 might work
            # But let's be more careful.
            # The maximum coefficient of x is max(A) = 2 * max(a) (for lags where a[k-i] is max)
            # The maximum coefficient of x^2 is 1 (at k=2i)
            # So f(x) >= x^2 + 2*max_a*x + min_C (for k=2i)
            # At x = x_old + 1: f >= (x_old+1)^2 + ... which should be > c_max for reasonable x_old
            # Let's use x_upper = x_old + max(1.0, 2 * c_max / (max(a) + 1e-15))
            x_upper = x_old + max(1.0, x_old)
            # Evaluate f at x_upper
            vals = C + A * x_upper + B * x_upper * x_upper
            if vals.max() < f_at_old:
                x_upper *= 2
                vals = C + A * x_upper + B * x_upper * x_upper
            
            # Golden section search on [0, x_upper]
            # But we need to be careful: the minimum might be at x=0
            # Let's check f(0) first
            f0 = float(C.max())  # f(0) = max_k C[k]
            
            if f0 < f_at_old - 1e-15:
                # Try golden section on [0, x_upper]
                gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
                lo = 0.0
                hi = x_upper
                # Evaluate at two interior points
                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)
                f1 = float((C + A * x1 + B * x1 * x1).max())
                f2 = float((C + A * x2 + B * x2 * x2).max())
                
                for _ in range(60):
                    if f1 < f2:
                        hi = x2
                        x2 = x1
                        f2 = f1
                        x1 = hi - gr * (hi - lo)
                        f1 = float((C + A * x1 + B * x1 * x1).max())
                    else:
                        lo = x1
                        x1 = x2
                        f1 = f2
                        x2 = lo + gr * (hi - lo)
                        f2 = float((C + A * x2 + B * x2 * x2).max())
                
                x_opt = (lo + hi) / 2
                f_opt = float((C + A * x_opt + B * x_opt * x_opt).max())
                
                if f_opt < f_at_old - 1e-15:
                    delta = f_at_old - f_opt
                    sweep_imp += delta
                    total_imp += delta
                    a[i] = x_opt
            # If f0 >= f_at_old, the minimum is at x_old (no improvement possible from this coordinate)
        
        # End of sweep
        c = _fft_convolve(a)
        cur_score = float(2.0 * n * c.max())
        
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            no_sweep_imp = 0
        else:
            no_sweep_imp += 1
        
        if no_sweep_imp > 50 and sweep_imp < 1e-14:
            break
        
        if sweep > 100 and total_imp < 1e-10 and sweep_imp < 1e-12:
            break
    
    return best_a

**Improved LP:**

Variables: `b_0, ..., b_{n-1}, t` (n+1 variables)
Objective: minimize `t`
Constraints:
1. For each lag `k` in `[0, 2n-2]`: `2 * conv(a, b)[k] + conv(a,a)[k] <= t`

   Where `c[k] = conv(a,a)[k]`
   
2. `sum(b) = 0` (maintain normalization)
3. `b[i] >= -a[i]` for all `i` (maintain non-negativity of `a+b`)

The constraint 1 can be written as:
`sum_{i'} a[i'] * b[k-i'] <= (t - c[k]) / 2`

This is linear in `b` and `t`.

In matrix form, constraint 1 for lag `k`:
`sum_{i=0}^{n-1} (a[i] if 0 <= k-i < n else 0) * b[i] - t <= -c[k]/2`

The coefficient of `b[i]` in constraint `k` is `a[max(0, k-i)]` if `0 <= k-i < n`, else 0.
Wait, it's `a[i']` where `i' = k - i`... no.

`conv(a, b)[k] = sum_{i=0}^{n-1} a[i] * b[k-i]` (with the convention that `b[j] = 0` for `j < 0` or `j >= n`).

So `conv(a, b)[k] = sum_{i: 0<=k-i<n} a[i] * b[k-i] = sum_{j: 0<=j<n, 0<=k-j<n} a[k-j] * b[j]`

where `j = k-i`.

So the coefficient of `b[j]` in constraint `k` is `a[k-j]` if `0 <= k-j < n`, else 0.

This is a Toeplitz-like matrix. For `n=768` and `2n-1 = 1535` constraints, the matrix is `1535 x 768`.

- Inequality constraints: `A_ub @ x <= b_ub`

"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / len(a)
    return a / s


def _coord_descent(a, deadline, max_sweeps=50000, verbose=False):
    """
    Coordinate descent exploiting 1D convex structure.
    For each coordinate i, exactly minimize max_k conv(a,a)[k] over a[i] >= 0.
    Uses incremental convolution updates.
    """
    a = _normalize(a)
    n = len(a)
    c = _fft_convolve(a)
    c_max = float(c.max())
    best_score = float(2.0 * n * c_max)
    best_a = a.copy()

    no_imp_sweeps = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        sweep_improved = False
        total_sweep_delta = 0.0
        c = _fft_convolve(a)
        cur_cmax = float(c.max())

        for i in range(n):
            if time.monotonic() > deadline:
                break

            x_old = a[i]

            # Affected lags: k in [i, min(i+n-1, 2n-2)]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            seg_len = k_hi - k_lo + 1

            # a[k-i] for k in [k_lo, k_hi] is a[0], a[1], ..., a[seg_len-1]
            a_seg = a[0:seg_len]
            c_seg = c[k_lo:k_hi + 1].copy()

            # Identify diagonal (k = 2i) within the segment
            diag_idx = 2 * i - k_lo  # position of k=2i in the segment
            is_diag = (diag_idx >= 0) and (diag_idx < seg_len)

            # Max of unaffected entries (computed once per coordinate)
            # For efficiency, just use c.max() as upper bound (slightly conservative)
            max_unaff = cur_cmax

            # Define f(x) using the segment
            # f(x) = max(max_unaff, max(c_seg + delta_seg(x)))
            # where delta_seg[k_lo+j] = 2*(x-x_old)*a_seg[j] for j != diag_idx
            #                          x^2 - x_old^2 for j == diag_idx

            if is_diag:
                c_non_diag = np.delete(c_seg, diag_idx)
                a_non_diag = np.delete(a_seg, diag_idx)
                c_diag_val = c_seg[diag_idx]
            else:
                c_non_diag = c_seg
                a_non_diag = a_seg
                c_diag_val = None

            # Precompute: max of unaffected (exclude affected range)
            if k_lo > 0:
                max_unaff = max(max_unaff, float(c[:k_lo].max()))
            if k_hi < 2 * n - 2:
                max_unaff = max(max_unaff, float(c[k_hi + 1:].max()))

            def f_eval(x):
                d = x - x_old
                if len(c_non_diag) > 0:
                    lin_vals = c_non_diag + 2.0 * d * a_non_diag
                    lin_max = float(lin_vals.max())
                else:
                    lin_max = -1e30
                if c_diag_val is not None:
                    quad_val = c_diag_val + x * x - x_old * x_old
                else:
                    quad_val = -1e30
                return max(max_unaff, lin_max, quad_val)

            f_old = f_eval(x_old)

            # Upper bound for search
            x_upper = max(2.0 * x_old + 0.01, 0.1)
            f_upper = f_eval(x_upper)
            guard = 0
            while f_upper < f_old and guard < 20:
                x_upper *= 2.0
                f_upper = f_eval(x_upper)
                guard += 1

            # Check if 0 is better
            f_zero = f_eval(0.0)

            # Golden section search on [0, x_upper]
            lo = 0.0
            hi = x_upper
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = f_eval(x1)
            f2 = f_eval(x2)

            for _ in range(45):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    f1 = f_eval(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    f2 = f_eval(x2)

            x_opt = (lo + hi) / 2.0
            f_opt = f_eval(x_opt)

            if f_opt < f_old - 1e-14:
                # Update a[i] and c incrementally
                a[i] = x_opt
                d = x_opt - x_old
                if abs(d) > 1e-16:
                    # Update c in the affected range
                    c[k_lo:k_hi + 1] += 2.0 * d * a_seg
                    if is_diag:
                        # Correct: the diagonal should get x^2-x_old^2, not 2*d*a_seg[diag_idx]
                        # a_seg[diag_idx] = a[i] which is now x_opt, so we added 2*d*x_opt
                        # but we should have added x_opt^2 - x_old^2
                        # Correction: subtract 2*d*x_opt, add x_opt^2-x_old^2
                        c[2 * i] -= 2.0 * d * x_opt
                        c[2 * i] += x_opt * x_opt - x_old * x_old
                cur_cmax = max(cur_cmax, f_opt)
                sweep_improved = True
                total_sweep_delta += (f_old - f_opt)

        # End sweep
        if not sweep_improved:
            no_imp_sweeps += 1
            if no_imp_sweeps > 5:
                break
        else:
            no_imp_sweeps = 0

        # Recompute c periodically for accuracy
        if sweep % 50 == 0:
            c = _fft_convolve(a)
            cur_cmax = float(c.max())
            sc = float(2.0 * n * cur_cmax)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
            if verbose and sweep % 100 == 0:
                print(f"    CD sweep {sweep}: score={sc:.10f}", flush=True)

    # Final
    a = _normalize(a)
    c = _fft_convolve(a)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a


def _improved_lp(a, deadline, max_iters=5000, verbose=False):
    """
    Improved LP: min t s.t. 2*conv(a,b)[k] + c[k] <= t for all k,
    sum(b)=0, a+b >= 0.
    Then try step sizes alpha in {0.3, 0.5, 0.7, 1.0} on direction b.
    """
    a = _normalize(a)
    n = len(a)
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    # Build LP matrix: for each lag k, coeff of b[j] is a[k-j] if 0<=k-j<n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Hankel-like matrix

    # For efficiency, build it row by row
    m = 2 * n - 1
    A_lp = np.zeros((m, n))
    for k in range(m):
        # b[j] appears in conv(a,b)[k] with coefficient a[k-j] if 0<=k-j<n
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A_lp[k, j] = a[k - j]

    # LP: min t
    # s.t. 2*A_lp @ b - t <= -c  (m constraints)
    #      -b[i] <= a[i]  (n constraints, non-negativity of a+b)
    #      sum(b) = 0  (1 equality)
    #
    # Variables: [b_0,...,b_{n-1}, t] (n+1 vars)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t

    # Inequality: 2*A_lp @ b - t <= -c
    A_ub_top = np.zeros((m, n + 1))
    A_ub_top[:, :n] = 2.0 * A_lp
    A_ub_top[:, -1] = -1.0
    b_ub_top = -c

    # Inequality: -b[i] <= a[i] => -e_i^T b <= a[i]
    A_ub_bot = np.zeros((n, n + 1))
    A_ub_bot[np.arange(n), np.arange(n)] = -1.0
    b_ub_bot = a

    A_ub = np.vstack([A_ub_top, A_ub_bot])
    b_ub = np.concatenate([b_ub_top, b_ub_bot])

    # Equality: sum(b) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = [(None, None)] * (n + 1)

    no_imp = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if not res.success:
            break

        b_dir = res.x[:n]
        t_val = res.x[-1]

        # Try different step sizes
        best_alpha = 0.0
        best_sc = best_score
        for alpha in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_new = a + alpha * b_dir
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_alpha = alpha

        if best_alpha == 0.0:
            no_imp += 1
            if no_imp > 20:
                break
            # Perturb: slightly inflate the max constraint
            c = _fft_convolve(a)
            b_ub_top = -c * (1.0 + 0.001 * (no_imp + 1))
            A_ub = np.vstack([A_ub_top, A_ub_bot])
            b_ub = np.concatenate([b_ub_top, b_ub_bot])
            continue
        else:
            a = _normalize(a + best_alpha * b_dir)
            no_imp = 0

        c = _fft_convolve(a)
        cur_sc = float(2.0 * n * c.max())
        if cur_sc < best_score - 1e-13:
            best_score = cur_sc
            best_a = a.copy()

        # Rebuild LP matrix for new a
        for k in range(m):
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            A_lp[k, :] = 0.0
            for j in range(j_start, j_end):
                A_lp[k, j] = a[k - j]
        A_ub_top = np.zeros((m, n + 1))
        A_ub_top[:, :n] = 2.0 * A_lp
        A_ub_top[:, -1] = -1.0
        A_ub = np.vstack([A_ub_top, A_ub_bot])

        if verbose and it % 10 == 0:
            print(f"    LP iter {it}: score={cur_sc:.10f}, alpha={best_alpha}", flush=True)

    return best_a


def _gradient_polish(a, deadline, max_iter=20000):
    """Smooth max gradient descent in log-space for final polish."""
    a = _normalize(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.02 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= z.mean()
    lr = 0.005
    no_imp = 0

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if not np.isfinite(s) or s <= 0:
            break
        w = ex / s

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= gz.mean()

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        sc = float(2.0 * n * c.max())
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr *= 0.9
                beta = max(1e-10, beta * 0.8)

        if no_imp > max(5000, 10 * n):
            break

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _alt_descent_lp(a, deadline, cd_sweeps=2000, lp_iters=200, verbose=False):

            a_pert = a * (1.0 + noise)
            a_pert = np.maximum(a_pert, 0.0)
            a = a_pert / a_pert.sum()
            no_final_imp = 0

        phase += 1

    return best_a


def _multi_length_search(parent, deadline, lengths, verbose=False):
    """Try optimization at multiple lengths, return best."""
    best_score = np.inf
    best_seq = None
    best_len = 0

    for n_t in lengths:
        if time.monotonic() > deadline - 60:
            break

        if parent is not None and parent.size == n_t:
            a0 = parent.copy()
        elif parent is not None:
            a0 = _interpolate_profile(parent, n_t)
        else:
            a0 = np.ones(n_t) / n_t

        sc0 = _score(a0)
        if verbose:
            print(f"  Length {n_t}: init score={sc0:.10f}", flush=True)

        # Allocate time proportional to remaining budget
        remaining = deadline - time.monotonic()
        time_left = remaining / max(1, len(lengths))
        phase_deadline = time.monotonic() + min(time_left, 400.0)

        a_opt = _alt_descent_lp(a0, phase_deadline, cd_sweeps=3000, lp_iters=300, verbose=verbose)
        sc_opt = _score(a_opt)

        if verbose:
            print(f"  Length {n_t}: after alt = {sc_opt:.10f}", flush=True)

        # Gradient polish
        if time.monotonic() < deadline - 20:
            gp_deadline = min(deadline - 15, time.monotonic() + 60.0)
            a_gp = _gradient_polish(a_opt, gp_deadline, max_iter=10000)
            sc_gp = _score(a_gp)
            if sc_gp < sc_opt:
                a_opt = a_gp
                sc_opt = sc_gp

        # Final coordinate descent
        if time.monotonic() < deadline - 10:
            cd_deadline = min(deadline - 5, time.monotonic() + 40.0)
            a_cd = _coord_descent(a_opt, cd_deadline, max_sweeps=5000)
            sc_cd = _score(a_cd)
            if sc_cd < sc_opt:
                a_opt = a_cd
                sc_opt = sc_cd

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            best_len = n_t

    return best_seq, best_score, best_len


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining coordinate descent, improved LP, and gradient polish
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_score = _score(parent)
        best_seq = parent.copy()

    # Determine lengths to try
    parent_n = parent.size
    lengths = sorted(set([parent_n, 512, 768, 1024]))
    # Remove lengths that are too far from parent
    lengths = [l for l in lengths if 0.5 * parent_n <= l <= 2.0 * parent_n]
    if not lengths:
        lengths = [768]

    print(f"Trying lengths: {lengths}, budget={budget_s-.0f}s", flush=True)

    # Main search
    seq, sc, ln = _multi_length_search(parent, deadline, lengths, verbose=True)
    if seq is not None and sc < best_score:
        best_score = sc
        best_seq = seq

    # Final: if time remains, do one more round of CD on the best
    if best_seq is not None and time.monotonic() < deadline - 5:
        cd_dead = deadline - 3
        a_final = _coord_descent(best_seq, cd_dead, max_sweeps=10000)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final
        print(f"After final CD: {best_score:.10f}", flush=True)

    # Validate with evaluate_sequence
    if best_seq is None:
        best_seq = parent.copy()

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = parent.copy() if parent is not None else np.ones(512) / 512

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
            print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
        else:
            print(f"evaluate_sequence: {sc_eval}", flush=True)
    except Exception as e:
        print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
