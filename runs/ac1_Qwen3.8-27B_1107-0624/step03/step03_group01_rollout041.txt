```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, fast O(n)-per-trial local search
with pair moves, and multi-scale exploration (n=2048, n=4096).
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1 (scale-invariant score)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _make_ramp(n, power=1.0):
    """Create a normalized ramp sequence."""
    t = np.arange(n, dtype=np.float64)
    a = t + 1.0
    a = a ** power
    return a / a.sum()


def _make_symmetric_window(n, width=0.3):
    """Create a normalized symmetric window."""
    t = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2*t - 1) / max(width, 0.01)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _interp_profile(p0, n_new):
    """Interpolate profile p0 to new length n_new."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-300:
        return np.ones(n_new) / n_new
    return p / s


def _lbfgs_smoothed_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_smoothed_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_starts=None):
    """Run L-BFGS with continuation on the smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_starts is None:
        beta_starts = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.001, 0.0005]
    
    betas = [b * max_c0 for b in beta_starts]
    
    total_time = max(1.0, deadline - time.monotonic())
    per_level = total_time / len(betas)
    
    for beta in betas:
        level_dl = time.monotonic() + per_level
        if level_dl > deadline:
            level_dl = deadline
        if time.monotonic() >= deadline:
            break
        
        try:
            result = minimize(
                _lbfgs_smoothed_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_smoothed_grad,
                options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 20, 'maxfun': 1000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _local_search(a, deadline, rng, use_simulated_annealing=True):
    """
    Fast O(n)-per-trial local search.
    a: normalized sequence (sum=1)
    Modifies a in place and returns (a, best_score).
    """
    n = len(a)
    conv = _full_convolve(a, a)
    current_max = float(conv.max())
    best_a = a.copy()
    best_max = current_max
    
    # Adaptive delta
    delta_base = 0.001 / n  # small relative to typical element values (which are ~1/n)
    
    # Temperature for simulated annealing
    T = 1e-6
    T_start = 1e-5
    T_end = 1e-8
    
    iterations = 0
    accepted = 0
    last_recompute = 0
    
    while time.monotonic() < deadline:
        # Gradually decrease temperature
        progress = min(1.0, iterations / 100000.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Choose random index
        i = rng.integers(0, n)
        
        # Choose delta (try both signs)
        delta = delta_base * rng.uniform(0.5, 2.0)
        if rng.random() < 0.5:
            delta = -delta
        
        # Compute new max
        # partner: conv positions i to i+n-1 get 2*delta*a added
        lo = i
        hi = i + n
        
        # Slice of conv that changes
        slice_conv = conv[lo:hi]
        new_slice = slice_conv + 2.0 * delta * a
        # Add delta^2 at position 2*i (which is at index 2*i - lo = i within the slice)
        if 2 * i < conv.size:
            new_slice[i] += delta * delta
        
        # Unaffected parts
        unaff_left = conv[:lo].max() if lo > 0 else -np.inf
        unaff_right = conv[hi:].max() if hi < conv.size else -np.inf
        new_max = max(unaff_left, unaff_right, float(new_slice.max()))
        
        # Decide whether to accept
        if new_max < current_max:
            accept = True
        elif use_simulated_annealing and T > 0:
            worsening = new_max - current_max
            if worsening < 0.01 * current_max and rng.random() < np.exp(-worsening / max(T, 1e-15)):
                accept = True
            else:
                accept = False
        else:
            accept = False
        
        if accept:
            # Update
            a[i] += delta
            if a[i] < 0:
                a[i] = 0
                delta = -a[i]  # correction
            conv[lo:hi] += 2.0 * delta * a
            if 2 * i < conv.size:
                conv[2 * i] += delta * delta
            current_max = new_max
            accepted += 1
            iterations += 10  # count as more work
            
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        else:
            iterations += 1
        
        # Periodic recompute to correct drift
        if accepted - last_recompute > 500:
            conv = _full_convolve(a, a)
            current_max = float(conv.max())
            last_recompute = accepted
            # Renormalize
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            conv = _full_convolve(a, a)
            current_max = float(conv.max())
    
    # Final: return best found
    final_score = _score_norm_fn(best_a)
    return best_a, final_score
```

```python
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, gradient-guided directed local search
with pair moves, random local search, and multi-scale exploration (n up to 4096).

from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _make_ramp(n, power=1.0):
    t = np.arange(n, dtype=np.float64)
    a = (t + 1.0) ** power
    return a / a.sum()


def _make_quadratic_ramp(n, p1, p2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + p1 * t + p2 * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-300:
        return np.ones(n_new) / n_new
    return p / s

    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_grad_func(z, args):
    beta, n = args
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())

    per_level = total_time / len(betas)

                    z = result.x.copy()

    k_star_cache = int(np.argmax(conv))

    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1

        # Recompute peak position periodically
        if iter_count % 50 == 1 or iter_count == 1:
            k_star_cache = int(np.argmax(conv))

        # Compute partner values at the peak
        # partner[i] = a[k_star - i] if 0 <= k_star - i <= n-1
        partner = np.zeros(n)
        lo = max(0, k_star_cache - n + 1)
        hi = min(n - 1, k_star_cache)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            partner[idx] = a[k_star_cache - idx]

        # Sort by partner value: we want to move from high-partner to low-partner
        # Pick a random "source" from top half and "destination" from bottom half
        sorted_idx = np.argsort(partner)
        half = n // 2
        # Source: high partner (end of sorted), Destination: low partner (start of sorted)
        src_pool = sorted_idx[half:]
        dst_pool = sorted_idx[:half]
        i = int(rng.integers(0, len(src_pool)))
        j = int(rng.integers(0, len(dst_pool)))
        i_idx = int(src_pool[i])
        j_idx = int(src_pool[j])  # Wait, j should be from dst_pool
        j_idx = int(dst_pool[j])

        if i_idx == j_idx:
            continue

        # Try several delta values
        found = False
        for d_scale in [1.0, 0.5, 2.0, 0.25, 4.0]:
            delta = delta_base * d_scale
            if a[i_idx] < delta:
                continue

            # Compute new conv: -2*delta*a_shifted_i + 2*delta*a_shifted_j
            new_max = current_max
            # Check the affected region
            # Positions i_idx to i_idx+n-1 get -2*delta*a added
            # Positions j_idx to j_idx+n-1 get +2*delta*a added
            lo_i = i_idx
            hi_i = i_idx + n
            lo_j = j_idx
            hi_j = j_idx + n

            # We need max of: conv + (-2*delta*a at [lo_i,hi_i)) + (2*delta*a at [lo_j,hi_j))
            # + delta^2*(delta at 2*i_idx - delta at 2*j_idx)
            # For efficiency, compute the max over the union of affected regions
            lo_union = min(lo_i, lo_j)
            hi_union = max(hi_i, hi_j)
            lo_union = max(0, lo_union)
            hi_union = min(2 * n - 1, hi_union)

            region = conv[lo_union:hi_union].copy()
            # Apply -2*delta*a at positions [lo_i, hi_i)
            overlap_lo_i = max(lo_i, lo_union) - lo_union
            overlap_hi_i = min(hi_i, hi_union) - lo_union
            if overlap_lo_i < overlap_hi_i:
                region[overlap_lo_i:overlap_hi_i] += -2.0 * delta * a[overlap_lo_i - lo_i:overlap_hi_i - lo_i]
            # Apply +2*delta*a at positions [lo_j, hi_j)
            overlap_lo_j = max(lo_j, lo_union) - lo_union
            overlap_hi_j = min(hi_j, hi_union) - lo_union
            if overlap_lo_j < overlap_hi_j:
                region[overlap_lo_j:overlap_hi_j] += 2.0 * delta * a[overlap_lo_j - lo_j:overlap_hi_j - lo_j]
            # Add delta^2 terms
            if 2 * i_idx >= lo_union and 2 * i_idx < hi_union:
                region[2 * i_idx - lo_union] += delta * delta
            if 2 * j_idx >= lo_union and 2 * j_idx < hi_union:
                region[2 * j_idx - lo_union] -= delta * delta

            region_max = float(region.max())
            # Also check unaffected regions
            if lo_union > 0:
                region_max = max(region_max, float(conv[:lo_union].max()))
            if hi_union < 2 * n - 1:
                region_max = max(region_max, float(conv[hi_union:].max()))

            if region_max < current_max:
                # Accept
                a[i_idx] -= delta
                a[j_idx] += delta
                conv[lo_i:hi_i] += -2.0 * delta * a
                conv[lo_j:hi_j] += 2.0 * delta * a
                conv[2 * i_idx] += delta * delta
                conv[2 * j_idx] -= delta * delta
                current_max = region_max
                accepted += 1
                stall_count = 0
                found = True
                if current_max < best_max:
                    best_max = current_max
                    best_a = a.copy()
                break
        if not found:
            stall_count += 1
            if stall_count > 200:
                # Random restart: try a random pair
                i_idx = int(rng.integers(0, n))
                j_idx = int(rng.integers(0, n))
                if i_idx != j_idx and a[i_idx] > 0:
                    delta = delta_base * rng.uniform(0.1, 1.0)
                    if a[i_idx] > delta:
                        lo_i = i_idx
                        hi_i = i_idx + n
                        lo_j = j_idx
                        hi_j = j_idx + n
                        lo_union = min(lo_i, lo_j)
                        hi_union = max(hi_i, hi_j)
                        lo_union = max(0, lo_union)
                        hi_union = min(2 * n - 1, hi_union)
                        region = conv[lo_union:hi_union].copy()
                        overlap_lo_i = max(lo_i, lo_union) - lo_union
                        overlap_hi_i = min(hi_i, hi_union) - lo_union
                        if overlap_lo_i < overlap_hi_i:
                            region[overlap_lo_i:overlap_hi_i] += -2.0 * delta * a[overlap_lo_i - lo_i:overlap_hi_i - lo_i]
                        overlap_lo_j = max(lo_j, lo_union) - lo_union
                        overlap_hi_j = min(hi_j, hi_union) - lo_union
                        if overlap_lo_j < overlap_hi_j:
                            region[overlap_lo_j:overlap_hi_j] += 2.0 * delta * a[overlap_lo_j - lo_j:overlap_hi_j - lo_j]
                        if 2 * i_idx >= lo_union and 2 * i_idx < hi_union:
                            region[2 * i_idx - lo_union] += delta * delta
                        if 2 * j_idx >= lo_union and 2 * j_idx < hi_union:
                            region[2 * j_idx - lo_union] -= delta * delta
                        region_max = float(region.max())
                        if lo_union > 0:
                            region_max = max(region_max, float(conv[:lo_union].max()))
                        if hi_union < 2 * n - 1:
                            region_max = max(region_max, float(conv[hi_union:].max()))
                        if region_max < current_max:
                            a[i_idx] -= delta
                            a[j_idx] += delta
                            conv[lo_i:hi_i] += -2.0 * delta * a
                            conv[lo_j:hi_j] += 2.0 * delta * a
                            conv[2 * i_idx] += delta * delta
                            conv[2 * j_idx] -= delta * delta
                            current_max = region_max
                            accepted += 1
                            stall_count = 0
                            if current_max < best_max:
                                best_max = current_max
                                best_a = a.copy()

        # Periodic recompute
        if accepted - last_recompute > 300:
            conv = _full_convolve(a, a)
            current_max = float(conv.max())
            k_star_cache = int(np.argmax(conv))
            last_recompute = accepted
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            conv = _full_convolve(a, a)
            current_max = float(conv.max())
            k_star_cache = int(np.argmax(conv))

    final_score = _score_norm_fn(best_a)
    return best_a, final_score


def _local_search_random(a, deadline, rng):
    """Simple random pair-move local search."""
    n = len(a)
    conv = _full_convolve(a, a)
    current_max = float(conv.max())
    best_a = a.copy()
    best_max = current_max
    delta_base = 1.0 / n
    accepted = 0
    last_recompute = 0

    while time.monotonic() < deadline:
        i_idx = int(rng.integers(0, n))
        j_idx = int(rng.integers(0, n))
        if i_idx == j_idx:
            continue
        delta = delta_base * rng.uniform(0.3, 3.0)
        if a[i_idx] < delta:
            continue

        lo_i = i_idx
        hi_i = i_idx + n
        lo_j = j_idx
        hi_j = j_idx + n
        lo_union = min(lo_i, lo_j)
        hi_union = max(hi_i, hi_j)
        lo_union = max(0, lo_union)
        hi_union = min(2 * n - 1, hi_union)

        region = conv[lo_union:hi_union].copy()
        oi_lo = max(lo_i, lo_union) - lo_union
        oi_hi = min(hi_i, hi_union) - lo_union
        if oi_lo < oi_hi:
            region[oi_lo:oi_hi] += -2.0 * delta * a[oi_lo - lo_i:oi_hi - lo_i]
        oj_lo = max(lo_j, lo_union) - lo_union
        oj_hi = min(hi_j, hi_union) - lo_union
        if oj_lo < oj_hi:
            region[oj_lo:oj_hi] += 2.0 * delta * a[oj_lo - lo_j:oj_hi - lo_j]
        if 2 * i_idx >= lo_union and 2 * i_idx < hi_union:
            region[2 * i_idx - lo_union] += delta * delta
        if 2 * j_idx >= lo_union and 2 * j_idx < hi_union:
            region[2 * j_idx - lo_union] -= delta * delta

        region_max = float(region.max())
        if lo_union > 0:
            region_max = max(region_max, float(conv[:lo_union].max()))
        if hi_union < 2 * n - 1:
            region_max = max(region_max, float(conv[hi_union:].max()))

        if region_max < current_max:
            a[i_idx] -= delta
            a[j_idx] += delta
            conv[lo_i:hi_i] += -2.0 * delta * a
            conv[lo_j:hi_j] += 2.0 * delta * a
            conv[2 * i_idx] += delta * delta
            conv[2 * j_idx] -= delta * delta
            current_max = region_max
            accepted += 1
            if current_max < best_max:
                best_max = current_max
                best_a = a.copy()

        if accepted - last_recompute > 500:
            conv = _full_convolve(a, a)
            current_max = float(conv.max())
            last_recompute = accepted
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            conv = _full_convolve(a, a)
            current_max = float(conv.max())

    final_score = _score_norm_fn(best_a)
    return best_a, final_score


def _adam_polish(a_init, deadline, rng, lr=0.02, beta_frac=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    c = _full_convolve(a_init, a_init)
    beta = max(1e-10, beta_frac * float(c.max()))

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 200 == 0:
            beta = max(1e-12, beta * 0.85)
        if t % 2000 == 0 and best_score < 1.51:
            lr = max(0.003, lr * 0.95)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS + directed local search + random local search + Adam.
    Targets score below 1.5027 by combining coarse optimization (L-BFGS) with
    fine-grained pair-move local search and multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _try_update(p, sc, label):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            p = np.asarray(p, dtype=np.float64)
            s = p.sum()
            if s < 1e-15:
                return
            p = p / s
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={_score_norm_fn(parent):.10f}")
        _try_update(parent, sc_parent, "parent")
    else:
        n_parent = 2048
        parent = _make_ramp(n_parent, 1.0)
        print(f"No parent. Using ramp n={n_parent}")
        _try_update(parent, _score_norm_fn(parent), "ramp_init")

    # --- Phase 1: L-BFGS (180s) ---
    print(f"\n=== Phase 1: L-BFGS (target 180s) ===")
    phase1_dl = t_start + 180.0
    a1, sc1 = _run_lbfgs(parent, phase1_dl)
    _try_update(a1, sc1, "lbfgs_p1")

    # --- Phase 1b: L-BFGS from ramp initialization (60s) ---
    if time.monotonic() < t_start + 240.0:
        ramp = _make_ramp(len(parent), 1.0)
        phase1b_dl = min(t_start + 240.0, deadline)
        a1b, sc1b = _run_lbfgs(ramp, phase1b_dl, beta_starts=[0.1, 0.05, 0.025, 0.01])
        _try_update(a1b, sc1b, "lbfgs_ramp")

    # --- Phase 2: Directed local search (500s) ---
    print(f"\n=== Phase 2: Directed local search (target 500s) ===")
    phase2_dl = t_start + 740.0
    a2 = best_seq.copy()
    a2, sc2 = _local_search_directed(a2, phase2_dl, rng, delta_scale=1.0)
    _try_update(a2, sc2, "directed_ls")

    # --- Phase 2b: Second directed search with different scale (200s) ---
    if time.monotonic() < t_start + 940.0:
        phase2b_dl = min(t_start + 940.0, deadline)
        a2b = best_seq.copy()
        a2b, sc2b = _local_search_directed(a2b, phase2b_dl, rng, delta_scale=0.3)
        _try_update(a2b, sc2b, "directed_ls_fine")

    # --- Phase 3: Random local search (200s) ---
    if time.monotonic() < t_start + 1140.0:
        print(f"\n=== Phase 3: Random local search (target 200s) ===")
        phase3_dl = min(t_start + 1140.0, deadline)
        a3 = best_seq.copy()
        a3, sc3 = _local_search_random(a3, phase3_dl, rng)
        _try_update(a3, sc3, "random_ls")

    # --- Phase 4: Try n=4096 (300s) ---
    print(f"\n=== Phase 4: Upscale to n=4096 (target 300s) ===")
    phase4_dl = min(t_start + 1440.0, deadline)
    if best_seq is not None and len(best_seq) < 4096:
        target_n = 4096
        p_interp = _interp_profile(best_seq, target_n)
        sc_interp = _score_norm_fn(p_interp)
        _try_update(p_interp, sc_interp, f"interp_n{target_n}")

        if time.monotonic() < phase4_dl - 120:
            a4, sc4 = _run_lbfgs(p_interp, phase4_dl - 150,
                                  beta_starts=[0.05, 0.025, 0.01, 0.005, 0.002, 0.001])
            _try_update(a4, sc4, "lbfgs_n4096")

        if time.monotonic() < phase4_dl - 20:
            a4b = best_seq.copy()
            a4b, sc4b = _local_search_directed(a4b, phase4_dl, rng, delta_scale=0.5)
            _try_update(a4b, sc4b, "directed_n4096")

    # --- Phase 5: Adam polish on best (150s) ---
    remaining = deadline - time.monotonic()
    if remaining > 50 and best_seq is not None:
        print(f"\n=== Phase 5: Adam polish (target {remaining:.0f}s) ===")
        adam_dl = min(deadline - 30, time.monotonic() + remaining * 0.5)
        a5, sc5 = _adam_polish(best_seq, adam_dl, rng, lr=0.015, beta_frac=0.003)
        _try_update(a5, sc5, "adam_polish")

    # --- Phase 6: Final directed search (remaining) ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 6: Final directed search ({remaining:.0f}s) ===")
        a6 = best_seq.copy()
        a6, sc6 = _local_search_directed(a6, deadline - 5, rng, delta_scale=0.1)
        _try_update(a6, sc6, "final_directed")

    # --- Final validation ---
    print(f"\n{'='*50}")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.0)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_ramp(256, 1.0)
    print(f"FALLBACK: score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
```
