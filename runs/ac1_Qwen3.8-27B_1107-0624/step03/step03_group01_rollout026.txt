```python
"""
Annealed log-sum-exp Adam optimization with FFT-based gradient computation
for minimizing 2n*max(conv(a,a))/sum(a)^2. Uses multi-phase temperature
continuation, multi-restart strategy, and L-BFGS final polish.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_grad_z(a, c, T, n):
    cm = float(c.max())
    c_shifted = (c - cm) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad_z = (4.0 * n / T) * a * (h - W)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _adam_annealed(z_init, n, T0, T_min, t_per_level, deadline, rng, lr_base=0.05):
    z = z_init.copy()
    best_a = _softmax_fn(z)
    best_score = _score_norm(best_a)
    a = best_a.copy()

    num_levels = int(np.log(T0 / T_min) / np.log(2.0)) + 1
    num_levels = max(num_levels, 1)

    for level in range(num_levels):
        T = T0 * (0.5 ** level)
        if T < T_min:
            T = T_min

        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        b1, b2, eps = 0.9, 0.999, 1e-8
        level_start = time.monotonic()
        level_deadline = min(deadline, level_start + t_per_level)
        no_improve_count = 0
        local_best = best_score

        while time.monotonic() < level_deadline:
            t += 1
            a = _softmax_fn(z)
            c = _full_convolve(a, a)
            grad = _compute_grad_z(a, c, T, n)

            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            mh = m / (1 - b1 ** t)
            vh = v / (1 - b2 ** t)
            z = z - lr_base * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -80, 80)
            z = z - z.mean()

            if t % 20 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < local_best:
                    local_best = sc
                    no_improve_count = 0
                else:
                    no_improve_count += 1

                if no_improve_count > 50:
                    break

                if local_best < best_score:
                    best_score = local_best
                    best_a = a.copy()

        if time.monotonic() >= deadline:
            break

    return best_a, best_score


def _adam_annealed_advanced(z_init, n, T0, T_min, deadline, rng, lr_base=0.03):
    z = z_init.copy()
    a = _softmax_fn(z)
    best_a = a.copy()
    best_score = _score_norm(a)

    c = _full_convolve(a, a)
    max_c = float(c.max())
    T = max(T0, 0.05 * max_c)
    T_floor = max(T_min, 0.001 * max_c)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    patience = 200
    no_improve = 0
    restart_count = 0
    max_restarts = 10

    lr = lr_base

    while time.monotonic() < deadline and restart_count < max_restarts:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        grad = _compute_grad_z(a, c, T, n)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 10 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= patience:
                if restart_count < max_restarts:
                    restart_count += 1
                    noise_level = 0.05 + 0.03 * restart_count
                    z = np.log(np.maximum(best_a, 1e-15))
                    z = z - z.mean()
                    z = z + noise_level * rng.normal(size=n)
                    z = np.clip(z, -60, 60)
                    z = z - z.mean()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    no_improve = 0
                    T = max(T_floor, T * 0.7)
                    lr = max(0.005, lr * 0.8)
                    continue
                else:
                    break

        if t % 100 == 0:
            T = max(T_floor, T * 0.92)
            if best_score < 1.51:
                lr = max(0.003, lr * 0.97)

    return best_a, best_score


def _lbfgs_polish(a_init, deadline, n):
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    def objective(z):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        T = 0.005 * max_c0
        c_shifted = (c - cm) / T
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + T * log_sum)

    def gradient(z):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        T = 0.005 * max_c0
        c_shifted = (c - cm) / T
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = (4.0 * n / T) * a * (h - W)
        if not np.all(np.isfinite(grad)):
            return np.zeros_like(z)
        return grad

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    result = minimize(
        objective, z, jac=gradient,
        method='L-BFGS-B',
        options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10,
                 'maxls': 30, 'maxfun': 1000}
    )

    if result.x is not None and np.all(np.isfinite(result.x)):
        a_new = _softmax_fn(result.x)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new

    return best_a, best_score


def _make_initial_profile(n, type_idx=0):
    if type_idx == 0:
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * t
    elif type_idx == 1:
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + 1.0 * u + 0.3 * u ** 2
    elif type_idx == 2:
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + 0.8 * u + 0.5 * u ** 3
    elif type_idx == 3:
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * t
    else:
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.2 * t + 0.3 * np.sin(2 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
```
