"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous: adaptive beta, coordinate descent polish,
symmetric sequence exploration, multi-length search, aggressive LP.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _smooth_max_weights(c, beta):
    """Compute weights for smooth max approximation."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 0))  # Numerical stability
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _conv_cross_corr(w, p):
    """Compute the gradient: cross-correlation of w with p, giving d(conv)/dp."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _lp_solve(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _lp_refine(a, deadline, t=0.3, max_solves=20):
    """LP-based refinement."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, t * 0.85)
    return current_best, current_score

def _adam_logit(p0, deadline, rng, lr=0.06, beta_frac=0.1, beta_decay=0.92,

    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _conv_cross_corr(w, p)
        
        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        # Adaptive beta
        if t_adam % 30 == 0:
            beta = max(1e-4, beta * beta_decay)
            # Check gap between top lags
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-4, beta * 0.5)
                elif gap > beta * 10:
                    beta = min(maxc * 0.5, beta * 1.5)
        
        # Restart

        if no_imp > limit:
            if noise_scale is None:
                if best_score < 1.51:

                elif best_score < 1.53:
                    ns = 0.15
                else:
                    ns = 0.3
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score

def _coord_descent(a, deadline, max_sweeps=50):
    """Coordinate descent polish."""
    n = len(a)
    current = a.copy()
    current_score = _score_norm(current)
    c = _full_convolve(current)
    M = float(c.max())
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Randomize coordinate order
        order = np.arange(n)
        np.random.shuffle(order)  # Actually use a fixed rng
        for j in order:
            if time.monotonic() > deadline:
                break
            # Try small perturbations
            old_val = current[j]
            for frac in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                trial = current.copy()
                trial[j] = old_val * frac
                s = trial.sum()
                if s <= 0:
                    continue
                trial = trial / s
                sc = _score_norm(trial)
                if sc < current_score - 1e-12:
                    current = trial
                    current_score = sc
                    c = _full_convolve(current)
                    M = float(c.max())
                    improved = True
                    break
        if not improved:
            break
    
    return current, current_score
