def _interpolate(a, n_target):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == n_target:

        return a / s if s > 1e-30 else np.ones(n_target) / n_target
    x0 = np.linspace(0.0, 1.0, n)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):

    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_spread_active(a, M, active_indices):
    """LP with only active (binding) constraints for speed."""
    n = len(a)
    c_obj = -np.ones(n)
    A_full = _make_conv_matrix(a)
    A_sub = A_full[active_indices, :]
    b_sub = np.full(len(active_indices), M)
    res = linprog(c_obj, A_ub=A_sub, b_ub=b_sub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine_v3(a, deadline, max_iter=30000, inflate_start=1.0008, blend_start=0.35):
    """
    Improved LP refinement with:
    - Active set tracking (only constrain binding lags)
    - Line search on blend parameter
    - Adaptive inflation
    - Periodic full-constraint checks
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend = blend_start
    inflate = inflate_start
    use_active = False
    active_idx = np.arange(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        # Determine active set: lags within 5% of max
        threshold = M * 1.05
        active_mask = c >= min(threshold, M + 1e-10)
        active_idx = np.where(active_mask)[0]

        # Use active set LP if we have enough constraints
        if len(active_idx) >= 3 and len(active_idx) < (2 * n - 1) * 0.5:
            g = _lp_spread_active(a, M_lp, active_idx)
            if g is not None:
                # Verify full constraints are satisfied
                A_check = _make_conv_matrix(a)
                conv_check = A_check @ g
                if conv_check.max() <= M * 1.01 + 1e-10:
                    pass
                else:
                    g = _lp_spread(a, M_lp)
        else:
            g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 15 == 0:
                inflate = min(1.03, inflate * 1.04)
            if no_imp > 800:
                break
            continue

        g = g / sum_g

        # Line search on blend parameter
        bestBlend = blend
        bestScoreLS = np.inf
        candidates = [blend, blend * 0.5, blend * 1.5, 0.1, 0.05, 0.5, 0.7]
        for b_try in candidates:
            b_try = min(0.9, max(0.01, b_try))
            a_new = (1.0 - b_try) * a + b_try * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = _fft_convolve(a_new)
            score_new = float(2.0 * n * c_new.max())
            if score_new < bestScoreLS:
                bestScoreLS = score_new
                bestBlend = b_try

        a_new = (1.0 - bestBlend) * a + bestBlend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            inflate = max(1.0003, inflate * 0.97)
            blend = bestBlend
        else:
            no_imp += 1
            if no_imp % 25 == 0:
                blend = max(0.03, blend * 0.88)
                inflate = min(1.05, inflate * 1.03)
            if no_imp > max(400, 4 * n):
                a = best_a.copy()
                blend = blend_start
                inflate = inflate_start
                no_imp = 0

        a = a_new

    return best_a


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c


def _momentum_polish(a, deadline, max_iter=25000, beta_scale=0.04):
    """Momentum-based gradient polish on simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, beta_scale * float(c.max()))
    lr = 0.025
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        a_nest = _project_simplex(a_nest)
        grad, maxc, c_arr = _gradient_smooth_max(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(1500, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, beta_scale * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 70 == 0:
            beta = max(1e-10, beta * 0.7)

    return best_a


def _adam_polish(a, deadline, max_iter=25000, lr=0.04, beta_scale=0.04):
    """Adam-based polish in log-space."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, beta_scale * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(2000, 3 * n):
            rng_r = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, beta_scale * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _interpolate(a, n_target):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == n_target:
        s = a.sum()
        return a / s if s > 1e-30 else np.ones(n_target) / n_target
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, n_target)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return np.ones(n_target) / n_target
    return a_new / s


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_bell(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, order=1):
    t = np.linspace(0.0, np.pi, n)
    a = np.cos(order * t) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_mixture(n, centers, sigmas, weights):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c, s, w in zip(centers, sigmas, weights):
        a += w * np.exp(-0.5 * ((t - c) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _smoothed_perturb(a, rng, strength=0.1):
    """Add smoothed noise to escape local minima."""
    n = len(a)
    noise = rng.normal(size=n)
    kernel_size = max(3, n // 40)
    kernel = np.ones(kernel_size) / kernel_size
    noise = np.convolve(noise, kernel, mode='same')
    a_pert = a * (1.0 + strength * noise)
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def _sinusoidal_perturb(a, rng, num_freqs=3):
    """Add sinusoidal perturbations at random frequencies."""
    n = len(a)
    a_pert = a.copy()
    for _ in range(num_freqs):
        freq = rng.integers(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.04)
        t = np.arange(n)
        a_pert += amp * np.sin(2.0 * np.pi * freq * t / n + phase)
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def _peak_targeted_lp(a, deadline, max_iter=8000, top_k=5):
    """LP that specifically targets the top-k peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    inflate = 1.001
    blend = 0.3

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        sorted_idx = np.argsort(c)[::-1]
        top_indices = sorted_idx[:top_k]

        A_full = _make_conv_matrix(a)
        A_sub = A_full[top_indices, :]
        b_sub = np.full(top_k, M * inflate)

        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A_sub, b_ub=b_sub,
                      bounds=[(0, None)] * n, method='highs')

        if res.success:
            g = res.x
            sum_g = float(g.sum())
            if sum_g > 1.0 + 1e-8:
                g = g / sum_g
                a_new = (1.0 - blend) * a + blend * g
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    c_new = _fft_convolve(a_new)
                    score_new = float(2.0 * n * c_new.max())
                    if score_new < best_score - 1e-12:
                        best_score = score_new
                        best_a = a_new.copy()
                        no_imp = 0
                        inflate = max(1.0003, inflate * 0.98)
                    else:
                        no_imp += 1
                    a = a_new
                else:
                    no_imp += 1
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp % 30 == 0:
            inflate = min(1.05, inflate * 1.04)
            blend = max(0.05, blend * 0.9)
        if no_imp > 500:
            a = best_a.copy()
            inflate = 1.001
            blend = 0.3
            no_imp = 0

    return best_a


def _full_pipeline_for_length(a_init, n, deadline):
    """Run the full optimization pipeline for a given initial sequence."""
    a = _interpolate(a_init, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_u_profile(n, 2.0)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    # Phase A: LP
    rem = deadline - time.monotonic() - 120.0
    if rem > 10:
        lp_budget = min(rem * 0.4, 400.0)
        if lp_budget > 10:
            lp_end = time.monotonic() + lp_budget
            a_lp = _lp_refine_v3(a, lp_end, max_iter=20000, inflate_start=1.001, blend_start=0.35)
            sc_lp = _score(a_lp)
            if sc_lp < best_score:
                best_score = sc_lp
                best_a = a_lp.copy()
            print(f"    LP: {sc_lp:.10f}")

    # Phase B: Momentum gradient
    rem = deadline - time.monotonic() - 60.0
    if rem > 10:
        gp_budget = min(rem * 0.25, 150.0)
        if gp_budget > 8:
            gp_end = time.monotonic() + gp_budget
            a_gp = _momentum_polish(best_a, gp_end, max_iter=15000)
            sc_gp = _score(a_gp)
            if sc_gp < best_score:
                best_score = sc_gp
                best_a = a_gp.copy()
            print(f"    Momentum: {sc_gp:.10f}")

    # Phase C: Peak-targeted LP
    rem = deadline - time.monotonic() - 30.0
    if rem > 10:
        pt_budget = min(rem * 0.2, 100.0)
        if pt_budget > 8:
            pt_end = time.monotonic() + pt_budget
            a_pt = _peak_targeted_lp(best_a, pt_end, max_iter=5000, top_k=5)
            sc_pt = _score(a_pt)
            if sc_pt < best_score:
                best_score = sc_pt
                best_a = a_pt.copy()
            print(f"    PeakTargeted: {sc_pt:.10f}")

    # Phase D: Adam
    rem = deadline - time.monotonic() - 15.0
    if rem > 10:
        ad_budget = min(rem * 0.2, 80.0)
        if ad_budget > 8:
            ad_end = time.monotonic() + ad_budget
            a_ad = _adam_polish(best_a, ad_end, max_iter=10000, lr=0.04, beta_scale=0.04)
            sc_ad = _score(a_ad)
            if sc_ad < best_score:
                best_score = sc_ad
                best_a = a_ad.copy()
            print(f"    Adam: {sc_ad:.10f}")

    # Phase E: Final LP
    rem = deadline - time.monotonic() - 5.0
    if rem > 8:
        lp2_budget = min(rem * 0.3, 60.0)
        if lp2_budget > 5:
            lp2_end = time.monotonic() + lp2_budget
            a_lp2 = _lp_refine_v3(best_a, lp2_end, max_iter=8000, inflate_start=1.0006, blend_start=0.2)
            sc_lp2 = _score(a_lp2)
            if sc_lp2 < best_score:
                best_score = sc_lp2
                best_a = a_lp2.copy()
            print(f"    LP2: {sc_lp2:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization with multi-length search, active-set LP,
    adaptive inflation with line search, diverse perturbations, and
    extended final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine target lengths to try
    if parent is not None:
        n_parent = parent.size
        lengths = [n_parent]
    else:
        lengths = [768]

    # Add other lengths to explore
    for ln in [512, 640, 896, 1024]:
        if ln not in lengths and ln <= 2048:
            lengths.append(ln)

    # Sort: parent length first, then others
    lengths = list(dict.fromkeys(lengths))
    print(f"Trying lengths: {lengths}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: Optimize at each length
    num_lengths = len(lengths)
    per_length_budget = (deadline - time.monotonic() - 200.0) / num_lengths

    for n_t in lengths:
        if time.monotonic() > deadline - 160:
            break

        remaining = deadline - time.monotonic() - 140.0
        if remaining < 30:
            break

        phase_budget = min(per_length_budget, remaining)
        phase_budget = max(30.0, phase_budget)

        print(f"\nLength {n_t}: budget={phase_budget:.0f}s")

        # Determine initialization
        if parent is not None:
            a_init = _interpolate(parent, n_t)
        else:
            a_init = _make_u_profile(n_t, 2.0)

        sc_init = _score(a_init)
        print(f"  Init score: {sc_init:.10f}")
        if np.isfinite(sc_init) and sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()

        a_best_n, sc_best_n = _full_pipeline_for_length(a_init, n_t, deadline)
        if sc_best_n < best_score:
            best_score = sc_best_n
            best_seq = a_best_n.copy()
            print(f"  ** New best at n={n_t}: {sc_best_n:.10f}")

    # Phase 2: Diverse initializations at the best length
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_best = best_seq.size
        print(f"\nPhase 2: Diverse inits at n={n_best}")

        inits = []
        # Perturbations of best
        inits.append(_smoothed_perturb(best_seq, rng, 0.08))
        inits.append(_smoothed_perturb(best_seq, rng, 0.15))
        inits.append(_smoothed_perturb(best_seq, rng, 0.25))
        inits.append(_sinusoidal_perturb(best_seq, rng, 3))
        inits.append(_sinusoidal_perturb(best_seq, rng, 5))
        # Profile-based
        inits.append(_make_u_profile(n_best, 3.0))
        inits.append(_make_u_profile(n_best, 5.0))
        inits.append(_make_gaussian_bell(n_best, 0.25))
        inits.append(_make_gaussian_bell(n_best, 0.15))
        inits.append(_make_cosine_profile(n_best, 1))
        inits.append(_make_gaussian_mixture(n_best, [0.15, 0.85], [0.12, 0.12], [1.0, 1.0]))
        inits.append(_make_gaussian_mixture(n_best, [0.1, 0.9], [0.1, 0.1], [1.0, 1.0]))
        # Blends
        if parent is not None:
            inits.append(0.7 * best_seq + 0.3 * parent)
            inits.append(0.5 * best_sequence_blend if False else 0.5 * best_seq + 0.5 * parent)

        # Ensure all are valid
        valid_inits = []
        for ini in inits:
            ini = np.asarray(ini, dtype=np.float64)
            ini = np.maximum(ini, 0.0)
            s = ini.sum()
            if s > 1e-10:
                valid_inits.append(ini / s)

        n_perts = len(valid_inits)
        rem = deadline - time.monotonic() - 50.0
        per_pert = max(10.0, (rem * 0.6) / max(1, n_perts))

        for i, a_pert in enumerate(valid_inits):
            if time.monotonic() > deadline - 40:
                break

            sc_pert = _score(a_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_seq = a_pert.copy()

            # Quick LP + gradient on this perturbation
            if time.monotonic() < deadline - 30:
                t1 = min(20.0, per_pert * 0.5)
                if t1 > 5:
                    a_res = _lp_refine_v3(a_pert, time.monotonic() + t1,
                                         max_iter=2000, inflate_start=1.002, blend_start=0.3)
                    sc_res = _score(a_res)
                    if sc_res < best_score:
                        best_score = sc_res
                        best_seq = a_res.copy()
                        print(f"    Pert {i}: {sc_res:.10f} (new best)")

                t2 = min(12.0, per_pert * 0.3)
                if t2 > 4 and time.monotonic() < deadline - 25:
                    a_res2 = _momentum_polish(best_seq, time.monotonic() + t2, max_iter=3000)
                    sc_res2 = _score(a_res2)
                    if sc_res2 < best_score:
                        best_score = sc_res2
                        best_seq = a_res2.copy()
                        print(f"    Pert {i} (grad): {sc_res2:.10f} (new best)")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 25:
        rem = deadline - time.monotonic() - 15.0
        print(f"\nPhase 3: Final polish (budget={rem:.0f}s)")

        # 3a: LP with tight inflation
        t1 = min(rem * 0.35, 120.0)
        if t1 > 8:
            a_f = _lp_refine_v3(best_seq, time.monotonic() + t1,
                               max_iter=15000, inflate_start=1.0006, blend_start=0.2)
            sc_f = _score(a_f)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  Final LP: {sc_f:.10f}")

        # 3b: Adam
        rem = deadline - time.monotonic() - 15.0
        t2 = min(rem * 0.25, 80.0)
        if t2 > 8:
            a_f2 = _adam_polish(best_seq, time.monotonic() + t2,
                               max_iter=12000, lr=0.03, beta_scale=0.03)
            sc_f2 = _score(a_f2)
            if sc_f2 < best_score:
                best_score = sc_f2
                best_seq = a_f2.copy()
            print(f"  Final Adam: {sc_f2:.10f}")

        # 3c: Peak-targeted LP
        rem = deadline - time.monotonic() - 15.0
        t3 = min(rem * 0.2, 60.0)
        if t3 > 5:
            a_f3 = _peak_targeted_lp(best_seq, time.monotonic() + t3,
                                    max_iter=4000, top_k=3)
            sc_f3 = _score(a_f3)
            if sc_f3 < best_score:
                best_score = sc_f3
                best_seq = a_f3.copy()
            print(f"  Final PeakTargeted: {sc_f3:.10f}")

        # 3d: Very brief LP for last-mile improvement
        rem = deadline - time.monotonic() - 5.0
        t4 = min(rem * 0.2, 40.0)
        if t4 > 4:
            a_f4 = _lp_refine_v3(best_seq, time.monotonic() + t4,
                                max_iter=5000, inflate_start=1.0003, blend_start=0.1)
            sc_f4 = _score(a_f4)
            if sc_f4 < best_score:
                best_score = sc_f4
                best_seq = a_f4.copy()
            print(f"  Final LP2: {sc_f4:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_u_profile(768, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_u_profile(768, 2.0)

    sc_final = _score(final_seq)
    print(f"\nFINAL: internal score={sc_final:.10f}, n={final_seq.size}")

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")

    return [float(x) for x in final_seq]
