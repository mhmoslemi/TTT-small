"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smooth max with L-BFGS continuation, Adam-based peak shaving
with top-P focus, and multi-scale exploration. Targets improvement below 1.5027.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _full_convolve_self(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve_self(a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    """Project v onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s


def _smooth_objective(z, beta, n, top_p=0):
    a = _softmax_fn(z)
    c = _full_convolve_self(a)
    if top_p > 0 and top_p < len(c):
        idx = np.argpartition(c, -top_p)[-top_p:]
        c_sub = c[idx]
        cm = float(c_sub.max())
        c_shifted = (c_sub - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _smooth_gradient(z, beta, n, top_p=0):
    a = _softmax_fn(z)
    c = _full_convolve_self(a)
    if top_p > 0 and top_p < len(c):
        idx = np.argpartition(c, -top_p)[-top_p:]
        c_sub = c[idx]
        cm = float(c_sub.max())
        c_shifted = (c_sub - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w_sub = np.zeros(top_p)
            w_sub[int(np.argmax(c_sub))] = 1.0
        else:
            w_sub = e / s
        w = np.zeros(len(c))
        w[idx] = w_sub
    else:
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

    W = float(np.dot(w, c))
    # Gradient: 2 * (w_rev * a)[n-1:2n-1][::-1] - W * ones
    w_rev = w[::-1]
    conv_wn = _full_convolve(w_rev, a)
    h = conv_wn[n - 1:2 * n - 1][::-1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_run(a_init, deadline, max_wall_time, top_p=0):
    """Run L-BFGS with smooth max objective. Returns (best_a, best_score)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve_self(a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        level_deadline = time.monotonic() + per_level

        try:
            def obj_wrapper(zw):
                return _smooth_objective(zw, beta, n, top_p)

            def grad_wrapper(zw):
                return _smooth_gradient(zw, beta, n, top_p)

            result = minimize(
                obj_wrapper, z, jac=grad_wrapper,
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 30, 'maxfun': 1500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

        if time.monotonic() > deadline:
            break

    return best_a, best_score


def _adam_peak_shave(a_init, deadline, top_p=5, initial_lr=0.1, beta_frac=0.005):
    """Adam optimizer targeting top-P peaks of convolution."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_v = np.zeros(n, dtype=np.float64)
    v_v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = initial_lr

    c = _full_convolve_self(a_init)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve_self(a)

        if top_p > 0 and top_p < len(c):
            idx = np.argpartition(c, -top_p)[-top_p:]
            c_sub = c[idx]
            cm = float(c_sub.max())
            c_shifted = (c_sub - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w_sub = np.zeros(top_p)
                w_sub[int(np.argmax(c_sub))] = 1.0
            else:
                w_sub = e / s
            w = np.zeros(len(c))
            w[idx] = w_sub
        else:
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

        W = float(np.dot(w, c))
        w_rev = w[::-1]
        conv_wn = _full_convolve(w_rev, a)
        h = conv_wn[n - 1:2 * n - 1][::-1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            continue

        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * (grad * grad)
        mh = m_v / (1 - b1 ** t)
        vh = v_v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        sc = _score_norm(_softmax_fn(z))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = _softmax_fn(z).copy()

        # Adaptive scheduling
        if t % 100 == 0:
            beta = max(1e-12, beta * 0.9)
        if t % 500 == 0:
            lr = max(0.005, lr * 0.97)

    return best_a, best_score


def _projected_gradient_peak_shave(a_init, deadline, n_iter_max=500000):
    """Direct projected gradient on the max of convolution (non-smooth)."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    alpha = 0.01
    momentum = np.zeros(n)
    mu = 0.9
    count = 0

    while time.monotonic() < deadline and count < n_iter_max:
        c = _full_convolve_self(a)
        cm = float(c.max())
        score = 2.0 * n * cm
        if score < best_score:
            best_score = score
            best_a = a.copy()
            alpha = min(alpha * 1.05, 0.1)
        else:
            alpha = max(alpha * 0.9, 0.001)

        k_star = int(c.argmax())
        # Gradient of c_{k_star}
        g = np.zeros(n)
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g[valid] = 2.0 * a[idx[valid]]
        if k_star % 2 == 0:
            mid = k_star // 2
            if 0 <= mid < n:
                g[mid] = a[mid]

        # Project gradient onto simplex tangent
        g = g - g.mean()
        momentum = mu * momentum + (1 - mu) * g
        direction = -momentum

        # Line search
        a_new = a
        for trial_alpha in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.125]:
            a_try = a + trial_alpha * direction
            a_try = _project_simplex(a_try)
            c_try = _full_convolve_self(a_try)
            if c_try.max() < cm * (1 + 1e-8):
                a_new = a_try
                break
        a = a_new
        count += 1

        if count % 10000 == 0:
            mu = min(0.95, mu + 0.01)

    return best_a, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine_profile(n, asym):
    """Raised cosine with asymmetry parameter."""
    t = np.linspace(0.0, 1.0, n)
    base = 0.5 * (1 + np.cos(2 * np.pi * t))
    asym_part = asym * (t - 0.5)
    a = base + asym_part
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_tapered_profile(n, taper_pow, asym):
    """Tapered profile with power-law taper and asymmetry."""
    t = np.linspace(0.0, 1.0, n)
    # Symmetric tapered window
    w = 1.0 - |2 * t - 1.0| ** taper_pow
    # Add asymmetry
    w = w + asym * (t - 0.5)
    w = np.maximum(w, 1e-12)
    return w / w.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: L-BFGS smooth max -> Adam peak shave -> 
    projected gradient -> fine polish -> multi-n.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 2048
        parent = _make_linear_profile(n, 1.5)
        print(f"No parent, using linear profile n={n}")

    # --- Phase 0: Evaluate diverse initial candidates (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    sc_parent = _score_norm(parent)
    candidates.append((sc_parent, parent.copy(), len(parent), "parent"))
    _update(parent, sc_parent, "parent")

    for sn in [512, 1024, 2048, 4096]:
        if time.monotonic() > phase0_end:
            break
        profiles = [
            _make_linear_profile(sn, 0.5),
            _make_linear_profile(sn, 1.0),
            _make_linear_profile(sn, 1.5),
            _make_linear_profile(sn, 2.0),
            _make_linear_profile(sn, 3.0),
            _make_quadratic_profile(sn, 1.0, 0.3),
            _make_quadratic_profile(sn, 1.5, -0.2),
            _make_quadratic_profile(sn, 2.0, 0.1),
            _make_cubic_profile(sn, 1.0, -0.5),
            _make_cubic_profile(sn, 1.5, -1.0),
            _make_raised_cosine_profile(sn, 0.3),
            _make_raised_cosine_profile(sn, 0.5),
            _make_tapered_profile(sn, 2.0, 0.3),
            _make_tapered_profile(sn, 3.0, 0.5),
        ]
        for idx, p in enumerate(profiles):
            if time.monotonic() > phase0_end:
                break
            sc = _score_norm(p)
            candidates.append((sc, p, sn, f"prof{idx}_n{sn}"))
            _update(p, sc, f"prof{idx}_n{sn}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: L-BFGS with log-sum-exp continuation (400s) ---
    phase1_end = min(deadline, t_start + 430.0)
    n_phase1 = 6
    per_cand = (phase1_end - time.monotonic()) / max(1, n_phase1)

    seen_keys = set()
    count = 0
    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end or count >= n_phase1:
            break
        if sc > best_score * 1.02:
            break
        key = (pn, label)
        if key in seen_keys:
            continue
        seen_keys.add(key)

        remaining = min(per_cand * (n_phase1 - count), phase1_end - time.monotonic() - 5)
        if remaining < 10:
            break
        dl = time.monotonic() + remaining
        p_best, sc_best = _lbfgs_run(p, dl, remaining, top_p=0)
        _update(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"Phase 1 done: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # --- Phase 2: Adam peak shaving (700s) ---
    phase2_end = min(deadline, t_start + 1130.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        n_best = len(best_seq)
        remaining = phase2_end - time.monotonic() - 5
        if remaining > 30:
            # Run Adam with top_p=5
            dl = time.monotonic() + remaining * 0.6
            p_best, sc_best = _adam_peak_shave(best_seq, dl, top_p=5, initial_lr=0.05, beta_frac=0.003)
            _update(p_best, sc_best, f"adam_top5_n{n_best}")

            if time.monotonic() < phase2_end - 10:
                # Run Adam with top_p=3 for more focus
                remaining2 = phase2_end - time.monotonic() - 5
                dl2 = time.monotonic() + remaining2
                p_best2, sc_best2 = _adam_peak_shave(best_seq, dl2, top_p=3, initial_lr=0.03, beta_frac=0.002)
                _update(p_best2, sc_best2, f"adam_top3_n{n_best}")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Projected gradient peak shaving (300s) ---
    phase3_end = min(deadline, t_start + 1430.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic() - 5
        if remaining > 10:
            dl = time.monotonic() + remaining
            p_best, sc_best = _projected_gradient_peak_shave(best_seq, dl)
            _update(p_best, sc_best, "proj_grad")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Multi-restart L-BFGS from best (200s) ---
    phase4_end = min(deadline, t_start + 1630.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = (phase4_end - time.monotonic()) / max(1, n_restarts) - 5

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise_scale = 0.05 * (i + 1)
            z = z + noise_scale * rng.standard_normal(n_best)
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            remaining = min(per_restart, phase4_end - time.monotonic() - 5)
            if remaining < 5:
                break
            dl = time.monotonic() + remaining
            p_best, sc_best = _lbfgs_run(p_start, dl, remaining, top_p=3)
            _update(p_best, sc_best, f"restart{i}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Try different n values (150s) ---
    phase5_end = min(deadline, t_start + 1780.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = [s for s in [1024, 4096] if s != n_curr]

        for tn in target_ns:
            if time.monotonic() > phase5_end - 10:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm(p_interp)
            _update(p_interp, sc_interp, f"interp_n{tn}")

            if sc_interp < best_score * 1.01:
                remaining = min(60.0, phase5_end - time.monotonic() - 5)
                if remaining > 10:
                    dl = time.monotonic() + remaining
                    p_best, sc_best = _adam_peak_shave(p_interp, dl, top_p=5, initial_lr=0.03, beta_frac=0.003)
                    _update(p_best, sc_best, f"adam_n{tn}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final fine polish (remaining time) ---
    if time.monotonic() < deadline - 5 and best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            # L-BFGS with very small beta
            dl = time.monotonic() + remaining * 0.5
            p_best, sc_best = _lbfgs_run(best_seq, dl, remaining * 0.5, top_p=3)
            _update(p_best, sc_best, "final_lbfgs")

            if time.monotonic() < deadline - 10:
                remaining2 = deadline - time.monotonic() - 5
                if remaining2 > 10:
                    dl2 = time.monotonic() + remaining2
                    p_best2, sc_best2 = _adam_peak_shave(best_seq, dl2, top_p=2, initial_lr=0.02, beta_frac=0.001)
                    _update(p_best2, sc_best2, "final_adam")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(2048, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Cross-check with evaluator if feasible
        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence check failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(2048, 1.5)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
