"""
Optimized autocorrelation peak minimization using exact-max gradient descent
with Adam optimizer in log-softmax space, top-k equalization, and multi-length search.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _grad_c_at_k(a, k):
    """Compute gradient of c[k] = (conv(a,a))[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    j_low = max(0, k - n + 1)
    j_high = min(n - 1, k)
    if j_low > j_high:
        return grad
    js = np.arange(j_low, j_high + 1)
    kj = k - js
    grad[js] = 2.0 * a[kj]
    # Fix diagonal: where 2*j == k
    diag_mask = (2.0 * js == k)
    if np.any(diag_mask):
        grad[js[diag_mask]] = a[js[diag_mask]]
    return grad


def _grad_z_from_grad_a(a, grad_a):
    """Convert gradient w.r.t. a to gradient w.r.t. z (log-softmax params)."""
    d = np.dot(grad_a, a)
    return a * (grad_a - d)


def _adam_update(z, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * (grad * grad)
    mh = m / (1 - b1 ** t)
    vh = v / (1 - b2 ** t)
    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    return z_new, m, v


def _run_adam_exact_max(a_init, deadline, lr_init=0.1, top_k=1, lr_decay=0.9999):
    """Adam optimization using exact max gradient in log-softmax space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    
    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        sc = 2.0 * n * cm
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Find top-k indices
        if top_k == 1:
            k_max = int(np.argmax(c))
            grad_c = _grad_c_at_k(a, k_max)
        else:
            # Use top-k weighted gradient
            thresh = cm - 1e-10 * cm  # within 0.1% of max
            active = np.where(c >= thresh)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])
            # Weight by how close to max
            weights = c[active] / c[active].sum()
            grad_c = np.zeros(n)
            for w, k in zip(weights, active):
                grad_c += w * _grad_c_at_k(a, int(k))
        
        grad_a = 2.0 * n * grad_c  # gradient of 2*n*c[k_max] w.r.t. a
        grad_z = _grad_z_from_grad_a(a, grad_a)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        t += 1
        z, m, v = _adam_update(z, grad_z, m, v, t, lr)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if t % 1000 == 0:
            lr = lr_init * (lr_decay ** 1000)
            # Adaptive lr based on top_k
            if cm > 0:
                cv = float(c.std() / cm)
                if cv < 0.01:
                    top_k = min(5, top_k + 1)
    
    return best_a, best_score

    z = z - z.mean()

    best_score = _score_norm_fn(a_init)

    v = np.zeros(n)
    t = 0

    lr = lr_init

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    x0 = np.linspace(0.0, 1.0, n0)

    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
