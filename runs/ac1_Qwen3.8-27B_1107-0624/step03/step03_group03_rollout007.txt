def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:

    return c

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _coordinate_descent(a, deadline, max_sweeps=200):

    s = a.sum()
    if s < 1e-10:

    a = a / s

    best_score = _score_normalized(a)

        if time.monotonic() > deadline:
            break

        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break

            old_val = a[i]
            a[i] = 0.0
            rest_sum = a.sum()
            if rest_sum < 1e-10:
                a[i] = old_val
                continue

            # conv with a[i]=0
            c_no = _fft_convolve(a)
            max_no = float(c_no.max())

            # Try a few values of a[i] in [0, 1] (rest renormalized)
            # The effect: conv values increase by a[i]*a[k-i] for each k
            # We want to find a[i] that minimizes max(conv)
            # conv with a[i]=v (and rest scaled to sum=1):
            # Actually let's just try: set a[i] = v, normalize, compute score
            # But that's expensive. Instead, use a simple heuristic:
            # If a[i] is at the "peak" of a, try reducing it.
            # If a[i] is at a "valley", try increasing it.

            # Simple approach: try old_val, 0, and a few perturbations
            candidates = [0.0, old_val * 0.5, old_val, old_val * 1.5, 1.0 / n]
            best_cand_score = _score_normalized(a / rest_sum) if rest_sum > 1e-10 else np.inf
            best_cand_val = 0.0

            for cv in candidates:
                if cv < -1e-10 or cv > 1.0:

                a[i] = cv

                if s_tmp < 1e-10:
                    continue
                a_norm = a / s_tmp
                sc = _score_normalized(a_norm)
                if sc < best_cand_score:
                    best_cand_score = sc
                    best_cand_val = cv

            a[i] = best_cand_val
            s_a = a.sum()
            if s_a < 1e-10:
                a = best_a.copy()
                break

            if best_cand_val != old_val:
                improved = True

        sc = _score_normalized(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

        print(f"  LP result n={n_t}: score={sc_lp:.10f}")

    # Phase 4: Subgradient polish on overall best
    print("\n=== Phase 4: Final subgradient polish ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        sp_budget = min(80.0, deadline - time.monotonic() - 60.0)
        if sp_budget > 15:
            sp_end = time.monotonic() + sp_budget
            a_sp = _subgradient_polish(best_seq, sp_end, max_iter=100000, lr=0.01, momentum=0.9)
            sc_sp = _score(a_sp)
            if np.isfinite(sc_sp) and sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            print(f"  Subgradient: score={sc_sp:.10f}")

    # Phase 5: Coordinate descent on best
    print("\n=== Phase 5: Coordinate descent ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        cd_budget = min(50.0, deadline - time.monotonic() - 30.0)
        if cd_budget > 10:
            cd_end = time.monotonic() + cd_budget
            a_cd = _coordinate_descent(best_seq, cd_end, max_sweeps=100)
            sc_cd = _score(a_cd)
            if np.isfinite(sc_cd) and sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            print(f"  CoordDescent: score={sc_cd:.10f}")

    # Phase 6: Final LSE micro-polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        lse_budget = min(30.0, deadline - time.monotonic() - 15.0)
        if lse_budget > 5:
            lse_end = time.monotonic() + lse_budget
            a_last = _lse_optimize(best_seq, lse_end, max_iter=20000,
                                   T_start=0.01, T_end=1e-7, lr=0.05)
            sc_last = _score(a_last)
            if np.isfinite(sc_last) and sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LSE: score={sc_last:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_raised_cosine(1024)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0

    # Validate
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
