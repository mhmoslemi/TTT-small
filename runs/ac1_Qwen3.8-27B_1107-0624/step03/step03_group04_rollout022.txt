def _subgrad_topk(a, c, top_k=5):
    """Compute subgradient using top-k peaks of the convolution."""
    n = len(a)
    k_idx = np.argsort(c)[::-1][:top_k]
    k_vals = c[k_idx]
    k_max = c.max()
    if k_max < 1e-15:
        return np.zeros(n)
    # Weights: exponential decay from max
    w = np.exp(-(k_max - k_vals) / (k_max * 0.1 + 1e-15))
    w = w / w.sum()

    g = np.zeros(n)
    for ki, wi in zip(k_idx, w):
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        for i in range(lo, hi + 1):
            j = ki - i
            if 0 <= j < n:
                g[i] += wi * a[j]
                if i == j:
                    g[i] += wi * a[i]
    return g


def _subgrad_single(a, k, n):
    """Compute exact subgradient for a single peak at position k."""
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        j = k - i
        if 0 <= j < n:
            g[i] += a[j]
            if i == j:
                g[i] += a[i]
    return g


def _make_conv_matrix(a):
    n = len(a)
    # For LP: we need conv(a, b) for variable b
    # conv(a,b)[k] = sum_i a[i]*b[k-i]
    # As a matrix: A[k,i] = a[k-i] if valid, else 0
    # But it's more efficient to build it as: for each i, a[i] contributes to A[i:i+n, i]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M, timeout=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = _make_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': timeout} if timeout else None)
    if res.success:
        return res.x
    return None


def _make_flat_spectrum(n, rng):
    """Generate sequence with flat magnitude spectrum."""
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_uniform(n):
    return np.ones(n) / n


def _interpolate_to_length(a, n_new):
    n = len(a)
    if n == n_new:
        return a.copy() / a.sum()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _proximal_subgradient(a, deadline, top_k=5, max_iter=100000):
    """
    Proximal subgradient method with Polyak step size.
    Minimizes max_k conv(a,a)[k] subject to a in simplex.
    """
    a = a.copy() / a.sum()
    n = len(a)
    f_lower = 1.0 / (2 * n - 1)  # Lower bound on max(c) when sum(a)=1

    c = _fft_convolve(a)
    f_curr = float(c.max())
    best_a = a.copy()
    best_f = f_curr

    # Polyak step: alpha = (f - f*) / ||g||^2
    alpha = 1.0 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        f_curr = float(c.max())

        # Compute subgradient
        if it < 100 or it % 50 == 0:
            # Use top-k for smoother direction initially
            g = _subgrad_topk(a, c, top_k=min(top_k, 10))
        else:
            # Use single peak for sharper direction later
            k_star = int(np.argmax(c))
            g = _subgrad_single(a, k_star, n)

        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-30:
            break

        # Polyak step size
        alpha = (f_curr - f_lower) / g_norm_sq
        alpha = min(alpha, 2.0 / np.sqrt(it + 1))  # Cap for safety
        alpha = max(alpha, 1e-10)

        # Projected step
        a_new = _project_simplex(a - alpha * g)

        c_new = _fft_convolve(a_new)
        f_new = float(c_new.max())

        if f_new < best_f - 1e-14:
            best_f = f_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        if no_imp > 5000:
            # Restart with perturbation
            a = _project_simplex(best_a + 0.1 * np.random.default_rng(it).standard_normal(n))
            no_imp = 0

    return best_a


def _lp_iterative(a, deadline, max_iter=5000, initial_inflate=1.002):
    """
    Iterative LP-based refinement.
    Repeatedly solve LP with tightening M to flatten the convolution.
    """
    a = a.copy() / a.sum()
    n = len(a)
    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    inflate = initial_inflate
    blend = 0.3
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Solve LP with current M inflated slightly
        M_lp = M * inflate
        timeout = max(1.0, (deadline - time.monotonic()) * 0.1)
        g = _lp_spread(a, M_lp, timeout=timeout)

        if g is None:
            # Try with larger inflation
            inflate = min(1.01, inflate * 1.5)
            if inflate > 1.01:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = max(1.0001, inflate * 0.9)
                blend = max(0.05, blend * 0.95)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            # Tighten
            inflate = max(1.0001, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend = max(0.05, blend * 0.9)
            if no_imp > 300:
                a = best_a.copy()
                blend = 0.3
                no_imp = 0

        a = a_new

    return best_a


def _smooth_max_optimize(a, deadline, beta_init=None, max_iter=50000):
    """Smooth max gradient descent in log-space for fine-tuning."""
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    n = len(a)

    z = np.log(a.copy())
    z -= np.mean(z)

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = 0.01 * c_max

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.03
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        mval = float(c.max())
        x = np.clip((c - mval) / beta, -50, 50)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex <= 0:
            break
        w = ex / s_ex

        # Gradient via FFT
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= np.mean(gz)

        if not np.all(np.isfinite(gz)):
            z = np.log(best_a) - np.mean(np.log(best_a))
            m, v, t = np.zeros(n), np.zeros(n), 0
            no_imp = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.001, lr * 0.9)

        # Adaptive beta
        if it % 200 == 0 and it > 100:
            top3 = np.sort(c)[::-1][:3]
            if len(top3) > 1 and top3[0] - top3[-1] < beta:
                beta = max(1e-10, beta * 0.6)

        if no_imp > max(5000, 5 * n):
            break

    return best_a


def _coordinate_descent(a, deadline, max_sweeps=50):
    """Coordinate descent: optimize one coordinate at a time."""
    a = a.copy() / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try a range of values for a[i]
            old_val = a[i]
            # The optimal a[i] for fixed others: we want to minimize max of
            # c[k] for k where i appears
            # c[k] = c_old[k] + 2*a[i]*a[k-i] + a[i]^2 (if k=2i)
            # Simple approach: try scaling a[i] by factors
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                a_test = a.copy()
                a_test[i] = old_val * factor
                a_test = np.maximum(a_test, 0)
                s = a_test.sum()
                if s < 1e-10:
                    continue
                a_test = a_test / s
                sc = _score(a_test)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_test.copy()
                    a = a_test.copy()
                    old_val = a[i]
                    improved = True
        if not improved:
            break

    return best_a


def _hybrid_optimize(a0, deadline, n):
    """
    Main optimization loop combining subgradient, LP, and smooth max.
    """
    a = a0.copy() / a0.sum()
    best_a = a.copy()
    best_score = _score(a)
    print(f"  Start: score={best_score:.10f}", flush=True)

    # Phase A: Proximal subgradient (coarse)
    sub_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
    a_sub = _proximal_subgradient(a, sub_deadline, top_k=5)
    sc_sub = _score(a_sub)
    print(f"  Subgradient: {sc_sub:.10f}", flush=True)
    if sc_sub < best_score:
        best_score = sc_sub
        best_a = a_sub.copy()
        a = a_sub.copy()

    # Phase B: LP iterative refinement
    lp_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
    a_lp = _lp_iterative(a, lp_deadline, max_iter=3000, initial_inflate=1.001)
    sc_lp = _score(a_lp)
    print(f"  LP refine: {sc_lp:.10f}", flush=True)
    if sc_lp < best_score:
        best_score = sc_lp
        best_a = a_lp.copy()
        a = a_lp.copy()

    # Phase C: Smooth max fine-tuning
    sm_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
    a_sm = _smooth_max_optimize(a, sm_deadline)
    sc_sm = _score(a_sm)
    print(f"  Smooth max: {sc_sm:.10f}", flush=True)
    if sc_sm < best_score:
        best_score = sc_sm
        best_a = a_sm.copy()
        a = a_sm.copy()

    # Phase D: Another LP pass on the best
    if time.monotonic() < deadline - 10:
        lp2_deadline = deadline - 5
        a_lp2 = _lp_iterative(a, lp2_deadline, max_iter=2000, initial_inflate=1.0005)
        sc_lp2 = _score(a_lp2)
        print(f"  LP round 2: {sc_lp2:.10f}", flush=True)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_a = a_lp2.copy()

    # Phase E: Smooth max final polish
    if time.monotonic() < deadline - 5:
        a_final = _smooth_max_optimize(best_a, deadline - 2, beta_init=1e-5)
        sc_final = _score(a_final)
        print(f"  Final polish: {sc_final:.10f}", flush=True)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()

    return best_a


def _try_length(n, deadline_remaining, seed, parent=None):
    """Try to optimize a sequence of length n."""
    if deadline_remaining < 60:
        return None, np.inf

    rng = np.random.default_rng(seed)
    best_a = None
    best_score = np.inf

    # Generate several initializations
    inits = []

    # 1. Flat spectrum
    inits.append(_make_flat_spectrum(n, rng))
    inits.append(_make_flat_spectrum(n, rng))

    # 2. Uniform
    inits.append(_make_uniform(n))

    # 3. Parent interpolation
    if parent is not None:
        inits.append(_interpolate_to_length(parent, n))

    # 4. Random
    r = rng.random(n)
    inits.append(r / r.sum())

    # Score all inits
    init_scores = []
    for i, a0 in enumerate(inits):
        sc = _score(a0)
        init_scores.append((sc, i, a0))
    init_scores.sort(key=lambda x: x[0])

    print(f"  n={n}: best init score={init_scores[0][0]:.8f}", flush=True)

    # Try top 2-3 initializations
    n_to_try = min(3, len(init_scores))
    per_budget = deadline_remaining / n_to_try

    for idx in range(n_to_try):
        if time.monotonic() >= deadline_remaining + time.monotonic() - 0:
            pass
        sc0, _, a0 = init_scores[idx]
        sub_deadline = time.monotonic() + per_budget - 5
        if sub_deadline < time.monotonic() + 30:
            sub_deadline = time.monotonic() + 30

        a_opt = _hybrid_optimize(a0, sub_deadline, n)
        sc = _score(a_opt)
        print(f"  n={n}, init {idx}: {sc:.10f}", flush=True)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

    Strategy:
    1. Collect parent sequence and generate diverse initializations
    2. For each candidate length, run hybrid optimization (subgradient + LP + smooth max)
    3. Final polish on the best solution found
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc_parent = _score(parent)
                best_score = sc_parent
                best_seq = parent.copy()
                print(f"Parent: score={sc_parent:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine which lengths to try and time allocation
    if parent is not None:
        lengths = [parent.size]
        for ln in [768, 512, 1024]:
            if ln not in lengths:
                lengths.append(ln)
    else:
        lengths = [768, 512]

    # Allocate time: parent length gets 50%, others get rest
    n_lengths = len(lengths)
    if n_lengths > 1:
        time_alloc = [0.45, 0.25, 0.15, 0.10][:n_lengths]
    else:
        time_alloc = [0.9]

    for i, n in enumerate(lengths[:4]):
        if time.monotonic() > deadline - 60:
            break
        alloc = time_alloc[i] * (deadline - time.monotonic())
        if alloc < 60:
            break
        print(f"\n=== Trying n={n}, budget={alloc:.0f}s ===", flush=True)
        a_try, sc_try = _try_length(n, deadline - 60 - (1 - i / n_lengths) * 30,
                                     seed + i * 1000, parent)
        if a_try is not None and sc_try < best_score:
            best_score = sc_try
            best_seq = a_try.copy()
            print(f"  New best: {best_score:.10f} (n={n})", flush=True)

    # Final: additional LP + smooth max polish on the global best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final polish, remaining={deadline - time.monotonic():.0f}s ===", flush=True)

        # LP polish
        if time.monotonic() < deadline - 15:
            a_lp = _lp_iterative(best_seq, deadline - 10, max_iter=2000, initial_inflate=1.0003)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
                print(f"  Final LP: {sc:.10f}", flush=True)

        # Smooth max polish
        if time.monotonic() < deadline - 5:
            a_sm = _smooth_max_optimize(best_seq, deadline - 2, beta_init=1e-6)
            sc = _score(a_sm)
            if sc < best_score:
                best_score = sc
                best_seq = a_sm.copy()
                print(f"  Final smooth: {sc:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = _make_uniform(512)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_uniform(512)

    # Verify with evaluate_sequence if possible
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval check skipped: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
