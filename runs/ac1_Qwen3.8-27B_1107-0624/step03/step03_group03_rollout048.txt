def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()

"""

Uses improved log-sum-exp gradient with adaptive temperature,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix_reduced(a, M, top_k=80):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    return A, b


def _lp_spread(a, M, use_reduced=True, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    if use_reduced and n > 128:
        A, b = _make_conv_matrix_reduced(a, M, top_k)
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search_blend(a, g, max_iters=15):
    n = len(a)
    best_score = _score_norm(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c * (g - a), 0))
    fd = _score_norm(np.maximum(a + d * (g - a), 0))
    for _ in range(max_iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c * (g - a), 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d * (g - a), 0))
    best_t = (lo + hi) / 2.0
    return best_t

def _lse_adam_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -50.0, 50.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()
    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n, order=1):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) ** order
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_with_edges(n, edge_weight=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bump(n, center=0.5, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 2.0 * np.exp(-((t - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_two_peaks(n, pos1=0.3, pos2=0.7, width=0.1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 3.0 * np.exp(-((t - pos1) / width) ** 2) + 3.0 * np.exp(-((t - pos2) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_smooth(n, seed=0):
    rng = np.random.default_rng(seed)
    freqs = np.arange(1, 8)
    coeffs = rng.normal(size=len(freqs)) * np.exp(-freqs / 5.0)
    t = np.linspace(0.0, 1.0, n)
    a = 1.0
    for i, f in enumerate(freqs):
        a += coeffs[i] * np.cos(2.0 * np.pi * f * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

                                  restart_every=5000)
            sc_lse = _score(a_lse)
            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse
            print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> LSE={sc_lse:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best for n={n_t}: {best_here_score:.10f}")

    # Phase 2: Deeper optimization on top lengths
    print("\n=== Phase 2: Deep LSE + mixed optimization ===")
    phase2_deadline = t_start + 1000.0
    sorted_lengths = sorted(best_per_length.keys(),
                            key=lambda ln: _score(best_per_length[ln]))

    for n_t in sorted_lengths[:3]:
        if time.monotonic() > phase2_deadline - 100:
            break
        a_in = best_per_length[n_t]
        sc_in = _score(a_in)
        print(f"  Deep opt n={n_t}: start score={sc_in:.10f}")

        remaining = phase2_deadline - 100 - time.monotonic()
        if remaining < 30:
            break
        end_t = time.monotonic() + remaining * 0.5
        a_deep = _lse_optimize(a_in, end_t, max_iter=150000,
                               T_start=0.02, T_end=1e-6, lr=0.08,
                               restart_every=10000)
        sc_deep = _score(a_deep)
        if np.isfinite(sc_deep) and sc_deep < best_score:
            best_score = sc_deep
            best_seq = a_deep.copy()

        end_t2 = time.monotonic() + remaining * 0.3
        a_mixed = _mixed_method_optimize(a_deep, end_t2, max_iter=80000)
        sc_mixed = _score(a_mixed)
        if np.isfinite(sc_mixed) and sc_mixed < best_score:
            best_score = sc_mixed
            best_seq = a_mixed.copy()

        end_t3 = time.monotonic() + remaining * 0.2
        a_mirror = _mirror_descent(a_mixed, end_t3, max_iter=60000,
                                   lr_start=0.03, lr_end=0.001)
        sc_mirror = _score(a_mirror)
        if np.isfinite(sc_mirror) and sc_mirror < best_score:
            best_score = sc_mirror
            best_seq = a_mirror.copy()

        best_per_length[n_t] = best_seq.copy() if best_seq.size == n_t else a_mirror
        print(f"  n={n_t}: LSE={sc_deep:.10f}, mixed={sc_mixed:.10f}, mirror={sc_mirror:.10f}")

    # Phase 3: LP refinement
    print("\n=== Phase 3: LP refinement ===")
    phase3_deadline = t_start + 1400.0
    for n_t in sorted_lengths[:2]:
        if time.monotonic() > phase3_deadline - 60:
            break
        a_in = best_per_length.get(n_t, best_seq[:n_t] if best_seq.size >= n_t else _interpolate_profile(best_seq, n_t))
        a_in = np.asarray(a_in, dtype=np.float64).copy()
        if a_in.size != n_t:
            a_in = _interpolate_profile(a_in, n_t)
        sc_in = _score(a_in)
        print(f"  LP n={n_t}: start={sc_in:.10f}")

        end_t = time.monotonic() + min(150.0, phase3_deadline - 60 - time.monotonic())
        a_lp = _lp_refine(a_in, end_t, max_iter=3000,
                          inflate=1.0005, blend=0.15, use_line_search=True)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP result n={n_t}: {sc_lp:.10f}")

    # Phase 4: Subgradient polish
    print("\n=== Phase 4: Subgradient polish ===")
    phase4_deadline = t_start + 1600.0
    if best_seq is not None and time.monotonic() < phase4_deadline - 30:
        end_t = time.monotonic() + min(150.0, phase4_deadline - 30 - time.monotonic())
        a_sp = _subgradient_polish(best_seq, end_t, max_iter=100000, lr=0.012)
        sc_sp = _score(a_sp)
        if np.isfinite(sc_sp) and sc_sp < best_score:
            best_score = sc_sp
            best_seq = a_sp.copy()
        print(f"  Subgradient: {sc_sp:.10f}")

    # Phase 5: Final LP micro-polish
    print("\n=== Phase 5: Final LP micro-polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        end_t = time.monotonic() + min(80.0, deadline - 30 - time.monotonic())
        a_final = _lp_refine(best_seq, end_t, max_iter=2000,
                             inflate=1.0002, blend=0.08, use_line_search=True)
        sc_final = _score(a_final)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final LP: {sc_final:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
