        gz -= np.mean(gz)
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -20.0, 20.0)
        z_new -= np.mean(z_new)
        
        p_new = np.exp(z_new)
        s = p_new.sum()
        if not np.isfinite(s) or s <= 0:
            continue
        p_new = p_new / s
        
        c_new = _full_convolve(p_new)
        maxc_new = float(c_new.max())
        score = float(2.0 * n * maxc_new)
        
        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p_new.copy()
                best_z = z_new.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
            
            z = z_new
            p = p_new
            c = c_new
            maxc = maxc_new
        else:
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp += 10
            continue
        
        if it % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
        
        if no_imp > no_imp_limit:
            if noise_scale is None:
                ns = 0.1
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -20, 20)
            z -= np.mean(z)
            p = np.exp(z)
            s = p.sum()
            if s > 0:
                p = p / s
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score

def _coordinate_descent(p0, deadline, rng, group_size=20, max_rounds=10000):
    """Coordinate descent: optimize groups of elements while keeping others fixed."""
    p = np.array(p0, dtype=np.float64)
    n = p.size
    best_p = p.copy()
    best_score = _score(p)
    round_i = 0
    
    while time.monotonic() < deadline and round_i < max_rounds:
        round_i += 1
        
        # Select a random group of indices
        idx = rng.integers(0, n, size=min(group_size, n))
        idx = np.unique(idx)
        
        # Get current values
        p_fixed = p.copy()
        p_fixed[idx] = 1e-15  # temporarily set to minimum
        
        # The contribution of the fixed part to the convolution
        # We want to optimize p[idx] to minimize max(conv)
        # This is hard in general, so we use a simple approach:
        # For each element in the group, try to reduce its contribution to the peak
        
        # Find the peak of the current convolution
        c = _full_convolve(p)
        peak_idx = int(np.argmax(c))
        
        # For the peak at position k, conv[k] = sum_i a[i]*a[k-i]
        # We want to reduce this. The elements that contribute most are those
        # where both a[i] and a[k-i] are large.
        
        # Simple heuristic: reduce elements at positions i where i or k-i is in our group
        # and a[i]*a[k-i] is large.
        
        # Alternative: just do a few steps of projected gradient on the subset
        # Fix all elements not in idx, optimize those in idx.
        
        # The objective as a function of p[idx]:
        # conv[k] = p[idx] @ A_k @ p[idx] + p[idx] @ b_k + c_k
        # where A_k, b_k, c_k depend on the fixed elements and k.
        # For the smooth max, this becomes a quadratic to minimize.
        
        # For simplicity, let's just do: for each i in idx, try reducing p[i]
        # if it helps reduce the peak.
        
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            # Try a small decrease
            old = p[i]
            if old < 1e-10:
                continue
            for factor in [0.9, 0.95, 0.8, 1.1, 1.05]:
                p[i] = min(1000.0, max(0.0, old * factor))
                # Renormalize
                s = p.sum()
                if s <= 0:
                    p[i] = old
                    break
                p_norm = p / s
                sc = _score(p_norm)
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                    p = p_norm
                    improved = True
                    old = p[i]
                    break
            p[i] = old
        
        if not improved and round_i % 100 == 0:
            # Reset to best
            p = best_p.copy()
    
    p = best_p / best_p.sum()
    return p, best_score

def _make_ramp(n, order=1.0, direction=1):
    """Make a power-law ramp profile."""
    t = np.arange(n, dtype=np.float64)
    if direction > 0:
        a = (t + 1.0) ** order
    else:
        a = (n - t) ** order
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_sigmoid(n, center=0.3, width=0.1):
    """Sigmoid-like profile."""
    t = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-10*(t - center))) 
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_bimodal_asym(n, c1=0.2, c2=0.8, s1=0.1, s2=0.15):
    """Asymmetric two-bump profile."""
    t = np.linspace(0, 1, n)
    a = 0.7 * np.exp(-0.5*((t-c1)/s1)**2) + 0.3 * np.exp(-0.5*((t-c2)/s2)**2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_random_smooth(n, rng, n_freq=5):
    """Random smooth profile from low-frequency components."""
    coeffs = rng.normal(size=n_freq)
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    for k, c in enumerate(coeffs, 1):
        a += c * np.cos(2*np.pi*k*t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
