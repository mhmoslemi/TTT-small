Smooth-max gradient descent with FFT for minimizing 2n*max(conv(a,a))/sum(a)^2.
Starts from linear ramp initializations at various lengths, uses FFT-based
gradient computation with adaptive learning rate and beta scheduling.
"""

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b=None):
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([])
    if n <= 300 and m <= 300:
        return np.convolve(a, b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project v onto the probability simplex (sum=1, all >= 0)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    """Compute the gradient of the smooth-max score for normalized a (sum=1)."""
    n = len(a)
    c = _fft_convolve(a)  # length 2n-1
    # Softmax weights
    c_shifted = c - c.max()
    # Avoid overflow
    if beta * c_shifted.max() > 700:
        c_shifted = c_shifted - c_shifted.max() / beta * 0.5
    exp_c = np.exp(beta * c_shifted)
    w = exp_c / exp_c.sum()
    
    # Compute g[i] = sum_m a[m] * w[m+i] for i = 0, ..., n-1
    # g[i] = conv(a, w_rev)[2n-2-i]
    w_rev = w[::-1]

    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w_rev, N)
    conv_aw = np.fft.irfft(fa * fw, N)[:n + 2 * n - 1]
    
    # g[i] = conv_aw[2n-2-i]
    g = conv_aw[n-1:2*n-1][::-1]
    
    # Score (smooth)
    s_max = (1.0 / beta) * np.log(exp_c.sum()) + c.max()
    score = 2.0 * n * s_max  # since sum(a) = 1
    
    # Gradient = 4n * g (as derived)
    grad = 4.0 * n * g
    return grad, float(score), w


def _make_ramp(n, alpha=1.0):

    a = t ** alpha + 1e-10
    return a / a.sum()

    a = 1.0 / (1.0 + np.exp(-steepness * (t - center)))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)  # momentum buffer
    
    no_imp = 0
    lr_decay = 1.0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, sm_score, w = _smooth_max_grad(a, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-3, beta * 0.5)
            v = np.zeros(n)
            continue
        
        # Gradient step with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = _project_simplex(a_new)
        
        # Evaluate
        sc = _score(a_new)
        
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        a = a_new
        
        # Adaptive learning rate
        if no_imp > 50:
            lr = max(0.001, lr * 0.95)
            momentum = min(0.95, momentum * 1.002)
        elif no_imp == 0 and lr < base_lr:
            lr = min(base_lr, lr * 1.05)
        
        # Decrease beta over time (sharper max approximation)
        if it % 200 == 199:
            beta = max(1e-2, beta * 0.7)
        
        # Occasionally reset momentum if stuck
        if no_imp > 500:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr
            beta = max(0.05, 2.0 / float(_fft_convolve(a).max()))
            no_imp = 0
    
    return best_a


def _anneal_restart(a0, deadline, max_iter=500000):
    """Gradient descent with periodic random perturbations (simulated annealing)."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    beta = max(0.1, 2.0 / float(_fft_convolve(a).max()))
    lr = 0.05
    base_lr = lr
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    temp = 0.1  # annealing temperature
    
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, sm_score, w = _smooth_max_grad(a, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-3, beta * 0.5)
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad
        a_new = a + v + temp * rng.standard_normal(n)
        a_new = _project_simplex(a_new)
        
        sc = _score(a_new)
        
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        a = a_new
        
        # Annealing
        temp = max(0.001, temp * 0.999)
        
        if no_imp > 100:
            lr = max(0.001, lr * 0.97)
        
        if it % 100 == 99:
            beta = max(1e-2, beta * 0.8)
        
        if no_imp > 1000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = base_lr
            temp = 0.05
            no_imp = 0
            beta = max(0.05, 2.0 / float(_fft_convolve(a).max()))
    
    return best_a
