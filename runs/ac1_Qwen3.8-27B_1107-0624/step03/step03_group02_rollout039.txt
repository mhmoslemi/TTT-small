import time

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w

The max of these is a convex function of x. Divided by (S+x)^2 (which is convex and increasing), the result might not be convex, but in practice it should be unimodal or close to it.

I'll use a simple approach: evaluate at a grid of x values and pick the best, then refine with golden section search.

The grid: x = 0, x_current/4, x_current/2, x_current, x_current*2, x_current*4, and maybe a few more.

Actually, for efficiency, let me think about what the optimal x typically looks like. If the current a[i] is "too large" (causing peaks), the optimal x would be smaller. If it's "too small", it might be larger. But since all the g_k functions are non-decreasing in x (they're either convex increasing, linear increasing, or constant), the max is also non-decreasing. So increasing x always increases the numerator.

The denominator (S+x)^2 is also increasing. So the question is which grows faster.

At x = 0: score = 2*n*max(conv_base) / S^2
As x increases: the numerator grows (at least quadratically for the x^2 term, or linearly for others) and the denominator grows quadratically.

If the dominant term in the numerator is the x^2 term (i.e., conv_base[2i] + x^2 is the max), then for large x, score ~ 2*n*x^2/x^2 = 2*n, which is constant. So the score approaches 2n from below (or above).

If the dominant term is linear (2*x*a[j] for some j), then for large x, score ~ 2*n*2*x*a[j]/x^2 = 4*n*a[j]/x -> 0. So the score approaches 0!

Wait, that can't be right. If we make a[i] very large, the convolution values involving a[i] become very large, but so does the sum. Let me recheck.

If a[i] = x is very large, then:
- conv[2i] = base + x^2 (dominant term is x^2)
- conv[i+j] = base + 2*x*a[j] for j != i (dominant term is 2*x*a[j])

The max convolution is dominated by max(x^2, 2*x*max_j a[j]) = x^2 (for x > 2*max_j a[j]).

So for very large x: max(conv) ~ x^2, and (sum)^2 ~ x^2. Score ~ 2*n*x^2/x^2 = 2*n.

For small x: max(conv) ~ max(conv_base) (the base values), and (sum)^2 ~ S^2. Score ~ 2*n*max(conv_base)/S^2.

So the score goes from 2*n*max(conv_base)/S^2 at x=0 to 2*n as x -> infinity. The minimum is somewhere in between (or at x=0).

This suggests that x=0 is often optimal (removing the coordinate entirely), unless the base values are already causing a high score.

This makes sense: if a coordinate is causing a peak, setting it to 0 removes the peak. The cost is that the sum decreases, which increases the score. But if the peak was the dominant factor, removing it helps.

So coordinate descent with the option to set coordinates to 0 is powerful. It's like a feature selection / sparsification step.

However, we need to be careful: if we set too many coordinates to 0, the sum becomes too small and the score blows up (or the sequence becomes trivial).

            # Try some candidate x values

                    best_x = x
                if time.monotonic() > deadline:

                    # So M3 should exclude m=i, and M2 handles m=i.

"""
Step function optimization using coordinate descent with FFT-accelerated convolution.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses exact per-coordinate optimization with adaptive grid, combined with Adam
smooth-max for global exploration.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s

def _grad_smooth_max(a, w):

    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)

    fw = np.fft.rfft(w_pad, N)

    conv_result = np.fft.irfft(fw * fa, N)

def _coord_descent(a, deadline, n_passes=200, n_grid=30, recompute_every=64):
    """Coordinate descent minimizing 2n*max(conv)/(sum)^2."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-10:
        return a
    conv = _fft_convolve(a)
    rng = np.random.default_rng(789)
    updates_since_recompute = 0
    
    for pass_num in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        pass_improved = 0
        
        for idx, i in enumerate(order):
            if time.monotonic() > deadline:
                break
            
            x_old = a[i]
            if x_old < 1e-12:
                continue
            
            # Compute conv_base by subtracting contribution of a[i]
            conv_base = conv.copy()
            conv_base[2 * i] -= x_old * x_old
            m_idx = np.arange(n)
            k_idx = i + m_idx
            mask = (k_idx != 2 * i)
            conv_base[k_idx[mask]] -= 2.0 * x_old * a[m_idx[mask]]
            
            S_base = S - x_old
            if S_base < 1e-10:
                a[i] = x_old
                continue
            
            # M1: max of conv_base outside [i, i+n-1]
            far_mask = np.ones(2 * n - 1, dtype=bool)
            far_mask[i:i+n] = False
            M1 = float(conv_base[far_mask].max()) if far_mask.any() else 0.0
            
            # Near values for M3
            near_conv = conv_base[i:i+n]
            # slopes[m] = 2 * a[m], but a[i] is excluded (set to 0 effectively)
            slopes = 2.0 * a.copy()
            slopes[i] = 0.0
            
            # Build x grid
            x_max = max(x_old * 5.0, 1.0)
            x_grid = np.geomspace(1e-10, x_max, n_grid)
            x_grid[0] = 0.0  # include x=0
            
            # Add x_old to grid
            x_grid = np.append(x_grid, x_old)
            x_grid = np.sort(x_grid)
            
            best_score_local = np.inf
            best_x = x_old
            
            for x in x_grid:
                M2 = float(conv_base[2 * i]) + x * x
                M3_vals = near_conv + x * slopes
                M3_vals[i] = -np.inf
                M3 = float(M3_vals.max())
                max_conv = max(M1, M2, M3)
                S_test = S_base + x
                if S_test < 1e-10:
                    continue
                sc = 2.0 * n * max_conv / (S_test * S_test)
                if sc < best_score_local:
                    best_score_local = sc
                    best_x = x
            
            if abs(best_x - x_old) > 1e-12:
                # Update conv
                conv = conv_base.copy()
                conv[2 * i] += best_x * best_x
                conv[k_idx[mask]] += 2.0 * best_x * a[m_idx[mask]]
                a[i] = best_x
                S = S_base + best_x
                updates_since_recompute += 1
                pass_improved += 1
            
            if updates_since_recompute >= recompute_every:
                conv = _fft_convolve(a)
                S = float(a.sum())
                updates_since_recompute = 0
        
    return a

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
