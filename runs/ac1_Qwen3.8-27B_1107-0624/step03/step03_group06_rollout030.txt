def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / len(a)

def _interpolate(a, n_new):

    n0 = a.size

    x0 = np.linspace(0.0, 1.0, n0)

def _solve_lp(f, M, n):
    """LP: maximize sum(g) subject to conv(f, g) <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _lp_refine(a0, deadline, max_iter=20000, inflate_start=1.005, inflate_min=1.0005):
    """LP-based refinement with line search."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start
    line_grid = 9

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 15 == 0:
                inflate = min(0.02, inflate * 1.3)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        # Line search
        best_t = 0.0
        best_sc = float(2.0 * n * _fft_convolve(a).max())
        ts = np.linspace(0.05, 1.0, line_grid)
        for t in ts:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = float(2.0 * n * _fft_convolve(a_new).max())
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new)
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 100 == 99:
                inflate = max(inflate_min, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                inflate = min(0.02, inflate * 1.15)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                if inflate > 1.002:
                    break

    return best_a


def _adam_polish(a0, deadline, max_iter=50000):
    """Adam-based gradient descent with smooth max."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.1 * float(c.max()))
    lr = 0.05
    base_lr = lr
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        grad, maxc = _smooth_max_grad_direct(a, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            m = np.zeros(n)
            v = np.zeros(n)
            continue

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to simplex
        a_new = _project_simplex(a_new)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                beta = max(1e-6, 0.1 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        # Adapt beta
        if it % 50 == 49:
            beta = max(1e-8, beta * 0.9)

    return best_a


def _momentum_polish(a0, deadline, max_iter=50000):
    """Momentum-based gradient descent (original approach)."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.05 * float(c.max()))
    lr = 0.03
    base_lr = lr
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        grad, maxc = _smooth_max_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.92)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 100 == 99:
            beta = max(1e-8, beta * 0.8)

    return best_a


def _basin_hop(a0, deadline, rng, n_hops=5):
    """Perturb and re-optimize to escape local minima."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    for hop in range(n_hops):
        if time.monotonic() > deadline:
            break

        # Create perturbation
        noise_scale = 0.1 * (1.0 + 0.5 * hop / max(1, n_hops))
        noise = rng.standard_normal(n) * noise_scale
        a_pert = a + noise
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = _normalize(a_pert)

        # Short LP refinement
        hop_deadline = min(deadline, time.monotonic() + 30.0)
        a_hopped = _lp_refine(a_pert, hop_deadline, max_iter=2000,
                             inflate_start=1.01, inflate_min=1.001)
        sc = _score(a_hopped)
        if sc < best_score:
            best_score = sc
            best_a = a_hopped.copy()
            a = a_hopped.copy()
            print(f"  Basin hop {hop}: score={sc:.10f} (improved)")
        else:
            print(f"  Basin hop {hop}: score={sc:.10f} (no improvement)")

    return best_a, best_score
