"""
Improved optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements: Chebyshev initialization, Nesterov-accelerated smooth max gradient,
improved LP inflation schedule, golden section line search, and mirror descent.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _make_chebyshev(n, degree=8):
    """Chebyshev-based initialization with equal ripple property."""
    t = np.linspace(0, np.pi, n)
    # T_0(x) = 1, T_1(x) = x, T_2(x) = 2x^2-1, etc.
    # Use a mix of low-degree Chebyshev polynomials for smoothness
    x = np.cos(t)  # maps [0, pi] to [1, -1]
    a = np.ones(n)
    # Add Chebyshev components with decreasing amplitude
    amplitude = 0.3
    for d in range(2, degree + 1):
        # T_d(x) = cos(d * arccos(x))
        c = np.cos(d * t)
        a += amplitude * c
        amplitude *= 0.7
    a = np.maximum(a, 1e-8)
    return a / a.sum()

def _make_chebyshev2(n, order=4):
    """Alternative Chebyshev: |T_order(cos(pi*t))| based."""
    t = np.linspace(0, 1, n)
    x = 2 * t - 1  # in [-1, 1]
    # T_order(x) = cos(order * arccos(x))
    a = np.abs(np.cos(order * np.arccos(np.clip(x, -1, 1)))) + 1e-6
    return a / a.sum()

def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return a / a.sum()

def _make_zolotarev(n, m=4):
    """Zolotarev-like: minimizes max deviation from zero on [-1,1] with free zeros."""
    # Approximate with a polynomial that has rapid transition
    t = np.linspace(0, 1, n)
    x = 2 * t - 1
    # Use (1 + x^m) / 2 which has a flat top
    a = 0.5 * (1 + np.sign(x) * np.abs(x) ** (1.0/m)) + 0.5 + 1e-6
    return a / a.sum()

def _make_tapered(n, taper=0.3):
    """Tapered (Hanning-like) window."""
    t = np.linspace(0, 1, n)
    a = 1.0 - taper * (1 - np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-8)
    return a / a.sum()

def _make_gauss(n, sigma=0.3):
    """Gaussian profile."""
    t = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2) + 1e-8
    return a / a.sum()

def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. convolve(f, g)[k] <= M for all k, g >= 0."""
    # The constraint matrix A has A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _golden_section_search(a, g, n_grid=20):
    """Golden section line search for blending a and g."""
    sc_a = _score(a)
    
    # First find a bracket
    t_hi = 1.0
    sc_hi = _score(np.maximum((1-t_hi)*a + t_hi*g, 0) / np.maximum((1-t_hi)*a + t_hi*g, 0).sum())
    if not np.isfinite(sc_hi) or sc_hi >= sc_a:
        # Direction is bad, return 0
        ts = np.linspace(0.01, 0.2, n_grid)
        best_t = 0.0
        best_sc = sc_a
        for t in ts:
            blend = np.maximum((1-t)*a + t*g, 0)
            s = blend.sum()
            if s < 1e-10:
                continue
            blend = blend / s
            sc = _score(blend)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        return best_t, best_sc
    
    t_lo = 0.0
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    t1 = t_lo + gr * (t_hi - t_lo)
    t2 = t_lo + (1 - gr) * (t_hi - t_lo)
    
    sc1 = sc_a  # at t=0
    blend2 = np.maximum((1-t2)*a + t2*g, 0)
    s2 = blend2.sum()
    sc2 = _score(blend2 / s2) if s2 > 1e-10 else np.inf
    
    for _ in range(n_grid):
        if sc1 < sc2:
            t_hi = t2
            t2 = t1
            sc2 = sc1
            t1 = t_lo + gr * (t_hi - t_lo)
            blend1 = np.maximum((1-t1)*a + t1*g, 0)
            s1 = blend1.sum()
            sc1 = _score(blend1 / s1) if s1 > 1e-10 else np.inf
        else:
            t_lo = t1
            t1 = t2
            sc1 = sc2
            t2 = t_lo + (1 - gr) * (t_hi - t_lo)
            blend2 = np.maximum((1-t2)*a + t2*g, 0)
            s2 = blend2.sum()
            sc2 = _score(blend2 / s2) if s2 > 1e-10 else np.inf
    
    t_best = (t_lo + t_hi) / 2
    blend = np.maximum((1-t_best)*a + t_best*g, 0)
    s = blend.sum()
    if s < 1e-10:
        return 0.0, sc_a
    sc_best = _score(blend / s)
    return t_best, sc_best

def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.003, inflate_min=1.0005):
    """LP-based refinement with improved inflation schedule."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start
    patient = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            # LP infeasible, increase inflation
            inflate = min(0.05, inflate * 1.5)
            patient += 1
            if patient > 50:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.03, inflate * 1.2)
            if no_imp > 500:
                # Reset and try with fresh inflation
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                patient = 0
                if patient > 200:
                    break
            continue

        g = g / sum_g

        # Golden section line search
        best_t, best_sc = _golden_section_search(a, g, n_grid=15)

        if best_t > 1e-4 and best_sc < best_score - 1e-14:
            a_new = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            patient = 0
            # Decrease inflation slowly
            if it % 50 == 49:
                inflate = max(inflate_min, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.03, inflate * 1.1)

    return best_a

def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of convolution."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -100, 100)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # Gradient: sum_k w_k * (a reversed and shifted)
    # dc_k/da_j = a_{k-j} for 0<=k-j<n
    # So grad_j = sum_k w_k * a_{k-j}
    # This is the cross-correlation of w with a
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval

def _gradient_polish(a0, deadline, max_iter=50000):
    """Gradient descent with momentum and adaptive step size."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = 0.01 * float(c.max())
    lr = 0.02
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    best_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov step
        a_nest = a + momentum * v
        grad, maxc = _smooth_max_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            if score < best_score * 0.999:
                best_lr = lr
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.01 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0
                lr = best_lr * 0.8

        if it % 50 == 49:
            beta = max(1e-8, beta * 0.85)

    return best_a

def _mirror_descent(a0, deadline, max_iter=30000):
    """Mirror descent with entropy (multiplicative updates)."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = 0.02 * float(c.max())
    lr = 0.01
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, maxc = _smooth_max_grad(a, beta)

        if not np.all(np.isfinite(grad)):

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Chebyshev initialization, LP refinement with golden-section line search,
Nesterov-accelerated smooth-max gradient, and mirror descent.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_chebyshev(n, degree=8):
    t = np.linspace(0, np.pi, n)
    a = np.ones(n)
    amplitude = 0.25
    for d in range(2, degree + 1):
        a += amplitude * np.cos(d * t)
        amplitude *= 0.65
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_chebyshev_abs(n, order=3):
    t = np.linspace(0, 1, n)
    x = 2 * t - 1
    a = np.abs(np.cos(order * np.arccos(np.clip(x, -1, 1)))) + 1e-6
    return a / a.sum()


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return a / a.sum()


def _make_tapered(n, taper=0.4):
    t = np.linspace(0, 1, n)
    a = 1.0 - taper * (1 - np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_gauss(n, sigma=0.35):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 1e-8
    return a / a.sum()


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _solve_lp(f, M, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g):
    """Golden section line search for blending a and g on the probability simplex."""
    sc_a = _score(a)

    def blend_score(t):
        blend = np.maximum((1 - t) * a + t * g, 0.0)
        s = blend.sum()
        if s < 1e-10:
            return np.inf
        return _score(blend / s)

    t_hi = 1.0
    sc_hi = blend_score(t_hi)
    if not np.isfinite(sc_hi) or sc_hi >= sc_a:
        # Try small steps
        best_t = 0.0
        best_sc = sc_a
        for t in np.linspace(0.01, 0.3, 10):
            sc = blend_score(t)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        return best_t, best_sc

    t_lo = 0.0
    gr = (np.sqrt(5) - 1) / 2
    t1 = t_lo + gr * (t_hi - t_lo)
    t2 = t_lo + (1 - gr) * (t_hi - t_lo)
    sc1 = sc_a
    sc2 = blend_score(t2)

    for _ in range(18):
        if sc1 < sc2:
            t_hi = t2
            t2 = t1
            sc2 = sc1
            t1 = t_lo + gr * (t_hi - t_lo)

            t1 = t2
            sc1 = sc2
            t2 = t_lo + (1 - gr) * (t_hi - t_lo)

def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -80, 80)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = 0.005 * float(c.max())
    lr = 0.015
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    best_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        grad, maxc = _smooth_max_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 4.0)
            v = np.zeros(n)
            continue

        # Adaptive step: reduce lr if norm is too large
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e6:
            lr = max(1e-5, lr * 0.5)
            grad = grad / grad_norm

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            if score < best_score * 0.9995:
                best_lr = lr
                base_lr = lr
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 80 == 0:
                lr = max(0.002, lr * 0.92)
            if no_imp > max(1500, 2 * n):

                v = np.zeros(n)
                beta = max(1e-6, 0.005 * float(_fft_convolve(a).max()))
                lr = best_lr
                no_imp = 0

        if it % 60 == 59:

            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.001, lr * 0.92)
            if no_imp > 4000:
                a = best_a.copy()
                lr = 0.008
                beta = 0.01 * float(_fft_convolve(a).max())
                no_imp = 0

        if it % 80 == 79:
            beta = max(1e-8, beta * 0.88)

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP refinement, Nesterov gradient,
    and mirror descent for minimizing the autocorrelation peak ratio.
    """
    t_start = time.monototic if hasattr(time, 'monototic') else time.monotonic
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initializations
    configs = []
    n_parent = parent.size if parent is not None else 768

    if parent is not None:
        for n_t in [n_parent, max(256, n_parent // 2), min(2048, n_parent * 2)]:
            configs.append(('parent', n_t, _interpolate(parent, n_t)))

    for n_t in [512, 768, 1024]:
        configs.append(('cheb8', n_t, _make_chebyshev(n_t, 8)))
        configs.append(('cheb_abs3', n_t, _make_chebyshev_abs(n_t, 3)))
        configs.append(('cheb_abs5', n_t, _make_chebyshev_abs(n_t, 5)))
        configs.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp2', n_t, _make_ramp(n_t, 2.0)))
        configs.append(('taper', n_t, _make_tapered(n_t, 0.4)))
        configs.append(('gauss', n_t, _make_gauss(n_t, 0.35)))

    # Score and sort
    scored_configs = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored_configs.append((sc, name, n_t, a0))
    scored_configs.sort(key=lambda x: x[0])

    print(f"Top initializations:")
    for sc, name, n_t, _ in scored_configs[:6]:
        print(f"  {name} n={n_t}: {sc:.10f}")

    # Phase 1: LP on top configs
    max_lp = 5
    for i, (sc0, name, n_t, a0) in enumerate(scored_configs[:max_lp]):
        if time.monotonic() > deadline - 250:
            break
        if sc0 >= best_score + 0.03:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        remaining = deadline - time.monotonic() - 180
        if remaining < 15:
            break

        if i == 0:
            lp_time = min(500.0, remaining * 0.45)
        elif i < 2:
            lp_time = min(250.0, remaining * 0.3)
        else:
            lp_time = min(120.0, remaining * 0.2)

        if lp_time < 10:
            break

        lp_end = time.monotonic() + lp_time
        print(f"LP [{name} n={n_t}] init={sc0:.8f} budget={lp_time:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=50000, inflate_start=1.002)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> {sc_lp:.10f} (improved)")
        else:
            print(f"  -> {sc_lp:.8f}")

    # Phase 2: Gradient polish
    if best_seq is not None and time.monotonic() < deadline - 120:
        gp_time = min(250.0, deadline - time.monotonic() - 80)
        if gp_time > 15:
            gp_end = time.monotonic() + gp_time
            print(f"Gradient: budget={gp_time:.0f}s")
            a_gp = _gradient_polish(best_seq, gp_end, max_iter=50000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
                print(f"  -> {sc_gp:.10f} (improved)")
            else:
                print(f"  -> {sc_gp:.8f}")

    # Phase 3: Mirror descent
    if best_seq is not None and time.monotonic() < deadline - 80:
        md_time = min(120.0, deadline - time.monotonic() - 50)
        if md_time > 10:
            md_end = time.monotonic() + md_time
            print(f"Mirror: budget={md_time:.0f}s")
            a_md = _mirror_descent(best_seq, md_end, max_iter=30000)
            sc_md = _score(a_md)
            if np.isfinite(sc_md) and sc_md < best_score:
                best_score = sc_md
                best_seq = a_md.copy()
                print(f"  -> {sc_md:.10f} (improved)")
            else:
                print(f"  -> {sc_md:.8f}")

    # Phase 4: Final tight LP
    if best_seq is not None and time.monotonic() < deadline - 40:
        lp_time = min(150.0, deadline - time.monotonic() - 30)
        if lp_time > 10:
            lp_end = time.monotonic() + lp_time
            print(f"Final LP: budget={lp_time:.0f}s")
            a_final = _lp_refine(best_seq, lp_end, max_iter=50000,
                                 inflate_start=1.001, inflate_min=1.0003)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  -> {sc_final:.10f} (improved)")
            else:
                print(f"  -> {sc_final:.8f}")

    # Phase 5: Final gradient
    if best_seq is not None and time.monotonic() < deadline - 15:
        gp_time = min(50.0, deadline - time.monotonic() - 8)
        if gp_time > 5:
            gp_end = time.monotonic() + gp_time
            a_g = _gradient_polish(best_seq, gp_end, max_iter=20000)
            sc_g = _score(a_g)
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
                print(f"  -> Final grad: {sc_g:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)

    # Validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
