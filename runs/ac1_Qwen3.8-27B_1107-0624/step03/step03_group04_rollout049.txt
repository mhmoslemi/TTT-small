        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        pressure[i] += a[j]
                    else:
                        pressure[i] += 2.0 * a[j]

        pmax = pressure.max()
        if pmax < 1e-15:
            break

        # Exponential tilting
        a_new = a * np.exp(-str_cur * pressure / pmax)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.3, str_cur * 1.005)
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                str_cur = max(0.02, str_cur * 0.9)
            if no_imp > 1500:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a


def _adam_smooth_max(a, deadline, lr=0.05, beta_init=None, max_iter=30000):
    """Adam optimizer in logit space with adaptive beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    if beta_init is not None:
        beta = beta_init
    else:
        beta = 0.05 * c_max

    best_a = a.copy()
    best_score = 2.0 * n * c_max

    z = np.log(a)
    z -= np.mean(z)

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        e = np.exp(z - np.max(z))
        p = e / e.sum()
        c = _fft_convolve(p)
        val, ga = _smooth_max_val_grad(c, p, beta)
        sga = float(np.dot(ga, p))
        gz = p * (ga - sga)

        if not np.all(np.isfinite(gz)):
            z = np.log(best_a + 1e-30)
            z -= np.mean(z)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - base_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z -= np.mean(z)

        e = np.exp(z - np.max(z))
        a = e / e.sum()
        c = _fft_convolve(a)
        sc = 2.0 * n * float(c.max())

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                base_lr = max(0.005, base_lr * 0.9)

        # Adaptive beta
        if it % 200 == 0 and it > 100:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                gap = top5[0] - top5[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.6)
                elif gap > beta * 5:
                    beta = min(0.3 * float(c.max()), beta * 1.3)

        # Restart if stuck
        if no_imp > max(5000, 10 * n):
            rng = np.random.default_rng(it * 104729 + 7)
            z = z + 0.2 * rng.standard_normal(n)
            z = np.clip(z, -60, 60)
            z -= np.mean(z)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            base_lr = lr
            no_imp = 0
            beta = max(1e-10, 0.05 * float(_fft_convolve(a).max()))

    return best_a


def _interpolate_profile(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return a_new / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Diversified initialization across lengths
    2. Adam coarse optimization
    3. L-BFGS fine optimization with adaptive beta
    4. Peak equalization
    5. Aggressive LP refinement
    6. Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    lengths = [768, 1024, 2048]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)

    # Phase 1: Generate initializations
    print(f"Phase 1: Initialization (deadline {deadline - time.monotonic():.0f}s)", flush=True)
    candidates = []

    for n_t in lengths:
        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
            candidates.append((f"parent_{n_t}", n_t, a_init))

        a_init = _make_linear_profile(n_t, 1.5)
        candidates.append((f"linear_{n_t}", n_t, a_init))

        a_init = _make_bell_profile(n_t, 0.35)
        candidates.append((f"bell_{n_t}", n_t, a_init))

        a_init = _flat_spectrum_init(n_t, rng)
        candidates.append((f"flat_{n_t}", n_t, a_init))

    # Quick score all inits
    init_scores = []
    for name, n_t, a0 in candidates:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print("  Top 8 initializations:", flush=True)
    for sc, name, n_t, _ in init_scores[:8]:
        print(f"    {name}: {sc:.8f}", flush=True)

    # Phase 2: Adam optimization on top candidates
    top_inits = init_scores[:6]
    per_candidate_time = min(300.0, (deadline - t_start - 600) / len(top_inits))

    for sc0, name, n_t, a0 in top_inits:
        if time.monotonic() > deadline - 400:
            break
        # Skip if clearly worse than best and not the parent's length
        if sc0 > best_score * 1.3 and parent is not None and n_t != parent.size:
            continue

        ad_budget = min(deadline - 350, time.monotonic() + per_candidate_time)
        if ad_budget - time.monotonic() < 30:
            break

        print(f"Phase 2: Adam on {name} (n={n_t}), start={sc0:.8f}, budget={ad_budget - time.monotonic():.0f}s", flush=True)
        a_opt = _adam_smooth_max(a0, ad_budget, lr=0.04, max_iter=20000)
        sc_opt = _score(a_opt)
        print(f"  Adam result: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {sc_opt:.10f} ***", flush=True)

        # Phase 3: L-BFGS refinement
        if time.monotonic() < deadline - 200:
            lb_budget = min(deadline - 180, time.monotonic() + min(150.0, per_candidate_time * 0.5))
            if lb_budget - time.monotonic() > 20:
                print(f"  L-BFGS, budget={lb_budget - time.monotonic():.0f}s", flush=True)
                a_lb = _lbfgs_optimize(a_opt, lb_budget, max_sessions=25)
                sc_lb = _score(a_lb)
                print(f"  L-BFGS result: {sc_lb:.10f}", flush=True)
                if sc_lb < best_score:
                    best_score = sc_lb
                    best_seq = a_lb.copy()
                    print(f"  *** New best: {sc_lb:.10f} ***", flush=True)
                a_opt = a_lb

    # Phase 4: Peak equalization on best
    if best_seq is not None and time.monotonic() < deadline - 100:
        pe_budget = min(deadline - 80, time.monotonic() + 80)
        if pe_budget - time.monotonic() > 20:
            print(f"Phase 4: Peak equalize, budget={pe_budget - time.monotonic():.0f}s", flush=True)
            a_pe = _peak_equalize(best_seq, pe_budget, max_iter=5000, n_peaks=15, strength=0.06)
            sc_pe = _score(a_pe)
            print(f"  Peak eq result: {sc_pe:.10f}", flush=True)
            if sc_pe < best_score:
                best_score = sc_pe
                best_seq = a_pe.copy()
                print(f"  *** New best: {sc_pe:.10f} ***", flush=True)

    # Phase 5: Aggressive LP refinement
    if best_seq is not None and time.monotonic() < deadline - 30:
        lp_budget = min(deadline - 20, time.monotonic() + 120)
        if lp_budget - time.monotonic() > 30:
            print(f"Phase 5: LP refine, budget={lp_budget - time.monotonic():.0f}s", flush=True)
            a_lp = _lp_refine(best_seq, lp_budget, max_iter=50000, inflate_start=1.0004, blend_start=0.12)
            sc_lp = _score(a_lp)
            print(f"  LP result: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  *** New best: {sc_lp:.10f} ***", flush=True)

    # Phase 6: Quick L-BFGS polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        pl_budget = deadline - 5
        if pl_budget - time.monotonic() > 30:
            print(f"Phase 6: Final L-BFGS polish, budget={pl_budget - time.monotonic():.0f}s", flush=True)
            a_final = _lbfgs_optimize(best_seq, pl_budget, max_sessions=15, beta_init=1e-8)
            sc_final = _score(a_final)
            print(f"  Final polish: {sc_final:.10f}", flush=True)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  *** New best: {sc_final:.10f} ***", flush=True)

    # Phase 7: Final LP if time remains
    if best_seq is not None and time.monotonic() < deadline - 5:
        if deadline - time.monotonic() > 15:
            a_last = _lp_refine(best_seq, deadline - 3, max_iter=10000, inflate_start=1.0002, blend_start=0.08)
            sc_last = _score(a_last)
            if sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LP: {sc_last:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(768, 1.5)
        best_score = _score(best_seq)

    # Final validation with evaluate_sequence
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final > 0.01:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
