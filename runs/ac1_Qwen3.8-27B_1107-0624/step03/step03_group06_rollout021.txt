"""

2n*max(conv(a,a))/sum(a)^2. Key improvements over previous:

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    return a / s


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    if n0 == 0:
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_ramp_rev(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** alpha + 1e-8
    return a / a.sum()


def _make_gauss(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    return a / a.sum()


def _make_lin(n, slope=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_sine(n, freq=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _solve_lp(f, M, n):
    """Solve: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    # Build constraint matrix: A[k, i] = f[k-i] if 0<=k-i<n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search_fine(a, g, score_a, n_points=20):
    """Finer line search with golden-section refinement."""
    best_t = 0.0
    best_sc = score_a
    # Coarse grid
    ts = np.linspace(0.02, 1.0, n_points)
    best_coarse_t = 0.0
    best_coarse_sc = score_a
    for t in ts:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_coarse_sc:
            best_coarse_sc = sc
            best_coarse_t = t
    if best_coarse_t == 0.0:
        return 0.0, best_sc
    # Golden section refinement around best coarse point
    idx = int(np.argmin(ts - best_coarse_t))
    lo = ts[max(0, idx - 1)]
    hi = ts[min(len(ts) - 1, idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = _line_eval(a, g, x1)
    f2 = _line_eval(a, g, x2)
    for _ in range(15):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = _line_eval(a, g, x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = _line_eval(a, g, x2)
    best_t = (lo + hi) / 2.0
    best_sc = min(f1, f2)
    if best_sc < score_a - 1e-13:
        return best_t, best_sc
    return 0.0, score_a


def _line_eval(a, g, t):
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return np.inf
    a_new = a_new / s
    return _score(a_new)


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.003, inflate_min=1.0002):

    a = np.asarray(a0, dtype=np.float64).copy()

    best_score = _score(a)

        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            consecutive_stuck += 1
            if consecutive_stuck % 10 == 0:
                inflate = min(0.05, inflate * 1.5)
            if consecutive_stuck > 100:
                # Try a different strategy: perturb and continue
                if no_imp > 50:
                    a = best_a.copy()
                    inflate = inflate_start
                    consecutive_stuck = 0
                    no_imp = 0
                    if inflate <= inflate_min + 0.001:
                        break
                else:
                    no_imp += 1
                    if no_imp > 200:
                        break
            continue

        consecutive_stuck = 0
        g = g / sum_g

        best_t, best_sc = _line_search_fine(a, g, best_score, n_points=15)

        if best_t > 0 and best_sc < best_score - 1e-13:

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish_adam(a0, deadline, max_iter=50000, top_k=3):
    """Adam-based gradient polish with deflation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score(a)

    c0 = _fft_convolve(a)
    beta = max(1e-6, 0.03 * float(c0.max()))
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t_count = 0
    no_imp = 0
    best_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_count += 1
        grad, maxc = _compute_grad_deflated(a, beta, top_k=top_k)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            continue

        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_unbias = m_hat / (1 - beta1 ** t_count)
        v_unbias = v_hat / (1 - beta2 ** t_count)
        a_new = a - lr * m_unbias / (np.sqrt(v_unbias) + eps)
        a_new = np.maximum(a_new, 0.0)
        a = _project_simplex(a_new)

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(2000, 4 * n):
                a = best_a.copy()
                m_hat = np.zeros(n)
                v_hat = np.zeros(n)
                t_count = 0
                c_temp = _fft_convolve(a)
                beta = max(1e-6, 0.03 * float(c_temp.max()))
                lr = best_lr
                no_imp = 0

        # Anneal beta every 200 iterations
        if it % 200 == 199:
            beta = max(1e-8, beta * 0.75)

    return best_a

        if time.monotonic() > deadline:
            break

        grad, _ = _compute_grad_deflated(a, beta, top_k=top_k)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.92)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                c_temp = _fft_convolve(a)
                beta = max(1e-6, 0.05 * float(c_temp.max()))
                lr = base_lr
                no_imp = 0

        if it % 100 == 99:
            beta = max(1e-8, beta * 0.8)

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP refinement + Adam gradient polish with deflation.
    Uses parent sequence and multiple initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Build initializations
    configs = []
    
    if parent is not None:
        n_parent = parent.size
        configs.append(('parent', n_parent, parent.copy()))
        for n_t in [512, 768, 1024, 1536]:
            if n_t != n_parent:
                configs.append((f'parent_{n_t}', n_t, _interpolate(parent, n_t)))
    
    # Additional profiles
    for n_t in [512, 768, 1024]:
        configs.append((f'ramp_{n_t}', n_t, _make_ramp(n_t, 1.0)))
        configs.append((f'gauss_{n_t}', n_t, _make_gauss(n_t, 0.35)))
        configs.append((f'lin_{n_t}', n_t, _make_lin(n_t, 3.0)))
        configs.append((f'sine_{n_t}', n_t, _make_sine(n_t, 1.0)))

    # Score all initializations
    scored_configs = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored_configs.append((sc, name, n_t, a0))
    scored_configs.sort(key=lambda x: x[0])

    print(f"Top 5 initializations:")
    for sc, name, n_t, _ in scored_configs[:5]:
        print(f"  {name}: score={sc:.8f}, n={n_t}")

    # Phase 1: LP refinement on top initializations
    # Give most time to the best 2-3
    n_configs = min(5, len(scored_configs))
    for i in range(n_configs):
        if time.monotonic() > deadline - 120:
            break
        sc0, name, n_t, a0 = scored_configs[i]
        
        if sc0 >= best_score + 0.3:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Allocate time: most to the first (best) config
        remaining = deadline - time.monotonic() - 90
        if remaining < 15:
            break
        
        if i == 0:
            lp_time = min(400.0, remaining * 0.5)
        elif i == 1:
            lp_time = min(250.0, remaining * 0.3)
        else:
            lp_time = min(120.0, remaining * 0.15)
        
        if lp_time < 10:
            break

        lp_end = time.monotonic() + lp_time
        print(f"\nLP phase: {name} n={n_t} init={sc0:.8f} budget={lp_time:.0f}s")

        # Use tighter inflation for parent-based starts
        if 'parent' in name:
            inf_start = 1.002
        else:
            inf_start = 1.005

        a_lp = _lp_refine(a0, lp_end, max_iter=50000, inflate_start=inf_start)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> LP score={sc_lp:.10f} (improved!)")
        else:
            print(f"  -> LP score={sc_lp:.8f} (no improvement)")

    # Phase 2: Adam gradient polish on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        gp_time = min(200.0, deadline - time.monotonic() - 50)
        if gp_time > 15:
            gp_end = time.monotonic() + gp_time
            print(f"\nAdam gradient polish: budget={gp_time:.0f}s")
            a_gp = _gradient_polish_adam(best_seq, gp_end, max_iter=50000, top_k=3)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
                print(f"  -> Adam score={sc_gp:.10f} (improved!)")
            else:
                print(f"  -> Adam score={sc_gp:.8f}")

    # Phase 3: Another LP pass with tighter parameters
    if best_seq is not None and time.monotonic() < deadline - 40:
        lp_time = min(150.0, deadline - time.monotonic() - 25)
        if lp_time > 10:
            lp_end = time.monotonic() + lp_time
            print(f"\nFinal LP (tight): budget={lp_time:.0f}s")
            a_final = _lp_refine(best_seq, lp_end, max_iter=50000,
                                 inflate_start=1.001, inflate_min=1.0001)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  -> Final LP score={sc_final:.10f} (improved!)")
            else:
                print(f"  -> Final LP score={sc_final:.8f}")

    # Phase 4: Momentum gradient polish (complementary to Adam)
    if best_seq is not None and time.monotonic() < deadline - 20:
        gp_time = min(100.0, deadline - time.monotonic() - 10)
        if gp_time > 10:
            gp_end = time.monotonic() + gp_time
            print(f"\nMomentum gradient: budget={gp_time:.0f}s")
            a_g2 = _gradient_polish_momentum(best_seq, gp_end, max_iter=50000, top_k=2)
            sc_g2 = _score(a_g2)
            if np.isfinite(sc_g2) and sc_g2 < best_score:
                best_score = sc_g2
                best_seq = a_g2.copy()
                print(f"  -> Momentum score={sc_g2:.10f} (improved!)")
            else:
                print(f"  -> Momentum score={sc_g2:.8f}")

    # Phase 5: Final tight LP squeeze
    if best_seq is not None and time.monotonic() < deadline - 5:
        lp_time = deadline - time.monotonic() - 3
        if lp_time > 5:
            lp_end = time.monotonic() + lp_time
            a_squeeze = _lp_refine(best_seq, lp_end, max_iter=50000,
                                   inflate_start=1.0005, inflate_min=1.0001)
            sc_sq = _score(a_squeeze)
            if np.isfinite(sc_sq) and sc_sq < best_score:
                best_score = sc_sq
                best_seq = a_squeeze.copy()
                print(f"  -> Squeeze LP score={sc_sq:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)

    # Verify with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nevaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print(f"\nevaluate_sequence returned inf, using internal score: {best_score:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
