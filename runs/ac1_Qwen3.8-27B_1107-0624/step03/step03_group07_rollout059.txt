def lp_refine(a, M, n):
    """
    Given sequence a and target max M, find g >= 0 maximizing sum(g)
    subject to conv(a, g)[k] <= M for all k.
    """
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    
    constraints = [cp.sum(g) <= cp.sum(a)]  # normalize
    
    # conv(a, g)[k] = sum_i a[i] * g[k-i] for valid i
    # For k = 0, ..., 2n-2:
    # conv[k] = sum_{i: max(0,k-n+1) <= i <= min(k,n-1)} a[i] * g[k-i]

    prob.solve(solver=cp.CBC)  # or another solver
    
    if g.value is None:

"""
LP-guided optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions, combined with
subgradient methods for fast iterations and adaptive step sizes.
"""
import numpy as np
import time
from scipy.optimize import linprog

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)

def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1"""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n-1:2*n-1]

    """Compute subgradient of max(conv(a,a)) with respect to a.
    Returns (max_value, subgradient_vector)."""
    n = len(a)
    c = full_convolve_fft(a)
    M = float(c.max())
    # Find all maximizers (within tolerance)
    tol = 1e-10 * M + 1e-15
    max_mask = c >= M - tol
    max_indices = np.where(max_mask)[0]
    
    # Subgradient: for each max position k, grad_k[i] = 2*a[k-i] (if valid)
    # Average over all max positions
    grad = np.zeros(n)
    for k in max_indices:
        # Valid i: 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= len(max_indices)
    return M, grad

def lp_solve_direction(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, conv(a,g)[k] <= M for all k.
    Returns the optimal g (or None if LP fails)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    
    # Build constraint matrix A for conv(a,g)[k] <= M
    # conv(a,g)[k] = sum_{i: valid} a[i] * g[k-i]
    # For each k = 0, ..., 2n-2, the constraint involves g[0], ..., g[n-1]
    # with coefficients a[i] where i = k - j for valid j
    
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    # Objective: maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Bounds: g[j] >= 0
    bounds = [(0, None)] * n
    
    # Constraints: A @ g <= M (vector of all M's)
    b = np.full(2*n-1, M)
    
    # Solve using HiGHS
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if result.success:
            g = np.maximum(result.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s  # normalize to sum=1
            return None
        else:
            return None
    except Exception:
        return None

def subgradient_step(a, lr, beta=0.0):
    """One subgradient step with softmax weighting if beta > 0."""
    n = len(a)
    c = full_convolve_fft(a)
    
    if beta > 1e-14:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a)
        grad = 2.0 * h
    else:
        M, grad = max_convolve_subgradient(a)
    
    # Project onto tangent space of simplex
    grad = grad - np.mean(grad)
    
    # Step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

"""
LP-guided optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find exact optimal descent directions, combined with
softmax-weighted subgradient methods for fast iterations and adaptive step sizes.
Explores multiple sequence lengths and profiles.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 1e-300)


def soft_max_weights(c, beta):
    """Compute soft-max value and softmax weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def softmax_subgradient(c, beta):
    """Compute gradient of soft_max(conv) using given convolution c."""
    _, w = soft_max_weights(c, beta)
    return w


def exact_subgradient(a):
    """Compute exact subgradient of max(conv(a,a)) at a."""
    n = len(a)
    c = full_convolve_fft(a)
    M = float(c.max())
    tol = max(1e-10 * M, 1e-15)
    max_idx = np.where(c >= M - tol)[0]
    grad = np.zeros(n)
    for k in max_idx:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        grad[i_start:i_end + 1] += 2.0 * a[i_end - (i_end - i_start):0:-1][:i_end - i_start + 1] if i_end >= i_start else 0
    # Simpler:
    grad = np.zeros(n)
    for k in max_idx:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= max(1, len(max_idx))
    return M, grad


def build_lp_matrix(a):
    """Build the constraint matrix A for conv(a,g)[k] <= M.
    A[k,j] = a[k-j] if 0 <= k-j < n, else 0.
    Returns dense matrix of shape (2n-1, n)."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j_idx = np.arange(n)[None, :].astype(np.int64)
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    diff_clipped = np.clip(diff, 0, n - 1)
    A = np.where(valid, a[diff_clipped], 0.0)
    return A


def lp_solve_direction(a, M, max_time=5.0):
    """Solve LP: max sum(g) s.t. g >= 0, conv(a,g)[k] <= M for all k.
    Returns normalized g (sum=1) or None."""
    n = len(a)
    t0 = time.monotonic()
    if time.monotonic() - t0 > max_time:
        return None

    A = build_lp_matrix(a)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'time_limit': max_time})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
        return None
    except Exception:
        return None


def subgradient_optimize(a0, deadline, rng, lr=0.02, beta_init=0.05,

    n = a0.size
    a0 = np.maximum(a0, 1e-15)

    beta = beta_init * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr

        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        a0 = np.exp(z - z.max())

            a0 = best_a.copy()
            m.fill(0.0)

            beta = max(beta_min, 0.05 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

    return a / a.sum()


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_asymmetric_bump(n, width=0.4, offset=0.7):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - offset) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    LP-guided multi-phase optimization for minimizing the convolution score.
    Phase 1: Quick subgradient exploration at multiple n values.
    Phase 2: LP-based optimization at the best n.
    Phase 3: Try larger n with LP.
    Phase 4: Final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting LP-guided search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Quick exploration =====
    phase1_end = min(deadline, t_start + 120.0)
    print(f"\n--- Phase 1: Quick exploration (until {phase1_end - t_start:.0f}s) ---")

    # Try multiple n values and profiles
    n_candidates = [512, 1024, 2048]
    profiles_per_n = {}

    for n in n_candidates:
        profiles = [make_power_ramp(n, p) for p in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]]
        profiles.append(make_linear_profile(n, 1.5))
        profiles.append(make_asymmetric_bump(n, 0.4, 0.7))
        if init_seq is not None:
            profiles.insert(0, interpolate_profile(init_seq, n))
        profiles_per_n[n] = profiles

    # Quick subgradient on each profile
    for n in n_candidates:
        for idx, p in enumerate(profiles_per_n[n]):
            if time.monotonic() > phase1_end - 5:
                break
            sc = compute_score(p)
            if sc >= best_score + 0.05:
                continue
            dl = min(phase1_end, time.monotonic() + 8.0)
            result, sc_r = subgradient_optimize(p, dl, rng, lr=0.04,
                                                 beta_init=0.08, beta_decay=0.95)
            sc_final = compute_score(result)
            if sc_final < best_score + 0.02:
                print(f"  n={n}, profile {idx}: score={sc_final:.8f}")
            update(result, sc_final)

    print(f"  Phase 1 best: {best_score:.10f} (n={best_seq.size if best_seq is not None else '?'})")

    # ===== Phase 2: LP optimization at best n =====
    n_main = best_seq.size if best_seq is not None else 2048
    phase2_end = min(deadline, t_start + 1400.0)
    print(f"\n--- Phase 2: LP optimization at n={n_main} ---")

    a_main = best_seq.copy() if best_seq is not None else make_power_ramp(n_main, 1.0)
    a_main = np.maximum(a_main, 1e-15)
    a_main = a_main / a_main.sum()

    result, sc = lp_based_optimize(a_main, phase2_end, rng,
                                    base_steps=[0.05, 0.1, 0.15, 0.25, 0.4, 0.6])
    sc_final = compute_score(result)
    update(result, sc_final)
    print(f"  After LP optimization: {sc_final:.10f}")

    # ===== Phase 3: Try larger n =====
    if time.monotonic() < deadline - 200:
        phase3_end = min(deadline, t_start + 1650.0)
        for n_large in [3072, 4096]:
            if time.monotonic() > phase3_end - 30:
                break
            print(f"\n  Trying n={n_large}...")
            p_large = interpolate_profile(best_seq, n_large)
            sc_large_init = compute_score(p_large)
            update(p_large, sc_large_init)
            print(f"  Interpolated to n={n_large}: {sc_large_init:.10f}")

            if sc_large_init < best_score + 0.02:
                # Quick subgradient warmup
                dl_warm = min(phase3_end, time.monotonic() + 30.0)
                r_warm, sc_warm = subgradient_optimize(p_large, dl_warm, rng,
                                                        lr=0.02, beta_init=0.04,
                                                        beta_decay=0.96)
                sc_warm = compute_score(r_warm)
                update(r_warm, sc_warm)

                # LP optimization
                dl_lp = min(phase3_end, time.monotonic() + 180.0)
                r_lp, sc_lp = lp_based_optimize(r_warm, dl_lp, rng,
                                                 base_steps=[0.05, 0.1, 0.2, 0.3, 0.5])
                sc_lp_final = compute_score(r_lp)
                update(r_lp, sc_lp_final)
                print(f"  After n={n_large} LP: {sc_lp_final:.10f}")

    # ===== Phase 4: Final polish at best n =====
    phase4_end = deadline
    print(f"\n--- Phase 4: Final polish ---")

    if best_seq is not None:
        n_polish = best_seq.size
        a_polish = best_seq.copy()
        a_polish = np.maximum(a_polish, 1e-15)
        a_polish = a_polish / a_polish.sum()

        # More LP iterations with smaller steps
        dl_polish = min(phase4_end, time.monotonic() + 100.0)
        r_polish, sc_polish = lp_based_optimize(a_polish, dl_polish, rng,
                                                 base_steps=[0.02, 0.05, 0.1, 0.15, 0.25])
        sc_polish_final = compute_score(r_polish)
        update(r_polish, sc_polish_final)
        print(f"  After polish LP: {sc_polish_final:.10f}")

        # Subgradient with very small beta
        if time.monotonic() < phase4_end - 20:
            dl_sg = min(phase4_end, time.monotonic() + 50.0)
            a_sg = best_seq + 0.0002 * rng.normal(size=n_polish)
            a_sg = np.maximum(a_sg, 1e-12)
            a_sg = a_sg / a_sg.sum()
            r_sg, sc_sg = subgradient_optimize(a_sg, dl_sg, rng, lr=0.008,
                                                beta_init=0.01, beta_decay=0.97,
                                                beta_min=1e-10)
            sc_sg_final = compute_score(r_sg)
            update(r_sg, sc_sg_final)
            print(f"  After polish subgradient: {sc_sg_final:.10f}")

    # ===== Check smaller n =====
    if time.monotonic() < deadline - 15 and best_seq is not None and best_seq.size > 1024:
        for n_try in [512, 1024]:
            if time.monotonic() > deadline - 10:
                break
            p_small = interpolate_profile(best_seq, n_try)
            sc_s = compute_score(p_small)
            update(p_small, sc_s)
            if sc_s < best_score + 0.005:
                dl_s = min(deadline, time.monotonic() + 20.0)
                r_s, _ = subgradient_optimize(p_small, dl_s, rng, lr=0.015,
                                               beta_init=0.02, beta_decay=0.97)
                sc_s2 = compute_score(r_s)
                update(r_s, sc_s2)
                print(f"  Smaller n={n_try}: {sc_s2:.10f}")

    # ===== Final validation and return =====
    print(f"\n  Final best score: {best_score:.10f}")

    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_power_ramp(2048, 1.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_power_ramp(512, 1.0)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
