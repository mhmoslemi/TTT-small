def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = c - c.min()
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):

    a = np.asarray(a, dtype=np.float64).ravel()

    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _full_conv_matrix(n):

    for j in range(n):
        # a[j] contributes to positions k = j, j+1, ..., j+n-1
        # A[k, j] = a[k-j] ... wait, that's not right.
        # A[k, j] should be such that sum_j A[k,j]*g[j] = sum_i a[i]*g[k-i]
        # Let i = k-j, so j = k-i. Then sum_i a[i]*g[k-i] = sum_j a[k-j]*g[j]
        # So A[k,j] = a[k-j] if 0 <= k-j < n
        # For fixed j, A[k,j] = a[k-j] for k in [j, j+n-1]
        A[j:j+n, j] = a[::-1]  # Wait, let me check
        # A[j+m, j] should equal a[j+m-j] = a[m] for m in [0, n-1]
        # So A[j:j+n, j] = a[0:n] = a
    # Let me redo:
    # A[k, j] = a[k-j] if 0 <= k-j < n
    # For fixed j, k ranges from j to j+n-1, and A[k,j] = a[k-j] = a[0], a[1], ..., a[n-1]
    # So column j of A is: zeros, then a[0], a[1], ..., a[n-1], then zeros
    # The non-zero part starts at row j and has length n.
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def _lp_full(a, M, max_iter_lp=100000):
    """Solve the full LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = _full_conv_matrix_fast(a)
    b = np.full(2*n-1, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'maxiter': max_iter_lp})
    if res.success:
        return res.x
    return None


def _lp_blend_line_search(a, g, n_iter=25):
    """Golden section search for optimal blend t in (0,1)."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c*(g-a), 0))
    fd = _score_norm(np.maximum(a + d*(g-a), 0))
    for _ in range(n_iter):
        if fc <= fd:
            hi, d, fd = d, c, fc
            c = hi - gr*(hi-lo)
            fc = _score_norm(np.maximum(a + c*(g-a), 0))
        else:
            lo, c, fc = c, d, fd
            d = lo + gr*(hi-lo)
            fd = _score_norm(np.maximum(a + d*(g-a), 0))
    t = (lo+hi)/2.0
    return t


def _lp_refine_full(a, deadline, max_iter=10000, inflate_start=1.001, inflate_min=1.0001):
    """LP refinement using ALL constraints."""
    a = np.asarray(a, dtype=np.float64).copy()

def _interpolate(a, n_new):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        return a / a.sum()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _momentum_gradient(a, deadline, max_iter=50000, lr=0.05, momentum=0.9, T=0.03):
    """Momentum-based gradient on LSE-smoothed objective in log-space."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    v = np.zeros(n)
    
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        w = ex / ex.sum()
        
        # Gradient in a-space: 2 * conv(w, reverse(a)) at the center
        N = _next_pow2(2*n-1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)[n-1:2*n-1]
        grad = 2.0 * h
        
        # Project to tangent space of simplex
        grad_tan = grad - (a @ grad) * np.ones(n)  # Hmm, this isn't quite right
        # Actually for the constraint sum(a)=1, the tangent space is {v : sum(v)=0}
        grad_tan = grad - (np.sum(grad) / n)
        
        # Update in z-space (log-space)
        gz = a * (grad - np.dot(a, grad))
        gz -= gz.mean()
        
        v = momentum * v + gz
        z = z - lr * v
        z = np.clip(z, -30, 30)
        z -= z.mean()
        
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        score = _score_norm(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 10000:
                a = best_a.copy()
                z = np.log(a)
                z -= z.mean()
                v = np.zeros(n)
                no_imp = 0
            if no_imp % 5000 == 0:
                lr = max(0.01, lr * 0.9)
        
        a = a_new
        T = max(0.005, T * 0.9999)
    
    return best_a


def _kkt_adjust(a, deadline, max_iter=100):
    """Enforce KKT conditions: for the active lag, make relevant values equal."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        k_star = int(np.argmax(c))
        M = c[k_star]
        
        # KKT: for the active lag k_star, a[k_star - i] should be constant
        # for all i where a[i] > 0 (and k_star - i is valid)
        # Find the set of pairs (i, k_star-i) where both are in [0, n-1]
        pairs = []
        for i in range(max(0, k_star - n + 1), min(n, k_star + 1)):
            j = k_star - i
            if 0 <= j < n and a[i] > 1e-10:
                pairs.append((i, j))
        
        if not pairs:
            break
        
        # Average the values at positions {k_star - i : i in support(a)}
        target_positions = set()
        for i, j in pairs:
            target_positions.add(j)
        
        if len(target_positions) < 2:
            break
        
        positions = sorted(target_positions)
        avg_val = np.mean(a[positions])
        
        # Blend towards equal values at these positions
        a_new = a.copy()
        for pos in positions:
            a_new[pos] = 0.9 * a_new[pos] + 0.1 * avg_val
        
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = _score_norm(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def _random_perturb(a, sigma=0.05):
    """Add random noise and re-normalize."""
    rng = np.random.default_rng()
    a_pert = a * (1.0 + sigma * rng.normal(size=len(a)))
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s
