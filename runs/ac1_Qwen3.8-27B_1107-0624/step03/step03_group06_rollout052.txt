"""
Hybrid optimization combining LP-directed updates, Nesterov accelerated
subgradient descent with smooth-max objective, and periodic perturbations
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v

def _next_pow2(n):

            best_a = a.copy()
            no_imp = 0
            imp_count += 1
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.001, lr * 0.93)
            if no_imp > 2000:
                # Restart from best
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.02 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        # Adaptive beta: decay over time
        if it % 200 == 199:
            beta = max(1e-10, beta * 0.85)

        # Periodically increase momentum
        if it % 500 == 499:
            momentum = min(0.9, momentum + 0.01)

    return best_a


def _subgrad_descent(a0, deadline, max_iter=50000, lr_init=0.05):
    """Subgradient descent using exact subgradient at the argmax."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_normalized(a)

    lr = lr_init
    base_lr = lr
    momentum = 0.5
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))
        grad = _subgrad_at(a, k_star)

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                lr = max(0.005, lr * 0.95)
            if no_imp > 3000:
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

        if it % 300 == 299:
            momentum = min(0.85, momentum + 0.02)

    return best_a


def _combined_optimize(a0, deadline, max_iter=20000):
    """Alternate between LP steps and gradient steps."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_normalized(a)

    beta = max(1e-8, 0.02 * float(_fft_convolve(a).max()))
    lr = 0.03
    momentum = 0.6
    v = np.zeros(n)
    no_imp = 0
    inflate = 1.003

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Every 20th iteration, do an LP step
        if it % 20 == 0:
            c = _fft_convolve(a)
            M = float(c.max()) * inflate
            g = _solve_lp(a, M, n)
            if g is not None:
                sg = float(g.sum())
                if sg > 1.0 + 1e-6:
                    g = g / sg
                    # Try blending
                    for t in [0.2, 0.5]:
                        a_try = (1.0 - t) * a + t * g
                        a_try = np.maximum(a_try, 0.0)
                        st = a_try.sum()
                        if st < 1e-10:
                            continue
                        a_try = a_try / st
                        sc = _score_normalized(a_try)
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a_try.copy()
                            a = a_try
                            no_imp = 0
                            v = np.zeros(n)
                            break
                    no_imp += 1
                    continue
            no_imp += 20
            if no_imp > 300:
                a = best_a.copy()
                v = np.zeros(n)
                no_imp = 0
                inflate = 1.003

        # Gradient step
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0.0)
        sl = a_look.sum()
        if sl < 1e-10:
            a_look = a
        else:
            a_look = a_look / sl

        grad, mval, _ = _smooth_max_grad(a_look, beta)
        if not np.all(np.isfinite(grad)):
            beta *= 2.0
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if it % 300 == 299:
            beta = max(1e-10, beta * 0.9)
            momentum = min(0.85, momentum + 0.01)

    return best_a


def _perturb(a, rng, magnitude=0.05):
    """Add random perturbation and re-project to simplex."""
    n = len(a)
    noise = rng.normal(0, magnitude, n)
    a_new = a + noise
    return _project_simplex(a_new)


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization: generate diverse initializations, refine with LP,
    polish with Nesterov accelerated gradient descent, and combine with
    periodic LP steps. Multiple lengths and restarts for robustness.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initializations at multiple lengths
    lengths = [512, 768, 1024, 1280]
    profiles_funcs = [
        ('ramp1', lambda n: _make_ramp(n, 1.0)),
        ('ramp2', lambda n: _make_ramp(n, 2.0)),
        ('ramp3', lambda n: _make_ramp(n, 3.0)),
        ('cos', lambda n: _make_cosine(n)),
        ('gauss', lambda n: _make_gaussian(n, 0.25)),
        ('bump', lambda n: _make_bump(n, 0.4)),
        ('double', lambda n: _make_double_bump(n)),
        ('power', lambda n: _make_power(n, 0.5)),
        ('semi', lambda n: _make_semicircle(n)),
        ('uniform', lambda n: _make_uniform(n)),
    ]

    candidates = []
    for n_t in lengths:
        if parent is not None:
            p_int = _interpolate(parent, n_t)
            candidates.append(('parent', n_t, p_int))
        for name, func in profiles_funcs:
            a0 = func(n_t)
            candidates.append((name, n_t, a0))

    # Score all candidates
    scored = []
    for name, n_t, a0 in candidates:
        sc = _score(a0)
        scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    # Keep top candidates (those better than best_score + 0.3)
    top_candidates = [(sc, name, n_t, a0) for sc, name, n_t, a0 in scored
                      if np.isfinite(sc) and sc < best_score + 0.3]
    if not top_candidates:
        top_candidates = scored[:5]

    # Update best from initializations
    for sc, name, n_t, a0 in top_candidates:
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a0.copy()

    print(f"Top init: score={top_candidates[0][0]:.8f}, n={top_candidates[0][2]}")
    print(f"Current best: {best_score:.10f}")

    # Phase 1: LP refinement on top 3-5 candidates
    phase1_budget = 350.0
    phase1_end = t_start + phase1_budget
    top_for_lp = top_candidates[:5]

    for sc0, name, n_t, a0 in top_for_lp:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 20:
            break

        lp_time = min(remaining * 0.9, 200.0)
        lp_end = time.monotonic() + lp_time
        print(f"LP: {name} n={n_t} score={sc0:.8f} time={lp_time:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=5000, inflate_start=1.003)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> LP: {sc_lp:.10f} (improved!)")
        else:
            print(f"  -> LP: {sc_lp:.8f}")

    # Phase 2: Nesterov gradient descent on best
    if best_seq is not None and time.monotonic() < deadline - 300:
        gp_time = min(700.0, deadline - time.monotonic() - 200)
        if gp_time > 50:
            gp_end = time.monotonic() + gp_time
            print(f"Nesterov GD: budget={gp_time:.0f}s, n={best_seq.size}")
            a_gp = _nesterov_grad_descent(best_seq, gp_end, max_iter=200000,
                                          beta_init=None, lr_init=0.02, momentum_init=0.6)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
                print(f"  -> Nesterov: {sc_gp:.10f} (improved!)")
            else:
                print(f"  -> Nesterov: {sc_gp:.8f}")

    # Phase 3: Combined LP + gradient
    if best_seq is not None and time.monotonic() < deadline - 150:
        comb_time = min(300.0, deadline - time.monotonic() - 100)
        if comb_time > 30:
            comb_end = time.monotonic() + comb_time
            print(f"Combined: budget={comb_time:.0f}s")
            a_comb = _combined_optimize(best_seq, comb_end, max_iter=30000)
            sc_comb = _score(a_comb)
            if np.isfinite(sc_comb) and sc_comb < best_score:
                best_score = sc_comb
                best_seq = a_comb.copy()
                print(f"  -> Combined: {sc_comb:.10f} (improved!)")
            else:
                print(f"  -> Combined: {sc_comb:.8f}")

    # Phase 4: Subgradient descent (different approach may find different local min)
    if best_seq is not None and time.monotonic() < deadline - 80:
        sg_time = min(150.0, deadline - time.monotonic() - 50)
        if sg_time > 20:
            sg_end = time.monotonic() + sg_time
            print(f"Subgrad: budget={sg_time:.0f}s")
            a_sg = _subgrad_descent(best_seq, sg_end, max_iter=50000, lr_init=0.05)
            sc_sg = _score(a_sg)
            if np.isfinite(sc_sg) and sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"  -> Subgrad: {sc_sg:.10f} (improved!)")
            else:
                print(f"  -> Subgrad: {sc_sg:.8f}")

    # Phase 5: Perturbation + re-optimize (try to escape local min)
    if best_seq is not None and time.monotonic() < deadline - 60:
        for mag in [0.03, 0.01]:
            if time.monotonic() > deadline - 40:
                break
            a_pert = _perturb(best_seq, rng, magnitude=mag)
            # Quick gradient polish
            pol_end = time.monotonic() + min(40.0, deadline - time.monotonic() - 30)
            if pol_end < time.monotonic() + 5:
                break
            a_pol = _nesterov_grad_descent(a_pert, pol_end, max_iter=50000,
                                           beta_init=None, lr_init=0.03, momentum_init=0.7)
            sc_pol = _score(a_pol)
            if np.isfinite(sc_pol) and sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"  -> Perturb({mag}) + polish: {sc_pol:.10f} (improved!)")
            else:
                print(f"  -> Perturb({mag}) + polish: {sc_pol:.8f}")

    # Phase 6: Final LP polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        lf_time = min(60.0, deadline - time.monotonic() - 10)
        if lf_time > 10:
            lf_end = time.monotonic() + lf_time
            a_final = _lp_refine(best_seq, lf_end, max_iter=2000,
                                 inflate_start=1.001, inflate_min=1.0001)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  -> Final LP: {sc_final:.10f} (improved!)")
            else:
                print(f"  -> Final LP: {sc_final:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 2.0)
        best_score = _score(best_seq)

    # Final validation and normalization
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 2.0)

    # Verify with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    # Try a few more quick variations at the end
    if time.monotonic() < deadline - 5:
        for n_try in [best_seq.size, best_seq.size + 128, best_seq.size - 128]:
            if n_try < 64 or n_try > 2048:
                continue
            a_try = _interpolate(best_seq, n_try)
            sc_try = _score(a_try)
            if np.isfinite(sc_try) and sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()
                print(f"  -> Resized to {n_try}: {sc_try:.10f} (improved!)")

    # Rebuild final
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
