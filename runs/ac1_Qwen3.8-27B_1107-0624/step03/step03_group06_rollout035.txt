def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    c = _fft_convolve(a)

def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:

        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)

    p = np.maximum(p, 0.0)

def _coord_desc(a0, deadline, max_sweeps=50000, n_grid=30, seed=42):

            break

            # new_conv[k] = conv[k] + 2*(x-x_old)*a[k-i] + (x-x_old)^2 * [k==2i]

            k_end = min(2 * n - 1, i + n)  # exclusive

                # No valid k, objective doesn't depend on x (only denominator)
                # Score = 2n * conv.max() / (1-x_old+x)^2, minimized at x=x_old
                continue

            ki_idx = np.arange(k_start, k_end) - i  # indices into a
            a_vals = a[ki_idx]  # shape: (k_end - k_start,)

            # deltas: shape (n_grid,)
            deltas = ts - x_old  # shape: (n_grid,)

            # new_conv for valid k: shape (k_end-k_start, n_grid)
            linear_part = 2.0 * np.outer(deltas, a_vals)  # (n_grid, k_end-k_start)
            # Quadratic part: only at k=2i
            quad_part = np.outer(deltas * deltas, np.zeros(k_end - k_start))
            if 2 * i >= k_start and 2 * i < k_end:
                quad_pos = 2 * i - k_start
                quad_part[:, quad_pos] = deltas * deltas

            new_conv_valid = linear_part.T + quad_part.T  # (k_end-k_start, n_grid)
            new_conv_valid = new_conv_valid + conv[k_start:k_end, np.newaxis]

            # For k outside valid range, new_conv = conv (unchanged)
            max_invalid = conv[:k_start].max() if k_start > 0 else 0.0
            max_invalid2 = conv[k_end:].max() if k_end < 2 * n - 1 else 0.0
            max_outside = max(max_invalid, max_invalid2, 0.0)

            max_conv = np.maximum(new_conv_valid.max(axis=0), max_outside)
            denom = (1.0 - x_old + ts) ** 2
            scores = 2.0 * n * max_conv / denom

            # Find best
            valid_mask = np.isfinite(scores) & (denom > 1e-20)
            if not valid_mask.any():
                continue
            scores[~valid_mask] = np.inf
            best_idx = np.argmin(scores)
            x_new = ts[best_idx]
            new_score = scores[best_idx]

                # Update convolution incrementally
                delta = x_new - x_old
                if abs(delta) > 1e-15:
                    conv[k_start:k_end] += 2.0 * delta * a_vals
                    if 2 * i >= k_start and 2 * i < k_end:
                        conv[2 * i] += delta * delta
                    a[i] = x_new
                    improved = True
                    best_score = new_score
                    best_a = a.copy()

            elif new_score < 2.0 * n * conv.max() / (1.0 - x_old + x_old) ** 2 - 1e-14:

                    a[i] = x_new
                    improved = True

            no_imp_sweeps = 0
            best_no_imp = 0
        else:
            no_imp_sweeps += 1
            best_no_imp += 1
            if best_no_imp > 8:

def _solve_lp(f, M, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

        sc_cd = _score(a_cd)
        if np.isfinite(sc_cd) and sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  -> CD score={sc_cd:.10f}")
        else:
            print(f"  -> CD score={sc_cd:.8f}")

    # Phase 2: Refined coordinate descent on best
    if best_seq is not None and time.monotonic() < deadline - 200:
        remaining = min(200.0, deadline - time.monotonic() - 150)
        if remaining > 30:
            cd_deadline = time.monotonic() + remaining
            print(f"Phase 2 refined CD: time={remaining:.0f}s")
            a_cd2 = _coord_desc_refined(best_seq, cd_deadline, max_sweeps=30000, n_grid=50, seed=200)
            sc_cd2 = _score(a_cd2)
            if np.isfinite(sc_cd2) and sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()
                print(f"  -> Refined CD score={sc_cd2:.10f}")
            else:
                print(f"  -> Refined CD score={sc_cd2:.8f}")

    # Phase 3: LP refinement on best
    if best_seq is not None and time.monotonic() < deadline - 100:
        remaining = min(150.0, deadline - time.monotonic() - 80)
        if remaining > 20:
            lp_deadline = time.monotonic() + remaining
            print(f"Phase 3 LP: time={remaining:.0f}s")
            a_lp = _lp_refine(best_seq, lp_deadline, max_iter=15000, inflate_start=1.003, inflate_min=1.0004)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  -> LP score={sc_lp:.10f}")
            else:
                print(f"  -> LP score={sc_lp:.8f}")

    # Phase 4: Coordinate descent after LP
    if best_seq is not None and time.monotonic() < deadline - 50:
        remaining = min(80.0, deadline - time.monotonic() - 30)
        if remaining > 15:
            cd_deadline = time.monotonic() + remaining
            print(f"Phase 4 CD: time={remaining:.0f}s")
            a_cd3 = _coord_desc_refined(best_seq, cd_deadline, max_sweeps=20000, n_grid=50, seed=500)
            sc_cd3 = _score(a_cd3)
            if np.isfinite(sc_cd3) and sc_cd3 < best_score:
                best_score = sc_cd3
                best_seq = a_cd3.copy()
                print(f"  -> Phase 4 CD score={sc_cd3:.10f}")
            else:
                print(f"  -> Phase 4 CD score={sc_cd3:.8f}")

    # Phase 5: Gradient polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = min(50.0, deadline - time.monotonic() - 10)
        if remaining > 10:
            gp_deadline = time.monotonic() + remaining
            print(f"Phase 5 gradient: time={remaining:.0f}s")
            a_gp = _gradient_polish(best_seq, gp_deadline, max_iter=30000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
                print(f"  -> Gradient score={sc_gp:.10f}")
            else:
                print(f"  -> Gradient score={sc_gp:.8f}")

    # Phase 6: Final coordinate descent polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = min(30.0, deadline - time.monotonic() - 2)
        if remaining > 5:
            cd_deadline = time.monotonic() + remaining
            a_final_cd = _coord_desc_refined(best_seq, cd_deadline, max_sweeps=10000, n_grid=60, seed=999)
            sc_final = _score(a_final_cd)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final_cd.copy()
                print(f"  -> Final CD score={sc_final:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)
        best_score = _score(final_seq)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
