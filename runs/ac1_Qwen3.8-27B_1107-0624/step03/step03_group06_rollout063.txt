    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0 using sparse matrix."""

    else:
        A = lil_matrix((2 * n - 1, n))
        for j in range(n):
            for k in range(j, j + n):
                A[k, j] = f[k - j]
        A = A.tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, base_score):
    best_t = 0.0
    best_sc = base_score
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def _lp_refine(a0, deadline, max_iter=100000, inflate_start=1.004, inflate_min=1.0002):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start
    lp_fail = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            lp_fail += 1
            if lp_fail > 10:
                break
            inflate = min(0.1, inflate * 1.5)
            continue

        lp_fail = 0
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-9:
            no_imp += 1
            if no_imp > 500:
                break
            if no_imp % 100 == 0:
                inflate = min(0.1, inflate * 1.3)
            continue

        g = g / sum_g
        best_t, best_sc = _line_search(a, g, best_score)

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 300 == 299:
                inflate = max(inflate_min, inflate * 0.96)
        else:
            no_imp += 1
            if no_imp % 80 == 0:
                inflate = min(0.1, inflate * 1.15)
            if no_imp > max(400, 3 * n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                lp_fail = 0

    return best_a


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=100000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.08 * float(c.max()))
    lr = 0.04
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    stiff_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad = _smooth_max_grad(a, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 5.0)
            v = np.zeros(n)
            stiff_count += 1
            if stiff_count > 15:
                break
            continue

        v = momentum * v - lr * grad
        a = _project_simplex(a + v)

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.003, lr * 0.93)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.08 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 300 == 299:
            beta = max(1e-9, beta * 0.82)

    return best_a


def _local_search(a0, deadline, rng, max_iter=50000):
    """Random coordinate perturbation local search."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    step = 0.05

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Pick random coordinates to perturb
        n_pert = max(3, n // 32)
        idx = rng.integers(0, n, size=n_pert)
        a_trial = a.copy()
        for i in idx:
            a_trial[i] *= (1.0 + rng.uniform(-step, step))
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        sc = _score(a_trial)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            step = min(0.1, step * 1.05)
        else:
            step = max(0.005, step * 0.97)

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: generate diverse initializations at multiple scales,
    refine via LP best-response iteration, polish with smooth-max gradient descent,
    and local search. Time-budgeted to fit within 1800s.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception as e:
        print(f"No parent available: {e}")

    # Generate candidates
    n_values = [256, 384, 512, 768, 1024]
    if parent is not None and 128 <= parent.size <= 1200:
        n_values.append(parent.size)
    n_values = sorted(set(n_values))

    candidates = []
    for n_t in n_values:
        # Beta distributions (various skewness)
        for p, q in [(0.5, 2.0), (0.3, 1.5), (1.0, 3.0), (0.7, 2.5),
                     (1.5, 0.5), (2.0, 0.5), (0.5, 1.0), (1.0, 1.5)]:
            candidates.append(_make_beta(n_t, p, q))
        # Skew ramps
        for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
            candidates.append(_make_skew_ramp(n_t, alpha))
        # Cosine variants
        for ph in [0.2, 0.35, 0.5]:
            candidates.append(_make_cosine(n_t, ph))
        # Parent interpolation
        if parent is not None:
            candidates.append(_interpolate(parent, n_t))

    # Score and sort candidates
    scored = []
    for c in candidates:
        sc = _score(c)
        if np.isfinite(sc):
            scored.append((sc, c))
    scored.sort(key=lambda x: x[0])

    print(f"Generated {len(scored)} candidates, best init score={scored[0][0]:.8f}")

    # Phase 1: LP refinement on top candidates
    # Use 65% of available time
    lp_budget = (deadline - time.monotonic()) * 0.65
    lp_deadline = t_start + 25 + lp_budget

    top_k = min(6, len(scored))
    for idx in range(top_k):
        if time.monotonic() > lp_deadline:
            break
        sc0, a0 = scored[idx]
        if sc0 >= best_score + 0.08:
            break

        n_t = len(a0)
        # Time for this LP: proportional to remaining budget
        remaining = max(0.0, lp_deadline - time.monotonic())
        frac = 1.0 / (top_k - idx)
        alloc = min(remaining * frac, 280.0) if n_t <= 600 else min(remaining * frac, 200.0)
        if alloc < 15:
            break

        lp_end = time.monotonic() + alloc
        print(f"LP phase: n={n_t}, init={sc0:.8f}, budget={alloc:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=80000, inflate_start=1.004)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> improved to {sc_lp:.10f}")
        else:
            print(f"  -> no improvement ({sc_lp:.8f})")

    # Phase 2: Gradient polish on best (20% of remaining)
    if best_seq is not None:
        rem = deadline - time.monotonic()
        gp_budget = min(rem * 0.25, 250.0)
        if gp_budget > 15:
            gp_end = time.monotonic() + gp_budget
            print(f"Gradient polish: n={len(best_seq)}, budget={gp_budget:.0f}s")
            a_gp = _gradient_polish(best_seq, gp_end, max_iter=80000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  -> score={sc_gp:.10f}")

    # Phase 3: Final LP with tighter params (15% of remaining)
    if best_seq is not None:
        rem = deadline - time.monotonic()
        lp2_budget = min(rem * 0.2, 200.0)
        if lp2_budget > 10:
            lp2_end = time.monotonic() + lp2_budget
            print(f"Final LP: n={len(best_seq)}, budget={lp2_budget:.0f}s")
            a_f = _lp_refine(best_seq, lp2_end, max_iter=50000,
                             inflate_start=1.0015, inflate_min=1.0001)
            sc_f = _score(a_f)
            if np.isfinite(sc_f) and sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  -> score={sc_f:.10f}")

    # Phase 4: Local search (10% of remaining)
    if best_seq is not None:
        rem = deadline - time.monotonic()
        ls_budget = min(rem * 0.15, 120.0)
        if ls_budget > 10:
            ls_end = time.monotonic() + ls_budget
            print(f"Local search: n={len(best_seq)}, budget={ls_budget:.0f}s")
            a_ls = _local_search(best_seq, ls_end, rng, max_iter=30000)
            sc_ls = _score(a_ls)
            if np.isfinite(sc_ls) and sc_ls < best_score:
                best_score = sc_ls
                best_seq = a_ls.copy()
            print(f"  -> score={sc_ls:.10f}")

    # Phase 5: Final gradient polish
    if best_seq is not None:
        rem = deadline - time.monotonic()
        gp2_budget = min(rem * 0.5, 80.0)
        if gp2_budget > 5:
            gp2_end = time.monotonic() + gp2_budget
            a_g2 = _gradient_polish(best_seq, gp2_end, max_iter=40000)
            sc_g2 = _score(a_g2)
            if np.isfinite(sc_g2) and sc_g2 < best_score:
                best_score = sc_g2
                best_seq = a_g2.copy()
            print(f"  -> final gradient score={sc_g2:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_beta(384, 1.0, 2.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_beta(256, 1.0, 2.0)

    if final_seq.size <= 5000:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence validation: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in final_seq]
