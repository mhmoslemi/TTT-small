def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w

def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:

    return p / s

def _solve_sdp(n):
    """Solve the SDP relaxation for the optimal shape at size n."""
    # Variables
    a = cp.Variable(n, nonneg=True)
    B = cp.Variable((n + 1, n + 1), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        B == cp.psd_wrap(B),  # B is PSD
        B[0, 0] == 1,
        cp.sum(a) == 1,
    ]
    
    # Link B[0,i] = a[i-1] for i=1,...,n
    for i in range(n):
        constraints.append(B[0, i + 1] == a[i])
        constraints.append(B[i + 1, 0] == a[i])
    
    # Link B[i+1,j+1] = a[i]*a[j] (relaxed to just PSD)
    # The convex relaxation: B >= 0 automatically implies B[i+1,j+1] <= B[i+1,i+1] = a[i]^2
    # But we don't have B[i+1,i+1] = a[i]^2 as a constraint in the relaxation
    # We need to add: B[i+1, i+1] <= a[i]^2 or use the LMI [a[i]^2, a[i]; a[i], 1] >= 0
    # Actually the standard lifting is:
    # B = [[1, a^T], [a, A]] where A >= 0 and A[i,j] >= a[i]*a[j] (relaxation of A = aa^T)
    # The tightest relaxation with the rank-1 constraint dropped is just A >= 0.
    # But we also need consistency: A[i,i] = a[i]^2 is NOT enforced in the relaxation.
    # Instead, the constraint [a[i]^2, a[i]; a[i], 1] >= 0 gives a[i]^2 >= a[i]^2 (=0 always) -- that's trivial.
    # The correct constraint is: the (i+1,i+1) entry of B should be >= 0 (from PSD) 
    # and the "correct" value is a[i]^2, but in the relaxation it can be anything >= 0.
    # 
    # Actually, the standard SDP relaxation for min max_k (aa^T)_k subject to 1^Ta=1, a>=0 is:
    # min t
    # s.t. sum_{i+j=k} B[i,j+1,j+1] <= t for all k
    #      B >= 0
    #      B[0,0] = 1
    #      B[i+1, 0] = B[0, i+1] = a[i] (linking)
    #      sum a[i] = 1
    #      a >= 0
    # 
    # The "gap" is that B[i+1,j+1] >= a[i]*a[j] is implied by PSD (since B >= 0 and 
    # B[0,i+1]=a[i], B[0,j+1]=a[j], we get B[i+1,j+1] >= a[i]*a[j] by the Schur complement
    # or just by the 2x2 principal minor [1, a[i]; a[j], B[i+1,j+1]] >= 0).
    # Wait, the 2x2 minor of B indexed by {0, i+1, j+1} gives:
    # det([[1, a[i]], [a[j], B[i+1,j+1]]]) >= 0 => B[i+1,j+1] >= a[i]*a[j]
    # (for the 2x2 case). For the 3x3 minor, we get a stronger condition.
    # 
    # So actually, the PSD constraint ALREADY implies B[i+1,j+1] >= a[i]*a[j].
    # The relaxation allows B[i+1,j+1] > a[i]*a[j], which makes the objective (sum of B entries)
    # potentially larger than the true (aa^T)_k. So the SDP gives an UPPER bound on the true optimum.
    # 
    # Wait no: we're MINIMIZING the max of sums of B entries. If B[i+1,j+1] >= a[i]*a[j],
    # then the SDP objective is >= true objective. So the SDP gives an upper bound on the 
    # minimum, which is what we want (it's a relaxation).
    # 
    # If the optimal B is rank-1 (i.e., B[i+1,j+1] = a[i]*a[j] at the optimum), the relaxation is tight.

    # Convolution constraints: (a*a)[k] = sum_{i+j=k} a[i]*a[j] <= t
    # In terms of B: sum_{i+j=k} B[i+1, j+1] <= t
    for k in range(2 * n - 1):
        # i ranges from max(0, k-(n-1)) to min(k, n-1), j = k - i
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        conv_expr = cp.sum([B[i + 1, k - i + 1] for i in range(i_start, i_end + 1)])
        constraints.append(conv_expr <= t)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    # Try to solve
    best_val = np.inf
    best_a = None
    
    solvers_to_try = ['MOSEK', 'SCS', 'CVXOPT', 'ECOS']
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
                val = float(t.value)
                a_val = np.asarray(a.value, dtype=np.float64).ravel()
                if val < best_val and np.all(a_val >= -1e-8):
                    a_val = np.maximum(a_val, 0.0)
                    s = a_val.sum()
                    if s > 1e-10:
                        best_a = a_val / s
                        best_val = val
                break
        except Exception as e:
            print(f"  SDP solver {solver} failed: {e}")
            continue
    
    if best_a is not None:
        print(f"  SDP n={n}: t={best_val:.10f}, score={_score(best_a):.10f}")
    return best_a

def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None
