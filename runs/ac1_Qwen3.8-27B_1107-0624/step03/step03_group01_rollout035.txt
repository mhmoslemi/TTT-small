"""
LP-based peak-shaving with gradient polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative LP to find flattest convolution direction, followed by projected gradient.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _lp_find_flattest(a, M):
    """
    Solve LP: maximize sum(b) s.t. b >= 0, conv(a,b) <= M
    Returns the optimal b.
    """
    n = len(a)
    # Constraint: A @ b <= M * ones(2n-1)
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # Build A as a sparse-like structure (but use dense for simplicity with HiGHS)
    
    # Actually, for n=2048, A is (4095, 2048) which is ~8.4M elements.
    # That's 67 MB as float64. Should be OK.
    
    A_ub = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return a.copy()  # fallback: return a itself (which is feasible)

def _next_pow2(n):

def _full_convolve(a, b):

    fb = np.fft.rfft(b, N)

def _lp_step(a, M):
    """Solve LP to find b maximizing sum(b) s.t. b>=0, conv(a,b) <= M."""
    n = len(a)
    # Build constraint matrix
    A_ub = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(2*n-2, j+n-1)
        A_ub[k_start:k_end+1, j] = a[:k_end+1-j]
    
    b_ub = np.full(2*n-1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return a.copy()

def _project_simplex(v):
    """Project v onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _gradient_descent_step(a, lr, top_k=10):
    """One step of projected gradient descent on the simplex."""
    n = len(a)
    c, peak, grad = _peak_and_grad(a, top_k)
    a_new = a - lr * grad
    a_new = _project_simplex(a_new)
    return a_new, c, peak

def _make_profile(n, type='linear', **kwargs):
    t = np.linspace(0, 1, n)
    if type == 'uniform':
        a = np.ones(n)
    elif type == 'linear':
        slope = kwargs.get('slope', 1.0)
        a = 1.0 + slope * (t - 0.5)
    elif type == 'quadratic':
        a1 = kwargs.get('a1', 1.0)
        a2 = kwargs.get('a2', 0.3)
        u = 2*t - 1
        a = 1.0 + a1*u + a2*u**2
    elif type == 'ramp':
        a = np.linspace(1.0, 1.0 + kwargs.get('ratio', 5.0), n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)

    return p / p.sum()
