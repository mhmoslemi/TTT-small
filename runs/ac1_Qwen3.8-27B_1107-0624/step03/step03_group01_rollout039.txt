def _next_pow2(n):

    while k < n:

    out_n = n + m - 1
    N = _next_pow2(out_n)

    return e / s

def _lbfgs_optimize(a_init, deadline, beta, n):
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    def objective(z_in):
        a_in = _softmax_fn(z_in)
        c = _full_convolve(a_in, a_in)
        return _smooth_max_value(c, beta) * 2.0 * n
    
    def gradient(z_in):
        a_in = _softmax_fn(z_in)
        c = _full_convolve(a_in, a_in)
        w = _smooth_max_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a_in[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        # dL/da_i = 4*n*a_i*(h_i - W)
        # dz_to_a: da/dz = a*(delta - a) for softmax
        # By chain rule: dL/dz_i = sum_j dL/da_j * da_j/dz_i
        # = dL/da_i * a_i + sum_{j!=i} dL/da_j * (-a_j * a_i)
        # = a_i * (dL/da_i - sum_j dL/da_j * a_j)
        g_a = 4.0 * n * a_in * (h - W)
        sum_ga_a = float(np.dot(g_a, a_in))
        g_z = a_in * (g_a - sum_ga_a)
        return g_z
    
    try:
        result = minimize(
            objective, z, jac=gradient,
            method='L-BFGS-B',
            options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxls': 50, 'maxfun': 5000}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            return a_new, _score_norm_fn(a_new)
    except Exception:
        pass
    return a_init, _score_norm_fn(a_init)

def _make_profile(n, shape='linear', param=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if shape == 'linear':
        a = 1.0 + param * u
    elif shape == 'quadratic':
        a = 1.0 + param * u * abs(u)
    elif shape == 'cubic':
        a = 1.0 + param * u ** 3
    elif shape == 'sinusoidal':
        a = 1.0 + param * np.sin(np.pi * t)
    elif shape == 'ramp':
        a = np.linspace(0.01, 2.0, n)
    elif shape == 'bimodal':
        a = 1.0 + param * np.cos(2 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: diverse initialization, Adam with adaptive logsumexp smoothing and noise,
polynomial manifold optimization, L-BFGS continuation, and fine polishing.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _smooth_max_and_weights(c, beta):
    cm = float(c.max())
    if beta < 1e-15:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return cm, w
    c_shifted = (c - cm) / beta
    c_max_sh = float(np.max(c_shifted))
    e = np.exp(c_shifted - c_max_sh)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    val = cm + beta * (c_max_sh + np.log(s))
    return float(val), w


def _grad_z(a, beta, n):
    """Gradient of 2n*smooth_max(conv(a,a)) w.r.t. z where a=softmax(z)."""
    c = _full_convolve(a, a)
    _, w = _smooth_max_and_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    g_a = 4.0 * n * a * (h - W)
    sum_ga = float(np.dot(g_a, a))
    g_z = a * (g_a - sum_ga)
    if not np.all(np.isfinite(g_z)):
        return np.zeros_like(a)
    return g_z


def _adam_optimize(a_init, deadline, rng, lr=0.02, beta_frac=0.05,
                   noise_interval=200, noise_amp=0.3, noise_decay=0.93,
                   n_max_iter=200000, lr_decay_interval=3000, lr_decay_factor=0.92):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    a = a_init.copy()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a, a)
    beta = max(1e-12, beta_frac * float(c.max()))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    cur_noise = noise_amp
    last_improve = 0

    while time.monotonic() < deadline:
        if t >= n_max_iter:
            break

        # Compute gradient
        g_z = _grad_z(a, beta, n)

        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * g_z
        v = b2 * v + (1 - b2) * (g_z * g_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)

        # Noise injection
        if t % noise_interval == 0 and cur_noise > 1e-6:
            z_new = z_new + cur_noise * rng.normal(size=n)
            cur_noise *= noise_decay

        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()
        a_new = _softmax_fn(z_new)

        z = z_new
        a = a_new

        # Evaluate (do every few steps to save time for large n)
        eval_interval = 1 if n <= 1024 else (2 if n <= 2048 else 3)
        if t % eval_interval == 0:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = t

        # Adaptive beta: decrease over time
        if t % 150 == 0:
            c_curr = _full_convolve(a, a)
            max_c = float(c_curr.max())
            progress = min(1.0, t / 20000.0)
            target_frac = beta_frac * (1.0 - progress) + 0.0005 * progress
            beta = max(1e-12, target_frac * max_c)

        # Learning rate decay
        if t % lr_decay_interval == 0:
            lr = max(0.00005, lr * lr_decay_factor)

        # Stall detection: if no improvement for a while, add a bigger kick
        if t - last_improve > 5000 and t % 200 == 0:
            z = z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a = _softmax_fn(z)

    return best_a, best_score


def _lbfgs_optimize(a_init, deadline, beta, n):
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    def objective(z_in):
        a_in = _softmax_fn(z_in)
        c = _full_convolve(a_in, a_in)
        val, _ = _smooth_max_and_weights(c, beta)
        return 2.0 * n * val

    def gradient(z_in):
        a_in = _softmax_fn(z_in)
        return _grad_z(a_in, beta, n)

    try:
        result = minimize(
            objective, z, jac=gradient,
            method='L-BFGS-B',
            options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxls': 50, 'maxfun': 10000}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc):
                return a_new, sc
    except Exception:
        pass
    return a_init, _score_norm_fn(a_init)


def _poly_optimize(n, degree, deadline, rng):
    """Optimize a polynomial-parameterized sequence."""
    u = 2.0 * np.linspace(0.0, 1.0, n) - 1.0
    # Build design matrix
    phi = np.ones((n, degree + 1))
    for d in range(1, degree + 1):
        phi[:, d] = u ** d

    def poly_objective(coeffs):
        z = phi @ coeffs
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Use smooth max with fixed beta
        beta_p = 0.01 * cm
        val, _ = _smooth_max_and_weights(c, beta_p)
        return 2.0 * n * val

    best_coeffs = None
    best_sc = np.inf

    # Try a few random initializations
    for trial in range(3):
        c0 = rng.normal(size=degree + 1) * 0.3
        try:
            res = minimize(poly_objective, c0, method='Nelder-Mead',
                           options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.fun < best_sc and np.isfinite(res.fun):
                best_sc = float(res.fun)
                best_coeffs = res.x.copy()
        except Exception:
            pass

    if best_coeffs is not None:
        z = phi @ best_coeffs
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        return a, sc
    return None, np.inf


def _make_profile(n, shape='linear', param=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if shape == 'linear':
        a = 1.0 + param * u
    elif shape == 'quadratic':
        a = 1.0 + param * u * np.abs(u)
    elif shape == 'cubic':
        a = 1.0 + param * u ** 3
    elif shape == 'sinusoidal':
        a = 1.0 + param * np.sin(np.pi * t)
    elif shape == 'bimodal':
        a = 1.0 + param * np.cos(2.0 * np.pi * t)
    elif shape == 'asym':
        a = 1.0 + param * (u + 0.3 * u ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining Adam, polynomial manifold search,
    L-BFGS continuation, and fine polishing to minimize the peak autocorrelation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")

    # === Phase 0: Diverse initialization (30s) ===
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    if parent is not None:
        candidates.append((_score_norm_fn(parent), parent.copy(), len(parent), "parent"))

    lengths = [512, 1024, 2048, 4096]
    shapes = [
        ('linear', 0.3), ('linear', 0.7), ('linear', 1.2), ('linear', 2.0),
        ('quadratic', 0.5), ('quadratic', 1.0),
        ('cubic', 0.5), ('cubic', 1.0),
        ('sinusoidal', 0.5), ('sinusoidal', 1.0),
        ('bimodal', 0.5), ('asym', 0.8),
    ]

    for sn in lengths:
        if time.monotonic() > phase0_end:
            break
        for shape, param in shapes:
            p = _make_profile(sn, shape, param)
            sc = _score_norm_fn(p)
            candidates.append((sc, p, sn, f"{shape}_{param}_n{sn}"))
            update_best(p, sc, f"{shape}_{param}_n{sn}")

    # Random initializations
    for sn in [512, 1024, 2048]:
        if time.monotonic() > phase0_end:
            break
        for trial in range(5):
            z = rng.normal(size=sn) * (0.3 + 0.1 * trial)
            p = _softmax_fn(z)
            sc = _score_norm_fn(p)
            candidates.append((sc, p, sn, f"rand{trial}_n{sn}"))
            update_best(p, sc, f"rand{trial}_n{sn}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 1: Adam on top candidates (450s) ===
    phase1_end = min(deadline, t_start + 480.0)
    top_k = min(5, len(candidates))
    seen = set()
    count = 0

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end - 60:
            break
        if count >= top_k:
            break
        if sc > best_score * 1.03:
            break
        key = pn
        if key in seen and count >= 3:
            continue
        seen.add(key)

        remaining = phase1_end - time.monotonic() - 20
        per_time = min(100.0, remaining / max(1, top_k - count))
        per_dl = time.monotonic() + per_time

        p_best, sc_best = _adam_optimize(
            p, per_dl, rng, lr=0.025, beta_frac=0.06,
            noise_interval=150, noise_amp=0.4, noise_decay=0.92,
            n_max_iter=100000, lr_decay_interval=2000, lr_decay_factor=0.93
        )
        update_best(p_best, sc_best, f"adam_{label}")
        count += 1

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Polynomial manifold optimization (200s) ===
    phase2_end = min(deadline, t_start + 680.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        n_best = len(best_seq)
        for deg in [4, 6, 8]:
            if time.monotonic() > phase2_end - 30:
                break
            p_poly, sc_poly = _poly_optimize(n_best, deg, phase2_end, rng)
            if p_poly is not None:
                update_best(p_poly, sc_poly, f"poly_deg{deg}_n{n_best}")

    # Also try polynomial on parent length if different
    if parent is not None and len(parent) != (len(best_seq) if best_seq is not None else 0):
        if time.monotonic() < phase2_end - 30:
            n_p = len(parent)
            for deg in [4, 6]:
                if time.monotonic() > phase2_end - 10:
                    break
                p_poly, sc_poly = _poly_optimize(n_p, deg, phase2_end, rng)
                if p_poly is not None:
                    update_best(p_poly, sc_poly, f"poly_deg{deg}_n{n_p}_parent")

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS continuation on best (350s) ===
    phase3_end = min(deadline, t_start + 1030.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        n_best = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        max_c_best = float(c_best.max())

        # Continuation: start with moderate beta, decrease
        beta_schedule = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        a_curr = best_seq.copy()

        for i, bf in enumerate(beta_schedule):
            remaining = phase3_end - time.monotonic() - 10
            if remaining < 10:
                break
            beta = bf * max_c_best
            per_dl = time.monotonic() + min(50.0, remaining / max(1, len(beta_schedule) - i))

            a_new, sc_new = _lbfgs_optimize(a_curr, per_dl, beta, n_best)
            if sc_new < best_score:
                update_best(a_new, sc_new, f"lbfgs_b{i}")
                a_curr = a_new
            elif a_new is not None and _score_norm_fn(a_new) < _score_norm_fn(a_curr):
                a_curr = a_new

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Adam fine polish (300s) ===
    phase4_end = min(deadline, t_start + 1330.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = time.monotonic() + min(280.0, phase4_end - time.monotonic() - 5)
        p_best, sc_best = _adam_optimize(
            best_seq, per_dl, rng, lr=0.008, beta_frac=0.008,
            noise_interval=500, noise_amp=0.1, noise_decay=0.95,
            n_max_iter=150000, lr_decay_interval=5000, lr_decay_factor=0.95
        )
        update_best(p_best, sc_best, "adam_fine")

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Try larger sequence (200s) ===
    phase5_end = min(deadline, t_start + 1530.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr < 4096:
            target_n = 4096
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            remaining = phase5_end - time.monotonic() - 10
            if remaining > 30:
                per_dl = time.monotonic() + min(150.0, remaining)
                p_best, sc_best = _adam_optimize(
                    p_interp, per_dl, rng, lr=0.015, beta_frac=0.04,
                    noise_interval=200, noise_amp=0.2, noise_decay=0.93,
                    n_max_iter=80000, lr_decay_interval=3000, lr_decay_factor=0.93
                )
                update_best(p_best, sc_best, f"adam_n{target_n}")

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final intensive polish (170s) ===
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        max_c_best = float(c_best.max())

        # Very fine L-BFGS
        remaining = phase6_end - time.monotonic() - 10
        if remaining > 20:
            beta = 0.0001 * max_c_best
            per_dl = time.monotonic() + min(60.0, remaining * 0.5)
            a_new, sc_new = _lbfgs_optimize(best_seq, per_dl, beta, n_best)
            if sc_new < best_score:
                update_best(a_new, sc_new, "final_lbfgs")
                best_seq = a_new

        # Final Adam with minimal noise
        if time.monotonic() < phase6_end - 10:
            per_dl2 = time.monotonic() + min(50.0, phase6_end - time.monotonic() - 5)
            p_best2, sc_best2 = _adam_optimize(
                best_seq, per_dl2, rng, lr=0.002, beta_frac=0.003,
                noise_interval=1000, noise_amp=0.03, noise_decay=0.9,
                n_max_iter=50000, lr_decay_interval=3000, lr_decay_factor=0.95
            )
            update_best(p_best2, sc_best2, "final_adam")

    print(f"Phase 6 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(256, 'linear', 1.0)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, total_time={time.monotonic()-t_start:.1f}s")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(512, 'linear', 1.0)
    print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
