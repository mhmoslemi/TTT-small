def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    """Compute full convolution of a with a using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-10:
        c = c - mn
    return c


def _score(a):
    """Compute the objective score."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project onto the probability simplex (sum=1, all >= 0)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weights(c, beta):
    """Compute softmax weights approximating the max."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s

    c = _fft_convolve(a)

def _lp_sparse(a, M, top_k=100):
    """
    LP: maximize sum(g) subject to conv(a,g)[k] <= M for top-k positions, g >= 0.
    This is much faster than using all 2n-1 constraints.
    """
    n = len(a)
    c = _fft_convolve(a)
    # Find top-k positions
    if top_k < 2 * n - 1:
        top_indices = np.argsort(c)[::-1][:top_k]
    else:
        top_indices = np.arange(2 * n - 1)
    
    # Build constraint matrix (sparse)
    # conv(a,g)[k] = sum_i a[i]*g[k-i] <= M
    # For each top constraint k, the row is: a[i] for valid i where k-i in [0,n-1]
    num_constraints = len(top_indices)
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    
    for ci, k in enumerate(top_indices):
        # conv(a,g)[k] = sum_{i: max(0,k-n+1) <= i <= min(k,n-1)} a[i]*g[k-i]
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i  # index into g
            rows.append(ci)
            cols.append(j)
            vals.append(a[i])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    
    c_obj = -np.ones(n)
    b = np.full(num_constraints, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_full_sparse(a, M, top_k=200):
    """LP with top-k constraints, returns normalized solution or None."""
    n = len(a)
    g = _lp_sparse(a, M, top_k)
    if g is None:

    s = float(g.sum())
    if s <= 0:
        return None
    return g / s

        for b_mult in [1.0, 2.0, 0.5]:
            b = max(1e-10, beta * b_mult)
            w = _smooth_max_weights(c, b)
            N = _next_pow2(3 * n - 2)
            fw = np.fft.rfft(w, N)
            fp_rev = np.fft.rfft(p[::-1], N)
            h_full = np.fft.irfft(fw * fp_rev, N)
            grad_p = 2.0 * h_full[n - 1:2 * n - 1]
            total_grad += grad_p
        total_grad /= num_betas
        
        # Small regularization to prevent collapse
        total_grad += 1e-8 * p
        
        # Softmax gradient: dz = p * (grad_p - dot(p, grad_p))
        dot = float(np.dot(p, total_grad))
        gz = p * (total_grad - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = np.log(best_a + 1e-30)
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = np.log(best_a + 1e-30)
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, lr * 0.97)
            if no_imp > max(3000, 4 * n):
                # Restart from best with perturbation
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                z = np.log(best_a + 1e-30) + 0.1 * rng.normal(size=n)
                z -= float(np.mean(z))
                z = np.clip(z, -60.0, 60.0)
                e = np.exp(z)
                a = e / e.sum()
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr = base_lr
                no_imp = 0
        
        if it % 200 == 0:
            beta = max(1e-12, beta * 0.85)
    
    return best_a


def _lp_refine_fast(a, deadline, top_k=150, inflate=1.002, blend=0.25, max_iter=5000):
    """LP-based refinement using only top-K constraints."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    
    no_imp = 0
    infl = inflate
    bl = blend
    
    for it in range(max_iter):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * infl
        
        g = _lp_full_sparse(a, M_lp, top_k=top_k)
        
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                infl = min(1.01, infl * 1.04)
            if no_imp > 300:
                break
            continue
        
        a_new = (1.0 - bl) * a + bl * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            infl = max(1.0005, infl * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                bl = max(0.05, bl * 0.93)
                infl = min(1.008, infl * 0.98)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                bl = blend
                infl = inflate
                no_imp = 0
        
        a = a_new
    
    return best_a


def _targeted_perturb(a, deadline, max_iter=20000):
    """
    Targeted moves: identify hot positions contributing to peaks,
    reduce them and redistribute to cold positions.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    rng = np.random.default_rng(999)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = _fft_convolve(a)
        # Find top peak position
        k_star = int(np.argmax(c))
        
        # Find hot positions: i where a[i]*a[k_star-i] is large
        i_range = range(max(0, k_star - n + 1), min(k_star, n - 1) + 1)
        if len(i_range) == 0:
            no_imp += 1
            continue
            break
        
        hot_vals = []
        hot_indices = []
        for i in i_range:
            j = k_star - i
            if 0 <= j < n:
                hot_vals.append(a[i] * a[j])
                hot_indices.append(i)
        
        hot_vals = np.array(hot_vals)
        hot_indices = np.array(hot_indices)
        
        if len(hot_indices) == 0:
            no_imp += 1
            continue
        
        # Pick a hot position to reduce
        # Weight probability by contribution
        total_hv = hot_vals.sum()
        if total_hv < 1e-20:
            no_imp += 1
            continue
        
        probs = hot_vals / total_hv
        idx = rng.choice(len(hot_indices), p=probs)
        i_hot = hot_indices[idx]
        
        # Find a cold position to increase
        # Prefer positions where a[j] is small AND a[k_star - j] is small
        cold_scores = a * 0 + 1.0  # base score
        for j in range(n):
            # If j is at a peak position, it's less good to increase
            penalty = 0.0
            c_peak = a[j]
            cold_scores[j] = -c_peak + 0.1 * rng.random()
        
        # Pick from positions with lowest a[j]
        cold_candidates = np.argsort(a)[:max(10, n // 20)]
        j_cold = rng.choice(cold_candidates)
        
        if j_cold == i_hot:
            continue
        
        # Try moving some mass
        best_local_score = best_score
        best_local_a = None
        for frac in [0.02, 0.05, 0.1, 0.2]:
            delta = frac * a[i_hot]
            if delta < 1e-15:
                continue
            a_try = a.copy()
            a_try[i_hot] -= delta
            a_try[j_cold] += delta
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_local_a = a_try.copy()
        
        if best_local_a is not None:
            a = best_local_a
            if best_local_score < best_score - 1e-12:
                best_score = best_local_score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        
        if no_imp > 3000:
            break
    
    return best_a


def _annealing_optimize(a, deadline, max_iter=50000, T0=0.1, T1=0.001, lr=0.03):
    """
    Simulated annealing on the simplex.
    Moves: reduce a hot position, increase a cold position.
    Accept worse moves with probability exp(-delta/T).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    rng = np.random.default_rng(77000)
    
    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    
    for it in range(max_iter):
        if time.monotonic() > deadline - 5.0:
            break
        
        T = T0 * (T1 / T0) ** (it / max_iter)
        
        c = _fft_convolve(a)
        k_star = int(np.argmax(c))
        
        # Pick hot and cold positions
        i_min = max(0, k_star - n + 1)
        i_max = min(k_star, n - 1)
        if i_min > i_max:
            no_imp = 1
            continue
        
        hot_range = list(range(i_min, i_max + 1))
        hot_weights = np.array([a[i] * a[k_star - i] for i in hot_range])
        tw = hot_weights.sum()
        if tw < 1e-20:
            continue
        
        probs = hot_weights / tw
        idx = rng.choice(len(hot_range), p=probs)
        i_hot = hot_range[idx]
        
        # Cold: pick from low-value positions
        low_mask = a < np.median(a)
        if not np.any(low_mask):
            low_mask = np.ones(n, dtype=bool)
        cold_candidates = np.where(low_mask)[0]
        j_cold = rng.choice(cold_candidates)
        
        if j_cold == i_hot:
            continue
        
        delta_move = 0.05 * a[i_hot] + 0.01
        a_try = a.copy()
        a_try[i_hot] = max(0, a_try[i_hot] - delta_move)
        a_try[j_cold] += delta_move
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        
        sc_try = _score(a_try)
        
        # Acceptance criterion
        if sc_try < cur_score or rng.random() < np.exp(-(sc_try - cur_score) / max(T, 1e-10)):
            a = a_try
            cur_score = sc_try
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
        else:
            # Reject
            pass
    
    return best_a


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_inverted_u(n, alpha=1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 - alpha * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, center_gap=0.3, width=0.15):
    """Bimodal: mass concentrated at two locations."""
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - center_gap
    c2 = 0.5 + center_gap
    a = np.exp(-0.5 * ((t - c1) / width) ** 2) + np.exp(-0.5 * ((t - c2) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quartic(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * u ** 4 - 0.5 * alpha * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _combined_optimize(a_init, deadline, n_adam_frac=0.4, n_lp_frac=0.3, n_target_frac=0.15, n_anneal_frac=0.15):
    """Run combined optimization pipeline on a single sequence."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score(a)
    remaining = deadline - time.monotonic()
    
    # Phase 1: Adam
    t1 = time.monotonic() + remaining * n_adam_frac
    if t1 > time.monotonic() + 10:
        a1 = _adam_optimize(a, t1, max_iter=200000, base_lr=0.04, beta0=0.03)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
        print(f"      Adam: {sc1:.10f}")
    
    # Phase 2: LP
    t2 = time.monotonic() + (deadline - time.monotonic()) * n_lp_frac
    if t2 > time.monotonic() + 10:
        a2 = _lp_refine_fast(best_a, t2, top_k=100, inflate=1.002, blend=0.25)
        sc2 = _score(a2)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
        print(f"      LP: {sc2:.10f}")
    
    # Phase 3: Targeted perturbation
    t3 = time.monotonic() + (deadline - time.monotonic()) * (n_target_frac / (n_target_frac + n_anneal_frac + 0.05))
    if t3 > time.monotonic() + 5:
        a3 = _targeted_perturb(best_a, t3, max_iter=30000)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
        print(f"      Targeted: {sc3:.10f}")
    
    # Phase 4: Annealing
    t4 = deadline - 10.0
    if t4 > time.monotonic() + 5:
        a4 = _annealing_optimize(best_a, t4, max_iter=50000, T0=0.05, T1=0.0005, lr=0.02)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
        print(f"      Anneal: {sc4:.10f}")
    
    # Phase 5: Final Adam polish
    t5 = deadline - 5.0
    if t5 > time.monotonic() + 10:
        a5 = _adam_optimize(best_a, t5, max_iter=100000, base_lr=0.02, beta0=0.01)
        sc5 = _score(a5)
        if sc5 < best_score:
            best_score = sc5
            best_a = a5.copy()
        print(f"      Final Adam: {sc5:.10f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses Adam with multi-temperature smooth max, sparse LP, targeted
    perturbation, and simulated annealing.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**32 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Candidate lengths to try
    candidate_lengths = []
    if parent is not None:
        candidate_lengths.append(parent.size)
    for L in [512, 768, 1024, 1280, 1536, 2048]:
        if L not in candidate_lengths and L >= 256:
            candidate_lengths.append(L)
    
    # Remove duplicates, sort by priority (parent first)
    candidate_lengths = list(dict.fromkeys(candidate_lengths))
    
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")
    print(f"Lengths to try: {candidate_lengths}")
    
    # Phase 1: Process each length
    num_lengths = len(candidate_lengths)
    # Allocate time: give more time to promising lengths
    total_opt_time = deadline - time.monotonic() - 300.0  # reserve 300s for phase 2+3
    
    for li, n_t in enumerate(candidate_lengths):
        if time.monotonic() > deadline - 250:
            break
        
        # Time allocation: earlier lengths get more time
        frac_remaining = max(0.3, 1.0 - 0.7 * li / max(1, num_lengths))
        per_time = max(20.0, total_opt_time * frac_remaining / max(1, num_lengths - li))
        per_time = min(per_time, deadline - time.monotonic() - 200.0)
        
        if per_time < 15:
            continue
        
        print(f"\n=== Length {n_t} (time={per_time:.0f}s) ===")
        
        # Create initializations for this length
        inits = []
        
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
            # Mirror parent
            inits.append(_interpolate_profile(parent[::-1], n_t))
        
        # Standard profiles
        inits.append(_make_u_profile(n_t, 2.0))
        inits.append(_make_u_profile(n_t, 3.0))
        inits.append(_make_inverted_u(n_t, 1.0))
        inits.append(_make_bimodal(n_t, 0.25, 0.15))
        inits.append(_make_linear(n_t, 2.0))
        inits.append(_make_quartic(n_t, 1.0))
        
        # Random structured
        a_rand = np.maximum(rng.normal(1.0, 0.4, n_t), 0.0)
        inits.append(a_rand / a_rand.sum())
        
        if best_seq is not None and best_seq.size != n_t:
            inits.append(_interpolate_profile(best_seq, n_t))
        
        # Remove duplicates (approximately)
        unique_inits = []
        for ini in inits:
            dup = False
            for u in unique_inits:
                if np.allclose(ini, u, atol=1e-6):
                    dup = True
                    break
            if not dup:
                unique_inits.append(ini)
        inits = unique_inits
        
        # Time per initialization
        n_inits = len(inits)
        time_per_init = per_time / n_inits
        time_per_init = max(10.0, time_per_init)
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline - 200:
                break
            
            a0 = _project_simplex(a0)
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            
            if time.monotonic() > deadline - 150:
                break
            
            init_deadline = time.monotonic() + time_per_init
            init_deadline = min(init_deadline, deadline - 150.0)
            
            if init_deadline - time.monotonic() < 10:
                continue
            
            a_res, sc_res = _combined_optimize(a0, init_deadline)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"    ** New best at n={n_t}: {sc_res:.10f}")
    
    # Phase 2: Intensive optimization on best found
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = best_seq.size
        remaining = deadline - time.monotonic() - 30.0
        print(f"\n=== Phase 2: Intensive on n={n_best}, time={remaining:.0f}s ===")
        
        # Sub-phase 2a: Long Adam
        t_a = time.monotonic() + remaining * 0.35
        a_a = _adam_optimize(best_seq, t_a, max_iter=500000, base_lr=0.03, beta0=0.02)
        sc_a = _score(a_a)
        if sc_a < best_score:
            best_score = sc_a
            best_seq = a_a.copy()
        print(f"  Long Adam: {sc_a:.10f}")
        
        # Sub-phase 2b: LP with varying top_k
        for tk in [50, 100, 200]:
            if time.monotonic() > deadline - 40:
                break
            t_lp = time.monotonic() + min(60.0, (deadline - time.monotonic() - 30.0) * 0.2)
            a_lp = _lp_refine_fast(best_seq, t_lp, top_k=tk, inflate=1.001, blend=0.2)
            sc_lp = _score(a_lp)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  LP (k={tk}): {sc_lp:.10f}")
        
        # Sub-phase 2c: Targeted
        if time.monotonic() < deadline - 30:
            t_t = deadline - 30.0
            a_t = _targeted_perturb(best_seq, t_t, max_iter=50000)
            sc_t = _score(a_t)
            if sc_t < best_score:
                best_score = sc_t
                best_seq = a_t.copy()
            print(f"  Targeted: {sc_t:.10f}")
        
        # Sub-phase 2d: Final Adam
        if time.monotonic() < deadline - 15:
            t_f = deadline - 10.0
            a_f = _adam_optimize(best_seq, t_f, max_iter=200000, base_lr=0.015, beta0=0.005)
            sc_f = _score(a_f)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  Final Adam: {sc_f:.10f}")
    
    # Phase 3: Try a few more diverse approaches if time remains
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = best_seq.size
        remaining = deadline - time.monotonic() - 30.0
        print(f"\n=== Phase 3: Diversity, time={remaining:.0f}s ===")
        
        # Try: best + random perturbation
        for _ in range(3):
            if time.monotonic() > deadline - 30:
                break
            a_pert = best_seq * (1.0 + 0.2 * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_sub = min(60.0, (deadline - time.monotonic() - 25.0) * 0.3)
            if t_sub < 10:
                break
            a_res, sc_res = _combined_optimize(a_pert, time.monotonic() + t_sub)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** Perturbed best: {sc_res:.10f}")
        
        # Try: interpolation with random profile
        a_rand = np.maximum(rng.exponential(1.0, n_best), 0.0)
        a_mixed = 0.5 * best_seq + 0.5 * a_rand / a_rand.sum()
        a_mixed = a_mixed / a_mixed.sum()
        if time.monotonic() < deadline - 30:
            t_sub = min(45.0, deadline - time.monotonic() - 25.0)
            a_res, sc_res = _combined_optimize(a_mixed, time.monotonic() + t_sub)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** Mixed best: {sc_res:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = _make_u_profile(768, 2.0)
        best_score = _score(best_seq)
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_u_profile(768, 2.0)
        s = float(final_seq.sum())
    
    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")
    
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
