    # A[k, j] = f_{k-j} if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        idx = range(start, end)
        A[idx, j] = f[:end - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _next_pow2(n):
    k = 1
    while k < n:

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):

def _build_lp_matrix(n):
    """Build the convolution constraint matrix A where A[k,j] = a_{k-j}."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        A[start:end, j] = np.arange(end - start)
    return A


def _lp_solve(f, M, n, A=None):
    """Solve: max sum(b) s.t. A @ (f-dependent) <= M, b >= 0."""
    # conv(f,b)_k = sum_j f_{k-j} * b_j
    # A[k,j] = f_{k-j} if 0 <= k-j < n else 0
    if A is None:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            start = j
            end = min(2 * n - 1, j + n)
            idx = slice(start, end)
            A[idx, j] = f[:end - j]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

        ps = (parent + parent[::-1]) / 2.0
        ps = ps / ps.sum()
        configs.append(('parent_sym', parent.size, ps))

    for n_t in [512, 640, 768, 896, 1024]:
        configs.append((f'ramp1_{n_t}', n_t, _make_profile(n_t, 'ramp', 1.0)))
        configs.append((f'ramp2_{n_t}', n_t, _make_profile(n_t, 'ramp', 1.5)))
        configs.append((f'lin1_{n_t}', n_t, _make_profile(n_t, 'lin', 1.0)))
        configs.append((f'lin3_{n_t}', n_t, _make_profile(n_t, 'lin', 3.0)))
        configs.append((f'bell2_{n_t}', n_t, _make_profile(n_t, 'bell', 2.0)))
        configs.append((f'cos_{n_t}', n_t, _make_profile(n_t, 'cos')))

    # Score and sort
    scored_configs = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        if np.isfinite(sc):
            scored_configs.append((sc, name, n_t, a0))
    scored_configs.sort(key=lambda x: x[0])

    print(f"\nTop 8 initializations:")
    for sc, name, n_t, a0 in scored_configs[:8]:
        print(f"  {name}: score={sc:.10f}, n={n_t}")

    # Optimize top candidates
    max_candidates = 10
    for i, (sc0, name, n_t, a0) in enumerate(scored_configs[:max_candidates]):
        if time.monotonic() > deadline - 90:
            break

        if sc0 > best_score + 0.02 and i > 2:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        remaining = deadline - time.monotonic() - 60
        if remaining < 30:
            break

        # Allocate time: more for better initializations
        if i == 0:
            opt_time = min(400.0, remaining * 0.45)
        elif i <= 2:
            opt_time = min(250.0, remaining * 0.35)
        elif i <= 5:
            opt_time = min(150.0, remaining * 0.25)
        else:
            opt_time = min(80.0, remaining * 0.15)

        if opt_time < 20:
            break

        opt_end = time.monotonic() + opt_time
        print(f"\n[{i}] Optimizing {name} (n={n_t}, init={sc0:.8f}), "
              f"budget={opt_time:.0f}s, remaining_after={deadline - opt_end - t_start:.0f}s")

        a_opt, sc_opt = _combined_optimize(a0, opt_end)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  => IMPROVED: score={sc_opt:.10f}")
        else:
            print(f"  => No improvement (best={best_score:.10f})")

    # Phase: Try random perturbations of the best to escape local optima
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\nRandom perturbation phase: current best={best_score:.10f}")
        for attempt in range(5):
            if time.monotonic() > deadline - 40:
                break
            eps = 0.01 * (attempt + 1)
            a_pert = _random_perturb(best_seq, eps, rng)
            # Quick gradient optimization
            pert_time = min(60.0, deadline - time.monotonic() - 30)
            if pert_time < 15:
                break
            a_p, _ = _combined_optimize(a_pert, time.monotonic() + pert_time)
            sc_p = _score(a_p)
            if np.isfinite(sc_p) and sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                print(f"  Perturb {attempt} (eps={eps}): score={sc_p:.10f} IMPROVED")
            else:
                print(f"  Perturb {attempt} (eps={eps}): score={sc_p:.8f}")

    # Final tight LP pass on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 10:
        final_time = min(80.0, deadline - time.monotonic() - 5)
        if final_time > 10:
            print(f"\nFinal tight LP: budget={final_time:.0f}s")
            a_final = _lp_frank_wolfe(best_seq, time.monotonic() + final_time,
                                       inflate_start=1.0005, inflate_min=1.0002)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LP score={sc_final:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_profile(512, 'ramp', 1.0)
        best_score = _score(best_seq)
        print(f"Fallback used: score={best_score:.10f}")

    # Final validation and cleanup
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_profile(512, 'ramp', 1.0)

    # Verify with evaluate_sequence if available
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print("evaluate_sequence returned inf, using fallback")
                final_seq = _make_profile(512, 'ramp', 1.0)
        except Exception as e:
            print(f"Eval check error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
