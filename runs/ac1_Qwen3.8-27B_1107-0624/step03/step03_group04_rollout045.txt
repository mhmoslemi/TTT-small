        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _grad_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    val, w = _smooth_max(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)[n - 1:2 * n - 1]
    grad = 2.0 * h
    return val, grad, float(c.max())


def _make_conv_matrix(a, subset_indices=None):
    n = len(a)
    if subset_indices is not None:
        A = np.zeros((len(subset_indices), n))
        for row, k in enumerate(subset_indices):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                A[row, i] = a[j]
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
    return A


def _lp_spread_sparse(a, M, n_top=200, n_rand=100):
    n = len(a)
    c = _fft_convolve(a)
    n_total = 2 * n - 1
    top_idx = np.argsort(c)[::-1][:min(n_top, n_total)]
    remaining = np.setdiff1d(np.arange(n_total), top_idx)
    if len(remaining) > 0 and n_rand > 0:
        rand_idx = np.random.default_rng(42).choice(remaining, size=min(n_rand, len(remaining)), replace=False)
        all_idx = np.concatenate([top_idx, rand_idx])
    else:
        all_idx = top_idx
    A = _make_conv_matrix(a, subset_indices=all_idx)
    b = np.full(len(all_idx), M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_spread_full(a, M):
    n = len(a)
    c = _fft_convolve(a)
    # Only use constraints near the peak (top 20%)
    threshold = np.percentile(c, 80)
    active = np.where(c >= threshold)[0]
    A = _make_conv_matrix(a, subset_indices=active)
    b = np.full(len(active), M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_symmetric_bell(n, width=0.4, edge_boost=0.5):
    t = np.linspace(0.0, 1.0, n)
    dist = np.abs(t - 0.5)
    a = np.exp(-(dist / width) ** 2) + edge_boost * np.exp(-dist ** 2 / 0.02)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _logit_adam_optimize(a, deadline, max_iter=50000, lr=0.05, beta_init=None):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    if beta_init is not None:
        beta = beta_init
    else:
        beta = max(1e-6, 0.05 * float(c.max()))

    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    best_z = z.copy()

    no_imp = 0
    base_lr = lr
    prev_z = z.copy()
    prev_grad = None
    bb_scale = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp_rev, N)[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            continue

        # BB step size
        if prev_grad is not None and np.all(np.isfinite(prev_grad)):
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20:
                bb_candidate = dot_ss / max(abs(dot_sg), 1e-20)
                bb_candidate = np.clip(bb_candidate, 0.1, 10.0)
                bb_scale = 0.7 * bb_scale + 0.3 * bb_candidate
        prev_z = z.copy()
        prev_grad = gz.copy()

        cur_lr = base_lr * bb_scale

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        # Line search every 50 iterations
        if it % 50 == 0:
            best_lr = cur_lr
            cur_score = float(2.0 * n * _fft_convolve(a).max())
            lr_try = cur_lr
            for _ls in range(8):
                z_new = z - lr_try * mh / (np.sqrt(vh) + eps)
                z_new = np.clip(z_new, -60.0, 60.0)
                z_new -= float(np.mean(z_new))
                e = np.exp(z_new)
                a_try = e / e.sum()
                sc_try = float(2.0 * n * _fft_convolve(a_try).max())
                if sc_try < cur_score:
                    best_lr = lr_try
                    break
                lr_try *= 0.5
            cur_lr = best_lr

        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
            base_lr = min(0.12, base_lr * 1.01)
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                base_lr = max(0.003, base_lr * 0.92)

        # Adaptive beta
        if it % 100 == 0 and it > 100:
            top5 = np.sort(c)[::-1][:min(5, len(c))]
            if len(top5) > 1:
                peak_gap = top5[0] - top5[-1]
                if peak_gap < beta * 0.4:
                    beta = max(1e-10, beta * 0.6)
                elif peak_gap > beta * 6.0:
                    beta = min(0.4 * c_max_new, beta * 1.3)

        # Annealing restart
        if no_imp > max(4000, 6 * n):
            rng_r = np.random.default_rng(it * 7919 + 1)
            noise_scale = 0.25 * best_score
            z_restart = best_z + noise_scale * rng_r.standard_normal(n)
            z_restart = np.clip(z_restart, -60.0, 60.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            bb_scale = 1.0
            prev_grad = None
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
            no_imp = 0

        z = z_new

    return best_a


def _gradient_polish(a, deadline, max_iter=40000, lr=0.008):
    """Fine gradient polish with small beta for final refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.005 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= np.mean(z)
    no_imp = 0
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        _, w = _smooth_max(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp_rev, N)[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr_cur * gz
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr_cur = max(0.0005, lr_cur * 0.93)

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.75)

        if no_imp > max(2000, 4 * n):
            break

    return best_a


def _lp_refine(a, deadline, max_iter=5000, inflate=1.0005, blend=0.2):
    """LP-based spreading refinement with sparse constraints."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        # Use sparse LP for speed
        if n <= 512:
            g = _lp_spread_full(a, M_lp)
        else:
            g = _lp_spread_sparse(a, M_lp, n_top=150, n_rand=80)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 5 == 0:
                inflate_cur = min(1.003, inflate_cur * 1.03)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.93)
                inflate_cur = min(1.001, inflate_cur * 0.97)
            if no_imp > 150:
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _peak_shave(a, deadline, max_iter=3000, n_peaks=5, strength=0.04):
    """Targeted peak shaving with pressure-based coordinate reduction."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    str_cur = strength

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_idx]
        med = np.median(c)
        weights = np.maximum(top_vals - med, 0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_peaks) / n_peaks
        else:
            weights = weights / wsum

        pressure = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * a[j]

        p_max = pressure.max() + 1e-15
        a_new = a * np.exp(-str_cur * pressure / p_max)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            str_cur = min(0.15, str_cur * 1.008)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                str_cur = max(0.005, str_cur * 0.9)
            if no_imp > 500:
                a = best_a.copy()
                str_cur = strength
                no_imp = 0

        a = a_new

    return best_a


def _coordinate_swap_optimize(a, deadline, n_trials=20000):
    """Try pairwise coordinate swaps to reduce peak."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    rng = np.random.default_rng(12345)

    # Pre-compute which pairs are "good" to try
    # Focus on high-value coordinates (they contribute more to peaks)
    sorted_idx = np.argsort(a)[::-1]
    top_half = sorted_idx[:n // 2]

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Pick a random pair, biased toward high-value coordinates
        if rng.random() < 0.7:
            i = top_half[rng.integers(len(top_half))]
            j = sorted_idx[rng.integers(n)]
        else:
            i = rng.integers(n)
            j = rng.integers(n)
        if i == j:
            continue

        # Try moving a small amount from i to j (or vice versa)
        for direction in [1, -1]:
            for frac in [0.01, 0.03, 0.05]:
                delta = direction * frac * a[i]
                if a[i] + delta < 1e-15 or a[j] - delta < 1e-15:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                a_try = a_try / a_try.sum()

                c_try = _fft_convolve(a_try)
                sc_try = float(2.0 * n * c_try.max())
                if sc_try < best_score - 1e-14:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a


def _spectral_flat_init(n, rng, n_modes=0):
    """Initialize with controlled spectral flatness."""
    N = _next_pow2(n)
    if n_modes == 0:
        n_modes = N // 2 + 1
    else:
        n_modes = min(n_modes, N // 2 + 1)

    # Create spectrum with controlled magnitude profile
    mags = np.ones(N // 2 + 1)
    # Slightly taper the high frequencies for better time-domain behavior
    t = np.linspace(0, 1, N // 2 + 1)
    mags = mags * (1.0 - 0.3 * t ** 2)

    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = mags[0]
    for k in range(1, N // 2 + 1):
        X[k] = mags[k] * np.exp(1j * phases[k])
        X[N - k] = mags[k] * np.exp(-1j * phases[k])

    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _multi_start_refine(a, deadline, n_starts=6, rng=None):
    """Try multiple perturbation patterns and refine the best."""
    if rng is None:
        rng = np.random.default_rng(99999)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    # Different perturbation patterns
    patterns = [
        lambda arr, r: arr * (1.0 + 0.03 * r.standard_normal(len(arr))),
        lambda arr, r: arr * (1.0 + 0.05 * r.standard_normal(len(arr))),
        lambda arr, r: np.maximum(arr + 0.02 * r.standard_normal(len(arr)), 1e-10),
        lambda arr, r: arr ** (1.0 + 0.1 * r.standard_normal()),
    ]

    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        pat = patterns[i % len(patterns)]
        a_pert = pat(a, rng)
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        if sc > best_score * 1.3:
            continue

        sub_deadline = min(deadline, time.monotonic() + 20.0)
        a_ref = _gradient_polish(a_pert, sub_deadline, max_iter=3000, lr=0.005)
        sc_ref = _score(a_ref)
        if sc_ref < best_score:
            best_score = sc_ref
            best_a = a_ref.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Phases: diverse init -> logit-Adam -> LP refine -> peak shave ->
    coordinate swap -> fine polish -> multi-start -> final LP.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Target length
    target_n = 768
    if parent is not None:
        target_n = parent.size

    # Phase 1: Generate diverse initializations
    print(f"Phase 1: Initialization, remaining={deadline - time.monotonic():.0f}s", flush=True)
    init_pool = []

    if parent is not None:
        init_pool.append(('parent', _interpolate_profile(parent, target_n)))

    init_pool.append(('linear1.0', _make_linear_profile(target_n, 1.0)))
    init_pool.append(('linear1.5', _make_linear_profile(target_n, 1.5)))
    init_pool.append(('linear2.0', _make_linear_profile(target_n, 2.0)))
    init_pool.append(('bell', _make_bell_profile(target_n, 0.3)))
    init_pool.append(('sym_bell', _make_symmetric_bell(target_n, 0.4, 0.3)))
    init_pool.append(('flat_spec', _make_flat_spectrum_init(target_n, rng)))
    init_pool.append(('spectral_flat', _spectral_flat_init(target_n, rng)))

    # Quick evaluation
    init_scores = []
    for name, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, a0))
    init_scores.sort(key=lambda x: x[0])

    print("  Top 5 inits:", flush=True)
    for sc, name, _ in init_scores[:5]:
        print(f"    {name}: {sc:.8f}", flush=True)

    # Phase 2: Main optimization on best candidates
    top_inits = init_scores[:3]
    for sc0, name, a0 in top_inits:
        if time.monotonic() > deadline - 400:
            break

        opt_deadline = min(deadline - 250, time.monotonic() + 350.0)
        print(f"Phase 2: Logit-Adam on {name} (n={target_n}), start={sc0:.8f}, budget={opt_deadline - time.monotonic():.0f}s", flush=True)
        a_opt = _logit_adam_optimize(a0, opt_deadline, max_iter=50000, lr=0.05)
        sc_opt = _score(a_opt)
        print(f"  Logit-Adam: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # LP refinement
        if time.monotonic() < deadline - 150:
            lp_deadline = min(deadline - 120, time.monotonic() + 150.0)
            print(f"  LP refine, budget={lp_deadline - time.monotonic():.0f}s", flush=True)
            a_lp = _lp_refine(a_opt, lp_deadline, max_iter=3000, inflate=1.0005, blend=0.2)
            sc_lp = _score(a_lp)
            print(f"  LP: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            a_opt = a_lp

        # Peak shaving
        if time.monotonic() < deadline - 80:
            ps_deadline = min(deadline - 60, time.monotonic() + 40.0)
            a_ps = _peak_shave(a_opt, ps_deadline, max_iter=3000, n_peaks=5, strength=0.03)
            sc_ps = _score(a_ps)
            print(f"  Peak shave: {sc_ps:.10f}", flush=True)
            if sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
            a_opt = a_ps

    # Phase 3: Fine gradient polish
    if best_seq is not None and time.monotonic() < deadline - 60:
        gp_deadline = min(deadline - 40, time.monotonic() + 80.0)
        print(f"Phase 3: Fine polish, budget={gp_deadline - time.monotonic():.0f}s", flush=True)
        a_fine = _gradient_polish(best_seq, gp_deadline, max_iter=40000, lr=0.005)
        sc_fine = _score(a_fine)
        print(f"  Fine polish: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()

    # Phase 4: Coordinate swap optimization
    if best_seq is not None and time.monotonic() < deadline - 40:
        cs_deadline = min(deadline - 25, time.monotonic() + 30.0)
        print(f"Phase 4: Coordinate swap, budget={cs_deadline - time.monotonic():.0f}s", flush=True)
        a_cs = _coordinate_swap_optimize(best_seq, cs_deadline, n_trials=15000)
        sc_cs = _score(a_cs)
        print(f"  Coord swap: {sc_cs:.10f}", flush=True)
        if sc_cs < best_score:
            best_score = sc_cs
            best_seq = a_cs.copy()

    # Phase 5: Multi-start local refinement
    if best_seq is not None and time.monotonic() < deadline - 25:
        ms_deadline = deadline - 12
        print(f"Phase 5: Multi-start, budget={ms_deadline - time.monotonic():.0f}s", flush=True)
        a_ms = _multi_start_refine(best_seq, ms_deadline, n_starts=5, rng=rng)
        sc_ms = _score(a_ms)
        print(f"  Multi-start: {sc_ms:.10f}", flush=True)
        if sc_ms < best_score:
            best_score = sc_ms
            best_seq = a_ms.copy()

    # Phase 6: Final LP pass
    if best_seq is not None and time.monotonic() < deadline - 5:
        if deadline - time.monotonic() > 10:
            a_last = _lp_refine(best_seq, deadline - 3, max_iter=2000, inflate=1.0003, blend=0.1)
            sc_last = _score(a_last)
            if sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LP: {sc_last:.10f}", flush=True)

    # Phase 7: One more gradient polish if time remains
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_final = _gradient_polish(best_seq, deadline - 1, max_iter=10000, lr=0.003)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final polish: {sc_final:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(target_n, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(target_n, 1.5)

    # Validate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
