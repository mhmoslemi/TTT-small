        key = (pn, label[:10])
        if key in seen:
            continue
        seen.add(key)

        remaining = phase1_end - time.monotonic()
        per_time = min(80.0, remaining / max(1, 5 - count))
        per_dl = min(phase1_end, time.monotonic() + per_time)
        p_best, sc_best = _run_lbfgs_continuation(p, per_dl, max_wall_time=per_time)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"Phase 1 done. Best={best_score:.10f}")

    # --- Phase 2: Nesterov PGD on best ---
    print(f"=== Phase 2: Nesterov PGD ===")
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        n_best = len(best_seq)
        # Try multiple tolerance values
        for tol in [0.001, 0.01, 0.05]:
            if time.monotonic() > phase2_end - 10:
                break
            per_dl = min(phase2_end - 30, time.monotonic() + 120.0)
            p_best, sc_best = _nesterov_pgd(best_seq, per_dl, lr_init=0.003 * (n_best / 1024), tol_frac=tol)
            update_best(p_best, sc_best, f"nesterov_tol{tol}")

    print(f"Phase 2 done. Best={best_score:.10f}")

    # --- Phase 3: Adam polish ---
    print(f"=== Phase 3: Adam Polish ===")
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 20, time.monotonic() + 200.0)
        p_best, sc_best = _adam_polish(best_seq, per_dl, lr=0.02, beta_frac=0.005)
        update_best(p_best, sc_best, "adam_polish_1")

    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 10, time.monotonic() + 150.0)
        p_best, sc_best = _adam_polish(best_seq, per_dl, lr=0.005, beta_frac=0.001)
        update_best(p_best, sc_best, "adam_polish_2")

    print(f"Phase 3 done. Best={best_score:.10f}")

    # --- Phase 4: Try interpolation to larger size ---
    print(f"=== Phase 4: Size Extension ===")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_curr = len(best_seq)
        for target_n in [2048, 4096]:
            if target_n <= n_curr:
                continue
            if time.monotonic() > phase4_end - 30:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase4_end - 60:
                per_dl = min(phase4_end - 30, time.monotonic() + 120.0)
                p_best, sc_best = _run_lbfgs_continuation(p_interp, per_dl, max_wall_time=120.0)
                update_best(p_best, sc_best, f"lbfgs_interp_n{target_n}")

    print(f"Phase 4 done. Best={best_score:.10f}")

    # --- Phase 5: Final combined polish ---
    print(f"=== Phase 5: Final Polish ===")
    phase5_end = deadline
    if time.monotonic() < phase5_end and best_seq is not None:
        # L-BFGS with very fine beta
        per_dl = min(phase5_end - 30, time.monotonic() + 80.0)
        fine_betas = [0.001, 0.0005, 0.0002, 0.0001]
        p_best, sc_best = _run_lbfgs_continuation(best_seq, per_dl, beta_starts=fine_betas, max_wall_time=80.0)
        update_best(p_best, sc_best, "final_lbfgs_fine")

        if time.monotonic() < phase5_end - 20:
            per_dl2 = min(phase5_end - 15, time.monotonic() + 50.0)
            p_best2, sc_best2 = _adam_polish(best_seq, per_dl2, lr=0.002, beta_frac=0.0005)
            update_best(p_best2, sc_best2, "final_adam")

        if time.monotonic() < phase5_end - 10:
            per_dl3 = min(phase5_end - 10, time.monotonic() + 30.0)
            p_best3, sc_best3 = _nesterov_pgd(best_seq, per_dl3, lr_init=0.001, tol_frac=0.001)
            update_best(p_best3, sc_best3, "final_nesterov")

    print(f"Phase 5 done. Best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(1024, 1.0)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(1024, 1.0)
    print(f"FALLBACK: n=1024, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
