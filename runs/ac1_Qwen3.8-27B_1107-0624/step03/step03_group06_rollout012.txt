"""
LP-based optimization with adaptive constraint sets, FFT convolution,
backtracking line search, gradient polish, and multi-scale exploration
for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements: reduced constraint LP, better initializations,
adaptive step sizes, symmetry breaking, and iterative refinement.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_ramp_rev(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** alpha + 1e-8
    return a / a.sum()


def _make_hann(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 1e-8
    return a / a.sum()


def _make_blackman(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.42 + 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_bell(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 1e-8
    return a / a.sum()


def _make_exponential(n, rate=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t) + 1e-8
    return a / a.sum()


def _make_exp_rev(n, rate=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * (1.0 - t)) + 1e-8
    return a / a.sum()


def _make_bimodal(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-20 * t * t) + np.exp(-20 * (t - 1) ** 2) + 1e-8
    return a / a.sum()


def _interpolate_seq(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _solve_lp_reduced(f, M, n, top_k=None):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M, g >= 0.
    Uses only top_k largest potential constraint rows for speed."""
    if top_k is None:
        top_k = min(2 * n - 1, max(n, 512))
    
    # Build constraint matrix A where A[k,j] = f[k-j] if valid
    # Only include top_k rows (those where f has largest values contributing)
    # For the convolution, row k has f[0..n-1] shifted
    # The constraint is: sum_j f[k-j] * g[j] <= M (where valid)
    
    # For speed, build only a subset of constraints
    # Focus on constraints near where the max of conv(f,f) is
    c_ff = _fft_convolve(f)
    peak_idx = int(np.argmax(c_ff))
    
    # Select constraints: peak and neighbors
    num_constraints = top_k
    idx = np.arange(2 * n - 1)
    # Distance from peak
    dist = np.abs(idx - peak_idx)
    # Select closest constraints
    selected = idx[np.argsort(dist)[:num_constraints]]
    
    # Build A matrix (num_constraints x n)
    A = np.zeros((num_constraints, n))
    for i, k in enumerate(selected):
        # A[i,j] = f[k-j] if 0 <= k-j < n
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            for j in range(j_start, j_end + 1):
                A[i, j] = f[k - j]
    
    c_obj = -np.ones(n)
    b = np.full(num_constraints, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None


def _solve_lp_full(f, M, n):
    """Solve the full LP with all 2n-1 constraints."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def _backtrack_line_search(a, g, alpha0=1.0, beta=0.5, max_iter=20):
    """Backtracking line search for a_new = (1-t)*a + t*g."""
    s = a.sum()
    c = _fft_convolve(a)
    f0 = float(c.max())
    
    best_t = 0.0
    best_a = a.copy()
    best_f = f0
    
    t = alpha0
    for _ in range(max_iter):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        st = a_try.sum()
        if st < 1e-10:
            t *= beta
            continue
        a_try = a_try / st
        c_try = _fft_convolve(a_try)
        f_try = float(c_try.max())
        
        if f_try < best_f - 1e-14:
            best_f = f_try
            best_a = a_try.copy()
            best_t = t
            return best_t, best_a, best_f
        t *= beta
        if t < 1e-8:
            break
    
    return best_t, best_a, best_f

def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _perturb(a, rng, strength=0.1):

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:

    return a_new / s

    return a_asym / s

            lp_time = min(100.0, remaining * 0.15)
        
        if lp_time < 5:
            break
        
        lp_end = time.monotonic() + lp_time
        print(f"\nPhase 1 LP: {name} n={a0.size} init={sc0:.8f} budget={lp_time:.0f}s")
        
        a_lp = _lp_refine(a0, lp_end, max_iter=50000)
        sc_lp = _score(a_lp)
        
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> LP score={sc_lp:.10f} (NEW BEST)")
        else:
            print(f"  -> LP score={sc_lp:.8f}")
    
    # Phase 2: Gradient polish on current best
    if best_seq is not None and time.monotonic() < deadline - 80:
        gp_time = min(120.0, deadline - time.monotonic() - 50)
        if gp_time > 10:
            gp_end = time.monotonic() + gp_time
            print(f"\nPhase 2 Gradient: budget={gp_time:.0f}s")
            a_gp = _gradient_polish(best_seq, gp_end, max_iter=50000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
                print(f"  -> Gradient score={sc_gp:.10f} (NEW BEST)")
            else:
                print(f"  -> Gradient score={sc_gp:.8f}")
    
    # Phase 3: LP again on best (tighter)
    if best_seq is not None and time.monotonic() < deadline - 50:
        lp_time = min(100.0, deadline - time.monotonic() - 30)
        if lp_time > 5:
            lp_end = time.monotonic() + lp_time
            print(f"\nPhase 3 LP (tight): budget={lp_time:.0f}s")
            a_final = _lp_refine(best_seq, lp_end, max_iter=30000)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  -> Final LP score={sc_final:.10f} (NEW BEST)")
            else:
                print(f"  -> Final LP score={sc_final:.8f}")
    
    # Phase 4: Try asymmetric perturbation + re-optimize
    if best_seq is not None and time.monotonic() < deadline - 40:
        for t_blend in [0.15, 0.25, 0.35]:
            if time.monotonic() > deadline - 20:
                break
            a_pert = _asymmetric_blend(best_seq, t_blend)
            # Quick LP refine
            pe = time.monotonic() + min(40.0, deadline - time.monotonic() - 15)
            a_r = _lp_refine(a_pert, pe, max_iter=20000)
            sc_r = _score(a_r)
            if np.isfinite(sc_r) and sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  -> Asym blend t={t_blend}: score={sc_r:.10f} (NEW BEST)")
            else:
                print(f"  -> Asym blend t={t_blend}: score={sc_r:.8f}")
    
    # Phase 5: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        gp_time = min(50.0, deadline - time.monotonic() - 5)
        if gp_time > 5:
            gp_end = time.monotonic() + gp_time
            a_g = _gradient_polish(best_seq, gp_end, max_iter=20000)
            sc_g = _score(a_g)
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
                print(f"\nPhase 5 Final gradient: score={sc_g:.10f} (NEW BEST)")
    
    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)
    
    # Final validation and normalization
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)
        s = final_seq.sum()
        final_seq = final_seq / s
    
    # Evaluate
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")
    
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
