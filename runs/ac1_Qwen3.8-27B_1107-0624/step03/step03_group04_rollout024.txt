import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 192:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def _lp_min_peak(a):
    """Find minimum M such that exists b>=0, sum(b)=1, a*b <= M."""
    n = len(a)
    # minimize M subject to a*b <= M (all k), sum(b)=1, b>=0
    # Variables: b_0,...,b_{n-1}, M (total n+1 vars)
    # Objective: minimize M -> c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A = _make_conv_matrix(a)  # (2n-1) x n
    # a*b - M <= 0 for each k
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return float(res.x[-1])
    return None


def _direct_max_grad(a, top_k=10):
    """Compute gradient of max(conv(a,a)) using top-k peaks."""
    n = len(a)
    c = _fft_convolve(a)
    c_max = float(c.max())
    top_idx = np.argsort(c)[::-1][:top_k]
    # Compute gradient: d(max_k c_k)/d(a_j) = sum over active k of (a_{k-j} + a_j) / n_active
    # But since we use top_k as a soft approximation, weight by proximity to max
    vals = c[top_idx]
    # Exponential weighting
    weights = np.exp(-10.0 * (vals - vals.min()) / (vals.max() - vals.min() + 1e-15))
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        for j in range(n):
            k_minus_j = k - j
            if 0 <= k_minus_j < n:
                grad[j] += w * (a[k_minus_j] + a[j])
    return c_max, grad, top_idx


def _smooth_max_grad(a, beta):
    """Compute smooth max value and gradient."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = np.clip((c - mval) / beta, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros(n)
    w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    val = mval + beta * float(np.log(s))
    return val, grad

        if it < 200 or no_imp < 200:
            # Use direct max gradient for sharper optimization
            c_max, grad_p, top_idx = _direct_max_grad(a, top_k=use_top_k)
            # Project to log-space gradient
            dot = float(np.dot(a, grad_p))
            gz = a * (grad_p - dot)
            gz -= float(np.mean(gz))
        else:
            val, grad_p = _smooth_max_grad(a, beta)
            dot = float(np.dot(a, grad_p))
            gz = a * (grad_p - dot)
            gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_arr = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            no_imp = 0
            continue

        adam_t += 1
        m_arr = b1 * m_arr + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m_arr / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.93)
                beta = max(1e-6, beta * 0.8)

        if no_imp > max(3000, 5 * n):
            # Anti-symmetric restart
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            d = rng.standard_normal(n)
            d = (d - d[::-1]) / 2.0  # make anti-symmetric
            z_restart = best_z + 0.15 * d
            z_restart = np.clip(z_restart, -50.0, 50.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m_arr = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            base_lr = lr
            no_imp = 0
            beta = 0.03 * float(_fft_convolve(a).max())

        z = z_new

    return best_a


def _lp_refine_intensive(a, deadline, max_rounds=500):
    """Intensive LP refinement with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    inflate = 1.0005
    blend = 0.2

    for it in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _lp_spread(a, M_lp)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(1.003, inflate * 1.03)
                blend = min(0.4, blend * 1.02)
            if no_imp > max(200, n):
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score = float(2.0 * n * M_new)

        if score < best_score - 1e-14:
            best_score = score
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            # Slowly tighten
            inflate = max(1.0001, inflate * 0.995)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend = max(0.05, blend * 0.95)
                inflate = max(1.0001, inflate * 0.99)
            if no_imp > max(100, n // 2):
                a = best_a.copy()
                no_imp = 0
                inflate = 1.0005
                blend = 0.2

        a = a_new

    return best_a


def _hybrid_optimize(a0, deadline):
    """
    Alternating gradient + LP optimization.
    Gradient runs for short bursts, LP runs for longer periods.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    # Track time allocation
    t_start = time.monotonic()

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        # Determine phase based on remaining time
        frac = remaining / (deadline - t_start) if (deadline - t_start) > 0 else 1.0

        # Phase: alternate between gradient (short) and LP (long)
        # Gradient gets ~30% of time, LP gets ~70%
        if frac > 0.7:
            # Early: more gradient to establish shape
            grad_time = min(40.0, remaining * 0.4)
            lp_time = remaining * 0.6
        elif frac > 0.3:
            # Middle: balanced
            grad_time = min(30.0, remaining * 0.25)
            lp_time = remaining * 0.75
        else:
            # Late: mostly LP
            grad_time = min(15.0, remaining * 0.15)
            lp_time = remaining * 0.85

        # Gradient step
        if grad_time > 3:
            grad_deadline = time.monotonic() + grad_time
            a_grad = _logit_adam_core(a, grad_deadline, max_iter=10000, lr=0.03)
            sc_grad = _score(a_grad)
            if sc_grad < best_score:
                best_score = sc_grad
                best_a = a_grad.copy()
            a = a_grad

        # LP step
        if time.monotonic() < deadline - 3:
            lp_deadline = min(deadline - 2, time.monotonic() + lp_time)
            if lp_deadline - time.monotonic() > 5:
                a_lp = _lp_refine_intensive(a, lp_deadline, max_rounds=300)
                sc_lp = _score(a_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_a = a_lp.copy()
                a = a_lp

    return best_a


def _anti_symmetric_perturb(a, strength=0.1, rng=None):
    """Perturb with anti-symmetric noise (exploits KKT structure)."""
    if rng is None:
        rng = np.random.default_rng()
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    d = rng.standard_normal(n)
    # Make anti-symmetric about center
    d = (d - d[::-1]) / 2.0
    a_pert = a * (1.0 + strength * d / (np.std(d) + 1e-15))
    a_pert = np.maximum(a_pert, 1e-15)
    a_pert = a_pert / a_pert.sum()
    return a_pert


def _try_length_interpolate(parent, n_new):
    """Interpolate parent to new length."""
    p0 = np.asarray(parent, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0.copy() / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    LP-dominant hybrid optimization exploiting anti-symmetric KKT structure.
    Primary length: 768 (matching parent). Secondary: 640, 896.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"[Init] Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception as e:
        print(f"[Init] No parent: {e}", flush=True)

    # Determine primary lengths
    if parent is not None:
        primary_n = parent.size
        lengths = [primary_n, 640, 896]
        if 768 not in lengths and primary_n != 768:
            lengths.append(768)
    else:
        primary_n = 768
        lengths = [768, 640]

    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for ln in lengths:
        if ln not in seen and ln >= 256:
            unique_lengths.append(ln)
            seen.add(ln)
    lengths = unique_lengths

    # Phase 1: Quick initialization and scoring (30s)
    t_phase1_end = t_start + 30.0
    print(f"[Phase 1] Initialization, budget={t_phase1_end - t_start:.0f}s", flush=True)

    candidates = []
    for n_t in lengths:
        if parent is not None:
            a_cand = _try_length_interpolate(parent, n_t)
            sc = _score(a_cand)
            candidates.append((sc, n_t, a_cand, f'parent_{n_t}'))

        a_lin = _make_linear_profile(n_t, 1.5)
        sc = _score(a_lin)
        candidates.append((sc, n_t, a_lin, f'linear_{n_t}'))

        a_flat = _make_flat_spectrum_init(n_t, rng)
        sc = _score(a_flat)
        candidates.append((sc, n_t, a_flat, f'flat_{n_t}'))

    candidates.sort(key=lambda x: x[0])
    print("[Phase 1] Top candidates:", flush=True)
    for sc, n_t, _, name in candidates[:5]:
        print(f"  {name}: {sc:.10f}", flush=True)

    # Phase 2: Main optimization on best candidates
    # Allocate time based on remaining budget
    t_phase2_end = deadline - 60.0
    remaining = t_phase2_end - time.monotonic()

    # Focus on top 3 candidates, weight toward primary length
    top_candidates = []
    for sc, n_t, a_cand, name in candidates[:6]:
        if n_t == primary_n or len(top_candidates) < 3:
            top_candidates.append((sc, n_t, a_cand, name))
        if len(top_candidates) >= 3:
            break

    # Time allocation: 50% to first (likely parent), 30% to second, 20% to third
    time_fracs = [0.5, 0.3, 0.2][:len(top_candidates)]

    for i, (sc0, n_t, a0, name) in enumerate(top_candidates):
        if time.monotonic() > t_phase2_end:
            break

        alloc = remaining * time_fracs[i]
        sub_deadline = time.monotonic() + alloc - 5.0

        print(f"[Phase 2.{i+1}] {name} (n={n_t}), start={sc0:.10f}, time={alloc:.0f}s", flush=True)

        # Hybrid optimization (gradient + LP alternating)
        a_opt = _hybrid_optimize(a0, sub_deadline)
        sc_opt = _score(a_opt)
        print(f"  Hybrid result: {sc_opt:.10f}", flush=True)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc_opt:.10f}", flush=True)

        # Additional LP pass if time remains
        if time.monotonic() < deadline - 40:
            extra_time = min(60.0, deadline - 40 - time.monotonic())
            if extra_time > 10:
                a_lp2 = _lp_refine_intensive(a_opt, time.monotonic() + extra_time, max_rounds=200)
                sc_lp2 = _score(a_lp2)
                if sc_lp2 < best_score:
                    best_score = sc_lp2
                    best_seq = a_lp2.copy()
                    print(f"  Extra LP: {sc_lp2:.10f} (NEW BEST)", flush=True)
                a_opt = a_lp2

        # Anti-symmetric perturb + quick refine
        if time.monotonic() < deadline - 30:
            a_pert = _anti_symmetric_perturb(a_opt, strength=0.05, rng=rng)
            quick_deadline = min(deadline - 25, time.monotonic() + 20.0)
            a_pert_opt = _hybrid_optimize(a_pert, quick_deadline)
            sc_pert = _score(a_pert_opt)
            if sc_pert < best_score:
                best_score = sc_pert
                best_seq = a_pert_opt.copy()
                print(f"  Perturb refine: {sc_pert:.10f} (NEW BEST)", flush=True)

    # Phase 3: Final intensive LP on the best (last 60s)
    if best_seq is not None:
        final_deadline = deadline
        remaining_final = final_deadline - time.monotonic()
        print(f"[Phase 3] Final LP polish, time={remaining_final:.0f}s", flush=True)

        if remaining_final > 10:
            a_final = _lp_refine_intensive(best_seq, final_deadline, max_rounds=500)
            sc_final = _score(a_final)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LP: {sc_final:.10f}", flush=True)

    # Phase 4: One more gradient + LP cycle if time allows
    if best_seq is not None and time.monotonic() < deadline - 5:
        quick_dl = deadline - 2
        a_q = _logit_adam_core(best_seq, quick_dl, max_iter=5000, lr=0.01)
        sc_q = _score(a_q)
        if sc_q < best_score:
            best_score = sc_q
            best_seq = a_q.copy()
        # Quick LP
        if time.monotonic() < deadline - 2:
            a_q2 = _lp_refine_intensive(best_seq, deadline, max_rounds=100)
            sc_q2 = _score(a_q2)
            if sc_q2 < best_score:
                best_score = sc_q2
                best_seq = a_q2.copy()
        print(f"[Phase 4] Final: {max(best_score, _score(best_seq)):.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(768, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(768, 1.5)

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"[Eval] evaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"[Eval] Error: {e}", flush=True)

    elapsed = time.monotonic() - t_start
    print(f"[DONE] score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s", flush=True)
    return [float(x) for x in final_seq]
