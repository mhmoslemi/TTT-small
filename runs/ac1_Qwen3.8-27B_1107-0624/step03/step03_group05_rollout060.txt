"""
Optimized step function search using alternating LP direction-finding
with entropic (KL) gradient descent and adaptive smooth-max temperature.
Key improvements over previous: KL projection for simplex constraint,
targeted peak elimination, multi-scale optimization, and aggressive
time management.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _kl_project(v, target_sum=1.0):
    """Project onto simplex using KL/natural gradient approach.
    Actually, for the simplex {x >= 0, sum(x) = 1}, the Euclidean projection
    is what we typically use. The KL projection (Bregman) onto the simplex
    isn't standard. Let me just use Euclidean projection but with a 
    'soft' version for the gradient step.
    """
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.ones(n) * target_sum / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - target_sum) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) * target_sum / n
    return w * target_sum / s

def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of convolution."""
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x - x.max())  # numerical stability
    w = ex / ex.sum()
    # grad[i] = 2 * sum_k w[k] * a[k-i] (with a[j]=0 outside [0,n-1])
    # This is 2 * (w convolved with reversed a), centered
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # We need h[i] = sum_k w[k] * a[k-i]
    # In FFT: this is the cross-correlation of w and a
    # h = ifft(conj(FW) * FA) ... let me think
    # Actually, (w * a)[k] = sum_i w[i] * a[k-i]
    # We want h[i] = sum_k w[k] * a[k-i] = (w_rev * a)[n-1+i] ... 
    # Let me use a different approach
    # h[i] = sum_k w[k] * a[k-i] (a[j]=0 for j not in [0,n-1])
    # = sum_j w[i+j] * a[j] (substituting k=i+j)
    # This is the cross-correlation of w and a, evaluated at lag i
    # cross_corr[i] = sum_j w[i+j] * a[j]
    # In FFT: cross_corr = ifft(conj(FA) * FW) ... 
    # Actually, let me just use the convolution directly
    # h = np.convolve(w, a[::-1]) gives h[i] = sum_j w[i-j] * a_rev[j] = sum_j w[i-j] * a[n-1-j]
    # That's not quite right either.
    
    # Let me think again. We want:
    # c[k] = sum_i a[i] * a[k-i]
    # d c[k] / d a[j] = a[k-j]
    # grad[j] = sum_k w[k] * d c[k] / d a[j] = sum_k w[k] * a[k-j]
    
    # So grad[j] = sum_k w[k] * a[k-j]
    # Let me define b[m] = a[m] for m in [0,n-1], 0 otherwise
    # grad[j] = sum_{k=0}^{2n-2} w[k] * b[k-j]
    # = sum_{m=-j}^{2n-2-j} w[m+j] * b[m]
    # = sum_{m=max(0,-j)}^{min(n-1, 2n-2-j)} w[m+j] * a[m]
    
    # This is the cross-correlation of w (padded) and a, at lag j
    # cross_corr_w_a[j] = sum_m w[m+j] * a[m]
    
    # Using FFT: cross_corr = ifft(FFT(w_padded) * conj(FFT(a_padded)))
    # where w_padded is w zero-padded to length >= 2n-1
    
    w_pad = np.zeros(N)
    w_pad[:len(w)] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    cross = np.fft.irfft(np.fft.rfft(w_pad) * np.conj(np.fft.rfft(a_pad)), N)
    # cross[j] = sum_m w_pad[m] * conj(a_pad[(m-j) mod N]) ... 
    # Actually numpy's irfft of product gives circular convolution
    # conv[i] = sum_j w_pad[j] * a_pad[(i-j) mod N]
    # We want cross[j] = sum_m w_pad[m+j] * a_pad[m] = sum_m w_pad[m'] * a_pad[m'-j] where m'=m+j
    # = conv_circ[-j mod N] (circular cross-correlation)
    # Let me verify: conv[i] = sum_j w[j] * a[i-j]
    # cross[j] = sum_m w[m+j] * a[m] = sum_m' w[m'] * a[m'-j] where m'=m+j
    # So cross[j] = conv[-j] (in circular sense) = conv[N-j] for j>0, conv[0] for j=0
    
    # Actually let me just compute it directly for small n or use the relation:
    # grad = 2 * cross where cross[j] = sum_k w[k] * a[k-j]
    # This equals 2 * (reverse of convolve(a, w)) [n-1:2n-1]
    # Because convolve(a, w)[k] = sum_i a[i] * w[k-i]
    # And we want grad[j] = sum_k w[k] * a[k-j] = sum_i a[i+j] * w[i] (sub i=k-j)
    # = convolve(a_shifted, w) where a_shifted[i] = a[i+j]... this is getting circular.
    
    # Let me just use the direct formula:
    # grad[j] = sum_{k=j}^{j+2n-2} w[k] * a[k-j] (with bounds)
    # = sum_{m=0}^{n-1} w[j+m] * a[m]
    
    # This is: (w cross-correlated with a) at lag j
    # = sum_m w[j+m] * a[m]
    
    # Using the full convolution: np.convolve(w, a[::-1]) gives
    # result[k] = sum_i w[i] * a_rev[k-i] = sum_i w[i] * a[n-1-k+i]
    # result[n-1+j] = sum_i w[i] * a[n-1-(n-1-j)+i] = sum_i w[i] * a[j+i]
    # = sum_i w[i] * a[j+i] ... that's what we want! (with m=i)
    
    # So grad[j] = convolve(w, a[::-1])[n-1+j] for j=0,...,n-1
    # i.e., grad = convolve(w, a[::-1])[n-1:2n-1]
    
    # For efficiency with FFT:
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n-1:2*n-1]
    
    return grad, c, mval

def _lp_direction(a, M, method='highs'):
    """Find direction g that maximizes sum(g) subject to convolve(a,g) <= M, g >= 0."""
    n = len(a)
    # Build the constraint matrix: for each lag k, sum_i A[k,i] * g[i] <= M
    # A[k,i] = a[k-i] if 0 <= k-i <= n-1, else 0
    # This is a (2n-1) x n matrix
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # a contributes to lags i to i+n-1
        A[i:i+n, i] = a
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method=method)
    if res.success:
        return res.x
    return None

def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _logit_adam_optimize(a, deadline, max_iter=50000, lr=0.03):
    """Optimize using Adam on logit parameterization with smooth max objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = _score(a)
    beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient
        grad, c, mval = _smooth_max_grad(a, beta)
        
        if not np.all(np.isfinite(grad)):
            beta *= 2.0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        # Convert to logit gradient
        # d score / dz[i] = a[i] * (grad[i] - sum(a * grad))
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))  # center for numerical stability
        
        # Adam update
        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1**t)
        vh = v / (1.0 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        a = np.exp(z)
        a = a / a.sum()
        
        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)
        
        # Adapt beta
        if it % 200 == 0:
            beta = max(1e-10, beta * 0.8)
        
        # Restart if stuck
        if no_imp > max(3000, 4 * n):
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.5 * np.log(best_a) + 0.5 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            a = np.exp(z)

peak inflation, sparse coordinate Averaging for diversification, and

import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of full convolution."""
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = x - x.max()
    ex = np.exp(x)
    w = ex / ex.sum()
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n - 1:2 * n - 1]
    return grad, c, mval


def _lp_direction(a, M, top_k_inflate=0.0):
    """LP: maximize sum(g) s.t. convolve(a,g) <= M (with optional targeted inflation)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a

    b = np.full(2 * n - 1, M)
    if top_k_inflate > 0:
        c = _fft_convolve(a)
        sorted_idx = np.argsort(c)[::-1][:int(top_k_inflate)]
        for idx in sorted_idx:
            b[idx] = c[idx] * 1.0005

    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

        a2b, sc2b = _sparse_coord_optimize(best_seq.copy(), hard_deadline - 60,
                                          sparsity=0.2, lr=0.015)
        if sc2b < best_score:
            best_score = sc2b
            best_seq = a2b.copy()
            print(f"  Sparse best: {sc2b:.10f}")
        else:
            print(f"  Sparse: no improvement ({sc2b:.10f})")

    # --- Phase 3: Try a different length ---
    if time.monotonic() < hard_deadline - 120:
        n_alt = 1024 if n_parent < 1024 else 512
        print(f"\nPhase 3: Trying n={n_alt}")
        a_alt = _interpolate_profile(best_seq, n_alt)
        p3_deadline = t_start + (hard_deadline - t_start) * 0.82
        a3, sc3 = _hybrid_optimize(a_alt, p3_deadline)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  Alt length best: {sc3:.10f}")
        else:
            print(f"  Alt length: no improvement ({sc3:.10f})")

    # --- Phase 4: Final intensive polish ---
    if time.monotonic() < hard_deadline - 20:
        print(f"\nPhase 4: Final polish on n={best_seq.size}")
        final_deadline = hard_deadline

        # Short hybrid run
        a4, sc4 = _hybrid_optimize(best_seq.copy(), final_deadline)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  Final hybrid: {sc4:.10f}")
        else:
            print(f"  Final hybrid: {sc4:.10f}")

    # --- Final validation ---
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(512, 1.5)
        s = float(final_seq.sum())
    final_seq = final_seq / s

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
