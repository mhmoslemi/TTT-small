import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 200:
        c = np.convolve(a, a)
        return c
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # Algorithm from "Euclidean projection on the simplex"
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a) using softmax weighting."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # Gradient of sum_k w_k * (a*a)_k with respect to a
    # (a*a)_k = sum_j a[j] * a[k-j]
    # d/d a[i] of (a*a)_k = a[k-i] + (a*a)[k] derivative... 
    # Actually: d/d a[i] sum_j a[j]a[k-j] = a[k-i] + a[k-i] (if i appears twice)
    # More carefully: (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d a[i]: a[i] appears when j=i (gives a[k-i]) and when k-j=i i.e. j=k-i (gives a[i])
    # So d/d a[i] (a*a)[k] = a[k-i] + a[i] if i != k-i, or 2*a[i] if i = k-i
    # Wait, that's not right. Let me be more careful.
    # (a*a)[k] = sum_j a[j] * a[k-j]
    # d/d a[i] = a[k-i] + a[k-i] ... no.
    # If i is in the range of j: term is a[i] * a[k-i], derivative w.r.t. a[i] is a[k-i].
    # Also if k-i is in the range of j (i.e., j = k-i): term is a[k-i] * a[i], derivative w.r.t. a[i] is a[k-i].
    # So d/d a[i] (a*a)[k] = 2 * a[k-i] (when both i and k-i are valid indices)
    # But we need to handle boundaries.
    
    # The gradient is: grad[i] = 2 * sum_k w_k * a[k-i] (where k-i is valid)
    # This is 2 * (w * a_rev)[something]... let me think in FFT terms.
    # Actually, the gradient of (a*a)[k] w.r.t. a is: 2 * (w convolved with reversed a)
    # Let me just use the FFT approach.
    
    # grad[i] = 2 * sum_k w_k * (d/d a[i]) (a*a)[k]
    # (a*a)[k] = sum_j a[j] a[k-j]
    # The subgradient w.r.t. a[i] is: 2 * sum_k w_k * a[k-i] (for valid k-i)
    
    # This is equivalent to: grad = 2 * (w * a_rev) where * is correlation
    # In FFT terms: if W = FFT(w), A_rev = FFT(a[::-1]), then grad = 2 * IFFT(W * A_rev)[n-1:2n-1]
    
    N_pad = _next_pow2(2 * n - 1)
    fw = np.zeros(N_pad // 2 + 1, dtype=np.complex128)
    # w has length 2n-1
    fw_full = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw_full * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval, c


def _make_profiles(n):

def _interpolate(a, n_new):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_solve_constraints(a, M, constraint_indices):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for k in constraint_indices, b >= 0."""
    n = len(a)
    if len(constraint_indices) == 0:
        return None
    
    # Build constraint matrix A where A[k_idx, j] = a[j] if constraint k is active at position j
    # conv(a,b)[k] = sum_j a[j]*b[k-j] (for valid j)
    # = sum_j b[j]*a[k-j] (for valid j)
    # So the constraint is: sum_j A[k, j] * b[j] <= M
    # where A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    n_constraints = len(constraint_indices)
    A = np.zeros((n_constraints, n))
    for idx, k in enumerate(constraint_indices):
        # For constraint at position k: conv(a,b)[k] = sum_j a[j]*b[k-j]
        # = sum_j b[j] * a[k-j]
        # b[j] is valid for j in [0, n-1]
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[idx, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
            # A[idx, j] = a[k-j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(n_constraints, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None


def _fast_lp_refine(a0, deadline, max_iter=10000, top_k=50):
    """LP refinement with constraint generation (only top-k constraints)."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    
    no_imp = 0
    inflate = 1.005
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        
        # Select top-k constraints (those with highest conv values)
        thresh = np.percentile(c, max(0, 100 - 100.0/top_k * 2))
        constraint_indices = np.where(c >= thresh)[0]
        # Always include the argmax
        constraint_indices = np.unique(np.append(constraint_indices, np.argmax(c)))
        
        M_lp = M * inflate
        g = _lp_solve_constraints(a, M_lp, constraint_indices)
        if g is None:
            break

def _adam_gradient(a0, deadline, max_iter=50000):

            a = a_new
            no_imp = 0

            a = a_new  # still move (Adam momentum helps)
