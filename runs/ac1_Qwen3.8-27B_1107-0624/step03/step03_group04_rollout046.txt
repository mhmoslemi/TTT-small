                    return np.inf

                        else:
                            affected_vals.append(c[k])

            # Golden section

"""
Optimized step function height sequence search.
Uses projected gradient with momentum, coordinate descent, and multi-length search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2*n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2*n-1]
    mn = c.min()
    if mn < 0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    """Project v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _smooth_max_and_grad(a, beta):
    """Smooth max of conv(a,a) and its gradient."""
    n = len(a)
    c = _fft_conv(a)
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return m, np.zeros(n)
    w = ex / s
    val = m + beta * float(np.log(s))
    
    # Gradient: d/d a_i [val] = 2 * sum_k w_k * a_{k-i} (for valid indices)
    # This is 2 * cross-correlation of w and a (reversed)
    N = _next_pow2(2*n - 1)
    fw = rfft(w, N)
    fa_rev = rfft(a[::-1], N)
    h = irfft(fw * fa_rev, N)[n-1:2*n-1]
    grad = 2.0 * h
    return val, grad

def _momentum_gradient(a, deadline, max_steps=100000):
    """Project gradient with heavy ball momentum and adaptive step size."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    c = _fft_conv(a)
    c_max = float(c.max())
    beta = 0.02 * c_max / n  # initial beta
    
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    lr = 0.1
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    no_imp = 0
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        val, grad = _smooth_max_and_grad(a, beta)
        
        # Project gradient onto tangent space of simplex
        # grad_perp = grad - mean(grad) (for the constraint sum(a)=1)
        grad_perp = grad - float(np.mean(grad))
        
        v = mu * v + (1 - mu) * grad_perp
        a_new = a - lr * v
        a_new = _project_simplex(a_new)
        
        c_new = _fft_conv(a_new)
        sc = float(2.0 * n * c_new.max())
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr *= 0.8
                v *= 0.5
                beta *= 0.7
            if no_imp > 20000:
                break
        
        # Adapt beta
        if step % 500 == 0 and step > 500:
            top3 = np.sort(c_new)[::-1][:3]
            if len(top3) > 1 and top3[0] - top3[-1] < beta:
                beta *= 0.8
            elif top3[0] - top3[-1] > 10 * beta:
                beta = min(beta * 1.3, 0.1 * c_max)
        
        a = a_new
    
    return best_a, best_score

            # First, let's estimate a good upper bound
            # At t = 0: affected entries decrease (since t - ai < 0)
            # Actually at t = 0: affected = c[k] - ai * a[k-i] for k != 2i
            # For k = 2i: affected = c[2i] - ai^2 + 0 = c[2i] - ai^2
            # These could be negative! So t=0 might give negative affected values.
            # The max would then be c_unaff.
            # f(0) = 2n * c_unaff / sa^2
            # f(ai) = current score (by construction)
            # We want to find t that minimizes.
            
            # Simple approach: try log-spaced values
            # t in [0, ai*0.5, ai, ai*1.5, ai*2, ...]
            # Then refine with golden section
            
            # Let's just do golden section with 40 iterations
            for _ in range(40):
                c_t = hi_t - gr * (hi_t - lo_t)
                d_t = lo_t + gr * (hi_t - lo_t)
                
                # Evaluate at c_t
                delta_c = c_t - ai
                new_sum_c = sa - ai + c_t
                if new_sum_c <= 1e-15:
                    f_c = np.inf
                else:
                    aff_c = c_aff + delta_c * a_aff
                    if has_2i:
                        idx_2i = k_2i - lo
                        aff_c[idx_2i] = c[k_2i] - ai*ai + c_t*c_t
                    aff_max_c = aff_c.max()
                    total_max_c = max(c_unaff, aff_max_c)
                    f_c = 2.0 * n * total_max_c / (new_sum_c * new_sum_c)
                
                # Evaluate at d_t
                delta_d = d_t - ai
                new_sum_d = sa - ai + d_t
                if new_sum_d <= 1e-15:
                    f_d = np.inf
                else:
                    aff_d = c_aff + delta_d * a_aff
                    if has_2i:
                        idx_2i = k_2i - lo
                        aff_d[idx_2i] = c[k_2i] - ai*ai + d_t*d_t
                    aff_max_d = aff_d.max()
                    total_max_d = max(c_unaff, aff_max_d)
                    f_d = 2.0 * n * total_max_d / (new_sum_d * new_sum_d)
                
                if f_c < f_d:
                    hi_t = d_t
                else:
                    lo_t = c_t
            
            t_opt = (lo_t + hi_t) / 2.0
            if t_opt < 1e-10:
                t_opt = 1e-10
            
            # Update
            a[i] = t_opt
            a = a / a.sum()
            
            new_score = float(2.0 * n * _fft_conv(a).max())
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                improved = True
                n_improved += 1
        
        if n_improved == 0:
            break
        elif n_improved < n // 10:
            # Convergence slowing
            if cycle > 10 and n_improved < n // 50:
                break
    
    return best_a, best_score

def _make_hanning(n):
    """Hanning window (raised cosine)."""
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * t / (n - 1)))
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_parabolic(n):
    """Parabolic (quadratic) window."""
    t = np.linspace(0, 1, n)
    a = 1.0 - (2*t - 1)**2
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_gaussian(n, sigma=0.3):
    """Gaussian window."""
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma)**2)
    return a / a.sum()

def _make_linear(n, slope=1.0):
    """Linear ramp."""
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * (2*t - 1)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_bimodal(n, gap=0.15):
    """Two peaks."""
    t = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-((t - (center - gap))/0.1)**2) + np.exp(-((t - (center + gap))/0.1)**2)
    return a / a.sum()

def _symmetrize(a):
    """Enforce a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return a_sym / a_sym.sum()
