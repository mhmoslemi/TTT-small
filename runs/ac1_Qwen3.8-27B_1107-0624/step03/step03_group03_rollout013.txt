"""
Multi-length optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LSE gradient descent, LP refinement, and a new informed local search
method. Explores lengths 1024, 2048, and 4096 with adaptive time allocation.

import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score for sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _make_conv_matrix_reduced(a, M, top_k=60):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    b = np.full(len(active), M)
    return A, b


def _lp_spread(a, M, use_reduced=False, top_k=60):
    n = len(a)
    c_obj = -np.ones(n)
    if use_reduced and n > 256:
        A, b = _make_conv_matrix_reduced(a, M, top_k)
    else:
        A = _make_conv_matrix(a)
        b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, max_iters=25):
    n = len(a)
    best_score = _score_norm(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c_pt * (g - a), 0))
    fd = _score_norm(np.maximum(a + d_pt * (g - a), 0))
    for _ in range(max_iters):
        if not np.isfinite(fc):
            fc = best_score
        if not np.isfinite(fd):
            fd = best_score
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c_pt * (g - a), 0))
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d_pt * (g - a), 0))
    return (lo + hi) / 2.0

def _lse_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()
    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val

def _informed_local_search(a, deadline, max_iter=500000, T_start=0.0005, T_end=1e-7, seed=42):
    """Informed local search: guided moves that reduce the peak of the convolution."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    c = _fft_convolve(a)
    current_max = float(c.max())
    best_a = a.copy()
    best_max = current_max
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    T = T_start
    no_imp = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        T = max(T * T_ratio, T_end)
        c = _fft_convolve(a)
        max_val = float(c.max())
        max_lags = np.where(c >= max_val - 1e-10 * max_val)[0]
        top_lags = max_lags[:min(5, len(max_lags))]
        # Compute responsibility for each position
        resp = np.zeros(n)
        for k in top_lags:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    resp[i] += a[j]
        resp_max = resp.max()
        if resp_max < 1e-15:
            break
        resp_norm = resp / resp_max
        # Choose source: high responsibility, weighted by value
        source_weights = resp_norm * a
        source_weights = np.maximum(source_weights, 1e-15)
        source_weights = source_weights / source_weights.sum()
        i = rng.choice(n, p=source_weights)
        # Choose destination: low responsibility
        dest_weights = 1.0 - resp_norm
        dest_weights = np.maximum(dest_weights, 1e-5)
        dest_weights[i] = 0.0
        if dest_weights.sum() < 1e-10:
            dest_weights = np.ones(n)
            dest_weights[i] = 0.0
        dest_weights = dest_weights / dest_weights.sum()
        j = rng.choice(n, p=dest_weights)
        if j == i:
            continue
        # Transfer amount
        frac = rng.uniform(0.01, 0.08)
        eps = frac * a[i]
        if eps < 1e-12:
            continue
        a_trial = a.copy()
        a_trial[i] -= eps
        a_trial[j] += eps
        if a_trial[i] < 0 or a_trial[j] < 0:
            continue
        c_trial = _fft_convolve(a_trial)
        new_max = float(c_trial.max())
        delta = new_max - max_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_trial
            c = c_trial
            current_max = new_max
            if current_max < best_max - 1e-14:
                best_max = current_max
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        if no_imp > max(5000, 10 * n):
            a = best_a.copy()
            c = _fft_convolve(a)
            current_max = float(c.max())
            no_imp = 0
    return best_a


def _multiplicative_update(a, deadline, max_iter=50000, eta_start=0.1, eta_end=0.001, seed=42):
    """Multiplicative update: a_i proportional to exp(-eta * responsibility_i)."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    eta_ratio = (eta_end / eta_start) ** (1.0 / max(max_iter - 1, 1))
    eta = eta_start
    no_imp = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        eta = max(eta * eta_ratio, eta_end)
        c = _fft_convolve(a)
        m_val = float(c.max())
        # Soft responsibility: weighted by softmax of c
        x = (c - m_val) / (0.05 * m_val + 1e-10)
        x = np.clip(x, -50, 0)
        w = np.exp(x)
        w = w / w.sum()
        # Compute responsibility of each position
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        resp = h_full[n - 1:2 * n - 1]
        resp = np.maximum(resp, 1e-15)
        # Multiplicative update
        a_new = a * np.exp(-eta * resp * 2.0)
        a_new = np.maximum(a_new, 1e-12)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        score = _score_norm(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 5 * n):

                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                no_imp = 0
                eta = max(eta_start, 0.05)
    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, edge_weight=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_two_bump(n, sep=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.25) / 0.12) ** 2) + np.exp(-0.5 * ((t - 0.75) / 0.12) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
