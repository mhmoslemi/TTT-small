def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:

        if s > 1e-30:

        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)

    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:

def _solve_lp(a, M, n):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    A = _build_lp_matrix_fast(a)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _golden_line_search(a, g, n_evals=15):
    """Golden section line search for t in (0,1) minimizing score of (1-t)*a + t*g."""
    best_sc = _score(a)
    best_t = 0.0

    if g is None:
        return 0.0, best_sc

    g = np.maximum(g, 0.0)
    s_g = g.sum()
    if s_g < 1e-10:
        return 0.0, best_sc
    g = g / s_g

    # Golden section search in [0.01, 1.0]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.01
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score(np.maximum((1.0 - c) * a + c * g, 0.0))
    fd = _score(np.maximum((1.0 - d) * a + d * g, 0.0))
    if np.isfinite(fc) and fc < best_sc:
        best_sc = fc
        best_t = c
    if np.isfinite(fd) and fd < best_sc:
        best_sc = fd
        best_t = d

    for _ in range(min(n_evals, 20)):
        if np.isfinite(fc) and fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score(np.maximum((1.0 - c) * a + c * g, 0.0))
            if np.isfinite(fc) and fc < best_sc:
                best_sc = fc
                best_t = c
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score(np.maximum((1.0 - d) * a + d * g, 0.0))
            if np.isfinite(fd) and fd < best_sc:
                best_sc = fd
                best_t = d

    return best_t, best_sc


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.002, inflate_min=1.0005):
    """LP-based iterative refinement with adaptive inflation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        cur_sum = float(a.sum())
        ratio = sum_g / cur_sum if cur_sum > 0 else 1.0

        if ratio <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(0.05, inflate * 1.4)
            if no_imp > 200:
                break
            continue

        g = g / sum_g

        best_t, best_sc = _golden_line_search(a, g, n_evals=12)

        if best_t > 0.005 and best_sc < best_score - 1e-12:
            a_new = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            # Tighten inflation gradually
            if it % 50 == 49:
                inflate = max(inflate_min, inflate * 0.92)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.05, inflate * 1.2)
            if no_imp > 150:
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                if inflate > 1.003:
                    break

    return best_a


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a) using FFT."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    if mval < 1e-30:
        return np.zeros(2 * n - 1), mval
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # grad_a of smooth_max(conv(a,a)) = 2 * (w * a_reversed) convolved appropriately
    # d/d a_i [sum_k w_k * conv(a,a)_k] = 2 * sum_k w_k * a_{k-i} (for valid k-i)
    # = 2 * (w * a_rev)_i where a_rev is reversed and we take the right part
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    # corr[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-(k-j)]
    # We want: grad_i = 2 * sum_k w_k * a_{k-i} (convolution pattern)
    # = 2 * sum_j w_{j+i} * a_j ... hmm let me think differently
    # 
    # conv(a,a)_k = sum_j a_j * a_{k-j}
    # d/d a_i: sum_k w_k * (delta(j,i) * a_{k-j} + a_j * delta(k-j, i))
    # = sum_k w_k * a_{k-i} + sum_k w_k * a_{k-i} (with proper bounds)
    # Actually: d/d a_i [conv(a,a)_k] = a_{k-i} (if valid) + a_{k-i} (if valid) = 2*a_{k-i}
    # No wait: conv(a,a)_k = sum_j a_j * a_{k-j}
    # d/d a_i = a_{k-i} (from the first factor when j=i) + a_j * delta(j, k-i) * ... 
    # = a_{k-i} + a_{k-i} = 2 * a_{k-i} (if both indices valid)
    # 
    # So grad_i = 2 * sum_k w_k * a_{k-i} (with valid range)
    # This is 2 * (w * a)_i in the "cross-correlation" sense
    # = 2 * sum_k w_k * a_{k-i}
    # Let's use: result_i = sum_k w_k * a_{k-i} = (w cross-correlate a)_i
    # = sum_k w_k * a_{k-i}
    # If we reverse w: (w_rev * a)_i = sum_k w_rev_k * a_{i-k} = sum_k w_{L-1-k} * a_{i-k}
    # Hmm, let me just use the FFT approach directly.
    
    # sum_k w_k * a_{k-i} for i = 0..n-1
    # This is the cross-correlation of w and a, evaluated at lag i
    # = (w flipped) convolved with a, at index (len(w)-1-i+n-1)... 
    # 
    # Actually: let's define h = w[0:n] (or w[0:2n-1]) and compute h * a_rev via FFT
    # conv(w, a_rev)[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-k+j]
    # For k = n-1-i: = sum_j w[j] * a[n-1-(n-1-i)+j] = sum_j w[j] * a[i+j]
    # That's not quite what we want.
    
    # Let me just do it directly for correctness (slower but n is at most ~2048)
    # grad_i = 2 * sum_k w_k * a_{k-i} where 0 <= k-i < n, i.e., i <= k < i+n
    grad = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(i + n, 2 * n - 1)
        if k_lo < 2 * n - 1:
            grad[i] = 2.0 * float(np.dot(w[k_lo:k_hi], a[k_hi - 1 - k_lo::-1]))
    # Wait, that's not right either. Let me be more careful.
    # grad_i = 2 * sum_{k: 0<=k-i<n} w_k * a_{k-i}
    # = 2 * sum_{k=i}^{min(i+n-1, 2n-2)} w_k * a_{k-i}
    
    grad = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        if k_start <= 2 * n - 2:
            a_slice = a[0:k_end - i]
            w_slice = w[k_start:k_end]
            grad[i] = 2.0 * float(np.dot(w_slice, a_slice))
    
    return grad, mval


def _smooth_max_grad_fast(a, beta):
    """Faster gradient using FFT."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    if mval < 1e-30:
        return np.zeros(n), mval
    x = (c - mval) / beta
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # grad_i = 2 * sum_k w_k * a_{k-i} for k in [i, i+n-1]
    # = 2 * cross-correlation of w and a at lag i
    # Use FFT: cross_corr[i] = sum_k w[k] * a[k-i]
    # = sum_k w[k] * a[k-i]
    # Let's compute via: if we define b[k] = w[k] for k=0..2n-2, then
    # cross_corr[i] = sum_k b[k] * a[k-i]
    # This is the correlation of b and a.
    # Using FFT: correlation = ifft(fft(b) * conj(fft(a))) -- but sizes differ
    # 
    # Alternative: cross_corr[i] = sum_j a[j] * b[j+i] = sum_j a[j] * w[j+i]
    # = (a * w_rev) but with different sizes...
    # 
    # Simplest: use direct computation but vectorized
    # For each i, we need sum_{j=0}^{n-1} a[j] * w[j+i]
    # This is a sliding dot product
    N_pad = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N_pad)
    fw = np.fft.rfft(w, N_pad)
    # cross_corr[i] = sum_j a[j] * w[j+i]
    # = sum_j a[j] * conj(conj(w[j+i]))
    # Let's use: ifft(fa * conj(fw))[i] gives sum_j a[j] * w[i+j] ... no
    # 
    # Actually: ifft(F * G)[n] = sum_k f[k] * g[n-k] (circular correlation)
    # We want sum_j a[j] * w[i+j] = sum_j a[j] * w[i+j]
    # Let m = i+j, j = m-i: = sum_m a[m-i] * w[m] (with appropriate bounds)
    # This is cross-correlation at lag i.
    # 
    # If I compute ifft(conj(fa) * fw), I get sum_k a[k] * w[k+i] (approximately, with circular wrap)
    # Or ifft(fa * conj(fw)) gives sum_k a[k+i] * w[k]... 
    # 
    # Let me just use: result = ifft(fa * np.conj(fw), N_pad)
    # result[k] = sum_j a[j+k] * w[j] (circular) = sum_j a[j+k] * w[j]
    # We want sum_j a[j] * w[j+i] = sum_j a[j] * w[j+i]
    # Let m = j+i, j = m-i: = sum_m a[m-i] * w[m]
    # = (cross-correlation of a and w at lag -i) 
    # = ifft(conj(fa) * fw)[i] ... 
    # 
    # I'll just do it the simple way for correctness.
    
    # Vectorized sliding dot product
    # For i = 0: sum_j a[j] * w[j]
    # For i = 1: sum_j a[j] * w[j+1]
    # ...
    # For i = n-1: sum_j a[j] * w[j+n-1]
    
    # This is equivalent to: for each i, dot(a, w[i:i+n])
    # We can build a matrix but that's expensive. Let's use FFT.
    
    # Actually the simplest correct approach:
    # grad_i = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # Build w_padded = w[0:2n-1]
    # For i=0: dot(a, w[0:n])
    # For i=1: dot(a, w[1:n+1])
    # ...
    # For i=n-1: dot(a, w[n-1:2n-1])
    
    # This is a matrix-vector product where the matrix has columns w[0:n], w[1:n+1], ...
    # Equivalently, it's the "sliding window dot product"
    # We can compute it as: result[i] = sum_j a[j] * w[i+j]
    # = (a * w_rev)[Len(w)-1 - i + n - 1] ... this is getting confusing.
    
    # Let me just use a simple loop with numpy (n is at most 2048, so n iterations of dot products of length n)
    # That's n^2 operations = ~4M, which is fine.
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * float(np.dot(a, w[i:i + n]))
    
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=50000):
    """Gradient polish using smooth max with adaptive beta."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    c = _fft_convolve(a)
    maxc = float(c.max())
    beta = 0.02 * maxc
    lr = 0.05
    base_lr = lr
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, mval = _smooth_max_grad_fast(a, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 5.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.9)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                beta = 0.02 * float(_fft_convolve(a).max())
                no_imp = 0

        if it % 50 == 49:
            beta = max(1e-6, beta * 0.85)

    return best_a


def _direct_gradient(a0, deadline, max_iter=30000):
    """Alternative: directly minimize max(conv) using subgradient methods."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    lr = 0.02
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_max = int(np.argmax(c))
        mval = c[k_max]

        # Subgradient of max(conv(a,a)) at a:
        # d/da_i max_k conv(a,a)_k = 2 * a_{k_max - i} (if valid)
        grad = np.zeros(n)
        for i in range(n):
            idx = k_max - i
            if 0 <= idx < n:
                grad[i] = 2.0 * a[idx]

        # Also add small regularization to keep a away from boundaries
        lr_decay = lr / (1.0 + it * 0.001)
        a = a - lr_decay * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        score = _score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 1000:
                a = best_a.copy()
                no_imp = 0
                lr = max(0.002, lr * 0.8)

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP refinement with tight inflation,
    gradient polish with adaptive smoothing, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initializations
    configs = []
    lengths = [512, 768, 1024, 1536]

    for n_t in lengths:
        if parent is not None:
            a_int = _interpolate(parent, n_t)
            configs.append((f'parent_n{n_t}', n_t, a_int))

        for alpha in [0.5, 1.0, 1.5, 2.0]:
            configs.append((f'ramp_a{alpha}_n{n_t}', n_t, _make_ramp(n_t, alpha)))
            configs.append((f'ramp_a{alpha}_r_n{n_t}', n_t, _make_ramp_increasing(n_t, alpha)))

    # Score and sort
    scored = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    print(f"Top 5 initializations:")
    for i, (sc, name, n_t, _) in enumerate(scored[:5]):
        print(f"  {i+1}. {name}: score={sc:.8f}")

    # Phase 1: LP refinement on top candidates
    n_lp_attempts = 0
    for sc0, name, n_t, a0 in scored:
        if time.monotonic() > deadline - 200:
            break
        if sc0 >= best_score + 0.2:
            continue
        if n_lp_attempts >= 8:
            break

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        remaining = deadline - time.monotonic() - 150
        if remaining < 20:
            break

        # Allocate more time to better candidates
        if sc0 < best_score + 0.02:
            lp_time = min(300.0, remaining * 0.4)
        elif sc0 < best_score + 0.1:
            lp_time = min(180.0, remaining * 0.3)
        else:
            lp_time = min(100.0, remaining * 0.2)

        if lp_time < 10:
            break

        lp_end = time.monotonic() + lp_time
        print(f"LP: {name} n={n_t} init={sc0:.8f} budget={lp_time:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=50000, inflate_start=1.002, inflate_min=1.0003)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> LP improved: {sc_lp:.10f}")
        else:
            print(f"  -> LP: {sc_lp:.8f} (no improvement)")
        n_lp_attempts += 1

    # Phase 2: Additional LP on the best with tighter settings
    if best_seq is not None and time.monotonic() < deadline - 100:
        gp_time = min(120.0, deadline - time.monotonic() - 60)
        if gp_time > 20:
            print(f"Phase 2 LP: budget={gp_time:.0f}s")
            a2 = _lp_refine(best_seq, time.monotonic() + gp_time,
                           max_iter=30000, inflate_start=1.001, inflate_min=1.0002)
            sc2 = _score(a2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a2.copy()
            print(f"  -> Phase 2: {sc2:.10f}")

    # Phase 3: Gradient polish
    if best_seq is not None and time.monotonic() < deadline - 60:
        gp_time = min(180.0, deadline - time.monotonic() - 40)
        if gp_time > 20:
            print(f"Gradient polish: budget={gp_time:.0f}s, n={len(best_seq)}")
            a_gp = _gradient_polish(best_seq, time.monotonic() + gp_time, max_iter=50000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  -> Gradient: {sc_gp:.10f}")

    # Phase 4: Try interpolating best to a longer length and re-optimizing
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = len(best_seq)
        n_long = n_best * 2 if n_best <= 1024 else n_best + 256
        if n_long <= 2048:
            a_long = _interpolate(best_seq, n_long)
            sc_long = _score(a_long)
            print(f"Interpolated to n={n_long}: score={sc_long:.8f}")

            lp_time = min(120.0, deadline - time.monotonic() - 40)
            if lp_time > 20:
                a_long2 = _lp_refine(a_long, time.monotonic() + lp_time,
                                    max_iter=20000, inflate_start=1.002, inflate_min=1.0003)
                sc_long2 = _score(a_long2)
                if np.isfinite(sc_long2) and sc_long2 < best_score:
                    best_score = sc_long2
                    best_seq = a_long2.copy()
                    print(f"  -> Long LP: {sc_long2:.10f}")
                else:
                    print(f"  -> Long LP: {sc_long2:.8f}")

    # Phase 5: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        gp_time = min(60.0, deadline - time.monotonic() - 15)
        if gp_time > 10:
            a_f = _gradient_polish(best_seq, time.monotonic() + gp_time, max_iter=20000)
            sc_f = _score(a_f)
            if np.isfinite(sc_f) and sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  -> Final gradient: {sc_f:.10f}")

    # Phase 6: One more tight LP
    if best_seq is not None and time.monotonic() < deadline - 15:
        lp_time = min(60.0, deadline - time.monotonic() - 10)
        if lp_time > 10:
            a_f = _lp_refine(best_seq, time.monotonic() + lp_time,
                            max_iter=10000, inflate_start=1.001, inflate_min=1.0002)
            sc_f = _score(a_f)
            if np.isfinite(sc_f) and sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
            print(f"  -> Final LP: {sc_f:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(768, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(768, 1.0)

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        print(f"evaluate_sequence: {sc_eval:.10f}")
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
