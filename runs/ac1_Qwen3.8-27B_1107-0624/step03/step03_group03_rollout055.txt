```python
"""
Optimized step function coefficient search combining:
- Nesterov-accelerated log-sum-exp gradient descent
- Directed stochastic coordinate descent with incremental convolution updates
- LP-based spreading for escaping local minima
- Multi-length exploration with adaptive time allocation
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _update_conv_incremental(c, a, i, di, j, dj):
    """Update convolution c = conv(a,a) after changing a[i]+=di, a[j]+=dj.
    Returns updated c. Assumes di and dj are the actual changes."""
    n = a.size
    c = c.copy()
    # First order: 2*(di*a[k-i] + dj*a[k-j]) for valid indices
    # a[k-i] is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
    lo1 = max(0, i)
    hi1 = min(2 * n - 1, i + n)
    c[lo1:hi1] += 2.0 * di * a[lo1 - i:hi1 - i]
    
    lo2 = max(0, j)
    hi2 = min(2 * n - 1, j + n)
    c[lo2:hi2] += 2.0 * dj * a[lo2 - j:hi2 - j]
    
    # Second order: di^2 at k=2i, dj^2 at k=2j, 2*di*dj at k=i+j
    if 2 * i < 2 * n - 1:
        c[2 * i] += di * di
    if 2 * j < 2 * n - 1:
        c[2 * j] += dj * dj
    if i + j < 2 * n - 1:
        c[i + j] += 2.0 * di * dj
    
    # Clean up tiny negative values from floating point
    c = np.maximum(c, 0.0)
    return c


def _find_optimal_t(c, a, i, j, max_t, n_steps=25):
    """Find t in [0, max_t] minimizing max_k [c[k] + 2t*(a[k-i]-a[k-j]) + t^2*q[k]].
    Uses ternary search on the convex function."""
    n = len(a)
    # Precompute slope array: s[k] = 2*(a[k-i] - a[k-j])
    s = np.zeros(2 * n - 1)
    lo1 = max(0, i)
    hi1 = min(2 * n - 1, i + n)
    if lo1 < hi1:
        s[lo1:hi1] += 2.0 * a[lo1 - i:hi1 - i]
    lo2 = max(0, j)
    hi2 = min(2 * n - 1, j + n)
    if lo2 < hi2:
        s[lo2:hi2] -= 2.0 * a[lo2 - j:hi2 - j]
    
    # Quadratic corrections at positions 2i, 2j, i+j
    q = np.zeros(2 * n - 1)
    if 2 * i < 2 * n - 1:
        q[2 * i] = 1.0
    if 2 * j < 2 * n - 1:
        q[2 * j] = 1.0
    if i + j < 2 * n - 1:
        q[i + j] = -2.0
    
    def eval_f(t):
        vals = c + t * s + t * t * q
        return float(vals.max())
    
    # Check if t=0 is already optimal
    # The peak at t=0 has slopes from active constraints
    M0 = float(c.max())
    peak_mask = c >= M0 - 1e-12
    peak_slopes = s[peak_mask]
    if len(peak_slopes) > 0 and np.all(peak_slopes >= -1e-12):
        return 0.0
    
    # Ternary search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max_t
    c_val = hi - gr * (hi - lo)
    d_val = lo + gr * (hi - lo)
    fc = eval_f(c_val)
    fd = eval_f(d_val)
    for _ in range(n_steps):
        if fc < fd:
            hi = d_val
            d_val = c_val
            fd = fc
            c_val = hi - gr * (hi - lo)
            fc = eval_f(c_val)
        else:
            lo = c_val
            c_val = d_val
            fc = fd
            d_val = lo + gr * (hi - lo)
            fd = eval_f(d_val)
    return (lo + hi) / 2.0

    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    
    no_imp = 0
    rng = np.random.default_rng(12345)
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        if use_directed and step % 3 != 2:
            # Directed: find i with max contribution to peak, j with min
            # Contribution of a[l] to lag k_star is a[l] * a[k_star - l]
            valid_l = np.arange(n)
            valid_ks = k_star - valid_l
            contrib = np.zeros(n)
            mask = (valid_ks >= 0) & (valid_ks < n)
            contrib[mask] = a[valid_l[mask]] * a[valid_ks[mask]]
            
            # i: max contribution (source of peak mass)
            i = int(np.argmax(contrib))
            # j: min contribution with a[j] > 0 (target for mass redistribution)
            min_contrib = contrib.min()
            candidate_j = np.where(contrib <= min_contrib + 0.1 * (contrib.max() - min_contrib + 1e-15))[0]
            candidate_j = candidate_j[candidate_j != i]
            if len(candidate_j) == 0:
                candidate_j = np.ones(n)
                candidate_j[i] = 0
            j = int(candidate_j[rng.integers(len(candidate_j))])
            
            # Normal direction: from i to j
            t = _find_optimal_t(c, a, i, j, a[i])
            if t > 1e-10:
                # Apply update
                c = _update_conv_incremental(c, a, i, -t, j, t)
                a[i] -= t
                a[j] += t
        else:
            # Random pair
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            # Try both directions
            t1 = _find_optimal_t(c, a, i, j, a[i])
            t2 = _find_optimal_t(c, a, j, i, a[j])
            if t1 >= t2 and t1 > 1e-10:
                c = _update_conv_incremental(c, a, i, -t1, j, t1)
                a[i] -= t1
                a[j] += t1
            elif t2 > 1e-10:
                c = _update_conv_incremental(c, a, j, -t2, i, t2)
                a[j] -= t2
                a[i] += t2
            else:
                no_imp += 1
                continue
        
        # Evaluate
        new_score = float(2.0 * n * c.max())
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 5000:
                # Reset to best and try different strategy
                a = best_a.copy()
                c = _fft_convolve(a)
                no_imp = 0
        
        # Clean up
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            c = _fft_convolve(a)
            best_score = float(2.0 * n * c.max())
    
    return best_a


def _lse_nesterov(a, deadline, max_iter=500000, T_start=0.03, T_end=1e-5, lr=0.15):
    """Log-sum-exp gradient with Nesterov acceleration."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    v = np.zeros(n)  # momentum
    t_nest = 1.0
    z_prev = z.copy()
    no_imp = 0
    base_lr = lr
    N = _next_pow2(2 * n - 1)
    
    if max_iter > 1:
        T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    else:
        T_ratio = 1.0
    T = T_start
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute LSE gradient
        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        else:
            w = ex / s_ex
        
        # Gradient via FFT
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        # Project to tangent space (sum constraint)
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        
        if not np.all(np.isfinite(grad_tan)):
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            v = np.zeros(n)
            z_prev = z.copy()
            continue
        
        # Nesterov step
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        y = a + ((t_nest - 1.0) / t_next) * (a - (np.exp(z_prev) / np.exp(z_prev).sum()))
        y = np.maximum(y, 1e-12)
        y = y / y.sum()
        
        # Recompute grad at y for Nesterov (optional, using old grad for speed)
        z_new = np.log(y) - lr * grad_tan / (y + 1e-10)
        z_new = np.clip(z_new, -40, 40)
        z_new -= float(np.mean(z_new))
        a_new = np.exp(z_new)
        a_new = a_new / a_new.sum()
        
        z_prev = z.copy()
        z = z_new
        t_nest = t_next
        a = a_new
        
        # Evaluate
        score = float(2.0 * n * a.sum() * 0 + 2.0 * n * _fft_convolve(a).max())
        # Actually score is just 2n*max(conv) since sum=1
        c_new = _fft_convolve(a)
        score = float(2.0 * n * c_new.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.02, lr * 0.95)
            if no_imp > max(5000, 5 * n):
                a = best_a.copy()
                z = np.log(np.maximum(a, 1e-12))
                z -= float(np.mean(z))
                z_prev = z.copy()
                t_nest = 1.0
                lr = base_lr
                no_imp = 0
        
        T = max(T * T_ratio, T_end)
    
    return best_a


def _lp_spread(a, M, top_k=100):
    """LP to find direction that spreads mass while keeping conv below M."""
    n = len(a)
    c_aa = _fft_convolve(a)
    
    # Use reduced constraints
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[row, lo:hi] = a[lo:hi]
    
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search_blend(a, g, max_iters=20):
    """Golden section search for optimal blend factor."""
    n = len(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_val = hi - gr * (hi - lo)
    d_val = lo + gr * (hi - lo)
    
    def eval_blend(t):
        a_new = np.maximum(a + t * (g - a), 0)
        s = a_new.sum()
        if s < 1e-10:
            return float('inf')
        a_new = a_new / s
        return _score_normalized(a_new)
    
    fc = eval_blend(c_val)
    fd = eval_blend(d_val)
    for _ in range(max_iters):
        if fc < fd:
            hi = d_val
            d_val = c_val
            fd = fc
            c_val = hi - gr * (hi - lo)
            fc = eval_blend(c_val)
        else:
            lo = c_val
            c_val = d_val
            fc = fd
            d_val = lo + gr * (hi - lo)
            fd = eval_blend(d_val)
    return (lo + hi) / 2.0


def _lp_refine(a, deadline, max_iter=500, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    inflate_cur = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        
        g = _lp_spread(a, M_lp)
        if g is None:
            inflate_cur = min(1.02, inflate_cur * 1.1)
            no_imp += 3
            continue
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        g = g / sum_g
        t_opt = _line_search_blend(a, g)
        if t_opt < 0.01:
            t_opt = 0.1
        
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.015, inflate_cur * 0.95)
            if no_imp > 150:
                a = best_a.copy()
                no_imp = 0
    
    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bell_edge(n, alpha=2.0):
    """Raised cosine with sharp edges."""
    t = np.linspace(0.0, 1.0, n)
    a = (0.5 * (1.0 + np.cos(2.0 * np.pi * t))) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_anti_palindromic(n):
    """Sequence that is anti-symmetric to reduce center peak."""
    t = np.linspace(0.0, 1.0, n)
    # Weight more at edges, less at center
    a = np.abs(t - 0.5) + 0.1
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LSE-Nesterov, directed coordinate descent,
    LP refinement. Explores multiple initializations and lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_full(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 1024, 2048]:
        if ln not in lengths:
            lengths.append(ln)
    
    # Generate candidates per length
    def get_candidates(n_t):
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            p_pert = cands[0] * (1.0 + 0.1 * rng.normal(size=n_t))
            p_pert = np.maximum(p_pert, 0)
            if p_pert.sum() > 1e-10:
                cands.append(p_pert / p_pert.sum())
            p_pert2 = cands[0] * (1.0 + 0.05 * rng.normal(size=n_t))
            p_pert2 = np.maximum(p_pert2, 0)
            if p_pert2.sum() > 1e-10:
                cands.append(p_pert2 / p_pert2.sum())
        cands.append(_make_gaussian(n_t, 0.35))
        cands.append(_make_bell_edge(n_t, 2.0))
        cands.append(_make_anti_palindromic(n_t))
        return cands
    
    # === Phase 1: LSE-Nesterov on best candidates ===
    print("\n=== Phase 1: LSE-Nesterov ===")
    phase1_deadline = t_start + min(500, budget_s * 0.28)
    best_per_length = {}
    
    for n_t in lengths:
        if time.monotonic() > phase1_deadline - 60:
            break
        
        cands = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf
        
        for ci, a0 in enumerate(cands):
            if time.monotonic() > phase1_deadline - 30:
                break
            
            sc0 = _score_full(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            
            # Run LSE-Nesterov
            lse_end = min(phase1_deadline - 10, time.monotonic() + 120)
            a_lse = _lse_nesterov(a0, lse_end, max_iter=100000,
                                  T_start=0.03, T_end=1e-5, lr=0.12)
            sc_lse = _score_full(a_lse)
            
            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse
            
            print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> LSE={sc_lse:.10f}")
        
        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best LSE for n={n_t}: {best_here_score:.10f}")
    
    # === Phase 2: Directed Coordinate Descent ===
    print("\n=== Phase 2: Coordinate Descent ===")
    phase2_deadline = t_start + min(1200, budget_s * 0.67)
    
    for n_t in lengths:
        if time.monotonic() > phase2_deadline - 60:
            break
        
        if n_t not in best_per_length:
            continue
        
        a_in = best_per_length[n_t]
        sc_in = _score_full(a_in)
        print(f"  CD start n={n_t}: score={sc_in:.10f}")
        
        cd_end = min(phase2_deadline - 20, time.monotonic() + 250)
        a_cd = _coordinate_descent(a_in, cd_end, max_steps=5000000)
        sc_cd = _score_full(a_cd)
        
        if np.isfinite(sc_cd) and sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
        best_per_length[n_t] = a_cd
        print(f"  CD result n={n_t}: score={sc_cd:.10f}")
        
        # Second CD pass with different strategy
        if time.monotonic() < phase2_deadline - 60:
            cd_end2 = min(phase2_deadline - 10, time.monotonic() + 200)
            a_cd2 = _coordinate_descent(a_cd, cd_end2, max_steps=5000000, use_directed=False)
            sc_cd2 = _score_full(a_cd2)
            if np.isfinite(sc_cd2) and sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()
            best_per_length[n_t] = a_cd2
            print(f"  CD2 result n={n_t}: score={sc_cd2:.10f}")
    
    # === Phase 3: LP Refinement ===
    print("\n=== Phase 3: LP Refinement ===")
    phase3_deadline = t_start + min(1500, budget_s * 0.83)
    
    for n_t in lengths:
        if time.monotonic() > phase3_deadline - 30:
            break
        if n_t not in best_per_length:
            continue
        
        a_in = best_per_length[n_t]
        sc_in = _score_full(a_in)
        print(f"  LP start n={n_t}: score={sc_in:.10f}")
        
        lp_end = min(phase3_deadline - 10, time.monotonic() + 150)
        a_lp = _lp_refine(a_in, lp_end, max_iter=300, inflate=1.001, blend=0.25)
        sc_lp = _score_full(a_lp)
        
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP result n={n_t}: score={sc_lp:.10f}")
    
    # === Phase 4: Final Coordinate Descent Polish ===
    print("\n=== Phase 4: Final CD Polish ===")
    if best_seq is not None and time.monotonic() < phase3_deadline:
        cd_end = min(deadline - 30, time.monotonic() + 200)
        a_final_cd = _coordinate_descent(best_seq, cd_end, max_steps=5000000)
        sc_final_cd = _score_full(a_final_cd)
        if np.isfinite(sc_final_cd) and sc_final_cd < best_score:
            best_score = sc_final_cd
            best_seq = a_final_cd.copy()
        print(f"  Final CD: score={sc_final_cd:.10f}")
    
    # === Phase 5: Short LSE polish on very best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        lse_end = min(deadline - 10, time.monotonic() + 80)
        a_final_lse = _lse_nesterov(best_seq, lse_end, max_iter=50000,
                                    T_start=0.01, T_end=1e-6, lr=0.05)
        sc_final_lse = _score_full(a_final_lse)
        if np.isfinite(sc_final_lse) and sc_final_lse < best_score:
            best_score = sc_final_lse
            best_seq = a_final_lse.copy()
        print(f"  Final LSE: score={sc_final_lse:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0
    
    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")
    
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
```
