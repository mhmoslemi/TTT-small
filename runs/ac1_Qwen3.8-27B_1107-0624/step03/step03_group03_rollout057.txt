    n = a.size
    c = _fft_convolve(a)

def _make_conv_matrix(a):
    """Build matrix A where (A @ delta)[k] = conv(delta, a)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_direct_minimize(a, c, active_lags=None, verbose=False):
    """
    Solve LP to find direction delta that minimizes linearized max convolution.
    minimize t
    s.t. c[k] + 2*(A@delta)[k] <= t for k in active_lags
         sum(delta) = 0
         delta[i] >= -a[i]
    Returns delta (direction) or None if infeasible.
    """
    n = len(a)
    m = 2 * n - 1

    if active_lags is None:
        active_lags = np.arange(m)

    A_sub = np.zeros((len(active_lags), n))
    for row, k in enumerate(active_lags):
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        if jlo <= jhi:
            A_sub[row, jlo:jhi + 1] = a[jlo:jhi + 1]

    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality constraints: c[k] + 2*A_sub@delta - t <= 0
    A_ub = np.zeros((len(active_lags), n + 1))
    A_ub[:, :n] = 2.0 * A_sub
    A_ub[:, -1] = -1.0
    b_ub = -c[active_lags]  # c[k] + 2*A@delta <= t => 2*A@delta - t <= -c[k]

    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    # Bounds: delta[i] >= -a[i], t >= 0
    bounds = [(-a[i], None) for i in range(n)]
    bounds.append((0, None))

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        return res.x[:n]
    return None

        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = _score_normalized(a_try)
        if sc < best_local_score:
            best_local_score = sc
            best_local = alpha

    if best_local == 0.0:
        return 0.0, best_score

    # Golden section around best_local
    lo = max(0.0, best_local - 0.15)
    hi = min(1.0, best_local + 0.15)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)

    def eval_at(alpha):
        a_try = np.maximum(a + alpha * delta, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            return np.inf
        return _score_normalized(a_try / s)

    fc = eval_at(c_pt)
    fd = eval_at(d_pt)
    for _ in range(8):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_at(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_at(d_pt)

    best_alpha = (lo + hi) / 2.0
    best_score = eval_at(best_alpha)
    return best_alpha, best_score


def _lp_refine_direct(a, deadline, max_iter=2000, top_k=80):
    """LP-based refinement using direct peak minimization."""
    a = np.asarray(a, dtype=np.float64).copy().ravel()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    k_cur = top_k

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Select active lags: those within a fraction of the max
        threshold = M * 0.995
        active_mask = c >= threshold
        active_idx = np.where(active_mask)[0]

        # Also include top k_cur by value
        top_idx = np.argsort(c)[::-1][:k_cur]
        active_set = np.sort(np.unique(np.concatenate([active_idx, top_idx])))

        delta = _lp_direct_minimize(a, c, active_set)

        if delta is None:
            no_imp += 5
            if no_imp > 200:
                break
            continue

        # Check if delta is meaningful
        delta_norm = np.linalg.norm(delta)
        if delta_norm < 1e-12:
            no_imp += 1
            if no_imp % 50 == 0:
                k_cur = max(20, k_cur // 2)
            if no_imp > 300:
                break
            continue

        alpha, new_score = _line_search_lp_step(a, delta)

        if alpha < 1e-4:
            no_imp += 1
            if no_imp > 200:
                a = best_a.copy()
                k_cur = top_k
                no_imp = 0
            if no_imp > 500:
                break
            continue

        a_new = np.maximum(a + alpha * delta, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                k_cur = max(20, k_cur - 5)
            if no_imp > 300:
                a = best_a.copy()
                k_cur = top_k
                no_imp = 0

    return best_a


def _lse_optimize(a, deadline, max_iter=50000, T_start=0.04, T_end=1e-5, lr=0.08):
    """Log-sum-exp gradient optimization with adaptive restarts."""
    a = np.asarray(a, dtype=np.float64).copy().ravel()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state_t = [0, np.zeros(n), np.zeros(n)]

    best_a = a.copy()
    best_score = _score_normalized(a)
    T = T_start
    no_imp = 0
    base_lr = lr

    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        x = np.clip((c - m_val) / T, -50, 50)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        else:
            w = ex / s_ex

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state_t = [0, np.zeros(n), np.zeros(n)]
            continue

        t = state_t[0] + 1
        b1, b2 = 0.9, 0.999
        m = b1 * state_t[1] + (1.0 - b1) * gz
        v = b2 * state_t[2] + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a_new = e / e.sum()

        state_t = [t, m, v]

        if not np.all(np.isfinite(a_new)):
            a = best_a.copy()
            continue

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 800 == 0:
                lr = max(0.01, lr * 0.92)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.1 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state_t = [0, np.zeros(n), np.zeros(n)]
                lr = base_lr
                no_imp = 0

        T = max(T * T_ratio, T_end)

    return best_a


def _subgradient_polish(a, deadline, max_iter=30000, lr=0.01):
    """Polish using subgradient at actual max points."""
    a = np.asarray(a, dtype=np.float64).copy().ravel()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_normalized(a)
    v = np.zeros(n)
    momentum = 0.7
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-10)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.001, lr * 0.93)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_profile(type_str, n):
    t = np.linspace(0.0, 1.0, n)
    if type_str == 'gaussian':
        a = np.exp(-0.5 * ((t - 0.5) / 0.25) ** 2)
    elif type_str == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif type_str == 'flat_edge':
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
    elif type_str == 'double_cos':
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t) + 0.15 * np.cos(4.0 * np.pi * t)
    elif type_str == 'sin_arch':
        a = np.sin(np.pi * t) ** 2
    elif type_str == 'linear':
        a = 1.0 + 1.5 * (t - 0.5)
    elif type_str == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _conv_shape_optimize(a, deadline, max_iter=5000, n_shape=64):
    """
    Optimize by targeting a specific convolution shape.
    Iteratively: compute current conv, define target (capped version),
    then find a that best matches the target via frequency-domain "square root".
    """
    a = np.asarray(a, dtype=np.float64).copy().ravel()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Define target: slightly below current peak, with some shaping
        # Target should be "flatter" than current c
        target = np.minimum(c, M * 0.99)
        # Add a small "floor" to avoid zeros (which cause issues in sqrt)
        floor = 1e-6 / n
        target = np.maximum(target, floor)

        # Frequency-domain "square root"
        N = _next_pow2(2 * n - 1)
        C_freq = np.fft.rfft(target, N)
        # Take element-wise square root (choose branch with positive real part)
        A_freq = np.sqrt(C_freq)
        # Ensure the "DC" component (sum) is correct
        a_candidate = np.fft.irfft(A_freq, N)[:n]

        # Make non-negative and normalize
        a_candidate = np.maximum(a_candidate, 0.0)
        s_cand = a_candidate.sum()
        if s_cand < 1e-10:
            no_imp += 10
            continue
        a_candidate = a_candidate / s_cand

        # Blend
        alpha = 0.3 if it < 100 else 0.1
        a_new = (1.0 - alpha) * a + alpha * a_candidate
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp > 200:
                a = best_a.copy()
                no_imp = 0

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Direct LP minimization for peak reduction
    Phase 2: LSE gradient for smooth refinement
    Phase 3: Convolution shape matching
    Phase 4: Subgradient polish
    Phase 5: Final LP + LSE
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [1024, 2048]:
        if ln not in lengths:
            lengths.append(ln)

    # Phase 1: LP direct minimization on best candidates
    print("\n=== Phase 1: LP direct minimization ===")
    for n_t in lengths:
        if time.monotonic() > deadline - 300:
            break

        # Create initial candidates for this length
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            for _ in range(2):
                p_int = _interpolate_profile(parent, n_t)
                p_pert = p_int * (1.0 + 0.06 * rng.normal(size=n_t))
                p_pert = np.maximum(p_pert, 0)
                if p_pert.sum() > 1e-10:
                    cands.append(p_pert / p_pert.sum())

        lp_budget = min(300.0, deadline - time.monotonic() - 300.0)
        if lp_budget < 30:
            break

        # Use the best initial candidate
        best_init = None
        best_init_score = np.inf
        for c in cands:
            sc = _score_normalized(c)
            if sc < best_init_score:
                best_init_score = sc
                best_init = c

        if best_init is None:
            best_init = _make_profile('raised_cosine', n_t)

        lp_end = time.monotonic() + lp_budget
        a_lp = _lp_refine_direct(best_init, lp_end, max_iter=5000, top_k=100)
        sc_lp = _score(a_lp)

        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"  LP n={n_t}: {sc_lp:.10f}")

    # Phase 2: LSE gradient on best
    print("\n=== Phase 2: LSE gradient ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        lse_budget = min(200.0, deadline - time.monotonic() - 200.0)
        if lse_budget > 20:
            lse_end = time.monotonic() + lse_budget
            a_lse = _lse_optimize(best_seq, lse_end, max_iter=60000,
                                  T_start=0.03, T_end=1e-6, lr=0.07)
            sc_lse = _score(a_lse)
            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            print(f"  LSE: {sc_lse:.10f}")

    # Phase 3: Convolution shape matching
    print("\n=== Phase 3: Conv shape matching ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        cs_budget = min(120.0, deadline - time.monotonic() - 150.0)
        if cs_budget > 20:
            cs_end = time.monotonic() + cs_budget
            a_cs = _conv_shape_optimize(best_seq, cs_end, max_iter=3000)
            sc_cs = _score(a_cs)
            if np.isfinite(sc_cs) and sc_cs < best_score:
                best_score = sc_cs
                best_seq = a_cs.copy()
            print(f"  ConvShape: {sc_cs:.10f}")

    # Phase 4: LP again (finer)
    print("\n=== Phase 4: Fine LP ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        lp_budget = min(150.0, deadline - time.monotonic() - 120.0)
        if lp_budget > 20:
            lp_end = time.monotonic() + lp_budget
            a_lp2 = _lp_refine_direct(best_seq, lp_end, max_iter=5000, top_k=60)
            sc_lp2 = _score(a_lp2)
            if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = a_lp2.copy()
            print(f"  Fine LP: {sc_lp2:.10f}")

    # Phase 5: Subgradient polish
    print("\n=== Phase 5: Subgradient polish ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        sp_budget = min(80.0, deadline - time.monotonic() - 60.0)
        if sp_budget > 10:
            sp_end = time.monotonic() + sp_budget
            a_sp = _subgradient_polish(best_seq, sp_end, max_iter=40000, lr=0.008)
            sc_sp = _score(a_sp)
            if np.isfinite(sc_sp) and sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            print(f"  Subgrad: {sc_sp:.10f}")

    # Phase 6: Final LSE
    print("\n=== Phase 6: Final LSE ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        lse_budget = min(50.0, deadline - time.monotonic() - 20.0)
        if lse_budget > 10:
            lse_end = time.monotonic() + lse_budget
            a_final = _lse_optimize(best_seq, lse_end, max_iter=30000,
                                    T_start=0.02, T_end=1e-7, lr=0.04)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LSE: {sc_final:.10f}")

    # Phase 7: Ultimate LP (one last shot)
    print("\n=== Phase 7: Ultimate LP ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        lp_end = deadline
        a_ul = _lp_refine_direct(best_seq, lp_end, max_iter=2000, top_k=40)
        sc_ul = _score(a_ul)
        if np.isfinite(sc_ul) and sc_ul < best_score:
            best_score = sc_ul
            best_seq = a_ul.copy()
        print(f"  Ultimate LP: {sc_ul:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
