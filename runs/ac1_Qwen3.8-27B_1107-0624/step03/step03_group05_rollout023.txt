"""
Optimized peak-reduction algorithm for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated subgradient on the auto-convolution peak, adaptive
smooth-max gradients, LP spreading, and structured diversification.
Key novelty: fast exact-subgradient Nesterov steps targeting the single peak,
combined with multi-scale beta adaptation and batch mass transfers.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _gradient_from_weights(a, w):
    """Compute gradient of sum_k w[k]*c[k] with respect to a, where c=conv(a,a)."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad


def _subgradient_peak(a, peak_pos):
    """Exact subgradient of max_k c[k] at the peak position."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        p = peak_pos - j
        if 0 <= p < n:
            grad[j] += a[p]
        # Also the symmetric contribution
        p2 = peak_pos - j
        if 0 <= j < n and 0 <= (peak_pos - j) < n:
            pass  # already counted above
    # Actually: d/d(a[j]) of c[peak_pos] = a[peak_pos-j] + a[peak_pos-j]
    # = 2*a[peak_pos-j] if j != peak_pos-j, or 2*a[j] if 2j=peak_pos
    # More simply: d/d(a[j]) sum_k a[k]*a[peak_pos-k]
    # = a[peak_pos-j] (from term k=j) + a[peak_pos-j] (from term k=peak_pos-j, i.e., a[peak_pos-j]*a[j])
    # Wait, the term k=peak_pos-j gives a[peak_pos-j]*a[j], derivative wrt a[j] is a[peak_pos-j]
    # And the term k=j gives a[j]*a[peak_pos-j], derivative wrt a[j] is a[peak_pos-j]
    # So total = 2*a[peak_pos-j] (when both terms are valid, i.e., 0<=peak_pos-j<n)
    # But if peak_pos-j == j (i.e., 2j = peak_pos), then it's a[j]*a[j] = a[j]^2, derivative = 2*a[j]
    # Which is the same as 2*a[peak_pos-j] = 2*a[j]. So the formula is uniform.
    
    # Recompute cleanly:
    grad = np.zeros(n)
    for j in range(n):
        p = peak_pos - j
        if 0 <= p < n:
            grad[j] = 2.0 * a[p]
    return grad


def _nesterov_subgradient(a, deadline, max_iter=50000, lr_init=0.02):
    """Nesterov-accelerated subgradient method targeting the main peak."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at lookahead point
        c_y = _fft_convolve(y)
        peak_pos = int(np.argmax(c_y))
        grad = _subgradient_peak(y, peak_pos)
        
        # Nesterov step
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new
        y = a + momentum * v
        
        # Evaluate
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.97)
            if no_imp > max(2000, 5 * n):
                a = best_a.copy()
                y = a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
        
        if it % 50 == 0:
            lr = max(0.001, lr * 0.995)
    
    return best_a, best_score

def _lp_refine(a, deadline, max_iter=15000, inflate=1.001, blend=0.3):

    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, _score(a)
    a = a / s
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    blend_cur = blend
    inflate_cur = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        
        # Build LP
        c_obj = -np.ones(n)
        A = _make_conv_matrix(a)
        b = np.full(2 * n - 1, M_lp)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.05)
            if no_imp > 600:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0005, inflate_cur * 0.98)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = min(1.01, inflate_cur * 0.97)
            if no_imp > max(400, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

    return best_a, best_score


def _batch_transfer(a, deadline, max_iter=20000, batch_size=10):
    """Batch mass transfer: move mass from top-contributing to bottom-contributing positions."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    c = _fft_convolve(a)
    M = c.max()
    peak_pos = int(np.argmax(c))
    
    no_imp = 0
    delta_frac = 0.03
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Update convolution periodically
        if it % 20 == 0:
            c = _fft_convolve(a)
            M = float(c.max())
            peak_pos = int(np.argmax(c))
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > 100:
                    break
        
        # Compute contribution of each position to the peak
        contrib = np.zeros(n)
        for j in range(n):
            p = peak_pos - j
            if 0 <= p < n:
                contrib[j] = 2.0 * a[j] * a[p] if j != p else a[j] * a[j]
        
        # Find top and bottom contributors
        sorted_idx = np.argsort(contrib)[::-1]
        top = sorted_idx[:batch_size]
        bottom = sorted_idx[-batch_size:]
        
        # Transfer from top to bottom
        total_top = a[top].sum()
        if total_top < 1e-10:
            continue
        
        for i in range(len(top)):
            src = top[i]
            dst = bottom[i]
            if src == dst:
                continue
            delta = delta_frac * a[src]
            if delta > 0.5 * a[src]:
                delta = 0.3 * a[src]
            a[src] -= delta
            a[dst] += delta
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
    
    c = _fft_convolve(a)
    sc = 2.0 * n * c.max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def _logit_adam_polish(a, deadline, max_iter=20000, lr=0.04):

    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    
    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.08 * float(c.max()))
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.96)
        
        if it % 100 == 0:
            beta = max(1e-10, beta * 0.7)
        
        if no_imp > max(3000, 5 * n):
            rng_r = np.random.default_rng(it * 7919 + n)
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.08 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_a, best_score


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, center_sep=0.3, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - center_sep
    c2 = 0.5 + center_sep
    a = np.exp(-0.5 * ((t - c1) / width) ** 2) + np.exp(-0.5 * ((t - c2) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _pipeline(a_init, n, deadline):
    """Run multi-method pipeline on a single sequence."""
    a = np.asarray(a_init, dtype=np.float64).ravel()[:n]
    if len(a) < n:
        a = np.resize(a, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_u_profile(n, 2.0)
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Phase 1: Nesterov subgradient (fast, targets single peak)
    t1 = time.monotonic()
    rem = deadline - t1 - 30
    if rem > 10:
        budget1 = min(rem * 0.25, 120.0)
        a1, sc1 = _nesterov_subgradient(best_a, t1 + budget1, max_iter=80000, lr_init=0.015)
        if sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
        print(f"      Nesterov: {sc1:.10f}")
    
    # Phase 2: Batch transfer
    t2 = time.monotonic()
    rem = deadline - t2 - 25
    if rem > 8:
        budget2 = min(rem * 0.15, 60.0)
        a2, sc2 = _batch_transfer(best_a, t2 + budget2, max_iter=30000, batch_size=8)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
        print(f"      BatchTransfer: {sc2:.10f}")
    
    # Phase 3: LP refinement
    t3 = time.monotonic()
    rem = deadline - t3 - 20
    if rem > 10:
        budget3 = min(rem * 0.25, 180.0)
        a3, sc3 = _lp_refine(best_a, t3 + budget3, max_iter=20000, inflate=1.0012, blend=0.25)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
        print(f"      LP: {sc3:.10f}")
    
    # Phase 4: Adaptive beta gradient
    t4 = time.monotonic()
    rem = deadline - t4 - 15
    if rem > 8:
        budget4 = min(rem * 0.2, 120.0)
        a4, sc4 = _adaptive_beta_gradient(best_a, t4 + budget4, max_iter=25000, beta_scale_init=0.08)
        if sc4 < best_score:
            best_score = sc4
            best_a = a4.copy()
        print(f"      GradBeta: {sc4:.10f}")
    
    # Phase 5: Adam polish
    t5 = time.monotonic()
    rem = deadline - t5 - 10
    if rem > 8:
        budget5 = min(rem * 0.15, 100.0)
        a5, sc5 = _logit_adam_polish(best_a, t5 + budget5, max_iter=15000, lr=0.035)
        if sc5 < best_score:
            best_score = sc5
            best_a = a5.copy()
        print(f"      Adam: {sc5:.10f}")
    
    # Phase 6: Final Nesterov
    t6 = time.monotonic()
    rem = deadline - t6 - 5
    if rem > 5:
        budget6 = min(rem * 0.15, 60.0)
        a6, sc6 = _nesterov_subgradient(best_a, t6 + budget6, max_iter=50000, lr_init=0.008)
        if sc6 < best_score:
            best_score = sc6
            best_a = a6.copy()
        print(f"      Nesterov2: {sc6:.10f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization: Nesterov subgradient + batch transfer + LP + Adam.
    Tries multiple lengths and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
        # Try nearby lengths
        for delta in [-64, -32, 32, 64, 128]:
            ln = parent.size + delta
            if 128 <= ln <= 4096 and ln not in lengths:
                lengths.append(ln)
    
    # Add some standard lengths
    for ln in [512, 768, 1024, 1280]:
        if ln not in lengths:
            lengths.append(ln)
    
    lengths = sorted(set(l for l in lengths if 128 <= l <= 4096))
    print(f"Lengths to try: {lengths}")
    print(f"Total budget: {budget_s}s, effective deadline in {deadline - time.monotonic():.0f}s")
    
    # Phase 1: Try each length with parent-based initialization
    num_lengths = len(lengths)
    # Allocate more time to parent length
    parent_len = parent.size if parent is not None else 768
    
    for idx, n_t in enumerate(lengths):
        if time.monotonic() > deadline - 120:
            break
        
        # Time allocation: more for parent length
        is_parent = (n_t == parent_len)
        if is_parent:
            time_frac = 0.25
        else:
            time_frac = 0.10
        
        remaining = deadline - time.monotonic() - 100
        phase_budget = max(30.0, min(remaining * time_frac, 300.0))
        
        print(f"\n=== Length {n_t} (parent={is_parent}), budget={phase_budget:.0f}s ===")
        
        # Initialize
        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
        else:
            a_init = _make_u_profile(n_t, 2.0)
        
        sc_init = _score(a_init)
        if np.isfinite(sc_init) and sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()
        
        a_best, sc_best = _pipeline(a_init, n_t, time.monotonic() + phase_budget)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best.copy()
            print(f"  ** New best at n={n_t}: {sc_best:.10f}")
    
    # Phase 2: Diverse initializations at the best length
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = best_seq.size
        print(f"\n=== Phase 2: Diverse inits at n={n_best} ===")
        
        inits = []
        # U-shaped
        for alpha in [1.0, 2.0, 3.0, 4.0, 6.0]:
            inits.append(_make_u_profile(n_best, alpha))
        # Bimodal
        for sep in [0.2, 0.3, 0.4]:
            for w in [0.1, 0.15, 0.2]:
                inits.append(_make_bimodal_profile(n_best, sep, w))
        # Linear
        for eta in [1.0, 2.0, 3.0, 4.0]:
            inits.append(_make_linear_profile(n_best, eta))
        # Perturbed best
        for scale in [0.1, 0.2, 0.3]:
            a_p = best_seq * (1.0 + scale * rng.normal(size=n_best))
            a_p = np.maximum(a_p, 0.0)
            a_p = a_p / a_p.sum()
            inits.append(a_p)
        # Random smooth
        for _ in range(3):
            a_r = np.maximum(rng.normal(1.0, 0.4, n_best), 0.0)
            # Smooth it
            kernel = np.ones(5) / 5.0
            a_r = np.convolve(a_r, kernel, mode='same')
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / a_r.sum()
            inits.append(a_r)
        
        # Limit number of inits based on remaining time
        rem_time = deadline - time.monotonic() - 50
        max_inits = max(1, int(rem_time / 45.0))
        inits = inits[:min(len(inits), max_inits)]
        
        print(f"  Trying {len(inits)} diverse initializations")
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline - 40:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            sub_budget = min(40.0, rem * 0.6)
            a_res, sc_res = _pipeline(a0, n_best, time.monotonic() + sub_budget)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best (init {i}): {sc_res:.10f}")
    
    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = best_seq.size
        rem = deadline - time.monotonic() - 10
        print(f"\n=== Phase 3: Final polish, budget={rem:.0f}s ===")
        
        # Sub-gradient Nesterov (long)
        b1 = min(rem * 0.3, 120.0)
        if b1 > 10:
            a1, sc1 = _nesterov_subgradient(best_seq, time.monotonic() + b1, max_iter=100000, lr_init=0.005)
            if sc1 < best_score:
                best_score = sc1
                best_seq = a1.copy()
            print(f"  Final Nesterov: {sc1:.10f}")
        
        # LP
        b2 = min(rem * 0.3, 120.0)
        if b2 > 10:
            a2, sc2 = _lp_refine(best_seq, time.monotonic() + b2, max_iter=20000, inflate=1.0008, blend=0.2)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a2.copy()
            print(f"  Final LP: {sc2:.10f}")
        
        # Batch transfer
        b3 = min(rem * 0.15, 60.0)
        if b3 > 5:
            a3, sc3 = _batch_transfer(best_seq, time.monotonic() + b3, max_iter=30000, batch_size=5)
            if sc3 < best_score:
                best_score = sc3
                best_seq = a3.copy()
            print(f"  Final Batch: {sc3:.10f}")
        
        # Adam
        b4 = min(rem * 0.2, 80.0)
        if b4 > 5:
            a4, sc4 = _logit_adam_polish(best_seq, time.monotonic() + b4, max_iter=15000, lr=0.02)
            if sc4 < best_score:
                best_score = sc4
                best_seq = a4.copy()
            print(f"  Final Adam: {sc4:.10f}")
        
        # One more Nesterov
        b5 = min(rem * 0.1, 40.0)
        if b5 > 5:
            a5, sc5 = _nesterov_subgradient(best_seq, time.monotonic() + b5, max_iter=50000, lr_init=0.003)
            if sc5 < best_score:
                best_score = sc5
                best_seq = a5.copy()
            print(f"  Final Nesterov2: {sc5:.10f}")
    
    # Final validation
    if best_seq is None:
        best_seq = _make_u_profile(768, 2.0)
        best_score = _score(best_seq)
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    
    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception:
        pass
    
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
