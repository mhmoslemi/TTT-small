from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    """Returns 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())

    """Returns 2*n*max(conv(a,a))/sum(a)^2."""

        return np.inf

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines aggressive Adam with smooth-max continuation, LP-guided blending,
direct subgradient steps, and multi-length exploration.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, ptype='linear', param=1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if ptype == 'linear':
        a = 1.0 + param * u
    elif ptype == 'quadratic':
        a = 1.0 + param * u ** 2
    elif ptype == 'cosine':
        a = 1.0 + param * np.cos(np.pi * t)
    elif ptype == 'gaussian':
        a = np.exp(-param * u ** 2)
    elif ptype == 'beta':
        from scipy.special import betainc
        a = t ** param * (1 - t) ** param
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _compute_gradient(a, beta):
    """Compute gradient of smoothed objective w.r.t. z (log-parameters)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad, c, w


def _adam_optimize(a_init, deadline, beta_frac=0.01, lr=0.1, n_max_iter=100000,
                   beta_decay=0.95, beta_decay_interval=50, min_beta_frac=1e-6,
                   lr_decay=0.999, lr_min=0.001):
    """Adam optimization in log-space with adaptive beta annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c0 = float(c.max())
    beta = max(1e-15, beta_frac * max_c0)
    min_beta = max(1e-15, min_beta_frac * max_c0)

    while time.monotonic() < deadline and t < n_max_iter:
        t += 1
        a = _softmax_fn(z)
        grad, c, w = _compute_gradient(a, beta)

        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % beta_decay_interval == 0:
            beta = max(min_beta, beta * beta_decay)

        if t % 1000 == 0:
            lr = max(lr_min, lr * lr_decay)

    return best_a, best_score


def _subgradient_step(a, n_steps=50, lr=0.01, deadline=None):
    """Direct subgradient step targeting the argmax of the convolution."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    beta = 1e-10

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        grad, c, w = _compute_gradient(a, beta)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _lp_blend_step(f, n_max_iter=20, t_step=0.3, deadline=None):
    """LP-guided blending: find g maximizing sum(g) s.t. (f*g)_k <= M, g>=0."""
    n = len(f)
    current = f.copy()
    best_a = f.copy()
    best_score = _score_norm(f)

    M = float(_full_convolve(current, current).max())

    for it in range(n_max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Build sparse constraint matrix
        # A[k, i] = f[k-i] if 0 <= k-i < n
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end >= i_start:
                idx_range = range(i_start, i_end + 1)
                rows.extend(range(k, k + 1) * (i_end - i_start + 1))
                cols.extend(idx_range)
                vals.extend(current[k - i] for i in idx_range)

        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        # LP: maximize sum(g) => minimize -sum(g)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if not result.success:
                break
            g = result.x
            g = np.maximum(g, 0)
            g_sum = float(g.sum())
            if g_sum < 1e-10:
                break
            g_norm = g / g_sum
        except Exception:
            break

        # Blend
        t_actual = t_step
        for t_try in [t_actual, t_actual * 0.5, t_actual * 0.25, t_actual * 0.1]:
            candidate = (1 - t_try) * current + t_try * g_norm
            candidate = np.maximum(candidate, 0)
            csum = float(candidate.sum())
            if csum < 1e-10:
                continue
            candidate = candidate / csum
            sc = _score_norm(candidate)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                current = candidate.copy()
                M = float(_full_convolve(current, current).max())
                break

        if time.monotonic() > (deadline if deadline else 0):
            break

    return best_a, best_score


def _cosine_anneal_adam(a_init, deadline, lr_start=0.15, lr_end=0.001,
                        beta_start_frac=0.05, beta_end_frac=1e-5,
                        n_max_iter=100000):
    """Adam with cosine-annealed learning rate and beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c0 = float(c.max())
    beta_start = beta_start_frac * max_c0
    beta_end = beta_end_frac * max_c0

    total_time_deadline = deadline
    est_iters_left = int((deadline - time.monotonic()) / 0.005)
    total_iters = max(n_max_iter, est_iters_left)

    while time.monotonic() < deadline and t < total_iters:
        t += 1
        a = _softmax_fn(z)
        grad, c, w = _compute_gradient(a, beta_start)

        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        # Cosine annealing
        progress = min(1.0, t / total_iters)
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))
        beta = beta_start + (beta_end - beta_start) * (1 - np.cos(np.pi * min(1.0, t / total_iters)))
        beta = max(beta_end, beta)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score

            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 2048
        parent = _make_profile(n, 'linear', 1.5)
        print(f"No parent, using linear profile n={n}")

    # --- Phase 0: Evaluate initial candidates (0-30s) ---
    phase0_end = min(deadline, t_start + 20.0)
    candidates = []

    sc_parent = _score_norm(parent)
    candidates.append((sc_parent, parent.copy(), len(parent), "parent"))
    update_best(parent, sc_parent, "parent")

    for sn in [512, 1024, 2048, 4096]:
        if time.monotonic() > phase0_end:
            break
        profiles = [
            _make_profile(sn, 'linear', 1.0),
            _make_profile(sn, 'linear', 1.5),
            _make_profile(sn, 'linear', 2.0),
            _make_profile(sn, 'quadratic', 1.0),
            _make_profile(sn, 'cosine', 0.8),
            _make_profile(sn, 'gaussian', 2.0),
        ]
        for idx, p in enumerate(profiles):
            sc = _score_norm(p)
            candidates.append((sc, p, sn, f"p{idx}_n{sn}"))
            update_best(p, sc, f"p{idx}_n{sn}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: Adam with aggressive continuation on parent (20s - 400s) ---
    phase1_end = min(deadline, t_start + 400.0)
    p1_deadline = phase1_end
    p_best, sc_best = _adam_optimize(parent, p1_deadline, beta_frac=0.05, lr=0.12,
                                     n_max_iter=100000, beta_decay=0.92,
                                     beta_decay_interval=30, min_beta_frac=1e-5)
    update_best(p_best, sc_best, "adam_phase1")

    # --- Phase 2: LP blending (400s - 600s) ---
    phase2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        n_best = len(best_seq)
        if n_best <= 2048:  # LP is expensive for large n
            p_lp, sc_lp = _lp_blend_step(best_seq, n_max_iter=30, t_step=0.2,
                                         deadline=phase2_end)
            update_best(p_lp, sc_lp, f"lp_blend_n{n_best}")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Cosine-annealed Adam (600s - 900s) ---
    phase3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        p_ca, sc_ca = _cosine_anneal_adam(best_seq, phase3_end,
                                          lr_start=0.08, lr_end=0.0005,
                                          beta_start_frac=0.02, beta_end_frac=1e-6,
                                          n_max_iter=100000)
        update_best(p_ca, sc_ca, "cosine_adam")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Try larger length (900s - 1200s) ---
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_curr = len(best_seq)
        target_n = min(4096, n_curr * 2)
        if target_n > n_curr:
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase4_end - 10:
                p_big, sc_big = _adam_optimize(p_interp, phase4_end,
                                               beta_frac=0.03, lr=0.08,
                                               n_max_iter=50000, beta_decay=0.93,
                                               beta_decay_interval=40, min_beta_frac=1e-5)
                update_best(p_big, sc_big, f"adam_n{target_n}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Subgradient polish (1200s - 1450s) ---
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        p_sg, sc_sg = _subgradient_step(best_seq, n_steps=200, lr=0.005,
                                        deadline=phase5_end)
        update_best(p_sg, sc_sg, "subgrad")

        # Additional Adam with very small beta
        if time.monotonic() < phase5_end - 5:
            p_fin, sc_fin = _adam_optimize(best_seq, phase5_end,
                                           beta_frac=0.002, lr=0.03,
                                           n_max_iter=30000, beta_decay=0.97,
                                           beta_decay_interval=20, min_beta_frac=1e-7)
            update_best(p_fin, sc_fin, "adam_small_beta")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Multi-restart (1450s - 1700s) ---
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = (phase6_end - time.monotonic()) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 5:
                break
            noise_scale = [0.05, 0.1, 0.15, 0.25][i]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase6_end - 5, time.monotonic() + max(5.0, per_restart))
            p_r, sc_r = _adam_optimize(p_start, per_dl, beta_frac=0.01, lr=0.06,
                                       n_max_iter=20000, beta_decay=0.93,
                                       beta_decay_interval=30, min_beta_frac=1e-5)
            update_best(p_r, sc_r, f"restart{i}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final polish (1700s - deadline) ---
    if time.monotonic() < deadline and best_seq is not None:
        per_dl = min(deadline - 5, time.monotonic() + 50.0)
        p_final, sc_final = _adam_optimize(best_seq, per_dl,
                                           beta_frac=0.001, lr=0.01,
                                           n_max_iter=10000, beta_decay=0.98,
                                           beta_decay_interval=15, min_beta_frac=1e-8)
        update_best(p_final, sc_final, "final_polish")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(256, 'linear', 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(2048, 'linear', 1.5)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
