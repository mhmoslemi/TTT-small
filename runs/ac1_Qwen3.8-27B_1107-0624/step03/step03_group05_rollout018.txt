def _project_simplex(v):
    """Project onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a_init, deadline, beta_start=0.03, max_iter=20000):
    """Momentum gradient descent on the simplex with smooth-max."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    beta = beta_start
    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        if no_imp > max(2000, 3 * n):
            a = best_a.copy()
            v = np.zeros(n)
            beta = beta_start
            lr = base_lr
            no_imp = 0
        
        _, grad_a = _smooth_max_and_grad_a(a, beta)
        # Project gradient to simplex tangent space
        grad = grad_a - float(np.mean(grad_a))
        
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new
        
        sc = _score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.95)
        
        if it % 200 == 0:
            beta = max(1e-6, beta * 0.7)
    
    return best_a, best_score


def _coord_descent_polish(a_init, deadline, max_iter=200000):
    """Coordinate descent: randomly transfer mass between pairs."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        if no_imp > 10000:
            break
        
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try transferring a small fraction of mass from i to j
        for frac in [0.02, 0.1, 0.3]:
            delta = frac * a[i]
            if delta <= 0:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_try.copy()
                a = a_try.copy()
                no_imp = 0
                break
        else:
            no_imp += 1
    
    return best_a, best_score


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
