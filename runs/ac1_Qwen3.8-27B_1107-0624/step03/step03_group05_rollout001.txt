def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]

        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()

def _cutting_plane_lp(a, top_k=50, max_iter=100, inflate=1.0005, blend=0.3):

    best_a = a.copy()

        M = float(c.max()) * inflate
        
        # Get top-k lags
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:top_k]

        A = np.zeros((len(active), n))

            # For b[j] to appear: k-j must be in [0, n-1], so j in [k-n+1, k]
            # A[idx, j] = a[j] if 0 <= k-j < n, else a[k-j] if 0 <= j-(n-1-k) ...

            # So (a*b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*b[k-j]
            # = sum_{j: max(0,k-n+1)<=j<=min(k,n-1)} a[j]*b[k-j]
            # The coefficient of b[j] in (a*b)[k] is: a[k-j] if 0<=k-j<n (i.e., k-n+1<=j<=k), else 0
            for j in range(max(0, k - n + 1), min(k + 1, n)):
                A[idx, j] = a[k - j]

        b_ub = np.full(len(active), M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-10:

            continue

            a = a_new

            a = best_a.copy()
    
    return best_a
