    n = len(a)

        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn
    return c

"""
Multi-scale adaptive smooth-max optimization with progressive length scaling,
adaptive temperature annealing, multi-peak weighting, and LP-based spreading.
Targets sub-1.503 score for the convolution flatness objective.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    u = a / s
    c = _fft_convolve(u)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta, top_k_bias=None):
    mval = float(c.max())
    x = np.clip((c - mval) / max(beta, 1e-15), -50.0, 50.0)
    ex = np.exp(x)
    if top_k_bias is not None and top_k_bias > 1:
        sorted_idx = np.argsort(c)[::-1]
        k = min(int(top_k_bias), len(c))
        for rank, idx in enumerate(sorted_idx[:k]):
            ex[idx] *= (1.0 + 0.5 / (rank + 1))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    return w


def _grad_conv_weighted(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_smooth_max(a, deadline, max_iter=50000, lr=0.02, beta_init=None,
                     beta_min=0.003, decay=0.985, top_k_bias=1.5,
                     restart_noise=0.4, verbose=False):
    n = len(a)
    a = np.maximum(a, 1e-12).copy()
    a = a / a.sum()

    c = _fft_convolve(a)
    if beta_init is None:
        beta = 0.04 * float(c.max())
    else:
        beta = beta_init

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _smooth_max_weights(c, beta, top_k_bias)
        grad_a = _grad_conv_weighted(a, w)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -55.0, 55.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        s_new = float(2.0 * n * c_new.max())

        if s_new < best_score - 1e-12:
            best_score = s_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr_cur = max(0.001, lr_cur * 0.95)
            if no_imp > max(4000, 5 * n):
                rng_r = np.random.default_rng(7919 * (1 + it))
                z = np.log(best_a) + restart_noise * rng_r.normal(size=n)
                z = np.clip(z, -55.0, 55.0)
                z -= float(np.mean(z))
                a = np.exp(z)
                a = a / a.sum()
                beta = max(0.03 * float(_fft_convolve(a).max()), beta_min * 5)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new
        beta = max(beta_min, beta * decay)

        if verbose and it % 10000 == 0:
            print(f"        Adam it={it}: best={best_score:.10f}, beta={beta:.3e}, n_imp={no_imp}")

    return best_a, best_score


def _multi_subgradient(a, deadline, max_iter=10000, top_k=5, verbose=False):
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    base_lr = 0.15 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k = min(top_k, len(c))
        top_indices = np.argsort(c)[::-1][:k]
        c_sorted = c[top_indices]
        weights = np.exp(5.0 * (c_sorted - c_sorted[0]) / max(c_sorted[0], 1e-10))
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            idx = int(idx)
            j1_start = max(0, idx - n + 1)
            j1_end = min(idx, n - 1)
            if j1_start <= j1_end:
                positions = np.arange(j1_start, j1_end + 1)
                sg[positions] += weights[i] * a[idx - positions]
            j2_start = max(idx, 0)
            j2_end = min(idx + n - 1, n - 1)
            if j2_start <= j2_end:
                positions = np.arange(j2_start, j2_end + 1)
                sg[positions] += weights[i] * a[positions - idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break
        a = a_new

    return best_a, best_score


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b_vec = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * len(a) * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        n = len(a)
        c = _fft_convolve(a)
        M = float(c.max()) * inflate_cur
        g = _lp_spread(a, M)
        if g is None:
            break
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.008, inflate_cur * 1.04)
            if no_imp > 100:
                break
            continue
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = min(1.003, inflate_cur * 0.95)
            if no_imp > max(150, n // 2):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        a = a_new

    return best_a, best_score


def _interpolate_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        s = p.sum()
        return p / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-10)
    s = p_new.sum()
    return p_new / s if s > 1e-30 else np.ones(n_new) / n_new


def _make_profile(n, kind="uniform", param=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == "linear":
        a = 1.0 + param * u
    elif kind == "cubic":
        a = 1.0 + param * u - 0.5 * param * u ** 3
    elif kind == "cosine":
        a = 1.0 + param * np.cos(np.pi * t)
    elif kind == "bell":
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * param ** 2))
    elif kind == "arch":
        a = 1.0 - param * u ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _optimization_phase(a, deadline, label=""):
    """Run a full optimization pipeline on sequence a."""
    n = len(a)
    c0 = _fft_convolve(a)
    s0 = float(2.0 * n * c0.max())
    best_a = a.copy()
    best_sc = s0
    if label:
        print(f"    {label} init: {s0:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_sc

    t_adam1 = min(remaining * 0.4, 300.0)
    t_lp = min(remaining * 0.2, 150.0)
    t_adam2 = min(remaining * 0.25, 200.0)
    t_sg = min(remaining * 0.1, 100.0)
    t_final = min(remaining * 0.05, 50.0)

    # Phase 1: Coarse Adam
    d1 = time.monotonic() + t_adam1
    a1, s1 = _adam_smooth_max(best_a, d1, max_iter=40000, lr=0.025,
                              beta_init=0.04 * c0.max(), beta_min=0.005,
                              decay=0.985, top_k_bias=1.5)
    if s1 < best_sc:
        best_sc = s1
        best_a = a1
    if label:
        print(f"    {label} Adam1: {best_sc:.10f}")

    # Phase 2: LP refinement
    if time.monotonic() < deadline - 30:
        d2 = time.monotonic() + min(t_lp, deadline - time.monotonic() - 30)
        a2, s2 = _lp_refine(best_a, d2, max_iter=1500, inflate=1.001, blend=0.25)
        if s2 < best_sc:
            best_sc = s2
            best_a = a2
        if label:
            print(f"    {label} LP: {best_sc:.10f}")

    # Phase 3: Fine Adam
    if time.monotonic() < deadline - 20:
        d3 = time.monotonic() + min(t_adam2, deadline - time.monotonic() - 20)
        c3 = _fft_convolve(best_a)
        a3, s3 = _adam_smooth_max(best_a, d3, max_iter=30000, lr=0.01,
                                  beta_init=0.015 * c3.max(), beta_min=0.003,
                                  decay=0.99, top_k_bias=1.2, restart_noise=0.2)
        if s3 < best_sc:
            best_sc = s3
            best_a = a3
        if label:
            print(f"    {label} Adam2: {best_sc:.10f}")

    # Phase 4: Multi-subgradient
    if time.monotonic() < deadline - 10:
        d4 = time.monotonic() + min(t_sg, deadline - time.monotonic() - 10)
        a4, s4 = _multi_subgradient(best_a, d4, max_iter=8000, top_k=5)
        if s4 < best_sc:
            best_sc = s4
            best_a = a4
        if label:
            print(f"    {label} SG: {best_sc:.10f}")

    # Phase 5: Very fine Adam
    if time.monotonic() < deadline - 5:
        d5 = deadline - 5
        c5 = _fft_convolve(best_a)
        a5, s5 = _adam_smooth_max(best_a, d5, max_iter=20000, lr=0.005,
                                  beta_init=0.008 * c5.max(), beta_min=0.002,
                                  decay=0.992, top_k_bias=1.0, restart_noise=0.1)
        if s5 < best_sc:
            best_sc = s5
            best_a = a5
        if label:
            print(f"    {label} Adam3: {best_sc:.10f}")

    return best_a, best_sc
