    # M = hankel(w[:n], w[n-1:])
    
    # Use cvxpy to solve
    a_var = cp.Variable(n)

    M = np.zeros((n, n))

    prob.solve()
    
    if a_var.value is not None:
        return np.maximum(a_var.value, 0)
    return np.ones(n) / n

def _hankel_matvec(w, a):
    """Compute M @ a where M[i,j] = w[i+j]."""
    n = len(a)
    # M[i,j] = w[i+j]
    # (M @ a)[i] = sum_j w[i+j] a[j]
    # Let b = reverse(a), so b[k] = a[n-1-k]
    # (M @ a)[i] = sum_j w[i+j] b[n-1-j]
    # Let t = n-1-j, j = n-1-t
    # = sum_t w[i + n-1-t] b[t]
    # This is the cross-correlation of w and b at lag (i + n - 1)
    # corr(w, b)[l] = sum_t w[t+l] b[t] (for valid t)
    # So (M @ a)[i] = corr(w, b)[i + n - 1]
    
    b = a[::-1]
    # Cross-correlation: corr[w, b] = ifft(conj(fft(w)) * fft(b))
    # But we need to be careful about the definition
    
    # Actually, let's use the definition:
    # (M @ a)[i] = sum_j w[i+j] * a[j] for j = 0, ..., n-1
    # This is a "sliding dot product" of w with a, starting at position i.
    
    # We can compute this using FFT:
    # Define W = w[0], w[1], ..., w[2n-2] (length 2n-1)
    # Define A = a[0], a[1], ..., a[n-1] (length n)
    # (M @ a)[i] = sum_j W[i+j] * A[j] for j=0..n-1
    # = sum_j W[i+j] * A[j]
    # Let's reverse A: A_rev[t] = A[n-1-t]
    # = sum_j W[i+j] * A_rev[n-1-j]
    # Let s = n-1-j, j = n-1-s:
    # = sum_s W[i+n-1-s] * A_rev[s]
    # This is the cross-correlation of W and A_rev at lag (i+n-1):
    # xcorr[W, A_rev][lag] = sum_s W[s+lag] * A_rev[s]
    # So (M @ a)[i] = xcorr[W, A_rev][i+n-1]
    
    # Using FFT: xcorr[W, A_rev] = ifft(fft(W) * conj(fft(A_rev)))
    # (with appropriate zero-padding)
    
    N = _next_pow2(2 * (2 * n - 1) - 1)  # pad to at least 2*(2n-1)
    # Actually, we need the cross-correlation at lags 0 to 2n-2
    # The full cross-correlation has length (2n-1) + n - 1 = 3n-2
    # We need lags n-1 to 2n-2 (that's n values)
    
    # Simpler: just use direct computation for n <= 1024
    # (M @ a)[i] = sum_j w[i+j] * a[j]
    result = np.zeros(n)
    for i in range(n):
        result[i] = np.dot(w[i:i+n], a)
    return result

    # or just use a fixed step size based on ||w||_inf
    # The Lipschitz constant of the gradient is 2 * lambda_max(M)
    # lambda_max(M) <= sum_k w[k] (by Gershgorin or simple bound)
    # More tightly: lambda_max(M) <= max_i sum_j w[i+j]
    
    # Simple bound: lambda_max(M) <= (sum w) (since all entries of M are non-negative)
    # Actually, lambda_max(M) <= max_row_sum (by Gershgorin for non-negative entries)
    # max_row_sum = max_i sum_j w[i+j]
    
    # For safety, use: lr = 1 / (2 * max_row_sum)
    row_sums = np.array([np.sum(w[i:i+n]) for i in range(n)])
    max_row_sum = row_sums.max()
    if max_row_sum < 1e-15:
        return a
    lr = 0.9 / (2.0 * max_row_sum)  # FISTA step
    
    # FISTA (accelerated projected gradient)
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_obj = float(np.dot(a, _hankel_matvec(w, a)))
    
    for it in range(max_iter):
        grad = 2.0 * _hankel_matvec(w, y)
        a_new = _project_simplex(y - lr * grad)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        t = t_new
        a = a_new
        
        obj = float(np.dot(a, _hankel_matvec(w, a)))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

        k <<= 1

    n = len(v)

        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _hankel_matvec(w, a):
    """Compute M @ a where M[i,j] = w[i+j]. Uses FFT cross-correlation."""
    n = len(a)
    N = _next_pow2(n + (2 * n - 1) - 1)
    fa = np.fft.fft(a, N)
    fw = np.fft.fft(w, N)
    xcorr = np.fft.ifft(fa * np.conj(fw)).real
    return xcorr[:n]

def _solve_qp_fista(w, max_iter=3000, tol=1e-10):
    """Solve min a^T M a s.t. a >= 0, sum(a) = 1 where M[i,j] = w[i+j]."""
    n = (len(w) + 1) // 2
    a = np.ones(n) / n
    
    # Step size
    row_sums = np.array([np.sum(w[i:i+n]) for i in range(n)])
    max_row_sum = float(row_sums.max())
    if max_row_sum < 1e-15:
        return a
    lr = 0.9 / max_row_sum  # Since gradient of a^T M a is 2Ma, lip = 2*max_row_sum, step = 1/lip * 0.9
    
    # Actually, the objective is a^T M a, gradient is (M + M^T) a = 2 M a (since M is symmetric)
    # Lipschitz constant of gradient is 2 * lambda_max(M) <= 2 * max_row_sum
    # Step size should be < 1 / (2 * max_row_sum) for convergence
    lr = 0.95 / (2.0 * max_row_sum)
    
    # FISTA
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_obj = float(np.dot(a, _hankel_matvec(w, a)))
    
    for it in range(max_iter):
        grad = 2.0 * _hankel_matvec(w, y)
        a_new = _project_simplex(y - lr * grad)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)

        a_prev = a_new
        a = a_new

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Convergence check
        if it > 0 and abs(best_obj - obj) < tol * (1 + abs(obj)):
            # Check if we've converged
            pass
    
    return best_a

def _continuation_optimize(a_init, deadline, max_outer=500, max_inner=3000):
    """Main continuation algorithm."""
    a = _project_simplex(a_init.copy())
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Temperature schedule: start low (near-uniform weights), increase
    # beta = 0 means uniform weights, beta → ∞ means only peak is weighted
    beta = 0.1
    
    for outer_it in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        c_max = float(c.max())
        c_min = float(c[c > 0].min()) if np.any(c > 0) else 0.0
        
        # Compute Boltzmann weights
        # w[k] = exp(beta * c[k] / c_max)
        # Normalize w so that sum(w) = 2n-1 (to keep the scale consistent)
        scaled_c = c / c_max
        w = np.exp(beta * scaled_c)
        w = w / w.sum() * (2 * n - 1)
        
        # Add a small regularization to avoid degenerate weights
        w = w + 1e-10
        
        # Solve QP
        a_qp = _solve_qp_fista(w, max_iter=max_inner)
        
        # Blend
        blend = min(0.5, 0.1 + 0.3 * outer_it / max_outer)
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = _project_simplex(a_new)
        
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            # Slow down beta increase on success
            beta *= 1.02
        else:
            a = a_new  # Still update to explore
            if sc < best_score * 1.01:
                beta *= 1.02
            else:
                beta *= 1.05
        
        # Increase beta (continuation)
        beta = min(beta * 1.03, 50.0)
        
        if outer_it % 10 == 0:
            print(f"  Outer {outer_it}: score={best_score:.10f}, beta={beta:.3f}")
    
    return best_a, best_score

def _gradient_polish(a, deadline, max_iter=10000):
    """Simple gradient polish using smooth max."""
    a = _project_simplex(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    
    beta = 0.02 * float(_fft_convolve(a).max())
    lr = 0.05
    v = np.zeros(n)
    momentum = 0.9
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        c_max = float(c.max())
        # Smooth max weights
        x = (c - c_max) / beta
        w = np.exp(x)
        w = w / w.sum()
        
        # Gradient of smooth max: grad = convolve(w, a) (approximately)
        # More precisely: d/d a[i] smooth_max(c) ≈ sum_k w[k] * 2 * a[k-i] (for valid k-i)
        # = 2 * (w_rev * a_convolve)[i] ... let me use FFT
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        a_nest = a + momentum * v
        c_nest = _fft_convolve(a_nest)
        c_max_nest = float(c_nest.max())
        x_nest = (c_nest - c_max_nest) / beta
        w_nest = np.exp(x_nest)
        w_nest = w_nest / w_nest.sum()
        fw_nest = np.fft.rfft(w_nest, N)
        fa_nest_rev = np.fft.rfft(a_nest[::-1], N)
        h_full_nest = np.fft.irfft(fw_nest * fa_nest_rev, N)
        h_nest = h_full_nest[n - 1:2 * n - 1]
        grad_nest = 2.0 * h_nest
        
        v = momentum * v - lr * grad_nest
        a_new = _project_simplex(a + v)
        a = a_new
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if it % 100 == 0:
            beta = max(1e-10, beta * 0.85)
    
    return best_a, best_score

    a = np.maximum(a, 1e-12)

"""
Hankel QP continuation method for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key insight: the surrogate QP min a^T M a s.t. a>=0, sum(a)=1 has M as a Hankel
matrix (M[i,j]=w[i+j]), enabling O(n log n) matvecs via FFT cross-correlation.
Continuation on Boltzmann weights gradually sharpens focus on peak locations.
Combined with projected gradient polish for final refinement.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:

    return w / s

            break

        a_nest = a + momentum * v
        c_nest = _fft_convolve(a_nest)
        c_max = float(c_nest.max())
        x = (c_nest - c_max) / beta
        ex = np.exp(x)
        w = ex / ex.sum()

        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_nest[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a, best_score


def _peak_targeted_polish(a, deadline, max_iter=15000, top_k=8):
    """Gradient that specifically targets the top-k peaks."""
    a = _project_simplex(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    c_init = _fft_convolve(a)
    beta = max(1e-8, 0.02 * float(c_init.max()))
    lr = 0.03
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0

    N = _next_pow2(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        c_nest = _fft_convolve(a_nest)
        c_max = float(c_nest.max())

        # Combine uniform smooth-max with peak-focused weight
        x = (c_nest - c_max) / beta
        w_smooth = np.exp(x)
        w_smooth = w_smooth / w_smooth.sum()

        sorted_idx = np.argsort(c_nest)[::-1]
        w_peak = np.zeros_like(c_nest)
        top_idx = sorted_idx[:top_k]
        w_peak[top_idx] = 1.0 / top_k
        w = 0.6 * w_smooth + 0.4 * w_peak
        w = w / w.sum()

        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_nest[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                a = best_a.copy()
                v = np.zeros(n)
                no_imp = 0

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.85)

    return best_a, best_score


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, sigma=0.12, sep=0.7):
    t = np.linspace(0.0, 1.0, n)
    c1 = (1.0 - sep) / 2.0
    c2 = (1.0 + sep) / 2.0
    a = np.exp(-0.5 * ((t - c1) / sigma) ** 2) + np.exp(-0.5 * ((t - c2) / sigma) ** 2)
    a = a + 0.05
    return a / a.sum()


def _interpolate(profile, n_new):
    n0 = profile.size
    if n0 == n_new:
        return profile / profile.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, profile)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Hankel QP continuation + peak-targeted gradient polish.
    Uses FFT-based Hankel matvec for fast QP subproblem solves.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                best_score = sc
                best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = list(dict.fromkeys([768, 640, 512, 896, 1024]))
    if parent is not None:
        lengths = list(dict.fromkeys([parent.size] + lengths))

    print(f"Lengths: {lengths}, budget: {budget_s}s")

    for n_t in lengths:
        if time.monotonic() > deadline - 80:
            break

        elapsed = time.monotonic() - t_start
        remaining = deadline - time.monotonic()
        n_left = lengths.index(n_t)
        # Allocate time: more for earlier lengths
        phase_time = remaining * 0.75 / max(1, len(lengths) - n_left)
        phase_time = max(30.0, min(phase_time, remaining * 0.6))
        phase_time = min(phase_time, remaining - 60)
        if phase_time < 20:
            continue

        print(f"\n=== n={n_t}, phase budget={phase_time:.0f}s ===")

        # Initialization
        if parent is not None:
            a_init = _interpolate(parent, n_t)
        else:
            a_init = _make_u_profile(n_t, 2.0)

        sc_init = _score(a_init)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()
        print(f"  Init: {sc_init:.10f}")

        # Strategy 1: Continuation with Boltzmann weights
        t1 = time.monotonic()
        dl1 = t1 + phase_time * 0.35
        a1, s1 = _continuation_optimize(a_init, dl1, max_outer=200, max_inner=4000,
                                        weight_mode='boltzmann', beta_init=0.3,
                                        beta_max=50.0, blend_init=0.4)
        print(f"  ContBoltz: {s1:.10f} ({time.monotonic()-t1:.1f}s)")
        if s1 < best_score:
            best_score = s1
            best_seq = a1.copy()

        # Strategy 2: Continuation with power weights
        t2 = time.monotonic()
        dl2 = t2 + phase_time * 0.2
        a2, s2 = _continuation_optimize(a1, dl2, max_outer=150, max_inner=3000,
                                        weight_mode='power', beta_init=1.0,
                                        beta_max=8.0, blend_init=0.3)
        print(f"  ContPower: {s2:.10f} ({time.monotonic()-t2:.1f}s)")
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()

        # Strategy 3: Active set continuation
        t3 = time.monotonic()
        dl3 = t3 + phase_time * 0.15
        a3, s3 = _continuation_optimize(a2, dl3, max_outer=100, max_inner=4000,
                                        weight_mode='active', beta_init=1.0,
                                        beta_max=25.0, blend_init=0.35)
        print(f"  ContActive: {s3:.10f} ({time.monotonic()-t3:.1f}s)")
        if s3 < best_score:
            best_score = s3
            best_seq = a3.copy()

        # Strategy 4: Peak-targeted gradient
        t4 = time.monotonic()
        dl4 = t4 + phase_time * 0.15
        a4, s4 = _peak_targeted_polish(a3, dl4, max_iter=15000, top_k=10)
        print(f"  PeakGrad: {s4:.10f} ({time.monotonic()-t4:.1f}s)")
        if s4 < best_score:
            best_score = s4
            best_seq = a4.copy()

        # Strategy 5: Standard gradient polish
        t5 = time.monotonic()
        dl5 = t5 + max(5.0, phase_time * 0.1)
        a5, s5 = _gradient_polish(a4, dl5, max_iter=15000)
        print(f"  GradPolish: {s5:.10f} ({time.monotonic()-t5:.1f}s)")
        if s5 < best_score:
            best_score = s5
            best_seq = a5.copy()

        # Strategy 6: One more continuation round from the best
        if time.monotonic() < deadline - 40:
            t6 = time.monotonic()
            dl6 = t6 + min(30.0, deadline - t6 - 20)
            a6, s6 = _continuation_optimize(a5, dl6, max_outer=80, max_inner=5000,
                                            weight_mode='boltzmann', beta_init=5.0,
                                            beta_max=60.0, blend_init=0.25)
            print(f"  ContFinal: {s6:.10f} ({time.monotonic()-t6:.1f}s)")
            if s6 < best_score:
                best_score = s6
                best_seq = a6.copy()

    # Final intensive polish on overall best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final Polish ===")
        t_final = time.monotonic()
        dl_final = deadline - 10

        # Gradient polish
        half_time = t_final + (dl_final - t_final) * 0.5
        a_f1, s_f1 = _peak_targeted_polish(best_seq, half_time, max_iter=20000, top_k=6)
        print(f"  Final PeakGrad: {s_f1:.10f}")
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()

        # One more gradient
        if time.monotonic() < dl_final - 5:
            a_f2, s_f2 = _gradient_polish(best_seq, dl_final - 3, max_iter=10000)
            print(f"  Final Grad: {s_f2:.10f}")
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()

    # Validate and return
    if best_seq is None:
        best_seq = _make_u_profile(768, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_u_profile(768, 2.0)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"  evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"  Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.0f}s")
    return [float(x) for x in final_seq]
