```python
"""
Improved multi-method optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LSE gradient, LP spreading, peak-shaving, and adaptive restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score_normalized(a):
    """Score with sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _score_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))
```

```python
"""
Convex optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Exploits the convexity of the objective (max of convex quadratics over simplex).
Uses projected subgradient with Nesterov acceleration, LSE smoothing,
LP-based direction finding, and adaptive restarts.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient(a):
    """Compute subgradient of max_k conv(a)_k at point a (sum a = 1)."""
    n = a.size
    c = _fft_convolve(a)
    m_val = float(c.max())
    # Use all lags within epsilon of max for a less noisy subgradient
    eps = 1e-10
    max_idx = np.where(c >= m_val - eps)[0]
    if len(max_idx) == 0:
        max_idx = np.array([np.argmax(c)])
    # Gradient: for each active lag k, grad_i += 2 * a_{k-i}
    # Use FFT for efficiency
    w = np.zeros(2 * n - 1)
    for k in max_idx:
        w[k] = 1.0 / len(max_idx)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n - 1:2 * n - 1]
    return grad, m_val


def _lse_gradient(a, T):
    """Compute smoothed gradient using log-sum-exp."""
    n = a.size
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n - 1:2 * n - 1]
    return grad, m_val


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_find_direction(a, M):
    """LP: maximize sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _nesterov_optimize(a, deadline, max_iter=50000, T_start=0.03, T_end=1e-4):
    """Nesterov-accelerated LSE gradient with projected steps."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    # Nesterov momentum
    a_prev = a.copy()
    y = a.copy()

    T = T_start
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    lr_base = 0.5
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        beta = (it - 1) / (it + 2.0) if it > 0 else 0.0
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient
        lr = lr_base / (1.0 + 0.01 * it)
        grad, m_val = _lse_gradient(y, T)

        step = y - lr * grad
        a_new = _project_simplex(step)

        a_new = np.maximum(a_new, 1e-12)

        a = a_new

                a_prev = a.copy()
                no_imp = 0

def _peak_shave_optimize(a, deadline, max_iter=30000, top_k=3):
    """Optimize by directly targeting top-k peaks with weighted subgradient."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size

    a = a / a.sum()

    best_a = a.copy()

    lr = 0.02
    momentum = np.zeros(n)
    mom_decay = 0.7
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())

        # Find top-k peaks and weight them
        sorted_idx = np.argsort(c)[::-1]
        top_indices = sorted_idx[:top_k]
        # Weight by how prominent each peak is
        weights = np.zeros(2 * n - 1)
        max_w = c[top_indices[0]]
        for i, k in enumerate(top_indices):
            weights[k] = (c[k] / max_w) ** 2  # Square to emphasize the top peak

        # Compute weighted subgradient
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(weights, N)
        fa_rev = np.fft.rfft(a[::-1], N)

        grad = 2.0 * h_full[n - 1:2 * n - 1]

        # Momentum step
        lr_cur = lr / (1.0 + 0.005 * it)
        momentum = mom_decay * momentum - lr_cur * grad
        a_new = a + momentum
        a_new = _project_simplex(a_new)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(5000, 5 * n):
                a = best_a.copy()
                momentum = np.zeros(n)
                lr = 0.02
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    except Exception:
        pass

    # Lengths to explore
    lengths = [1024, 2048]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)

    print(f"Lengths to try: {lengths}")

    # Generate diverse initializations for each length
    def get_candidates(n_t):
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            # Perturbed versions
            for pi in range(3):
                p = _interpolate_profile(parent, n_t)
                p_pert = p * (1.0 + 0.1 * rng.normal(size=n_t))
                p_pert = np.maximum(p_pert, 0)
                if p_pert.sum() > 1e-10:
                    cands.append(p_pert / p_pert.sum())
        # Analytical profiles
        cands.append(np.ones(n_t) / n_t)
        cands.append(_make_gaussian(n_t, 0.35))
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_flat_with_edges(n_t, 0.4))
        cands.append(_make_two_bump(n_t, 0.2))
        cands.append(_make_triangle(n_t))
        return cands

    # ===== Phase 1: Quick Nesterov optimization on diverse candidates =====
    print("\n=== Phase 1: Nesterov LSE optimization ===")
    best_per_length = {}

    for n_t in lengths:
        if time.monotonic() > deadline - 600:
            break

        cands = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 600:
                break

            sc0 = _score_norm(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick Nesterov optimization
            nest_budget = min(80.0, deadline - time.monotonic() - 600.0)
            if nest_budget < 15:
                break
            nest_end = time.monotonic() + nest_budget

            a_nest = _nesterov_optimize(a0, nest_end, max_iter=30000,
                                        T_start=0.03, T_end=1e-4)
            sc_nest = _score_norm(a_nest)

            if np.isfinite(sc_nest) and sc_nest < best_score:
                best_score = sc_nest
                best_seq = a_nest.copy()
            if sc_nest < best_here_score:
                best_here_score = sc_nest
                best_here = a_nest

            print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> nesterov={sc_nest:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best Phase 1 for n={n_t}: {best_here_score:.10f}")

    # ===== Phase 2: Hybrid LP+LSE optimization =====
    print("\n=== Phase 2: Hybrid LP+LSE optimization ===")
    for n_t in lengths:
        if n_t not in best_per_length:
            # Use best candidate from phase 1 or default
            cands = get_candidates(n_t)
            a0 = cands[0]
            sc0 = _score_norm(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            best_per_length[n_t] = a0
            continue

        if time.monotonic() > deadline - 400:
            break

        a_in = best_per_length[n_t]
        sc_in = _score_norm(a_in)
        print(f"  n={n_t} start: score={sc_in:.10f}")

        hyb_budget = min(200.0, deadline - time.monotonic() - 400.0)
        if hyb_budget < 30:
            continue
        hyb_end = time.monotonic() + hyb_budget

        a_hyb = _hybrid_optimize(a_in, hyb_end, max_iter=50000)
        sc_hyb = _score_norm(a_hyb)

        if np.isfinite(sc_hyb) and sc_hyb < best_score:
            best_score = sc_hyb
            best_seq = a_hyb.copy()
        best_per_length[n_t] = a_hyb
        print(f"  n={n_t} hybrid: score={sc_hyb:.10f}")

    # ===== Phase 3: Peak-shaving optimization =====
    print("\n=== Phase 3: Peak-shaving subgradient ===")
    for n_t in lengths:
        if n_t not in best_per_length:
            continue
        if time.monotonic() > deadline - 250:
            break

        a_in = best_per_length[n_t]
        sc_in = _score_norm(a_in)
        print(f"  n={n_t} peak-shave start: score={sc_in:.10f}")

        ps_budget = min(100.0, deadline - time.monotonic() - 250.0)
        if ps_budget < 20:
            continue
        ps_end = time.monotonic() + ps_budget

        a_ps = _peak_shave_optimize(a_in, ps_end, max_iter=30000, top_k=3)
        sc_ps = _score_norm(a_ps)

        if np.isfinite(sc_ps) and sc_ps < best_score:
            best_score = sc_ps
            best_seq = a_ps.copy()
        best_per_length[n_t] = a_ps
        print(f"  n={n_t} peak-shave: score={sc_ps:.10f}")

    # ===== Phase 4: Fine-tuning with long Nesterov runs =====
    print("\n=== Phase 4: Fine-tuning Nesterov ===")
    for n_t in lengths:
        if n_t not in best_per_length:
            continue
        if time.monotonic() > deadline - 100:
            break

        a_in = best_per_length[n_t]
        sc_in = _score_norm(a_in)

        nt_budget = min(80.0, deadline - time.monotonic() - 100.0)
        if nt_budget < 20:
            continue
        nt_end = time.monotonic() + nt_budget

        a_nt = _nesterov_optimize(a_in, nt_end, max_iter=30000,
                                  T_start=0.01, T_end=1e-5)
        sc_nt = _score_norm(a_nt)

        if np.isfinite(sc_nt) and sc_nt < best_score:
            best_score = sc_nt
            best_seq = a_nt.copy()
        best_per_length[n_t] = a_nt
        print(f"  n={n_t} fine-tune: score={sc_nt:.10f}")

    # ===== Phase 5: Final LP polish on the overall best =====
    print("\n=== Phase 5: Final LP polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_final = best_seq.size
        if n_final <= 2048:
            lp_budget = min(60.0, deadline - time.monotonic() - 30.0)
            if lp_budget > 15:
                lp_end = time.monotonic() + lp_budget
                # Run hybrid with small blend for final polish
                a_final = _hybrid_optimize(best_seq, lp_end, max_iter=20000)
                sc_final = _score_norm(a_final)
                if np.isfinite(sc_final) and sc_final < best_score:
                    best_score = sc_final
                    best_seq = a_final.copy()
                print(f"  Final LP: score={sc_final:.10f}")

    # ===== Final validation =====
    if best_seq is None:
        best_seq = np.ones(1024) / 1024.0
        best_score = _score_norm(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(1024) / 1024.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
```
