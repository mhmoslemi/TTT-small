Active-set KKT optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 512:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _cross_corr(a, w):
    """Compute g_j = sum_m a_m * w_{j+m} for j=0,...,n-1.
    a has length n, w has length 2n-1.
    Uses: g = (reversed_a * w)[n-1 : 2n-1]
    """
    n = len(a)
    N = _next_pow2(3 * n - 1)
    ra = np.zeros(N)
    ra[:n] = a[::-1]
    wp = np.zeros(N)
    wp[:2*n-1] = w
    ra_fft = np.fft.rfft(ra)
    w_fft = np.fft.rfft(wp)
    conv = np.fft.irfft(ra_fft * w_fft, N)
    return conv[n-1 : 2*n-1]

def _grad_weighted_conv(w, a):
    """Gradient of sum_k w_k * c_k(a) with respect to a.
    Result is 2 * cross_corr(a, w).

def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    Minimize non-flatness of weighted gradient on support.

    # So the matrix G has rows indexed by support indices and columns by K

    sup_idx = np.where(support)[0]

    for ki, k in enumerate(K):

            val = k - i
            if 0 <= val <= n - 1:
                G[si, ki] = 2.0 * a[val]
    
    # Center G: subtract row means (to enforce flatness)
    G_mean = G.mean(axis=1, keepdims=True)
    G_c = G - G_mean
