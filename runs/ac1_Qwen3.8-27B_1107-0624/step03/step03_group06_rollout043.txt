```python
import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b=None):
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = len(a), len(b)
    if n * m <= 100000:
        return np.convolve(a, b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n+m-1]
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_cross_corr(a, w):
    """Compute grad[i] = sum_j a[j] * w[j+i] for i = 0..n-1.
    a has length n, w has length 2n-1.
    Returns vector of length n.

    n = len(a)

    N = _next_pow2(2 * n - 1 + n - 1)  # enough for full conv of length n and 2n-1

    fa = np.fft.rfft(a[::-1], N)
```

```python
def _dpss_first(n, W):
    """Compute the first discrete prolate spheroidal sequence of length n with bandwidth W.
    W is the half-bandwidth as a fraction of the sampling rate (0 < W < 0.5).
    """
    # Build the Toeplitz matrix T[j,k] = sinc(W*(j-k)) = sin(pi*W*(j-k)) / (pi*W*(j-k))
    # (with sinc(0) = 1)
    idx = np.arange(n)
    diff = idx[:, None] - idx[None, :]
    T = np.ones((n, n))
    mask = diff != 0
    T[mask] = np.sin(np.pi * W * diff[mask]) / (np.pi * W * diff[mask])
    
    # Eigendecomposition
    eigvals, eigvecs = np.linalg.eigh(T)
    # The largest eigenvalue corresponds to the first DPSS
    idx_max = np.argmax(eigvals)
    dpss = eigvecs[:, idx_max]
    dpss = np.abs(dpss)  # Make it positive (the first DPSS should be positive)
    # Actually, we should be careful - the sign might matter
    # The first DPSS has all same-sign elements
    if dpss.sum() < 0:
        dpss = -dpss
    dpss = np.maximum(dpss, 0)
    s = dpss.sum()
    if s > 0:
        dpss = dpss / s
    return dpss
```

```python
Minimizes 2n*max(conv(a,a))/sum(a)^2 using FISTA-based smooth alternating
minimization with DPSS initializations, multi-scale optimization, and LP polish.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b=None):
    """FFT-based linear convolution."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64).ravel()
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    if n * m <= 50000:
        return np.convolve(a, b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    return c


def _score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    idx = rho[-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_grad(a, w):
    """Compute grad[i] = sum_j a[j]*w[j+i] for i=0..n-1.
    
    a: length n
    w: length 2n-1
    Returns: length n
    """
    n = a.size
    N = _next_pow2(3 * n - 2)
    fa = np.fft.rfft(a[::-1], N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(fa * fw, N)
    return corr[n - 1:2 * n - 1]


def _fista_optimize_g(a, beta, max_iter=200, verbose=False):
    """Optimize g: minimize beta*log(sum exp((a*g)[k]/beta)) s.t. g>=0, sum(g)=1.
    Uses FISTA (accelerated projected gradient).
    """
    n = a.size
    g = np.ones(n) / n
    y = g.copy()
    t_fista = 1.0
    
    # Step size
    lr = 0.5 * beta  # safe step size
    
    for it in range(max_iter):
        # Compute c = conv(a, y)
        c = _fft_convolve(a, y)
        
        # Softmax weights
        c_shift = c - c.max()
        ex = np.exp(c_shift / beta)
        w = ex / ex.sum()
        
        # Gradient
        grad = _compute_grad(a, w)
        
        # FISTA update
        g_new = _project_simplex(y - lr * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        y = g_new + (t_fista - 1.0) / t_new * (g_new - g)
        y = _project_simplex(y)
        g = g_new
        t_fista = t_new
    
    return g


def _dpss_first(n, W):
    """Compute first DPSS (Slepian sequence) of length n with bandwidth W.
    
    W is half-bandwidth as fraction of sampling rate (0 < W < 0.5).
    Returns a non-negative sequence summing to 1.
    """
    idx = np.arange(n)
    diff = idx[:, None].astype(np.float64) - idx[None, :].astype(np.float64)
    T = np.ones((n, n), dtype=np.float64)
    mask = diff != 0
    T[mask] = np.sin(np.pi * W * diff[mask]) / (np.pi * W * diff[mask])
    
    eigvals, eigvecs = np.linalg.eigh(T)
    # Largest eigenvalue
    k = np.argmax(eigvals)
    v = np.abs(eigvecs[:, k])
    s = v.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return v / s
```

```python
"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using FISTA-based smooth alternating minimization
with DPSS (Slepian) initializations, multi-scale optimization, random restarts, and LP polish.

Key ideas:
1. DPSS sequences concentrate spectral energy in low frequencies -> flatter autocorrelation.
2. FISTA on the log-sum-exp smoothed subproblem is O(n log n) per iteration (vs O(n^2) for LP).
3. Multi-scale (coarse-to-fine) optimization escapes local minima.
4. Random perturbations from the best found provide diversity.
5. Final LP polish for exact subproblem accuracy.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64).ravel()
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    if n * m <= 100000:
        return np.convolve(a, b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    idx = rho[-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    """grad[i] = sum_j a[j]*w[j+i] for i=0..n-1, via FFT convolution."""
    n = a.size
    N = _next_pow2(3 * n - 2)
    fa = np.fft.rfft(a[::-1], N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(fa * fw, N)
    return corr[n - 1:2 * n - 1]


def _fista_optimize_g(a, beta, max_iter, lr_scale=0.5):
    """FISTA for: min_g beta*log(sum exp((a*g)[k]/beta)) s.t. g>=0, sum(g)=1."""
    n = a.size
    g = np.ones(n) / n
    y = g.copy()
    t_f = 1.0
    lr = lr_scale * beta
    lr_min = 1e-8

    prev_h = None
    for it in range(max_iter):
        c = _fft_convolve(a, y)
        c_shift = c - c.max()
        ex = np.exp(c_shift / beta)
        w = ex / ex.sum()
        grad = _compute_grad(a, w)

        g_new = _project_simplex(y - lr * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_f * t_f)) / 2.0
        y = g_new + (t_f - 1.0) / t_new * (g_new - g)
        y = _project_simplex(y)
        g = g_new
        t_f = t_new

    return g


def _dpss_first(n, W):
    """First DPSS/Slepian sequence of length n, bandwidth W (fraction of Nyquist)."""
    idx = np.arange(n, dtype=np.float64)
    diff = idx[:, None] - idx[None, :]
    T = np.ones((n, n), dtype=np.float64)
    mask = diff != 0
    arg = np.pi * W * diff[mask]
    T[mask] = np.sin(arg) / arg
    eigvals, eigvecs = np.linalg.eigh(T)
    k = np.argmax(eigvals)
    v = np.abs(eigvecs[:, k])
    s = v.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return v / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_ramp_rev(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** alpha + 1e-8
    return a / a.sum()


def _make_triangle(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.maximum(1.0 - np.abs(2.0 * t - 1.0), 1e-8)
    return a / a.sum()


def _make_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.maximum(0.5 * (1.0 + np.cos(np.pi * t)), 1e-8)
    return a / a.sum()


def _make_gauss(n, sigma=0.3):
    t = np.linspace(-1.0, 1.0, n)
    a = np.exp(-0.5 * (t / sigma) ** 2)
    return a / a.sum()


def _make_random(n, rng):
    a = rng.random(n) + 0.05
    return a / a.sum()


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _line_search(a, g, n_grid=15):
    """Find best t: a_new = (1-t)*a + t*g."""
    best_t = 0.0
    best_sc = _score(a)
    a_n = a / a.sum()
    g_n = g / g.sum()
    ts = np.linspace(0.02, 1.0, n_grid)
    for t in ts:
        a_new = (1.0 - t) * a_n + t * g_n
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t > 0:
        a_b = (1.0 - best_t) * a_n + best_t * g_n
        a_b = np.maximum(a_b, 0.0)
        a_b = a_b / a_b.sum()
        return a_b, best_sc
    return a_n, best_sc


def _fista_alterning(a0, deadline, max_outer=400, fista_iters=120,
                     beta_start=0.1, beta_end=0.002, beta_decay=0.97):
    """FISTA-based alternating minimization."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    beta = beta_start
    no_imp = 0

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        g = _fista_optimize_g(a, beta, max_iter=fista_iters)
        a_new, sc_new = _line_search(a, g)
        if sc_new < best_score - 1e-12:
            a = a_new
            best_score = sc_new
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                beta = max(beta_end, beta * beta_decay * 0.6)
            if no_imp > 100:
                beta = max(beta_end, beta * 0.5)
                no_imp = 0
        if outer % 40 == 39:
            beta = max(beta_end, beta * beta_decay)

    return best_a, best_score


def _lp_refine(a0, deadline, max_iter=400, inflate_start=1.01):
    """LP-based refinement."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a, a).max())
    no_imp = 0
    cur_inf = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        M = float(c.max())
        M_lp = M * cur_inf

        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success or res.x is None:
            no_imp += 1
        else:
            g = res.x
            sum_g = float(g.sum())
            if sum_g > 1.0 + 1e-6:
                g = g / sum_g
                a_n = a / a.sum()
                best_t = 0.0
                best_sc = best_score
                for t in np.linspace(0.05, 1.0, 9):
                    a_new = (1.0 - t) * a_n + t * g
                    a_new = np.maximum(a_new, 0.0)
                    s = a_new.sum()
                    if s < 1e-10:
                        continue
                    a_new = a_new / s
                    sc = float(2.0 * n * _fft_convolve(a_new, a_new).max())
                    if sc < best_sc:
                        best_sc = sc
                        best_t = t
                if best_t > 0 and best_sc < best_score - 1e-12:
                    a = (1.0 - best_t) * a_n + best_t * g
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_score = best_sc
                    best_a = a.copy()
                    no_imp = 0
                    if no_imp == 0 and it % 20 == 0:
                        cur_inf = max(1.001, cur_inf * 0.95)
                else:
                    no_imp += 1
            else:
                no_imp += 1

        if no_imp > 0 and no_imp % 25 == 0:
            cur_inf = min(0.05, cur_inf * 1.3)
        if no_imp > 150:
            a = best_a.copy()
            cur_inf = inflate_start
            no_imp = 0
            if inflate_start < 1.005:
                break

    return best_a, best_score


def _gradient_polish(a0, deadline, max_iter=5000):
    """Smooth-max gradient descent with momentum for final polish."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a, a).max())
    beta = 0.05 * float(_fft_convolve(a, a).max())
    lr = 0.02
    base_lr = lr
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a_mom = a + momentum * v
        c = _fft_convolve(a_mom, a_mom)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex
        grad = _compute_grad(a_mom, w)
        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue
        v = momentum * v - lr * grad
        a = _project_simplex(a + v)
        sc = float(2.0 * n * _fft_convolve(a, a).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = 0.05 * float(_fft_convolve(a, a).max())
                lr = base_lr
                no_imp = 0
        if it % 100 == 99:
            beta = max(1e-6, beta * 0.85)

    return best_a, best_score


def _random_perturb(a, rng, strength=0.1):
    """Add random perturbation to a (maintaining non-negativity and sum=1)."""
    n = a.size
    noise = rng.standard_normal(n) * strength
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s
```
