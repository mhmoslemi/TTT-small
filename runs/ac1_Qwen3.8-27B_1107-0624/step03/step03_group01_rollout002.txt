"""
Multi-phase optimization for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses smooth-max (logsumexp) approximation with adaptive beta continuation,
L-BFGS-B in log-softmax space, Adam polishing, Nesterov momentum,
and multi-restart strategy. Leverages parent sequence as warm start.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _smooth_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_phase(a_init, deadline, rng, beta_start_frac=0.05, n_levels=8, beta_ratio=0.6):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    for lev in range(n_levels):
        if time.monotonic() > deadline:
            break
        beta = max_c0 * beta_start_frac * (beta_ratio ** lev)
        beta = max(beta, 1e-8)

        level_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) / max(1, n_levels - lev))

        try:
            result = minimize(
                _smooth_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_smooth_grad,
                options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 40, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x.copy()
                a_new = _softmax_fn(z_new)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new
        except Exception:
            pass

        if time.monotonic() > deadline:
            break

    return best_a, best_score


def _adam_polish(a_init, deadline, rng, lr=0.04, beta_start_frac=0.005, beta_decay=0.9):

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = max(1e-10, beta_start_frac * max_c)

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

def _make_profile(n, type='linear', slope=1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if type == 'linear':
        a = 1.0 + slope * u
    elif type == 'quadratic':
        a = 1.0 + slope * u + 0.3 * u ** 2
    elif type == 'cubic':
        a = 1.0 + slope * u - 0.5 * u ** 3
    elif type == 'sine':
        a = 1.0 + slope * np.sin(np.pi * t)
    elif type == 'raised_cos':
        a = 1.0 - slope * np.cos(2 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

            p_best, sc_best = _lbfgs_phase(best_seq, per_dl, rng, beta_start_frac=0.02, n_levels=5, beta_ratio=0.5)
            update_best(p_best, sc_best, "final_lbfgs")

        if time.monotonic() < phase6_end - 10:
            per_dl2 = min(phase6_end - 3, time.monotonic() + (phase6_end - time.monotonic()) * 0.5)
            p_best2, sc_best2 = _adam_polish(best_seq, per_dl2, rng, lr=0.01, beta_start_frac=0.002, beta_decay=0.8)
            update_best(p_best2, sc_best2, "final_adam")

        print(f"Phase 6: best={best_score:.10f}")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profile(2048, 'linear', 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Verify with evaluate_sequence if feasible
        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(2048, 'linear', 1.5)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
