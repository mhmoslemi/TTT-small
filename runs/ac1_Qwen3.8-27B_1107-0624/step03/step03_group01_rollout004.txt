"""
Minimizing 2*n*max(conv(a,a))/sum(a)^2 using projected gradient descent with
adaptive logsumexp smoothing, Adam optimizer, multi-restart, and simplex projection.
Works directly on the probability simplex (allowing exact zeros), unlike softmax.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    cond = u * np.arange(1, n + 1) > cssv
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.zeros(n)
    rho = int(idx[-1])
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 2.0 * h
    smooth_val = cm + beta * float(np.log(max(s, 1e-300)))
    return smooth_val, grad, c

def _make_uniform(n):
    return np.ones(n) / n


def _make_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.cos(np.pi * (t - center) / n) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return _make_uniform(n)
    return a / s


def _make_linear_taper(n, taper_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half = center
    a = np.ones(n)
    taper = int(taper_frac * n)
    for i in range(taper):
        scale = (i + 1) / (taper + 1)
        a[i] = scale
        a[n - 1 - i] = scale
    return a / a.sum()


def _make_symmetric(n, profile_fn):
    half = n // 2
    h = profile_fn(half)
    if n % 2 == 0:
        a = np.concatenate([h, h[::-1]])
    else:
        a = np.concatenate([h, h[-2::-1]])
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return p / s


def _random_positive(n, rng, smoothness=3):
    a = rng.random(n)
    for _ in range(smoothness):
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 1e-10)
    return a / a.sum()

        update_best(p_best, sc_best, f"adam_{label}_n{pn}")
        n_optimized += 1

    print(f"Phase 1 done. Best: {best_score:.10f}")

    # --- Phase 2: L-BFGS on best ---
    print("Phase 2: L-BFGS refinement...")
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        per_dl = min(phase2_end - 10, time.monotonic() + 200.0)
        p_best, sc_best = _lbfgs_projected(best_seq, per_dl, rng, beta_init=None)
        update_best(p_best, sc_best, "lbfgs_best")

    print(f"Phase 2 done. Best: {best_score:.10f}")

    # --- Phase 3: Multi-restart with perturbations ---
    print("Phase 3: Multi-restart...")
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        n_restarts = 6
        n_best = len(best_seq)
        per_restart = min(60.0, (phase3_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 20:
                break

            noise_scale = 0.05 + 0.03 * i
            a_pert = best_seq.copy()
            noise = noise_scale * rng.normal(size=n_best)
            a_pert = a_pert + noise
            a_pert = _project_simplex(a_pert)

            per_dl = min(phase3_end - 10, time.monotonic() + per_restart)
            p_best, sc_best = _adam_projected(
                a_pert, per_dl, rng,
                lr=0.015, beta_init=None, beta_decay=0.93, beta_period=150,
                n_max_iter=30000, patience=3000
            )
            update_best(p_best, sc_best, f"restart{i}_n{n_best}")

    print(f"Phase 3 done. Best: {best_score:.10f}")

    # --- Phase 4: Try larger n via interpolation ---
    print("Phase 4: Larger n...")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = [n_curr * 2, n_curr * 4]
        for tgt_n in target_ns:
            if tgt_n > 8192 or time.monotonic() > phase4_end - 60:
                break
            p_interp = _interp_profile(best_seq, tgt_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tgt_n}")

            per_dl = min(phase4_end - 30, time.monotonic() + 150.0)
            p_best, sc_best = _adam_projected(
                p_interp, per_dl, rng,
                lr=0.01, beta_init=None, beta_decay=0.94, beta_period=200,
                n_max_iter=40000, patience=4000
            )
            update_best(p_best, sc_best, f"adam_n{tgt_n}")

    print(f"Phase 4 done. Best: {best_score:.10f}")

    # --- Phase 5: L-BFGS on best (if n allows) ---
    print("Phase 5: Final L-BFGS...")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end and best_seq is not None and len(best_seq) <= 4096:
        per_dl = min(phase5_end - 10, time.monotonic() + 120.0)
        p_best, sc_best = _lbfgs_projected(best_seq, per_dl, rng, beta_init=None)
        update_best(p_best, sc_best, "final_lbfgs")

    print(f"Phase 5 done. Best: {best_score:.10f}")

    # --- Phase 6: Final Adam polish ---
    print("Phase 6: Final polish...")
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        per_dl = min(phase6_end - 8, time.monotonic() + 100.0)
        p_best, sc_best = _adam_projected(
            best_seq, per_dl, rng,
            lr=0.005, beta_init=0.01 * float(_full_convolve(best_seq, best_seq).max()),
            beta_decay=0.96, beta_period=300,
            n_max_iter=30000, patience=5000
        )
        update_best(p_best, sc_best, "final_adam_polish")

    # --- Final validation ---
    print(f"\nFinal best score: {best_score:.10f}")

    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_uniform(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={final_seq.sum():.6f}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_uniform(256)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
