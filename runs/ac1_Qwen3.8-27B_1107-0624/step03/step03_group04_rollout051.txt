```python
def _alt_min_step(a_fixed, n):
    """Solve min t s.t. conv(a_new, a_fixed) <= t, a_new >= 0, sum(a_new) = 1."""
    # conv(a_new, a_fixed)[k] = sum_i a_new[i] * a_fixed[k-i]
    # This is linear in a_new!
    # The matrix A has A[k, i] = a_fixed[k-i] (for valid i, k-i)
    
    A = np.zeros((2*n-1, n))
    for i in range(n):
        for k in range(max(0, i), min(2*n-1, i+n)):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a_fixed[j]
    
    # LP: min t s.t. A @ a_new <= t * 1, sum(a_new) = 1, a_new >= 0
    # In standard form: min [0, ..., 0, 1] @ [a_new, t]
    # s.t. A @ a_new - t <= 0  =>  [A, -1] @ [a_new, t] <= 0
    #      sum(a_new) = 1
    #      a_new >= 0, t free (or t >= 0)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.hstack([np.ones(n), [[0]]])
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    return res.x[:n] if res.success else None
```

```python
    """Solve LP: min t s.t. conv(a_new, a_fixed)[k] <= t for all k, a_new>=0, sum=1."""
    A = _make_conv_matrix(a_fixed, n)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None


def _alt_lp_refine(a, deadline, max_iter=200, blend=0.5):
    """Alternating LP: at each step, reoptimize one side of the bilinear autocorrelation."""
    a = _normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a_new = _alt_lp_step(a, n)
        if a_new is None:
            break
        a_new = _normalize(a_new)
        a_blend = (1.0 - blend_cur) * a + blend_cur * a_new
        a_blend = _normalize(a_blend)
        sc = _score(a_blend)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_blend.copy()
            no_imp = 0
            # Keep blend moderate when improving
            blend_cur = min(0.7, blend_cur)
        else:
            no_imp += 1
            blend_cur = max(0.15, blend_cur * 0.92)
            if no_imp > max(30, n // 10):
                break
        a = a_blend
        if it % 10 == 0:
            print(f"    AltLP {it}: score={sc:.10f}", flush=True)

    return best_a


def _nesterov_grad_optimize(a, deadline, max_iter=20000, lr=0.03, beta_init=None):
    """Nesterov-accelerated gradient descent in logit space with adaptive smooth-max."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))
    z_prev = z.copy()
    mu = 0.85  # Nesterov momentum

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = beta_init if beta_init is not None else max(1e-6, 0.05 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        y = z + mu * (z - z_prev)
        e_y = np.exp(np.clip(y, -60, 60))
        a_y = e_y / e_y.sum()

        # Compute smooth max gradient
        c_y = _fft_convolve(a_y)
        mval = float(c_y.max())
        x_sm = (c_y - mval) / beta
        x_sm = np.clip(x_sm, -50.0, 50.0)
        ex = np.exp(x_sm)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c_y)
            w[np.argmax(c_y)] = 1.0
        else:
            w = ex / s_ex

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a_y[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * a_y
        dot = float(np.dot(a_y, vp))
        gz = a_y * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = np.log(best_a)
            z -= np.mean(z)
            z_prev = z.copy()
            continue

        # Nesterov update
        z_new = z + mu * (z - z_prev) - lr * gz
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))
        z_prev = z
        z = z_new

        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                base_lr = max(0.003, base_lr * 0.93)
                lr = base_lr
            if no_imp > max(5000, 8 * n):
                z = np.log(best_a)
                z -= np.mean(z)
                z_prev = z.copy()
                lr = base_lr
                no_imp = 0

        # Adaptive beta
        if it % 300 == 0 and it > 200:
            c_chk = _fft_convolve(a)
            top3 = np.sort(c_chk)[::-1][:3]
            if len(top3) > 1:
                gap = float(top3[0] - top3[-1])
                if gap < beta * 0.5:
                    beta = max(1e-10, beta * 0.6)
                elif gap > beta * 5.0:
                    beta = min(0.3 * float(c_chk.max()), beta * 1.4)
        lr = base_lr * (1.0 + 0.5 * max(0, 1.0 - it / max(1, max_iter // 3)))

    return best_a


def _error_reduction(a, deadline, max_iter=5000):
    """Spectral error reduction: flatten autocorrelation in frequency domain."""
    a = _normalize(a.copy())
    n = len(a)
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    target = 0.88

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        c_max = float(c.max())
        c_target = np.minimum(c, c_max * target)
        C = np.fft.rfft(c_target, N)
        C = np.maximum(C, 0)
        A_mag = np.sqrt(C)
        fa = np.fft.rfft(a, N)
        mag = np.abs(fa)
        phase = np.zeros_like(mag)
        mask = mag > 1e-15
        phase[mask] = np.angle(fa[mask])
        A_new = A_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.irfft(A_new, N)[:n])
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new)
        sc = _score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            target = min(0.995, target + 0.005)
            if no_imp > 800:
                break
        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_flat_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_triangular(n, alpha=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - alpha * np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine(n, sharpness=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t - 1.0) * sharpness))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _perturb_and_refine(a, deadline, rng, n_tries=5):
    """Try random perturbations and keep the best after quick refinement."""
    a = _normalize(a.copy())
    n = len(a)
    best = a.copy()
    best_sc = _score(a)

    for i in range(n_tries):
        if time.monotonic() > deadline:
            break
        scale = 0.03 * (i + 1)
        noise = rng.standard_normal(n)
        a_p = a * (1.0 + scale * noise)
        a_p = _normalize(a_p)
        sc_p = _score(a_p)
        if sc_p < best_sc * 1.2:
            sub_dl = min(deadline, time.monotonic() + 30.0)
            a_r = _nesterov_grad_optimize(a_p, sub_dl, max_iter=3000, lr=0.02)
            sc_r = _score(a_r)
            if sc_r < best_sc:
                best_sc = sc_r
                best = a_r.copy()

    return best


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. Generate diverse initializations across multiple lengths
    2. Run alternating LP + Nesterov GD + error reduction on top candidates
    3. Intensively refine the global best with multiple ALP/GD rounds
    4. Final fine polish with small learning rates
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Lengths to explore
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 640, 768, 896, 1024]:
        if ln not in lengths:
            lengths.append(ln)

    # Phase 1: Initialize
    print(f"\nPhase 1: Initialization", flush=True)
    candidates = []
    for n_t in lengths:
        if parent is not None:
            candidates.append(('parent', n_t, _interpolate_profile(parent, n_t)))
        candidates.append(('tri03', n_t, _make_triangular(n_t, 0.3)))
        candidates.append(('tri05', n_t, _make_triangular(n_t, 0.5)))
        candidates.append(('cos1', n_t, _make_cosine(n_t, 1.0)))
        candidates.append(('cos2', n_t, _make_cosine(n_t, 2.0)))
        candidates.append(('flat', n_t, _make_flat_init(n_t, rng)))

    scored = []
    for name, n_t, a0 in candidates:
        sc = _score(a0)
        scored.append((sc, name, n_t, a0))
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
    scored.sort(key=lambda x: x[0])

    print("  Top initializations:", flush=True)
    for sc, name, n_t, _ in scored[:8]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Phase 2: Optimize top candidates
    t_phase2 = time.monotonic()
    time_remaining = deadline - t_phase2
    n_top = min(5, len(scored))
    budget_each = time_remaining / n_top - 90.0
    budget_each = max(90.0, budget_each)

    print(f"\nPhase 2: Optimizing top {n_top} candidates ({budget_each:.0f}s each)", flush=True)

    for idx in range(n_top):
        if time.monotonic() > deadline - 150:
            break
        sc0, name, n_t, a0 = scored[idx]
        if sc0 > best_score * 1.25 and idx > 0:
            continue

        print(f"\n  [{idx}] {name} (n={n_t}), init={sc0:.8f}", flush=True)
        t_c = time.monotonic()

        # Nesterov GD (40% of budget)
        gd_dl = t_c + budget_each * 0.35
        if gd_dl > time.monotonic() + 15:
            a1 = _nesterov_grad_optimize(a0, gd_dl, max_iter=8000, lr=0.04)
            sc1 = _score(a1)
            print(f"    GD: {sc1:.10f}", flush=True)
            if sc1 < best_score:
                best_score = sc1
                best_seq = a1.copy()
            a0 = a1

        # Alternating LP (35% of budget)
        lp_dl = t_c + budget_each * 0.70
        if lp_dl > time.monotonic() + 15:
            a2 = _alt_lp_refine(a0, lp_dl, max_iter=100, blend=0.5)
            sc2 = _score(a2)
            print(f"    ALP: {sc2:.10f}", flush=True)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a2.copy()
            a0 = a2

        # Error reduction (15% of budget)
        er_dl = t_c + budget_each * 0.85
        if er_dl > time.monotonic() + 10:
            a3 = _error_reduction(a0, er_dl, max_iter=2000)
            sc3 = _score(a3)
            print(f"    ER: {sc3:.10f}", flush=True)
            if sc3 < best_score:
                best_score = sc3
                best_seq = a3.copy()
            a0 = a3

        # Final GD polish (10% of budget)
        if time.monotonic() < deadline - 140:
            a4 = _nesterov_grad_optimize(a0, deadline - 140, max_iter=4000, lr=0.015)
            sc4 = _score(a4)
            print(f"    Polish: {sc4:.10f}", flush=True)
            if sc4 < best_score:
                best_score = sc4
                best_seq = a4.copy()

    # Phase 3: Intensive refinement on global best
    if best_seq is not None and time.monotonic() < deadline - 80:
        print(f"\nPhase 3: Intensive refinement (current best={best_score:.10f})", flush=True)
        n_rounds = 4
        for r in range(n_rounds):
            if time.monotonic() > deadline - 40:
                break

            # ALP round
            if time.monotonic() < deadline - 50:
                a_lp = _alt_lp_refine(best_seq, deadline - 50, max_iter=40, blend=0.4)
                sc_lp = _score(a_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                    print(f"  ALP r{r}: {best_score:.10f}", flush=True)

            # GD round
            if time.monotonic() < deadline - 40:
                a_gd = _nesterov_grad_optimize(best_seq, deadline - 40, max_iter=4000, lr=0.01)
                sc_gd = _score(a_gd)
                if sc_gd < best_score:
                    best_score = sc_gd
                    best_seq = a_gd.copy()
                    print(f"  GD  r{r}: {best_score:.10f}", flush=True)

            # ER round
            if time.monotonic() < deadline - 35:
                a_er = _error_reduction(best_seq, deadline - 35, max_iter=1000)
                sc_er = _score(a_er)
                if sc_er < best_score:
                    best_score = sc_er
                    best_seq = a_er.copy()
                    print(f"  ER  r{r}: {best_score:.10f}", flush=True)

    # Phase 4: Perturbation escape + refinement
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\nPhase 4: Perturbation search", flush=True)
        a_pert = _perturb_and_refine(best_seq, deadline - 20, rng, n_tries=4)
        sc_pert = _score(a_pert)
        if sc_pert < best_score:
            best_score = sc_pert
            best_seq = a_pert.copy()
            print(f"  Perturb: {best_score:.10f}", flush=True)

    # Phase 5: Ultra-fine final polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin = _nesterov_grad_optimize(best_seq, deadline - 5, max_iter=8000, lr=0.003, beta_init=1e-5)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
        print(f"\nPhase 5: Final = {best_score:.10f}", flush=True)

    # Ensure valid output
    if best_seq is None:
        best_seq = _make_triangular(512, 0.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_triangular(512, 0.5)

    # Authoritative evaluation
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
```
