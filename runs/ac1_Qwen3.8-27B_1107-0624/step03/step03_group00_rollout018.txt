from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

    # grad[i] = sum_k w[k] * (p[k-i] + p[i-k]) (with 0 when out of bounds)
    # = sum_k w[k] * p[k-i] + sum_k w[k] * p[i-k]
    # First term: cross-correlation of w and p, evaluated at position i
    #   = sum_k w[k] p[k-i] = (w cross-correlated with p)[i]
    # Second term: sum_k w[k] p[i-k] = (w convolved with p)[i] (reversed)
    #   = sum_k w[k] p[i-k] = (w * p_reversed)[i] where * is convolution
    # Hmm, let me just use FFT:
    # grad = ifft(fft(w) * conj(fft(p_reversed))) ... this is getting confusing.
    
    # Simplest: grad = cross_corr(w_padded, p_padded) where we handle the symmetry
    # Actually, the cleanest way:
    # sum_k w[k] * c[k] where c = ifft(fft(p)^2)
    # d/dp = 2 * ifft(fft(p) * fft(w)) ... no that's not right either.
    
    # Let me just do it the direct way for clarity:
    # The gradient of f(p) = sum_k w[k] * (p*p)[k] w.r.t. p[i] is:
    # g[i] = sum_k w[k] * (p[k-i] + p[i-k])  (with p[j]=0 for j out of [0,n-1])
    # = sum_k w[k]*p[k-i] + sum_k w[k]*p[i-k]
    
    # First sum: let l = k-i, so k = l+i, sum_l w[l+i]*p[l] = (w cross-correlated with p) at offset i
    # This is the cross-correlation (w ⊛ p)[i] = sum_l w[l+i] p[l]
    
    # Second sum: let m = i-k, so k = i-m, sum_m w[i-m]*p[m] = (w convolved with p) at offset i
    # This is the convolution (w * p)[i] = sum_m w[i-m] p[m]
    
    # So g = (w ⊛ p) + (w * p)
    
    # Using FFT:
    # (w ⊛ p)[i] = cross-correlation = ifft(conj(fft(w)) * fft(p)) ... 
    # Actually in numpy, the cross-correlation can be computed as:
    # np.correlate(w, p, mode='full') shifted appropriately
    
    # Let me just compute it as:
    # g = ifft(fft(w) * fft(p_reversed)) reversed + ifft(fft(w) * fft(p))
    # ... this is getting too complex. Let me just use the formula:
    
    # Actually, the simplest correct formula:
    # g[i] = 2 * sum_k w[k] * p[i + (k - (2n-2)/2)] ... no.
    
    # OK let me just think about it from scratch.
    # c = conv(p, p), so c[k] = sum_j p[j] * p[k-j] for k = 0, ..., 2n-2
    # f = sum_k w[k] * c[k] = sum_k w[k] sum_j p[j] p[k-j]
    # df/dp[i] = sum_k d/dp[i] [w[k] sum_j p[j] p[k-j]]
    # = sum_k w[k] * (p[k-i] * 1[j=i in first sum] + p[i] * 1[k-j=i in second sum])
    # = sum_k w[k] * p[k-i] + sum_k w[k] * p[i-k]  (where p[x]=0 for x not in [0,n-1])
    
    # Let's compute:
    # A[i] = sum_k w[k] * p[k-i]  (k from 0 to 2n-2, p[k-i] nonzero only when 0<=k-i<=n-1)
    # B[i] = sum_k w[k] * p[i-k]  (similar)
    
    # Note: A[i] = sum_j w[i+j] * p[j] (letting j = k-i)
    # This is the cross-correlation of w and p: (w ⊛ p)[i] = sum_j w[i+j] p[j]
    
    # B[i] = sum_j w[i-j] * p[j] (letting j = i-k... wait, k = i-j)
    # = sum_j w[i-j] * p[j]
    # This is the convolution of w and p: (w * p)[i]
    
    # So g[i] = (w ⊛ p)[i] + (w * p)[i]
    
    # In terms of FFT:
    # (w ⊛ p)[i] = ifft(conj(fft(w)) * fft(p))[i] ... hmm, need to be careful with conventions
    # (w * p)[i] = ifft(fft(w) * fft(p))[i]
    
    # Actually in numpy:
    # fft(w) * fft(p) gives the DFT of the convolution w*p (circular, but we pad so it's linear)
    # fft(w) * conj(fft(p)) gives the DFT of the cross-correlation w⊛p
    
    # So: g = ifft(fft(w) * (fft(p) + conj(fft(p)))) = ifft(fft(w) * 2*real(fft(p)))
    # Wait: g = cross_corr(w,p) + conv(w,p)
    # cross_corr(w,p) = ifft(conj(fft(w)) * fft(p))
    # conv(w,p) = ifft(fft(w) * fft(p))
    # g = ifft(fft(p) * (conj(fft(w)) + fft(w))) = ifft(fft(p) * 2*Re(fft(w)))
    
    # Hmm, but this assumes w and p have the same length and we're doing full (linear) operations.
    # Since w has length 2n-1 and p has length n, we need to pad p.
    
    # Let me just do it with padded FFT:
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    g = np.fft.irfft(fp * 2.0 * np.real(fw), N)
    return g[:n]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _score_norm(p):
    """Compute 2*n*max(conv(p,p)) assuming sum(p)=1."""
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve_given_a(a, M):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns b or None if infeasible."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                no_imp_limit=300, noise_scale=None, use_momentum=True):
    """Adam optimizer in logit space with adaptive smooth-max temperature."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    if n < 2:
        return p0, 2.0

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    n_iter = 0

    while time.monotonic() < deadline:
        # Compute smooth max and weights
        _, w = _smooth_max_and_weights(c, beta)
        
        # Compute gradient
        g = _conv_cross_corr(w, p)

        # Project gradient to simplex tangent space
        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        n_iter += 1

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        # Adaptive beta
        if n_iter % 30 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-4, beta * 0.5)

        # Restart
        limit = max(50, no_imp_limit)
        if no_imp > limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.06
                elif best_score < 1.53:
                    ns = 0.12
                else:
                    ns = 0.25
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_refine(a, deadline, t_start=0.3, max_solves=15):
    """LP-based refinement."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_start
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve_given_a(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
            print(f"    LP improve: {sc_new:.10f} (iter {n_solves})")
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, t * 0.85)

    return current_best, current_score


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / (n - 1))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_multisine(n, freqs, amps):
    t = np.arange(n)
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        a += amp * np.cos(2.0 * np.pi * f * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 1e-12)
    return result / result.sum()
