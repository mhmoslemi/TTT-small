from scipy.optimize import linprog
import numpy as np
import time

def solve_lp(n):
    # Variables: a_0, ..., a_{n-1}, t
    # Minimize t
    # Subject to: (Aa)_k <= t for k = 0, ..., 2n-2
    #             sum(a) = 1
    #             a_i >= 0, t >= 0
    
    # c = [0, 0, ..., 0, 1] (objective: minimize t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: (Aa)_k - t <= 0
    # A is (2n-1) x n, so the full constraint matrix is (2n-1) x (n+1)
    A_full = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_full[k, j] = 1.0
        A_full[k, n] = -1.0  # coefficient of t
    
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c_obj, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        a = res.x[:n]
        t = res.x[n]
        score = 2.0 * n * t
        return a, t, score
    return None, None, None

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))

def _make_linear_profile(n, eta=1.5):

def _lse_step(a, z, state, T, lr):
    """One Adam step in logit space using log-sum-exp smoothing."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v

    return a_new, z, state, m_val

def _lp_spread_full(a, M):
    """LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[j]
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _lp_spread_reduced(a, M, top_k=80):
    """LP with reduced constraints."""
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def _subgradient_polish(a, deadline, max_iter=50000, lr=0.02):
    """Polish using subgradient at the actual max."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    v = np.zeros(n)
    momentum = 0.8
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-12)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(5000, 10 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a

        lse_budget = min(40.0, deadline - time.monotonic() - 20.0)
        if lse_budget > 5:
            lse_end = time.monotonic() + lse_budget
            a_last = _lse_optimize_adaptive(best_seq, lse_end, max_iter=20000,
                                            T_start=0.015, lr=0.04)
            sc_last = _score(a_last)
            if np.isfinite(sc_last) and sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LSE: score={sc_last:.10f}")

    # Phase 6: One more subgradient at the very end
    if best_seq is not None and time.monotonic() < deadline - 5:
        sp_end = time.monotonic() + min(20.0, deadline - time.monotonic() - 5.0)
        if sp_end - time.monotonic() > 3:
            a_last2 = _subgradient_polish(best_seq, sp_end, max_iter=20000, lr=0.005)
            sc_last2 = _score(a_last2)
            if np.isfinite(sc_last2) and sc_last2 < best_score:
                best_score = sc_last2
                best_seq = a_last2.copy()
            print(f"  Final subgrad: score={sc_last2:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
