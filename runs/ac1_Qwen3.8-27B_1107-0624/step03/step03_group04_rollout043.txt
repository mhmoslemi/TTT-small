import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    """Compute a*a using FFT."""
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Clamp tiny negatives from FFT roundoff
    c[np.abs(c) < 1e-12] = 0.0
    return c


def _score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * c.max() / (s * s)


def _smooth_max_and_grad(c, beta):
    """Compute smooth max value and weights for convolution c."""
    m = c.max()
    x = (c - m) / beta
    x = np.clip(x, -100, 0)  # Only values at or below max matter
    ex = np.exp(x)
    s = ex.sum()
    w = ex / s
    val = m + beta * np.log(s)
    return val, w, m


def _grad_smooth_max(a, w):
    """Compute gradient of smooth max w.r.t. a, using weights w."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    # grad_j = 2 * sum_k w_k * a_{k-j} (for valid k-j)
    # This is 2 * (w * a_rev)_j where * is correlation
    fw = np.fft.rfft(w, N)
    fa_r = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_r, N)[n-1:2*n-1]
    return 2.0 * h

def _lp_spread(a, M, deadline):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = a.size
    # Build constraint matrix: (a*g)_k = sum_j A[k,j] * g[j]
    # A[k,j] = a[j] if 0 <= k-j < n, else 0
    # This is the convolution matrix of a
    
    # For efficiency, use the fact that A is Toeplitz
    # A[k,j] = a[k-j] for 0 <= k-j <= n-1
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_lin = -np.ones(n)
    
    # Constraints: A @ g <= M (for all k = 0, ..., 2n-2)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_lin, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': max(1, deadline - time.monotonic())})
    
    if res.success and res.x is not None:
        return res.x
    return None

def _spectral_flatten(a, deadline, max_iter=2000, alpha=0.1):
    """Gerchberg-Saxton style spectral flattening."""
    a = np.maximum(a, 1e-15).copy()

def _init_flat_spectrum(n, rng):

    X[0] = 1.0

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    idx = rho[-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w / w.sum()

    a = a / a.sum()
    n = a.size

    c = _fft_conv(a)
    c_max = c.max()
    beta = max(1e-8, 0.03 / n)

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = 2.0 * n * c_max
    no_imp = 0
    cur_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _fft_conv(a)
        val, w = _smooth_max_weights(c, beta)
        g = _grad_conv_max(a, w)

        # Project gradient to log-space: g_z = a * (g - mean(a*g))
        mean_dot = float(a @ g)
        g_z = a * (g - mean_dot)
        g_z -= g_z.mean()

        if not np.all(np.isfinite(g_z)):
            break

        t += 1
        m_arr = b1 * m_arr + (1 - b1) * g_z
        v_arr = b2 * v_arr + (1 - b2) * g_z * g_z
        m_hat = m_arr / (1 - b1 ** t)
        v_hat = v_arr / (1 - b2 ** t)

        z_new = z - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        z_new -= z_new.mean()
        z_new = np.clip(z_new, -40.0, 40.0)

        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()

        c_new = _fft_conv(a_new)
        sc = 2.0 * n * c_new.max()

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        z = z_new

        # Adaptive beta
        if it % 300 == 150:
            sorted_c = np.sort(c_new)
            top_k = min(10, len(sorted_c))
            top = sorted_c[-top_k:]
            gap = top[0] - top[-1]
            if gap < beta * 0.5:
                beta = max(1e-10, beta * 0.6)
            elif gap > beta * 5.0:
                beta = min(0.5 / n, beta * 1.5)

        # LR scheduling
        if no_imp > 3000:
            cur_lr = max(0.001, cur_lr * 0.92)
            if no_imp > 20000:
                break

    return best_a


def _penalty_optimize(a, deadline, max_iter=100000, lr=0.01):
    """Penalty-based optimization: minimize sum max(0, (a*a)_k - T)^2."""
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    n = a.size

    c = _fft_conv(a)
    c_max = c.max()
    T = c_max * 0.95  # Start slightly below current max

    z = np.log(a)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_score = 2.0 * n * c_max
    no_imp = 0
    cur_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = np.exp(z - z.max())
        a = a / a.sum()

        g, c, pen_val = _penalty_grad(a, T)

        if pen_val < 1e-10:
            # All constraints satisfied, lower T
            T = max(c.max() * 0.99, T * 0.99)
            continue

        mean_dot = float(a @ g)
        g_z = a * (g - mean_dot)
        g_z -= g_z.mean()

        if not np.all(np.isfinite(g_z)):
            break

        t += 1
        m_arr = b1 * m_arr + (1 - b1) * g_z
        v_arr = b2 * v_arr + (1 - b2) * g_z * g_z
        m_hat = m_arr / (1 - b1 ** t)
        v_hat = v_arr / (1 - b2 ** t)

        z_new = z - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        z_new -= z_new.mean()
        z_new = np.clip(z_new, -40.0, 40.0)

        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()

        c_new = _fft_conv(a_new)
        sc = 2.0 * n * c_new.max()

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0

            # Gradually lower T towards the target
            T = max(T * 0.999, c_new.max() * 0.95)
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                # Raise T slightly to escape
                T = min(c_new.max() * 1.05, T * 1.01)
                cur_lr = max(0.002, cur_lr * 0.95)
            if no_imp > 15000:
                break

        z = z_new

    return best_a


def _lp_spread(a, M, time_limit=10.0):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = a.size
    c_lin = -np.ones(n)

    # Build Toeplitz matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)

    res = linprog(c_lin, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})

    if res.success and res.x is not None:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=3000, inflate=1.0005, blend=0.15):
    """Iterative LP spreading."""
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    cur_blend = blend
    cur_inflate = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline - 10:
            break

        c = _fft_conv(a)
        M = c.max()
        M_lp = M * cur_inflate

        time_left = deadline - time.monotonic() - 10
        if time_left < 5:
            break

        tl = min(8.0, time_left * 0.3)
        g = _lp_spread(a, M_lp, time_limit=tl)
        if g is None:
            break

        s_g = g.sum()
        if s_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 30:
                cur_inflate = min(1.005, cur_inflate * 1.03)
            if no_imp > 300:
                break
            continue

        g = g / s_g
        a_new = (1.0 - cur_blend) * a + cur_blend * g
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                cur_blend = max(0.05, cur_blend * 0.92)

        a = a_new

    return best_a


def _spectral_flatten(a, deadline, max_iter=5000, alpha=0.08):
    """Spectral domain flattening (Gerchberg-Saxton style)."""
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    n = a.size
    N = _npow2(2 * n + 1)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    cur_alpha = alpha

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute target magnitude for flat convolution
        # We want |A|^2 to be as flat as possible
        power = mag * mag
        # Exclude DC
        non_dc_power = power[1:]
        target_mean = non_dc_power.mean()

        # Smoothly blend towards flat
        new_mag = mag.copy()
        new_mag[1:] = (1.0 - cur_alpha) * mag[1:] + cur_alpha * np.sqrt(target_mean)
        # Keep DC fixed
        new_mag[0] = mag[0]

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                cur_alpha = max(0.01, cur_alpha * 0.85)
            if no_imp > 3000:
                break

        a = a_new

    return best_a


def _frank_wolfe_step(a, c_max_idx, deadline):
    """Frank-Wolfe: move towards the vertex that most reduces the objective."""
    n = a.size
    c = _fft_conv(a)
    k_star = c.max()

    # Gradient of (a*a)_{k_star}
    g = _grad_conv_max(a, np.zeros(2 * n - 1))
    # Set weight only on k_star
    w = np.zeros(2 * n - 1)
    w[int(k_star)] = 1.0
    g = _grad_conv_max(a, w)

    # FW: minimize g^T s over simplex => s = e_{argmin g}
    j_min = int(np.argmin(g))
    s = np.zeros(n)
    s[j_min] = 1.0

    # Step size: gamma = 2/(t+2) or line search
    # Use simple step
    gamma = 0.1
    a_new = (1 - gamma) * a + gamma * s
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new


def _sim_anneal(a, deadline, max_iter=8000, T_init=0.03):
    """Simulated annealing with structured perturbations."""
    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    T = T_init
    rng = np.random.default_rng(12345)
    no_best = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Structured perturbation: mix of global and local
        mode = rng.random()
        if mode < 0.3:
            # Global multiplicative noise
            scale = 0.02 + 0.05 * (T / T_init)
            a_pert = a * np.exp(scale * rng.standard_normal(n))
        elif mode < 0.6:
            # Block perturbation
            block = max(4, n // 16)
            start = rng.integers(0, n - block)
            a_pert = a.copy()
            a_pert[start:start + block] *= np.exp(0.1 * rng.standard_normal(block))
        else:
            # Swap-like perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            a_pert = a.copy()
            swap = 0.05 * min(a[i], a[j])
            a_pert[i] = max(a[i] - swap, 1e-15)
            a_pert[j] = a[j] + swap

        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        delta = sc - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_pert
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_best = 0
            else:
                no_best += 1
        else:
            no_best += 1

        T *= 0.9995
        if T < 0.005 or no_best > 5000:
            T = T_init * 0.3
            no_best = 0
            a = best_a.copy()
            cur_score = best_score

    return best_a


def _init_flat_spec(n, rng):
    N = _npow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    A = np.zeros(N, dtype=complex)
    A[0] = 1.0
    for k in range(1, N // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        A[N - k] = np.exp(-1j * phases[k])
    a = np.fft.ifft(A).real[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _init_shaped(n, shape='cos2'):
    t = np.linspace(0, 1, n)
    if shape == 'cos2':
        a = (np.cos(np.pi * (t - 0.5)) ** 2 + 0.2)
    elif shape == 'cos4':
        a = (np.cos(np.pi * (t - 0.5)) ** 4 + 0.1)
    elif shape == 'u_shape':
        a = (np.cos(np.pi * t) ** 2 + 0.5)
    elif shape == 'center_heavy':
        a = (np.exp(-((t - 0.5) * 4) ** 2) + 0.1)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _interpolate(profile, n_new):
    p0 = np.asarray(profile, dtype=np.float64)
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0.copy() / s if s > 0 else np.ones(n_new) / n_new
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-10)
    s = p.sum()
    return p / s if s > 0 else np.ones(n_new) / n_new

            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()

        pass
