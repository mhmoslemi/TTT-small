```python
    freqs = np.linspace(0, 1, len(P))
```

```python
"""
Optimized step function search using top-k subgradient descent with 
spectral flattening in logit space. Improves upon previous state of the art
by using sharper peak targeting, spectral domain corrections, and adaptive 
learning rate schedules.
"""
import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_gradient_topk(a, c, k=10):
    """
    Compute subgradient using top-k peaks.
    Returns the gradient of the weighted sum of top-k convolution values.
    
    Gradient of (a*a)[k] wrt a[i] = a[k-i] + a[i] (with boundary handling)
    which is equivalent to the cross-correlation of the weight vector with a.
    """
    n = len(a)
    # Find top-k peak indices
    if k >= len(c):
        k = len(c)
    
    # Get top-k values and normalize weights
    top_indices = np.argpartition(c, -k)[-k:]
    top_values = c[top_indices]
    
    # Weight proportional to value (or uniform)
    w = np.zeros_like(c)
    # Use softmax-like weighting on the top-k values
    max_tv = top_values.max()
    exp_vals = np.exp((top_values - max_tv) / (max_tv * 0.1 + 1e-10))
    weights = exp_vals / exp_vals.sum()
    w[top_indices] = weights
    
    # Compute gradient: grad[i] = 2 * sum_k w[k] * a_converted[k-i]
    # This is 2 * cross_correlation(w, a)
    # cross_correlation(w, a)[i] = sum_j w[j] * a[j-i] (with valid indices)
    # Equivalent to: convolve(reversed(w), a) shifted appropriately
    
    # Use FFT for efficiency
    N = next_pow2(n + len(c) - 1)
    af = a[::-1]  # reverse a
    fa = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa * fw, N)
    
    # Extract the relevant portion
    # cross_corr[i] = sum_j w[j] * a[j-i] = conv_reversed[i + n - 1]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    return grad


def spectral_flatten_step(a, alpha=0.05):
    """
    Apply a small spectral flattening step.
    Modifies the magnitude of the DFT to make the power spectrum flatter.
    """
    n = len(a)
    N = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    P = A.real ** 2 + A.imag ** 2
    
    # Smooth target spectrum
    freqs = np.linspace(0.0, 1.0, len(P))
    # Target: relatively flat but with some low-frequency content
    P_target = 1.0 + 5.0 * np.exp(-10.0 * freqs)
    
    # Compute adjustment
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = P_target / np.maximum(P, 1e-15)
    ratio = np.clip(ratio, 0.1, 10.0)
    
    mag = np.sqrt(P)
    mag_new = mag * (ratio ** alpha)
    phase = np.arctan2(A.imag, A.real)
    A_new = mag_new * np.exp(1j * phase)
    
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new


def logit_adam_topk(a0, deadline, rng, lr=0.03, k_start=5, k_end=30,
                    k_ramp_steps=500, spectral_every=300, spectral_alpha=0.03,
                    restart_period=800):
    """
    Logit-space Adam optimization using top-k subgradient.
    Periodically applies spectral flattening.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    c = full_convolve_fft(a0)
    max_c = float(c.max())
    
    z = np.log(a0)
    z = z - z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr
    
    iters = 0
    while time.monotonic() < deadline:
        # Adaptive k
        progress = min(1.0, iters / k_ramp_steps)
        k = int(k_start + (k_end - k_start) * progress)
        k = max(1, min(k, len(c) - 1))
        
        # Compute top-k subgradient
        grad_a = compute_gradient_topk(a0, c, k)
        
        # Project to tangent space of simplex
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)
        
        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        
        # Stabilize
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            iters += 1
            continue
        
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        
        # Softmax to get a
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            iters += 1
            continue
        a0 = a0 / s
        
        # Evaluate
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iters += 1
        
        # Spectral flattening step
        if spectral_every > 0 and iters % spectral_every == 0:
            a_spec = spectral_flatten_step(a0, spectral_alpha)
            c_spec = full_convolve_fft(a_spec)
            score_spec = float(2.0 * n * c_spec.max())
            if score_spec < best_score:
                best_score = score_spec
                best_a = a_spec.copy()
                z = np.log(np.maximum(a_spec, 1e-15))
                z = z - z.mean()
                a0 = a_spec.copy()
                c = c_spec
                no_improve = 0
            else:
                # Don't accept, just use as perturbation
                a0 = 0.9 * a0 + 0.1 * a_spec
                a0 = np.maximum(a0, 1e-15)
                a0 = a0 / a0.sum()
                z = np.log(a0)
                z = z - z.mean()
        
        # Learning rate schedule (cosine with restarts)
        if no_improve > 200 and no_improve % 100 == 0:
            current_lr = base_lr * max(0.3, 1.0 - no_improve / restart_period)
        
        # Random restart
        if no_improve > restart_period:
            noise_scale = 0.1
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0
        
        # Progress reporting
        if iters % 500 == 0:
            print(f"    iter={iters}, score={score:.8f}, best={best_score:.8f}, k={k}, lr={current_lr:.5f}")
    
    return best_a, best_score


def make_bathtub_profile(n, depth=0.5):
    """U-shaped profile: high at edges, low in middle."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - depth * np.sin(np.pi * t)
    a = np.maximum(a, 0.05)
    return a / a.sum()


def make_smooth_profile(n, rng, n_fourier=8):
    """Random smooth profile using Fourier series."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_fourier + 1):
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result
```

```python
"""
Step function optimization using top-k subgradient descent with spectral
flattening in logit space. Uses adaptive peak targeting, coarse-to-fine
multi-resolution search, and periodic frequency-domain corrections to
minimize 2n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_gradient_topk(a, c, k=10, threshold_frac=0.5):
    """
    Compute subgradient using top peaks above a threshold.
    Returns gradient of weighted sum of selected convolution values.
    """
    n = len(a)
    max_c = c.max()
    if max_c < 1e-30:
        return np.zeros(n)
    
    # Select peaks above threshold
    threshold = max_c * threshold_frac
    mask = c >= threshold
    n_peaks = int(np.count_nonzero(mask))
    
    if n_peaks == 0:
        idx = int(np.argmax(c))
        mask[idx] = True
        n_peaks = 1
    elif n_peaks > k:
        # Keep only top k
        top_indices = np.argsort(c)[-k:]
        mask = np.zeros_like(c)
        mask[top_indices] = True
        n_peaks = k
    
    # Weight proportional to value
    peak_vals = c[mask]
    exp_vals = np.exp((peak_vals - max_c) / (max_c * 0.15 + 1e-15))
    weights = exp_vals / exp_vals.sum()
    
    w = np.zeros_like(c)
    w[mask] = weights
    
    # Compute gradient via cross-correlation
    N = next_pow2(n + len(c) - 1)
    af = a[::-1]
    fa = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa * fw, N)
    
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def spectral_flatten_step(a, alpha=0.03):
    """
    Apply spectral flattening: modify DFT magnitude to reduce convolution peak.
    """
    n = len(a)
    N = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    P = A.real ** 2 + A.imag ** 2
    
    freqs = np.linspace(0.0, 1.0, len(P))
    P_target = 1.0 + 3.0 * np.exp(-8.0 * freqs)
    
    with np.errstate(divide='ignore', invalid='ignore'):
        ratio = P_target / np.maximum(P, 1e-15)
    ratio = np.clip(ratio, 0.2, 5.0)
    
    mag = np.sqrt(P)
    mag_new = mag * (ratio ** alpha)
    phase = np.arctan2(A.imag, A.real)
    A_new = mag_new * np.exp(1j * phase)
    
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new


def direct_peak_shave(a, c, strength=0.005):
    """
    Directly shave the peak by reducing elements that contribute to it.
    """
    n = len(a)
    peak_idx = int(np.argmax(c))
    
    a_new = a.copy()
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            reduction = strength * a[i] * a[j] / (a.sum() + 1e-15)
            a_new[i] -= reduction * 0.5
            if j != i:
                a_new[j] -= reduction * 0.5
    
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new


def logit_adam_optimize(a0, deadline, rng, lr=0.025, k_start=5, k_end=40,
                        k_ramp=400, spectral_every=250, spectral_alpha=0.02,
                        restart_period=1200, threshold_frac=0.6):
    """
    Logit-space Adam optimization with top-k subgradient and spectral steps.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    c = full_convolve_fft(a0)
    max_c = float(c.max())
    
    z = np.log(a0)
    z = z - z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    base_lr = lr
    current_lr = lr
    
    iters = 0
    while time.monotonic() < deadline:
        progress = min(1.0, iters / k_ramp)
        k = max(1, int(k_start + (k_end - k_start) * progress))
        
        grad_a = compute_gradient_topk(a0, c, k, threshold_frac)
        
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            iters += 1
            continue
        
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            iters += 1
            continue
        a0 = a0 / s
        
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iters += 1
        
        # Spectral flattening
        if spectral_every > 0 and iters % spectral_every == 0:
            a_spec = spectral_flatten_step(a0, spectral_alpha)
            c_spec = full_convolve_fft(a_spec)
            score_spec = float(2.0 * n * c_spec.max())
            if score_spec < best_score:
                best_score = score_spec
                best_a = a_spec.copy()
                a0 = a_spec.copy()
                c = c_spec
                z = np.log(np.maximum(a0, 1e-15))
                z = z - z.mean()
                no_improve = 0
            else:
                # Blend slightly
                a0 = 0.85 * a0 + 0.15 * a_spec
                a0 = np.maximum(a0, 1e-15)
                a0 = a0 / a0.sum()
                z = np.log(a0)
                z = z - z.mean()
                c = full_convolve_fft(a0)
        
        # Direct peak shave every 100 iterations
        if iters % 100 == 0 and iters > 50:
            a_shaved = direct_peak_shave(a0, c, strength=0.002)
            c_shaved = full_convolve_fft(a_shaved)
            score_shaved = float(2.0 * n * c_shaved.max())
            if score_shaved < best_score:
                best_score = score_shaved
                best_a = a_shaved.copy()
                a0 = a_shaved.copy()
                c = c_shaved
                z = np.log(np.maximum(a0, 1e-15))
                z = z - z.mean()
                no_improve = 0
        
        # Learning rate schedule
        if no_improve > 150:
            decay = max(0.2, 1.0 - no_improve / (restart_period * 0.5))
            current_lr = base_lr * decay
        
        # Random restart
        if no_improve > restart_period:
            noise_scale = 0.08
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -30.0, 30.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0
        
        if iters % 400 == 0:
            print(f"    iter={iters}, score={score:.8f}, best={best_score:.8f}, k={k}, lr={current_lr:.5f}")
    
    return best_a, best_score


def make_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    
    # Bathtub (U-shaped)
    for depth in [0.3, 0.5, 0.7]:
        a = 1.0 - depth * np.sin(np.pi * t)
        a = np.maximum(a, 0.05)
        profiles.append(a / a.sum())
    
    # Cosine-modulated
    for k in [1, 2, 3]:
        for amp in [0.2, 0.4]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 0.05)
            profiles.append(a / a.sum())
    
    # Power-law ramps
    for p in [0.5, 1.0, 2.0]:
        a = (1.0 - t) ** p + 0.1
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())
    
    # Random smooth
    for _ in range(5):
        a = np.ones(n)
        for k in range(1, 6):
            amp = rng.uniform(0.05, 0.25) / k
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t + phase)
        a = np.maximum(a, 0.01)
        profiles.append(a / a.sum())
    
    return profiles


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

            best_seq = np.asarray(p, dtype=np.float64).copy()

            if s > 0.01:

        pass
    
    print(f"Starting search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
```
