def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):

        return np.array([], dtype=np.float64)

    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    return c

def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]

    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

    """Project v onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_h(w, a):
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def hard_peaks_subgradient(c, a, n_peaks=3):
    n = len(a)
    c_max = c.max()
    idx = np.argsort(c)[-n_peaks:]
    w = np.zeros_like(c)
    w[idx] = 1.0 / n_peaks
    h = compute_h(w, a)
    return 2.0 * h


def adaptive_peaks_subgradient(c, a, n_peaks_min=2, n_peaks_max=10, frac=0.9):
    n = len(a)
    c_max = c.max()
    threshold = c_max * frac
    active = np.where(c >= threshold)[0]
    n_active = min(len(active), n_peaks_max)
    if n_active == 0:
        n_active = 1
        active = np.array([np.argmax(c)])
    n_use = max(n_peaks_min, n_active)
    n_use = min(n_use, len(c))
    idx = np.argsort(c)[-n_use:]
    w = np.zeros_like(c)
    w[idx] = 1.0 / n_use
    h = compute_h(w, a)
    return 2.0 * h

            a0 = best_a.copy()
            m.fill(0.0)

            a0 = best_a.copy()

def active_set_nudge(a0, deadline, rng, max_rounds=2000, frac=0.92):
    """
    Active-set peak targeting: identify top peaks, nudge contributing coordinates down
    and spread mass to low-contribution coordinates.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    best_a = a0.copy()
    best_score = float(2.0 * n * c.max())
    no_improve = 0
    step_scale = 0.005

    for _ in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c_max = c.max()
        threshold = c_max * frac
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute "damage vector": for each coordinate i, how much it contributes
        # to the active peaks
        damage = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    damage[i] += a[j]

        # Also compute "safety vector": how much increasing coordinate i would
        # increase the active peaks
        safety = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    safety[i] += 2.0 * a[j]

        # Donor: high damage, high a[i] (lots of mass contributing to peaks)
        # Receiver: low safety (adding mass here won't hurt peaks much)
        donor_scores = damage * a0
        receiver_scores = safety  # we want low safety

        # Apply step: decrease donors, increase receivers
        n_donors = max(1, min(5, n // 100))
        n_receivers = max(1, min(5, n // 100))

        top_donors = np.argsort(donor_scores)[-n_donors:]
        top_receivers = np.argsort(receiver_scores)[:n_receivers]

        # Transfer mass
        total_transfer = 0.0
        for i in top_donors:
            if a0[i] > 1e-10:
                delta = step_scale * a0[i] * damage[i] / (c_max + 1e-10)
                delta = min(delta, a0[i] * 0.1)
                a0[i] -= delta
                total_transfer += delta

        if total_transfer < 1e-15:
            no_improve += 1
            if no_improve > 200:
                break
            continue

        # Distribute to receivers proportional to (1/safety)
        recip = 1.0 / (safety[top_receivers] + 1e-10)
        recip = recip / recip.sum()
        for idx, j in enumerate(top_receivers):
            a0[j] += total_transfer * recip[idx]

        a0 = np.maximum(a0, 0)
        a0 = a0 / a0.sum()

        c = full_convolve_fft(a0)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
            step_scale = min(0.02, step_scale * 1.05)
        else:
            no_improve += 1
            # Revert
            for i in top_donors:
                pass  # We already modified a0, need to revert
            # Simpler: just reduce step
            step_scale = max(0.0005, step_scale * 0.9)
            if no_improve > 300:
                break

    return best_a, best_score


def pairwise_refine(a0, deadline, rng, max_iters=5000):
    """
    Pairwise coordinate descent: find the best (donor, receiver) pair to
    transfer mass, reduce the max convolution.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    best_a = a0.copy()
    best_score = float(2.0 * n * c.max())
    no_improve = 0
    base_delta = 0.002

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c_max = c.max()
        k_star = int(np.argmax(c))

        # The peak is at k_star. c[k_star] = sum_i a[i]*a[k_star-i]
        # To reduce it, decrease a[i] where a[k_star-i] is large.
        # To not create new peaks, increase a[j] where it has minimal effect on all peaks.

        # Compute effect of each coordinate on the main peak
        peak_effect = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                peak_effect[i] = a0[j]

        # Compute "global damage" if we increase coordinate j:
        # It increases c[k] by a[k-j] + a[j-k] for each k
        # Approximate by: sum over top few peaks of (a[k-j] + a[j-k])
        top_k = np.argsort(c)[-min(5, n):]
        global_damage = np.zeros(n)
        for k in top_k:
            for j in range(n):
                if 0 <= k - j < n:
                    global_damage[j] += a0[k - j]
                if 0 <= j - k < n:
                    global_damage[j] += a0[j - k]

        # Best donor: high peak_effect * a0[i] (large contribution to peak, has mass)
        donor_metric = peak_effect * a0
        # Best receiver: low global_damage, some mass (not zero)
        # Use a small mass threshold to avoid dividing by zero
        recv_metric = global_damage  # want minimum

        # Sample candidate donors and receivers
        n_cand = min(20, n)
        donor_candidates = np.argsort(donor_metric)[-n_cand:]
        # For receivers, pick from coordinates with small global_damage
        # but also with some existing mass (to avoid creating spikes)
        recv_pool = np.where(a0 > 1e-6)[0]
        if len(recv_pool) > n_cand:
            recv_indices = np.argsort(global_damage[recv_pool])[:n_cand]
            receiver_candidates = recv_pool[recv_indices]
        else:
            receiver_candidates = recv_pool

        best_improve = 0
        best_pair = None
        best_delta = 0

        for i in donor_candidates:
            if a0[i] < 1e-8:
                continue
            # Estimate reduction in c[k_star]
            reduction = peak_effect[i]  # per unit of a[i] decreased

            for j in receiver_candidates:
                if j == i:
                    continue
                # Estimate increase in max due to adding to j
                increase = global_damage[j]
                # Net effect (rough): reduction - increase
                # We want reduction > increase
                net = reduction - increase * 0.5
                if net > best_improve:
                    best_improve = net
                    best_pair = (i, j)
                    best_delta = min(a0[i] * 0.01, base_delta * a0[i])

        if best_pair is None or best_delta < 1e-15:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        i, j = best_pair
        a0[i] -= best_delta
        a0[j] += best_delta
        a0 = np.maximum(a0, 0)
        a0 = a0 / a0.sum()

        c = full_convolve_fft(a0)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            # Revert
            a0[i] += best_delta
            a0[j] -= best_delta
            a0 = a0 / a0.sum()
            no_improve += 1
            if no_improve > 100:
                base_delta = max(1e-5, base_delta * 0.8)
                if base_delta < 1e-6:
                    break

    return best_a, best_score


def make_smooth_profile(n, rng, seed_offset=0):
    """Generate a smooth, non-negative, normalized profile."""
    t = np.linspace(0.0, 1.0, n)
    # Random low-frequency content
    n_harmonics = rng.integers(2, 6)
    a = np.ones(n)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 0.4) / k
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2 * np.pi * k * t + phase)
    # Add a linear trend
    trend = rng.uniform(-0.5, 0.5)
    a = a + trend * (0.5 - t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def fine_grained_polish(a0, deadline, rng, n_rounds=50):
    """
    Very fine-grained polish: tiny perturbations with score evaluation.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()

    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        # Perturb a few random coordinates
        n_perturb = rng.integers(3, 10)
        idx = rng.choice(n, size=n_perturb, replace=False)

        a_perturbed = a0.copy()
        deltas = rng.normal(0, step_size, size=n_perturb)
        for i, idx_i in enumerate(idx):
            a_perturbed[idx_i] += deltas[i]

        a_perturbed = np.maximum(a_perturbed, 0)
        s = a_perturbed.sum()
        if s < 0.01:
            continue
        a_perturbed = a_perturbed / s

        c_new = full_convolve_fft(a_perturbed)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_perturbed.copy()
            a0 = a_perturbed.copy()
            c = c_new
        elif score_new > best_score * 1.001:
            # Reduce step size if things are getting worse
            step_size = max(1e-5, step_size * 0.9)

    return best_a, best_score

        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting refined search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # Determine which n values to try
    # Focus on n values near the initial length, plus some exploration
    n_candidates = [512, 768, 1024, 1536, 2048, 2560, 3072, 4096]

    # Allocate time: prioritize n values that are likely to be best
    # The previous best was at n=2048, so weight toward that
    time_weights = {}
    for n in n_candidates:
        if init_seq is not None:
            dist = abs(n - init_seq.size) / max(init_seq.size, 1)
        else:
            dist = abs(n - 2048) / 2048
        time_weights[n] = 1.0 / (1.0 + 2.0 * dist)

    total_weight = sum(time_weights.values())
    eff_budget = deadline - t_start
    time_alloc = {}
    for n in n_candidates:
        time_alloc[n] = max(30.0, eff_budget * time_weights[n] / total_weight)

    print(f"Time allocation: {{{', '.join(f'{n}:{t:.0f}s' for n,t in time_alloc.items())}}}")

    # Process n values in order of priority (highest time allocation first)
    n_order = sorted(n_candidates, key=lambda x: -time_alloc[x])

    for n in n_order:
        if time.monotonic() > deadline - 30:
            break

        n_deadline = min(deadline, t_start + 10 + time_alloc[n])
        elapsed = time.monotonic() - t_start
        print(f"\n--- n={n} (time={time_alloc[n]:.0f}s, elapsed={elapsed:.0f}s) ---")

        # Get starting profile
        if init_seq is not None:
            p0 = interpolate_profile(init_seq, n)
        else:
            p0 = make_smooth_profile(n, rng)

        sc0 = compute_score(p0)
        update(p0, sc0)

        # Phase A: Logit-space Adam with hard peaks (main optimization)
        if time.monotonic() < n_deadline - 10:
            dl_a = min(n_deadline, time.monotonic() + time_alloc[n] * 0.5)
            result_a, sc_a = logit_adam_optimize(
                p0, dl_a, rng, lr=0.03, n_peaks=3,
                decay=0.999, restart_period=350)
            sc_a = compute_score(result_a)
            update(result_a, sc_a)
            print(f"  After logit Adam: {sc_a:.10f}")
            p0 = result_a

        # Phase B: Active-set nudging
        if time.monotonic() < n_deadline - 10:
            dl_b = min(n_deadline, time.monotonic() + time_alloc[n] * 0.25)
            result_b, sc_b = active_set_nudge(
                p0, dl_b, rng, max_rounds=3000, frac=0.93)
            sc_b = compute_score(result_b)
            update(result_b, sc_b)
            print(f"  After active-set nudge: {sc_b:.10f}")
            if sc_b < compute_score(p0):
                p0 = result_b

        # Phase C: Pairwise coordinate descent
        if time.monotonic() < n_deadline - 5:
            dl_c = min(n_deadline, time.monotonic() + time_alloc[n] * 0.15)
            result_c, sc_c = pairwise_refine(p0, dl_c, rng, max_iters=8000)
            sc_c = compute_score(result_c)
            update(result_c, sc_c)
            print(f"  After pairwise: {sc_c:.10f}")
            if sc_c < compute_score(p0):
                p0 = result_c

        # Phase D: Fine-grained polish
        if time.monotonic() < n_deadline - 2:
            dl_d = min(n_deadline, time.monotonic() + time_alloc[n] * 0.1)
            result_d, sc_d = fine_grained_polish(p0, dl_d, rng, n_rounds=100)
            sc_d = compute_score(result_d)
            update(result_d, sc_d)
            print(f"  After fine polish: {sc_d:.10f}")

        print(f"  Best so far: {best_score:.10f} (n={best_seq.size if best_seq is not None else '?'})")

    # ===== Final global polish on the best sequence found =====
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n--- Final global polish (n={best_seq.size}) ---")
        n_final = best_seq.size
        remaining = deadline - time.monotonic()

        # Multiple rounds of different strategies
        strategies = [
            ("logit_adam", 0.4),
            ("active_set", 0.25),
            ("pairwise", 0.2),
            ("fine_polish", 0.15),
        ]

        for strat_name, frac_time in strategies:
            if time.monotonic() > deadline - 5:
                break
            dl = min(deadline, time.monotonic() + remaining * frac_time)
            p0 = best_seq.copy()

            # Add tiny noise to avoid getting stuck
            p0 = p0 + 0.0001 * rng.normal(size=n_final)
            p0 = np.maximum(p0, 1e-10)
            p0 = p0 / p0.sum()

            if strat_name == "logit_adam":
                r, _ = logit_adam_optimize(p0, dl, rng, lr=0.015,
                                           n_peaks=2, decay=0.9995,
                                           restart_period=300)
            elif strat_name == "active_set":
                r, _ = active_set_nudge(p0, dl, rng, max_rounds=2000, frac=0.95)
            elif strat_name == "pairwise":
                r, _ = pairwise_refine(p0, dl, rng, max_iters=10000)
            else:
                r, _ = fine_grained_polish(p0, dl, rng, n_rounds=200)

            sc_r = compute_score(r)
            update(r, sc_r)
            print(f"  {strat_name}: {sc_r:.10f}")

    # ===== Try a few more n values at the end if time permits =====
    if best_seq is not None and time.monotonic() < deadline - 20:
        print("\n--- Quick n exploration at end ---")
        n_final = best_seq.size
        # Try n slightly larger and smaller
        for n_try in [n_final + 256, n_final - 256, n_final + 512]:
            if n_try < 128 or n_try > 4096:
                continue
            if time.monotonic() > deadline - 15:
                break
            p_try = interpolate_profile(best_seq, n_try)
            sc_try = compute_score(p_try)
            update(p_try, sc_try)
            if sc_try < best_score + 0.005:
                dl = min(deadline - 10, time.monotonic() + 30)
                r, _ = logit_adam_optimize(p_try, dl, rng, lr=0.02, n_peaks=3)
                sc_r = compute_score(r)
                update(r, sc_r)
                print(f"  n={n_try}: {sc_r:.10f}")

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        # Fallback: uniform sequence
        final_seq = np.ones(512) / 512.0

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(256) / 256.0
        s = 1.0

    # Normalize for the evaluation (score is scale-invariant, but keep sum reasonable)
    # Don't renormalize to sum=1 to avoid issues; just ensure validity
    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        else:
            print(f"WARNING: evaluate_sequence returned {eval_score}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check error: {e}")

    return [float(x) for x in final_seq]
