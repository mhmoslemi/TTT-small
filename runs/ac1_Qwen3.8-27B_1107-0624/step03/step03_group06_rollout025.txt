"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return a / a.sum()

    a = 1.0 - w * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-6)

def _smooth_max_and_grad(a, beta):
    """Compute soft-max value and gradient w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    cm = float(c.max())

    # Clip for numerical stability
    x = np.clip(x, -50.0, 0.0)

    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:

        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    soft_max_val = cm + beta * np.log(s)
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n)

    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return soft_max_val, grad

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        k_star = int(c.argmax())
        cv = float(c.max())
        s = float(a.sum())
        cur_score = 2.0 * n * cv

        # Subgradient: grad[j] = 2*a[k_star - j] if 0<=k_star-j<n
        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_max >= j_min:
            idx = k_star - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] = 2.0 * a[idx]

        # Update with step size
        a_new = a - eta * grad / (s * s)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            eta = eta * 0.5
            continue
        a_new = a_new / sn
        c_new = _fft_convolve(a_new)
        sc_new = 2.0 * n * float(c_new.max())

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            eta = min(eta * 1.02, 0.3)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                eta = max(eta * 0.9, 0.005)
            if no_imp > 5000:
                if it > 2000:
                    break
                a = best_a.copy()
                eta = 0.05
                no_imp = 0

    return best_a


def _solve_lp(f, M, n):
    """Solve LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=20000, inflate_start=0.008, inflate_min=0.001):
    """LP-based refinement with line search."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * (1.0 + inflate)

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.02, inflate * 1.2)
            if no_imp > 200:
                inflate = min(0.03, inflate * 1.5)
                if no_imp > 500:
                    break
            continue

        g = g / sum_g

        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            sc = float(2.0 * n * _fft_convolve(a_new).max())
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            a_new = a_new / sn
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            inflate = max(inflate_min, inflate * 0.92)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.02, inflate * 1.1)

    return best_a


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=30000):
    """Momentum-based gradient polishing with smooth max."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    c_range = float(c.max() - c.min()) + 1e-10
    beta = 0.03 * c_range
    lr = 0.05
    momentum = 0.7
    v = np.zeros(n)

    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_try = a + momentum * v
        a_try = np.maximum(a_try, 0.0)
        sn = a_try.sum()
        if sn < 1e-10:
            v = np.zeros(n)
            continue
        a_try = a_try / sn

        # Compute smooth max gradient
        c_try = _fft_convolve(a_try)
        cm = float(c_try.max())
        x = (c_try - cm) / beta
        x = np.clip(x, -50.0, 0.0)
        ex = np.exp(x)
        ss = float(ex.sum())
        if not np.isfinite(ss) or ss <= 0:
            break
        w = ex / ss

        N_pad = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N_pad)
        fa_rev = np.fft.rfft(a_try[::-1], N_pad)
        corr = np.fft.irfft(fw * fa_rev, N_pad)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        if not np.all(np.isfinite(grad)):
            beta = beta * 2.0
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a = _project_simplex(a + v)

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.9)
            if no_imp > 5000:
                a = best_a.copy()
                v = np.zeros(n)
                beta = 0.03 * (float(_fft_convolve(a).max()) + 1e-10)
                lr = 0.05
                no_imp = 0

        if it % 200 == 199:
            beta = max(1e-6 * c_range, beta * 0.7)

    return best_a


def _perturb(a, rng, amount=0.1):
    """Add random perturbation and re-normalize."""
    n = len(a)
    noise = rng.normal(0, amount, size=n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS-B, peak reduction, LP refinement,
    gradient polish, with multiple n values and perturbations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is None:
        parent = _make_ramp(768, 1.0)
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"No parent, using ramp init: score={best_score:.10f}")

    n_parent = parent.size

    # === Phase 1: L-BFGS-B on parent ===
    if time.monotonic() < deadline - 300:
        lb_end = time.monotonic() + min(400.0, deadline - time.monotonic() - 200)
        print(f"Phase 1: L-BFGS-B on parent n={n_parent}, budget={lb_end - time.monotonic():.0f}s")
        a_lb = _lbfgs_optimize(parent, lb_end, beta_start=0.05, beta_end=0.003, max_iter=3000)
        sc_lb = _score(a_lb)
        print(f"  L-BFGS result: score={sc_lb:.10f}")
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()

    # === Phase 2: Peak reduction ===
    if best_seq is not None and time.monotonic() < deadline - 200:
        pr_end = time.monotonic() + min(250.0, deadline - time.monotonic() - 150)
        print(f"Phase 2: Peak reduction, budget={pr_end - time.monotonic():.0f}s")
        a_pr = _peak_reduce(best_seq, pr_end, max_iter=100000)
        sc_pr = _score(a_pr)
        print(f"  Peak reduce result: score={sc_pr:.10f}")
        if sc_pr < best_score:
            best_score = sc_pr
            best_seq = a_pr.copy()

    # === Phase 3: LP refinement ===
    if best_seq is not None and time.monotonic() < deadline - 150:
        lp_end = time.monotonic() + min(200.0, deadline - time.monotonic() - 100)
        print(f"Phase 3: LP refinement, budget={lp_end - time.monotonic():.0f}s")
        a_lp = _lp_refine(best_seq, lp_end, max_iter=20000, inflate_start=0.005, inflate_min=0.0005)
        sc_lp = _score(a_lp)
        print(f"  LP result: score={sc_lp:.10f}")
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()

    # === Phase 4: L-BFGS-B again (second pass) ===
    if best_seq is not None and time.monotonic() < deadline - 100:
        lb2_end = time.monotonic() + min(150.0, deadline - time.monotonic() - 80)
        print(f"Phase 4: L-BFGS-B pass 2, budget={lb2_end - time.monotonic():.0f}s")
        a_lb2 = _lbfgs_optimize(best_seq, lb2_end, beta_start=0.02, beta_end=0.001, max_iter=3000)
        sc_lb2 = _score(a_lb2)
        print(f"  L-BFGS-2 result: score={sc_lb2:.10f}")
        if sc_lb2 < best_score:
            best_score = sc_lb2
            best_seq = a_lb2.copy()

    # === Phase 5: Gradient polish ===
    if best_seq is not None and time.monotonic() < deadline - 50:
        gp_end = time.monotonic() + min(100.0, deadline - time.monotonic() - 30)
        print(f"Phase 5: Gradient polish, budget={gp_end - time.monotonic():.0f}s")
        a_gp = _gradient_polish(best_seq, gp_end, max_iter=30000)
        sc_gp = _score(a_gp)
        print(f"  Gradient polish result: score={sc_gp:.10f}")
        if sc_gp < best_score:
            best_score = sc_gp
            best_seq = a_gp.copy()

    # === Phase 6: Try perturbations + quick re-optimization ===
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_perturb = min(5, int((deadline - time.monotonic() - 30) / 30))
        print(f"Phase 6: {n_perturb} perturbations")
        for p_i in range(n_perturb):
            if time.monotonic() > deadline - 30:
                break
            pert_amount = 0.05 * (p_i + 1) * 0.3
            a_pert = _perturb(best_seq, rng, amount=pert_amount)
            # Quick peak reduction
            pe_end = time.monotonic() + 20.0
            if time.monotonic() > deadline - 30:
                break
            a_pf = _peak_reduce(a_pert, pe_end, max_iter=20000)
            sc_pf = _score(a_pf)
            if sc_pf < best_score:
                best_score = sc_pf
                best_seq = a_pf.copy()
                print(f"  Perturb {p_i}: score={sc_pf:.10f} (improved!)")
            else:
                print(f"  Perturb {p_i}: score={sc_pf:.10f}")

    # === Phase 7: Try different n values ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        # Try a couple of nearby n values
        for n_try in [n_parent - 32, n_parent + 32]:
            if n_try < 64 or time.monotonic() > deadline - 40:
                continue
            a_interp = _interpolate(best_seq, n_try)
            a_interp = _symmetrize(a_interp)
            a_interp = np.maximum(a_interp, 0.0)
            a_interp = a_interp / a_interp.sum()
            sc_int = _score(a_interp)
            print(f"  n={n_try} interp: score={sc_int:.10f}")
            if sc_int < best_score:
                best_score = sc_int
                best_seq = a_interp.copy()
            # Quick optimize
            if time.monotonic() < deadline - 40:
                pe_end = time.monotonic() + min(30.0, deadline - time.monotonic() - 30)
                a_opt = _peak_reduce(a_interp, pe_end, max_iter=30000)
                sc_opt = _score(a_opt)
                print(f"  n={n_try} optimized: score={sc_opt:.10f}")
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_seq = a_opt.copy()

    # === Final: one more gradient polish if time remains ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        gp2_end = time.monotonic() + min(30.0, deadline - time.monotonic() - 10)
        a_final_gp = _gradient_polish(best_seq, gp2_end, max_iter=10000)
        sc_final_gp = _score(a_final_gp)
        if sc_final_gp < best_score:
            best_score = sc_final_gp
            best_seq = a_final_gp.copy()
        print(f"  Final polish: score={sc_final_gp:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(768, 1.0)
        best_score = _score(best_seq)

    # Final validation and normalization
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(768, 1.0)

    # Verify with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    total_time = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={total_time:.1f}s")
    return [float(x) for x in final_seq]
