from scipy.optimize import linprog
from scipy.sparse import csc_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b=None):
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size + b.size - 1
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n]
    # Remove tiny negative values from numerical errors
    c = np.maximum(c, 0)
    return c

def _score_normalized(a):
    """Score assuming sum(a) = 1: 2n * max(conv(a,a))."""
    n = len(a)

def _gradient_step(a, T, lr):
    """One gradient step using log-sum-exp smoothing."""
    n = len(a)
    c = _fft_convolve(a)
    x = (c - c.max()) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    w = ex / ex.sum()
    
    # Gradient: 2 * conv(a_reversed, w)[n-1 : 2n-1]
    a_rev = a[::-1]
    h = _fft_convolve(a_rev, w)
    grad = 2.0 * h[n-1:2*n-1]
    
    # Project to tangent space of simplex
    grad_tan = grad - np.mean(grad)
    
    # Update
    a_new = a - lr * grad_tan
    # Project to simplex
    a_new = _project_simplex(a_new)

LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses full-constraint LP with adaptive inflation, line search,
and gradient-based polishing.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n_out = a.size + b.size - 1
    N = _next_pow2(n_out)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n_out]
    # Clean up tiny negatives
    c[~np.isfinite(c)] = 0
    c = np.maximum(c, 0)
    return c

def _score_norm(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _build_A(a):
    """Build sparse convolution constraint matrix."""
    n = len(a)
    m = 2 * n - 1
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    return csc_matrix((data, (rows, cols)), shape=(m, n))

    """LP: max sum(b) s.t. (a*b)_k <= M, b >= 0. Returns b or None."""
    n = len(a)
    A = _build_A(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

    """Find best t in [0,1] for (1-t)*a + t*g."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:

        x = x / s
        return _score_norm(x)

            c = hi - gr * (hi - lo)
            fc = f(c)

            d = lo + gr * (hi - lo)
