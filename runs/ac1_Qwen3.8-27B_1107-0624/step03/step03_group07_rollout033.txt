def compute_score(a):

    if s < 0.01:

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)

        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: exact subgradient with Nesterov acceleration,
adaptive top-K peak weighting, direct a-space polish, and more time at final scale.

import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_h(w, a):
    """h[i] = sum_k w[k]*a[k-i] via FFT cross-correlation."""
    n = len(a)
    N = next_pow2(n + len(w) - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fw * np.conj(fa), N)
    return corr[:n]


def softmax_fn(z):
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-300:
        return np.ones(z.size) / z.size
    return a / s


def get_adaptive_weights(c, min_gap=1e-9, max_k=5):
    """Get uniform weights on peaks within min_gap of the max, up to max_k."""
    max_c = c.max()
    top_k = min(max_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_indices = top_indices[c[top_indices] >= max_c - min_gap]
    w = np.zeros(len(c))
    w[top_indices] = 1.0 / len(top_indices)
    return w


def nestrov_optimize(a0, deadline, rng, lr=0.03, momentum0=0.9, momentum_max=0.97,
                     print_interval=60, restart_every=2000, lr_floor=0.001):
    """Nesterov-accelerated subgradient in logit space."""
    n = a0.size
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()
    v = np.zeros(n)
    beta = momentum0

    best_a = a0.copy()
    best_z = z.copy()
    c = full_convolve_fft(a0)
    best_maxc = float(c.max())

    no_improve = 0
    t = 0
    current_lr = lr
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        if time.monotonic() - last_print > print_interval:
            print(f"      nestrov: t={t}, maxc={best_maxc:.10f}, score={2.0*n*best_maxc:.10f}, lr={current_lr:.6f}")
            last_print = time.monotonic()

        y = z + beta * v
        a_y = softmax_fn(y)

        c_y = full_convolve_fft(a_y)
        w = get_adaptive_weights(c_y, min_gap=1e-9, max_k=3)

        h = compute_h(w, a_y)
        g_a = 2.0 * h
        dot = float(np.dot(a_y, g_a))
        g_z = a_y * (g_a - dot)

        v = beta * v - current_lr * g_z
        z = z + v
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        a = softmax_fn(z)
        c = full_convolve_fft(a)
        maxc = float(c.max())

        if maxc < best_maxc - 1e-14:
            best_maxc = maxc
            best_a = a.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        t += 1

        if no_improve % 500 == 0 and no_improve > 0:
            current_lr = max(lr_floor, current_lr * 0.99)
            beta = min(momentum_max, beta + 0.0003)

        if no_improve > restart_every:
            noise = 0.03 * rng.normal(size=n)
            z = best_z + noise
            z = z - z.mean()
            v = np.zeros(n)
            current_lr = lr
            beta = momentum0
            no_improve = 0

    return best_a, best_z, 2.0 * n * best_maxc

def soft_max_optimize(a0, deadline, rng, lr=0.05, beta_factor=0.05,
                      beta_decay=0.95, beta_min=1e-10,
                      print_interval=60, restart_every=2000):
    """Soft-max smoothed gradient with Adam (for exploration phase)."""
    n = a0.size
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    beta = beta_factor * max_c

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_maxc = max_c
    no_improve = 0
    current_lr = lr
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        if time.monotonic() - last_print > print_interval:
            print(f"      softmax: t={t_adam}, maxc={best_maxc:.10f}, score={2.0*n*best_maxc:.10f}, beta={beta:.2e}")
            last_print = time.monotonic()

        x = (c - max_c) / beta
        x = np.clip(x, -60.0, 0.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        h = compute_h(w, a0)
        g_a = 2.0 * h
        dot = float(np.dot(a0, g_a))
        dz = a0 * (g_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(best_a + 1e-30)
            z = z - z.mean()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = softmax_fn(z)
        c = full_convolve_fft(a0)
        max_c = float(c.max())

        if max_c < best_maxc - 1e-14:
            best_maxc = max_c
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 300:
            current_lr = max(0.005, current_lr * 0.97)

        if no_improve > restart_every:
            z = np.log(best_a + 1e-30) + 0.1 * rng.normal(size=n)
            z = z - z.mean()
            z = np.clip(z, -40.0, 40.0)
            a0 = softmax_fn(z)
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = beta_factor * max_c
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = lr
            no_improve = 0

    return best_a, 2.0 * n * best_maxc


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_diverse_profiles(n, rng):
    profiles = []
    for eta in [0.5, 1.0, 2.0, 3.0]:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for k in [1, 2, 3]:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for _ in range(4):
        coeffs = rng.uniform(-0.3, 0.3, 4)
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2.0 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search using Nesterov-accelerated exact subgradient,
    soft-max exploration, and direct a-space polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Try to load previous best
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"  Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Quick exploration at n=512 =====
    n1 = 512
    phase1_end = t_start + 120.0
    print(f"\n--- Phase 1: n={n1} exploration ---")

    profiles_1 = make_diverse_profiles(n1, rng)
    if init_seq is not None:
        profiles_1.insert(0, interpolate_profile(init_seq, n1))

    best_p1 = None
    best_s1 = np.inf
    for idx, p in enumerate(profiles_1[:8]):
        if time.monotonic() > phase1_end - 5:
            break
        dl = min(phase1_end, time.monotonic() + 12.0)
        r, sc = soft_max_optimize(p, dl, rng, lr=0.04, beta_factor=0.08, beta_decay=0.92)
        sc_f = compute_score(r)
        if sc_f < best_s1:
            best_s1 = sc_f
            best_p1 = r.copy()
        update(r, sc_f)
        if idx < 4:
            print(f"    Profile {idx}: {sc_f:.8f}")

    if best_p1 is not None:
        print(f"  Phase 1 best: {best_s1:.10f}")

    # ===== Phase 2: Main optimization at n=1024 =====
    n2 = 1024
    phase2_end = t_start + 600.0
    print(f"\n--- Phase 2: n={n2} main (Nesterov) ---")

    if init_seq is not None and init_seq.size >= n2:
        p2 = interpolate_profile(init_seq, n2)
    elif best_p1 is not None:
        p2 = interpolate_profile(best_p1, n2)
    else:
        p2 = interpolate_profile(profiles_1[0], n2)

    sc2_init = compute_score(p2)
    update(p2, sc2_init)
    print(f"  Interpolated: {sc2_init:.10f}")

    # Nesterov at n=1024
    dl2 = min(phase2_end, time.monotonic() + 300.0)
    r2, _, sc2 = nestrov_optimize(p2, dl2, rng, lr=0.02, momentum0=0.85, momentum_max=0.95)
    sc2_f = compute_score(r2)
    update(r2, sc2_f)
    print(f"  After Nesterov n=1024: {sc2_f:.10f}")

    # Also try from different starting point
    if time.monotonic() < phase2_end - 60:
        p2b = make_diverse_profiles(n2, rng)[1]
        dl2b = min(phase2_end, time.monotonic() + 60.0)
        r2b, _, sc2b = nestrov_optimize(p2b, dl2b, rng, lr=0.02, momentum0=0.85)
        sc2b_f = compute_score(r2b)
        update(r2b, sc2b_f)
        print(f"  Alt start n=1024: {sc2b_f:.10f}")

    # ===== Phase 3: Optimization at n=2048 =====
    n3 = 2048
    phase3_end = t_start + 1300.0
    print(f"\n--- Phase 3: n={n3} (Nesterov + direct) ---")

    if best_seq is not None:
        p3 = interpolate_profile(best_seq, n3)
    else:
        p3 = interpolate_profile(r2, n3)

    sc3_init = compute_score(p3)
    update(p3, sc3_init)
    print(f"  Interpolated: {sc3_init:.10f}")

    if init_seq is not None and init_seq.size == n3:
        p3 = init_seq.copy()
        sc3_init = compute_score(p3)
        update(p3, sc3_init)
        print(f"  Using previous best directly: {sc3_init:.10f}")

    # Long Nesterov at n=2048
    dl3 = min(phase3_end, time.monotonic() + 350.0)
    r3, _, sc3 = nestrov_optimize(p3, dl3, rng, lr=0.01, momentum0=0.85, momentum_max=0.95)
    sc3_f = compute_score(r3)
    update(r3, sc3_f)
    print(f"  After Nesterov n=2048: {sc3_f:.10f}")

    # Direct a-space optimization
    if time.monotonic() < phase3_end - 100:
        dl3b = min(phase3_end, time.monotonic() + 100.0)
        r3b, sc3b = direct_a_optimize(r3, dl3b, rng, lr=0.0005, momentum0=0.85)
        sc3b_f = compute_score(r3b)
        update(r3b, sc3b_f)
        print(f"  After direct n=2048: {sc3b_f:.10f}")

    # ===== Phase 4: Try n=4096 if time allows =====
    n4 = 4096
    if time.monotonic() < phase3_end - 150 and best_score > 1.503:
        phase4_end = min(phase3_end, t_start + 1500.0)
        print(f"\n--- Phase 4: n={n4} ---")
        p4 = interpolate_profile(best_seq if best_seq is not None else r3, n4)
        sc4_init = compute_score(p4)
        update(p4, sc4_init)
        print(f"  Interpolated: {sc4_init:.10f}")

        dl4 = min(phase4_end, time.monotonic() + 150.0)
        r4, _, sc4 = nestrov_optimize(p4, dl4, rng, lr=0.005, momentum0=0.8, momentum_max=0.9)
        sc4_f = compute_score(r4)
        update(r4, sc4_f)
        print(f"  After Nesterov n=4096: {sc4_f:.10f}")

    # ===== Phase 5: Final polish =====
    phase5_end = deadline
    print(f"\n--- Phase 5: Final polish ---")

    if best_seq is not None:
        n_polish = best_seq.size
        p_polish = best_seq.copy()

        # Multiple Nesterov rounds with decreasing lr
        for round_idx, (lr_val, mom) in enumerate([
            (0.005, 0.85), (0.003, 0.9), (0.001, 0.95),
        ]):
            if time.monotonic() > phase5_end - 30:
                break
            dl = min(phase5_end, time.monotonic() + 80.0)
            r, _, sc = nestrov_optimize(p_polish, dl, rng, lr=lr_val, momentum0=mom,
                                         momentum_max=min(0.99, mom + 0.05),
                                         print_interval=30, restart_every=1000)
            sc_f = compute_score(r)
            update(r, sc_f)
            if sc_f < compute_score(p_polish) - 1e-10:
                p_polish = r
            print(f"    Polish {round_idx} (lr={lr_val}): {sc_f:.10f}")

        # Direct optimization final
        if time.monotonic() < phase5_end - 20:
            dl = min(phase5_end, time.monotonic() + 40.0)
            r, sc = direct_a_optimize(p_polish, dl, rng, lr=0.0002, momentum0=0.9)
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"    Direct final: {sc_f:.10f}")

    # ===== Try smaller n if larger isn't better =====
    if best_seq is not None and best_seq.size > 512:
        for n_try in [512, 1024]:
            if time.monotonic() > deadline - 15:
                break
            p_small = interpolate_profile(best_seq, n_try)
            sc_s = compute_score(p_small)
            update(p_small, sc_s)
            if sc_s < best_score + 0.005:
                dl_s = min(deadline, time.monotonic() + 20.0)
                r_s, _, sc_s2 = nestrov_optimize(p_small, dl_s, rng, lr=0.01, momentum0=0.85)
                sc_s2_f = compute_score(r_s)
                update(r_s, sc_s2_f)
                print(f"    n={n_try}: {sc_s2_f:.10f}")

    print(f"\n  Final best score: {best_score:.10f}")

    if best_seq is None:
        best_seq = np.ones(512) / 512.0

    final_seq = np.maximum(best_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Eval check: {e}")

    return [float(x) for x in final_seq]
