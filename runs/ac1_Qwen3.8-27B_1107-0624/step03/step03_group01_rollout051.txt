def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _full_convolve_self(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve_self(a)
    return float(2.0 * n * c.max())


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve_self(a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    c_shifted = np.clip(c_shifted, -50, 50)
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve_self(a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    c_shifted = np.clip(c_shifted, -50, 50)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _make_gaussian_bump(n, width):
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    a = np.exp(-((t - center) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_sine_bump(n, k=1):

    a = np.maximum(a, 1e-15)
    a[0] = max(a[0], 1e-15)
    a[-1] = max(a[-1], 1e-15)
    return a / a.sum()

    a = np.cos(np.pi * (t - 0.5)) ** (2.0 * power)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_asymmetric_bump(n, alpha=2.0, beta_p=1.0):

    a = np.maximum(a, 1e-15)
    a[0] = max(a[0], 1e-15)
    a[-1] = max(a[-1], 1e-15)

    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)

def _make_chirp_profile(n, freq_start=0.5, freq_end=3.0):
    t = np.linspace(0.0, 1.0, n)
    phases = 2.0 * np.pi * (freq_start * t + (freq_end - freq_start) * t ** 2 / 2.0)
    a = np.abs(np.cos(phases))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _run_lbfgs_continuation(a_init, deadline, rng, max_wall_time=None):

    z = np.log(np.maximum(a_init, 1e-15))

    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve_self(a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_starts]

    if max_wall_time is not None:
        per_level = max_wall_time / len(betas)
    else:
        per_level = 60.0

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level * 1.5)

        try:
            maxiter = max(100, int(200 * (1.0 - bi * 0.05)))
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 40, 'maxfun': 3000}

            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)

                if np.isfinite(s_new) and s_new < best_score:

                    best_a = a_new.copy()
                    z = z_new.copy()
                else:
                    z = z_new.copy()
        except Exception:
            pass

        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            grad = np.zeros_like(z)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 500 == 0:
            a_check = _softmax_fn(z)
            sc = _score_norm_fn(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            beta = max(1e-12, beta * 0.9)
            lr = base_lr * max(0.1, 1.0 - 0.01 * (t // 5000))

    a_check = _softmax_fn(z)
    sc = _score_norm_fn(a_check)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_check.copy()

    return best_a, best_score


def _refined_lbfgs(a_init, deadline, rng):
    """Multiple L-BFGS runs with progressively smaller beta, with restarts."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve_self(a_init)
    max_c0 = float(c0.max())

    # Start with smaller betas for fine polishing
    betas = [0.001 * max_c0, 0.0005 * max_c0, 0.0002 * max_c0, 0.0001 * max_c0, 0.00005 * max_c0]
    per_level = max(5.0, (deadline - time.monotonic()) / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 2:
            break
        level_deadline = min(deadline - 1, time.monotonic() + per_level)

        # Try 2 restarts at each beta level
        for restart in range(2):
            if time.monotonic() > level_deadline:
                break
            z_try = z.copy()
            if restart > 0:
                z_try = z_try + 0.01 * rng.normal(size=n)
                z_try = z_try - z_try.mean()

            try:
                result = minimize(
                    _lbfgs_objective, z_try, args=(beta, n),
                    method='L-BFGS-B', jac=_lbfgs_gradient,
                    options={'maxiter': 2000, 'ftol': 1e-17, 'gtol': 1e-11,
                             'maxls': 50, 'maxfun': 5000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
            except Exception:
                pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining diverse initializations, adaptive L-BFGS
    with logsumexp continuation, Adam-based peak equalization, and length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}, t={elapsed:.1f}s) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={_score_fn(parent):.10f}")
    else:
        n_parent = 2048
        parent = _make_cosine_window(n_parent, 1.0)
        print(f"No parent, using cosine window n={n_parent}")

    # --- Phase 0: Diverse initialization (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    if parent is not None:
        sc = _score_norm_fn(parent)
        candidates.append((sc, parent.copy(), len(parent), "parent"))
        update_best(parent, sc, "parent")

    n_base = len(parent)

    # Generate diverse profiles at n_base
    profiles_n = []
    profiles_n.append(_make_cosine_window(n_base, 1.0))
    profiles_n.append(_make_cosine_window(n_base, 0.7))
    profiles_n.append(_make_cosine_window(n_base, 1.5))
    profiles_n.append(_make_gaussian_bump(n_base, 0.3))
    profiles_n.append(_make_gaussian_bump(n_base, 0.4))
    profiles_n.append(_make_gaussian_bump(n_base, 0.5))
    profiles_n.append(_make_sine_bump(n_base, 1))
    profiles_n.append(_make_sine_bump(n_base, 2))
    profiles_n.append(_make_asymmetric_bump(n_base, 1.5, 1.0))
    profiles_n.append(_make_asymmetric_bump(n_base, 1.0, 1.5))
    profiles_n.append(_make_linear_profile(n_base, 1.0))
    profiles_n.append(_make_linear_profile(n_base, 2.0))
    profiles_n.append(_make_chirp_profile(n_base, 0.5, 2.0))
    profiles_n.append(_make_chirp_profile(n_base, 1.0, 3.0))

    # Perturb parent
    z_p = np.log(np.maximum(parent, 1e-15))
    z_p = z_p - z_p.mean()
    for noise_level in [0.05, 0.1, 0.2]:
        z_noisy = z_p + noise_level * rng.normal(size=n_base)
        z_noisy = np.clip(z_noisy, -50, 50)
        z_noisy = z_noisy - z_noisy.mean()
        profiles_n.append(_softmax_fn(z_noisy))

    for idx, p in enumerate(profiles_n):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, n_base, f"init{idx}"))
        update_best(p, sc, f"init{idx}")

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0: {len(candidates)} candidates, best={best_score:.10f}")
    for i, (sc, p, pn, lbl) in enumerate(candidates[:5]):
        print(f"    {i+1}. {sc:.10f} n={pn} [{lbl}]")

    # --- Phase 1: L-BFGS continuation on top candidates (400s) ---
    phase1_end = min(deadline, t_start + 430.0)
    count = 0
    seen = set()

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= 5:
            break
        if sc > best_score * 1.08:
            break
        key = (pn, label[:8])
        if key in seen:
            continue
        seen.add(key)

        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        per_time = min(80.0, remaining / max(1, 5 - count))
        level_deadline = time.monotonic() + per_time
        p_best, sc_best = _run_lbfgs_continuation(p, level_deadline, rng, max_wall_time=per_time)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"Phase 1: best={best_score:.10f}")

    # --- Phase 2: Refined L-BFGS on best (250s) ---
    phase2_end = min(deadline, t_start + 680.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        per_dl = min(phase2_end - 5, time.monotonic() + 180.0)
        p_best, sc_best = _refined_lbfgs(best_seq, per_dl, rng)
        update_best(p_best, sc_best, "refined_lbfgs")

        # Another round with tiny perturbation
        if time.monotonic() < phase2_end - 30:
            n_best = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + 0.005 * rng.normal(size=n_best)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            per_dl2 = min(phase2_end - 5, time.monotonic() + 80.0)
            p_best2, sc_best2 = _refined_lbfgs(p_start, per_dl2, rng)
            update_best(p_best2, sc_best2, "refined_perturb")

    print(f"Phase 2: best={best_score:.10f}")

    # --- Phase 3: Adam optimization (300s) ---
    phase3_end = min(deadline, t_start + 980.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 5, time.monotonic() + 200.0)
        p_best, sc_best = _adam_optimize(best_seq, per_dl, rng, lr=0.02, beta_init_frac=0.005)
        update_best(p_best, sc_best, "adam")

        # Second Adam round with different params
        if time.monotonic() < phase3_end - 40:
            per_dl2 = min(phase3_end - 5, time.monotonic() + 100.0)
            p_best2, sc_best2 = _adam_optimize(best_seq, per_dl2, rng, lr=0.01, beta_init_frac=0.002)
            update_best(p_best2, sc_best2, "adam_fine")

    print(f"Phase 3: best={best_score:.10f}")

    # --- Phase 4: Length exploration - try 4096 (400s) ---
    phase4_end = min(deadline, t_start + 1380.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr < 4096:
            target_n = 4096
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interpolated to n={target_n}: score={sc_interp:.10f}")
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase4_end - 60:
                per_dl = min(phase4_end - 30, time.monotonic() + 200.0)
                p_best, sc_best = _run_lbfgs_continuation(p_interp, per_dl, rng, max_wall_time=per_dl - time.monotonic())
                update_best(p_best, sc_best, f"lbfgs_n{target_n}")

            if time.monotonic() < phase4_end - 30:
                per_dl2 = min(phase4_end - 10, time.monotonic() + 100.0)
                p_best2, sc_best2 = _adam_optimize(p_interp, per_dl2, rng, lr=0.015, beta_init_frac=0.004)
                update_best(p_best2, sc_best2, f"adam_n{target_n}")

        # Also try n=3072 if we haven't
        if len(best_seq) < 3072 and time.monotonic() < phase4_end - 60:
            target_n2 = 3072
            p_interp2 = _interp_profile(best_seq, target_n2)
            sc_interp2 = _score_norm_fn(p_interp2)
            update_best(p_interp2, sc_interp2, f"interp_n{target_n2}")

            if time.monotonic() < phase4_end - 30:
                per_dl3 = min(phase4_end - 10, time.monotonic() + 80.0)
                p_best3, sc_best3 = _run_lbfgs_continuation(p_interp2, per_dl3, rng, max_wall_time=80.0)
                update_best(p_best3, sc_best3, f"lbfgs_n{target_n2}")

    print(f"Phase 4: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # --- Phase 5: Multi-restart with diverse perturbations (250s) ---
    phase5_end = min(deadline, t_start + 1630.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = max(10.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            noise_levels = [0.002, 0.005, 0.01, 0.02]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_levels[i % len(noise_levels)] * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            # Use refined L-BFGS (small beta)
            per_dl = min(phase5_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _refined_lbfgs(p_start, per_dl, rng)
            update_best(p_best, sc_best, f"restart{i}_n{n_best}")

    print(f"Phase 5: best={best_score:.10f}")

    # --- Phase 6: Final fine polish (160s) ---
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        # Final L-BFGS with very small beta
        n_best = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()

        c_final = _full_convolve_self(best_seq)
        max_c_final = float(c_final.max())

        # Very fine beta
        beta_fine = 0.00005 * max_c_final
        per_dl = min(phase6_end - 10, time.monotonic() + 100.0)
        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta_fine, n_best),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_seq = a_new.copy()
                    print(f"  Final L-BFGS: {best_score:.10f}")
        except Exception:
            pass

        # Final Adam with very small beta
        if time.monotonic() < phase6_end - 10:
            per_dl2 = min(phase6_end - 5, time.monotonic() + 50.0)
            p_best, sc_best = _adam_optimize(best_seq, per_dl2, rng, lr=0.005, beta_init_frac=0.001)
            update_best(p_best, sc_best, "final_adam")

    print(f"Phase 6: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_cosine_window(2048, 1.0)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_cosine_window(2048, 1.0)
    print(f"FALLBACK: n=2048, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
