Given current sequence f (normalized to sum 1) and M = max(f*f):
LP: maximize sum(g) subject to:
- g >= 0

The optimal g* has sum(g*) >= sum(f) = 1 (since f itself is feasible: (f*f)[k] <= M for all k).

def compute_conv_and_score(a):
    """Compute full convolution and score."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64), np.inf
    s = float(a.sum())

        return np.convolve(a, a), np.inf
    c = np.convolve(a, a)
    score = 2.0 * n * c.max() / (s * s)

def fft_convolve(a):
    """Fast full linear convolution using FFT."""
    n = len(a)
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)

"""
Multi-scale optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 using
exact active-set gradients, LP-based structured updates, and Adam in logit space.
Combines diverse initialization, periodic LP refinement, and coordinate descent polish.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a):
    """Full linear convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def compute_active_gradient(a, c, rel_tol=0.001):
    """
    Compute gradient of max(c) w.r.t. a using active set of near-max lags.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    max_c = float(c.max())
    if max_c < 1e-30:
        return np.ones(n)
    threshold = max_c * (1.0 - rel_tol)
    active = np.where(c >= threshold)[0]
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j_start = k - hi
            j_end = k - lo
            grad[lo:hi + 1] += a[j_start:j_end + 1]
    grad *= 2.0 / float(len(active))
    return grad


def logit_adam_optimize(a0, deadline, rng, lr=0.03, rel_tol=0.001,
                        restart_period=1500, noise_scale=0.12):
    """
    Adam optimizer in logit space using exact active-set gradient.
    Minimizes 2n*max(conv(a,a))/S^2 over a >= 0.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = fft_convolve(a0)
    max_c = float(c.max())
    S = float(a0.sum())

    z = np.log(a0)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a0.copy()
    best_score = 2.0 * n * max_c / (S * S)
    no_improve = 0
    current_lr = lr

    while time.monotonic() < deadline:
        grad_max = compute_active_gradient(a0, c, rel_tol)
        S = float(a0.sum())
        grad_F = S * grad_max - 2.0 * max_c

        dot = float(np.dot(a0, grad_F))
        dz = a0 * (grad_F - dot)

        t += 1
        m = b1 * m + (1 - b1) * dz
        v = b2 * v + (1 - b2) * (dz * dz)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z -= current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = z - z.mean()
            z = np.clip(z, -50.0, 50.0)

        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        a0 = np.exp(z - z.max())
        S_new = a0.sum()
        if S_new < 1e-15:
            a0 = best_a.copy()
            continue
        a0 /= S_new

        c = fft_convolve(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c / (S_new * S_new)

            a0 = np.exp(z - z.max())

            c = fft_convolve(a0)
            max_c = float(c.max())
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            current_lr = lr
            no_improve = 0

    return best_a, best_score


def lp_update(a, max_time=30.0):
    """
    LP-based update: find g maximizing sum(g) subject to g>=0, ||f*g||_inf <= M.
    Returns updated sequence or None if LP fails.
    """
    t0 = time.monotonic()
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel()
    c = fft_convolve(a)
    M = float(c.max())
    if M < 1e-15:
        return None

    # Build constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    # Shape: (2n-1) x n
    # For efficiency with large n, use sparse
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            j_start = k - hi
            j_end = k - lo
            rows.extend(range(lo, hi + 1))
            cols.extend(range(k - hi, k - lo + 1))
            vals.extend(a[j_start:j_end + 1].tolist())

    if len(vals) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': max_time, 'presolve': True})

    if time.monotonic() - t0 > max_time * 1.5:
        return None

    if res.status != 0 or res.x is None:
        return None

    g = res.x
    sum_g = float(g.sum())
    if sum_g <= 0:
        return None

    g = g / sum_g
    return g


def lp_step(f, t_blend, deadline_lp=20.0):
    """
    Perform one LP-based blended update. Returns new sequence or None.
    """
    g = lp_update(f, max_time=deadline_lp)
    if g is None:
        return None
    f_new = (1.0 - t_blend) * f + t_blend * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return None
    return f_new / s


def coordinate_descent(a, deadline, rng, n_sweeps=5, step_fraction=0.01):
    """
    Simple coordinate descent: for each coordinate, try small perturbations.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_score = compute_score(a)
    indices = np.arange(n)

    while time.monotonic() < deadline:
        rng.shuffle(indices)
        for i in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            # Try decreasing
            a[i] = old_val * (1.0 - step_fraction)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
            else:
                a[i] = old_val
                # Try increasing
                a[i] = old_val * (1.0 + step_fraction)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                else:
                    a[i] = old_val

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def diverse_profiles(n, rng):

def interpolate_profile(p0, n_new):
    """Interpolate profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

        final_seq = np.exp(-((t - 0.5) / 0.3) ** 2)
        final_seq = final_seq / final_seq.sum()

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Eval check: {e}")

    return [float(x) for x in final_seq]
