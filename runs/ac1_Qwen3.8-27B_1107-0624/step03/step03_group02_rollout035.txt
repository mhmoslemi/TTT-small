import numpy as np

from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0 and mn > -1e-10:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval
    return mval + beta * np.log(s / len(c))


def _smooth_max_grad(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return ex / s


def _grad_conv(a, c_weights):
    """Gradient of sum_k w[k]*conv(a,a)[k] with respect to a. Uses FFT."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = c_weights
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lbfgs_smooth_max(a0, deadline, max_iter=200, lr_scale=1.0):
    """L-BFGS optimization in log-space for smooth max of convolution."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a) - np.log(a).mean()
    c = _fft_convolve(a)
    beta = max(0.01 * float(c.max()), 1e-10)

    best_z = z.copy()
    c_best = _fft_convolve(a)
    best_obj = 2.0 * n * float(c_best.max())

    def objective(zv):
        nonlocal beta
        e = np.exp(np.clip(zv, -50, 50))
        av = e / e.sum()
        cv = _fft_convolve(av)
        return _smooth_max(cv, beta)

    def gradient(zv):
        nonlocal beta
        e = np.exp(np.clip(zv, -50, 50))
        av = e / e.sum()
        cv = _fft_convolve(av)
        w = _smooth_max_grad(cv, beta)
        grad_conv_val = _grad_conv(av, w)
        dot = float(np.dot(av, grad_conv_val))
        grad_z = av * (grad_conv_val - dot)
        return grad_z

    result = minimize(objective, z, jac=gradient, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'maxls': 20,
                               'iprint': -1, 'ftol': 1e-12, 'gtol': 1e-8})

    e = np.exp(np.clip(result.x, -50, 50))
    a_new = e / e.sum()
    c_new = _fft_convolve(a_new)
    obj_new = 2.0 * n * float(c_new.max())

    # Also track best during iterations via callback isn't available,
    # so just use the final result and compare with initial
    if obj_new < best_obj:
        best_obj = obj_new
        best_z = result.x.copy()
    # If we didn't improve, try again with smaller beta
    if obj_new >= best_obj - 1e-10:
        beta = max(0.005 * float(_fft_convolve(a_new).max()), 1e-10)

        def objective2(zv):
            e2 = np.exp(np.clip(zv, -50, 50))
            av2 = e2 / e2.sum()
            cv2 = _fft_convolve(av2)
            return _smooth_max(cv2, beta)

        def gradient2(zv):
            e2 = np.exp(np.clip(zv, -50, 50))
            av2 = e2 / e2.sum()
            cv2 = _fft_convolve(av2)
            w2 = _smooth_max_grad(cv2, beta)
            gc = _grad_conv(av2, w2)
            dot2 = float(np.dot(av2, gc))
            return av2 * (gc - dot2)

        result2 = minimize(objective2, result.x, jac=gradient2, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'maxls': 20,
                                    'iprint': -1, 'ftol': 1e-12, 'gtol': 1e-8})
        e3 = np.exp(np.clip(result2.x, -50, 50))
        a_new2 = e3 / e3.sum()
        c_new2 = _fft_convolve(a_new2)
        obj_new2 = 2.0 * n * float(c_new2.max())
        if obj_new2 < best_obj:
            best_obj = obj_new2
            a_new = a_new2

    c_final = _fft_convolve(a_new)
    final_score = 2.0 * n * float(c_final.max())
    return a_new, final_score


def _coordinate_descent(a0, deadline, max_sweeps=500, random_order=True, seed=42):
    """Coordinate descent with exact 1D subproblem solving."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0, np.inf
    a = a / s

    conv = _fft_convolve(a).copy()
    best_a = a.copy()
    best_max = float(conv.max())
    rng = np.random.default_rng(seed)

    order = np.arange(n) if not random_order else None

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if random_order:
            rng.shuffle(order)

        improved = False
        for j in order:
            old_val = a[j]
            if old_val < 1e-15:
                # Try setting to a small positive value
                # Compute what conv would be
                pass

            # Compute c[k] and d[k] for the 1D subproblem
            # conv[k] = c[k] + d[k]*t + q[k]*t^2 where t = a[j]
            # d[k] = 2*a[k-j] if 0<=k-j<n and k!=2j else 2*a[j] if k==2j else 0
            # q[k] = 1 if k==2j else 0
            # c[k] = conv[k] - d[k]*old_val - q[k]*old_val^2

            # Vectorized computation of d and c
            kj = np.arange(2 * n - 1) - j  # k - j for each k
            valid = (kj >= 0) & (kj < n)
            d = np.zeros(2 * n - 1)
            d[valid] = 2.0 * a[kj[valid]]
            # Fix k=2j case: d[2j] should be 2*a[j] (already correct since kj[2j]=j, a[j])
            # Actually d[2j] = 2*a[2j-j] = 2*a[j]. But the actual term is a[j]^2, not 2*a[j]*t.
            # So for k=2j: conv[2j] has a[j]^2, meaning the "linear coeff" is 2*a[j] and
            # the "quadratic coeff" is 1. Let's handle it:
            q = np.zeros(2 * n - 1)
            if 2 * j < 2 * n - 1:
                q[2 * j] = 1.0
                # For k=2j, the actual contribution is a[j]^2 = t^2
                # d[2j] should be 0 for the linear part, and q[2j]=1 for quadratic
                # Wait: conv[2j] = ... + a[j]*a[j] + ... = ... + t^2 + ...
                # So d[2j] = 0, q[2j] = 1
                d[2 * j] = 0.0

            c_vals = conv - d * old_val - q * old_val * old_val

            # Now solve: min_t max_k (c_vals[k] + d[k]*t + q[k]*t^2) s.t. t >= 0
            # The q term is nonzero only at k=2j, so:
            # f(t) = max(max_k≠2j (c[k]+d[k]*t), c[2j]+d[2j]*t+t^2)
            # = max(max_k≠2j (c[k]+d[k]*t), c[2j]+t^2)  [since d[2j]=0]

            # Use golden section search on [0, t_hi]
            # Find t_hi: start at t=0, increase until f is increasing
            def f_1d(t):
                linear_part = c_vals + d * t
                # Handle quadratic term
                if 2 * j < len(c_vals):
                    linear_part[2 * j] = c_vals[2 * j] + t * t
                return float(linear_part.max())

            t_hi = 1.0
            f_hi = f_1d(t_hi)
            f_0 = f_1d(0.0)
            # Expand t_hi until f is clearly increasing
            for _ in range(20):
                if f_hi < f_0 or (t_hi > 1.0 and f_hi >= f_1d(t_hi * 0.5) * 0.999):
                    break
                t_hi *= 2.0
                f_hi = f_1d(t_hi)
                if t_hi > 10.0:
                    break

            # Golden section search
            if t_hi < 1e-15:
                new_val = 0.0
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                lo = 0.0
                hi = t_hi
                f_lo = f_0
                f_hi = f_1d(hi)
                if f_hi < f_lo:
                    # Minimum might be at hi or beyond, but we'll search [0, hi*2]
                    lo = hi
                    hi = 2.0 * t_hi
                    f_lo = f_hi
                    f_hi = f_1d(hi)

                for _ in range(40):
                    m1 = lo + gr * (hi - lo)
                    m2 = hi - gr * (hi - lo)
                    f_m1 = f_1d(m1)
                    f_m2 = f_1d(m2)
                    if f_m1 < f_m2:
                        hi = m2
                        f_hi = f_m2
                    else:
                        lo = m1
                        f_lo = f_m1
                new_val = (lo + hi) / 2.0

            # Clamp to reasonable range
            new_val = max(0.0, min(new_val, 1.0))

            if abs(new_val - old_val) > 1e-14:
                a[j] = new_val
                # Update conv incrementally
                delta = new_val - old_val
                # conv[k] changes by: delta * 2*a[k-j] for k!=2j, and delta*(2*a[j]+delta) for k=2j
                for k_idx in np.where(valid)[0]:
                    k = k_idx
                    if k != 2 * j:
                        conv[k] += delta * 2.0 * a[k - j]
                    # else k==2j handled below
                if 2 * j < len(conv):
                    # conv[2j] = old_conv[2j] - old_val^2 + new_val^2
                    conv[2 * j] += new_val * new_val - old_val * old_val
                improved = True

        # Renormalize periodically
        if sweep % 10 == 0:
            s_cur = a.sum()
            if s_cur > 0:
                a = a / s_cur
                conv = _fft_convolve(a)

        cur_max = float(conv.max())
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
        elif not improved:
            if sweep > 5:
                break

    s_final = best_a.sum()
    if s_final > 0:
        best_a = best_a / s_final
    c_final = _fft_convolve(best_a)
    final_score = 2.0 * n * float(c_final.max()) / (s_final * s_final) if s_final > 0 else np.inf
    return best_a, final_score


def _fictitious_play(a0, deadline, max_iter=5000, n_inner=50):
    """Game-theoretic fictitious play."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0, np.inf
    a = a / s

    # p is distribution over lags (2n-1 of them)
    p = np.ones(2 * n - 1) / (2 * n - 1)

    best_a = a.copy()
    conv = _fft_convolve(a)
    best_score = 2.0 * n * float(conv.max())

    a_avg = a.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Inner: minimize a^T Q_p a = sum_k p[k] * conv(a,a)[k]
        # Using projected gradient on simplex
        a_inner = a_avg.copy()
        for inner_it in range(n_inner):
            conv_i = _fft_convolve(a_inner)
            obj_val = float(np.dot(p, conv_i))
            # Gradient: 2 * sum_k p[k] * (a shifted by k)
            # grad[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
            # This is 2 * conv(p, a_rev) reversed... let me compute directly
            # grad[j] = 2 * sum_{k: 0<=k-j<2n-1} p[k] * a[k-j]
            # = 2 * (p * a)[j] where (p*a) is correlation
            # Actually: sum_k p[k]*a[k-j] = (a convolve p)[j] ... not quite
            # Let me use FFT
            N = _next_pow2(3 * n)
            p_pad = np.zeros(N)
            p_pad[:2 * n - 1] = p
            a_pad = np.zeros(N)
            a_rev = a_inner[::-1]
            a_pad[:n] = a_rev
            grad = 2.0 * np.fft.irfft(np.fft.rfft(p_pad, N) * np.fft.rfft(a_pad, N), N)[n-1:2*n-1]

            lr = 1.0 / (float(np.dot(grad, grad)) + 1e-10) * 0.5
            a_new = a_inner - lr * grad
            a_new = _project_simplex(a_new)

            # Check convergence
            if np.max(np.abs(a_new - a_inner)) < 1e-10:
                a_inner = a_new
                break
            a_inner = a_new

        # Update running average
        a_avg = a_avg + (a_inner - a_avg) / (it + 1)

        # Player B's best response: k* = argmax conv(a_avg, a_avg)
        conv_avg = _fft_convolve(a_avg)
        k_star = int(np.argmax(conv_avg))

        # Update p (fictitious play)
        p_new = p.copy()
        p_new[k_star] += 1.0
        p_new /= p_new.sum()
        p = p_new

        # Track best
        cur_score = 2.0 * n * float(conv_avg.max())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_avg.copy()

        # Also check a_inner
        conv_inner = _fft_convolve(a_inner)
        inner_score = 2.0 * n * float(conv_inner.max())
        if inner_score < best_score - 1e-12:
            best_score = inner_score
            best_a = a_inner.copy()

    s_final = best_a.sum()
    if s_final > 0:
        best_a = best_a / s_final
    c_final = _fft_convolve(best_a)
    final_score = 2.0 * n * float(c_final.max()) / (s_final * s_final)
    return best_a, final_score


def _fourier_perturb(a, seed=0, strength=0.1):
    """Add perturbation in Fourier domain."""
    n = len(a)
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    rng = np.random.default_rng(seed)
    # Perturb the phase of A (keeping magnitude roughly fixed)
    # Add small random phase noise
    phase_noise = rng.normal(0, strength * 0.1, size=A.size)
    A_pert = A * np.exp(1j * phase_noise)
    a_new = np.fft.irfft(A_pert, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _adam_smooth_max(a0, deadline, max_iter=50000, beta_factor=0.05):
    """Adam optimizer in logit-space for smooth max of autocorrelation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = 0.03
    lr_cur = base_lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_grad(c_curr, beta)
        grad_a = _grad_conv(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr_cur = max(0.001, lr_cur * 0.93)
            if no_imp > max(4000, 6 * n):
                rng_r = np.random.default_rng(1000 + it)
                z = best_z + 0.3 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.75)

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    profiles = []

    # Linear profiles
    for eta in [1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Quadratic profiles
    for c1 in [0.5, 1.0, 2.0]:
        a = 1.0 + c1 * (2 * t - 1) ** 2
        profiles.append(a / a.sum())

    # Cosine profiles
    for alpha in [0.3, 0.7, 1.0, 1.5]:
        a = 1.0 + alpha * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Multi-hump
    for freq in [3, 5]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential decay
    for rate in [1.0, 2.0, 3.0]:
        a = np.exp(-rate * np.abs(t - 0.5))
        profiles.append(a / a.sum())

    return profiles


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: coordinate descent, L-BFGS, fictitious play,
    Adam in log-space, and Fourier perturbations. Time-managed for 1800s budget.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p1024 = _interpolate_profile(parent, 1024)
        p256 = _interpolate_profile(parent, 256)
    else:
        p512 = _make_profiles(512)[1]  # linear eta=1.5
        p1024 = _make_profiles(1024)[1]
        p256 = _make_profiles(256)[1]

    # === Phase 1: Coordinate Descent on parent ===
    print("=== Phase 1: Coordinate Descent n=512 ===")
    d1 = min(deadline - 1200, t_start + 400)
    a_cd, s_cd = _coordinate_descent(p512, d1, max_sweeps=300, seed=123)
    if s_cd < best_score:
        best_score = s_cd
        best_seq = a_cd.copy()
    print(f"  CD n=512: {s_cd:.10f} (best={best_score:.10f})")

    # === Phase 2: L-BFGS on parent ===
    if time.monotonic() < deadline - 1000:
        print("=== Phase 2: L-BFGS n=512 ===")
        d2 = min(deadline - 1000, time.monotonic() + 300)
        a_lb, s_lb = _lbfgs_smooth_max(best_seq, d2, max_iter=100)
        if s_lb < best_score:
            best_score = s_lb
            best_seq = a_lb.copy()
        print(f"  L-BFGS n=512: {s_lb:.10f} (best={best_score:.10f})")

    # === Phase 3: Coordinate Descent n=1024 ===
    if time.monotonic() < deadline - 800:
        print("=== Phase 3: Coordinate Descent n=1024 ===")
        best_1024 = _interpolate_profile(best_seq, 1024)
        d3 = min(deadline - 800, time.monotonic() + 400)
        a_cd2, s_cd2 = _coordinate_descent(best_1024, d3, max_sweeps=200, seed=456)
        if s_cd2 < best_score:
            best_score = s_cd2
            best_seq = a_cd2.copy()
        print(f"  CD n=1024: {s_cd2:.10f} (best={best_score:.10f})")

    # === Phase 4: Fictitious Play ===
    if time.monotonic() < deadline - 500:
        print("=== Phase 4: Fictitious Play n=256 ===")
        d4 = min(deadline - 500, time.monotonic() + 250)
        a_fp, s_fp = _fictitious_play(p256, d4, max_iter=3000, n_inner=30)
        if s_fp < best_score:
            best_score = s_fp
            best_seq = a_fp.copy()
        print(f"  FP n=256: {s_fp:.10f} (best={best_score:.10f})")

    # === Phase 5: Adam on best ===
    if time.monotonic() < deadline - 300:
        print("=== Phase 5: Adam n=512 ===")
        best_512 = _interpolate_profile(best_seq, 512)
        d5 = min(deadline - 300, time.monotonic() + 250)
        a_ad, s_ad = _adam_smooth_max(best_512, d5, max_iter=30000, beta_factor=0.03)
        if s_ad < best_score:
            best_score = s_ad
            best_seq = a_ad.copy()
        print(f"  Adam n=512: {s_ad:.10f} (best={best_score:.10f})")

    # === Phase 6: L-BFGS on best at n=1024 ===
    if time.monotonic() < deadline - 200:
        print("=== Phase 6: L-BFGS n=1024 ===")
        best_1024 = _interpolate_profile(best_seq, 1024)
        d6 = min(deadline - 200, time.monotonic() + 200)
        a_lb2, s_lb2 = _lbfgs_smooth_max(best_1024, d6, max_iter=80)
        if s_lb2 < best_score:
            best_score = s_lb2
            best_seq = a_lb2.copy()
        print(f"  L-BFGS n=1024: {s_lb2:.10f} (best={best_score:.10f})")

    # === Phase 7: Fourier perturbations + re-optimize ===
    if time.monotonic() < deadline - 100:
        print("=== Phase 7: Fourier perturbations ===")
        for trial in range(3):
            if time.monotonic() > deadline - 50:
                break
            a_pert = _fourier_perturb(best_seq, seed=100 + trial * 7,
                                       strength=0.05 + 0.05 * trial)
            # Quick L-BFGS
            d7 = min(deadline - 50 - trial * 20, time.monotonic() + 80)
            a_opt, s_opt = _lbfgs_smooth_max(a_pert, d7, max_iter=50)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
            print(f"  Perturb {trial}: {s_opt:.10f} (best={best_score:.10f})")

    # === Phase 8: Final coordinate descent polish ===
    if time.monotonic() < deadline - 30:
        print("=== Phase 8: Final CD polish ===")
        d8 = deadline - 10
        a_final, s_final = _coordinate_descent(best_seq, d8, max_sweeps=200, seed=789)
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
        print(f"  Final CD: {s_final:.10f} (best={best_score:.10f})")

    # === Phase 9: Ultra-fine Adam ===
    if time.monotonic() < deadline - 5:
        d9 = deadline - 3
        a_uf, s_uf = _adam_smooth_max(best_seq, d9, max_iter=10000,
                                       beta_factor=0.01, )
        if s_uf < best_score:
            best_score = s_uf
            best_seq = a_uf.copy()
        print(f"  Ultra-fine Adam: {s_uf:.10f} (best={best_score:.10f})")

    # Final validation
    if best_seq is None:
        best_seq = p256
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = p256

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
