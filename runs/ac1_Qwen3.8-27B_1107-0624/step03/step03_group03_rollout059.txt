"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS in logit space, LP-based spreading with full constraints,
and subgradient polishing. Explores multiple lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def _line_search(a, g, n_evals=25):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    best_score = _score_norm(a)
    if not np.isfinite(best_score):
        return 0.5
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c * (g - a), 0))
    fd = _score_norm(np.maximum(a + d * (g - a), 0))
    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c * (g - a), 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d * (g - a), 0))
    best_t = (lo + hi) / 2.0
    return best_t


def _lp_refine(a, deadline, max_iter=15000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)

    no_imp = 0
    inflate = 1.0001
    blend = 0.3
    lp = _lp_spread(a, float(_fft_convolve(a).max()) * inflate)
    prev_lp_sum = float(lp.sum()) if lp is not None else 0.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _lp_spread(a, M_lp)

        if g is None:
            inflate = min(1.01, inflate * 1.5)
            no_imp += 10
            if no_imp > 500:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-10:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(1.01, inflate * 1.3)
            if no_imp > 500:
                break
            continue

        g = g / sum_g
        t = _line_search(a, g, n_evals=22)
        if t < 0.003:
            t = blend * 0.5

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        sc = _score_norm(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0

        else:

            a = a_new
            if no_imp % 80 == 0:
                blend = max(0.05, blend * 0.9)
                inflate = min(1.005, inflate * 0.97)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                inflate = 1.0001
                blend = 0.3
                no_imp = 0

    return best_a


def _lbfgs_optimize(a, deadline, T_values=(0.05, 0.03, 0.01, 0.005, 0.002)):
    """Optimize using L-BFGS in logit space with decreasing temperature."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for T in T_values:
        if time.monotonic() > deadline:
            break

        z0 = np.log(a)
        z0 -= float(np.mean(z0))

        def obj_grad(z):
            z_shifted = z - z.max()
            e = np.exp(z_shifted)
            a_curr = e / e.sum()

            c = _fft_convolve(a_curr)
            m_val = float(c.max())
            x = (c - m_val) / T
            x = np.clip(x, -50, 50)
            ex = np.exp(x)
            ex_sum = float(ex.sum())
            w = ex / ex_sum
            val = 2.0 * n * (m_val + T * np.log(ex_sum))

            N_fft = _next_pow2(2 * n - 1)
            fw = np.fft.rfft(w, N_fft)
            fa_rev = np.fft.rfft(a_curr[::-1], N_fft)
            h = np.fft.irfft(fw * fa_rev, N_fft)[n - 1:2 * n - 1]
            ga = 2.0 * h

            dot = float(np.dot(a_curr, ga))
            gz = a_curr * ga - a_curr * dot
            gz = gz - gz.mean()

            if not np.all(np.isfinite(gz)):
                gz = np.zeros(n)

            return val, gz

        try:
            res = minimize(obj_grad, z0, method='L-BFGS-B',
                           jac=True, options={'maxiter': 5000,
                                              'ftol': 1e-12, 'gtol': 1e-10,
                                              'maxls': 40})
            if res.success or res.fun < 1e10:
                z_opt = res.x
                z_opt = z_opt - z_opt.max()
                e = np.exp(np.clip(z_opt, -40, 40))
                a_new = e / e.sum()
                a_new = np.maximum(a_new, 1e-12)
                a_new = a_new / a_new.sum()
                sc = _score_norm(a_new)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
        except Exception:
            pass

    return best_a


def _lse_adam_refine(a, deadline, max_iter=80000, T_start=0.04, T_end=1e-5, lr=0.08):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    no_imp = 0
    base_lr = lr
    beta = 0.9

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        ex_sum = float(ex.sum())
        if ex_sum <= 0 or not np.isfinite(ex_sum):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / ex_sum

        N_fft = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N_fft)
        fa_rev = np.fft.rfft(a[::-1], N_fft)
        h = np.fft.irfft(fw * fa_rev, N_fft)[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        # Nesterov-style momentum
        state['t'] += 1
        b1, b2 = 0.9, 0.999
        state['m'] = b1 * state['m'] + (1 - b1) * grad_tan
        state['v'] = b2 * state['v'] + (1 - b2) * (grad_tan * grad_tan)
        mh = state['m'] / (1 - b1 ** state['t'])
        vh = state['v'] / (1 - b2 ** state['t'])
        step = lr * mh / (np.sqrt(vh) + 1e-8)

        a_new = a - step
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()

        sc = _score_norm(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            a = a_new
            if no_imp % 3000 == 0:
                lr = max(0.01, lr * 0.9)
            if no_imp > max(10000, 12 * n):
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.06 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0

        T = max(T * T_ratio, T_end)

    return best_a


def _subgradient_polish(a, deadline, max_iter=50000, top_k=8):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    lr = 0.008
    beta = 0.85
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        k = min(top_k, 2 * n - 1)
        sorted_idx = np.argsort(c)[::-1]
        top_lags = sorted_idx[:k]
        weights = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_lags):
            weights[idx] = 1.0 / (i + 1)
        weights /= weights.sum()

        N_fft = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(weights, N_fft)
        fa_rev = np.fft.rfft(a[::-1], N_fft)
        h = np.fft.irfft(fw * fa_rev, N_fft)[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = beta * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a

        sc = _score_norm(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            a = a_new
            if no_imp % 400 == 0:
                lr = max(0.0005, lr * 0.9)
            if no_imp > max(5000, 8 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

        return a

    return best_a

        sc2048_init = _score(a2048)
        print(f"  n=2048 init: {sc2048_init:.10f}")

        # LP on 2048
        budget_left = deadline - time.monotonic() - 60.0
        if budget_left > 80:
            a_lp2048 = _lp_refine(a2048, time.monotonic() + min(150, budget_left - 20), max_iter=5000)
            sc_lp2048 = _score(a_lp2048)
            print(f"  LP (n=2048): {sc_lp2048:.10f}")
            if sc_lp2048 < best_score:
                best_score = sc_lp2048
                best_seq = a_lp2048.copy()

        # L-BFGS on 2048 if time allows
        budget_left = deadline - time.monotonic() - 40.0
        if budget_left > 60 and best_seq.size == n2:
            a_lb2048 = _lbfgs_optimize(best_seq, time.monotonic() + min(80, budget_left - 20),
                                       T_values=(0.02, 0.008))
            sc_lb2048 = _score(a_lb2048)
            print(f"  L-BFGS (n=2048): {sc_lb2048:.10f}")
            if sc_lb2048 < best_score:
                best_score = sc_lb2048
                best_seq = a_lb2048.copy()

    # Phase 6: Subgradient polish on best
    print("\n=== Phase 6: Subgradient polish ===")
    sp_budget = deadline - time.monotonic() - 30.0
    if sp_budget > 20:
        a_sp = _subgradient_polish(best_seq, time.monotonic() + sp_budget, max_iter=80000, top_k=10)
        sc_sp = _score(a_sp)
        print(f"  Subgradient: {sc_sp:.10f}")
        if sc_sp < best_score:
            best_score = sc_sp
            best_seq = a_sp.copy()

    # Phase 7: Final LP micro-polish
    final_budget = deadline - time.monotonic() - 15.0
    if final_budget > 15:
        a_final = _lp_refine(best_seq, time.monotonic() + final_budget, max_iter=5000)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final LP: {sc_final:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(1024) / 1024.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(1024) / 1024.0
        s = 1.0
    final_seq = final_seq / s

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
