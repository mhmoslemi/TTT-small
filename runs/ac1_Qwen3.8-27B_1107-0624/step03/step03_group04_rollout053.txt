    #   The output has S[i] at position i for i = 0, ..., 2n-2 and S[i] at position N+i for i = -(n-1), ..., -1

    # So S[0:n] = ifft(fft(w_padded)*conj(fft(a_padded)))[0:n].real
    
    S_full = np.fft.ifft(fw3 * np.conj(fa3)).real
    S = S_full[:n]  # S[i] for i = 0, ..., n-1
    
    # Now: grad_i = 2*S[i] - w[2*i]*a[i]
    w_shifted = w[0:2*n:2]  # w[2i] for i = 0, ..., n-1 (since 2(n-1) = 2n-2 < 2n-1)
    grad = 2.0 * S - w_shifted * a
    
    # For the simplex constraint, the gradient in the "unconstrained" sense is grad.
    # For the log-space parameterization, we need the projected gradient.
    
    return val, grad, float(c.max())

def _fista_logspace(a, deadline, max_iter=50000, lr=0.05, beta_init=None):
    """
    Fista (Nesterov accelerated proximal gradient) in log-space.
    Objective: J(a) = 2n * max_k c_k / (sum(a))^2, minimized over a >= 0.
    Since J is scale-invariant, we fix sum(a) = 1 (project onto simplex).
    Then minimize max_k c_k (which equals J/2n).
    
    We parameterize a = exp(z) / sum(exp(z)) and optimize over z (unconstrained).
    The proximal step is the exponential map (softmax).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    # Initialize z = log(a)
    z = np.log(a)
    z -= float(np.mean(z))  # Center for numerical stability
    z_old = z.copy()
    
    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = beta_init if beta_init is not None else max(1e-6, 0.1 * c_max)
    
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()
    
    no_imp = 0
    t_fista = 1.0  # Fista momentum counter
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        p = a  # Current point (Fista uses p for gradient evaluation)
        
        # Compute gradient
        c_p = _fft_convolve(p)
        _, w = _smooth_max_and_weights(c_p, beta)
        
        # Compute S[i] = sum_j w[j+i] * a[j]
        N3 = _next_pow2(3 * n)
        fw3 = np.fft.rfft(w, N3)
        fp3 = np.fft.rfft(p, N3)
        S_full = np.fft.ifft(fw3 * np.conj(fp3)).real
        S = S_full[:n]
        
        w_shifted = w[0:2*n:2]
        grad_p = 2.0 * S - w_shifted * p
        
        # Project gradient onto the tangent space of the simplex (remove mean)
        # For the log-space parameterization, the gradient w.r.t. z is:
        # dJ/dz_i = a_i * (grad_p_i - sum_j a_j * grad_p_j)
        dot = float(np.dot(a, grad_p))  # Use a (current) for the projection
        gz = a * (grad_p - dot)
        gz -= float(np.mean(gz))  # Extra centering for stability
        
        if not np.all(np.isfinite(gz)):
            # Reset momentum
            z = best_z.copy()
            z_old = z.copy()
            t_fista = 1.0
            continue
        
        # Fista update
        z_old_current = z_old.copy()
        z_new = z - lr * gz
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        
        # Momentum step
        z = z_new + ((t_fista - 1.0) / t_new) * (z_new - z_old_current)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        
        # Exponential map (softmax)
        e = np.exp(z)
        a = e / e.sum()
        
        # Evaluate
        c_new = _fft_convolve(a)
        score = float(2.0 * n * c_new.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive beta
        if it % 200 == 0 and it > 100:
            top5 = np.sort(c_new)[::-1][:min(5, len(c_new))]
            if len(top5) > 1:
                gap = top5[0] - top5[-1]
                if gap < beta * 0.5:
                    beta = max(1e-10, beta * 0.7)
                elif gap > beta * 5:
                    beta = min(0.5 * c_new.max(), beta * 1.3)
        
        # Restart if stuck
        if no_imp > max(3000, 5 * n):
            noise = 0.2 * np.random.standard_normal(n)
            z = best_z + noise
            z = np.clip(z, -50.0, 50.0)
            z -= np.mean(z)
            e = np.exp(z)
            a = e / e.sum()
            z_old = z.copy()
            t_fista = 1.0
            no_imp = 0
            beta = max(1e-6, 0.1 * c_new.max())
        
        z_old = z.copy()
        t_fista = t_new
    
    return best_a

def _coordinate_descent(a, deadline, n_passes=10, rng=None):
    """
    Coordinate descent: for each coordinate i, find the optimal value of a_i
    given all other coordinates fixed, by binary searching on the target score.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    for pass_idx in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            a_i = a[i]
            # Compute base_k = c_k with a_i terms removed
            # c_k = sum_j a_j * a_{k-j}
            # Terms with a_i: j=i gives a_i * a_{k-i}, j=k-i gives a_{k-i} * a_i
            # base_k = c_k - 2*a_i*a_{k-i} (for k != 2i) or c_k - a_i^2 (for k = 2i)
            
            # Actually, let's compute c without a_i:
            a_no_i = np.delete(a, i)
            # But this is O(n) per coordinate, and we do this O(n) times = O(n^2) total
            # For n=1024, that's ~1M operations per pass, which is fine.
            
            # Faster: c_k = sum_{j != i} a_j * a_{k-j} + (terms with i)
            # Let c_full = conv(a, a)
            # c_k - a_i contribution:
            #   if 0 <= k-i < n: subtract a_i * a_{k-i}
            #   if 0 <= k-i < n and k-i != i: subtract another a_i * a_{k-i}
            # So: base_k = c_k - a_i * a_{k-i} - (a_i * a_{k-i} if k-i != i else a_i * a_i)
            #     = c_k - 2*a_i*a_{k-i} if k-i != i
            #     = c_k - a_i^2 if k-i = i (k = 2i)
            
            c_full = _fft_convolve(a)
            
            # Build base
            base = c_full.copy()
            # Subtract a_i contributions
            for k in range(2 * n - 1):
                j = k - i
                if 0 <= j < n:
                    base[k] -= a[i] * a[j]
                    if j != i:
                        base[k] -= a[i] * a[j]  # Second term (j=k-i as the "other" index)
            
            # Hmm, this is O(n) per coordinate, O(n^2) total per pass. For n=1024, that's ~1M ops per pass.
            # With 10 passes, that's 10M ops. Should be fast enough.
            
            # Now find optimal x = a_i to minimize max_k f_k(x) subject to x >= 0
            # where f_k(x) = (base_k + delta_k(x)) / (S + x)^2 * 2n
            # and S = sum(a) - a_i
            # delta_k(x) = 2*x*a_{k-i} if k-i != i, x^2 if k-i = i, 0 otherwise
            
            S = a.sum() - a[i]
            
            # Binary search on target T = max_k (numerator_k)
            # We want: for all k, base_k + delta_k(x) <= T * (S+x)^2 / (2n)
            # Let ratio = T / (2n)
            # base_k + delta_k(x) <= ratio * (S+x)^2
            
            # For linear terms (k where k-i is valid and k-i != i):
            #   base_k + 2*x * a_{k-i} <= ratio * (S+x)^2
            #   ratio*x^2 + (2*ratio*S - 2*a_{k-i})*x + (ratio*S^2 - base_k) <= 0
            # For quadratic term (k = 2i):
            #   base_{2i} + x^2 <= ratio * (S+x)^2
            #   (ratio-1)*x^2 + 2*ratio*S*x + (ratio*S^2 - base_{2i}) <= 0  [if ratio < 1]
            #   or if ratio >= 1: -x^2 + 2*ratio*S*x + (ratio*S^2 - base_{2i}) <= 0
            #     which is always true for large x... actually not, let me be more careful.
            
            # Actually, the score is 2n * max_k c_k / (sum(a))^2.
            # If we set a_i = x and keep others fixed, sum(a) = S + x.
            # c_k(x) = base_k + delta_k(x)
            # score(x) = 2n * max_k c_k(x) / (S+x)^2
            
            # We want to minimize score(x) over x >= 0.
            # Equivalent to: minimize T = max_k [c_k(x) / (S+x)^2] over x >= 0.
            
            # For a fixed x, T(x) = max_k [c_k(x)] / (S+x)^2
            # = max_k [base_k + delta_k(x)] / (S+x)^2
            
            # Binary search on T: find smallest T such that there exists x >= 0 with
            #   base_k + delta_k(x) <= T*(S+x)^2 for all k.
            
            # For each k, this gives a constraint on x (an interval).
            # The feasible x is the intersection of all intervals.
            
            # Let's implement this:
            
            def _check_feasible(x, ratio):
                """Check if x is feasible for given ratio (= T)."""
                # For all k: base_k + delta_k(x) <= ratio * (S+x)^2
                # delta_k(x) = 2*x*a[k-i] if k-i valid and k-i != i
                #             = x^2 if k == 2i
                #             = 0 otherwise
                
                rhs = ratio * (S + x) ** 2
                # Check all k where a_{k-i} is valid
                lo = max(0, i - (n - 1))  # Not needed, k ranges from 0 to 2n-2
                # k ranges from 0 to 2n-2, and k-i ranges from -i to 2n-2-i
                # Valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
                for k in range(i, i + n):  # k-i ranges from 0 to n-1
                    j = k - i
                    if j == i:  # k = 2i
                        lhs = base[k] + x * x
                    else:
                        lhs = base[k] + 2.0 * x * a[j]
                    if lhs > rhs + 1e-15:
                        return False
                # Also check k = 2i if it's in range
                if 2 * i < 2 * n - 1:
                    k = 2 * i
                    lhs = base[k] + x * x
                    if lhs > rhs + 1e-15:
                        return False
                return True
            
            # This is O(n) per check. Binary search takes ~30 iterations.
            # So each coordinate takes O(30*n) = O(n). Total per pass: O(n^2).
            # For n=1024: 1M ops per pass. 10 passes: 10M. Fine.
            
            # Actually, the above _check_feasible is O(n) which makes the total O(n^2 * 30) per pass.
            # For n=1024: 1024 * 30 * 1024 = 31M per pass. 10 passes = 310M. Might be slow in Python.
            
            # Let me vectorize:
            def _check_feasible_fast(x, ratio):
                """Vectorized feasibility check."""
                k_range = np.arange(i, i + n)
                j_range = k_range - i  # 0 to n-1
                rhs = ratio * (S + x) ** 2
                # delta
                delta = np.where(j_range == i, x * x, 2.0 * x * a[j_range])
                lhs = base[k_range] + delta
                # Also check k = 2i if not already in k_range
                if 2 * i < i or 2 * i >= i + n:
                    if 2 * i < 2 * n - 1:
                        lhs_extra = base[2 * i] + x * x
                        if lhs_extra > rhs + 1e-15:
                            return False
                return np.all(lhs <= rhs + 1e-15)
            
            # Binary search on ratio (= T, the target max c_k / (S+x)^2)
            # Lower bound: 0, Upper bound: current score / (2n)
            T_lo = 0.0

            # Actually, T is the value of max_k c_k(x) / (S+x)^2 that we're trying to achieve.
            # The minimum achievable T tells us the best we can do for this coordinate.
            
            # But we also need to find the x that achieves it.
            # At the optimal T*, the feasible set is a single point (or interval).
            # The optimal x is in that feasible set.
            
            # Strategy: binary search on T. For each T, find the feasible interval for x.
            # The feasible interval is [x_min, x_max] where:
            #   For each k: base_k + delta_k(x) <= T*(S+x)^2
            #   This gives a constraint on x.
            
            # For linear constraint k (j = k-i != i):
            #   base_k + 2*x*a_j <= T*(S+x)^2
            #   T*x^2 + (2T*S - 2*a_j)*x + (T*S^2 - base_k) <= 0
            #   Quadratic in x: Ax^2 + Bx + C <= 0
            #   where A = T, B = 2T*S - 2*a_j, C = T*S^2 - base_k
            #   A > 0 (since T > 0), so the feasible set is between the two roots.
            
            # For quadratic constraint (k = 2i):
            #   base_{2i} + x^2 <= T*(S+x)^2
            #   T*x^2 + 2T*S*x + T*S^2 - base_{2i} - x^2 <= 0
            #   (T-1)*x^2 + 2T*S*x + (T*S^2 - base_{2i}) <= 0
            #   If T < 1: A = T-1 < 0, so feasible set is OUTSIDE the roots.
            #     But we also need x >= 0, so x <= root_min or x >= root_max.
            #     Since x >= 0 and the quadratic opens downward, for large x the quadratic is negative.
            #     So feasible: x in [0, root_min] union [root_max, inf)
            #   If T >= 1: A = T-1 >= 0, feasible set is between roots.
            
            # This is getting complex. Let me use a simpler approach:
            # Just do ternary search on x in [0, x_max] where x_max is some upper bound.
            # The function score(x) = 2n * max_k c_k(x) / (S+x)^2 is NOT unimodal in general,
            # but it should be "mostly" unimodal (decreasing then increasing, or vice versa).
            
            # Actually, let me think about behavior:
            # As x -> 0: c_k -> base_k, sum -> S, score -> 2n * max(base_k) / S^2
            # As x -> inf: c_k ~ 2*x*a_{k-i} (dominated by linear term), sum ~ x
            #   score ~ 2n * 2*x*max(a_j) / x^2 = 4n*max(a_j) / x -> 0
            # So score(x) -> 0 as x -> inf! That means we should make a_i as large as possible?!
            
            # Wait, that can't be right. Let me reconsider.
            # If a_i is very large, then c_k for k near i (where a_{k-i} is large) will be ~ 2*x*a_{k-i} ~ 2*x^2 (when k-i = i, i.e., k=2i, c_{2i} ~ x^2).
            # And sum ~ x. So score ~ 2n * x^2 / x^2 = 2n. That's terrible!
            
            # I made an error. Let me redo:
            # c_{2i}(x) = base_{2i} + x^2
            # For large x: c_{2i} ~ x^2, sum ~ x, so c_{2i}/sum^2 ~ x^2/x^2 = 1
            # So score ~ 2n * 1 = 2n. Terrible.
            
            # For the linear terms: c_k ~ 2*x*a_j, sum ~ x, so c_k/sum^2 ~ 2*a_j/x -> 0
            # But c_{2i} dominates for large x.
            
            # So score(x) starts at 2n*max(base_k)/S^2 (at x=0), decreases initially (since the linear terms grow slower than x^2 in the denominator), then increases (due to the x^2 term at k=2i).
            
            # So score(x) IS unimodal (decreases then increases)! We can use ternary search.
            
            # Let's find a good x_max: at x = S (i.e., a_i = S, which doubles the sum), the score is bounded.
            x_max = S  # Reasonable upper bound
            
            # Ternary search
            def _eval_score(x):
                """Evaluate score with a_i = x."""
                if x < 0:
                    x = 0
                new_sum = S + x
                if new_sum < 1e-15:
                    return np.inf
                # c_k with a_i = x
                # c_k = base_k + 2*x*a_{k-i} (k != 2i) or base_k + x^2 (k = 2i)
                k_range = np.arange(i, min(i + n, 2 * n - 1))
                j_range = k_range - i
                delta = np.where(j_range == i, x * x, 2.0 * x * a[j_range])
                c_new = base[k_range] + delta
                # Also check k = 2i if not in k_range
                if 2 * i < 2 * n - 1 and not (i <= 2 * i < i + n):
                    c_2i = base[2 * i] + x * x
                    c_max = max(c_new.max(), c_2i)
                else:
                    c_max = c_new.max()
                # Also need to check k values NOT in [i, i+n):
                # For k < i: k-i < 0, so no contribution from a_i. c_k = base_k.
                # For k >= i+n: k-i >= n, so no contribution from a_i. c_k = base_k.
                c_other = base[np.concatenate([np.arange(0, i), np.arange(i+n, 2*n-1)])] if i + n < 2*n-1 else base[:i]
                if len(c_other) > 0:
                    c_max = max(c_max, c_other.max())
                else:
                    # Only k in [i, i+n) matter
                    pass
                
                return 2.0 * n * c_max / (new_sum * new_sum)
            
            # Ternary search for minimum of _eval_score(x) on [0, x_max]
            a_left = 0.0
            a_right = x_max
            for _ in range(60):  # ~1e-18 precision
                m1 = a_left + (a_right - a_left) / 3.0
                m2 = a_right - (a_right - a_left) / 3.0
                f1 = _eval_score(m1)
                f2 = _eval_score(m2)
                if f1 < f2:
                    a_right = m2
                else:
                    a_left = m1
            
            x_opt = (a_left + a_right) / 2.0
            score_opt = _eval_score(x_opt)
            
            # Update
            if score_opt < best_score - 1e-12:
                a[i] = max(x_opt, 1e-15)
                best_a = a.copy()
                best_score = _score(a)  # Recompute to be safe
                improved =
