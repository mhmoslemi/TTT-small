Trust-region QP + LP spreading hybrid for minimizing 2n*max(conv(a,a))/sum(a)^2.
Novel approach: sequential convexification via trust-region subproblems solved with CVXPY,
combined with LP-based spreading and spectral-aware initialization.
"""

import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _conv_gradient_single(a, k):

    #   If j=i: term is a[k-i] (valid if 0<=k-i<=n-1)

    # The term a[j]*a[k-j] involves a[i] when j=i (contributing a[k-i]) or when k-j=i i.e. j=k-i (contributing a[k-(k-i)]=a[i])
    # So grad[i] = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=k-i<n) [wait, that's not right either]
    
    # Let me just be explicit:
    # d/d a[i] {a[j] * a[k-j]} is non-zero when j=i or k-j=i
    # When j=i: d/d a[i] {a[i] * a[k-i]} = a[k-i] (if k-i != i) or 2*a[i] (if k-i == i)
    # When k-j=i, i.e., j=k-i: d/d a[i] {a[k-i] * a[i]} = a[k-i] (if k-i != i) or 2*a[i] (if k-i == i)
    # But we only count each term once in the sum!
    # The sum is over j, so the term for j=i is a[i]*a[k-i], and the term for j=k-i is a[k-i]*a[i].

    # When i != k-i, they're different j values, so we get two contributions: a[k-i] + a[i]...

    # When j=k-i (and k-i != i), the term is a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i].
    # So total: 2*a[k-i] when i != k-i and both j=i and j=k-i are valid.
    # When i = k-i (k=2i): only one term, derivative is 2*a[i].
    
    # Hmm wait, that's not right. Let me redo.
    # c[k] = sum_j a[j] * a[k-j]
    # Take d/d a[i]:
    # = sum_j [d a[j]/d a[i] * a[k-j] + a[j] * d a[k-j]/d a[i]]
    # = sum_j [δ(j,i) * a[k-j] + a[j] * δ(k-j, i)]
    # = a[k-i] + a[k-i]  ... wait
    # = a[k-i] (from j=i term, if k-i is valid) + a[j] where k-j=i, i.e., j=k-i (if k-i is valid)
    # = a[k-i] + a[k-i] = 2*a[k-i] (if 0 <= k-i < n)
    # But when k-i = i (k=2i): we're counting the same term twice!
    # Actually no: when k=2i, the j=i term gives a[k-i] = a[i], and the j=k-i=i term gives a[j] = a[i].
    # But they're the same j! So we should only count it once.
    # The issue is that δ(j,i) * a[k-j] and a[j] * δ(k-j,i) both fire at j=i when k=2i.
    # So we get a[i] + a[i] = 2*a[i], which is correct (derivative of a[i]^2 is 2*a[i]).
    # When k != 2i: j=i fires (giving a[k-i]) and j=k-i fires (giving a[k-i]).

    # Wait, that gives 2*a[k-i] in both cases? Let me check with a simple example.

    # k=1: c[1] = a[0]*a[1] + a[1]*a[0] = 2*a0*a1. grad = [2*a1, 2*a0, 0].
    #   Using formula: for i=0: 2*a[1-0] = 2*a1 ✓. For i=1: 2*a[0] ✓. For i=2: k-i = -1, invalid, 0 ✓.
    # k=2: c[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0] = 2*a0*a2 + a1^2. grad = [2*a2, 2*a1, 2*a0].
    #   Using formula: for i=0: 2*a[2] ✓. For i=1: 2*a[1] ✓. For i=2: 2*a[0] ✓.
    # k=3: c[3] = a[1]*a[2] + a[2]*a[1] = 2*a1*a2. grad = [0, 2*a2, 2*a1].
    #   Using formula: for i=1: 2*a[2] ✓. For i=2: 2*a[1] ✓.
    
    # So the formula is: grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    # Wait, that can't be right for k=1, i=0: 2*a[1-0] = 2*a1. ✓
    # For k=1, i=1: 2*a[0]. ✓
    # For k=2, i=1: 2*a[1]. ✓ (this is the a1^2 term)
    
    # Great! So the gradient is simply: grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    # This is 2 * (a shifted and reversed around position k/2).
    
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad

def _trust_region_step(a, c, peak_indices, delta):
    """Solve trust-region subproblem using CVXPY.
    Minimize t subject to:
        c[k] + grad_k^T * 2 * d <= t for k in peak_indices
        sum(d) = 0
        a + d >= 0
        ||d||_2 <= delta
    """
    n = len(a)
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        sum(d) == 0,
        a + d >= 0,
        cp.norm(d, 2) <= delta
    ]
    
    for k in peak_indices:
        grad = _conv_gradient_single(a, k)
        constraints.append(c[k] + grad @ d <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS)
        if d.value is not None:
            return d.value
    except:
        pass
    
    try:
        prob.solve(solver=cp.CLARABEL)
        if d.value is not None:
            return d.value
    except:
        pass
    
    return None

def _tr_qp_optimize(a, deadline, max_iter=500, init_delta=0.1, top_m=5):
    """Trust-region QP optimization."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    delta = init_delta
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        
        # Find top-m peak indices
        peak_indices = np.argsort(c)[::-1][:top_m]
        
        # Solve trust-region subproblem
        d = _trust_region_step(a, c, peak_indices, delta)
        
        if d is None:
            break
        
        d = np.clip(d, -a, 1 - a)  # Ensure a+d stays in [0, 1]
        d = d - d.sum() / n  # Re-center to maintain sum
        
        a_new = a + d
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            # Increase delta slightly
            delta = min(0.5, delta * 1.05)
        else:
            no_imp += 1
            delta *= 0.8
            if no_imp > 100 or delta < 1e-4:
                break
            a = best_a.copy()
    
    return best_a

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    # Build constraint matrix: conv(a, g) <= M
    # conv(a, g)[k] = sum_i a[i] * g[k-i] (with bounds)
    # This is linear in g.
    # For each k, the constraint is: sum_i a[i] * g[k-i] <= M
    # The coefficient of g[j] in constraint k is: a[k-j] (if 0<=k-j<n)
    # So the matrix A has A[k, j] = a[k-j] if 0<=k-j<n, else 0
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in conv(a,g)[k] with coefficient a[k-j] for k=j, ..., j+n-1
        ks = range(j, min(j + n, 2 * n - 1))
        for k in ks:
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _lp_refine(a, deadline, max_iter=10000, inflate=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.05, inflate_cur * 1.08)
            if no_imp > 500:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0003, inflate_cur * 0.95)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.02, inflate_cur * 0.98)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a

def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w

def _adam_smooth_max_optimize(a, deadline, max_iter=20000, lr=0.05):
    """Adam optimizer on logit parameterization with smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.03 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(3000, 5 * n):
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.6 * best_z + 0.4 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.03 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a

    a = np.maximum(a, 1e-12)

def _make_spectrum_flat_init(n):
    """Initialize by inverse FFT of a flat spectrum."""
    # We want |A(k)|^2 to be as flat as possible.
    # If we set A(k) = 1 for all k (in some band), then a = IDFT of that.
    # But a must be non-negative.
    # Try: a = |IDFT(ones)| = |rectangular| which is just the delta... not useful.
    # Better: use a sinc-like function.
    # Or: use the "flat-top" window in frequency domain.
    t = np.linspace(0.0, 1.0, n)
    # Raised cosine (Hann window in time domain has flat-ish spectrum)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_bimodal_init(n, sigma=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.2) / sigma) ** 2) + np.exp(-0.5 * ((t - 0.8) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
