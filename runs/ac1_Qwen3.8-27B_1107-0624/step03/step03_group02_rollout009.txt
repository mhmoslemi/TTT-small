    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _nesterov_smooth_max(a, deadline, max_iter=50000, lr=0.025, beta_factor=0.04):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    v = np.zeros(n)
    momentum = 0.85

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        z_look = z + momentum * v
        a_look = np.exp(z_look - np.max(z_look))
        a_look = a_look / a_look.sum()

        c_curr = _fft_convolve(a_look)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a_look, w)

        dot = float(np.dot(a_look, grad_a))
        grad_z = a_look * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            v = np.zeros(n)
            continue

        v = momentum * v - lr_cur * grad_z
        z_new = z + v
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                momentum = max(0.5, momentum * 0.96)
                lr_cur = max(0.001, lr_cur * 0.92)
            if no_imp > max(6000, 10 * n):
                rng_r = np.random.default_rng(2000 + it)
                z = best_z + 0.25 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                v = np.zeros(n)
                momentum = 0.85
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 250 == 0:
            beta = max(1e-13, beta * 0.78)

    return best_a, best_score


def _adam_smooth_max(a, deadline, max_iter=50000, lr=0.025, beta_factor=0.04):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr_cur = max(0.001, lr_cur * 0.93)
            if no_imp > max(5000, 8 * n):
                rng_r = np.random.default_rng(3000 + it)
                z = best_z + 0.3 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 200 == 0:
            beta = max(1e-13, beta * 0.77)

    return best_a, best_score


def _simulated_annealing(a, deadline, max_iter=8000, n_moves=5):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    current_score = best_score

    t_start = time.monotonic()
    total_iter = max_iter
    t0 = float(n * 0.005)
    t_end = 0.001 * t0

    rng = np.random.default_rng(777)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        alpha = t0 * (t_end / t0) ** (it / max(1, total_iter))

        move_type = rng.integers(0, 3)

        if move_type == 0:
            i = rng.integers(0, n)
            delta = rng.normal(0, 0.01)
            a_new = a.copy()
            a_new[i] = max(0, a_new[i] + delta)
        elif move_type == 1:
            i = rng.integers(0, n - 1)
            delta = rng.normal(0, 0.008)
            a_new = a.copy()
            a_new[i] = max(0, a_new[i] + delta)
            a_new[i + 1] = max(0, a_new[i + 1] - delta)
        else:
            idx = rng.integers(0, n, size=min(3, n))
            eps = rng.normal(0, 0.005, size=len(idx))
            a_new = a.copy()
            a_new[idx] = np.maximum(0, a_new[idx] + eps)

        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        delta_score = score_new - current_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / max(alpha, 1e-15)):
            a = a_new
            current_score = score_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

    return best_a, best_score


def _lp_spread_step(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=5000, inflate=1.0005, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread_step(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.005, inflate_cur * 1.03)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = min(1.002, inflate_cur * 0.95)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _subgradient_polish(a, deadline, max_iter=5000, top_k=5):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.15 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        weights = np.array([1.0 / (1.0 + 0.5 * i) for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _make_profile(n, shape, **kwargs):
    if shape == 'linear':
        eta = kwargs.get('eta', 1.5)
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + eta * (t - 0.5)
    elif shape == 'cosine':
        alpha = kwargs.get('alpha', 1.0)
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + alpha * np.cos(np.pi * t)
    elif shape == 'bell':
        width = kwargs.get('width', 1.0)
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    elif shape == 'triangular':
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
    elif shape == 'exp_decay':
        alpha = kwargs.get('alpha', 3.0)
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-alpha * t)
    elif shape == 'cubic':
        a1 = kwargs.get('a1', 1.5)
        a3 = kwargs.get('a3', -1.0)
        t = np.linspace(0.0, 1.0, n)
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a3 * u ** 3
    elif shape == 'two_peak':
        t = np.linspace(0.0, 1.0, n)
        a = np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return np.ones(n_new) / n_new
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _quick_pipeline(a0, deadline, label=""):
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0

    if time.monotonic() < deadline - 30:
        d1 = min(deadline - 30, time.monotonic() + 120)
        a1, s1 = _adam_smooth_max(a0, d1, max_iter=15000, lr=0.025, beta_factor=0.04)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam: {best_sc:.10f}")

    if time.monotonic() < deadline - 20:
        d2 = min(deadline - 20, time.monotonic() + 60)
        a2, s2 = _nesterov_smooth_max(best_a, d2, max_iter=10000, lr=0.02, beta_factor=0.03)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} Nesterov: {best_sc:.10f}")

    if time.monotonic() < deadline - 15:
        d3 = min(deadline - 15, time.monotonic() + 80)
        a3 = _lp_refine(best_a, d3, max_iter=2000, inflate=1.0005, blend=0.25)
        s3 = _score(a3)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} LP: {best_sc:.10f}")

    if time.monotonic() < deadline - 10:
        d4 = min(deadline - 10, time.monotonic() + 60)
        a4, s4 = _subgradient_polish(best_a, d4, max_iter=3000, top_k=5)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} SubGrad: {best_sc:.10f}")

    return best_a, best_sc


def _full_pipeline(a0, deadline, label=""):
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: Adam smooth max (coarse)
    if time.monotonic() < deadline - 180:
        d1 = min(deadline - 180, time.monotonic() + 200)
        a1, s1 = _adam_smooth_max(a0, d1, max_iter=30000, lr=0.03, beta_factor=0.05)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam1: {best_sc:.10f}")

    # Phase 2: Nesterov (medium)
    if time.monotonic() < deadline - 120:
        d2 = min(deadline - 120, time.monotonic() + 100)
        a2, s2 = _nesterov_smooth_max(best_a, d2, max_iter=20000, lr=0.02, beta_factor=0.03)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} Nesterov: {best_sc:.10f}")

    # Phase 3: Simulated annealing
    if time.monotonic() < deadline - 100:
        d3 = min(deadline - 100, time.monotonic() + 60)
        a3, s3 = _simulated_annealing(best_a, d3, max_iter=5000)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} SA: {best_sc:.10f}")

    # Phase 4: LP refinement
    if time.monotonic() < deadline - 60:
        d4 = min(deadline - 60, time.monotonic() + 100)
        a4 = _lp_refine(best_a, d4, max_iter=3000, inflate=1.0003, blend=0.2)
        s4 = _score(a4)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} LP: {best_sc:.10f}")

    # Phase 5: Fine Adam
    if time.monotonic() < deadline - 30:
        d5 = min(deadline - 30, time.monotonic() + 80)
        a5, s5 = _adam_smooth_max(best_a, d5, max_iter=15000, lr=0.01, beta_factor=0.015)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} Adam2: {best_sc:.10f}")

    # Phase 6: Final LP + subgradient
    if time.monotonic() < deadline - 10:
        d6 = min(deadline - 10, time.monotonic() + 40)
        a6 = _lp_refine(best_a, d6, max_iter=1500, inflate=1.0002, blend=0.15)
        s6 = _score(a6)
        if s6 < best_sc:
            best_a, best_sc = a6, s6
        print(f"  {label} LP2: {best_sc:.10f}")

    if time.monotonic() < deadline - 5:
        d7 = min(deadline - 5, time.monotonic() + 30)
        a7, s7 = _subgradient_polish(best_a, d7, max_iter=2000, top_k=5)
        if s7 < best_sc:
            best_a, best_sc = a7, s7
        print(f"  {label} FinalSG: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc

                print(f"Parent: score={sc:.10f}, n={parent.size}")

        p1024 = _make_profile(1024, 'linear', eta=1.5)

        print(f"  Phase B best: {best_score:.10f}")

    # === Phase C: Diverse initializations ===
    if time.monotonic() < deadline - 200:
        print("\n=== Phase C: Diverse inits ===")
        diverse_init = [
            ('linear_1', 'linear', {'eta': 1.0}),
            ('linear_2', 'linear', {'eta': 2.5}),
            ('cos_05', 'cosine', {'alpha': 0.5}),
            ('cos_15', 'cosine', {'alpha': 1.5}),
            ('bell_35', 'bell', {'width': 0.35}),
            ('bell_6', 'bell', {'width': 0.6}),
            ('tri', 'triangular', {}),
            ('exp3', 'exp_decay', {'alpha': 3.0}),
            ('cubic', 'cubic', {'a1': 2.0, 'a3': -2.0}),
            ('two_peak', 'two_peak', {}),
        ]

        # Also try perturbed parent at 512
        p_pert = p512 * (1.0 + 0.1 * rng.normal(size=n))
        p_pert = np.maximum(p_pert, 0.0)
        p_pert = p_pert / p_pert.sum()

        for name, shape, kw in diverse_init:
            if time.monotonic() > deadline - 150:
                break
            per_budget = min(35.0, (deadline - 150 - time.monotonic()) / max(1, 1))
            if per_budget < 15:
                break
            a0 = _make_profile(n, shape, **kw)
            d_ci = min(deadline - 150, time.monotonic() + per_budget)
            a_c, s_c = _quick_pipeline(a0, d_ci, f"C-{name}")
            if s_c < best_score:
                best_score = s_c
                best_seq = a_c.copy()

        # Perturbed parent
        if time.monotonic() < deadline - 100:
            d_cp = min(deadline - 100, time.monotonic() + 40)
            a_cp, s_cp = _quick_pipeline(p_pert, d_cp, "C-pert")
            if s_cp < best_score:
                best_score = s_cp
                best_seq = a_cp.copy()

    # === Phase D: Fine-tune best at its native length ===
    if best_seq is not None and time.monotonic() < deadline - 50:
        print(f"\n=== Phase D: Final focus (best={best_score:.10f}) ===")
        bn = best_seq.size

        # If best is at 1024, also try interpolating to 512 and back
        if bn == 1024:
            best_512 = _interpolate_profile(best_seq, 512)
            if time.monotonic() < deadline - 50:
                d_d1 = min(deadline - 50, time.monotonic() + 50)
                a_d1, s_d1 = _adam_smooth_max(best_512, d_d1, max_iter=10000, lr=0.01, beta_factor=0.01)
                if s_d1 < best_score:
                    best_score = s_d1
                    best_seq = a_d1.copy()
                print(f"  D 512 Adam: {best_score:.10f}")

        # Fine Nesterov
        if time.monotonic() < deadline - 40:
            d_d2 = min(deadline - 40, time.monotonic() + 40)
            a_d2, s_d2 = _nesterov_smooth_max(best_seq, d_d2, max_iter=10000, lr=0.008, beta_factor=0.01)
            if s_d2 < best_score:
                best_score = s_d2
                best_seq = a_d2.copy()
            print(f"  D Nesterov: {best_score:.10f}")

        # Fine LP
        if time.monotonic() < deadline - 25:
            d_d3 = min(deadline - 25, time.monotonic() + 30)
            a_d3 = _lp_refine(best_seq, d_d3, max_iter=2000, inflate=1.0002, blend=0.1)
            s_d3 = _score(a_d3)
            if s_d3 < best_score:
                best_score = s_d3
                best_seq = a_d3.copy()
            print(f"  D LP: {best_score:.10f}")

        # Simulated annealing polish
        if time.monotonic() < deadline - 15:
            d_d4 = min(deadline - 15, time.monotonic() + 20)
            a_d4, s_d4 = _simulated_annealing(best_seq, d_d4, max_iter=3000)
            if s_d4 < best_score:
                best_score = s_d4
                best_seq = a_d4.copy()
            print(f"  D SA: {best_score:.10f}")

        # Final subgradient
        if time.monotonic() < deadline - 5:
            d_d5 = min(deadline - 5, time.monotonic() + 15)
            a_d5, s_d5 = _subgradient_polish(best_seq, d_d5, max_iter=2000, top_k=7)
            if s_d5 < best_score:
                best_score = s_d5
                best_seq = a_d5.copy()
            print(f"  D SubGrad: {best_score:.10f}")

    # === Phase E: Multi-restart from best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Phase E: Multi-restart ===")
        n_restart = best_seq.size
        for r in range(3):
            if time.monotonic() > deadline - 20:
                break
            pert = best_seq * (1.0 + 0.05 * rng.normal(size=n_restart))
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            d_er = min(deadline - 20, time.monotonic() + 25)
            a_r, s_r = _adam_smooth_max(pert, d_er, max_iter=8000, lr=0.015, beta_factor=0.02)
            if s_r < best_score:
                best_score = s_r
                best_seq = a_r.copy()
            print(f"  E restart {r}: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_profile(512, 'linear', eta=1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_profile(512, 'linear', eta=1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
