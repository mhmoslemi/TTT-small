import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a) = 1, so score = 2*n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -500, 500)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _gradient_smooth_max(a, beta):
    """Compute gradient of softmax(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    # Gradient of sum_k w_k * (a*a)[k] = 2 * (w_rev * a) [truncated]
    # (a*a)[k] = sum_j a[j]*a[k-j], so d/dx[a[i]] (a*a)[k] = a[k-i] (if valid)
    # Total gradient: grad[i] = sum_k w_k * a[k-i] = (w * a)[i] in circular sense
    # But we want linear convolution, so: grad[i] = sum_{k} w[k] * a[k-i] where k-i in [0,n-1]
    # This is (w_full * a)[n-1:n-1+n] where w_full is w zero-padded to length 2n-1
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # We need h[i] = sum_k w[k] * a[k-i] for i in [0, n-1]
    # This is the cross-correlation of w and a, which is conv(w, a_flipped)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, c


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):

    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)

    a = 1.0 + 0.5 * np.sin(2 * np.pi * rate * t)
    a = np.maximum(a, 1e-12)

def _make_multiband_profile(n, n_bands=3, bandwidth=0.3):
    """Multi-band spectral profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for i in range(n_bands):
        center = (i + 0.5) / n_bands
        a += np.exp(-0.5 * ((t - center) / bandwidth) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_block_profile(n, n_blocks=4, heights=None):

        heights = np.random.default_rng(42).uniform(0.5, 2.0, n_blocks)

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_spread(a, M):
    """LP: max sum(b) s.t. a*b <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 10})
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=10000, inflate=1.001, blend=0.3):
    """LP-based refinement with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.05)
            if no_imp > 600:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0003, inflate_cur * 0.97)
            # Gradually increase blend for faster convergence
            blend_cur = min(0.5, blend_cur * 1.01)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.01, inflate_cur * 0.97)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _multi_scale_gradient_polish(a, deadline, max_iter=20000):
    """Gradient descent using multiple scales of smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)

    c = _fft_convolve(a)

    # Multiple beta scales
    c_max = float(c.max())
    betas = [0.01 * c_max, 0.05 * c_max, 0.15 * c_max]
    weights = [0.5, 0.35, 0.15]  # Focus on sharper scale

    lr = 0.04
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Multi-scale gradient
        c_cur = _fft_convolve(a)
        c_max_cur = float(c_cur.max())

        # Update betas relative to current max
        betas_cur = [b / c_max * c_max_cur for b in betas]
        betas_cur = [max(1e-10, b) for b in betas_cur]

        grad_total = np.zeros(n)
        for beta, w in zip(betas_cur, weights):
            g, _ = _gradient_smooth_max(a, beta)
            grad_total += w * g

        # Normalize gradient
        g_norm = float(np.linalg.norm(grad_total))
        if g_norm < 1e-15:
            break
        grad_normed = grad_total / g_norm

        if not np.all(np.isfinite(grad_normed)):
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad_normed * g_norm
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
            # Decay momentum slightly on progress
            momentum = max(0.7, momentum * 0.999)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.003, lr * 0.97)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

        if it % 200 == 0:
            # Gradually focus on the sharpest scale
            weights[0] = min(0.8, weights[0] * 1.02)
            total_w = sum(weights)
            weights = [w / total_w for w in weights]

    return best_a


def _peak_targeted_polish(a, deadline, max_iter=15000, top_k=5):
    """Focus gradient descent on the top-k peaks specifically."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    lr = 0.03
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        # Create peak-focused weight
        sorted_idx = np.argsort(c)[::-1]
        peak_indices = sorted_idx[:top_k]
        peak_vals = c[peak_indices]
        # Weight by how much above average
        avg = float(c.mean())
        excess = np.maximum(peak_vals - avg, 0)
        if excess.sum() < 1e-15:
            excess = np.ones(top_k)
        w_peaks = excess / excess.sum()

        # Compute gradient for each peak and combine
        grad = np.zeros(n)
        N = _next_pow2(2 * n - 1)
        fa_rev = np.fft.rfft(a[::-1], N)

        for idx, w in zip(peak_indices, w_peaks):
            # Gradient of (a*a)[idx] w.r.t. a
            # d(a*a)[idx]/da[i] = a[idx-i] if 0<=idx-i<n
            lo = max(0, idx - n + 1)
            hi = min(idx, n - 1)
            if lo <= hi:
                # grad[i] += w * a[idx - i] for i in [lo, hi]
                grad[lo:hi+1] += w * a[idx - hi:idx - lo + 1][::-1]
            # Also the symmetric part
            # Actually for (a*a)[k] = sum_j a[j]*a[k-j]:
            # d/da[i] = a[k-i] (if k-i valid) + a[k-i] (if i != k-i, from the other term)
            # = a[k-i] * (1 + [i != k-i]) ... no, let me just use the full formula
            # (a*a)[k] = sum_{j valid} a[j] * a[k-j]
            # d/da[i]: appears when j=i (gives a[k-i]) and when k-j=i i.e. j=k-i (gives a[i])
            # If i == k-i (i.e., k=2i): only one term, gives 2*a[i]
            # If i != k-i: two terms, gives a[k-i] + a[i]... wait no
            # Let me just be careful:
            # (a*a)[k] = sum_j a[j]*a[k-j]
            # d/da[i] = a[k-i] (from j=i term, if k-i in [0,n-1])
            #           + sum over j where k-j=i, i.e., j=k-i: a[j]*da[k-j]/da[i] = a[k-i]*1 (if k-i in [0,n-1])
            # But if i = k-i (k=2i): the term a[i]*a[i] appears once, derivative is 2*a[i]
            # If i != k-i: a[i]*a[k-i] and a[k-i]*a[i] are the same term counted once
            #   derivative = a[k-i] (from a[i] factor) + a[i] (from a[k-i] factor)
            #   Wait no, (a*a)[k] = sum_j a[j]*a[k-j]. The term with a[i] appears as:
            #   - j=i: a[i]*a[k-i]
            #   - j=k-i: a[k-i]*a[i] (if k-i != i, this is a different j)
            #   So d/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if i != k-i
            #   Or d/da[i] = 2*a[i] if i = k-i
            #   In both cases: d/da[i] = 2*a[k-i] (since if i=k-i, then a[k-i]=a[i])
            pass  # The FFT-based computation handles this correctly

        # Use FFT-based gradient for the weighted sum
        w_vec = np.zeros(2 * n - 1)
        for idx, w in zip(peak_indices, w_peaks):
            w_vec[idx] = w
        fw = np.fft.rfft(w_vec, N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            continue

        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-15:
            break

        v = momentum * v - lr * (grad / g_norm) * 0.1
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.97)
            if no_imp > max(1500, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.03
                no_imp = 0

    return best_a


def _block_perturbation_optimize(a, deadline, max_iter=3000):
    """Random block swaps and transfers to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(54321)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        move_type = rng.integers(0, 3)

        if move_type == 0:
            # Block swap
            block_size = int(rng.integers(max(2, n // 64), max(4, n // 16)))
            i = int(rng.integers(0, n - block_size))
            j = int(rng.integers(0, n - block_size))
            if abs(i - j) < block_size:
                continue
            a_new = a.copy()
            a_new[i:i+block_size], a_new[j:j+block_size] = \
                a_new[j:j+block_size].copy(), a_new[i:i+block_size].copy()
        elif move_type == 1:
            # Mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = float(rng.uniform(0.05, 0.3))
            a_new = a.copy()
            delta = frac * a[i]
            a_new[i] -= delta
            a_new[j] += delta
            a_new = np.maximum(a_new, 0.0)
        else:
            # Smooth perturbation
            sigma = float(rng.uniform(0.01, 0.05))
            a_new = a * (1.0 + sigma * rng.standard_t(5, n))
            a_new = np.maximum(a_new, 0.0)

        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 2000:
                break

    return best_a


def _adam_polish(a, deadline, max_iter=20000, lr=0.04):
    """Logit-space Adam optimization with smooth max."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.03 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        pp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * pp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.97)

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(2000, 3 * n):
            rng_r = np.random.default_rng((it * 7 + n) % (2**32))
            z = 0.8 * best_z + 0.2 * rng_r.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.03 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _full_pipeline(a_init, deadline, budget):
    """Run optimization pipeline on a single sequence."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    # Phase A: LP refinement
    lp_time = min(budget * 0.3, 60.0)
    rem = deadline - time.monotonic()
    if lp_time > 5 and rem > lp_time + 20:
        a_lp = _lp_refine(a, time.monotonic() + lp_time, max_iter=8000, inflate=1.0012, blend=0.25)
        sc = _score(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
        print(f"      LP: {sc:.10f}")

    # Phase B: Multi-scale gradient
    gp_time = min(budget * 0.25, 50.0)
    rem = deadline - time.monotonic()
    if gp_time > 5 and rem > gp_time + 20:
        a_gp = _multi_scale_gradient_polish(best_a, time.monotonic() + gp_time, max_iter=8000)
        sc = _score(a_gp)
        if sc < best_score:
            best_score = sc
            best_a = a_gp.copy()
        print(f"      MultiScale: {sc:.10f}")

    # Phase C: Peak-targeted
    pt_time = min(budget * 0.2, 40.0)
    rem = deadline - time.monotonic()
    if pt_time > 5 and rem > pt_time + 20:
        a_pt = _peak_targeted_polish(best_a, time.monotonic() + pt_time, max_iter=6000, top_k=5)
        sc = _score(a_pt)
        if sc < best_score:
            best_score = sc
            best_a = a_pt.copy()
        print(f"      PeakTarget: {sc:.10f}")

    # Phase D: Adam polish
    ad_time = min(budget * 0.15, 35.0)
    rem = deadline - time.monotonic()
    if ad_time > 5 and rem > ad_time + 15:
        a_ad = _adam_polish(best_a, time.monotonic() + ad_time, max_iter=8000, lr=0.035)
        sc = _score(a_ad)
        if sc < best_score:
            best_score = sc
            best_a = a_ad.copy()
        print(f"      Adam: {sc:.10f}")

    # Phase E: Block perturbation
    bp_time = min(budget * 0.08, 15.0)
    rem = deadline - time.monotonic()
    if bp_time > 3 and rem > bp_time + 10:
        a_bp = _block_perturbation_optimize(best_a, time.monotonic() + bp_time, max_iter=1500)
        sc = _score(a_bp)
        if sc < best_score:
            best_score = sc
            best_a = a_bp.copy()
        print(f"      BlockPert: {sc:.10f}")

    # Phase F: Final LP
    lp2_time = min(budget * 0.07, 15.0)
    rem = deadline - time.monotonic()
    if lp2_time > 3 and rem > lp2_time + 5:
        a_lp2 = _lp_refine(best_a, time.monotonic() + lp2_time, max_iter=3000, inflate=1.0005, blend=0.15)
        sc = _score(a_lp2)
        if sc < best_score:
            best_score = sc
            best_a = a_lp2.copy()
        print(f"      FinalLP: {sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining LP spreading, multi-scale gradient,
    peak-targeted optimization, Adam in logit space, and block perturbations.
    Explores multiple sequence lengths with diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [384, 512, 640, 768, 896, 1024, 1280]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))

    print(f"Lengths to try: {lengths}")
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: Quick screening - evaluate diverse inits quickly
    print("\n=== Phase 1: Quick screening ===")
    screen_results = []  # (score, n, a_init)
    screen_time = min(120.0, (deadline - t_start) * 0.1)
    screen_deadline = t_start + screen_time

    for n_t in lengths:
        if time.monotonic() > screen_deadline:
            break

        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
        inits.append(_make_linear_profile(n_t, 1.5))
        inits.append(_make_linear_profile(n_t, 2.5))
        inits.append(_make_u_profile(n_t, 2.0))
        inits.append(_make_u_profile(n_t, 4.0))
        inits.append(_make_chirp_profile(n_t, 3.0))
        inits.append(_make_chirp_profile(n_t, 5.0))
        inits.append(_make_multiband_profile(n_t, 3, 0.25))
        inits.append(_make_multiband_profile(n_t, 5, 0.15))
        # Random structured
        for _ in range(3):
            a_r = np.maximum(rng.normal(1.0, 0.4, n_t), 0.1)
            inits.append(a_r / a_r.sum())

        for a0 in inits:
            if time.monotonic() > screen_deadline:
                break
            sc = _score(a0)
            screen_results.append((sc, n_t, a0))

    # Sort by score
    screen_results.sort(key=lambda x: x[0])
    print(f"  Screened {len(screen_results)} inits, best: {screen_results[0][0]:.10f} (n={screen_results[0][1]})")

    if screen_results[0][0] < best_score:
        best_score = screen_results[0][0]
        best_seq = screen_results[0][2].copy()

    # Phase 2: Focused optimization on top candidates
    print("\n=== Phase 2: Focused optimization ===")
    # Group by length, take best 2 per length
    by_length = {}
    for sc, n_t, a0 in screen_results[:20]:
        if n_t not in by_length:
            by_length[n_t] = []
        if len(by_length[n_t]) < 2:
            by_length[n_t].append(a0)

    # Also add the parent if not already there
    if parent is not None and parent.size not in by_length:
        by_length[parent.size] = [parent.copy()]

    num_candidates = sum(len(v) for v in by_length.values())
    # Allocate time per candidate
    phase2_deadline = deadline - 120.0  # Reserve 120s for phase 3
    per_candidate = max(20.0, (phase2_deadline - time.monotonic()) / max(1, num_candidates))

    candidate_list = []
    for n_t, inits in sorted(by_length.items()):
        for a0 in inits:
            candidate_list.append((n_t, a0))

    # Sort by screening score (approximate)
    # We already have them roughly sorted, just process in order
    for i, (n_t, a0) in enumerate(candidate_list):
        if time.monotonic() > phase2_deadline:
            break

        # Dynamic budget: give more time to promising candidates
        frac = (deadline - time.monotonic() - 100.0) / max(1, len(candidate_list) - i)
        cand_budget = min(per_candidate * 1.5, frac)
        cand_budget = max(15.0, min(cand_budget, 90.0))

        print(f"\n  Candidate {i+1}/{len(candidate_list)}: n={n_t}, budget={cand_budget:.0f}s")
        a_best, sc_best = _full_pipeline(a0, deadline, cand_budget)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best.copy()
            print(f"  ** New best: {sc_best:.10f} (n={n_t})")

    # Phase 3: Final intensive polish on the overall best
    print(f"\n=== Phase 3: Final polish (score={best_score:.10f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        final_budget = deadline - time.monotonic() - 20.0
        print(f"  Final budget: {final_budget:.0f}s")

        # 3a: Aggressive LP
        lp_b = min(final_budget * 0.3, 60.0)
        if lp_b > 10:
            a_lp = _lp_refine(best_seq, time.monotonic() + lp_b,
                             max_iter=15000, inflate=1.0005, blend=0.2)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  FinalLP: {sc:.10f}")

        # 3b: Multi-scale gradient
        gs_b = min(final_budget * 0.2, 40.0)
        if gs_b > 5:
            a_gs = _multi_scale_gradient_polish(best_seq, time.monotonic() + gs_b, max_iter=10000)
            sc = _score(a_gs)
            if sc < best_score:
                best_score = sc
                best_seq = a_gs.copy()
            print(f"  FinalMultiScale: {sc:.10f}")

        # 3c: Peak targeted
        pt_b = min(final_budget * 0.2, 35.0)
        if pt_b > 5:
            a_pt = _peak_targeted_polish(best_seq, time.monotonic() + pt_b, max_iter=8000, top_k=3)
            sc = _score(a_pt)
            if sc < best_score:
                best_score = sc
                best_seq = a_pt.copy()
            print(f"  FinalPeak: {sc:.10f}")

        # 3d: Adam
        ad_b = min(final_budget * 0.15, 30.0)
        if ad_b > 5:
            a_ad = _adam_polish(best_seq, time.monotonic() + ad_b, max_iter=10000, lr=0.025)
            sc = _score(a_ad)
            if sc < best_score:
                best_score = sc
                best_seq = a_ad.copy()
            print(f"  FinalAdam: {sc:.10f}")

        # 3e: Block perturbation
        bp_b = min(final_budget * 0.1, 20.0)
        if bp_b > 3:
            a_bp = _block_perturbation_optimize(best_seq, time.monotonic() + bp_b, max_iter=2000)
            sc = _score(a_bp)
            if sc < best_score:
                best_score = sc
                best_seq = a_bp.copy()
            print(f"  FinalBlock: {sc:.10f}")

        # 3f: One more LP with very tight inflation
        lp_f_b = min(final_budget * 0.05, 15.0)
        if lp_f_b > 3:
            a_lpf = _lp_refine(best_seq, time.monotonic() + lp_f_b,
                              max_iter=3000, inflate=1.0003, blend=0.1)
            sc = _score(a_lpf)
            if sc < best_score:
                best_score = sc
                best_seq = a_lpf.copy()
            print(f"  FinalLPtight: {sc:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        # Keep the natural scale - don't renormalize since score is scale-invariant
        pass
    else:
        final_seq = _make_linear_profile(512, 1.5)
        s = final_seq.sum()
        final_seq = final_seq / s

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"  Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.0f}s")
    print(f"{'='*50}")
    return [float(x) for x in final_seq]
