    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_smooth_max(a, deadline, max_iter=30000, lr=0.03, beta_factor=0.04):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max_fft(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr_cur = max(0.001, lr_cur * 0.92)
            if no_imp > max(5000, 8 * n):
                rng_r = np.random.default_rng(1000 + it)
                z = best_z + 0.25 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.75)

    return best_a, best_score


def _bottleneck_projected_gradient(a, deadline, max_iter=5000, step=0.3):
    """Targeted projected gradient on the current bottleneck lag."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        max_idx = int(np.argmax(c_curr))

        sg = _correct_subgradient(a, max_idx)
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        a_new = a - step * sg / sg_norm
        a_new = _project_simplex(a_new)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
        else:
            step = step * 0.95

        a = a_new

        if it % 500 == 0:
            step = min(0.3, step * 1.02)

    return best_a, best_score


def _multi_subgradient_corrected(a, deadline, max_iter=10000, top_k=3):
    """Multi-peak subgradient with corrected gradients."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.15 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]

        weights = np.array([(1.0 + 0.4 * i) for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _correct_subgradient(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = _project_simplex(a_new)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30})
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.002, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.05)
                blend_cur = min(0.6, blend_cur * 1.1)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
                inflate_cur = min(1.005, inflate_cur * 0.93)
            if no_imp > max(150, int(1.5 * n)):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _make_decreasing_profile(n, gamma=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** gamma + 0.01
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_decay(n, rate=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t) + 0.01
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_two_bump(n, pos1=0.2, pos2=0.8, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - pos1) ** 2) / (2 * width ** 2)) + np.exp(-((t - pos2) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _hot_potato_optimization(a, deadline, max_iter=3000):
    """Iteratively target the current bottleneck lag with projected gradient,
    switching when the bottleneck moves."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    step = 0.2
    current_k = int(np.argmax(c))
    same_k_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        max_idx = int(np.argmax(c_curr))

        if max_idx != current_k:
            same_k_count = 0
            current_k = max_idx
            step = 0.2
        else:
            same_k_count += 1

        sg = _correct_subgradient(a, current_k)
        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        a_new = a - step * sg / sg_norm
        a_new = _project_simplex(a_new)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
        else:
            step = max(0.01, step * 0.9)

        a = a_new

        if same_k_count > 500:
            step = max(0.01, step * 0.7)

        if it % 200 == 0 and same_k_count > 100:
            step = min(0.3, step * 1.1)

    return best_a, best_score


def _full_pipeline(a0, deadline, label=""):
    """Run full optimization pipeline on a0."""
    sc0 = _score_normalized(a0)
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: Adam smooth max
    if time.monotonic() < deadline - 30:
        d1 = min(deadline - 30, time.monotonic() + 200)
        a1, s1 = _adam_smooth_max(a0, d1, max_iter=20000, lr=0.025, beta_factor=0.04)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam1: {best_sc:.10f}")

    # Phase 2: Hot potato (bottleneck targeting)
    if time.monotonic() < deadline - 20:
        d2 = min(deadline - 20, time.monotonic() + 150)
        a2, s2 = _hot_potato_optimization(best_a, d2, max_iter=5000)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} HotPotato: {best_sc:.10f}")

    # Phase 3: LP refinement
    if time.monotonic() < deadline - 15:
        d3 = min(deadline - 15, time.monotonic() + 120)
        a3 = _lp_refine(best_a, d3, max_iter=1500, inflate=1.0015, blend=0.3)
        s3 = _score_normalized(a3)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} LP: {best_sc:.10f}")

    # Phase 4: Fine Adam
    if time.monotonic() < deadline - 10:
        d4 = deadline - 10
        a4, s4 = _adam_smooth_max(best_a, d4, max_iter=15000, lr=0.008, beta_factor=0.015)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} Adam2: {best_sc:.10f}")

    # Phase 5: Final hot potato
    if time.monotonic() < deadline - 5:
        d5 = deadline - 5
        a5, s5 = _hot_potato_optimization(best_a, d5, max_iter=3000)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} FinalHP: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining corrected smooth-max Adam,
    bottleneck-targeted projected gradient (hot potato), and LP refinement.
    Searches at multiple sequence lengths (512-4096) with time-aware allocation.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_normalized(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is not None:
        base_512 = _interpolate_profile(parent, 512)
        base_1024 = _interpolate_profile(parent, 1024)
        base_2048 = _interpolate_profile(parent, 2048)
    else:
        base_512 = _make_decreasing_profile(512, 1.5)
        base_1024 = _make_decreasing_profile(1024, 1.5)
        base_2048 = _make_decreasing_profile(2048, 1.5)

    # === Strategy 1: Full pipeline on parent @ n=1024 (most reliable) ===
    remaining = deadline - time.monotonic()
    if remaining > 300:
        print("=== Strategy 1: Parent @ n=1024 ===")
        d1 = min(deadline - 400, t_start + 400)
        a_best, s_best = _full_pipeline(base_1024, d1, "S1-1024")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()

    # === Strategy 2: Full pipeline on parent @ n=2048 (higher potential) ===
    remaining = deadline - time.monotonic()
    if remaining > 350:
        print("=== Strategy 2: Parent @ n=2048 ===")
        d2 = min(deadline - 250, time.monotonic() + 350)
        a_best2, s_best2 = _full_pipeline(base_2048, d2, "S2-2048")
        if s_best2 < best_score:
            best_score = s_best2
            best_seq = a_best2.copy()

    # === Strategy 3: Diverse initializations at n=1024 ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        print("=== Strategy 3: Diverse inits @ n=1024 ===")
        n = 1024
        candidates = [
            _make_decreasing_profile(n, 0.8),
            _make_decreasing_profile(n, 2.0),
            _make_exp_decay(n, 1.5),
            _make_exp_decay(n, 3.0),
            _make_two_bump(n, 0.3, 0.7, 0.2),
        ]
        # Perturbed parent
        p_pert = base_1024 * (1.0 + 0.1 * rng.normal(size=n))
        p_pert = np.maximum(p_pert, 0.0)
        candidates.append(p_pert / p_pert.sum())

        # Asymmetric linear
        t = np.linspace(0, 1, n)
        asym = (1 - t) ** 1.2 + 0.05 * t
        candidates.append(asym / asym.sum())

        time_per = min(30.0, (deadline - 200 - time.monotonic()) / max(1, len(candidates)))
        for i, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 200:
                break
            if time_per < 10:
                break
            d3i = min(deadline - 200, time.monotonic() + time_per)
            a_q, s_q = _adam_smooth_max(a0, d3i, max_iter=4000, lr=0.02, beta_factor=0.05)
            # Quick hot potato
            if time.monotonic() < deadline - 200:
                a_q2, s_q2 = _hot_potato_optimization(a_q, min(deadline - 200, time.monotonic() + 10), max_iter=1000)
                if s_q2 < s_q:
                    a_q, s_q = a_q2, s_q2
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
            print(f"  S3-{i}: {s_q:.10f}")

    # === Strategy 4: Focused final optimization ===
    remaining = deadline - time.monotonic()
    if best_seq is not None and remaining > 30:
        print(f"=== Strategy 4: Final focus (best={best_score:.10f}, n={best_seq.size}) ===")

        # Fine-tune at current length
        d4a = min(deadline - 10, time.monotonic() + 120)
        a_f1, s_f1 = _adam_smooth_max(best_seq, d4a, max_iter=10000, lr=0.005, beta_factor=0.01)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Final Adam: {best_score:.10f}")

        # Hot potato polish
        if time.monotonic() < deadline - 8:
            d4b = min(deadline - 8, time.monotonic() + 80)
            a_f2, s_f2 = _hot_potato_optimization(best_seq, d4b, max_iter=3000)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Final HotPotato: {best_score:.10f}")

        # LP polish
        if time.monotonic() < deadline - 5:
            d4c = min(deadline - 5, time.monotonic() + 60)
            a_f3 = _lp_refine(best_seq, d4c, max_iter=1000, inflate=1.0008, blend=0.2)
            s_f3 = _score_normalized(a_f3)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  Final LP: {best_score:.10f}")

        # Ultra-fine Adam
        if time.monotonic() < deadline - 3:
            a_f4, s_f4 = _adam_smooth_max(best_seq, deadline - 3, max_iter=5000, lr=0.002, beta_factor=0.005)
            if s_f4 < best_score:
                best_score = s_f4
                best_seq = a_f4.copy()
            print(f"  Ultra-final Adam: {best_score:.10f}")

    # Also try: take best and interpolate to a different length, then quick optimize
    if best_seq is not None and time.monotonic() < deadline - 20:
        best_n = best_seq.size
        target_n = 2048 if best_n <= 1024 else 4096
        if target_n != best_n:
            print(f"=== Strategy 5: Try n={target_n} from best ===")
            interp = _interpolate_profile(best_seq, target_n)
            d5 = min(deadline - 10, time.monotonic() + 80)
            a5, s5 = _adam_smooth_max(interp, d5, max_iter=5000, lr=0.01, beta_factor=0.02)
            if time.monotonic() < deadline - 5:
                a5b, s5b = _hot_potato_optimization(a5, min(deadline - 5, time.monotonic() + 30), max_iter=1000)
                if s5b < s5:
                    a5, s5 = a5b, s5b
            if s5 < best_score:
                best_score = s5
                best_seq = a5.copy()
            print(f"  Strategy 5 (n={target_n}): {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_decreasing_profile(256, 1.5)
        best_score = _score_normalized(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_decreasing_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
