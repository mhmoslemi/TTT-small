2. LP-based refinement (Sinkhorn-like)
3. QP-based fictitious play (minimax approach)
4. Multi-scale search
5. KKT-based active set identification
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if mval <= 0:
        return 0.0, np.zeros_like(c)
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    """Compute cross-correlation: result[i] = sum_j w[i+j] * p[j]"""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    # h has length N, and h[n-1 + i] = sum_j w[i+j] * p[j] for valid i,j
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve(a, M, max_time=30.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    t0 = time.monotonic()
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=20):
    """LP-based refinement loop."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
            print(f"    LP step {n_solves+1}: score={sc_new:.10f}")
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, t * 0.85)
        n_solves += 1
    return current_best, current_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=300, noise_scale=None, lr_decay=0.95):
    """Adam optimization in logit space with smooth max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        # Project gradient onto tangent space of simplex
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

        # Adaptive beta
        if t_adam % 50 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.5:
                    beta = max(1e-5, beta * 0.5)

        # Restart if stuck
        if no_imp > no_imp_limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.05
                elif best_score < 1.53:
                    ns = 0.1
                else:
                    ns = 0.2
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

        # Learning rate decay
        if t_adam % 200 == 0:
            lr = max(0.002, lr * lr_decay)

    return best_p, best_score


def _qp_fictitious_play(p0, deadline, rng, n_iter=50, lr_q=0.1):
    """
    Fictitious play: minimize max_k conv(a,a)[k] via alternating QP and q-update.
    
    Primal: solve min a^T Q_a a s.t. sum(a)=1, a>=0 where Q_a[i,j] = q[i+j]
    Dual:   update q towards argmax of conv(a,a)
    """
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    
    best_p = p.copy()
    best_score = _score_norm(p)
    
    # Initialize q as slightly peaked at the current max lag
    c = _full_convolve(p)
    # q is a distribution over lags 0..2n-2
    q = np.ones(2 * n - 1) / (2 * n - 1)
    # Add some weight to the current peak
    peak_idx = int(np.argmax(c))
    q[peak_idx] += 0.5
    q = q / q.sum()
    
    for it in range(n_iter):
        if time.monotonic() > deadline - 5:
            break
        
        # Primal step: solve QP min a^T Q a s.t. sum(a)=1, a>=0
        # Q[i,j] = q[i+j] (with q_k = 0 for k >= 2n-1)
        t_qp = time.monotonic()
        
        # Build Q as a matrix (for n up to 1024, this is 1024x1024)
        # Q[i,j] = q[i+j] if i+j < 2n-1 else 0
        idx = np.add.outer(np.arange(n), np.arange(n))
        Q = np.where(idx < 2 * n - 1, q[np.minimum(idx, 2 * n - 2)], 0.0)
        
        # Solve QP using scipy minimize
        def obj(a):
            return a @ Q @ a
        
        def grad(a):
            return 2.0 * Q @ a
        
        constraints = [{'type': 'eq', 'fun': lambda a: a.sum() - 1.0,
                        'jac': lambda a: np.ones(n)}]
        bounds = [(0, None)] * n
        
        res = minimize(obj, p, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-12})
        
        if res.success:
            p_new = np.maximum(res.x, 0.0)
            s = p_new.sum()
            if s > 1e-10:
                p_new = p_new / s
                sc = _score_norm(p_new)
                if sc < best_score:
                    best_score = sc
                    best_p = p_new.copy()
                p = p_new
        
        # Dual step: update q
        c = _full_convolve(p)
        # softmax-like update on q
        beta_q = max(0.01, 0.5 * (c.max() - c.min()))
        q_raw = np.exp((c - c.max()) / beta_q)
        q_new = (1 - lr_q) * q + lr_q * q_raw / q_raw.sum()
        q = q_new
    
    return best_p, best_score

    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve(a, M, time_limit=None):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    opts = {'time_limit': time_limit} if time_limit else {}
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds,
                  method='highs', options=opts)
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t0=0.4, max_solves=25, decay=0.88):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t0
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        remain = deadline - time.monotonic()
        if remain < 5:
            break
        tl = max(1.0, min(remain * 0.4, 60.0))
        b = _lp_solve(a, M, time_limit=tl)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        # Line search over t values
        best_t = t
        best_a = a
        best_sc = current_score
        for tt in [t * 0.5, t, t * 1.5]:
            tt = min(tt, 0.9)
            a_new = (1.0 - tt) * a + tt * b_norm
            a_new = np.maximum(a_new, 1e-14)
            a_new = a_new / a_new.sum()
            sc_new = _score_norm(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new
                best_t = tt
        if best_sc < current_score:
            current_score = best_sc
            current_best = best_a.copy()
            print(f"    LP step {n_solves+1}: score={best_sc:.10f} t={best_t:.3f}")
        a = best_a
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.03, best_t * decay)
        n_solves += 1
    return current_best, current_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.07, beta_decay=0.93,
                no_imp_limit=250, noise_scale=None, lr_decay=0.96):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1

        if t_adam % 40 == 0:
            beta = max(1e-6, beta * beta_decay)
            if c.size >= 2:
                sorted_c = np.sort(c)[::-1]
                gap = sorted_c[0] - sorted_c[1]
                if gap < beta:
                    beta = max(1e-6, beta * 0.6)

        if no_imp > no_imp_limit:
            if noise_scale is None:
                if best_score < 1.505:
                    ns = 0.03
                elif best_score < 1.51:
                    ns = 0.06
                elif best_score < 1.53:
                    ns = 0.12
                else:
                    ns = 0.25
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-4, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

        if t_adam % 300 == 0:
            lr = max(0.002, lr * lr_decay)

    return best_p, best_score


def _qp_fictitious_play(p0, deadline, rng, n_iter=40, lr_q=0.15):
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    n = p.size
    p = np.maximum(p, 1e-12)
    p = p / p.sum()

    best_p = p.copy()
    best_score = _score_norm(p)

    c = _full_convolve(p)
    q = np.ones(2 * n - 1) / (2 * n - 1)
    peak_idx = int(np.argmax(c))
    q[peak_idx] *= 3.0
    q = q / q.sum()

    for it in range(n_iter):
        if time.monotonic() > deadline - 5:
            break

        # Build Q matrix: Q[i,j] = q[i+j]
        idx = np.add.outer(np.arange(n), np.arange(n))
        Q = np.where(idx < 2 * n - 1, q[np.minimum(idx, 2 * n - 2)], 0.0)

        def _qp_obj(a):
            return float(a @ Q @ a)

        def _qp_grad(a):
            return 2.0 * (Q @ a)

        constraints = [{'type': 'eq', 'fun': lambda a: float(a.sum() - 1.0),
                        'jac': lambda a: np.ones(n)}]
        bounds = [(0.0, None)] * n

        maxit = 150 if n <= 1024 else 80
        res = sp_minimize(_qp_obj, p, jac=_qp_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxit, 'ftol': 1e-14})

        if res.success:
            p_new = np.maximum(res.x, 0.0)
            s = float(p_new.sum())
            if s > 1e-10:
                p_new = p_new / s
                sc = _score_norm(p_new)
                if sc < best_score:
                    best_score = sc
                    best_p = p_new.copy()
                p = p_new

        c = _full_convolve(p)
        beta_q = max(0.005, 0.3 * (c.max() - c.min()))
        q_raw = np.exp((c - c.max()) / beta_q)
        q_raw = q_raw / q_raw.sum()
        q = (1.0 - lr_q) * q + lr_q * q_raw

    return best_p, best_score


def _local_search(a, deadline, rng, n_trials=5000):
    n = len(a)
    current = a.copy()
    current_score = _score_norm(current)
    best = current.copy()
    best_score = current_score
    c = _full_convolve(current)

    eps_base = 0.01
    trial = 0
    while trial < n_trials and time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        eps = eps_base * rng.uniform(0.1, 1.0)
        # Try moving mass from j to i
        if current[j] < eps:
            continue
        a_test = current.copy()
        a_test[i] += eps
        a_test[j] -= eps
        a_test = np.maximum(a_test, 0.0)
        a_test = a_test / a_test.sum()
        sc = _score_norm(a_test)
        if sc < current_score:
            current = a_test
            current_score = sc
            if sc < best_score:
                best_score = sc
                best = current.copy()
        elif current_score < best_score * (1 - 1e-6):
            # Occasionally accept worse moves (simulated annealing)
            if rng.random() < 0.01:
                current = a_test
                current_score = sc
        trial += 1

    return best, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # Phase 1: Setup (first 30s)
    start_p = None
    if parent is not None:
        start_p = parent.copy()
        sc0 = _score_norm(start_p)
        print(f"  Parent score: {sc0:.10f}")
        _update(start_p, sc0, 'parent')
    else:
        start_p = _make_profile_linear(512, 1.0)
        _update(start_p, _score_norm(start_p), 'default')

    # Phase 2: Adam optimization (300s)
    phase2_end = t_start + 300.0
    print(f"\n=== Phase 2: Adam (until t+300s) ===")
    p2, sc2 = _adam_logit(start_p, phase2_end, rng, lr=0.05, beta_frac=0.07,
                          beta_decay=0.93, no_imp_limit=200)
    _update(p2, sc2, 'adam_phase2')

    # Phase 3: LP refinement (300s)
    phase3_end = min(t_start + 700.0, deadline)
    print(f"\n=== Phase 3: LP refinement (until t+700s) ===")
    if best_seq is not None and best_seq.size <= 2048:
        p3, sc3 = _lp_refine(best_seq, phase3_end, t0=0.4, max_solves=15, decay=0.85)
        _update(p3, sc3, 'lp_phase3')
        # Adam after LP
        if time.monotonic() < phase3_end - 10:
            p3a, sc3a = _adam_logit(p3, phase3_end, rng, lr=0.03, beta_frac=0.05,
                                     beta_decay=0.94, no_imp_limit=200)
            _update(p3a, sc3a, 'adam_post_lp3')

    # Phase 4: QP Fictitious Play (200s)
    phase4_end = min(t_start + 1000.0, deadline)
    print(f"\n=== Phase 4: QP Fictitious Play ===")
    if best_seq is not None and best_seq.size <= 1536:
        p4, sc4 = _qp_fictitious_play(best_seq, phase4_end, rng, n_iter=30, lr_q=0.15)
        _update(p4, sc4, 'qp_fictitious')
        # Polish after QP
        if time.monotonic() < phase4_end - 10:
            p4a, sc4a = _adam_logit(best_seq, phase4_end, rng, lr=0.02, beta_frac=0.04,
                                     beta_decay=0.95, no_imp_limit=150)
            _update(p4a, sc4a, 'adam_post_qp')

    # Phase 5: LP + Adam alternation (200s)
    phase5_end = min(t_start + 1300.0, deadline)
    print(f"\n=== Phase 5: LP+Adam alternation ===")
    if best_seq is not None and best_seq.size <= 2048:
        n_alt = 2
        per_alt = max(20.0, (phase5_end - time.monotonic()) / max(1, n_alt * 2))
        for alt_i in range(n_alt):
            if time.monotonic() > phase5_end - 10:
                break
            # LP step
            lp_end = time.monotonic() + min(per_alt, phase5_end - time.monotonic() - 30)
            if lp_end > time.monotonic() + 10:
                p5, sc5 = _lp_refine(best_seq, lp_end, t0=0.3, max_solves=5, decay=0.85)
                _update(p5, sc5, f'lp_alt{alt_i}')
            # Adam step
            ad_end = time.monotonic() + min(per_alt, phase5_end - time.monotonic() - 10)
            if ad_end > time.monotonic() + 10:
                p5a, sc5a = _adam_logit(best_seq, ad_end, rng, lr=0.02, beta_frac=0.04,
                                         beta_decay=0.95, no_imp_limit=150)
                _update(p5a, sc5a, f'adam_alt{alt_i}')

    # Phase 6: Try n=2048 escalation
    phase6_end = min(t_start + 1550.0, deadline)
    print(f"\n=== Phase 6: n=2048 escalation ===")
    if best_seq is not None and best_seq.size <= 1024 and time.monotonic() < phase6_end - 60:
        p_int = _interp_profile(best_seq, 2048)
        if p_int is not None:
            sc_int = _score_norm(p_int)
            print(f"  Interp to n=2048: {sc_int:.10f}")
            _update(p_int, sc_int, 'interp_2048')
            ad_end = min(phase6_end - 10, time.monotonic() + 120)
            p6, sc6 = _adam_logit(p_int, ad_end, rng, lr=0.04, beta_frac=0.06,
                                   beta_decay=0.93, no_imp_limit=200)
            _update(p6, sc6, 'adam_2048')

    # Phase 7: Local search
    phase7_end = min(t_start + 1700.0, deadline)
    print(f"\n=== Phase 7: Local search ===")
    if best_seq is not None and best_seq.size <= 2048:
        p7, sc7 = _local_search(best_seq, phase7_end, rng, n_trials=3000)
        _update(p7, sc7, 'local_search')

    # Phase 8: Final polish
    print(f"\n=== Phase 8: Final polish ===")
    final_end = deadline - 5.0
    if best_seq is not None and time.monotonic() < final_end - 10:
        per = final_end - time.monotonic() - 5
        if per > 15:
            p8, sc8 = _adam_logit(best_seq, time.monotonic() + per, rng, lr=0.01,
                                   beta_frac=0.03, beta_decay=0.96, no_imp_limit=100)
            _update(p8, sc8, 'final_polish')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(512, 1.5)

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    fb = _make_profile_linear(512, 1.5)
    print(f"FALLBACK n=512")
    return [float(x) for x in fb]
