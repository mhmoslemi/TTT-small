"""
Optimization of step function coefficients to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses spectral initialization, LSE gradient with adaptive annealing, LP refinement,
and subgradient polish. Explores multiple lengths and diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-15:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    return _normalize(p)


def _make_spectral_init(n, rng):
    """Create initialization from random phase in frequency domain."""
    # We want |A(f)|^2 to be as flat as possible
    # Start with flat magnitude, random phase
    N = _next_pow2(2 * n - 1)
    magnitude = np.ones(N // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, size=N // 2 + 1)
    phase[0] = 0
    phase[-1] = 0 if (N % 2 == 0) else phase[-1]
    spectrum = magnitude * np.exp(1j * phase)
    # Make it conjugate symmetric
    spectrum = np.concatenate([spectrum, np.conj(spectrum[N // 2:0:-1])])
    a = np.fft.irfft(spectrum, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, separation=0.3):
    """Bimodal: peaks at edges."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - (0.5 - separation)) / 0.1) ** 2) + \
        np.exp(-0.5 * ((t - (0.5 + separation)) / 0.1) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, edge_weight=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_linear_profile(n, eta=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, alpha=2.0):
    """a_i proportional to (i+1)^(-alpha) - heavy at start."""
    t = np.arange(n, dtype=np.float64) + 1.0
    a = t ** (-alpha)
    return a / a.sum()


def _make_conv_matrix_reduced(a, M, top_k=80):
    """Create reduced constraint matrix using only near-active constraints."""
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    b = np.full(len(active), M)
    return A, b


def _lp_spread(a, M, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    if n > 32:
        A, b = _make_conv_matrix_reduced(a, M, top_k)
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lse_gradient(a, T):
    """Compute gradient of log-sum-exp approximation of max(conv)."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # grad of sum_k w_k * (a*a)[k] w.r.t. a
    # = 2 * sum_k w_k * conv(w_indicator_at_k, a_rev)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n - 1:2 * n - 1]
    
    return grad, c, m_val


def _lse_optimize(a, deadline, max_iter=50000, T_start=0.03, T_end=1e-6, lr=0.08, momentum=0.9):
    """Optimize using LSE gradient with Adam and geometric annealing."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    z = np.log(a)
    z -= float(np.mean(z))
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    b1, b2 = 0.9, 0.999
    
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    T = T_start
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, c, m_val = _lse_gradient(a, T)
        
        # Project gradient onto tangent space (sum-preserving)
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            break
        
        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * gz
        v_adam = b2 * v_adam + (1.0 - b2) * (gz * gz)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.95)
            if no_imp > max(3000, 5 * n):
                # Restart from best with perturbation
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_adam = 0
                lr = base_lr
                no_imp = 0
        
        T *= T_ratio
        T = max(T, T_end)
    
    return best_a


def _lp_refine(a, deadline, max_iter=3000, inflate=1.0003, blend=0.15):
    """LP-based spreading with line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp, top_k=80)
        
        if g is None:
            inflate_cur = min(1.02, inflate_cur * 1.3)
            no_imp += 5
            if no_imp > 200:
                break
            continue
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.1)
            if no_imp > 300:
                break
            continue
        
        g = g / sum_g
        
        # Line search
        best_t = 0.0
        best_sc = best_score
        for t_try in [blend_cur, blend_cur * 0.5, blend_cur * 0.25, blend_cur * 2.0]:
            t_try = min(max(t_try, 0.005), 0.5)
            a_try = np.maximum((1.0 - t_try) * a + t_try * g, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t_try
        
        if best_t < 0.005:
            best_t = blend_cur * 0.5
        
        a_new = np.maximum((1.0 - best_t) * a + best_t * g, 0)
        a_new = a_new / a_new.sum()
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend_cur = max(0.02, blend_cur * 0.9)
                inflate_cur = min(1.01, inflate_cur * 0.95)
            if no_imp > max(150, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        
        a = a_new
    
    return best_a


def _subgradient_polish(a, deadline, max_iter=30000, lr=0.02, momentum=0.85):
    """Polish using subgradient at the actual max."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-12)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)
        
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        grad = 2.0 * h_full[n - 1:2 * n - 1]
        
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        
        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.92)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
    
    return best_a

"""
Optimization of step function coefficients to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses spectral initialization, LSE gradient with adaptive annealing, full-constraint LP,
subgradient polish, and coordinate descent refinement. Explores multiple sequence
lengths and diverse initializations to avoid local optima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-14:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    return _normalize(p)


def _make_spectral_init(n, rng):
    N = _next_pow2(2 * n - 1)
    magnitude = np.ones(N // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, size=N // 2 + 1)
    phase[0] = 0
    spectrum = magnitude * np.exp(1j * phase)
    spectrum_full = np.concatenate([spectrum, np.conj(spectrum[1:N // 2 + 1][::-1])])
    a = np.fft.irfft(spectrum_full, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, separation=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = (np.exp(-0.5 * ((t - (0.5 - separation)) / 0.12) ** 2) +
         np.exp(-0.5 * ((t - (0.5 + separation)) / 0.12) ** 2))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, edge_weight=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_power_profile(n, alpha=1.5):
    t = np.arange(n, dtype=np.float64) + 1.0
    a = t ** (-alpha)
    return a / a.sum()


def _make_symmetric_power(n, alpha=1.5):
    """Symmetric heavy-tail: large at both ends."""
    t = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = (1.0 + np.abs(t - mid) / mid) ** alpha
    return a / a.sum()


def _build_full_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _build_reduced_conv_matrix(a, M, top_k=200):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-4, 5):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[row, lo:hi] = a[lo:hi]
    b = np.full(len(active), M)
    return A, b


def _lp_solve(a, M, use_full=False, top_k=200):
    n = len(a)
    c_obj = -np.ones(n)
    if use_full or n <= 256:
        A = _build_full_conv_matrix(a)
        b = np.full(2 * n - 1, M)
    else:
        A, b = _build_reduced_conv_matrix(a, M, top_k)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lse_grad(a, T):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h_full[n - 1:2 * n - 1]
    return grad, c, m_val


def _lse_optimize(a, deadline, max_iter=40000, T_start=0.025, T_end=1e-7, lr=0.07):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    z = np.log(a)
    z -= float(np.mean(z))
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    b1, b2 = 0.9, 0.999

    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    T = T_start
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, c, m_val = _lse_grad(a, T)
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * gz
        v_adam = b2 * v_adam + (1.0 - b2) * (gz * gz)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-11:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.01, lr * 0.94)
            if no_imp > max(2500, 4 * n):
                rng_r = np.random.default_rng(99999 + it)
                a = best_a * (1.0 + 0.04 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_adam = 0
                lr = base_lr
                no_imp = 0

        T *= T_ratio
        T = max(T, T_end)

    return best_a


def _lp_refine(a, deadline, max_iter=2500, inflate=1.0002, blend=0.1, top_k=200):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        use_full = (n <= 256)
        g = _lp_solve(a, M_lp, use_full=use_full, top_k=top_k)

        if g is None:
            inflate_cur = min(1.02, inflate_cur * 1.4)
            no_imp += 5
            if no_imp > 200:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.15)
            if no_imp > 250:
                break
            continue

        g = g / sum_g

        # Multi-point line search
        best_t = 0.0
        best_sc = best_score
        for t_try in [blend_cur, blend_cur * 0.5, blend_cur * 0.25,
                      blend_cur * 1.5, blend_cur * 2.0]:
            t_try = min(max(t_try, 0.003), 0.6)
            a_try = np.maximum((1.0 - t_try) * a + t_try * g, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_norm(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t_try

        if best_t < 0.003:
            best_t = blend_cur * 0.3

        a_new = np.maximum((1.0 - best_t) * a + best_t * g, 0)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 15 == 0:
                blend_cur = max(0.015, blend_cur * 0.88)
                inflate_cur = min(1.01, inflate_cur * 0.92)
            if no_imp > max(120, n // 2):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _subgrad_polish(a, deadline, max_iter=40000, lr=0.015, momentum=0.87):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-13)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        grad = 2.0 * h_full[n - 1:2 * n - 1]

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 250 == 0:
                lr = max(0.002, lr * 0.93)
            if no_imp > max(2500, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _coord_descent(a, deadline, max_rounds=20):
    """Coordinate descent: optimize each coordinate while keeping sum=1."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # To change a[i], we must adjust another coordinate to maintain sum=1
            # Try: increase a[i] by delta, decrease a[j] by delta for some j
            # Best j: the one whose decrease most reduces the max convolution
            # Simplification: just try small increases and decreases
            base = a[i]
            for frac in [1.05, 1.1, 0.95, 0.9]:
                new_val = base * frac
                if new_val <= 0:
                    continue
                # Distribute the difference to another coordinate
                diff = new_val - base
                if abs(diff) < 1e-15:
                    continue
                # Find the coordinate with max value (other than i) to adjust
                a_temp = a.copy()
                a_temp[i] = new_val
                # Redistribute to maintain sum: scale all others
                others = a_temp[1:] if i == 0 else np.concatenate([a_temp[:i], a_temp[i+1:]])
                target_sum = 1.0 - new_val
                if target_sum <= 0:
                    continue
                if others.sum() < 1e-15:
                    continue
                others = others * (target_sum / others.sum())
                if i == 0:
                    a_temp = np.concatenate([[new_val], others])
                else:
                    a_temp = np.concatenate([a_temp[:i], [new_val], a_temp[i+1:]])

                a_temp = np.maximum(a_temp, 0)
                s = a_temp.sum()
                if s < 1e-10:
                    continue
                a_temp = a_temp / s

                sc = _score_norm(a_temp)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_temp.copy()
                    a = a_temp.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_a


def _alternating_refine(a, deadline):
    """Alternate between LSE and LP for sustained improvement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    while time.monotonic() < deadline:
        # LSE phase: ~30 seconds
        lse_end = min(deadline, time.monotonic() + 30.0)
        if deadline - lse_end <= 5:
            break
        a_lse = _lse_optimize(a, lse_end, max_iter=15000,
                              T_start=0.015, T_end=1e-7, lr=0.05)
        sc = _score_norm(a_lse)
        if sc < best_score:
            best_score = sc
            best_a = a_lse.copy()
            a = a_lse
        else:
            a = best_a

        # LP phase: ~25 seconds
        lp_end = min(deadline, time.monotonic() + 25.0)
        if deadline - lp_end <= 3:
            break
        a_lp = _lp_refine(a, lp_end, max_iter=1000,
                          inflate=1.0002, blend=0.08, top_k=250)
        sc = _score_norm(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
            a = a_lp
        else:
            a = best_a

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimization: spectral init, LSE gradient, full-constraint LP,
    subgradient polish, and alternating refinement. Tries multiple lengths and
    diverse starting points to find the global minimum of 2n*max(conv)/sum^2.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Lengths to try
    lengths = [1024]
    if parent is not None and parent.size != 1024:
        lengths.insert(0, parent.size)
    lengths.append(2048)

    def get_cands(n_t):
        cands = []
        if parent is not None:
            cands.append(_interp_profile(parent, n_t))
            for _ in range(2):
                p = _interp_profile(parent, n_t)
                pp = p * (1.0 + 0.05 * rng.normal(size=n_t))
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    cands.append(pp / pp.sum())
        cands.append(_make_gaussian(n_t, 0.3))
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_bimodal(n_t, 0.2))
        cands.append(_make_flat_edges(n_t, 0.6))
        cands.append(_make_spectral_init(n_t, rng))
        cands.append(_make_symmetric_power(n_t, 1.0))
        return cands

    # Phase 1: LSE on diverse candidates
    print("\n=== Phase 1: LSE optimization ===", flush=True)
    results = {}

    for n_t in lengths:
        if time.monotonic() > deadline - 100:
            break
        cands = get_cands(n_t)
        best_here = None
        best_here_sc = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 100:
                break
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            bud = min(100.0, deadline - time.monotonic() - 100.0)
            if bud < 12:
                break
            end = time.monotonic() + bud

            a_lse = _lse_optimize(a0, end, max_iter=25000,
                                  T_start=0.025, T_end=1e-7, lr=0.07)
            sc_lse = _score(a_lse)
            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_sc:
                best_here_sc = sc_lse
                best_here = a_lse

            print(f"  n={n_t} c{ci}: {sc0:.8f} -> {sc_lse:.10f}", flush=True)

        if best_here is not None:
            results[n_t] = best_here
            print(f"  Best n={n_t}: {best_here_sc:.10f}", flush=True)

    # Phase 2: LP refinement (full constraints for small n)
    print("\n=== Phase 2: LP refinement ===", flush=True)
    for n_t in sorted(results.keys(), key=lambda x: _score_norm(results[x])):
        if time.monotonic() > deadline - 80:
            break
        a_in = results[n_t]
        sc_in = _score(a_in)
        print(f"  LP n={n_t} start: {sc_in:.10f}", flush=True)

        bud = min(100.0, deadline - time.monotonic() - 80.0)
        if bud < 10:
            continue
        end = time.monotonic() + bud

        a_lp = _lp_refine(a_in, end, max_iter=2000,
                          inflate=1.0002, blend=0.1, top_k=300)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        results[n_t] = a_lp
        print(f"  LP n={n_t} end: {sc_lp:.10f}", flush=True)

    # Phase 3: Alternating LSE+LP on best
    print("\n=== Phase 3: Alternating refinement ===", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 40:
        alt_end = min(deadline - 40.0, time.monotonic() + 150.0)
        a_alt = _alternating_refine(best_seq, alt_end)
        sc_alt = _score(a_alt)
        if np.isfinite(sc_alt) and sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt.copy()
        print(f"  Alternating: {sc_alt:.10f}", flush=True)

    # Phase 4: Subgradient polish
    print("\n=== Phase 4: Subgradient polish ===", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 20:
        sp_end = min(deadline - 10.0, time.monotonic() + 60.0)
        a_sp = _subgrad_polish(best_seq, sp_end, max_iter=30000,
                               lr=0.01, momentum=0.9)
        sc_sp = _score(a_sp)
        if np.isfinite(sc_sp) and sc_sp < best_score:
            best_score = sc_sp
            best_seq = a_sp.copy()
        print(f"  Subgrad: {sc_sp:.10f}", flush=True)

    # Phase 5: Final fine LP
    if best_seq is not None and time.monotonic() < deadline - 5:
        lp_end = min(deadline, time.monotonic() + 30.0)
        a_fin = _lp_refine(best_seq, lp_end, max_iter=1000,
                           inflate=1.0001, blend=0.05, top_k=300)
        sc_fin = _score(a_fin)
        if np.isfinite(sc_fin) and sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
        print(f"  Final LP: {sc_fin:.10f}", flush=True)

    # Final LSE
    if best_seq is not None and time.monotonic() < deadline:
        lse_end = deadline
        a_last = _lse_optimize(best_seq, lse_end, max_iter=15000,
                               T_start=0.008, T_end=1e-8, lr=0.03)
        sc_last = _score(a_last)
        if np.isfinite(sc_last) and sc_last < best_score:
            best_score = sc_last
            best_seq = a_last.copy()
        print(f"  Final LSE: {sc_last:.10f}", flush=True)

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(512) / 512.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}", flush=True)
    return [float(x) for x in final_seq]
