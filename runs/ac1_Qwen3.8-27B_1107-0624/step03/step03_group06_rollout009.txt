import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_poly_ramp(n, order=1, side=1):
    """Create a polynomial ramp. side=1 means increasing, side=-1 means decreasing."""
    t = np.linspace(0.0, 1.0, n)
    if side == -1:
        t = 1.0 - t
    a = 1.0 + order * (t ** 1) 
    if order >= 2:
        a = t ** (order - 1) + 1e-8
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_sym_ramp(n, order=2):
    """Symmetric polynomial: high in middle, low at edges (or vice versa)."""
    t = np.linspace(0.0, 1.0, n)
    # High at edges, low in middle
    a = (t ** order + (1.0 - t) ** order) + 1e-8
    return a / a.sum()


def _make_plateau(n, width=0.6):
    """Plateau function: flat in the middle, tapering at edges."""
    t = np.linspace(0.0, 1.0, n)
    a = np.maximum(1.0 - np.abs(t - 0.5) / (width / 2.0), 0.0)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_cosine(n):
    """Cosine bell: smooth bump."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t - 1.0))) + 1e-8
    return a / a.sum()


def _make_gaussian(n, sigma=0.3):
    """Gaussian bump."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 1e-8
    return a / a.sum()


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)


def _solve_lp(f, M, n):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

        a_new = a_new / s

            best_t = t
    return best_t, best_sc

def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of convolution."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # clip to prevent overflow
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    """Project onto probability simplex (non-negative, sum=1)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

            else:
                print(f"  -> Gradient: {sc_gp:.8f}")

    # Phase 3: Perturbation + re-optimization (escape local minima)
    print(f"\nPhase 3: Perturbation restarts (score={best_score:.10f})")
    pert_end = t_start + budget_s * 0.80
    if best_seq is not None and time.monotonic() < pert_end:
        n_pert = min(5, int((pert_end - time.monotonic()) / 40))
        for i in range(n_pert):
            if time.monotonic() > pert_end:
                break
            print(f"  Perturbation {i+1}/{n_pert}...")
            a_p = _perturb_and_optimize(best_seq, rng, pert_end, perturb_size=0.03 + 0.02 * i)
            sc_p = _score(a_p)
            if np.isfinite(sc_p) and sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                print(f"    -> {sc_p:.10f} (NEW BEST)")
            else:
                print(f"    -> {sc_p:.8f}")

    # Phase 4: Final LP on the best (if length allows)
    print(f"\nPhase 4: Final LP (score={best_score:.10f})")
    if best_seq is not None and time.monotonic() < deadline:
        n_best = len(best_seq)
        if n_best <= 1024:
            lp_time = min(120.0, deadline - time.monotonic() - 20)
            if lp_time > 10:
                a_final = _lp_refine(best_seq, time.monotonic() + lp_time,
                                     max_iter=30000, inflate_start=1.0005, inflate_min=1.0001)
                sc_f = _score(a_final)
                if np.isfinite(sc_f) and sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_final.copy()
                    print(f"  -> Final LP: {sc_f:.10f} (NEW BEST)")
                else:
                    print(f"  -> Final LP: {sc_f:.8f}")

    # Phase 5: Final gradient polish
    print(f"\nPhase 5: Final gradient polish (score={best_score:.10f})")
    if best_seq is not None and time.monotonic() < deadline - 10:
        gp_time = min(80.0, deadline - time.monotonic() - 5)
        if gp_time > 5:
            a_g = _gradient_polish(best_seq, time.monotonic() + gp_time, max_iter=20000)
            sc_g = _score(a_g)
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
                print(f"  -> Final gradient: {sc_g:.10f} (NEW BEST)")
            else:
                print(f"  -> Final gradient: {sc_g:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_poly_ramp(512, 1, 1)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_poly_ramp(512, 1, 1)

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final_seq]
