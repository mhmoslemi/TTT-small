```python
def lp_minimize_peak(a, M_init=None):
    """Solve LP to minimize max of conv(a, b) subject to sum(b)=1, b>=0."""
    n = len(a)
    # Variables: [b_0, ..., b_{n-1}, t]
    # Objective: [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequality constraints: for each k, sum_j a_j * b_{k-j} - t <= 0
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        # conv(a,b)[k] = sum_j a[j] * b[k-j] for valid j
        # j ranges from max(0, k-(n-1)) to min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[j]
        A_ub[k, n] = -1.0  # -t
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = c - c.min()
    return c
```

```python
"""
Multi-resolution alternating LP optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses a tight LP formulation (minimize peak cross-convolution directly), coarse-to-fine
resolution strategy, adaptive blending, and gradient polishing to break local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _build_lp_matrix(a):
    n = len(a)
    num_rows = 2 * n - 1
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, n] = -1.0
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, k - j] = a[j]
    b_ub = np.zeros(num_rows)
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    return A_ub, b_ub, A_eq, b_eq


def _lp_minimize_peak(a, x0=None):
    n = len(a)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub, b_ub, A_eq, b_eq = _build_lp_matrix(a)
    bounds = [(0, None)] * (n + 1)
    kwargs = {'method': 'highs', 'options': {'presolve': True}}
    if x0 is not None and len(x0) == n + 1:
        kwargs['x0'] = x0
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, **kwargs)
    if res.success:
        b = np.maximum(res.x[:n], 0.0)
        sb = b.sum()
        if sb > 1e-12:
            b /= sb
        else:
            b = np.ones(n) / n
        return b, float(res.x[n])
    return None, None


def _alt_lp_optimize(a, deadline, max_iter=2000, blend_start=0.5, blend_min=0.05,
                     name=""):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    blend = blend_start
    no_imp = 0
    x0 = np.append(a, best_score / (2.0 * n))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        b, t_opt = _lp_minimize_peak(a, x0)

        if b is None:
            no_imp += 3
            if no_imp > 80:
                break
            continue

        a_new = (1.0 - blend) * a + blend * b
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            break
        a_new /= sn

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new
        x0 = np.append(a, t_opt)

        if no_imp > 15:
            blend = max(blend_min, blend * 0.92)
        if no_imp > 60:
            # Perturb to escape
            rng_p = np.random.default_rng(it * 7919 + n)
            a = best_a * (1.0 + 0.05 * rng_p.normal(size=n))
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_imp = 20
            blend = max(blend_min, blend * 1.3)

        if it % 100 == 0:
            elapsed = time.monotonic()
            print(f"    {name} iter {it}: score={best_score:.10f}, t_opt={t_opt:.8f}, blend={blend:.3f}")

    return best_a, best_score


def _gradient_optimize(a, deadline, max_iter=50000, lr=0.08, T_start=0.03, T_end=1e-5):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    z = np.log(a)
    z -= np.mean(z)

    best_a = a.copy()
    best_score = _score_norm(a)

    state_t = 0
    state_m = np.zeros(n)
    state_v = np.zeros(n)
    b1, b2 = 0.9, 0.999

    T = T_start
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        else:
            w = ex / s_ex

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        z_grad = a * (grad - dot)
        z_grad -= np.mean(z_grad)

        if not np.all(np.isfinite(z_grad)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-15))
            z -= np.mean(z)
            state_t = 0
            state_m = np.zeros(n)
            state_v = np.zeros(n)
            continue

        state_t += 1
        state_m = b1 * state_m + (1 - b1) * z_grad
        state_v = b2 * state_v + (1 - b2) * (z_grad * z_grad)
        mh = state_m / (1.0 - b1 ** state_t)
        vh = state_v / (1.0 - b2 ** state_t)

        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -35, 35)
        z -= np.mean(z)

        a_new = np.exp(z)
        a_new /= a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        T *= T_ratio
        T = max(T, T_end)

    return best_a, best_score


def _upsample(a, factor):
    n = len(a)
    new_n = n * factor
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-30:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def _interpolate_to(a, new_n):
    n = len(a)
    if n == new_n:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(new_n) / new_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, new_n)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    return p / s


def _make_gaussian(n, sigma=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_edge_peaked(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, center_sep=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5 + center_sep) / 0.15) ** 2) + \
        np.exp(-0.5 * ((t - 0.5 - center_sep) / 0.15) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear(n, eta=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    a = np.maximum(a, 0.01)
    return a / a.sum()
```
