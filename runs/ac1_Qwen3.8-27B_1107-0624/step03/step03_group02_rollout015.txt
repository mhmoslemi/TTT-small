import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

Key innovation: exact coordinate descent that solves each 1D subproblem optimally

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_norm(a):
    n = len(a)
    if n == 0:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())

        ex[int(np.argmax(c))] = 1.0

            idx = np.arange(j1_start, j1_end + 1)
            sg[idx] += a[max_idx - idx]
        j2_start = max(max_idx, 0)
        j2_end = min(max_idx + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            sg[idx] += a[idx - max_idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        gap = f_curr - lb
        if gap < 1e-8:
            break
        alpha = gap / (sg_norm * sg_norm)
        alpha = min(alpha, 1.0 / n)

        a_new = a - alpha * sg
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

        a = a_new

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _full_pipeline(a0, deadline, label=""):
    sc0 = _score_norm(a0)
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.10f}")

    best_a = a0.copy()
    best_sc = sc0
    n = len(a0)

    # Phase A: Adam
    if time.monotonic() < deadline - 60:
        d1 = min(deadline - 60, time.monotonic() + 200)
        a1, s1 = _adam_smooth_max(a0, d1, max_iter=20000, lr=0.025, beta_factor=0.04)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam1: {best_sc:.10f}")

    # Phase B: Coordinate Descent
    if time.monotonic() < deadline - 40:
        d2 = min(deadline - 40, time.monotonic() + 150)
        a2, s2 = _coord_descent(best_a, d2, max_sweeps=100)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} CD1: {best_sc:.10f}")

    # Phase C: Adam fine
    if time.monotonic() < deadline - 30:
        d3 = min(deadline - 30, time.monotonic() + 120)
        a3, s3 = _adam_smooth_max(best_a, d3, max_iter=15000, lr=0.01, beta_factor=0.015)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} Adam2: {best_sc:.10f}")

    # Phase D: Coordinate Descent again
    if time.monotonic() < deadline - 20:
        d4 = min(deadline - 20, time.monotonic() + 100)
        a4, s4 = _coord_descent(best_a, d4, max_sweeps=100)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} CD2: {best_sc:.10f}")

    # Phase E: Subgradient Polyak
    if time.monotonic() < deadline - 10:
        d5 = min(deadline - 10, time.monotonic() + 60)
        a5, s5 = _subgradient_polyak(best_a, d5, max_iter=5000)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} Subgrad: {best_sc:.10f}")

    # Phase F: Final CD polish
    if time.monotonic() < deadline - 5:
        d6 = deadline - 5
        a6, s6 = _coord_descent(best_a, d6, max_sweeps=50)
        if s6 < best_sc:
            best_a, best_sc = a6, s6
        print(f"  {label} CD3: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_norm(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is not None:
        p512 = _interpolate_profile(parent, 512)
        p1024 = _interpolate_profile(parent, 1024)
    else:
        p512 = _make_linear_profile(512, 1.5)
        p1024 = _make_linear_profile(1024, 1.5)

    # === Strategy 1: Full pipeline on parent @ 1024 ===
    print("=== Strategy 1: Parent @ n=1024 ===")
    d1 = min(deadline - 600, t_start + 600)
    a_b1, s_b1 = _full_pipeline(p1024, d1, "S1")
    if s_b1 < best_score:
        best_score = s_b1
        best_seq = a_b1.copy()
    print(f"  After S1: best={best_score:.10f}")

    # === Strategy 2: Full pipeline on parent @ 512 ===
    if time.monotonic() < deadline - 400:
        print("=== Strategy 2: Parent @ n=512 ===")
        d2 = min(deadline - 400, time.monotonic() + 400)
        a_b2, s_b2 = _full_pipeline(p512, d2, "S2")
        if s_b2 < best_score:
            best_score = s_b2
            best_seq = a_b2.copy()
        print(f"  After S2: best={best_score:.10f}")

    # === Strategy 3: Diverse initializations at n=1024 ===
    if time.monotonic() < deadline - 200:
        print("=== Strategy 3: Diverse inits @ n=1024 ===")
        n = 1024
        candidates = []
        # Perturbed parent
        p_pert = p1024 * (1.0 + 0.1 * rng.normal(size=n))
        p_pert = np.maximum(p_pert, 0.0)
        candidates.append(p_pert / p_pert.sum())

        # Anti-symmetric perturbation (exploit structure)
        p_anti = p1024.copy()
        p_anti[:n//2] *= 1.2
        p_anti[n//2:] *= 0.85
        p_anti = np.maximum(p_anti, 0.0)
        candidates.append(p_anti / p_anti.sum())

        # Center-weighted
        t = np.linspace(0, 1, n)
        p_center = 1.0 + 0.5 * np.cos(np.pi * t)
        p_center = np.maximum(p_center, 0.01)
        candidates.append(p_center / p_center.sum())

        # Edge-suppressed
        p_edge = p1024.copy()
        edge_decay = np.linspace(0.3, 1.0, n)
        edge_decay = np.minimum(edge_decay, 1.0 / edge_decay)
        p_edge = p1024 * edge_decay
        p_edge = np.maximum(p_edge, 0.0)
        candidates.append(p_edge / p_edge.sum())

        for i, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 200:
                break
            per_budget = min(50.0, (deadline - 200 - time.monotonic()) / max(1, len(candidates) - i))
            if per_budget < 15:
                break
            d3i = min(deadline - 200 + i * 10, time.monotonic() + per_budget)
            a_q, s_q = _adam_smooth_max(a0, d3i, max_iter=8000, lr=0.02, beta_factor=0.04)
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
            # Quick CD on the result
            if time.monotonic() < deadline - 200:
                d_cd = min(deadline - 200, time.monotonic() + 20)
                a_cd, s_cd = _coord_descent(a_q, d_cd, max_sweeps=30, verbose=False)
                if s_cd < best_score:
                    best_score = s_cd
                    best_seq = a_cd.copy()
            print(f"  S3-{i}: {best_score:.10f}")

    # === Strategy 4: Focused final optimization ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"=== Strategy 4: Final focus (current best={best_score:.10f}) ===")
        best_n = best_seq.size

        # Coordinate descent final push
        d4a = min(deadline - 15, time.monotonic() + 120)
        a_f1, s_f1 = _coord_descent(best_seq, d4a, max_sweeps=100)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  Final CD: {best_score:.10f}")

        # Tiny Adam
        if time.monotonic() < deadline - 10:
            d4b = min(deadline - 10, time.monotonic() + 60)
            a_f2, s_f2 = _adam_smooth_max(best_seq, d4b, max_iter=8000, lr=0.005, beta_factor=0.008)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  Final Adam: {best_score:.10f}")

        # Another CD
        if time.monotonic() < deadline - 5:
            d4c = deadline - 5
            a_f3, s_f3 = _coord_descent(best_seq, d4c, max_sweeps=50)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  Final CD2: {best_score:.10f}")

    # === Strategy 5: Try slightly different length ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        best_n = best_seq.size
        if best_n == 1024:
            alt_n = 1536
        elif best_n == 512:
            alt_n = 768
        else:
            alt_n = best_n

        if alt_n != best_n:
            print(f"=== Strategy 5: Try n={alt_n} ===")
            p_alt = _interpolate_profile(best_seq, alt_n)
            d5 = min(deadline - 10, time.monotonic() + 80)
            a_alt, s_alt = _wander_optimize(p_alt, d5)
            if s_alt < best_score:
                best_score = s_alt
                best_seq = a_alt.copy()
            print(f"  After S5: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score_norm(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print("WARNING: evaluate_sequence returned inf, using fallback")
                final_seq = _make_linear_profile(256, 1.5)
                best_score = _score_norm(final_seq)
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]


def _wander_optimize(a0, deadline):
    """Quick optimization: Adam + CD."""
    a = np.asarray(a0, dtype=np.float64).copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    d1 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.5)
    a1, s1 = _adam_smooth_max(a, d1, max_iter=10000, lr=0.02, beta_factor=0.03)
    if s1 < best_sc:
        best_a, best_sc = a1, s1

    if time.monotonic() < deadline:
        a2, s2 = _coord_descent(best_a, deadline, max_sweeps=50, verbose=False)
        if s2 < best_sc:
            best_a, best_sc = a2, s2

    return best_a, best_sc
