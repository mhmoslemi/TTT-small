"""
Multi-phase optimization for step function sequence minimization.
Uses LP-based peak spreading, smooth-max Adam in log-space, and a
coarsen-then-refine strategy with simulated annealing restarts.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:

    return p / s


def _smooth_max_grad(a, beta):
    """Compute gradient of soft-max of conv(a,a) w.r.t. a. Returns (grad, soft_max_value)."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())

    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    # Gradient of sum_k w[k] * c[k] where c = conv(a, a)
    # d(c[k])/d(a[i]) = a[k-i] + a[i-k] (with valid indices)
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # This is 2 * conv(w, a_rev)[i] where a_rev = a reversed, padded appropriately
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    smooth_val = mval + beta * np.log(s) if s > 0 else mval
    return grad, float(smooth_val)

        grad_z = a * (grad - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * grad_z**2
        mh = m / (1.0 - b1**t)
        vh = v / (1.0 - b2**t)

        z_new = np.clip(z_new, -80.0, 80.0)

            best_z = z_new.copy()
            no_imp = 0

                m = np.zeros(n)

def _lp_spread(a, M, max_iter_lp=50000):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)[k] <= M for all k."""
    n = len(a)
    c_obj = -np.zeros(n)
    c_obj[:] = -1.0

    # Build constraint matrix: (a*g)[k] = sum_j a[j]*g[k-j] for valid j
    # Row k has non-zeros at positions j where 0<=j<n and 0<=k-j<n
    # i.e., j in [max(0, k-n+1), min(k, n-1)]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])

    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)

    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  options={'maxiter': max_iter_lp, 'presolve': True},
                  method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_outer=200, inflate=1.0005, blend=0.4, verbose=False, tag=""):
    """Iterative LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_outer):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        g = _lp_spread(a, M_lp)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-9:
            no_imp += 1
            if no_imp > 30:
                inflate_cur = min(1.01, inflate_cur * 1.1)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend_cur = max(0.1, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(50, n // 10):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

        if verbose and it % 20 == 0:
            print(f"    {tag} LP it={it}: score={best_score:.10f}")

    return best_a, best_score


def _peak_targeted_opt(a, deadline, max_iter=10000, verbose=False, tag=""):
    """Target peaks directly: find top peaks, construct gradient from them, use accelerated method."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score(a)

    # Momentum-based accelerated gradient
    y = a.copy()
    alpha = 1.0 / 4.0
    base_lr = 0.5 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(y)
        max_val = float(c.max())
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]

        # Weighted subgradient from top peaks
        weights = np.exp(-0.5 * np.arange(top_k).astype(np.float64))
        weights /= weights.sum()

        grad = np.zeros(n)
        for w_i, idx in zip(weights, top_idx):
            idx = int(idx)
            # dc[idx]/da[i] = a[idx-i] (if valid) + a[i-idx] (if valid)
            for i in range(n):
                v = 0.0
                if 0 <= idx - i < n:
                    v += y[idx - i]
                if 0 <= i - idx < n:
                    v += y[i - idx]
                grad[i] += w_i * v

        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        x_new = y - lr * grad / grad_norm
        x_new = np.maximum(x_new, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-10:
            break
        x_new = x_new / s_new

        t_new = (it + 1) / (it + 1 + alpha)
        y = t_new * x_new + (1.0 - t_new) * y

        sc = _score(x_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = x_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break

    return best_a, best_score


def _coarsen_refine(a, deadline, coarse_n=64, verbose=False, tag=""):
    """Optimize at coarse resolution, then upsample and refine."""
    n = len(a)
    a_coarse = _interpolate_profile(a, coarse_n)
    sc_coarse = _score(a_coarse)
    if verbose:
        print(f"    {tag} coarse n={coarse_n}: {sc_coarse:.10f}")

    # Optimize coarse
    deadline_coarse = min(deadline - 60, time.monotonic() + min(120, (deadline - time.monotonic()) * 0.3))
    a_c_opt, sc_c = _adam_logspace(a_coarse, deadline_coarse, max_iter=10000,
                                   lr=0.05, beta_init=0.03, beta_decay=0.95, tag=tag+"c")
    # Upsample back
    a_up = _interpolate_profile(a_c_opt, n)
    sc_up = _score(a_up)
    if verbose:
        print(f"    {tag} after upsample n={n}: {sc_up:.10f}")

    # Refine at full resolution
    deadline_fine = deadline - 10
    a_fine, sc_fine = _adam_logspace(a_up, deadline_fine, max_iter=20000,
                                     lr=0.02, beta_init=0.01, beta_decay=0.96, tag=tag+"f")
    if sc_fine < sc_up:
        return a_fine, sc_fine
    return a_up, sc_up


def _random_perturb_search(a, deadline, max_rounds=50, tag=""):
    """Simulated annealing style: perturb, locally optimize, accept if better."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(12345)
    temp = 1.0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Perturb
        scale = 0.05 + 0.1 * temp
        a_pert = a + scale * rng.normal(size=n)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        # Quick local optimization (few Adam steps)
        dl = min(deadline - 5, time.monotonic() + 20)
        a_opt, sc_opt = _adam_logspace(a_pert, dl, max_iter=2000,
                                       lr=0.03, beta_init=0.02, beta_decay=0.9)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            temp *= 0.95
        else:
            # Metropolis acceptance
            delta = sc_opt - best_score
            if delta < 0 or (delta < 0.1 and rng.random() < np.exp(-delta / (0.01 + temp * 0.005))):
                a = a_opt.copy()
            temp *= 0.97

    return best_a, best_score


def _symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i])/2."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    return _normalize(a_sym)


def _make_candidate(n, profile_type=0, rng=None):
    if rng is None:
        rng = np.random.default_rng(n)
    t = np.linspace(0.0, 1.0, n)
    if profile_type == 0:
        a = np.ones(n)
    elif profile_type == 1:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t)
    elif profile_type == 2:
        a = np.exp(-2.0 * (t - 0.5)**2)
    elif profile_type == 3:
        a = 1.0 + 0.3 * np.sin(4 * np.pi * t) + rng.normal(size=n) * 0.05
    elif profile_type == 4:
        a = 1.0 / (1.0 + 3.0 * (t - 0.5)**2)
    else:
        a = 1.0 + 0.2 * rng.normal(size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: starts from parent, uses coarsen-refine, LP spreading,
    smooth-max Adam, and peak-targeted methods. Explores multiple lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: n={parent.size}, score={sc:.10f}")
    except Exception as e:
        print(f"No parent available: {e}")

    n_parent = parent.size if parent is not None else 1024

    # If no parent, create one
    if parent is None:
        parent = _make_candidate(1024, 0, rng)
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"Created parent: n={1024}, score={best_score:.10f}")

    # Phase 1: Coarse-to-fine at parent's resolution
    print("\n=== Phase 1: Coarse-to-fine at parent resolution ===")
    if time.monotonic() < deadline - 200:
        d1 = min(deadline - 200, time.monotonic() + 250)
        try:
            a1, s1 = _coarsen_refine(parent, d1, coarse_n=64, verbose=True, tag="P1")
            if s1 < best_score:
                best_score = s1
                best_seq = a1.copy()
            print(f"  Phase 1 result: {best_score:.10f}")
        except Exception as e:
            print(f"  Phase 1 error: {e}")

    # Phase 2: LP-based refinement
    print("\n=== Phase 2: LP refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        d2 = min(deadline - 150, time.monotonic() + 200)
        try:
            a2, s2 = _lp_refine(best_seq, d2, max_outer=100, inflate=1.001,
                                blend=0.35, verbose=True, tag="P2")
            if s2 < best_score:
                best_score = s2
                best_seq = a2.copy()
            print(f"  Phase 2 result: {best_score:.10f}")
        except Exception as e:
            print(f"  Phase 2 error: {e}")

    # Phase 3: Try different sequence lengths
    print("\n=== Phase 3: Multi-length exploration ===")
    lengths_to_try = [256, 512, 1024, 2048]
    # Remove parent's length from the front (already tried)
    lengths_to_try = [l for l in lengths_to_try if l != n_parent] + [n_parent]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 100:
            break
        if time.monotonic() < deadline - 80:
            a_init = _interpolate_profile(best_seq, n_try)
            d3i = min(deadline - 80, time.monotonic() + 100)
            print(f"  Trying n={n_try}...")
            try:
                a3i, s3i = _adam_logspace(a_init, d3i, max_iter=15000,
                                          lr=0.03, beta_init=0.02,
                                          beta_decay=0.96, tag=f"P3-{n_try}")
                if s3i < best_score:
                    best_score = s3i
                    best_seq = a3i.copy()
                    print(f"    n={n_try}: improved to {best_score:.10f}")
                else:
                    print(f"    n={n_try}: {s3i:.10f} (no improvement)")
            except Exception as e:
                print(f"    n={n_try} error: {e}")

    # Phase 4: Peak-targeted optimization on best
    print("\n=== Phase 4: Peak-targeted optimization ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        d4 = min(deadline - 60, time.monotonic() + 80)
        try:
            a4, s4 = _peak_targeted_opt(best_seq, d4, max_iter=5000, tag="P4")
            if s4 < best_score:
                best_score = s4
                best_seq = a4.copy()
            print(f"  Phase 4 result: {best_score:.10f}")
        except Exception as e:
            print(f"  Phase 4 error: {e}")

    # Phase 5: Symmetrization exploitation
    print("\n=== Phase 5: Symmetrization ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_sym = _symmetrize(best_seq)
        sc_sym = _score(a_sym)
        if sc_sym < best_score:
            print(f"  Symmetrization helps: {sc_sym:.10f}")
            best_score = sc_sym
            best_seq = a_sym.copy()
            # Further optimize the symmetric version
            d5 = min(deadline - 30, time.monotonic() + 30)
            try:
                a5, s5 = _adam_logspace(best_seq, d5, max_iter=5000,
                                        lr=0.01, beta_init=0.005, beta_decay=0.95, tag="P5")
                if s5 < best_score:
                    best_score = s5
                    best_seq = a5.copy()
            except Exception:
                pass
        else:
            print(f"  Symmetrization doesn't help ({sc_sym:.10f} vs {best_score:.10f})")

    # Phase 6: Final Adam polish
    print("\n=== Phase 6: Final Adam polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        d6 = deadline - 5
        try:
            a6, s6 = _adam_logspace(best_seq, d6, max_iter=20000,
                                    lr=0.005, beta_init=0.005, beta_decay=0.97, tag="P6")
            if s6 < best_score:
                best_score = s6
                best_seq = a6.copy()
            print(f"  Phase 6 result: {best_score:.10f}")
        except Exception as e:
            print(f"  Phase 6 error: {e}")

    # Phase 7: Final LP polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        try:
            a7, s7 = _lp_refine(best_seq, deadline - 2, max_outer=50,
                                inflate=1.0003, blend=0.2, tag="P7")
            if s7 < best_score:
                best_score = s7
                best_seq = a7.copy()
            print(f"  Phase 7 result: {best_score:.10f}")
        except Exception as e:
            print(f"  Phase 7 error: {e}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = 2.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
