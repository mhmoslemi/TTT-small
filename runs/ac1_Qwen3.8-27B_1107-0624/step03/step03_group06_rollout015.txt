import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

    a = t ** alpha + 1e-8
    return a / a.sum()

    # Smooth ramp: high at one end, low at the other, with smooth transition
    t = np.linspace(0.0, 1.0, n)

Step function optimization combining LP-based alternating updates with 
fine-grained smooth-max gradient polishing. Key improvements over previous:
- Adaptive beta for smooth max (geometric decay)
- Preconditioned gradient updates
- Targeted LP with active-set acceleration
- Multiple length exploration with parent interpolation
- Random restarts with structured perturbations

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_grad_direct(a, w):
    """Compute gradient of sum_k w_k * conv(a,a)_k w.r.t. a, using direct method."""
    n = len(a)

    # Efficient computation:
    # T1_i = sum_k w_k * a_{k-i} = sum_j w_{j+i} * a_j (letting j = k-i)
    # T2_i = sum_k w_k * a_{i-k} = sum_j w_{i-j} * a_j (letting j = i-k) = conv(w,a)_i
    # So grad = T1 + T2 where:
    # T1[i] = sum_j a[j] * w[j+i]  (cross-correlation of a and w at lag i)
    # T2[i] = conv(w, a)[i]        (standard convolution)
    
    N_pad = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N_pad)
    fw = np.fft.rfft(w, N_pad)
    
    # T2 = conv(w, a)
    T2 = np.fft.irfft(fw * fa, N_pad)[:n]

    # = ifft(FFT(a) * conj(FFT(w)))[N-i] ... 
    # Actually: ifft(FFT(a) * conj(FFT(w)))[m] = sum_j a[j] * w[(j-m) mod N]
    # We want: sum_j a[j] * w[j+i] = ifft(FFT(a) * conj(FFT(w)))[(N-i) mod N]
    corr = np.fft.irfft(fa * np.conj(fw), N_pad)
    # corr[m] = sum_j a[j] * w[(j-m) mod N_pad]
    # T1[i] = corr[(N_pad - i) % N_pad] for i = 0, ..., n-1
    T1 = np.array([corr[(N_pad - i) % N_pad] for i in range(n)])
    
    return T1 + T2


def _smooth_max_grad(a, beta):
    """Compute gradient of smooth_max(conv(a,a)) using softmax weighting."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = np.clip((c - mval) / beta, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    grad = _conv_grad_direct(a, w)
    return grad, mval, c


def _project_nonneg_sum1(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t**alpha + 1e-10
    return a / a.sum()

        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_refine(f, M_target, n, inflation=1.001):
    """Solve LP: maximize sum(b) s.t. conv(f,b) <= M_target, b >= 0."""
    # Build constraint matrix: conv(f,b)_k = sum_j f[(k-j) mod ?] * b_j
    # For k = 0, ..., 2n-2: conv(f,b)_k = sum_{j} f[j] * b[k-j] (valid indices)
    # This is a Toeplitz-like matrix.
    
    # A[i, j] = f[(i-j) mod n] if 0 <= i-j <= n-1 else 0
    # Actually: conv(f,b)_i = sum_j f[j] * b[i-j] where both indices must be in [0, n-1]
    # So A[i, j] = f[j] if 0 <= i-j <= n-1, i.e., i-n+1 <= j <= i and 0 <= j <= n-1
    
    # For efficiency with large n, we won't build the full matrix explicitly.
    # Instead, use the structure: each row i has nonzeros in columns max(0, i-n+1) to min(n-1, i).
    
    # For n up to ~1000, the matrix is (2n-1) x n which is manageable.
    if n > 2000:
        return None
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_iteration(a, inflation=1.001):
    """One LP iteration: compute M, solve LP, return new b if improvement."""
    n = len(a)
    c = _fft_convolve(a)
    M = float(c.max())
    M_lp = M * inflation
    
    b = _lp_refine(a, M_lp, n)
    if b is None:
        return None, M
    
    sum_b = float(b.sum())
    if sum_b <= 1.0 + 1e-8:
        return None, M
    
    b_norm = b / sum_b
    return b_norm, M


def _gradient_descent(a0, deadline, max_iter=50000):
    """Smooth-max gradient descent with adaptive step size and preconditioning."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = len(a)
    a = a / a.sum()
    
    c = _fft_convolve(a)
    c_max = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    
    beta = 0.01 * c_max
    min_beta = 1e-8
    lr = 0.05
    base_lr = lr
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0
    imp_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Adaptive beta: decrease slowly
        beta = max(min_beta, beta * 0.999)
        
        # Compute gradient
        a_mom = a + momentum * v
        a_mom = np.maximum(a_mom, 0)
        sm = a_mom.sum()
        if sm > 1e-10:
            a_mom = a_mom / sm
        else:
            a_mom = a
        
        grad, mval, c = _smooth_max_grad(a_mom, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(min_beta, beta * 5.0)
            v = np.zeros(n)
            continue
        
        # Preconditioning: scale by 1/sqrt(a_i + epsilon) to handle scale issues
        # Actually, let's use a simpler approach: scale the gradient by the max
        grad_max = float(np.abs(grad).max())
        if grad_max < 1e-30:
            break
        grad = grad / grad_max
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = _project_nonneg_sum1(a_new)
        
        # Evaluate
        c_new = _fft_convolve(a_new)
        c_max_new = float(c_new.max())
        score_new = float(2.0 * n * c_max_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            imp_count += 1
        else:
            no_imp += 1
            if no_imp > 500:
                # Reset
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                beta = max(0.005 * c_max, min_beta)
                no_imp = 0
        
        # Adaptive learning rate
        if no_imp > 100:
            lr = max(0.002, lr * 0.95)
        elif imp_count > 0 and no_imp == 0:
            lr = min(0.1, lr * 1.02)
    
    return best_a


def _combined_optimize(a0, deadline, lp_time_frac=0.6):
    """Combine LP and gradient methods."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Phase 1: LP iterations
    lp_deadline = time.monotonic() + (deadline - time.monotonic()) * lp_time_frac
    no_lp_imp = 0
    inflation = 1.002
    
    while time.monotonic() < lp_deadline:
        b, M = _lp_iteration(a, inflation)
        if b is None:
            no_lp_imp += 1
            if no_lp_imp % 20 == 0:
                inflation = min(0.05, inflation * 1.2)
            if no_lp_imp > 200:
                break
            continue
        
        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in np.array([0.1, 0.3, 0.5, 0.7, 1.0]):
            a_try = (1 - t) * a + t * b
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t > 0 and best_sc < best_score - 1e-12:
            a = (1 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = best_sc
            best_a = a.copy()
            no_lp_imp = 0
            inflation = max(1.0005, inflation * 0.97)
        else:
            no_lp_imp += 1
    
    # Phase 2: Gradient polish
    grad_deadline = deadline - 5.0
    if time.monotonic() < grad_deadline:
        a_grad = _gradient_descent(a, grad_deadline, max_iter=30000)
        sc_grad = _score(a_grad)
        if sc_grad < best_score:
            best_a = a_grad
            best_score = sc_grad
    
    return best_a, best_score
