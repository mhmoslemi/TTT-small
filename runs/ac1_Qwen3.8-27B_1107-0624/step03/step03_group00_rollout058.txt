"""
Optimization of non-negative step function heights to minimize
2*n*max(conv(a,a))/sum(a)^2 using Adam in logit space with 
smooth-max, L-BFGS polishing, peak-peeling heuristics, and 
multi-scale escalation. Combines gradient-based methods with 
combinatorial refinement.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_grad(c, beta):
    """Returns (smooth_max_value, gradient_weights)"""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _cross_corr_weighted(w, a):
    """
    Compute the gradient of <w, conv(a,a)> w.r.t. a.
    w: weights in the convolution domain (length 2n-1)
    a: sequence (length n)
    Returns: gradient vector of length n
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _score_from_conv(c, n):
    return float(2.0 * n * c.max())


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return _score_from_conv(c, n)


def _lp_solve(a, M):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    p0: initial probability vector (sums to 1)
    Returns: (best_p, best_score)
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    if noise_scales is None:

    noise_idx = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_grad(c, beta)
        g = _cross_corr_weighted(w, p)

        gz = p * (g - float(np.dot(p, g)))
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

            m.fill(0.0)

        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)

        score = float(2.0 * n * maxc)

                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:

                if no_imp % 500 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        # Adaptive beta
        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)

        # Check if multiple peaks are close
        if t_adam % 100 == 0 and c.size >= 2:
            sidx = np.argsort(c)[::-1]
            gap = c[sidx[0]] - c[sidx[1]]
            if gap < beta * 0.05:
                beta = max(1e-5, beta * 0.7)

        if no_imp > no_imp_limit:
            ns = noise_scales[min(noise_idx, len(noise_scales) - 1)]
            noise_idx += 1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lbfgs_polish(p0, deadline, beta=1e-3, maxiter=200):
    """
    L-BFGS polishing of the solution in logit space.
    Uses smooth-max with fixed small beta.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z0 = np.log(p0)
    z0 -= float(np.mean(z0))

    def objective_and_grad(z):
        if not np.all(np.isfinite(z)):
            return 1e10, np.zeros(n)
        z_clipped = np.clip(z, -60.0, 60.0)
        z_clipped = z_clipped - float(np.max(z_clipped))
        e = np.exp(z_clipped)
        p = e / e.sum()
        c = _full_convolve(p)
        val, w = _smooth_max_and_grad(c, beta)
        score = 2.0 * n * val
        g = _cross_corr_weighted(w, p)
        gz = p * (g - float(np.dot(p, g)))
        gz -= float(np.mean(gz))
        return score, gz

    # Wrap for scipy
    def fun(z):
        val, _ = objective_and_grad(z)
        return val

    def jac(z):
        _, g = objective_and_grad(z)
        return g

    # Check time
    t0 = time.monotonic()
    if time.monotonic() - t0 > deadline - t0 - 5:
        return p0, float(2.0 * n * _full_convolve(p0).max())

    best_p = p0.copy()
    best_score = float(2.0 * n * _full_convolve(p0).max())

    try:
        res = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': maxiter * 5,
                                'gtol': 1e-8, 'ftol': 1e-10})
        if res.success or res.fun < best_score:
            z_res = np.clip(res.x, -60.0, 60.0)
            z_res = z_res - float(np.max(z_res))
            e = np.exp(z_res)
            p_res = e / e.sum()
            sc_res = float(2.0 * n * _full_convolve(p_res).max())
            if sc_res < best_score:
                best_p = p_res
                best_score = sc_res
    except Exception as ex:
        print(f"  L-BFGS error: {ex}")

    return best_p, best_score


def _lp_refine(a, deadline, t=0.3, max_solves=20):
    """LP-based refinement."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, t * 0.85)
    return current_best, current_score


def _peak_peeling(a, deadline, max_iters=100):
    """
    Peak-peeling heuristic: iteratively reduce entries that contribute most to the peak.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for itr in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]
        
        # Find which a[m] contributes most to c[k_star]
        # c[k_star] = sum_j a[j] * a[k_star - j]
        # Contribution of a[m]: a[m] * a[k_star - m]
        
        # Valid range: j in [max(0, k_star-n+1), min(n-1, k_star)]
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        
        if j_start > j_end:
            break
            
        j_range = np.arange(j_start, j_end + 1)
        contributions = a[j_range] * a[k_star - j_range]
        
        # Find the entry with maximum contribution
        m_idx = j_range[int(np.argmax(contributions))]
        
        # Reduce this entry slightly
        reduction = 0.01 * a[m_idx]
        a[m_idx] -= reduction
        a[m_idx] = max(a[m_idx], 0)
        
        # Redistribute to entries that don't contribute to the peak
        # Find entries far from the peak's contributing range
        a_sum = a.sum()
        if a_sum < 1e-10:
            break
        a = a / a_sum
        
        sc = _score_norm(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def _make_profile_exponential(n, lam):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_flat_top(n):
    """Flat-top window (raised cosine to the 4th power)"""
    t = np.linspace(0.0, 1.0, n)
    a = 0.21557895 + 0.41663158 * np.cos(2 * np.pi * t) + \
        0.277263158 * np.cos(4 * np.pi * t) + \
        0.083578947 * np.cos(6 * np.pi * t) + \
        0.006947368 * np.cos(8 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_kaiser(n, alpha):
    """Kaiser window"""
    from scipy.special import i0
    t = np.linspace(-1.0, 1.0, n)
    a = i0(alpha * np.sqrt(np.maximum(1 - t**2, 0))) / i0(alpha)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_symmetric_two_bump(n, c, s):
    """Symmetric two-bump profile"""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c) / s) ** 2) + np.exp(-0.5 * ((t - (1-c)) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 0:
            return p0 / s
        return None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

multi-restart. Key novelty: L-BFGS second-order polishing phases that
exploit the smoothness of the log-sum-exp approximation near the optimum.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
from scipy.special import i0
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_grad(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _cross_corr_weighted(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        ctx = np.ones_like(e) / float(e.size)
    else:
        ctx = p / s
    return ctx


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _sort_candidate(x):
    return x[0]


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=20):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, t * 0.85)
    return current_best, current_score


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                   no_imp_limit=500, noise_scales=None):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    if noise_scales is None:
        noise_scales = [0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5]

    noise_idx = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_grad(c, beta)
        g = _cross_corr_weighted(w, p)

        gz = p * (g - float(np.dot(p, g)))
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(5e-5, beta * beta_decay)

        if t_adam % 100 == 0 and c.size >= 2:
            sidx = np.argsort(c)[::-1]
            gap = c[sidx[0]] - c[sidx[1]]
            if gap < beta * 0.05:
                beta = max(5e-5, beta * 0.7)

        if no_imp > no_imp_limit:
            ns = noise_scales[min(noise_idx, len(noise_scales) - 1)]
            noise_idx += 1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lbfgs_objective(z, beta, n):
    z_clipped = np.clip(z, -60.0, 60.0)
    z_clipped = z_clipped - float(np.max(z_clipped))
    e = np.exp(z_clipped)
    p = e / e.sum()
    c = _full_convolve(p)
    val, w = _smooth_max_and_grad(c, beta)
    score = 2.0 * n * val
    g = _cross_corr_weighted(w, p)
    gz = p * (g - float(np.dot(p, g)))
    gz -= float(np.mean(gz))
    return score, gz


def _lbfgs_polish(p0, deadline, beta=2e-3, maxiter=150):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z0 = np.log(p0)
    z0 -= float(np.mean(z0))

    best_p = p0.copy()
    best_score = float(2.0 * n * _full_convolve(p0).max())

    def _fun(zv):
        val, _ = _lbfgs_objective(zv, beta, n)
        return val

    def _jac(zv):
        _, g = _lbfgs_objective(zv, beta, n)
        return g

    try:
        res = minimize(_fun, z0, jac=_jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': maxiter * 3,
                                'gtol': 1e-9, 'ftol': 1e-12})
        sc_res, _ = _lbfgs_objective(res.x, beta, n)
        z_res = np.clip(res.x, -60.0, 60.0)
        z_res = z_res - float(np.max(z_res))
        e = np.exp(z_res)
        p_res = e / e.sum()
        exact_sc = _score_norm(p_res)
        if exact_sc < best_score:
            best_p = p_res
            best_score = exact_sc
    except Exception as ex:
        print(f"  L-BFGS error: {ex}")

    return best_p, best_score


def _peak_peeling(a, deadline, max_iters=80):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, _score_norm(a)
    a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)

    for itr in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        if j_start > j_end:
            break

        j_range = np.arange(j_start, j_end + 1)
        contributions = a[j_range] * a[k_star - j_range]
        total_contrib = contributions.sum()
        if total_contrib < 1e-15:
            break

        m_idx = j_range[int(np.argmax(contributions))]
        frac = contributions.max() / total_contrib

        reduction = min(0.02, 0.1 * frac) * a[m_idx]
        a[m_idx] = max(a[m_idx] - reduction, 0.0)

        a_sum = a.sum()
        if a_sum < 1e-10:
            break
        a = a / a_sum

        sc = _score_norm(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_flat_top(n):
    t = np.linspace(0.0, 1.0, n)
    a = (0.21557895 + 0.41663158 * np.cos(2 * np.pi * t)
         + 0.277263158 * np.cos(4 * np.pi * t)
         + 0.083578947 * np.cos(6 * np.pi * t)
         + 0.006947368 * np.cos(8 * np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_kaiser(n, alpha):
    t = np.linspace(-1.0, 1.0, n)
    a = i0(alpha * np.sqrt(np.maximum(1 - t ** 2, 0))) / i0(alpha)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_symmetric_two_bump(n, c, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c) / s) ** 2) + np.exp(-0.5 * ((t - (1 - c)) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 0:
            return p0 / s
        return None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    # --- Phase 7: Multi-restart (1450s-1700s) ---
    restart_end = t_start + 1700.0
    if time.monotonic() < restart_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(15.0, (restart_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > restart_end - 5:
                break
            per = min(per_r, restart_end - time.monotonic() - 5)
            if per < 8:
                break
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.05 + 0.08 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            p_r, sc_r = _adam_optimize(p_s, time.monotonic() + per, rng, lr=0.04,
                                       beta_frac=0.07, beta_decay=0.93,
                                       no_imp_limit=int(0.3 * n_best))
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p_r.copy()
                print(f"  RESTART {i}: {best_score:.10f}")

            if sc_r < best_score + 0.01 and time.monotonic() < restart_end - 5:
                p_rb, sc_rb = _lbfgs_polish(best_seq, min(time.monotonic() + 15, restart_end - 5),
                                             beta=1e-3, maxiter=30)
                if sc_rb < best_score:
                    best_score = sc_rb
                    best_seq = p_rb.copy()
                    print(f"  RESTART {i} L-BFGS: {best_score:.10f}")

    # --- Phase 8: Final exhaustive polish (1700s-deadline) ---
    final_end = deadline - 10.0
    if time.monotonic() < final_end and best_seq is not None:
        for beta_final in [5e-3, 2e-3, 8e-4, 3e-4]:
            if time.monotonic() > final_end - 5:
                break
            per = min(final_end, time.monotonic() + 20.0)
            if per - time.monotonic() < 3:
                break
            p_f, sc_f = _lbfgs_polish(best_seq, per, beta=beta_final, maxiter=50)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = p_f.copy()
                print(f"  FINAL L-BFGS b={beta_final}: {best_score:.10f}")

        if time.monotonic() < final_end - 10:
            per = final_end - 5
            if per > 10:
                p_fa, sc_fa = _adam_optimize(best_seq, per, rng, lr=0.01,
                                             beta_frac=0.03, beta_decay=0.96,
                                             no_imp_limit=200)
                if sc_fa < best_score:
                    best_score = sc_fa
                    best_seq = p_fa.copy()
                    print(f"  FINAL ADAM: {best_score:.10f}")

    # --- Final validation ---
    print(f"\nFinal best: {best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(256, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    fb = _make_profile_linear(512, 1.5)
    print(f"FALLBACK n=512")
    return [float(x) for x in fb]
