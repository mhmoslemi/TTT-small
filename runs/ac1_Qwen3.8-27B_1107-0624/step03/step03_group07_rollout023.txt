```python
def lp_step(f, M):
    """
    Solve: max sum(b) subject to b >= 0, (f*b)(i) <= M for all i.
    Returns optimal b.
    """
    n = len(f)
    # Build constraint matrix: A[i, j] = f[i-j] if 0 <= i-j < n
    # This is a (2n-1) x n matrix
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    # Build the matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(2*n-1):
            idx = i - j
            if 0 <= idx < n and f[idx] > 0:
                rows.append(i)
                cols.append(j)
                vals.append(f[idx])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(b)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
LP-based and hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find descent directions, combined with 
gradient-based refinement and anti-peak heuristics.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ... helper functions ...

def lp_direction(f, M):
    """Solve LP to find direction g that reduces peaks."""
    n = len(f)
    # Build constraint matrix
    # A[i, j] = f[i-j] if 0 <= i-j < n else 0
    # Shape: (2n-1) x n
    
    # Use dense for simplicity (n=2048 gives ~67MB which is fine)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f  # A[i, j] = f[i-j] for i = j, ..., j+n-1
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s*s))
```

```python
def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def softmax_weights(c, beta):
    """Compute soft-max value and softmax weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def interpolate_profile(p0, n_new):

    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def lp_direction(f, M, max_time=30.0):
    """
    Solve LP: max sum(b) s.t. f*b <= M, b >= 0.
    Returns optimal b or None if infeasible/timeout.
    """
    n = len(f)
    t0 = time.monotonic()
    if time.monotonic() - t0 > max_time:
        return None

    # Build constraint matrix A (2n-1) x n
    # A[i, j] = f[i-j] if 0 <= i-j < n else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': max_time, 'presolve': True})
        if res.success:
            sol = res.x
            sol = np.maximum(sol, 0)
            s = sol.sum()
            if s > 1e-10:
                return sol
            return None
    except Exception:
        pass
    return None

def anti_peak_step(a, c, n_peaks=5, n_moves=10):
    """
    Targeted anti-peak moves: identify peaks, move mass from contributing
    positions to non-contributing positions.
    """
    n = len(a)
    a_new = a.copy()
    
    for _ in range(n_moves):
        c_new = full_convolve_fft(a_new)
        peak_idx = int(np.argmax(c_new))
        
        # Find which positions contribute most to this peak
        # (a*a)[peak_idx] = sum_j a[j] * a[peak_idx - j]
        contrib = np.zeros(n)
        j_min = max(0, peak_idx - n + 1)
        j_max = min(n - 1, peak_idx)
        for j in range(j_min, j_max + 1):
            contrib[j] = a_new[j] * a_new[peak_idx - j]
        
        # Find the position with highest contribution
        j_high = int(np.argmax(contrib))
        if contrib[j_high] < 1e-15:
            break
        
        # Find a low-contribution position to move mass to
        # Prefer positions that don't contribute to any peak
        peak_set = set(np.argsort(c_new)[-n_peaks * 3:].tolist())
        low_contrib = []
        for j in range(n):
            if j == j_high:
                continue
            # Check if j contributes to any peak
            max_c = 0
            for p in list(peak_set)[:n_peaks]:
                if 0 <= p - j < n:
                    max_c = max(max_c, a_new[p - j])
            low_contrib.append((max_c, j))
        
        if not low_contrib:
            break
        low_contrib.sort()
        j_low = low_contrib[0][1]
        
        # Move a small amount of mass
        frac = min(0.02, a_new[j_high] * 0.1)
        a_new[j_high] -= frac
        a_new[j_low] += frac
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def logit_adam_optimize(a0, deadline, rng, lr=0.03, beta_factor=0.06,

    Logit-space Adam optimization with soft-max approximation.
    """

    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = softmax_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = z + current_lr * mh / (np.sqrt(vh) + eps)
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 500:
            current_lr = max(0.003, current_lr * 0.95)
            if no_improve > 1500:
                z = best_z = np.log(best_a)
                z = z - z.mean()
                z = z + 0.1 * rng.normal(size=n)
                z = np.clip(z, -40, 40)
                z = z - z.mean()
                a0 = np.exp(z - z.max())
                a0 = a0 / a0.sum()
                c = full_convolve_fft(a0)
                max_c = float(c.max())
                beta = max(1e-6, 0.06 * max_c)
                m.fill(0)
                v.fill(0)
                t_adam = 0
                current_lr = lr
                no_improve = 0

    return best_a, best_score


def projected_subgradient_optimize(a0, deadline, rng, lr=0.01, n_argmax=10):
    """
    Optimize using exact subgradient + simplex projection.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    best_a = a0.copy()
    best_score = float(2.0 * n * c.max())
    
    lr_current = lr
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = c.max()
        
        # Compute subgradient
        g = subgradient_step_fast(a0, c)
        
        # Projected gradient step
        a_new = project_simplex(a0 - lr_current * g / (n * max_c + 1e-10))
        
        # Line search
        best_t = 0.0
        best_sc = float(2.0 * n * max_c)
        for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a0 + t * a_new
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = compute_score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t > 0:
            a0 = (1 - best_t) * a0 + best_t * a_new
            a0 = np.maximum(a0, 0)
            s = a0.sum()
            if s > 1e-10:
                a0 = a0 / s
        
        sc = compute_score(a0)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a0.copy()
            no_improve = 0
        else:
            no_improve += 1
```
