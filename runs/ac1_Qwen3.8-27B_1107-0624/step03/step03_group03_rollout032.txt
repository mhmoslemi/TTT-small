def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_sparse(a):
    """Build sparse convolution matrix A where (A @ b)_k = conv(a,b)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_spread_full(a, M_lp):
    """LP: max sum(b) s.t. b >= 0, conv(a,b)_k <= M_lp for all k."""
    n = len(a)
    A = _build_conv_sparse(a)
    M = A.toarray()  # For n <= 2048, dense is fine
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=M, b_ub=np.full(2 * n - 1, M_lp),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search_quad(a, g, c_aa, c_ag, c_gg, max_iter=30):
    """
    Line search for (1-t)*a + t*g.
    Score(t) = 2n * max_k [(1-t)^2 c_aa_k + 2t(1-t) c_ag_k + t^2 c_gg_k]
    Each term is quadratic in t, max of quadratics is convex.
    Use golden section search.
    """
    def f(t):
        val = (1-t)**2 * c_aa + 2*t*(1-t) * c_ag + t*t * c_gg
        return val.max()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f(c_pt)
    fd = f(d_pt)
    
    for _ in range(max_iter):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = f(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f(d_pt)
    
    return (lo + hi) / 2.0

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        A[k, j_start:j_end] = a[j_start:j_end]
    return A


def _lp_spread(a, M_lp):
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M_lp),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search_quad(c_aa, c_ag, c_gg, max_iter=30):
    def f(t):
        val = (1.0 - t) ** 2 * c_aa + 2.0 * t * (1.0 - t) * c_ag + t * t * c_gg
        return val.max()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f(c_pt)
    fd = f(d_pt)
    for _ in range(max_iter):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = f(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f(d_pt)
    return (lo + hi) / 2.0

    sc_p2 = _score(a_p2)
    print(f"  Phase 2: {sc_p2:.10f}")
    if sc_p2 < best_score:
        best_score = sc_p2
        best_seq = a_p2.copy()
    a_work = a_p2 if sc_p2 < sc_p1 else a_p1

    # Phase 3: LSE gradient (fine-tuning)
    print(f"\n=== Phase 3: LSE gradient ===")
    phase3_end = min(t_start + 1000.0, deadline - 350.0)
    a_p3 = _lse_optimize(a_work, phase3_end, max_iter=40000,
                         T_start=0.015, T_end=1e-7, lr=0.05)
    sc_p3 = _score(a_p3)
    print(f"  Phase 3: {sc_p3:.10f}")
    if sc_p3 < best_score:
        best_score = sc_p3
        best_seq = a_p3.copy()
    a_work = a_p3 if sc_p3 < best_score else best_seq

    # Phase 4: LP refinement (conservative)
    print(f"\n=== Phase 4: LP refinement (conservative) ===")
    phase4_end = min(t_start + 1400.0, deadline - 200.0)
    a_p4 = _lp_refine(a_work, phase4_end, max_iter=3000,
                      initial_inflate=1.0004, blend_init=0.12)
    sc_p4 = _score(a_p4)
    print(f"  Phase 4: {sc_p4:.10f}")
    if sc_p4 < best_score:
        best_score = sc_p4
        best_seq = a_p4.copy()
    a_work = a_p4 if sc_p4 < best_score else best_seq

    # Phase 5: Final coordinate descent polish
    print(f"\n=== Phase 5: Final CD polish ===")
    if time.monotonic() < deadline - 80:
        phase5_end = min(t_start + 1650.0, deadline - 50.0)
        a_p5 = _coordinate_descent(a_work, phase5_end, max_sweeps=25)
        sc_p5 = _score(a_p5)
        print(f"  Phase 5: {sc_p5:.10f}")
        if sc_p5 < best_score:
            best_score = sc_p5
            best_seq = a_p5.copy()
        a_work = a_p5 if sc_p5 < best_score else a_work

    # Phase 6: Tiny final LP
    print(f"\n=== Phase 6: Final tiny LP ===")
    if time.monotonic() < deadline - 25:
        a_p6 = _lp_refine(a_work, deadline - 5.0, max_iter=500,
                          initial_inflate=1.0001, blend_init=0.05)
        sc_p6 = _score(a_p6)
        print(f"  Phase 6: {sc_p6:.10f}")
        if sc_p6 < best_score:
            best_score = sc_p6
            best_seq = a_p6.copy()
        a_work = a_p6 if sc_p6 < best_score else a_work

    # Final validation
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
