"""
Multi-scale optimization combining LSE gradient with Nesterov momentum,
LP refinement, KKT-based polishing, and spectral flattening.
Explores n=1024 and n=2048 with diverse initializations.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    """Score with sum(a)=1: returns 2n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_conv_matrix_full(n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = 1.0
    return A


def _make_conv_matrix_reduced(a, M, top_k=80):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    return A, b


def _lp_spread(a, M, use_reduced=False, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    if use_reduced and n > 128:
        A, b = _make_conv_matrix_reduced(a, M, top_k)
    else:
        A = _make_conv_matrix_full(n)
        b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def _line_search_blend(a, g, n_evals=15):
    """Golden section search for best blend factor."""
    best_t = 0.0
    best_score = _score_normalized(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_normalized(np.maximum(a + c * (g - a), 0))
    fd = _score_normalized(np.maximum(a + d * (g - a), 0))
    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_normalized(np.maximum(a + c * (g - a), 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_normalized(np.maximum(a + d * (g - a), 0))
    best_t = (lo + hi) / 2.0
    return best_t


def _lp_refine(a, deadline, max_iter=5000, inflate=1.0004, blend=0.2, use_line_search=True, top_k=80):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score_normalized(a)

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        use_red = n > 128
        g = _lp_spread(a, M_lp, use_reduced=use_red, top_k=top_k)

        if g is None:
            inflate_cur = min(1.01, inflate_cur * 1.15)
            no_imp += 5
            if no_imp > 400:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.05)
            if no_imp > 500:
                break
            continue

        g = g / sum_g

        if use_line_search:
            t_opt = _line_search_blend(a, g, n_evals=12)
            if t_opt < 0.01:
                t_opt = blend_cur * 0.5
        else:
            t_opt = blend_cur

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.02, blend_cur * 0.85)
                inflate_cur = min(1.006, inflate_cur * 0.92)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _lse_nesterov_step(a, z, state, T, lr):
    """One Nesterov Adam step in logit space using log-sum-exp."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    # Nesterov lookahead
    z_nest = z - lr * state['m'] / (np.sqrt(state['v']) + 1e-8)
    # Recompute gradient at Nesterov point (approximation: use current gradient)
    z = z_nest - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z - z.max())
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v

    return a_new, z, state, m_val

    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score_normalized(a)

    step = 0.01

    for it in range(max_iter):
        if time.monotonic() > deadline:

        c = _fft_convolve(a)
        m_val = float(c.max())

        threshold = m_val * 0.99
        active_idx = np.where(c >= threshold)[0]

        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])

        # Compute gradient as average over active lags
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        grad = np.zeros(n)
        for k in active_idx:
            # conv(w_delta_k, a_rev) gives the gradient contribution
            # For a single lag k, grad_j = 2 * sum_i a[i] * delta(i + j == k)
            # Actually, d/d a_j of c[k] = a[j + (n-1-k)] + a[j - (n-1-k)] (if in range)
            # More precisely: c[k] = sum_j a[j]*a[k-j], so d/d a_j c[k] = a[k-j] (if 0<=k-j<n) + a[k-j] (symmetric)
            # Simplified: for the full convolution c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # d c[k] / d a[j] = a[k-j] + a[k-j] = 2*a[k-j] if both j and k-j are in [0,n-1)
            # Wait, that's only if j != k-j, i.e., j != k/2. If j == k/2, it's just a[j] (the square term).
            # Actually for the quadratic form: c[k] = sum_j a[j]*a[k-j]
            # d c[k] / d a[j] = a[k-j] (from the first factor) + a[k-j] (from the second factor, when the index shifts)
            # No: d/d a[j] of sum_i a[i]*a[k-i] = a[k-j] (derivative of a[j]*a[k-j] wrt a[j] gives a[k-j])
            #   + (term where k-i = j, i.e., i = k-j: derivative of a[k-j]*a[j] wrt a[j] gives a[k-j])
            # So total: 2*a[k-j] if 0 <= k-j < n, else a[k-j] if k-j == j (the diagonal)
            # More carefully: c[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
            # d c[k] / d a[j] = a[k-j] (if max(0,k-n+1) <= j <= min(n-1,k))
            #   + a[k-j] (if max(0,k-n+1) <= k-j <= min(n-1,k), i.e., max(0,n-1-k) <= j <= min(n-1,k))
            # For j in the valid range for the first sum: contributes a[k-j]
            # For j in the valid range for the second sum (i = k-j): contributes a[k-j]
            # These overlap when both j and k-j are in [0, n-1]
            # If j == k-j (j = k/2), the two contributions are the same term (a[j]^2), so derivative is 2*a[j]
            # Actually no: c[k] = sum_i a[i]*a[k-i]. The term with i=j is a[j]*a[k-j].
            # The term with i=k-j is a[k-j]*a[j]. These are the same term if j = k-j.
            # So if j != k-j: c[k] contains a[j]*a[k-j] twice (once with i=j, once with i=k-j)
            #   Wait no! c[k] = sum_i a[i]*a[k-i]. Each i appears once.
            #   If j is in range: the term i=j gives a[j]*a[k-j]
            #   If k-j is in range: the term i=k-j gives a[k-j]*a[j]
            #   These are two different terms in the sum (unless j = k-j)
            #   So d c[k] / d a[j] = a[k-j] (from i=j term) + a[k-j] (from i=k-j term, if k-j != j)
            #   = 2*a[k-j] if j != k-j and both j, k-j in [0,n-1]
            #   = a[k-j] if j = k-j (i.e., the square term) and j in [0,n-1]
            #   More precisely: d c[k] / d a[j] = a[k-j] * (1 if k-j in [0,n-1]) + a[k-j] * (1 if k-j in [0,n-1] and k-j != j)
            #   Hmm, I'm overcomplicating this. Let me just use the FFT approach.
            pass
        # Use FFT to compute gradient at active lags
        w = np.zeros(2 * n - 1)
        w[active_idx] = 1.0 / len(active_idx)
        fw = np.fft.rfft(w, N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Project gradient onto tangent space of simplex
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        # Take step
        a_new = a - step * grad_tan
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            step *= 0.5
            continue
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
            step = min(0.05, step * 1.05)
        else:
            no_imp += 1
            step *= 0.9
            if no_imp > max(2000, 5 * n):
                break

    return best_a


def _spectral_flatten(a, max_iter=200, alpha=0.7):
    """Iterative spectral flattening (IAM-like)."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score_normalized(a)

    for it in range(max_iter):
        if time.monotonic() > deadline_global:
            break
        A = np.fft.rfft(a, 2 * n - 1)
        mag = np.abs(A)
        mag_flat = mag ** alpha
        # Normalize to keep sum(a) = 1
        # The DC component controls the sum
        phase = np.angle(A)
        A_new = mag_flat * phase / np.abs(phase + 1e-30) * (phase + 1e-30) / (np.abs(phase + 1e-30) + 1e-30)
        # Simpler: keep phase, modify magnitude
        A_new = mag_flat * (A / (mag + 1e-30))
        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def _subgradient_polish(a, deadline, max_iter=50000, lr=0.02):
    """Polish using subgradient at the actual max lags."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score_normalized(a)

    v = np.zeros(n)
    momentum = 0.7
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-10)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > max(3000, 8 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    return _normalize(p)


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    return _normalize(np.maximum(a, 1e-10))


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    return _normalize(np.maximum(a, 1e-10))


def _make_flat_edges(n, edge_weight=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    return _normalize(np.maximum(a, 1e-10))


def _make_asymmetric(n, eta=1.0):
    """Asymmetric profile shifted to one side."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-eta * (t - 0.3)**2 * 4.0)
    return _normalize(np.maximum(a, 1e-12))


def _make_bimodal(n, gap=0.3):
    """Two peaks."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-8.0*(t-0.3)**2) + np.exp(-8.0*(t-0.7)**2)
    return _normalize(np.maximum(a, 1e-12))


deadline_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    global deadline_global
    """
    Multi-scale optimization combining LSE+Nesterov, LP refinement,
    KKT polishing, and spectral flattening. Explores n=1024 and n=2048.
    """
    t_start = time.monotonic()
    deadline_global = t_start + budget_s - 25.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    # Determine which lengths to try
    lengths = [1024, 2048]
    if parent is not None and parent.size not in lengths:
        lengths.insert(0, parent.size)

    # Phase 1: LSE optimization on diverse candidates
    print("\n=== Phase 1: LSE+Nesterov optimization ===")
    best_per_length = {}
    phase1_deadline = t_start + budget_s * 0.45

    for n_t in lengths:
        if time.monotonic() > phase1_deadline:
            break

        # Generate candidates
        cands = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            for _ in range(3):
                p = _interpolate_profile(parent, n_t)
                p_pert = p * (1.0 + 0.06 * rng.normal(size=n_t))
                cands.append(_normalize(p_pert))
        cands.append(_make_gaussian(n_t, 0.35))
        cands.append(_make_raised_cosine(n_t))
        cands.append(_make_flat_edges(n_t, 0.4))
        cands.append(_make_asymmetric(n_t, 1.0))
        cands.append(_make_bimodal(n_t, 0.3))

        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > phase1_deadline - 20:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # LSE optimization
            remaining = min(phase1_deadline - time.monotonic() - 10, 150.0)
            if remaining < 15:
                break
            lse_end = time.monotonic() + remaining

            # Use different T and lr for different candidates
            T_s = 0.04 if ci < 3 else 0.06
            lr_v = 0.08 if ci < 3 else 0.06

            a_lse = _lse_optimize(a0, lse_end, max_iter=60000,
                                  T_start=T_s, T_end=1e-5, lr=lr_v, nesterov=True)
            sc_lse = _score(a_lse)

            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse

            print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> LSE={sc_lse:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best for n={n_t}: {best_here_score:.10f}")

    # Phase 2: LP refinement
    print("\n=== Phase 2: LP refinement ===")
    phase2_deadline = t_start + budget_s * 0.65

    for n_t in sorted(best_per_length.keys()):
        if time.monotonic() > phase2_deadline:
            break

        a_in = best_per_length[n_t]
        sc_in = _score(a_in)
        print(f"  LP start n={n_t}: score={sc_in:.10f}")

        lp_remaining = min(phase2_deadline - time.monotonic() - 10, 200.0)
        if lp_remaining < 15:
            continue
        lp_end = time.monotonic() + lp_remaining

        # Adjust LP params based on n
        top_k = 60 if n_t <= 1024 else 100
        a_lp = _lp_refine(a_in, lp_end, max_iter=3000,
                          inflate=1.0003, blend=0.15, use_line_search=True, top_k=top_k)
        sc_lp = _score(a_lp)

        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP result n={n_t}: score={sc_lp:.10f}")

    # Phase 3: KKT polish + subgradient on best
    print("\n=== Phase 3: KKT + Subgradient polish ===")
    phase3_deadline = t_start + budget_s * 0.82

    if best_seq is not None and time.monotonic() < phase3_deadline:
        for n_t in sorted(best_per_length.keys()):
            if time.monotonic() > phase3_deadline - 10:
                break
            a_cur = best_per_length[n_t]
            # KKT polish
            kkt_end = min(time.monotonic() + 60.0, phase3_deadline - 5)
            a_kkt = _kkt_polish(a_cur, kkt_end, max_iter=5000)
            sc_kkt = _score(a_kkt)
            if sc_kkt < _score(a_cur):
                best_per_length[n_t] = a_kkt
                if sc_kkt < best_score:
                    best_score = sc_kkt
                    best_seq = a_kkt.copy()
            print(f"  KKT n={n_t}: score={sc_kkt:.10f}")

            # Subgradient
            if time.monotonic() < phase3_deadline - 10:
                sp_end = min(time.monotonic() + 40.0, phase3_deadline - 5)
                a_sp = _subgradient_polish(best_per_length[n_t], sp_end, max_iter=30000, lr=0.012)
                sc_sp = _score(a_sp)
                if sc_sp < _score(best_per_length[n_t]):
                    best_per_length[n_t] = a_sp
                    if sc_sp < best_score:
                        best_score = sc_sp
                        best_seq = a_sp.copy()
                print(f"  Subgrad n={n_t}: score={sc_sp:.10f}")

    # Phase 4: Final LP polish (very fine)
    print("\n=== Phase 4: Final LP polish ===")
    phase4_deadline = t_start + budget_s * 0.93

    if best_seq is not None and time.monotonic() < phase4_deadline:
        n_best = len(best_seq)
        a_in = _normalize(best_seq)
        lp_end = min(time.monotonic() + 60.0, phase4_deadline - 5)
        a_final = _lp_refine(a_in, lp_end, max_iter=3000,
                             inflate=1.0002, blend=0.08, use_line_search=True,
                             top_k=80 if n_best <= 1024 else 120)
        sc_final = _score(a_final)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final LP: score={sc_final:.10f}")

    # Phase 5: Last LSE pass
    if best_seq is not None and time.monotonic() < deadline_global:
        lse_end = min(time.monotonic() + 30.0, deadline_global)
        a_last = _lse_optimize(_normalize(best_seq), lse_end, max_iter=20000,
                               T_start=0.02, T_end=1e-6, lr=0.04, nesterov=True)
        sc_last = _score(a_last)
        if np.isfinite(sc_last) and sc_last < best_score:
            best_score = sc_last
            best_seq = a_last.copy()
        print(f"  Final LSE: score={sc_last:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
