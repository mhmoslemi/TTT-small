def compute_score(a):

    n = a.size
    if n == 0:
        return np.inf

    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def compute_h(w, a):
    """h[i] = sum_{m} a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def hard_max_weights(c):
    """Uniform weights on argmax positions."""
    mval = float(c.max())
    mask = c >= mval - 1e-12
    w = np.zeros_like(c)
    w[mask] = 1.0 / np.sum(mask)
    return mval, w

def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        return hard_max_weights(c)
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)

    s = float(np.sum(ex))
    if s < 1e-300:
        return hard_max_weights(c)
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def gradient_convolution(a, w):
    """Gradient of sum_k w[k] * (a*a)[k] with respect to a."""
    # d/d a[i] sum_k w[k] * sum_j a[j]*a[k-j]
    # = 2 * sum_k w[k] * a[k-i] (summed over valid k)
    # = 2 * h[i] where h[i] = sum_k w[k] * a[k-i]
    n = len(a)
    h = compute_h(w, a)
    return 2.0 * h

    z0 = np.log(np.maximum(a0, 1e-15))

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_power_profile(n, p=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + 0.05
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_bimodal_profile(n, width=0.2):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.25) / width)**2) + np.exp(-((t - 0.75) / width)**2) + 0.05
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_smooth_random(n, rng, n_freq=8):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_freq + 1):
        amp = rng.uniform(-0.3, 0.3) / k
        a = a + amp * np.cos(2 * np.pi * k * t + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result
