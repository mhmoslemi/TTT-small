```python
def _spectral_flatten(a, alpha=0.5):
    """
    Partially flatten the power spectrum while preserving phases.
    alpha in (0,1): 0 means no change, 1 means full flattening.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    
    # Target: make |A_k| as flat as possible for k > 0
    # Keep A[0] fixed (it's the sum)
    # For k > 0, blend |A_k| toward the geometric mean
    
    mags = np.abs(A)
    mags_nonzero = mags[1:]
    if len(mags_nonzero) == 0:
        return a.copy()
    
    # Geometric mean of non-DC magnitudes
    log_mags = np.log(np.maximum(mags_nonzero, 1e-30))
    target_mag = float(np.exp(np.mean(log_mags)))
    
    # Blend
    new_mags = mags.copy()
    new_mags[1:] = (1 - alpha) * mags_nonzero + alpha * target_mag
    
    # Reconstruct with original phases
    phases = np.angle(A)
    A_new = new_mags * np.exp(1j * phases)
    
    a_new = np.real(np.fft.irfft(A_new, N))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s
```

```python
    """Find optimal delta for coordinate j."""
    n = len(a)
    # Compute g_k = 2 * a_{k-j} for valid k
    g = np.zeros(2 * n - 1)
    lo = max(0, j)  # k-j >= 0 => k >= j
    hi = min(2*n-2, j + n - 1)  # k-j <= n-1 => k <= j+n-1
    for k in range(lo, hi + 1):
        idx = k - j
        if 0 <= idx < n:
            g[k] = 2.0 * a[idx]

    quad_k = 2 * j
    if 0 <= quad_k < 2 * n - 1:

        pass  # handle separately

    def obj(delta):

        if 0 <= quad_k < len(vals):
            vals[quad_k] += delta * delta
```
