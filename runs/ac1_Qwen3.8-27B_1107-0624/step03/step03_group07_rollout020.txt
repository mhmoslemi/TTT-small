    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(n_new) / n_new
    return result


def lp_optimize(a0, deadline, max_iters=5000, verbose=False, verbose_every=50):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 < 1e-15:
        a0 = np.ones(n) / n
    a0 = a0 / a0.sum()

    c0 = full_conv(a0)
    M0 = float(c0.max())
    best_a = a0.copy()
    best_M = M0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = float(full_conv(a0).max())
        b = lp_direction(a0, M, max_time=8.0)
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb

        n_a = a0.copy()
        n_b = b.copy()
        t_opt, _ = golden_line_search(a0, b, n_a, n_b, n_iter=35)

        a_new = (1.0 - t_opt) * a0 + t_opt * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        M_new = float(full_conv(a_new).max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

        a0 = a_new

        if verbose and (it + 1) % verbose_every == 0:
            print(f"    LP iter {it+1}: M={M_new:.10f}, best_M={best_M:.10f}, t={t_opt:.4f}")

    return best_a, best_M


def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10 * max(mval, 1.0)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def compute_grad_h(w, a):
    n = len(a)
    af = a[::-1]
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def gradient_optimize(a0, deadline, wg, lr=0.03, beta_factor=0.05, beta_decay=0.95,
                      restart_period=500, verbose=False):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_conv(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    beta = beta_factor * max_c
    if beta < 1e-10:
        beta = 1e-5

    best_a = a0.copy()
    best_z = z.copy()
    best_M = max_c
    no_improve = 0
    current_lr = lr

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    while time.monotonic() < deadline:
        w = soft_max_weights(c, beta)
        h = compute_grad_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            continue
        a0 = a0 / s

        c = full_conv(a0)
        max_c = float(c.max())

        if max_c < best_M - 1e-14:
            best_M = max_c
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(1e-10, beta * beta_decay)

        if no_improve > 200:
            current_lr = max(0.003, current_lr * 0.95)

        if no_improve > restart_period:
            z = best_z + 0.1 * wg.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_conv(a0)
            max_c = float(c.max())
            beta = max(1e-8, beta_factor * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = lr
            no_improve = 0

    return best_a, best_M


def coordinate_descent_refine(a0, deadline, n_passes=200):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    best_a = a0.copy()
    best_M = float(full_conv(a0).max())

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a0[j]
            if old_val < 1e-15:
                try_vals = [1e-8, 1e-6, 1e-4, 1e-3]
            else:
                try_vals = [old_val * 0.7, old_val * 0.85, old_val, old_val * 1.15, old_val * 1.3]
            best_v = old_val
            best_M_local = best_M

            for tv in try_vals:
                a0[j] = max(0.0, tv)
                s = a0.sum()
                if s < 1e-15:
                    a0[j] = old_val
                    continue
                a0 = a0 / s
                M_local = float(full_conv(a0).max())
                if M_local < best_M_local - 1e-15:
                    best_M_local = M_local
                    best_v = tv
                a0[j] = old_val

            if best_v != old_val:
                a0[j] = max(0.0, best_v)
                s = a0.sum()
                a0 = a0 / s
                improved = True
                M_now = float(full_conv(a0).max())
                if M_now < best_M:
                    best_M = M_now
                    best_a = a0.copy()

        if not improved:
            break

    return best_a, best_M


def diverse_starts(n, rng):
    profiles = []
    profiles.append(make_cosine_profile(n, 1, 0.0))
    profiles.append(make_gaussian_profile(n, 0.35))
    profiles.append(make_gaussian_profile(n, 0.5))
    profiles.append(make_linear_profile(n, 1.0))
    profiles.append(make_linear_profile(n, 2.0))
    profiles.append(make_poly_profile(n, 2))
    profiles.append(make_poly_profile(n, 3))
    for _ in range(8):
        profiles.append(make_random_smooth(n, rng, n_harm=rng.integers(3, 8)))
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc_init = compute_score(init_seq)
                print(f"Loaded initial: n={init_seq.size}, score={sc_init:.10f}")
                update(init_seq, sc_init)
    except Exception:
        pass

    print(f"Budget: {budget_s}s, deadline in {deadline - t_start:.1f}s")

    # ===== Phase 1: LP optimization at n=256 =====
    n1 = 256
    phase1_end = min(deadline, t_start + 250.0)
    print(f"\n=== Phase 1: LP at n={n1} ===")

    starts_1 = diverse_starts(n1, rng)
    if init_seq is not None:
        starts_1.insert(0, interpolate_profile(init_seq, n1))

    best_p1 = None
    best_M1 = np.inf

    for idx, p in enumerate(starts_1):
        if time.monotonic() > phase1_end - 10:
            break
        sc = compute_score(p)
        M_init = float(full_conv(p).max())
        if M_init >= best_M1 + 0.001:
            continue
        dl = min(phase1_end, time.monotonic() + 60.0)
        result, M_res = lp_optimize(p, dl, max_iters=300, verbose=(idx < 3))
        sc_res = compute_score(result)
        if sc_res < best_score:
            update(result, sc_res)
        if M_res < best_M1:
            best_M1 = M_res
            best_p1 = result.copy()
        if idx < 3 or sc_res < 1.52:
            print(f"  Start {idx}: score={sc_res:.10f}, M={M_res:.8f}")

    if best_p1 is not None:
        print(f"  Phase 1 best: score={compute_score(best_p1):.10f}")

    # ===== Phase 2: LP optimization at n=512 =====
    n2 = 512
    phase2_end = min(deadline, t_start + 700.0)
    print(f"\n=== Phase 2: LP at n={n2} ===")

    if best_p1 is not None:
        p2_base = interpolate_profile(best_p1, n2)
    elif init_seq is not None:
        p2_base = interpolate_profile(init_seq, n2)
    else:
        p2_base = make_cosine_profile(n2, 1, 0.0)

    sc_p2 = compute_score(p2_base)
    update(p2_base, sc_p2)
    print(f"  Interpolated to n={n2}: score={sc_p2:.10f}")

    dl2 = min(phase2_end, time.monotonic() + 400.0)
    result2, M2 = lp_optimize(p2_base, dl2, max_iters=200, verbose=True, verbose_every=20)
    sc2 = compute_score(result2)
    update(result2, sc2)
    print(f"  After LP at n={n2}: score={sc2:.10f}, M={M2:.8f}")

    # Try a few alternative starts at n=512
    for idx, p in enumerate(starts_1[:4]):
        if time.monotonic() > phase2_end - 10:
            break
        p_alt = interpolate_profile(p, n2)
        sc_alt_init = compute_score(p_alt)
        if sc_alt_init < best_score + 0.05:
            dl_alt = min(phase2_end, time.monotonic() + 60.0)
            r_alt, M_alt = lp_optimize(p_alt, dl_alt, max_iters=100)
            sc_alt = compute_score(r_alt)
            update(r_alt, sc_alt)
            if sc_alt < best_score + 0.01:
                print(f"  Alt start {idx} at n={n2}: score={sc_alt:.10f}")

    # ===== Phase 3: LP + gradient at n=1024 =====
    n3 = 1024
    phase3_end = min(deadline, t_start + 1350.0)
    print(f"\n=== Phase 3: LP + gradient at n={n3} ===")

    base3 = best_seq if best_seq is not None else result2
    p3_base = interpolate_profile(base3, n3)
    sc3 = compute_score(p3_base)
    update(p3_base, sc3)
    print(f"  Interpolated to n={n3}: score={sc3:.10f}")

    # LP at n=1024 (fewer iterations due to cost)
    if time.monotonic() < phase3_end - 300:
        dl3_lp = min(phase3_end - 300, time.monotonic() + 200.0)
        result3_lp, M3 = lp_optimize(p3_base, dl3_lp, max_iters=80, verbose=True, verbose_every=10)
        sc3_lp = compute_score(result3_lp)
        update(result3_lp, sc3_lp)
        print(f"  After LP at n={n3}: score={sc3_lp:.10f}, M={M3:.8f}")
        if sc3_lp < sc3:
            p3_base = result3_lp
            sc3 = sc3_lp

    # Gradient refinement at n=1024
    if time.monotonic() < phase3_end - 50:
        dl3_grad = min(phase3_end, time.monotonic() + 300.0)
        result3_g, M3g = gradient_optimize(p3_base, dl3_grad, rng, lr=0.02,
                                            beta_factor=0.04, beta_decay=0.96)
        sc3_g = compute_score(result3_g)
        update(result3_g, sc3_g)
        print(f"  After gradient at n={n3}: score={sc3_g:.10f}")
        if sc3_g < sc3:
            p3_base = result3_g
            sc3 = sc3_g

    # ===== Phase 4: Gradient at n=2048 =====
    n4 = 2048
    phase4_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 4: Gradient at n={n4} ===")

    if time.monotonic() < phase4_end - 30 and best_score > 1.503:
        p4_base = interpolate_profile(best_seq if best_seq is not None else p3_base, n4)
        sc4 = compute_score(p4_base)
        update(p4_base, sc4)
        print(f"  Interpolated to n={n4}: score={sc4:.10f}")

        dl4 = min(phase4_end, time.monotonic() + 250.0)
        result4, M4 = gradient_optimize(p4_base, dl4, rng, lr=0.015,
                                         beta_factor=0.03, beta_decay=0.965,
                                         restart_period=800)
        sc4 = compute_score(result4)
        update(result4, sc4)
        print(f"  After gradient at n={n4}: score={sc4:.10f}")

    # ===== Phase 5: Polish =====
    print(f"\n=== Phase 5: Polish ===")
    polish_end = deadline

    if best_seq is not None and time.monotonic() < polish_end - 20:
        n_polish = best_seq.size
        a_polish = best_seq.copy()

        # Coordinate descent refinement
        if time.monotonic() < polish_end - 60:
            dl_cd = min(polish_end, time.monotonic() + 120.0)
            r_cd, M_cd = coordinate_descent_refine(a_polish, dl_cd, n_passes=50)
            sc_cd = compute_score(r_cd)
            update(r_cd, sc_cd)
            print(f"  After coord descent (n={n_polish}): score={sc_cd:.10f}")
            if sc_cd < best_score:
                a_polish = r_cd

        # Small perturbation + re-optimize
        for pert_scale in [0.001, 0.005, 0.01]:
            if time.monotonic() > polish_end - 10:
                break
            a_pert = a_polish + pert_scale * rng.normal(size=n_polish)
            a_pert = np.maximum(a_pert, 1e-12)
            a_pert = a_pert / a_pert.sum()
            sc_pert = compute_score(a_pert)
            if sc_pert < best_score:
                dl_pert = min(polish_end, time.monotonic() + 40.0)
                r_pert, _ = gradient_optimize(a_pert, dl_pert, rng, lr=0.005,
                                               beta_factor=0.01, beta_decay=0.97)
                sc_pert2 = compute_score(r_pert)
                update(r_pert, sc_pert2)
                print(f"  Perturbation {pert_scale}: score={sc_pert2:.10f}")

        # Try LP polish if n is small enough
        if n_polish <= 1024 and time.monotonic() < polish_end - 30:
            dl_lp_polish = min(polish_end, time.monotonic() + 80.0)
            r_lp_p, M_lp_p = lp_optimize(a_polish, dl_lp_polish, max_iters=50)
            sc_lp_p = compute_score(r_lp_p)
            update(r_lp_p, sc_lp_p)
            print(f"  LP polish (n={n_polish}): score={sc_lp_p:.10f}")

    # Try different n values for final evaluation
    if best_seq is not None:
        for n_try in [128, 256, 384, 512, 768, 1024, 1536, 2048]:
            if time.monotonic() > polish_end - 5:
                break
            if n_try == best_seq.size:
                continue
            p_try = interpolate_profile(best_seq, n_try)
            sc_try = compute_score(p_try)
            update(p_try, sc_try)
            if sc_try < best_score + 0.005 and time.monotonic() < polish_end - 15:
                dl_try = min(polish_end, time.monotonic() + 20.0)
                r_try, _ = gradient_optimize(p_try, dl_try, rng, lr=0.01,
                                              beta_factor=0.02, beta_decay=0.97)
                sc_try2 = compute_score(r_try)
                update(r_try, sc_try2)
                if sc_try2 < best_score + 0.005:
                    print(f"  n={n_try}: score={sc_try2:.10f}")

    print(f"\nFinal best score: {best_score:.10f}")
    if best_seq is not None:
        print(f"Final best n: {best_seq.size}")

    # Final validation
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_cosine_profile(256, 1, 0.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_cosine_profile(256, 1, 0.0)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        else:
            print(f"WARNING: evaluate_sequence returned {eval_score}")
            fallback = make_cosine_profile(256, 1, 0.0)
            print(f"Fallback score: {compute_score(fallback):.10f}")
            return [float(x) for x in fallback]
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check error: {e}")

    return [float(x) for x in final_seq]
