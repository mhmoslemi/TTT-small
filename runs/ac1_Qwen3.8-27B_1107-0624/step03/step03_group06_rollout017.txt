    # Phase 1: LP refinement at original length
    n_orig = len(parent)
    print(f"\n=== Phase 1: LP at n={n_orig} ===")
    lp1_deadline = min(deadline, t_start + budget_s * 0.35)
    a1 = _lp_refine(parent, lp1_deadline, max_iter=50000, initial_inflate=1.002)
    sc1 = _score(a1)
    print(f"Phase 1 LP: score={sc1:.10f}")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()

    # Phase 2: Try larger lengths (1024, 1536)
    for n_try in [1024, 1536]:
        if time.monotonic() > deadline - 200:
            break
        if n_try <= n_orig:
            continue
        print(f"\n=== Phase 2: LP at n={n_try} ===")
        a_interp = _interpolate_seq(best_seq, n_try)
        sc_interp = _score(a_interp)
        print(f"  Interpolated: score={sc_interp:.10f}")

        lp2_deadline = min(deadline, time.monotonic() + 250)
        a2 = _lp_refine(a_interp, lp2_deadline, max_iter=50000, initial_inflate=1.002)
        sc2 = _score(a2)
        print(f"  LP result: score={sc2:.10f}")
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()

    # Phase 3: Random restarts on best
    print(f"\n=== Phase 3: Random restarts (best score so far: {best_score:.10f}) ===")
    n_best = len(best_seq)
    temperatures = [0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]
    restart_budget = min(300.0, max(60.0, deadline - time.monotonic() - 200))
    per_restart = restart_budget / len(temperatures)

    for temp in temperatures:
        if time.monotonic() > deadline - 200:
            break
        n_restarts = max(1, int(per_restart / 25))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 200:
                break
            a_restart = _random_restart(best_seq, n_best, rng, temp, deadline - 200)
            sc_r = _score(a_restart)
            if sc_r < best_score - 1e-10:
                best_score = sc_r
                best_seq = a_restart.copy()
                print(f"  Restart (T={temp}, r={r}): NEW BEST score={sc_r:.10f}")

    # Phase 4: Gradient polish
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Gradient polish ===")
        gp_deadline = min(deadline, time.monotonic() + 150)
        a_gp = _gradient_polish(best_seq, gp_deadline, max_iter=50000)
        sc_gp = _score(a_gp)
        print(f"  Gradient: score={sc_gp:.10f}")
        if sc_gp < best_score:
            best_score = sc_gp
            best_seq = a_gp.copy()

    # Phase 5: Final LP pass with very tight parameters
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Final tight LP ===")
        lp5_deadline = min(deadline, time.monotonic() + 120)
        a_final = _lp_refine(best_seq, lp5_deadline, max_iter=30000, initial_inflate=1.001)
        sc_final = _score(a_final)
        print(f"  Final LP: score={sc_final:.10f}")
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # Phase 6: Quick gradient on final
    if time.monotonic() < deadline - 10:
        gp2_deadline = min(deadline, time.monotonic() + 40)
        a_g2 = _gradient_polish(best_seq, gp2_deadline, max_iter=20000)
        sc_g2 = _score(a_g2)
        print(f"  Final gradient: score={sc_g2:.10f}")
        if sc_g2 < best_score:
            best_score = sc_g2
            best_seq = a_g2.copy()

    # Validation
    final_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        pass  # Already normalized
    else:
        final_seq = _make_ramp(768, 1.0)
        best_score = _score(final_seq)

    # Final evaluation check
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print(f"WARNING: evaluate_sequence returned {sc_eval}")
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
