Now let me think about the LP more carefully. The LP at each iteration is:
- Maximize sum(g)
- Subject to: conv(a, g)[k] <= M for all k
- g >= 0

The constraint matrix is the "convolution matrix" of a:
A[k, j] = a[k-j] (if 0 <= k-j < n, else 0)

This is a (2n-1) × n matrix. For n=512, that's 1023 × 512, which is fine for HiGHS.

The key improvement I can make is to **only include active constraints**. At each iteration, I know which convolution indices are near the maximum. If I only include (say) the top 50% of constraints (those where conv(a,a)[k] > 0.5 * M), the LP becomes much faster.

But wait - the LP is finding a direction g that has `conv(a,g)[k] <= M` for ALL k. If I drop constraints, the solution might violate them. However, I can check afterwards and if violated, add those constraints back.

Actually, a better approach: since we're doing a line search `(1-t)*a + t*g`, the new max convolution is at most `(1-t)*M + t * max(conv(a,g))` (by linearity of convolution and the bound on conv(a,g)). So if `max(conv(a,g)) <= M`, the new max is <= M. But if we only enforce a subset of constraints, we might have `max(conv(a,g)) > M` at some dropped constraint. In that case, the line search would still find a valid t where the score improves.

Actually, the LP guarantees `conv(a,g)[k] <= M` for all included k. For excluded k, we don't have a guarantee. But in practice, if a is well-optimized, the convolution values that are far from the max won't be a problem. So the "active set" approach should work well.

Multi-resolution optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses active-set LP, Adam-type gradient descent, multi-resolution scaling,
symmetric constraint, and golden-section line search for improved convergence.

def _fft_convolve(a):

    n = a.size
    s = float(a.sum())

    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_unit(a):

def _symmetrize(a):

        s = a.sum()

    p = np.maximum(p, 0.0)

    return p / s

    t = np.linspace(0.0, 1.0, n)

    elif kind == 'triangle':

    elif kind == 'sinc_like':

        a = np.sinc(x / np.pi)

    a /= a.sum()

def _smooth_max_grad(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    shift_x = x - x.max()
    ex = np.exp(shift_x)
    s = float(ex.sum())
    if not np.isfinite(s) or s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    return mval, w

def _conv_gradient(w, a):
    n = len(a)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)

def _adam_step(a, grad, m_v, v_v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m_v[:] = b1 * m_v + (1.0 - b1) * grad
    v_v[:] = b2 * v_v + (1.0 - b2) * grad * grad
    m_hat = m_v / (1.0 - b1 ** t)
    v_hat = v_v / (1.0 - b2 ** t)
    step = m_hat / (np.sqrt(v_hat) + eps)
    return a - lr * step

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    idx = rho[-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _lp_solve_active(a, M, n, top_frac=1.0, use_symmetry=False):
    c = _fft_convolve(a)
    if top_frac < 1.0:
        threshold = np.percentile(c, 100.0 * (1.0 - top_frac))
        active = c >= threshold
    else:
        active = np.ones(2 * n - 1, dtype=bool)
    n_active = int(active.sum())
    if n_active == 0:
        return None
    A_rows = np.zeros((n_active, n))
    for j in range(n):
        idx = np.arange(j, j + n)
        valid = (idx >= 0) & (idx < 2 * n - 1) & active[idx]
        if valid.any():
            A_rows[active[idx[valid]], j] = a[idx[valid]]
    c_obj = -np.ones(n)
    b = np.full(n_active, M)
    res = linprog(c_obj, A_ub=A_rows, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _golden_section_line_search(a, g, n_grid=15):
    best_t = 0.0
    best_sc = _normed_score(a)
    try:
        lo, hi = 0.0, 1.0
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        x1 = hi - gr * (hi - lo)
        x2 = lo + gr * (hi - lo)
        f1 = _eval_blend(a, g, x1)
        f2 = _eval_blend(a, g, x2)
        for _ in range(n_grid):
            if f1 < f2:
                hi = x2
                x2 = x1
                f2 = f1
                x1 = hi - gr * (hi - lo)
                f1 = _eval_blend(a, g, x1)
            else:
                lo = x1
                x1 = x2
                f1 = f2
                x2 = lo + gr * (hi - lo)
                f2 = _eval_blend(a, g, x2)
        best_t = (lo + hi) / 2.0
        best_sc = _eval_blend(a, g, best_t)
        if best_sc >= best_sc:
            best_sc = _eval_blend(a, g, best_t)
        if best_sc < _normed_score(a) - 1e-15:
            return best_t, best_sc
    except Exception:
        pass
    return 0.0, best_sc

def _eval_blend(a, g, t):

    a_new = a_new / s

            if no_imp > max(150, n):

                inflate = inflate_start
                top_frac_cur = top_frac
                no_imp = 0
                if no_imp == 0 and time.monotonic() > deadline:
                    break

    return best_a

def _gradient_polish(a0, deadline, max_iter=50000, beta_start=None):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _symmetrize(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(_score_unit(a))
    mval0 = float(c.max())

    if beta_start is None:
        beta = max(1e-6, 0.03 * mval0)
    else:
        beta = beta_start

    lr = 0.05
    base_lr = lr
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        mval, w = _smooth_max_grad(c, beta)
        grad = _conv_gradient(w, a)
        grad = 2.0 * grad
        grad_mean = grad.mean()
        grad = grad - grad_mean

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 5.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        a_new = _symmetrize(a_new)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            v = np.zeros(n)
            lr = max(0.005, lr * 0.5)
            continue
        a = a_new / s_new

        c = _fft_convolve(a)
        score = float(_score_unit(a))

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.93)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                c_reset = _fft_convolve(a)
                beta = max(1e-6, 0.03 * float(c_reset.max()))
                lr = base_lr
                no_imp = 0

        if it % 50 == 49:
            beta = max(1e-8, beta * 0.85)

    return best_a

def _adam_gradient_polish(a0, deadline, max_iter=30000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _symmetrize(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(_score_unit(a))

    mval0 = float(c.max())
    beta = max(1e-6, 0.02 * mval0)
    lr = 0.02
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t_count = 0
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_count += 1

        c = _fft_convolve(a)
        mval, w = _smooth_max_grad(c, beta)
        grad = 2.0 * _conv_gradient(w, a)
        grad = grad - grad.mean()

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            m_v[:] = 0
            v_v[:] = 0
            continue

        a_new = _adam_step(a, grad, m_v, v_v, t_count, lr)
        a_new = np.maximum(a_new, 0.0)
        a_new = _symmetrize(a_new)
        s_new = a_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a = a_new / s_new

        c = _fft_convolve(a)
        score = float(_score_unit(a))

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.9)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                m_v[:] = 0
                v_v[:] = 0
                t_count = 0
                beta = max(1e-6, 0.02 * float(_fft_convolve(a).max()))
                lr = 0.02
                no_imp = 0

        if t_count % 100 == 99:
            beta = max(1e-8, beta * 0.8)

    return best_a

def _multi_resolution(a_init, n_final, deadline, budget_per_stage=180):
    stages = []
    n_cur = max(64, n_final // 16)
    while n_cur < n_final:
        stages.append(n_cur)
        n_cur = min(n_final, n_cur * 2)
    stages.append(n_final)
    stages = sorted(set(stages))

    a = a_init
    for i, n_stage in enumerate(stages):
        if time.monotonic() > deadline:
            break
        a = _interp_to(a, n_stage)
        a = _symmetrize(a)
        a = a / a.sum()
        remaining = deadline - time.monotonic()
        stage_budget = min(budget_per_stage, remaining * 0.4)
        if stage_budget < 20:
            stage_budget = max(10, remaining * 0.3)
        stage_end = time.monotonic() + stage_budget

        a_lp = _lp_refine(a, stage_end, max_iter=2000, inflate_start=1.003,
                          top_frac=0.6)
        sc_lp = _normed_score(a_lp)
        sc_cur = _normed_score(a)
        if sc_lp < sc_cur:
            a = a_lp

        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            gp_end = time.monotonic() + min(60, remaining2 * 0.3)
            a_gp = _gradient_polish(a, gp_end, max_iter=5000)
            sc_gp = _normed_score(a_gp)
            if sc_gp < _normed_score(a):
                a = a_gp

        sc = _normed_score(a)
        print(f"  MR stage n={n_stage}: score={sc:.8f}")

    return a

def _perturb_reoptimize(a, deadline, rng, max_perturbations=5):
    best_a = a.copy()
    best_sc = _normed_score(a)
    n = len(a)
    for p in range(max_perturbations):
        if time.monotonic() > deadline:
            break
        noise_scale = 0.01 / (p + 1)
        noise = rng.standard_normal(n) * noise_scale
        a_pert = a + noise
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = _symmetrize(a_pert)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s
        perturb_end = time.monotonic() + min(30, (deadline - time.monotonic()) * 0.2)
        a_opt = _lp_refine(a_pert, perturb_end, max_iter=500, inflate_start=1.005)
        sc = _normed_score(a_opt)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            a = a_opt
    return best_a

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _normed_score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define target lengths
    target_lengths = [512, 768, 1024]

    initializations = []
    if parent is not None:
        for n_t in target_lengths:
            a_p = _interp_to(parent, n_t)
            a_p = _symmetrize(a_p)
            initializations.append(('parent', n_t, a_p))

    for n_t in target_lengths:
        for prof in ['raised_cos', 'gaussian', 'flat_top', 'bell', 'triangle']:
            a_init = _make_init(n_t, prof)
            a_init = _symmetrize(a_init)
            initializations.append((prof, n_t, a_init))

    # Score and sort initializations
    scored_inits = []
    for name, n_t, a0 in initializations:
        sc = _normed_score(a0)
        scored_inits.append((sc, name, n_t, a0))
    scored_inits.sort(key=lambda x: x[0])

    print(f"Top initializations:")
    for sc, name, n_t, _ in scored_inits[:5]:
        print(f"  {name} n={n_t}: {sc:.8f}")

    # Phase 1: Multi-resolution from best initializations
    phase1_deadline = t_start + min(600, (deadline - t_start) * 0.35)
    for sc0, name, n_t, a0 in scored_inits[:6]:
        if time.monotonic() > phase1_deadline:
            break
        if sc0 > best_score + 0.3:
            continue
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        remaining = phase1_deadline - time.monotonic()
        if remaining < 60:
            break
        stage_budget = min(100, remaining * 0.4)
        if stage_budget < 20:
            break

        cand = _multi_resolution(a0, n_t, phase1_deadline, budget_per_stage=stage_budget)
        sc = _normed_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"  MR {name} n={n_t}: score={sc:.10f}")

    # Phase 2: Intensive LP refinement on best
    if best_seq is not None:
        n_best = len(best_seq)
        phase2_deadline = t_start + min(1200, (deadline - t_start) * 0.7)
        remaining = phase2_deadline - time.monotonic()
        if remaining > 30:
            print(f"Phase 2: LP refine n={n_best}, budget={remaining:.0f}s")
            a_lp = _lp_refine(best_seq, phase2_deadline, max_iter=10000,
                              inflate_start=1.002, inflate_min=1.0001, top_frac=0.5)
            sc = _normed_score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
                print(f"  LP score={sc:.10f}")

    # Phase 3: Adam gradient polish
    if best_seq is not None:
        phase3_deadline = t_start + min(1500, (deadline - t_start) * 0.85)
        remaining = phase3_deadline - time.monotonic()
        if remaining > 20:
            print(f"Phase 3: Adam gradient, budget={remaining:.0f}s")
            a_adam = _adam_gradient_polish(best_seq, phase3_deadline, max_iter=30000)
            sc = _normed_score(a_adam)
            if sc < best_score:
                best_score = sc
                best_seq = a_adam.copy()
                print(f"  Adam score={sc:.10f}")

    # Phase 4: Standard gradient polish
    if best_seq is not None:
        phase4_deadline = deadline - 80
        remaining = phase4_deadline - time.monotonic()
        if remaining > 20:
            print(f"Phase 4: Momentum gradient, budget={remaining:.0f}s")
            a_gp = _gradient_polish(best_seq, phase4_deadline, max_iter=50000)
            sc = _normed_score(a_gp)
            if sc < best_score:
                best_score = sc
                best_seq = a_gp.copy()
                print(f"  Momentum score={sc:.10f}")

    # Phase 5: Perturbation and re-optimization
    if best_seq is not None:
        perturb_deadline = deadline - 50
        remaining = perturb_deadline - time.monotonic()
        if remaining > 30:
            print(f"Phase 5: Perturbation, budget={remaining:.0f}s")
            a_pert = _perturb_reoptimize(best_seq, perturb_deadline, rng, max_perturbations=8)
            sc = _normed_score(a_pert)
            if sc < best_score:
                best_score = sc
                best_seq = a_pert.copy()
                print(f"  Perturb score={sc:.10f}")

    # Phase 6: Try different length if we have time
    if best_seq is not None:
        final_deadline = deadline - 30
        remaining = final_deadline - time.monotonic()
        if remaining > 60:
            n_try = len(best_seq)
            n_new = n_try * 2 if n_try <= 768 else n_try // 2
            if n_new != n_try and n_new >= 128 and n_new <= 2048:
                a_new_len = _interp_to(best_seq, n_new)
                a_new_len = _symmetrize(a_new_len)
                a_new_len = a_new_len / a_new_len.sum()
                opt_end = time.monotonic() + min(60, remaining * 0.5)
                a_new_len = _lp_refine(a_new_len, opt_end, max_iter=2000,
                                       inflate_start=1.003, top_frac=0.6)
                sc = _normed_score(a_new_len)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_new_len.copy()
                    print(f"  New length n={n_new}: score={sc:.10f}")

    # Final: one more LP pass with very tight params
    if best_seq is not None:
        final_deadline = deadline
        remaining = final_deadline - time.monotonic()
        if remaining > 15:
            a_final = _lp_refine(best_seq, final_deadline, max_iter=3000,
                                 inflate_start=1.001, inflate_min=1.0001, top_frac=0.4)
            sc = _normed_score(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"  Final LP: score={sc:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_init(512, 'raised_cos')
        best_score = _normed_score(best_seq)

    # Final validation and normalization
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_init(512, 'raised_cos') / _make_init(512, 'raised_cos').sum()

    # Evaluate with the actual function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
