Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses projected subgradient descent, L-BFGS with continuation, Adam polishing,

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _logit_fn(a):
    return np.log(np.maximum(a, 1e-15)) - np.log(np.maximum(a, 1e-15)).mean()

    return 2.0 * n * (cm + beta * log_sum)

    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _subgradient_step(a, lr, n):
    """
    Projected subgradient descent step for minimizing max(conv(a,a))
    subject to sum(a)=1, a>=0. Uses the exact subgradient at the active lags.
    """
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Find active lags (within 1% of max)
    threshold = c_max * 0.999
    active = c >= threshold
    w = active.astype(np.float64)
    w = w / w.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    # Gradient of sum w_k * c_k w.r.t. a
    grad = 2.0 * a * h
    # Projected gradient step on simplex
    a_new = a - lr * grad
    # Project onto simplex {x: sum(x)=1, x>=0}
    a_new = _project_simplex(a_new)
    return a_new, c_max


def _project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def _adam_simplex(a_init, deadline, rng, lr=0.01, n_max_iter=50000, beta_frac=0.005):
    """
    Adam optimizer on the simplex with subgradient of smooth max.
    """
    n = len(a_init)
    a = a_init.copy()
    z = _logit_fn(a)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 500 == 0:
            beta = max(1e-12, beta * 0.8)
            if sc < 1.52:
                lr = max(0.001, lr * 0.95)

    return best_a, best_score


def _active_set_descent(a_init, deadline, rng, n_max_iter=30000):
    """
    Direct projected subgradient descent targeting active lags.
    Uses adaptive step size (Barzilai-Borwein style).
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    lr = 0.1 / n
    prev_a = a.copy()
    prev_c_max = _score_norm_fn(a)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        a_new, c_max = _subgradient_step(a, lr, n)

        # Barzilai-Borwein step size update
        s = a_new - a
        y = c_max - prev_c_max
        ss = float(np.dot(s, s))
        sy = float(np.dot(s, y))
        if ss > 1e-30:
            if sy > 1e-15:
                lr_bb = ss / sy
            elif sy < -1e-15:
                lr_bb = ss / (-sy) * 0.5
            else:
                lr_bb = lr
            lr = np.clip(lr_bb, 1e-6, 10.0 / n)

        a = a_new
        prev_a = a.copy()
        prev_c_max = c_max

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _lbfgs_continuation(a_init, deadline, rng, max_wall_time=None):
    """L-BFGS with beta continuation."""
    n = len(a_init)
    z = _logit_fn(a_init)

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004]
    betas = [b * max_c0 for b in beta_starts]

    if max_wall_time is not None:
        per_level = max_wall_time / len(betas)
    else:
        per_level = 60.0

    for beta in betas:
        if time.monotonic() > deadline:
            break

        level_start = time.monotonic()
        level_deadline = level_start + per_level

        try:
            result = minimize(
                _smooth_max_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_smooth_max_grad,
                options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 40, 'maxfun': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    # Uniform
    profiles.append(np.ones(n) / n)

    # Linear
    for slope in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        a2 = 1.0 + slope * (0.5 - t)
        a2 = np.maximum(a2, 1e-12)
        profiles.append(a2 / a2.sum())

    # Quadratic
    for a2 in [-0.5, -0.2, 0.2, 0.5]:
        a = 1.0 + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cubic
    for a3 in [-0.5, -0.3, 0.3, 0.5]:
        a = 1.0 + a3 * u ** 3
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Sinusoidal
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Cosine bump
    a = np.cos(np.pi * (t - 0.5)) ** 2
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Exponential decay (two sides)
    a = np.exp(-np.abs(u) * 2.0)
    profiles.append(a / a.sum())
    a = np.exp(-np.abs(u) * 4.0)
    profiles.append(a / a.sum())

    # Power law
    a = 1.0 / (1.0 + np.abs(u) ** 2)
    profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining L-BFGS continuation, Adam,
    active-set subgradient, and multi-restart exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 1024
        parent = np.ones(n) / n
        print(f"No parent, using uniform n={n}")

    # --- Phase 0: Evaluate candidates (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    candidates = []

    sc_parent = _score_norm_fn(parent)
    candidates.append((sc_parent, parent.copy(), len(parent), "parent"))
    update_best(parent, sc_parent, "parent")

    for sn in [512, 1024, 2048]:
        if time.monotonic() > phase0_end:
            break
        profiles = _make_profiles(sn)
        for idx, p in enumerate(profiles):
            sc = _score_norm_fn(p)
            candidates.append((sc, p, sn, f"prof{idx}_n{sn}"))
            update_best(p, sc, f"prof{idx}_n{sn}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f} at n={len(best_seq) if best_seq is not None else 0}")

    # --- Phase 1: L-BFGS on top candidates (400s) ---
    phase1_end = min(deadline, t_start + 430.0)
    count = 0
    seen_n = set()

    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end - 60:
            break
        if count >= 4:
            break
        if sc > best_score * 1.02:
            break
        if pn in seen_n and count >= 2:
            continue
        seen_n.add(pn)

        per_time = min(80.0, phase1_end - time.monotonic() - 30)
        if per_time < 20:
            break
        p_best, sc_best = _lbfgs_continuation(p, time.monotonic() + per_time, rng, max_wall_time=per_time)
        update_best(p_best, sc_best, f"lbfgs_{label}")
        count += 1

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Adam polishing (300s) ---
    phase2_end = min(deadline, t_start + 730.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        per_dl = min(phase2_end - 10, time.monotonic() + 250.0)
        p_best, sc_best = _adam_simplex(best_seq, per_dl, rng, lr=0.05, beta_frac=0.008, n_max_iter=40000)
        update_best(p_best, sc_best, "adam1")

        if time.monotonic() < phase2_end - 30:
            per_dl2 = min(phase2_end - 10, time.monotonic() + 100.0)
            p_best2, sc_best2 = _adam_simplex(best_seq, per_dl2, rng, lr=0.02, beta_frac=0.003, n_max_iter=20000)
            update_best(p_best2, sc_best2, "adam2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Active-set subgradient (200s) ---
    phase3_end = min(deadline, t_start + 930.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 10, time.monotonic() + 180.0)
        p_best, sc_best = _active_set_descent(best_seq, per_dl, rng, n_max_iter=50000)
        update_best(p_best, sc_best, "active_set")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Multi-restart L-BFGS (300s) ---
    phase4_end = min(deadline, t_start + 1230.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = min(70.0, (phase4_end - time.monotonic() - 20) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break
            noise_levels = [0.05, 0.1, 0.15, 0.3]
            z = _logit_fn(best_seq)
            z = z + noise_levels[i % len(noise_levels)] * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(phase4_end - 15, time.monotonic() + per_restart)
            p_best, sc_best = _lbfgs_continuation(p_start, per_dl, rng, max_wall_time=per_restart)
            update_best(p_best, sc_best, f"restart{i}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Try different length (200s) ---
    phase5_end = min(deadline, t_start + 1430.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)
        target_n = 2048 if n_curr < 2048 else (4096 if n_curr < 4096 else n_curr)
        if target_n != n_curr:
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase5_end - 60:
                per_dl = min(phase5_end - 20, time.monotonic() + 150.0)
                p_best, sc_best = _lbfgs_continuation(p_interp, per_dl, rng, max_wall_time=150.0)
                update_best(p_best, sc_best, f"lbfgs_n{target_n}")

                if time.monotonic() < phase5_end - 30:
                    per_dl2 = min(phase5_end - 15, time.monotonic() + 80.0)
                    p_best2, sc_best2 = _adam_simplex(p_best, per_dl2, rng, lr=0.02, beta_frac=0.004, n_max_iter=15000)
                    update_best(p_best2, sc_best2, f"adam_n{target_n}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final polish with remaining time ---
    if time.monotonic() < deadline - 10 and best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 60:
            per_dl = time.monotonic() + min(remaining * 0.6, 120.0)
            p_best, sc_best = _adam_simplex(best_seq, per_dl, rng, lr=0.005, beta_frac=0.002, n_max_iter=20000)
            update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < deadline - 10:
            remaining2 = deadline - time.monotonic() - 10
            if remaining2 > 30:
                per_dl2 = time.monotonic() + min(remaining2, 60.0)
                p_best2, sc_best2 = _lbfgs_continuation(best_seq, per_dl2, rng, max_wall_time=min(remaining2, 60.0))
                update_best(p_best2, sc_best2, "final_lbfgs")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(256, dtype=np.float64) / 256.0

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned: {sc_eval}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(256, dtype=np.float64) / 256.0
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
