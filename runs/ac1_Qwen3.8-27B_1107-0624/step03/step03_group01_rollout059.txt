```python
"""
Multi-stage optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines Fourier parameterization, adaptive-beta gradient descent,
L-BFGS with fine continuation, and peak-targeted local search.
Key insight: optimal sequence is asymmetric, reducing center peak
while keeping off-center peaks controlled.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _compute_grad_z(a, w, n):
    W = float(np.dot(w, a * 0 + _full_convolve(a, a)))
    # Actually W = dot(w, c) where c = conv(a,a)
    # But we need h_j = sum_k w_k * a[k-j]
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return h, W


def _grad_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _obj_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta)
    return 2.0 * n * (cm + beta * float(np.log(np.sum(np.exp((c - cm) / beta)))))


def _adaptive_beta(c):
    """Choose beta based on peak structure."""
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(20, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.05 * cm:
        # Peaks are balanced, use moderate beta to see all of them
        return max(1e-10, 0.1 * cm)
    elif gap > 0.5 * cm:
        # One peak dominates, use small beta to focus on it
        return max(1e-10, 0.01 * cm)
    else:
        # Intermediate, use half the gap
        return max(1e-10, 0.5 * gap)


def _run_adaptive_gd(a_init, deadline, rng, lr_init=0.3, max_iter=50000):
    """Custom gradient descent with adaptive beta and Adam moments."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        beta = _adaptive_beta(c)
        w, cm = _compute_weights(c, beta)
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Adaptive learning rate: scale by current peak magnitude
        peak_scale = cm / max(1e-15, _score_norm_fn(a_init) / (2 * n))
        lr_adapt = lr * min(1.0, max(0.1, peak_scale))

        z = z - lr_adapt * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Decay learning rate slowly
        if t % 1000 == 0:
            lr = max(0.01, lr * 0.95)

        # Occasionally try a "kick" if stuck
        if t % 5000 == 0:
            z = z + 0.05 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _run_lbfgs_fine(a_init, deadline, rng, beta_list=None):

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

                     0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                     0.0002, 0.00015, 0.0001]

        if time.monotonic() > deadline:

        # Allocate more time to smaller betas
        remaining_betas = n_betas - idx
        time_per_beta = (deadline - time.monotonic()) / max(1, remaining_betas)
        time_per_beta = min(time_per_beta, 120.0)
        level_deadline = time.monotonic() + time_per_beta

        try:
            result = minimize(
                _obj_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_grad_objective,
                options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-12,

        except Exception:

def _fourier_optimize(n, deadline, rng):
    """Optimize a smooth Fourier-parameterized profile."""
    u = 2.0 * np.linspace(0, 1, n) - 1.0  # in [-1, 1]
    # Basis: 1, u, u^2, u^3, u^4, u^5 (or cos terms)
    K = 5
    basis = np.ones((n, K + 1))
    for k in range(1, K + 1):
        basis[:, k] = u ** k

    best_score = np.inf
    best_alpha = np.zeros(K + 1)

    # Try multiple initial alpha values
    inits = [
        np.zeros(K + 1),
        np.array([0, 1.0, 0, 0, 0, 0]),
        np.array([0, -1.0, 0, 0, 0, 0]),
        np.array([0, 0, 1.0, 0, 0, 0]),
        np.array([0, 1.0, 0.5, 0, 0, 0]),
        np.array([0, -1.0, 0.5, 0, 0, 0]),
        np.array([0, 2.0, -1.0, 0, 0, 0]),
        np.array([0, -2.0, -1.0, 0, 0, 0]),
        np.array([0, 0.5, 0.3, 0.1, 0, 0]),
        np.array([0, -0.5, 0.3, -0.1, 0, 0]),
    ]

    def fourier_score(alpha):
        log_a = basis @ alpha
        log_a = log_a - np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s < 1e-10:
            return np.inf
        a = a / s
        return _score_norm_fn(a)

    for init in inits:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(fourier_score, init, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.fun < best_score:
                best_score = res.fun
                best_alpha = res.x.copy()
        except Exception:
            pass

    # Construct the sequence
    log_a = basis @ best_alpha
    log_a = log_a - np.max(log_a)
    a = np.exp(log_a)
    a = a / a.sum()
    return a, best_score


def _peak_surgery(a, deadline, rng):
    """Targeted reduction of the highest peaks."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for outer in range(50):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        # Find contributing indices
        valid_i = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        contributions = a_cur[valid_i] * a_cur[k_star - valid_i]
        top_idx = valid_i[np.argsort(contributions)[::-1][:10]]

        # Try reducing each top contributor
        improved = False
        for idx in top_idx:
            # Try small reductions
            for delta in [0.001, 0.002, 0.005, 0.01, 0.02]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try adding to small values (redistribute)
            small_idx = np.argsort(a_cur)[:5]
            for idx in small_idx:
                for delta in [0.001, 0.005, 0.01]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    # Linear ramps
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Quadratic
    for a1, a2 in [(1.0, 0.5), (2.0, -0.3), (1.5, 0.2), (-1.0, 0.5)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    # Cosine power
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization combining Fourier parameterization,
    adaptive-beta gradient descent, and L-BFGS with fine continuation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_parent = len(parent) if parent is not None else 2048
    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, norm_score={sc_parent:.10f}, actual_score={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_profiles(2048)[5]  # linear slope 1.5
        update_best(parent, _score_norm_fn(parent), "default")

    # --- Phase 1: Fourier + profile exploration (80s) ---
    phase1_end = min(deadline, t_start + 80.0)
    print("Phase 1: Fourier + profile exploration")

    for n_try in [n_parent]:
        sc_fourier, _ = _fourier_optimize(n_try, phase1_end, rng)
        a_fourier = _fourier_optimize(n_try, phase1_end, rng)[0]
        sc_f = _score_norm_fn(a_fourier)
        update_best(a_fourier, sc_f, f"fourier_n{n_try}")

        # Also try profiles
        for idx, p in enumerate(_make_profiles(n_try)[:6]):
            sc_p = _score_norm_fn(p)
            update_best(p, sc_p, f"prof{idx}_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: L-BFGS fine continuation (700s) ---
    phase2_end = min(deadline, t_start + 780.0)
    print("Phase 2: L-BFGS fine continuation")

    n_best = len(best_seq)
    a_start = best_seq.copy()

    # Multi-restart L-BFGS
    n_restarts = 4
    per_restart = (phase2_end - time.monotonic()) / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > phase2_end - 10:
            break

        if r == 0:
            a_cur = a_start.copy()
        else:
            # Perturb the best so far
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise_scale = 0.05 * (r + 1)
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)

        per_dl = min(phase2_end - 5, time.monotonic() + per_restart)
        label = f"lbfgs_r{r}" if r == 0 else f"lbfgs_restart{r}"
        p_best, sc_best = _run_lbfgs_fine(a_cur, per_dl, rng)
        update_best(p_best, sc_best, label)

    print(f"  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adaptive gradient descent (300s) ---
    phase3_end = min(deadline, t_start + 1080.0)
    print("Phase 3: Adaptive GD")

    if time.monotonic() < phase3_end and best_seq is not None:
        per_dl = min(phase3_end - 5, time.monotonic() + 280.0)
        p_best, sc_best = _run_adaptive_gd(best_seq, per_dl, rng, lr_init=0.15)
        update_best(p_best, sc_best, "adaptive_gd")

        # Second round with different params
        if time.monotonic() < phase3_end - 30:
            per_dl2 = min(phase3_end - 5, time.monotonic() + 200.0)
            p_best2, sc_best2 = _run_adaptive_gd(best_seq, per_dl2, rng, lr_init=0.05)
            update_best(p_best2, sc_best2, "adaptive_gd_slow")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Peak surgery (150s) ---
    phase4_end = min(deadline, t_start + 1230.0)
    print("Phase 4: Peak surgery")

    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = min(phase4_end - 5, time.monotonic() + 130.0)
        p_best, sc_best = _peak_surgery(best_seq, per_dl, rng)
        update_best(p_best, sc_best, "peak_surgery")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Try n=4096 (200s) ---
    phase5_end = min(deadline, t_start + 1430.0)
    print("Phase 5: Try n=4096")

    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr <= 2048:
            n_new = 4096
            p_interp = _interp_profile(best_seq, n_new)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{n_new}")

            if time.monotonic() < phase5_end - 60:
                per_dl = min(phase5_end - 30, time.monotonic() + 150.0)
                p_best, sc_best = _run_lbfgs_fine(p_interp, per_dl, rng)
                update_best(p_best, sc_best, f"lbfgs_n{n_new}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final fine polish (remaining) ---
    phase6_end = deadline
    print("Phase 6: Final fine polish")

    if time.monotonic() < phase6_end and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        # Very fine L-BFGS with the smallest beta
        beta_final = 1e-5 * max_c_final
        per_dl = min(phase6_end - 10, time.monotonic() + (phase6_end - time.monotonic()) * 0.6)
        p_best, sc_best = _run_lbfgs_single_beta(best_seq, beta_final, n_final, per_dl, maxiter=10000)
        update_best(p_best, sc_best, "final_lbfgs_tiny_beta")

        # Final Adam polish
        if time.monotonic() < phase6_end - 10:
            per_dl2 = min(phase6_end - 5, time.monotonic() + (phase6_end - time.monotonic()) * 0.4)
            p_best2, sc_best2 = _run_adaptive_gd(best_seq, per_dl2, rng, lr_init=0.03)
            update_best(p_best2, sc_best2, "final_adam")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(256)[2]

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator
        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(256)[2]
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
```
