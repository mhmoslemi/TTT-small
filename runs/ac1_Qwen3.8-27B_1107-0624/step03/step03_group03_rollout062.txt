Let me be more specific about the spectral domain method:

The key insight: in the frequency domain, the autocorrelation c has DFT |A|^2. The peak of c is determined by the "coherent" addition of the frequency components. To reduce the peak, we want to:
1. Reduce the magnitude of A at frequencies that contribute coherently to the peak
2. This can be done by multiplying A[k] by a factor that reduces the large values

Specifically, if the peak is at location p, then:
c[p] = (1/N) * sum_k |A[k]|^2 * exp(2πikp/N)

To reduce c[p], we want to reduce the contribution of the terms that add constructively. The "constructive" terms are those where exp(2πikp/N) is close to 1, i.e., k is close to 0 mod N/p.

A simple approach: multiply A[k] by exp(-α * |A[k]|^2 * Re(exp(2πikp/N))) for k ≠ 0. This reduces the amplitude at frequencies that contribute most to the peak.

But this might not maintain non-negativity of a. So we need to project back.

Actually, a simpler spectral approach:
1. Compute A = DFT(a)
2. Compute target spectral shape: |A_target[k]|^2 = (1/M_target) * |A[k]|^2 * (1 - β * cos(2πkp/N))
   This reduces the amplitude at frequencies that align with the peak direction.
3. Set A_new[k] = A[k] * sqrt(|A_target[k]|^2 / |A[k]|^2) = A[k] * sqrt(1 - β * cos(2πkp/N))
   (preserving the phase of A)
4. Inverse DFT to get a_new
5. Project to non-negative simplex

"""
Optimization combining spectral domain methods, LSE gradient, LP refinement,
and pair-based coordinate descent to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return _score_normalized(a / s)

def _project_simplex(v):
    """Project v onto the probability simplex (sum=1, all >= 0)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    # Simple projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _spectral_step(a, n_peaks=5, alpha=0.1, deadline=None):
    """One spectral domain step to reduce peak."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    c = _fft_convolve(a)
    
    # Find top peaks
    peak_indices = np.argsort(c)[::-1][:n_peaks]
    
    # For each peak, compute the "direction" in frequency domain
    # and reduce amplitude in that direction
    A_new = A.copy()
    for p in peak_indices:
        # Phase alignment for peak at position p
        phase = np.exp(2j * np.pi * np.arange(N//2+1) * p / N)
        # Reduce amplitude where phase aligns with peak
        reduction = 1.0 - alpha * np.abs(np.conj(A) * phase).real / (np.abs(A).max() + 1e-30)
        reduction = np.clip(reduction, 0.5, 1.0)
        A_new *= np.sqrt(reduction)
    
    # Inverse transform
    a_new = np.fft.irfft(A_new, N)[:n]
    # Project to simplex
    a_new = _project_simplex(a_new)
    return a_new

def _spectral_optimize(a, deadline, max_iter=5000):
    """Iterative spectral domain optimization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _score_normalized(a)
    alpha = 0.05
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a_new = _spectral_step(a, n_peaks=5, alpha=alpha)
        sc = _score_normalized(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                alpha *= 0.9
            if no_imp > 500:
                a = best_a.copy()
                alpha = 0.05
                no_imp = 0
    
    return best_a

def _lse_optimize_v2(a, deadline, T_start=0.03, T_end=1e-6, lr=0.15, max_iter=50000):

    a = np.asarray(a, dtype=np.float64).copy()

    z -= float(np.mean(z))

    best_score = _score_normalized(a)

            break

        else:
            w = ex / s

                z = np.log(a)

                lr = base_lr
                no_imp = 0
        
        a = a_new

    return best_a

def _lp_spread_v2(a, M, top_k=30):
    """LP to find spreading direction with reduced constraints."""
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _lp_refine_v2(a, deadline, max_iter=3000, inflate_factor=1.001):
    """LP refinement with line search."""
    a = np.asarray(a, dtype=np.float64).copy()

    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    inflate = inflate_factor
    
    for it in range(max_iter):

        M = float(c.max())
        M_lp = M * inflate
        g = _lp_spread_v2(a, M_lp)
        
        if g is None:
            inflate = min(1.01, inflate * 1.1)
            no_imp += 3
            if no_imp > 200:
                break
            continue
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp > 300:
                a = best_a.copy()
                no_imp = 0
                inflate = inflate_factor
            continue
        
        g = g / sum_g
        
        # Line search
        best_t = 0.0
        best_sc = best_score
        for t_try in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_try = a + t_try * (g - a)
            a_try = _project_simplex(a_try)

def _pair_optimization(a, deadline, max_iter=5000):
    """Optimize the split of each symmetric pair."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _score_normalized(a)
    rng = np.random.default_rng(999)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Pick a random pair
        i = rng.integers(0, n // 2)
        j = n - 1 - i
        
        # Current values
        ai = a[i]
        aj = a[j]
        total = ai + aj
        
        if total < 1e-15:
            continue
        
        # Try different splits
        best_split = 0.5  # current ratio
        best_sc = _score_normalized(a)
        
        # Try a few splits
        for frac in [0.1, 0.2, 0.3, 0.35, 0.4, 0.45, 0.55, 0.6, 0.65, 0.7, 0.8, 0.9]:
            a_try = a.copy()
            a_try[i] = total * frac
            a_try[j] = total * (1 - frac)
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_split = frac
        
        if best_split != 0.5:
            a[i] = total * best_split
            a[j] = total * (1 - best_split)
            sc = _score_normalized(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
    
    return best_a

"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines antisymmetric push, top-K gradient, spectral methods, LP refinement,
and stochastic search with adaptive restart strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return _score_normalized(a / s)


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _lse_adam_step(a, z, state, T, lr):
    """One Adam step in logit space using log-sum-exp."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state


def _lse_optimize(a, deadline, T_start=0.04, T_end=1e-6, lr=0.12, max_iter=50000):
    """LSE gradient optimization with adaptive restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

    best_a = a.copy()
    best_score = _score_normalized(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    rng = np.random.default_rng(888)

    if max_iter > 1:
        T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    else:
        T_ratio = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a, z, state = _lse_adam_step(a, z, state, T, lr)

        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue

        sc = _score_normalized(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.02, lr * 0.93)
            if no_imp > max(4000, 6 * n):
                a = best_a * (1.0 + 0.06 * rng.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0

        T = max(T * T_ratio, T_end)

    return best_a


def _make_conv_matrix_reduced(a, M, top_k=40):
    """Create reduced constraint matrix for LP."""
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    return A, b


def _lp_spread(a, M, top_k=40):
    """LP to find spreading direction."""
    n = len(a)
    A, b = _make_conv_matrix_reduced(a, M, top_k)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=3000, inflate_start=1.002):
    """LP-based refinement with adaptive inflation and line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        g = _lp_spread(a, M_lp, top_k=40)

        if g is None:
            inflate = min(1.02, inflate * 1.15)
            no_imp += 3
            if no_imp > 300:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 50 == 0:
                inflate = min(1.02, inflate * 1.05)
            if no_imp > 400:
                a = best_a.copy()
                no_imp = 0
                inflate = inflate_start
            continue

        g = g / sum_g

        # Multi-point line search
        best_t = 0.0
        best_sc = best_score
        t_candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4]
        for t_try in t_candidates:
            a_try = a + t_try * (g - a)
            a_try = _project_simplex(a_try)
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t_try

        if best_t == 0.0:
            no_imp += 1
            inflate = min(1.02, inflate * 1.03)
            if no_imp > max(300, n):
                a = best_a.copy()
                no_imp = 0
                inflate = inflate_start
            continue

        a_new = a + best_t * (g - a)
        a_new = _project_simplex(a_new)
        sc_new = _score_normalized(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if inflate > 1.001:
                inflate = max(1.0008, inflate * 0.92)
        else:
            no_imp += 1
            inflate = min(1.02, inflate * 1.05)
            if no_imp > max(300, n):
                a = best_a.copy()
                no_imp = 0
                inflate = inflate_start

    return best_a


def _stochastic_optimize(a, deadline, max_iter=30000, n_perturb=8, base_scale=0.12):
    """Stochastic local search with adaptive scale."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    best_a = a.copy()
    best_score = _score_normalized(a)
    rng = np.random.default_rng(4242)
    scale = base_scale
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        idx = rng.choice(n, size=min(n_perturb, n), replace=False)
        a_pert = a.copy()
        perturb = 1.0 + scale * rng.normal(size=len(idx))
        a_pert[idx] = np.maximum(a_pert[idx] * perturb, 0)
        a_pert = _project_simplex(a_pert)

        sc = _score_normalized(a_pert)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_pert.copy()
            a = a_pert
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                scale = max(0.005, scale * 0.98)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                scale = base_scale
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, weight=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_normalized(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    n_base = 1024 if parent is None else parent.size

    # Phase 1: Quick multi-start (120s)
    print("\n=== Phase 1: Multi-start evaluation ===")
    end1 = t_start + 120

    if parent is not None:
        candidates.append(("parent", parent.copy()))

    candidates.append(("raised_cos", _make_raised_cosine(n_base)))
    candidates.append(("flat_edges", _make_flat_edges(n_base, 0.4)))

    # Add a perturbed parent
    if parent is not None:
        p_pert = parent * (1.0 + 0.1 * rng.normal(size=n_base))
        p_pert = np.maximum(p_pert, 0)
        if p_pert.sum() > 1e-10:
            candidates.append(("parent_pert", p_pert / p_pert.sum()))

    for name, a0 in candidates:
