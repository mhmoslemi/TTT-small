    """Compute smooth max of convolve(a,a) and its gradient."""
    b = fft_convolve(a, a)
    smooth_max_val = beta * log(sum(exp(b/beta)))
    weights = exp(b/beta) / sum(exp(b/beta))
    return smooth_max_val, weights

def objective_and_grad(z, beta):
    """z is the log-space parameter, a = softmax(z)."""
    a = softmax(z)
    val, weights = smooth_max_obj(a, beta)
    # Gradient w.r.t. z: da/dz = a (I - a a^T)
    # dval/da = 2 * cross_corr(weights, a_reversed) / n ... 
    # Actually, dval/da = 2 * (P * a) where P is the "peak operator"
    # More precisely: dval/da[j] = 2 * sum_k weights[k] * db[k]/da[j]
    # db[k]/da[j] = a[k-j] (with boundary handling)
    # So dval/da = 2 * cross_correlation(weights, a)
    # And dz/da gives the Jacobian of softmax
    
    # In log-space: dval/dz = (dval/da / a) - mean(dval/da / a * a)
    # (This is the standard softmax gradient transformation)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    c = np.maximum(c, 0.0)
    return c

    c = np.asarray(c, dtype=np.float64)

        w = np.zeros_like(c)
        w[mask] = 1.0 / max(cnt, 1)

```python
"""
Multi-strategy optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) L-BFGS with multi-scale smooth max in log-space,
(2) LP-based refinement, (3) multi-restart with structured perturbations,
(4) multi-scale escalation (increasing n).
"""
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    c = np.maximum(c, 0.0)
    return c

def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        mask = c >= mval - 1e-10
        cnt = max(int(mask.sum()), 1)
        w = np.zeros_like(c)
        w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _softmax_from_z(z, floor=1e-30):
    z = np.clip(z.astype(np.float64), -50.0, 50.0)
    z = z - float(z.max())
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    return p / s

def _grad_smooth_max_conv(p, w):
    """
    Compute gradient of smooth_max(convolve(p,p)) w.r.t. p.
    p: array of length n
    w: weights from smooth max, length 2n-1
    Returns: gradient array of length n
    """
    n = p.size
    N = _next_pow2(2 * n - 1)
    A = np.zeros(N)
    A[:n] = p
    w_pad = np.zeros(N)
    w_pad[:2*n-1] = w
    # grad_A = 2 * real(ifft(fft(w_pad) * conj(fft(A))))
    fa = np.fft.fft(A, N)
    fw = np.fft.fft(w_pad, N)
    grad_A = 2.0 * np.real(np.fft.ifft(fw * np.conj(fa), N))
    return grad_A[:n]

def _score(p):
    n = p.size
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _lp_solve(a, M):
    """Solve LP: max sum(b) s.t. convolve(a,b) <= M, b >= 0."""
    n = len(a)
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _multi_scale_lbfgs(p0, deadline, rng, max_beta_iters=15):
    """
    Multi-scale L-BFGS optimization.
    Starts with large beta (smooth), decreases beta over iterations.
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= z.mean()

    c = _full_convolve(p0)
    best_p = p0.copy()
    best_score = float(2.0 * n * c.max())

    beta = 0.1 * c.max()
    
    for it in range(max_beta_iters):
        if time.monotonic() > deadline - 30:
            break
            
        def obj_grad(z_in):
            p = _softmax_from_z(z_in)
            c_val = _full_convolve(p)
            sm_val, w = _smooth_max_and_weights(c_val, beta)
            obj = 2.0 * n * sm_val
            ga = _grad_smooth_max_conv(p, w)
            dot_ga_a = float(np.dot(ga, p))
            grad_z = p * (ga - dot_ga_a)
            # Clip gradient for stability
            grad_z = np.clip(grad_z, -10.0, 10.0)
            if not np.all(np.isfinite(obj)) or not np.all(np.isfinite(grad_z)):
                return np.inf, np.zeros_like(z_in)
            return obj, grad_z

        # Run L-BFGS
        maxfun = max(500, min(20000, int(3000000 / n)))
        res = minimize(obj_grad, z, jac=True, method='L-BFGS-B',
                       options={'maxiter': min(500, maxfun // 10),
                                'maxls': 20,
                                'ftol': 1e-14, 'gtol': 1e-8,
                                'maxfun': maxfun, 'iprint': -1})

        if np.all(np.isfinite(res.x)):
            z = res.x.astype(np.float64)
            p = _softmax_from_z(z)
            sc = _score(p)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p.copy()

        # Evaluate exact score (not smooth max)
        p_check = _softmax_from_z(z)
        sc_check = _score(p_check)
        if np.isfinite(sc_check) and sc_check < best_score:
            best_score = sc_check
            best_p = p_check.copy()

        # Decrease beta
        beta *= 0.65
        if beta < 1e-10 * c.max():
            break

    return best_p, best_score

def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=300):
    """Adam optimizer in logit space with smooth max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-4, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        ga = _grad_smooth_max_conv(p, w)
        dot_ga_a = float(np.dot(ga, p))
        grad_z = p * (ga - dot_ga_a)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m_buf.fill(0.0)
            v_buf.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t_adam)
        vh = v_buf / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_buf.fill(0.0)
            v_buf.fill(0.0)
            t_adam = 0
            continue

        z = np
