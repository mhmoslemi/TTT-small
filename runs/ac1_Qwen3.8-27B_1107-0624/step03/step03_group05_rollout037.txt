def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]

"""
Multi-phase step function optimizer using L-BFGS-B with adaptive log-sum-exp
continuation, iterative LP spreading, multi-length exploration, and diverse
initializations. Key improvements: quasi-Newton optimization for faster
convergence, tighter LP iterations with adaptive inflation, and exploration
across multiple sequence lengths (768-2048).

import numpy as np
import time
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M_target):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M_target)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _compute_grad_a(a, w, beta):
    """Compute grad_a[i] = (2/beta) * sum_j w[j] * a[j-i] (valid indices)."""
    n = len(a)
    L = _next_pow2(2 * (2 * n - 1) + n - 1)
    fw = np.fft.rfft(w, L)
    fa = np.fft.rfft(a, L)
    R = np.fft.irfft(np.conj(fw) * fa, L)[:n]
    return 2.0 / beta * R


def _lgbfgs_optimize(a_init, deadline, num_rounds=6, iters_per_round=300):
    """L-BFGS-B with log-sum-exp continuation on softmax parameterization."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a) - np.log(a).mean()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = 0.03 * float(c.max())
    if beta < 1e-10:
        beta = 1e-6

    for rnd in range(num_rounds):
        if time.monotonic() > deadline:
            break

        beta_cur = beta

        def objective(z_vec):
            e = np.exp(z_vec - z_vec.max())
            a_vec = e / e.sum()
            c_vec = _fft_convolve(a_vec)
            m = float(c_vec.max())
            shifted = (c_vec - m) / beta_cur
            ex = np.exp(shifted)
            s_ex = float(ex.sum())
            if s_ex < 1e-30:
                s_ex = 1e-30
            lse = m + beta_cur * np.log(s_ex)
            w = ex / s_ex
            grad_a = _compute_grad_a(a_vec, w, beta_cur)
            dot = float(np.dot(a_vec, grad_a))
            grad_z = a_vec * (grad_a - dot)
            return float(lse), grad_z

        try:
            res = minimize(objective, z, method='L-BFGS-B', jac=True,
                          options={'maxiter': iters_per_round,
                                   'maxfun': iters_per_round * 4,
                                   'ftol': 1e-16, 'gtol': 1e-12})
            z = res.x.copy()
        except Exception as e:
            print(f"  L-BFGS round {rnd} error: {e}")
            break

        e = np.exp(z - z.max())
        a_new = e / e.sum()
        c_new = _fft_convolve(a_new)
        sc = float(2.0 * n * c_new.max())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        print(f"    L-BFGS round {rnd+1}/{num_rounds}: beta={beta_cur:.6f}, "
              f"score={best_score:.10f}", flush=True)

        beta = max(beta * 0.5, 1e-10)

    return best_a


def _lp_refine(a_init, deadline, max_iter=20000, inflate=1.001, blend=0.25):
    """Iterative LP spreading with adaptive inflation and blending."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        g = _lp_spread(a, M * inflate_cur)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 15 == 0:
                inflate_cur = min(1.03, inflate_cur * 1.04)
            if no_imp > 600:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0003, inflate_cur * 0.97)
            blend_cur = min(0.4, blend_cur * 1.03)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.015, inflate_cur * 0.96)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _adam_logit_polish(a_init, deadline, max_iter=20000, lr=0.04):
    """Adam optimization on logit parameterization with adaptive smooth max."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a) - np.log(a).mean()

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.04 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c_arr = _fft_convolve(p)
        m_val = float(c_arr.max())
        shifted = (c_arr - m_val) / beta
        ex = np.exp(shifted)
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c_arr)
            ex[int(np.argmax(c_arr))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        grad_a = _compute_grad_a(p, w, beta)
        dot = float(np.dot(p, grad_a))
        gz = p * (grad_a - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c_arr = _fft_convolve(a)
        score = float(2.0 * n * c_arr.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 100 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(1500, 3 * n):
            rng_r = np.random.default_rng(it * 7919 + n)
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-8, 0.04 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _active_set_newton(a_init, deadline, max_iter=5000):
    """Newton-like method focusing on the active (top) lags."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        # Find top lags (active set)
        threshold = M * 0.995
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute gradient emphasizing active lags
        w = np.zeros_like(c)
        w[active] = 1.0
        w = w / w.sum()

        grad_a = _compute_grad_a(a, w, 1.0)
        dot = float(np.dot(a, grad_a))
        grad_simplex = a * (grad_a - dot)

        # Line search
        best_step = 0.0
        best_sc = best_score
        for step in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
            a_try = a - step * grad_simplex
            a_try = _project_simplex(a_try)
            sc = _score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_step = step

        if best_step == 0.0:
            no_imp += 1
            if no_imp > 500:
                break
            continue

        a = a - best_step * grad_simplex
        a = _project_simplex(a)
        sc = _score(a)

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

    return best_a


def _block_swap_optimize(a_init, deadline, max_iter=5000):
    """Random block swaps to escape local minima."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        block_size = int(rng.integers(max(2, n // 32), max(3, n // 8)))
        i = int(rng.integers(0, n - block_size))
        j = int(rng.integers(0, n - block_size))
        if abs(i - j) < block_size:
            continue

        a_new = a.copy()
        a_new[i:i + block_size], a_new[j:j + block_size] = \
            a_new[j:j + block_size].copy(), a_new[i:i + block_size].copy()
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break

    return best_a

        if time.monotonic() > deadline - 120:
            break

        frac = time_alloc.get(n_t, 0.15) / total_alloc
        phase_budget = remaining_budget * frac
        phase_budget = max(30.0, min(phase_budget, deadline - time.monotonic() - 80.0))

        if phase_budget < 20:
            continue

        print(f"\n=== Length {n_t}: budget={phase_budget:.0f}s ===", flush=True)

        # Initialize from parent (interpolated)
        a_init = _interpolate_profile(parent, n_t)
        sc_init = _score(a_init)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()
        print(f"  Init score: {sc_init:.10f}", flush=True)

        a_best_n, sc_best_n = _full_pipeline(a_init, n_t, deadline, phase_budget)
        if sc_best_n < best_score:
            best_score = sc_best_n
            best_seq = a_best_n.copy()
            print(f"  ** New best at n={n_t}: {sc_best_n:.10f}", flush=True)

    # Phase 2: Diverse initializations at the best length
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = best_seq.size
        print(f"\n=== Phase 2: Diverse inits at n={n_best} ===", flush=True)

        inits = []
        # U-shaped profiles with different curvatures
        for alpha in [1.5, 2.5, 4.0, 6.0]:
            inits.append(_make_u_profile(n_best, alpha))
        # Linear profiles
        for eta in [1.0, 2.0, 3.5]:
            inits.append(_make_linear_profile(n_best, eta))
        # Bimodal (two peaks)
        t = np.linspace(0, 1, n_best)
        for c1, c2, s1, s2 in [(0.1, 0.9, 0.12, 0.12),
                                (0.15, 0.85, 0.15, 0.15),
                                (0.05, 0.95, 0.08, 0.08)]:
            a_gm = np.exp(-0.5*((t-c1)/s1)**2) + np.exp(-0.5*((t-c2)/s2)**2)
            a_gm = np.maximum(a_gm, 1e-12)
            inits.append(a_gm / a_gm.sum())
        # Perturbed best
        for scale in [0.1, 0.2, 0.3]:
            a_pert = best_seq * (1.0 + scale * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            inits.append(a_pert)

        # Allocate time for diverse inits
        div_budget = deadline - time.monotonic() - 80.0
        per_init = max(15.0, div_budget / len(inits))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick optimization
            sub_budget = min(per_init, deadline - time.monotonic() - 40.0)
            if sub_budget < 10:
                break
            a_res, sc_res = _full_pipeline(a0, n_best, deadline, sub_budget)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  Init {idx+1}: ** New best: {sc_res:.10f}", flush=True)
            else:
                print(f"  Init {idx+1}: {sc_res:.10f} (best: {best_score:.10f})", flush=True)

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        final_budget = deadline - time.monotonic() - 20.0
        print(f"\n=== Phase 3: Final polish, budget={final_budget:.0f}s ===", flush=True)

        # LP with very tight parameters
        t = time.monotonic()
        b = min(final_budget * 0.3, 100.0)
        if b > 10:
            a_lp = _lp_refine(best_seq, t + b, max_iter=20000,
                              inflate=1.0005, blend=0.15)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  Final LP: {sc:.10f}", flush=True)

        # L-BFGS-B with very small beta
        t = time.monotonic()
        b = min(final_budget * 0.3, 80.0)
        if b > 10:
            a_lb = _lgbfgs_optimize(best_seq, t + b,
                                    num_rounds=4, iters_per_round=500)
            sc = _score(a_lb)
            if sc < best_score:
                best_score = sc
                best_seq = a_lb.copy()
            print(f"  Final L-BFGS-B: {sc:.10f}", flush=True)

        # Adam
        t = time.monotonic()
        b = min(final_budget * 0.2, 60.0)
        if b > 5:
            a_ad = _adam_logit_polish(best_seq, t + b, max_iter=15000, lr=0.025)
            sc = _score(a_ad)
            if sc < best_score:
                best_score = sc
                best_seq = a_ad.copy()
            print(f"  Final Adam: {sc:.10f}", flush=True)

        # Active set
        t = time.monotonic()
        b = min(final_budget * 0.15, 40.0)
        if b > 5:
            a_an = _active_set_newton(best_seq, t + b, max_iter=5000)
            sc = _score(a_an)
            if sc < best_score:
                best_score = sc
                best_seq = a_an.copy()
            print(f"  Final ActiveSet: {sc:.10f}", flush=True)

        # One more LP
        t = time.monotonic()
        b = min(final_budget * 0.1, 30.0)
        if b > 3:
            a_lp2 = _lp_refine(best_seq, t + b, max_iter=5000,
                               inflate=1.0003, blend=0.1)
            sc = _score(a_lp2)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp2.copy()
            print(f"  Final LP2: {sc:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(768, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(768, 1.5)

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"  Evaluation error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
