def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""

    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(a):
    """Build matrix A where (A @ b) = conv(a, b) (full linear convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max 1^T b s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = scipy.optimize.linprog(
        c_obj, A_ub=A, b_ub=b_ub,
        bounds=[(0, None)] * n, method='highs',
        options={'presolve': True, 'time_limit': time_limit}
    )
    if res.success:
        return res.x

def soft_max_weights(c, beta):
    """Compute soft-max weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    Otherwise, use top 'active_fractions' constraints by value.

        top_k = min(active_fractions, 2*n-1)
        active_indices = np.argsort(c)[-top_k:]

                                  bounds=[(0, None)]*n, method='highs',

        return None

    full_conv = A_full @ b

"""
Hybrid LP best-response + gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding (from SOTA description) combined with logit-space
gradient polishing. Explores multiple sequence lengths.

import numpy as np
import time
import scipy.optimize


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = full_convolve_fft(a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(a):
    """Build matrix A where (A @ b) = conv(a, b) (full linear convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def solve_lp(a, M, time_limit=15):
    """Solve LP: max 1^T b s.t. conv(a,b) <= M, b >= 0. Returns b array or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = scipy.optimize.linprog(
        c_obj, A_ub=A, b_ub=b_ub,
        bounds=[(0, None)] * n, method='highs',
        options={'presolve': True, 'time_limit': time_limit}
    )
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-10:
        a = np.ones(n) * target_sum / n
    else:
        a = a * target_sum / ss

    best_a = a.copy()
    best_sc = compute_score(a)
    print(f"    LP start: n={n}, score={best_sc:.10f}")

    t = t0
    stall = 0

        if time.monotonic() > deadline:
            break

        c = full_convolve_fft(a)
        M = float(c.max())

        b = solve_lp(a, M, time_limit=12)
        lp_time = time.monotonic() - t_start

        if b is None:
            print(f"    LP failed at iter {it}, time={lp_time:.1f}s")
            break

        b_sum = float(b.sum())

            print(f"    LP returned near-zero at iter {it}")
            break
        b = b * target_sum / b_sum

        # Find step size via line search or schedule
        if use_line_search:
            chosen_t = t
            current_sc = compute_score(a)
            for trial_t in [t, t * 0.5, t * 0.25, t * 0.125, 0.05]:
                new_a = (1.0 - trial_t) * a + trial_t * b
                new_a = np.maximum(new_a, 0.0)
                nss = new_a.sum()
                if nss < 1e-10:
                    continue
                new_a = new_a * target_sum / nss
                sc_trial = compute_score(new_a)
                if sc_trial < current_sc - 1e-12:
                    chosen_t = trial_t
                    break
        else:
            chosen_t = t

        new_a = (1.0 - chosen_t) * a + chosen_t * b
        new_a = np.maximum(new_a, 0.0)
        nss = new_a.sum()
        if nss < 1e-10:
            break
        new_a = new_a * target_sum / nss

        sc = compute_score(new_a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = new_a.copy()
            stall = 0
        else:
            stall += 1

        a = new_a
        t *= t_decay

        if it % 10 == 0 or sc < best_sc + 1e-6:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_sc:.10f}, M={M:.6f}, t={chosen_t:.4f}, time={lp_time:.1f}s")

        if stall > 50:
            print(f"    LP stalled at iter {it}")
            break

    print(f"    LP done: best score={best_sc:.10f}")
    return best_a, best_sc


def soft_max_weights(c, beta):
    """Compute soft-max weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def gradient_optimize(a0, deadline, rng, lr=0.02, beta_factor=0.04,
                      beta_decay=0.94, beta_min=1e-10, restart_period=800):
    """Logit-space Adam optimization with soft-max for the max function."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = full_convolve_fft(a)
    max_c = float(c.max())

    z = np.log(a)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a)
        grad_a = 2.0 * h
        dot = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot)

        t_adam += 1
        m_arr = b1 * m_arr + (1.0 - b1) * dz
        v_arr = b2 * v_arr + (1.0 - b2) * (dz * dz)
        mh = m_arr / (1.0 - b1 ** t_adam)
        vh = v_arr / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_arr.fill(0.0)
            v_arr.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-15:
            z = best_z.copy()
            a = best_a.copy()
            m_arr.fill(0.0)
            v_arr.fill(0.0)
            t_adam = 0
            continue
        a = a / s

        c = full_convolve_fft(a)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 200:
            current_lr = max(0.004, current_lr * 0.96)

        if no_improve > restart_period:
            noise_scale = 0.08
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = full_convolve_fft(a)
            max_c = float(c.max())
            beta = max(beta_min, 0.05 * max_c)
            m_arr.fill(0.0)
            v_arr.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_bell_profile(n, sigma=0.25):
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid LP + gradient optimization for minimizing the convolution peak score.
    Phase 1: LP best-response at n=1024
    Phase 2: LP best-response at n=2048
    Phase 3: Gradient polishing at best n
    Phase 4: Additional exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc_init = compute_score(init_seq)
                print(f"Initial from height_sequence_1: n={init_seq.size}, score={sc_init:.10f}")
                update(init_seq, sc_init)
    except Exception:
        pass

    if init_seq is None:
        init_seq = make_linear_profile(1024, 1.5)
        sc_init = compute_score(init_seq)
        print(f"Using default linear init: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: LP at n=1024 =====
    n1 = 1024
    p1 = interpolate_profile(init_seq, n1)
    sc_p1 = compute_score(p1)
    update(p1, sc_p1)
    print(f"\n{'='*60}")
    print(f"Phase 1: LP optimization at n={n1} (init score={sc_p1:.10f})")
    print(f"{'='*60}")

    ph1_deadline = min(deadline, t_start + 350.0)
    result1, sc1 = lp_optimize(p1, ph1_deadline, t0=0.5, t_decay=0.998, max_iters=100)
    sc1_final = compute_score(result1)
    update(result1, sc1_final)
    print(f"  Phase 1 best: n={n1}, score={sc1_final:.10f}")

    # Also try a bell-shaped start at n=1024
    if time.monotonic() < deadline - 200:
        p1_bell = make_bell_profile(n1, 0.2)
        sc1b = compute_score(p1_bell)
        if sc1b < best_score + 0.1:
            ph1b_deadline = min(deadline, t_start + 450.0)
            r1b, sc1b_r = lp_optimize(p1_bell, ph1b_deadline, t0=0.4, t_decay=0.998, max_iters=50)
            sc1b_f = compute_score(r1b)
            update(r1b, sc1b_f)
            print(f"  Bell start n={n1}: score={sc1b_f:.10f}")

    # ===== Phase 2: LP at n=2048 =====
    n2 = 2048
    p2 = interpolate_profile(best_seq, n2)
    sc_p2 = compute_score(p2)
    update(p2, sc_p2)
    print(f"\n{'='*60}")
    print(f"Phase 2: LP optimization at n={n2} (init score={sc_p2:.10f})")
    print(f"{'='*60}")

    ph2_deadline = min(deadline, t_start + 1100.0)
    result2, sc2 = lp_optimize(p2, ph2_deadline, t0=0.4, t_decay=0.997, max_iters=100)
    sc2_final = compute_score(result2)
    update(result2, sc2_final)
    print(f"  Phase 2 best: n={n2}, score={sc2_final:.10f}")

    # ===== Phase 3: Gradient polishing =====
    print(f"\n{'='*60}")
    print(f"Phase 3: Gradient polishing (current best score={best_score:.10f})")
    print(f"{'='*60}")

    ph3_deadline = min(deadline, t_start + 1550.0)

    if best_seq is not None:
        n_polish = best_seq.size
        for attempt in range(4):
            if time.monotonic() > ph3_deadline - 20:
                break
            p0 = best_seq.copy()
            noise = 0.0005 * (attempt + 1)
            p0 = p0 + noise * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()

            lr_val = 0.02 / (1 + 0.3 * attempt)
            beta_f = 0.03 / (1 + 0.2 * attempt)
            dl = min(ph3_deadline, time.monotonic() + 100.0)
            r, sc_r = gradient_optimize(p0, dl, rng, lr=lr_val, beta_factor=beta_f,
                                         beta_decay=0.95)
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  Gradient attempt {attempt}: n={n_polish}, lr={lr_val:.4f}, score={sc_f:.10f}")

    # ===== Phase 4: Additional exploration =====
    print(f"\n{'='*60}")
    print(f"Phase 4: Additional exploration")
    print(f"{'='*60}")

    if best_seq is not None:
        # Try different n values
        for n_try in [512, 1536, 3072]:
            if time.monotonic() > deadline - 30:
                break
            if n_try == best_seq.size:
                continue
            p_try = interpolate_profile(best_seq, n_try)
            sc_try = compute_score(p_try)
            update(p_try, sc_try)
            print(f"  Interpolated n={n_try}: score={sc_try:.10f}")

            # Quick gradient polish at this n
            if sc_try < best_score + 0.02 and time.monotonic() < deadline - 50:
                dl_try = min(deadline, time.monotonic() + 60.0)
                r_try, _ = gradient_optimize(p_try, dl_try, rng, lr=0.015, beta_factor=0.02,
                                              beta_decay=0.95)
                sc_try2 = compute_score(r_try)
                update(r_try, sc_try2)
                print(f"    After gradient: n={n_try}, score={sc_try2:.10f}")

        # Final LP from best point at its native n
        if time.monotonic() < deadline - 60:
            n_final = best_seq.size
            p_final = best_seq.copy()
            p_final = p_final / p_final.sum()
            dl_final = min(deadline, time.monotonic() + 80.0)
            r_final, sc_final = lp_optimize(p_final, dl_final, t0=0.2, t_decay=0.995,
                                             max_iters=30, use_line_search=True)
            sc_final_f = compute_score(r_final)
            update(r_final, sc_final_f)
            print(f"  Final LP at n={n_final}: score={sc_final_f:.10f}")

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(1024, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_score = compute_score(final_seq)
    print(f"\n{'='*60}")
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")
    print(f"{'='*60}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
