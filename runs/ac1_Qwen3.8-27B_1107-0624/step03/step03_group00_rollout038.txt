"""
LP-based optimization with active constraint selection for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses the dual certificate approach:
solves a restricted LP (only top-m active constraints) to find the
"spread" direction, then blends toward it. Combined with multi-scale
escalation and Hann-window initialization.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    """Compute objective assuming sum(a) = 1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _make_hann(n):
    t = np.arange(n)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _make_sine_sq(n):
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    w = np.sin(np.pi * t) ** 2
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _make_super_gaussian(n, p):
    t = np.linspace(-1.0, 1.0, n)
    w = np.exp(-np.abs(t) ** p)
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _make_ramp(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    w = 1.0 + alpha * (2.0 * t - 1.0)
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _make_u_shape(n, p):
    t = np.linspace(-1.0, 1.0, n)
    w = 1.0 + p * t ** 2
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _solve_lp_active(a, M, active_indices):
    """
    Solve LP: maximize sum(b) subject to:
      conv(a, b)[k] <= M for k in active_indices
      b >= 0
    Returns b (or None if infeasible).
    """
    n = len(a)
    m = len(active_indices)
    
    # Build constraint matrix: for each active index k, the constraint is
    # sum_j b[j] * a[k-j] <= M (where a is treated as 0 outside [0,n-1])
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_indices):
        # conv(a, b)[k] = sum_j b[j] * a[k-j]
        # a[k-j] is non-zero when 0 <= k-j <= n-1, i.e., k-(n-1) <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            a_vals = a[k - j_range]
            rows.extend([idx] * len(j_range))
            cols.extend(j_range.tolist())
            vals.extend(a_vals.tolist())
    
    if not rows:
        return None
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(m)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _interpolate(a, n_new):
    """Interpolate a normalized sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()

def _adam_gradient(a, deadline, lr=0.05, m_top=5):
    """
    Adam optimization using subgradient at top-m positions.
    Works in logit space.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score(a)
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    no_imp = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(a)
        M = float(c.max())
        
        # Top-m subgradient
        top_m = min(m_top, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:top_m]
        
        # Gradient: g[i] = sum_{k in top} (2 * a[k-i]) for valid k-i
        # Using the cross-correlation approach
        w = np.zeros(2 * n - 1)
        w[top_indices] = 1.0 / top_m
        
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        g = h[n - 1:2 * n - 1]
        g = 2.0 * g  # factor of 2 from auto-convolution
        
        # Project to tangent space of simplex (remove mean)
        g = g - g.mean()
        
        # Transform to logit space gradient
        gz = a * g  # chain rule: d(da)/dz = a for softmax
        
        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * gz
        v_adam = b2 * v_adam + (1.0 - b2) * (gz * gz)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        e = np.exp(z)
        a_new = e / e.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = _score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.9)
        
        # Reset if no improvement for too long
        z = np.log(np.maximum(best_a, 1e-12))
        z = z - z.mean()
        if no_imp > 300:
            z = z + 0.1 * np.random.default_rng().normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            lr = 0.05
            no_imp = 0
        
        a = a_new
    
    return best_a, best_score

"""
LP-based optimization with active constraint selection for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses Hann-window initialization at large n,
fast LP with top-m active constraints (dual certificate approach),
multi-scale escalation, and Adam gradient polishing.
The key insight: for non-negative sequences, we solve a restricted LP that
only enforces the convolution bound at the top-m "active" positions, making
each LP solve extremely fast and allowing thousands of iterations.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _make_hann(n):
    t = np.arange(n)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1)))
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _make_sine_sq(n):
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    w = np.sin(np.pi * t) ** 2
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _make_super_gaussian(n, p):
    t = np.linspace(-1.0, 1.0, n)
    w = np.exp(-np.abs(t) ** p)
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _make_ramp(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    w = 1.0 + alpha * (2.0 * t - 1.0)
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _make_u_shape(n, p):
    t = np.linspace(-1.0, 1.0, n)
    w = 1.0 + p * t ** 2
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _make_bell(n, width):
    """Gaussian bell centered, with given width parameter."""
    t = np.linspace(-1.0, 1.0, n)
    w = np.exp(-t ** 2 / (2.0 * width ** 2))
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _solve_lp_active(a, M, active_indices):
    """
    Solve LP: maximize sum(b) subject to:
      conv(a, b)[k] <= M for k in active_indices
      0 <= b[i] <= n (upper bound for safety)
    Returns b or None.
    """
    n = len(a)
    m = len(active_indices)
    if m == 0 or M <= 0:
        return None

    rows = []
    cols = []
    vals = []

    for idx, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            a_vals = a[k - j_range]
            # Only add non-zero entries
            mask = a_vals > 1e-15
            if mask.any():
                jj = j_range[mask]
                aa = a_vals[mask]
                rows.extend([idx] * len(jj))
                cols.extend(jj.tolist())
                vals.extend(aa.tolist())

    if not rows:
        return None

    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(m)
    bounds = [(0.0, float(n))] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None

    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _interpolate(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _optimize_lp(a, deadline, m_start=30, m_end=3, t_blend=0.3, max_iter=10000):

    Optimize using LP with active constraints.
    Returns (best_a, best_score).

        M = float(c.max())
        current_score = float(2.0 * n * M)

        frac = min(1.0, iter_count / 500.0)
        m = max(m_end, int(m_start - frac * (m_start - m_end)))
        m = min(m, 2 * n - 1)

        # Find active indices: top-m positions of convolution
        if m >= 2 * n - 1:
            active = np.arange(2 * n - 1)
        else:

            threshold = M * (1.0 - 0.02 * (m_start / max(m, 1)))
            threshold = max(threshold, M * 0.9)
            active_mask = c >= threshold
            active = np.where(active_mask)[0]
            if len(active) < m:
                sorted_idx = np.argsort(c)[::-1]
                active = sorted_idx[:m]
            elif len(active) > m:
                active = active[np.argsort(c[active])[::-1]][:m]

        # Solve LP
        b = _solve_lp_active(a, M, active)
        if b is None:
            no_imp_count += 1
            if no_imp_count > 100:
                break
            iter_count += 1
            continue

        s_b = float(b.sum())
        if s_b < 1.0 + 1e-5:
            no_imp_count += 1
            if no_imp_count > 100:
                break
            iter_count += 1
            continue

        # Compute adaptive step size

        conv_bb = _full_convolve(b_norm)
        max_bb = float(conv_bb.max())
        if max_bb > 1e-10:
            t_max = min(1.0, (s_b - 1.0) / (2.0 * max_bb + 1e-10))

            t = min(current_t, 1.0)
        t = max(t, 0.01)

        # Blend

        a_new = np.maximum(a_new, 1e-15)

        sc_new = _score(a_new)
        if sc_new < current_score - 1e-13:
            a = a_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
            no_imp_count = 0
            # Slowly decrease step size
            current_t = max(0.05, current_t * 0.999)
        else:

            a_try = (1.0 - t2) * a + t2 * b_norm
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < current_score - 1e-13:
                a = a_try
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a.copy()
                no_imp_count = 0
                current_t = max(0.02, current_t * 0.9)
            else:
                no_imp_count += 1
                current_t = max(0.02, current_t * 0.85)

        iter_count += 1
        if iter_count % 500 == 0:
            elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
            print(f"    LP iter={iter_count}, best={best_score:.10f}, M={M:.8f}, s_b={s_b:.4f}, t={t:.4f}")

    return best_a, best_score


def _adam_gradient(a, deadline, lr=0.04, m_top=5, restart_noise=0.08):
    """
    Adam optimization in logit space using subgradient at top-m positions.
    Returns (best_a, best_score).
    """
    n = len(a)
    a = np.maximum(a, 1e-12).astype(np.float64)
    a = a / a.sum()

    z = np.log(a)
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score(a)
    best_z = z.copy()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    no_imp = 0
    base_lr = lr
    rng_local = np.random.default_rng(12345)

    while time.monotonic() < deadline:
        c = _full_convolve(a)
        M = float(c.max())

        # Top-m subgradient
        top_m = min(m_top, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:top_m]

        # Compute gradient via cross-correlation
        w = np.zeros(2 * n - 1)
        w[top_indices] = 1.0 / top_m

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        g = h[n - 1:2 * n - 1]
        g = 2.0 * g

        # Project to tangent of simplex
        g = g - g.mean()

        # Logit-space gradient
        gz = a * g

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp += 1
            continue

        t_adam += 1
        m_adam = b1 * m_adam + (1.0 - b1) * gz
        v_adam = b2 * v_adam + (1.0 - b2) * (gz * gz)
        mh = m_adam / (1.0 - b1 ** t_adam)
        vh = v_adam / (1.0 - b2 ** t_adam)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        e = np.exp(z)
        s_e = e.sum()
        if not np.isfinite(s_e) or s_e <= 0:
            z = best_z.copy()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            no_imp += 1
            continue
        a_new = e / s_e
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, lr * 0.9)

        if no_imp > 400:
            z = best_z + restart_noise * rng_local.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            e = np.exp(z)
            a_new = e / e.sum()
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            m_adam.fill(0)
            v_adam.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

        a = a_new

    return best_a, best_score

        do_update(a_opt, sc_opt, f"lp_{label}")

    # --- Phase 3: Multi-scale escalation ---
    print("Phase 3: Multi-scale Escalation")
    phase3_end = t0 + budget_s * 0.78
    if state['best_seq'] is not None and time.monotonic() < phase3_end - 60:
        n_cur = len(state['best_seq'])
        target_sizes = []
        if n_cur <= 2048:
            target_sizes = [4096, 8192]
        elif n_cur <= 4096:
            target_sizes = [8192, 16384]
        elif n_cur <= 8192:
            target_sizes = [16384]

        for tn in target_sizes:
            if time.monotonic() > phase3_end - 60:
                break
            p_int = _interpolate(state['best_seq'], tn)
            sc_int = _score(p_int)
            print(f"  Escalate to n={tn}: score={sc_int:.10f}")
            do_update(p_int, sc_int, f"interp_{tn}")

            per = min(150.0, phase3_end - time.monotonic() - 30)
            if per > 30:
                a_e, sc_e = _optimize_lp(p_int, time.monotonic() + per,
                                         m_start=30, m_end=3, t_blend=0.25)
                do_update(a_e, sc_e, f"lp_escal_{tn}")

    # --- Phase 4: Adam gradient polish on best ---
    print("Phase 4: Adam Polish")
    if state['best_seq'] is not None and time.monotonic() < deadline - 60:
        per = min(120.0, deadline - time.monotonic() - 30)
        if per > 30:
            a_ad, sc_ad = _adam_gradient(state['best_seq'], time.monotonic() + per,
                                         lr=0.03, m_top=3, restart_noise=0.06)
            do_update(a_ad, sc_ad, "adam_polish")

    # --- Phase 5: Final LP polish ---
    print("Phase 5: Final LP")
    if state['best_seq'] is not None and time.monotonic() < deadline - 10:
        per = deadline - time.monotonic() - 5
        if per > 20:
            a_f, sc_f = _optimize_lp(state['best_seq'], time.monotonic() + per,
                                     m_start=15, m_end=2, t_blend=0.15, max_iter=5000)
            do_update(a_f, sc_f, "final_lp")

    # --- Final output ---
    if state['best_seq'] is not None:
        final = np.clip(state['best_seq'], 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_hann(4096)
        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fb = _make_hann(4096)
    print(f"FALLBACK: n=4096, score={_score(fb):.10f}")
    return [float(x) for x in fb]
