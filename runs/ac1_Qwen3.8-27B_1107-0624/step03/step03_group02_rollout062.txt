  - k-i=j → i=k-j: a[k-j]*a[j], valid when 0<=k-j<n → j<=k<j+n AND 0<=i=n... wait, i=k-j must be in [0,n-1], so 0<=k-j<=n-1 → j<=k<=j+n-1.
  
  Both conditions give j<=k<=j+n-1. So the terms involving a[j] in c[k] are:
  - If j<=k<=j+n-1: a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] (when k-j≠j, i.e., k≠2j)
  - If k=2j (and 2j<=2n-2, i.e., j<=n-1 which is always true): a[j]*a[j] = a[j]^2
  
  So:
  - For k=2j: contribution = a[j]^2
  - For j<=k<=j+n-1, k≠2j: contribution = 2*a[j]*a[k-j]
  - Otherwise: 0
  
  When we change a[j] to x:
  - c_new[2j] = c_current[2j] - a[j]^2 + x^2
  - c_new[k] = c_current[k] - 2*a[j]*a[k-j] + 2*x*a[k-j] for j<=k<=j+n-1, k≠2j
  - c_new[k] = c_current[k] for other k
  
  And S_new = S - a[j] + x
  
  Objective(x) = 2n * max(c_new) / S_new^2
  
  We want to minimize this over x >= 0.
  
  This is a 1D problem that can be solved with golden section search or by finding the minimum analytically.
  
  The function max(c_new) is the maximum of:
  - c_current[k] for k not in [j, j+n-1] (constant)
  - c_current[2j] - a[j]^2 + x^2 (quadratic in x, convex)
  - c_current[k] - 2*a[j]*a[k-j] + 2*x*a[k-j] for j<=k<=j+n-1, k≠2j (linear in x)
  
  So max(c_new) is the maximum of a constant, a convex quadratic, and several linear functions. It's a convex function of x (maximum of convex functions is convex).
  
  And S_new = S - a[j] + x is linear and increasing in x.
  
  So the objective is 2n * (convex function) / (linear function)^2. This is not necessarily convex, so we can't just use bisection on the derivative. But we can use golden section search or a similar 1D optimization method.
  
  Actually, for practical purposes, we can just use a simple approach:
  1. Evaluate the objective at x=0, x=a[j], and a few other points.
  2. Use golden section search in the interval [0, upper_bound] where upper_bound might be 2*a[j] or 3*a[j].

def coord_descent_step(a, c, S, j):
    """Try to improve a[j] given current c and S."""
    n = len(a)
    old = a[j]
    
    # Compute affected range and coefficients
    k_lo = max(0, j - n + 1)
    k_hi = min(2*n - 2, j + n - 1)
    
    # coeff[k] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    k_range = np.arange(k_lo, k_hi + 1)
    coeff = np.zeros(k_hi - k_lo + 1)
    for idx, k in enumerate(k_range):
        c1 = 1 if (0 <= k - j < n) else 0
        c2 = 1 if (0 <= j - k < n) else 0
        if c1:
            coeff[idx] += a[k - j]
        if c2:
            coeff[idx] += a[j - k]
    
    c_affected = c[k_lo:k_hi+1].copy()
    
    # max of unaffected part
    if k_lo > 0:
        max_unaffected_left = c[:k_lo].max()
    else:
        max_unaffected_left = 0
    if k_hi < 2*n - 2:
        max_unaffected_right = c[k_hi+1:].max()
    else:
        max_unaffected_right = 0
    max_unaffected = max(max_unaffected_left, max_unaffected_right, 0)
    
    def obj(x):
        d = x - old
        S_new = S + d
        if S_new < 1e-10:
            return np.inf
        c_new_affected = c_affected + d * coeff
        max_c = max(max_unaffected, c_new_affected.max())
        return 2.0 * n * max_c / (S_new * S_new)
    
    # Try candidate values
    candidates = [0.0, old]
    if old > 0:
        candidates.extend([old*0.5, old*0.25, old*1.5, old*2.0, old*0.1, old*3.0])
    else:
        # old is 0, try positive values
        avg = S / n
        candidates.extend([avg*0.1, avg*0.5, avg, avg*2.0])
    
    best_x = old
    best_obj = obj(old)
    for x in candidates:
        if x < 0:
            x = 0.0
        o = obj(x)
        if o < best_obj:
            best_obj = o
            best_x = x
    
    # Refine around best_x with golden section
    if best_x > 0:
        lo = max(0, best_x * 0.5)
        hi = best_x * 1.5
        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(20):
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = obj(x1)
            f2 = obj(x2)
            if f1 < f2:
                hi = x2
            else:
                lo = x1
        best_x = (lo + hi) / 2

            if x > 0 and obj(x) < best_obj:
                best_obj = obj(x)
                best_x = x
    
    return best_x, best_obj

"""
Optimized step function search using Adam smooth-max + coordinate descent.
Explores multiple initializations at various sequence lengths (512, 1024, 2048).

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max_grad(c, beta):
    """Compute weights for smooth max."""
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s

def _grad_conv(c_weights, a):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c=conv(a,a)."""
    n = len(a)
    # c[k] = sum_j a[j]*a[k-j]
    # dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # grad_a[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    # First sum: let m = k-i, so k = m+i, range of m: 0<=m<n and 0<=k= m+i < 2n-1
    # = sum_m a[m] * w[m+i] for m in [max(0,-i), min(n-1, 2n-2-i)]
    # Second sum: let m = i-k, so k = i-m, range of m: 0<=m<n and 0<=k=i-m<2n-1
    # = sum_m a[m] * w[i-m] for m in [max(0, i-(2n-2)), min(n-1, i)]
    
    # Use FFT: grad_a = 2 * conv(w, a_reversed)[n-1:2n-1]
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = c_weights
    a_rev = a[::-1]
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    result = np.fft.irfft(fw * fa, N)
    h = result[n - 1:2 * n - 1]
    return 2.0 * h

    a = a / a.sum()

    best_z = z.copy()

    lr_cur = lr
    base_lr = lr

        if time.monotonic() > deadline:
            break
        
        c_curr = _fft_convolve(a)
        w = _smooth_max_grad(c_curr, beta)
        grad_a = _grad_conv(w, a)
        
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))
        
        if not np.all(np.isfinite(grad_z)):

            m, v_arr, t = np.zeros(n), np.zeros(n), 0

            k_lo = max(0, j - n + 1)

"""
Improved step function optimization using:
1. Arcsine-inspired initialization (theoretical insight: minimizes sum-density peak)
2. Adam smooth-max optimization with adaptive restarts
3. Incremental coordinate descent (exact 1D optimization per coordinate)
4. Multi-length exploration (512, 1024, 2048)
5. Shape sweeping to find best parametric form before detailed optimization
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unit(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_conv(w, a):
    """Gradient of sum_k w[k]*c[k] where c=conv(a,a)."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    result = np.fft.irfft(fw * fa, N)
    h = result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_optimize(a, deadline, max_iter=50000, lr=0.03, beta_factor=0.05):
    """Adam in logit-space for smooth max of convolution."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    lr_cur = lr
    base_lr = lr
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_conv(w, a)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m, v_arr, t = np.zeros(n), np.zeros(n), 0
            continue

        t += 1
        m = b1 * m + (1 - b1) * grad_z
        v_arr = b2 * v_arr + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v_arr / (1 - b2 ** t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr_cur = max(0.001, lr_cur * 0.93)
            if no_imp > max(4000, 6 * n):
                rng_r = np.random.default_rng(1000 + it)
                z = best_z + 0.3 * rng_r.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                e2 = np.exp(z)
                a = e2 / e2.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m, v_arr, t = np.zeros(n), np.zeros(n), 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        if it % 150 == 0:
            beta = max(1e-12, beta * 0.78)

    return best_a, best_score


def _coordinate_descent(a, max_passes=200, deadline=None):
    """Incremental coordinate descent for min 2n*max(conv)/sum^2."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    c = _fft_convolve(a)
    S = 1.0
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for p in range(max_passes):
        if deadline and time.monotonic() > deadline:
            break
        improved = 0

        for j in range(n):
            if deadline and time.monotonic() > deadline:
                break
            old = a[j]

            k_lo = max(0, j - n + 1)
            k_hi = min(2 * n - 2, j + n - 1)
            k_range = np.arange(k_lo, k_hi + 1)

            coeff = np.zeros(k_hi - k_lo + 1)
            m1 = (k_range >= j) & (k_range < j + n)
            if np.any(m1):
                coeff[m1] = a[k_range[m1] - j]
            m2 = (k_range >= j - n + 1) & (k_range <= j)
            if np.any(m2):
                coeff[m2] += a[j - k_range[m2]]

            c_aff = c[k_lo:k_hi + 1]

            max_unaff = 0.0
            if k_lo > 0:
                max_unaff = max(max_unaff, float(c[:k_lo].max()))
            if k_hi < 2 * n - 2:
                max_unaff = max(max_unaff, float(c[k_hi + 1:].max()))

            # Find best d = x - old
            avg = S / n
            d_cands = [0.0]
            if old > 1e-15:
                d_cands.extend([-old * 0.4, -old * 0.7, old * 0.5, old])
            d_cands.extend([avg * 0.3 - old, avg - old, 2 * avg - old, avg * 0.1 - old])
            d_cands = [d for d in d_cands if (old + d) >= 0 and (S + d) > 0.01]

            best_d = 0.0
            cur_max = max(max_unaff, float(c_aff.max()))
            min_f = 2.0 * n * float(c.max()) / (S * S)

            for d in d_cands:
                if abs(d) < 1e-15:
                    continue
                S_new = S + d
                if S_new < 0.01:
                    continue
                new_max_c = max(max_unaff, float((c_aff + d * coeff).max()))
                f_val = 2.0 * n * new_max_c / (S_new * S_new)
                if f_val < min_f:
                    min_f = f_val
                    best_d = d

            # Golden section refine
            if abs(best_d) > 1e-15:
                if best_d > 0:
                    lo, hi = 0.0, best_d * 2.0
                else:
                    lo, hi = best_d * 2.0, 0.0
                for _ in range(20):
                    x1 = hi - gr * (hi - lo)
                    x2 = lo + gr * (hi - lo)
                    S1, S2 = S + x1, S + x2
                    if S1 < 0.01 or S2 < 0.01:
                        break
                    f1 = 2.0 * n * max(max_unaff, float((c_aff + x1 * coeff).max())) / (S1 * S1)
                    f2 = 2.0 * n * max(max_unaff, float((c_aff + x2 * coeff).max())) / (S2 * S2)
                    if f1 < f2:
                        hi = x2
                    else:
                        lo = x1
                dmid = (lo + hi) / 2.0
                S_mid = S + dmid
                if S_mid > 0.01:
                    f_mid = 2.0 * n * max(max_unaff, float((c_aff + dmid * coeff).max())) / (S_mid * S_mid)
                    if f_mid < min_f:
                        min_f = f_mid
                        best_d = dmid

            if abs(best_d) > 1e-15:
                a[j] = old + best_d
                c[k_lo:k_hi + 1] += best_d * coeff
                S += best_d
                improved += 1

        # Renormalize
        S_actual = a.sum()
        if S_actual > 0.01:
            if abs(S_actual - S) > 1e-10:
                S = S_actual
            else:
                S = S_actual
        if S < 0.01:
            a = best_a.copy()
            c = _fft_convolve(a)
            S = 1.0

        cur_score = 2.0 * n * float(c.max()) / (S * S)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if improved == 0 and p > 3:
            break

    s_final = best_a.sum()
    if s_final > 0.01:
        best_a = best_a / s_final
    return best_a, _score_unit(best_a)


def _make_arcsine(n, alpha=0.5):
    j = np.arange(n, dtype=np.float64)
    x = (j + alpha) * (n - j - 1 + alpha)
    x = np.maximum(x, 1e-10)
    a = 1.0 / np.sqrt(x)
    return a / a.sum()


def _make_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, c2=1.0, c0=0.3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = c0 + c2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_inverse_linear(n, t=2.0):
    j = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + t * j / (n - 1))
    return a / a.sum()


def _make_power(n, p=0.3):
    j = np.arange(n, dtype=np.float64)
    a = (j + 0.5) ** p
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _shape_sweep(n, deadline):
    """Try multiple parametric shapes, return the best one."""
    shapes = [
        _make_arcsine(n, 0.3),
        _make_arcsine(n, 0.5),
        _make_arcsine(n, 1.0),
        _make_arcsine(n, 2.0),
        _make_linear(n, 1.0),
        _make_linear(n, 1.5),
        _make_linear(n, 2.0),
        _make_linear(n, 2.5),
        _make_quadratic(n, 1.0, 0.2),
        _make_quadratic(n, 2.0, 0.3),
        _make_inverse_linear(n, 1.5),
        _make_inverse_linear(n, 3.0),
        _make_power(n, 0.3),
        _make_power(n, 0.5),
        _make_power(n, 0.7),
    ]
    best_a = None
    best_s = np.inf
    for sh in shapes:
        s = _score_unit(sh)
        if s < best_s:
            best_s = s
            best_a = sh.copy()
    return best_a, best_s


def _full_optimize(a0, deadline, label=""):
    """Full pipeline: Adam -> Coord Descent -> Adam -> Coord Descent."""
    a0 = np.asarray(a0, dtype=np.float64).copy()
    a0 = a0 / a0.sum()
    n = len(a0)
    s0 = _score_unit(a0)
    if np.isfinite(s0):
        print(f"  {label} init: {s0:.10f}")

    best_a = a0.copy()
    best_s = s0

    # Phase 1: Adam
    d1 = min(deadline - 30, time.monotonic() + 0.4 * (deadline - time.monotonic()))
    a1, s1 = _adam_optimize(a0, d1, max_iter=30000, lr=0.03, beta_factor=0.05)
    if s1 < best_s:
        best_a, best_s = a1, s1
    print(f"  {label} Adam1: {best_s:.10f}")

    # Phase 2: Coordinate descent
    d2 = min(deadline - 20, time.monotonic() + 60)
    a2, s2 = _coordinate_descent(best_a, max_passes=200, deadline=d2)
    if s2 < best_s:
        best_a, best_s = a2, s2
    print(f"  {label} CD1: {best_s:.10f}")

    # Phase 3: Fine Adam
    if time.monotonic() < deadline - 10:
        d3 = min(deadline - 10, time.monotonic() + 0.3 * (deadline - time.monotonic()))
        a3, s3 = _adam_optimize(best_a, d3, max_iter=20000, lr=0.01, beta_factor=0.02)
        if s3 < best_s:
            best_a, best_s = a3, s3
        print(f"  {label} Adam2: {best_s:.10f}")

    # Phase 4: Final coordinate descent
    if time.monotonic() < deadline - 5:
        d4 = deadline - 5
        a4, s4 = _coordinate_descent(best_a, max_passes=300, deadline=d4)
        if s4 < best_s:
            best_a, best_s = a4, s4
        print(f"  {label} CD2: {best_s:.10f}")

    return best_a, best_s
