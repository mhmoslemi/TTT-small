Improved optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

import numpy as np

from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)

    return a_sym / a_sym.sum()


def _make_cosine_mod(n, base_slope=2.0, mod_freq=1.0, mod_amp=0.1):
    t = np.linspace(0.0, 1.0, n)
    base = 1.0 + base_slope * t
    mod = 1.0 + mod_amp * np.cos(2.0 * np.pi * mod_freq * t)
    a = base * mod
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _solve_lp(f, M, n):
    """Solve: maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _ls_objective_and_grad(z, beta, n):
    """
    Objective and gradient for L-BFGS.
    z are the free parameters (no constraint).
    a = softmax(z) ensures a > 0, sum(a) = 1.
    We minimize: 2n * logsumexp(beta * conv(a,a))
    
    Actually, to handle scale invariance better, let's just work with
    a >= 0 directly and normalize. Use a = max(z, 0) normalized.
    """
    a = np.maximum(z, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(z)
    a = a / s
    c = _fft_convolve(a)
    c_max = float(c.max())
    # Log-sum-exp
    x = beta * (c - c_max)
    x_max = x.max()
    lse_val = c_max + (x_max + np.log(np.exp(x - x_max).sum())) / beta
    score = 2.0 * n * lse_val
    
    # Gradient of lse_val w.r.t. a:
    w = np.exp(x - x_max)
    w = w / w.sum()
    
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad_a = 2.0 * corr[n - 1:2 * n - 1]
    
    # Chain rule: a = z / sum(z) where z_j = max(z_j, 0)
    # grad_z = (grad_a - a * sum(grad_a * a)) / s  (for the normalization)
    # But we also have the max(., 0) nonlinearity
    grad_z = (grad_a - a * np.dot(grad_a, a)) / s
    # Zero out gradient where z <= 0 (subgradient of max)
    grad_z[z <= 0] = 0.0
    
    return score, grad_z


def _lbfgs_optimize(a0, deadline, beta_start=50.0, beta_end=500.0, max_iter=2000):
    """
    Optimize using L-BFGS-B on the log-sum-exp of the convolution.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_c = _fft_convolve(a)
    best_score = float(2.0 * n * best_c.max())
    
    # Schedule: split into phases with increasing beta
    n_phases = 5
    phase_iter = max(20, max_iter // n_phases)
    
    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        frac = (phase + 1) / n_phases
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Initialize z from current a
        z = a + 1e-6  # ensure positive for the initial point
        
        def fun_and_grad(z):
            return _ls_objective_and_grad(z, beta, n)
        
        try:
            res = minimize(fun_and_grad, z, method='L-BFGS-B',
                          jac=True, options={'maxiter': phase_iter,
                                             'ftol': 1e-12, 'gtol': 1e-8,
                                             'maxls': 50})
        except Exception:
            break
        
        if res.x is not None:
            a_new = np.maximum(res.x, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                c_new = _fft_convolve(a_new)
                sc_new = float(2.0 * n * c_new.max())
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                a = a_new
    
    return best_a


def _peak_shaving(a, deadline, n_steps=200):
    """
    Directly reduce the top peaks of the convolution.
    At each step, find the dominant peak and nudge coefficients to reduce it.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        
        # Gradient of conv[peak_idx] w.r.t. a:
        # d(conv[a,a])[k]/da[m] = 2 * a[k-m] (with bounds)
        # For k = peak_idx: grad[m] = 2 * a[peak_idx - m] if 0 <= peak_idx - m < n
        #                     = 2 * a[m - peak_idx] if... 
        # Actually: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/da[m] = a[k-m] + a[k-m] = 2*a[k-m] (when both in range)
        
        grad = np.zeros(n)
        for m in range(n):
            j1 = peak_idx - m
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            # Second term: a[k-j] when k-j = m => j = k-m, same as above
            # Wait, I already counted both. Let me redo.
            # conv[k] = sum_j a[j] * a[k-j]
            # d/d a[m]: 
            #   Term 1: j = m, gives a[k-m] (if valid)
            #   Term 2: k-j = m => j = k-m, gives a[k-m] (if valid)
            # So d(conv[k])/da[m] = a[k-m] * [0<=k-m<n] + a[k-m] * [0<=m<n and 0<=k-m<n]
            # Wait no. Let me be very careful.
            # conv[k] = sum_j a[j] * a[k-j] where a[i]=0 for i notin [0,n-1]
            # d(conv[k])/da[m] = d/d a[m] sum_j a[j]*a[k-j]
            # = (da[m]/da[m]) * a[k-m] + sum_{j != m} 0 * ... + sum_j a[j] * da[k-j]/da[m]
            # = a[k-m] + a[k-m] (the second sum is a[j] * delta[k-j, m] = a[k-m])
            # = 2 * a[k-m] (if 0 <= k-m < n, else 0)
            if 0 <= j1 < n:
                val = 2.0 * a[j1]
            grad[m] = val
        
        # We want to reduce conv[peak_idx], so move in -grad direction
        # But maintain non-negativity and sum=1
        # Use a small step: a_new = a - alpha * grad, then project
        alpha = 0.001
        a_trial = a - alpha * grad
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            break
        a_trial = a_trial / s
        
        sc = _score(a_trial)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            # Gradually increase step
            alpha = min(0.01, alpha * 1.05)
        else:
            alpha = max(0.0001, alpha * 0.7)
            if alpha < 0.0002:
                break
    
    return best_a


def _lp_refine(a0, deadline, max_iter=20000, inflate_start=1.003, inflate_min=1.0002):
    """LP-based refinement with adaptive inflation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(0.02, inflate * 1.2)
            if no_imp > 200:
                break
            continue

        g = g / sum_g

        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = _fft_convolve(a_new)
            sc = float(2.0 * n * c_new.max())
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            inflate = max(inflate_min, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.02, inflate * 1.1)
            if no_imp > max(150, n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                if inflate > 1.001:
                    break

    return best_a


def _gradient_polish(a0, deadline, max_iter=30000):
    """Smooth-max gradient descent with adaptive parameters."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.03 * float(c.max()))
    lr = 0.02
    base_lr = lr
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0.0)
        s_nest = a_nest.sum()
        if s_nest < 1e-10:
            a_nest = a
            s_nest = a.sum()
        a_nest = a_nest / s_nest

        c = _fft_convolve(a_nest)
        mval = float(c.max())
        x = (c - mval) / beta
        x_shift = x - x.max()
        ex = np.exp(x_shift)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        N_pad = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N_pad)
        fa_rev = np.fft.rfft(a_nest[::-1], N_pad)
        corr = np.fft.irfft(fw * fa_rev, N_pad)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
            s_new = a.sum()
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 200 == 199:
            beta = max(1e-8, beta * 0.85)

    return best_a


def _multi_peak_shaving(a, deadline, top_k=5, n_steps=300):
    """Reduce top K peaks simultaneously."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        # Get top k peak indices
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute combined gradient to reduce all top peaks
        grad = np.zeros(n)
        for pi in peak_idx:
            pi = int(pi)
            for m in range(n):
                j1 = pi - m
                if 0 <= j1 < n:
                    grad[m] += 2.0 * a[j1]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad = grad / gnorm
        
        # Try multiple step sizes
        best_sc = best_score
        best_alpha = -1
        for alpha in [0.0005, 0.001, 0.002, 0.005, 0.01]:
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            sc = _score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha
        
        if best_alpha > 0:
            a_new = a - best_alpha * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            a_new = a_new / s
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try adding to reduce peaks (sometimes the gradient direction
            # needs adjustment for non-negativity)
            break
    
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Multiple initializations at various lengths
    2. LP refinement
    3. L-BFGS on log-sum-exp
    4. Gradient polish
    5. Peak shaving
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Build candidate pool
    candidates = []
    
    if parent is not None:
        for n_t in [384, 512, 768, 1024]:
            a_int = _interpolate(parent, n_t)
            sc = _score(a_int)
            candidates.append((sc, f'parent_n{n_t}', n_t, a_int))
    
    # Various structured initializations
    for n_t in [256, 384, 512, 768]:
        inits = [
            (_make_ramp(n_t, 1.0), f'ramp1_n{n_t}'),
            (_make_ramp(n_t, 1.5), f'ramp15_n{n_t}'),
            (_make_ramp(n_t, 2.0), f'ramp2_n{n_t}'),
            (_make_ramp2(n_t, 1.5), f'ramp2r_n{n_t}'),
            (_make_sym_ramp(n_t), f'symramp_n{n_t}'),
            (_make_cosine_mod(n_t, 2.0, 1.0, 0.1), f'cosmod_n{n_t}'),
            (_make_cosine_mod(n_t, 3.0, 2.0, 0.15), f'cosmod2_n{n_t}'),
        ]
        for a0, name in inits:
            sc = _score(a0)
            candidates.append((sc, name, n_t, a0))
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    
    # Try top candidates with LP
    n_attempts = 6
    for idx in range(min(n_attempts, len(candidates))):
        if time.monotonic() > deadline - 300:
            break
        
        sc0, name, n_t, a0 = candidates[idx]
        
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()
            print(f"New best from init: {name} score={sc0:.10f}")
        
        if sc0 >= best_score + 0.05:
            continue
        
        remaining = deadline - time.monotonic() - 200
        if remaining < 30:
            break
        
        # LP refinement
        lp_time = min(120.0, remaining * 0.4)
        if lp_time > 10:
            lp_end = time.monotonic() + lp_time
            print(f"LP: {name} n={n_t} init={sc0:.8f} budget={lp_time:.0f}s")
            a_lp = _lp_refine(a0, lp_end)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  LP -> {sc_lp:.10f}")
            else:
                print(f"  LP -> {sc_lp:.8f}")

    # Now focus on the best sequence found
    if best_seq is None:
        best_seq = _make_ramp(512, 1.5)
        best_score = _score(best_seq)
        print(f"Fallback: score={best_score:.10f}")

    print(f"\n=== Focused optimization on best (score={best_score:.10f}, n={best_seq.size}) ===")
    
    # Phase 2: L-BFGS optimization
    if time.monotonic() < deadline - 180:
        lb_time = min(180.0, deadline - time.monotonic() - 150)
        if lb_time > 30:
            print(f"\nL-BFGS: budget={lb_time:.0f}s, n={best_seq.size}")
            a_lb = _lbfgs_optimize(best_seq, time.monotonic() + lb_time,
                                   beta_start=30.0, beta_end=300.0, max_iter=500)
            sc_lb = _score(a_lb)
            if np.isfinite(sc_lb) and sc_lb < best_score:
                best_score = sc_lb
                best_seq = a_lb.copy()
                print(f"  L-BFGS -> {best_score:.10f}")
            else:
                print(f"  L-BFGS -> {sc_lb:.8f} (no improvement)")

    # Phase 3: LP refinement on the best
    if time.monotonic() < deadline - 120:
        lp_time = min(120.0, deadline - time.monotonic() - 100)
        if lp_time > 20:
            print(f"\nLP pass 2: budget={lp_time:.0f}s, n={best_seq.size}")
            a_lp2 = _lp_refine(best_seq, time.monotonic() + lp_time,
                               inflate_start=1.002, inflate_min=1.0001)
            sc_lp2 = _score(a_lp2)
            if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = a_lp2.copy()
                print(f"  -> {best_score:.10f}")
            else:
                print(f"  -> {sc_lp2:.8f} (no improvement)")

    # Phase 4: Gradient polish
    if time.monotonic() < deadline - 80:
        gp_time = min(80.0, deadline - time.monotonic() - 60)
        if gp_time > 10:
            print(f"\nGradient polish: budget={gp_time:.0f}s")
            a_gp = _gradient_polish(best_seq, time.monotonic() + gp_time, max_iter=20000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
                print(f"  -> {best_score:.10f}")
            else:
                print(f"  -> {sc_gp:.8f} (no improvement)")

    # Phase 5: Peak shaving
    if time.monotonic() < deadline - 40:
        ps_time = min(40.0, deadline - time.monotonic() - 30)
        if ps_time > 5:
            print(f"\nPeak shaving: budget={ps_time:.0f}s")
            a_ps = _multi_peak_shaving(best_seq, time.monotonic() + ps_time,
                                       top_k=3, n_steps=200)
            sc_ps = _score(a_ps)
            if np.isfinite(sc_ps) and sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
                print(f"  -> {best_score:.10f}")
            else:
                print(f"  -> {sc_ps:.8f} (no improvement)")

    # Phase 6: Final LP squeeze
    if time.monotonic() < deadline - 10:
        lp_time = min(20.0, deadline - time.monotonic() - 5)
        if lp_time > 5:
            print(f"\nFinal LP: budget={lp_time:.0f}s")
            a_fin = _lp_refine(best_seq, time.monotonic() + lp_time,
                               inflate_start=1.001, inflate_min=1.0001)
            sc_fin = _score(a_fin)
            if np.isfinite(sc_fin) and sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
                print(f"  -> {best_score:.10f}")

    # Phase 7: Try different length from the best shape
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_best = best_seq.size
        for n_try in [n_best // 2, n_best * 2]:
            if n_try < 64 or n_try > 2048:
                continue
            a_try = _interpolate(best_seq, n_try)
            sc_try = _score(a_try)
            if np.isfinite(sc_try) and sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()
                print(f"  Length change to {n_try}: {best_score:.10f}")
            # Quick gradient polish on new length
            if time.monotonic() < deadline - 5:
                gp = _gradient_polish(a_try, deadline, max_iter=5000)
                sc_gp = _score(gp)
                if np.isfinite(sc_gp) and sc_gp < best_score:
                    best_score = sc_gp
                    best_seq = gp.copy()
                    print(f"  Polished n={n_try}: {best_score:.10f}")

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.5)

    # Symmetrize for final (often helps)
    final_seq = (final_seq + final_seq[::-1]) / 2.0
    final_seq = np.maximum(final_seq, 0.0)
    s = final_seq.sum()
    if s > 1e-10:
        final_seq = final_seq / s
    
    sc_final = _score(final_seq)
    if sc_final < best_score:
        best_score = sc_final
        final_seq = final_seq.copy()
        print(f"  Symmetrized: {best_score:.10f}")
    
    # Also check without symmetrization
    final_seq2 = np.clip(best_seq, 0.0, 1000.0)
    s2 = float(final_seq2.sum())
    if s2 > 0.01:
        final_seq2 = final_seq2 / s2
    sc2 = _score(final_seq2)
    if sc2 < best_score:
        best_score = sc2
        final_seq = final_seq2
        print(f"  Asymmetric better: {best_score:.10f}")

    # Evaluate with the actual function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence: {sc_eval:.10f}")
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
