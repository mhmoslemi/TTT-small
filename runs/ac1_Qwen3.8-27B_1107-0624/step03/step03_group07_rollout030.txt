def lp_refine(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M, b >= 0."""
    n = len(a)
    # conv(a,b)[k] = sum_i a[i] * b[k-i] for valid i
    # This is a linear function of b.
    # For k = 0, 1, ..., 2n-2:
    #   conv(a,b)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i] * b[k-i]
    
    # Variables: b[0], ..., b[n-1]
    # Constraints: for each k, the linear expression <= M
    # Objective: maximize sum(b)
    
    from scipy.optimize import linprog
    import numpy as np
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    # Build constraint matrix A_ub such that A_ub @ b <= M
    # A_ub has (2n-1) rows and n columns
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(k, n-1) + 1):
            j = k - i  # b[j]
            A_ub[k, j] += a[i]
    
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

import numpy as np

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def convolve_fft(a):
    """Full linear convolution using FFT for speed."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

`sum_{j'} a[j'] * w[j+j'] = conv(a_pad, w_shifted)[n-1+j]`

where `w_shifted[m] = w[m - (n-1)]` for m in [n-1, n-1+2n-2] = [n-1, 3n-3], and 0 elsewhere.

L needs to be >= n + (2n-1) = 3n-1.

This is getting complicated. Let me just use a simpler (but still O(n log n)) approach:

- If |S| is large, use the FFT approach.

In practice, for this problem, the argmax set is usually small (1-5 elements), so the direct approach is fine.

def lp_min_cross_peak(a, M_ref):
    """
    Solve: min M s.t. conv(a,b)[k] <= M for all k, sum(b) = 1, b >= 0.
    Returns (b, M_opt).
    """
    n = a.size
    # Variables: b[0..n-1], M (total n+1 variables)
    # Objective: minimize M (index n in the variable vector)
    # Constraints:
    #   conv(a,b)[k] - M <= 0 for k = 0..2n-2
    #   sum(b) = 1 (equality)
    #   b[i] >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: sum_i A[k,i] * b[i] - M <= 0
    # A_ub is (2n-1) x (n+1)
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(k, n-1)+1):
            j = k - i
            A_ub[k, j] = a[i]
        A_ub[k, n] = -1.0  # -M
    
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b[i] >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
```

"""
LP-based direction finding + projected subgradient for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions (exact, no soft-max bias),
projected subgradient on the simplex for polishing, adaptive step sizes, and
multi-scale exploration to escape local minima.
"""
import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def convolve_fft(a):
    """Full linear convolution using FFT for speed."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def cross_convolve_fft(a, b):
    """Compute conv(a, b) using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    m = b.size
    if n * m <= 20000:
        return np.convolve(a, b)
    N = 1
    while N < n + m - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    return c


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def subgradient_of_max_conv(a):
    """
    Compute the subgradient of max_k conv(a,a)[k] at point a.
    Returns (g, M, peak_set).
    """
    n = a.size
    c = convolve_fft(a)
    M = float(c.max())
    tol = max(1e-10 * M, 1e-15)
    S = np.where(c >= M - tol)[0]

    g = np.zeros(n)
    for k in S:
        w_k = 1.0 / len(S)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        g[js] += w_k * (a[k - js] + a[js])

    return g, M, S


def lp_find_direction(a):
    """
    Solve LP: min M s.t. conv(a,b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b, M_opt) or (None, None) on failure.
    """
    n = a.size
    # Variables: [b_0, ..., b_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M

    # Build constraint matrix: conv(a,b)[k] - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            js = np.arange(i_start, i_end + 1)
            A_ub[k, k - js] = a[js]
        A_ub[k, n] = -1.0

    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        return res.x[:n].copy(), float(res.x[n])
    return None, None


def lp_with_target(a, M_target):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= M_target, b >= 0.
    Returns (b, sum_b) or (None, 0) on failure.
    """
    n = a.size
    c_obj = -np.ones(n)

    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            js = np.arange(i_start, i_end + 1)
            A_ub[k, k - js] = a[js]

    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if res.success:
        b = res.x.copy()
        b = np.maximum(b, 0)
        return b, float(b.sum())
    return None, 0.0


def lp_refine_step(a, lr_base=0.1, max_backtrack=10):
    """
    One LP-based refinement step.
    Returns (a_new, success, M_old, M_new).
    """
    n = a.size
    c = convolve_fft(a)
    M_old = float(c.max())

    # Solve LP to find direction minimizing cross-peak
    b, M_cross = lp_find_direction(a)
    if b is None:
        return a, False, M_old, M_old

    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-10:
        return a, False, M_old, M_old
    b = b / s

    # Line search: try different step sizes
    t = lr_base
    for _ in range(max_backtrack):
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        a_new = a_new / s_new

        c_new = convolve_fft(a_new)
        M_new = float(c_new.max())

        if M_new < M_old:
            return a_new, True, M_old, M_new
        t *= 0.5

    return a, False, M_old, M_old


def subgradient_step(a, lr):
    """One projected subgradient step on the simplex."""
    g, M, S = subgradient_of_max_conv(a)
    a_new = a - lr * g
    a_new = project_simplex(a_new)
    return a_new, M


def multi_start_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0, 1, n)

    # Linear
    for eta in [0.5, 1.0, 2.0, 4.0]:
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cosine modulated
    for k in [1, 2, 3, 5]:
        for amp in [0.3, 0.6, 0.9]:
            a = 1.0 + amp * np.cos(2 * np.pi * k * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Gaussian bumps
    for width in [0.15, 0.3, 0.5, 0.8]:
        for center in [0.25, 0.5, 0.75]:
            a = np.exp(-((t - center) / width) ** 2)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Power ramps
    for p in [0.5, 1.0, 2.0, 5.0, 10.0]:
        a = (1.0 - t) ** p + 0.05
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        a2 = t ** p + 0.05
        a2 = np.maximum(a2, 1e-12)
        profiles.append(a2 / a2.sum())

    # Random low-freq
    for _ in range(8):
        coeffs = rng.uniform(-0.5, 0.5, 6)
        a = np.ones(n)
        for i in range(1, 7):
            a += coeffs[i-1] * np.cos(2 * np.pi * i * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def lp_optimize(a, deadline, lr_base=0.1, adaptive=True, verbose=True, tag=""):
    """
    Main LP-based optimizer. Iteratively applies LP direction finding.
    """
    n = a.size
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    c = convolve_fft(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    no_improve = 0
    lr = lr_base
    lp_count = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline:
        a_new, success, M_old, M_new = lp_refine_step(a, lr_base=lr)

        if success:
            a = a_new
            if M_new < best_M - 1e-14:
                best_M = M_new
                best_score = float(2.0 * n * best_M)
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        lp_count += 1

        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.95
        elif no_improve > 100:
            # Try subgradient step as alternative
            lr_sg = lr * 0.01
            a_sg, _ = subgradient_step(a, lr_sg)
            c_sg = convolve_fft(a_sg)
            M_sg = float(c_sg.max())
            if M_sg < best_M:
                a = a_sg
                best_a = a.copy()
                best_M = M_sg
                best_score = float(2.0 * n * best_M)
                no_improve = 0

        if no_improve > 200:
            # Random restart with small perturbation
            noise = 0.05 * np.random.randn(n)
            a = best_a + noise
            a = project_simplex(a)
            c = convolve_fft(a)
            no_improve = 0
            lr = lr_base

        if verbose and lp_count % 50 == 0:

                  f"no_improve={no_improve}, t={el:.1f}s")

def subgradient_optimize(a, deadline, lr_init=0.001, lr_decay=0.999,
                         verbose=False, tag=""):
    """
    Projected subgradient method on the simplex.
    Uses the exact subgradient (uniform over argmax set).
    """
    n = a.size
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    c = convolve_fft(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    lr = lr_init
    iters = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline:
        g, M, S = subgradient_of_max_conv(a)
        a_new = a - lr * g
        a_new = project_simplex(a_new)

        c = convolve_fft(a_new)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_score = float(2.0 * n * best_M)
            best_a = a_new.copy()

        a = a_new
        iters += 1
        lr *= lr_decay

        if verbose and iters % 1000 == 0:
            el = time.monotonic() - t_start
            print(f"    {tag} SG iter {iters}: M={best_M:.10f}, score={best_score:.10f}, "

    return best_a, best_score


def hybrid_optimize(a, deadline, verbose=True, tag=""):
    """
    Hybrid: alternate between LP steps and subgradient steps.
    LP for large improvements, SG for fine polishing.
    """
    n = a.size
    a = a.copy()
    a = project_simplex(a)

    best_a = a.copy()
    c = convolve_fft(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    t_start = time.monotonic()
    lp_count = 0
    sg_count = 0
    phase = "lp"  # alternate phases
    consecutive_no_improve = 0

    while time.monotonic() < deadline:
        if phase == "lp":
            a_new, success, M_old, M_new = lp_refine_step(a, lr_base=0.1)
            if success and M_new < best_M - 1e-14:
                a = a_new
                best_M = M_new
                best_score = float(2.0 * n * best_M)
                best_a = a.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
                if consecutive_no_improve > 5:
                    phase = "sg"
                    consecutive_no_improve = 0
            lp_count += 1
        else:
            # Subgradient phase: do a batch of SG steps
            lr = 0.001
            batch_end = min(deadline, time.monotonic() + 5.0)
            while time.monotonic() < batch_end:
                g, M, S = subgradient_of_max_conv(a)
                a = project_simplex(a - lr * g)
                c = convolve_fft(a)
                M_new = float(c.max())
                if M_new < best_M - 1e-14:
                    best_M = M_new
                    best_score = float(2.0 * n * best_M)
                    best_a = a.copy()
                lr *= 0.999
                sg_count += 1

            phase = "lp"
            consecutive_no_improve = 0

        if verbose and lp_count % 30 == 0:
            el = time.monotonic() - t_start
            print(f"    {tag} Hybrid: LP={lp_count}, SG={sg_count}, "
                  f"M={best_M:.10f}, score={best_score:.10f}, t={el:.1f}s")

    phase2_deadline = min(deadline, t_start + 800.0)

    if best_seq is not None and best_seq.size == n1:
        a2 = best_seq.copy()
    else:
        a2 = result.copy()

    result2, sc2 = hybrid_optimize(a2, phase2_deadline, tag="P2")
    sc2_final = compute_score(result2)
    update(result2, sc2_final)
    print(f"  Phase 2 best: score={sc2_final:.10f}")

    # Also try subgradient-only for polishing
    if time.monotonic() < phase2_deadline - 30:
        dl_sg = min(phase2_deadline, time.monotonic() + 60.0)
        a_sg = best_seq.copy() if best_seq is not None else result2.copy()
        # Add tiny noise to break symmetry
        a_sg = a_sg + 0.001 * rng.normal(size=a_sg.size)
        a_sg = project_simplex(a_sg)
        r_sg, sc_sg = subgradient_optimize(a_sg, dl_sg, lr_init=0.0005,
                                           lr_decay=0.9999, tag="P2-SG")
        sc_sg_f = compute_score(r_sg)
        update(r_sg, sc_sg_f)
        print(f"  SG polish: score={sc_sg_f:.10f}")

    # ===== Phase 3: Multi-scale exploration =====
    print(f"\n--- Phase 3: Multi-scale exploration ---")
    phase3_deadline = min(deadline, t_start + 1400.0)

    if best_seq is not None:
        n_best = best_seq.size

        # Try larger n
        for n_try in [n_best * 2, n_best * 4]:
            if n_try > 8192 or time.monotonic() > phase3_deadline - 60:
                break
            print(f"  Trying n={n_try}...")
            p_interp = interpolate_profile(best_seq, n_try)
            sc_interp = compute_score(p_interp)
            update(p_interp, sc_interp)
            print(f"    Interpolated: score={sc_interp:.10f}")

            # LP at this scale (only if feasible time-wise)
            lp_time_est = n_try * n_try / 1e6 * 2  # rough estimate in seconds
            if lp_time_est < 10 and time.monotonic() < phase3_deadline - 100:
                dl = min(phase3_deadline, time.monotonic() + 150.0)
                r, sc_r = lp_optimize(p_interp, dl, lr_base=0.05, tag=f"P3-n{n_try}")
                sc_f = compute_score(r)
                update(r, sc_f)
                print(f"    After LP: score={sc_f:.10f}")
            else:
                # Use subgradient for larger n
                dl = min(phase3_deadline, time.monotonic() + 100.0)
                r, sc_r = subgradient_optimize(p_interp, dl, lr_init=0.001,
                                               lr_decay=0.9995, tag=f"P3-SG-n{n_try}")
                sc_f = compute_score(r)
                update(r, sc_f)
                print(f"    After SG: score={sc_f:.10f}")

        # Try smaller n (sometimes smaller is better due to the factor 2n)
        for n_try in [n_best // 2, n_best // 4]:
            if n_try < 64 or time.monotonic() > phase3_deadline - 30:
                break
            print(f"  Trying n={n_try}...")
            p_small = interpolate_profile(best_seq, n_try)
            sc_small = compute_score(p_small)
            update(p_small, sc_small)
            print(f"    Interpolated: score={sc_small:.10f}")

            if sc_small < best_score + 0.05 and time.monotonic() < phase3_deadline - 50:
                dl = min(phase3_deadline, time.monotonic() + 60.0)
                r, sc_r = lp_optimize(p_small, dl, lr_base=0.1, tag=f"P3-n{n_try}")
                sc_f = compute_score(r)
                update(r, sc_f)
                print(f"    After LP: score={sc_f:.10f}")

    # ===== Phase 4: Diverse restarts at best n =====
    print(f"\n--- Phase 4: Diverse restarts ---")
    phase4_deadline = deadline

    if best_seq is not None:
        n4 = best_seq.size
        profiles = multi_start_profiles(n4, rng)

        # Interpolate best as top candidate
        p_best_interp = interpolate_profile(best_seq, n4)
        profiles.insert(0, p_best_interp)

        for idx, p in enumerate(profiles[:12]):
            if time.monotonic() > phase4_deadline - 20:
                break
            sc_p = compute_score(p)
            if sc_p >= best_score + 0.02:
                continue

            # Try LP if n is small enough
            if n4 <= 2048 and time.monotonic() < phase4_deadline - 40:
                dl = min(phase4_deadline, time.monotonic() + 30.0)
                r, sc_r = lp_optimize(p, dl, lr_base=0.08, verbose=False, tag=f"P4-{idx}")
            else:
                dl = min(phase4_deadline, time.monotonic() + 30.0)
                r, sc_r = subgradient_optimize(p, dl, lr_init=0.001,
                                               lr_decay=0.9995, verbose=False, tag=f"P4-{idx}")
            sc_f = compute_score(r)
            update(r, sc_f)
            if sc_f < best_score + 0.01:
                print(f"  Restart {idx}: score={sc_f:.10f}")

    # ===== Phase 5: Final fine polish =====
    print(f"\n--- Phase 5: Final polish ---")
    if best_seq is not None:
        n_pol = best_seq.size
        a_pol = best_seq.copy()

        # Multiple short subgradient passes with decreasing lr
        for lr_pass in [0.0003, 0.0001, 0.00003, 0.00001]:
            if time.monotonic() > deadline - 10:
                break
            dl = min(deadline, time.monotonic() + 30.0)
            a_pol = a_pol + 0.0001 * rng.normal(size=n_pol)
            a_pol = project_simplex(a_pol)
            r, sc = subgradient_optimize(a_pol, dl, lr_init=lr_pass,
                                         lr_decay=0.99999, verbose=False, tag="polish")
            sc_f = compute_score(r)
            update(r, sc_f)

        # One more LP pass if time allows
        if n_pol <= 2048 and time.monotonic() < deadline - 60:
            dl_final = min(deadline, time.monotonic() + 50.0)
            r, sc = lp_optimize(best_seq.copy(), dl_final, lr_base=0.05,
                                verbose=False, tag="final-LP")
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  Final LP: score={sc_f:.10f}")

    print(f"\n  Overall best score: {best_score:.10f}")

    # ===== Final validation and return =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = np.ones(512) / 512

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
