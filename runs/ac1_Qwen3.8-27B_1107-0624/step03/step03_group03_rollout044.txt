"""
Symmetry-enforced multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines top-k LSE gradient with periodic reheating, symmetry projection,
LP refinement, and multi-scale initialization.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _make_conv_matrix(a, M, top_k=None):
    n = len(a)
    c_aa = _fft_convolve(a)
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(c_aa)[::-1]
        active = set()
        for i in idx[:top_k]:
            for offset in range(-3, 4):
                j = i + offset
                if 0 <= j < 2 * n - 1:
                    active.add(j)
        active = sorted(active)
    else:
        active = list(range(2 * n - 1))
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[row, j] = a[j]
    return A, b, active


def _lp_spread(a, M, top_k=None):
    n = len(a)
    c_obj = -np.ones(n)
    if n <= 256:
        A, b, _ = _make_conv_matrix(a, M, top_k=None)
    else:
        A, b, _ = _make_conv_matrix(a, M, top_k=top_k or 30)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lse_step_topk(a, z, state, T, lr, top_k=20):
    """LSE gradient step using top-k softmax for efficiency."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())

    # Top-k softmax
    sorted_idx = np.argsort(c)[::-1]
    k = min(top_k, len(c))
    active = sorted_idx[:k]
    x_active = (c[active] - m_val) / T
    x_active = np.clip(x_active, -50, 50)
    ex = np.exp(x_active)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = np.zeros_like(c)
        w[active] = ex / s

    # Gradient: conv(w, reverse(a))
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    # Symmetrize gradient
    grad = _symmetrize(grad)

    # Logit-space gradient (project to tangent of simplex)
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    gz = _symmetrize(gz)

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t_idx = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t_idx)
    vh = v / (1.0 - b2 ** t_idx)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z = _symmetrize(z)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t_idx
    state['m'] = m
    state['v'] = v

    return a_new, z, state, m_val

    best_score = _score_normalized(a)

            z = np.log(np.maximum(a, 1e-12))

    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    if use_symmetry:
        a = _symmetrize(a)
        a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    inflate_cur = inflate
    blend_cur = blend
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp, top_k=top_k)

        if g is None:
            inflate_cur = min(1.02, inflate_cur * 1.3)
            no_imp += 3
            if no_imp > 200:
                break
            continue

        if use_symmetry:
            g = _symmetrize(g)

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 15 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.08)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        # Line search for optimal blend
        best_t = blend_cur
        best_sc = _score_normalized(a)
        for t in [blend_cur, blend_cur * 0.5, blend_cur * 1.5, 0.05, 0.3, 0.5]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            if use_symmetry:
                a_try = _symmetrize(a_try)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_normalized(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0.0)
        if use_symmetry:
            a_new = _symmetrize(a_new)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend_cur = max(0.03, blend_cur * 0.9)
            if no_imp > 150:
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _subgradient_polish(a, deadline, max_iter=30000, lr=0.01, use_symmetry=True):
    """Subgradient polish at the actual max."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    if use_symmetry:
        a = _symmetrize(a)
        a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    v = np.zeros(n)
    momentum = 0.85
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-10)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        if use_symmetry:
            grad = _symmetrize(grad)

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        if use_symmetry:
            grad_tan = _symmetrize(grad_tan)

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        if use_symmetry:
            a_new = _symmetrize(a_new)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_sym_gaussian(n, sigma_frac=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_sym_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_sym_flat_modulated(n, amp=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_sym_dual_gaussian(n, sigma1=0.2, sigma2=0.5, weight=0.3):
    t = np.linspace(0.0, 1.0, n)
    g1 = np.exp(-0.5 * ((t - 0.5) / (sigma1 * 0.5)) ** 2)
    g2 = np.exp(-0.5 * ((t - 0.5) / (sigma2 * 0.5)) ** 2)
    a = weight * g1 + (1 - weight) * g2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization with symmetry enforcement, top-k LSE gradient
    with periodic reheating, LP refinement, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = [1024, 512, 2048]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)

    def get_candidates(n_t, rng_local):
        cands = []
        if parent is not None:
            cands.append(_symmetrize(_interpolate_profile(parent, n_t)))
            for _ in range(3):
                p = _interpolate_profile(parent, n_t)
                p_pert = p * (1.0 + 0.06 * rng_local.normal(size=n_t))
                p_pert = np.maximum(p_pert, 0)
                if p_pert.sum() > 1e-10:
                    p_pert = p_pert / p_pert.sum()
                    p_pert = _symmetrize(p_pert)
                    p_pert = p_pert / p_pert.sum()
                    cands.append(p_pert)
        cands.append(_make_sym_gaussian(n_t, 0.4))
        cands.append(_make_sym_gaussian(n_t, 0.3))
        cands.append(_make_sym_raised_cosine(n_t))
        cands.append(_make_sym_flat_modulated(n_t, 0.3))
        cands.append(_make_sym_dual_gaussian(n_t, 0.25, 0.5, 0.4))
        return cands

    # Phase 1: Quick exploration
    print("\n=== Phase 1: Quick exploration ===")
    best_per_length = {}
    p1_end = t_start + (budget_s - 20.0) * 0.25

    for n_t in lengths:
        if time.monotonic() > p1_end - 30:
            break
        cands = get_candidates(n_t, rng)
        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > p1_end - 30:
                break
            a0 = _symmetrize(a0)
            a0 = a0 / a0.sum()
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Short LSE
            lse_budget = min(60.0, p1_end - time.monotonic() - 30.0)
            if lse_budget < 15:
                break
            lse_end = time.monotonic() + lse_budget
            a_lse = _lse_optimize(a0, lse_end, max_iter=20000,
                                   T_start=0.1, T_end=1e-3, lr=0.1,
                                   top_k_start=40, top_k_end=10,
                                   reheat_period=300, reheat_factor=1.5)
            sc_lse = _score(a_lse)
            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse
            print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> LSE={sc_lse:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best for n={n_t}: {best_here_score:.10f}")

    # Phase 2: Deep LSE optimization
    print("\n=== Phase 2: Deep LSE optimization ===")
    p2_end = t_start + (budget_s - 20.0) * 0.60

    for n_t in sorted(best_per_length.keys(), key=lambda x: -best_per_length[x].size):
        if time.monotonic() > p2_end - 60:
            break
        a_in = best_per_length[n_t]
        sc_in = _score(a_in)

        lse_budget = min(180.0, p2_end - time.monotonic() - 60.0)
        if lse_budget < 30:
            continue
        lse_end = time.monotonic() + lse_budget

        print(f"  Deep LSE n={n_t}: start score={sc_in:.10f}, budget={lse_budget:.0f}s")
        a_deep = _lse_optimize(a_in, lse_end, max_iter=100000,
                                T_start=0.05, T_end=5e-5, lr=0.08,
                                top_k_start=25, top_k_end=5,
                                reheat_period=800, reheat_factor=2.0)
        sc_deep = _score(a_deep)
        if np.isfinite(sc_deep) and sc_deep < best_score:
            best_score = sc_deep
            best_seq = a_deep.copy()
        best_per_length[n_t] = a_deep
        print(f"  Deep LSE n={n_t}: score={sc_deep:.10f}")

    # Phase 3: LP refinement
    print("\n=== Phase 3: LP refinement ===")
    p3_end = t_start + (budget_s - 20.0) * 0.80

    for n_t in sorted(best_per_length.keys(), key=lambda x: _score(best_per_length[x])):
        if time.monotonic() > p3_end - 30:
            break
        a_in = best_per_length[n_t]
        sc_in = _score(a_in)
        print(f"  LP n={n_t}: start score={sc_in:.10f}")

        lp_budget = min(120.0, p3_end - time.monotonic() - 30.0)
        if lp_budget < 20:
            continue
        lp_end = time.monotonic() + lp_budget

        a_lp = _lp_refine(a_in, lp_end, max_iter=3000,
                          inflate=1.0003, blend=0.12, top_k=50,
                          use_symmetry=True)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP n={n_t}: score={sc_lp:.10f}")

    # Phase 4: Final polish
    print("\n=== Phase 4: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        # Subgradient polish
        sp_budget = min(60.0, deadline - time.monotonic() - 15.0)
        if sp_budget > 15:
            sp_end = time.monotonic() + sp_budget
            a_sp = _subgradient_polish(best_seq, sp_end, max_iter=50000,
                                        lr=0.008, use_symmetry=True)
            sc_sp = _score(a_sp)
            if np.isfinite(sc_sp) and sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            print(f"  Subgradient: score={sc_sp:.10f}")

        # Final LP
        if time.monotonic() < deadline - 15:
            lp_budget = min(40.0, deadline - time.monotonic() - 10.0)
            if lp_budget > 10:
                lp_end = time.monotonic() + lp_budget
                a_final = _lp_refine(best_seq, lp_end, max_iter=1500,
                                      inflate=1.0002, blend=0.08, top_k=60,
                                      use_symmetry=True)
                sc_final = _score(a_final)
                if np.isfinite(sc_final) and sc_final < best_score:
                    best_score = sc_final
                    best_seq = a_final.copy()
                print(f"  Final LP: score={sc_final:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_sym_gaussian(1024, 0.4)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_sym_gaussian(1024, 0.4)

    # Ensure sum is sufficient
    if float(final_seq.sum()) < 0.01:
        final_seq = np.ones(1024) / 1024.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
