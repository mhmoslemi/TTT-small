"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses spectral initializations, adaptive LP refinement with peak targeting,
asymmetric profile design, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _next_pow2(n):
    k = 1
    while k < n:

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c -= c.min()
    return c

    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:

    return w / s

    return None

I want to be really careful about the LP. Let me think about the exact formulation:

Given current `a` (sum = 1), the LP is:
- Variables: `b_0, ..., b_{n-1} >= 0`
- Objective: maximize `sum_j b_j`
- Constraints: `sum_j a_j * b_{k-j} <= M` for all k = 0, ..., 2n-2

Where `b_{k-j} = 0` if `k-j < 0` or `k-j >= n`.

The constraint matrix A has:
- A[k, j] = a[k-j] if `0 <= k-j < n`, else 0.

For a given k, the non-zero entries are:
- j = max(0, k-n+1), ..., min(k, n-1)
- A[k, j] = a[k-j]

So A is a "banded" matrix where each row k has non-zero entries in a specific range of columns.

For n=768, the matrix is 1535 × 768 with about 768*768/2 ≈ 295K non-zeros (roughly half the entries are non-zero). This is a moderate-sized LP.

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using:

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < 0:
        c = c - mn
    return c


def _fft_cross_conv(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size + b.size - 1
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n]
    mn = c.min()
    if mn < 0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _gradient_conv(a, beta, top_k=None):
    """Gradient of smooth-max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = _fft_conv(a)
    w = _smooth_max_weight(c, beta)

    if top_k is not None and top_k < len(c):
        sorted_idx = np.argsort(c)[::-1]
        w_peak = np.zeros_like(c)
        w_peak[sorted_idx[:top_k]] = 1.0 / top_k
        w = 0.6 * w + 0.4 * w_peak
        w = w / w.sum()

    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    fw = np.fft.rfft(w, N)

    conv1 = np.fft.irfft(fa_rev * fw, N)
    conv2 = np.fft.irfft(fa * fw, N)

    grad = conv1[n - 1:2 * n - 1] + conv2[0:n]
    return grad, float(c.max()), c


def _make_conv_matrix_rows(a, lags):
    """Build LP constraint rows for given lags. A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    rows = []
    for k in lags:
        row = np.zeros(n)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            row[js] = a[k - js]
        rows.append(row)
    return np.array(rows) if rows else np.zeros((0, n))


def _solve_lp(a, M, lags):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for k in lags, b >= 0."""
    n = len(a)
    A = _make_conv_matrix_rows(a, lags)
    if A.size == 0:
        return np.ones(n) / n
    b_ub = np.full(len(lags), M)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def _adam_polish(a, deadline, max_iter=15000, lr=0.04):
    """Adam-based gradient polish using log-parameterization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_conv(a)
    best_score = 2.0 * n * float(c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_conv(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(3 * n)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        conv1 = np.fft.irfft(fp_rev * fw, N)
        conv2 = np.fft.irfft(fp * fw, N)
        vp = conv1[n - 1:2 * n - 1] + conv2[0:n]
        vp = 2.0 * vp + 1e-7 * p

        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_conv(a)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(2000, 3 * n):
            rng_r = np.random.default_rng((it * 7919 + n) % (2**32))
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_conv(a)
            beta = max(1e-8, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a, best_score


def _peak_chase(a, deadline, max_iter=20000):
    """Coordinate descent targeting the current peak lag."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a)
        peak_lag = int(np.argmax(c))
        M = float(c[peak_lag])

        # The peak at lag k: c[k] = sum_j a[j]*a[k-j]
        # To reduce it, move mass from high-contributing positions to low-contributing ones
        j_min = max(0, peak_lag - n + 1)
        j_max = min(n - 1, peak_lag)
        js = np.arange(j_min, j_max + 1)
        contributions = a[js] * a[peak_lag - js]
        total_contrib = float(contributions.sum())

        if total_contrib < 1e-15:
            break

        # Find the position with highest contribution
        max_local = int(np.argmax(contributions))
        i_src = js[max_local]

        # Find position with lowest "impact" on the peak (and overall)
        # Positions far from the peak lag have less impact
        if peak_lag == 0:
            # Only a[0] contributes; move mass away from a[0]
            i_src = 0
        elif peak_lag == 2 * n - 2:
            i_src = n - 1
        else:
            i_src = js[max_local]

        # Choose destination: position with minimum a[j] that doesn't contribute to peak
        peak_mask = np.zeros(n, dtype=bool)
        if 0 <= peak_lag:
            j_min2 = max(0, peak_lag - n + 1)
            j_max2 = min(n - 1, peak_lag)
            peak_mask[j_min2:j_max2 + 1] = True

        # Also mask positions that would create new peaks (lag 0 and 2n-2)
        # Actually, let's just pick a random position not in the peak region
        candidates = np.where(~peak_mask)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        i_dst = candidates[rng.integers(0, len(candidates))]

        # Determine transfer amount
        frac = rng.choice([0.02, 0.05, 0.1, 0.15, 0.2])
        delta = frac * a[i_src]
        if delta <= 0:
            no_imp += 1
            continue

        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break

    return best_a, best_score


def _spectral_init(n, rng):
    """Multi-frequency cosine initialization with Golomb-like frequencies."""
    n_freqs = int(rng.integers(3, min(15, n // 8) + 1))
    max_freq = max(2, n // 3)
    freqs = rng.choice(range(1, max_freq + 1), size=n_freqs, replace=False)
    phases = 2 * np.pi * rng.random(n_freqs)
    amplitudes = np.abs(rng.standard_normal(n_freqs)) + 0.5
    amplitudes = amplitudes / amplitudes.sum()

    t = np.arange(n)
    a = np.ones(n) * 3.0
    for f, p, amp in zip(freqs, phases, amplitudes):
        a += amp * np.cos(2.0 * np.pi * f * t / n + p)

    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _asymmetric_init(n, slope=2.0, rng=None):
    """Linearly asymmetric profile."""
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _u_shape_init(n, alpha=2.0):
    """U-shaped profile (large at both ends)."""
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = a / a.sum()
    return a


def _bimodal_init(n, center_sep=0.6, width=0.12):
    """Two Gaussian peaks."""
    t = np.linspace(0, 1, n)
    c1 = 0.5 - center_sep / 2
    c2 = 0.5 + center_sep / 2
    a = np.exp(-0.5 * ((t - c1) / width) ** 2) + np.exp(-0.5 * ((t - c2) / width) ** 2)
    a += 0.1
    a = a / a.sum()
    return a


def _random_smooth_init(n, rng, n_points=20):
    """Random smooth profile via spline-like interpolation."""
    x_pts = np.sort(rng.random(n_points))
    y_pts = np.abs(rng.normal(1.0, 0.5, n_points)) + 0.1
    x = np.linspace(0, 1, n)
    a = np.interp(x, x_pts, y_pts)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _pipeline_for_init(a0, n, deadline):
    """Run full optimization pipeline for a single initialization."""
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return None, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    if not np.isfinite(best_score):
        return None, np.inf

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 1: LP refinement (40% of budget)
    lp_end = time.monotonic() + remaining * 0.40
    a_lp, sc_lp = _lp_refine(a, lp_end, max_iter=10000, initial_blend=0.25)
    if sc_lp < best_score:
        best_score = sc_lp
        best_a = a_lp.copy()
    print(f"      LP: {sc_lp:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 2: Adam polish (25% of remaining)
    ad_end = time.monotonic() + remaining * 0.25
    a_ad, sc_ad = _adam_polish(best_a, ad_end, max_iter=10000, lr=0.03)
    if sc_ad < best_score:
        best_score = sc_ad
        best_a = a_ad.copy()
    print(f"      Adam: {sc_ad:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 3: LP again (20% of remaining)
    lp2_end = time.monotonic() + remaining * 0.20
    a_lp2, sc_lp2 = _lp_refine(best_a, lp2_end, max_iter=5000, initial_blend=0.15)
    if sc_lp2 < best_score:
        best_score = sc_lp2
        best_a = a_lp2.copy()
    print(f"      LP2: {sc_lp2:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 4: Peak chase (15% of remaining)
    pc_end = time.monotonic() + remaining * 0.15
    a_pc, sc_pc = _peak_chase(best_a, pc_end, max_iter=10000)
    if sc_pc < best_score:
        best_score = sc_pc
        best_a = a_pc.copy()
    print(f"      PeakChase: {sc_pc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score

    # Phase 5: Final Adam (remaining)
    a_fin, sc_fin = _adam_polish(best_a, time.monotonic() + remaining * 0.6,
                                 max_iter=5000, lr=0.02)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    print(f"      FinalAdam: {sc_fin:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: spectral/asymmetric inits, adaptive LP,
    Adam polish, peak-chase, multi-length search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define target lengths
    candidate_lengths = [384, 512, 640, 768, 1024, 1280, 1536]
    if parent is not None:
        pn = parent.size
        if pn not in candidate_lengths:
            candidate_lengths.insert(0, pn)

    print(f"Target lengths: {candidate_lengths}")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Multi-length search
    # Allocate time: more time for lengths near the parent
    n_lengths = len(candidate_lengths)
    base_time = (deadline - time.monotonic() - 250.0) / n_lengths

    for n_t in candidate_lengths:
        if time.monotonic() > deadline - 200:
            break

        # Allocate time based on proximity to parent length
        if parent is not None:
            proximity = 1.0 / (1.0 + abs(n_t - parent.size) / 256.0)
        else:
            proximity = 1.0 / (1.0 + abs(n_t - 768) / 256.0)
        time_alloc = base_time * (0.5 + 0.5 * proximity)
        time_alloc = max(30.0, min(time_alloc, deadline - time.monotonic() - 150.0))

        if time_alloc < 20:
            continue

        print(f"\n--- Length n={n_t}, time={time_alloc:.0f}s ---")
        phase_deadline = time.monotonic() + time_alloc

        # Determine number of initializations based on time
        num_inits = max(1, int(time_alloc / 40))

        for init_idx in range(num_inits):
            if time.monotonic() > phase_deadline - 15:
                break

            # Generate initialization
            if init_idx == 0 and parent is not None:
                a0 = _interpolate_profile(parent, n_t)
            elif init_idx == 1:
                a0 = _spectral_init(n_t, rng)
            elif init_idx == 2:
                a0 = _asymmetric_init(n_t, slope=rng.uniform(1.0, 3.0))
            elif init_idx == 3:
                a0 = _u_shape_init(n_t, alpha=rng.uniform(1.5, 4.0))
            elif init_idx == 4:
                a0 = _bimodal_init(n_t, center_sep=rng.uniform(0.4, 0.8),
                                   width=rng.uniform(0.08, 0.2))
            elif init_idx == 5:
                a0 = _random_smooth_init(n_t, rng, n_points=rng.integers(10, 30))
            else:
                # Perturbed parent or best
                if parent is not None:
                    base = _interpolate_profile(parent, n_t)
                elif best_seq is not None:
                    base = _interpolate_profile(best_seq, n_t)
                else:
                    base = _spectral_init(n_t, rng)
                pert = base * (1.0 + 0.2 * rng.standard_normal(n_t))
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                a0 = pert

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Run pipeline
            a_res, sc_res = _pipeline_for_init(a0, n_t, phase_deadline)
            if a_res is not None and sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"    ** New best at n={n_t}, init {init_idx}: {sc_res:.10f}")

    print(f"\nAfter Phase 1: best={best_score:.10f}")

    # Phase 2: Diverse inits at the best length
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = best_seq.size
        print(f"\nPhase 2: Diverse inits at n={n_best}")

        inits = []
        # More spectral inits with different parameters
        for _ in range(3):
            inits.append(_spectral_init(n_best, rng))
        # More asymmetric
        for slope in [0.8, 1.2, 1.8, 2.5, 3.5]:
            inits.append(_asymmetric_init(n_best, slope=slope))
        # U-shape
        for alpha in [1.0, 1.5, 2.5, 4.0, 6.0]:
            inits.append(_u_shape_init(n_best, alpha=alpha))
        # Bimodal
        for sep in [0.3, 0.5, 0.7]:
            for w in [0.08, 0.15, 0.25]:
                inits.append(_bimodal_init(n_best, center_sep=sep, width=w))
        # Perturbed best
        for _ in range(4):
            pert = best_seq * (1.0 + 0.1 * rng.standard_normal(n_best))
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            inits.append(pert)
        # Random smooth
        for _ in range(3):
            inits.append(_random_smooth_init(n_best, rng, n_points=rng.integers(15, 40)))

        # How many can we try?
        per_init_time = 50.0
        max_inits = max(1, int((deadline - time.monotonic() - 80.0) / per_init_time))
        inits = inits[:max_inits]

        for a0 in inits:
            if time.monotonic() > deadline - 80:
                break
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            rem = deadline - time.monotonic() - 60.0
            if rem < 20:
                break
            sub_deadline = time.monotonic() + min(50.0, rem * 0.8)
            a_res, sc_res = _pipeline_for_init(a0, n_best, sub_deadline)
            if a_res is not None and sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"    ** New best: {sc_res:.10f}")

    print(f"\nAfter Phase 2: best={best_score:.10f}")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        final_budget = deadline - time.monotonic() - 20.0
        print(f"\nPhase 3: Final polish, budget={final_budget:.0f}s, n={n_best}")

        # 3a: Aggressive LP
        lp_b = min(final_budget * 0.35, 150.0)
        if lp_b > 10:
            a_lp, sc = _lp_refine(best_seq, time.monotonic() + lp_b,
                                  max_iter=20000, initial_blend=0.2)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  Final LP: {sc:.10f}")

        # 3b: Adam
        ad_b = min(final_budget * 0.20, 80.0)
        if ad_b > 10:
            a_ad, sc = _adam_polish(best_seq, time.monotonic() + ad_b,
                                    max_iter=10000, lr=0.02)
            if sc < best_score:
                best_score = sc
                best_seq = a_ad.copy()
            print(f"  Final Adam: {sc:.10f}")

        # 3c: LP again
        lp_b2 = min(final_budget * 0.20, 60.0)
        if lp_b2 > 5:
            a_lp2, sc = _lp_refine(best_seq, time.monotonic() + lp_b2,
                                   max_iter=10000, initial_blend=0.1)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp2.copy()
            print(f"  Final LP2: {sc:.10f}")

        # 3d: Peak chase
        pc_b = min(final_budget * 0.10, 30.0)
        if pc_b > 5:
            a_pc, sc = _peak_chase(best_seq, time.monotonic() + pc_b, max_iter=5000)
            if sc < best_score:
                best_score = sc
                best_seq = a_pc.copy()
            print(f"  Final PeakChase: {sc:.10f}")

        # 3e: One more LP with very tight params
        lp_b3 = min(final_budget * 0.10, 30.0)
        if lp_b3 > 5:
            a_lp3, sc = _lp_refine(best_seq, time.monotonic() + lp_b3,
                                   max_iter=5000, initial_blend=0.08)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp3.copy()
            print(f"  Final LP3: {sc:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _asymmetric_init(512, slope=2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _asymmetric_init(512, slope=2.0)

    # Verify with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
