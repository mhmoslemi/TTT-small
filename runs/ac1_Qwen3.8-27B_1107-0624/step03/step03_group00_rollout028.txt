"""
LP-driven optimization with dual certificates for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Uses primal-dual LP fixed-point iteration with line search,
Adam smooth-max for fine-tuning, and multi-scale escalation.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _cross_correlate(w, a):
    """Compute (w * a_flipped)[n-1:2n-1] which gives the cross-correlation."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]

def _score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())

def _smooth_max_weights(c, beta):
    """Compute soft weights for smooth max."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w

def _build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (Ab)_k = (a*b)_k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_primal(a, M, n):
    """Solve max sum(b) s.t. A*b <= M, b >= 0."""
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _lp_dual(a, n):
    """Solve dual: min sum(y) s.t. A^T y >= 1, y >= 0.
    Returns (dual_obj, y_star)."""
    A = _build_conv_matrix(n, a)
    # Dual: min 1^T y s.t. A^T y >= 1, y >= 0
    # In standard form for linprog: min c^T y s.t. A_eq y = b_eq, A_ub y <= b_ub, bounds
    # A^T y >= 1  =>  -A^T y <= -1
    c_obj = np.ones(2 * n - 1)
    A_ub = -A.T
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return float(res.fun), np.maximum(res.x, 0.0)
    return None, None

def _lp_fixed_point(a, deadline, max_iters=50, use_line_search=True):
    """Iterative LP fixed-point: a <- b*/sum(b*) where b* solves the LP."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a)
        M = float(c.max())
        
        b = _lp_primal(a, M, n)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b = b / s_b
        
        if use_line_search:
            # Line search for best t
            best_t = 0.5
            best_sc = _score_norm(a)
            for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                a_t = (1 - t) * a + t * b
                a_t = np.maximum(a_t, 1e-15)
                a_t = a_t / a_t.sum()
                sc_t = _score_norm(a_t)
                if sc_t < best_sc:
                    best_sc = sc_t
                    best_t = t
            a = (1 - best_t) * a + best_t * b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
        else:
            a = b
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None, 
                     beta_decay=0.95, no_imp_limit=200):
    """Adam optimization in logit space with smooth max."""
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    
    z = np.log(a)
    z -= z.mean()
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = _full_convolve(a)
    maxc = float(c.max())
    beta = beta_init if beta_init is not None else 0.1 * maxc
    
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_correlate(w, a)
        
        # Gradient in logit space: a_j * (g_j - sum(a_j * g_j))
        dot = float(np.dot(a, g))
        gz = a * (g - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        c = _full_convolve(a)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if score < best_score - 1e-12:
            best_score = score
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Decay beta
        if t_adam % 50 == 0:
            beta = max(1e-4, beta * beta_decay)
        
        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = 0.1 if best_score < 1.51 else 0.2
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a)
            beta = 0.1 * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0
    
    # Return best
    a_best = np.exp(best_z - best_z.max())
    a_best = a_best / a_best.sum()
    return a_best, best_score

def _coord_descent(a, deadline, max_sweeps=100):
    """Coordinate descent: for each coordinate, try to optimize it while fixing others."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Current value
            old_val = a[j]
            # Try a few perturbations
            for factor in [0.5, 0.8, 1.2, 2.0]:
                a[j] = old_val * factor
                a[j] = max(a[j], 1e-15)
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
            a[j] = old_val
        if not improved:
            break
    
    return best_a, best_score

def _interp(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new

def _make_profile(n, profile_type, **params):
    """Generate various profile shapes."""
    if profile_type == 'linear':
        t = np.linspace(0, 1, n)
        eta = params.get('eta', 1.5)
        a = 1.0 + eta * (t - 0.5)
    elif profile_type == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + params.get('amp', 2.0) * (2*t - 1)**2
    elif profile_type == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2*t - 1
        a = 1.0 + params.get('a1', 1.5)*u + params.get('a3', -1.0)*u**3
    elif profile_type == 'exp':
        t = np.arange(n)
        a = np.exp(-params.get('lam', 1.0) * t / n)
    elif profile_type == 'cosine':
        t = np.arange(n)
        a = 1.0 + params.get('amp', 0.8) * np.cos(2*np.pi*params.get('freq', 1)*t/n)
    elif profile_type == 'gaussian':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5*((t-0.5)/params.get('sigma', 0.3))**2)
    elif profile_type == 'two_gaussian':
        t = np.linspace(0, 1, n)
        c1, c2 = params.get('c1', 0.3), params.get('c2', 0.7)
        s = params.get('s', 0.15)
        a = np.exp(-0.5*((t-c1)/s)**2) + np.exp(-0.5*((t-c2)/s)**2)
    elif profile_type == 'sine':
        t = np.arange(n)
        a = 1.0 + params.get('amp', 0.5) * np.sin(2*np.pi*params.get('freq', 2)*t/n)
    elif profile_type == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5)**2 + params.get('floor', 0.5)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 1e-12)
    return a / a.sum()

"""
LP-driven optimization with dual certificates and multi-scale escalation
for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines primal LP fixed-point with line search, Adam smooth-max in logit space,
coordinate descent, and dual LP lower bound estimation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _cross_correlate(w, a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * float(np.log(s)), w


def _build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_primal(a, M, n):
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_dual(a, n):
    A = _build_conv_matrix(n, a)
    c_obj = np.ones(2 * n - 1)
    A_ub = -A.T
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return float(res.fun), np.maximum(res.x, 0.0)
    return None, None


def _lp_fixed_point(a, deadline, max_iters=50, use_line_search=True):

    a = a / a.sum()

        M = float(c.max())

        s_b = float(b.sum())
        if s_b < 1e-10:

        b = b / s_b

        else:
            a = b

def _adam_smooth_max(a, deadline, rng, lr=0.05, beta_init=None,

    v = np.zeros(n)

        gz = a * (g - dot)
        gz -= gz.mean()

            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)

        a = a / a.sum()

            else:

    a_best = a_best / a_best.sum()
    return a_best, best_score


def _coord_descent(a, deadline, max_sweeps=50):

    a = np.maximum(a, 1e-15).copy()
    a = a / a.sum()

    best_a = a.copy()

                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
                    break
            a[j] = old_val
        if not improved:
            break

    return best_a, best_score


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    a_new = a_new / a_new.sum()
    return a_new


def _make_profile(n, profile_type, param1=1.0, param2=0.0, param3=0.0):
    if profile_type == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (t - 0.5)
    elif profile_type == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 + param1 * (2 * t - 1) ** 2
    elif profile_type == 'cubic':
        t = np.linspace(0, 1, n)
        u = 2 * t - 1
        a = 1.0 + param1 * u + param2 * u ** 3
    elif profile_type == 'exp':
        t = np.arange(n)
        a = np.exp(-param1 * t / n)
    elif profile_type == 'cosine':
        t = np.arange(n)
        a = 1.0 + param1 * np.cos(2 * np.pi * param2 * t / n)
    elif profile_type == 'sine':
        t = np.arange(n)
        a = 1.0 + param1 * np.sin(2 * np.pi * param2 * t / n)
    elif profile_type == 'parabola':
        t = np.linspace(0, 1, n)
        a = (t - 0.5) ** 2 + param1
    elif profile_type == 'two_gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - param1) / param3) ** 2) + np.exp(-0.5 * ((t - param2) / param3) ** 2)
    elif profile_type == 'gauss':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / param1) ** 2)
    elif profile_type == 'U_shape':
        t = np.linspace(0, 1, n)
        a = (2 * t - 1) ** 2 + param1
    elif profile_type == 'edge_heavy':
        t = np.arange(n) / max(1, n - 1)
        a = 1.0 / (0.01 + abs(t - 0.5) * 2.0) ** param1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _sort_candidates(candidates):
    return sorted(candidates, key=_sort_key)


def _sort_key(x):
    return (x[2], x[0])


def _sort_by_score(candidates):
    return sorted(candidates, key=_score_key)


def _score_key(x):
    return x[0]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining LP fixed-point with line search,
    Adam smooth-max, coordinate descent, dual LP certificates, and multi-scale
    escalation to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s, start time={t_start:.1f}")

    # === Phase 1: Generate diverse initial candidates ===
    print("Phase 1: Generating initial candidates...")
    candidates = []

    n_sizes = [128, 256, 512, 1024]
    profile_defs = [
        ('lin1.0', 'linear', 1.0, 0, 0),
        ('lin1.5', 'linear', 1.5, 0, 0),
        ('lin2.0', 'linear', 2.0, 0, 0),
        ('lin2.5', 'linear', 2.5, 0, 0),
        ('lin3.0', 'linear', 3.0, 0, 0),
        ('quad1', 'quadratic', 1.0, 0, 0),
        ('quad2', 'quadratic', 2.0, 0, 0),
        ('quad3', 'quadratic', 3.0, 0, 0),
        ('cubic1', 'cubic', 1.5, -1.0, 0),
        ('cubic2', 'cubic', 2.0, -1.5, 0),
        ('cubic3', 'cubic', 2.5, -2.0, 0),
        ('exp0.5', 'exp', 0.5, 0, 0),
        ('exp1.0', 'exp', 1.0, 0, 0),
        ('exp2.0', 'exp', 2.0, 0, 0),
        ('cos1_08', 'cosine', 0.8, 1, 0),
        ('cos2_06', 'cosine', 0.6, 2, 0),
        ('cos3_05', 'cosine', 0.5, 3, 0),
        ('sin2_05', 'sine', 0.5, 2, 0),
        ('sin3_07', 'sine', 0.7, 3, 0),
        ('para05', 'parabola', 0.5, 0, 0),
        ('para10', 'parabola', 1.0, 0, 0),
        ('g03', 'gauss', 0.3, 0, 0),
        ('tg30', 'two_gauss', 0.3, 0.7, 0.15),
        ('tg25', 'two_gauss', 0.25, 0.75, 0.2),
        ('U03', 'U_shape', 0.3, 0, 0),
        ('U05', 'U_shape', 0.5, 0, 0),
        ('edge1', 'edge_heavy', 1.0, 0, 0),
        ('edge15', 'edge_heavy', 1.5, 0, 0),
        ('edge2', 'edge_heavy', 2.0, 0, 0),
    ]

    for n in n_sizes:
        for (label, ptype, p1, p2, p3) in profile_defs:
            p = _make_profile(n, ptype, p1, p2, p3)
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in [256, 512, 1024, 2048]:
            pi = _interp(parent, n)
            sc = _score_norm(pi)
            candidates.append((sc, pi, n, 'parent'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent_exact'))

    candidates = _sort_by_score(candidates)
    print(f"  {len(candidates)} candidates, best init: {candidates[0][0]:.8f} n={candidates[0][2]} [{candidates[0][3]}]")

    if candidates[0][1] is not None and candidates[0][0] < best_score:
        best_seq = np.asarray(candidates[0][1], dtype=np.float64).copy()
        best_score = float(candidates[0][0])
        print(f"  BEST: {best_score:.10f} n={best_seq.size} [init]")

    # === Phase 2: LP fixed-point on top small-n candidates ===
    print("Phase 2: LP fixed-point optimization (small n)...")
    phase2_end = t_start + 500.0
    polished = []

    # Prefer smaller n for LP speed, then by score
    candidates_lp = _sort_candidates(candidates)

    count = 0
    for (sc, p, n, label) in candidates_lp:
        if time.monotonic() > phase2_end or count >= 10:
            break
        if n > 1024:
            continue
        # Allocate time based on n
        time_alloc = max(30.0, min(80.0, 500.0 * (256.0 / float(n))))
        per_end = min(phase2_end, time.monotonic() + time_alloc)

        a_lp, sc_lp = _lp_fixed_point(p, per_end, max_iters=40, use_line_search=True)
        if sc_lp < best_score:
            best_seq = np.asarray(a_lp, dtype=np.float64).copy()
            best_score = float(sc_lp)
            print(f"  BEST: {best_score:.10f} n={a_lp.size} [LP_{label}_n{n}]")
        polished.append((sc_lp, a_lp, n, label + '_lp'))
        count += 1

    polished = _sort_by_score(polished)
    print(f"  After Phase 2: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: Adam refinement on best LP results ===
    print("Phase 3: Adam smooth-max refinement...")
    phase3_end = t_start + 900.0

    for (sc, p, n, label) in polished[:4]:
        if time.monotonic() > phase3_end - 30:
            break
        per = min(100.0, phase3_end - time.monotonic() - 10)
        if per < 20:
            break
        a_ad, sc_ad = _adam_smooth_max(p, time.monotonic() + per, rng,
                                       lr=0.05, beta_decay=0.95, no_imp_limit=200)
        if sc_ad < best_score:
            best_seq = np.asarray(a_ad, dtype=np.float64).copy()
            best_score = float(sc_ad)
            print(f"  BEST: {best_score:.10f} n={a_ad.size} [Adam_{label}]")

        # LP after Adam for fine-tuning
        if n <= 1024 and time.monotonic() < phase3_end - 30:
            per2 = min(50.0, phase3_end - time.monotonic() - 10)
            if per2 > 15:
                a_lp2, sc_lp2 = _lp_fixed_point(a_ad, time.monotonic() + per2,
                                                max_iters=15, use_line_search=True)
                if sc_lp2 < best_score:
                    best_seq = np.asarray(a_lp2, dtype=np.float64).copy()
                    best_score = float(sc_lp2)
                    print(f"  BEST: {best_score:.10f} n={a_lp2.size} [LPpostAdam_{label}]")

    # === Phase 4: Multi-scale escalation ===
    print("Phase 4: Multi-scale escalation...")
    phase4_end = t_start + 1300.0

    if best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp(best_seq, tn)
            sc_int = _score_norm(p_int)
            print(f"  Interpolated to n={tn}: {sc_int:.8f}")
            if sc_int < best_score:
                best_seq = p_int.copy()
                best_score = sc_int
                print(f"  BEST: {best_score:.10f} n={tn} [interp]")

            # Adam on interpolated
            if time.monotonic() < phase4_end - 60:
                per = min(80.0, phase4_end - time.monotonic() - 30)
                if per > 20:
                    a_e, sc_e = _adam_smooth_max(p_int, time.monotonic() + per, rng,
                                                 lr=0.04, beta_decay=0.95, no_imp_limit=300)
                    if sc_e < best_score:
                        best_seq = a_e.copy()
                        best_score = sc_e
                        print(f"  BEST: {best_score:.10f} n={tn} [Adam_esc]")

            # LP at new scale (only if feasible)
            if tn <= 1024 and time.monotonic() < phase4_end - 30:
                per2 = min(50.0, phase4_end - time.monotonic() - 15)
                if per2 > 15:
                    a_le, sc_le = _lp_fixed_point(p_int, time.monotonic() + per2,
                                                  max_iters=12, use_line_search=True)
                    if sc_le < best_score:
                        best_seq = a_le.copy()
                        best_score = sc_le
                        print(f"  BEST: {best_score:.10f} n={tn} [LP_esc]")

    # === Phase 5: Dual LP for certificate-guided search ===
    print("Phase 5: Dual LP certificate search...")
    phase5_end = t_start + 1550.0

    if best_seq is not None and time.monotonic() < phase5_end:
        n_test = min(best_seq.size, 1024)
        p_test = _interp(best_seq, n_test) if best_seq.size != n_test else best_seq.copy()

        dual_obj, y_star = _lp_dual(p_test, n_test)
        if dual_obj is not None and np.isfinite(dual_obj):
            lower_bound = 2.0 * n_test / dual_obj
            print(f"  Dual LP lower bound (n={n_test}): {lower_bound:.10f}")
            print(f"  Current best: {best_score:.10f}")

            # Identify support from dual
            A_mat = _build_conv_matrix(n_test, p_test)
            Ay = np.asarray(A_mat.T @ y_star).ravel()
            Ay_max = float(Ay.max())
            threshold = Ay_max - 0.05 * Ay_max
            support = Ay >= threshold

            if 0 < support.sum() < n_test:
                p_masked = p_test.copy()
                p_masked[~support] = 1e-10
                p_masked = p_masked / p_masked.sum()
                sc_m = _score_norm(p_masked)
                print(f"  Masked (support={support.sum()}): {sc_m:.10f}")
                if sc_m < best_score:
                    best_seq = p_masked.copy()
                    best_score = sc_m
                    print(f"  BEST: {best_score:.10f} n={n_test} [dual_masked]")

                # Refine masked version
                if time.monotonic() < phase5_end - 20 and n_test <= 1024:
                    per = min(50.0, phase5_end - time.monotonic() - 10)
                    if per > 15:
                        a_dm, sc_dm = _lp_fixed_point(p_masked, time.monotonic() + per,
                                                      max_iters=15, use_line_search=True)
                        if sc_dm < best_score:
                            best_seq = a_dm.copy()
                            best_score = sc_dm
                            print(f"  BEST: {best_score:.10f} n={n_test} [LP_dual_masked]")

    # === Phase 6: Random restarts ===
    print("Phase 6: Random restarts...")
    phase6_end = deadline - 80.0

    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(15.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 10:
                break

            ns = 0.05 + 0.08 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()

            a_r, sc_r = _adam_smooth_max(a_s, time.monotonic() + per, rng,
                                         lr=0.05, beta_decay=0.94, no_imp_limit=150)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
                print(f"  BEST: {best_score:.10f} n={n_best} [restart{i}]")

            # Quick LP polish on restart result
            if n_best <= 1024 and time.monotonic() < phase6_end - 10:
                per2 = min(20.0, phase6_end - time.monotonic() - 5)
                if per2 > 10:
                    a_rl, sc_rl = _lp_fixed_point(a_r, time.monotonic() + per2,
                                                  max_iters=8, use_line_search=True)
                    if sc_rl < best_score:
                        best_seq = a_rl.copy()
                        best_score = sc_rl
                        print(f"  BEST: {best_score:.10f} n={n_best} [LP_restart{i}]")

    # === Phase 7: Final polish ===
    print("Phase 7: Final polish...")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 10
        if per > 15:
            if best_seq.size <= 1024:
                a_fp, sc_fp = _lp_fixed_point(best_seq, time.monotonic() + per,
                                              max_iters=25, use_line_search=True)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_LP]")
            else:
                a_fp, sc_fp = _adam_smooth_max(best_seq, time.monotonic() + per, rng,
                                               lr=0.02, beta_decay=0.96, no_imp_limit=100)
                if sc_fp < best_score:
                    best_seq = a_fp.copy()
                    best_score = sc_fp
                    print(f"  BEST: {best_score:.10f} n={best_seq.size} [final_Adam]")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile(512, 'linear', 1.5, 0, 0)

        exact_sc = _score_norm(final)
        print(f"\nFINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile(512, 'linear', 1.5, 0, 0)
    print("FALLBACK n=512")
    return [float(x) for x in fb]
