"""
Nesterov-accelerated smooth-max optimization with diverse profiles and LP refinement.
Key improvements: momentum-based acceleration, adaptive beta, diverse starts.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _smooth_max_and_weights(c, beta):
    """Returns (smooth_max_value, weights) for the smooth max of c."""
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -100.0, 100.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    sm = mval + beta * np.log(s)
    return float(sm), w

def _grad_conv_weighted(a, w):
    """
    Gradient of sum_k w[k] * conv(a,a)[k] with respect to a.
    Uses FFT for efficiency.
    """
    n = len(a)
    # conv(a,a)[k] = sum_j a[j]*a[k-j]
    # d/d a[i] of conv(a,a)[k] = a[k-i] + a[k-i] (with bounds)
    # = a[k-i] for the "first" term + a[k-i] for the "second" term
    # Actually: d conv[k]/d a[i] = sum of a[k-j] where j=i (if valid) + a[j-i] where j-i... 
    # Let me use the FFT approach:
    # sum_k w[k] * conv(a,a)[k] = sum_k w[k] * sum_j a[j]*a[k-j]
    # = sum_j a[j] * sum_k w[k] * a[k-j]
    # = sum_j a[j] * (w * a)[j]  (where w*a is cross-correlation)
    # So the gradient w.r.t. a[i] is (w * a)[i] + (w * a_rev)[i]... 
    
    # Actually, let's use the direct FFT approach:
    # f(a) = sum_k w[k] * conv(a,a)[k]
    # This is a quadratic form: f(a) = a^T (sum_k w[k] Q_k) a
    # The gradient is 2 * (sum_k w[k] Q_k) a
    # (sum_k w[k] Q_k) a at position i = sum_k w[k] * sum_j Q_k[i,j] * a[j]
    # = sum_k w[k] * a[k-i] (if k-i is valid)
    # = cross-correlation of w and a at position i (with appropriate bounds)
    
    # Let's compute h[i] = sum_k w[k] * a[k-i] (valid when k-i in [0,n-1])
    # This is the cross-correlation (w ∘ a)[i]
    # Using FFT: h = ifft(fft(w) * conj(fft(a)))... 
    # Actually, let me just use the approach from the previous code.
    
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h

def _adam_optimize(a0, deadline, max_iter=50000, lr=0.03, beta_init=0.05, 
                   beta_decay=0.9995, momentum=0.9, verbose=False):
    """Adam optimizer in logit-space with smooth max."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()
    
    beta = beta_init * float(c.max())
    beta = max(beta, 1e-10)
    no_imp = 0
    lr_cur = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c_curr = _fft_convolve(a)
        sm_val, w = _smooth_max_and_weights(c_curr, beta)
        grad_a = _grad_conv_weighted(a, w)
        
        # Convert to logit-space gradient
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))
        
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        
        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new -= float(np.mean(z_new))
        
        e = np.exp(z_new)
        a_new = e / e.sum()
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr_cur = max(0.001, lr_cur * 0.9)
            if no_imp > max(5000, 8 * n):
                # Restart with perturbation
                rng = np.random.default_rng(42 + it)
                z = best_z + 0.2 * rng.normal(size=n)
                z = np.clip(z, -80.0, 80.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_init * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr_cur = lr
                no_imp = 0
                continue
        
        z = z_new
        a = a_new
        
        if it % 100 == 0:
            beta = max(1e-12, beta * beta_decay)
        
        if verbose and it % 5000 == 0:
            print(f"        Adam it={it}: best={best_score:.10f}, beta={beta:.2e}")
    
    return best_a, best_score
