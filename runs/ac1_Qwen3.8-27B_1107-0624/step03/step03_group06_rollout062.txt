    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_n1(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        if (time.monotonic() - t_start_local) / max(1, it + 1) < min_time_per_iter and it > 50:
            # Check if we're running out of time
            pass

        a_new = _sca_lp_step(a)
        if a_new is None:
            break

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc = _score_n1(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                break

    return best_a


def _direction_lp_refine(a0, deadline, max_iter=2000, inflate_start=1.005):
    """Refine using direction-finding LP with line search."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_n1(a)
    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _direction_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.02, inflate * 1.3)
            if no_imp > 200:
                break
            continue

        g = g / sum_g

        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score_n1(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            inflate = max(1.0005, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.02, inflate * 1.2)
            if no_imp > max(150, n):
                break

    return best_a


def _mirror_descent(a0, deadline, max_iter=5000):
    """Mirror descent (multiplicative weights) with adaptive step size."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_n1(a)

    eta = 0.5
    beta = max(0.01, 0.1 * float(_fft_convolve(a).max()))
    no_imp = 0
    patience = 200

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        # Compute softmax weights
        x = (c - c_max) / beta
        ex = np.exp(x)
        w = ex / ex.sum()

        # Compute gradient: grad[j] = sum_k w[k] * a[k-j] (valid)
        # = sum_{k=0}^{n-1} a[k] * w[k+j]
        grad = np.zeros(n)
        for k in range(n):
            grad += a[k] * w[k:k + n]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score_n1(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > patience:
                # Reset and try different parameters
                a = best_a.copy()
                eta = max(0.01, eta * 0.5)
                beta = max(0.001, beta * 0.5)
                no_imp = 0
                if eta < 0.01:
                    break

        # Adapt beta (decrease over time for sharper weights)
        if it % 100 == 99:
            beta = max(0.001, beta * 0.9)
        # Adapt eta
        if it % 500 == 499:
            eta = max(0.01, eta * 0.95)

    return best_a


def _gradient_polish(a0, deadline, max_iter=20000):
    """Projected gradient descent with smooth max."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_n1(a)

    beta = max(0.001, 0.05 * float(_fft_convolve(a).max()))
    lr = 0.02
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        # Smooth max gradient
        x = (c - c_max) / beta
        ex = np.exp(x)
        w = ex / ex.sum()

        # grad[j] = sum_k w[k] * a[k-j]
        grad = np.zeros(n)
        for k in range(n):
            grad += a[k] * w[k:k + n]

        # Momentum update
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        sc = _score_n1(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                beta = max(0.001, 0.05 * float(_fft_convolve(a).max()))
                no_imp = 0
                if lr < 0.005:
                    break

        if it % 50 == 49:
            beta = max(0.0005, beta * 0.9)

    return best_a


def _combined_refine(a0, deadline):
    """Multi-stage refinement: SCA-LP -> direction LP -> mirror descent -> gradient polish -> SCA-LP."""
    n = len(a0)
    a = a0.copy()

    # Stage 1: SCA-LP (quick, few iterations)
    if time.monotonic() < deadline - 60:
        t_end = min(deadline - 30, time.monotonic() + 60)
        a_sc = _sca_lp_refine(a, t_end, max_iter=

50)
        if _score_n1(a_sc) < _score_n1(a):
            a = a_sc

    # Stage 2: Direction LP with line search
    if time.monotonic() < deadline - 120:
        t_end = min(deadline - 60, time.monotonic() + 120)
        a_dl = _direction_lp_refine(a, t_end, max_iter=2000)
        if _score_n1(a_dl) < _score_n1(a):
            a = a_dl

    # Stage 3: Mirror descent
    if time.monotonic() < deadline - 90:
        t_end = min(deadline - 45, time.monotonic() + 90)
        a_md = _mirror_descent(a, t_end, max_iter=5000)
        if _score_n1(a_md) < _score_n1(a):
            a = a_md

    # Stage 4: Gradient polish
    if time.monotonic() < deadline - 60:
        t_end = min(deadline - 30, time.monotonic() + 60)
        a_gp = _gradient_polish(a, t_end, max_iter=15000)
        if _score_n1(a_gp) < _score_n1(a):
            a = a_gp

    # Stage 5: Final SCA-LP
    if time.monotonic() < deadline - 20:
        t_end = min(deadline - 10, time.monotonic() + 30)
        a_final = _sca_lp_refine(a, t_end, max_iter=30)
        if _score_n1(a_final) < _score_n1(a):
            a = a_final

    return a

def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimizer: SCA-LP + direction LP + mirror descent + gradient polish.
    Tries multiple initializations and lengths, combines methods for best result.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initializations at various lengths
    lengths = [384, 512, 640, 768, 896, 1024]
    configs = []

    for n_t in lengths:
        if parent is not None:
            p_interp = _interpolate(parent, n_t)
            configs.append(('parent', n_t, p_interp))

        configs.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp1.5', n_t, _make_ramp(n_t, 1.5)))
        configs.append(('ramp2', n_t, _make_ramp(n_t, 2.0)))
        configs.append(('ramp_rev1', n_t, _make_ramp_rev(n_t, 1.0)))
        configs.append(('gauss1', n_t, _make_gaussian(n_t, 0.3)))
        configs.append(('gauss2', n_t, _make_gaussian(n_t, 0.2)))
        configs.append(('lin2', n_t, _make_lin(n_t, 2.0)))
        configs.append(('lin4', n_t, _make_lin(n_t, 4.0)))
        configs.append(('quad3', n_t, _make_quad(n_t, 3.0)))
        configs.append(('cosine', n_t, _make_cosine(n_t)))

    # Score and sort initializations
    scored = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        if np.isfinite(sc):
            scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    print(f"Top 5 initializations:")
    for i, (sc, name, n_t, _) in enumerate(scored[:5]):
        print(f"  {i+1}. {name} n={n_t} score={sc:.8f}")

    # Phase 1: Quick SCA-LP on top candidates
    for i, (sc0, name, n_t, a0) in enumerate(scored[:12]):
        if time.monotonic() > deadline - 300:
            break
        if sc0 >= best_score + 0.15:
            continue

        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Allocate time: more for better initializations
        if sc0 < best_score + 0.05:
            lp_time = min(120.0, (deadline - time.monotonic() - 200) * 0.3)
        else:
            lp_time = min(60.0, (deadline - time.monotonic() - 200) * 0.2)

        if lp_time < 10:
            break

        lp_end = time.monotonic() + lp_time
        print(f"SCA-LP: {name} n={n_t} score={sc0:.8f} budget={lp_time:.0f}s")

        a_sc = _sca_lp_refine_fixed(a0, lp_end, max_iter=80)
        sc_sc = _score(a_sc)
        if sc_sc < best_score:
            best_score = sc_sc
            best_seq = a_sc.copy()
            print(f"  -> SCA score={sc_sc:.10f}")

    # Phase 2: Combined refinement on the best
    if best_seq is not None and time.monotonic() < deadline - 150:
        comb_end = deadline - 60
        print(f"Combined refine: budget={comb_end - time.monotonic():.0f}s")
        a_comb = _combined_refine(best_seq, comb_end)
        sc_comb = _score(a_comb)
        if sc_comb < best_score:
            best_score = sc_comb
            best_seq = a_comb.copy()
            print(f"  -> Combined score={sc_comb:.10f}")

    # Phase 3: Try re-interpolating to nearby lengths
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = best_seq.size
        nearby_lengths = [n_best - 128, n_best - 64, n_best + 64, n_best + 128]
        nearby_lengths = [x for x in nearby_lengths if 128 <= x <= 2048]

        for n_try in nearby_lengths:
            if time.monotonic() > deadline - 50:
                break
            a_try = _interpolate(best_seq, n_try)
            sc_try = _score(a_try)
            if sc_try < best_score - 0.001:
                print(f"  Re-interp n={n_try}: score={sc_try:.8f}")
                # Quick refinement
                t_end = min(deadline - 30, time.monotonic() + 60)
                a_ref = _sca_lp_refine_fixed(a_try, t_end, max_iter=30)
                sc_ref = _score(a_ref)
                if sc_ref < best_score:
                    best_score = sc_ref
                    best_seq = a_ref.copy()
                    print(f"  -> refined score={sc_ref:.10f}")

    # Phase 4: Final polish on the very best
    if best_seq is not None and time.monotonic() < deadline - 30:
        gp_end = min(deadline - 10, time.monotonic() + 40)
        a_gp = _gradient_polish(best_seq, gp_end, max_iter=10000)
        sc_gp = _score(a_gp)
        if sc_gp < best_score:
            best_score = sc_gp
            best_seq = a_gp.copy()
            print(f"  Final gradient: score={sc_gp:.10f}")

    # Phase 5: One more SCA-LP
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_end = min(deadline - 5, time.monotonic() + 15)
        a_final = _sca_lp_refine_fixed(best_seq, t_end, max_iter=20)
        sc_final = _score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  Final SCA: score={sc_final:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.5)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.5)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
