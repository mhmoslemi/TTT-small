def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _adam_logit(p0, deadline, rng, lr=0.06, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=300, noise_scale=None):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.5)

        limit = max(80, no_imp_limit)
        if no_imp > limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.06
                elif best_score < 1.53:
                    ns = 0.12
                else:
                    ns = 0.25
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _peak_flood(a, deadline, eta=0.05, n_iter_max=5000):
    """Multiplicative peak-flooding: reduce mass at positions contributing to peaks."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)

    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break

        M = float(c.max())
        score = float(2.0 * n * M)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Identify hot positions (near-peak convolution values)
        threshold = M * 0.95
        hot_mask = c >= threshold
        n_hot = int(hot_mask.sum())
        if n_hot == 0:
            hot_mask[int(np.argmax(c))] = True

        # Compute heat for each position i: sum of a[k-i] over hot k
        heat = np.zeros(n)
        hot_indices = np.where(hot_mask)[0]
        for k in hot_indices:
            # conv[k] involves a[i] * a[k-i] for valid i
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                partner = k - np.arange(i_start, i_end + 1)
                heat[i_start:i_end + 1] += a[partner]

        # Normalize heat
        heat_max = float(heat.max())
        if heat_max < 1e-15:
            break
        heat = heat / heat_max

        # Multiplicative update: reduce mass at hot-contributing positions
        factor = 1.0 - eta * heat
        factor = np.maximum(factor, 0.01)  # floor to prevent zeroing
        a = a * factor
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        c = _full_convolve(a)

        # Adaptive eta: decrease if not improving
        if it % 100 == 0 and it > 0:
            if score >= best_score:
                eta = max(0.005, eta * 0.9)
            else:
                eta = min(0.15, eta * 1.05)

    return best_a, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine_extended(a, deadline, t_start=0.4, max_solves=50, m_shrink=0.995):
    """Extended LP refinement with decreasing M schedule."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    t = t_start

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1

        b = _lp_solve(a, M)
        if b is None:
            # Try with slightly larger M
            b = _lp_solve(a, M * 1.01)
            if b is None:
                break

        s_b = float(b.sum())
        if s_b < 1e-10:
            break

        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a)
        M_new = float(c.max())

        # Decrease M target to push peaks down further
        M = max(M_new * m_shrink, best_score / (2.0 * n) * 1.005)
        t = max(0.15, t * 0.92)

    return best_a, best_score


def _coordinate_descent(a, deadline, n_coord=500):
    """Targeted coordinate descent: adjust pairs of positions to reduce peak."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for it in range(n_coord):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        M = float(c.max())
        peak_k = int(np.argmax(c))
        score = float(2.0 * n * M)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Find the two most contributing positions to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        contrib = a[i_start:i_end + 1] * a[peak_k - (np.arange(i_start, i_end + 1))]
        top2_local = np.argsort(contrib)[::-1][:2]
        i1 = i_start + top2_local[0]
        i2 = i_start + top2_local[1] if len(top2_local) > 1 else (i_start + (i_end - i_start) // 2)

        # Move mass from i1 to i2 (or a random position)
        # Choose a random target position that doesn't contribute to the peak
        valid_targets = np.arange(n)
        # Exclude positions that are partners in the peak
        partner_set = set()
        for k_val in range(max(0, peak_k - 2), min(2 * n - 1, peak_k + 3)):
            for i_val in range(max(0, k_val - n + 1), min(n, k_val + 1)):
                partner_set.add(k_val - i_val)
        valid_targets = [x for x in valid_targets if x not in partner_set and x != i1]

        if len(valid_targets) == 0:
            continue

        target = valid_targets[rng.integers(0, len(valid_targets))]
        # Move a small fraction of mass
        frac = 0.01 + 0.02 * (1.0 - a[i1] * n)
        amount = a[i1] * frac
        a[i1] -= amount
        a[target] += amount
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

    return best_a, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc):
            sc = _score_norm(p)  # Re-verify with exact computation
            if sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = float(sc)
                print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent score: {sc_parent:.10f}")

    # --- Phase 1: Quick Adam polish on parent (150s) ---
    phase1_end = t_start + 150.0
    if parent is not None:
        print("Phase 1: Adam polish on parent...")
        p1, sc1 = _adam_logit(parent, phase1_end, rng, lr=0.05, beta_frac=0.07,
                              beta_decay=0.93, no_imp_limit=200)
        _update(p1, _score_norm(p1), 'adam_parent')
    else:
        p1 = _make_profile_linear(1024, 2.0)
        _update(p1, _score_norm(p1), 'linear_init')

    # --- Phase 2: Peak flooding (300s) ---
    phase2_end = t_start + 450.0
    print(f"Phase 2: Peak flooding, score={best_score:.10f}...")
    if best_seq is not None:
        p2, sc2 = _peak_flood(best_seq, phase2_end, eta=0.04)
        _update(p2, sc2, 'peak_flood')

    # --- Phase 3: Extended LP refinement (400s) ---
    phase3_end = t_start + 850.0
    print(f"Phase 3: LP refinement, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048:
        p3, sc3 = _lp_refine_extended(best_seq, phase3_end, t_start=0.35,
                                       max_solves=40, m_shrink=0.997)
        _update(p3, sc3, 'lp_ext')

        # Quick Adam after LP
        if time.monotonic() < phase3_end - 10:
            p3a, sc3a = _adam_logit(best_seq, min(phase3_end, time.monotonic() + 60),
                                     rng, lr=0.03, beta_frac=0.05,
                                     beta_decay=0.95, no_imp_limit=150)
            _update(p3a, sc3a, 'post_lp_adam')

    # --- Phase 4: Coordinate descent (150s) ---
    phase4_end = t_start + 1000.0
    print(f"Phase 4: Coordinate descent, score={best_score:.10f}...")
    if best_seq is not None and best_seq.size <= 2048:
        p4, sc4 = _coordinate_descent(best_seq, phase4_end, n_coord=300)
        _update(p4, sc4, 'coord_descent')

        # Another LP pass
        if time.monotonic() < phase4_end - 10 and best_seq.size <= 2048:
            p4l, sc4l = _lp_refine_extended(best_seq, min(phase4_end, time.monotonic() + 60),
                                             t_start=0.25, max_solves=10, m_shrink=0.999)
            _update(p4l, sc4l, 'lp_pass2')

    # --- Phase 5: Try n=2048 (250s) ---
    phase5_end = t_start + 1250.0
    print(f"Phase 5: Multi-scale, score={best_score:.10f}...")
    if time.monotonic() < phase5_end and best_seq is not None and best_seq.size <= 1024:
        n_new = 2048
        p_int = _interp_profile(best_seq, n_new)
        if p_int is not None:
            sc_int = _score_norm(p_int)
            print(f"  Interpolated to n={n_new}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{n_new}')

            # Adam polish at new scale
            per = min(80.0, phase5_end - time.monotonic() - 20)
            if per > 30:
                p5, sc5 = _adam_logit(p_int, time.monotonic() + per, rng,
                                       lr=0.04, beta_frac=0.06,
                                       beta_decay=0.94, no_imp_limit=200)
                _update(p5, sc5, f'adam_n{n_new}')

            # Peak flood at new scale
            if time.monotonic() < phase5_end - 20 and best_seq is not None:
                p5f, sc5f = _peak_flood(best_seq, min(phase5_end, time.monotonic() + 60), eta=0.03)
                _update(p5f, sc5f, f'flood_n{best_seq.size}')

    # --- Phase 6: Alternative initializations (200s) ---
    phase6_end = t_start + 1450.0
    print(f"Phase 6: Alt inits, score={best_score:.10f}...")
    if time.monotonic() < phase6_end:
        n_try = 1024
        alts = [
            ('lin2.5', _make_profile_linear(n_try, 2.5)),
            ('lin3.0', _make_profile_linear(n_try, 3.0)),
            ('cubic_2', _make_profile_cubic(n_try, 2.0, -1.5)),
            ('cubic_3', _make_profile_cubic(n_try, 2.5, -2.0)),
            ('cos1_08', _make_profile_cosine(n_try, 1, 0.8)),
            ('cos1_12', _make_profile_cosine(n_try, 1, 1.2)),
            ('cos2_05', _make_profile_cosine(n_try, 2, 0.5)),
        ]
        per_alt = max(15.0, (phase6_end - time.monotonic()) / len(alts))
        for label, pa in alts:
            if time.monotonic() > phase6_end - 10:
                break
            # Quick Adam from this init
            per = min(per_alt, phase6_end - time.monotonic() - 5)
            if per < 10:
                break
            p_alt, sc_alt = _adam_logit(pa, time.monotonic() + per, rng,
                                         lr=0.05, beta_frac=0.08,
                                         beta_decay=0.93, no_imp_limit=150)
            _update(p_alt, sc_alt, f'alt_{label}')

    # --- Phase 7: Final intensive polish (remaining time) ---
    phase7_end = deadline
    print(f"Phase 7: Final polish, score={best_score:.10f}...")
    if time.monotonic() < phase7_end and best_seq is not None:
        # Alternating: Adam -> Flood -> LP -> Adam
        remaining = phase7_end - time.monotonic()
        if remaining > 60:
            # Adam
            per1 = remaining * 0.3
            p7a, sc7a = _adam_logit(best_seq, time.monotonic() + per1, rng,
                                     lr=0.02, beta_frac=0.04,
                                     beta_decay=0.96, no_imp_limit=100)
            _update(p7a, sc7a, 'final_adam1')

        if time.monotonic() < phase7_end - 20 and best_seq.size <= 2048:
            per2 = min(60.0, phase7_end - time.monotonic() - 10)
            if per2 > 20:
                p7f, sc7f = _peak_flood(best_seq, time.monotonic() + per2, eta=0.02)
                _update(p7f, sc7f, 'final_flood')

        if time.monotonic() < phase7_end - 20 and best_seq.size <= 2048:
            per3 = min(60.0, phase7_end - time.monotonic() - 10)
            if per3 > 20:
                p7l, sc7l = _lp_refine_extended(best_seq, time.monotonic() + per3,
                                                 t_start=0.2, max_solves=10, m_shrink=0.999)
                _update(p7l, sc7l, 'final_lp')

        if time.monotonic() < phase7_end - 5:
            per4 = phase7_end - time.monotonic() - 5
            if per4 > 15:
                p7b, sc7b = _adam_logit(best_seq, time.monotonic() + per4, rng,
                                         lr=0.01, beta_frac=0.03,
                                         beta_decay=0.97, no_imp_limit=50)
                _update(p7b, sc7b, 'final_adam2')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 2.0)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(1024, 2.0)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
