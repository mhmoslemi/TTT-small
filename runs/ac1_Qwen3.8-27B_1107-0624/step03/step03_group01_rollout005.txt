LP-based direction finding + coordinate descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Combined with coordinate descent for fine-tuning.

from scipy.optimize import linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

        b_ub.append(M)

    if n == 0:
        return v

    s = v.sum()
    if s > 0:
        v = v * (target_sum / s)
    else:
        return np.full(n, target_sum / n)
    
    # Project: subtract threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]

    w = np.maximum(v - theta, 0)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    c_obj = -np.ones(n)
    A_ub_rows = []
    b_ub_vals = []
    for k in binding_lags:
        row = np.zeros(n)
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            row[j] = a[k - j]
        A_ub_rows.append(row)
        b_ub_vals.append(M)
    A_ub_rows.append(np.ones(n))
    b_ub_vals.append(cap)
    A_arr = np.array(A_ub_rows)
    b_arr = np.array(b_ub_vals)
    bounds = [(0, None)] * n
    try:

            s = b.sum()
            if s > 1e-10:

def _coordinate_descent(a, deadline, max_rounds=200):
    """Coordinate descent: try adjusting pairs of coordinates."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            # Try increasing a[i] and decreasing a[j] for a few j
            c = _full_convolve(a, a)
            M = c.max()
            # Find j that would best benefit from being decreased
            # Approximate: decrease the coordinate with highest "contribution to peak"
            j_candidates = rng.choice(n, size=min(5, n), replace=False)
            for j in j_candidates:
                if j == i:

                d = a[i] * 0.1
                a_trial = a.copy()
                a_trial[i] += d
                a_trial[j] -= d
                if a_trial[j] < 0:
                    d = -a_trial[j] + a[j]
                    a_trial = a.copy()
                    a_trial[i] += d
                    a_trial[j] -= d
                if a_trial[j] < 0:
                    continue
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
        if not improved and rd > 5:
            break

    return best_a, best_score


def _adam_refine(a, deadline, lr=0.02, n_max_iter=50000):
    """Adam-based refinement in log-softmax space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    a_curr = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        c = _full_convolve(a_curr, a_curr)
        cm = float(c.max())
        # Softmax-smoothed gradient
        shift = c - cm
        beta = max(1e-6, 0.005 * cm)
        c_s = shift / beta
        e = np.exp(c_s)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a_curr[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a_curr * (h - W)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a_curr = _softmax_fn(z)

        sc = _score_norm(a_curr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
        if t % 1000 == 0:
            lr = max(0.001, lr * 0.95)

    return best_a, best_score

def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bump_profile(n, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_tapered(n, power):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 - power * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP direction finding + coordinate descent + Adam polish.
    Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None:
            sc = float(sc)
            if np.isfinite(sc) and sc < best_score:
                best_seq = np.asarray(p, dtype=np.float64).copy()
                best_score = sc
                print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
    else:
        print("No parent available, using profiles.")

    # Phase 0: Evaluate initial candidates (30s)
    phase0_end = t_start + 30.0
    candidates = []

    if parent is not None:
        sc_p = _score_norm(parent)
        candidates.append((sc_p, parent.copy(), len(parent), "parent"))
        update_best(parent, sc_p, "parent")

    for n_try in [512, 1024, 2048]:
        if time.monotonic() > phase0_end:
            break
        profiles = [
            _make_linear_profile(n_try, 1.0),
            _make_linear_profile(n_try, 1.5),
            _make_linear_profile(n_try, 2.0),
            _make_linear_profile(n_try, 2.5),
            _make_quadratic_profile(n_try, 1.0, 0.3),
            _make_quadratic_profile(n_try, 1.5, -0.2),
            _make_cubic_profile(n_try, 1.0, -0.5),
            _make_bump_profile(n_try, 0.3),
            _make_bump_profile(n_try, 0.4),
            _make_tapered(n_try, 0.5),
            _make_tapered(n_try, 0.7),
        ]
        for idx, p in enumerate(profiles):
            sc = _score_norm(p)
            candidates.append((sc, p, n_try, f"prof{idx}_n{n_try}"))
            update_best(p, sc, f"prof{idx}_n{n_try}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # Phase 1: LP iteration on top candidates (700s)
    phase1_end = t_start + 730.0
    seen_labels = set()
    count = 0
    for sc, p, pn, label in candidates:
        if time.monotonic() > phase1_end:
            break
        if count >= 8:
            break
        if sc > best_score * 1.02 + 0.01:
            break
        key = f"{pn}_{label[:6]}"
        if key in seen_labels:
            continue
        seen_labels.add(key)

        per_deadline = min(phase1_end, time.monotonic() + 80.0)
        print(f"  LP on {label} (n={pn}), current best from this: {best_score:.10f}")
        p_best, sc_best = _lp_iterate(p, per_deadline, verbose=True)
        update_best(p_best, sc_best, f"LP_{label}")
        count += 1

    print(f"Phase 1 done: best={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # Phase 2: LP on best with more time (400s)
    phase2_end = t_start + 1130.0
    if time.monotonic() < phase2_end and best_seq is not None:
        per_dl = min(phase2_end, time.monotonic() + 350.0)
        p_best, sc_best = _lp_iterate(best_seq, per_dl, verbose=True)
        update_best(p_best, sc_best, "LP_deep")

    # Phase 3: Multi-restart with perturbation (200s)
    phase3_end = t_start + 1330.0
    if time.monotonic() < phase3_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = min(45.0, (phase3_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 5:
                break
            noise_scale = 0.05 + 0.03 * i
            a_pert = best_seq * (1 + noise_scale * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s < 1e-10:
                continue
            a_pert = a_pert / s
            per_dl = min(phase3_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _lp_iterate(a_pert, per_dl, verbose=False)
            update_best(p_best, sc_best, f"restart{i}_n{n_best}")

    # Phase 4: Adam polish (200s)
    phase4_end = t_start + 1530.0
    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = min(phase4_end - 5, time.monotonic() + 180.0)
        p_best, sc_best = _adam_refine(best_seq, per_dl, lr=0.015, n_max_iter=40000)
        update_best(p_best, sc_best, "adam_polish")

    # Phase 5: Try larger n by interpolation (150s)
    phase5_end = t_start + 1680.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr <= 2048:
            target_n = n_curr * 2
            if target_n <= 4096:
                p_interp = _interp_profile(best_seq, target_n)
                sc_interp = _score_norm(p_interp)
                update_best(p_interp, sc_interp, f"interp_n{target_n}")
                if time.monotonic() < phase5_end - 30:
                    per_dl = min(phase5_end - 10, time.monotonic() + 140.0)
                    p_best, sc_best = _lp_iterate(p_interp, per_dl, verbose=False)
                    update_best(p_best, sc_best, f"LP_n{target_n}")

    # Phase 6: Final LP + coordinate descent (50s)
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        per_dl = min(phase6_end - 10, time.monotonic() + 30.0)
        p_best, sc_best = _lp_iterate(best_seq, per_dl, verbose=False)
        update_best(p_best, sc_best, "final_LP")

        if time.monotonic() < phase6_end - 5:
            per_dl2 = min(phase6_end - 3, time.monotonic() + 15.0)
            p_best2, sc_best2 = _coordinate_descent(best_seq, per_dl2, max_rounds=50)
            update_best(p_best2, sc_best2, "final_CD")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(512, 1.5)

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_linear_profile(512, 1.5)
    print(f"FALLBACK: n=512, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
