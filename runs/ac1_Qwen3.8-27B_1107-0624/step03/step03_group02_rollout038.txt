"""
Hybrid optimization: coordinate descent + refined smooth-max Adam for minimizing
autocorrelation peak of non-negative sequences. Uses proportional redistribution
coordinate descent (equivalent to log-space descent) combined with adaptive
smooth-max gradient methods and Fourier-based diversification.

import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def _coord_descent_step(a, i, b, n):
    t = a[i]
    if t < 1e-14:
        return 0.0
    if t > 1.0 - 1e-10:
        return t

    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    two_i = 2 * i

        alpha = (1.0 - x) / (1.0 - t)
        alpha2 = alpha * alpha
        vals = alpha2 * b
        if k_start <= k_end:

            a_idx = ks - i
            av = a[a_idx]

            if mask.any():

def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / max(beta, 1e-30)
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h

def _subgradient_opt(a, deadline, max_iter=10000, top_k=3, verbose=False):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.4 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_idx = np.argsort(c_curr)[::-1][:k]

        sg = np.zeros(n)
        for rank, idx in enumerate(top_idx):
            idx = int(idx)
            weight = 1.0 / (1.0 + 0.5 * rank)
            j1_s = max(0, idx - n + 1)
            j1_e = min(idx, n - 1)
            if j1_s <= j1_e:
                jj = np.arange(j1_s, j1_e + 1)
                sg[jj] += weight * a[idx - jj]
            j2_s = max(idx, 0)
            j2_e = min(idx + n - 1, n - 1)
            if j2_s <= j2_e:
                jj = np.arange(j2_s, j2_e + 1)
                sg[jj] += weight * a[jj - idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-12:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(4000, 5 * n):
                break
        a = a_new

    return best_a, best_score


def _fourier_perturb(a, rng, magnitude=0.1):
    n = len(a)
    fa = np.fft.rfft(a)
    freqs = np.arange(len(fa))
    mask = freqs >= 2
    perturb = np.zeros_like(fa)
    perturb[mask] = magnitude * rng.normal(size=mask.sum())
    fa_pert = fa + perturb
    a_pert = np.fft.irfft(fa_pert, n)
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _full_pipeline(a0, deadline, cd_time_frac=0.4, adam_time_frac=0.35, sg_time_frac=0.15, verbose=False):
    a0 = np.asarray(a0, dtype=np.float64).ravel().copy()
    a0 = np.maximum(a0, 0.0)
    a0 = _normalize(a0)

    best_a = a0.copy()
    best_sc = _score(a0)
    n = len(a0)

    remaining = deadline - time.monotonic()
    if remaining <= 10:
        return best_a, best_sc

    cd_deadline = time.monotonic() + remaining * cd_time_frac
    a1, s1 = _coord_descent(a0, cd_deadline, max_sweeps=50000, recompute_every=16, verbose=verbose)
    if s1 < best_sc:
        best_a, best_sc = a1, s1
    if verbose:
        print(f"    CD: {best_sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        adam_deadline = time.monotonic() + remaining * (adam_time_frac / (adam_time_frac + sg_time_frac))
        a2, s2 = _adam_smooth_max(best_a, adam_deadline, max_iter=20000, lr=0.015, beta_factor=0.03, verbose=verbose)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        if verbose:
            print(f"    Adam: {best_sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        sg_deadline = time.monotonic() + remaining * 0.6
        a3, s3 = _subgradient_opt(best_a, sg_deadline, max_iter=8000, top_k=3)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        if verbose:
            print(f"    SG: {best_sc:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        a4, s4 = _coord_descent(best_a, time.monotonic() + remaining * 0.8, max_sweeps=10000, recompute_every=8)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        if verbose:
            print(f"    CD2: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"No parent, using uniform: score={best_score:.10f}")

    n_parent = parent.size

    # Try multiple lengths
    n_candidates = [n_parent]
    if n_parent != 512:
        n_candidates.append(512)
    if n_parent != 1024:
        n_candidates.append(1024)
    if n_parent != 2048:
        n_candidates.append(2048)

    n_candidates = sorted(set(n_candidates))

    # Phase 1: Optimize at parent length (highest priority)
    print(f"=== Phase 1: n={n_parent} ===")
    d1 = min(deadline - 400, t_start + 400)
    a1, s1 = _full_pipeline(parent, d1, cd_time_frac=0.5, adam_time_frac=0.3, sg_time_frac=0.2, verbose=True)
    if s1 < best_score:
        best_score = s1
        best_seq = a1.copy()
    print(f"Phase 1 best: {best_score:.10f}")

    # Phase 2: Try other lengths
    for nc in n_candidates:
        if nc == n_parent:
            continue
        if time.monotonic() > deadline - 200:
            break
        print(f"=== Phase 2: n={nc} ===")
        base = _interpolate_profile(best_seq, nc)
        d2 = min(deadline - 200, time.monotonic() + 180)
        a2, s2 = _full_pipeline(base, d2, cd_time_frac=0.45, adam_time_frac=0.35, sg_time_frac=0.2, verbose=False)
        if s2 < best_score:
            best_score = s2
            best_seq = a2.copy()
            print(f"  Improved at n={nc}: {best_score:.10f}")
        else:
            print(f"  n={nc}: {s2:.10f} (no improvement)")

    # Phase 3: Fourier-perturbed restarts at best length
    if time.monotonic() < deadline - 100:
        print(f"=== Phase 3: Fourier perturbations ===")
        n_best = best_seq.size
        for pi in range(5):
            if time.monotonic() > deadline - 100:
                break
            perturbed = _fourier_perturb(best_seq, rng, magnitude=0.08 + 0.03 * pi)
            d3 = min(deadline - 100, time.monotonic() + 60)
            a3, s3 = _full_pipeline(perturbed, d3, cd_time_frac=0.5, adam_time_frac=0.3, sg_time_frac=0.2, verbose=False)
            if s3 < best_score:
                best_score = s3
                best_seq = a3.copy()
                print(f"  Perturb {pi}: {best_score:.10f} (improved!)")
            else:
                print(f"  Perturb {pi}: {s3:.10f}")

    # Phase 4: Final focused optimization on best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"=== Phase 4: Final polish (score={best_score:.10f}) ===")
        remaining = deadline - time.monotonic()

        # Coordinate descent polish
        cd_dead = time.monotonic() + min(remaining * 0.5, 120)
        a4, s4 = _coord_descent(best_seq, cd_dead, max_sweeps=50000, recompute_every=4)
        if s4 < best_score:
            best_score = s4
            best_seq = a4.copy()
        print(f"  CD polish: {best_score:.10f}")

        # Adam fine-tune
        if time.monotonic() < deadline - 15:
            ad = time.monotonic() + min(deadline - time.monotonic() - 15, 80)
            a5, s5 = _adam_smooth_max(best_seq, ad, max_iter=10000, lr=0.005, beta_factor=0.01)
            if s5 < best_score:
                best_score = s5
                best_seq = a5.copy()
            print(f"  Adam fine: {best_score:.10f}")

        # Final CD
        if time.monotonic() < deadline - 10:
            a6, s6 = _coord_descent(best_seq, time.monotonic() + min(deadline - time.monotonic() - 10, 60), max_sweeps=30000, recompute_every=4)
            if s6 < best_score:
                best_score = s6
                best_seq = a6.copy()
            print(f"  Final CD: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(1024) / 1024.0

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final > 0.01:
        final_seq = final_seq / s_final
    else:
        final_seq = np.ones(1024) / 1024.0

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
