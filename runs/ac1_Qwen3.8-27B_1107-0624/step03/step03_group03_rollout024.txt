    return None

    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _lse_grad(a, T):
    """Compute gradient of 2n*logsumexp(conv(a,a)/T) with respect to a.
    Assumes sum(a)=1. Returns (smoothed_obj, grad)."""
    n = len(a)
    c = _fft_convolve(a)
    c_shift = (c - c.max()) / T
    c_shift = np.clip(c_shift, -50.0, 0.0)
    w = np.exp(c_shift)
    w_sum = w.sum()
    w = w / w_sum

    obj = 2.0 * n * (float(c.max()) + T * np.log(w_sum))

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)
    grad = 2.0 * h[n - 1:2 * n - 1]

    return obj, grad


def _adam_lse(a, deadline, max_iter=80000, T_start=0.08, T_end=5e-5, lr=0.06):
    """LSE-smoothed Adam optimization on the simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    T = T_start
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter, 2))

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999

    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        _, grad = _lse_grad(a, T)

        # Project to maintain sum=1: subtract weighted mean
        grad = grad - np.dot(a, grad) * a

        t += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        score = _score_norm(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 3000 == 0:
                lr = max(0.005, lr * 0.9)
            if no_imp > max(8000, 8 * n):
                rng = np.random.default_rng(777 + it)
                a_pert = best_a * (1.0 + 0.08 * rng.normal(size=n))
                a_pert = np.maximum(a_pert, 1e-15)
                a = a_pert / a_pert.sum()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = base_lr
                no_imp = 0

        a = a_new
        T = max(T * T_ratio, T_end)

    return best_a


def _lp_spread(a, M, top_k=120):
    """LP: maximize sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    c_aa = _fft_convolve(a)

    if n <= 256:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a[j]
        b_vec = np.full(2 * n - 1, M)
    else:
        idx = np.argsort(c_aa)[::-1][:top_k]
        active = set()
        for i in idx:
            for offset in range(-3, 4):
                j = i + offset
                if 0 <= j < 2 * n - 1:
                    active.add(j)
        active = sorted(active)
        A = np.zeros((len(active), n))
        for row, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                A[row, j] = a[j]
        b_vec = np.full(len(active), M)

    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.15):
    """LP-based spreading with line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    no_imp = 0
    cur_inflate = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        g = _lp_spread(a, M * cur_inflate)

        if g is None:
            cur_inflate = min(1.01, cur_inflate * 1.15)
            no_imp += 5
            if no_imp > 500:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 30 == 0:
                cur_inflate = min(1.01, cur_inflate * 1.08)
            if no_imp > 400:
                break
            continue

        g = g / sum_g

        # Line search with multiple candidates
        best_t = 0.0
        best_s = best_score
        for t in [0.03, 0.07, 0.12, 0.2, 0.35, 0.5]:
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            s_val = _score_norm(a_try)
            if s_val < best_s:
                best_s = s_val
                best_t = t

        if best_t == 0.0:
            no_imp += 1
            if no_imp > 200:
                a = best_a.copy()
                cur_inflate = inflate
                no_imp = 0
            continue

        a_new = np.maximum((1 - best_t) * a + best_t * g, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score = _score_norm(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 300:
                a = best_a.copy()
                cur_inflate = inflate
                no_imp = 0

        a = a_new

    return best_a


def _subgrad_polish(a, deadline, max_iter=40000, lr=0.012):
    """Subgradient polish with soft-max weighting."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    v = np.zeros(n)
    momentum = 0.85
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        T_sg = 0.005
        c_shift = (c - M) / T_sg
        c_shift = np.clip(c_shift, -50.0, 0.0)
        w = np.exp(c_shift)
        w_sum = w.sum()
        w = w / w_sum

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)
        grad = 2.0 * h[n - 1:2 * n - 1]

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        score = _score_norm(a_new)
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.001, lr * 0.95)
            if no_imp > max(5000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

        a = a_new

    return best_a


def _make_power_law(n, alpha):
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t ** (-alpha)
    return a / a.sum()


def _make_exponential(n, beta):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-beta * t / max(n - 1, 1))
    return a / a.sum()


def _make_cosine(n):
    t = np.linspace(0.0, np.pi, n)
    a = np.sin(t)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _make_symmetric_cosine(n):
    """Raised cosine: 0.5*(1+cos(2*pi*t))"""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interpolate(a0, n_new):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n0 = a0.size
    if n0 == n_new:
        s = a0.sum()
        if s > 1e-30:
            return a0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: diverse initializations + LSE-Adam + LP + subgradient.
    Uses beam search to keep top candidates and tries multiple lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 1024, 2048]:
        if ln not in lengths:
            lengths.append(ln)

    def generate_candidates(n_t):
        cands = []
        if parent is not None:
            c0 = _interpolate(parent, n_t)
            cands.append(c0)
            # Perturbed parent
            p1 = c0 * (1.0 + 0.1 * rng.normal(size=n_t))
            p1 = np.maximum(p1, 1e-15)
            cands.append(p1 / p1.sum())
        # Power laws
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cands.append(_make_power_law(n_t, alpha))
        # Exponential
        for beta in [1.0, 2.0, 3.0]:
            cands.append(_make_exponential(n_t, beta))
        # Cosine / Gaussian
        cands.append(_make_cosine(n_t))
        cands.append(_make_symmetric_cosine(n_t))
        cands.append(_make_gaussian(n_t, 0.3))
        cands.append(_make_gaussian(n_t, 0.45))
        cands.append(_make_uniform(n_t))
        return cands

    # Phase 1: LSE-Adam on diverse candidates (beam search)
    print("\n=== Phase 1: LSE-Adam optimization ===")
    beam = []  # list of (score, sequence)

    for n_t in lengths:
        if time.monotonic() > deadline - 300:
            break

        cands = generate_candidates(n_t)
        local_best = []

        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 300:
                break

            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            lse_budget = min(180.0, deadline - time.monotonic() - 300.0)
            if lse_budget < 15:
                break
            lse_end = time.monotonic() + lse_budget

            a_opt = _adam_lse(a0, lse_end, max_iter=60000,
                              T_start=0.08, T_end=5e-5, lr=0.06)
            sc = _score(a_opt)
            if np.isfinite(sc):
                local_best.append((sc, a_opt.copy()))
                if sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()

            print(f"  n={n_t} cand[{ci}]: init={sc0:.8f} -> LSE={sc:.10f}")

        # Keep top 3 per length
        local_best.sort(key=lambda x: x[0])
        beam.extend(local_best[:3])
        if local_best:
            print(f"  Best n={n_t}: {local_best[0][0]:.10f}")

    # Phase 2: LP refinement on beam candidates
    print("\n=== Phase 2: LP refinement ===")
    beam.sort(key=lambda x: x[0])
    top_candidates = beam[:6]

    for i, (sc, a_in) in enumerate(top_candidates):
        if time.monotonic() > deadline - 200:
            break

        n_t = len(a_in)
        lp_budget = min(200.0, deadline - time.monotonic() - 200.0)
        if lp_budget < 20:
            break
        lp_end = time.monotonic() + lp_budget

        a_lp = _lp_refine(a_in, lp_end, max_iter=4000, inflate=1.001, blend=0.15)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"  LP[{i}] n={n_t}: {sc:.10f} -> {sc_lp:.10f}")

        # Second LP pass with tighter inflate
        if time.monotonic() < deadline - 150:
            lp_budget2 = min(100.0, deadline - time.monotonic() - 150.0)
            if lp_budget2 > 15:
                a_lp2 = _lp_refine(a_lp, time.monotonic() + lp_budget2,
                                   max_iter=2000, inflate=1.0005, blend=0.1)
                sc_lp2 = _score(a_lp2)
                if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                    best_score = sc_lp2
                    best_seq = a_lp2.copy()
                print(f"  LP2[{i}]: {sc_lp:.10f} -> {sc_lp2:.10f}")

    # Phase 3: Subgradient polish
    print("\n=== Phase 3: Subgradient polish ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        sp_budget = min(150.0, deadline - time.monotonic() - 100.0)
        if sp_budget > 20:
            a_sp = _subgrad_polish(best_seq, time.monotonic() + sp_budget,
                                   max_iter=50000, lr=0.01)
            sc_sp = _score(a_sp)
            if np.isfinite(sc_sp) and sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            print(f"  Subgrad: {best_score:.10f} -> {sc_sp:.10f}")

    # Phase 4: Final LSE on best
    if best_seq is not None and time.monotonic() < deadline - 60:
        lse_budget = min(80.0, deadline - time.monotonic() - 60.0)
        if lse_budget > 10:
            a_final = _adam_lse(best_seq, time.monotonic() + lse_budget,
                                max_iter=40000, T_start=0.03, T_end=1e-5, lr=0.03)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LSE: -> {sc_final:.10f}")

    # Phase 5: Final LP micro-polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        lp_budget = min(40.0, deadline - time.monotonic() - 20.0)
        if lp_budget > 10:
            a_micro = _lp_refine(best_seq, time.monotonic() + lp_budget,
                                 max_iter=1000, inflate=1.0003, blend=0.08)
            sc_micro = _score(a_micro)
            if np.isfinite(sc_micro) and sc_micro < best_score:
                best_score = sc_micro
                best_seq = a_micro.copy()
            print(f"  Micro LP: -> {sc_micro:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    # Final normalization and validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
