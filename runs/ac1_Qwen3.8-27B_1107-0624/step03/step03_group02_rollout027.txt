    c = _fft_convolve(a)
    cm = c.max()
    x = np.clip((c - cm) / beta, -50, 50)
    ex = np.exp(x)
    s = ex.sum()
    smax = cm + beta * np.log(s)

    v[:2*n-1] = w

    af[:n] = a[::-1]

    e = np.exp(z)
    return e / e.sum()

    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return smax, grad_z

"""
L-BFGS quasi-Newton optimization with smooth-max annealing for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses multiple initializations, sequence lengths,
and a beta-cooling schedule to approach the global optimum. Includes subgradient
polishing and LP refinement as secondary strategies.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    """Score for a sequence normalized to sum=1: 2*n*max_conv."""
    n = len(a)
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _smooth_max_vals(c, beta):
    """Compute smooth max value and softmax weights."""
    cm = float(c.max())
    x = np.clip((c - cm) / beta, -80.0, 80.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return cm, w
    smax = cm + beta * np.log(s)
    w = ex / s
    return smax, w


def _grad_smooth_max_a(a, w):
    """Gradient of smooth_max(conv(a,a)) w.r.t. a, given weights w on convolution coefficients."""
    n = len(a)
    N = _next_pow2(3 * n)
    v = np.zeros(N)
    v[:2 * n - 1] = w
    af = np.zeros(N)
    af[:n] = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(v, N) * np.fft.rfft(af, N), N)
    return 2.0 * conv[n - 1:2 * n - 1]


def _softmax_z(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-30:
        return np.ones_like(z) / len(z)
    return e / s


def _obj_and_grad_z(z, beta):
    """Objective and gradient in log-space (z = log a, a = softmax(z))."""
    a = _softmax_z(z)
    c = _fft_convolve(a)
    smax, w = _smooth_max_vals(c, beta)
    grad_a = _grad_smooth_max_a(a, w)
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)
    return smax, grad_z


def _lbfgs_anneal(a0, deadline, beta_init=None, n_beta_steps=15, max_iter_per_beta=300, verbose=False):
    """Run L-BFGS with decreasing beta (smooth max temperature)."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    z = np.log(a0)
    z -= z.mean()

    c0 = _fft_convolve(a0)
    max_c0 = float(c0.max())
    best_a = a0.copy()
    best_score = 2.0 * n * max_c0
    best_smax = max_c0

    if beta_init is None:
        beta_init = max_c0 * 0.25
    beta = beta_init
    min_beta = max_c0 * 0.0005

    for step in range(n_beta_steps):
        if time.monotonic() > deadline:
            break
        if beta < min_beta:
            break

        def fun(z_in):
            val, _ = _obj_and_grad_z(z_in, beta)
            return val

        def gfun(z_in):
            _, g = _obj_and_grad_z(z_in, beta)
            return g

        try:
            res = minimize(fun, z, jac=gfun, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_beta, 'gtol': 1e-10,
                                    'maxls': 30, 'maxfun': 5000})
        except Exception:
            break

        a_new = _softmax_z(res.x)
        c_new = _fft_convolve(a_new)
        smax_new = float(c_new.max())

        if smax_new < best_smax - 1e-15:
            best_smax = smax_new
            best_a = a_new.copy()
            best_score = 2.0 * n * smax_new
            z = res.x.copy()
        else:
            # Try current z evaluation
            a_curr = _softmax_z(z)
            c_curr = _fft_convolve(a_curr)
            smax_curr = float(c_curr.max())
            if smax_curr < best_smax - 1e-15:
                best_smax = smax_curr
                best_a = a_curr.copy()
                best_score = 2.0 * n * smax_curr

        beta *= 0.65
        if verbose:
            print(f"      beta step {step}: beta={beta:.2e}, best_score={best_score:.10f}, "
                  f"res_msg={res.message}")

    return best_a, best_score


def _subgradient_polish(a0, deadline, max_iter=5000, verbose=False):
    """Subgradient method with backtracking on the exact max."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(c.max())
    best_cmax = float(c.max())

    base_lr = 0.5 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())
        max_idx = int(np.argmax(c_curr))

        # Subgradient at max_idx
        sg = np.zeros(n)
        j_start = max(0, max_idx - n + 1)
        j_end = min(max_idx, n - 1)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += a[max_idx - idx]
        j2_start = max(max_idx, 0)
        j2_end = min(max_idx + n - 1, n - 1)
        if j2_start <= j2_end:
            idx = np.arange(j2_start, j2_end + 1)
            sg[idx] += a[idx - max_idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        # Backtracking line search
        lr = base_lr / np.sqrt(it + 1)
        improved = False
        for _bt in range(20):
            a_new = a - lr * sg / sg_norm
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                lr *= 0.5
                continue
            a_new = a_new / s_new
            c_new = _fft_convolve(a_new)
            new_cmax = float(c_new.max())
            if new_cmax < max_val * (1.0 - 1e-4):
                a = a_new
                improved = True
                break
            lr *= 0.7

        if improved:
            c_check = _fft_convolve(a)
            cmax_check = float(c_check.max())
            if cmax_check < best_cmax - 1e-12:
                best_cmax = cmax_check
                best_score = 2.0 * n * cmax_check
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 2000:
            break

    return best_a, best_score


def _lp_refine(a0, deadline, max_iter=2000, inflate=1.002, blend=0.3):
    """LP-based refinement: find g maximizing sum(g) s.t. a*g <= M, blend with a."""
    a = np.maximum(a0, 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * inflate_cur

        # LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0
        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.03)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = 2.0 * n * float(c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
                inflate_cur = min(1.01, inflate_cur * 0.9)
            if no_imp > max(100, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _interpolate_to_length(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_random_init(n, rng, sparsity=0.0):
    """Random initialization with optional sparsity."""
    a = rng.random(n) + 1e-6
    if sparsity > 0:
        mask = rng.random(n) < (1.0 - sparsity)
        a *= mask
        a += 1e-10
    return a / a.sum()


def _make_sinusoidal_init(n, freq=1.0, amp=0.3):
    t = np.linspace(0, 1, n)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_polynomial_init(n, coeffs=None):
    t = np.linspace(0, 1, n)
    if coeffs is None:
        a = 1.0 + 0.5 * t - 0.3 * t * t
    else:
        a = np.zeros(n)
        for i, c in enumerate(coeffs):
            a += c * t ** i
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _gradient_descent_momentum(a0, deadline, lr=0.01, momentum=0.9, verbose=False):
    """Momentum gradient descent in log-space with smooth max."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)

    z = np.log(a0)

        if score_new < best_score - 1e-12:

            z = z_new.copy()

        else:

            z += 0.2 * np.random.default_rng(t).normal(size=n)

    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(256) / 256
    else:
        final_seq = final_seq / s

    # Validate with evaluate_sequence
    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
            if not np.isfinite(sc_eval):
                # Fallback
                final_seq = np.ones(256) / 256
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                print(f"Fallback evaluate_sequence: {sc_eval:.10f}")
        except Exception as e:
            print(f"Evaluation error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
