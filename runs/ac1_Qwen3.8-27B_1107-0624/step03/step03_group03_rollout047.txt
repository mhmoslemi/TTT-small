- For each element i (in random or importance order):
  - Try a small perturbation: a_i +- delta
  - Renormalize
  - If score improves, accept
  - Use a decreasing delta schedule

This is simple but can find very fine improvements that gradient methods miss.

Let me also add a "paired swap" operation:
- Pick two elements i, j
- Move mass from i to j (or vice versa): a_i -= delta, a_j += delta
- Renormalize
- Accept if score improves

This preserves the sum automatically and can find improvements that single-element changes miss.

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines smoothed gradient descent, LP-based KKT refinement,
coordinate descent polish, and multi-length exploration.
Key new ideas: KKT-informed LP, paired-swap coordinate descent,
arcsine initializations, and aggressive fine polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix(a, top_k=200):
    n = len(a)
    c = _fft_convolve(a)
    idx = np.argsort(c)[::-1][:top_k]
    active = sorted(set(idx))
    A = np.zeros((len(active), n))
    b = np.full(len(active), 0.0)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
        b[row] = c[k]
    return A, b


def _lp_spread_kkt(a, M_target, top_k=100):
    """LP: maximize sum(b) subject to conv(a,b) <= M_target on active lags, b>=0."""
    n = len(a)
    c = _fft_convolve(a)
    idx = np.argsort(c)[::-1][:top_k]
    active = sorted(set(idx))
    A = np.zeros((len(active), n))
    b_vec = np.full(len(active), M_target)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_spread_full(a, M_target, n_constraints=300):
    """LP with more constraints for better accuracy."""
    n = len(a)
    c = _fft_convolve(a)
    idx = np.argsort(c)[::-1][:n_constraints]
    active = sorted(set(idx))
    A = np.zeros((len(active), n))
    b_vec = np.full(len(active), M_target)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[row, lo:hi] = a[lo:hi]
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search_blend(a, g, max_iters=25):
    n = len(a)
    best_score = _score_normalized(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = _score_normalized(np.maximum(a + c_pt * (g - a), 0))
    fd = _score_normalized(np.maximum(a + d_pt * (g - a), 0))
    for _ in range(max_iters):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = _score_normalized(np.maximum(a + c_pt * (g - a), 0))
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = _score_normalized(np.maximum(a + d_pt * (g - a), 0))
    best_t = (lo + hi) / 2.0
    return best_t


def _lp_refine(a, deadline, max_iter=3000, inflate=1.0005, blend=0.2,
               n_constraints=200, use_line_search=True):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread_full(a, M_lp, n_constraints=n_constraints)
        if g is None:
            inflate_cur = min(1.01, inflate_cur * 1.3)
            no_imp += 5
            if no_imp > 400:
                break
            continue
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.1)
            if no_imp > 500:
                break
            continue
        g = g / sum_g
        if use_line_search:
            t_opt = _line_search_blend(a, g)
            if t_opt < 0.01:
                t_opt = blend_cur * 0.5
        else:
            t_opt = blend_cur
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        score_new = _score_normalized(a_new)
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.02, blend_cur * 0.85)
                inflate_cur = max(1.0001, inflate_cur * 0.9)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        a = a_new
    return best_a


def _lse_adam_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()
    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a, deadline, max_iter=100000, T_start=0.04, T_end=1e-6, lr=0.08):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    best_a = a.copy()
    best_score = _score_normalized(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    if max_iter > 1:
        T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    else:
        T_ratio = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a, z, state, m_val = _lse_adam_step(a, z, state, T, lr)
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1500 == 0:
                lr = max(0.005, lr * 0.85)
            if no_imp > max(8000, 12 * n):
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0
        T *= T_ratio
        T = max(T, T_end)
    return best_a


def _coordinate_descent_polish(a, deadline, max_rounds=500, delta_scale=0.01):
    """Polish using coordinate descent with paired swaps."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_normalized(a)
    rng = np.random.default_rng(999)
    no_imp = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        delta = delta_scale / (1.0 + rnd * 0.1)
        improved = False

        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            # Try single element change
            a_try = a.copy()
            a_try[i] += delta
            s_try = a_try.sum()
            if s_try > 0:
                a_try = a_try / s_try
                sc = _score_normalized(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try
                    improved = True

            a_try = a.copy()
            a_try[i] = max(0, a_try[i] - delta)
            s_try = a_try.sum()
            if s_try > 0:
                a_try = a_try / s_try
                sc = _score_normalized(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try
                    improved = True

            # Paired swap: move mass from i to j
            j = rng.integers(0, n)
            if j != i:
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try > 0:
                    a_try = a_try / s_try
                    sc = _score_normalized(a_try)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_try.copy()
                        a = a_try
                        improved = True

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break

    return best_a


def _subgradient_polish(a, deadline, max_iter=50000, lr=0.01):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    v = np.zeros(n)
    momentum = 0.7
    no_imp = 0
    base_lr = lr
    top_k = 50

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        # Use top-k soft weighting
        idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros_like(c)
        c_top = c[idx]
        w_top = (c_top - c_top.min() + 1e-10)
        w_top = w_top / w_top.sum()
        w[idx] = w_top

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        score = _score_normalized(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.001, lr * 0.85)
            if no_imp > max(5000, 10 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
    return best_a


def _make_arcsine(n, alpha=0.3):
    """Regularized arcsine distribution (heavy at edges)."""
    t = np.linspace(0.5 / n, 1.0 - 0.5 / n, n)
    a = 1.0 / np.power(t * (1.0 - t), alpha)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_edge_weighted(n, weight=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, sep=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-10.0 * (t - 0.3) ** 2) + np.exp(-10.0 * (t - 0.7) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _refine_sequence(a, deadline):
    """Full refinement pipeline for a single sequence."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best = a
    best_sc = _score_normalized(a)

    # Phase A: LSE gradient
    t_a = min(120.0, (deadline - time.monotonic()) * 0.35)
    if t_a > 10:
        a_lse = _lse_optimize(a, time.monotonic() + t_a, max_iter=60000,
                              T_start=0.03, T_end=1e-6, lr=0.06)
        sc = _score_normalized(a_lse)
        if sc < best_sc:
            best_sc = sc
            best = a_lse
        a = a_lse

    # Phase B: LP refinement (coarse)
    rem = deadline - time.monotonic()
    t_b = min(180.0, rem * 0.35)
    if t_b > 15:
        a_lp = _lp_refine(a, time.monotonic() + t_b, max_iter=2000,
                          inflate=1.001, blend=0.15, n_constraints=300)
        sc = _score_normalized(a_lp)
        if sc < best_sc:
            best_sc = sc
            best = a_lp
        a = a_lp

    # Phase C: Subgradient polish
    rem = deadline - time.monotonic()
    t_c = min(100.0, rem * 0.25)
    if t_c > 10:
        a_sp = _subgradient_polish(a, time.monotonic() + t_c, max_iter=40000, lr=0.008)
        sc = _score_normalized(a_sp)
        if sc < best_sc:
            best_sc = sc
            best = a_sp
        a = a_sp

    # Phase D: LP refinement (fine)
    rem = deadline - time.monotonic()
    t_d = min(120.0, rem * 0.25)
    if t_d > 10:
        a_lp2 = _lp_refine(a, time.monotonic() + t_d, max_iter=3000,
                           inflate=1.0003, blend=0.08, n_constraints=500,
                           use_line_search=True)
        sc = _score_normalized(a_lp2)
        if sc < best_sc:
            best_sc = sc
            best = a_lp2
        a = a_lp2

    # Phase E: Coordinate descent polish
    rem = deadline - time.monotonic()
    t_e = min(60.0, rem * 0.3)
    if t_e > 5:
        a_cd = _coordinate_descent_polish(a, time.monotonic() + t_e,
                                          max_rounds=n, delta_scale=0.005)
        sc = _score_normalized(a_cd)
        if sc < best_sc:
            best_sc = sc
            best = a_cd

    # Phase F: Final fine LP
    rem = deadline - time.monotonic()
    t_f = min(30.0, rem * 0.5)
    if t_f > 5:
        a_final = _lp_refine(best, time.monotonic() + t_f, max_iter=1000,
                             inflate=1.0001, blend=0.05, n_constraints=800)
        sc = _score_normalized(a_final)
        if sc < best_sc:
            best_sc = sc
            best = a_final

    return best

            if time.monotonic() > screen_end:
                break
            sc0 = _score_normalized(a0)
            if sc0 < best_here_sc:
                best_here_sc = sc0
                best_here = a0.copy()

            # Quick LSE
            qe = min(25.0, screen_end - time.monotonic() - 5.0)
            if qe > 5:
                a_lse = _lse_optimize(a0, time.monotonic() + qe,
                                      max_iter=20000, T_start=0.05, T_end=1e-4, lr=0.1)
                sc_lse = _score_normalized(a_lse)
                if sc_lse < best_here_sc:
                    best_here_sc = sc_lse
                    best_here = a_lse

        if best_here is not None:
            results[n_t] = (best_here, best_here_sc, 0)
            sc_full = _score(best_here)
            print(f"  n={n_t}: screening score={best_here_sc:.10f} (full={sc_full:.10f})")
            if sc_full < best_score:
                best_score = sc_full
                best_seq = best_here.copy()

    # Phase 2: Deep optimization on top lengths
    print("\n=== Phase 2: Deep optimization ===")
    # Sort lengths by screening score (best first)
    sorted_lengths = sorted(results.keys(), key=lambda x: results[x][1])

    for n_t in sorted_lengths:
        rem = deadline - time.monotonic() - 40.0
        if rem < 60:
            break
        # Allocate time based on rank
        rank = sorted_lengths.index(n_t)
        frac = max(0.15, 0.5 - rank * 0.1)
        deep_budget = min(rem * frac, 400.0)
        if deep_budget < 30:
            continue

        a_init = results[n_t][0]
        print(f"  Deep optimizing n={n_t}, budget={deep_budget:.0f}s")

        a_deep = _refine_sequence(a_init, time.monotonic() + deep_budget)
        sc = _score(a_deep)
        sc_norm = _score_normalized(a_deep)
        print(f"  n={n_t}: deep score={sc:.10f} (norm={sc_norm:.10f})")

        if sc < best_score:
            best_score = sc
            best_seq = a_deep.copy()
        results[n_t] = (a_deep, sc_norm, deep_budget)

    # Phase 3: Final polish on the overall best
    print("\n=== Phase 3: Final polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10.0
        if rem > 30:
            n_best = len(best_seq)
            # Try a few more refinements
            a_p = _refine_sequence(best_seq, time.monotonic() + min(rem * 0.7, 120.0))
            sc_p = _score(a_p)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p
                print(f"  After polish: {sc_p:.10f}")

            # Coordinate descent on the very best
            rem2 = deadline - time.monotonic() - 10.0
            if rem2 > 15:
                a_cd = _coordinate_descent_polish(best_seq, time.monotonic() + min(rem2 * 0.8, 60.0),
                                                  max_rounds=len(best_seq)//2, delta_scale=0.003)
                sc_cd = _score(a_cd)
                if sc_cd < best_score:
                    best_score = sc_cd
                    best_seq = a_cd
                    print(f"  After coord descent: {sc_cd:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_arcsine(1024, 0.3)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(1024) / 1024.0

    sc_eval = _score(final_seq)
    print(f"\nFinal internal score: {sc_eval:.10f}, n={final_seq.size}")

    try:
        sc_official = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_official):
            print(f"Official evaluate_sequence score: {sc_official:.10f}")
            if sc_official < sc_eval:
                sc_eval = sc_official
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={sc_eval:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
