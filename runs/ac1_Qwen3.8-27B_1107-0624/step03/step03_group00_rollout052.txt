"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) spectral and multi-scale initialization, (2) Adam smooth-max in logit
space with adaptive temperature, (3) LP-based refinement, (4) spectral equalization,
(5) multi-length exploration (1024/2048/4096), and (6) aggressive restarts.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=15):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.1, t * 0.9)
    return current_best, current_score


def _spectral_flatten(p, n_iter=20, tau=0.5):
    """Flatten the magnitude spectrum via iterative projection."""
    n = len(p)
    N = _next_pow2(n)
    for _ in range(n_iter):
        A = np.fft.rfft(p, N)
        mag = np.abs(A)
        # Soft-clip the magnitude to flatten the spectrum
        m_ref = np.median(mag)
        if m_ref < 1e-15:
            break
        # Scale: reduce peaks, boost troughs
        ratio = mag / m_ref
        # Soft-cap at (1+tau) and soft-floor at (1-tau)
        ratio_new = np.clip(ratio, 1.0 - tau, 1.0 + tau)
        # Reconstruct with modified magnitude, same phase
        phase = np.angle(A)
        A_new = ratio_new * mag / mag * np.exp(1j * phase)  # keep mag, modulate
        # Actually: new_mag = m_ref * ratio_new
        new_mag = m_ref * ratio_new
        # Interpolate to get smooth transition
        A_new = new_mag * np.exp(1j * phase)
        p_new = np.fft.irfft(A_new, N)[:n].real
        p_new = np.maximum(p_new, 1e-15)
        p_new = p_new / p_new.sum()
        p = p_new
    return p


def _adam_logit(p0, deadline, rng, lr=0.06, beta_frac=0.1, beta_decay=0.92,
                no_imp_limit=400, noise_scale=None, spectral_flatten_every=0):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_store = np.zeros(n)
    v_store = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    iter_count = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_store.fill(0.0)
            v_store.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        iter_count += 1
        m_store = b1 * m_store + (1.0 - b1) * gz
        v_store = b2 * v_store + (1.0 - b2) * (gz * gz)
        mh = m_store / (1.0 - b1 ** t_adam)
        vh = v_store / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_store.fill(0.0)
            v_store.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        # Spectral flatten periodically
        if spectral_flatten_every > 0 and iter_count % spectral_flatten_every == 0:
            p_flat = _spectral_flatten(p, n_iter=5, tau=0.3)
            sc_flat = _score_norm(p_flat)
            if sc_flat < score:
                p = p_flat
                z = np.log(np.maximum(p, 1e-12))
                z -= z.mean()
                c = _full_convolve(p)
                maxc = float(c.max())
                score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.003, lr * 0.9)
        else:
            z = best_z.copy()
            m_store.fill(0.0)
            v_store.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 40 == 0:
            beta = max(1e-5, beta * beta_decay)
            # Adaptive: if top-2 gap is small relative to beta, decrease faster
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.4)

        limit = max(100, no_imp_limit)
        if no_imp > limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.06
                elif best_score < 1.53:
                    ns = 0.12
                else:
                    ns = 0.25
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m_store.fill(0.0)
            v_store.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
            iter_count = 0

    return best_p, best_score


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_multi_cos(n, rng, n_terms=3):
    """Random multi-cosine profile."""
    t = np.arange(n)
    a = np.ones(n)
    for _ in range(n_terms):
        freq = rng.integers(1, 5)
        amp = rng.uniform(0.1, 0.6)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_block(n, block_size, rng):
    """Multi-scale block structure."""
    a = np.ones(n)
    n_blocks = n // block_size
    for b in range(n_blocks):
        scale = rng.uniform(0.3, 1.5)
        a[b * block_size:(b + 1) * block_size] *= scale
    # Add smooth envelope
    t = np.linspace(0, 1, n)
    a *= (0.5 + 0.5 * np.cos(np.pi * (t - 0.5)) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_fourier(n, rng):
    """Initialize in Fourier domain for flat spectrum."""
    N = _next_pow2(n)
    A = np.zeros(N // 2 + 1, dtype=np.complex128)
    A[0] = 1.0
    # Add random low-frequency components
    for k in range(1, min(10, N // 4)):
        A[k] = rng.normal(0, 0.3) + 1j * rng.normal(0, 0.3)
        A[N - k] = np.conj(A[k])
    a = np.fft.irfft(A, N)[:n].real
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _adam_logit_v2(p0, deadline, rng, lr=0.08, beta_init_frac=0.15, beta_decay=0.9,
                   no_imp_limit=300, restart_noise=None, use_spectral=False):
    """Improved Adam with faster beta decay and optional spectral steps."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.85, 0.999, 1e-8
    m_store = np.zeros(n)
    v_store = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_init_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    spec_counter = 0

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m_store.fill(0.0)
            v_store.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m_store = b1 * m_store + (1.0 - b1) * gz
        v_store = b2 * v_store + (1.0 - b2) * (gz * gz)
        mh = m_store / (1.0 - b1 ** t_adam)
        vh = v_store / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m_store.fill(0.0)
            v_store.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        # Periodic spectral flatten
        if use_spectral and t_adam % 200 == 0:
            spec_counter += 1
            p_flat = _spectral_flatten(p, n_iter=3, tau=0.2 * max(0.1, 1.0 - spec_counter * 0.1))
            sc_flat = _score_norm(p_flat)
            if sc_flat < score - 1e-8:
                p = p_flat
                z = np.log(np.maximum(p, 1e-12))
                z -= z.mean()
                c = _full_convolve(p)
                maxc = float(c.max())
                score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 400 == 0:
                    lr = max(0.005, lr * 0.93)
        else:
            z = best_z.copy()
            m_store.fill(0.0)
            v_store.fill(0.0)
            t_adam = 0
            continue

        # Fast beta decay
        if t_adam % 30 == 0:
            beta = max(1e-6, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.03:
                    beta = max(1e-6, beta * 0.3)

        limit = max(80, no_imp_limit)
        if no_imp > limit:
            if restart_noise is None:
                if best_score < 1.51:
                    ns = 0.05
                elif best_score < 1.53:
                    ns = 0.1
                else:
                    ns = 0.2
            else:
                ns = restart_noise
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_init_frac * maxc)
            m_store.fill(0.0)
            v_store.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score

    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # --- Phase 1: Generate and evaluate diverse candidates ---
    phase1_end = t_start + 90.0
    candidates = []

    lengths = [1024, 2048, 4096]
    if parent is not None and 0 < pn <= 4096:
        lengths = list(set(lengths + [pn]))
        lengths.sort()

    for n in lengths:
        profs = [
            ('lin1.5', _make_profile_linear(n, 1.5)),
            ('lin2.0', _make_profile_linear(n, 2.0)),
            ('lin3.0', _make_profile_linear(n, 3.0)),
            ('cubic1', _make_profile_cubic(n, 1.5, -1.0)),
            ('cubic2', _make_profile_cubic(n, 2.5, -1.5)),
            ('cubic3', _make_profile_cubic(n, 3.0, -2.0)),
            ('exp0.5', _make_profile_exponential(n, 0.5)),
            ('exp1.0', _make_profile_exponential(n, 1.0)),
            ('exp2.0', _make_profile_exponential(n, 2.0)),
            ('exp3.0', _make_profile_exponential(n, 3.0)),
            ('twobump1', _make_profile_two_bump(n, 0.3, 0.7, 0.15)),
            ('twobump2', _make_profile_two_bump(n, 0.25, 0.75, 0.2)),
            ('twobump3', _make_profile_two_bump(n, 0.2, 0.8, 0.25)),

        ]
        # Random Fourier and block initializations
        for i in range(3):

        for label, p in profs:

            candidates.append((sc, p, n, label))

    # Parent-based candidates
    if parent is not None:

    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f} n={candidates[0][2]}")
