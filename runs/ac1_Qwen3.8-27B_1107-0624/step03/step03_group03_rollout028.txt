def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c

def _score_norm(a):
    """Score for normalized (sum=1) sequence: 2n * max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix A where (A*g)_k = (a*g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a[j]
    return A

def _lp_spread_full(a, M):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = _build_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

    """Golden section search for best blend (1-t)*a + t*g."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    fd = _score_norm(np.maximum(a + d * (g - a), 0))

def _lse_step(a, z, state, T, lr):

    c = _fft_convolve(a)
    m_val = float(c.max())

    ex = np.exp(x)

    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n-1:2*n-1]
    grad = 2.0 * h
    
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    
    t = state['t'] + 1

    state['m'] = m

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LSE gradient for initialization, full-constraint LP spreading with adaptive
inflate and optimal line search for refinement, and stochastic coordinate
descent for final polish. Key improvement: exact-constraint LP with many
iterations and careful inflate management.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a = a / s
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[j]
    return A


def _lp_spread(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _line_search(a, g, n_evals=22):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c * (g - a), 0))
    fd = _score_norm(np.maximum(a + d * (g - a), 0))
    for _ in range(n_evals):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c * (g - a), 0))
        else:
            lo, c, fc = c, d, fc
            d = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d * (g - a), 0))
    return (lo + hi) / 2.0


def _lp_phase(a, deadline, max_iter=100000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    inflate = 1.0
    no_imp = 0
    it = 0

    while it < max_iter and time.monotonic() < deadline:
        c = _fft_convolve(a)
        M = float(c.max())

        g = _lp_spread(a, M * inflate)
        if g is None:
            inflate = min(1.05, inflate * 1.1 + 1e-6)
            no_imp += 20
            if no_imp > 1000:
                break
            it += 1
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-9:
            if inflate < 1.005:
                inflate = min(1.005, inflate * 1.005 + 1e-5)
            else:
                no_imp += 1
                if no_imp > 500:
                    # Perturb and restart
                    rng_r = np.random.default_rng(42 + it)
                    a = best_a * (1.0 + 0.03 * rng_r.normal(size=n))
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    inflate = 1.0
                    no_imp = 0
            it += 1
            continue

        g = g / sum_g

        t = _line_search(a, g, n_evals=20)
        if t < 0.003:
            t = 0.01

        a_new = np.maximum(a + t * (g - a), 0.0)

            break
        a_new = a_new / s_new

        it += 1

def _lse_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = np.clip((c - m_val) / T, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a, deadline, lr=0.1, T_start=0.05, T_end=1e-6):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        a, z, state, _ = _lse_step(a, z, state, T, lr)
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-15))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue

        sc = _score_norm(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 8 * n):
                rng_r = np.random.default_rng(7 + it)
                a = best_a * (1.0 + 0.04 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                no_imp = 0

        T = max(T_end, T * 0.99995)
        it += 1

    return best_a


def _stochastic_polish(a, deadline, n_trials=200000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(999)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        n_pert = int(rng.integers(1, max(2, n // 32)))
        idx = rng.integers(0, n, size=n_pert)
        a_try = a.copy()
        for i in idx:
            factor = 1.0 + float(rng.uniform(-0.5, 0.5))
            a_try[i] = max(0.0, a_try[i] * factor)

        s = float(a_try.sum())
        if s < 1e-15:

        a_try = a_try / s
        sc = _score_norm(a_try)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_try.copy()
            a = a_try

    return best_a


def _subgradient_polish(a, deadline, lr=0.01):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    momentum = 0.7
    no_imp = 0
    base_lr = lr

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-13)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.001, lr * 0.95)
            if no_imp > max(5000, 6 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, sigma=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
