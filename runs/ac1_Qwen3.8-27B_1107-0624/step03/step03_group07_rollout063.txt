"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses logit-space Adam, L-BFGS-B, top-k exact gradients, and multi-scale search.
"""
import numpy as np
import time
import scipy.optimize

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.fft.irfft(np.fft.rfft(a, 2*n) ** 2, 2*n)[:2*n-1]
    return float(2.0 * n * c.max() / (s * s))

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def compute_h(w, a):
    """h[i] = sum_{m} a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def top_k_weights(c, k=5):
    """Uniform weights on top-k positions."""
    idx = np.argsort(c)[-k:]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    return float(c.max()), w

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

def logit_adam(a0, deadline, rng, lr=0.04, beta_factor=0.08,
               beta_decay=0.93, beta_min=1e-10, restart_period=2000,
               use_topk=False, topk=5):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size

def lbfgs_optimize(a0, deadline, rng, beta=0.01, max_iter=5000):
    """Use L-BFGS-B to optimize the soft-max objective in logit space."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    c = full_convolve_fft(a0)
    max_c = float(c.max())
    best_score = float(2.0 * n * max_c)
    best_z0 = np.log(a0) - np.log(a0).mean()
    best_a0 = a0.copy()
    
    def objective_and_grad(z):
        z = z - z.mean()
        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-15:
            return 1e10, np.zeros(n)
        a = a / s
        c = full_convolve_fft(a)
        _, w = soft_max_weights(c, beta)
        soft_val = float(c.max()) + beta * float(np.log(float(np.sum(np.exp((c - c.max())/beta)))))
        h = compute_h(w, a)
        grad_a = 2.0 * h
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        return soft_val, grad_z
    
    # Run L-BFGS-B
    z = best_z0.copy()
    
    # We need to run this with a time limit
    # L-BFGS-B doesn't have a built-in time limit, so we'll use a wrapper
    iterations = [0]
    
    def callback(z_ind):
        iterations[0] += 1
        z_local = z_ind - z_ind.mean()
        a_local = np.exp(z_local - z_local.max())
        a_local = a_local / a_local.sum()
        c_local = full_convolve_fft(a_local)
        score = float(2.0 * n * c_local.max())
        if score < best_score:
            best_score = score
            best_a0 = a_local.copy()
            best_z0 = z_ind - z_ind.mean()
        if time.monotonic() > deadline:
            raise TimeExpired()
        if iterations[0] % 100 == 0:
            print(f"    L-BFGS-B iter {iterations[0]}: score={score:.8f}")
        return None
    
    class TimeExpired(Exception):
        pass
    
    try:
        result = scipy.optimize.minimize(
            lambda z: objective_and_grad(z)[0],
            z,
            jac=lambda z: objective_and_grad(z)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': max_iter, 'gtol': 1e-8},
            callback=callback
        )
    except TimeExpired:
        pass
    except Exception as e:
        print(f"    L-BFGS-B error: {e}")
    
    # Evaluate best found
    final_score = compute_score(best_a0)
    return best_a0, final_score

def make_diverse_profiles(n, rng):
    profiles = []
    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Reverse ramps
    for eta in [1.0, 2.0, 3.0]:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Cosine
    for k in [1, 2, 3]:
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Power ramps
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        t = np.linspace(0.0, 1.0, n)
        a = (1.0 - t) ** p + 0.05
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Random smooth
    for _ in range(8):
        num_harmonics = rng.integers(2, 6)
        coeffs = rng.uniform(-0.5, 0.5, num_harmonics)
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for i, c_val in enumerate(coeffs, 1):
            a = a + c_val * np.cos(2 * np.pi * i * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Triangular
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - 2.0 * np.abs(t - 0.5)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    # Inverted triangular
    a2 = 2.0 * np.abs(t - 0.5) + 0.1
    a2 = np.maximum(a2, 1e-12)
    profiles.append(a2 / a2.sum())
    
    return profiles

        contributions = a0[j_range] * a0[partners]
        
        # Find the pair with the largest contribution
        # We want to reduce this by moving mass from one to the other
        # or from one to some other position

        max_idx = int(np.argmax(contributions))
        j = j_range[max_idx]

        if a0[j] > a0[partner]:

        else:

        eps = 0.001 * a0[src]
        a0[src] -= eps

        targets = rng.choice(n, size=min(5, n), replace=False)
        for t in targets:

        a0 = np.maximum(a0, 0)

"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines logit-space Adam, SLSQP for small n, L-BFGS-B polishing,
top-k exact gradients, and multi-scale search across diverse n values.

import numpy as np
import time
import scipy.optimize
import functools


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[m+i] for i=0,...,n-1 via FFT (cross-correlation)."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def soft_max_weights(c, beta):
    """Compute soft-max value and softmax weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def top_k_weights(c, k=5):
    """Uniform weights on top-k positions."""
    k = min(k, len(c))
    idx = np.argsort(c)[-k:]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    return float(c.max()), w


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def logit_adam(a0, deadline, rng, lr=0.04, beta_factor=0.08,
               beta_decay=0.93, beta_min=1e-10, restart_period=2000,
               use_topk=False, topk=5):
    """Logit-space Adam optimization with soft-max or top-k gradient."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()
    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr

    while time.monotonic() < deadline:
        if use_topk:
            _, w = top_k_weights(c, topk)
        else:
            _, w = soft_max_weights(c, beta)

        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 400:
            current_lr = max(0.003, current_lr * 0.97)

        if no_improve > restart_period:
            z = best_z + 0.12 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.08 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


# State class for L-BFGS-B optimization
class LbfgsState:
    def __init__(self, n, beta, deadline):
        self.n = n
        self.beta = beta
        self.deadline = deadline
        self.best_a = None
        self.best_score = float('inf')
        self.iter_count = 0
        self.expired = False


def lbfgs_obj_func(z, state):
    """Objective for L-BFGS-B: soft-max of convolution in logit space."""
    state.iter_count += 1
    if state.expired or time.monotonic() > state.deadline:
        state.expired = True
        return 1e10
    z_c = z - z.mean()
    a = np.exp(z_c - z_c.max())
    s = a.sum()
    if s < 1e-15:
        return 1e10
    a = a / s
    c = full_convolve_fft(a)
    mval = c.max()
    x = (c - mval) / state.beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    ss = float(np.sum(ex))
    if ss < 1e-300:
        val = mval
    else:
        val = mval + state.beta * np.log(ss)
    score = float(2.0 * state.n * c.max())
    if score < state.best_score:
        state.best_score = score
        state.best_a = a.copy()
    return val


def lbfgs_grad_func(z, state):
    """Gradient for L-BFGS-B."""
    n = state.n
    z_c = z - z.mean()
    a = np.exp(z_c - z_c.max())
    s = a.sum()
    if s < 1e-15:
        return np.zeros(n)
    a = a / s
    c = full_convolve_fft(a)
    _, w = soft_max_weights(c, state.beta)
    h = compute_h(w, a)
    grad_a = 2.0 * h
    dot = float(np.dot(a, grad_a))
    return a * (grad_a - dot)


def lbfgs_callback(z, state):
    """Callback to check time limit."""
    if time.monotonic() > state.deadline:
        state.expired = True
    return state.expired


def lbfgs_optimize(a0, deadline, rng, beta=0.005, max_iter=5000):
    """L-BFGS-B optimization in logit space."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    best_score = float(2.0 * n * max_c)
    best_a = a0.copy()

    state = LbfgsState(n, beta, deadline)
    state.best_a = a0.copy()
    state.best_score = best_score

    z0 = np.log(a0)
    z0 = z0 - z0.mean()

    try:
        result = scipy.optimize.minimize(
            functools.partial(lbfgs_obj_func, state=state),
            z0,
            jac=functools.partial(lbfgs_grad_func, state=state),
            method='L-BFGS-B',
            callback=functools.partial(lbfgs_callback, state=state),
            options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'gtol': 1e-6,
                     'ftol': 1e-10}
        )
    except Exception as e:
        pass

    if state.best_a is not None and state.best_score < best_score:
        best_a = state.best_a
        best_score = state.best_score

    final_score = compute_score(best_a)
    return best_a, final_score


def slsqp_optimize(a0, deadline):
    """SLSQP optimization for small n - directly minimizes max of self-convolution."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-10)
    a0 = a0 / a0.sum()

    c0 = full_convolve_fft(a0)
    max_c0 = float(c0.max())
    best_score = float(2.0 * n * max_c0)
    best_a = a0.copy()
    c_len = 2 * n - 1

    def slsqp_obj(x):
        return x[-1]

    def slsqp_obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def slsqp_cons(x):
        a = x[:n]
        t = x[-1]
        if time.monotonic() > deadline:
            return np.full(c_len + n, -1e6)
        c = full_convolve_fft(a)
        return np.concatenate([c - t, -a])

    def slsqp_cons_jac(x):
        a = x[:n]
        t = x[-1]
        jac = np.zeros((c_len + n, n + 1))
        for k in range(c_len):
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_min <= i_max:
                idx = k - np.arange(i_min, i_max + 1)
                jac[k, i_min:i_max + 1] = 2.0 * a[idx]
            jac[k, n] = -1.0
        for i in range(n):
            jac[c_len + i, i] = -1.0
        return jac

    def slsqp_eq(x):
        return float(np.sum(x[:n])) - 1.0

    def slsqp_eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    constraints = [
        {'type': 'ineq', 'fun': slsqp_cons, 'jac': slsqp_cons_jac},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac},
    ]

    x0 = np.concatenate([a0, [max_c0]])

    def slsqp_cb(x):
        a = x[:n]
        t = x[-1]
        if t < best_score / (2.0 * n):
            c = full_convolve_fft(a)
            actual_score = float(2.0 * n * c.max())
            if actual_score < best_score:
                best_score = actual_score
                best_a = a.copy()
        if time.monotonic() > deadline:
            return True
        return None

    try:
        result = scipy.optimize.minimize(
            slsqp_obj, x0, jac=slsqp_obj_grad,
            method='SLSQP', bounds=[(1e-10, None)] * n + [(1e-10, None)],
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-10},
            callback=slsqp_cb
        )
    except Exception:
        pass

    final_score = compute_score(best_a)
    return best_a, final_score


def make_diverse_profiles(n, rng):

    profiles = []
    t = np.linspace(0.0, 1.0, n)

        a = 1.0 + eta * (0.5 - t)

        profiles.append(a / a.sum())

    # Reverse ramps
    for eta in [1.0, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Power ramps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = (1.0 - t) ** p + 0.03
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cosine modulations
    for k in range(1, 5):
        for amp in [0.2, 0.4, 0.6]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Triangular
    a = 1.0 - 1.5 * np.abs(t - 0.5)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Inverted triangular
    a = 0.3 + 1.0 * np.abs(t - 0.5)
    profiles.append(a / a.sum())

    # Gaussian bumps
    for width in [0.2, 0.4, 0.6]:
        for offset in [0.2, 0.35, 0.5, 0.65, 0.8]:
            a = np.exp(-((t - offset) / width) ** 2) + 0.05
            profiles.append(a / a.sum())

    # Random smooth (low-frequency Fourier)
    for _ in range(10):
        num_harm = int(rng.integers(2, 7))
        coeffs = rng.uniform(-0.5, 0.5, num_harm)
        phases = rng.uniform(0, 2 * np.pi, num_harm)
        a = np.ones(n)
        for i in range(num_harm):
            a = a + coeffs[i] * np.cos(2 * np.pi * (i + 1) * t + phases[i])
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Edge-concentrated
    for w in [0.1, 0.2, 0.3]:
        a = np.exp(-((t - 0.0) / w) ** 2) + 0.05
        profiles.append(a / a.sum())
        a2 = np.exp(-((t - 1.0) / w) ** 2) + 0.05
        profiles.append(a2 / a2.sum())

    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search: Adam + SLSQP + L-BFGS-B across multiple n values.
    Uses diverse profiles, multi-scale interpolation, and iterative refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting optimization. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        try_update(init_seq, sc_init)

    # ===== Phase 1: SLSQP at small n (exact, fast) =====
    for n_small in [64, 128]:
        if time.monotonic() > deadline - 100:
            break
        print(f"\n--- SLSQP at n={n_small} ---")
        dl = min(deadline - 50, time.monotonic() + 60.0)
        # Start from interpolated initial or a good profile
        if init_seq is not None:
            p0 = interpolate_profile(init_seq, n_small)
        else:
            p0 = make_diverse_profiles(n_small, rng)[3]  # linear ramp
        r, sc = slsqp_optimize(p0, dl)
        sc_f = compute_score(r)
        try_update(r, sc_f)
        print(f"  SLSQP n={n_small}: {sc_f:.10f}")

    # ===== Phase 2: Adam exploration at n=256,512 =====
    for n_exp in [256, 512]:
        if time.monotonic() > deadline - 300:
            break
        phase_end = min(deadline - 200, time.monotonic() + 180.0)
        print(f"\n--- Phase 2: Adam exploration n={n_exp} ---")

        profiles = make_diverse_profiles(n_exp, rng)
        if init_seq is not None:
            profiles.insert(0, interpolate_profile(init_seq, n_exp))
        if best_seq is not None:
            profiles.insert(0, interpolate_profile(best_seq, n_exp))

        best_p = None
        best_s = np.inf

        for idx, p in enumerate(profiles):
            if time.monotonic() > phase_end - 5:
                break
            sc = compute_score(p)
            if sc >= best_s + 0.12:
                continue
            dl = min(phase_end, time.monotonic() + 12.0)
            result, sc_r = logit_adam(p, dl, rng, lr=0.05, beta_factor=0.1,
                                       beta_decay=0.90, restart_period=1000)
            sc_final = compute_score(result)
            if sc_final < best_s:
                best_s = sc_final
                best_p = result.copy()
            try_update(result, sc_final)
            if idx < 6 or sc_final < best_s + 0.03:
                print(f"  Profile {idx}: {sc_final:.8f}")

        if best_p is not None:
            print(f"  Phase 2 n={n_exp} best: {best_s:.10f}")
            try_update(best_p, best_s)

    # ===== Phase 3: Focused refinement at best small n =====
    if best_seq is not None:
        n_refine = best_seq.size
        if n_refine > 512:
            n_refine = 512
    else:
        n_refine = 512

    phase3_end = min(deadline, t_start + 600.0)
    print(f"\n--- Phase 3: Refinement at n={n_refine} ---")

    p3 = interpolate_profile(best_seq if best_seq is not None else make_diverse_profiles(n_refine, rng)[0], n_refine)

    # Long Adam run
    dl3a = min(phase3_end, time.monotonic() + 200.0)
    r3, sc3 = logit_adam(p3, dl3a, rng, lr=0.025, beta_factor=0.04,
                          beta_decay=0.94, restart_period=2000)
    sc3_f = compute_score(r3)
    try_update(r3, sc3_f)
    print(f"  After Adam: {sc3_f:.10f}")

    # L-BFGS-B polish
    if time.monotonic() < phase3_end - 40:
        dl3b = min(phase3_end, time.monotonic() + 120.0)
        beta_lb = 0.003 * (sc3_f / (2.0 * n_refine))
        r3b, sc3b = lbfgs_optimize(r3, dl3b, rng, beta=max(beta_lb, 0.001), max_iter=3000)
        sc3b_f = compute_score(r3b)
        try_update(r3b, sc3b_f)
        print(f"  After L-BFGS-B: {sc3b_f:.10f}")
        if sc3b_f < sc3_f:
            r3 = r3b

    # Top-k variant
    if time.monotonic() < phase3_end - 30:
        dl3c = min(phase3_end, time.monotonic() + 80.0)
        r3c, sc3c = logit_adam(r3, dl3c, rng, lr=0.015, use_topk=True, topk=3,
                                restart_period=3000)
        sc3c_f = compute_score(r3c)
        try_update(r3c, sc3c_f)
        print(f"  After top-k: {sc3c_f:.10f}")
        if sc3c_f < sc3_f:
            r3 = r3c

    print(f"  Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Scale to larger n =====
    large_n_list = [768, 1024, 1536, 2048]
    for n4 in large_n_list:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        phase4_end = min(deadline - 40, time.monotonic() + remaining * 0.35)
        print(f"\n--- Phase 4: n={n4} (remaining={remaining:.0f}s) ---")

        p4 = interpolate_profile(best_seq if best_seq is not None else r3, n4)
        sc4_init = compute_score(p4)
        try_update(p4, sc4_init)
        print(f"  Interpolated: {sc4_init:.10f}")

        # Adam refinement
        dl4a = min(phase4_end, time.monotonic() + 150.0)
        r4, sc4 = logit_adam(p4, dl4a, rng, lr=0.012, beta_factor=0.025,
                              beta_decay=0.95, restart_period=3000)
        sc4_f = compute_score(r4)
        try_update(r4, sc4_f)
        print(f"  After Adam: {sc4_f:.10f}")

        # L-BFGS-B
        if time.monotonic() < phase4_end - 20:
            dl4b = min(phase4_end, time.monotonic() + 80.0)
            r4b, sc4b = lbfgs_optimize(r4, dl4b, rng, beta=0.002, max_iter=2000)
            sc4b_f = compute_score(r4b)
            try_update(r4b, sc4b_f)
            print(f"  After L-BFGS-B: {sc4b_f:.10f}")
            if sc4b_f < sc4_f:
                r4 = r4b

        # Top-k
        if time.monotonic() < phase4_end - 15:
            dl4c = min(phase4_end, time.monotonic() + 40.0)
            r4c, sc4c = logit_adam(r4, dl4c, rng, lr=0.01, use_topk=True,
                                    topk=5, restart_period=5000)
            sc4c_f = compute_score(r4c)
            try_update(r4c, sc4c_f)
            print(f"  After top-k: {sc4c_f:.10f}")

    # ===== Phase 5: Final intensive polish =====
    phase5_end = deadline
    remaining = phase5_end - time.monotonic()
    print(f"\n--- Phase 5: Final polish (time={remaining:.0f}s) ---")

    if best_seq is not None and remaining > 20:
        n_polish = best_seq.size

        # Multiple polish rounds with decreasing LR
        polish_configs = [
            (0.008, 0.012, "adam", 60),
            (0.005, 0.008, "lbfgs", 50),
            (0.004, 0.006, "adam", 40),
            (0.003, 0.004, "lbfgs", 40),
            (0.002, 0.003, "adam_topk", 30),
        ]

        for lr_val, beta_f, method, alloc in polish_configs:
            if time.monotonic() > phase5_end - 15:
                break
            alloc = min(alloc, phase5_end - time.monotonic() - 10)
            if alloc < 10:
                break
            dl = time.monotonic() + alloc
            p0 = best_seq.copy()

            if method == "adam":
                r, sc_r = logit_adam(p0, dl, rng, lr=lr_val, beta_factor=beta_f,
                                      beta_decay=0.96, restart_period=5000)
            elif method == "lbfgs":
                r, sc_r = lbfgs_optimize(p0, dl, rng, beta=beta_f * 0.3, max_iter=2000)
            else:  # adam_topk
                r, sc_r = logit_adam(p0, dl, rng, lr=lr_val, use_topk=True,
                                      topk=3, restart_period=5000)
            sc_f = compute_score(r)
            try_update(r, sc_f)
            print(f"  Polish ({method}, lr={lr_val}): {sc_f:.10f}")

    # ===== Check alternative n values =====
    if best_seq is not None:
        for n_alt in [256, 384, 512, 768]:
            if time.monotonic() > deadline - 20:
                break
            if n_alt == best_seq.size:
                continue
            p_alt = interpolate_profile(best_seq, n_alt)
            sc_alt = compute_score(p_alt)
            try_update(p_alt, sc_alt)
            if sc_alt < best_score + 0.015:
                dl = min(deadline - 10, time.monotonic() + 15.0)
                r_alt, _ = logit_adam(p_alt, dl, rng, lr=0.02,
                                       beta_factor=0.03, beta_decay=0.93)
                sc_alt2 = compute_score(r_alt)
                try_update(r_alt, sc_alt2)
                print(f"  Alt n={n_alt}: {sc_alt2:.10f}")

    # ===== Final output =====
    print(f"\n  Best score found: {best_score:.10f}")

    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_diverse_profiles(256, rng)[0]

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_diverse_profiles(256, rng)[0]

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
