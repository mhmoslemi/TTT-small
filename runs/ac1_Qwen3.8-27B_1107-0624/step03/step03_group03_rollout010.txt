"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LSE gradient descent, LP refinement, spectral ICR, and subgradient methods
with adaptive hyperparameters and diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s


def _make_conv_matrix_reduced(a, M, top_k=80):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[row, lo:hi] = a[lo:hi]
    return A, b


def _lp_spread(a, M, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    A, b = _make_conv_matrix_reduced(a, M, top_k)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, max_iters=25):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c * (g - a), 0))
    fd = _score_norm(np.maximum(a + d * (g - a), 0))
    for _ in range(max_iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c * (g - a), 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d * (g - a), 0))
    return (lo + hi) / 2.0


def _lp_refine(a, deadline, max_iter=5000, inflate=1.0005, blend=0.2, top_k=80):
    a = _normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp, top_k)
        if g is None:
            inflate_cur = min(1.015, inflate_cur * 1.3)
            no_imp += 5
            if no_imp > 400:
                break
            continue
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.015, inflate_cur * 1.1)
            if no_imp > 500:
                break
            continue
        g = g / sum_g
        t_opt = _line_search(a, g)
        if t_opt < 0.01:
            t_opt = blend_cur * 0.5
        a_new = _normalize((1.0 - t_opt) * a + t_opt * g)
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.03, blend_cur * 0.85)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        a = a_new

    return best_a


def _lse_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a, deadline, T_start=0.05, T_end=1e-5, lr=0.1, max_iter=100000):
    a = _normalize(a.copy())

def _icr_optimize(a, deadline, alpha=0.5, max_iter=50000, beta=0.1):

        phase = np.angle(fa)

            fa_new = mag_new * np.exp(1j * phase)

            a_new = np.maximum(a_new, 0)

            if s > 1e-10:

def _subgradient_polish(a, deadline, max_iter=50000, lr=0.02, momentum=0.8):
    a = _normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    N = _next_pow2(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-12)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = _normalize(a_new)
        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.9)
            if no_imp > max(10000, 10 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _hybrid_optimize(a, deadline, n_phases=5):
    """Alternate between LSE, LP, and subgradient with adaptive parameters."""
    a = _normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)

            a_new = _lse_optimize(a, phase_end, T_start=T_start,

        else:

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    return _normalize(p)


def _make_gaussian(n, sigma=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma * 0.5)) ** 2)

def _make_raised_cosine(n):

def _make_bimodal(n, sep=0.3):
