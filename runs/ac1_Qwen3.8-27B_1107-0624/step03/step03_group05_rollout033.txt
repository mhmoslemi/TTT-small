def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _cross_correlate(w, f):
    """Compute (w * f)(i) = sum_k w_k * f_{k-i} for valid indices.
    This is the 'gradient direction' for the smooth max of the convolution."""
    n = len(f)
    if n <= 128:
        # Direct computation
        result = np.zeros(n)
        for i in range(n):
            # sum_k w_k * f_{k-i} where 0 <= k-i < n, 0 <= k < 2n-1
            lo = max(0, i)
            hi = min(2 * n - 1, i + n - 1)
            if lo <= hi:
                idx = np.arange(lo, hi + 1) - i
                result[i] = np.dot(w[lo:hi+1], f[idx])
        return result
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    ff_rev = np.fft.rfft(f[::-1], N)
    h_full = np.fft.irfft(fw * ff_rev, N)

    # Let j = n-1-i, so i = n-1-j: h_full[k] = sum_j w[k] * f[j] (over valid j)
    # Actually let me think more carefully.

    # ff_rev = FFT(f[::-1], N) where f has length n, reversed and padded to N
    # h_full = IFFT(fw * ff_rev)
    # h_full[m] = sum_k w[k] * f_rev[(m-k) mod N] = sum_k w[k] * f[n-1-((m-k) mod N)]

    # So the result is h_full[n-1 : 2n-1]

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

    beta = max(1e-6, beta)

    no_imp = 0
    best_no_imp = 0

    eps = 1e-8

        # Subtract mean to stay on the "sum z = 0" constraint

def _lp_spread(a, M):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = res.x
            # Also get dual variables for guidance
            duals = res.ineqlin.marginals if hasattr(res, 'ineqlin') else None
            return g, duals
    except Exception:
        pass
    return None, None


def _lp_refine(a, deadline, max_iter=20000, inflate=1.001, blend=0.3):
    """Iterative LP refinement with adaptive inflation and blending."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    dual_weights = None

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g, duals = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.05, inflate_cur * 1.03)
            if no_imp > 500:
                break
            continue

        g = g / sum_g

        # Use dual weights if available for a more informed blend
        if duals is not None and np.all(np.isfinite(duals)) and np.any(duals > 0):
            # Weight the blend by dual importance
            dual_pos = np.maximum(duals, 0)
            dual_sum = dual_pos.sum()
            if dual_sum > 0:
                # Could use this to inform a targeted update
                pass

        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0003, inflate_cur * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.93)
                inflate_cur = min(1.02, inflate_cur * 0.98)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _dual_guided_mirror_descent(a, deadline, max_iter=30000, beta_init=0.02, lr_init=0.04):
    """Mirror descent guided by LP dual variables for constraint emphasis."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = beta_init * float(c.max())
    beta = max(1e-6, beta)
    lr = lr_init
    no_imp = 0

    b1, b2 = 0.85, 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    # Get initial dual guidance from LP
    dual_guide = None
    M_init = float(c.max())
    _, duals_init = _lp_spread(a, M_init * 1.001)
    if duals_init is not None and np.any(duals_init > 0):
        dual_guide = np.maximum(duals_init, 0)
        dual_guide /= dual_guide.sum()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w_sm = _softmax_weights(c, beta)

        # Combine softmax weights with dual guidance
        if dual_guide is not None:
            w = 0.5 * w_sm + 0.5 * dual_guide
            w = w / w.sum()
        else:
            w = w_sm

        grad_conv = _cross_correlate(w, a)
        grad_z = 2.0 * a * grad_conv
        grad_z -= float(np.mean(grad_z))
        grad_z += 1e-8 * a

        if not np.all(np.isfinite(grad_z)):
            z = np.log(best_a) - float(np.mean(np.log(best_a)))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            no_imp += 100
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        s = e.sum()
        if not np.isfinite(s) or s < 1e-30:
            z = np.log(best_a) - float(np.mean(np.log(best_a)))
            a = np.exp(z) / np.exp(z).sum()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            no_imp += 50
            continue
        a = e / s

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.96)
            if no_imp % 1000 == 0:
                # Refresh dual guidance
                M_cur = float(_fft_convolve(a).max())
                _, duals_new = _lp_spread(a, M_cur * 1.0005)
                if duals_new is not None and np.any(duals_new > 0):
                    dual_guide = np.maximum(duals_new, 0)
                    dual_guide /= dual_guide.sum()

            if no_imp > max(2000, 4 * n):
                z = np.log(best_a) - float(np.mean(np.log(best_a)))
                rng = np.random.default_rng(it * 31 + n)
                z += 0.08 * rng.normal(size=n)
                z = np.clip(z, -60.0, 60.0)
                z -= float(np.mean(z))
                a = np.exp(z) / np.exp(z).sum()
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr = lr_init
                beta = beta_init * float(_fft_convolve(a).max())
                no_imp = 0

        if it % 300 == 0 and it > 0:
            beta = max(1e-9, beta * 0.9)

    return best_a


def _peak_targeting(a, deadline, max_iter=10000):
    """Directly target the top peaks of the convolution."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)

    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        # Find top peaks
        sorted_idx = np.argsort(c)[::-1]
        top_k = sorted_idx[:min(5, len(c))]

        # For each top peak, find the positions contributing most
        for k_star in top_k[:3]:
            # Contribution from position i to c[k_star] is a_i * a_{k_star - i}
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            contrib = a[lo:hi+1] * a[k_star - hi:k_star - lo + 1]
            # Position with max contribution
            i_max = lo + int(np.argmax(contrib))

            # Try reducing a[i_max] and distributing to random positions
            for _ in range(5):
                delta = 0.05 * a[i_max]
                if delta <= 0:
                    continue
                j = rng.integers(0, n)
                if j == i_max:
                    continue
                a_try = a.copy()
                a_try[i_max] -= delta
                a_try[j] += delta
                if a_try[i_max] < 0:
                    continue
                a_try = a_try / a_try.sum()
                sc = _score_normalized(a_try)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                    no_imp = 0
                    break

        else:
            no_imp += 1

        if no_imp > 2000:
            break

    return best_a


def _multi_scale_search(a_parent, deadline, lengths):
    """Try optimization at multiple scales, upsampling from smaller to larger."""
    best_a = None
    best_score = np.inf

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        if best_a is not None and best_a.size != n:
            a_init = _interpolate_profile(best_a, n)
        elif a_parent is not None and a_parent.size != n:
            a_init = _interpolate_profile(a_parent, n)
        else:
            a_init = _u_profile(n, 2.0)

        # Quick mirror descent
        rem = deadline - time.monotonic() - 30
        if rem < 20:
            break
        budget = min(rem * 0.4, 120.0)
        a_md = _mirror_descent(a_init, time.monotonic() + budget,
                               max_iter=20000, beta_init=0.03, lr_init=0.04)
        sc = _score(a_md)
        print(f"    MD at n={n}: {sc:.10f}")

        if sc < best_score:
            best_score = sc
            best_a = a_md.copy()

        # Quick LP
        rem = deadline - time.monotonic() - 30
        if rem < 15:
            break
        budget = min(rem * 0.3, 60.0)
        a_lp = _lp_refine(best_a if best_a.size == n else a_md,
                          time.monotonic() + budget,
                          max_iter=5000, inflate=1.001, blend=0.25)
        sc = _score(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()

    return best_a, best_score


def _perturbation_search(a, deadline, n_pert=10):
    """Try random perturbations of the current best and re-optimize."""
    best_a = a.copy()
    best_score = _score(a)
    n = len(a)
    rng = np.random.default_rng(12345)

    for i in range(n_pert):
        if time.monotonic() > deadline - 30:
            break

        # Different perturbation strategies
        strategy = i % 4
        if strategy == 0:
            # Gaussian noise
            a_p = a * (1.0 + 0.1 * rng.normal(size=n))
        elif strategy == 1:
            # Block swap
            bs = max(2, n // 16)
            p = a.copy()
            i1 = rng.integers(0, n - bs)
            i2 = rng.integers(0, n - bs)
            if abs(i1 - i2) < bs:
                continue
            p[i1:i1+bs], p[i2:i2+bs] = p[i2:i2+bs].copy(), p[i1:i1+bs].copy()
            a_p = p
        elif strategy == 2:
            # Selective scaling: scale up the bottom half, scale down the top half
            a_sorted_idx = np.argsort(a)
            half = n // 2
            a_p = a.copy()
            a_p[a_sorted_idx[:half]] *= 1.1
            a_p[a_sorted_idx[-half:]] *= 0.9
        else:
            # Mirror with noise
            a_p = a[::-1].copy() * (1.0 + 0.05 * rng.normal(size=n))

        a_p = np.maximum(a_p, 0.0)
        s = a_p.sum()
        if s < 1e-10:
            continue
        a_p = a_p / s

        sc_p = _score(a_p)
        if sc_p < best_score:
            best_score = sc_p
            best_a = a_p.copy()

        # Quick re-optimization
        rem = deadline - time.monotonic() - 20
        if rem < 10:
            break
        budget = min(rem * 0.3, 45.0)
        a_opt = _mirror_descent(a_p, time.monotonic() + budget,
                                max_iter=5000, beta_init=0.04, lr_init=0.03)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"    Perturbation {i}: {sc_opt:.10f}")

    return best_a, best_score
