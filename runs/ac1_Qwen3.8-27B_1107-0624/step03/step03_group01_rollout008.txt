import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Assume sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

"""
Multi-phase convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, subgradient descent in softmax space
with adaptive active sets, momentum, and Barzilai-Borwein step sizes.
Exploits convexity of the problem for reliable convergence.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_subgradient(a, w):
    """Compute subgradient of max_k c_k w.r.t. a.
    a: sequence of length n (sum to 1)
    w: weights on convolution lags 0..2n-2
    Returns gradient vector of length n.
    grad[j] = 2 * sum_l a[l] * w[j+l] for j=0..n-1
    """
    n = len(a)
    N = _next_pow2(2 * n + len(w))
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:len(w)] = w
    cross = np.fft.ifft(np.fft.fft(a_pad) * np.conj(np.fft.fft(w_pad))).real
    grad = 2.0 * cross[:n]
    return grad


def _softmax_gradient(a, g):
    """Transform gradient w.r.t. a to gradient w.r.t. z (softmax params).
    grad_z[i] = a[i] * (g[i] - sum_j a[j]*g[j])
    """
    dot = float(np.dot(a, g))
    return a * (g - dot)


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    g = _compute_subgradient(a, w)
    grad_z = _softmax_gradient(a, g)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _run_lbfgs_continuation(a_init, deadline, max_wall_time=60.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # More aggressive continuation schedule
    beta_starts = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        level_start = time.monotonic()
        level_deadline = min(deadline, level_start + per_level * 2)

        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                if np.all(a_new > -1e-10) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

        if time.monotonic() > deadline:
            break

    return best_a, best_score


def _subgradient_phase(a_init, deadline, rng, top_k=16, momentum_decay=0.9):
    """Subgradient descent in softmax space with adaptive active set and BB step."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    current_score = best_score

    momentum = np.zeros(n)
    lr = 0.1
    lr_min = 1e-6
    lr_max = 1.0
    patience = 0
    max_patience = 200

    # BB step tracking
    prev_z = None
    prev_grad_z = None

    it = 0
    while time.monotonic() < deadline:
        it += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        current_score = 2.0 * n * cm

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive active set: top K lags with weights
        c_sorted_idx = np.argsort(c)[::-1]
        k_actual = min(top_k, len(c))
        top_indices = c_sorted_idx[:k_actual]
        top_values = c[top_indices]

        # Exponential weights based on proximity to max
        gaps = top_values - top_values[-1]
        w_local = np.exp(gaps / (0.1 * cm + 1e-10))
        w_local = w_local / w_local.sum()

        w = np.zeros(len(c))
        w[top_indices] = w_local

        # Compute subgradient
        g = _compute_subgradient(a, w)
        grad_z = _softmax_gradient(a, g)

        if not np.all(np.isfinite(grad_z)):
            z = np.zeros(n)
            continue

        # BB step size (if we have previous info)
        if prev_z is not None and prev_grad_z is not None:
            s_vec = z - prev_z
            y_vec = grad_z - prev_grad_z
            sy = float(np.dot(s_vec, y_vec))
            yy = float(np.dot(y_vec, y_vec))
            if yy > 1e-20 and sy > 1e-15:
                lr_bb = sy / yy
                lr = 0.5 * lr + 0.5 * np.clip(lr_bb, lr_min, lr_max)

        prev_z = z.copy()
        prev_grad_z = grad_z.copy()

        # Momentum update
        momentum = momentum_decay * momentum - lr * grad_z
        z_new = z + momentum

        # Clip for numerical stability
        z_new = np.clip(z_new, -50, 50)
        z_new = z_new - z_new.mean()

        # Evaluate
        a_new = _softmax_fn(z_new)
        score_new = _score_norm_fn(a_new)

        if np.isfinite(score_new) and score_new < current_score:
            z = z_new
        else:
            # Backtrack
            for backtrack in range(5):
                lr *= 0.5
                momentum = momentum_decay * momentum - lr * grad_z
                z_new = z + momentum
                z_new = np.clip(z_new, -50, 50)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                score_new = _score_norm_fn(a_new)
                if np.isfinite(score_new) and score_new < current_score:
                    z = z_new
                    break
            else:
                # No improvement, reset momentum
                momentum = np.zeros(n)
                lr = max(lr_min, lr * 0.5)

        if patience > max_patience:
            # Restart with perturbation
            noise_scale = 0.05 + 0.05 * (patience // max_patience)
            z = z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            momentum = np.zeros(n)
            lr = 0.1
            patience = 0
            max_patience = min(max_patience + 100, 1000)

        if it % 500 == 0:
            # Gradually increase top_k for more stability
            top_k = min(top_k + 1, 32)
            momentum_decay = min(momentum_decay + 0.001, 0.95)

    return best_a, best_score


def _adam_simplex_phase(a_init, deadline, rng, lr=0.05, top_k=8):
    """Adam-style optimizer in softmax space for the exact max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        score = 2.0 * n * cm

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Active set
        c_sorted_idx = np.argsort(c)[::-1]
        k_actual = min(top_k, len(c))
        top_indices = c_sorted_idx[:k_actual]
        top_values = c[top_indices]
        gaps = top_values - top_values[-1]
        temp = max(0.05 * cm, 1e-10)
        w_local = np.exp(gaps / temp)
        w_local = w_local / w_local.sum()
        w = np.zeros(len(c))
        w[top_indices] = w_local

        g = _compute_subgradient(a, w)
        grad_z = _softmax_gradient(a, g)

        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)
            continue

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if t % 1000 == 0:
            lr = max(0.001, lr * 0.97)
            top_k = min(top_k + 1, 32)

    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []
    # Uniform
    profiles.append(np.ones(n) / n)
    # Linear ramps
    for slope in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Quadratic
    for a1, a2 in [(1.0, 0.5), (2.0, -0.5), (0.5, 1.0), (1.5, 0.3)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Cubic / sinusoidal
    a = 1.0 + 0.5 * np.sin(np.pi * t)
    profiles.append(a / a.sum())
    a = 1.0 + 0.3 * np.cos(2 * np.pi * t)
    profiles.append(a / a.sum())
    # Tent (Fejer-like)
    a = np.maximum(1.0 - np.abs(u), 1e-12)
    profiles.append(a / a.sum())
    # Parabolic
    a = np.maximum(1.0 - 0.5 * u ** 2, 1e-12)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS smoothing -> subgradient refinement -> Adam polish.
    Exploits convexity of the problem for reliable progress.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:6.1f}s] NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_default = 2048
    if parent is not None:
        n_default = len(parent)
        print(f"Parent: n={n_default}, score={_score_fn(parent):.10f}")
        a0 = parent.copy()
        update_best(parent, _score_norm_fn(parent), "parent")
    else:
        a0 = _make_profiles(n_default)[5]  # tent profile
        print(f"No parent, using tent profile n={n_default}")

    # --- Phase 1: L-BFGS on parent (200s) ---
    p1_end = min(deadline, t_start + 200.0)
    if time.monotonic() < p1_end:
        p_best, sc_best = _run_lbfgs_continuation(a0, p1_end, max_wall_time=190.0)
        update_best(p_best, sc_best, "lbfgs_parent")

    # --- Phase 2: Subgradient refinement (500s) ---
    p2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < p2_end and best_seq is not None:
        p_best, sc_best = _subgradient_phase(best_seq, p2_end, rng, top_k=16, momentum_decay=0.85)
        update_best(p_best, sc_best, "subgrad")

    # --- Phase 3: Adam polish (300s) ---
    p3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < p3_end and best_seq is not None:
        p_best, sc_best = _adam_simplex_phase(best_seq, p3_end, rng, lr=0.03, top_k=8)
        update_best(p_best, sc_best, "adam")

    # --- Phase 4: Subgradient with different params (200s) ---
    p4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < p4_end and best_seq is not None:
        p_best, sc_best = _subgradient_phase(best_seq, p4_end, rng, top_k=32, momentum_decay=0.92)
        update_best(p_best, sc_best, "subgrad2")

    # --- Phase 5: Try different n values (200s) ---
    p5_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < p5_end and best_seq is not None:
        n_curr = len(best_seq)
        target_ns = []
        if n_curr < 4096:
            target_ns.append(4096)
        if n_curr > 1024:
            target_ns.append(1024)

        for tn in target_ns:
            if time.monotonic() > p5_end - 30:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{tn}")

            per_dl = min(p5_end - 10, time.monotonic() + 80.0)
            p_best, sc_best = _run_lbfgs_continuation(p_interp, per_dl, max_wall_time=75.0)
            update_best(p_best, sc_best, f"lbfgs_n{tn}")

            if time.monotonic() < p5_end - 20:
                per_dl2 = min(p5_end - 10, time.monotonic() + 50.0)
                p_best2, sc_best2 = _subgradient_phase(p_best, per_dl2, rng, top_k=16)
                update_best(p_best2, sc_best2, f"subgrad_n{tn}")

    # --- Phase 6: Multi-restart from best (200s) ---
    p6_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < p6_end and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 4
        per_restart = min(45.0, (p6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > p6_end - 5:
                break
            noise_scale = 0.03 + 0.03 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            per_dl = min(p6_end - 5, time.monotonic() + per_restart)
            p_best, sc_best = _subgradient_phase(p_start, per_dl, rng, top_k=12, momentum_decay=0.88)
            update_best(p_best, sc_best, f"restart{i}")

    # --- Phase 7: Final fine polish (remaining time) ---
    p7_end = deadline
    if time.monotonic() < p7_end and best_seq is not None:
        per_dl = min(p7_end - 5, time.monotonic() + 80.0)
        p_best, sc_best = _adam_simplex_phase(best_seq, per_dl, rng, lr=0.01, top_k=64)
        update_best(p_best, sc_best, "final_adam")

        if time.monotonic() < p7_end - 10:
            per_dl2 = min(p7_end - 5, time.monotonic() + 50.0)
            p_best2, sc_best2 = _subgradient_phase(best_seq, per_dl2, rng, top_k=64, momentum_decay=0.95)
            update_best(p_best2, sc_best2, "final_subgrad")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(256)[5]

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, time={time.monotonic()-t_start:.1f}s")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(256)[5]
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
