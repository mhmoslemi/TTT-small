def compute_score(a):

    n = a.size

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_h_exact(w, a):
    """h[i] = sum_m a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def compute_topk_gradient(c, a, k=None, tol=None):
    """
    Compute the exact subgradient of max(c) using top-k active set.
    Returns (grad_a, active_indices, max_val).
    """
    n = len(a)
    max_val = float(c.max())
    
    if tol is not None:
        threshold = max_val - tol
        active = np.where(c >= threshold)[0]
        k = len(active)
        if k == 0:
            active = np.array([np.argmax(c)])
            k = 1
    elif k is None:
        active = np.array([np.argmax(c)])
        k = 1
    else:
        active = np.argsort(c)[-k:]
    
    w = np.zeros_like(c)
    w[active] = 1.0 / k
    h = compute_h_exact(w, a)
    grad_a = 2.0 * h
    return grad_a, active, max_val


def softmax_normalize(z):
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return a / s

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

        z = z - z.mean() + step
        a0 = softmax_normalize(z)
    
    c = full_convolve_fft(a0)
    score = float(2.0 * n * c.max())
    return a0, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid optimization:
    Phase 1: Quick exploration at multiple n values
    Phase 2: Focused refinement with exact subgradient
    Phase 3: Multi-resolution polishing
    Phase 4: Final coordinate descent polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
    
    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc_init = compute_score(init_seq)
                update(init_seq, sc_init)
                print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
    except Exception:
        pass
    
    if init_seq is None:
        init_seq = np.ones(256) / 256
        update(init_seq, compute_score(init_seq))
    
    # ===== Phase 1: Multi-n exploration (120s) =====
    phase1_deadline = min(deadline, t_start + 120.0)
    print(f"\n=== Phase 1: Multi-n exploration ===")
    
    n_values = [128, 256, 512, 1024]
    if init_seq is not None:
        n_values.insert(0, init_seq.size)
        # Deduplicate
        n_values = list(dict.fromkeys(n_values))
    
    phase1_best = None
    phase1_best_score = np.inf
    
    for n_try in n_values:
        if time.monotonic() > phase1_deadline - 5:
            break
        
        # Get starting point for this n
        if init_seq is not None:
            p0 = interpolate_profile(init_seq, n_try)
        else:
            p0 = np.ones(n_try) / n_try
        
        sc0 = compute_score(p0)
        update(p0, sc0)
        
        # Quick optimization
        dl = min(phase1_deadline - 3, time.monotonic() + 25.0)
        result, sc_r = hybrid_optimize(p0, dl, rng, mode="auto", 
                                        lr_init=0.04, topk_init=3)
        sc_final = compute_score(result)
        update(result, sc_final)
        print(f"  n={n_try}: score={sc_final:.10f}")
        
        if sc_final < phase1_best_score:
            phase1_best_score = sc_final
            phase1_best = result.copy()
    
    print(f"  Phase 1 best: n={phase1_best.size if phase1_best is not None else '?'}, "
          f"score={best_score:.10f}")
    
    # ===== Phase 2: Focused refinement with exact subgradient (400s) =====
    phase2_deadline = min(deadline, t_start + 550.0)
    print(f"\n=== Phase 2: Focused refinement (exact subgradient) ===")
    
    n_refine = best_seq.size if best_seq is not None else 512
    p0 = best_seq.copy() if best_seq is not None else np.ones(n_refine) / n_refine
    
    # Run exact subgradient optimization
    dl2 = min(phase2_deadline - 10, time.monotonic() + 300.0)
    result2, sc2 = hybrid_optimize(p0, dl2, rng, mode="exact",
                                    lr_init=0.02, topk_init=5, topk_max=500,
                                    patience=300)
    sc2_final = compute_score(result2)
    update(result2, sc2_final)
    print(f"  After exact optimization: n={n_refine}, score={sc2_final:.10f}")
    
    # Try with different topk settings
    if time.monotonic() < phase2_deadline - 30:
        for topk_init in [1, 2, 10, 50]:
            if time.monotonic() > phase2_deadline - 15:
                break
            p0_mod = best_seq + 0.001 * rng.normal(size=best_seq.size)
            p0_mod = np.maximum(p0_mod, 1e-10)
            p0_mod = p0_mod / p0_mod.sum()
            dl = min(phase2_deadline - 10, time.monotonic() + 60.0)
            r, sc_r = hybrid_optimize(p0_mod, dl, rng, mode="exact",
                                       lr_init=0.015, topk_init=topk_init, topk_max=300)
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  topk_init={topk_init}: score={sc_f:.10f}")
    
    # ===== Phase 3: Multi-resolution (400s) =====
    phase3_deadline = min(deadline, t_start + 1000.0)
    print(f"\n=== Phase 3: Multi-resolution ===")
    
    # Try larger n
    if best_seq is not None:
        for n_up in [2 * best_seq.size, 4 * best_seq.size]:
            if n_up > 4096 or time.monotonic() > phase3_deadline - 50:
                break
            p_up = interpolate_profile(best_seq, n_up)
            sc_up = compute_score(p_up)
            update(p_up, sc_up)
            print(f"  Interpolated to n={n_up}: score={sc_up:.10f}")
            
            if sc_up < best_score + 0.01:
                dl_up = min(phase3_deadline - 20, time.monotonic() + 120.0)
                r_up, sc_up_r = hybrid_optimize(p_up, dl_up, rng, mode="exact",
                                                 lr_init=0.01, topk_init=5, topk_max=200)
                sc_up_f = compute_score(r_up)
                update(r_up, sc_up_f)
                print(f"  After optimization at n={n_up}: score={sc_up_f:.10f}")
    
    # Try smaller n (sometimes smaller is better)
    if best_seq is not None and best_seq.size > 256:
        for n_down in [best_seq.size // 2, best_seq.size // 4, 256, 128]:
            if n_down < 64 or time.monotonic() > phase3_deadline - 20:
                break
            p_down = interpolate_profile(best_seq, n_down)
            sc_down = compute_score(p_down)
            update(p_down, sc_down)
            
            if sc_down < best_score + 0.05:
                dl_down = min(phase3_deadline - 10, time.monotonic() + 30.0)
                r_down, sc_down_r = hybrid_optimize(p_down, dl_down, rng, mode="auto",
                                                     lr_init=0.03, topk_init=3)
                sc_down_f = compute_score(r_down)
                update(r_down, sc_down_f)
                print(f"  n={n_down}: score={sc_down_f:.10f}")
    
    # ===== Phase 4: Final polish (remaining time) =====
    print(f"\n=== Phase 4: Final polish ===")
    
    if best_seq is not None:
        n_polish = best_seq.size
        
        # Long exact optimization
        dl_p1 = min(deadline - 30, time.monotonic() + 200.0)
        p0 = best_seq + 0.0005 * rng.normal(size=n_polish)
        p0 = np.maximum(p0, 1e-10)
        p0 = p0 / p0.sum()
        r, sc_r = hybrid_optimize(p0, dl_p1, rng, mode="exact",
                                   lr_init=0.008, topk_init=10, topk_max=1000,
                                   patience=1000)
        sc_f = compute_score(r)
        update(r, sc_f)
        print(f"  Polish 1: score={sc_f:.10f}")
        
        # Coordinate descent
        if time.monotonic() < deadline - 30:
            dl_cd = min(deadline - 20, time.monotonic() + 60.0)
            r_cd, sc_cd = coordinate_descent_polish(best_seq, dl_cd, rng, max_iters=20000)
            sc_cd_f = compute_score(r_cd)
            update(r_cd, sc_cd_f)
            print(f"  CD polish: score={sc_cd_f:.10f}")
        
        # Flat top equalization
        if time.monotonic() < deadline - 10:
            a_eq, sc_eq = flat_top_equalize(best_seq, n_active=5)
            sc_eq_f = compute_score(a_eq)
            update(a_eq, sc_eq_f)
            print(f"  Equalize: score={sc_eq_f:.10f}")
    
    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = np.ones(256) / 256
    
    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    
    final_score = compute_score(final_seq)
    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}, time={elapsed:.1f}s")
    
    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Evaluation check: {e}")
    
    return [float(x) for x in final_seq]
