def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _logit_space_opt(a0, deadline, max_iter=30000, lr=0.05, beta_init=None,

    if beta_final is None:
        beta_final = max(1e-10, 0.001 * beta_init)

    z = np.log(a)
    z -= float(np.mean(z))
    z_prev = z.copy()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    best_no_imp_lr = lr
    base_lr = lr

        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        # Convert to logit-space gradient

            continue

        z_new -= float(np.mean(z_new))
        z_prev = z
        z = z_new

        e = np.exp(z - float(np.max(z)))
        a_new = e / e.sum()
        a = a_new

        c_new = _fft_convolve(a)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:

        if verbose and it % 5000 == 0:

def _peak_chase(a0, deadline, max_iter=20000, top_k=3, verbose=False):
    """Directly minimize the top-k convolution peaks using subgradient."""
    a = np.asarray(a0, dtype=np.float64).copy()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.5 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        max_val = float(c_curr.max())

        # Get top-k peaks
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]

        # Weighted subgradient: weight peaks exponentially by their value
        peak_vals = c_curr[top_indices]
        peak_range = peak_vals[0] - peak_vals[-1] + 1e-15
        weights = np.exp(-2.0 * (peak_vals[-1] - peak_vals[::-1]) / peak_range)
        weights = weights / weights.sum()

            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))

        lr = base_lr / np.sqrt(it + 1)

        s_new = float(a_new.sum())
        if s_new < 1e-10:

        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                # Try a different top_k
                if top_k < 5:
                    # Restart with larger top_k handled externally
                    pass
                break

        a = a_new

        if verbose and it % 5000 == 0:
            print(f"      PeakChase it={it}: best={best_score:.10f}")

    return best_a, best_score


def _lp_spread(a, M):
    """Find g >= 0 maximizing sum(g) s.t. conv(a,g)[k] <= M for all k."""
    n = len(a)
    c_obj = -np.ones(n)
    # A has rows for each convolution constraint
    # conv(a, g)[k] = sum_i a[i]*g[k-i] <= M
    # Build sparse representation using Toeplitz-like structure
    A_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            row[k - i] += a[i]
        A_rows.append(row)
    A = np.array(A_rows)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'maxiter': 500})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iter=1000, inflate=1.002, blend=0.3,
               verbose=False):
    """Iterative LP-based spreading to reduce convolution peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.03)
                blend_cur = max(0.05, blend_cur * 0.9)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate_cur = min(1.005, inflate_cur * 0.95)
                blend_cur = max(0.05, blend_cur * 0.9)
            if no_imp > 80:
                break

        a = a_new

    return best_a, best_score


def _make_ramp_profile(n, p=1.0):
    """Power-law ramp: high on right, low on left."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** p
    return a / a.sum()


def _make_exp_ramp(n, alpha=2.0):
    """Exponential ramp."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * i / n)
    return a / a.sum()


def _make_sine_bump(n, p=1.0):
    """Sine bump: zero at edges, peak in middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** p
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_two_hump(n, center_frac=0.3):
    """Two humps: one large, one small, asymmetric."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-((t - 0.5) ** 2) / 0.02) + 0.3 * np.exp(-((t - center_frac) ** 2) / 0.01)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_asym_bell(n, skew=1.5):
    """Asymmetric bell: skewed to one side."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    # Skewed Gaussian
    sign = np.sign(t - 0.5)
    beta_val = skew * sign
    a = np.exp(-0.5 * ((t - 0.5) / 0.25) ** 2 * np.exp(beta_val * (t - 0.5)))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _optimize_sequence(a0, deadline, label=""):
    """Multi-phase optimization on a single sequence."""
    n = len(a0)
    best_a = a0.copy()
    c = _fft_convolve(best_a)
    best_score = float(2.0 * n * c.max())
    if np.isfinite(best_score):
        print(f"  {label} init: {best_score:.10f} (n={n})")

    # Phase 1: Coarse Nesterov with large beta
    remaining = deadline - time.monotonic()
    if remaining > 30:
        d1 = time.monotonic() + min(remaining * 0.4, 200.0)
        a1, s1 = _logit_space_opt(best_a, d1, max_iter=20000, lr=0.05,
                                  beta_init=0.2 * float(c.max()),
                                  beta_final=0.02 * float(c.max()))
        if s1 < best_score:
            best_a, best_score = a1, s1
        print(f"  {label} Phase1 (Nesterov coarse): {best_score:.10f}")

    # Phase 2: Finer Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 20:
        c2 = _fft_convolve(best_a)
        d2 = time.monotonic() + min(remaining * 0.3, 150.0)
        a2, s2 = _logit_space_opt(best_a, d2, max_iter=15000, lr=0.02,
                                  beta_init=0.05 * float(c2.max()),
                                  beta_final=0.005 * float(c2.max()))
        if s2 < best_score:
            best_a, best_score = a2, s2
        print(f"  {label} Phase2 (Nesterov fine): {best_score:.10f}")

    # Phase 3: Peak chasing
    remaining = deadline - time.monotonic()
    if remaining > 15:
        d3 = time.monotonic() + min(remaining * 0.25, 100.0)
        a3, s3 = _peak_chase(best_a, d3, max_iter=10000, top_k=3)
        if s3 < best_score:
            best_a, best_score = a3, s3
        print(f"  {label} Phase3 (PeakChase): {best_score:.10f}")

    # Phase 4: ULG refinement
    remaining = deadline - time.monotonic()
    if remaining > 10:
        d4 = time.monotonic() + min(remaining * 0.3, 80.0)
        a4, s4 = _lp_refine(best_a, d4, max_iter=500, inflate=1.001, blend=0.25)
        if s4 < best_score:
            best_a, best_score = a4, s4
        print(f"  {label} Phase4 (LP): {best_score:.10f}")

    # Phase 5: Final fine Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 5:
        c5 = _fft_convolve(best_a)
        d5 = time.monotonic() + min(remaining * 0.7, 60.0)
        a5, s5 = _logit_space_opt(best_a, d5, max_iter=10000, lr=0.005,
                                  beta_init=0.01 * float(c5.max()),
                                  beta_final=0.001 * float(c5.max()))
        if s5 < best_score:
            best_a, best_score = a5, s5
        print(f"  {label} Phase5 (Final Nesterov): {best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses Nesterov-accelerated gradient in logit-space with adaptive smooth-max,
    peak-chasing subgradient, and LP spreading. Explores n=1024 and n=2048
    with diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Prepare interpolated versions
    if parent is not None:
        p1024 = _interpolate_profile(parent, 1024)
        p2048 = _interpolate_profile(parent, 2048)
    else:
        p1024 = _make_ramp_profile(1024, 1.0)
        p2048 = _make_ramp_profile(2048, 1.0)

    # === Strategy 1: Optimize parent at n=1024 (highest priority) ===
    print("=== Strategy 1: Parent @ n=1024 ===")
    d1 = min(deadline - 600, t_start + 400)
    a_best, s_best = _optimize_sequence(p1024, d1, "S1")
    if s_best < best_score:
        best_score = s_best
        best_seq = a_best.copy()
    print(f"  S1 final: {best_score:.10f}")

    # === Strategy 2: Optimize parent at n=2048 ===
    if time.monotonic() < deadline - 400:
        print("=== Strategy 2: Parent @ n=2048 ===")
        d2 = min(deadline - 400, time.monotonic() + 350)
        a_best2, s_best2 = _optimize_sequence(p2048, d2, "S2")
        if s_best2 < best_score:
            best_score = s_best2
            best_seq = a_best2.copy()
        print(f"  S2 final: {best_score:.10f}")

    # === Strategy 3: Diverse initializations at n=1024 ===
    if time.monotonic() < deadline - 200:
        print("=== Strategy 3: Diverse inits @ n=1024 ===")
        n = 1024
        candidates = [
            _make_ramp_profile(n, 0.5),
            _make_ramp_profile(n, 2.0),
            _make_exp_ramp(n, 3.0),
            _make_exp_ramp(n, 5.0),
            _make_sine_bump(n, 2.0),
            _make_asym_bell(n, 2.0),
            _make_asym_bell(n, -2.0),
            _make_two_hump(n, 0.25),
        ]
        # Perturbed parent
        p_pert = p1024 * (1.0 + 0.2 * rng.normal(size=n))
        p_pert = np.maximum(p_pert, 0.0)
        candidates.append(p_pert / p_pert.sum())

        # Mirror of parent (reverse)
        candidates.append(p1024[::-1] / p1024[::-1].sum())

        for i, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 150:
                break
            remaining_for_cands = deadline - 150 - time.monotonic()
            per_budget = remaining_for_cands / max(1, len(candidates) - i)
            if per_budget < 8:
                break
            d3i = time.monotonic() + min(per_budget * 0.9, 45.0)
            # Quick optimization: just Nesterov + peak chase
            a_q, s_q = _logit_space_opt(a0, d3i, max_iter=8000, lr=0.04,
                                        beta_init=0.15 * float(_fft_convolve(a0).max()),
                                        beta_final=0.01 * float(_fft_convolve(a0).max()))
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
            print(f"  S3-{i}: {s_q:.10f} (best={best_score:.10f})")

    # === Strategy 4: Diverse initializations at n=2048 ===
    if time.monotonic() < deadline - 150:
        print("=== Strategy 4: Diverse inits @ n=2048 ===")
        n = 2048
        candidates = [
            _make_ramp_profile(n, 1.0),
            _make_exp_ramp(n, 4.0),
            _make_asym_bell(n, 3.0),
        ]
        # Interpolated best from S1/S2
        if best_seq is not None and best_seq.size != n:
            candidates.append(_interpolate_profile(best_seq, n))

        for i, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 120:
                break
            remaining_for_cands = deadline - 120 - time.monotonic()
            per_budget = remaining_for_cands / max(1, len(candidates) - i)
            if per_budget < 15:
                break
            d4i = time.monotonic() + min(per_budget * 0.85, 50.0)
            a_q, s_q = _logit_space_opt(a0, d4i, max_iter=10000, lr=0.03,
                                        beta_init=0.15 * float(_fft_convolve(a0).max()),
                                        beta_final=0.01 * float(_fft_convolve(a0).max()))
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
            print(f"  S4-{i}: {s_q:.10f} (best={best_score:.10f})")

    # === Strategy 5: Final focused optimization on the best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"=== Strategy 5: Final focus (best={best_score:.10f}, n={best_seq.size}) ===")
        best_n = best_seq.size

        # Fine Nesterov with very small beta
        c_best = _fft_convolve(best_seq)
        d5a = time.monotonic() + min(deadline - 15 - time.monotonic(), 80.0)
        if d5a > time.monotonic() + 10:
            a_f1, s_f1 = _logit_space_opt(best_seq, d5a, max_iter=15000, lr=0.003,
                                          beta_init=0.005 * float(c_best.max()),
                                          beta_final=0.0005 * float(c_best.max()))
            if s_f1 < best_score:
                best_score = s_f1
                best_seq = a_f1.copy()
            print(f"  Final Nesterov: {best_score:.10f}")

        # Peak chase
        if time.monotonic() < deadline - 10:
            d5b = time.monotonic() + min(deadline - 8 - time.monotonic(), 50.0)
            if d5b > time.monotonic() + 5:
                a_f2, s_f2 = _peak_chase(best_seq, d5b, max_iter=8000, top_k=5)
                if s_f2 < best_score:
                    best_score = s_f2
                    best_seq = a_f2.copy()
                print(f"  Final PeakChase: {best_score:.10f}")

        # LP polish
        if time.monotonic() < deadline - 5:
            d5c = time.monotonic() + min(deadline - 3 - time.monotonic(), 40.0)
            if d5c > time.monotonic() + 5:
                a_f3, s_f3 = _lp_refine(best_seq, d5c, max_iter=300, inflate=1.0005, blend=0.15)
                if s_f3 < best_score:
                    best_score = s_f3
                    best_seq = a_f3.copy()
                print(f"  Final LP: {best_score:.10f}")

        # One more tiny Nesterov
        if time.monotonic() < deadline - 3:
            c_fin = _fft_convolve(best_seq)
            d5d = deadline - 3
            a_f4, s_f4 = _logit_space_opt(best_seq, d5d, max_iter=5000, lr=0.001,
                                          beta_init=0.002 * float(c_fin.max()),
                                          beta_final=0.0002 * float(c_fin.max()))
            if s_f4 < best_score:
                best_score = s_f4
                best_seq = a_f4.copy()
            print(f"  Ultra-final: {best_score:.10f}")

        # Try growing the best to a larger size and refitting
        if time.monotonic() < deadline - 30 and best_seq.size <= 1024:
            print("  Trying to grow to larger n...")
            grown = _interpolate_profile(best_seq, best_seq.size * 2)
            d5e = deadline - 25
            a_grown, s_grown = _logit_space_opt(grown, d5e, max_iter=8000, lr=0.02,
                                                 beta_init=0.05 * float(_fft_convolve(grown).max()),
                                                 beta_final=0.005 * float(_fft_convolve(grown).max()))
            if s_grown < best_score:
                best_score = s_grown
                best_seq = a_grown.copy()
            print(f"  Grown: {best_score:.10f} (n={best_seq.size})")

    # Final validation
    if best_seq is None:
        best_seq = _make_ramp_profile(512, 1.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp_profile(512, 1.0)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
                else:
                    # Internal score might differ slightly; use the evaluator's
                    print(f"  Note: internal={_score(final_seq):.10f} vs eval={sc_eval:.10f}")
        except Exception as e:
            print(f"  Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
