def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_norm(a):
    """Score assuming sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()

    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

    """Build the convolution constraint matrix A where (A@x)_k = (a*x)_k."""
    n = len(a)

def _interpolate(a, n_new):
    """Interpolate sequence a to length n_new, normalized to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        return a / a.sum()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return a_new / s
