"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam gradient descent in logit space with smooth-max, LP-based direction
finding, block coordinate descent, and multi-scale escalation.
Target: beat 1.506320, aim for <= 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def np2(k):
    """Next power of 2 >= k."""
    r = 1
    while r < k:
        r <<= 1
    return max(2, r)


def full_conv(a):
    """Compute full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = np2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def smax_weights(c, beta):
    """Compute smooth max value and softmax weights for weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def grad_conv(w, p):
    """Compute gradient of weighted convolution: g[i] = sum_k w[k] * p[k-i]."""
    n = len(p)
    N = np2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_z(z, floor=1e-12):
    """Convert logit vector to probability vector."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def normalize_nn(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-30:
        return None
    return a / s


def score_of(p):
    """Compute 2n * max(conv(p,p)) assuming sum(p)=1."""
    n = len(p)
    c = full_conv(p)
    return float(2.0 * n * c.max())


def lp_find_direction(a, M, n_solves=1):
    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0. Returns b or None."""
    n = len(a)
    # Build constraint matrix: A[k, i] = a[k-i] if 0 <= k-i < n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def profile_linear(n, eta):
    """Linear ramp profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def profile_exp(n, lam):
    """Exponential decay profile."""
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def profile_cubic(n, a1, a3):
    """Cubic polynomial profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def profile_twobump(n, c1, c2, s):
    """Two Gaussian bumps."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def profile_cos(n, freq, amp):
    """Cosine-modulated profile."""
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interp_profile(p0, n_new):
    """Interpolate profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nn(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    return normalize_nn(p1)


def adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                  no_imp_limit=300, noise_scales=None, verbose=False):
    """
    Adam optimization in logit space with smooth-max.
    Returns (best_p, best_score).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n = p0.size
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_ad = 0

    p = p0.copy()
    c = full_conv(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    if noise_scales is None:
        noise_scales = [0.1, 0.2, 0.3, 0.5, 0.8]
    noise_idx = 0

    it = 0
    while time.monotonic() < deadline:
        _, w = smax_weights(c, beta)
        g = grad_conv(w, p)

        # Project gradient to simplex (remove mean component)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_ad = 0
            it += 1
            continue

        t_ad += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_ad)
        vh = v / (1.0 - b2 ** t_ad)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_ad = 0
            it += 1
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = softmax_z(z)
        c = full_conv(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-12:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
                if verbose and it % 100 == 0:
                    print(f"    Adam it={it}: score={score:.10f}")
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_ad = 0
            it += 1
            continue

        # Adaptive temperature
        if it % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.5)

        # Restart if stuck
        if no_imp > no_imp_limit:
            ns = noise_scales[noise_idx % len(noise_scales)]
            noise_idx += 1
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = softmax_z(z)
            c = full_conv(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_ad = 0
            lr = base_lr
            no_imp = 0
            if verbose:
                print(f"    Adam restart: noise={ns:.2f}")

        it += 1

    return best_p, best_score


def lp_refine(a, deadline, t=0.3, max_solves=15, verbose=False):
    """LP-based refinement: iteratively find direction and blend."""
    n = len(a)
    current_best = a.copy()
    current_score = score_of(a)
    c = full_conv(a)
    M = float(c.max())
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = lp_find_direction(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = score_of(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
            if verbose:
                print(f"    LP iter {n_solves}: score={sc_new:.10f}")
        a = a_new
        c = full_conv(a)
        M = float(c.max())
        t = max(0.05, t * 0.95)

    return current_best, current_score


def block_cd(p, deadline, rng, block_size=16, n_blocks=100, verbose=False):
    """Block coordinate descent: optimize blocks of consecutive elements."""
    n = len(p)
    current = p.copy()
    current_score = score_of(current)
    c = full_conv(current)
    M = float(c.max())
    n_iter = 0

    while time.monotonic() < deadline and n_iter < n_blocks:
        n_iter += 1
        # Pick a random block
        start = int(rng.integers(0, max(1, n - block_size)))
        end = start + block_size
        idx = np.arange(start, end)

        # Build sub-problem: fix all elements except block
        # c_k = sum_i a[i]a[k-i] = c_k_fixed + c_k_block
        # where c_k_block involves a[i] for i in [start, end)
        # c_k_block = 2 * sum_{i in block} a[i] * (sum_{j not in block} a[j] * delta[k-i-j])
        #            + sum_{i1, i2 in block} a[i1] * a[i2] * delta[k-i1-i2]
        # This is quadratic! Not an LP.
        # For now, just do a simple projected gradient step on the block.
        
        # Compute gradient restricted to block
        _, w = smax_weights(c, 0.01 * M)
        g_full = grad_conv(w, current)
        g_block = g_full[idx]
        
        # Try a few step sizes
        best_local = None
        best_local_score = current_score
        for step in [0.01, 0.005, 0.002, 0.001]:
            new_block = current[idx] - step * g_block
            new_block = np.maximum(new_block, 0.0)
            if new_block.sum() < 1e-15:
                continue
            candidate = current.copy()
            candidate[idx] = new_block
            candidate = candidate / candidate.sum()
            sc = score_of(candidate)
            if sc < best_local_score:
                best_local_score = sc
                best_local = candidate.copy()
        
        if best_local is not None:
            current = best_local
            current_score = best_local_score
            c = full_conv(current)
            M = float(c.max())
            if verbose and n_iter % 10 == 0:
                print(f"    BlockCD iter {n_iter}: score={current_score:.10f}")
        else:
            # Try increasing the block
            for step in [0.01, 0.005, 0.002]:
                new_block = current[idx] + step * g_block
                new_block = np.maximum(new_block, 0.0)
                candidate = current.copy()
                candidate[idx] = new_block
                candidate = candidate / candidate.sum()
                sc = score_of(candidate)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = candidate.copy()
            if best_local is not None:
                current = best_local
                current_score = best_local_score
                c = full_conv(current)
                M = float(c.max())

    return current, current_score
