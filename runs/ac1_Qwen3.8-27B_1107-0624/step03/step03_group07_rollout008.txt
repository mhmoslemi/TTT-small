def exact_subgradient_step(a, c, n, active_thresh):
    """Compute exact subgradient using active set."""
    max_c = c.max()
    active = c >= max_c - active_thresh
    if not active.any():
        active[int(np.argmax(c))] = True
    w = active / active.sum()
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
    g = np.zeros(n)
    for i in range(n):
        total = 0.0
        for k in np.where(active)[0]:
            j = k - i
            if 0 <= j < n:
                total += w[k] * a[j]
        g[i] = 2.0 * total
    return g

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_h(w, a):
    """h[i] = sum_{m=0}^{n-1} a[m]*w[i+m] for i=0,...,n-1 via FFT."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_bump(n, width=0.6, offset=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - offset) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_ramp_profile(n, p=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_random_smooth(n, rng, n_fourier=4):
    coeffs = rng.uniform(-0.3, 0.3, n_fourier)
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i, c_val in enumerate(coeffs, 1):
        a = a + c_val * np.cos(2 * np.pi * i * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def diverse_profiles(n, rng):
    profiles = []
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(make_linear_profile(n, eta))
    for k in [1, 2, 3]:
        profiles.append(make_cosine_profile(n, k, 0.4))
    for width in [0.3, 0.5, 0.7]:
        for offset in [0.2, 0.4, 0.6]:
            profiles.append(make_smooth_bump(n, width, offset))
    for p in [0.5, 1.0, 2.0, 3.0]:
        profiles.append(make_ramp_profile(n, p))
    for _ in range(8):
        profiles.append(make_random_smooth(n, rng, n_fourier=rng.integers(2, 8)))
    return profiles


def get_active_weights(c, top_k=None, relative_thresh=None):
    """Get weights on active peaks. Uses top-k or relative threshold."""
    c = np.asarray(c, dtype=np.float64)
    max_c = c.max()
    if max_c <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w

    if top_k is not None and top_k > 0:
        k = min(top_k, len(c))
        top_indices = np.argpartition(c, -k)[-k:]
        mask = np.zeros_like(c, dtype=bool)
        mask[top_indices] = True
        w_cand = c * mask
        s = w_cand.sum()
        if s > 1e-30:
            return w_cand / s
    elif relative_thresh is not None:
        threshold = max_c - relative_thresh * max_c
        mask = c >= threshold
        if mask.sum() == 0:
            mask[int(np.argmax(c))] = True
        w_cand = c[mask]
        s = w_cand.sum()
        if s > 1e-30:
            w = np.zeros_like(c)
            w[mask] = w_cand / s
            return w

    w = np.zeros_like(c)
    w[int(np.argmax(c))] = 1.0
    return w


def soft_max_weights(c, beta):
    """Compute soft-max weights for smooth gradient."""
    mval = float(c.max())
    if beta < 1e-14:
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = np.zeros_like(c)
            w[mask] = 1.0 / cnt
        return w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def optimize_logit_exact(a0, deadline, rng, lr=0.03, top_k_start=50,
                         top_k_end=3, adaptive=False, nesterov_momentum=0.9,
                         lr_decay=0.9995, patience=500):
    """
    Logit-space optimization with exact subgradient (top-K peaks),
    Adam + Nesterov acceleration, and adaptive learning rate.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    base_score = 2.0 * n * max_c

    z = np.log(a0)
    z = z - z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    vel = np.zeros(n)  # Nesterov velocity

    best_a = a0.copy()
    best_z = z.copy()
    best_score = base_score
    no_improve = 0
    current_lr = lr
    total_iters = 0

    # Adaptive top-k schedule
    iters_so_far = 0
    est_iters = 5000
    k_schedule = "adaptive" if adaptive else "linear"

    while time.monotonic() < deadline:
        iters_so_far += 1
        # Determine current top-k
        if k_schedule == "adaptive":
            frac = min(1.0, iters_so_far / est_iters)
            top_k = max(top_k_end, int(top_k_start * (1.0 - frac) + top_k_end * frac))
        else:
            frac = min(1.0, iters_so_far / est_iters)
            top_k = max(top_k_end, int(top_k_start * (1.0 - 0.95 * frac) + top_k_end * 0.95 * frac))

        # Get weights
        w = get_active_weights(c, top_k=top_k)

        # Gradient
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # Nesterov lookahead
        vel = nesterov_momentum * vel + dz
        dz_nest = vel  # Use momentum-smoothed gradient

        # Adam
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz_nest
        v = b2 * v + (1.0 - b2) * (dz_nest * dz_nest)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        step = current_lr * mh / (np.sqrt(vh) + eps)

        # Clip step for stability
        step_max = np.max(np.abs(step))
        if step_max > 1.0:
            step = step / step_max

        z = z - step
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        # New a
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            vel.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        # Evaluate
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        total_iters += 1

        # Adaptive learning rate
        if no_improve > 100:
            current_lr = max(0.002, current_lr * lr_decay)

        # Restart
        if no_improve > patience:
            noise_scale = 0.1 + 0.05 * (no_improve / patience)
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            m.fill(0.0)
            v.fill(0.0)
            vel.fill(0.0)
            t_adam = 0
            current_lr = max(0.005, lr * 0.8)
            no_improve = 0

    return best_a, best_score


def optimize_logit_hybrid(a0, deadline, rng, lr=0.03, beta_start=0.1,
                          beta_end=0.005, decay='exp', patience=400):
    """
    Hybrid: starts with soft-max (smooth) and transitions to exact subgradient (sharp).
    This gives a smooth gradient early on and precise gradient near convergence.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    base_score = 2.0 * n * max_c

    z = np.log(a0)
    z = z - z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = base_score
    no_improve = 0
    current_lr = lr
    iters = 0
    est_total = 8000

    while time.monotonic() < deadline:
        iters += 1
        # Adaptive beta: starts large (smooth), decreases to small (sharp)
        frac = min(1.0, iters / est_total)
        if decay == 'exp':
            beta = beta_start * (beta_end / beta_start) ** frac
        else:
            beta = beta_start + (beta_end - beta_start) * frac

        # Use soft-max weights
        w = soft_max_weights(c, beta)

        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # Adam
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        step = current_lr * mh / (np.sqrt(vh) + eps)

        step_max = np.max(np.abs(step))
        if step_max > 1.0:
            step = step / step_max

        z = z - step
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            current_lr = max(0.003, current_lr * 0.998)

        if no_improve > patience:
            noise = 0.15 * rng.normal(size=n)
            z = best_z + noise
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = max(0.008, lr * 0.7)
            no_improve = 0

    return best_a, best_score


def optimize_frank_wolfe(a0, deadline, rng, lr_init=0.5, patience=300):
    """
    Frank-Wolfe (conditional gradient) method for minimizing max-conv on simplex.
    At each step: compute gradient, move towards minimizer of linearization (a vertex).
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    best_score = 2.0 * n * max_c
    best_a = a0.copy()
    no_improve = 0
    iters = 0
    t_fw = 0

    while time.monotonic() < deadline:
        iters += 1
        t_fw += 1

        c = full_convolve_fft(a0)
        max_c = float(c.max())

        # Use top-10 peaks for gradient
        k = min(10, 2 * n - 1)
        top_idx = np.argpartition(c, -k)[-k:]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / len(top_idx)

        h = compute_h(w, a0)
        grad_a = 2.0 * h

        # Frank-Wolfe: direction is e_j - a where j = argmin grad
        j_min = int(np.argmin(grad_a))
        s = np.zeros(n)
        s[j_min] = 1.0
        direction = s - a0

        # Step size: gamma = 2/(t+2) for FW, or line search
        gamma = 2.0 / (t_fw + 2.0)

        # Try line search
        a_trial = a0 + gamma * direction
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial > 1e-15:
            a_trial = a_trial / s_trial
            c_trial = full_convolve_fft(a_trial)
            score_trial = 2.0 * n * c_trial.max()

            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()
                no_improve = 0
            else:
                no_improve += 1

            a0 = a_trial
        else:
            a0 = best_a.copy()
            no_improve += 1

        if no_improve > patience:
            break

    return best_a, best_score


def line_search_step(a0, direction, n, max_trials=10):
    """Backtracking line search on the simplex."""
    a0 = np.asarray(a0)
    c0 = full_convolve_fft(a0)
    f0 = 2.0 * n * c0.max()
    step = 1.0
    for _ in range(max_trials):
        a_try = a0 + step * direction
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            step *= 0.5
            continue
        a_try = a_try / s
        c_try = full_convolve_fft(a_try)
        f_try = 2.0 * n * c_try.max()
        if f_try <= f0 - 1e-12 or step < 1e-10:
            return a_try
        step *= 0.5
    return a0


def polish_exact(a0, deadline, rng, top_k=5):
    """
    Final polishing: very small steps with exact top-k subgradient.
    Uses a fixed small learning rate and no restarts.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    best_score = 2.0 * n * max_c
    best_a = a0.copy()
    best_z = np.log(a0)
    best_z = best_z - best_z.mean()

    z = best_z.copy()
    lr = 0.005
    b1, b2, eps = 0.95, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    while time.monotonic() < deadline:
        # Top-k weights
        k = min(top_k, 2 * n - 1)
        top_idx = np.argpartition(c, -k)[-k:]
        w = np.zeros_like(c)
        # Weight by proximity to max
        vals = c[top_idx]
        max_v = vals.max()
        min_v = vals.min()
        if max_v - min_v > 1e-12:
            weights = (vals - min_v) / (max_v - min_v) + 0.1
        else:
            weights = np.ones_like(vals)
        weights = weights / weights.sum()
        for idx, wt in zip(top_idx, weights):
            w[idx] = wt

        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        step = lr * mh / (np.sqrt(vh) + eps)
        step = np.clip(step, -0.05, 0.05)

        z = z - step
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining exact subgradient optimization, hybrid soft/sharp
    gradients, Frank-Wolfe, and final polishing. Explores multiple dimensions.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Try to load previous best
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
                sc_init = compute_score(init_seq)
                print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
                update(init_seq, sc_init)
    except Exception:
        pass

    print(f"Starting search. Budget={budget_s}s, deadline in {deadline - t_start:.1f}s")

    # ===== Phase 1: Explore at n=256 and n=512 =====
    phase1_deadline = min(deadline, t_start + 300.0)
    print(f"\n=== Phase 1: Exploration (0 to {phase1_deadline - t_start:.0f}s) ===")

    for n_try in [256, 512]:
        if time.monotonic() > phase1_deadline - 20:
            break
        print(f"\n  -- n={n_try} --")

        # Build starting profiles
        profiles = []
        if init_seq is not None:
            profiles.append(interpolate_profile(init_seq, n_try))
        profiles.extend(diverse_profiles(n_try, rng)[:8])

        # Sort by initial score
        profile_scores = []
        for p in profiles:
            sc = compute_score(p)
            profile_scores.append((sc, p))
        profile_scores.sort(key=lambda x: x[0])

        # Optimize top profiles
        max_profiles = 5
        for idx, (sc_p, p) in enumerate(profile_scores[:max_profiles]):
            if time.monotonic() > phase1_deadline - 10:
                break
            alloc = min(50.0, max(10.0, (phase1_deadline - time.monotonic()) / (max_profiles - idx)))
            dl = time.monotonic() + alloc

            # Method 1: Hybrid soft-to-sharp
            r1, sc1 = optimize_logit_hybrid(p, dl, rng, lr=0.04,
                                             beta_start=0.12, beta_end=0.005)
            sc1 = compute_score(r1)
            update(r1, sc1)

            if time.monotonic() < dl - 5 and sc1 > best_score - 0.005:
                # Method 2: Exact subgradient
                dl2 = min(deadline, time.monotonic() + 25.0)
                r2, sc2 = optimize_logit_exact(r1, dl2, rng, lr=0.02,
                                                top_k_start=20, top_k_end=3)
                sc2 = compute_score(r2)
                update(r2, sc2)

            if idx < 3:
                print(f"    Profile {idx} (init={sc_p:.6f}): best so far={best_score:.10f}")

    print(f"  Phase 1 best: {best_score:.10f}")

    # ===== Phase 2: Refinement at n=512 =====
    phase2_deadline = min(deadline, t_start + 700.0)
    print(f"\n=== Phase 2: Refinement n=512 ===")

    n2 = 512
    if best_seq is not None:
        p2 = interpolate_profile(best_seq, n2)
    else:
        p2 = make_linear_profile(n2, 1.5)

    sc2_init = compute_score(p2)
    update(p2, sc2_init)
    print(f"  Interpolated to n={n2}: {sc2_init:.10f}")

    # Long hybrid optimization
    dl2a = min(phase2_deadline, time.monotonic() + 250.0)
    r2, sc2 = optimize_logit_hybrid(p2, dl2a, rng, lr=0.025,
                                     beta_start=0.08, beta_end=0.003,
                                     decay='exp', patience=500)
    sc2 = compute_score(r2)
    update(r2, sc2)
    print(f"  After hybrid: {sc2:.10f}")

    # Exact subgradient refinement
    if time.monotonic() < phase2_deadline - 30:
        dl2b = min(phase2_deadline, time.monotonic() + 150.0)
        r2b, sc2b = optimize_logit_exact(r2, dl2b, rng, lr=0.015,
                                          top_k_start=30, top_k_end=3,
                                          adaptive=True, nesterov_momentum=0.92)
        sc2b = compute_score(r2b)
        update(r2b, sc2b)
        print(f"  After exact: {sc2b:.10f}")
        if sc2b < sc2:
            r2 = r2b

    # Try Frank-Wolfe as alternative
    if time.monotonic() < phase2_deadline - 20 and best_score > 1.504:
        dl_fw = min(phase2_deadline, time.monotonic() + 40.0)
        r_fw, sc_fw = optimize_frank_wolfe(r2, dl_fw, rng)
        sc_fw = compute_score(r_fw)
        update(r_fw, sc_fw)
        print(f"  After FW: {sc_fw:.10f}")

    # Final polish at n=512
    if time.monotonic() < phase2_deadline - 10:
        dl_pol = min(phase2_deadline, time.monotonic() + 50.0)
        r_pol, sc_pol = polish_exact(r2, dl_pol, rng, top_k=5)
        sc_pol = compute_score(r_pol)
        update(r_pol, sc_pol)
        print(f"  After polish: {sc_pol:.10f}")

    print(f"  Phase 2 best: {best_score:.10f}")

    # ===== Phase 3: Large scale n=1024 =====
    phase3_deadline = min(deadline, t_start + 1350.0)
    print(f"\n=== Phase 3: Large scale n=1024 ===")

    n3 = 1024
    if best_seq is not None:
        p3 = interpolate_profile(best_seq, n3)
    else:
        p3 = interpolate_profile(r2, n3)

    sc3_init = compute_score(p3)
    update(p3, sc3_init)
    print(f"  Interpolated to n={n3}: {sc3_init:.10f}")

    # Long optimization
    dl3a = min(phase3_deadline, time.monotonic() + 400.0)
    r3, sc3 = optimize_logit_hybrid(p3, dl3a, rng, lr=0.02,
                                     beta_start=0.06, beta_end=0.002,
                                     decay='exp', patience=600)
    sc3 = compute_score(r3)
    update(r3, sc3)
    print(f"  After hybrid: {sc3:.10f}")

    # Exact refinement
    if time.monotonic() < phase3_deadline - 50:
        dl3b = min(phase3_deadline, time.monotonic() + 200.0)
        r3b, sc3b = optimize_logit_exact(r3, dl3b, rng, lr=0.01,
                                          top_k_start=20, top_k_end=3,
                                          adaptive=True, nesterov_momentum=0.93)
        sc3b = compute_score(r3b)
        update(r3b, sc3b)
        print(f"  After exact: {sc3b:.10f}")
        if sc3b < sc3:
            r3 = r3b

    # Polish
    if time.monotonic() < phase3_deadline - 20:
        dl3c = min(phase3_deadline, time.monotonic() + 80.0)
        r3c, sc3c = polish_exact(r3, dl3c, rng, top_k=3)
        sc3c = compute_score(r3c)
        update(r3c, sc3c)
        print(f"  After polish: {sc3c:.10f}")

    print(f"  Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Try n=2048 if promising =====
    n4 = 2048
    if time.monotonic() < phase3_deadline - 60 and best_score > 1.503:
        print(f"\n=== Phase 4: n={n4} ===")
        p4 = interpolate_profile(best_seq, n4)
        sc4_init = compute_score(p4)
        update(p4, sc4_init)
        print(f"  Interpolated to n={n4}: {sc4_init:.10f}")

        dl4 = min(deadline, time.monotonic() + 150.0)
        r4, sc4 = optimize_logit_hybrid(p4, dl4, rng, lr=0.015,
                                         beta_start=0.05, beta_end=0.002,
                                         decay='exp', patience=400)
        sc4 = compute_score(r4)
        update(r4, sc4)
        print(f"  After optimization: {sc4:.10f}")

    # ===== Phase 5: Final intensive polish =====
    print(f"\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        n_pol = best_seq.size
        remaining = deadline - time.monotonic()
        print(f"  n={n_pol}, remaining time={remaining:.1f}s")

        # Multiple polishing rounds with decreasing top_k
        for round_idx, (lr_val, top_k) in enumerate([
            (0.008, 10),
            (0.005, 5),
            (0.003, 3),
            (0.002, 2),
        ]):
            if time.monotonic() > deadline - 5:
                break
            alloc = min(remaining / (4 - round_idx), 120.0)
            if alloc < 5:
                break
            dl_pol = time.monotonic() + alloc
            p0 = best_seq + 0.0002 * rng.normal(size=n_pol)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()

            r_pol, sc_pol = polish_exact(p0, dl_pol, rng, top_k=top_k)
            sc_pol = compute_score(r_pol)
            update(r_pol, sc_pol)
            print(f"  Polish round {round_idx} (lr={lr_val}, k={top_k}): {sc_pol:.10f}")
            remaining = deadline - time.monotonic()

    # ===== Phase 6: Try smaller n alternatives =====
    if best_seq is not None and best_seq.size > 512:
        for n_try in [512, 256]:
            if time.monotonic() > deadline - 15:
                break
            p_small = interpolate_profile(best_seq, n_try)
            sc_s = compute_score(p_small)
            update(p_small, sc_s)
            if sc_s < best_score + 0.002:
                dl_s = min(deadline, time.monotonic() + 30.0)
                r_s, _ = polish_exact(p_small, dl_s, rng, top_k=5)
                sc_s2 = compute_score(r_s)
                update(r_s, sc_s2)
                print(f"  Smaller n={n_try}: {sc_s2:.10f}")

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
