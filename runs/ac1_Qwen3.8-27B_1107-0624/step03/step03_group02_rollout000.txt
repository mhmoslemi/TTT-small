import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_grad(a, beta):
    """Compute gradient of soft-max of conv(a,a) wrt a, using FFT."""
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    # Gradient of sum_k w[k] * c[k] where c[k] = sum_j a[j]*a[k-j]
    # d/dc[k] * c[k]/da[i] = a[i] appears in c[k] as a[j]*a[k-j]
    # dc[k]/da[i] = a[k-i] + a[i-k] (with appropriate bounds)
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h, float(mval + beta * np.log(s))


def _subgradient_at_max(a, k):
    """Exact subgradient of max convolution at index k."""
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, coeff=2.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + coeff * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_shaped(n, depth=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - depth * np.cos(4.0 * np.pi * t) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    # Phase 2: Adam smooth max
    if time.monotonic() < deadline - 20:
        d2 = min(deadline - 20, time.monotonic() + 180)
        a2, s2 = _adam_smooth_max_v2(best_a, d2, max_iter=25000, lr=0.02,
                                      beta_factor=0.03, beta_decay=0.94)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        if verbose:
            print(f"    After Adam: {best_sc:.10f}")

    # Phase 3: Peak pushing
    if time.monotonic() < deadline - 10:
        d3 = min(deadline - 10, time.monotonic() + 120)
        a3, s3 = _peak_pushing(best_a, d3, max_iter=15000)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        if verbose:
            print(f"    After PeakPush: {best_sc:.10f}")

    # Phase 4: Fine Nesterov
    if time.monotonic() < deadline - 5:
        d4 = deadline - 5
        a4, s4 = _nesterov_smooth_max(best_a, d4, max_iter=20000, lr=0.008,
                                       beta_init=0.03, beta_decay=0.98)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        if verbose:
            print(f"    After FineNesterov: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: Nesterov smooth-max + Adam + peak-pushing.
    Explores diverse initializations at multiple sequence lengths with
    adaptive time allocation and aggressive plateau-breaking.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Build candidate initializations
    candidates = []
    lengths_to_try = [256, 384, 512, 640, 768, 1024]

    if parent is not None:
        for n_try in lengths_to_try:
            candidates.append(_interpolate_profile(parent, n_try))

    # Diverse profiles
    profile_generators = [
        lambda n: _make_linear_profile(n, 1.0),
        lambda n: _make_linear_profile(n, 2.0),
        lambda n: _make_linear_profile(n, 3.0),
        lambda n: _make_cubic_profile(n, 2.0, -1.5),
        lambda n: _make_cubic_profile(n, 1.0, -0.5),
        lambda n: _make_cosine_profile(n, 0.5),
        lambda n: _make_cosine_profile(n, 1.5),
        lambda n: _make_bell_profile(n, 0.4),
        lambda n: _make_bell_profile(n, 0.8),
        lambda n: _make_quadratic_profile(n, 1.5),
        lambda n: _make_w_shaped(n, 0.3),
    ]

    # Add structured + random profiles at key lengths
    for n_try in [256, 512, 768, 1024]:
        for gen in profile_generators:
            candidates.append(gen(n_try))
        # Random smooth profile
        t = np.linspace(0, 1, n_try)
        for f_base in [1, 2, 3]:
            a_rand = 1.0 + 0.3 * rng.normal(size=n_try) * np.cos(2 * np.pi * f_base * t)
            a_rand = np.maximum(a_rand, 1e-8)
            candidates.append(a_rand / a_rand.sum())

    # Evaluate all candidates quickly and sort
    scored_candidates = []
    for c in candidates:
        sc = _score(c)
        if np.isfinite(sc):
            scored_candidates.append((sc, c))
            if sc < best_score:
                best_score = sc
                best_seq = c.copy()

    scored_candidates.sort(key=lambda x: x[0])
    print(f"Best initial: score={best_score:.10f}")
    print(f"Top 10 initial scores: {[f'{s:.6f}' for s, _ in scored_candidates[:10]]}")

    # Time allocation: dedicate more time to better candidates
    # Strategy: try top candidates with full optimization, then refine best

    # Allocate time: ~400s per top candidate, remaining for final polish
    top_k = min(5, len(scored_candidates))
    per_candidate_time = min(350.0, (deadline - t_start - 200) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > deadline - 100:
            break
        sc_init, a_init = scored_candidates[i]
        n_c = a_init.size
        label = f"Top{i+1}(n={n_c},init={sc_init:.6f})"
        print(f"\n=== Optimizing {label} ===")

        d_cand = min(deadline - 100, time.monotonic() + per_candidate_time)
        a_opt, s_opt = _full_optimize(a_init, d_cand, verbose=True)

        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {best_score:.10f}")

    # Final refinement on the best found
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final refinement (best={best_score:.10f}) ===")

        # Try interpolating best to different lengths and re-optimizing
        best_n = best_seq.size
        try_lengths = [512, 768, 1024]
        if best_n not in try_lengths:
            try_lengths = [best_n] + try_lengths

        for n_try in try_lengths[:3]:
            if time.monotonic() > deadline - 30:
                break
            a_interp = _interpolate_profile(best_seq, n_try)
            d_fin = min(deadline - 20, time.monotonic() + 150)
            a_fin, s_fin = _nesterov_smooth_max(a_interp, d_fin, max_iter=20000,
                                                 lr=0.01, beta_init=0.03, beta_decay=0.98)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = a_fin.copy()
                print(f"  Final refinement (n={n_try}): {best_score:.10f}")

        # One more Adam pass
        if time.monotonic() < deadline - 10:
            a_fin2, s_fin2 = _adam_smooth_max_v2(best_seq, deadline - 10,
                                                  max_iter=10000, lr=0.008,
                                                  beta_factor=0.015, beta_decay=0.99)
            if s_fin2 < best_score:
                best_score = s_fin2
                best_seq = a_fin2.copy()
                print(f"  Final Adam: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    # Final cleanup
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)
        best_score = _score(final_seq)

    # Validate with evaluate_sequence
    final_list = [float(x) for x in final_seq]
    try:
        sc_eval = float(evaluate_sequence(final_list))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
        else:
            print(f"WARNING: evaluate_sequence returned inf, using fallback")
            final_seq = _make_linear_profile(256, 1.5)
            final_list = [float(x) for x in final_seq]
            best_score = _score(final_seq)
    except Exception as e:
        print(f"Evaluate error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return final_list
