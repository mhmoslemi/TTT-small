"""
Multi-scale L-BFGS-B optimization with soft-max smoothing for minimizing
2n*max(conv(a,a))/sum(a)^2. Uses Frank-Wolfe warmup, scipy L-BFGS-B with
adaptive beta schedule, multi-scale interpolation, and diverse restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def cross_corr(w, a):
    """Compute h[i] = sum_k w[k] * a[k-i] via FFT (valid range)."""
    n = len(a)
    nw = len(w)
    # h[i] = sum_k w[k] * a[k-i] for i = 0..n-1
    # This is (w flipped) convolved with a, taking the middle n elements
    wf = w[::-1]
    N = next_pow2(n + nw - 1)
    fa = np.fft.rfft(a, N)
    fwf = np.fft.rfft(wf, N)
    conv = np.fft.irfft(fa * fwf, N)
    return conv[nw - 1:nw - 1 + n]


def soft_max_value_and_weights(c, beta):
    """Compute soft-max value and softmax weights."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def gradient_in_z(a, c, beta, n):
    """Compute gradient of 2n*soft_max(conv(a,a),beta) with respect to z (log-space)."""
    _, w = soft_max_value_and_weights(c, beta)
    h = cross_corr(w, a)
    g_a = 2.0 * h  # d(soft_max)/da
    dot = float(np.dot(a, g_a))
    grad_z = a * (g_a - dot)
    return 2.0 * n * grad_z


def objective_and_grad(z, a_cache, beta, n):
    """Objective and gradient in z-space for L-BFGS-B."""
    z = np.asarray(z, dtype=np.float64)
    z_centered = z - z.mean()
    z_shifted = z_centered - z_centered.max()
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-30:
        return 1e10, np.zeros(n)
    a = a / s
    c = full_convolve_fft(a)
    _, w = soft_max_value_and_weights(c, beta)
    val = 2.0 * n * (float(c.max()) + beta * float(np.log(np.sum(np.exp((c - c.max()) / beta)))))
    h = cross_corr(w, a)
    g_a = 2.0 * h
    dot = float(np.dot(a, g_a))
    grad_z = 2.0 * n * a * (g_a - dot)
    return val, grad_z


def lbfgsb_optimize(a0, deadline, rng, beta_start=0.1, beta_stages=None, max_iter_per_stage=200):
    """
    Multi-stage L-BFGS-B optimization in log-space.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    if beta_stages is None:
        c0 = full_convolve_fft(a0)
        max_c0 = float(c0.max())
        beta_stages = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        beta_stages = [b * max_c0 for b in beta_stages]

    best_a = a0.copy()
    best_score = float(2.0 * n * full_convolve_fft(a0).max())

    for stage, beta in enumerate(beta_stages):
        if time.monotonic() > deadline:
            break

        state = {'z': z.copy()}

        def local_obj(gradz):
            z_cur = state['z']
            a_cur = np.exp(z_cur - z_cur.max())
            s = a_cur.sum()
            if s < 1e-30:
                return 1e10, np.zeros(n)
            a_cur = a_cur / s
            c = full_convolve_fft(a_cur)
            sv, w = soft_max_value_and_weights(c, beta)
            val = 2.0 * n * sv
            h = cross_corr(w, a_cur)
            g_a = 2.0 * h
            dot = float(np.dot(a_cur, g_a))
            g_z = 2.0 * n * a_cur * (g_a - dot)
            return val, g_z

        def obj_wrapper(zv):
            state['z'] = np.asarray(zv, dtype=np.float64).copy()
            v, g = local_obj(True)
            return v

        def objgrad_wrapper(zv):
            state['z'] = np.asarray(zv, dtype=np.float64).copy()
            v, g = local_obj(True)
            return v, g

        # L-BFGS-B
        time_stage = max(1.0, (deadline - time.monotonic()) / max(1, len(beta_stages) - stage))
        res = minimize(
            objgrad_wrapper, z.copy(), method='L-BFGS-B',
            options={'maxiter': max_iter_per_stage, 'maxfun': max_iter_per_stage * 3,
                     'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 20}
        )

        z_new = res.x
        z_new = z_new - z_new.mean()
        a_new = np.exp(z_new - z_new.max())
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            c_new = full_convolve_fft(a_new)
            sc_new = float(2.0 * n * c_new.max())
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                z = z_new
        else:
            a_new = best_a.copy()
            z = np.log(a_new)
            z = z - z.mean()

    return best_a, best_score


def frank_wolfe_optimize(a0, deadline, rng, max_inner=5):
    """
    Frank-Wolfe (conditional gradient) method for the non-smooth objective.
    Very fast per iteration, good for warmup.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    best_a = a0.copy()
    c = full_convolve_fft(a0)
    best_score = float(2.0 * n * c.max())

    t_start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())

        # Find top K peaks
        K = min(max_inner, 2 * n - 1)
        peak_idx = np.argsort(c)[-K:]

        # Uniform weights on peaks
        w = np.zeros(2 * n - 1)
        w[peak_idx] = 1.0 / K

        # Gradient
        h = cross_corr(w, a0)
        grad = 2.0 * h

        # Frank-Wolfe: find e_j that minimizes <grad, e_j> = grad[j]
        j_star = int(np.argmin(grad))

        # Step size: exact line search (try a few)
        scores = []
        for t in [2.0 / (it + 2), 0.1, 0.2, 0.3, 0.5, 0.7]:
            a_trial = (1 - t) * a0 + t * 0
            a_trial[j_star] += t
            # Re-normalize
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
                c_trial = full_convolve_fft(a_trial)
                scores.append((float(c_trial.max()), a_trial.copy()))
        if scores:
            scores.sort(key=lambda x: x[0])
            best_trial = scores[0]
            if best_trial[0] < max_c * (1 - 1e-12):
                a0 = best_trial[1]
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # If stalled, try a random vertex
        if stall > 50:
            j_rand = rng.integers(n)
            t = 0.1
            a0 = (1 - t) * a0
            a0[j_rand] += t
            a0 = a0 / a0.sum()
            stall = 0

        if it % 100 == 0:
            c_check = full_convolve_fft(a0)
            sc_check = float(2.0 * n * c_check.max())
            if sc_check < best_score:
                best_score = sc_check
                best_a = a0.copy()

        it += 1

    return best_a, best_score


def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_bump_profile(n, width=0.5, offset=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - offset) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_ramp_profile(n, p=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + 0.05
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_symmetric_triangle(n):
    t = np.linspace(-1.0, 1.0, n)
    a = np.maximum(0.1, 1.0 - np.abs(t))
    return a / a.sum()


def diverse_profiles(n, rng):
    profiles = []
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        profiles.append(make_linear_profile(n, eta))
    for k in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            profiles.append(make_cosine_profile(n, k, amp))
    for width in [0.2, 0.4, 0.6, 0.8]:
        for offset in [0.3, 0.5, 0.7]:
            profiles.append(make_bump_profile(n, width, offset))
    for p in [0.5, 1.0, 2.0, 4.0]:
        profiles.append(make_ramp_profile(n, p))
    profiles.append(make_symmetric_triangle(n))
    # Random smooth
    for _ in range(8):
        coeffs = rng.uniform(-0.5, 0.5, 6)
        t = np.linspace(0.0, 1.0, n)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def adam_optimize(a0, deadline, rng, lr=0.03, beta_factor=0.05, beta_decay=0.93,
                  beta_min=1e-12, restart_period=500):
    """Adam optimizer in log-space with soft-max."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())
    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-8

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_value_and_weights(c, beta)
        h = cross_corr(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        if no_improve > 400:
            current_lr = max(0.003, current_lr * 0.98)

        if no_improve > restart_period:
            noise = 0.1 * rng.normal(size=n)
            z = best_z + noise
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, 0.08 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = lr
            no_improve = 0

    return best_a, best_score


def coordinate_descent_polish(a0, deadline, rng, n_sweep=100):
    """Coordinate descent: try adjusting individual elements."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    best_a = a0.copy()
    c = full_convolve_fft(a0)
    best_score = float(2.0 * n * c.max())

    it = 0
    while time.monotonic() < deadline:
        # Random subset of coordinates
        idx = rng.choice(n, size=min(n_sweep, n), replace=False)
        improved = False
        for j in idx:
            # Try small perturbations
            for delta in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                a_trial = a0.copy()
                a_trial[j] = max(1e-15, a_trial[j] + delta)
                a_trial = a_trial / a_trial.sum()
                c_trial = full_convolve_fft(a_trial)
                sc_trial = float(2.0 * n * c_trial.max())
                if sc_trial < best_score - 1e-12:
                    a0 = a_trial
                    best_score = sc_trial
                    best_a = a0.copy()
                    improved = True
                    break

        it += 1
        if not improved and it % 5 == 0:
            # Add small noise to escape
            a0 = a0 + 0.0001 * rng.normal(size=n)
            a0 = np.maximum(a0, 1e-15)
            a0 = a0 / a0.sum()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):

    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    print(f"Starting search. Budget={budget_s}s")

    # Load initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Fast exploration with Frank-Wolfe at n=512 =====
    n1 = 512
    phase1_deadline = min(deadline, t_start + 200.0)
    print(f"\n--- Phase 1: n={n1} Frank-Wolfe exploration ---")

    profiles_1 = diverse_profiles(n1, rng)
    if init_seq is not None:
        profiles_1.insert(0, interpolate_profile(init_seq, n1))
    # Also add uniform
    profiles_1.insert(0, np.ones(n1) / n1)

    fw_best = None
    fw_best_score = np.inf
    for idx, p in enumerate(profiles_1[:15]):
        if time.monotonic() > phase1_deadline - 5:
            break
        t0 = time.monotonic()
        result, sc_r = frank_wolfe_optimize(p.copy(), min(phase1_deadline, t0 + 10.0), rng)
        sc_f = compute_score(result)
        if sc_f < fw_best_score:
            fw_best_score = sc_f
            fw_best = result.copy()
        update(result, sc_f)
        print(f"  FW profile {idx}: score={sc_f:.8f} ({time.monotonic()-t0:.1f}s)")

    if fw_best is not None:
        print(f"  Phase 1 FW best: {fw_best_score:.10f}")

    # ===== Phase 2: L-BFGS-B refinement at n=512 =====
    phase2_deadline = min(deadline, t_start + 600.0)
    print(f"\n--- Phase 2: n={n1} L-BFGS-B refinement ---")

    # Start from best of Frank-Wolfe or init
    if fw_best is not None:
        p2_start = fw_best.copy()
    elif init_seq is not None:
        p2_start = interpolate_profile(init_seq, n1)
    else:
        p2_start = make_linear_profile(n1, 2.0)

    # Multi-stage L-BFGS-B
    dl2 = min(phase2_deadline, time.monotonic() + 300.0)
    result, sc_r = lbfgsb_optimize(p2_start, dl2, rng)
    sc_f = compute_score(result)
    update(result, sc_f)
    print(f"  After L-BFGS-B: {sc_f:.10f}")

    # Adam follow-up
    if time.monotonic() < phase2_deadline - 30:
        dl2b = min(phase2_deadline, time.monotonic() + 200.0)
        result2, sc2 = adam_optimize(result, dl2b, rng, lr=0.02, beta_factor=0.03,
                                     beta_decay=0.94)
        sc2_f = compute_score(result2)
        update(result2, sc2_f)
        print(f"  After Adam polish: {sc2_f:.10f}")
        if sc2_f < sc_f:
            result = result2

    # Try a few other starting points with L-BFGS-B
    for idx in range(1, min(6, len(profiles_1))):
        if time.monotonic() > phase2_deadline - 20:
            break
        p_alt = profiles_1[idx].copy()
        dl_alt = min(phase2_deadline, time.monotonic() + 40.0)
        r_alt, sc_alt = lbfgsb_optimize(p_alt, dl_alt, rng)
        sc_alt_f = compute_score(r_alt)
        update(r_alt, sc_alt_f)
        if sc_alt_f < best_score + 0.05:
            print(f"  Alt profile {idx}: {sc_alt_f:.8f}")

    print(f"  Phase 2 best: {best_score:.10f}")

    # ===== Phase 3: Multi-scale to n=1024 =====
    n3 = 1024
    phase3_deadline = min(deadline, t_start + 1200.0)
    print(f"\n--- Phase 3: n={n3} refinement ---")

    if best_seq is not None:
        p3 = interpolate_profile(best_seq, n3)
    else:
        p3 = interpolate_profile(result, n3)

    sc3_init = compute_score(p3)
    update(p3, sc3_init)
    print(f"  Interpolated to n={n3}: {sc3_init:.10f}")

    # L-BFGS-B at n=1024
    dl3a = min(phase3_deadline, time.monotonic() + 300.0)
    result3, sc3 = lbfgsb_optimize(p3, dl3a, rng)
    sc3_f = compute_score(result3)
    update(result3, sc3_f)
    print(f"  After L-BFGS-B n=1024: {sc3_f:.10f}")

    # Adam at n=1024
    if time.monotonic() < phase3_deadline - 30:
        dl3b = min(phase3_deadline, time.monotonic() + 200.0)
        r3a, sc3a = adam_optimize(result3, dl3b, rng, lr=0.015, beta_factor=0.02,
                                  beta_decay=0.94)
        sc3a_f = compute_score(r3a)
        update(r3a, sc3a_f)
        print(f"  After Adam n=1024: {sc3a_f:.10f}")
        if sc3a_f < sc3_f:
            result3 = r3a

    print(f"  Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Try n=2048 =====
    n4 = 2048
    phase4_deadline = min(deadline, t_start + 1600.0)
    print(f"\n--- Phase 4: n={n4} ---")

    if best_seq is not None:
        p4 = interpolate_profile(best_seq, n4)
    else:
        p4 = interpolate_profile(result3, n4)

    sc4_init = compute_score(p4)
    update(p4, sc4_init)
    print(f"  Interpolated to n={n4}: {sc4_init:.10f}")

    dl4a = min(phase4_deadline, time.monotonic() + 200.0)
    result4, sc4 = lbfgsb_optimize(p4, dl4a, rng)
    sc4_f = compute_score(result4)
    update(result4, sc4_f)
    print(f"  After L-BFGS-B n=2048: {sc4_f:.10f}")

    if time.monotonic() < phase4_deadline - 20:
        dl4b = min(phase4_deadline, time.monotonic() + 100.0)
        r4a, sc4a = adam_optimize(result4, dl4b, rng, lr=0.01, beta_factor=0.015,
                                  beta_decay=0.95)
        sc4a_f = compute_score(r4a)
        update(r4a, sc4a_f)
        print(f"  After Adam n=2048: {sc4a_f:.10f}")

    # ===== Phase 5: Final polish at best n =====
    print(f"\n--- Phase 5: Final polish ---")
    polish_deadline = deadline

    if best_seq is not None:
        n_polish = best_seq.size
        p_polish = best_seq.copy()

        # Small noise + Adam
        if time.monotonic() < polish_deadline - 30:
            p_noisy = p_polish + 0.0002 * rng.normal(size=n_polish)
            p_noisy = np.maximum(p_noisy, 1e-12)
            p_noisy = p_noisy / p_noisy.sum()
            dl_pol = min(polish_deadline, time.monotonic() + 80.0)
            r_pol, sc_pol = adam_optimize(p_noisy, dl_pol, rng, lr=0.005,
                                          beta_factor=0.01, beta_decay=0.95)
            sc_pol_f = compute_score(r_pol)
            update(r_pol, sc_pol_f)
            print(f"  Noise+Adam polish: {sc_pol_f:.10f}")

        # Coordinate descent
        if time.monotonic() < polish_deadline - 20:
            dl_cd = min(polish_deadline, time.monotonic() + 40.0)
            r_cd, sc_cd = coordinate_descent_polish(best_seq, dl_cd, rng, n_sweep=200)
            sc_cd_f = compute_score(r_cd)
            update(r_cd, sc_cd_f)
            print(f"  Coord descent: {sc_cd_f:.10f}")

    # ===== Also check smaller n in case it's better =====
    if best_seq is not None and best_seq.size > 512:
        for n_try in [256, 384, 512, 768]:
            if time.monotonic() > deadline - 10:
                break
            p_s = interpolate_profile(best_seq, n_try)
            sc_s = compute_score(p_s)
            update(p_s, sc_s)
            if sc_s < best_score + 0.02:
                dl_s = min(deadline, time.monotonic() + 20.0)
                r_s, _ = adam_optimize(p_s, dl_s, rng, lr=0.01, beta_factor=0.02,
                                       beta_decay=0.94)
                sc_s2 = compute_score(r_s)
                update(r_s, sc_s2)
                print(f"  Smaller n={n_try}: {sc_s2:.10f}")

    print(f"\n  Best score so far: {best_score:.10f}")

    # ===== Final validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 2.0)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
