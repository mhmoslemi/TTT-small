"""
L_p continuation + soft-max sharpening + subgradient polish for minimizing
2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FFT-based exact gradients, Adam in logit space, and Polyak averaging.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def compute_cross_corr(w, a):
    """h[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i=0,...,n-1 via FFT."""
    n = a.size
    N = next_pow2(3 * n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return corr[n - 1:2 * n - 1]


def softmax_from_z(z):
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return a / s


def compute_lp_gradient(a, p):
    """Compute gradient of (sum b^p)^(1/p) w.r.t. a (up to positive constant)."""
    b = full_convolve_fft(a)
    if p == 1:
        w = np.ones_like(b)
    elif p == 2:
        w = b.copy()
    else:
        # Work in log space to avoid overflow
        max_b = b.max()
        if max_b < 1e-300:
            w = np.zeros_like(b)
        else:
            log_b = np.where(b > 0, np.log(np.maximum(b, 1e-300)), -np.inf)
            log_w = (p - 1) * log_b
            max_log_w = log_w.max()
            if max_log_w > 500:
                log_w = log_w - max_log_w
            w = np.exp(log_w)
    h = compute_cross_corr(w, a)
    return 2.0 * h, b


def compute_softmax_gradient(a, beta):
    """Compute gradient of soft-max(beta) w.r.t. a (up to positive constant)."""
    b = full_convolve_fft(a)
    max_b = b.max()
    if beta < 1e-14:
        mask = b >= max_b - 1e-12
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0
        else:
            w = mask.astype(np.float64) / cnt
    else:
        x = (b - max_b) / beta
        x = np.clip(x, -60.0, 0.0)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-300:
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0
        else:
            w = ex / s
    h = compute_cross_corr(w, a)
    return 2.0 * h, b


def adam_logit_optimize(a0, deadline, rng, mode='lp', p=2, beta_init=None,
                        lr=0.05, lr_decay=0.999, beta_decay=0.95,
                        beta_min=1e-12, max_no_improve=500, verbose=False):
    """
    Adam optimization in logit space.
    mode: 'lp' for L_p objective, 'softmax' for soft-max objective.
    Returns (best_a, best_score_intermediate).
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a + 1e-30)
    z = z - z.mean()

    if mode == 'softmax':
        b = full_convolve_fft(a)
        if beta_init is None:
            beta = 0.08 * max(b.max(), 1e-6)
        else:
            beta = beta_init
    else:
        beta = 1.0  # Not used for L_p

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    b0 = full_convolve_fft(a)
    if mode == 'lp':
        best_val = float(np.sum(b0 ** p)) ** (1.0 / p)
    else:
        best_val = float(b0.max())
    best_a = a.copy()
    best_z = z.copy()
    no_improve = 0
    current_lr = lr

    while time.monotonic() < deadline:
        t += 1

        if mode == 'lp':
            grad_a, b = compute_lp_gradient(a, p)
            curr_val = float(np.sum(b ** p)) ** (1.0 / p)
        else:
            grad_a, b = compute_softmax_gradient(a, beta)
            curr_val = float(b.max())

        # Transform to logit gradient
        dot = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot)

        # Adam update
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        # Clip and renormalize z
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a = softmax_from_z(z)

        if curr_val < best_val - 1e-14:
            best_val = curr_val
            best_a = a.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt beta for softmax mode
        if mode == 'softmax' and t % 50 == 0:
            beta = max(beta_min, beta * beta_decay)

        # Decay learning rate slowly
        if no_improve > 100:
            current_lr = max(0.001, current_lr * lr_decay)

        # Restart if stuck
        if no_improve > max_no_improve:
            noise = 0.1 * rng.normal(size=n)
            z = best_z + noise
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a = softmax_from_z(z)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            current_lr = lr
            if mode == 'softmax':
                b_tmp = full_convolve_fft(a)
                beta = 0.08 * max(b_tmp.max(), 1e-6)
            no_improve = 0

    return best_a, best_val


def subgradient_polish(a0, deadline, rng, n_avg=100):
    """
    Exact subgradient method with Polyak averaging.
    Uses uniform weights over argmax set.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a + 1e-30)
    z = z - z.mean()

    b = full_convolve_fft(a)
    max_b = float(b.max())
    best_score = float(2.0 * n * max_b)
    best_a = a.copy()

    # Polyak average
    avg_z = z.copy()
    count = 0
    lr_base = 0.02
    t = 0

    while time.monotonic() < deadline:
        t += 1
        b = full_convolve_fft(a)
        max_b = float(b.max())
        curr_score = float(2.0 * n * max_b)

        if curr_score < best_score - 1e-14:
            best_score = curr_score
            best_a = a.copy()

        # Find argmax set (with tolerance)
        tol = max_b * 0.001 + 1e-12
        mask = b >= max_b - tol
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        # Subgradient
        h = compute_cross_corr(w, a)
        grad_a = 2.0 * h

        # Logit gradient
        dot = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot)

        # Step with annealed lr
        lr = lr_base / (1.0 + 0.01 * t)
        z = z - lr * dz
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        a = softmax_from_z(z)

        # Polyak averaging
        count += 1
        avg_z = avg_z + (z - avg_z) / count

        # Occasionally evaluate the average
        if t % 100 == 0:
            a_avg = softmax_from_z(avg_z)
            b_avg = full_convolve_fft(a_avg)
            score_avg = float(2.0 * n * b_avg.max())
            if score_avg < best_score:
                best_score = score_avg
                best_a = a_avg.copy()

    return best_a, best_score

def interpolate_profile(p0, n_new):

                    print(f"      L-BFGS L_{p}: score={sc_new:.10f} ({time.monotonic()-t_stage_start:.1f}s)")
                except Exception as e:
                    print(f"      L-BFGS failed: {e}")
                    # Fallback to Adam
                    dl = min(phase1_deadline, time.monotonic() + 10.0)
                    a_adam, _ = adam_logit_optimize(a, dl, rng, mode='lp', p=p, lr=0.08)
                    sc_new = compute_score_normalized(a_adam)
                    if sc_new < sc:
                        a = a_adam
                        sc = sc_new
                        update(a, sc)
                    print(f"      Adam L_{p}: score={sc_new:.10f}")
            else:
                # Adam for L_8
                dl = min(phase1_deadline, time.monotonic() + 15.0)
                a_new, _ = adam_logit_optimize(a, dl, rng, mode='lp', p=p, lr=0.05)
                sc_new = compute_score_normalized(a_new)
                if sc_new < sc:
                    a = a_new
                    sc = sc_new
                    update(a, sc)
                print(f"    L_{p} (Adam): score={sc_new:.10f} ({time.monotonic()-t_stage_start:.1f}s)")

        print(f"  After L_p stages: {sc:.10f}")

        # --- Phase 2: Soft-max sharpening ---
        phase2_deadline = min(deadline - 120, t_phase_start + budget_s * 0.55)
        print(f"\n  Phase 2: Soft-max sharpening")

        b = full_convolve_fft(a)
        max_b_init = float(b.max())

        beta_schedule = [0.10, 0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001]
        lr_schedule = [0.04, 0.035, 0.03, 0.025, 0.02, 0.015, 0.012, 0.008, 0.006]

        for i, (bf, lr_val) in enumerate(zip(beta_schedule, lr_schedule)):
            if time.monotonic() > phase2_deadline - 10:
                break

            beta_val = bf * max_b_init

            # Determine time allocation for this sub-stage
            remaining = phase2_deadline - time.monotonic()
            stages_left = len(beta_schedule) - i
            alloc = remaining / stages_left
            alloc = min(alloc, 150.0)  # Cap per-stage at 150s

            dl = time.monotonic() + alloc
            print(f"    Beta factor={bf:.4f} (beta={beta_val:.6f}), lr={lr_val}, alloc={alloc:.0f}s")

            a_new, _ = adam_logit_optimize(
                a, dl, rng, mode='softmax', beta_init=beta_val,
                lr=lr_val, beta_decay=0.96, max_no_improve=300
            )
            sc_new = compute_score_normalized(a_new)
            if sc_new < sc - 1e-12:
                a = a_new
                sc = sc_new
            update(a, sc_new)
            print(f"      Score: {sc_new:.10f}")

        print(f"  After soft-max: {sc:.10f}")

        # --- Phase 3: Subgradient polish ---
        phase3_deadline = min(deadline - 30, time.monotonic() + 200.0)
        if time.monotonic() < phase3_deadline - 10:
            print(f"\n  Phase 3: Subgradient polish")
            a_poly, sc_poly = subgradient_polish(a, phase3_deadline, rng)
            sc_poly = compute_score_normalized(a_poly)
            if sc_poly < sc:
                a = a_poly
                sc = sc_poly
            update(a, sc_poly)
            print(f"  After subgradient: {sc_poly:.10f}")

        # --- Phase 4: Final Adam polish with very small beta ---
        if time.monotonic() < deadline - 30 and n_target == max(target_ns):
            print(f"\n  Phase 4: Final polish")
            b = full_convolve_fft(a)
            max_b = float(b.max())
            for beta_f in [0.0005, 0.0003, 0.0001]:
                dl = min(deadline - 20, time.monotonic() + 60.0)
                a_new, _ = adam_logit_optimize(
                    a, dl, rng, mode='softmax',
                    beta_init=beta_f * max_b, lr=0.005,
                    beta_decay=0.98, max_no_improve=500
                )
                sc_new = compute_score_normalized(a_new)
                if sc_new < sc:
                    a = a_new
                    sc = sc_new
                update(a, sc_new)
                print(f"    beta_f={beta_f}: {sc_new:.10f}")

        print(f"\n  Final for n={n_target}: {sc:.10f}")

    # ===== Try to improve best found with additional optimization =====
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        print(f"\n\nFinal refinement at n={n_best}")

        # Multiple restarts with small perturbations
        for restart in range(3):
            if time.monotonic() > deadline - 20:
                break
            a_pert = best_seq + 0.002 * rng.normal(size=n_best)
            a_pert = np.maximum(a_pert, 1e-12)
            a_pert = a_pert / a_pert.sum()

            b_pert = full_convolve_fft(a_pert)
            max_bp = float(b_pert.max())

            dl = min(deadline - 20, time.monotonic() + 60.0)
            a_new, _ = adam_logit_optimize(
                a_pert, dl, rng, mode='softmax',
                beta_init=0.003 * max_bp, lr=0.005,
                beta_decay=0.98, max_no_improve=400
            )
            sc_new = compute_score_normalized(a_new)
            update(a_new, sc_new)
            print(f"  Restart {restart}: {sc_new:.10f}")

    print(f"\n{'='*50}")
    print(f"FINAL BEST: n={best_seq.size if best_seq is not None else 0}, score={best_score:.10f}")

    # ===== Validation =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_initial_profile(512, rng, 'mixed')

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_initial_profile(256, rng, 'mixed')

    # Verify with evaluate_sequence
    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        print(f"evaluate_sequence confirms: {eval_score:.10f}")
        if not np.isfinite(eval_score):
            print("WARNING: evaluate_sequence returned inf, using fallback")
            final_seq = make_initial_profile(512, rng, 'mixed')
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check error: {e}")

    print(f"FINAL: n={final_seq.size}, score={compute_score_normalized(final_seq):.10f}")
    return [float(x) for x in final_seq]
