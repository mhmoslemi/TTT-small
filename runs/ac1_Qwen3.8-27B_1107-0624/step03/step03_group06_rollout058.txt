"""
Optimized step function search using geometric initialization, aggressive LP refinement
with tight inflation, Nesterov-accelerated smooth-max gradient polish, and peak-focused
coordinate updates. Targets score < 1.5027 by exploiting the asymmetric (geometric-like)
structure of optimal sequences where the convolution peak is off-center.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-10:
        return a
    return a / s


def _make_geometric(n, r):
    a = np.power(r, np.arange(n, dtype=np.float64))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_beta(n, alpha, beta):
    t = np.linspace(1e-6, 1.0 - 1e-6, n)
    a = t ** alpha * (1.0 - t) ** beta + 1e-8
    return a / a.sum()


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _solve_lp(f, M, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.success or (res.status == 1 and res.x is not None and np.all(res.x >= -1e-9)):
        x = np.maximum(res.x, 0.0) if res.x is not None else None
        return x
    return None


def _line_search_fine(a, g, base_score, n_grid=17):
    best_t = 0.0
    best_sc = base_score
    ts = np.geomspace(0.02, 1.0, n_grid)
    for t in ts:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.003, inflate_min=1.0005):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start
    stall_count = 0

    for it in range(max_iter):

            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(0.01, inflate * 1.4)
            if no_imp > 200:
                inflate = min(0.02, inflate * 1.2)
            if no_imp > 400:
                break
            continue

        g = g / sum_g

        best_t, best_sc = _line_search_fine(a, g, best_score, n_grid=13)

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:

                no_imp = 0

                    break

def _peak_grad(a, top_k=5):
    """Compute gradient focusing on the top_k peaks of the convolution."""
    n = len(a)
    c = _fft_convolve(a)
    max_c = float(c.max())

    sorted_idx = np.argsort(c)[::-1][:top_k]
    weights = np.zeros(2 * n - 1)
    peak_vals = c[sorted_idx]
    w = np.exp((peak_vals - max_c) / (max_c * 0.01 + 1e-12))
    w = w / w.sum()
    weights[sorted_idx] = w

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(weights, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, max_c


def _nesterov_polish(a0, deadline, max_iter=80000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    lr = 0.05
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    beta_schedules = [5, 10, 20]
    top_k = 3

    for it in range(max_iter):
        if time.monotonic() > deadline - 5:
            break

        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0.0)
        s_nest = a_nest.sum()
        if s_nest > 1e-10:
            a_nest = a_nest / s_nest

        grad, maxc = _peak_grad(a_nest, top_k=top_k)

        if not np.all(np.isfinite(grad)):
            v = np.zeros(n)
            lr *= 0.5
            continue

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c = _fft_convolve(a_new)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = 0.05
                no_imp = 0

        a = a_new

        if it % 500 == 499:
            top_k = min(top_k + 1, 8)

    return best_a


def _coordinate_descent(a0, deadline, max_iter=100000):
    """Reduce the peak by adjusting individual coordinates."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline - 5:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))

        # Find which indices contribute to c[k_star]
        # c[k_star] = sum_j a[j] * a[k_star - j] for valid j
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)

        if j_lo > j_hi:
            break

        # Try reducing the largest contributor
        contributions = a[j_lo:j_hi + 1] * a[k_star - j_hi:k_star - j_lo + 1][::-1]
        j_best = j_lo + int(np.argmax(contributions))
        j_other = k_star - j_best

        # Try reducing a[j_best] or a[j_other]
        delta = min(a[j_best], a[j_other]) * 0.1
        trials = []

        # Reduce a[j_best]
        a_try = a.copy()
        a_try[j_best] = max(0, a_try[j_best] - delta)
        if a_try[j_best] < a[j_best]:
            sc = _score(a_try)
            trials.append((sc, a_try.copy()))

        # Reduce a[j_other]
        a_try = a.copy()
        a_try[j_other] = max(0, a_try[j_other] - delta)
        if a_try[j_other] < a[j_other]:
            sc = _score(a_try)
            trials.append((sc, a_try.copy()))

        # Try increasing a small element far from peak
        j_small = int(np.argmin(a))
        a_try = a.copy()
        a_try[j_small] += delta * 0.5
        sc = _score(a_try)
        trials.append((sc, a_try.copy()))

        best_trial_sc = min(t[0] for t in trials) if trials else best_score

        if best_trial_sc < best_score - 1e-12:
            best_score = best_trial_sc
            best_a = [t[1] for t in trials if t[0] == best_trial_sc][0]
            a = best_a.copy()
            no_imp = 0
        else:
            a = a + best_a * 1e-4  # small perturbation
            a = np.maximum(a, 0)
            a = _normalize(a)
            no_imp += 1
            if no_imp > 500:
                break

    return best_a
