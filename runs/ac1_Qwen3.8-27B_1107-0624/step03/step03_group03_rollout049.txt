def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize_to_one(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _make_conv_A(a, lags):
    n = len(a)
    A = np.zeros((len(lags), n))
    for row, k in enumerate(lags):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            vals = a[k - j_hi:k - j_lo + 1][::-1]
            A[row, j_lo:j_hi + 1] = vals
    return A


def _lp_spread(a, M, n_active=200, margin_frac=0.02):
    n = len(a)
    c_aa = _fft_convolve(a)
    max_val = float(c_aa.max())
    threshold = max_val * (1.0 - margin_frac)
    active_mask = c_aa >= threshold
    active_idx = np.where(active_mask)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c_aa)])
    top_idx = np.argsort(c_aa)[::-1][:n_active]
    active_idx = np.unique(np.concatenate([active_idx, top_idx]))
    active_idx = np.sort(active_idx)
    A = _make_conv_A(a, active_idx)
    b_vec = np.full(len(active_idx), M)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _line_search(a, b, max_iter=30):
    best_t = 0.0
    best_sc = _score_normalized(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_normalized(np.maximum(a + c * (b - a), 0))
    fd = _score_normalized(np.maximum(a + d * (b - a), 0))
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d, fd = c, fc
            c = hi - gr * (hi - lo)
            fc = _score_normalized(np.maximum(a + c * (b - a), 0))
        else:
            lo = c
            c, fc = d, fd
            d = lo + gr * (hi - lo)
            fd = _score_normalized(np.maximum(a + d * (b - a), 0))
    best_t = (lo + hi) / 2.0
    return best_t

def _lse_momentum_step(a, z, state, T, lr):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    fa_rev = np.fft.rfft(a[::-1], N)
    fw = np.fft.rfft(w, N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    if not np.all(np.isfinite(gz)):
        return a, z, state
    t_idx = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m_v = b1 * state['m'] + (1.0 - b1) * gz
    v_v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m_v / (1.0 - b1 ** t_idx)
    vh = v_v / (1.0 - b2 ** t_idx)
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -50.0, 50.0)
    z -= float(np.mean(z))
    e = np.exp(z - z.max())
    a_new = e / e.sum()
    state['t'] = t_idx
    state['m'] = m_v
    state['v'] = v_v
    return a_new, z, state

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_full(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    n_base = 1024
    if parent is not None and parent.size > 0:
        n_base = parent.size

    print(f"\nBase length: {n_base}, budget: {budget_s}s")
    print(f"Deadline in: {deadline - time.monotonic():.1f}s")

    # Phase 1: Multi-temperature mirror descent from parent
    print("\n=== Phase 1: Multi-temperature MD from parent ===")
    if parent is not None and time.monotonic() < deadline - 120:
        a1_deadline = min(deadline - 120, t_start + 600)
        a1 = _multi_temperature_md(parent, a1_deadline)
        sc1 = _score_normalized(a1)
        if sc1 < best_score - 1e-10:
            best_score = sc1
            best_seq = a1.copy()
        print(f"  Phase 1 result: {sc1:.10f}")

    # Phase 2: LP refinement on best
    print("\n=== Phase 2: LP refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        lp_deadline = min(deadline - 100, t_start + 1100)
        a2 = _lp_refine(best_seq, lp_deadline, max_iter=4000,
                        inflate_start=0.0008, inflate_min=0.0001)
        sc2 = _score_normalized(a2)
        if sc2 < best_score - 1e-10:
            best_score = sc2
            best_seq = a2.copy()
        print(f"  Phase 2 result: {sc2:.10f}")

    # Phase 3: Try different initialization (bimodal)
    print("\n=== Phase 3: Alternative initialization ===")
    if time.monotonic() < deadline - 200:
        alt_deadline = min(deadline - 200, t_start + 1300)
        inits = [
            _make_bimodal(n_base, 0.3),
            _make_raised_cosine(n_base),
            _make_gaussian(n_base, 0.28),
        ]
        for ii, a_init in enumerate(inits):
            if time.monotonic() > alt_deadline:
                break
            sc_init = _score_normalized(a_init)
            if sc_init >= best_score:
                continue
            a_alt = _mirror_descent_optimize(a_init, alt_deadline, T_start=0.04,
                                             T_end=1e-4, lr=0.12, max_iter=80000,
                                             restart_threshold=2500)
            sc_alt = _score_normalized(a_alt)
            if sc_alt < best_score - 1e-10:
                best_score = sc_alt
                best_seq = a_alt.copy()
            print(f"  Alt init {ii}: init={sc_init:.8f} -> {sc_alt:.10f}")

    # Phase 4: Subgradient polish
    print("\n=== Phase 4: Subgradient polish ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        sp_deadline = min(deadline - 60, t_start + 1550)
        a4 = _subgradient_polish(best_seq, sp_deadline, max_iter=100000, lr_start=0.015)
        sc4 = _score_normalized(a4)
        if sc4 < best_score - 1e-10:
            best_score = sc4
            best_seq = a4.copy()
        print(f"  Phase 4 result: {sc4:.10f}")

    # Phase 5: Final LP (very tight)
    print("\n=== Phase 5: Final tight LP ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        lp2_deadline = min(deadline - 30, t_start + 1700)
        a5 = _lp_refine(best_seq, lp2_deadline, max_iter=3000,
                        inflate_start=0.0003, inflate_min=0.00005)
        sc5 = _score_normalized(a5)
        if sc5 < best_score - 1e-10:
            best_score = sc5
            best_seq = a5.copy()
        print(f"  Phase 5 result: {sc5:.10f}")

    # Phase 6: Try n=2048
    print("\n=== Phase 6: Try n=2048 ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n2 = 2048
        a_2048 = _interpolate_profile(best_seq, n2)
        sc_2048_init = _score_normalized(a_2048)
        print(f"  Interpolated to 2048: {sc_2048_init:.10f}")
        if time.monotonic() < deadline - 15:
            a_2048 = _mirror_descent_optimize(a_2048, deadline, T_start=0.02,
                                              T_end=1e-5, lr=0.08, max_iter=50000,
                                              restart_threshold=3000)
            sc_2048 = _score_normalized(a_2048)
            if sc_2048 < best_score - 1e-10:
                best_score = sc_2048
                best_seq = a_2048.copy()
            print(f"  n=2048 result: {sc_2048:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_raised_cosine(1024)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_raised_cosine(1024)

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
