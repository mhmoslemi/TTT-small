def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a) = 1, so just 2n * max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return 2.0 * n * float(c.max())


def _geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _optimize_geometric_ratio(n, n_samples=500):
    """Find optimal geometric ratio for given length."""
    best_sc = np.inf
    best_r = 1.0
    # Search in log space for r
    log_rs = np.linspace(np.log(1.01), np.log(3.0), n_samples)
    for lr in log_rs:

        a = _geometric(n, r)
        sc = _score(a)
        if sc < best_sc:

            best_r = r
    # Refine around best

            a = _geometric(n, r)

def _make_exp_profile(n, rate):
    """Exponential profile: a[i] = exp(-rate * i / n)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = np.exp(-rate * t)
    s = a.sum()
    return a / s


def _make_ramp(n, alpha=1.0):

def _make_ramp_rev(n, alpha=1.0):

def _make_bathtub(n, alpha=2.0):
    """Bathtub shape: low in middle, high at edges."""
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    a = 1.0 + alpha * (t - center) ** 2
    return a / a.sum()


def _make_cosine(n, freq=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _solve_lp_direction(f, M, n):
    """Solve max sum(b) s.t. (f*b)(k) <= M, b >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _solve_peak_lp(a, k_peak, M, n):
    """Solve LP that specifically targets reducing peak at position k_peak."""
    conv = _fft_convolve(a)
    # Find direction d that reduces conv[k_peak] while not increasing others too much
    # Linearized: minimize (T_kpeak * a) . d subject to (T_k * a) . d <= slack_k
    # where slack_k = M - conv[k]
    T_kpeak_a = np.zeros(n)
    # (T_k * a)[i] = sum_j a[j] * T_k[i,j] = conv_cross between a and reversed shifted
    # Actually, gradient of conv(a,a)[k] w.r.t. a[i] = (T_k a)[i]
    # T_k a = the k-th "shifted" cross-correlation
    # For conv(a,a)[k] = sum_i a[i]*a[k-i+n-1] (valid range)
    # d/d a[m] = a[k-m+n-1] + a[m+k-n+1] (for valid indices)
    
    # Simpler: just use the gradient of the max coefficient
    grad = np.zeros(n)
    kp = k_peak
    for i in range(n):
        j1 = kp - i
        j2 = 2 * (n - 1) - kp + i
        val = 0.0
        if 0 <= j1 < n:
            val += a[j1]
        if 0 <= j2 < n:
            val += a[j2]
        grad[i] = val
    
    # LP: minimize grad . d subject to (grad_k . d) <= slack_k for all k, sum(d)=0, d >= -a
    # This is complex; simplify by just minimizing grad . d with bounds
    # Use a simpler approach: just negate gradient and project
    d = -grad
    d -= d.mean()  # keep sum zero
    d = np.maximum(d, -a)
    s = d.sum()
    if abs(s) > 1e-10:
        d = d / s * 0  # maintain sum=0
    return d


def _line_search_2nd_order(a, g, n_grid=15):
    """Line search with quadratic interpolation for better step size."""
    best_t = 0.0
    best_sc = _score(a)
    ts = np.linspace(0.02, 1.0, n_grid)
    scores = []
    for t in ts:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            scores.append(np.inf)
            continue
        a_new = a_new / s
        sc = _score(a_new)
        scores.append(sc)
    
    if not np.isfinite(min(scores)):
        return 0.0, best_sc
    
    scores = np.array(scores)
    idx = np.argmin(scores)
    best_t = ts[idx]
    best_sc = scores[idx]
    
    # Try to refine around best
    if 0 < idx < len(ts) - 1:
        t0, t1, t2 = ts[idx-1], ts[idx], ts[idx+1]
        f0, f1, f2 = scores[idx-1], scores[idx], scores[idx+1]
        # Quadratic interpolation
        denom = (f0 - 2*f1 + f2)
        if abs(denom) > 1e-15:
            t_star = t1 - 0.5 * (f0 - f1) * (t2 - t1) / denom  # approximate
            t_star = np.clip(t_star, t0, t2)
            a_new = (1.0 - t_star) * a + t_star * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                sc = _score(a_new)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_star
    
    return best_t, best_sc

def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of convolution coefficients."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval, w

def _peak_surgery(a0, deadline, n_iterations=100):
    """Directly target the peak convolution coefficients for reduction."""
    a = np.asarray(a0, dtype=np.float64).copy()

    for it in range(n_iterations):

            break

        c = _fft_convolve(a)
        max_val = float(c.max())
        # Find all positions near the max
        threshold = max_val * 0.999
        peak_positions = np.where(c >= threshold)[0]

        if len(peak_positions) == 0:
            break

        # For each peak position, compute the gradient direction to reduce it
        # and take a step that reduces the worst peak
        k_peak = int(np.argmax(c))
        
        # Compute gradient of conv(a,a)[k_peak] w.r.t. a
        grad = np.zeros(n)
        kp = k_peak
        for i in range(n):
            j1 = kp - i
            j2 = 2 * (n - 1) - kp + i
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                val += a[j2]
            grad[i] = val
        
        # Also compute gradients for other near-peak positions to avoid increasing them
        # Take a safe direction: negative gradient of peak, projected to be small change
        step_size = max_val * 1e-4
        d = -grad / (np.linalg.norm(grad) + 1e-10) * step_size
        d -= d.mean()  # maintain zero mean change
        
        a_new = a + d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            d2 = d * 0.1
            a_new2 = a + d2
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-10:
