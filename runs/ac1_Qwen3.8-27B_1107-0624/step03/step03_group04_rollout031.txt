import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    """Compute soft-max weights for convolution entries."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _compute_gradient(a, beta):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_weights(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c


def _project_steepest_direction(grad, a):
    """
    Project -grad onto {d: sum(d)=0, d_j >= -a_j}.
    Equivalently, find x = -d minimizing ||x - grad||^2 s.t. sum(x)=0, x_j <= a_j.
    Solution: x_j = min(grad_j - lambda, a_j), lambda chosen so sum(x)=0.
    """
    n = len(a)
    # Bisection on lambda
    lo = float(grad.min()) - float(a.max()) - 1.0
    hi = float(grad.max()) + 1.0

    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.minimum(grad - mid, a)
        s = float(x.sum())
        if s > 0:
            lo = mid
        else:
            hi = mid

    lam = (lo + hi) / 2.0
    x = np.minimum(grad - lam, a)
    x = x - x.sum() / n  # enforce sum=0 exactly
    # Re-clip
    x = np.minimum(x, a)
    x = x - x.sum() / n
    d = -x
    return d

            if sc < best_score - 1e-12:
                best_score = sc
                best_t = t
                best_a = a_new.copy()

    return best_t, best_a, best_score


def _steepest_descent_optimize(a, deadline, max_iter=20000, beta_init=None):
    """Exact steepest descent with projection and line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = beta_init if beta_init is not None else max(1e-6, 0.05 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, c_max, c = _compute_gradient(a, beta)

        # Normalize gradient (remove mean for sum-zero constraint)
        grad_centered = grad - float(np.mean(grad))

        # Compute steepest descent direction
        d = _project_steepest_direction(grad_centered, a)

        # Check if direction is meaningful
        if np.max(np.abs(d)) < 1e-12:
            no_imp += 100
            if no_imp > 500:
                break
            continue

        # Line search
        t, a_new, sc_new = _line_search(a, d, c, best_score)

        if sc_new < best_score - 1e-13:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                beta = max(1e-10, beta * 0.8)
            if no_imp > max(500, n):
                # Annealing restart
                rng = np.random.default_rng(it * 31 + 7)
                noise = 0.1 * best_score * rng.standard_normal(n)
                z = np.log(np.maximum(a, 1e-15)) + noise
                a = np.exp(z)
                a = a / a.sum()
                beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
                no_imp = 0

        a = a_new

        # Adapt beta
        if it % 200 == 0 and it > 100:
            c_cur = _fft_convolve(a)
            c_max_cur = float(c_cur.max())
            top5 = np.sort(c_cur)[::-1][:min(5, len(c_cur))]
            if len(top5) > 1:
                gap = top5[0] - top5[-1]
                if gap < beta * 0.5:
                    beta = max(1e-10, beta * 0.7)
                elif gap > beta * 5.0:
                    beta = min(0.5 * c_max_cur, beta * 1.3)

    return best_a


def _coordinate_descent(a, deadline, max_iter=50000, n_top=15, n_pairs=200):
    """
    Peak-aware coordinate descent: transfer mass between pairs (i,j)
    to reduce the maximum of the convolution.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(99991)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        # Identify peak entries
        top_idx = np.argsort(c)[::-1][:n_top]
        top_c = c[top_idx]

        # Compute "pressure" at each position from peak entries
        pressure = np.zeros(n)
        for k in top_idx:
            w = max(top_c[0] - c[k] + c[k], c[k])  # weight by peak value
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                pressure[i] += a[j]

        # Sample random pairs (i, j) where moving from high to low pressure helps
        found_move = False
        for _ in range(n_pairs):
            # Sample i from high-pressure positions, j from low-pressure
            if rng.random() < 0.7:
                # Biased sampling
                p_weights = np.maximum(pressure, 1e-10)
                p_weights = p_weights ** 2
                p_w = p_weights / p_weights.sum()
                i = rng.choice(n, p=p_w)
                q_weights = np.maximum(1.0 + pressure.max() - pressure, 1e-10)
                q_w = q_weights / q_weights.sum()
                j = rng.choice(n, p=q_w)
            else:
                i = rng.integers(n)
                j = rng.integers(n)

            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue

            # Compute the effect of moving mass from i to j
            # conv(a')_k = c_k + 2t*(a_{k-i} - a_{k-j}) + t^2*(quadratic terms)
            # For small t, focus on linear term
            d_vec = np.zeros(2 * n - 1)
            # a_{k-i} for valid k
            for k in range(i, i + n):
                if k < 2 * n - 1:
                    d_vec[k] += a[k - i]
            for k in range(j, j + n):
                if k < 2 * n - 1:
                    d_vec[k] -= a[k - j]

            # Find optimal t (ignore quadratic for now)
            # We want to minimize max_k (c_k + 2*t*d_vec_k)
            # subject to 0 <= t <= a[i] (can't move more than a[i])
            # and t >= -a[j] (but t > 0 means we're adding to j, always ok)

            t_limit = a[i]

            # The upper envelope of lines c_k + 2*t*d_k has minimum where
            # two active lines cross. Find the two most relevant peaks.
            # Simple approach: evaluate at a few candidate t values
            best_t = 0.0
            best_local_max = float('inf')

            # Candidate t values: where top peaks become equal
            for k1 in top_idx[:8]:
                for k2 in top_idx[:8]:
                    if k1 == k2:
                        continue
                    dd = 2.0 * (d_vec[k1] - d_vec[k2])
                    if abs(dd) < 1e-15:
                        continue
                    t_cross = (c[k2] - c[k1]) / dd
                    if 0 < t_cross < t_limit:
                        # Evaluate max at t_cross
                        vals = c + 2.0 * t_cross * d_vec
                        lm = float(vals.max())
                        if lm < best_local_max:
                            best_local_max = lm
                            best_t = t_cross

            # Also check t = 0 and t = t_limit
            vals_0 = c
            lm_0 = float(vals_0.max())
            if lm_0 < best_local_max:
                best_local_max = lm_0
                best_t = 0.0

            vals_lim = c + 2.0 * t_limit * d_vec
            lm_lim = float(vals_lim.max())
            if lm_lim < best_local_max:
                best_local_max = lm_lim
                best_t = t_limit

            # Also try a few random t values
            for _ in range(5):
                t_rand = rng.uniform(0, t_limit)
                vals_r = c + 2.0 * t_rand * d_vec
                lm_r = float(vals_r.max())
                if lm_r < best_local_max:
                    best_local_max = lm_r
                    best_t = t_rand

            if best_t < 1e-10:
                continue

            # Verify with actual convolution (including quadratic terms)
            a_new = a.copy()
            a_new[i] -= best_t
            a_new[j] += best_t
            if a_new.min() < -1e-12:
                continue
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()

            c_new = _fft_convolve(a_new)
            sc_new = float(2.0 * n * c_new.max())

            if sc_new < best_score - 1e-10:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                c = c_new
                no_imp = 0
                found_move = True
                break  # restart with new a

        if not found_move:
            no_imp += 1
            if no_imp > 500:
                break

    return best_a


def _lp_refine(a, deadline, max_iter=10000, inflate=1.001, blend=0.2):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    # Precompute convolution matrix structure for efficiency
    # Using banded approach for large n
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        # Build the constraint matrix A (size (2n-1) x n)
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        # This is the "convolution with a" matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a

        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M_lp),
                      bounds=[(0, None)] * n, method='highs')

        if res.success:
            g = res.x
            sum_g = float(g.sum())
            if sum_g > 1.0 + 1e-6:
                g = g / sum_g
                a_new = (1.0 - blend_cur) * a + blend_cur * g
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    break
                a_new = a_new / s_new

                c_new = _fft_convolve(a_new)
                sc_new = float(2.0 * n * c_new.max())

                if sc_new < best_score - 1e-12:
                    best_score = sc_new
                    best_a = a_new.copy()
                    no_imp = 0
                else:
                    no_imp += 1
                    if no_imp % 20 == 0:
                        blend_cur = max(0.05, blend_cur * 0.95)
                        inflate_cur = min(1.005, inflate_cur * 1.03)
                    if no_imp > 200:
                        a = best_a.copy()
                        blend_cur = blend
                        inflate_cur = inflate
                        no_imp = 0
                        if no_imp > 500:
                            break
                a = a
            else:
                no_imp += 1
                if no_imp > 100:
                    break
        else:
            break

    return best_a


def _lp_refine_fast(a, deadline, max_iter=5000, inflate=1.0005, blend=0.15):
    """Faster LP refinement using sparse structure."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        # Only constraints near the peak matter
        c_sorted = np.argsort(c)[::-1]
        n_constraints = min(n, 2 * n - 1)

        A = np.zeros((n_constraints, n))
        for row in range(n_constraints):
            k = c_sorted[row]
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                A[row, j] = a[k - j]

        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(n_constraints, M_lp),
                      bounds=[(0, None)] * n, method='highs')

        if res.success:
            g = res.x
            sum_g = float(g.sum())
            if sum_g > 1.0 + 1e-6:
                g = g / sum_g
                a_new = (1.0 - blend) * a + blend * g
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()

                c_new = _fft_convolve(a_new)
                sc_new = float(2.0 * n * c_new.max())

                if sc_new < best_score - 1e-12:
                    best_score = sc_new
                    best_a = a_new.copy()
                    no_imp = 0
                else:
                    no_imp += 1
                    if no_imp > 100:
                        break
                a = a_new
            else:
                no_imp += 1
                if no_imp > 50:
                    break
        else:
            break

    return best_a
