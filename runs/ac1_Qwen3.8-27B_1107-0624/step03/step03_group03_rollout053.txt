"""
Optimized step function minimization using adaptive-constraint LP, 
peak-targeted coordinate updates, and multi-scale exploration.
Combines fast LP with constraint generation, LSE gradient, and 
direct peak deflation to push below 1.5027.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score with sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _lp_single(a, M, active_idx, budget_s=5.0):
    """Solve LP max sum(b) s.t. conv(a,b)[active_idx] <= M, b>=0.
    Uses constraint generation: start with subset, add violated constraints."""
    n = len(a)
    c_obj = -np.ones(n)
    deadline = time.monotonic() + budget_s

    active = list(active_idx)
    min_active = 20
    max_active = min(2 * n - 1, 300)

    for cg_iter in range(5):
        if time.monotonic() > deadline:
            break
        A = np.zeros((len(active), n))
        for row, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if lo < hi:
                A[row, lo:hi] = a[lo:(hi - (k - lo))]
                # Fix: A[row, j] = a[k - j] for j in [lo, hi)
                # So A[row, lo:hi] = a[k-lo : k-hi+1] reversed = a[k-hi+1 : k-lo+1][::-1]
                # Actually: j ranges from lo to hi-1, k-j ranges from k-lo to k-(hi-1) = k-hi+1
                # So A[row, j] = a[k-j], meaning A[row, lo:hi] = a[k-hi+1:k-lo+1][::-1]
                A[row, lo:hi] = a[k - (hi - 1): k - lo + 1][::-1]
        b_ub = np.full(len(active), M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            return None
        x = res.x

        # Check which non-active constraints are violated
        c_full = np.array([float(np.dot(a, x[i:(i + n - k) if k >= 0 else 0:n][k - (i + n - k) if k >= 0 else 0:n])) for k in range(2 * n - 1)]) if n <= 100 else None

        # Faster: compute conv(a, x) fully
        c_ax = _fft_convolve(a) * 0 + _fft_convolve(np.zeros(n))  # placeholder
        # Actually just compute conv(a, x) directly
        c_ax = np.convolve(a, x) if n <= 500 else _fft_convolve(np.concatenate([np.zeros(n), a]))[:2*n-1]
        # Use proper FFT for large n
        if n > 500:
            N = _next_pow2(2 * n - 1)
            fa = np.fft.rfft(a, N)
            fx = np.fft.rfft(x, N)
            c_ax = np.fft.irfft(fa * fx, N)[:2 * n - 1]
            if c_ax.min() < 0:
                c_ax -= c_ax.min()

        violated = [k for k in range(2 * n - 1) if k not in set(active) and c_ax[k] > M * 1.001]
        if not violated:
            return x
        # Add most violated constraints
        violated.sort(key=lambda k: c_ax[k], reverse=True)
        new_active = sorted(set(active) | set(violated[:min(50, len(violated))]))
        if len(new_active) > max_active:
            # Keep only top max_active by violation
            all_idx = sorted(set(active) | set(violated))
            all_idx.sort(key=lambda k: c_ax[k] if k in set(violated) else 0, reverse=True)
            new_active = all_idx[:max_active]
        active = new_active

    # Final solve with current active set
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            A[row, lo:hi] = a[k - (hi - 1): k - lo + 1][::-1]
    b_ub = np.full(len(active), M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_spread_fast(a, M, top_k=100):
    """LP to find b maximizing sum(b) s.t. conv(a,b) <= M, b>=0.
    Uses reduced constraint set for speed."""
    n = len(a)
    c_aa = _fft_convolve(a)
    # Get top constraints
    idx = np.argsort(c_aa)[::-1][:top_k]
    # Include neighbors
    active = set()
    for i in idx:
        for off in range(-3, 4):
            j = i + off
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)

    c_obj = -np.ones(n)
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            A[row, lo:hi] = a[k - (hi - 1): k - lo + 1][::-1]
    b_ub = np.full(len(active), M)

    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _golden_line_search(a, g, max_iter=30):
    """Golden section search for t in (0,1) minimizing score of (1-t)*a + t*g."""
    d = g - a
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = _score_norm(np.maximum(a + c_pt * d, 0))
    fd = _score_norm(np.maximum(a + d_pt * d, 0))
    for _ in range(max_iter):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = _score_norm(np.maximum(a + c_pt * d, 0))
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = _score_norm(np.maximum(a + d_pt * d, 0))
    return (lo + hi) / 2.0


def _lp_optimize(a, deadline, max_iter=5000):
    """Main LP-based optimization with adaptive M and line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = 1.0003
    blend_default = 0.3
    lp_budget = 3.0

    for it in range(max_iter):
        if time.monotonic() > deadline - 5:
            break
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        # Scale LP budget based on remaining time
        cur_budget = min(lp_budget, rem / 50)
        if cur_budget < 0.5:
            cur_budget = 0.5

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        # Get active constraints
        top_k = min(80, 2 * n - 1)
        idx_top = np.argsort(c)[::-1][:top_k]
        active = set()
        for i in idx_top:
            for off in range(-3, 4):
                j = i + off
                if 0 <= j < 2 * n - 1:
                    active.add(j)
        active = sorted(active)

        g = _lp_single(a, M_lp, active, budget_s=cur_budget)
        if g is None:
            inflate = min(1.01, inflate * 1.5)
            no_imp += 3
            if no_imp > 200:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(1.01, inflate * 1.1)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        # Line search

        if t_opt < 0.005:
            t_opt = blend_default * 0.3

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())

            best_a = a_new.copy()
            no_imp = 0
            # Decrease inflate on success

        else:

            if no_imp % 20 == 0:

            if no_imp > max(150, n):
                a = best_a.copy()
                inflate = 1.0003
                no_imp = 0

def _lse_adam_step(a, z, state, T, lr):
    """One Adam step in logit space using log-sum-exp smoothed max."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    # Gradient: 2 * (w * a) centered (cross-correlation)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1: 2 * n - 1]
    grad = 2.0 * h

    # Project to tangent space of simplex
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -50.0, 50.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a, deadline, max_iter=200000, T_start=0.03, T_end=1e-6, lr=0.08):
    """LSE gradient with Adam and geometric annealing."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0
    base_lr = lr

    if max_iter > 2:
        T_ratio = (T_end / T_start) ** (1.0 / (max_iter - 1))
    else:
        T_ratio = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline - 3:
            break
        batch = 50  # Do multiple steps between checks
        for _ in range(batch):
            if time.monotonic() > deadline - 3:
                break
            a, z, state, _ = _lse_adam_step(a, z, state, T, lr)

        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-15))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue

        score = _score_norm(a)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.005, lr * 0.9)
            if no_imp > max(2000, 5 * n):
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0

        T = max(T * T_ratio, T_end)

    return best_a


def _peak_deflate(a, deadline, max_iter=50000, lr=0.01):
    """Directly deflate peaks using subgradient at the actual max."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    momentum = 0.85
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline - 3:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        # Use top few peaks with weights
        threshold = m_val * 0.99
        active_peaks = c[c >= threshold]
        if len(active_peaks) == 0:
            active_peaks = np.array([m_val])
        w = np.zeros_like(c)
        peak_mask = c >= threshold
        w[peak_mask] = (c[peak_mask] - threshold + 1e-10)
        w = w / w.sum()

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1: 2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        score = _score_norm(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.001, lr * 0.92)
            if no_imp > max(3000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_init_profiles(n, rng):
    """Generate diverse initial profiles."""
    profiles = []
    # Flat
    profiles.append(np.ones(n) / n)
    # Raised cosine
    t = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * t))
    rc = np.maximum(rc, 1e-10)
    profiles.append(rc / rc.sum())
    # Skewed raised cosine (asymmetric)
    for skew in [0.3, 0.5, 0.7]:
        sc = 0.5 * (1 + np.cos(2 * np.pi * (t + skew * 0.3)))
        sc *= (1.0 + 0.5 * (t - 0.5) * skew * 2)
        sc = np.maximum(sc, 1e-10)
        profiles.append(sc / sc.sum())
    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    profiles.append(ramp / ramp.sum())
    # Reverse ramp
    profiles.append(ramp[::-1] / ramp.sum())
    # Gaussian centered
    gc = np.exp(-0.5 * ((t - 0.5) / 0.25) ** 2)
    gc = np.maximum(gc, 1e-10)
    profiles.append(gc / gc.sum())
    # Skewed Gaussian
    for mu in [0.35, 0.65]:
        sg = np.exp(-0.5 * ((t - mu) / 0.2) ** 2)
        sg = np.maximum(sg, 1e-10)
        profiles.append(sg / sg.sum())
    # Power law
    for alpha in [0.5, 2.0]:
        pl = t**alpha + 1e-10
        profiles.append(pl / pl.sum())
    # Random (Dirichlet)
    for conc in [0.5, 2.0, 10.0]:
        rv = rng.dirichlet(np.full(n, conc))
        profiles.append(rv)
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Try multiple initializations with LP optimization
    Phase 2: LSE gradient refinement on best candidates
    Phase 3: Peak deflation polish
    Phase 4: Final LP micro-optimization
    Tries n=512, 1024, 2048.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = [1024]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)
    for ln in [512, 2048]:
        if ln not in lengths:
            lengths.append(ln)

    # Phase 1: LP optimization on diverse initializations
    print("\n=== Phase 1: LP optimization on diverse inits ===")
    results = {}  # key: (n, init_idx) -> (score, seq)

    for n_t in lengths:
        if time.monotonic() > deadline - 200:
            break
        print(f"\n--- n = {n_t} ---")
        profiles = _make_init_profiles(n_t, rng)

        # If we have a parent, add it
        if parent is not None:
            profiles.insert(0, _interpolate_profile(parent, n_t))

        # Limit profiles for speed
        max_profiles = 6 if n_t >= 1024 else 4
        profiles = profiles[:max_profiles]

        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(profiles):
            if time.monotonic() > deadline - 150:
                break
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            print(f"  Init {ci}: score={sc0:.10f}")

            # LP optimization
            lp_budget = min(180.0, deadline - time.monotonic() - 150.0)
            if lp_budget < 30:
                break
            lp_deadline = time.monotonic() + lp_budget
            a_lp = _lp_optimize(a0, lp_deadline, max_iter=3000)
            sc_lp = _score(a_lp)

            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            if sc_lp < best_here_score:
                best_here_score = sc_lp
                best_here = a_lp
            results[(n_t, ci)] = (sc_lp, a_lp)
            print(f"  Init {ci} -> LP: score={sc_lp:.10f} (time used: {time.monotonic()-t_start:.0f}s)")

        if best_here is not None:
            results[(n_t, 'best')] = (best_here_score, best_here)

    # Phase 2: LSE gradient on top candidates
    print("\n=== Phase 2: LSE gradient refinement ===")
    # Sort candidates by score
    cands = [(k, v) for k, v in results.items() if k[1] == 'best']
    cands.sort(key=lambda x: x[1][0])

    for (n_t, _), (sc, a_c) in cands[:3]:
        if time.monotonic() > deadline - 100:
            break
        lse_deadline = time.monotonic() + min(120.0, deadline - time.monotonic() - 100)
        a_lse = _lse_optimize(a_c, lse_deadline, max_iter=100000,
                              T_start=0.02, T_end=1e-6, lr=0.06)
        sc_lse = _score(a_lse)
        if np.isfinite(sc_lse) and sc_lse < best_score:
            best_score = sc_lse
            best_seq = a_lse.copy()
        print(f"  n={n_t}: LSE score={sc_lse:.10f}")

    # Phase 3: Peak deflation on the best
    if best_seq is not None and time.monotonic() < deadline - 80:
        print("\n=== Phase 3: Peak deflation ===")
        pd_deadline = time.monotonic() + min(100.0, deadline - time.monotonic() - 60)
        a_pd = _peak_deflate(best_seq, pd_deadline, max_iter=50000, lr=0.008)
        sc_pd = _score(a_pd)
        if np.isfinite(sc_pd) and sc_pd < best_score:
            best_score = sc_pd
            best_seq = a_pd.copy()
        print(f"  Peak deflation: score={sc_pd:.10f}")

    # Phase 4: Final LP micro-optimization
    if best_seq is not None and time.monotonic() < deadline - 40:
        print("\n=== Phase 4: Final LP ===")
        lp_deadline = time.monotonic() + min(60.0, deadline - time.monotonic() - 20)
        a_final = _lp_optimize(best_seq, lp_deadline, max_iter=2000)
        sc_final = _score(a_final)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final LP: score={sc_final:.10f}")

    # Phase 5: One more LSE pass with very low temperature
    if best_seq is not None and time.monotonic() < deadline - 15:
        lse_deadline = time.monotonic() + min(40.0, deadline - time.monotonic() - 10)
        a_last = _lse_optimize(best_seq, lse_deadline, max_iter=50000,
                               T_start=0.005, T_end=1e-7, lr=0.03)
        sc_last = _score(a_last)
        if np.isfinite(sc_last) and sc_last < best_score:
            best_score = sc_last
            best_seq = a_last.copy()
        print(f"  Final LSE: score={sc_last:.10f}")

    # Final: try a quick coordinate-descent style polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_polish = _peak_deflate(best_seq, time.monotonic() + (deadline - time.monotonic() - 3),
                                 max_iter=20000, lr=0.003)
        sc_polish = _score(a_polish)
        if np.isfinite(sc_polish) and sc_polish < best_score:
            best_score = sc_polish
            best_seq = a_polish.copy()
        print(f"  Polish: score={sc_polish:.10f}")

    # Validation and return
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
        else:
            print(f"WARNING: evaluate returned non-finite: {sc_eval}")
            # Fallback to parent if available
            if parent is not None:
                final_seq = parent
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                best_score = sc_eval
                print(f"Fallback to parent: {sc_eval:.10f}")
    except Exception as e:
        print(f"Evaluation error: {e}")
        if parent is not None:
            final_seq = parent
            best_score = _score(parent)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
