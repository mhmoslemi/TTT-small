def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    return _normalize(p)

def _nesterov_lse_step(a, z, prev_z, t_momentum, T, lr):
    """One Nesterov-accelerated LSE gradient step in logit space."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, prev_z, t_momentum, m_val

    # Nesterov: compute momentum
    nesterov_z = z + t_momentum * (z - prev_z) if prev_z is not None else z
    a_nest = np.exp(nesterov_z - np.max(nesterov_z))
    a_nest = a_nest / a_nest.sum()

    # Recompute gradient at Nesterov point
    if not np.all(a_nest == a):
        c_n = _fft_convolve(a_nest)
        m_n = float(c_n.max())
        x_n = (c_n - m_n) / T
        x_n = np.clip(x_n, -50, 0)
        ex_n = np.exp(x_n)
        s_n = float(ex_n.sum())
        if s_n > 0 and np.isfinite(s_n):
            w_n = ex_n / s_n
            fw_n = np.fft.rfft(w_n, N)
            fa_n_rev = np.fft.rfft(a_nest[::-1], N)
            h_n_full = np.fft.irfft(fw_n * fa_n_rev, N)
            h_n = h_n_full[n - 1:2 * n - 1]
            grad_n = 2.0 * h_n
            dot_n = float(np.dot(a_nest, grad_n))
            gz_n = a_nest * (grad_n - dot_n)
            gz_n -= float(np.mean(gz_n))
            if np.all(np.isfinite(gz_n)):
                gz = gz_n
                m_val = m_n

    z_new = z - lr * gz
    z_new = np.clip(z_new, -40.0, 40.0)
    z_new -= float(np.mean(z_new))
    e = np.exp(z_new)
    a_new = e / e.sum()

    return a_new, z_new, z, t_momentum, m_val

def _lp_spread_full(a, M, n):
    """LP with full constraint set (for smaller n)."""
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs', options={'time_limit': 30})
    if res.success:
        return res.x
    return None


def _lp_spread_cutting_plane(a, M, max_constraints=200, max_rounds=5):
    """LP with cutting plane: start with important constraints, add violated ones."""
    n = len(a)
    c_obj = -np.ones(n)
    # Start with top constraints
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1][:min(max_constraints, 2 * n - 1)]
    active = sorted(idx)

    b_spread = None
    for round_i in range(max_rounds):
        A = np.zeros((len(active), n))
        b = np.full(len(active), M)
        for row, k in enumerate(active):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                A[row, j] = a[j]
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            return None
        b_spread = res.x

        # Check for violated constraints
        c_b = _fft_convolve(b_spread)
        # Find lags where conv(a, b_spread) might exceed M
        # Actually we need conv(a, b_spread), not conv(b_spread, b_spread)
        # Let me compute it properly
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        fb = np.fft.rfft(b_spread, N)
        conv_ab = np.fft.irfft(fa * fb, N)[:2 * n - 1]

        violated = np.where(conv_ab > M + 1e-8)[0]
        if len(violated) == 0:
            break
        # Add most violated constraints
        violated_sorted = violated[np.argsort(conv_ab[violated])[::-1]]
        for v in violated_sorted:
            if v not in active:
                active.append(v)
        active = sorted(active)
        if len(active) > 2 * max_constraints:

def _random_perturb_blend(a1, a2, rng, strength=0.3):

                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    target_n = 1024

    # Determine lengths to try (multi-resolution)
    small_n = 256
    med_n = 512
    if parent is not None:
        parent_n = parent.size
    else:
        parent_n = target_n

    # Phase 0: Multi-resolution initialization
    print("\n=== Phase 0: Multi-resolution initialization ===")
    t_phase = time.monotonic()

    # Solve at small_n first to get good shape
    small_candidates = [
        _make_gaussian(small_n, 0.3),
        _make_gaussian(small_n, 0.5),
        _make_raised_cosine(small_n),
        _make_flat_with_edges(small_n, 0.3),
        _make_flat_with_edges(small_n, 0.6),
        _make_bimodal(small_n, 0.1),
        _make_bimodal(small_n, 0.2),
        _make_trapezoid(small_n, 0.15),
        _make_trapezoid(small_n, 0.3),
        _make_spectral_flat(small_n, 15, seed=1),
        _make_spectral_flat(small_n, 30, seed=2),
    ]
    if parent is not None:
        small_candidates.append(_interpolate_profile(parent, small_n))
        small_candidates.append(_interpolate_profile(parent, small_n) * (1 + 0.05 * rng.normal(size=small_n)))
        small_candidates = [c for c in small_candidates if np.all(c >= 0)]

    best_small = None
    best_small_score = np.inf

    for ci, a0 in enumerate(small_candidates):
        if time.monotonic() > deadline - (budget_s * 0.85):
            break
        a0 = _normalize(a0)
        sc0 = _score(a0)

        # Quick Nesterov optimization at small scale
        budget_here = min(15.0, deadline - time.monotonic() - (budget_s * 0.85))
        if budget_here < 3:
            break
        a_opt = _nesterov_optimize(a0, time.monotonic() + budget_here,
                                    max_iter=20000, T_start=0.1, T_end=1e-4, lr=0.1)
        sc = _score(a_opt)
        if sc < best_small_score:
            best_small_score = sc
            best_small = a_opt.copy()
        print(f"  Small n={small_n}, cand {ci}: {sc0:.6f} -> {sc:.8f}")

    if best_small is not None:
        print(f"  Best small: {best_small_score:.10f}")

    # Upsample to target_n
    if best_small is not None:
        upsampled = _interpolate_profile(best_small, target_n)
        sc_up = _score(upsampled)
        if sc_up < best_score:
            best_score = sc_up
            best_seq = upsampled.copy()
        print(f"  Upsampled to {target_n}: {sc_up:.10f}")

    # Also prepare medium-scale candidates
    med_candidates = []
    if best_small is not None:
        med_candidates.append(_interpolate_profile(best_small, med_n))
    med_candidates.extend([
        _make_gaussian(med_n, 0.3),
        _make_raised_cosine(med_n),
        _make_flat_with_edges(med_n, 0.4),
        _make_bimodal(med_n, 0.15),
    ])
    if parent is not None:
        med_candidates.append(_interpolate_profile(parent, med_n))

    best_med = None
    best_med_score = np.inf
    for ci, a0 in enumerate(med_candidates[:4]):  # Limit to 4 for time
        if time.monotonic() > deadline - (budget_s * 0.82):
            break
        a0 = _normalize(a0)
        budget_here = min(20.0, deadline - time.monotonic() - (budget_s * 0.82))
        if budget_here < 3:
            break
        a_opt = _nesterov_optimize(a0, time.monotonic() + budget_here,
                                    max_iter=30000, T_start=0.08, T_end=1e-5, lr=0.08)
        sc = _score(a_opt)
        if sc < best_med_score:
            best_med_score = sc
            best_med = a_opt.copy()
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            # Upsample this too
            up = _interpolate_profile(a_opt, target_n)
            sc_up = _score(up)
            if sc_up < best_score:
                best_score = sc_up
                best_seq = up.copy()
        print(f"  Med n={med_n}, cand {ci}: {sc:.8f}")

    print(f"  Phase 0 done in {time.monotonic() - t_phase:.1f}s")

    # Phase 1: Nesterov LSE at target_n with multiple starts
    print("\n=== Phase 1: Nesterov LSE at target_n ===")
    t_phase = time.monotonic()

    # Build target-n candidates
    target_candidates = []
    if best_seq is not None:
        target_candidates.append(_interpolate_profile(best_seq, target_n) if len(best_seq) != target_n else best_seq.copy())
        # Perturbed versions
        for _ in range(3):
            pert = _interpolate_profile(best_seq, target_n) * (1 + 0.04 * rng.normal(size=target_n))
            target_candidates.append(_normalize(pert))
    if best_med is not None:
        target_candidates.append(_interpolate_profile(best_med, target_n))
    if parent is not None:
        target_candidates.append(_interpolate_profile(parent, target_n))
    target_candidates.extend([
        _make_gaussian(target_n, 0.35),
        _make_raised_cosine(target_n),
        _make_flat_with_edges(target_n, 0.35),
        _make_bimodal(target_n, 0.12),
        _make_trapezoid(target_n, 0.2),
        _make_spectral_flat(target_n, 25, seed=42),
        _make_spectral_flat(target_n, 50, seed=99),
    ])

    # Remove duplicates (roughly)
    unique_cands = [target_candidates[0]]
    for c in target_candidates[1:]:
        if all(np.sum(np.abs(c - uc)) > 0.1 for uc in unique_cands):
            unique_cands.append(c)
    target_candidates = unique_cands

    # Allocate time per candidate
    n_cands = len(target_candidates)
    time_per_cand = min(80.0, (deadline - time.monotonic() - budget_s * 0.35) / max(n_cands, 1))

    lse_results = []
    for ci, a0 in enumerate(target_candidates):
        if time.monotonic() > deadline - budget_s * 0.35:
            break

        a0 = _normalize(a0)
        sc0 = _score(a0)
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        budget_here = min(time_per_cand, deadline - time.monotonic() - budget_s * 0.35)
        if budget_here < 5:
            break

        a_opt = _nesterov_optimize(a0, time.monotonic() + budget_here,
                                    max_iter=40000, T_start=0.06, T_end=1e-5, lr=0.07)
        sc = _score(a_opt)
        lse_results.append((sc, a_opt))

        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
        print(f"  LSE cand {ci}/ {n_cands}: init={sc0:.8f} -> {sc:.10f} (budget={budget_here:.0f}s)")

    # Sort by score
    lse_results.sort(key=lambda x: x[0])
    if lse_results:
        print(f"  Best LSE: {lse_results[0][0]:.10f}")
    print(f"  Phase 1 done in {time.monotonic() - t_phase:.1f}s")

    # Phase 2: LP refinement on top candidates
    print("\n=== Phase 2: LP refinement ===")
    t_phase = time.monotonic()

    # Take top 3 diverse candidates
    top_cands = []
    for sc, a in lse_results:
        if not top_cands or all(np.sum(np.abs(a - ta)) > 0.05 for _, ta in top_cands):
            top_cands.append((sc, a))
        if len(top_cands) >= 3:
            break
    # Also include current best
    if best_seq is not None:
        sc_best = _score(best_seq)
        if not any(np.sum(np.abs(best_seq - ta)) < 0.05 for _, ta in top_cands):
            top_cands.append((sc_best, best_seq.copy()))

    for ci, (sc_in, a_in) in enumerate(top_cands[:3]):
        if time.monotonic() > deadline - budget_s * 0.25:
            break

        a_in = _normalize(a_in)
        budget_here = min(120.0, deadline - time.monotonic() - budget_s * 0.25)
        if budget_here < 10:
            break

        a_lp = _lp_refine_adaptive(a_in, time.monotonic() + budget_here,
                                    max_iter=2000, inflate=1.0008, blend=0.12)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        print(f"  LP cand {ci}: {sc_in:.10f} -> {sc_lp:.10f} (budget={budget_here:.0f}s)")

    print(f"  Phase 2 done in {time.monotonic() - t_phase:.1f}s")

    # Phase 3: Peak coordinate descent
    print("\n=== Phase 3: Peak coordinate descent ===")
    t_phase = time.monotonic()
    if best_seq is not None:
        cd_budget = min(150.0, deadline - time.monotonic() - 30.0)
        if cd_budget > 10:
            a_cd = _peak_coordinate_descent(best_seq, time.monotonic() + cd_budget,
                                             max_iter=50000, line_search_iters=8)
            sc_cd = _score(a_cd)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            print(f"  Coord descent: {sc_cd:.10f} (budget={cd_budget:.0f}s)")
    print(f"  Phase 3 done in {time.monotonic() - t_phase:.1f}s")

    # Phase 4: Short LP polish on best
    print("\n=== Phase 4: Final LP polish ===")
    t_phase = time.monotonic()
    if best_seq is not None:
        lp_budget = min(80.0, deadline - time.monotonic() - 15.0)
        if lp_budget > 10:
            a_final = _lp_refine_adaptive(best_seq, time.monotonic() + lp_budget,
                                           max_iter=1500, inflate=1.0004, blend=0.08)
            sc_final = _score(a_final)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LP: {sc_final:.10f}")
    print(f"  Phase 4 done in {time.monotonic() - t_phase:.1f}s")

    # Phase 5: One more Nesterov pass on the very best
    if best_seq is not None:
        lse_budget = min(40.0, deadline - time.monotonic() - 10.0)
        if lse_budget > 5:
            a_last = _nesterov_optimize(best_seq, time.monotonic() + lse_budget,
                                         max_iter=20000, T_start=0.03, T_end=1e-6, lr=0.04)
            sc_last = _score(a_last)
            if sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LSE: {sc_last:.10f}")

    # Final validation
    print(f"\n  Internal best: {best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    if best_seq is None:
        best_seq = np.ones(target_n) / target_n
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(target_n) / target_n

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"  evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"  Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, took {time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final_seq]
