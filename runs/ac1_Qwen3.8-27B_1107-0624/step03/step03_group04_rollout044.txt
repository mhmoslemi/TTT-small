    """Alternating LP: minimize max_k (a*b)_k over b >= 0, sum(b)=1."""
    n = len(fixed_a)
    # Convolution matrix: (a*b)_k = sum_j a_j * b_{k-j}
    # For k = 0, ..., 2n-2:
    # (a*b)_k = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} a_j * b_{k-j}
    
    # Build constraint matrix A where A[k, i] = a_{k-i} if 0 <= k-i < n, else 0
    # Then (a*b)_k = (A @ b)_k
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = fixed_a[j]
    
    # LP: min t s.t. A@b <= t, sum(b) = 1, b >= 0
    # Variables: [b_0, ..., b_{n-1}, t]
    # Objective: [0, ..., 0, 1]
    # Inequalities: A@b - t <= 0  =>  [A, -I] @ [b; t] <= 0
    # Equality: sum(b) = 1  =>  [1, ..., 1, 0] @ [b; t] = 1
    # Bounds: b_i >= 0, t >= 0 (or t free?)
    
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.hstack([A, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    return res.x[:n] if res.success else None

"""
Alternating LP + gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Main approach: Alternating linear programming where we fix one sequence and
optimize the other to minimize the peak of their cross-convolution. This finds
fixed points where both sequences are mutual optimal, which correspond to
local minima of the auto-convolution peak. Combined with gradient polishing
and multi-start strategies.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        c = np.convolve(a, a)
    else:
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        mn = float(c.min())
        if mn < 0.0:
            c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(fixed_seq):
    n = len(fixed_seq)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # (fixed * x)_k = sum_i fixed[k-i] * x[i] for valid i
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A[k, i] = fixed_seq[k - i]
    return A


def _alt_lp_solve(fixed_seq):
    """Solve: min t s.t. (fixed_seq * b)_k <= t for all k, sum(b)=1, b>=0."""
    n = len(fixed_seq)
    A = _make_conv_matrix(fixed_seq)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # A @ b - t <= 0 for each row
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    # sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        b_opt = np.maximum(res.x[:n], 0.0)
        s = b_opt.sum()
        if s > 0:
            b_opt = b_opt / s
        return b_opt, float(res.x[-1])
    return None, None


def _alt_lp_optimize(a_init, deadline, max_iter=5000):
    """Alternating LP optimization."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    
    print(f"  AltLP start: score={best_score:.10f}, n={n}", flush=True)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Step 1: Fix a, find b that minimizes max(a*b)
        b, t_b = _alt_lp_solve(a)
        if b is None:
            break

        a_new, t_a = _alt_lp_solve(b)

            no_imp = 0

            no_imp = 0

def _project_simplex(v):
    """Project v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_gradient(a, beta):
    """Compute smooth max value and gradient w.r.t. a (with sum normalization)."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros(n)
    w = ex / s
    
    # Gradient of sum_k w_k * (a*a)_k w.r.t. a_i
    # = sum_k w_k * (d(a*a)_k / da_i)
    # = sum_k w_k * 2*a_{k-i} (with appropriate bounds)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    
    return mval, grad


def _gradient_polish(a, deadline, max_iter=20000, lr_init=0.01):
    """Gradient polish using smooth max in logit space."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-10, 0.01 * c_max)
    
    z = np.log(a)
    z -= np.mean(z)
    
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    lr = lr_init
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        mval, grad = _smooth_max_gradient(a, beta)
        
        # Project gradient: direction on the tangent space of the simplex
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= np.mean(gz)
        
        if not np.all(np.isfinite(gz)):
            break
        
        z = z - lr * gz
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.9)
        
        # Slowly decrease beta
        if it % 200 == 0:
            beta = max(1e-14, beta * 0.7)
        
        if no_imp > max(5000, 10 * n):
            break
    
    return best_a
