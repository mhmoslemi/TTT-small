    # 
    # Hmm, actually in the sum sum_i p[i]*p[k-i], each term has two factors of p.
    # d/dp_j of p[i]*p[k-i] is:
    #   p[k-i] if i = j
    #   p[i] if k-i = j (i.e., i = k-j)
    #   0 otherwise
    # But if i = j AND k-i = j (i.e., k = 2j), then both conditions hold, giving 2*p[j].
    # 
    # So: d(p*p)[k]/dp_j = p[k-j]*1_{0<=k-j<n} + p[k-j]*1_{0<=k-j<n} = 2*p[k-j]*1_{0<=k-j<n}
    # Wait that's not right either. Let me just use the fact that:
    # d/dp_j sum_k w_k (p*p)[k] = 2 * (w * p)[j] ... no
    # 
    # OK let me just use the cross-convolution approach:
    # grad = 2 * convolve(w, p) ... but with the right indexing.
    
    # Actually the simplest way: 
    # sum_k w_k (p*p)[k] = sum_k w_k sum_i p[i] p[k-i]
    # = sum_i p[i] sum_k w_k p[k-i]
    # = sum_i p[i] (w * p_flip)[i]  ... hmm this is getting circular.
    
    # Let me just use the FFT to compute the gradient directly.
    # The gradient of sum_k w_k (p*p)[k] w.r.t. p is:
    # g[j] = sum_k w_k * d(p*p)[k]/dp[j]
    # 
    # Since (p*p)[k] = sum_i p[i]*p[k-i], by the chain rule:
    # g[j] = sum_k w_k * [p[k-j] + p[j-k+n... no]]
    # 
    # I think the cleanest way is:
    # Let h[k] = (p*p)[k]. Then sum_k w_k h[k] = sum_k w_k sum_i p[i] p[k-i]
    # = sum_i p[i] sum_k w_k p[k-i]
    # 
    # Let v[i] = sum_k w_k p[k-i] = (w * p)[i] where (w*p) is the cross-correlation (not convolution).
    # Actually v[i] = sum_k w[k] * p[k-i] which is the cross-correlation of w and p, evaluated at lag i.
    # 
    # So the gradient w.r.t. p[i] is v[i] = (w ⋆ p)[i] where ⋆ denotes cross-correlation.
    # 
    # Using FFT: cross-correlation of w and p = ifft(conj(fft(w)) * fft(p))
    # But we need to be careful with the definition.
    
    # Let me use a more direct approach:
    # g[j] = 2 * sum_k w[k] * p[k-j] (with appropriate bounds)
    # This is essentially 2 * (the cross-correlation of w and p)
    
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p, N)
    # cross-correlation: (w ⋆ p)[j] = sum_k w[k+j] * p[k] = ifft(conj(fft(w)) * fft(p))[j]
    # ... this is getting complicated with indexing. Let me just use the simpler formula.
    
    # Actually, for the objective L = sum_k w_k (p*p)[k]:
    # dL/dp[j] = sum_k w_k * d(p*p)[k]/dp[j]
    # 
    # (p*p)[k] = sum_i p[i] p[k-i] (with valid indices)
    # d/dp[j] = p[k-j] (from the i=j term) + p[k-j] (from the k-i=j term, if different from i=j)
    # 
    # Wait, I think I'm overcomplicating this. Let me use the fact that for full convolution:
    # (p*p)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} p[i] p[k-i]
    # 
    # d(p*p)[k] / dp[j] = p[k-j] * 1_{j in [max(0,k-n+1), min(k,n-1)]}
    #                   + p[k-j] * 1_{k-j in [max(0,k-n+1), min(k,n-1)]}
    # 
    # The first condition is j >= max(0,k-n+1) and j <= min(k,n-1), which means 0<=j<n and 0<=k-j<=n-1.
    # The second condition is k-j >= max(0,k-n+1) and k-j <= min(k,n-1), which means 0<=k-j<n and 0<=j<=n-1.
    # Both conditions are equivalent: 0<=j<n and 0<=k-j<n.
    # 
    # But when these are the same (i.e., j = k-j, meaning k = 2j), we're counting the same term twice.
    # In the original sum, when i=j=k/2, the term is p[j]*p[j] = p[j]^2.
    # d/dp[j] of p[j]^2 = 2*p[j].
    # In our formula, we get p[k-j] + p[k-j] = 2*p[j] (when k=2j). ✓
    # 
    # So: d(p*p)[k]/dp[j] = 2*p[k-j] if 0<=k-j<n, else 0.
    # 
    # Therefore: g[j] = sum_k w[k] * 2*p[k-j] * 1_{0<=k-j<n}
    #                 = 2 * sum_{k: 0<=k-j<n} w[k]*p[k-j]
    #                 = 2 * sum_{m: 0<=m<n} w[m+j]*p[m]
    #                 = 2 * (w_shifted ⋆ p)[j]
    
    # This is 2 times the cross-correlation of shifted w and p.
    # In FFT terms: ifft(fft(w) * conj(fft(p))) gives correlation... 
    # 
    # Actually, let me just compute it directly using the convolution structure.
    # g[j] = 2 * sum_m w[m+j] * p[m] for m = 0..n-1, where w is defined on 0..2n-2.
    # 
    # This is equivalent to: take w, flip it, convolve with p, and extract the right part.
    # 
    # w_flipped[m] = w[2n-2-m]
    # (w_flipped * p)[j] = sum_m w_flipped[m] * p[j-m] = sum_m w[2n-2-m] * p[j-m]
    # 
    # Hmm, this is getting messy. Let me just use the approach from the previous program:
    # The gradient of sum_k w_k (p*p)[k] w.r.t. p is:
    # g = 2 * convolve(w, p reversed, 'valid')... 
    # 
    # Actually, the previous program uses _conv_cross_corr(w, p) which computes:
    # h = ifft(fft(w) * fft(p[::-1]))[n-1:2n-1]
    # This should give the gradient (up to a factor of 2).
    
    # Let me just use the same approach as before:
    h = _cross_convolve_correl(w, p)
    return 2.0 * h

def _cross_convolve_correl(w, p):
    """
    Compute the gradient vector g[j] = sum_k w[k]*p[k-j] (0<=k-j<n)
    which is what we need for the gradient of weighted convolution.
    """
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _normalize(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 1e-15, 1000.0)
    s = float(p.sum())
    if s <= 1e-30:
        return None
    return p / s

def _lp_solve(a, M):
    """Solve: max sum(b) s.t. a*b <= M (element-wise), b >= 0"""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _line_search_conv(a, b, c_aa, c_ab, c_bb, t_range=(0.0, 1.0), n_iter=30):
    """
    Golden section search for t that minimizes max_k ((1-t)^2 c_aa[k] + 2t(1-t) c_ab[k] + t^2 c_bb[k])
    """
    lo, hi = t_range
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    def f(t):
        c = (1.0 - t) ** 2 * c_aa + 2.0 * t * (1.0 - t) * c_ab + t * t * c_bb
        return float(c.max())
    
    d = hi - lo
    x1 = hi - gr * d
    x2 = lo + gr * d
    f1 = f(x1)
    f2 = f(x2)
    
    best_t = x1 if f1 < f2 else x2
    best_f = min(f1, f2)
    
    for _ in range(n_iter):
        if f1 < f2:
            hi = x2
            x2 = x1
            x1 = hi - gr * (hi - lo)
            f2 = f1
            f1 = f(x1)
        else:
            lo = x1
            x1 = x2
            x2 = lo + gr * (hi - lo)
            f1 = f2
            f2 = f(x2)
        cur_best_t = x1 if f1 < f2 else x2
        cur_best_f = min(f1, f2)
        if cur_best_f < best_f:
            best_f = cur_best_f
            best_t = cur_best_t
    
    return best_t, best_f

def _make_profiles(n, rng):
    """Generate diverse initial profiles"""
    profiles = []
    
    # Uniform
    profiles.append(('uniform', np.ones(n) / n))
    
    # Linear
    t = np.linspace(0, 1, n)
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + eta * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append((f'lin_{eta}', a / a.sum()))
    
    # Cosine
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
            a = np.maximum(a, 1e-12)
            profiles.append((f'cos_{freq}_{amp}', a / a.sum()))
    
    # Two bump
    for c1 in [0.2, 0.3, 0.4]:
        c2 = 1.0 - c1
        for s in [0.1, 0.15, 0.2, 0.3]:
            a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
            a = np.maximum(a, 1e-12)
            profiles.append((f'bump_{c1}_{s}', a / a.sum()))
    
    # Exponential (one-sided)
    for lam in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = np.exp(-lam * t)
        a = np.maximum(a, 1e-12)
        profiles.append((f'exp_{lam}', a / a.sum()))
    
    # U-shape
    for c in [1.0, 2.0, 5.0, 10.0]:
        a = c * (t - 0.5) ** 2 + 1.0
        a = np.maximum(a, 1e-12)
        profiles.append((f'u_{c}', a / a.sum()))
    
    # Random smooth
    for _ in range(5):
        a = 1.0 + 0.5 * rng.normal(size=n)
        # Smooth it
        kernel = np.exp(-0.5 * (np.arange(-10, 11) / 3.0) ** 2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-12)
        profiles.append(('rand', a / a.sum()))
    
    return profiles

def _adam_smoothmax(p, deadline, rng, lr=0.05, K=30, beta_init=None, beta_decay=0.95,
                    no_imp_limit=200, use_line_search=False):
    """
    Adam optimization in log-space with top-K focused smooth-max.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.maximum(p, 1e-12)
    p = p / p.sum()
    n = p.size
    
    z = np.log(p)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    c = _full_convolve(p)
    maxc = float(c.max())
    if beta_init is None:
        beta = max(1e-3, 0.1 * maxc)
    
    best_p = p.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    K_cur = K
    
    while time.monotonic() < deadline:
        w = _topk_weights(c, K=K_cur, beta=beta)
        g = _cross_convolve_correl(w, p)
        
        # Project gradient to log-space
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p_new = _softmax_from_z(z)
        c_new = _full_convolve(p_new)
        maxc_new = float(c_new.max())
        score = float(2.0 * n * maxc_new)
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p_new.copy()
                best_z = z.copy()
                p = p_new
                c = c_new
                maxc = maxc_new
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        # Adapt beta and K
        if t_adam % 50 == 0:
            beta = max(1e-4, beta * beta_decay)
            # Increase K as we converge (focus on fewer peaks initially)
            if best_score < 1.52:
                K_cur = max(5, K_cur // 2)
            elif t_adam > 1000 and K_cur > 10:
                K_cur = max(10, K_cur - 2)
        
        # Restart
        limit = max(50, no_imp_limit)
        if no_imp > limit:
            ns = 0.05 if best_score < 1.51 else (0.1 if best_score < 1.53 else 0.2)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, 0.1 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
            K_cur = K
    
    return best_p, best_score

def _lp_refine_with_linesearch(a, deadline, max_solves=30):
    """LP refinement with exact line search along the LP direction."""
    n = len(a)
    current_best = a.copy()
    current_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        
        # Compute cross-convolutions for line search
        c_aa = _full_convolve(a)
        c_bb = _full_convolve(b_norm)
        c_ab = _cross_convolve(a, b_norm)
        # Note: ( (1-t)a + t*b ) * ( (1-t)a + t*b )
        # = (1-t)^2 (a*a) + 2t(1-t)(a*b) + t^2 (b*b)
        # where a*b means the cross-convolution sum_i a[i]*b[k-i]
        
        # Line search
        t_opt, f_opt = _line_search_conv(a, b_norm, c_aa, c_ab, c_bb,
                                          t_range=(0.0, 0.8), n_iter=25)
        
        a_new = (1.0 - t_opt) * a + t_opt * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc_new = _score(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
            a = a_new
            c = _full_convolve(a)
            M = float(c.max())
        else:
            # Try smaller step
            a_try = (1.0 - 0.1) * a + 0.1 * b_norm
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < current_score:
                current_score = sc_try
                current_best = a_try.copy()
                a = a_try
                c = _full_convolve(a)
                M = float(c.max())
            else:
                # Try the convex combination with even smaller step
                for t_try in [0.05, 0.02, 0.01]:
                    a_try = (1.0 - t_try) * a + t_try * b_norm
                    a_try = np.maximum(a_try, 1e-15)
                    a_try = a_try / a_try.sum()
                    sc_try = _score(a_try)
                    if sc_try < current_score:
                        current_score = sc_try
                        current_best = a_try.copy()
                        a = a_try
                        c = _full_convolve(a)
                        M = float(c.max())
                        break
    
    return current_best, current_score

def _alternating_lp_gradient(a, deadline, max_iters=50):
    """
    Alternate between LP direction and gradient direction for faster convergence.
    """
    n = len(a)
    current_best = a.copy()
    current_score = _score(a)
    
    c = _full_convolve(a)
    M = float(c.max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # LP step
        b = _lp_solve(a, M)
        if b is not None:
            s_b = float(b.sum())
            if s_b > 1e-10:
                b_norm = b / s_b
                c_aa = _full_convolve(a)
                c_bb = _full_convolve(b_norm)
                c_ab = _cross_convolve(a, b_norm)
                t_opt, _ = _line_search_conv(a, b_norm, c_aa, c_ab, c_bb,
                                              t_range=(0.0, 0.5), n_iter=20)
                a_new = (1.0 - t_opt) * a + t_opt * b_norm
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                sc_new = _score(a_new)
                if sc_new < current_score:
                    current_score = sc_new
                    current_best = a_new.copy()
                    a = a_new
                    c = _full_convolve(a)
                    M = float(c.max())
        
        # Gradient step (top-K)
        w = _topk_weights(c, K=20, beta=0.05 * M)
        g = _cross_convolve_correl(w, a)
        # Simple descent step
        eta = 0.01
        a_new = a - eta * g
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
            a = a_new
            c = _full_convolve(a)
            M = float(c.max())
    
    return current_best, current_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  *** BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")
    
    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")
    
    # === Phase 1: Initialize and score candidates (first 60s) ===
    phase1_end = t_start + 60.0
    candidates = []
    
    ns_to_try = [512, 1024]
    if pn > 0 and pn <= 2048:
        ns_to_try.append(pn)
    
    for n in ns_to_try:
        profs = _make_profiles(n, rng)
        for label, p in profs:
            if time.monotonic() > phase1_end:
                break
            sc = _score(p)
            candidates.append((sc, p, n, label))
    
    # Add parent at various scales
    if parent is not None:
        for n in [512, 1024, 2048]:
            if pn != n:
                x0 = np.linspace(0, 1, pn)
                x1 = np.linspace(0, 1, n)
                p_int = np.interp(x1, x0, parent)
                p_int = np.maximum(p_int, 1e-12)
                p_int = p_int / p_int.sum()
                sc = _score(p_int)
                candidates.append((sc, p_int, n, 'parent_interp'))
        if 0 < pn <= 4096:
            sc = _score(parent.copy())
            candidates.append((sc, parent.copy(), pn, 'parent'))
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')
    
    # === Phase 2: Adam polish top candidates (60s to 600s) ===
    phase2_end = t_start + 600.0
    count = 0
    for sc, p, n, label in candidates:
