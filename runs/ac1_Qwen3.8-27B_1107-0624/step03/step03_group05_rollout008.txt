    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_mixture(n, centers, sigmas, weights):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c, s, w in zip(centers, sigmas, weights):
        a += w * np.exp(-0.5 * ((t - c) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_optimize(a, deadline, max_iter=50000, inflate_init=1.001, blend_init=0.4):
    """Aggressive LP-based optimization with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    no_imp = 0
    blend_cur = blend_init
    inflate_cur = inflate_init
    base_blend = blend_init
    base_inflate = inflate_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        g = g / sum_g

        # Compute similarity to current a
        sim = float(np.dot(a, g))
        # If very similar, reduce inflate or break
        if sim > 1.0 - 1e-10:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.03)
            if no_imp > 500:
                break
            continue

        # Adaptive blend: larger when improvement is expected
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            # Tighten on success
            inflate_cur = max(1.0003, inflate_cur * 0.95)
            blend_cur = min(0.6, blend_cur * 1.02)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.015, inflate_cur * 0.98)
            if no_imp > max(300, 2 * n):
                a = best_a.copy()
                blend_cur = base_blend
                inflate_cur = base_inflate
                no_imp = 0

        a = a_new

    return best_a


def _spectral_flatten(a, deadline, max_iter=2000, n_fft=None):
    """
    Gerchberg-Saxton style: alternate between enforcing flat power spectrum
    and non-negativity. This provides a novel escape from local minima.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    if n_fft is None:
        n_fft = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0

    # Target: flat magnitude spectrum
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Forward transform
        A = np.fft.rfft(a, n_fft)

        # Enforce flat magnitude (keep phase)
        mag_target = 1.0 / np.sqrt(len(A))
        phase = A / np.abs(A + 1e-30 * (1 + 1j * 0))
        # Handle zeros
        phase[np.abs(A) < 1e-15] = 1.0 + 0j
        A_flat = mag_target * phase

        # Inverse transform and take real part
        a_new = np.fft.irfft(A_flat, n_fft)[:n]

        # Enforce non-negativity (soft clipping to allow smoother transitions)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = np.maximum(a_new + 0.1, 0.0)
            s_new = a_new.sum()
        a_new = a_new / s_new

        # Blend to avoid oscillation
        a = 0.7 * a + 0.3 * a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 150:
            break

    return best_a


def _spectral_flatten_aggressive(a, deadline, max_iter=3000):
    """More aggressive spectral flattening with larger steps."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    n_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    alpha = 0.5  # Blend factor

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, n_fft)
        mag_target = 1.0 / np.sqrt(len(A))

        # Smooth transition: partially flatten
        mag_current = np.abs(A)
        mag_target_smooth = mag_current * (1.0 - alpha) + mag_target * alpha

        phase = np.zeros_like(A)
        mask = mag_current > 1e-15
        phase[mask] = A[mask] / mag_current[mask]
        phase[~mask] = 1.0 + 0j

        A_new = mag_target_smooth * phase
        a_flat = np.fft.irfft(A_new, n_fft)[:n]
        a_flat = np.maximum(a_flat, 0.0)
        s_flat = a_flat.sum()
        if s_flat < 1e-10:
            a_flat = np.full(n, 1.0 / n)
        a_flat = a_flat / s_flat

        a = a_flat

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                alpha = max(0.1, alpha * 0.9)

        if no_imp > 200:
            break

    return best_a


def _gradient_polish(a, deadline, max_iter=20000, beta_init_scale=0.03):
    """Smooth-max gradient with momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, beta_init_scale * float(c.max()))
    lr = 0.02
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.003, lr * 0.95)
            if no_imp > max(1000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, beta_init_scale * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 80 == 0:
            beta = max(1e-10, beta * 0.75)

    return best_a


def _adam_polish(a, deadline, max_iter=20000, lr=0.04):
    """Adam optimizer in log-space."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.03 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.95)

        if it % 120 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > max(1500, 3 * n):
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.8 * best_z + 0.2 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.03 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _random_perturb_escape(a, best_score, deadline, rng, n_perturbs=10):
    """Try random perturbations to escape local minima, returning best."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)

    best_a = a.copy()
    best_sc = best_score

    for _ in range(n_perturbs):
        if time.monotonic() > deadline:
            break
        # Random multiplicative noise
        noise = np.exp(rng.normal(0, 0.3, n))
        a_pert = a * noise
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s
        sc = _score(a_pert)
        if sc < best_sc:
            best_sc = sc
            best_a = a_pert.copy()

    return best_a


def _full_optimize(a_init, n, deadline, budget):
    """Full optimization pipeline for a single candidate."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_u_profile(n, 2.0)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    t_start = time.monotonic()

    # Phase 1: LP (main workhorse) - give it the most time
    lp_budget = budget * 0.35
    lp_end = t_start + lp_budget
    if lp_end > time.monotonic() + 5:
        a_lp = _lp_optimize(a, lp_end, max_iter=30000, inflate_init=1.001, blend_init=0.4)
        sc = _score(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
        print(f"      LP: {sc:.10f}")

    # Phase 2: Spectral flattening (novel)
    spec_budget = budget * 0.15
    spec_end = time.monotonic() + spec_budget
    if spec_end > time.monotonic() + 3:
        a_spec = _spectral_flatten(best_a, spec_end, max_iter=2000)
        sc = _score(a_spec)
        if sc < best_score:
            best_score = sc
            best_a = a_spec.copy()
        print(f"      Spectral: {sc:.10f}")

        # Post-spectral LP
        a_spec_lp = _lp_optimize(best_a, time.monotonic() + min(30, spec_budget * 0.5),
                                 max_iter=10000, inflate_init=1.0008, blend_init=0.3)
        sc = _score(a_spec_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_spec_lp.copy()
        print(f"      Post-Spectral LP: {sc:.10f}")

    # Phase 3: Further spectral (aggressive)
    if time.monotonic() < deadline - 60:
        spec2_budget = min(budget * 0.1, deadline - time.monotonic() - 60)
        if spec2_budget > 3:
            a_spec2 = _spectral_flatten_aggressive(best_a, time.monotonic() + spec2_budget, max_iter=2000)
            sc = _score(a_spec2)
            if sc < best_score:
                best_score = sc
                best_a = a_spec2.copy()
            print(f"      Spectral2: {sc:.10f}")

    # Phase 4: Gradient polish
    gp_budget = budget * 0.15
    gp_end = time.monotonic() + gp_budget
    if gp_end > time.monotonic() + 3:
        a_gp = _gradient_polish(best_a, gp_end, max_iter=15000)
        sc = _score(a_gp)
        if sc < best_score:
            best_score = sc
            best_a = a_gp.copy()
        print(f"      Gradient: {sc:.10f}")

    # Phase 5: Adam polish
    ad_budget = budget * 0.15
    ad_end = time.monotonic() + ad_budget
    if ad_end > time.monotonic() + 3:
        a_ad = _adam_polish(best_a, ad_end, max_iter=15000, lr=0.03)
        sc = _score(a_ad)
        if sc < best_score:
            best_score = sc
            best_a = a_ad.copy()
        print(f"      Adam: {sc:.10f}")

    # Phase 6: Final LP
    if time.monotonic() < deadline - 30:
        final_lp_budget = min(budget * 0.1, deadline - time.monotonic() - 25)
        if final_lp_budget > 3:
            a_final_lp = _lp_optimize(best_a, time.monotonic() + final_lp_budget,
                                       max_iter=10000, inflate_init=1.0005, blend_init=0.25)
            sc = _score(a_final_lp)
            if sc < best_score:
                best_score = sc
                best_a = a_final_lp.copy()
            print(f"      Final LP: {sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: aggressive LP + spectral flattening + 
    gradient polish, with smart multi-length search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine target lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    # Try a range of lengths
    for ln in [512, 640, 768, 896, 1024, 1280]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = sorted(set(lengths))

    print(f"Target lengths: {lengths}")
    print(f"Budget: {budget_s}s")

    # Phase 1: Quick screening - evaluate initializations for all lengths
    # and pick the most promising 2-3 lengths to invest in
    length_scores = {}
    for n_t in lengths:
        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
        else:
            a_init = _make_u_profile(n_t, 2.0)
        sc = _score(a_init)
        length_scores[n_t] = sc
        if sc < best_score:
            best_score = sc
            best_seq = a_init.copy()
    print(f"Initial scores: {[(n, f'{s:.6f}') for n, s in length_scores.items()]}")

    # Phase 2: Full optimization on top lengths
    # Sort by score, try best 3 first
    sorted_lengths = sorted(lengths, key=lambda x: length_scores[x])
    top_lengths = sorted_lengths[:3]

    # Allocate time: invest more in the best length
    total_budget = deadline - time.monotonic() - 100.0
    # 60% to best, 25% to second, 15% to third
    allocs = [0.55, 0.25, 0.20]
    while len(allocs) < len(top_lengths):
        allocs.append(0.0)

    for idx, n_t in enumerate(top_lengths):
        if time.monotonic() > deadline - 80:
            break

        budget = abs(total_budget * allocs[idx])
        budget = max(30.0, min(budget, deadline - time.monotonic() - 60.0))
        if budget < 20:
            continue

        print(f"\n=== Length {n_t}: budget={budget:.0f}s ===")

        # Multiple initializations for this length
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
        inits.append(_make_u_profile(n_t, 1.5))
        inits.append(_make_u_profile(n_t, 3.0))
        inits.append(_make_u_profile(n_t, 5.0))
        inits.append(_make_linear_profile(n_t, 1.0))
        inits.append(_make_gaussian_mixture(n_t, [0.1, 0.9], [0.2, 0.2], [1.0, 1.0]))
        inits.append(_make_gaussian_mixture(n_t, [0.05, 0.95], [0.1, 0.1], [1.0, 1.0]))

        # Random initializations
        for _ in range(3):
            a_rand = np.maximum(rng.normal(1.0, 0.4, n_t), 0.01)
            inits.append(a_rand / a_rand.sum())

        # Sub-divide budget among initializations
        # Give more to the parent-interpolated one
        n_inits = len(inits)
        # First 2 get 40% each, rest split remaining
        init_budgets = []
        remaining = budget
        for i in range(n_inits):
            if i < 2:
                b = remaining * 0.35
            else:
                b = remaining * (0.8 / (n_inits - 2)) if n_inits > 2 else remaining
            init_budgets.append(b)
            remaining -= b

        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break

            sc_init = _score(a_init)
            if sc_init < best_score:
                best_score = sc_init
                best_seq = a_init.copy()

            # Skip if clearly worse than current best
            if sc_init > best_score * 1.1:
                continue

            ib = min(init_budgets[i], deadline - time.monotonic() - 50)
            if ib < 15:
                continue

            print(f"    Init {i}: score={sc_init:.8f}, budget={ib:.0f}s")
            a_opt, sc_opt = _full_optimize(a_init, n_t, deadline, ib)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"    ** New best at n={n_t}: {sc_opt:.10f}")

    # Phase 3: Intensive refinement on the best found
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = best_seq.size
        remaining = deadline - time.monotonic() - 30.0
        print(f"\n=== Final refinement: n={n_best}, budget={remaining:.0f}s, score={best_score:.10f} ===")

        # Multiple rounds of LP + spectral + gradient
        for round_id in range(5):
            if time.monotonic() > deadline - 30:
                break

            sub_budget = min(remaining / (5 - round_id), deadline - time.monotonic() - 25)
            if sub_budget < 10:
                break

            # LP
            a_lp = _lp_optimize(best_seq, time.monotonic() + sub_budget * 0.4,
                                max_iter=15000, inflate_init=1.0005 + 0.0002 * round_id,
                                blend_init=0.3 - 0.03 * round_id)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"    Round {round_id} LP: {sc:.10f}")

            # Spectral
            a_sp = _spectral_flatten(best_seq, time.monotonic() + sub_budget * 0.3, max_iter=1500)
            sc = _score(a_sp)
            if sc < best_score:
                best_score = sc
                best_seq = a_sp.copy()
            print(f"    Round {round_id} Spectral: {sc:.10f}")

            # Gradient
            a_gp = _gradient_polish(best_seq, time.monotonic() + sub_budget * 0.2, max_iter=10000)
            sc = _score(a_gp)
            if sc < best_score:
                best_score = sc
                best_seq = a_gp.copy()
            print(f"    Round {round_id} Gradient: {sc:.10f}")

            # Adam
            a_ad = _adam_polish(best_seq, time.monotonic() + sub_budget * 0.1, max_iter=10000, lr=0.02)
            sc = _score(a_ad)
            if sc < best_score:
                best_score = sc
                best_seq = a_ad.copy()
            print(f"    Round {round_id} Adam: {sc:.10f}")

    # Phase 4: Try escape with random perturbations and quick re-optimization
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Escape attempts ===")
        for attempt in range(5):
            if time.monotonic() > deadline - 20:
                break
            a_pert = _random_perturb_escape(best_seq, best_score,
                                            time.monotonic() + 20, rng, n_perturbs=20)
            sc = _score(a_pert)
            if sc < best_score:
                best_score = sc
                best_seq = a_pert.copy()
                print(f"    Escape {attempt}: {sc:.10f}")
                # Quick LP to refine
                a_lp = _lp_optimize(best_seq, time.monotonic() + 15,
                                    max_iter=5000, inflate_init=1.0008, blend_init=0.3)
                sc2 = _score(a_lp)
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = a_lp.copy()
                    print(f"    Escape {attempt} LP: {sc2:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_u_profile(512, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_u_profile(512, 2.0)

    # Cross-check with evaluate_sequence
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence check: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
