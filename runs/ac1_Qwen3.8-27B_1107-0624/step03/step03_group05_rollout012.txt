        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 100:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _cross_convolve(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _make_conv_matrix(a):
    n = len(a)
    # Sparse construction: A[k, j] = a[j] if 0 <= k-j <= n-1
    # Using scipy sparse would be ideal, but let's use dense for small n
    # For n up to ~1000, dense is 2000x1000 = 2M entries, manageable
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M elementwise, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='hires')
    if res.success:
        return res.x
    return None


def _lp_spread_sparse(a, M):
    """Solve LP using sparse matrix for better performance."""
    from scipy.sparse import csr_matrix
    n = len(a)
    n_rows = 2 * n - 1
    # Build sparse convolution matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if a[j] < 1e-15:
            continue
        for k_offset in range(n):
            k = j + k_offset
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    A = csr_matrix((vals, (rows, cols)), shape=(n_rows, n))
    c_obj = -np.ones(n)
    b_ub = np.full(n_rows, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='hires')
    if res.success:
        return res.x
    return None


def _smooth_max(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return mval + beta * np.log(s / len(c) * len(c))  # just return the value


def _smooth_max_weight(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _gradient_smooth_max(a, beta):
    """Gradient of smooth_max(conv(a,a)) with respect to a."""
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    # grad_j = 2 * sum_k w_k * a_{k-j} (with appropriate bounds)
    # This is 2 * cross_convolve(w, a_rev)[n-1:2n-1]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h, float(c.max()), c


def _project_nonneg_sum1(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # Use bisection on threshold
    lo = -float(v.max()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        w = np.maximum(v - mid, 0.0)
        s = w.sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    w = np.maximum(v - mid, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, center_sep=0.3, width=0.15):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5 + center_sep) / width) ** 2) + \
        np.exp(-0.5 * ((t - 0.5 - center_sep) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_to_length(profile, n_new):
    profile = np.asarray(profile, dtype=np.float64).ravel()
    n0 = profile.size
    if n0 == n_new:
        s = profile.sum()
        if s > 1e-30:
            return profile / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, profile)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_iterate(a, deadline, max_iter=50000, min_inflate=1.00001,
                max_inflate=1.01, initial_blend=0.5, min_blend=0.01):
    """Core LP-based iteration with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, _score(a)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    inflate = 1.001
    blend = initial_blend
    no_imp = 0
    imp_count = 0

    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Solve LP
        g = _lp_spread(a, M * inflate)
        if g is None:
            inflate = min(max_inflate, inflate * 1.02)
            no_imp += 1
            if no_imp > 500:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            # Try to reduce inflate to get closer to exact
            inflate = max(min_inflate, inflate * 0.995)
            if no_imp > 1000:
                # Try with exact M
                g = _lp_spread(a, M)
                if g is not None and float(g.sum()) > 1.0 + 1e-6:
                    sum_g = float(g.sum())
                else:
                    break
            else:
                continue

        g = g / sum_g
        # Blend
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_score - 1e-10:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
            imp_count += 1
            # On improvement, reduce inflate and keep blend
            inflate = max(min_inflate, inflate * 0.99)
            blend = max(min_blend, blend * 0.995)
        else:
            no_imp += 1
            # On non-imp, reduce blend and slightly increase inflate
            blend = max(min_blend, blend * 0.95)
            if no_imp % 100 == 0:
                inflate = min(max_inflate, inflate * 1.01)
            if no_imp > 3000:
                # Reset to best and try different params
                a = best_a.copy()
                inflate = 1.002
                blend = 0.3
                no_imp = 0
                # Try a kick
                rng = np.random.default_rng(it * 7 + n)
                a_kick = a * (1.0 + 0.05 * rng.normal(size=n))
                a_kick = np.maximum(a_kick, 0.0)
                a_kick = a_kick / a_kick.sum()
                if _score(a_kick) < best_score + 0.01:
                    a = a_kick

        a = a_new

        if it % 5000 == 0 or (time.monotonic() - last_print > 30):
            elapsed = time.monotonic() - t0
            print(f"      LP iter {it}: score={best_score:.10f}, "
                  f"inflate={inflate:.5f}, blend={blend:.4f}, "
                  f"no_imp={no_imp}, time={elapsed:.0f}s")
            last_print = time.monotonic()

    return best_a, best_score


def _gradient_polish(a, deadline, max_iter=30000, beta_init=0.02):
    """Smooth-max gradient descent with adaptive beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_nonneg_sum1(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = beta_init * float(c.max())
    lr = 0.02
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, maxc, c_arr = _gradient_smooth_max(a, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-10, beta * 2.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_nonneg_sum1(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = beta_init * float(_fft_convolve(a).max())
                lr = base_lr
                no_imp = 0

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.85)

    return best_a, best_score


def _adam_polish(a, deadline, max_iter=20000, lr=0.03):
    """Adam optimizer on logit parametrization."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.03 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-10, beta * 0.8)

        if no_imp > max(3000, 3 * n):
            rng_r = np.random.default_rng(it * 13 + 7)
            z = 0.8 * best_z + 0.2 * rng_r.normal(size=n) * 0.3
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a, best_score


def _pair_exchange(a, deadline, max_iter=100000):
    """Targeted pair-exchange local search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    # Pre-compute which pairs to try
    # Focus on pairs where one has high value and moving to the other reduces peak
    c = _fft_convolve(a)
    peak_idx = np.argmax(c)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if a[i] < 1e-10:
            continue

        # Try several transfer amounts
        best_local_score = best_score
        best_local = None
        for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            delta = frac * a[i]
            if delta >= a[i]:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            sc = _score(a_try)
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_local = a_try.copy()

        if best_local is not None:
            a = best_local
            if best_local_score < best_score - 1e-12:
                best_score = best_local_score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 5000:
            break

        if it % 10000 == 0:
            print(f"      PairExchange iter {it}: best={best_score:.10f}, no_imp={no_imp}")

    return best_a, best_score


def _block_swap(a, deadline, max_iter=3000):
    """Random block swaps to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(999)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        block_size = int(rng.integers(max(2, n // 32), max(3, n // 6)))
        i = int(rng.integers(0, n - block_size))
        j = int(rng.integers(0, n - block_size))
        if abs(i - j) < block_size:
            continue

        a_new = a.copy()
        tmp = a_new[i:i + block_size].copy()
        a_new[i:i + block_size] = a_new[j:j + block_size]
        a_new[j:j + block_size] = tmp
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break

    return best_a, best_score


def _full_optimize(a_init, n, deadline):
    """Run full optimization pipeline for a given sequence length."""
    a = np.asarray(a_init, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score(a)
    remaining = deadline - time.monotonic()

    if remaining < 20:
        return best_a, best_score

    print(f"    Initial score: {best_score:.10f}")

    # Phase 1: LP iteration (main workhorse)
    lp_deadline = time.monotonic() + min(remaining * 0.4, 600.0)
    a1, s1 = _lp_iterate(a, lp_deadline, max_iter=100000,
                         min_inflate=1.00001, max_inflate=1.005,
                         initial_blend=0.4, min_blend=0.02)
    if s1 < best_score:
        best_score = s1
        best_a = a1.copy()
    print(f"    After LP: {best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_score

    # Phase 2: Gradient polish
    gp_deadline = time.monotonic() + min(remaining * 0.2, 200.0)
    a2, s2 = _gradient_polish(best_a, gp_deadline, max_iter=20000, beta_init=0.02)
    if s2 < best_score:
        best_score = s2
        best_a = a2.copy()
    print(f"    After Gradient: {best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_score

    # Phase 3: Adam polish
    ad_deadline = time.monotonic() + min(remaining * 0.2, 200.0)
    a3, s3 = _adam_polish(best_a, ad_deadline, max_iter=15000, lr=0.025)
    if s3 < best_score:
        best_score = s3
        best_a = a3.copy()
    print(f"    After Adam: {best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_score

    # Phase 4: More LP (with best as start)
    lp2_deadline = time.monotonic() + min(remaining * 0.2, 200.0)
    a4, s4 = _lp_iterate(best_a, lp2_deadline, max_iter=50000,
                         min_inflate=1.00001, max_inflate=1.003,
                         initial_blend=0.2, min_blend=0.01)
    if s4 < best_score:
        best_score = s4
        best_a = a4.copy()
    print(f"    After LP2: {best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_score

    # Phase 5: Pair exchange
    pe_deadline = time.monotonic() + min(remaining * 0.3, 300.0)
    a5, s5 = _pair_exchange(best_a, pe_deadline, max_iter=200000)
    if s5 < best_score:
        best_score = s5
        best_a = a5.copy()
    print(f"    After PairExchange: {best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_score

    # Phase 6: Final gradient polish
    gp2_deadline = time.monotonic() + min(remaining * 0.5, 100.0)
    a6, s6 = _gradient_polish(best_a, gp2_deadline, max_iter=10000, beta_init=0.01)
    if s6 < best_score:
        best_score = s6
        best_a = a6.copy()
    print(f"    Final Gradient: {best_score:.10f}")

    return best_a, best_score


def _concentrated_lp(a, M, peak_positions, tolerance=0.1):
    """LP that only enforces constraints near peak positions (for speed)."""
    n = len(a)
    n_rows = 2 * n - 1
    # Only enforce constraints at peak positions and nearby
    row_idx = []
    for k in range(n_rows):
        if min(abs(k - pk) for pk in peak_positions) <= tolerance * n:
            row_idx.append(k)
    if len(row_idx) < 10:
        row_idx = list(range(n_rows))
    A = _make_conv_matrix(a)
    A_sub = A[row_idx, :]
    b_sub = np.full(len(row_idx), M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_sub, b_ub=b_sub,
                  bounds=[(0, None)] * n, method='hires')
    if res.success:
        return res.x
    return None


def _peak_focused_lp(a, deadline, max_iter=30000):
    """LP iteration focusing on peak constraints for speed."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    inflate = 1.001
    blend = 0.3
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        # Find peak positions
        peak_thresh = M * 0.99
        peak_positions = list(np.where(c >= peak_thresh)[0])

        # Solve relaxed LP (only peak constraints)
        g = _concentrated_lp(a, M * inflate, peak_positions, tolerance=0.05)
        if g is None:
            # Fall back to full LP
            g = _lp_spread(a, M * inflate)
            if g is None:
                inflate = min(1.01, inflate * 1.02)
                no_imp += 1
                if no_imp > 500:
                    break
                continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            inflate = max(1.00001, inflate * 0.998)
            if no_imp > 500:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            inflate = max(1.00001, inflate * 0.995)
            blend = max(0.02, blend * 0.997)
        else:
            no_imp += 1
            blend = max(0.01, blend * 0.93)
            if no_imp % 200 == 0:
                inflate = min(1.005, inflate * 1.01)
            if no_imp > 2000:
                a = best_a.copy()
                inflate = 1.001
                blend = 0.25
                no_imp = 0
                # Kick
                rng_k = np.random.default_rng(it * 31 + n)
                a_k = a * (1.0 + 0.03 * rng_k.normal(size=n))
                a_k = np.maximum(a_k, 0.0)
                a_k = a_k / a_k.sum()
                if _score(a_k) < best_score + 0.005:
                    a = a_k

        a = a_new

        if it % 5000 == 0:
            print(f"      PeakLP iter {it}: best={best_score:.10f}, no_imp={no_imp}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP spreading, gradient methods,
    and pair exchange to minimize the step function score.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = [768]
    if parent is not None:
        pn = parent.size
        if pn not in lengths:
            lengths.insert(0, pn)
        # Add nearby lengths
        for dl in [-64, -32, 32, 64]:
            ln = pn + dl
            if ln >= 128 and ln not in lengths:
                lengths.append(ln)
    # Add some other lengths
    for ln in [640, 800, 896]:
        if ln not in lengths:
            lengths.append(ln)

    # Sort: parent length first, then by distance from parent
    if parent is not None:
        lengths.sort(key=lambda x: abs(x - parent.size))
    else:
        lengths.sort()

    print(f"Lengths to try: {lengths}")
    print(f"Total budget: {budget_s}s, effective deadline in {deadline - time.monotonic():.0f}s")

    # Allocate time: give more to shorter lists (faster LP)
    # Rough time per LP iteration scales as n^2 (matrix construction) + n*log(n) (FFT)
    # Prioritize lengths close to parent
    remaining = deadline - time.monotonic()
    # Give 70% of time to the first (best) length, 20% to second, 10% to rest
    if len(lengths) == 1:
        budgets = [remaining - 60]
    elif len(lengths) == 2:
        budgets = [remaining * 0.65 - 60, remaining * 0.30 - 30]
    else:
        first = remaining * 0.50 - 60
        second = remaining * 0.20 - 30
        rest = remaining * 0.15 - 20
        budgets = [first, second] + [rest / (len(lengths) - 2)] * (len(lengths) - 2)

    for idx, n_t in enumerate(lengths):
        if time.monotonic() > deadline - 60:
            break

        alloc = max(30.0, min(budgets[idx], deadline - time.monotonic() - 60.0))
        if alloc < 30:
            continue

        print(f"\n=== Length {n_t} (budget {alloc:.0f}s) ===")

        # Initialization
        if parent is not None:
            a_init = _interpolate_to_length(parent, n_t)
        else:
            a_init = _make_u_profile(n_t, 1.5)

        sc_init = _score(a_init)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()

        phase_deadline = time.monotonic() + alloc
        a_best, sc_best = _full_optimize(a_init, n_t, phase_deadline)
        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best.copy()
            print(f"  ** New best at n={n_t}: {sc_best:.10f}")

    # Phase 2: Diverse initializations at best length
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_best = best_seq.size
        print(f"\n=== Phase 2: Diverse inits at n={n_best} ===")

        inits = [
            _make_u_profile(n_best, 1.0),
            _make_u_profile(n_best, 2.5),
            _make_u_profile(n_best, 4.0),
            _make_linear_profile(n_best, 1.0),
            _make_linear_profile(n_best, 2.5),
            _make_bimodal(n_best, 0.25, 0.18),
            _make_bimodal(n_best, 0.35, 0.12),
        ]
        # Perturbed best
        for _ in range(2):
            a_p = best_seq * (1.0 + 0.1 * rng.normal(size=n_best))
            a_p = np.maximum(a_p, 0.0)
            a_p = a_p / a_p.sum()
            inits.append(a_p)

        rem = deadline - time.monotonic() - 50
        per_init = max(20.0, rem / len(inits) * 0.8)

        for a0 in inits:
            if time.monotonic() > deadline - 50:
                break
            sc0 = _score(a0)
            print(f"  Init: {sc0:.8f}")
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick optimization
            init_deadline = time.monotonic() + min(per_init, deadline - time.monotonic() - 40)
            if init_deadline - time.monotonic() > 20:
                a_res, sc_res = _full_optimize(a0, n_best, init_deadline)
                if sc_res < best_score:
                    best_score = sc_res
                    best_seq = a_res.copy()
                    print(f"  ** New best from init: {sc_res:.10f}")

    # Phase 3: Intensive final polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        final_remain = deadline - time.monotonic() - 20
        print(f"\n=== Phase 3: Final polish (budget {final_remain:.0f}s) ===")

        # Sub-phase: Peak-focused LP
        pf_deadline = time.monotonic() + min(final_remain * 0.3, 300.0)
        a_pf, s_pf = _peak_focused_lp(best_seq, pf_deadline, max_iter=50000)
        if s_pf < best_score:
            best_score = s_pf
            best_seq = a_pf.copy()
        print(f"  After PeakLP: {best_score:.10f}")

        # Sub-phase: Full LP
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            lp_deadline = time.monotonic() + min(remaining * 0.25, 200.0)
            a_fl, s_fl = _lp_iterate(best_seq, lp_deadline, max_iter=50000,
                                     min_inflate=1.00001, max_inflate=1.002,
                                     initial_blend=0.15, min_blend=0.01)
            if s_fl < best_score:
                best_score = s_fl
                best_seq = a_fl.copy()
            print(f"  After FinalLP: {best_score:.10f}")

        # Sub-phase: Adam
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            ad_deadline = time.monotonic() + min(remaining * 0.2, 150.0)
            a_ad, s_ad = _adam_polish(best_seq, ad_deadline, max_iter=10000, lr=0.02)
            if s_ad < best_score:
                best_score = s_ad
                best_seq = a_ad.copy()
            print(f"  After FinalAdam: {best_score:.10f}")

        # Sub-phase: Gradient with very small beta
        remaining = deadline - time.monotonic() - 20
        if remaining > 20:
            gp_deadline = time.monotonic() + min(remaining * 0.2, 100.0)
            a_gp, s_gp = _gradient_polish(best_seq, gp_deadline, max_iter=10000, beta_init=0.005)
            if s_gp < best_score:
                best_score = s_gp
                best_seq = a_gp.copy()
            print(f"  After FinalGrad: {best_score:.10f}")

        # Sub-phase: Pair exchange
        remaining = deadline - time.monotonic() - 20
        if remaining > 20:
            pe_deadline = time.monotonic() + min(remaining * 0.3, 200.0)
            a_pe, s_pe = _pair_exchange(best_seq, pe_deadline, max_iter=300000)
            if s_pe < best_score:
                best_score = s_pe
                best_seq = a_pe.copy()
            print(f"  After FinalPair: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_u_profile(768, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_u_profile(768, 1.5)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
