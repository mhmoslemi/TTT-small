        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unnorm(a):
    """Score assuming sum(a)=1, returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return a / a.sum()


def _make_lin(n, slope=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_quad(n, curvature=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + curvature * t + curvature * t * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_bell(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2) + 1e-8
    return a / a.sum()


def _make_sine(n, freq=2.0, amp=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_cosine_envelope(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * t / 2.0) + 1e-8
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_power_asym(n, p=0.5):
    """Asymmetric power law: larger on one side."""
    t = np.linspace(0.0, 1.0, n)
    a = t ** p + (1.0 - t) ** (1.0 / p) + 1e-8
    return a / a.sum()


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    """Compute gradient of soft-max of self-convolution w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # grad_j = 2 * sum_m w[m+j] * a[m]  (cross-correlation, valid mode)
    corr = np.correlate(w, a, mode='valid')
    grad = 2.0 * corr
    return grad, mval


def _solve_lp_active(f, M, n, thresh=0.9):
    """Solve LP: max sum(g) s.t. (f*g)_i <= M for near-active i, g >= 0."""
    c = _fft_convolve(f)
    active = c > thresh * M
    n_active = int(active.sum())
    if n_active < 3:
        active = c > 0.5 * M
        n_active = int(active.sum())
    if n_active == 0:
        return None

    # Cap constraints for speed
    if n_active * n > 3_000_000:
        idx = np.where(active)[0]
        c_sub = c[idx]
        top = np.argsort(c_sub)[::-1][:2000]
        active2 = np.zeros_like(active)
        active2[idx[top]] = True
        active = active2
        n_active = 2000

    rows = np.where(active)[0]
    A = np.zeros((n_active, n))
    for k, i in enumerate(rows):
        j_start = max(0, i - n + 1)
        j_end = min(n - 1, i)
        A[k, j_start:j_end + 1] = f[j_start:j_end + 1]

    c_obj = -np.ones(n)
    b = np.full(n_active, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _line_search(a, g, n=10):
    """Find best blend coefficient."""
    best_t = 0.0
    best_sc = _score_unnorm(a)
    ts = np.logspace(-2, 0, n)  # 0.01 to 1.0
    for t in ts:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score_unnorm(a_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.005, inflate_min=1.001):
    """LP refinement with active set constraint reduction."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start
    thresh = 0.9
    consecutive_breaks = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp_active(a, M_lp, n, thresh=thresh)
        if g is None:
            consecutive_breaks += 1
            if consecutive_breaks > 5:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 15 == 0:
                inflate = min(0.05, inflate * 1.25)
                thresh = max(0.7, thresh * 0.95)
            if no_imp > 150:
                a = best_a.copy()
                inflate = inflate_start
                thresh = 0.9
                no_imp = 0
                consecutive_breaks += 1
                if consecutive_breaks > 3:
                    break
            continue

        g = g / sum_g
        best_t, best_sc = _line_search(a, g)

        if best_t > 0 and best_sc < best_score - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            consecutive_breaks = 0
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.05, inflate * 1.1)
            if no_imp > 80:
                a = best_a.copy()
                inflate = inflate_start
                thresh = 0.9
                no_imp = 0

    return best_a


def _nesterov_gradient(a0, deadline, max_iter=50000):
    """Nesterov-accelerated projected gradient descent on simplex."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    cmax = float(c.max())
    beta = max(1e-6, 0.02 * cmax)
    lr = 0.05
    base_lr = lr
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    it = 0

    while it < max_iter:
        if time.monotonic() > deadline:
            break
        it += 1

        a_nest = a + momentum * v
        grad, mval = _smooth_max_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-7, 0.02 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 200 == 199:
            beta = max(1e-8, beta * 0.85)

    return best_a


def _perturb_and_reopt(a0, rng, deadline, n_perturb=5):
    """Add structured sinusoidal perturbations and re-optimize."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best = a.copy()
    best_sc = _score_unnorm(a)

    for p in range(n_perturb):
        if time.monotonic() > deadline:
            break

        # Random sinusoidal perturbation
        freq = rng.uniform(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.08)
        t = np.linspace(0, 1, n)
        pert = amp * np.sin(2 * np.pi * freq * t + phase)
        
        a_pert = a + pert
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = a_pert / a_pert.sum()

        sc_pert = _score_unnorm(a_pert)
        
        # Short gradient polish on perturbed
        gp_end = min(time.monotonic() + 30, deadline)
        a_opt = _nesterov_gradient(a_pert, gp_end, max_iter=2000)
        sc_opt = _score_unnorm(a_opt)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best = a_opt.copy()
            a = a_opt.copy()

    return best


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimization with active-set LP, sinusoidal perturbations,
    and Nesterov gradient. Explores sequence lengths 128-2048.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define initializations
    def generate_initializations(n):
        inits = []
        inits.append(('ramp1', _make_ramp(n, 1.0)))
        inits.append(('ramp2', _make_ramp(n, 2.0)))
        inits.append(('lin2', _make_lin(n, 2.0)))
        inits.append(('lin3', _make_lin(n, 3.0)))
        inits.append(('quad2', _make_quad(n, 2.0)))
        inits.append(('bell', _make_bell(n, 0.3)))
        inits.append(('sine2', _make_sine(n, 2.0, 0.3)))
        inits.append(('cos_env', _make_cosine_envelope(n)))
        inits.append(('pwr_asym', _make_power_asym(n, 0.5)))
        inits.append(('pwr_asym2', _make_power_asym(n, 2.0)))
        if parent is not None:
            inits.append(('parent_interp', _interpolate(parent, n)))
        return inits

    # Phase 1: Coarse optimization at small scale (n=128)
    print(f"\n=== Phase 1: Coarse at n=128 ===")
    n_small = 128
    inits_small = generate_initializations(n_small)
    
    # Score all initializations
    scored = []
    for name, a0 in inits_small:
        sc = _score(a0)
        scored.append((sc, name, a0))
    scored.sort(key=lambda x: x[0])
    
    # Take top 3 and do quick gradient polish
    for sc0, name, a0 in scored[:3]:
        if time.monotonic() > deadline - 600:
            break
        if sc0 >= best_score + 0.5:
            continue
        gp_end = min(time.monotonic() + 30, deadline)
        a_opt = _nesterov_gradient(a0, gp_end, max_iter=3000)
        sc_opt = _score(a_opt)
        print(f"  {name}: init={sc0:.6f} -> grad={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

    # Phase 2: Multi-scale expansion
    print(f"\n=== Phase 2: Multi-scale expansion ===")
    target_sizes = [256, 512, 1024]
    
    for n_t in target_sizes:
        if time.monotonic() > deadline - 300:
            break
        
        # Interpolate best to new size
        if best_seq is not None:
            a_scale = _interpolate(best_seq, n_t)
        else:
            a_scale = _make_ramp(n_t, 1.0)
        
        # Also try interpolating second-best
        candidates = [a_scale]
        second_best = None
        for sc0, name, a0 in scored:
            if name != scored[0][1]:
                second_best = _interpolate(a0, n_t)
                break
        if second_best is not None:
            candidates.append(second_best)
        
        # Add a fresh initialization at this scale
        candidates.append(_make_lin(n_t, 3.0))
        candidates.append(_make_sine(n_t, 3.0, 0.2))
        
        # Quick gradient on each candidate
        best_at_scale = None
        best_sc_at_scale = np.inf
        for i, cand in enumerate(candidates):
            if time.monotonic() > deadline - 300:
                break
            gp_end = min(time.monotonic() + 20, deadline)
            a_opt = _nesterov_gradient(cand, gp_end, max_iter=2000)
            sc = _score(a_opt)
            print(f"  n={n_t} candidate {i}: score={sc:.6f}")
            if sc < best_sc_at_scale:
                best_sc_at_scale = sc
                best_at_scale = a_opt
        
        if best_at_scale is not None:
            # LP refinement at this scale
            if time.monotonic() < deadline - 200:
                lp_end = min(time.monotonic() + 120, deadline - 100)
                a_lp = _lp_refine(best_at_scale, lp_end, max_iter=20000)
                sc_lp = _score(a_lp)
                print(f"  n={n_t} LP: {best_sc_at_scale:.6f} -> {sc_lp:.6f}")
                if sc_lp < best_sc_at_scale:
                    best_at_scale = a_lp
                    best_sc_at_scale = sc_lp
            
            # Gradient polish
            if time.monotonic() < deadline - 100:
                gp_end = min(time.monotonic() + 30, deadline - 50)
                a_gp = _nesterov_gradient(best_at_scale, gp_end, max_iter=5000)
                sc_gp = _score(a_gp)
                if sc_gp < best_sc_at_scale:
                    best_at_scale = a_gp
                    best_sc_at_scale = sc_gp
            
            if best_sc_at_scale < best_score:
                best_score = best_sc_at_scale
                best_seq = best_at_scale.copy()
                print(f"  New best: {best_score:.8f} at n={n_t}")

    # Phase 3: Scale up to 1536 or 2048 if time allows
    print(f"\n=== Phase 3: Large scale refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        for n_large in [1536, 2048]:
            if time.monotonic() > deadline - 150:
                break
            a_large = _interpolate(best_seq, n_large)
            sc0 = _score(a_large)
            print(f"  n={n_large}: interp score={sc0:.6f}")
            
            # Gradient polish
            gp_end = min(time.monotonic() + 60, deadline - 80)
            a_gp = _nesterov_gradient(a_large, gp_end, max_iter=10000)
            sc_gp = _score(a_gp)
            print(f"  n={n_large} gradient: {sc_gp:.6f}")
            
            # LP refinement
            if time.monotonic() < deadline - 100:
                lp_end = min(time.monotonic() + 90, deadline - 50)
                a_lp = _lp_refine(a_gp, lp_end, max_iter=20000, inflate_start=1.003)
                sc_lp = _score(a_lp)
                print(f"  n={n_large} LP: {sc_lp:.6f}")
                if sc_lp < sc_gp:
                    a_gp = a_lp
                    sc_gp = sc_lp
            
            if sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
                print(f"  New best: {best_score:.8f} at n={n_large}")

    # Phase 4: Perturbation escape on the best
    print(f"\n=== Phase 4: Perturbation escape ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        pert_end = min(time.monotonic() + 120, deadline - 40)
        a_pert = _perturb_and_reopt(best_seq, rng, pert_end, n_perturb=8)
        sc_pert = _score(a_pert)
        print(f"  Perturb: {best_score:.8f} -> {sc_pert:.8f}")
        if sc_pert < best_score:
            best_score = sc_pert
            best_seq = a_pert.copy()

    # Phase 5: Final LP and gradient polish
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Tight LP
        if time.monotonic() < deadline - 50:
            lp_end = min(time.monotonic() + 60, deadline - 20)
            a_final_lp = _lp_refine(best_seq, lp_end, max_iter=20000,
                                     inflate_start=1.002, inflate_min=1.0005)
            sc_flp = _score(a_final_lp)
            print(f"  Final LP: {best_score:.8f} -> {sc_flp:.8f}")
            if sc_flp < best_score:
                best_score = sc_flp
                best_seq = a_final_lp.copy()

        # Final gradient
        if time.monotonic() < deadline - 15:
            gp_end = min(time.monotonic() + 40, deadline - 5)
            a_final_gp = _nesterov_gradient(best_seq, gp_end, max_iter=10000)
            sc_fgp = _score(a_final_gp)
            print(f"  Final gradient: {best_score:.8f} -> {sc_fgp:.8f}")
            if sc_fgp < best_score:
                best_score = sc_fgp
                best_seq = a_final_gp.copy()

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(256, 1.0)
        best_score = _score(best_seq)

    # Final validation and cleanup
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_ramp(256, 1.0)
    
    # Try to use evaluate_sequence for final check
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence check: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.0f}s")
    return [float(x) for x in final_seq]
