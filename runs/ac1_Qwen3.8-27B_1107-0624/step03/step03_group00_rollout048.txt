def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

        return np.array([])

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _subgradient_fft(a, K, gap_frac=0.01):
    """
    Compute weighted subgradient using FFT for efficiency.
    grad[m] = sum_k w_k * (a[k-m] + a[m-k]) where sum uses a[j]=0 outside [0,n).
    """
    n = len(a)
    c = _full_convolve(a)
    maxc = float(c.max())
    if maxc < 1e-30:
        return np.zeros(n), 0.0
    threshold = maxc - gap_frac * maxc
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    active = active[:K]
    diffs = maxc - c[active]
    wks = np.exp(-diffs / (maxc * 0.01 + 1e-15))
    wks = wks / wks.sum()
    
    # Build filter h of length 2n-1 where h[k] = w_k for active k
    h = np.zeros(2 * n - 1)
    for i, k in enumerate(active):
        h[int(k)] = wks[i]
    
    # First term: sum_k w_k * a[k-m] for m = 0..n-1
    # a[k-m] is nonzero when 0 <= k-m < n, i.e., m <= k and m >= k-n+1
    # This is cross-correlation: (a cross h)[m] = sum_k h[k] * a[k-m]
    # In FFT terms: IFFT(FFT(a) * conj(FFT(h)))... 
    # Actually: cross-correlation x ⊛ y = IFFT(conj(FFT(x)) * FFT(y))
    # sum_k h[k] a[k-m] = (h ⊛ a)[-m]... this is getting confusing.
    # Let me just use direct computation since K is typically small (1-10)
    
    grad = np.zeros(n)
    for i, k in enumerate(active):
        k = int(k)
        wk = wks[i]
        # a[k-m]: m ranges over [max(0, k-n+1), min(k, n-1)]
        m1 = max(0, k - n + 1)
        m2 = min(k, n - 1)
        if m1 <= m2:
            ms = np.arange(m1, m2 + 1)
            grad[ms] += wk * a[k - ms]
        # a[m-k]: m ranges over [max(k, 0), min(k+n-1, n-1)]
        m3 = max(k, 0)
        m4 = min(k + n - 1, n - 1)
        if m3 <= m4:
            ms = np.arange(m3, m4 + 1)
            grad[ms] += wk * a[ms - k]
    return grad, maxc


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _lp_solve(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 60})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, t_init=0.3, max_solves=30):
    """LP-based refinement: iteratively solve LP and blend."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, t * 0.85)
    return current_best, current_score


def _adam_optimize(p0, deadline, rng, lr_init=0.04, beta_frac_init=0.05,

    Adam optimization in logit space with exact top-K subgradient.
    Uses cosine LR schedule and periodic LP refinement.

    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)

    total_steps = max(1, int((deadline - time.monotonic()) / 0.05))

    K_cur = 1
    gap_frac = 0.005

    while time.monotonic() < deadline:
        # Cosine LR schedule
        progress = min(1.0, step_count / max(1, total_steps))
        if cosine_decay:
            lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
            lr = max(lr, lr_init * 0.05)
        else:
            lr = lr_init

        # Compute subgradient with adaptive K
        # If the top is flat (small gap), use more lags
        c_sorted = np.sort(c)[::-1]
        if c_sorted.size >= 2:
            gap = c_sorted[0] - c_sorted[1]
            if gap < maxc * 0.002:
                K_cur = min(K_max, K_cur + 1)
            elif gap > maxc * 0.01:
                K_cur = max(1, K_cur - 1)
            # Adaptive gap threshold
            if gap < maxc * 0.001:
                gap_frac = max(0.001, gap_frac * 0.9)
            elif gap > maxc * 0.02:
                gap_frac = min(0.05, gap_frac * 1.1)
        
        vp, cur_maxc = _subgradient_fft(p, K_cur, gap_frac)
        
        # Project to tangent space of simplex
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        # Adam update
        t_adam += 1
        step_count += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0

            m.fill(0.0)

            t_adam = 0

                best_p = p_lp.copy()
                z = np.log(np.maximum(p_lp, 1e-12))
                z -= float(np.mean(z))

                m.fill(0.0)
                v.fill(0.0)

                no_imp = 0

        if no_imp > limit:
            ns = restart_noise * (1.0 + 0.5 * min(5, (no_imp - limit) // 100))
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            no_imp = 0
            K_cur = 1
            gap_frac = 0.01

    return best_p, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _try_different_n(p_best, n_new, deadline, rng):
    """Try interpolating to a different n and optimizing there."""
    p_int = _interp_profile(p_best, n_new)
    if p_int is None:
        return None, np.inf
    sc_int = _score_norm(p_int)
    print(f"  Interp to n={n_new}: score={sc_int:.10f}")
    if time.monotonic() < deadline - 30:
        end = min(deadline - 5, time.monotonic() + min(120.0, deadline - time.monotonic() - 10))
        p_opt, sc_opt = _adam_optimize(p_int, end, rng, lr_init=0.03,
                                        K_max=6, lp_every=200,
                                        no_imp_limit=300)
        if sc_opt < sc_int:
            print(f"  Optimized n={n_new}: score={sc_opt:.10f}")
            return p_opt, sc_opt
    return p_int, sc_int


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function. Minimizes 2*n*max(conv(a,a))/sum(a)^2."""
    t_start = time.monotonic()
    deadline = t_start + max(50.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.1f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")
    print(f"Starting at t=0, deadline at t={deadline - t_start:.1f}s")

    # --- Phase 1: Initialization (0-60s) ---
    phase1_end = t_start + 60.0
    candidates = []

    if parent is not None:
        sc_parent = _score_norm(parent)
        candidates.append((sc_parent, parent.copy(), pn, 'parent'))
        print(f"  Parent score: {sc_parent:.10f} (n={pn})")
        # Also try interpolating parent to other sizes
        for n_try in [512, 1024, 2048]:
            if n_try != pn:
                p_int = _interp_profile(parent, n_try)
                if p_int is not None:
                    sc_int = _score_norm(p_int)
                    candidates.append((sc_int, p_int, n_try, f'parent_int_{n_try}'))

    # Diverse initial candidates
    for n in [512, 1024, 2048]:
        profs = [
            ('lin1.0', _make_profile_linear(n, 1.0)),
            ('lin2.0', _make_profile_linear(n, 2.0)),
            ('cubic', _make_profile_cubic(n, 1.5, -1.0)),
            ('cubic2', _make_profile_cubic(n, 2.0, -1.5)),
            ('cos2', _make_profile_cosine(n, 2, 0.7)),
            ('cos3', _make_profile_cosine(n, 3, 0.5)),
            ('twobump', _make_profile_two_bump(n, 0.3, 0.7, 0.15)),
            ('exp1', _make_profile_exponential(n, 1.0)),
        ]
        for label, p in profs:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: Main optimization on top candidates (60s - 1200s) ---
    phase2_end = t_start + 1200.0
    top_cands = candidates[:6]
    n_top = len(top_cands)
    time_per = (phase2_end - phase1_end) / max(1, n_top)
    
    for ci, (sc, p, n, label) in enumerate(top_cands):
        if time.monotonic() > phase2_end:
            break
        per_end = min(phase2_end - 30, time.monotonic() + time_per)
        print(f"  Phase 2 [{label} n={n}]: optimizing from {sc:.10f}, time={time_per:.0f}s")
        p_opt, sc_opt = _adam_optimize(p, per_end, rng, lr_init=0.04,
                                        K_max=8, cosine_decay=True,
                                        lp_every=150, restart_noise=0.04,
                                        no_imp_limit=300)
        _update(p_opt, sc_opt, f'phase2_{label}_n{n}')

        # If this is the best so far, do additional LP refinement
        if sc_opt < best_score and best_seq is not None and best_seq.size <= 2048:
            lp_end = min(phase2_end - 10, time.monotonic() + 60.0)
            if lp_end > time.monotonic() + 20:
                p_lp, sc_lp = _lp_refine(best_seq, lp_end, t_init=0.25, max_solves=15)
                _update(p_lp, sc_lp, f'lp_{label}_n{n}')
                # Quick Adam polish after LP
                if time.monotonic() < phase2_end - 20:
                    ad_end = min(phase2_end - 5, time.monotonic() + 40.0)
                    if ad_end > time.monotonic() + 15:
                        p_ap, sc_ap = _adam_optimize(p_lp, ad_end, rng, lr_init=0.02,
                                                      K_max=6, lp_every=300,
                                                      no_imp_limit=200)
                        _update(p_ap, sc_ap, f'postlp_{label}_n{n}')

    # --- Phase 3: Intensive refinement on current best (1200s - 1500s) ---
    phase3_end = t_start + 1500.0
    if best_seq is not None and time.monotonic() < phase3_end:
        print(f"  Phase 3: Intensive refinement, current best={best_score:.10f}")
        # Run Adam with very small steps for fine-tuning
        end3 = min(phase3_end - 10, deadline - 60)
        p_fine, sc_fine = _adam_optimize(best_seq, end3, rng, lr_init=0.015,
                                          K_max=10, cosine_decay=True,
                                          lp_every=100, restart_noise=0.02,
                                          no_imp_limit=500)
        _update(p_fine, sc_fine, 'phase3_fine')

        # LP refinement
        if time.monotonic() < phase3_end - 30 and best_seq.size <= 2048:
            lp_end = min(phase3_end - 5, time.monotonic() + 90.0)
            if lp_end > time.monotonic() + 20:
                p_lp2, sc_lp2 = _lp_refine(best_seq, lp_end, t_init=0.15, max_solves=20)
                _update(p_lp2, sc_lp2, 'phase3_lp')
                # Final Adam polish
                if time.monotonic() < phase3_end - 10:
                    ad_end = min(phase3_end - 3, time.monotonic() + 60.0)
                    if ad_end > time.monotonic() + 15:
                        p_fp, sc_fp = _adam_optimize(p_lp2, ad_end, rng, lr_init=0.01,
                                                      K_max=8, lp_every=500,
                                                      no_imp_limit=300)
                        _update(p_fp, sc_fp, 'phase3_final')

    # --- Phase 4: Multi-scale exploration (1500s - 1650s) ---
    phase4_end = t_start + 1650.0
    if best_seq is not None and time.monotonic() < phase4_end:
        n_best = best_seq.size
        target_ns = []
        if n_best <= 512:
            target_ns = [1024, 2048]
        elif n_best <= 1024:
            target_ns = [2048]
        elif n_best <= 2048:
            target_ns = [4096]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            print(f"  Phase 4: Trying n={tn}")
            p_new, sc_new = _try_different_n(best_seq, tn, phase4_end - 10, rng)
            _update(p_new, sc_new, f'scale_n{tn}')

    # --- Phase 5: Multi-restart (1650s - 1750s) ---
    phase5_end = t_start + 1750.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_best = best_seq.size
        n_restarts = 3
        per_r = max(15.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 5:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 10:
                break
            ns = 0.03 + 0.03 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_optimize(p_s, time.monotonic() + per, rng,
                                        lr_init=0.03, K_max=6,
                                        lp_every=300, no_imp_limit=200)
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

    # --- Phase 6: Final ultra-fine polish (1750s - deadline) ---
    if time.monotonic() < deadline - 3 and best_seq is not None:
        per = deadline - time.monotonic() - 3
        if per > 10:
            print(f"  Phase 6: Final polish, per={per:.1f}s")
            p_f, sc_f = _adam_optimize(best_seq, time.monotonic() + per, rng,
                                        lr_init=0.005, K_max=10,
                                        cosine_decay=True, lp_every=500,
                                        no_imp_limit=1000, restart_noise=0.01)
            _update(p_f, sc_f, 'final_polish')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size} sum={s:.6f}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
            else:
                print(f"EVAL: {eval_sc} (inf/nan) - using fallback")
        except Exception as e:
            print(f"EVAL error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
