Hybrid peak-reduction optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines GR-style iterative peak correction, Adam in logit space,
simulated annealing, spectral recombination, and multi-resolution search.

import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_gradient_weights(c, a, beta=0.05, n_top=5):
    """Compute gradient weights for the max of convolution.
    Uses a blend of softmax and top-k for better peak targeting."""
    n = len(a)
    conv_len = 2 * n - 1
    mval = float(c.max())

    if beta < 1e-14:
        # Hard top-k
        w = np.zeros(conv_len)
        k = min(n_top, conv_len)
        top_idx = np.argpartition(c, -k)[-k:]
        w[top_idx] = 1.0 / k
        return mval, w

    # Softmax weights
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros(conv_len)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w_soft = ex / s

    # Top-k weights
    k = min(n_top, conv_len)
    w_top = np.zeros(conv_len)
    top_idx = np.argpartition(c, -k)[-k:]
    # Weight by excess above median
    med = float(np.median(c))
    excess = np.maximum(c[top_idx] - med, 0.0)
    if excess.sum() > 1e-15:
        w_top[top_idx] = excess / excess.sum()
    else:
        w_top[top_idx] = 1.0 / k

    # Blend
    alpha = 0.5  # 50/50 blend
    w = alpha * w_soft + (1.0 - alpha) * w_top
    return mval, w


def compute_h(w, a):
    """h[i] = sum_{m} a[m]*w[i+m] for i=0,...,n-1 via FFT.
    This gives d(max conv)/da[i]."""
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        n_new_max = np.ones(n_new) / n_new
        return n_new_max
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 1e-10)
            profiles.append(a / a.sum())

    # Gaussian bumps
    for width in [0.2, 0.4, 0.7]:
        for offset in [0.1, 0.3, 0.5, 0.7, 0.9]:
            a = np.exp(-((t - offset) / width) ** 2)
            a = np.maximum(a, 1e-10)
            profiles.append(a / a.sum())

    # Power ramps
    for p in [0.5, 1.0, 2.0, 4.0]:
        a = (1.0 - t + 0.05) ** p
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
        a2 = (t + 0.05) ** p
        a2 = np.maximum(a2, 1e-10)
        profiles.append(a2 / a2.sum())

    # Random low-freq
    for _ in range(8):
        coeffs = rng.uniform(-0.5, 0.5, 5)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    return profiles


def gr_peak_correction(a, n_corr=10, lr=0.01):
    """GR-style peak correction: directly reduce the top peaks.
    For each top peak in conv, identify contributing pairs and
    redistribute mass to reduce the peak."""
    n = len(a)
    a = a.copy()
    c = full_convolve_fft(a)
    max_c = float(c.max())

    for _ in range(n_corr):
        # Find top peak
        k_star = int(np.argmax(c))

        # Find contributing pairs: a[j] * a[k_star - j] for valid j
        # In np.convolve indexing: c[k] = sum_j a[j] * a[k-j] where 0<=j<n, 0<=k-j<n
        j_min = max(0, k_star - (n - 1))
        j_max = min(n - 1, k_star)
        js = np.arange(j_min, j_max + 1)
        partners = k_star - js
        products = a[js] * a[partners]

        if len(js) == 0:
            break

        # Find the top contributors
        n_top = max(1, len(js) // 4)
        top_local = np.argpartition(products, -n_top)[-n_top:]
        top_js = js[top_local]
        top_partners = partners[top_local]

        # Reduce mass at top contributors, add to low contributors
        total_reduce = 0.0
        low_js = js[np.array([i for i in range(len(js)) if i not in top_local])]
        if len(low_js) > 0:
            for j in top_js:
                partner = k_star - j
                # Reduce a[j] and a[partner] proportionally
                red_j = a[j] * lr * 0.5
                red_p = a[partner] * lr * 0.5 if partner != j else a[partner] * lr
                a[j] = max(0.0, a[j] - red_j)
                a[partner] = max(0.0, a[partner] - red_p)
                total_reduce += red_j + red_p

            # Distribute to low contributors
            for j in low_js:
                partner = k_star - j
                add_amt = total_reduce / (2.0 * len(low_js))
                a[j] += add_amt
                if partner != j:
                    a[partner] += add_amt
        else:
            # All are top contributors; do a uniform perturbation
            a[js] = a[js] * (1.0 - lr * 0.1)
            a[js[0]] += lr * 0.1 * a[js].sum()

        s = a.sum()
        if s > 0:
            a = a / s
        c = full_convolve_fft(a)

    return a


def adam_logit_optimize(a0, deadline, rng, lr=0.03, beta_factor=0.06,
                        beta_decay=0.93, beta_min=1e-10,
                        restart_period=1500, n_peaks=5,
                        use_gr=False, gr_every=50):
    """Adam optimization in logit space with adaptive softmax and GR correction."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())

    z = np.log(a0)
    z = z - z.mean()

    beta = beta_factor * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m_mom = np.zeros(n)
    v_mom = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr
    base_lr = lr
    iter_count = 0

    while time.monotonic() < deadline:
        # Compute gradient
        _, w = compute_gradient_weights(c, a0, beta=beta, n_top=n_peaks)
        h = compute_h(w, a0)
        grad_a = 2.0 * h

        # Project to tangent space of simplex (in logit space)
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # Adam update
        t_adam += 1
        m_mom = b1 * m_mom + (1.0 - b1) * dz
        v_mom = b2 * v_mom + (1.0 - b2) * (dz * dz)
        m_hat = m_mom / (1.0 - b1 ** t_adam)
        v_hat = v_mom / (1.0 - b2 ** t_adam)
        z = z - current_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Stability
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            a0 = best_a.copy()
            m_mom.fill(0.0)
            v_mom.fill(0.0)
            t_adam = 0
            no_improve += 1
            continue

        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m_mom.fill(0.0)
            v_mom.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        # Optional GR correction
        if use_gr and iter_count % gr_every == 0 and iter_count > 0:
            a0 = gr_peak_correction(a0, n_corr=3, lr=0.005)
            z = np.log(np.maximum(a0, 1e-15))
            z = z - z.mean()

        # Evaluate
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        # Adapt beta
        if t_adam % 30 == 0:
            beta = max(beta_min, beta * beta_decay)

        # Adapt learning rate
        if no_improve > 200:
            current_lr = max(0.003, current_lr * 0.98)
        elif no_improve % 100 == 0 and no_improve < 200:
            current_lr = max(0.005, current_lr * 0.995)

        # Restart
        if no_improve > restart_period:
            # Diverse restart
            rtype = rng.integers(0, 5)
            if rtype == 0:
                # Gaussian noise
                z = best_z + 0.15 * rng.normal(size=n)
            elif rtype == 1:
                # Structured perturbation: add low-freq sinusoid
                freq = rng.integers(1, 5)
                amp = 0.1 * rng.random()
                phase = rng.uniform(0, 2 * np.pi)
                t = np.linspace(0, 1, n)
                pert = amp * np.cos(2 * np.pi * freq * t + phase)
                z = best_z + pert
            elif rtype == 2:
                # Swap-based perturbation
                z_new = best_z.copy()
                n_swaps = rng.integers(n // 10, n // 3) + 1
                idx1 = rng.integers(0, n, n_swaps)
                idx2 = rng.integers(0, n, n_swaps)
                z_new[idx1], z_new[idx2] = z_new[idx2], z_new[idx1]
                z = z_new + 0.05 * rng.normal(size=n)
            elif rtype == 3:
                # Mix with a new random profile
                t = np.linspace(0, 1, n)
                mix = 0.15 + 0.2 * rng.random()
                rand_profile = 1.0 + 0.5 * rng.normal() * np.cos(2 * np.pi * t)
                rand_profile = np.maximum(rand_profile, 0.1)
                rand_z = np.log(rand_profile)
                rand_z = rand_z - rand_z.mean()
                z = (1.0 - mix) * best_z + mix * rand_z
            else:
                # Temperature bump: increase beta temporarily
                z = best_z + 0.2 * rng.normal(size=n)
                beta = max(0.02 * max_c, beta * 5)

            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(beta_min, beta_factor * max_c)
            m_mom.fill(0.0)
            v_mom.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0

    return best_a, best_score


def anneal_optimize(a0, deadline, rng, n_sweeps=50, cooling=0.97):
    """Simulated annealing: accept worse solutions with probability exp(-delta/T)."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    current_score = float(2.0 * n * c.max())
    best_a = a0.copy()
    best_score = current_score

    temperature = 0.5 * current_score
    t_arr = 1.0 / temperature

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        # Generate perturbation
        pert_type = rng.integers(0, 4)
        a_try = a0.copy()

        if pert_type == 0:
            # Random pair swap with scaling
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = a0[i] * rng.uniform(-0.3, 0.3)
                a_try[i] = max(0, a0[i] - delta)
                a_try[j] = max(0, a0[j] + delta)
        elif pert_type == 1:
            # Local smoothing
            i = rng.integers(1, n - 1)
            w = rng.uniform(0.1, 0.5)
            a_try[i] = (1 - w) * a0[i] + w * 0.5 * (a0[i-1] + a0[i+1])
        elif pert_type == 2:
            # Small random perturbation
            n_pert = rng.integers(1, max(2, n // 20))
            idx = rng.integers(0, n, n_pert)
            scale = rng.uniform(0.01, 0.1)
            a_try[idx] = np.maximum(0, a0[idx] * (1 + scale * rng.normal(size=n_pert)))
        else:
            # Peak-targeted reduction
            c_tmp = full_convolve_fft(a0)
            k_star = int(np.argmax(c_tmp))
            j_min = max(0, k_star - (n - 1))
            j_max = min(n - 1, k_star)
            js = np.arange(j_min, j_max + 1)
            if len(js) > 0:
                products = a0[js] * a0[k_star - js]
                top_j = js[np.argmax(products)]
                partner = k_star - top_j
                red = a0[top_j] * rng.uniform(0.1, 0.3)
                a_try[top_j] = max(0, a0[top_j] - red)
                if partner != top_j:
                    a_try[partner] = max(0, a0[partner] - red * 0.5)
                # Redistribute
                low_js = js[js != top_j]
                if len(low_js) > 0:
                    target_j = low_js[rng.integers(0, len(low_js))]
                    a_try[target_j] += red

        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s

        c_try = full_convolve_fft(a_try)
        try_score = float(2.0 * n * c_try.max())

        if try_score < current_score:
            current_score = try_score
            a0 = a_try.copy()
            if try_score < best_score:
                best_score = try_score
                best_a = a_try.copy()
        else:
            # Metropolis acceptance
            delta = try_score - current_score
            if np.log(rng.random()) < -delta * t_arr:
                current_score = try_score
                a0 = a_try.copy()

        # Cool
        temperature *= cooling
        t_arr = 1.0 / max(temperature, 1e-10)

        # Record best
        if try_score < best_score:
            best_score = try_score
            best_a = a_try.copy()

    return best_a, best_score


def spectral_recombine(a1, a2, rng, n):
    """Recombine two sequences in frequency domain for diversity."""
    N = next_pow2(2 * n - 1)
    f1 = np.fft.rfft(a1, N)
    f2 = np.fft.rfft(a2, N)

    # Random phase mixing
    phase = rng.uniform(0, 2 * np.pi, size=len(f1))
    mix = 0.3 + 0.4 * rng.random()
    f_mix = mix * f1 * np.exp(1j * phase) + (1 - mix) * f2

    a_mix = np.fft.irfft(f_mix, N)[:n]
    a_mix = np.maximum(a_mix, 0.0)
    s = a_mix.sum()
    if s > 1e-10:
        a_mix = a_mix / s
    else:
        a_mix = interpolate_profile(a1, n)
    return a_mix


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid search combining Adam logit optimization,
    GR peak correction, simulated annealing, and spectral recombination.
    Explores multiple sequence lengths and maintains a solution portfolio.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    portfolio = []  # List of (score, sequence)

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: n={p.size}, score={sc:.10f} at t={time.monotonic()-t_start:.1f}s")

    def add_to_portfolio(p, sc):
        nonlocal best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        portfolio.append((sc, np.asarray(p, dtype=np.float64).copy()))
        portfolio.sort(key=lambda x: x[0])
        if len(portfolio) > 8:
            portfolio = portfolio[:8]

    # Load initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting hybrid search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)
        add_to_portfolio(init_seq, sc_init)

    # ===== Phase 1: Quick exploration at n=512 =====
    n1 = 512
    phase1_deadline = min(deadline, t_start + 200.0)
    print(f"\n=== Phase 1: n={n1} (exploration) ===")

    if init_seq is not None:
        p1_base = interpolate_profile(init_seq, n1)
    else:
        p1_base = make_initial_profiles(n1, rng)[0]

    # Optimize the base
    dl = min(phase1_deadline, time.monotonic() + 80.0)
    r1, sc1 = adam_logit_optimize(p1_base, dl, rng, lr=0.04,
                                   beta_factor=0.08, beta_decay=0.92,
                                   restart_period=800, n_peaks=5)
    sc1_final = compute_score(r1)
    update(r1, sc1_final)
    add_to_portfolio(r1, sc1_final)

    # Try a few diverse profiles
    profiles_1 = make_initial_profiles(n1, rng)
    for idx, p in enumerate(profiles_1[:8]):
        if time.monotonic() > phase1_deadline - 5:
            break
        sc_p = compute_score(p)
        if sc_p > best_score + 0.1:
            continue
        dl = min(phase1_deadline, time.monotonic() + 15.0)
        r, sc_r = adam_logit_optimize(p, dl, rng, lr=0.05,
                                       beta_factor=0.1, beta_decay=0.90,
                                       restart_period=400)
        sc_f = compute_score(r)
        update(r, sc_f)
        if sc_f < best_score + 0.05:
            add_to_portfolio(r, sc_f)
    print(f"Phase 1 best: {best_score:.10f}")

    # ===== Phase 2: Main optimization at n=1024 =====
    n2 = 1024
    phase2_deadline = min(deadline, t_start + 800.0)
    print(f"\n=== Phase 2: n={n2} (main optimization) ===")

    if best_seq is not None:
        p2_base = interpolate_profile(best_seq, n2)
    else:
        p2_base = interpolate_profile(p1_base, n2)

    sc2_init = compute_score(p2_base)
    update(p2_base, sc2_init)
    add_to_portfolio(p2_base, sc2_init)

    # Main Adam run with GR
    dl2a = min(phase2_deadline, time.monotonic() + 300.0)
    r2, sc2 = adam_logit_optimize(p2_base, dl2a, rng, lr=0.025,
                                   beta_factor=0.04, beta_decay=0.94,
                                   restart_period=1500, n_peaks=3,
                                   use_gr=True, gr_every=40)
    sc2_final = compute_score(r2)
    update(r2, sc2_final)
    add_to_portfolio(r2, sc2_final)
    print(f"  After main Adam+GR: {sc2_final:.10f}")

    # Annealing refinement
    if time.monotonic() < phase2_deadline - 60:
        dl2b = min(phase2_deadline, time.monotonic() + 120.0)
        r2a, sc2a = anneal_optimize(r2, dl2b, rng, n_sweeps=200, cooling=0.96)
        sc2a_final = compute_score(r2a)
        update(r2a, sc2a_final)
        add_to_portfolio(r2a, sc2a_final)
        print(f"  After annealing: {sc2a_final:.10f}")
        if sc2a_final < sc2_final:
            r2 = r2a

    # Spectral recombination with portfolio
    if time.monotonic() < phase2_deadline - 30 and len(portfolio) >= 2:
        scores_port = [p[0] for p in portfolio[:4]]
        seqs_port = [p[1] for p in portfolio[:4]]
        for i in range(min(3, len(seqs_port))):
            if time.monotonic() > phase2_deadline - 10:
                break
            a_rec = spectral_recombine(seqs_port[0], seqs_port[i+1], rng, n2)
            dl_rec = min(phase2_deadline, time.monotonic() + 30.0)
            r_rec, sc_rec = adam_logit_optimize(a_rec, dl_rec, rng, lr=0.02,
                                                 beta_factor=0.03, beta_decay=0.95)
            sc_rec_f = compute_score(r_rec)
            update(r_rec, sc_rec_f)
            add_to_portfolio(r_rec, sc_rec_f)
            print(f"  Recombination {i}: {sc_rec_f:.10f}")

    print(f"Phase 2 best: {best_score:.10f}")

    # ===== Phase 3: Large n (2048, 2560, 3072) =====
    phase3_deadline = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 3: Large n ===")

    for n3 in [2048, 2560, 3072]:
        if time.monotonic() > phase3_deadline - 100:
            break
        print(f"\n  --- n={n3} ---")
        if best_seq is not None:
            p3 = interpolate_profile(best_seq, n3)
        else:
            p3 = interpolate_profile(p2_base, n3)

        sc3_init = compute_score(p3)
        update(p3, sc3_init)
        add_to_portfolio(p3, sc3_init)

        # Determine time budget for this n
        time_left_for_opt = min(phase3_deadline, time.monotonic() + 250.0) - time.monotonic()
        if time_left_for_opt < 30:
            time_left_for_opt = max(30, (phase3_deadline - time.monotonic()) / 3)

        r3, sc3 = adam_logit_optimize(p3, time.monotonic() + time_left_for_opt * 0.6,
                                       rng, lr=0.015, beta_factor=0.025,
                                       beta_decay=0.95, restart_period=2000,
                                       n_peaks=5, use_gr=True, gr_every=60)
        sc3_final = compute_score(r3)
        update(r3, sc3_final)
        add_to_portfolio(r3, sc3_final)
        print(f"  After Adam at n={n3}: {sc3_final:.10f}")

        # Annealing
        if time.monotonic() < phase3_deadline - 50:
            dl_a = min(phase3_deadline, time.monotonic() + 80.0)
            r3a, sc3a = anneal_optimize(r3, dl_a, rng, n_sweeps=150, cooling=0.96)
            sc3a_final = compute_score(r3a)
            update(r3a, sc3a_final)
            add_to_portfolio(r3a, sc3a_final)
            print(f"  After annealing at n={n3}: {sc3a_final:.10f}")
            if sc3a_final < sc3_final:
                r3 = r3a

    print(f"Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Also try n=768 which might be optimal =====
    if time.monotonic() < deadline - 100:
        n_alt = 768
        print(f"\n=== Try n={n_alt} ===")
        if best_seq is not None:
            p_alt = interpolate_profile(best_seq, n_alt)
        else:
            p_alt = interpolate_profile(p2_base, n_alt)
        sc_alt_init = compute_score(p_alt)
        update(p_alt, sc_alt_init)

        dl_alt = min(deadline - 50, time.monotonic() + 150.0)
        r_alt, sc_alt = adam_logit_optimize(p_alt, dl_alt, rng, lr=0.02,
                                             beta_factor=0.03, beta_decay=0.94,
                                             restart_period=1000, n_peaks=3)
        sc_alt_final = compute_score(r_alt)
        update(r_alt, sc_alt_final)
        print(f"  n={n_alt} result: {sc_alt_final:.10f}")

    # ===== Phase 5: Final intensive polish =====
    phase5_deadline = deadline
    print(f"\n=== Phase 5: Final polish ===")

    if best_seq is not None:
        n_pol = best_seq.size
        a_pol = best_seq.copy()

        # Multiple polish rounds with decreasing lr
        polish_configs = [
            (0.012, 0.02, 0.96, 5, False),
            (0.008, 0.015, 0.97, 3, True),
            (0.005, 0.01, 0.98, 3, False),
            (0.003, 0.008, 0.99, 2, True),
        ]

        for round_idx, (lr_val, beta_f, decay, n_pk, gr) in enumerate(polish_configs):
            if time.monotonic() > phase5_deadline - 20:
                break
            # Small perturbation to escape exact local minimum
            a_try = a_pol + 0.001 * rng.normal(size=n_pol)
            a_try = np.maximum(a_try, 1e-12)
            a_try = a_try / a_try.sum()

            time_per_round = max(20, (phase5_deadline - time.monotonic()) / (len(polish_configs) - round_idx))
            dl = min(phase5_deadline - 10, time.monotonic() + time_per_round)

            r, sc_r = adam_logit_optimize(a_try, dl, rng, lr=lr_val,
                                           beta_factor=beta_f, beta_decay=decay,
                                           restart_period=3000, n_peaks=n_pk,
                                           use_gr=gr, gr_every=80)
            sc_f = compute_score(r)
            update(r, sc_f)
            add_to_portfolio(r, sc_f)
            print(f"  Polish round {round_idx} (lr={lr_val}, gr={gr}): {sc_f:.10f}")
            if sc_f < best_score + 1e-12:
                a_pol = r

        # Final GR correction pass
        if time.monotonic() < phase5_deadline - 10:
            a_gr = gr_peak_correction(a_pol, n_corr=20, lr=0.003)
            sc_gr = compute_score(a_gr)
            update(a_gr, sc_gr)
            if sc_gr < best_score + 1e-12:
                a_pol = a_gr
            print(f"  Final GR: {compute_score(a_pol):.10f}")

        # Final tiny anneal
        if time.monotonic() < deadline - 5:
            dl_fin = min(deadline - 2, time.monotonic() + 20.0)
            r_fin, sc_fin = anneal_optimize(a_pol, dl_fin, rng,
                                             n_sweeps=50, cooling=0.95)
            sc_fin_f = compute_score(r_fin)
            update(r_fin, sc_fin_f)
            print(f"  Final anneal: {sc_fin_f:.10f}")

    # Try smaller n values one more time
    if best_seq is not None and best_seq.size > 512:
        for n_try in [512, 768, 1024]:
            if time.monotonic() > deadline - 10:
                break
            p_s = interpolate_profile(best_seq, n_try)
            sc_s = compute_score(p_s)
            update(p_s, sc_s)
            if sc_s < best_score + 0.01 and time.monotonic() < deadline - 10:
                dl_s = min(deadline - 5, time.monotonic() + 20.0)
                r_s, _ = adam_logit_optimize(p_s, dl_s, rng, lr=0.015,
                                              beta_factor=0.02, beta_decay=0.96)
                sc_s2 = compute_score(r_s)
                update(r_s, sc_s2)
                print(f"  Smaller n={n_try}: {sc_s2:.10f}")

    # ===== Final validation and return =====
    print(f"\n  Overall best score: {best_score:.10f}")
    print(f"  Portfolio size: {len(portfolio)}")
    for i, (sc, seq) in enumerate(portfolio[:5]):
        print(f"    #{i}: n={seq.size}, score={sc:.10f}")

    if best_seq is not None:
        final_seq = best_seq
    else:
        t = np.linspace(0, 1, 512)
        final_seq = 1.0 + 1.5 * (0.5 - t)
        final_seq = np.maximum(final_seq, 1e-10)
        final_seq = final_seq / final_seq.sum()

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    final_score = compute_score(final_seq)
    print(f"\nFINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        else:
            print("WARNING: evaluate_sequence returned non-finite!")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check error: {e}")

    return [float(x) for x in final_seq]
