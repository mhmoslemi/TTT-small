    # Let me just use the direct formula:
    # grad_j proportional to sum_k c_k^(p-1) * a_{k-j} (with appropriate bounds)
    # This is the cross-correlation of c^(p-1) and a, evaluated at lag j.
    # cross_corr[j] = sum_k c[k] * a[k+j] (for j >= 0) or sum_k c[k] * a[k-j] (for j < 0)
    # 
    # Actually, let's define: for j in [0, n-1], 
    #   grad_j = sum_{k: 0<=k<2n-1, 0<=k-j<n} c_k^(p-1) * a[k-j]
    # This is (conv(c^(p-1), a))[j + something]... 
    
    # Simplest approach: use the same trick as the LSE gradient
    # The gradient of max_k conv(a,a)_k at the max is 2*conv(delta, reverse(a)) centered
    # For power sum, replace delta with weights = c^(p-1) / sum(c^(p-1))
    
    # Actually, let me just use the LSE-style gradient with weights proportional to c^(p-1):
    norm_w = float(weights.sum())
    if norm_w < 1e-30:
        return np.zeros(n)
    w = weights / norm_w
    
    # grad_j = 2 * sum_k w[k] * a[k-j] (for valid indices)
    # This is 2 * conv(w, reverse(a))[n-1 : 2n-1] ... let me verify
    # conv(w, reverse(a))[m] = sum_i w[i] * reverse(a)[m-i] = sum_i w[i] * a[n-1-m+i]
    # For m = n-1+j: sum_i w[i] * a[n-1-n+1+j+i] = sum_i w[i] * a[j+i]
    # That's not quite what we want.
    
    # Let me use: grad_j = 2 * (w * a reversed correlation)
    # sum_k w[k] * a[k-j] = cross-correlation at lag j
    # = ifft(fft(w) * conj(fft(a))) at index j (circular)
    # But our sequences are shorter than the FFT length, so circular = linear for small lags.
    
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # Cross-correlation: ifft(conj(FW) * FA) gives sum_k w[k-j] * a[k] = we want sum_k w[k] * a[k-j]
    # = ifft(FW * conj(FA)) at index -j = ifft(FW * conj(FA)) at index N-j (for j > 0)
    # Actually: ifft(FW * conj(FA))[j] = sum_k w[k] * a[k-j] (circular)
    # Let me verify: FW[k] = sum_i w[i] exp(-2pi i k i/N)
    # conj(FA)[k] = sum_j a[j] exp(2pi i k j/N)
    # FW[k] * conj(FA)[k] = sum_i sum_j w[i] a[j] exp(2pi i k (j-i)/N)
    # ifft of that at position m: (1/N) sum_k sum_i sum_j w[i] a[j] exp(2pi i k (j-i)/N) exp(2pi i k m/N)
    # = (1/N) sum_i sum_j w[i] a[j] sum_k exp(2pi i k (j-i+m)/N)
    # = sum_i sum_j w[i] a[j] delta(j-i+m mod N)
    # = sum_i w[i] a[i-m mod N]
    # 
    # So ifft(FW * conj(FA))[m] = sum_i w[i] * a[(i-m) mod N]
    # For m = 0: sum_i w[i] * a[i] (circular)
    # For m = j (small): sum_i w[i] * a[i-j] (circular, but if N > 2n-1, no wraparound for valid ranges)
    
    # So grad_j = 2 * ifft(FW * conj(FA))[j] for j = 0, ..., n-1
    
    prod = fw * np.conj(fa)
    corr = np.fft.irfft(prod, N)
    grad = 2.0 * corr[:n]
    
    # Scale factor
    scale = float(np.sum(weights) ** (1.0/p - 1.0))
    grad *= scale
    
    return grad

def _lse_gradient_step(a, T, lr, state):
    """One Adam step using log-sum-exp gradient in log-space."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    prod = fw * np.conj(fa)
    corr = np.fft.irfft(prod, N)
    grad = 2.0 * corr[:n]

    # Log-space gradient
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, state

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    z = state['z'] - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -40.0, 40.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    state['z'] = z

    return a_new, state

def _lse_optimize(a, deadline, max_iter=100000, T_start=0.03, T_end=1e-5, lr=0.08, restart_interval=5000):
    """Optimize using log-sum-exp gradient with Adam in log-space."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n), 'z': z}

    best_a = a.copy()
    best_score = _score_normalized(a)
    T = T_start
    no_imp = 0
    base_lr = lr

    if max_iter > 1:
        T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    else:
        T_ratio = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a, state = _lse_gradient_step(a, T, lr, state)

        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            state['z'] = np.log(np.maximum(a, 1e-12))
            state['z'] -= float(np.mean(state['z']))
            state['t'] = 0
            state['m'] = np.zeros(n)
            state['v'] = np.zeros(n)
            continue

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.005, lr * 0.92)
            if no_imp > restart_interval:
                rng_r = np.random.default_rng(42 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n), 'z': z}
                lr = base_lr
                no_imp = 0

        T *= T_ratio
        T = max(T, T_end)

    return best_a

def _power_sum_optimize(a, deadline, max_iter=50000, p_start=100, p_end=1000, lr=0.05):
    """Optimize using power-sum gradient with Adam."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = p_start * (p_end / p_start) ** (it / max(max_iter - 1, 1))
        grad = _power_sum_grad(a, p)
        
        if not np.all(np.isfinite(grad)):
            break

        # Log-space gradient
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        t += 1
        b1, b2 = 0.9, 0.999
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a_new = e / e.sum()

        if not np.all(np.isfinite(a_new)):
            break

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                a = best_a.copy()
                z = np.log(a)
                z -= float(np.mean(z))
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = max(0.01, lr * 0.8)
                no_imp = 0

    return best_a

def _make_conv_matrix_reduced(a, M, top_k=80):
    """Create reduced constraint matrix using only near-active constraints."""
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    for row, k in enumerate(active):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        for j in range(jmin, jmax + 1):
            A[row, j] = a[j]
    b = np.full(len(active), M)
    return A, b

def _lp_spread(a, M, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    if n > 64:
        A, b = _make_conv_matrix_reduced(a, M, top_k)
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a[j]
        b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _lp_refine(a, deadline, max_iter=3000, inflate=1.0003, blend=0.15):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            inflate_cur = min(1.01, inflate_cur * 1.3)
            no_imp += 3
            if no_imp > 200:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.05)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        # Line search
        best_t = 0.0
        best_s = best_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, blend_cur]:
            a_try = np.maximum((1.0 - t) * a + t * g, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            s_try_score = _score_normalized(a_try)
            if s_try_score < best_s:
                best_s = s_try_score
                best_t = t

        if best_t < 0.01:
            no_imp += 1
            if no_imp % 20 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
            if no_imp > 300:
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
            continue

        a_new = np.maximum((1.0 - best_t) * a + best_t * g, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

    return best_a

def _kkt_solve(a, tol=1e-6, max_iter=100):
    """
    Try to solve the KKT conditions directly.
    At optimum: for j in support(a), sum_{k in S} a[k-j] / |S| = constant
    where S is the set of active lags.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    for _ in range(max_iter):
        c = _fft_convolve(a)
        M = float(c.max())
        # Active set: lags where conv is within tol of max
        active_idx = np.where(c >= M - tol * M)[0]
        S = len(active_idx)
        if S == 0:
            break
        
        # Compute the "average column" vector
        # avg_col[j] = (1/|S|) * sum_{k in active_idx} a[k-j] (for valid k-j)
        avg_col = np.zeros(n)
        for k in active_idx:
            jmin = max(0, k - n + 1)
            jmax = min(n - 1, k)
            for j in range(jmin, jmax + 1):
                avg_col[j] += a[j]  # a[k-j] where j is the index in a... 
            # Wait, I need to be more careful
            # conv_k = sum_i a[i] * a[k-i]
            # d conv_k / d a[j] = a[k-j] (if valid)
            # So the "gradient contribution" from lag k to position j is a[k-j]
            # But this means avg_col[j] = (1/|S|) * sum_{k in active} a[k-j]
            # Which is the same as the cross-correlation of "indicator of active set" with a, at lag j
        # Let me redo this more carefully
        avg_col = np.zeros(n)
        for k in active_idx:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    avg_col[j] += a[idx]
        avg_col /= S
        
        # At the KKT point, avg_col should be constant on the support of a
        # The support is where a > 0
        support = a > 1e-10
        if not np.any(support):
            break
        
        # Try to adjust a so that avg_col is constant on support
        # This is a fixed-point iteration:
        # New a[j] proportional to 1/avg_col[j] for j in support (to flatten avg_col)
        # Or: a[j] = a[j] * (mean(avg_col[support]) / avg_col[j])
        
        mean_ac = float(np.mean(avg_col[support]))
        if mean_ac < 1e-20:
            break
        
        # Damped update
        a_new = a.copy()
        a_new[support] = a[support] * np.clip(mean_ac / np.maximum(avg_col[support], 1e-20), 0.5, 2.0)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-10:
            a = a_new
            break
        a = a_new
    
    return a

def _simulated_perturb(a, n_perturbs=50, deadline=None):
    """Try random perturbations and keep improvements."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score_normalized(a)
    rng = np.random.default_rng(12345)
    
    for i in range(n_perturbs):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Random perturbation
        scale = 0.02 * (1.0 + rng.random())
        a_pert = a * (1.0 + scale * rng.normal(size=n))
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s
        
        sc = _score_normalized(a_pert)
        if sc < best_score:
            best_score = sc
            best_a = a_pert.copy()
            a = a_pert
    
    return best_a

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
