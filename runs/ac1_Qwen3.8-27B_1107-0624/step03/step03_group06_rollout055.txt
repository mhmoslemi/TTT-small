import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    if res.success:
        return res.x
    return None


def _line_search_parabolic(a, g, max_evals=15):
    best_t = 0.0
    best_sc = _score(a)
    t = 1.0
    prev_t = None
    prev_sc = None
    prev2_t = None
    prev2_sc = None

    for _ in range(max_evals):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_new = a_new / s
        sc = _score(a_new)

        if sc < best_sc:
            best_sc = sc
            best_t = t
        else:
            # Try smaller step
            t *= 0.5
            if t < 0.001:
                break
            continue

        # Try to extrapolate (parabolic)
        if prev_t is not None and prev2_t is not None:
            d1 = prev_sc - sc
            d2 = prev2_sc - prev_sc
            if d1 > 0 and d2 > 0:
                num = 0.5 * (d1 * (prev_t - t) ** 2 - d2 * (prev2_t - t) ** 2)
                den = d1 * (prev2_t - prev_t) - d2 * (prev2_t - prev_t)
                # Skip parabolic for stability, just use geometric
            prev2_t = prev_t
            prev2_sc = prev_sc
        prev_t = t
        prev_sc = sc

        # Keep it simple: geometric decrease always
        t *= 0.5
        if t < 0.001:
            break

    return best_t, best_sc


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.005, inflate_min=1.0002):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start
    consecutive_no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            consecutive_no_imp += 1
            if consecutive_no_imp > 100:
                inflate = min(0.05, inflate * 1.5)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                no_imp = 0
                consecutive_no_imp = 0
                inflate = inflate_start
                continue
            if no_imp % 50 == 0:
                inflate = min(0.05, inflate * 1.1)
            continue

        g = g / sum_g

        best_t, best_sc = _line_search_parabolic(a, g)

        if best_t > 0 and best_sc < best_score - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            consecutive_no_imp = 0
            # Gradually tighten inflation
            if it % 30 == 29:
                inflate = max(inflate_min, inflate * 0.93)
        else:
            no_imp += 1
            consecutive_no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.03, inflate * 1.08)

    return best_a


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=50000):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.03 * float(c.max()))
    lr = 0.05
    base_lr = lr
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_moment = a + momentum * v
        if a_moment.sum() < 1e-10:
            v = np.zeros(n)
            continue
        grad, maxc = _smooth_max_grad(a_moment, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 50 == 49:
            beta = max(1e-8, beta * 0.85)

    return best_a


def _random_perturb(a, rng, strength=0.1):
    n = len(a)
    noise = rng.normal(0, strength, n)
    a_new = a * (1.0 + noise)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    LP-based optimization with adaptive inflation, parabolic line search,
    multiple sequence lengths, gradient polish, and random restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 60.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate initializations
    configs = []
    lengths_to_try = [512, 768, 1024, 1536, 2048]

    for n_t in lengths_to_try:
        if parent is not None:
            configs.append(('parent_interp', n_t, _interpolate(parent, n_t)))

        # Diverse profiles
        configs.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp2', n_t, _make_ramp(n_t, 2.0)))
        configs.append(('ramp3', n_t, _make_ramp(n_t, 3.0)))
        configs.append(('ramp_half', n_t, _make_ramp(n_t, 0.5)))
        configs.append(('u2', n_t, _make_u_shape(n_t, 2.0)))
        configs.append(('u3', n_t, _make_u_shape(n_t, 3.0)))
        configs.append(('u4', n_t, _make_u_shape(n_t, 4.0)))
        configs.append(('bell', n_t, _make_bell(n_t, 0.3)))
        configs.append(('bell_wide', n_t, _make_bell(n_t, 0.5)))

        # Random perturbations of parent (if available)
        if parent is not None:
            configs.append(('perturb1', n_t, _random_perturb(_interpolate(parent, n_t), rng, 0.15)))
            configs.append(('perturb2', n_t, _random_perturb(_interpolate(parent, n_t), rng, 0.3)))

    # Score and sort
    scored_configs = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored_configs.append((sc, name, n_t, a0))
    scored_configs.sort(key=lambda x: x[0])

    print(f"Top 5 initializations:")
    for i, (sc, name, n_t, _) in enumerate(scored_configs[:5]):
        print(f"  {name} n={n_t}: {sc:.8f}")

    # Phase 1: LP refinement on top initializations
    used_names = set()
    for sc0, name, n_t, a0 in scored_configs:
        if time.monotonic() > deadline - 200:
            break
        if sc0 > best_score + 0.05:
            continue

        # Skip duplicate lengths+profiles
        key = (name.split('_')[0], n_t)
        if key in used_names and (name == 'parent_interp' or 'perturb' in name):
            continue
        used_names.add(key)

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        remaining = deadline - time.monotonic() - 150
        if remaining < 30:
            break

        # Allocate time: give more to better and longer sequences
        if n_t >= 1024:
            lp_time = min(300.0, remaining * 0.4)
        elif n_t >= 768:
            lp_time = min(200.0, remaining * 0.35)
        else:
            lp_time = min(120.0, remaining * 0.25)

        if lp_time < 10:
            break

        lp_end = time.monotonic() + lp_time
        print(f"LP: {name} n={n_t} init={sc0:.8f} budget={lp_time:.0f}s")

        a_lp = _lp_refine(a0, lp_end, max_iter=30000, inflate_start=1.004, inflate_min=1.0002)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  -> LP score={sc_lp:.10f}")
        else:
            print(f"  -> LP score={sc_lp:.8f} (no improvement over {best_score:.8f})")

    # Phase 2: Second LP pass on the best with tighter parameters
    if best_seq is not None and time.monotonic() < deadline - 100:
        gp_time = min(250.0, deadline - time.monotonic() - 80)
        if gp_time > 15:
            gp_end = time.monotonic() + gp_time
            print(f"Phase 2 LP (tight): budget={gp_time:.0f}s, current best={best_score:.10f}")
            a_p2 = _lp_refine(best_seq, gp_end, max_iter=40000,
                              inflate_start=1.002, inflate_min=1.0001)
            sc_p2 = _score(a_p2)
            if np.isfinite(sc_p2) and sc_p2 < best_score:
                best_score = sc_p2
                best_seq = a_p2.copy()
            print(f"  -> Phase 2 score={sc_p2:.10f}")

    # Phase 3: Gradient polish
    if best_seq is not None and time.monotonic() < deadline - 60:
        gp_time = min(150.0, deadline - time.monotonic() - 40)
        if gp_time > 10:
            gp_end = time.monotonic() + gp_time
            print(f"Gradient polish: budget={gp_time:.0f}s")
            a_gp = _gradient_polish(best_seq, gp_end, max_iter=40000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  -> Gradient score={sc_gp:.10f}")

    # Phase 4: Another LP pass after gradient (to escape gradient's local min)
    if best_seq is not None and time.monotonic() < deadline - 40:
        lp_time = min(120.0, deadline - time.monotonic() - 30)
        if lp_time > 10:
            lp_end = time.monotonic() + lp_time
            print(f"Phase 4 LP (post-gradient): budget={lp_time:.0f}s")
            a_p4 = _lp_refine(best_seq, lp_end, max_iter=20000,
                              inflate_start=1.003, inflate_min=1.0001)
            sc_p4 = _score(a_p4)
            if np.isfinite(sc_p4) and sc_p4 < best_score:
                best_score = sc_p4
                best_seq = a_p4.copy()
            print(f"  -> Phase 4 score={sc_p4:.10f}")

    # Phase 5: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        gp_time = min(80.0, deadline - time.monotonic() - 10)
        if gp_time > 5:
            gp_end = time.monotonic() + gp_time
            a_g = _gradient_polish(best_seq, gp_end, max_iter=20000)
            sc_g = _score(a_g)
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            print(f"  -> Final gradient score={sc_g:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(512, 1.0)

    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
