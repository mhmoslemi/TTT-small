"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth-max Adam, adaptive LP spreading, coordinate descent, and peak equalization.
Balances exploration (diverse inits) and exploitation (fine polishing) with strict time management.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _adam_optimize(a0, deadline, max_iter=50000, lr=0.03, beta_factor=0.05,
                   beta_decay=0.80, restart_patience=5000, noise_scale=0.2):

    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -70.0, 70.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

            best_z = z_new.copy()
            no_imp = 0
        else:

            if no_imp % 500 == 0:
                lr_cur = max(0.0005, lr_cur * 0.92)
            if no_imp > restart_patience:
                rng_r = np.random.default_rng(7000 + it)
                z = best_z + noise_scale * rng_r.normal(size=n)
                z = np.clip(z, -70.0, 70.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0

        z = z_new
        a = a_new

        if it % 200 == 0:
            beta = max(1e-14, beta * beta_decay)

def _multi_subgradient(a0, deadline, max_iter=10000, top_k=5):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.25 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]

        max_val = float(c_curr[top_indices[0]])
        # Weight peaks by their relative height
        rel_heights = c_curr[top_indices] / max_val
        weights = rel_heights ** 2
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(4000, 5 * n):
                break

        a = a_new

    return best_a, best_score


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=2000, inflate=1.0005, blend=0.3):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-10:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.005, inflate_cur * 1.03)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.03, blend_cur * 0.85)
                inflate_cur = min(1.002, inflate_cur * 0.9)
            if no_imp > max(150, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _coordinate_descent(a0, deadline, max_sweeps=50, n_try=5):
    """Try small perturbations of individual elements to reduce max convolution."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        # Random subset of indices for diversity
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-10:
                continue

            # Try several fractions of the current value
            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[i] = orig * frac
                # Renormalize
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    continue
                a = a / s

                c = _fft_convolve(a)
                sc = float(2.0 * n * c.max())

                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = orig
                    a = a / a.sum()

        if not improved and sweep > 5:
            break

    return best_a


def _peak_equalization_adam(a0, deadline, max_iter=20000, lr=0.01,
                            beta_factor=0.02, top_k=5):
    """Adam optimization with weights focused on top-k peaks for equalization."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    z = np.log(np.maximum(a, 1e-14))
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-12)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        mval = float(c_curr.max())

        # Compute weights: emphasize top peaks more
        x = (c_curr - mval) / beta
        x = np.clip(x, -60.0, 60.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-10 or not np.isfinite(s_ex):
            ex = np.zeros_like(c_curr)
            ex[int(np.argmax(c_curr))] = 1.0
            s_ex = 1.0
        w = ex / s_ex

        grad_a = _grad_smooth_max(a, w)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -70.0, 70.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                beta = max(1e-14, beta * 0.7)
            if no_imp > 8000:
                break

        z = z_new
        a = a_new

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_profile(n, kind="linear", **kwargs):
    t = np.linspace(0.0, 1.0, n)
    if kind == "linear":
        eta = kwargs.get("eta", 1.5)
        a = 1.0 + eta * (t - 0.5)
    elif kind == "quadratic":
        c = kwargs.get("c", 2.0)
        a = 1.0 + c * (t - 0.5) ** 2 - c * 0.25
    elif kind == "cosine":
        alpha = kwargs.get("alpha", 1.0)
        a = 1.0 + alpha * np.cos(2 * np.pi * t)
    elif kind == "bell":
        w = kwargs.get("width", 0.8)
        a = np.exp(-((t - 0.5) ** 2) / (2.0 * w ** 2))
    elif kind == "sine":
        a = np.sin(np.pi * t)
    elif kind == "power":
        p = kwargs.get("p", 2.0)
        a = t ** p + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _quick_pipeline(a0, deadline, label=""):
    """Rapid optimization: Adam + subgradient."""
    a_best = a0.copy()
    c = _fft_convolve(a_best)
    s_best = float(2.0 * len(a_best) * c.max())

    d1 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.6)
    a1, s1 = _adam_optimize(a_best, d1, max_iter=8000, lr=0.03, beta_factor=0.05)
    if s1 < s_best:
        a_best, s_best = a1, s1

    if time.monotonic() < deadline - 5:
        d2 = deadline
        a2, s2 = _multi_subgradient(a_best, d2, max_iter=3000, top_k=3)
        if s2 < s_best:
            a_best, s_best = a2, s2

    return a_best, s_best


def _full_refine(a0, deadline, label=""):
    """Full multi-phase refinement."""
    a_best = a0.copy()
    n = len(a_best)
    c = _fft_convolve(a_best)
    s_best = float(2.0 * n * c.max())

    if label:
        print(f"  {label} start: {s_best:.10f}")

    # Phase A: Peak equalization Adam
    remaining = deadline - time.monotonic()
    if remaining > 30:
        d_a = time.monotonic() + remaining * 0.35
        a_a, s_a = _peak_equalization_adam(a_best, d_a, max_iter=15000, lr=0.015, beta_factor=0.03)
        if s_a < s_best:
            a_best, s_best = a_a, s_a
            print(f"  {label} peak-eq Adam: {s_best:.10f}")

    # Phase B: LP refine
    remaining = deadline - time.monotonic()
    if remaining > 20:
        d_b = time.monotonic() + min(remaining * 0.3, 120)
        a_b = _lp_refine(a_best, d_b, max_iter=1500, inflate=1.0005, blend=0.25)
        s_b = float(2.0 * n * _fft_convolve(a_b).max())
        if s_b < s_best:
            a_best, s_best = a_b, s_b
            print(f"  {label} LP: {s_best:.10f}")

    # Phase C: Fine Adam with very low lr
    remaining = deadline - time.monotonic()
    if remaining > 15:
        d_c = time.monotonic() + remaining * 0.3
        a_c, s_c = _adam_optimize(a_best, d_c, max_iter=20000, lr=0.005, beta_factor=0.01,
                                  beta_decay=0.9, restart_patience=8000, noise_scale=0.1)
        if s_c < s_best:
            a_best, s_best = a_c, s_c
            print(f"  {label} fine Adam: {s_best:.10f}")

    # Phase D: Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 10:
        d_d = time.monotonic() + min(remaining * 0.3, 60)
        a_d = _coordinate_descent(a_best, d_d, max_sweeps=20)
        s_d = float(2.0 * n * _fft_convolve(a_d).max())
        if s_d < s_best:
            a_best, s_best = a_d, s_d
            print(f"  {label} coord descent: {s_best:.10f}")

    # Phase E: Final LP polish
    remaining = deadline - time.monotonic()
    if remaining > 5:
        d_e = time.monotonic() + remaining * 0.5
        a_e = _lp_refine(a_best, d_e, max_iter=500, inflate=1.0002, blend=0.1)
        s_e = float(2.0 * n * _fft_convolve(a_e).max())
        if s_e < s_best:
            a_best, s_best = a_e, s_e
            print(f"  {label} final LP: {s_best:.10f}")

    return a_best, s_best


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initialization, smooth-max Adam, peak equalization,
    LP spreading, multi-peak subgradient, and coordinate descent.
    Time-managed to always return the best sequence found before the deadline.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # --- Prepare candidates at multiple lengths ---
    lengths_to_try = [512, 1024]
    if parent is not None:
        candidates = {}
        for n_try in lengths_to_try:
            p_n = _interpolate_profile(parent, n_try)
            candidates[n_try] = p_n
    else:
        candidates = {512: _make_profile(512, "linear", eta=1.5),
                      1024: _make_profile(1024, "linear", eta=1.5)}

    # --- Phase 1: Quick exploration (first 400s) ---
    print("=== Phase 1: Quick exploration ===")
    p1_deadline = min(deadline, t_start + 400)

    # Try parent at 512
    if parent is not None:
        d = min(p1_deadline - 150, t_start + 200)
        a1, s1 = _quick_pipeline(candidates[512], d, "P1-512")
        if s1 < best_score:
            best_score, best_seq = s1, a1.copy()
        print(f"  Parent@512: {s1:.10f}")

    # Try parent at 1024
    if time.monotonic() < p1_deadline - 100:
        d = min(p1_deadline - 100, time.monotonic() + 150)
        a2, s2 = _quick_pipeline(candidates[1024], d, "P1-1024")
        if s2 < best_score:
            best_score, best_seq = s2, a2.copy()
        print(f"  Parent@1024: {s2:.10f}")

    # Try diverse profiles at 512
    if time.monotonic() < p1_deadline - 50:
        profiles = [
            _make_profile(512, "linear", eta=1.0),
            _make_profile(512, "linear", eta=2.0),
            _make_profile(512, "quadratic", c=2.0),
            _make_profile(512, "cosine", alpha=0.5),
            _make_profile(512, "sine"),
            _make_profile(512, "power", p=1.5),
        ]
        # Perturbed parent
        if parent is not None:
            pp = candidates[512] * (1.0 + 0.1 * rng.normal(size=512))
            pp = np.maximum(pp, 0.0)
            profiles.append(pp / pp.sum())

        for i, a0 in enumerate(profiles):
            if time.monotonic() > p1_deadline - 30:
                break
            remaining = p1_deadline - 30 - time.monotonic()
            per_budget = min(30.0, remaining / max(1, len(profiles) - i))
            if per_budget < 8:
                break
            d = time.monotonic() + per_budget
            a_q, s_q = _quick_pipeline(a0, d, f"P1-prof{i}")
            if s_q < best_score:
                best_score, best_seq = s_q, a_q.copy()
            print(f"  Profile {i}: {s_q:.10f}")

    # --- Phase 2: Focused optimization on best (next 500s) ---
    print(f"=== Phase 2: Focused optimization (best={best_score:.10f}) ===")
    p2_deadline = min(deadline, t_start + 900)

    if best_seq is not None:
        best_n = best_seq.size
        # Work at the best length
        a_work = best_seq.copy()

        # Also try interpolating to other lengths
        alt_length = 1024 if best_n == 512 else 512
        a_alt = _interpolate_profile(best_seq, alt_length)

        # Full refine on primary
        remaining = p2_deadline - time.monotonic()
        if remaining > 50:
            d_primary = time.monotonic() + remaining * 0.55
            a_r1, s_r1 = _full_refine(a_work, d_primary, "P2-primary")
            if s_r1 < best_score:
                best_score, best_seq = s_r1, a_r1.copy()

        # Full refine on alternate length
        remaining = p2_deadline - time.monotonic()
        if remaining > 30:
            d_alt = time.monotonic() + remaining * 0.4
            a_r2, s_r2 = _full_refine(a_alt, d_alt, f"P2-alt{alt_length}")
            if s_r2 < best_score:
                best_score, best_seq = s_r2, a_r2.copy()

    # --- Phase 3: Intensive polishing (next 500s) ---
    print(f"=== Phase 3: Intensive polishing (best={best_score:.10f}) ===")
    p3_deadline = min(deadline, t_start + 1400)

    if best_seq is not None:
        current_n = best_seq.size

        # Multi-restart with tiny perturbations
        for restart in range(3):
            if time.monotonic() > p3_deadline - 50:
                break
            noise = 0.005 * (restart + 1)
            a_pert = best_seq * (1.0 + noise * rng.normal(size=current_n))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()

            remaining = p3_deadline - 50 - time.monotonic()
            per_budget = remaining / (3 - restart)
            d = time.monotonic() + min(per_budget, 150)
            a_r, s_r = _full_refine(a_pert, d, f"P3-restart{restart}")
            if s_r < best_score:
                best_score, best_seq = s_r, a_r.copy()
                print(f"  Restart {restart}: {s_r:.10f}")

    # --- Phase 4: Ultra-fine final polish ---
    print(f"=== Phase 4: Ultra-fine polish (best={best_score:.10f}) ===")

    if best_seq is not None:
        final_n = best_seq.size

        # Very low learning rate Adam
        d_final1 = min(deadline - 30, time.monotonic() + 120)
        a_f1, s_f1 = _adam_optimize(best_seq, d_final1, max_iter=30000, lr=0.002,
                                     beta_factor=0.005, beta_decay=0.92,
                                     restart_patience=10000, noise_scale=0.05)
        if s_f1 < best_score:
            best_score, best_seq = s_f1, a_f1.copy()
        print(f"  Ultra Adam: {s_f1:.10f} (best={best_score:.10f})")

        # Peak equalization with very low temperature
        if time.monotonic() < deadline - 40:
            d_f2 = min(deadline - 25, time.monotonic() + 80)
            a_f2, s_f2 = _peak_equalization_adam(best_seq, d_f2, max_iter=20000,
                                                  lr=0.003, beta_factor=0.003)
            if s_f2 < best_score:
                best_score, best_seq = s_f2, a_f2.copy()
            print(f"  Peak-eq: {s_f2:.10f} (best={best_score:.10f})")

        # Multi-subgradient with many peaks
        if time.monotonic() < deadline - 30:
            d_f3 = min(deadline - 20, time.monotonic() + 60)
            a_f3, s_f3 = _multi_subgradient(best_seq, d_f3, max_iter=5000, top_k=10)
            if s_f3 < best_score:
                best_score, best_seq = s_f3, a_f3.copy()
            print(f"  MultiSG: {s_f3:.10f} (best={best_score:.10f})")

        # LP final polish
        if time.monotonic() < deadline - 15:
            d_f4 = min(deadline - 10, time.monotonic() + 45)
            a_f4 = _lp_refine(best_seq, d_f4, max_iter=500, inflate=1.0001, blend=0.05)
            s_f4 = float(2.0 * len(a_f4) * _fft_convolve(a_f4).max())
            if s_f4 < best_score:
                best_score, best_seq = s_f4, a_f4.copy()
            print(f"  Final LP: {s_f4:.10f} (best={best_score:.10f})")

        # Coordinate descent
        if time.monotonic() < deadline - 10:
            d_f5 = min(deadline - 5, time.monotonic() + 30)
            a_f5 = _coordinate_descent(best_seq, d_f5, max_sweeps=30)
            s_f5 = float(2.0 * len(a_f5) * _fft_convolve(a_f5).max())
            if s_f5 < best_score:
                best_score, best_seq = s_f5, a_f5.copy()
            print(f"  CoordDesc: {s_f5:.10f} (best={best_score:.10f})")

    # --- Final validation ---
    if best_seq is None:
        best_seq = _make_profile(256, "linear", eta=1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final_seq.sum())
    if s_final > 0.01:
        final_seq = final_seq / s_final
    else:
        final_seq = _make_profile(256, "linear", eta=1.5)

    # Verify with evaluate_sequence
    final_list = [float(x) for x in final_seq]
    try:
        eval_score = float(evaluate_sequence(final_list))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence: {eval_score:.10f}")
            if eval_score < best_score:
                best_score = eval_score
        else:
            # Fallback
            print("evaluate_sequence returned inf, using fallback")
            final_seq = _make_profile(256, "linear", eta=1.5)
            final_list = [float(x) for x in final_seq]
    except Exception as e:
        print(f"Error in evaluate_sequence: {e}")

    print(f"FINAL: score={best_score:.10f}, n={len(final_list)}")
    return final_list
