"""
Optimized step function minimization using:
1. Diverse profile initialization at multiple scales
2. Adam gradient descent in logit space with adaptive smooth-max temperature
3. LP-based fixed-point refinement (Brouwer argument)
4. Multi-scale escalation
5. Final polish with tight temperature

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

    """Compute cross-correlation a * b using FFT, returning only valid range."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:2 * n - 1]
    return c

def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _logit_gradient(c, p, beta):
    """Compute gradient of smooth max of convolution w.r.t. logit parameters."""
    _, w = _smooth_max_weights(c, beta)
    # d/dc_k (smooth max) = w_k
    # dc_k / dp_j = p_{k-j} (for valid range) -> this is cross-correlation
    n = len(p)
    h = _fft_cross(w, p)  # (w * p)_k = sum_j w_j * p_{k-j}
    return h[n-1:2*n-1]  # extract the "valid" range for gradient w.r.t. p

def _softmax(z, floor=1e-15):
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(e) / e.size
    return p / s

def _score(p):
    n = len(p)
    c = _fft_convolve(p)
    return 2.0 * n * float(c.max())

def _lp_solve(a, M, method='highs'):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def _adam_optimize(p0, deadline, rng, lr=0.05, beta_init_frac=0.1, 
                   beta_decay=0.93, no_imp_limit=200, momentum_reset=300):
    """Adam optimization in logit space with adaptive beta."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    p = p0.copy()
    c = _fft_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_init_frac * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        # Compute gradient
        g_p = _logit_gradient(c, p, beta)
        # Convert to logit gradient (mirror descent on simplex)
        dot = float(np.dot(p, g_p))
        g_z = p * (g_p - dot)
        g_z -= float(np.mean(g_z))
        
        if not np.all(np.isfinite(g_z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t = 0
            continue
        
        t += 1
        m = b1 * m + (1 - b1) * g_z
        v = b2 * v + (1 - b2) * (g_z * g_z)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t = 0
            continue
        
        z = np.clip(z, -50, 50)
        z -= float(np.mean(z))
        p = _softmax(z)
        c = _fft_convolve(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc
        
        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        
        # Adaptive beta
        if t % 20 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sorted_c = np.sort(c)[::-1]
                gap = sorted_c[0] - sorted_c[1]
                if gap < beta * 0.2:
                    beta = max(1e-4, beta * 0.5)
        
        # Restart
        if no_imp > no_imp_limit:
            ns = 0.05 + 0.05 * (no_imp // no_imp_limit)
            ns = min(ns, 0.3)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= float(np.mean(z))
            p = _softmax(z)
            c = _fft_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_init_frac * maxc)
            m.fill(0); v.fill(0); t = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score

def _lp_iterate(a, deadline, t_step=0.3, max_iters=50, M_margin=1.0):
    """LP-based fixed point iteration."""
    n = len(a)
    best = a.copy()
    best_sc = _score(a)
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max()) * M_margin
        
        b = _lp_solve(a, M)
        if b is None:
            break
        sb = float(b.sum())
        if sb < 1e-10:
            break
        b = b / sb
        
        a_new = (1 - t_step) * a + t_step * b
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best = a_new.copy()
        
        a = a_new
        t_step = max(0.1, t_step * 0.95)
    
    return best, best_sc

def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = {}
    t = np.linspace(0, 1, n)
    ar = np.arange(n)
    
    profiles['uniform'] = np.ones(n)
    profiles['linear'] = np.maximum(0.1, 1.0 + 2.0*(t - 0.5))
    profiles['linear_neg'] = np.maximum(0.1, 1.0 - 2.0*(t - 0.5))
    profiles['quadratic'] = np.maximum(0.01, 1.0 - 4.0*(t-0.5)**2)
    profiles['cubic'] = np.maximum(0.01, 1.0 + 1.5*(2*t-1) - 2.0*(2*t-1)**3)
    profiles['two_bump'] = np.exp(-50*(t-0.25)**2) + np.exp(-50*(t-0.75)**2)
    profiles['three_bump'] = np.exp(-30*(t-0.2)**2) + np.exp(-30*(t-0.5)**2) + np.exp(-30*(t-0.8)**2)
    profiles['cos1'] = np.maximum(0.1, 1.0 + 0.7*np.cos(2*np.pi*t))
    profiles['cos2'] = np.maximum(0.1, 1.0 + 0.7*np.cos(4*np.pi*t))
    profiles['exp_decay'] = np.exp(-2.0*ar/np.maximum(n,1))
    profiles['exp_decay2'] = np.exp(-1.0*ar/np.maximum(n,1))
    profiles['sinc'] = np.abs(np.sin(np.pi*t)/(np.pi*t + 1e-10))
    
    for k in profiles:
        profiles[k] = profiles[k] / profiles[k].sum()
    
    return profiles

def _interp(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-12)
    return p1 / p1.sum()

"""
Optimized step function minimization: minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses diverse initialization, Adam in logit space with adaptive smooth-max,
LP-based fixed-point refinement, multi-scale escalation, and multi-restart.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _grad_wrt_p(p, beta):
    """Gradient of smooth_max_k (p*p)_k w.r.t. p (on the simplex)."""
    n = len(p)
    c = _fft_convolve(p)
    _, w = _smooth_max_weights(c, beta)
    # h_j = sum_k w_k * p_{k-j} = np.convolve(w, p[::-1])[n-1:2n-1]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h_full = np.fft.irfft(fw * fp, N)[:2 * n - 1]
    return h_full[n - 1:2 * n - 1]


def _softmax(z, floor=1e-15):
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / float(e.size)
    return p / s


def _score(p):
    n = len(p)
    c = _fft_convolve(p)
    return 2.0 * n * float(c.max())


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.08,
                   beta_decay=0.93, no_imp_limit=200):
    """Adam optimization in logit space with adaptive smooth-max temperature."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    p = p0.copy()
    c = _fft_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * maxc
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        g_p = _grad_wrt_p(p, beta)
        dot = float(np.dot(p, g_p))
        g_z = p * (g_p - dot)
        g_z -= float(np.mean(g_z))

        if not np.all(np.isfinite(g_z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        t += 1
        m = b1 * m + (1.0 - b1) * g_z
        v = b2 * v + (1.0 - b2) * (g_z * g_z)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax(z)
        c = _fft_convolve(p)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            continue

        if t % 20 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sc = np.sort(c)[::-1]
                gap = sc[0] - sc[1]
                if gap < beta * 0.15:
                    beta = max(1e-4, beta * 0.5)

        if no_imp > no_imp_limit:
            ns = min(0.05 + 0.04 * (no_imp // no_imp_limit), 0.25)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax(z)
            c = _fft_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_iterate(a, deadline, t_step=0.35, max_iters=40, M_margin=1.0):
    """LP-based fixed-point iteration for refinement."""
    n = len(a)
    best = a.copy()
    best_sc = _score(a)
    current_t = t_step

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max()) * M_margin

        b = _lp_solve(a, M)
        if b is None:
            # Try slightly larger M
            M2 = M * 1.01
            b = _lp_solve(a, M2)
            if b is None:
                break

        sb = float(b.sum())
        if sb < 1e-10:
            break
        b = b / sb

        a_new = (1.0 - current_t) * a + current_t * b
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best = a_new.copy()

        a = a_new
        current_t = max(0.1, current_t * 0.93)

    return best, best_sc


def _make_profiles(n):
    """Generate diverse initial profiles."""
    profiles = {}
    t = np.linspace(0.0, 1.0, n)
    ar = np.arange(n, dtype=np.float64)
    nm = max(n, 1)

    profiles['uniform'] = np.ones(n)
    profiles['lin_up'] = np.maximum(0.05, 1.0 + 2.5 * (t - 0.5))
    profiles['lin_dn'] = np.maximum(0.05, 1.0 - 2.5 * (t - 0.5))
    profiles['quad'] = np.maximum(0.01, 1.0 - 3.0 * (t - 0.5) ** 2)
    profiles['cubic'] = np.maximum(0.01, 1.0 + 2.0 * (2 * t - 1) - 3.0 * (2 * t - 1) ** 3)
    profiles['two_bump'] = np.exp(-40 * (t - 0.25) ** 2) + np.exp(-40 * (t - 0.75) ** 2)
    profiles['three_bump'] = np.exp(-25 * (t - 0.2) ** 2) + np.exp(-25 * (t - 0.5) ** 2) + np.exp(-25 * (t - 0.8) ** 2)
    profiles['cos1'] = np.maximum(0.05, 1.0 + 0.7 * np.cos(2 * np.pi * t))
    profiles['cos2'] = np.maximum(0.05, 1.0 + 0.7 * np.cos(4 * np.pi * t))
    profiles['cos3'] = np.maximum(0.05, 1.0 + 0.5 * np.cos(6 * np.pi * t))
    profiles['exp1'] = np.exp(-1.5 * ar / nm)

    profiles['center'] = np.exp(-15 * (t - 0.5) ** 2)

    for k in profiles:
        profiles[k] = profiles[k] / profiles[k].sum()

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 1e-12)
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 1e-12)
    return p1 / p1.sum()

    # Final validation and return
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        else:
            final = final / s
        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        try:
            esc = float(evaluate_sequence([float(x) for x in final]))
            print(f"EVAL: {esc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    # Fallback
    fb = np.ones(256) / 256.0
    print("FALLBACK: uniform n=256")
    return [float(x) for x in fb]
