Let me think about the spectral projection method more carefully.

**Spectral Projection (Flat Spectrum) Method:**

Goal: Find a >= 0, sum(a) = 1, such that |FFT(a)| is as constant as possible.

Algorithm (POCS-like):
1. a = initial (e.g., uniform)
2. Repeat:
   a. Compute A = FFT(a, N) where N >= 2n-1
   b. Compute target magnitude: |A_target| = sqrt(sum(|A|^2) / N) (constant)
   c. New phase-preserving update: A_new = A / |A| * |A_target| (where |A| > 0)
   d. a_new = IFFT(A_new).real
   e. Project to non-negative: a_new = max(a_new, 0)
   f. Normalize: a_new = a_new / sum(a_new)
   g. a = a_new (or blend)

The issue is that step (c) might produce negative values in step (d), and the projection in (e) might undo the spectral improvement. This is a standard trade-off in POCS.

To improve convergence, I can:
- Use a soft projection: `a_new = a_new + alpha * max(0, -a_new)` (gradually push negative values to 0)
- Use a relaxed spectral update: `A_new = (1-lambda)*A + lambda*(A/|A|*|A_target|)`
- Use over-relaxation in the non-negativity projection

        target_mag = np.sqrt(np.mean(np.abs(A[:n])**2))
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        phase = A / mag
        new_mag = (1.0 - n_flat) * mag + n_flat * target_mag
        A_new = phase * new_mag
        a_spec = np.fft.irfft(A_new, N)[:n]
        a_spec = np.maximum(a_spec, 0.0)
        s = a_spec.sum()
        if s < 1e-15:
            a_spec = a.copy()
        else:
            a_spec = a_spec / s

        # Step 2: Blend with current (relaxation)
        a_new = (1.0 - n_nonneg) * a + n_nonneg * a_spec
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        sc = _score_normalized(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            a = a_new  # still move forward for exploration
            if no_imp > 200:
                # Gradually increase relaxation for more aggressive updates
                n_flat = min(1.0, n_flat * 1.001)
                n_nonneg = min(1.0, n_nonneg * 1.001)
            if no_imp > 2000:
                break

    return best_a


def _lse_optimize(a, deadline, max_iter=100000, T_start=0.03, T_end=1e-5, lr=0.1, cyclic=False):
    """LSE gradient with Adam in logit space, optional cyclic annealing."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= z.mean()
    state_t = 0
    state_m = np.zeros(n)
    state_v = np.zeros(n)

    best_a = a.copy()
    best_score = _score_normalized(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    b1, b2 = 0.9, 0.999

    if cyclic:
        T_cycle = T_start
        T_decay_rate = 0.9995
        T_min_cycle = 0.005
    else:
        if max_iter > 1:
            T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
        else:
            T_ratio = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = ex.sum()
        if not np.isfinite(s) or s <= 0:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

        grad = _compute_grad(w, a)
        if not np.all(np.isfinite(grad)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= z.mean()
            state_t = 0
            state_m = np.zeros(n)
            state_v = np.zeros(n)
            continue

        # Natural gradient on simplex
        dot = float(np.dot(a, grad))
        gz = grad / np.maximum(a, 1e-12) - dot
        gz -= gz.mean()
        gz = np.clip(gz, -100, 100)

        # Adam
        state_t += 1
        state_m = b1 * state_m + (1 - b1) * gz
        state_v = b2 * state_v + (1 - b2) * (gz * gz)
        mh = state_m / (1 - b1 ** state_t)
        vh = state_v / (1 - b2 ** state_t)

        z = z - lr * mh / (np.sqrt(vh) + 1e-8)
        z = np.clip(z, -30, 30)
        z -= z.mean()
        e = np.exp(z)
        a_new = e / e.sum()

        if not np.all(np.isfinite(a_new)):
            a_new = best_a.copy()
            z = np.log(np.maximum(a_new, 1e-12))
            z -= z.mean()
            state_t = 0
            state_m = np.zeros(n)
            state_v = np.zeros(n)

        a = a_new
        sc = _score_normalized(a)

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.92)
            if no_imp > max(3000, 8 * n):
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.1 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= z.mean()
                state_t = 0
                state_m = np.zeros(n)
                state_v = np.zeros(n)
                lr = base_lr
                no_imp = 0

        # Temperature schedule
        if cyclic:
            T *= T_decay_rate
            if T < T_min_cycle:
                T = T_cycle  # reset
        else:
            T *= T_ratio
            T = max(T, T_end)

    return best_a


def _multiplicative_update(a, deadline, max_iter=50000, eta=0.5, T=0.02):
    """
    Multiplicative update: a_new[i] = a[i] * (C / grad[i])^eta
    where grad[i] = 2 * sum_k w[k] * a[k-i], w = softmax((conv-max)/T).
    Naturally maintains non-negativity.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    current_eta = eta

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        w = ex / ex.sum()

        grad = _compute_grad(w, a)
        grad = np.maximum(grad, 1e-12)

        # Multiplicative update: equalize the gradient across coordinates
        C = float(np.mean(grad))
        ratio = C / grad
        a_new = a * (ratio ** current_eta)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s

        sc = _score_normalized(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            a = a_new  # still move for exploration
            if no_imp % 200 == 0:
                current_eta = max(0.1, current_eta * 0.95)
            if no_imp > max(5000, 10 * n):
                a = best_a.copy()
                current_eta = eta
                no_imp = 0

    return best_a


def _lp_refine(a, deadline, max_iter=3000, inflate=1.0003, blend=0.2, use_line_search=True, top_k=80):
    """LP-based refinement with reduced constraints and line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur

        # Reduced constraint set
        if n <= 128:
            idx_all = np.arange(2 * n - 1)
            active = idx_all.tolist()
        else:
            idx = np.argsort(c)[::-1]
            active_set = set()
            for i in idx[:top_k]:
                for offset in range(-3, 4):
                    j = i + offset
                    if 0 <= j < 2 * n - 1:
                        active_set.add(j)
            active = sorted(active_set)

        # Build LP
        A = np.zeros((len(active), n))
        for row, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                A[row, j] = a[j]
        b = np.full(len(active), M_lp)
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if not res.success:
            inflate_cur = min(1.01, inflate_cur * 1.3)
            no_imp += 3
            if no_imp > 200:
                break
            continue

        g = np.maximum(res.x, 0)
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 50 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.1)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        if use_line_search:
            # Simple bisection line search
            best_t = 0.0
            best_sc = _score_normalized(a)
            lo, hi = 0.0, 1.0
            for _ in range(12):
                mid = (lo + hi) / 2
                a_mid = np.maximum(a + mid * (g - a), 0)
                sm = a_mid.sum()
                if sm > 1e-10:
                    a_mid = a_mid / sm
                    sc_mid = _score_normalized(a_mid)
                    if sc_mid < best_sc:
                        best_sc = sc_mid
                        best_t = mid
                        lo = mid
                    else:
                        hi = mid
                else:
                    hi = mid
            t_opt = best_t
            if t_opt < 0.01:
                t_opt = blend_cur * 0.3
        else:
            t_opt = blend_cur

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.03, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return np.maximum(p0, 0) / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, np.maximum(p0, 0))
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_gaussian(n, sigma_frac=0.35):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_u_shaped(n, alpha=1.0):
    """U-shaped: high at edges, low in center."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_edges(n, edge_w=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_w * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, peak_width=0.15):
    """Two peaks at 1/4 and 3/4 positions."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.25) / peak_width) ** 2) + np.exp(-0.5 * ((t - 0.75) / peak_width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_profile(n, rng, sparsity=0.0):
    a = rng.random(n)
    if sparsity > 0:
        mask = rng.random(n) > sparsity
        a[~mask] = 0
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method search combining spectral projection, LSE gradient,
    multiplicative updates, and LP refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 40.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                    print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 1024, 2048]:
        if ln not in lengths:
            lengths.append(ln)

    print(f"Lengths to explore: {lengths}")

    # Generate initial candidates per length
    def get_candidates(n_t):
        cands = []
        labels = []
        if parent is not None:
            cands.append(_interpolate_profile(parent, n_t))
            labels.append("parent")
            for _ in range(2):
                p = _interpolate_profile(parent, n_t)
                p_p = p * (1.0 + 0.1 * rng.normal(size=n_t))
                p_p = np.maximum(p_p, 0)
                if p_p.sum() > 1e-10:
                    cands.append(p_p / p_p.sum())
                    labels.append("parent_pert")
        cands.append(_make_gaussian(n_t, 0.35))
        labels.append("gauss")
        cands.append(_make_raised_cosine(n_t))
        labels.append("cosine")
        cands.append(_make_u_shaped(n_t, 0.7))
        labels.append("u_shaped")
        cands.append(_make_bimodal(n_t, 0.15))
        labels.append("bimodal")
        cands.append(_make_flat_edges(n_t, 0.4))
        labels.append("flat_edges")
        cands.append(_make_random_profile(n_t, rng))
        labels.append("random")
        return cands, labels

    # Per-length best tracking
    best_per_length = {}

    # Phase 1: Spectral projection
    print("\n=== Phase 1: Spectral Projection ===")
    total_phase1_budget = min(350.0, (deadline - t_start) * 0.25)
    per_cand_budget = total_phase1_budget / (len(lengths) * 7)

    for n_t in lengths:
        cands, labels = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 150:
                break
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            sp_end = min(time.monotonic() + per_cand_budget, deadline - 150)
            if sp_end - time.monotonic() < 10:
                continue

            a_sp = _spectral_projection(a0, sp_end, max_iter=5000,
                                        n_flat=0.85, n_nonneg=0.85)
            sc_sp = _score(a_sp)
            if np.isfinite(sc_sp) and sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            if sc_sp < best_here_score:
                best_here_score = sc_sp
                best_here = a_sp
            print(f"  n={n_t}, {labels[ci]}: init={sc0:.6f} -> spec={sc_sp:.8f}")

        if best_here is not None:
            best_per_length[n_t] = best_here

    # Phase 2: LSE gradient with cyclic annealing
    print("\n=== Phase 2: LSE Gradient (cyclic) ===")
    phase2_start = time.monotonic()
    phase2_budget = min(450.0, (deadline - phase2_start) * 0.3)

    for n_t in lengths:
        if time.monotonic() > deadline - 200:
            break
        a0 = best_per_length.get(n_t)
        if a0 is None and parent is not None:
            a0 = _interpolate_profile(parent, n_t)
        if a0 is None:
            a0 = _make_gaussian(n_t, 0.35)

        lse_budget = min(phase2_budget / len(lengths), deadline - time.monotonic() - 200)
        if lse_budget < 20:
            continue
        lse_end = time.monotonic() + lse_budget

        # Try cyclic and monotone
        a_cyc = _lse_optimize(a0, lse_end, max_iter=80000,
                              T_start=0.03, T_end=1e-5, lr=0.1, cyclic=True)
        sc_cyc = _score(a_cyc)
        if np.isfinite(sc_cyc) and sc_cyc < best_score:
            best_score = sc_cyc
            best_seq = a_cyc.copy()

        # Quick monotone pass on the cyclic result
        mono_end = min(time.monotonic() + lse_budget * 0.3, deadline - 200)
        if mono_end - time.monotonic() > 15:
            a_mono = _lse_optimize(a_cyc, mono_end, max_iter=30000,
                                   T_start=0.01, T_end=1e-5, lr=0.05, cyclic=False)
            sc_mono = _score(a_mono)
            if np.isfinite(sc_mono) and sc_mono < best_score:
                best_score = sc_mono
                best_seq = a_mono.copy()
            a_best_here = a_mono if sc_mono < sc_cyc else a_cyc
        else:
            a_best_here = a_cyc

        if n_t in best_per_length:
            if _score(a_best_here) < _score(best_per_length[n_t]):
                best_per_length[n_t] = a_best_here
        else:
            best_per_length[n_t] = a_best_here

        print(f"  n={n_t}: LSE score={min(sc_cyc, sc_mono if np.isfinite(sc_mono) else sc_cyc):.8f}")

    # Phase 3: Multiplicative updates
    print("\n=== Phase 3: Multiplicative Updates ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        mu_budget = min(200.0, deadline - time.monotonic() - 150)
        if mu_budget > 20:
            mu_end = time.monotonic() + mu_budget
            a_mu = _multiplicative_update(best_seq, mu_end, max_iter=50000, eta=0.4, T=0.02)
            sc_mu = _score(a_mu)
            if np.isfinite(sc_mu) and sc_mu < best_score:
                best_score = sc_mu
                best_seq = a_mu.copy()
            print(f"  MU: score={sc_mu:.8f}")

    # Phase 4: LP refinement
    print("\n=== Phase 4: LP Refinement ===")
    for n_t in lengths:
        if time.monotonic() > deadline - 100:
            break
        if n_t not in best_per_length:
            continue
        a_in = best_per_length[n_t]
        sc_in = _score(a_in)

        lp_budget = min(300.0, deadline - time.monotonic() - 100)
        if lp_budget < 15:
            continue
        lp_end = time.monotonic() + lp_budget

        a_lp = _lp_refine(a_in, lp_end, max_iter=2000,
                          inflate=1.0003, blend=0.15, use_line_search=True, top_k=80)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP n={n_t}: {sc_in:.8f} -> {sc_lp:.8f}")

    # Phase 5: Final combined polish
    print("\n=== Phase 5: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 20
        # Quick LSE
        if remaining > 40:
            lse_end = time.monotonic() + remaining * 0.4
            a_f1 = _lse_optimize(best_seq, lse_end, max_iter=30000,
                                 T_start=0.01, T_end=1e-6, lr=0.05, cyclic=True)
            sc_f1 = _score(a_f1)
            if np.isfinite(sc_f1) and sc_f1 < best_score:
                best_score = sc_f1
                best_seq = a_f1.copy()
            print(f"  Final LSE: {sc_f1:.8f}")

        # Quick MU
        if time.monotonic() < deadline - 20:
            mu_end = time.monotonic() + (deadline - time.monotonic() - 20) * 0.4
            if mu_end - time.monotonic() > 15:
                a_f2 = _multiplicative_update(best_seq, mu_end, max_iter=30000, eta=0.3, T=0.015)
                sc_f2 = _score(a_f2)
                if np.isfinite(sc_f2) and sc_f2 < best_score:
                    best_score = sc_f2
                    best_seq = a_f2.copy()
                print(f"  Final MU: {sc_f2:.8f}")

        # Quick LP
        if time.monotonic() < deadline - 10:
            lp_end = time.monotonic() + (deadline - time.monotonic() - 10) * 0.8
            if lp_end - time.monotonic() > 10:
                a_f3 = _lp_refine(best_seq, lp_end, max_iter=500,
                                  inflate=1.0002, blend=0.1, use_line_search=True, top_k=60)
                sc_f3 = _score(a_f3)
                if np.isfinite(sc_f3) and sc_f3 < best_score:
                    best_score = sc_f3
                    best_seq = a_f3.copy()
                print(f"  Final LP: {sc_f3:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
