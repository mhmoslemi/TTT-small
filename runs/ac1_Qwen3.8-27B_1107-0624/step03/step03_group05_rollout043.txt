"""
Multi-strategy optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP spreading, adaptive smooth-max gradient descent, simulated annealing
with diverse moves, and peak-directed local search. Explores multiple
sequence lengths up to 2048 for more degrees of freedom.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v, target_sum=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # Project onto {x : x >= 0, sum(x) = target_sum}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - target_sum) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    return w * (target_sum / s)

def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s

def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h

def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

            break

def _peak_shaving(a, deadline, max_iter=10000):

    no_imp = 0

        c = _fft_convolve(a)
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for pk in peak_indices:
            pk = int(pk)
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            
            # Find the pair with max product at this peak
            best_prod = 0
            best_i = -1
            for k in range(lo, hi + 1):
                prod = a[k] * a[pk - k]
                if prod > best_prod:
                    best_prod = prod
                    best_i = k
            
            if best_i < 0 or best_prod < 1e-10:
                continue
            
            # Try moving mass from best_i to random positions
            for _ in range(3):
                j = rng.integers(0, n)
                if j == best_i:
                    continue
                # Check if j contributes to this peak

                    continue
                
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[best_i]
                    if delta > a[best_i] or delta < 1e-10:
                        continue
                    a_try = a.copy()
                    a_try[best_i] -= delta
                    a_try[j] += delta
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:

                    a_try = a_try / s_try
                    sc = _score(a_try)
                    if sc < _score(a) - 1e-12:
                        a = a_try
                        improved = True

                if improved:

        if improved:
            sc_new = _score(a)
            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
            if no_imp > 2000:

    return best_a

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new

"""
Multi-strategy optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LP spreading, adaptive smooth-max gradient, simulated annealing
with diverse moves (including peak-directed), peak-shaving local search,
and multi-length exploration up to n=2048 for more degrees of freedom.
Key novelties: proper SA cooling schedule with 6 move types, targeted
peak-shaving that identifies worst contributing pairs, and structured
initializations (sinusoidal, alternating, concentrated) to escape
local minima found by previous methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v, target_sum=1.0):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - target_sum) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    return w * (target_sum / s)


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.25):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):

            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

                break
            continue
        g = g / sum_g

        if s_new < 1e-10:

def _gradient_polish(a, deadline, max_iter=20000, beta_scale=0.05):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a, 1.0)
    best_a = a.copy()
    best_score = _score(a)
    beta = max(1e-8, beta_scale * float(_fft_convolve(a).max()))
    lr = 0.03
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a_nest = a + momentum * v
        grad = _gradient_smooth_max(a_nest, beta)
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue
        v = momentum * v - lr * grad
        a = _project_simplex(a + v, 1.0)
        score = _score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:

            if no_imp % 300 == 0:

                v = np.zeros(n)

                lr = base_lr
                no_imp = 0
        if it % 150 == 0:

def _simulated_annealing(a, deadline, max_iter=50000, T0=0.008, alpha=0.9998, seed=42):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score
    T = T0
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_try = a.copy()
        move_type = rng.integers(0, 6)

        if move_type == 0:
            # Single transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = rng.uniform(0.005, 0.05) * max(a[i], 1e-6)
                delta = min(delta, a[i] * 0.5)
                if delta > 1e-12:
                    a_try[i] -= delta
                    a_try[j] += delta
        elif move_type == 1:
            # Block swap
            L = rng.integers(max(2, n // 64), max(3, n // 16))
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            if abs(i - j) >= L:
                a_try[i:i + L], a_try[j:j + L] = a_try[j:j + L].copy(), a_try[i:i + L].copy()
        elif move_type == 2:
            # Gaussian perturbation
            sigma = max(T * 0.5, 1e-4)
            a_try += rng.normal(0, sigma, n)
        elif move_type == 3:
            # Block mirror
            L = rng.integers(max(4, n // 32), max(8, n // 8))
            i = rng.integers(0, n - L)
            a_try[i:i + L] = a_try[i:i + L][::-1]
        elif move_type == 4 and it % 10 == 0:
            # Peak-directed: move mass from highest contributing pair
            c = _fft_convolve(a)
            idx = int(np.argmax(c))
            lo = max(0, idx - n + 1)
            hi = min(n - 1, idx)
            best_prod = 0.0
            best_i = lo
            for k in range(lo, hi + 1):
                prod = a[k] * a[idx - k]
                if prod > best_prod:
                    best_prod = prod
                    best_i = k
            if best_prod > 1e-10:
                j = rng.integers(0, n)
                if j != best_i:
                    delta = 0.1 * a[best_i]
                    a_try[best_i] -= delta
                    a_try[j] += delta
        else:
            # Block copy with blending
            L = rng.integers(max(3, n // 32), max(5, n // 12))
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            if abs(i - j) >= L:
                blend = rng.uniform(0.3, 0.7)
                a_try[j:j + L] = blend * a_try[j:j + L] + (1.0 - blend) * a_try[i:i + L]

        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        sc_try = _score(a_try)
        delta_sc = sc_try - cur_score

        if delta_sc < 0 or (T > 1e-10 and rng.random() < np.exp(-delta_sc / T)):
            a = a_try
            cur_score = sc_try
            if sc_try < best_score - 1e-12:
                best_score = sc_try
                best_a = a_try.copy()
                no_imp = 0
                T = max(1e-8, T * 0.999)
            else:
                no_imp += 1
        else:
            no_imp += 1

        T = max(1e-10, T * alpha)

        if no_imp > 4000:
            a = best_a.copy()
            cur_score = best_score
            T = max(T0 * 0.3, T * 5.0)
            no_imp = 0

    return best_a


def _peak_shaving(a, deadline, max_iter=10000, seed=999):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        top_k = min(3, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        improved = False
        for pk in peak_indices:
            pk = int(pk)
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)

            best_prod = 0.0
            best_i = -1
            for k in range(lo, hi + 1):
                prod = a[k] * a[pk - k]
                if prod > best_prod:
                    best_prod = prod
                    best_i = k

            if best_i < 0 or best_prod < 1e-12:
                continue

            for _ in range(5):
                j = rng.integers(0, n)
                if j == best_i:
                    continue
                if lo <= j <= hi and (pk - j) != best_i and lo <= (pk - j) <= hi:
                    continue

                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    delta = frac * a[best_i]
                    if delta >= a[best_i] or delta < 1e-12:
                        continue
                    a_try = a.copy()
                    a_try[best_i] -= delta
                    a_try[j] += delta
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    sc = _score(a_try)
                    if sc < _score(a) - 1e-12:
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            sc_new = _score(a)
            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
            if no_imp > 3000:
                break

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _structured_init(n, rng):
    type_id = rng.integers(0, 7)
    t = np.linspace(0, 1, n)
    if type_id == 0:
        a = 1.0 + rng.uniform(0.5, 3.0) * (t - 0.5)
    elif type_id == 1:
        a = 1.0 + rng.uniform(1.0, 5.0) * (2.0 * t - 1.0) ** 2
    elif type_id == 2:
        k = rng.integers(1, max(2, n // 10))
        a = 1.0 + rng.uniform(0.3, 0.9) * np.sin(2.0 * np.pi * k * t)
    elif type_id == 3:
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 0.9)
    elif type_id == 4:
        c1, c2 = sorted(rng.uniform(0, 1, 2))
        s1, s2 = rng.uniform(0.05, 0.3, 2)
        a = np.exp(-0.5 * ((t - c1) / s1) ** 2) + np.exp(-0.5 * ((t - c2) / s2) ** 2)
    elif type_id == 5:
        a = np.zeros(n)
        a[:n // 2] = 1.0
        a += 0.1 * rng.random(n)
    else:
        a = np.maximum(rng.normal(1.0, 0.4, n), 0.01)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _adam_polish(a, deadline, max_iter=20000, lr=0.05):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    best_score = _score(a)
    beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.005, lr * 0.96)

        if it % 150 == 0:
            beta = max(1e-10, beta * 0.75)

        if no_imp > max(2000, 4 * n):
            rng_r = np.random.default_rng(hash((it, n)) % (2**32))
            z = 0.7 * best_z + 0.3 * rng_r.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Default init: score={best_score:.10f}")

    n_parent = parent.size

    # Phase 1: Polish parent with gradient + LP + peak shaving
    if time.monotonic() < deadline - 400:
        phase1_end = deadline - 400
        print(f"\nPhase 1: Polish parent (n={n_parent}), budget={phase1_end - time.monotonic():.0f}s")

        a = parent.copy()

        # Gradient polish
        gp_end = min(time.monotonic() + 150, phase1_end - 200)
        a = _gradient_polish(a, gp_end, max_iter=10000, beta_scale=0.04)
        sc = _score(a)
        print(f"  Gradient: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

        # LP
        if n_parent <= 1024:
            lp_end = min(time.monotonic() + 120, phase1_end - 100)
            a = _lp_refine(a, lp_end, max_iter=8000, inflate=1.001, blend=0.25)
            sc = _score(a)
            print(f"  LP: {sc:.10f}")
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

        # Peak shaving
        ps_end = min(time.monotonic() + 80, phase1_end - 50)
        a = _peak_shaving(a, ps_end, max_iter=8000)
        sc = _score(a)
        print(f"  PeakShave: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

        # Adam
        ad_end = min(time.monotonic() + 80, phase1_end)
        a = _adam_polish(a, ad_end, max_iter=10000, lr=0.04)
        sc = _score(a)
        print(f"  Adam: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # Phase 2: Multi-length search
    lengths_to_try = [ln for ln in [1024, 1536, 2048] if ln > n_parent]
    # Also re-try parent length with different inits
    print(f"\nPhase 2: Multi-length search: {lengths_to_try} + parent re-init")
    all_lengths = lengths_to_try + [n_parent]

    rem = deadline - time.monotonic() - 200
    if rem > 60:
        per_length = max(20.0, rem / len(all_lengths))

        for n_t in all_lengths:
            if time.monotonic() > deadline - 180:
                break
            budget_n = min(per_length, deadline - time.monotonic() - 180)
            if budget_n < 20:
                break

            print(f"\n  n={n_t}: budget={budget_n:.0f}s")

            num_inits = min(3, max(1, int(budget_n / 60)))
            for init_id in range(num_inits):
                if time.monotonic() > deadline - 170:
                    break
                sub_b = deadline - time.monotonic() - 170
                if sub_b < 15:
                    break
                sub_b = min(sub_b, budget_n / (num_inits - init_id))
                if sub_b < 15:
                    break

                if init_id == 0 and n_t == n_parent:
                    a_init = _interpolate_profile(best_seq, n_t)
                elif init_id == 0:
                    a_init = _interpolate_profile(parent, n_t)
                else:
                    a_init = _structured_init(n_t, rng)

                sc_init = _score(a_init)
                if sc_init < best_score:
                    best_score = sc_init
                    best_seq = a_init.copy()
                    print(f"    Init{init_id} ({n_t}): {sc_init:.10f} *")

                # Gradient
                gp_e = time.monotonic() + sub_b * 0.35
                a = _gradient_polish(a_init, min(gp_e, deadline - 160), max_iter=5000)
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"    Init{init_id} grad ({n_t}): {sc:.10f} *")

                # LP for smaller n
                if n_t <= 1024:
                    lp_e = time.monotonic() + sub_b * 0.25
                    a = _lp_refine(a, min(lp_e, deadline - 150), max_iter=3000, inflate=1.001, blend=0.2)
                    sc = _score(a)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a.copy()
                        print(f"    Init{init_id} lp ({n_t}): {sc:.10f} *")

                # SA
                sa_e = time.monotonic() + sub_b * 0.25
                a = _simulated_annealing(a, min(sa_e, deadline - 140), max_iter=30000, T0=0.005, alpha=0.9998, seed=100 + init_id)
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"    Init{init_id} sa ({n_t}): {sc:.10f} *")

                # Peak shaving
                ps_e = time.monotonic() + sub_b * 0.15
                a = _peak_shaving(a, min(ps_e, deadline - 130), max_iter=5000, seed=200 + init_id)
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"    Init{init_id} ps ({n_t}): {sc:.10f} *")

    # Phase 3: Intensive final polish
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_best = best_seq.size
        final_budget = deadline - time.monotonic() - 25
        print(f"\nPhase 3: Final polish (n={n_best}), budget={final_budget:.0f}s")

        # SA (most time)
        sa_e = time.monotonic() + final_budget * 0.35
        a = _simulated_annealing(best_seq, min(sa_e, deadline - 120), max_iter=60000, T0=0.003, alpha=0.9999, seed=777)
        sc = _score(a)
        print(f"  Final SA: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

        # Gradient
        gp_e = time.monotonic() + final_budget * 0.25
        a = _gradient_polish(best_seq, min(gp_e, deadline - 100), max_iter=25000, beta_scale=0.02)
        sc = _score(a)
        print(f"  Final Grad: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

        # Peak shaving
        ps_e = time.monotonic() + final_budget * 0.2
        a = _peak_shaving(best_seq, min(ps_e, deadline - 80), max_iter=20000, seed=555)
        sc = _score(a)
        print(f"  Final PS: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

        # Adam
        ad_e = time.monotonic() + final_budget * 0.12
        a = _adam_polish(best_seq, min(ad_e, deadline - 60), max_iter=15000, lr=0.03)
        sc = _score(a)
        print(f"  Final Adam: {sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()

        # LP if small enough
        if n_best <= 1024:
            lp_e = time.monotonic() + final_budget * 0.1
            a = _lp_refine(best_seq, min(lp_e, deadline - 40), max_iter=10000, inflate=1.0005, blend=0.15)
            sc = _score(a)
            print(f"  Final LP: {sc:.10f}")
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(768) / 768
        best_score = _score(final_seq)

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
        else:
            print(f"WARNING: evaluate returned {sc_eval}")
    except Exception as e:
        print(f"WARNING: eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
