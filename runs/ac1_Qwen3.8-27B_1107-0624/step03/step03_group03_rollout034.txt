Multi-phase optimization combining LSE gradient (with Nesterov momentum), 
exact coordinate descent, and LP-based spreading to minimize 2n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn
    return c


def _score_norm(a):
    """Score for normalized (sum=1) sequence: 2n * max(conv)."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _max_conv(a):
    """Just the max of the convolution."""
    return float(_fft_convolve(a).max())


def _lse_step_nesterov(a, z, state, T, lr):
    """LSE gradient step with Nesterov momentum in logit space."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = np.clip((c - m_val) / T, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # Gradient = 2 * conv(w, reverse(a))[n-1:2n-1]
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)[n - 1:2 * n - 1]
    grad = 2.0 * h

    # Project to tangent space of simplex
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    b1, b2, b3 = 0.9, 0.999, 0.9  # b3 for Nesterov
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    # Nesterov: use look-ahead
    z_new = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z_new = np.clip(z_new, -40, 40)
    z_new -= float(np.mean(z_new))
    
    # Nesterov momentum
    v_nest = b3 * state['v_n'] + (1.0 - b3) * (z_new - z)
    z_new = z_new + v_nest

    e = np.exp(z_new)
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    state['v_n'] = v_nest
    return a_new, z_new, state, m_val

    """LSE optimization with Nesterov momentum and adaptive restarts."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n), 'v_n': np.zeros(n)}

    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0
    base_lr = lr

    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a, z, state, _ = _lse_step_nesterov(a, z, state, T, lr)

        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n), 'v_n': np.zeros(n)}
            continue

        score = _score_norm(a)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0

            # Let me be very explicit. conv[k] = sum_j a[j]*a[k-j]
            # The terms in this sum that involve a[i] are:
            #   j = i: a[i]*a[k-i]
            #   j = k-i: a[k-i]*a[i] (same as above, it's the same term)
            # Wait, no. In the sum over j, each j appears once. So:
            #   When j = i: term is a[i]*a[k-i]
            #   When j = k-i (and k-i != i): term is a[k-i]*a[i]
            # These are two different terms in the sum (for different values of j).
            # 
            # After change: a'[i] = x, a'[j] = a[j] for j != i.
            # conv_new[k] = conv_old[k] - a[i]*a[k-i] - a[k-i]*a[i] + x*a[k-i] + a[k-i]*x
            #             = conv_old[k] - 2*a[i]*a[k-i] + 2*x*a[k-i]  (when i != k-i, i.e., k != 2i)
            # When k = 2i:
            # conv_new[2i] = conv_old[2i] - a[i]*a[i] + x*x = conv_old[2i] - c^2 + x^2
            # 
            # For k where k-i is out of range [0,n-1]: no terms involve a[i], so no change.
            # For k where i is out of range [0,n-1]: impossible since i is in range.
            # 
            # So the affected k's are those where 0 <= k-i <= n-1, i.e., i <= k <= i+n-1.
            # Among these, k=2i is special (quadratic), others are linear.
            # 
            # This confirms my formula. The update is:
            # conv_new[k] = conv_old[k] + 2*(x-c)*a[k-i] for k in [i, i+n-1], k != 2i
            # conv_new[2i] = conv_old[2i] - c^2 + x^2
            # 
            # And a[k-i] in the formula is the ORIGINAL a[k-i] (which equals a'[k-i] since k-i != i).
            
            # So my code above is correct. The line:
            # conv_cur[k_lin] = intercepts + slopes * x_opt
            # is correct because intercepts + slopes*x = conv_old + 2*(x-c)*a[k-i]
            # and a[k-i] is unchanged (verified above).
            
            # Check improvement
            new_max = float(conv_cur.max())
            new_score = 2.0 * n * new_max
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
            improved = True
        
        if not improved:

                  bounds=[(0, None)] * n, method='highs')

    return None

Looking at this problem, I need to minimize `2n * max(conv(a,a)) / sum(a)^2` for a non-negative sequence. The key insight from my analysis is that an **exact coordinate descent** approach can be very powerful here: fixing all coordinates except one, the objective becomes a 1D convex optimization problem (max of affine functions plus one quadratic), which can be solved exactly via subgradient bisection.

I'll combine:
1. **LSE gradient with Nesterov momentum** for global exploration
2. **Exact coordinate descent** (the key innovation) for precise local optimization
3. **LP-based spreading** for additional exploitation
4. **Multiple alternating phases** to escape local minima

"""
Multi-phase optimization: LSE (Nesterov) + exact coordinate descent + LP.
Key innovation: coordinate descent with exact 1D minimization per coordinate,
exploiting that fixing all but one coordinate makes the max-conv objective
a 1D convex function (max of affine + one quadratic).
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn
    return c


def _score_norm(a):
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _lse_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = np.clip((c - m_val) / T, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa_rev, N)[n - 1:2 * n - 1]
    grad = 2.0 * h
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    t = state['t'] + 1
    b1, b2, b3 = 0.9, 0.999, 0.9
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    z_new = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z_new = np.clip(z_new, -40, 40)
    z_new -= float(np.mean(z_new))
    v_nest = b3 * state['v_n'] + (1.0 - b3) * (z_new - z)
    z_new = z_new + v_nest
    e = np.exp(z_new)
    a_new = e / e.sum()
    state['t'] = t
    state['m'] = m
    state['v'] = v
    state['v_n'] = v_nest
    return a_new, z_new, state, m_val


def _lse_optimize(a, deadline, max_iter=100000, T_start=0.05, T_end=1e-5, lr=0.1):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n), 'v_n': np.zeros(n)}
    best_a = a.copy()
    best_score = _score_norm(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a, z, state, _ = _lse_step(a, z, state, T, lr)
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n), 'v_n': np.zeros(n)}
            continue
        score = _score_norm(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.95)
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(42 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n), 'v_n': np.zeros(n)}
                lr = base_lr
                no_imp = 0
        T = max(T * T_ratio, T_end)
    return best_a


def _optimal_x(i, c_old, conv_cur, a, n):
    """Find optimal value for coordinate i given current state."""
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)
    has_quad = (2 * i >= 0 and 2 * i <= 2 * n - 2)
    quad_base = conv_cur[2 * i] - c_old * c_old if has_quad else 0.0

    # Unaffected lags
    max_unaff = 0.0
    if k_lo > 0:
        max_unaff = max(max_unaff, float(conv_cur[:k_lo].max()))
    if k_hi < 2 * n - 2:
        max_unaff = max(max_unaff, float(conv_cur[k_hi + 1:].max()))

    # Linear part: k in [k_lo, k_hi], k != 2i
    k_idx = np.arange(k_lo, k_hi + 1)
    if has_quad:
        mask = k_idx != 2 * i
        k_lin = k_idx[mask]
    else:
        k_lin = k_idx

    if len(k_lin) > 0:
        a_shift = a[k_lin - i]
        intercepts = conv_cur[k_lin] - 2.0 * c_old * a_shift
        slopes = 2.0 * a_shift
    else:
        a_shift = np.array([])
        intercepts = np.array([])
        slopes = np.array([])

    def f_val(x):
        val = max_unaff
        if has_quad:
            val = max(val, x * x + quad_base)
        if len(k_lin) > 0:
            val = max(val, float((intercepts + slopes * x).max()))
        return val

    def subgrad_bounds(x):
        f_active = f_val(x)
        sg_lo = 0.0
        sg_hi = 0.0
        if has_quad and abs(x * x + quad_base - f_active) < 1e-9:
            sg_lo = min(sg_lo, 2.0 * x)
            sg_hi = max(sg_hi, 2.0 * x)
        if len(k_lin) > 0:
            lin_vals = intercepts + slopes * x
            active_idx = np.where(np.abs(lin_vals - f_active) < 1e-9)[0]
            if len(active_idx) > 0:
                sg_lo = min(sg_lo, float(slopes[active_idx].min()))
                sg_hi = max(sg_hi, float(slopes[active_idx].max()))
        return sg_lo, sg_hi

    # Check x=0
    _, sg_hi0 = subgrad_bounds(0.0)
    if sg_hi0 >= -1e-11:
        return 0.0

    # Upper bound
    hi_x = max(c_old * 4, 3.0 / n)
    sg_lo_hi, _ = subgrad_bounds(hi_x)
    if sg_lo_hi <= 1e-11:
        return hi_x

    # Bisection
    lo_x = 0.0
    for _ in range(50):
        mid = (lo_x + hi_x) * 0.5
        sg_lo, sg_hi = subgrad_bounds(mid)
        if sg_hi < -1e-11:
            lo_x = mid
        elif sg_lo > 1e-11:
            hi_x = mid
        else:
            break
    return (lo_x + hi_x) * 0.5


def _coordinate_descent(a, deadline, max_sweeps=500, order='sequential'):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    conv_cur = _fft_convolve(a)
    best_score = float(2.0 * n * conv_cur.max())
    best_a = a.copy()
    rng = np.random.default_rng(99999)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if order == 'random':
            seq = rng.permutation(n)
        else:
            seq = range(n)

        any_change = False
        for i in seq:
            if time.monotonic() > deadline:
                break
            c_old = a[i]
            if c_old < 1e-15:
                continue
            x_opt = _optimal_x(i, c_old, conv_cur, a, n)
            x_opt = max(0.0, x_opt)
            if abs(x_opt - c_old) < 1e-14:
                continue

            # Update convolution
            k_lo = max(i, 0)
            k_hi = min(i + n - 1, 2 * n - 2)
            has_quad = (2 * i <= 2 * n - 2 and 2 * i >= 0)
            if has_quad:
                conv_cur[2 * i] = x_opt * x_opt + (conv_cur[2 * i] - c_old * c_old)
            k_idx = np.arange(k_lo, k_hi + 1)
            if has_quad:
                mask = k_idx != 2 * i
                k_aff = k_idx[mask]
            else:
                k_aff = k_idx
            if len(k_aff) > 0:
                a_shift = a[k_aff - i]
                conv_cur[k_aff] = conv_cur[k_aff] + 2.0 * (x_opt - c_old) * a_shift

            a[i] = x_opt
            any_change = True

            new_max = float(conv_cur.max())
            new_score = 2.0 * n * new_max
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_a = a.copy()

        if not any_change:
            break

    return best_a

    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score_norm(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        best_score = _score_norm(parent)
        best_seq = parent.copy()
        print(f"No parent, uniform: score={best_score:.10f}")

    # Phase 1: LSE optimization
    print("\n=== Phase 1: LSE (Nesterov) ===")
    p1_dl = t_start + 280
    a1 = _lse_optimize(parent, p1_dl, max_iter=80000, T_start=0.05, T_end=1e-5, lr=0.1)
    sc1 = _score_norm(a1)
    print(f"  LSE: {sc1:.10f}")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()

    # Phase 2: Coordinate descent (sequential)
    print("\n=== Phase 2: Coordinate Descent ===")
    p2_dl = t_start + 700
    a2 = _coordinate_descent(a1, p2_dl, max_sweeps=300, order='sequential')
    sc2 = _score_norm(a2)
    print(f"  CD: {sc2:.10f}")
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()

    # Phase 3: LSE on improved
    print("\n=== Phase 3: LSE (finer) ===")
    p3_dl = t_start + 950
    a3 = _lse_optimize(a2, p3_dl, max_iter=50000, T_start=0.03, T_end=1e-6, lr=0.05)
    sc3 = _score_norm(a3)
    print(f"  LSE2: {sc3:.10f}")
    if sc3 < best_score:
        best_score = sc3
        best_seq = a3.copy()

    # Phase 4: Coordinate descent (random order)
    print("\n=== Phase 4: CD (random) ===")
    p4_dl = t_start + 1300
    a4 = _coordinate_descent(a3, p4_dl, max_sweeps=300, order='random')
    sc4 = _score_norm(a4)
    print(f"  CD2: {sc4:.10f}")
    if sc4 < best_score:
        best_score = sc4
        best_seq = a4.copy()

    # Phase 5: LP refinement
    print("\n=== Phase 5: LP ===")
    p5_dl = t_start + 1550
    a5 = _lp_refine(a4, p5_dl, max_iter=1000, inflate=1.001)
    sc5 = _score_norm(a5)
    print(f"  LP: {sc5:.10f}")
    if sc5 < best_score:
        best_score = sc5
        best_seq = a5.copy()

    # Phase 6: Final coordinate descent
    print("\n=== Phase 6: Final CD ===")
    a6 = _coordinate_descent(a5, deadline, max_sweeps=500, order='random')
    sc6 = _score_norm(a6)
    print(f"  Final CD: {sc6:.10f}")
    if sc6 < best_score:
        best_score = sc6
        best_seq = a6.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    sc_eval = float(evaluate_sequence(final_seq.tolist()))
    print(f"\nEvaluate_sequence: {sc_eval:.10f}")
    print(f"FINAL: score={min(best_score, sc_eval):.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final_seq]
