"""

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    if n <= 250:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _pnorm_objective(a, p):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    val = (c ** p).sum() ** (1.0 / p)
    return float(2.0 * n * val / (s * s))


def _pnorm_gradient(a, p):
    """Gradient of the p-norm score (assuming sum(a)=1 for simplicity)."""
    n = len(a)
    c = _fft_convolve(a)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    c_p = c ** (p - 1)
    sum_cp = (c ** p).sum()
    if sum_cp < 1e-30:
        return np.zeros(n)
    scale = 2.0 * p / sum_cp
    # grad_j = scale * sum_k c[k]^(p-1) * 2*a[k-j] for valid k-j
    # = 2*scale * cross-correlation of c_p and a
    # Compute via FFT: need (c_p flipped * a) extracted properly
    N2 = _next_pow2(3 * n)
    cp_flip = c_p[::-1].copy()
    cpf = np.zeros(N2)
    cpf[:2 * n - 1] = cp_flip
    ap = np.zeros(N2)
    ap[:n] = a
    conv = np.fft.irfft(np.fft.rfft(cpf) * np.fft.rfft(ap), N2)
    # cross_corr[j] = sum_k c_p[k] * a[k-j] = conv[2n-2+j]
    cross = conv[2 * n - 2: 2 * n - 2 + n]
    grad = 2.0 * scale * cross / (s * s)
    # Subtract mean to account for normalization constraint (sum(a)=1)
    grad = grad - np.mean(grad)
    return grad


def _smooth_max_score(a, beta):
    """Smooth approximation of the max-based score using log-sum-exp."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0
    x = (c - c_max) / beta
    ex = np.exp(x)
    log_sum = c_max + beta * np.log(ex.sum())
    return float(2.0 * n * log_sum / (s * s))


def _smooth_max_grad(a, beta):
    """Gradient of smooth max score."""
    n = len(a)
    s = a.sum()
    c = _fft_convolve(a)
    c_max = c.max()
    x = (c - c_max) / beta
    ex = np.exp(x)
    total = ex.sum()
    if total < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = ex / total
    # grad = 2n * (dw/da) / s^2
    # w_k = exp((c_k - cmax)/beta) / sum(exp((c-cmax)/beta))
    # dw_k/da_j = (w_k/beta) * (dc_k/da_j) - w_k * sum_l (w_l/beta) * (dc_l/da_j)
    # Simplified: grad_j = (2n/s^2) * sum_k w_k * (dc_k/da_j) / beta * beta
    # Actually: d logsumexp / da_j = sum_k w_k * dc_k/da_j

    grad = 2.0 * 2.0 * n * cross / (s * s)
    grad = grad - np.mean(grad)
    return grad, float(2.0 * n * (c_max + beta * np.log(total)) / (s * s))


def _project_simplex(v):
    """Project onto the probability simplex (sum=1, all >= 0)."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_profile(n, kind='ramp', param=1.0):
    t = np.linspace(0.0, 1.0, n)
    if kind == 'ramp':
        a = t ** param + 1e-8
    elif kind == 'ramp_r':
        a = (1.0 - t) ** param + 1e-8
    elif kind == 'lin':
        a = 1.0 + param * t
    elif kind == 'quad':
        a = 1.0 + param * t + 0.5 * param * t * t
    elif kind == 'exp':
        a = np.exp(-param * t) + 1e-8
    elif kind == 'exp_r':
        a = np.exp(-param * (1.0 - t)) + 1e-8
    elif kind == 'cos':
        a = 0.5 + 0.5 * np.cos(np.pi * t) + 1e-8
    elif kind == 'bell':
        a = np.exp(-param * (t - 0.5) ** 2) + 1e-8
    elif kind == 'u_shape':
        a = (t - 0.5) ** 2 * param + 1e-8
    elif kind == 'v_shape':
        a = np.abs(t - 0.5) * param + 1e-8
    elif kind == 'sqrt':
        a = np.sqrt(t + 1e-8)
    elif kind == 'inv_sqrt':
        a = 1.0 / np.sqrt(t + 0.1)
    else:
        a = t + 1e-8
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0

        if time.monotonic() > deadline:

        p = p_schedule[p_idx]

def _solve_lp(f, M, n):
    """Solve LP: maximize sum(g) subject to conv(f,g) <= M, g >= 0."""
    # Build constraint matrix: (2n-1) constraints, n variables
    # Constraint k: sum_j f[j] * g[k-j] <= M (for valid k-j)
    # This is: A[k, j] = f[j] if 0 <= k-j < n (i.e., k-n+1 <= j <= k)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = f[:hi - lo]
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def _coordinate_descent(a0, deadline, max_sweep=100):
    """Cyclic coordinate descent for local optimization."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for sweep in range(max_sweep):
        if time.monotonic() > deadline:
            break

        improved = False
        c = _fft_convolve(a)
        M = c.max()

        for j in range(n):
            # Optimize a[j] with others fixed
            # c_k = a_j * (a[k-j] if valid) + rest
            # The contribution of a_j to c_k is:
            #   If 0 <= k-j < n and k-j != j: a_j * a[k-j]
            #   If k-j == j (k=2j): a_j^2
            #   If 0 <= j-k < n and j-k != j: a_j * a[j-k]
            # So c_k = a_j^2 * I(2j=k) + a_j * L_k + R_k
            # where L_k = a[k-j]*I(0<=k-j<n, k!=2j) + a[j-k]*I(0<=j-k<n, k!=2j)
            # and R_k = c_k - a_j * L_k - a_j^2 * I(2j=k)

            # We want to minimize max_k c_k by choosing a_j >= 0
            # subject to sum(a) = 1 (adjust later)
            # For simplicity, just do a small step: reduce a_j if it contributes to the max

            # Find which c_k are at or near the max
            # The marginal contribution of increasing a_j by delta to c_k is:
            # dc_k/da_j = 2*a_j*I(2j=k) + L_k
            # We want to reduce a_j if it's increasing the max

            # Simple approach: if a_j contributes significantly to the max, reduce it
            contrib = np.zeros(2 * n - 1)
            for k in range(max(0, j - (n - 1)), min(2 * n - 1, j + n - 1) + 1):
                dj = k - j
                if 0 <= dj < n:
                    if dj == j:  # k = 2j
                        contrib[k] += 2.0 * a[j]
                    else:
                        contrib[k] += a[dj]
                dj2 = j - k
                if 0 <= dj2 < n and dj2 != j:
                    contrib[k] += a[dj2]

            # If any contributing c_k is near the max, reduce a_j
            max_contrib = contrib.max()
            if max_contrib > 0 and c.min() > 0:
                # Reduce a_j proportionally to its contribution to the peak
                factor = 1.0 - 0.01 * (max_contrib / (M + 1e-15))
                old_val = a[j]
                a[j] = max(0.0, old_val * factor)
                if abs(a[j] - old_val) > 1e-15:
                    improved = True

        # Renormalize
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        sc = _score(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                a = best_a.copy()
                break

        if not improved:
            break

    return best_a


def _random_perturb(a, sigma=0.01, seed=0):
    """Add small random perturbation and renormalize."""
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.normal(0, sigma, n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

        for name, kind, param in profiles:
            a = _make_profile(n_t, kind, param)
            candidates.append((f'{name}_n{n_t}', n_t, a))

    # Score and sort candidates
    scored = []
    for name, n_t, a0 in candidates:
        sc = _score(a0)
        if np.isfinite(sc):
            scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])
    print(f"Initial candidates: {len(scored)}")
    if scored:
        print(f"Best initial: {scored[0][1]} score={scored[0][0]:.8f}")

    # Phase 1: Quick p-norm descent on top candidates
    top_k = min(12, len(scored))
    refined = []
    for i in range(top_k):
        if time.monotonic() > deadline - 300:
            break
        sc0, name, n_t, a0 = scored[i]
        if sc0 > best_score + 0.15:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Allocate time for p-norm descent
        remaining = deadline - time.monotonic() - 200
        if remaining < 30:
            break
        t_alloc = min(60.0, remaining * 0.15)
        if t_alloc < 5:
            break

        t_end = time.monotonic() + t_alloc
        a_pn = _pnorm_descent(a0, t_end, max_iter=2000)
        sc_pn = _score(a_pn)
        print(f"  pnorm: {name} -> {sc_pn:.10f}")
        if np.isfinite(sc_pn) and sc_pn < best_score:
            best_score = sc_pn
            best_seq = a_pn.copy()
        refined.append((sc_pn, name, n_t, a_pn))

    # Phase 2: LP refinement on best candidates
    refined.sort(key=lambda x: x[0])
    lp_top = min(6, len(refined))
    for i in range(lp_top):
        if time.monotonic() > deadline - 150:
            break
        sc0, name, n_t, a0 = refined[i]
        if sc0 > best_score + 0.05:
            continue

        remaining = deadline - time.monotonic() - 100
        if remaining < 20:
            break
        t_alloc = min(120.0, remaining * 0.3)
        if t_alloc < 10:
            break

        t_end = time.monotonic() + t_alloc
        a_lp = _lp_refine(a0, t_end, max_iter=15000, inflate_start=1.003)
        sc_lp = _score(a_lp)
        print(f"  LP: {name} -> {sc_lp:.10f}")
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()

        # Also try smooth-max on the LP result
        if time.monotonic() < deadline - 60:
            t_end2 = time.monotonic() + min(40.0, deadline - time.monotonic() - 50)
            a_sm = _smooth_max_descent(a_lp, t_end2, max_iter=3000)
            sc_sm = _score(a_sm)
            if np.isfinite(sc_sm) and sc_sm < best_score:
                best_score = sc_sm
                best_seq = a_sm.copy()
                print(f"    smooth_max -> {sc_sm:.10f}")

    # Phase 3: Try different lengths for the best sequence
    if best_seq is not None and time.monotonic() < deadline - 100:
        best_n = best_seq.size
        # Try extending to nearby lengths
        for extend_n in [best_n + 128, best_n + 256, max(512, best_n - 128)]:
            if extend_n <= 0 or extend_n > 2048:
                continue
            if time.monotonic() > deadline - 50:
                break
            a_ext = _interpolate(best_seq, extend_n)
            t_end = time.monotonic() + min(60.0, deadline - time.monotonic() - 40)
            a_pn = _pnorm_descent(a_ext, t_end, max_iter=1500)
            sc = _score(a_pn)
            print(f"  Extend to n={extend_n}: {sc:.10f}")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_pn.copy()

    # Phase 4: Coordinate descent polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        t_end = time.monotonic() + min(40.0, deadline - time.monotonic() - 20)
        a_cd = _coordinate_descent(best_seq, t_end, max_sweep=50)
        sc_cd = _score(a_cd)
        print(f"  Coord descent: {sc_cd:.10f}")
        if np.isfinite(sc_cd) and sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()

    # Phase 5: Final LP with tight inflation
    if best_seq is not None and time.monotonic() < deadline - 20:
        t_end = time.monotonic() + min(50.0, deadline - time.monotonic() - 10)
        a_final = _lp_refine(best_seq, t_end, max_iter=10000,
                             inflate_start=1.001, inflate_min=1.0002)
        sc_final = _score(a_final)
        print(f"  Final LP: {sc_final:.10f}")
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # Phase 6: Random perturbation attempts
    if best_seq is not None and time.monotonic() < deadline - 10:
        for trial in range(5):
            if time.monotonic() > deadline - 5:
                break
            a_pert = _random_perturb(best_seq, sigma=0.005, seed=1000 + trial)
            t_end = time.monotonic() + min(15.0, deadline - time.monotonic() - 3)
            a_r = _smooth_max_descent(a_pert, t_end, max_iter=1000)
            sc_r = _score(a_r)
            if np.isfinite(sc_r) and sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Perturb trial {trial}: {sc_r:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_profile(512, 'ramp', 1.5)
        best_score = _score(best_seq)

    # Final validation and output
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s * s  # keep same sum, just normalize
    else:
        final_seq = _make_profile(512, 'ramp', 1.5)

    # Ensure sum >= 0.01
    if final_seq.sum() < 0.01:
        final_seq = final_seq * (0.1 / final_seq.sum())

    # Evaluate
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
