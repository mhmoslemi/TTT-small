"""
Optimized step-function sequence search using Riemannian Adam on softmax space,
adaptive smooth-max, multi-peak subgradient, and coordinate descent polish at large n (2048-4096).
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg

                z = best_z + 0.2 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                c_curr = _fft_convolve(a)
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                no_imp = 0
                continue

        z = z_new
        a = a_new
        c_curr = c_new

        if verbose and it % 5000 == 0:
            print(f"      Adam it={it}: best={best_score:.10f}, beta={beta:.2e}")

    return best_a, best_score


def _multi_subgradient(a_init, deadline, max_iter=10000, top_k=5):
    """Subgradient targeting top-k peaks with adaptive learning rate."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    c_curr = _fft_convolve(a)
    best_score = float(2.0 * n * c_curr.max())
    best_a = a.copy()
    no_imp = 0
    base_lr = 0.4 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])

        # Exponential weighting for top peaks
        weights = np.array([np.exp(-0.2 * i) for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(4000, 5 * n):
                break

        a = a_new

    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_sweeps=50):
    """Coordinate descent: for each coordinate, find optimal value via bisection."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    c_curr = _fft_convolve(a)
    best_score = float(2.0 * n * c_curr.max())
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Random order for diversification
        order = np.random.default_rng(sweep % 1000).permutation(n)

        for pos in order:
            i = int(pos)
            a_i = a[i]
            others_sum = float(a.sum()) - a_i

            # Compute R_k for all k (convolution terms not involving a[i])
            # c[k] contribution from a[i]:
            #   if 0 <= k-i < n: a[i]*a[k-i]
            #   if 0 <= i-k < n (i.e., k <= i): a[i-k]*a[i]
            #   if k == 2i: a[i]^2 (single term, not double)

            # For efficiency, compute the effect on max
            # We want to find a_i that minimizes max_k c(k; a_i)
            # c(k; a_i) = a_i * a[k-i] + a[i-k] * a_i + R_k (for k != 2i)
            #            = a_i^2 + R_k (for k = 2i)
            # where R_k = c(k) - terms involving a[i]

            # The current c is computed with current a[i]. Let's compute R_k.
            # c(k) current includes:
            #   term1 = a[i]*a[k-i] if 0 <= k-i < n
            #   term2 = a[k-i]*a[i] if 0 <= k-i < n and k-i != i (same as term1 when indices differ)
            # Actually c(k) = sum_j a[j]*a[k-j]. The terms with a[i] are:
            #   j=i: a[i]*a[k-i] (if 0<=k-i<n)
            #   j=k-i: a[k-i]*a[i] (if 0<=k-i<n and k-i != i)
            # So contribution to c[k] from a[i] is:
            #   if k==2i: a[i]^2
            #   elif 0<=k-i<n: 2*a[i]*a[k-i]  (two terms: j=i gives a[i]*a[k-i], j=k-i gives a[k-i]*a[i])
            #   Wait, that's only if both j=i and j=k-i are in [0,n-1]
            #   if k-i is in [0,n-1] and k-i != i: two terms
            #   if k-i is NOT in [0,n-1]: zero terms (but j=i might still be valid if k-i is out of range)
            #   Actually j=i is valid iff 0<=i<n (always true) AND 0<=k-i<n
            #   j=k-i is valid iff 0<=k-i<n AND 0<=i<n (always true)
            #   So if 0<=k-i<n: we get a[i]*a[k-i] from j=i, and a[k-i]*a[i] from j=k-i
            #     These are the same term (a[i]*a[k-i]) counted twice UNLESS i==k-i
            #     If i == k-i (k=2i): only one term a[i]^2
            #     If i != k-i: two terms, both equal a[i]*a[k-i]
            #   If k-i NOT in [0,n-1]: j=i gives a[i]*a[k-i] but k-i out of range so a[k-i]=0 → 0
            #     j=k-i: k-i out of range → not a valid j → 0
            #     So no contribution.

            # Simplification: contribution to c[k] from coordinate i is:
            #   k == 2i: a[i]^2
            #   0 <= k-i < n and k-i != i: 2*a[i]*a[k-i]
            #   otherwise: 0

            # For bisection, we need: for target T, what's the max feasible a[i]?
            # Constraints:
            #   a[i]^2 + R_{2i} <= T  (for k=2i)
            #   2*a[i]*a[k-i] + R_k <= T  (for k where 0<=k-i<n, k!=2i)
            #   a[i] >= 0
            #   a[i] <= 1 (since sum=1 and others >= 0)

            # R_{2i} = c[2i] - a[i]^2
            # R_k = c[k] - 2*a[i]*a[k-i]  (for k where contribution is 2*a[i]*a[k-i])

            # This is O(n) per coordinate which is O(n^2) per sweep. For n=2048 that's ~4M per sweep.
            # Too slow for many sweeps. Let's do a simpler version: just try a few candidate values.

            # Simple approach: try a[i] = 0, current, and a few perturbations
            # Use the subgradient direction to suggest a good perturbation
            target_max = float(c_curr.max())
            # Find which k's are at or near the max
            # For coordinate i, the "pressure" is how much c[k] would increase if a[i] increases

            # Quick heuristic: if increasing a[i] would increase the max, decrease it; else increase
            # Compute dc/d a[i] at current max location
            max_k = int(np.argmax(c_curr))

            # dc[max_k]/da[i]:
            dk_di = 0.0
            if 0 <= max_k - i < n:
                dk_di += a[max_k - i]
            if max_k != 2 * i and 0 <= max_k - i < n:
                dk_di += a[max_k - i]  # second term (same value but different j)
            # Wait, this isn't quite right for the subgradient. Let me use the formula:
            # dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)  [for the two j terms]
            # Actually from the convolution: c[k] = sum_j a[j]*a[k-j]
            # dc[k]/da[i] = a[k-i] (from j=i term) + a[i-k] (from j=k-i term, which is a[k-i]*a[i])
            # Wait, the j=k-i term is a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i]
            # The j=i term is a[i]*a[k-i], derivative w.r.t. a[i] is a[k-i]
            # So dc[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n and k!=2i
            #                 = a[i] if k==2i (since a[k-i]=a[i], and it's a[i]^2, derivative is 2*a[i])
            # Wait no: if k=2i, then k-i=i, so a[k-i]=a[i], and we get 2*a[i] from the formula
            # But c[2i] includes a[i]^2 term, whose derivative is 2*a[i]. And the "two terms" argument
            # gives j=i: a[i]*a[i] derivative 2*a[i]? No, it gives a[i] (derivative of a[i]*a[k-i] w.r.t a[i] when k-i=i is a[i]... no.
            # c[2i] = ... + a[i]*a[i] + ... = ... + a[i]^2 + ...
            # dc[2i]/da[i] = 2*a[i]
            # From the formula: a[k-i] + a[k-i] where k-i=i: a[i]+a[i] = 2*a[i]. Correct.

            # So the subgradient is:
            # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0
            # (with the understanding that when k=2i, a[k-i]=a[i], giving 2*a[i])

            dk_di = 0.0
            if 0 <= max_k - i < n:
                dk_di = 2.0 * a[max_k - i]

            if dk_di > 0:
                # Increasing a[i] increases the max → try decreasing
                new_ai = max(0.0, a_i * 0.7)
            else:
                # Try increasing
                new_ai = min(1.0 - others_sum, a_i * 1.3 + 0.001)

            a[i] = new_ai
            s = float(a.sum())
            if s < 1e-10:
                a[i] = a_i
                continue
            a = a / s
            c_curr = _fft_convolve(a)
            sc = float(2.0 * n * c_curr.max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _lp_spread_fast(a, M):
    """LP: maximize sum(g) s.t. g>=0, a*g convolve <= M."""
    n = len(a)
    c_obj = -np.ones(n)
    # Build constraint matrix: (a*g)[k] <= M for all k
    # (a*g)[k] = sum_j a[j]*g[k-j]
    # For each k, the coefficient of g[j] is a[k-j] (if 0<=k-j<n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a_init, deadline, max_iter=2000, inflate=1.002, blend=0.3):
    """LP-based refinement."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    c_curr = _fft_convolve(a)
    best_score = float(2.0 * n * c_curr.max())
    best_a = a.copy()
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        M = float(c_curr.max())
        M_lp = M * inflate_cur
        g = _lp_spread_fast(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 40 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.08)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 60 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
                inflate_cur = min(1.005, max(inflate_cur * 0.9, 1.001))
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sigmoid(n, sharp=5.0, shift=0.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 / (1.0 + np.exp(-sharp * (t - 0.5 + shift)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power(n, p=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** p + 0.1
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):

            s = p.sum()
            if s > 0.01:

                if np.isfinite(sc) and sc < best_score:
                    best_score = sc

    s_alloc = min(700.0, (deadline - t_start) * 0.45)
    d1 = t_start + s_alloc

            best_seq = a4.copy()
        print(f"  CoordDesc: {best_score:.10f}")

    # Fine Adam
    if time.monotonic() < deadline - 60:
        d5 = min(deadline - 60, time.monotonic() + 100)
        a5, s5 = _adam_riemannian(best_seq, d5, max_iter=20000, lr=0.01,
                                   beta_init=0.01 * float(_fft_convolve(best_seq).max()),
                                   cosine_anneal=True)
        if s5 < best_score:
            best_score = s5
            best_seq = a5.copy()
        print(f"  Fine Adam: {best_score:.10f}")

        ]

        per_time = min(30.0, (deadline - 150 - time.monotonic()) / max(1, len(inits)))

            if time.monotonic() > deadline - 150:

        a_fp, s_fp = _adam_riemannian(best_seq, d_fp, max_iter=10000, lr=0.005,
                                       beta_init=0.005 * float(_fft_convolve(best_seq).max()),
                                       cosine_anneal=True)
        if s_fp < best_score:
            best_score = s_fp
            best_seq = a_fp.copy()
        print(f"  Final: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_linear(512, 1.5)
        best_score = _score(best_seq)

    # Ensure validity
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:

    # Validate with evaluator
    if final_seq.size <= 8192:

                if sc_eval < best_score:
                    best_score = sc_eval
            else:
                print(f"Evaluate returned non-finite: {sc_eval}")
        except Exception as e:
            print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
