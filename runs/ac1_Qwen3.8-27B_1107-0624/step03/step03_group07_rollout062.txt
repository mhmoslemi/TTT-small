"""
Projected subgradient optimization with Nesterov acceleration for minimizing
2n*max(conv(a,a))/sum(a)^2. Uses exact top-K peak subgradients, simplex projection,
and multi-resolution search with diverse one-sided initial profiles.
"""
import numpy as np
import time


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2*n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)


def grad_from_peaks(w, a):
    """Compute subgradient: 2 * sum_j w[j+m] * a[m] for j=0..n-1."""
    n = len(a)
    # w has length 2n-1, a has length n
    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # This is the cross-correlation of w and reversed a, shifted
    ar = a[::-1]
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    faf = np.fft.rfft(ar, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    # conv[n-1 + j] = sum_m ar[m] * w[n-1+j-m] = sum_m a[n-1-m] * w[n-1+j-m]
    # Let k = n-1-m: sum_k a[k] * w[j+k] 
    # So conv[n-1+j] = sum_k a[k] * w[j+k] = grad[j]/2
    return 2.0 * conv[n-1:2*n-1]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_subgradient_optimize(a0, deadline, rng, K=5, lr=None, restart_period=500):
    """
    Nesterov accelerated projected subgradient on the simplex.
    Uses exact top-K peak subgradients.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    if lr is None:
        lr = 1.0 / np.sqrt(n)
    
    base_lr = lr
    best_a = a.copy()
    
    c = full_convolve_fft(a)
    best_score = float(2.0 * n * c.max())
    
    # Nesterov state
    a_prev = a.copy()
    t = 1.0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Nesterov extrapolation
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        y = a + (t_prev - 1.0) / t * (a - a_prev)
        y = np.maximum(y, 0.0)
        y = y / max(y.sum(), 1e-15)
        
        # Compute convolution and find top-K peaks
        c = full_convolve_fft(y)
        max_c = float(c.max())
        
        # Top-K peak selection
        if K >= 2*(n-1):
            # Uniform over all peaks
            w = np.ones(2*n-1) / (2*n-1)
        else:
            threshold_idx = np.argsort(c)[-K:]
            c_max = c.max()
            # Use a soft threshold: include peaks within 5% of max
            mask = c >= c_max * 0.95
            if np.count_nonzero(mask) < K:
                mask = np.zeros_like(c)
                mask[threshold_idx] = 1.0
            w = mask.astype(np.float64) / mask.sum()
        
        # Subgradient
        g = grad_from_peaks(w, y)
        
        # Step size adaptation
        # Use a diminishing step: lr / sqrt(t)
        step_lr = base_lr / (1.0 + 0.1 * t)
        
        # Projected step
        a_new = project_simplex(y - step_lr * g)
        
        a_prev = a
        a = a_new
        
        # Evaluate
        c = full_convolve_fft(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > restart_period:
            # Jitter and restart
            a = best_a + 0.05 * rng.normal(size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            a_prev = a.copy()
            t = 1.0
            no_improve = 0
            base_lr = 0.8 * base_lr
        
        # Reduce K over time (sharpen peak focusing)
        if no_improve % 200 == 0 and K > 1:
            K = max(1, K // 2)
            base_lr = lr  # Reset lr
    
    return best_a, best_score

Hybrid L-BFGS-B + projected subgradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses scipy L-BFGS-B in logit space with adaptive soft-max beta annealing,
projected Nesterov subgradient as secondary method, polynomial initial profiles,

import numpy as np

def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        cv = np.convolve(a, a)
        return np.maximum(cv, 0.0)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def grad_from_peaks(w, a):
    """Compute 2 * sum_m w[j+m] * a[m] for j=0..n-1 via FFT."""
    n = len(a)
    ar = a[::-1]
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    faf = np.fft.rfft(ar, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return 2.0 * conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_obj_and_grad(z, beta):
    """Objective and gradient for soft-max of conv(a,a) in logit space."""
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s
    c = full_convolve_fft(a)
    csh = c / beta
    csh = csh - csh.max()
    csh = np.clip(csh, -50.0, 0.0)
    ex = np.exp(csh)
    s_ex = ex.sum()
    if s_ex < 1e-300:
        ex = np.zeros_like(c)
        ex[np.argmax(c)] = 1.0
        s_ex = 1.0
    log_sum = np.log(s_ex)
    val = beta * (log_sum + (c / beta).max())
    w = ex / s_ex
    g_a = 2.0 * grad_from_peaks(w, a)
    dot = np.dot(a, g_a)
    g_z = a * (g_a - dot)
    return val, g_z

def subgradient_nesterov_optimize(a0, deadline, rng, K=10, base_lr=0.02):
    """
    Nesterov accelerated projected subgradient with exact top-K peaks.
    """
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 0.0)
    a = a / max(a.sum(), 1e-15)

    best_score = float(2.0 * n * c.max())

    a_prev = a.copy()
    t = 1.0
    no_improve = 0
    current_lr = base_lr
    current_K = K

    while time.monotonic() < deadline:
        # Nesterov extrapolation
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        y = a + (t_prev - 1.0) / t * (a - a_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = best_a.copy()
        else:
            y = y / ys

        # Convolution and top-K
        c = full_convolve_fft(y)
        c_max = float(c.max())

        # Adaptive K: use peaks within 10% of max, at most current_K
        mask = c >= c_max * 0.90
        n_mask = np.count_nonzero(mask)
        if n_mask == 0:
            mask = np.zeros_like(c)
            mask[np.argmax(c)] = 1.0
        elif n_mask > current_K:
            idx = np.argsort(c)[-current_K:]
            mask = np.zeros_like(c)
            mask[idx] = 1.0
        w = mask.astype(np.float64) / mask.sum()

        # Subgradient
        g = grad_from_peaks(w, y)

        # Step with diminishing lr
        step_lr = current_lr / (1.0 + 0.05 * np.log1p(t))
        a_new = project_simplex(y - step_lr * g)

        a_prev = a
        a = a_new

        # Evaluate
        c = full_convolve_fft(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt

            current_lr *= 0.95
        if no_improve > 300:
            current_K = max(1, current_K // 2)
        if no_improve > 1000:
            # Restart with noise
            a = best_a + 0.03 * rng.normal(size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            a_prev = a.copy()
            t = 1.0
            no_improve = 0
            current_lr = base_lr

    return best_a, best_score


def make_polynomial_profile(n, p):
    """a[j] proportional to (1 + j/n)^p, normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 + 0.5 * t) ** p + 0.01
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_increasing_profile(n, p=1.0):
    """a[j] proportional to (j+1)^p, normalized (one-sided)."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** p
    return a / a.sum()


def make_bell_profile(n, center=0.7, width=0.3):
    """Gaussian-like profile centered off-center."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - center) / width) ** 2)
    a = a + 0.01
    return a / a.sum()


def make_random_smooth(n, rng, n_freq=5):
    """Random smooth profile via low-frequency Fourier coefficients."""
    coeffs = rng.uniform(-0.3, 0.3, n_freq)
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for i, c_val in enumerate(coeffs, 1):
        a = a + c_val * np.cos(2.0 * np.pi * i * t)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result
