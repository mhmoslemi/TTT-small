The projected subgradient step is:

where Proj_{simplex} is the Euclidean projection onto {a >= 0, sum(a) = 1}.

For the smooth max approximation:

∇f_smooth(a) = sum_k (exp(c_k/beta) / sum_j exp(c_j/beta)) * ∇c_k(a)

where w_k = exp(c_k/beta) / sum_j exp(c_j/beta).

This is a weighted average of the gradients of all c_k, weighted by their "importance" (higher c_k gets more weight).

For the projection onto the simplex, I'll use the standard algorithm:

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using asymmetric initialization,
projected subgradient descent, and LP-based spreading refinement.
Key insight: optimal sequences are asymmetric (symmetric sequences have score >= 2).
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    n = len(a)
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_conv_matrix(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

def _make_exp_profile(n, tau_frac=0.3):
    """Exponential decay profile: a[i] proportional to exp(-i/(tau*n))."""
    t = np.arange(n) / n
    tau = tau_frac
    a = np.exp(-t / tau)
    a = a / a.sum()
    return a


def _make_power_profile(n, p=0.5):
    """Power law profile: a[i] proportional to (i+1)^p."""
    i = np.arange(n) + 1
    a = i ** p
    a = a / a.sum()
    return a


def _make_skew_profile(n, alpha=2.0, beta_val=3.0):
    """Beta distribution profile (skewed)."""
    # Use beta distribution PDF
    i = np.arange(n) / max(n - 1, 1)
    i = np.clip(i, 1e-6, 1 - 1e-6)
    a = i ** (alpha - 1) * (1 - i) ** (beta_val - 1)
    a = a / a.sum()
    return a


def _make_trend_noise_profile(n, rng, base='exp', noise_level=0.3):
    """Add noise to a base profile."""
    t = np.arange(n) / n
    if base == 'exp':
        a = np.exp(-t / 0.3)
    elif base == 'power':
        a = (np.arange(n) + 1) ** 0.5
    else:
        a = np.ones(n)
    a = a / a.sum()
    a = a * (1.0 + noise_level * rng.standard_normal(n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using:

2. Projected subgradient descent with smooth max and cross-correlation gradient

import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    """Fast convolution using FFT."""
    n = len(a)
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(a):
    """Compute the objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_conv_matrix(a):
    """Build (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    """LP: max sum(g) s.t. conv_matrix(a) @ g <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def _compute_smooth_max_grad(a, beta):
    """Compute smooth max value and gradient w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    c_max = c.max()
    x = (c - c_max) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    w = ex / ex.sum()
    val = c_max + beta * float(np.log(ex.sum()))
    
    # Gradient: dF/da[j] = 2 * sum_i w[i+j] * a[i] (cross-correlation)
    N2 = _next_pow2(3 * n)
    fw = np.fft.rfft(w, N2)
    fa = np.fft.rfft(a, N2)
    corr = np.fft.irfft(fw * np.conj(fa), N2)
    grad = 2.0 * corr[:n]
    return val, grad, c, w, c_max

def _make_exp_profile(n, tau_frac=0.3):
    """Exponential decay: a[i] ~ exp(-i/(tau*n))."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-t / tau_frac)
    return a / a.sum()


def _make_power_profile(n, p=0.5):
    """Power law: a[i] ~ (i+1)^p."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** p
    return a / a.sum()


def _make_beta_profile(n, alpha=2.0, beta_val=3.0):
    """Skewed beta distribution profile."""
    i = np.linspace(1e-6, 1 - 1e-6, n)
    a = i ** (alpha - 1) * (1 - i) ** (beta_val - 1)
    return a / a.sum()


def _make_bimodal_profile(n, center1=0.3, center2=0.7, width=0.1):
    """Bimodal profile."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - center1) / width) ** 2) + np.exp(-((t - center2) / width) ** 2)
    return a / a.sum()


def _upsample(a, n_new):
    """Upsample a profile to a new length using interpolation."""
    n = len(a)
    if n == n_new:
        return a.copy() / a.sum()
    x0 = np.linspace(0, 1, n)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s

        gz = a * (grad - np.dot(a, grad))
        gz -= np.mean(gz)
        
        if not np.all(np.isfinite(gz)):
            break
        
        t += 1
        m = b1 * m + (1 - b1) * gz
        v_arr = b2 * v_arr + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t)
        vh = v_arr / (1 - b2 ** t)
        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= np.mean(z_new)
        
        e = np.exp(z_new)
        a = e / e.sum()
        z = z_new
    
    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Key insight: optimal sequences are asymmetric (symmetric => score >= 2).
    Uses subgradient descent + LP refinement with multiple asymmetric inits.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 45.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_score = np.inf
    best_seq = None
    
    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass
    
    # Lengths to explore
    lengths = [768, 1024, 1536]
    if parent is not None:
        pn = parent.size
        if pn not in lengths and pn <= 2048:
            lengths.insert(0, pn)
    
    # Phase 1: Diversified initialization
    print(f"Phase 1: Generating initializations...", flush=True)
    init_pool = []
    
    for n_t in lengths:
        # Parent interpolation
        if parent is not None:
            init_pool.append((f'parent_{n_t}', n_t, _upsample(parent, n_t)))
        
        # Exponential profiles (asymmetric)
        for tau in [0.15, 0.2, 0.3, 0.4, 0.6]:
            init_pool.append((f'exp_{tau}_{n_t}', n_t, _make_exp_profile(n_t, tau)))
        
        # Power law profiles
        for p in [-1.0, -0.3, 0.2, 0.7, 1.5]:
            init_pool.append((f'power_{p}_{n_t}', n_t, _make_power_profile(n_t, p)))
        
        # Beta (skewed) profiles
        for a_, b_ in [(1.5, 2.5), (2.0, 4.0), (0.7, 1.5), (3.0, 2.0)]:
            init_pool.append((f'beta_{a_}_{b_}_{n_t}', n_t, _make_beta_profile(n_t, a_, b_)))
        
        # Bimodal
        init_pool.append((f'bimodal_{n_t}', n_t, _make_bimodal_profile(n_t, 0.25, 0.7, 0.12)))
    
    # Quick evaluation and sorting
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])
    
    print(f"  Top 10 initializations:", flush=True)
    for sc, name, n_t, _ in init_scores[:10]:
        print(f"    {name}: {sc:.8f}", flush=True)
    
    # Select diverse top candidates (at most 2 per length)
    selected = []
    count_per_len = {}
    for sc, name, n_t, a0 in init_scores:
        if len(selected) >= 6:
            break
        if count_per_len.get(n_t, 0) >= 2:
            continue
        if sc > best_score * 1.2 and n_t != lengths[0]:
            continue
        selected.append((sc, name, n_t, a0))
        count_per_len[n_t] = count_per_len.get(n_t, 0) + 1
    
    if len(selected) == 0:
        selected = init_scores[:3]
    
    print(f"\nSelected {len(selected)} candidates for optimization.", flush=True)
    
    # Phase 2: Main optimization loop
    for idx, (sc0, name, n_t, a0) in enumerate(selected):
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        
        # Allocate time per candidate
        per_candidate = min(remaining / max(1, len(selected) - idx), 500.0)
        
        print(f"\n--- Candidate {idx+1}: {name} (n={n_t}), init={sc0:.8f}, budget={per_candidate:.0f}s ---", flush=True)
        
        a_curr = a0.copy()
        
        # Step A: Subgradient descent (main workhorse)
        sg_deadline = time.monotonic() + per_candidate * 0.45
        a_sg = _subgradient_descent(a_curr, sg_deadline, lr0=0.015, max_iter=200000, kick_interval=4000)
        sc_sg = _score(a_sg)
        print(f"  Subgradient: {sc_sg:.10f} (improved by {sc0 - sc_sg:.6f})", flush=True)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
        a_curr = a_sg
        
        # Step B: LP refinement
        if time.monotonic() < deadline - 80:
            lp_budget = min(per_candidate * 0.35, deadline - time.monotonic() - 60)
            if lp_budget > 20:
                a_lp = _lp_refine(a_curr, time.monotonic() + lp_budget,
                                  max_iter=5000, inflate=1.001, blend=0.3)
                sc_lp = _score(a_lp)
                print(f"  LP refine: {sc_lp:.10f} (improved by {sc_sg - sc_lp:.6f})", flush=True)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                a_curr = a_lp
        
        # Step C: Adam logit optimization
        if time.monotonic() < deadline - 60:
            ad_budget = min(per_candidate * 0.15, deadline - time.monotonic() - 40)
            if ad_budget > 15:
                a_ad = _adam_logit_optimize(a_curr, time.monotonic() + ad_budget,
                                            max_iter=30000, lr=0.02)
                sc_ad = _score(a_ad)
                print(f"  Adam logit: {sc_ad:.10f}", flush=True)
                if sc_ad < best_score:
                    best_score = sc_ad
                    best_seq = a_ad.copy()
                a_curr = a_ad
        
        # Step D: Second LP pass
        if time.monotonic() < deadline - 40:
            lp2_budget = min(per_candidate * 0.15, deadline - time.monotonic() - 30)
            if lp2_budget > 15:
                a_lp2 = _lp_refine(a_curr, time.monotonic() + lp2_budget,
                                   max_iter=3000, inflate=1.0008, blend=0.2)
                sc_lp2 = _score(a_lp2)
                print(f"  LP refine 2: {sc_lp2:.10f}", flush=True)
                if sc_lp2 < best_score:
                    best_score = sc_lp2
                    best_seq = a_lp2.copy()
    
    # Phase 3: Extended aggressive LP on the overall best
    if best_seq is not None and time.monotonic() < deadline - 20:
        lp3_deadline = min(deadline - 10, time.monotonic() + 300.0)
        remaining3 = lp3_deadline - time.monotonic()
        if remaining3 > 30:
            print(f"\nPhase 3: Aggressive LP on best (score={best_score:.10f}, budget={remaining3:.0f}s)", flush=True)
            a_final = _lp_refine_aggressive(best_seq, lp3_deadline, max_iter=15000)
            sc_final = _score(a_final)
            print(f"  Aggressive LP result: {sc_final:.10f}", flush=True)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
    
    # Phase 4: Final subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        sg4_deadline = deadline - 5
        remaining4 = sg4_deadline - time.monotonic()
        if remaining4 > 20:
            print(f"\nPhase 4: Final subgradient polish (budget={remaining4:.0f}s)", flush=True)
            a_polish = _subgradient_descent(best_seq, sg4_deadline, lr0=0.005, max_iter=50000, kick_interval=8000)
            sc_polish = _score(a_polish)
            print(f"  Polish result: {sc_polish:.10f}", flush=True)
            if sc_polish < best_score:
                best_score = sc_polish
                best_seq = a_polish.copy()
    
    # Final validation
    if best_seq is None:
        best_seq = _make_exp_profile(768, 0.3)
        best_score = _score(best_seq)
    
    final_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    s = float(final_seq.sum())
    if s <= 0.01:
        final_seq = _make_exp_profile(768, 0.3)
    
    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence score: {sc_eval:.10f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}", flush=True)
    
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
