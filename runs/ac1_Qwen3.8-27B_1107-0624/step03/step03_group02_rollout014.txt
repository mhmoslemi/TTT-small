Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam in log-domain with
smooth-max, subgradient, and LP refinement. Multiple sequence lengths explored.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)  # Clip tiny negative values from FFT

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:

        else:
            w = ex / s_ex

        grad_a = _compute_grad(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= grad_z.mean()

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            no_imp += 100
            continue

        t_count += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_count)
        vh = v / (1.0 - b2 ** t_count)

        lr_cur = base_lr * (0.96 ** (no_imp // 100))
        lr_cur = max(0.0005, lr_cur)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= z_new.mean()

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = 2.0 * n * float(c_new.max())

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > max(2000, 3 * n):
            rng = np.random.default_rng(12345 + it * 7)
            z = best_z + 0.4 * rng.standard_normal(n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            e = np.exp(z)
            a = e / e.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            no_imp = 0
        else:
            z = z_new
            a = a_new

    return best_a, best_score


def _subgradient_opt(a0, deadline, max_iter=10000, top_k=3):
    """Subgradient method targeting top-k peaks."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, _score(a)
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(c.max())

    base_lr = 0.3 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_idx = np.argsort(c_curr)[::-1][:k]
        weights = np.array([1.0 / (1.0 + 0.5 * i) for i in range(k)])
        weights /= weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_idx):
            idx = int(idx)
            j1s = max(0, idx - n + 1)
            j1e = min(idx, n - 1)
            if j1s <= j1e:
                jj = np.arange(j1s, j1e + 1)
                sg[jj] += weights[i] * a[idx - jj]
            j2s = max(idx, 0)
            j2e = min(idx + n - 1, n - 1)
            if j2s <= j2e:
                jj = np.arange(j2s, j2e + 1)
                sg[jj] += weights[i] * a[jj - idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = 2.0 * n * float(c_new.max())

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _lp_spread(a, M):
    """LP: maximize sum(x) s.t. conv(a,x) <= M, x >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=1000, blend=0.3):
    """Iterative LP-based spreading."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(c.max())

    no_imp = 0
    inflate = 1.002

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max()) * inflate
        g = _lp_spread(a, M)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp > 100:
                break
            inflate = min(1.01, inflate * 1.03)
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = 2.0 * n * float(c_new.max())

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend = max(0.05, blend * 0.85)
                inflate = max(1.0005, inflate * 0.9)
            if no_imp > 200:
                break

        a = a_new

    return best_a


def _interpolate_to(a0, n_new):
    p0 = np.asarray(a0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: Adam smooth-max, multi-peak subgradient,
    and LP refinement. Explores parent sequence and multiple lengths.
    """
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            if p.sum() > 0.01:
                parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        n_parent = parent.size
        p_score = _score(parent)
        print(f"Parent: n={n_parent}, score={p_score:.10f}")
        if p_score < best_score:
            best_score = p_score
            best_seq = parent.copy()
    else:
        n_parent = 1024

    # Strategy 1: Uniform (theoretical optimum)
    uniform = np.ones(n_parent)
    u_score = _score(uniform)
    print(f"Uniform n={n_parent}: score={u_score:.10f}")
    if u_score < best_score:
        best_score = u_score
        best_seq = uniform.copy()

    # Strategy 2: Adam from parent at native length
    if parent is not None and time.monotonic() < deadline - 300:
        print("=== Adam on parent ===")
        d1 = min(deadline - 300, t0 + 400)
        a1, s1 = _adam_opt(parent, d1, max_iter=25000, lr=0.025, beta0=0.05)
        if s1 < best_score:
            best_score = s1
            best_seq = a1.copy()
        print(f"Adam parent: {best_score:.10f}")

        # Subgradient refinement
        if time.monotonic() < deadline - 200:
            d1b = min(deadline - 200, time.monotonic() + 150)
            a1b, s1b = _subgradient_opt(best_seq, d1b, max_iter=5000, top_k=3)
            if s1b < best_score:
                best_score = s1b
                best_seq = a1b.copy()
            print(f"Subgrad: {best_score:.10f}")

        # LP refinement
        if time.monotonic() < deadline - 150:
            d1c = min(deadline - 150, time.monotonic() + 100)
            a1c = _lp_refine(best_seq, d1c, max_iter=500, blend=0.3)
            s1c = _score(a1c)
            if s1c < best_score:
                best_score = s1c
                best_seq = a1c.copy()
            print(f"LP: {best_score:.10f}")

    # Strategy 3: Explore different lengths
    lengths_to_try = [256, 512, 1024, 2048]
    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent and parent is not None:
            continue

        if parent is not None:
            init = _interpolate_to(parent, n_try)
        else:
            init = np.ones(n_try)

        # Add some diversity
        init_div = init * (1.0 + 0.1 * rng.standard_normal(n_try))
        init_div = np.maximum(init_div, 1e-10)
        init_div /= init_div.sum()

        d3 = min(deadline - 100, time.monotonic() + 180)
        a3, s3 = _adam_opt(init_div, d3, max_iter=12000, lr=0.02, beta0=0.04)
        if s3 < best_score:
            best_score = s3
            best_seq = a3.copy()
        print(f"n={n_try} Adam: {s3:.10f} (best={best_score:.10f})")

        # Quick subgradient
        if time.monotonic() < deadline - 80:
            d3b = min(deadline - 80, time.monotonic() + 80)
            a3b, s3b = _subgradient_opt(a3, d3b, max_iter=3000, top_k=3)
            if s3b < best_score:
                best_score = s3b
                best_seq = a3b.copy()
            print(f"n={n_try} Subgrad: {best_score:.10f}")

    # Strategy 4: Focus on best with fine-tuning
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"=== Final focus: {best_score:.10f}, n={best_seq.size} ===")

        # Fine Adam
        d4 = min(deadline - 15, time.monotonic() + 120)
        a4, s4 = _adam_opt(best_seq, d4, max_iter=15000, lr=0.008, beta0=0.015)
        if s4 < best_score:
            best_score = s4
            best_seq = a4.copy()
        print(f"Fine Adam: {best_score:.10f}")

        # Subgradient polish
        if time.monotonic() < deadline - 10:
            d4b = min(deadline - 10, time.monotonic() + 60)
            a4b, s4b = _subgradient_opt(best_seq, d4b, max_iter=3000, top_k=5)
            if s4b < best_score:
                best_score = s4b
                best_seq = a4b.copy()
            print(f"Final Subgrad: {best_score:.10f}")

        # LP polish
        if time.monotonic() < deadline - 5:
            d4c = min(deadline - 5, time.monotonic() + 40)
            a4c = _lp_refine(best_seq, d4c, max_iter=300, blend=0.2)
            s4c = _score(a4c)
            if s4c < best_score:
                best_score = s4c
                best_seq = a4c.copy()
            print(f"Final LP: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(1024)
        best_score = 2.0

    final = np.maximum(best_seq, 0.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(1024)
        best_score = 2.0

    result = [float(x) for x in final]
    print(f"FINAL: score={best_score:.10f}, n={len(result)}")

    # Verify
    try:
        ev = evaluate_sequence(result)
        print(f"evaluate_sequence: {ev:.10f}")
    except Exception:
        pass

    return result
