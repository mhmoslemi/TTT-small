"""
Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Larger sequence lengths (4096) with FFT-based convolution
2. Adam smooth-max optimization in logit space
3. Coordinate descent with incremental convolution updates
4. LP-based refinement
5. Multi-restart with diverse initializations
Target: beat 1.506320, aim for ~1.5027.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=15):
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score_norm(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.1, t * 0.85)
    return current_best, current_score


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.08, beta_decay=0.93,
                no_imp_limit=300, noise_scale=None):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 40 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.5)

        limit = max(80, no_imp_limit)
        if no_imp > limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.06
                elif best_score < 1.53:
                    ns = 0.12
                else:
                    ns = 0.25
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _coord_descent(a, c, n_passes=1, deadline=None, n_ternary=6):
    """Coordinate descent minimizing max convolution. Modifies a and c in place."""
    n = len(a)
    two_n_1 = 2 * n - 1

    for _pass in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            # Affected indices: k from j to min(2n-2, j+n-1)
            k_lo = j
            k_hi = min(two_n_1 - 1, j + n - 1)
            L = k_hi - k_lo + 1
            if L <= 0:
                continue

            affected_k = np.arange(k_lo, k_hi + 1)
            # a[k-j] for k in affected range
            akj = a[affected_k - j]

            # Current values at affected positions
            c_affected = c[affected_k].copy()
            cur_a = a[j]

            # Base values (removing contribution of a[j])
            # For k = 2j (if in range): contribution is a[j]^2
            # For k != 2j: contribution is 2*a[j]*a[k-j]
            base = c_affected.copy()
            # Determine which affected k has k == 2j
            diag_mask = (affected_k == 2 * j)
            if np.any(diag_mask):
                base[diag_mask] -= cur_a * cur_a
            else:
                base -= 2.0 * cur_a * akj
                # But for the diagonal, we subtracted too much; fix it
                # Actually if diag is not in range, no fix needed
            if not np.any(diag_mask):
                # All affected k have k != 2j, so base = c - 2*a[j]*a[k-j]
                pass
            else:
                # For non-diagonal, base = c - 2*a[j]*a[k-j]
                # For diagonal, base = c - a[j]^2
                non_diag = ~diag_mask
                base[non_diag] = c_affected[non_diag] - 2.0 * cur_a * akj[non_diag]
                base[diag_mask] = c_affected[diag_mask] - cur_a * cur_a

            # C = max of unaffected convolution values
            if k_lo > 0 or k_hi < two_n_1 - 1:
                mask_unaffected = np.ones(two_n_1, dtype=bool)
                mask_unaffected[k_lo:k_hi + 1] = False
                C = float(c[mask_unaffected].max()) if mask_unaffected.any() else 0.0
            else:
                C = 0.0

            # Now find optimal x in [0, x_upper] minimizing:
            # f(x) = max(C, x^2 + B_diag (if exists), max(2*x*akj + B_i))
            has_diag = np.any(diag_mask)
            if has_diag:
                B_diag = float(base[diag_mask][0])
            else:
                B_diag = -np.inf

            # Quick check: at x=0, what's the max?
            f_at_0 = max(C, B_diag, float(np.max(base))) if has_diag else max(C, float(np.max(base)))

            if f_at_0 <= C + 1e-12:
                # x=0 already makes all affected <= C
                new_x = 0.0
            else:
                # Do ternary search in [0, x_upper]
                # x_upper: ensure quadratic doesn't blow up
                x_upper = 2.0 * cur_a + 1.0 / max(n, 1)
                # Also limit by linear constraints: 2*x*max(akj) + max(base) <= f_at_0
                max_akj = float(akj.max()) if len(akj) > 0 else 0.0
                if max_akj > 0:
                    x_upper = min(x_upper, 2.0 * f_at_0 / max_akj)
                x_upper = max(x_upper, cur_a * 2.0)
                x_upper = min(x_upper, 2.0 / max(n, 1))  # cap reasonably

                def _eval_f(x):
                    val_quad = x * x + B_diag if has_diag else -np.inf
                    val_lin = np.max(2.0 * x * akj + base)
                    return max(C, val_quad, val_lin)

                # Ternary search
                a1, b1s = 0.0, x_upper
                for _t in range(n_ternary):
                    m1 = a1 + (b1s - a1) / 3.0
                    m2 = b1s - (b1s - a1) / 3.0
                    f1 = _eval_f(m1)
                    f2 = _eval_f(m2)
                    if f1 < f2:
                        b1s = m2
                    else:
                        a1 = m1
                new_x = (a1 + b1s) / 2.0

            # Update a[j] and c
            delta = new_x - cur_a
            if abs(delta) > 1e-15:
                a[j] = new_x
                # Update convolution
                if has_diag:
                    diag_idx = affected_k[diag_mask][0]
                    c[diag_idx] += 2.0 * cur_a * delta + delta * delta
                non_diag_mask = ~diag_mask if has_diag else np.ones(L, dtype=bool)
                if np.any(non_diag_mask):
                    c[affected_k[non_diag_mask]] += 2.0 * delta * akj[non_diag_mask]

    return a, c


def _coord_descent_vectorized(a, max_time=None, n_ternary=5, batch_size=256):
    """More efficient coordinate descent processing coordinates in batches."""
    n = len(a)
    two_n_1 = 2 * n - 1
    c = _full_convolve(a)
    t_start = time.monotonic()
    passes = 0

    while max_time is None or time.monotonic() - t_start < max_time:
        passes += 1
        for j_start in range(0, n, batch_size):
            j_end = min(j_start + batch_size, n)
            for j in range(j_start, j_end):
                if max_time is not None and time.monotonic() - t_start > max_time:
                    return a, c

                k_lo = j
                k_hi = min(two_n_1 - 1, j + n - 1)
                L = k_hi - k_lo + 1
                if L <= 0:
                    continue

                affected_k = np.arange(k_lo, k_hi + 1)
                akj = a[affected_k - j]
                cur_a = a[j]

                # Base: remove a[j]'s contribution
                base = c[affected_k] - 2.0 * cur_a * akj
                diag_pos = 2 * j
                if k_lo <= diag_pos <= k_hi:
                    # Diagonal: we subtracted 2*cur_a*cur_a but should subtract cur_a^2
                    # So base at diagonal = c - 2*cur_a^2 + cur_a^2 = c - cur_a^2
                    di = diag_pos - k_lo
                    base[di] = c[diag_pos] - cur_a * cur_a

                # C = max unaffected
                if k_lo > 0:
                    C = float(max(c[:k_lo].max(), c[k_hi + 1:].max())) if k_hi + 1 < two_n_1 else float(c[:k_lo].max())
                elif k_hi < two_n_1 - 1:
                    C = float(c[k_hi + 1:].max())
                else:
                    C = 0.0

                # Find optimal x
                f_at_0 = float(np.max(base))
                if f_at_0 <= C + 1e-12:
                    new_x = 0.0
                else:
                    # Quick balance: try to set max base + 2*x*akj = C
                    # x_candidate = (C - max(base)) / (2 * max(akj)) if positive
                    max_base = float(np.max(base))
                    if max_base > C:
                        # Need to reduce: best x might be negative, but x>=0
                        # Try x=0 first
                        new_x = 0.0
                        # Check if a small x helps
                        # At x=0: max = max(C, max_base) = max_base
                        # We want to find x that minimizes max(C, 2*x*akj + base, x^2 + B_diag)
                        # Simple: try x that makes the top element = C
                        top_idx = int(np.argmax(base))
                        if akj[top_idx] > 1e-15:
                            x_cand = max(0.0, (C - base[top_idx]) / (2.0 * akj[top_idx]))
                            if x_cand > 0 and x_cand < 1.0 / n:
                                # Evaluate a few candidates
                                x_cands = [0.0, x_cand, cur_a * 0.5, cur_a]
                                x_cands = [max(0.0, min(x, 2.0 / n)) for x in x_cands in [x_cands]]
                                best_x = 0.0
                                best_f = _eval_coord_f(0.0, C, base, akj, k_lo, k_hi, j, two_n_1)
                                for xc in x_cands:
                                    fc = _eval_coord_f(xc, C, base, akj, k_lo, k_hi, j, two_n_1)
                                    if fc < best_f:
                                        best_f = fc
                                        best_x = xc
                                new_x = best_x
                            else:
                                new_x = 0.0
                        else:
                            new_x = 0.0
                    else:
                        new_x = 0.0

                delta = new_x - cur_a
                if abs(delta) > 1e-15:
                    a[j] = new_x
                    c[affected_k] += 2.0 * delta * akj
                    diag_pos = 2 * j
                    if k_lo <= diag_pos <= k_hi:
                        di = diag_pos - k_lo
                        c[diag_pos] -= 2.0 * delta * cur_a  # fix double count
                        # Actually: the correct update for diagonal is delta*(2*cur_a + delta)
                        # We added 2*delta*cur_a (from the vectorized update), but should add delta*(2*cur_a+delta)
                        # So we need to add delta*delta extra
                        c[diag_pos] += delta * delta
                # Re-verify: at diagonal, true contribution change = new^2 - old^2 = (new-old)(new+old) = delta*(2*cur_a + delta)
                # We did: 2*delta*akj[di] = 2*delta*cur_a (since akj[di] = a[diag_pos - j] = a[j] = cur_a)
                # So we're missing delta*delta. Fix:
                if k_lo <= diag_pos <= k_hi:
                    di = diag_pos - k_lo
                    c[diag_pos] += delta * delta

    return a, c


def _eval_coord_f(x, C, base, akj, k_lo, k_hi, j, two_n_1):
    """Evaluate the max convolution value for a given x at coordinate j."""
    val_lin = float(np.max(2.0 * x * akj + base))
    diag_pos = 2 * j
    if k_lo <= diag_pos <= k_hi:
        di = diag_pos - k_lo
        # At diagonal, the value is x^2 + (base[di] + 2*cur_a*akj[di] - 2*x*akj[di])
        # Actually base already has the correct "rest" value
        # The value at diagonal = x^2 + base[di] (since base[di] = c[diag] - a[j]^2)
        # Wait, base[di] = c[diag] - cur_a^2 (the rest without a[j])
        # New value at diagonal = x^2 + base[di]
        val_quad = x * x + float(base[di])
        return max(C, val_quad, val_lin)
    else:
        return max(C, val_lin)


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_symmetric_cos(n, freq, amp):
    """Symmetric cosine profile: 1 + amp*cos(2*pi*freq*t) where t in [0,1], symmetric about 0.5"""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(np.pi * freq * (2.0 * t - 1.0))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_thickened(n, center_width):
    """Thickened center: higher in middle, lower at edges."""
    t = np.linspace(-1.0, 1.0, n)
    a = 1.0 + center_width * np.exp(-4.0 * t * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def _fast_coord_pass(a, c, n_ternary=4):
    """Single fast coordinate descent pass with minimal overhead."""
    n = len(a)
    two_n_1 = 2 * n - 1
    cur_time = time.monotonic()

    for j in range(n):
        k_lo = j
        k_hi = min(two_n_1 - 1, j + n - 1)
        akj = a[k_lo:k_hi + 1 + (k_hi - k_lo) - j + j]  # a[affected_k - j]
        # Actually: affected_k = arange(k_lo, k_hi+1), so akj = a[affected_k - j] = a[k_lo-j : k_hi+1-j]
        akj = a[k_lo - j:k_hi + 1 - j]
        if akj.size == 0:
            continue

        cur_a = a[j]
        base = c[k_lo:k_hi + 1] - 2.0 * cur_a * akj

        diag_pos = 2 * j
        if k_lo <= diag_pos <= k_hi:
            di = diag_pos - k_lo
            base[di] = c[diag_pos] - cur_a * cur_a

        # C: max of unaffected
        if k_lo == 0 and k_hi == two_n_1 - 1:
            C = 0.0
        elif k_lo == 0:
            C = float(c[k_hi + 1:].max())
        elif k_hi == two_n_1 - 1:
            C = float(c[:k_lo].max())
        else:
            C = float(max(c[:k_lo].max(), c[k_hi + 1:].max()))

        # Quick optimization
        max_base = float(np.max(base))
        if max_base <= C + 1e-13:
            new_x = 0.0
        else:
            # Find x that balances top constraint with C
            # Try a few candidates
            x_cands = np.array([0.0, cur_a * 0.3, cur_a * 0.6, cur_a, cur_a * 1.4])
            # Also try the "balancing" value
            top_i = int(np.argmax(base))
            if akj[top_i] > 1e-15 and base[top_i] > C:
                x_bal = (C - base[top_i]) / (2.0 * akj[top_i])
                if x_bal > 0:
                    x_cands = np.append(x_cands, x_bal)
            x_cands = np.clip(x_cands, 0.0, 4.0 / n)
            x_cands = np.unique(x_cands)

            best_x = 0.0
            best_f = np.inf
            for xc in x_cands:
                val_lin = float(np.max(2.0 * xc * akj + base))
                if k_lo <= diag_pos <= k_hi:
                    val_quad = xc * xc + base[diag_pos - k_lo]
                    f = max(C, val_quad, val_lin)
                else:
                    f = max(C, val_lin)
                if f < best_f:
                    best_f = f
                    best_x = xc
            new_x = best_x

        delta = new_x - cur_a
        if abs(delta) > 1e-14:
            a[j] = new_x
            c[k_lo:k_hi + 1] += 2.0 * delta * akj
            if k_lo <= diag_pos <= k_hi:
                c[diag_pos] += delta * delta

    return a, c


def _adapt_coord_pass(a, c, max_time, n_ternary=4):
    """Run coordinate descent passes until time runs out."""
    n = len(a)
    t_start = time.monotonic()
    pass_count = 0
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0

    while time.monotonic() - t_start < max_time:
        a, c = _fast_coord_pass(a, c, n_ternary)
        pass_count += 1
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 5:
                # Try a perturbation
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:.0f}s] BEST: {best_score:.10f} n={p.size} [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s")

    # --- Target lengths ---
    target_ns = [2048, 4096]

    # === PHASE 1:_initialize candidates at target lengths ===
    phase1_end = t_start + 60.0
    candidates = []

    # Interpolate parent to target lengths
    if parent is not None:
        for tn in target_ns:
            pi = _interp_profile(parent, tn)
            if pi is not None:
                sc = _score_norm(pi)
                candidates.append((sc, pi, tn, 'parent_interp'))
        # Also keep parent at its own length
        sc_p = _score_norm(parent)
        candidates.append((sc_p, parent.copy(), pn, 'parent'))

    # Structured initializations at target lengths
    for tn in target_ns:
        profs = [
            ('lin1.5', _make_profile_linear(tn, 1.5)),
            ('lin2.5', _make_profile_linear(tn, 2.5)),
            ('cos1', _make_profile_symmetric_cos(tn, 1, 0.6)),
            ('cos2', _make_profile_symmetric_cos(tn, 2, 0.4)),
            ('cos3', _make_profile_symmetric_cos(tn, 3, 0.3)),
            ('thick', _make_profile_thickened(tn, 0.8)),
            ('thick2', _make_profile_thickened(tn, 1.5)),
        ]
        for label, p in profs:
            sc = _score_norm(p)
            candidates.append((sc, p, tn, label))

    # Random initializations
    for tn in [2048]:
        for trial in range(3):
            p = np.abs(rng.normal(size=tn)) + 0.1
            p = p / p.sum()
            sc = _score_norm(p)
            candidates.append((sc, p, tn, f'rand{trial}'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # === PHASE 2: Adam optimization on top candidates ===
    phase2_end = t_start + 700.0
    n_to_polish = 5
    count = 0
    for sc, p, n, label in candidates:
        if time.monotonic() > phase2_end or count >= n_to_polish:
            break
        per_end = min(phase2_end, time.monotonic() + 120.0)
        if per_end - time.monotonic() < 15:
            break
        print(f"  Adam polishing n={n} [{label}] from {sc:.10f}")
        p_b, sc_b = _adam_logit(p, per_end, rng, lr=0.05, beta_frac=0.08,
                                beta_decay=0.93, no_imp_limit=200)
        _update(p_b, sc_b, f'adam_{label}_n{n}')
        count += 1

    print(f"Phase 2 done: best={best_score:.10f}")

    # === PHASE 3: Coordinate descent on best ===
    phase3_end = t_start + 1200.0
    if best_seq is not None:
        n_cd = best_seq.size
        cd_time = min(phase3_end - time.monotonic(), 400.0)
        if cd_time > 20:
            print(f"  Coord descent n={n_cd}, time={cd_time:.0f}s")
            a_cd = best_seq.copy()
            c_cd = _full_convolve(a_cd)
            a_cd, sc_cd = _adapt_coord_pass(a_cd, c_cd, cd_time)
            _update(a_cd, sc_cd, 'coord_descent')

            # Polish after CD
            if time.monotonic() < phase3_end - 30:
                per = min(60.0, phase3_end - time.monotonic() - 10)
                if per > 15:
                    p_ap, sc_ap = _adam_logit(a_cd, time.monotonic() + per, rng,
                                             lr=0.03, beta_frac=0.05,
                                             beta_decay=0.94, no_imp_limit=150)
                    _update(p_ap, sc_ap, 'post_CD_adam')

    # === PHASE 4: Try larger n escalation ===
    phase4_end = t_start + 1500.0
    if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase4_end - 60:
        tn_new = 4096
        p_int = _interp_profile(best_seq, tn_new)
        if p_int is not None:
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn_new}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn_new}')

            # Adam at new scale
            if time.monotonic() < phase4_end - 30:
                per = min(120.0, phase4_end - time.monotonic() - 20)
                if per > 15:
                    p_e, sc_e = _adam_logit(p_int, time.monotonic() + per, rng,
                                           lr=0.04, beta_frac=0.07,
                                           beta_decay=0.93, no_imp_limit=200)
                    _update(p_e, sc_e, f'adam_n{tn_new}')

            # CD at new scale
            if best_seq is not None and time.monotonic() < phase4_end - 10:
                cd_t = min(60.0, phase4_end - time.monotonic() - 5)
                if cd_t > 10:
                    a_cd2 = best_seq.copy()
                    c_cd2 = _full_convolve(a_cd2)
                    a_cd2, sc_cd2 = _adapt_coord_pass(a_cd2, c_cd2, cd_t)
                    _update(a_cd2, sc_cd2, f'CD_n{tn_new}')

    # === PHASE 5: LP refinement ===
    phase5_end = deadline - 60.0
    if best_seq is not None and best_seq.size <= 4096 and time.monotonic() < phase5_end - 30:
        lp_time = min(120.0, phase5_end - time.monotonic() - 20)
        if lp_time > 20:
            print(f"  LP refinement n={best_seq.size}")
            p_lp, sc_lp = _lp_refine(best_seq, time.monotonic() + lp_time, t=0.3, max_solves=10)
            _
