```python
"""
Improved LP + gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: tighter LP inflation, better line search,
subgradient polish, multi-length exploration, and parent-seed initialization.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 500:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s

def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
```

```python
"""
LP + subgradient + multiplicative optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Improvements: tighter LP inflation, parabolic line search, subgradient polish with
smooth-max, multiplicative weight updates, multi-length exploration, parent seeding.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 500:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _build_conv_matrix(a, n):
    """Build the constraint matrix A where A[k,j] = a[j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, 0)
        hi = min(2 * n - 2, 0) + j
        # A[k, j] = a[j] when 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            A[k_lo:k_hi + 1, j] = a
    return A


def _solve_lp(a, M, n):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    A = _build_conv_matrix(a, n)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'max_time': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _solve_lp_subset(a, M, n, active_indices):
    """Solve LP with only a subset of constraints (for speed)."""
    n_constraints = len(active_indices)
    A = np.zeros((n_constraints, n))
    for idx, k in enumerate(active_indices):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[idx, j] = a[k - j]
    c_obj = -np.ones(n)
    b = np.full(n_constraints, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'max_time': 10})
        if res.success:
            g = res.x
            # Verify against all constraints
            c_full = _fft_convolve(a)
            conv_check = _fft_convolve(a)  # a*g, not a*a
            # Actually need a*g
            N = _next_pow2(2 * n - 1)
            fa = np.fft.rfft(a, N)
            fg = np.fft.rfft(g, N)
            ag = np.fft.irfft(fa * fg, N)[:2 * n - 1]
            if np.max(np.abs(ag)) <= M * 1.001:
                return g
    except Exception:
        pass
    return None


def _parabolic_line_search(a, g, max_evals=8):
    """Line search along a + t*(g-a) using parabolic interpolation."""
    a = np.maximum(a, 0.0)
    s_a = a.sum()
    if s_a < 1e-10:
        return 0.0, _score(a)
    a = a / s_a

    best_t = 0.0
    best_sc = _score(a)

    # Initial grid
    ts = [0.1, 0.3, 0.5, 0.7, 1.0]
    scs = [best_sc]
    for t in ts:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            scs.append(np.inf)
            continue
        a_new = a_new / s
        sc = _score(a_new)
        scs.append(sc)
        if sc < best_sc:
            best_sc = sc
            best_t = t

    # Find the best 3 consecutive points and do parabolic fit
    all_ts = [0.0] + ts
    all_scs = scs
    # Find min in the grid
    idx = int(np.argmin(all_scs))
    if 0 < idx < len(all_scs) - 1:
        t0, t1, t2 = all_ts[idx - 1], all_ts[idx], all_ts[idx + 1]
        f0, f1, f2 = all_scs[idx - 1], all_scs[idx], all_scs[idx + 1]
        denom = (f0 - 2 * f1 + f2)
        if abs(denom) > 1e-15:
            t_par = t1 - 0.5 * (f0 - f2) * (t1 - t0) / (f0 - 2 * f1 + f2)
            if t_par > 0 and t_par < 1:
                a_new = (1.0 - t_par) * a + t_par * g
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s > 1e-10:
                    a_new = a_new / s
                    sc_par = _score(a_new)
                    if sc_par < best_sc:
                        best_sc = sc_par
                        best_t = t_par

    return best_t, best_sc


def _lp_refine(a0, deadline, max_iter=50000):
    """LP-based refinement with tight inflation and adaptive stepping."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = 1.0003
    max_stall = max(100, 3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 50 == 0:
                inflate = min(0.05, inflate * 1.2)
            if no_imp > max_stall:
                a = best_a.copy()
                inflate = 1.0003
                no_imp = 0
                if it > 1000:
                    break
            continue

        g = g / sum_g

        best_t, best_sc = _parabolic_line_search(a, g)

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            # Slowly decrease inflation as we improve
            if no_imp == 0:
                inflate = max(1.0001, inflate * 0.995)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.05, inflate * 1.1)
            if no_imp > max_stall:
                a = best_a.copy()
                inflate = 1.0003
                no_imp = 0
                if it > 2000:
                    break

    return best_a


def _subgradient_polish(a0, deadline, max_iter=50000):
    """Subgradient method with smooth-max for the peak."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = 0.02 * float(c.max())
    base_lr = 0.05
    lr = base_lr
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        mval = float(c.max())

        # Smooth max weights
        x = (c - mval) / beta
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if not np.isfinite(s_ex) or s_ex <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # Gradient: sum_k w[k] * 2 * a shifted
        # grad[j] = 2 * sum_k w[k] * a[k-j] (where valid)
        # This is 2 * (w * a_rev) cross-correlation
        N_pad = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N_pad)
        fa_rev = np.fft.rfft(a[::-1], N_pad)
        corr = np.fft.irfft(fw * fa_rev, N_pad)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Adjust for sum constraint (project gradient)
        grad = grad - grad.mean()

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        sc = _score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.002, lr * 0.9)
            if no_imp > max(2000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = 0.02 * float(_fft_convolve(a).max())
                lr = base_lr
                no_imp = 0

        if it % 50 == 49:
            beta = max(1e-8, beta * 0.85)

    return best_a


def _multiplicative_update(a, max_iter=10000, deadline=None):
    """Multiplicative weight updates to flatten the convolution peak."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    lr = 0.1
    no_imp = 0

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))
        M = float(c.max())

        # Contribution of each j to the peak: a[j] * a[k_star - j]
        # We want to reduce high-contribution indices
        contrib = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contrib[j] = a[j] * a[k_j]
            else:
                contrib[j] = 0.0

        avg_contrib = M / n
        # Multiplicative update: increase where contribution is low, decrease where high
        ratio = avg_contrib / (contrib + 1e-15)
        ratio = np.clip(ratio, 0.1, 10.0)
        a = a * ratio ** lr
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        sc = _score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.01, lr * 0.8)
            if no_imp > 2000:
                break

    return best_a


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_symmetric_ramp(n):
    if n % 2 == 0:
        half = n // 2
        t = np.linspace(0.5, 1.0, half)
        a = np.concatenate([t, t[::-1]])
    else:
        half = n // 2
        t = np.linspace(0.5, 1.0, half)
        a = np.concatenate([t, [1.0], t[::-1]])
    a = a + 1e-8
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma_frac) ** 2) + 1e-8
    return a / a.sum()

        candidates.append((f'ramp1_n{n_try}', _make_ramp(n_try, 1.0)))
        candidates.append((f'sym_ramp_n{n_try}', _make_symmetric_ramp(n_try)))
        candidates.append((f'gauss_n{n_try}', _make_gaussian(n_try, 0.3)))
        candidates.append((f'gauss_w_n{n_try}', _make_gaussian(n_try, 0.4)))

    # Score and sort candidates
    scored = []
    for name, a0 in candidates:
        sc0 = _score(a0)
        if np.isfinite(sc0):
            scored.append((sc0, name, a0))
    scored.sort(key=lambda x: x[0])

    print(f"\nTop 10 initializations:")
    for sc0, name, _ in scored[:10]:
        print(f"  {name}: {sc0:.8f}")

    # Phase 1: Refine top candidates
    # Give more time to better initializations
    for i, (sc0, name, a0) in enumerate(scored):
        if time.monotonic() > deadline - 120:
            break
        if sc0 >= best_score + 0.05:
            continue  # Not promising enough

        remaining = deadline - time.monotonic() - 60
        if remaining < 30:
            break

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Allocate time based on promise
        if sc0 < best_score + 0.02:
            alloc = min(200.0, remaining * 0.3)
        elif sc0 < best_score + 0.05:
            alloc = min(100.0, remaining * 0.2)
        else:
            alloc = min(50.0, remaining * 0.1)

        if alloc < 15:
            break

        alloc_end = time.monotonic() + alloc
        print(f"\nRefining {name} (init={sc0:.8f}, alloc={alloc:.0f}s):")
        a_ref = _combined_refine(a0, alloc_end)
        sc_ref = _score(a_ref)
        if np.isfinite(sc_ref) and sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"  NEW BEST: {sc_ref:.10f} from {name}")
        else:
            print(f"  No improvement: {sc_ref:.8f}")

        # If we've already beaten the target significantly, focus on the best
        if best_score < 1.5030:
            break

    # Phase 2: Focused refinement on the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 15
        if remaining > 30:
            print(f"\n=== Focused refinement on best (score={best_score:.10f}) ===")
            focus_end = time.monotonic() + remaining
            a_focus = _combined_refine(best_seq, focus_end)
            sc_focus = _score(a_focus)
            if np.isfinite(sc_focus) and sc_focus < best_score:
                best_score = sc_focus
                best_seq = a_focus.copy()
                print(f"  Focused best: {sc_focus:.10f}")

    # Phase 3: Try a fresh aggressive LP from best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            a_fresh = _lp_refine(best_seq, time.monotonic() + remaining, max_iter=50000)
            sc_fresh = _score(a_fresh)
            if np.isfinite(sc_fresh) and sc_fresh < best_score:
                best_score = sc_fresh
                best_seq = a_fresh.copy()
                print(f"  Fresh LP: {sc_fresh:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(256, 1.0)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")

    # Verify with evaluate_sequence if available
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval check: {e}")

    return [float(x) for x in final_seq]
```
