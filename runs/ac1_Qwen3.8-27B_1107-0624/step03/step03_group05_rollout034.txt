def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

        # Random operation: swap, scale, or shift blocks

            a = a_new.copy()
            no_imp = 0

                break

Multi-strategy optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses active-set LP with adaptive constraint growth, mirror descent,
smooth-max gradient with adaptive beta, and multi-length search.
Key improvements: faster active-set LP allowing more iterations,

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _build_active_set_matrix(a, active_lags):
    """Build constraint matrix for given active lags only."""
    n = len(a)
    k = len(active_lags)
    A = np.zeros((k, n))
    for i, lag in enumerate(active_lags):
        for jj in range(n):
            kk = lag - jj
            if 0 <= kk < n:
                A[i, jj] = a[kk]
    return A


def _active_set_lp(a, M, max_constraints=50):
    """LP with adaptive active set: start with top lags, grow if needed."""
    n = len(a)
    c = _fft_convolve(a)
    
    # Start with top lags
    sorted_lags = np.argsort(c)[::-1]
    active_lags = sorted_lags[:max_constraints].tolist()
    
    for attempt in range(5):
        A = _build_active_set_matrix(a, active_lags)
        c_obj = -np.ones(n)
        b_ub = np.full(len(active_lags), M)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            # Try full set
            A_full = _build_active_set_matrix(a, sorted_lags.tolist())
            res = linprog(c_obj, A_ub=A_full, b_ub=np.full(2*n-1, M),
                          bounds=[(0, None)] * n, method='highs')
            if not res.success:
                return None
            return res.x
        
        x = res.x
        # Check if solution violates non-active constraints
        c_check = _fft_convolve(x)
        violations = np.where(c_check > M + 1e-10)[0]
        if len(violations) == 0:
            return x
        # Add violated constraints to active set
        for v in violations[:10]:
            if v not in active_lags:
                active_lags.append(v)
        # Keep only top by importance
        active_lags = sorted(active_lags, key=lambda l: -c_check[l])[:max_constraints]
    
    # Final try with all constraints
    A_full = _build_active_set_matrix(a, np.arange(2*n-1).tolist())
    res = linprog(c_obj, A_ub=A_full, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

    A = _build_active_set_matrix(a, np.arange(2*n-1).tolist())
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs')

        return res.x
    return None


def _lp_refine_v3(a, deadline, max_iter=30000, use_active_set=True):
    """LP refinement with active set and adaptive inflation."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    inflate = 1.0003
    blend = 0.2
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        if use_active_set:
            g = _active_set_lp(a, M_lp, max_constraints=min(80, 2*n-1))
        else:
            g = _full_lp(a, M_lp)

        if g is None:
            inflate = min(1.01, inflate * 1.3)
            if inflate > 1.02:
                break
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-9:
            no_imp += 1
            if no_imp % 50 == 0:
                inflate = min(1.015, inflate * 1.02)
            if no_imp > 500:
                break
            continue

        no_imp = 0
        g = g / sum_g

        # Adaptive blend
        improvement = sum_g - 1.0
        blend_cur = min(0.4, max(0.05, 0.1 + 0.3 * min(1.0, improvement * 10)))

        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            inflate = max(1.0002, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                inflate = min(1.01, inflate * 1.01)
            if no_imp > 300:
                a = best_a.copy()
                inflate = 1.0005
                no_imp = 0

        a = a_new

    return best_a


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _gradient_smooth_max(a, beta, top_k_boost=0):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    
    if top_k_boost > 0:
        sorted_idx = np.argsort(c)[::-1]
        top = sorted_idx[:top_k_boost]
        w[top] *= 3.0
        w = w / w.sum()
    
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c


def _gradient_polish_v3(a, deadline, max_iter=30000, mode='smooth'):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.015 * float(c.max()))
    lr = 0.04
    momentum = 0.9
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr

    top_k = 0 if mode == 'smooth' else 3

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        grad, maxc, c_arr = _gradient_smooth_max(a, beta, top_k_boost=top_k)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue

        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr_eff = lr * min(1.0, 0.05 / grad_norm)
        else:
            lr_eff = lr

        v = momentum * v - lr_eff * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.002, lr * 0.94)
            if no_imp > max(1500, 2 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-8, 0.015 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 300 == 0:
            beta = max(1e-10, beta * 0.8)

    return best_a


def _mirror_descent_v2(a, deadline, max_iter=20000, lr_init=0.08):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, 0.02 * float(c.max()))
    lr = lr_init
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Mirror descent: multiplicative update
        log_scale = -lr * grad
        log_scale -= np.max(log_scale)
        a_new = a * np.exp(log_scale)
        s = a_new.sum()
        if s < 1e-10 or not np.isfinite(s):
            a_new = a.copy()
        else:
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.003, lr * 0.93)

        if it % 150 == 0:
            beta = max(1e-10, beta * 0.82)

        if no_imp > max(2000, 3 * n):
            a = best_a.copy()
            beta = max(1e-8, 0.02 * float(_fft_convolve(a).max()))
            lr = lr_init
            no_imp = 0

    return best_a


def _subgradient_method(a, deadline, max_iter=50000, avg_window=1000):
    """Subgradient method with iterate averaging."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _score(a)

    lr_init = 0.1
    lr = lr_init
    t = 0
    no_imp = 0
    
    # Averaging
    avg_a = np.zeros(n)
    avg_count = 0
    avg_best_a = a.copy()
    avg_best_score = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        c = _fft_convolve(a)
        max_idx = int(np.argmax(c))
        M = c[max_idx]

        # Subgradient of (a*a)_{max_idx}
        # d/d a_j of sum_k a_k * a_{max_idx - k} = 2 * a_{max_idx - j} (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            k = max_idx - j
            if 0 <= k < n:
                grad[j] = 2.0 * a[k]

        # Subgradient descent with projection
        alpha = lr_init / np.sqrt(t + 1)
        a_new = a - alpha * grad
        a_new = _project_simplex(a_new)
        a = a_new

        # Track average
        avg_a += a
        avg_count += 1
        if avg_count >= avg_window:
            avg_candidate = avg_a / avg_count
            avg_score = _score(avg_candidate)
            if avg_score < avg_best_score:
                avg_best_score = avg_score
                avg_best_a = avg_candidate.copy()
            
            if avg_score < best_score - 1e-12:
                best_score = avg_score
                best_a = avg_candidate.copy()
            
            # Reset averaging window
            avg_a = np.zeros(n)
            avg_count = 0

        score = _score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 5000:
            break

    # Final check of average
    if avg_count > 0:
        avg_candidate = avg_a / avg_count
        avg_score = _score(avg_candidate)
        if avg_score < best_score:
            best_score = avg_score
            best_a = avg_candidate.copy()

    return best_a


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, width=0.1, boost=4.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + boost * np.exp(-((t) / width) ** 2) + boost * np.exp(-((t - 1.0) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, freq=1, depth=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _sinusoidal_perturb(a, deadline, max_iter=5000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(42)
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    t = np.linspace(0, 1, n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        freq = rng.integers(1, max(2, n // 16))
        amp = rng.uniform(0.02, 0.25)
        phase = rng.uniform(0, 2 * np.pi)
        pert = amp * np.sin(2 * np.pi * freq * t + phase)
        a_new = a * (1.0 + pert)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break
    return best_a


def _block_perturb(a, deadline, max_iter=5000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(99)
    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        op = rng.integers(0, 3)
        block_size = rng.integers(max(2, n // 64), max(3, n // 16))
        a_new = a.copy()

        if op == 0:
            i = rng.integers(0, max(1, n - block_size))
            j = rng.integers(0, max(1, n - block_size))
            if abs(i - j) < block_size:
                continue
            a_new[i:i+block_size], a_new[j:j+block_size] = a_new[j:j+block_size].copy(), a_new[i:i+block_size].copy()
        elif op == 1:
            i = rng.integers(0, max(1, n - block_size))
            factor = rng.uniform(0.5, 2.0)
            a_new[i:i+block_size] *= factor
        else:
            i = rng.integers(0, max(1, n - block_size))
            j = rng.integers(0, max(1, n - block_size))
            if abs(i - j) < block_size:
                continue
            delta = 0.1 * a_new[i:i+block_size].sum()
            a_new[i:i+block_size] -= delta / block_size
            a_new[j:j+block_size] += delta / block_size

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 3000:
                break
    return best_a


def _full_pipeline(a_init, n, deadline, phase_budget, use_active=True):
    a = _interpolate_profile(a_init, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_u_profile(n, 2.0)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)

    # Phase A: LP (primary)
    lp_budget = min(phase_budget * 0.35, deadline - time.monotonic() - 60.0)
    if lp_budget > 5:
        a_lp = _lp_refine_v3(a, time.monotonic() + lp_budget, max_iter=30000, use_active_set=use_active)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
        print(f"    LP: {sc_lp:.10f}")

    # Phase B: Gradient
    gp_budget = min(phase_budget * 0.2, deadline - time.monotonic() - 40.0)
    if gp_budget > 5:
        a_gp = _gradient_polish_v3(best_a, time.monotonic() + gp_budget, max_iter=20000)
        sc_gp = _score(a_gp)
        if sc_gp < best_score:
            best_score = sc_gp
            best_a = a_gp.copy()
        print(f"    Gradient: {sc_gp:.10f}")

    # Phase C: Mirror descent
    md_budget = min(phase_budget * 0.15, deadline - time.monotonic() - 30.0)
    if md_budget > 5:
        a_md = _mirror_descent_v2(best_a, time.monotonic() + md_budget, max_iter=15000)
        sc_md = _score(a_md)
        if sc_md < best_score:
            best_score = sc_md
            best_a = a_md.copy()
        print(f"    Mirror: {sc_md:.10f}")

    # Phase D: Perturbations
    pt_budget = min(phase_budget * 0.1, deadline - time.monotonic() - 20.0)
    if pt_budget > 3:
        a_sin = _sinusoidal_perturb(best_a, time.monotonic() + pt_budget * 0.5, max_iter=2000)
        a_blk = _block_perturb(best_a, time.monotonic() + pt_budget * 0.5, max_iter=2000)
        sc = _score(a_sin)
        if sc < best_score:
            best_score = sc
            best_a = a_sin.copy()
        sc = _score(a_blk)
        if sc < best_score:
            best_score = sc
            best_a = a_blk.copy()
        print(f"    Perturb: {best_score:.10f}")

    # Phase E: Second LP
    lp2_budget = min(phase_budget * 0.15, deadline - time.monotonic() - 10.0)
    if lp2_budget > 5:
        a_lp2 = _lp_refine_v3(best_a, time.monotonic() + lp2_budget, max_iter=15000, use_active_set=use_active)
        sc_lp2 = _score(a_lp2)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_a = a_lp2.copy()
        print(f"    LP2: {sc_lp2:.10f}")

    # Phase F: Final gradient with top-k boost
    gp2_budget = min(phase_budget * 0.1, deadline - time.monotonic() - 5.0)
    if gp2_budget > 3:
        a_gp2 = _gradient_polish_v3(best_a, time.monotonic() + gp2_budget, max_iter=10000, mode='top_k')
        sc_gp2 = _score(a_gp2)
        if sc_gp2 < best_score:
            best_score = sc_gp2
            best_a = a_gp2.copy()
        print(f"    Gradient2: {sc_gp2:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining active-set LP, mirror descent,
    smooth-max gradient with adaptive beta, subgradient method, and
    perturbations to minimize 2n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine target lengths
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 640, 768, 1024, 1280]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))

    print(f"Trying lengths: {lengths}")
    print(f"Time budget: {budget_s}s")

    # Phase 1: For each length, run pipeline
    # Give priority to parent length and nearby lengths
    available_time = deadline - time.monotonic() - 250.0
    num_lengths = len(lengths)
    
    # Weight: more time for parent length, less for extreme lengths
    weights = np.ones(num_lengths)
    if parent is not None:
        weights[0] = 2.0
    for i, ln in enumerate(lengths):
        if ln > 1024:
            weights[i] *= 0.5
        elif ln < 512:
            weights[i] *= 0.7
    
    total_w = weights.sum()
    
    for idx, n_t in enumerate(lengths):
        if time.monotonic() > deadline - 200:
            break
        phase_budget = available_time * weights[idx] / total_w
        phase_budget = max(25.0, min(phase_budget, deadline - time.monotonic() - 150.0))
        if phase_budget < 15:
            continue

        print(f"\nLength {n_t}: budget={phase_budget:.0f}s")

        if parent is not None:
            a_init = _interpolate_profile(parent, n_t)
        else:
            a_init = _make_u_profile(n_t, 2.0)

        sc_init = _score(a_init)
        if np.isfinite(sc_init) and sc_init < best_score:
            best_score = sc_init
            best_seq = a_init.copy()

        a_best_n, sc_best_n = _full_pipeline(a_init, n_t, deadline, phase_budget, use_active=True)
        if sc_best_n < best_score:
            best_score = sc_best_n
            best_seq = a_best_n.copy()
            print(f"  ** New best at n={n_t}: {sc_best_n:.10f}")

    # Phase 2: Diverse initializations at best length
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_best = best_seq.size
        print(f"\nPhase 2: Diverse inits at n={n_best}")

        inits = []
        for alpha in [1.5, 2.5, 4.0, 6.0, 8.0]:
            inits.append(_make_u_profile(n_best, alpha))
        for eta in [1.0, 2.0, 3.5, 5.0]:
            inits.append(_make_linear_profile(n_best, eta))
        for w in [0.05, 0.1, 0.2]:
            for b in [3.0, 5.0, 8.0]:
                inits.append(_make_bimodal_profile(n_best, w, b))
        for f in [1, 2, 3]:
            for d in [0.2, 0.4, 0.6]:
                inits.append(_make_cosine_profile(n_best, f, d))

        # Perturbed best
        for _ in range(4):
            a_pert = best_seq * (1.0 + 0.15 * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            inits.append(a_pert)

        rem = deadline - time.monotonic() - 80.0
        per_init = max(12.0, rem / max(1, len(inits)))

        for a0 in inits:
            if time.monotonic() > deadline - 70:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            rem = deadline - time.monotonic() - 50.0
            if rem < 15:
                break
            sub_budget = min(per_init, rem * 0.5)
            if sub_budget < 10:
                continue
            a_res, sc_res = _full_pipeline(a0, n_best, deadline, sub_budget, use_active=True)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best: {sc_res:.10f}")

    # Phase 3: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        final_budget = deadline - time.monotonic() - 20.0
        print(f"\nPhase 3: Final polish, budget={final_budget:.0f}s")

        # Intensive LP
        lp_b = min(final_budget * 0.3, 180.0)
        if lp_b > 10:
            a_lp = _lp_refine_v3(best_seq, time.monotonic() + lp_b, max_iter=50000, use_active_set=True)
            sc = _score(a_lp)
            if sc < best_score:
                best_score = sc
                best_seq = a_lp.copy()
            print(f"  Final LP: {sc:.10f}")

        # Gradient
        gp_b = min(final_budget * 0.2, 120.0)
        if gp_b > 10:
            a_gp = _gradient_polish_v3(best_seq, time.monotonic() + gp_b, max_iter=30000)
            sc = _score(a_gp)
            if sc < best_score:
                best_score = sc
                best_seq = a_gp.copy()
            print(f"  Final Gradient: {sc:.10f}")

        # Mirror descent
        md_b = min(final_budget * 0.2, 100.0)
        if md_b > 10:
            a_md = _mirror_descent_v2(best_seq, time.monotonic() + md_b, max_iter=20000)
            sc = _score(a_md)
            if sc < best_score:
                best_score = sc
                best_seq = a_md.copy()
            print(f"  Final Mirror: {sc:.10f}")

        # Subgradient with averaging
        sg_b = min(final_budget * 0.1, 60.0)
        if sg_b > 10:
            a_sg = _subgradient_method(best_seq, time.monotonic() + sg_b, max_iter=30000)
            sc = _score(a_sg)
            if sc < best_score:
                best_score = sc
                best_seq = a_sg.copy()
            print(f"  Final SubGrad: {sc:.10f}")

        # Perturb + quick LP
        pt_b = min(final_budget * 0.1, 40.0)
        if pt_b > 5:
            a_pt = _sinusoidal_perturb(best_seq, time.monotonic() + pt_b * 0.4, max_iter=3000)
            a_bl = _block_perturb(best_seq, time.monotonic() + pt_b * 0.4, max_iter=3000)
            sc_pt = _score(a_pt)
            sc_bl = _score(a_bl)
            if sc_pt < best_score:
                best_score = sc_pt
                best_seq = a_pt.copy()
            if sc_bl < best_score:
                best_score = sc_bl
                best_seq = a_bl.copy()
            
            lp3_b = min(final_budget * 0.05, 20.0)
            if lp3_b > 5:
                a_lp3 = _lp_refine_v3(best_seq, time.monotonic() + lp3_b, max_iter=5000, use_active_set=True)
                sc = _score(a_lp3)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_lp3.copy()
            print(f"  After Perturb: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_u_profile(768, 2.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_u_profile(768, 2.0)

    # Evaluate with authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
