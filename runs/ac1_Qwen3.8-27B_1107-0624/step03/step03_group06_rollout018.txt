```python
def _solve_lp_minimize_conv(a, n):
    """Minimize t s.t. conv(a,b)_j <= t, sum(b) = 1, b >= 0"""
    # Variables: b_0, ..., b_{n-1}, t (total n+1 variables)
    # Constraints: 
    #   conv(a,b)_j <= t for j=0,...,2n-2 (2n-1 constraints)
    #   sum(b) = 1 (equality)
    #   b_i >= 0
    
    # In linprog form: minimize c^T x s.t. A_ub @ x <= b_ub, A_eq @ x = b_eq, bounds
    
    # x = [b_0, ..., b_{n-1}, t]
    # c = [0, ..., 0, 1] (minimize t)
    
    # conv(a,b)_j = sum_i a_{j-i} * b_i (with valid indices)
    # This is: sum_i A[j,i] * b_i - t <= 0
    # So A_ub[j, 0:n] = A[j,:], A_ub[j, n] = -1
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, n] = -1.0  # -t coefficient
    
    for j in range(2*n-1):
        for i in range(n):
            idx = j - i
            if 0 <= idx < n:
                A_ub[j, i] = a[idx]
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, 0:n] = 1.0  # sum(b) = 1
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # t is free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(2*n-1),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
```

The key question is: does the gradient method get stuck in local minima? For this problem, the objective is not convex (it's max of linear functions over a simplex, which is convex in `a` if we consider the normalized version). Wait, actually:

`f(a) = max_j conv(a,a)_j` where `conv(a,a)_j` is quadratic in `a`. So `f` is the max of quadratic functions, which is convex. And we're minimizing a convex function over a convex set (simplex ∩ non-negative orthant). So the problem is convex!

Wait, is `conv(a,a)_j` convex in `a`? `conv(a,a)_j = sum_i a_{j-i} * a_i` (with valid indices). This is a quadratic form, specifically `a^T T_j a` where `T_j` is a symmetric matrix. For the max to be convex, each `T_j` needs to be positive semi-definite.

Is `T_j` PSD? `T_j[i,k] = 1/2 * (delta_{i=j-k} + delta_{k=j-i})` (for valid indices). This is a permutation matrix (up to symmetry), which has eigenvalues ±1. So it's NOT PSD in general!

So `conv(a,a)_j` is NOT convex in `a` (it's an indefinite quadratic form). The max of indefinite quadratic forms is also not convex. So the problem is non-convex!

This explains why gradient methods might get stuck. The LP approach is better because it only requires linearity in the variable being optimized (we fix one argument of the convolution).

```python
"""
Multi-scale LP best-response optimization with gradient polish for minimizing
2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses LP-based best response, adaptive line search, multi-scale refinement,
and coordinate descent local search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_exp(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-alpha * (1.0 - t)) + 1e-8
    return a / a.sum()


def _make_cos(n):
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * t / 2.0) + 1e-8
    return a / a.sum()


def _make_lin(n, slope=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_poly(n, p=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + t ** (p + 1) + 1e-8
    return a / a.sum()


def _build_conv_matrix(a, n):
    """Build (2n-1) x n matrix where M[j,i] = a[j-i] if 0 <= j-i < n else 0."""
    A = np.zeros((2 * n - 1, n))
    col_idx = np.repeat(np.arange(n), n)
    row_idx = (np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)).ravel()
    vals = np.tile(a, n)
    A[row_idx, col_idx] = vals
    return A


def _lp_best_response(a, n):
    """
    Solve: minimize t s.t. conv(a,b)_j <= t, sum(b)=1, b>=0, t free.
    Returns (b, t) or (None, None).
    """
    A_conv = _build_conv_matrix(a, n)
    # A_ub: (2n-1) x (n+1), last column is -1 for t
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={"presolve": True, "dual_feasibility_tolerance": 1e-9,
                           "primal_feasibility_tolerance": 1e-9})
    if res.success:
        b = res.x[:n]
        t_val = res.x[n]
        b = np.maximum(b, 0)
        s = b.sum()
        if s > 1e-15:
            b = b / s
        return b, t_val
    return None, None


def _line_search_conv(a, b, n_alphas=6):
    """Find alpha in [0,1] minimizing score of (1-alpha)*a + alpha*b."""
    best_alpha = 0.0
    best_sc = _score(a)
    alphas = np.array([0.1, 0.25, 0.4, 0.5, 0.7, 1.0])
    for alpha in alphas:
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_alpha = alpha
    return best_alpha, best_sc

def _smooth_max_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a) using FFT."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    
    # grad_i = sum_j w_j * dc_j/da_i = sum_j w_j * a_{j-i} (valid indices)
    # This is conv(w, a_reversed)[n-1:2n-1]
    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=50000):
    """Gradient descent with smooth max and momentum."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    beta = max(1e-6, 0.03 * float(_fft_convolve(a).max()))
    lr = 0.05
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a_mom = a + momentum * v
        a_mom = np.maximum(a_mom, 0)
        s_mom = a_mom.sum()
        if s_mom < 1e-10:
            a_mom = a
        else:
            a_mom = a_mom / s_mom
        
        grad, maxc = _smooth_max_grad(a_mom, beta)
        
        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new
        
        if it % 5 == 4:
            c = _fft_convolve(a)
            score = float(2.0 * n * c.max())
            
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 100 == 0:
                    lr = max(0.002, lr * 0.95)
                if no_imp > max(2000, 3 * n):
                    a = best_a.copy()
                    v = np.zeros(n)
                    beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
                    lr = base_lr
                    no_imp = 0
            
            if it % 200 == 199:
                beta = max(1e-10, beta * 0.85)
    
    return best_a

                    best_score = sc
                    best_a = a.copy()
                    no_imp = 0
                    break
            else:
                no_imp += 1
            continue
        
        # Try decreasing and increasing
        for frac in [0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0]:
            new_val = old_val * frac
            a_try = a.copy()
            a_try[i] = new_val
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_score - 1e-14:
                a = a_try
                best_score = sc
                best_a = a.copy()
                no_imp = 0
                break
        
        if no_imp > max(5000, 5 * n):
            break
    
    return best_a


def _pairwise_swap(a0, deadline, max_rounds=20):
    """Try swapping adjacent pairs of elements."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n - 1):
            if time.monotonic() > deadline:
                break
            # Try swapping a[i] and a[i+1]
            a_try = a.copy()
            a_try[i], a_try[i + 1] = a_try[i + 1], a_try[i]
            sc = _score(a_try)
            if sc < best_score - 1e-12:
                a = a_try
                best_score = sc
                best_a = a.copy()
                improved = True
        if not improved:
            break
    
    return best_a
```
