"""
Hybrid Fienup-Adam-LP optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Frequency-domain iterative spectrum flattening (Fienup/Gerchberg-Saxton
style) combined with gradient-based Adam in logit space and LP refinement.
The Fienup step alternates between flattening the power spectrum and clipping to
the non-negative cone, driving the autocorrelation peak below what pure gradient
methods achieve. Multi-scale escalation and diverse initialization ensure robustness.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return None
    return a / s


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs',
                  options={'time_limit': 60, 'mip_rel_gap': 0.01})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine(a, deadline, t=0.3, max_solves=20):
    n = len(a)
    current_best = a.copy()
    current_score = _score(a)
    c = _full_convolve(a)
    M = float(c.max())
    n_solves = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = _score(a_new)
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.1, t * 0.9)
    return current_best, current_score


def _adam_logit(p0, deadline, rng, lr=0.06, beta_frac=0.1, beta_decay=0.92,
                no_imp_limit=400, noise_scale=None):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.003, lr * 0.92)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 50 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-4, beta * 0.5)

        limit = max(100, no_imp_limit)
        if no_imp > limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.08
                elif best_score < 1.53:
                    ns = 0.15
                else:
                    ns = 0.3
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _fienup_iterate(p0, deadline, rng, gamma_start=0.85, gamma_end=0.5,
                    n_iters=500, blend=0.7, every_k=1):
    """
    Fienup-style iteration: alternate between spectrum flattening and time-domain clipping.
    gamma controls compression: |A| -> |A|^gamma (gamma < 1 compresses dynamic range).
    blend controls how much of the flattened spectrum we use vs original.
    """
    p = _normalize_nonneg(p0)
    if p is None:
        return p0, _score(p0)
    n = p.size
    L = _next_pow2(2 * n - 1)

    best_p = p.copy()
    best_score = _score(p)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Progress gamma from gamma_start to gamma_end
        frac = min(1.0, (it + 1) / n_iters)
        gamma = gamma_start + (gamma_end - gamma_start) * frac

        # Compute spectrum
        A = np.fft.fft(p, L)
        mag = np.abs(A)
        ph = np.angle(A)

        # Flatten: power compression
        # Protect DC component (sum)
        mag_flat = mag ** gamma
        # Blend with original to avoid too drastic changes
        mag_new = blend * mag_flat + (1.0 - blend) * mag
        # Preserve DC: renormalize so sum is maintained
        if mag_new[0] > 0:
            dc_ratio = mag[0] / mag_new[0]
            mag_new = mag_new * dc_ratio

        A_new = mag_new * np.exp(1j * ph)
        p_new = np.fft.ifft(A_new).real[:n]

        # Clip to non-negative
        p_new = np.maximum(p_new, 0.0)
        s = float(p_new.sum())
        if s < 1e-15:
            break
        p_new = p_new / s

        sc = _score(p_new)
        if sc < best_score:
            best_score = sc
            best_p = p_new.copy()

        p = p_new

    return best_p, best_score


def _fienup_soft(p0, deadline, rng, alpha=2.0, n_iters=500, blend=0.7):
    """
    Alternative Fienup with soft thresholding: |A| -> |A| / (1 + alpha*|A|)
    This compresses large spectral components more aggressively.
    """
    p = _normalize_nonneg(p0)
    if p is None:
        return p0, _score(p0)
    n = p.size
    L = _next_pow2(2 * n - 1)

    best_p = p.copy()
    best_score = _score(p)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.fft(p, L)
        mag = np.abs(A)
        ph = np.angle(A)

        # Soft compression
        mag_new = mag / (1.0 + alpha * mag / mag.max())
        # Blend
        mag_new = blend * mag_new + (1.0 - blend) * mag
        # Preserve DC
        if mag_new[0] > 0:
            dc_ratio = mag[0] / mag_new[0]
            mag_new = mag_new * dc_ratio

        A_new = mag_new * np.exp(1j * ph)
        p_new = np.fft.ifft(A_new).real[:n]
        p_new = np.maximum(p_new, 0.0)
        s = float(p_new.sum())
        if s < 1e-15:
            break
        p_new = p_new / s

        sc = _score(p_new)
        if sc < best_score:
            best_score = sc
            best_p = p_new.copy()

        p = p_new

    return best_p, best_score


def _fienup_reverse_wf(p0, deadline, rng, n_iters=500):
    """
    Reverse water-filling: raise low spectral components, lower high ones.
    Target: make |A| as flat as possible while maintaining non-negativity.
    """
    p = _normalize_nonneg(p0)
    if p is None:
        return p0, _score(p0)

    L = _next_pow2(2 * n - 1)

        A = np.fft.fft(p, L)
        mag = np.abs(A)
        ph = np.angle(A)

        # Target: geometric mean of magnitudes (flat spectrum at GM level)
        log_mag = np.log(np.maximum(mag, 1e-30))
        target = np.exp(np.mean(log_mag))

        # Blend magnitudes toward target

        mag_new = mag * (target / np.maximum(mag, 1e-30)) ** t_param
        # Ensure DC is preserved

            mag_new = mag_new * dc_ratio

        p_new = np.fft.ifft(A_new).real[:n]
        p_new = np.maximum(p_new, 0.0)
        s = float(p_new.sum())
        if s < 1e-15:

        sc = _score(p_new)

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))

        ('exp0.5', lambda n: _make_profile_exponential(n, 0.5)),
        ('exp1.0', lambda n: _make_profile_exponential(n, 1.0)),
        ('exp2.0', lambda n: _make_profile_exponential(n, 2.0)),
        ('geometric0.7', lambda n: _make_profile_geometric(n, 0.7)),
        ('geometric0.5', lambda n: _make_profile_geometric(n, 0.5)),
        ('twobump', lambda n: _make_profile_two_bump(n, 0.3, 0.7, 0.15)),
        ('twobump2', lambda n: _make_profile_two_bump(n, 0.25, 0.75, 0.2)),
        ('cos1', lambda n: _make_profile_cosine(n, 1, 0.8)),
        ('cos2', lambda n: _make_profile_cosine(n, 2, 0.6)),
        ('cos4', lambda n: _make_profile_cosine(n, 4, 0.5)),
        ('bell_edge', lambda n: _make_profile_bell_edge(n, 0.2)),
    ]

    for n in sizes:
        for label, fn in profiles_factory:
            p = fn(n)
            sc = _score(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in [256, 512, 1024, 2048, 4096]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score(pi)
                    candidates.append((sc, pi, n, 'parent_interp'))
        sc = _score(parent)
        candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # ===== Phase 2: Fienup refinement (new! 300s) =====
    phase2_end = t_start + 420.0
    print(f"Phase 2: Fienup refinement...")

    top_fienup = []
    count = 0
    for sc, p, n, label in candidates[:15]:
        if time.monotonic() > phase2_end or count >= 8:
            break
        per = min(phase2_end, time.monotonic() + 50.0)
        p_f, sc_f = _fienup_hybrid(p, per, rng, method='all')
        _update(p_f, sc_f, f'fienup_{label}_n{n}')
        top_fienup.append((sc_f, p_f, n, label + '_fin'))
        count += 1

    top_fienup.sort(key=lambda x: x[0])
    print(f"Phase 2: best={best_score:.10f}")

    # ===== Phase 3: Adam refinement on best (400s) =====
    phase3_end = t_start + 820.0
    print(f"Phase 3: Adam refinement...")

    if best_seq is not None:
        # Adam from best Fienup result
        per = min(200.0, phase3_end - time.monotonic() - 20)
        if per > 20:
            p_a, sc_a = _adam_logit(best_seq, time.monotonic() + per, rng,
                                     lr=0.06, beta_frac=0.1, beta_decay=0.92,
                                     no_imp_limit=300)
            _update(p_a, sc_a, 'adam_after_fienup')

        # Also try Adam from a few other initializations
        for sc_c, p_c, n_c, lab_c in candidates[:5]:
            if time.monotonic() > phase3_end - 60:
                break
            per = min(80.0, phase3_end - time.monotonic() - 10)
            if per < 20:
                break
            p_a2, sc_a2 = _adam_logit(p_c, time.monotonic() + per, rng,
                                       lr=0.05, beta_frac=0.08, beta_decay=0.93,
                                       no_imp_limit=250)
            _update(p_a2, sc_a2, f'adam_{lab_c}_n{n_c}')

    # ===== Phase 4: LP refinement (200s) =====
    phase4_end = t_start + 1020.0
    print(f"Phase 4: LP refinement...")

    if best_seq is not None and best_seq.size <= 2048:
        n_lp = best_seq.size
        print(f"  LP refine n={n_lp}, score={best_score:.10f}")
        p_lp, sc_lp = _lp_refine(best_seq, min(phase4_end, time.monotonic() + 100),
                                  t=0.4, max_solves=20)
        _update(p_lp, sc_lp, 'LP_refine')

        # Adam polish after LP
        if time.monotonic() < phase4_end - 10:
            per = min(60.0, phase4_end - time.monotonic() - 5)
            if per > 15:
                p_ap, sc_ap = _adam_logit(p_lp, time.monotonic() + per, rng,
                                           lr=0.04, beta_frac=0.08, beta_decay=0.93,
                                           no_imp_limit=200)
                _update(p_ap, sc_ap, 'post_LP_polish')

    # ===== Phase 5: Multi-scale escalation (300s) =====
    phase5_end = t_start + 1320.0
    print(f"Phase 5: Multi-scale escalation...")

    if time.monotonic() < phase5_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]
        else:
            target_ns = []

        for tn in target_ns:
            if time.monotonic() > phase5_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            # Fienup at new scale
            per = min(80.0, phase5_end - time.monotonic() - 60)
            if per > 20:
                p_f2, sc_f2 = _fienup_hybrid(p_int, time.monotonic() + per, rng,
                                               method='power')
                _update(p_f2, sc_f2, f'fienup_escape_n{tn}')

            # Adam at new scale
            per2 = min(80.0, phase5_end - time.monotonic() - 30)
            if per2 > 20:
                p_e, sc_e = _adam_logit(best_seq, time.monotonic() + per2, rng,
                                         lr=0.05, beta_frac=0.08, beta_decay=0.92,
                                         no_imp_limit=300)
                _update(p_e, sc_e, f'adam_escape_n{tn}')

            # LP at new scale (only if small enough)
            if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase5_end - 30:
                p_lp2, sc_lp2 = _lp_refine(best_seq, min(phase5_end - 10, time.monotonic() + 60),
                                            t=0.3, max_solves=8)
                _update(p_lp2, sc_lp2, f'LP_n{best_seq.size}')

    # ===== Phase 6: Multi-restart with Fienup+Adam (250s) =====
    phase6_end = deadline - 50.0
    print(f"Phase 6: Multi-restart...")

    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(20.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 15:
                break

            # Create perturbed starting point
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.08 + 0.06 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)

            # Fienup on perturbed
            per_f = per * 0.4
            p_rf, sc_rf = _fienup_iterate(p_s, time.monotonic() + per_f, rng,
                                           gamma_start=0.9, gamma_end=0.7,
                                           n_iters=400, blend=0.7)
            _update(p_rf, sc_rf, f'restart_fienup{i}_n{n_best}')

            # Adam on Fienup result
            per_a = per * 0.5
            p_ra, sc_ra = _adam_logit(p_rf, time.monotonic() + per_a, rng, lr=0.05,
                                       beta_frac=0.07, beta_decay=0.93,
                                       no_imp_limit=int(0.3 * n_best))
            _update(p_ra, sc_ra, f'restart_adam{i}_n{n_best}')

    # ===== Phase 7: Final polish (50s) =====
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 15:
            p_f, sc_f = _adam_logit(best_seq, time.monotonic() + per, rng,
                                     lr=0.02, beta_frac=0.05, beta_decay=0.95,
                                     no_imp_limit=150)
            _update(p_f, sc_f, 'final_polish')

            # One more Fienup pass
            if time.monotonic() < deadline - 5:
                p_ff, sc_ff = _fienup_iterate(best_seq, deadline - 5, rng,
                                               gamma_start=0.95, gamma_end=0.85,
                                               n_iters=300, blend=0.8)
                _update(p_ff, sc_ff, 'final_fienup')

    # ===== Final validation =====
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(512, 1.5)

        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(512, 1.5)
    print(f"FALLBACK n=512, score={_score(fb):.10f}")
    return [float(x) for x in fb]
