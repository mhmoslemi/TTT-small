"""
Optimized multi-strategy search for low-autocorrelation non-negative sequences.
Uses adaptive smooth-max Adam, weighted multi-peak subgradient, LP spreading,
and coordinate descent polish across multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 48:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _smooth_max_grad(a, c, beta):
    """Compute gradient of smooth max of c with temperature beta."""
    mval = float(c.max())
    if beta < 1e-10:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        x = np.clip((c - mval) / beta, -50.0, 50.0)
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
    # Gradient: dc[k]/da[j] = a[k-j] + a[j-k] (with bounds)
    # Vectorized: grad[j] = 2 * sum_k w[k] * (a[k-j] + a[j-k])
    # Which equals 2 * (w * a_rev)[j] convolved appropriately
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = np.zeros(N)
    a_rev[:n] = a[::-1]
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_rev, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_optimize(a0, deadline, max_iter=80000, lr=0.04, beta_init=0.04,
                   cool_rate=0.995, restart_patience=5000, seed=0):
    """Adam optimization in log-space with adaptive smooth-max temperature."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    beta = beta_init * float(c.max())
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        cmax = float(c_curr.max())
        beta = max(beta_init * cmax * 0.5, 1e-12)

        grad_a = _smooth_max_grad(a, c_curr, beta)
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m, v_arr, adam_t = np.zeros(n), np.zeros(n), 0
            continue

        adam_t += 1
        m = b1 * m + (1 - b1) * grad_z
        v_arr = b2 * v_arr + (1 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        # Cosine-like LR with decay
        lr_cur = lr * (1.0 + np.cos(np.pi * min(it, max_iter) / max_iter)) * 0.5
        lr_cur = max(lr_cur, lr * 0.01)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > restart_patience:
                # Warm restart
                z = best_z + 0.2 * rng.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                m, v_arr, adam_t = np.zeros(n), np.zeros(n), 0
                beta = beta_init * float(_fft_convolve(a).max())
                no_imp = 0
                continue

        z = z_new
        a = a_new

    return best_a, best_score


def _multi_peak_subgradient(a0, deadline, max_iter=20000, top_k=5, base_lr=0.15):
    """Subgradient method targeting top-k peaks with exponential weighting."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c_curr)[::-1][:k]
        top_vals = c_curr[top_idx]
        max_val = top_vals[0]

        # Exponential weighting: heavier weight on higher peaks
        weights = np.exp(-0.5 * (top_vals - max_val) / max(max_val * 0.1, 1e-10))
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_idx):
            idx = int(idx)
            # dc[idx]/da[j] = a[idx-j] + a[j-idx] (with bounds)
            j1_s = max(0, idx - n + 1)
            j1_e = min(idx, n - 1)
            if j1_s <= j1_e:
                sg[j1_s:j1_e + 1] += weights[i] * a[idx - np.arange(j1_s, j1_e + 1)]
            j2_s = max(idx, 0)
            j2_e = min(idx + n - 1, n - 1)
            if j2_s <= j2_e:
                idx_arr = np.arange(j2_s, j2_e + 1)
                sg[idx_arr] += weights[i] * a[idx_arr - idx]

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, 3 * n):
                break

        a = a_new

    return best_a, best_score


def _lp_spread(a, M, n_constraints=None):
    """LP: maximize sum(g) subject to conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    n_con = 2 * n - 1
    if n_constraints is not None and n_constraints < n_con:
        # Only use top constraint constraints for speed
        c = _fft_convolve(a)
        top_idx = np.argsort(c)[::-1][:n_constraints]
        A_rows = []
        b_rows = []
        for k in top_idx:
            row = np.zeros(n)
            k = int(k)
            j1_s = max(0, k - n + 1)
            j1_e = min(k, n - 1)
            if j1_s <= j1_e:
                row[j1_s:j1_e + 1] = a[k - np.arange(j1_s, j1_e + 1)]
            j2_s = max(k, 0)
            j2_e = min(k + n - 1, n - 1)
            if j2_s <= j2_e:
                idx_arr = np.arange(j2_s, j2_e + 1)
                row[idx_arr] += a[idx_arr - k]
            A_rows.append(row)
            b_rows.append(M)
        A = np.array(A_rows)
        b = np.array(b_rows)
    else:
        A = np.zeros((n_con, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(n_con, M)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

def _coord_descent_polish(a0, deadline, max_passes=20):

    a = np.asarray(a0, dtype=np.float64).copy()

            break
        improved = False

            if time.monotonic() > deadline:

            peak = float(c_curr.max())

            # Sensitivity: da[j] by delta changes c[k] by delta*(a[k-j]+a[j-k])

def _make_profiles(n, rng):

    a = 1.0 + 1.5 * u - 0.5 * u ** 3
    profiles.append(np.maximum(a, 0.01))

    return [p / p.sum() for p in profiles]


def _interpolate_to_length(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)

                a2, s2 = _multi_peak_subgradient(a1, d2, max_iter=10000, top_k=5,
                                                 base_lr=0.12)
                sc2 = _score(a2)
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = a2.copy()
                    print(f"  n={n} init={i} Subgrad: {sc2:.10f}")
                a1 = a2 if sc2 < sc1 else a1

            if time.monotonic() > deadline - 30:
                break

            # Phase 3: LP refinement (only for promising candidates)
            if sc1 < best_score * 1.01 and n <= 768:
                t3 = time.monotonic()
                lp_budget = min(per_init_time * 0.15, 40.0)
                if lp_budget > 10:
                    d3 = t3 + lp_budget
                    a3 = _lp_refine(a1, d3, max_iter=200, inflate=1.002, blend=0.35)
                    sc3 = _score(a3)
                    if sc3 < best_score:
                        best_score = sc3
                        best_seq = a3.copy()
                        print(f"  n={n} init={i} LP: {sc3:.10f}")
                    a1 = a3 if sc3 < sc1 else a1

        # After all inits for this length, do a final Adam on the best
        if time.monotonic() < deadline - 60:
            best_n = best_seq.size
            if best_n == n:
                best_at_n = best_seq.copy()
            else:
                best_at_n = _interpolate_to_length(best_seq, n)

            t_final = time.monotonic()
            final_budget = min(120.0, deadline - 60 - t_final)
            if final_budget > 30:
                d_final = t_final + final_budget
                af, sf = _adam_optimize(best_at_n, d_final, max_iter=50000, lr=0.015,
                                        beta_init=0.015, restart_patience=4000,
                                        seed=seed + 9999 + n)
                scf = _score(af)
                if scf < best_score:
                    best_score = scf
                    best_seq = af.copy()
                    print(f"  n={n} Final Adam: {scf:.10f}")

    # Final polish phase on the overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final polish phase (best={best_score:.10f}, n={best_seq.size}) ===")

        # Subgradient polish
        t_p1 = time.monotonic()
        p1_budget = min(60.0, deadline - 30 - t_p1)
        if p1_budget > 10:
            ap1, sp1 = _multi_peak_subgradient(best_seq, t_p1 + p1_budget,
                                               max_iter=8000, top_k=8, base_lr=0.05)
            scp1 = _score(ap1)
            if scp1 < best_score:
                best_score = scp1
                best_seq = ap1.copy()
                print(f"  Polish Subgrad: {scp1:.10f}")

        # LP polish (only if n isn't too large)
        if best_seq.size <= 512 and time.monotonic() < deadline - 15:
            t_p2 = time.monotonic()
            p2_budget = min(40.0, deadline - 15 - t_p2)
            if p2_budget > 10:
                ap2 = _lp_refine(best_seq, t_p2 + p2_budget, max_iter=100,
                                 inflate=1.001, blend=0.2)
                scp2 = _score(ap2)
                if scp2 < best_score:
                    best_score = scp2
                    best_seq = ap2.copy()
                    print(f"  Polish LP: {scp2:.10f}")

        # Ultra-fine Adam
        if time.monotonic() < deadline - 5:
            t_p3 = time.monotonic()
            p3_budget = deadline - 5 - t_p3
            if p3_budget > 5:
                ap3, sp3 = _adam_optimize(best_seq, t_p3 + p3_budget, max_iter=20000,
                                          lr=0.005, beta_init=0.005,
                                          restart_patience=8000, seed=seed + 77777)
                scp3 = _score(ap3)
                if scp3 < best_score:
                    best_score = scp3
                    best_seq = ap3.copy()
                    print(f"  Ultra Adam: {scp3:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = _score(best_seq)

    # Final validation and normalization
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s

    # Verify with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence confirms: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
        else:
            print(f"\nWARNING: evaluate_sequence returned {sc_eval}, using fallback")
            final_seq = _normalize(best_seq)
            sc_eval2 = float(evaluate_sequence(final_seq.tolist()))
            print(f"  Fallback score: {sc_eval2:.10f}")
    except Exception as e:
        print(f"\nEvaluation error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
