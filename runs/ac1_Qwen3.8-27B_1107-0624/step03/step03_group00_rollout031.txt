def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.maximum(np.convolve(a, a), 0.0)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return None
    return a / s

def _score(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if mval < 1e-15:
        return mval, np.zeros_like(c)
    x = (c - mval) / beta
    ex = np.exp(np.maximum(x, -50))
    s = float(ex.sum())
    if s < 1e-15:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return mval, w
    w = ex / s
    return mval, w

def _cross_corr(w, p):
    """Compute sum_k w[k] * conv_rev(p, k) = cross-correlation."""
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    # We need: g[j] = sum_k w[k] * p[k-j] (with appropriate bounds)
    # This is the cross-correlation of w and p
    # Using FFT: cross_corr = ifft(fft(w) * conj(fft(p)))
    # but we need to be careful about the definition
    fp = np.fft.rfft(p, N)
    cc = np.fft.irfft(fw * np.conj(fp), N)
    # cc[k] = sum_j w[(k-j) mod N] * p[j]
    # We want g[j] = sum_k w[k] * p[k-j] for j = 0,...,n-1
    # which is cc shifted
    return cc[n-1:2*n-1] if n <= N//2 else cc[:n]

def _adam_step(p, c, beta, z, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    _, w = _smooth_max_weights(c, beta)
    g = _cross_corr(w, p)
    n = len(p)
    # Project gradient to simplex tangent
    vp = g
    dot = float(np.dot(p, vp))
    gz = p * (vp - dot)
    gz -= float(np.mean(gz))
    
    t += 1
    m = b1 * m + (1.0 - b1) * gz
    v = b2 * v + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    z = z - lr * mh / (np.sqrt(vh) + eps)
    return z, m, v, t

def _lp_solve(a, M, n_active=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None
