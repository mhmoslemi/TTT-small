def _full_convolve(a, b):
    """Compute full convolution using FFT for efficiency."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _safe_softmax(z, max_threshold=10000.0):
    """Softmax but ensure no element exceeds max_threshold."""
    max_z = max(z)
    if max_z is None:
        return None
    z_adj = z - max_z
    exp_z = np.exp(z_adj)
    if np.any(np.isinf(exp_z)):
        # Revert to safer computation with capped exp
        exp_z_new = np.where(z_adj < math.log(max_threshold), exp_z, max(exp_z))
        total = np.sum(exp_z_new)
        return exp_z_new / total
    total = np.sum(exp_z)
    return exp_z / total
