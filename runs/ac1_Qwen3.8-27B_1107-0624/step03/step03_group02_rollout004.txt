"""
Improved step function optimization combining adaptive tilted smooth-max Adam,
Nesterov momentum subgradient, frequency-aware initialization, and LP refinement.
Targets n=2048 for improved score over previous n=1024 baseline of 1.506659.
Key innovations: spectral initialization, temperature annealing schedule,
momentum-based subgradient, and efficient multi-start exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_max_weights(c, beta):
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 80.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _tilted_smooth_max_weights(c, beta, tilt=1.5):
    """Weights that emphasize the top peaks more strongly."""
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -80.0, 80.0)
    # Apply tilt: raise to power > 1 to emphasize peaks
    ex = np.exp(x * tilt)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s

    """Gradient of sum_k w[k]*(a*a)[k] with respect to a. Uses FFT."""
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = a[::-1].copy()
    a_pad = np.zeros(N)
    a_pad[:n] = a_rev
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_pad, N)
    conv_result = np.fft.irfft(fw * fa, N)
    h = conv_result[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient_at_max(a, k):
    n = len(a)
    sg = np.zeros(n)
    j1_start = max(0, k - n + 1)
    j1_end = min(k, n - 1)
    if j1_start <= j1_end:
        idx = np.arange(j1_start, j1_end + 1)
        sg[idx] += a[k - idx]
    j2_start = max(k, 0)
    j2_end = min(k + n - 1, n - 1)
    if j2_start <= j2_end:
        idx = np.arange(j2_start, j2_end + 1)
        sg[idx] += a[idx - k]
    return sg


def _adam_logit(a, deadline, max_iter=30000, lr=0.025, beta_init=0.04,
                beta_min=0.003, beta_decay_per_100=0.88, tilt=1.3, verbose=False):
    """Adam in logit-space with adaptive tilted smooth max and temperature annealing."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-14)
    a = a / a.sum()

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = beta_init * float(c.max())
    no_imp = 0
    base_lr = lr
    lr_cur = lr
    best_no_imp_lr = 0.001

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _tilted_smooth_max_weights(c_curr, beta, tilt=tilt)
        grad_a = _grad_weighted_conv(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        gnorm = float(np.linalg.norm(grad_z))
        if gnorm < 1e-18:
            break

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr_cur = base_lr
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        e_max = float(e.max())
        if e_max > 45.0:
            z_new = z_new - e_max + 45.0
            e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr_cur = max(best_no_imp_lr, lr_cur * 0.95)
            if no_imp > max(5000, 8 * n):
                rng_r = np.random.default_rng(5000 + it)
                z = best_z + 0.4 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(beta_min, beta * 1.5)
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        # Temperature annealing
        if it % 100 == 0 and it > 0:
            beta *= beta_decay_per_100
            beta = max(beta_min, beta)

        if verbose and it % 5000 == 0:
            print(f"      Adam it={it}: best={best_score:.10f}, beta={beta:.3e}, lr={lr_cur:.5f}")

    return best_a, best_score


def _nesterov_subgradient(a, deadline, max_iter=10000, top_k=2, verbose=False):
    """Nesterov momentum subgradient targeting top-k peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    v = np.zeros(n)  # momentum
    mu = 0.6  # momentum parameter

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    base_lr = 0.4 / np.sqrt(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        a_look = a - mu * v
        a_look = np.maximum(a_look, 0.0)
        s_look = float(a_look.sum())
        if s_look > 1e-10:
            a_look = a_look / s_look
        else:
            a_look = a

        c_curr = _fft_convolve(a_look)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]

        # Weighted subgradient with exponential decay
        weights = np.exp(-0.5 * np.arange(k))
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a_look, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a_look - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        v = mu * v + (a_new - a)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 5 * n):
                break

        a = a_new

        if verbose and it % 3000 == 0:
            print(f"      NesterovSG it={it}: best={best_score:.10f}")

    return best_a, best_score


def _lp_spread_fast(a, M, max_iter_lp=50):
    """Fast LP: maximize sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if n > 4096:
        return None
    c_obj = -np.ones(n)
    # Build constraint matrix sparsely
    # (a*g)[k] = sum_j a[j]*g[k-j] <= M
    # This is a convolution constraint matrix (Toeplitz-like)
    # For efficiency, limit to n constraints (most important ones)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'maxiter': max_iter_lp * 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.0008, blend=0.3):
    """LP-based refinement: spread mass to reduce peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread_fast(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.005, inflate_cur * 1.03)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.003, inflate_cur * 0.93)
            if no_imp > max(150, int(1.5 * n)):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, alpha=1.0, freq=1):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(2 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quad_profile(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + alpha * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=0.8):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _spectral_init(n, target_flatness=0.3, rng=None):
    """Initialize based on spectral considerations.
    Create a sequence whose autocorrelation is close to flat."""
    if rng is None:
        rng = np.random.default_rng(0)
    # Start with a random phase in frequency domain
    A_mag = np.ones(n)  # flat magnitude
    A_phase = rng.uniform(0, 2 * np.pi, n)
    A = A_mag * np.exp(1j * A_phase)
    a = np.fft.irfft(A, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    # Blend with uniform for stability
    a = (1 - target_flatness) * a + target_flatness * np.ones(n) / n
    return a / a.sum()


def _full_pipeline(a0, deadline, label=""):
    """Run multi-phase optimization pipeline."""
    sc0 = _score(a0)
    if np.isfinite(sc0):
        print(f"  {label} init: {sc0:.10f}", flush=True)

    best_a = a0.copy()
    best_sc = sc0

    # Phase 1: Adam smooth max (coarse)
    if time.monotonic() < deadline - 80:
        d1 = min(deadline - 80, time.monotonic() + 200)
        a1, s1 = _adam_logit(a0, d1, max_iter=20000, lr=0.03,
                             beta_init=0.05, beta_min=0.005, tilt=1.3)
        if s1 < best_sc:
            best_a, best_sc = a1, s1
        print(f"  {label} Adam1: {best_sc:.10f}", flush=True)

    # Phase 2: Nesterov subgradient
    if time.monotonic() < deadline - 60:
        d2 = min(deadline - 60, time.monotonic() + 120)
        a2, s2 = _nesterov_subgradient(best_a, d2, max_iter=6000, top_k=2)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} NSG: {best_sc:.10f}", flush=True)

    # Phase 3: LP refinement
    if time.monotonic() < deadline - 40:
        d3 = min(deadline - 40, time.monotonic() + 100)
        a3 = _lp_refine(best_a, d3, max_iter=1500, inflate=1.0008, blend=0.25)
        s3 = _score(a3)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} LP: {best_sc:.10f}", flush=True)

    # Phase 4: Fine Adam
    if time.monotonic() < deadline - 20:
        d4 = deadline - 20
        a4, s4 = _adam_logit(best_a, d4, max_iter=15000, lr=0.012,
                             beta_init=0.02, beta_min=0.002, tilt=1.5)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} Adam2: {best_sc:.10f}", flush=True)

    # Phase 5: Final subgradient
    if time.monotonic() < deadline - 8:
        d5 = deadline - 8
        a5, s5 = _nesterov_subgradient(best_a, d5, max_iter=4000, top_k=3)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} FinalSG: {best_sc:.10f}", flush=True)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization for step function coefficient minimization.
    Uses n=2048 as primary length with spectral init, adaptive Adam,
    Nesterov subgradient, and LP refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # === Primary strategy: n=2048 ===
    n_primary = 2048

    if parent is not None:
        p_primary = _interpolate_profile(parent, n_primary)
    else:
        p_primary = _make_linear_profile(n_primary, 1.5)

    print(f"\n=== Strategy 1: Primary @ n={n_primary} ===", flush=True)
    d1 = min(deadline - 600, t_start + 550)
    a_best, s_best = _full_pipeline(p_primary, d1, "S1")
    if s_best < best_score:
        best_score = s_best
        best_seq = a_best.copy()
    print(f"  S1 final: {best_score:.10f}", flush=True)

    # === Strategy 2: Diverse initializations at n=2048 ===
    if time.monotonic() < deadline - 300:
        print(f"\n=== Strategy 2: Diverse inits @ n={n_primary} ===", flush=True)
        candidates = [
            _make_linear_profile(n_primary, 1.0),
            _make_linear_profile(n_primary, 2.5),
            _make_cosine_profile(n_primary, 0.7, 1),
            _make_cosine_profile(n_primary, 1.2, 2),
            _make_quad_profile(n_primary, 1.5),
            _make_bell_profile(n_primary, 0.6),
            _spectral_init(n_primary, 0.4, rng),
        ]
        # Perturbed parent
        if parent is not None:
            p_pert = p_primary * (1.0 + 0.2 * rng.normal(size=n_primary))
            p_pert = np.maximum(p_pert, 0.0)
            p_pert = p_pert / p_pert.sum()
            candidates.append(p_pert)

        time_per_cand = min(35.0, (deadline - 300 - time.monotonic()) / max(1, len(candidates)))
        for i, a0 in enumerate(candidates):
            if time.monotonic() > deadline - 300:
                break
            if time_per_cand < 8:
                break
            d2i = min(deadline - 300, time.monotonic() + time_per_cand)
            a_q, s_q = _adam_logit(a0, d2i, max_iter=6000, lr=0.02,
                                   beta_init=0.04, beta_min=0.005, tilt=1.3)
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
            print(f"  S2-{i} (t={time.monotonic()-t_start:.0f}s): {s_q:.10f}", flush=True)

    # === Strategy 3: Refined optimization on best ===
    if best_seq is not None and time.monotonic() < deadline - 100:
        print(f"\n=== Strategy 3: Refine best (score={best_score:.10f}) ===", flush=True)
        # Adam fine-tune
        d3a = min(deadline - 60, time.monotonic() + 150)
        a_f1, s_f1 = _adam_logit(best_seq, d3a, max_iter=12000, lr=0.008,
                                  beta_init=0.015, beta_min=0.001, tilt=1.5)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1.copy()
        print(f"  S3 Adam: {best_score:.10f}", flush=True)

        # Subgradient
        if time.monotonic() < deadline - 40:
            d3b = min(deadline - 40, time.monotonic() + 80)
            a_f2, s_f2 = _nesterov_subgradient(best_seq, d3b, max_iter=4000, top_k=3)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2.copy()
            print(f"  S3 NSG: {best_score:.10f}", flush=True)

        # LP polish
        if time.monotonic() < deadline - 15:
            d3c = deadline - 15
            a_f3 = _lp_refine(best_seq, d3c, max_iter=1000, inflate=1.0005, blend=0.15)
            s_f3 = _score(a_f3)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3.copy()
            print(f"  S3 LP: {best_score:.10f}", flush=True)

    # === Strategy 4: Try n=4096 if time allows ===
    if time.monotonic() < deadline - 200:
        n_large = 4096
        print(f"\n=== Strategy 4: Try n={n_large} ===", flush=True)
        if best_seq is not None:
            p_large = _interpolate_profile(best_seq, n_large)
        else:
            p_large = _make_linear_profile(n_large, 1.5)

        # Quick Adam on the larger sequence
        d4 = min(deadline - 50, time.monotonic() + 180)
        a_large, s_large = _adam_logit(p_large, d4, max_iter=8000, lr=0.02,
                                        beta_init=0.03, beta_min=0.003, tilt=1.3)
        if s_large < best_score:
            best_score = s_large
            best_seq = a_large.copy()
        print(f"  S4 large: {best_score:.10f}", flush=True)

        # Subgradient on large
        if time.monotonic() < deadline - 30:
            d4b = min(deadline - 30, time.monotonic() + 80)
            a_l2, s_l2 = _nesterov_subgradient(best_seq, d4b, max_iter=3000, top_k=2)
            if s_l2 < best_score:
                best_score = s_l2
                best_seq = a_l2.copy()
            print(f"  S4 NSG: {best_score:.10f}", flush=True)

    # === Strategy 5: Final micro-optimization ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Strategy 5: Final micro-opt ===", flush=True)
        d5 = deadline - 5
        a_final, s_final = _adam_logit(best_seq, d5, max_iter=5000, lr=0.004,
                                        beta_init=0.008, beta_min=0.001, tilt=1.8)
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
        print(f"  S5 final: {best_score:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)
        best_score = _score(final_seq)

    # Evaluate with authoritative function
    if final_seq.size <= 8192:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
