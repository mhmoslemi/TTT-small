def _solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0"""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _make_profiles(n):
    """Generate diverse profile initializations."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    
    # Power law ramps
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = t ** alpha + 1e-8
        profiles.append(a / a.sum())
    
    # Inverse ramps
    for alpha in [0.5, 1.0, 2.0]:
        a = (1.0 - t) ** alpha + 1e-8
        profiles.append(a / a.sum())
    
    # Symmetric profiles
    for alpha in [1.0, 2.0, 3.0]:
        a = (1.0 - np.abs(2.0 * t - 1.0)) ** alpha + 1e-8
        profiles.append(a / a.sum())
    
    # Cosine profiles
    for freq in [1, 2, 3]:
        a = 0.5 + 0.5 * np.cos(np.pi * freq * t) + 1e-8
        a = np.maximum(a, 0)
        if a.sum() > 0.01:
            profiles.append(a / a.sum())
    
    # Quadratic
    a = 1.0 + 3.0 * t * t + 1e-8
    profiles.append(a / a.sum())
    
    # Sinc-like (positive part)
    x = np.linspace(-4, 4, n)
    a = np.sinc(x / (2 * np.pi)) ** 2 + 1e-8
    a = np.maximum(a, 0)
    profiles.append(a / a.sum())
    
    # Gaussian
    a = np.exp(-3.0 * (t - 0.5) ** 2) + 1e-8
    profiles.append(a / a.sum())
    
    # Two bumps
    a = np.exp(-10.0 * (t - 0.25) ** 2) + np.exp(-10.0 * (t - 0.75) ** 2) + 1e-8
    profiles.append(a / a.sum())
    
    # Arnold-like (raised cosine)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t) + 1e-8
    a = np.maximum(a, 0)
    if a.sum() > 0.01:
        profiles.append(a / a.sum())
    
    return profiles

def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

def _random_smooth(n, rng, n_fourier=5):
    """Generate a random smooth non-negative profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 0.5
    for k in range(1, n_fourier + 1):
        amp = rng.uniform(0.1, 0.5) / k
        freq = rng.uniform(1, 3) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines fine-grained gradient descent with adaptive smooth-max, LP direction finding,
peak-shaving heuristics, and simulated-annealing restarts to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 500:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N_pad = _next_pow2(2 * n)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _subgrad_at_argmax(a):
    n = len(a)
    c = _fft_convolve(a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for j in range(n):
        m = k_star - j
        if 0 <= m < n:
            grad[j] = 2.0 * a[m]
    return grad, k_star


def _backtracking_line_search(a, direction, initial_lr, c1=0.01):

    base_score = _score(a)
    base_grad, _ = _smooth_max_grad(a, 0.01 * float(_fft_convolve(a).max()))
    lr = initial_lr
    for _ in range(40):
        a_new = np.maximum(a - lr * direction, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc <= base_score - 1e-15:
            return a_new, sc, lr
        lr *= 0.5
        if lr < 1e-14:

    return a, base_score, 0.0


def _gradient_descent(a0, deadline, max_iter=80000, use_subgrad_mix=True):

    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_beta = beta
    lr = 0.05
    base_lr = lr

    v = np.zeros(n)

        else:

def _solve_lp(f, M, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.002, inflate_min=1.0001):

    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.1, inflate * 1.15)
            if no_imp > 500:
                break
            continue

        g = g / sum_g

        # Fine line search
        best_t = 0.0
        best_sc = best_score
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:

            a_new = a_new / s_new
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t

            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 100 == 99:

            if no_imp > max(200, 3 * n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                if inflate < 1.0005:

def _peak_shave(a0, deadline, max_iter=20000):
    """Iteratively shave the peak of the convolution."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    step = 0.02

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))

        # Find the most "responsible" index
        resp = np.zeros(n)
        for j in range(n):
            m = k_star - j
            if 0 <= m < n:
                resp[j] = a[j] * a[m]

        # Try reducing the top responsible index
        top_k = np.argsort(resp)[::-1][:5]
        improved = False
        for idx in top_k:
            for frac in [step, step * 0.5, step * 0.25]:
                a_new = a.copy()
                a_new[idx] *= (1.0 - frac)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                sc = _score(a_new)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    break
            if improved:
                break

        if not improved:
            step *= 0.9
            if step < 1e-6:
                break
            # Try increasing a non-responsible index
            low_resp = np.argsort(resp)[:5]
            for idx in low_resp:
                for frac in [step, step * 2.0]:
                    a_new = a.copy()
                    a_new[idx] *= (1.0 + frac)
                    s = a_new.sum()
                    if s < 1e-10:
                        continue
                    a_new = a_new / s
                    sc = _score(a_new)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_new.copy()
                        a = a_new
                        improved = True
                        break
                if improved:
                    break

    return best_a


def _make_profiles(n):
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    for alpha in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = t ** alpha + 1e-8
        profiles.append(a / a.sum())

    for alpha in [1.0, 2.0]:
        a = (1.0 - t) ** alpha + 1e-8
        profiles.append(a / a.sum())

    for alpha in [1.0, 2.0]:
        a = (1.0 - np.abs(2.0 * t - 1.0)) ** alpha + 1e-8
        profiles.append(a / a.sum())

    a = 1.0 + 2.0 * t + 1e-8
    profiles.append(a / a.sum())
    a = 1.0 + 4.0 * t * t + 1e-8
    profiles.append(a / a.sum())

    x = np.linspace(-3, 3, n)
    a = np.exp(-x * x) + 1e-8
    profiles.append(a / a.sum())

    a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 1e-8
    profiles.append(a / a.sum())

    a = 0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t) + 1e-8
    a = np.maximum(a, 0)
    if a.sum() > 0.01:
        profiles.append(a / a.sum())

    return profiles


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _random_smooth(n, rng, n_harmonics=4):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 1.0
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 0.4) / k
        freq = rng.uniform(0.5, 2.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _anneal_step(a, temp, rng):
    """Add a random perturbation scaled by temperature."""
    n = len(a)
    pert = rng.normal(0, temp, n)
    a_new = _project_simplex(a + pert)
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse starts, gradient descent with subgradient mixing,
    LP refinement, peak shaving, and annealing restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate candidates at multiple lengths
    lengths = [640, 768, 896, 1024]
    candidates = []

    for n_t in lengths:
        if parent is not None:
            p_int = _interpolate(parent, n_t)
            sc = _score(p_int)
            candidates.append((sc, f'parent_{n_t}', n_t, p_int))

        for i, prof in enumerate(_make_profiles(n_t)):
            sc = _score(prof)
            candidates.append((sc, f'p{i}_{n_t}', n_t, prof))

        for ri in range(2):
            rs = _random_smooth(n_t, rng)
            sc = _score(rs)
            candidates.append((sc, f'r{ri}_{n_t}', n_t, rs))

    candidates.sort(key=lambda x: x[0])

    print("Top candidates:")
    for sc, name, n_t, _ in candidates[:8]:
        print(f"  {name}: {sc:.10f}")

    # Phase 1: Gradient descent on top candidates
    phase1_deadline = deadline - 700
    n_try = min(5, len(candidates))

    for idx in range(n_try):
        if time.monotonic() > phase1_deadline:
            break
        sc0, name, n_t, a0 = candidates[idx]

        if sc0 >= best_score + 0.03:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        remaining = phase1_deadline - time.monotonic()
        if remaining < 20:
            break

        if idx == 0:
            gd_time = min(150.0, remaining * 0.4)
        elif idx < 3:
            gd_time = min(100.0, remaining * 0.3)
        else:
            gd_time = min(60.0, remaining * 0.2)

        if gd_time < 15:
            break

        gd_end = time.monotonic() + gd_time
        print(f"GD: {name} n={n_t} init={sc0:.8f} t={gd_time:.0f}s")

        a_gd = _gradient_descent(a0, gd_end, max_iter=80000)
        sc_gd = _score(a_gd)
        if np.isfinite(sc_gd) and sc_gd < best_score:
            best_score = sc_gd
            best_seq = a_gd.copy()
            print(f"  -> {sc_gd:.10f}")
        else:
            print(f"  -> {sc_gd:.8f} (no imp)")

    # Phase 2: LP on best
    if best_seq is not None and time.monotonic() < deadline - 350:
        lp_time = min(250.0, deadline - time.monotonic() - 150)
        if lp_time > 30:
            lp_end = time.monotonic() + lp_time
            print(f"LP phase: t={lp_time:.0f}s, best={best_score:.10f}")
            a_lp = _lp_refine(best_seq, lp_end, max_iter=50000, inflate_start=1.002)
            sc_lp = _score(a_lp)
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  -> LP: {sc_lp:.10f}")

    # Phase 3: Gradient polish
    if best_seq is not None and time.monotonic() < deadline - 200:
        gp_time = min(150.0, deadline - time.monotonic() - 100)
        if gp_time > 20:
            gp_end = time.monotonic() + gp_time
            print(f"GD polish: t={gp_time:.0f}s")
            a_gp = _gradient_descent(best_seq, gp_end, max_iter=60000)
            sc_gp = _score(a_gp)
            if np.isfinite(sc_gp) and sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  -> {sc_gp:.10f}")

    # Phase 4: Peak shave
    if best_seq is not None and time.monotonic() < deadline - 100:
        ps_time = min(80.0, deadline - time.monotonic() - 50)
        if ps_time > 15:
            ps_end = time.monotonic() + ps_time
            print(f"Peak shave: t={ps_time:.0f}s")
            a_ps = _peak_shave(best_seq, ps_end, max_iter=20000)
            sc_ps = _score(a_ps)
            if np.isfinite(sc_ps) and sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps.copy()
            print(f"  -> {sc_ps:.10f}")

    # Phase 5: Final tight LP
    if best_seq is not None and time.monotonic() < deadline - 50:
        lp_time = min(40.0, deadline - time.monotonic() - 25)
        if lp_time > 10:
            lp_end = time.monotonic() + lp_time
            a_fl = _lp_refine(best_seq, lp_end, max_iter=30000,
                              inflate_start=1.001, inflate_min=1.0001)
            sc_fl = _score(a_fl)
            if np.isfinite(sc_fl) and sc_fl < best_score:
                best_score = sc_fl
                best_seq = a_fl.copy()
            print(f"  -> Final LP: {sc_fl:.10f}")

    # Phase 6: Anneal + re-optimize best
    if best_seq is not None and time.monotonic() < deadline - 30:
        anneal_budget = min(20.0, deadline - time.monotonic() - 15)
        if anneal_budget > 8:
            print(f"Anneal: t={anneal_budget:.0f}s")
            a_cur = best_seq.copy()
            temp = 0.01
            for _ in range(200):
                if time.monotonic() > deadline - 15:
                    break
                a_try = _anneal_step(a_cur, temp, rng)
                sc_try = _score(a_try)
                if sc_try < best_score - 1e-14:
                    best_score = sc_try
                    best_seq = a_try.copy()
                    a_cur = a_try
                    temp = min(0.05, temp * 1.1)
                else:
                    temp = max(0.001, temp * 0.95)
            print(f"  -> After anneal: {best_score:.10f}")

    # Fallback
    if best_seq is None:
        n = 768
        t = np.linspace(0, 1, n)
        best_seq = (t + 1e-8) / (t + 1e-8).sum()
        best_score = _score(best_seq)

    # Finalize
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        n = 768
        t = np.linspace(0, 1, n)
        final_seq = (t + 1e-8) / (t + 1e-8).sum()

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
