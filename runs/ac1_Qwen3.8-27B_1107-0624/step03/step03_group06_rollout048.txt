        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s

def _symmetrize(a):
    n = len(a)
    return (a + a[::-1]) / 2.0

def _interpolate_to(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _normalize(p)

def _make_profile(n, type='ramp', param=1.0):
    t = np.linspace(0.0, 1.0, n)
    if type == 'ramp':
        a = t ** param + 1e-8
    elif type == 'ramp_rev':
        a = (1.0 - t) ** param + 1e-8
    elif type == 'lin':
        a = 1.0 + param * t
        a = np.maximum(a, 1e-8)
    elif type == 'quad':
        a = 1.0 + param * t + param * t * t
        a = np.maximum(a, 1e-8)
    elif type == 'cosine':
        a = 0.5 + 0.5 * np.cos(np.pi * t) + 1e-8
    elif type == 'bump':
        center = 0.5
        width = 0.3
        a = np.exp(-((t - center) / width) ** 2) + 1e-8
    elif type == 'flat_edges':
        # Flat in middle, decay at edges
        a = np.ones(n)
        edge = int(0.2 * n)
        for i in range(edge):
            a[i] = 0.5 + 0.5 * np.sin(np.pi * i / (2 * edge))
            a[n-1-i] = a[i]
        a = np.maximum(a, 1e-8)
    elif type == 'uniform':
        a = np.ones(n)
    elif type == 'superlinear':
        a = t ** 2 + 0.5 * t + 0.1
    else:
        a = t + 1e-8
    return _normalize(a)

def _solve_lp(f, M, n, rng=None):
    """Solve max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    # Build the TOEPLITZ constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # So (A @ g)[k] = sum_j f[k-j] * g[j] = (f*g)[k]
    
    # Use FFT-based approach for efficiency if n is large
    # For the LP, we need the full matrix, so let's build it
    
    if n <= 1024:
        # Build full matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            # A[k, j] = f[k-j] for valid k-j
            lo = max(0, j)
            hi = min(2 * n - 2, j + n - 1)
            A[lo:hi+1, j] = f[j:j + (hi-lo+1)]
        # Wait, let me think again.
        # (f*g)_k = sum_j f_{k-j} * g_j where f_i = 0 for i<0 or i>=n
        # So A[k,j] = f[k-j] if 0 <= k-j < n
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            # k-j must be in [0, n-1], so k in [j, j+n-1]
            lo = j
            hi = min(2*n-2, j+n-1)
            A[lo:hi+1, j] = f[0:hi-lo+1]
        
        c_obj = -np.ones(n)
        b = np.full(2*n-1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if res.success:
            return res.x
        return None
    else:
        # For large n, use a sparser approach
        # The matrix is banded (banded Toeplitz)
        # Use scipy's sparse matrix
        from scipy.sparse import lil_matrix, csr_matrix
        A = lil_matrix((2*n-1, n))
        for j in range(n):
            lo = j
            hi = min(2*n-2, j+n-1)
            A[lo:hi+1, j] = f[0:hi-lo+1]
        A = A.tocsr()
        c_obj = -np.ones(n)
        b = np.full(2*n-1, M)
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if res.success:
            return res.x
        return None

def _golden_section_line_search(a, g, n_iter=20):
    """Golden section line search for blending a and g."""
    golden = (np.sqrt(5) - 1) / 2  # ~0.618
    
    def f(t):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return _score(a_new)
    
    lo = 0.0
    hi = 1.0
    c = hi - golden * (hi - lo)
    d = lo + golden * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - golden * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + golden * (hi - lo)
            fd = f(d)
    
    best_t = (lo + hi) / 2
    best_sc = f(best_t)
    return best_t, best_sc

def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.01, inflate_min=1.0005):
    """LP-based refinement with adaptive inflation and perturbation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    
    no_imp = 0
    inflate = inflate_start
    perturbation_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        
        g = _solve_lp(a, M_lp, n)
        if g is None:
            # Try with larger inflation
            inflate = min(0.1, inflate * 1.5)
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        sum_g = float(g.sum())
        sum_a = float(a.sum())
        
        if sum_g <= sum_a * (1.0 + 1e-6):
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.05, inflate * 1.3)
                perturbation_count += 1
                if perturbation_count > 5:
                    # Add random perturbation
                    eps = 1e-4
                    a = a + eps * np.random.default_rng(it).uniform(-1, 1, n)
                    a = np.maximum(a, 0.0)
                    a = _normalize(a)
                    perturbation_count = 0
                if no_imp > 300:
                    break
            continue
        
        g = _normalize(g)
        
        best_t, best_sc = _golden_section_line_search(a, g, n_iter=15)
        
        if best_t > 0 and best_sc < best_score - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a = _normalize(a_new)
            best_score = best_sc
            best_a = a.copy()
            no_imp = 0
            inflate = max(inflate_min, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                inflate = min(0.05, inflate * 1.2)
            if no_imp > max(200, n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
                perturbation_count = 0
                if inflate > 1.002:
                    break
    
    return best_a

def _gradient_polish(a0, deadline, max_iter=20000):
    """Smooth-max gradient polish with momentum."""
    a = np.asarray(a0, dtype=np.float64).copy()

    a = _normalize(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    beta = max(1e-6, 0.03 * float(c.max()))
    lr = 0.05
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        # Clip to avoid overflow
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
        
        # Gradient of smooth max of (a*a) w.r.t. a
        # d/d a_i [sum_k w_k * (a*a)_k] = 2 * sum_k w_k * (T_k a)_i
        # where (T_k a)_i = sum_j w_k * a_{k+i-j} ... 
        # Actually, (a*a)_k = sum_j a_j * a_{k-j}
        # d(a*a)_k / da_i = a_{k-i} + a_{k-i} (if k-i != i) or 2*a_i (if k-i==i)

        # (a*a)_k = sum_j a_j * a_{k-j}
        # d/d a_i: the term j=i gives a_{k-i}, and if k-i = i (i.e., j=k-i=i), we get another a_i

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s
