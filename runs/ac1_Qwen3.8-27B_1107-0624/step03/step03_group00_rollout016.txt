Optimized multi-phase search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive LP-based refinement with adaptive M reduction, combined with
Adam smooth-max polish and multi-scale escalation.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())

def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p

def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s

def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

                best_a = a_try
                best_t = t_try

        current_peak = best_peak
        t = best_t
        
        if current_peak * 2.0 * n < current_score - 1e-10:
            current_score = 2.0 * n * current_peak

                no_improve = 0

    z = np.log(p0)
    z -= float(np.mean(z))

    v = np.zeros(n)
    t_adam = 0

        g = _conv_cross_corr(w, p)

            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        t_adam += 1

        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:

            m.fill(0.0)
            v.fill(0.0)

def _make_profile_linear(n, eta):

    t = np.arange(n)

def _make_profile_two_bump(n, c1, c2, s):

    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)

def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3

    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)

    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: progressive M tightening in LP, adaptive step-size line search,
quartic/cubic profile initialization, better time allocation, and scaled Adam polish.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_norm(p):
    n = len(p)
    c = _full_convolve(p)
    return float(2.0 * n * c.max())


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _conv_cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize_nonneg(a):

    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:

    return a / s


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)

            m.fill(0.0)

def _make_profile_flat_edges(n, w):

    a[:edge] = taper
    a[-edge:] = taper[::-1]

    return a / a.sum()

        try_update(p_lp, sc_lp, f'lp_{label}')

        # Quick Adam polish after LP
        if time.monotonic() < phase2_end - 20:
            per_ad = min(40.0, phase2_end - time.monotonic() - 10)
            if per_ad > 15:
                p_ad, sc_ad = _adam_logit(p_lp, time.monotonic() + per_ad, rng,
                                           lr=0.05, beta_frac=0.08, beta_decay=0.93,
                                           no_imp_limit=200)
                try_update(p_ad, sc_ad, f'lp_adam_{label}')

    # === Phase 3: Intensive Adam on best (300s) ===
    phase3_end = t_start + 1100.0
    if best_seq is not None and time.monotonic() < phase3_end:
        per = phase3_end - time.monotonic()
        if per > 30:
            print(f"Phase 3 Adam: n={best_seq.size} score={best_score:.10f} budget={per:.0f}s")
            p_ad, sc_ad = _adam_logit(best_seq, time.monotonic() + per, rng,
                                       lr=0.03, beta_frac=0.06, beta_decay=0.94,
                                       no_imp_limit=500)
            try_update(p_ad, sc_ad, 'adam_main')

            # Second LP pass on the Adam result
            if time.monotonic() < phase3_end - 30 and best_seq.size <= 2048:
                per_lp = min(80.0, phase3_end - time.monotonic() - 10)
                if per_lp > 20:
                    p_lp2, sc_lp2 = _lp_refine_progressive(best_seq, time.monotonic() + per_lp,
                                                            max_solves=15, initial_t=0.3)
                    try_update(p_lp2, sc_lp2, 'lp_after_adam')

    # === Phase 4: Multi-scale escalation (300s) ===
    phase4_end = t_start + 1500.0
    if time.monotonic() < phase4_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 512:
            target_ns = [1024, 2048]
        elif n_cur <= 1024:
            target_ns = [2048]

        for tn in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            try_update(p_int, sc_int, f'interp_n{tn}')

            if time.monotonic() < phase4_end - 40:
                per_lp = min(100.0, phase4_end - time.monotonic() - 40)
                if per_lp > 25:
                    p_lp3, sc_lp3 = _lp_refine_progressive(p_int, time.monotonic() + per_lp,
                                                            max_solves=12, initial_t=0.35)
                    try_update(p_lp3, sc_lp3, f'LP_n{tn}')

            if time.monotonic() < phase4_end - 25:
                per_ad = min(50.0, phase4_end - time.monotonic() - 25)
                if per_ad > 15:
                    p_ad3, sc_ad3 = _adam_logit(best_seq, time.monotonic() + per_ad, rng,
                                                 lr=0.03, beta_frac=0.05, beta_decay=0.94,
                                                 no_imp_limit=300)
                    try_update(p_ad3, sc_ad3, f'adam_n{tn}')

    # === Phase 5: Multi-restart (200s) ===
    phase5_end = deadline - 50.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 4
        per_r = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase5_end:
                break
            per = min(per_r, phase5_end - time.monotonic() - 5)
            if per < 15:
                break
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.06 + 0.05 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per, rng, lr=0.04,
                                     beta_frac=0.07, beta_decay=0.93, no_imp_limit=200)
            try_update(p_r, sc_r, f'restart{i}_n{n_best}')

            # Quick LP on restart if it improved
            if sc_r < best_score and best_seq.size <= 2048 and time.monotonic() < phase5_end - 30:
                per_lr = min(40.0, phase5_end - time.monotonic() - 10)
                if per_lr > 15:
                    p_rl, sc_rl = _lp_refine_progressive(best_seq, time.monotonic() + per_lr,
                                                          max_solves=8, initial_t=0.3)
                    try_update(p_rl, sc_rl, f'restart_lp{i}')

    # === Phase 6: Final polish ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 15:
            p_f, sc_f = _adam_logit(best_seq, time.monotonic() + per, rng,
                                     lr=0.01, beta_frac=0.03, beta_decay=0.96,
                                     no_imp_limit=100)
            try_update(p_f, sc_f, 'final_polish')

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass

        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
