import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    n = a.size
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = c - c.min()
    return c

def _score(a):
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def _project_simplex(v):
    # Project v onto {x : x >= 0, sum(x) = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w / w.sum()

"""
Multi-scale subgradient optimization with adaptive top-k peak focusing,
momentum-based updates, logit-space Adam, LP refinement, and iterated local
search for minimizing 2n * max(conv(a,a)) / (sum(a))^2.

Key innovations over previous approach:
- Exact (non-smooth) subgradient with adaptive top-k averaging
- Iterated local search with diverse perturbations
- Larger sequence lengths (up to 2048) for better asymptotic behavior
- Two-timescale optimization: coarse peak-finding then fine polish
- Logit-space parameterization prevents boundary issues
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 192:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 1e-12:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _subgradient_topk(a, c, top_k=3):
    """Compute subgradient of max(c) averaged over top-k peaks.
    Returns gradient direction (not normalized)."""
    n = a.size
    c_len = len(c)
    # Get top-k indices in c
    if top_k >= c_len:
        top_k = c_len
    idx = np.argpartition(c, -top_k)[-top_k:]
    
    # For each peak at position p: grad[i] = a[p-i] if 0<=p-i<n
    grad = np.zeros(n)
    for p in idx:
        p = int(p)
        # i such that 0 <= p-i < n, i.e., p-n+1 <= i <= p, intersected with [0, n-1]
        i_lo = max(0, p - n + 1)
        i_hi = min(p, n - 1)
        if i_lo <= i_hi:
            j = p - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += a[j]
    
    return grad


def _gradient_smooth_max(a, beta=0.01):
    """Smooth max gradient (fallback for stability)."""
    n = a.size
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    # Numerical stability
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_v_profile(n, alpha=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.sin(np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p=1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + p * (0.5 + 0.5 * u) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_mixture(n, centers, sigmas, weights):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c, s, w in zip(centers, sigmas, weights):
        a += w * np.exp(-0.5 * ((t - c) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_spread(a, M):
    n = len(a)
    if n > 1500:
        return None
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _subgradient_optimize(a, deadline, max_iter=50000, lr_init=0.05, top_k_init=3):
    """Exact subgradient method with momentum and adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    lr = lr_init
    momentum = 0.80
    v = np.zeros(n)
    no_imp = 0
    top_k = top_k_init
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        
        # Adaptive top-k: use more peaks when they're close together
        sorted_c = np.sort(c)[::-1]
        if len(sorted_c) > top_k + 1:
            gap = M - sorted_c[top_k]
            if gap < 0.1 * M:
                top_k = min(top_k + 1, 10)
            elif gap > 0.5 * M and top_k > 2:
                top_k = max(2, top_k - 1)
        
        grad = _subgradient_topk(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            grad = _gradient_smooth_max(a, 0.01 * M)
            gnorm = np.linalg.norm(grad)
        
        if gnorm > 0:
            grad = grad / gnorm
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        
        sc_new = _score(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.002, lr * 0.95)
            if no_imp > max(3000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr * 0.7
                top_k = top_k_init
                no_imp = 0
        
        a = a_new
    
    return best_a, best_score


def _logit_adam_optimize(a, deadline, max_iter=30000, lr=0.04):

    best_z = z.copy()

            best_z = z.copy()

def _iterated_local_search(a, deadline, max_rounds=10):
    """Perturb + re-optimize to escape local minima."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**32))
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Generate perturbation
        noise_scale = 0.05 + 0.05 * (rnd / max(1, max_rounds))
        
        # Mix of perturbations
        choice = rnd % 4
        if choice == 0:
            # Gaussian noise
            pert = best_a * (1.0 + noise_scale * rng.normal(size=n))
            pert = np.maximum(pert, 0.0)
        elif choice == 1:
            # Move mass randomly
            pert = best_a.copy()
            for _ in range(n // 16):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                amt = noise_scale * best_a[i] * rng.random()
                pert[i] -= amt
                pert[j] += amt
            pert = np.maximum(pert, 0.0)
        elif choice == 2:
            # Smoothing (blur)
            kernel_size = max(3, n // 32)
            kernel = np.ones(kernel_size) / kernel_size
            pert = np.convolve(best_a, kernel, mode='same')
            pert = np.maximum(pert, 0.0)
        else:
            # Endpoint emphasis
            t = np.linspace(0, 1, n)
            weight = 1.0 + noise_scale * (2*t - 1)**2
            pert = best_a * weight
            pert = np.maximum(pert, 0.0)
        
        s = pert.sum()
        if s < 1e-10:
            continue
        pert = pert / s
        
        sc_pert = _score(pert)
        if sc_pert < best_score - 1e-8:
            best_a = pert.copy()
            best_score = sc_pert
        
        # Quick local optimization from perturbation
        sub_deadline = min(time.monotonic() + 15.0, deadline - 30.0)
        if sub_deadline > time.monotonic() + 5:
            a_opt, sc_opt = _subgradient_optimize(
                pert, sub_deadline, max_iter=2000, lr_init=0.03, top_k_init=3)
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
    
    return best_a, best_score

    best_score = _score(a)
    
    # Stage 1: Subgradient (global-ish)
    sg_deadline = min(time.monotonic() + 40.0, deadline - 60.0)
    if sg_deadline > time.monotonic() + 5:
        a_sg, sc_sg = _subgradient_optimize(a, sg_deadline, max_iter=10000,
                                            lr_init=0.06, top_k_init=5)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
        print(f"      Subgrad: {sc_sg:.10f}")
    
    # Stage 2: LP (if applicable)
    if use_lp and n <= 1500:
        lp_deadline = min(time.monotonic() + 30.0, deadline - 45.0)
        if lp_deadline > time.monotonic() + 5:
            a_lp = _lp_refine(best_a, lp_deadline, max_iter=5000,
                             inflate=1.001, blend=0.25)
            sc_lp = _score(a_lp)
            if sc_lp < best_score:
                best_score = sc_lp
                best_a = a_lp.copy()
            print(f"      LP: {sc_lp:.10f}")
    
    # Stage 3: Adam polish
    ad_deadline = min(time.monotonic() + 30.0, deadline - 30.0)
    if ad_deadline > time.monotonic() + 5:
        a_ad, sc_ad = _logit_adam_optimize(best_a, ad_deadline, max_iter=10000, lr=0.03)
        if sc_ad < best_score:
            best_score = sc_ad
            best_a = a_ad.copy()
        print(f"      Adam: {sc_ad:.10f}")
    
    # Stage 4: Subgradient again (finer)
    sg2_deadline = min(time.monotonic() + 20.0, deadline - 15.0)
    if sg2_deadline > time.monotonic() + 5:
        a_sg2, sc_sg2 = _subgradient_optimize(best_a, sg2_deadline, max_iter=5000,
                                              lr_init=0.02, top_k_init=3)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_a = a_sg2.copy()
        print(f"      Subgrad2: {sc_sg2:.10f}")
    
    # Stage 5: Iterated local search
    ils_deadline = min(time.monotonic() + 30.0, deadline - 10.0)
    if ils_deadline > time.monotonic() + 10:
        a_ils, sc_ils = _iterated_local_search(best_a, ils_deadline, max_rounds=5)
        if sc_ils < best_score:
            best_score = sc_ils
            best_a = a_ils.copy()
        print(f"      ILS: {sc_ils:.10f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale subgradient optimization with adaptive peak focusing,
    iterated local search, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    
    best_a = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = float(p.sum())
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    if best_a is None:
        best_a = _make_u_profile(768, 1.5)
        best_score = _score(best_a)
        print(f"Default init: score={best_score:.10f}")
    
    # Determine lengths to try
    if parent is not None:
        parent_n = parent.size
    else:
        parent_n = 768
    
    # Try parent length and nearby lengths
    lengths_to_try = []
    for ln in [parent_n, 640, 896, 1024, 1280, 1536, 2048]:
        if ln not in lengths_to_try and 128 <= ln <= 2500:
            lengths_to_try.append(ln)
    
    print(f"Lengths to try: {lengths_to_try}")
    print(f"Time budget: {budget_s}s, remaining: {deadline - time.monotonic():.0f}s")
    
    # Phase 1: Quick evaluation of initializations at each length
    # Allocate time: 40% to Phase 1, 40% to Phase 2 (refinement), 20% to Phase 3 (final polish)
    phase1_end = t_start + (budget_s - 30) * 0.35
    phase2_end = t_start + (budget_s - 30) * 0.75
    
    # Phase 1: Explore multiple lengths with quick optimization
    print("\n=== Phase 1: Multi-length exploration ===")
    
    # Generate initializations
    def get_inits(n):
        inits = []
        if parent is not None:
            inits.append(_interpolate_to_length(parent, n))
        inits.append(_make_u_profile(n, 1.5))
        inits.append(_make_u_profile(n, 2.5))
        inits.append(_make_v_profile(n, 1.0))
        inits.append(_make_gaussian_mixture(n, [0.15, 0.85], [0.15, 0.15], [1, 1]))
        inits.append(_make_gaussian_mixture(n, [0.1, 0.9], [0.12, 0.12], [1, 1]))
        # Random structured
        rng = np.random.default_rng(int(seed) + n)
        for _ in range(2):
            a_r = np.maximum(1.0 + 0.5 * rng.normal(size=n), 0.0)
            inits.append(a_r / a_r.sum())
        return inits
    
    # For each length, try initializations with quick optimization
    for n_t in lengths_to_try:
        if time.monotonic() > phase1_end - 60:
            break
        
        inits = get_inits(n_t)
        use_lp = n_t <= 1000
        
        # Quick score all inits
        best_init = None
        best_init_score = np.inf
        for a0 in inits:
            sc0 = _score(a0)
            if sc0 < best_init_score:
                best_init_score = sc0
                best_init = a0
        
        if best_init is None:
            continue
        
        print(f"  n={n_t}: best init score={best_init_score:.10f}")
        
        # Run pipeline on best init (time-limited)
        sub_deadline = min(time.monotonic() + 25.0, phase1_end - 30.0)
        if sub_deadline > time.monotonic() + 10:
            a_res, sc_res = _full_pipeline(best_init, sub_deadline, use_lp=use_lp)
            if sc_res < best_score:
                best_score = sc_res
                best_a = a_res.copy()
                print(f"  ** New global best at n={n_t}: {sc_res:.10f}")
    
    # Phase 2: Intensive refinement on best candidates
    print(f"\n=== Phase 2: Intensive refinement (score={best_score:.10f}) ===")
    
    n_best = best_a.size
    
    # Run full pipeline with more time
    phase2_sub = min(time.monotonic() + 80.0, phase2_end - 20.0)
    a_res, sc_res = _full_pipeline(best_a, phase2_sub, use_lp=(n_best <= 1200))
    if sc_res < best_score:
        best_score = sc_res
        best_a = a_res.copy()
        print(f"  After pipeline: {sc_res:.10f}")
    
    # Try diverse restarts at the best length
    if time.monotonic() < phase2_end - 30:
        print(f"  Trying diverse restarts at n={n_best}...")
        rng = np.random.default_rng(int(seed) + 999)
        
        restarts = [
            _make_u_profile(n_best, 1.0),
            _make_u_profile(n_best, 2.0),
            _make_u_profile(n_best, 3.5),
            _make_v_profile(n_best, 0.8),
            _make_gaussian_mixture(n_best, [0.08, 0.92], [0.1, 0.1], [1, 1]),
            _make_gaussian_mixture(n_best, [0.2, 0.8], [0.18, 0.18], [1, 1]),
        ]
        
        # Perturbed best
        for _ in range(3):
            pert = best_a * (1.0 + 0.1 * rng.normal(size=n_best))
            pert = np.maximum(pert, 0.0)
            restarts.append(pert / pert.sum())
        
        # Random
        for _ in range(2):
            a_r = np.maximum(1.0 + 0.3 * rng.normal(size=n_best), 0.0)
            restarts.append(a_r / a_r.sum())
        
        for a0 in restarts:
            if time.monotonic() > phase2_end - 20:
                break
            sc0 = _score(a0)
            if sc0 > best_score + 0.01:
                continue  # Skip clearly worse inits
            
            sub_d = min(time.monotonic() + 20.0, phase2_end - 15.0)
            if sub_d > time.monotonic() + 8:
                a_r, sc_r = _full_pipeline(a0, sub_d, use_lp=(n_best <= 1000))
                if sc_r < best_score:
                    best_score = sc_r
                    best_a = a_r.copy()
                    print(f"    ** New best from restart: {sc_r:.10f}")
    
    # Phase 3: Final intensive polish
    print(f"\n=== Phase 3: Final polish (score={best_score:.10f}) ===")
    
    final_deadline = deadline - 10.0
    rem = final_deadline - time.monotonic()
    print(f"  Remaining time: {rem:.0f}s")
    
    # 3a: Extended subgradient
    sg_d = min(time.monotonic() + rem * 0.25, final_deadline - 60.0)
    if sg_d > time.monotonic() + 10:
        a_sg, sc_sg = _subgradient_optimize(best_a, sg_d, max_iter=50000,
                                            lr_init=0.015, top_k_init=5)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
        print(f"  Subgrad final: {sc_sg:.10f}")
    
    # 3b: Extended Adam
    ad_d = min(time.monotonic() + rem * 0.25, final_deadline - 50.0)
    if ad_d > time.monotonic() + 10:
        a_ad, sc_ad = _logit_adam_optimize(best_a, ad_d, max_iter=30000, lr=0.02)
        if sc_ad < best_score:
            best_score = sc_ad
            best_a = a_ad.copy()
        print(f"  Adam final: {sc_ad:.10f}")
    
    # 3c: LP (if applicable)
    if n_best <= 1200:
        lp_d = min(time.monotonic() + rem * 0.2, final_deadline - 30.0)
        if lp_d > time.monotonic() + 10:
            a_lp = _lp_refine(best_a, lp_d, max_iter=20000,
                             inflate=1.0005, blend=0.15)
            sc_lp = _score(a_lp)
            if sc_lp < best_score:
                best_score = sc_lp
                best_a = a_lp.copy()
            print(f"  LP final: {sc_lp:.10f}")
    
    # 3d: ILS
    ils_d = min(time.monotonic() + rem * 0.15, final_deadline - 15.0)
    if ils_d > time.monotonic() + 10:
        a_ils, sc_ils = _iterated_local_search(best_a, ils_d, max_rounds=8)
        if sc_ils < best_score:
            best_score = sc_ils
            best_a = a_ils.copy()
        print(f"  ILS final: {sc_ils:.10f}")
    
    # 3e: One more Adam pass
    ad2_d = final_deadline - 5.0
    if ad2_d > time.monotonic() + 5:
        a_ad2, sc_ad2 = _logit_adam_optimize(best_a, ad2_d, max_iter=20000, lr=0.01)
        if sc_ad2 < best_score:
            best_score = sc_ad2
            best_a = a_ad2.copy()
        print(f"  Adam2 final: {sc_ad2:.10f}")
    
    # Final validation
    if best_a is None:
        best_a = _make_u_profile(768, 1.5)
        best_score = _score(best_a)
    
    final_seq = np.clip(best_a, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s * s  # Keep the sum (evaluate_sequence normalizes)
    else:
        final_seq = _make_u_profile(768, 1.5)
    
    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception:
        pass
    
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
