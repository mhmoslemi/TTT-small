def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    idx = rho[-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

def _smooth_max_and_weights(c, beta):
    """Compute smooth max and softmax weights."""
    m = float(c.max())
    x = (c - m) / beta
    # Clip for numerical stability
    x = np.clip(x, -50, 50)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = ex / s
    smax = m + beta * float(np.log(s))
    return smax, w

def _grad_from_weights(w, a):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a, given weights w."""
    n = len(a)
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # This is 2 * correlate(w, a, 'valid')
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g

def _subgradient_optimize(a0, deadline, max_iter=50000):
    """Subgradient descent with log-sum-exp smooth approximation."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(c.max())  # we minimize this (score = 2n * this since sum=1)
    
    beta = 0.05 * float(c.max())
    base_lr = 0.02
    lr = base_lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    imp_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        smax, w = _smooth_max_and_weights(c, beta)

        grad = _grad_from_weights(w, a)

        v = momentum * v - lr * grad

        a_new = _project_simplex(a_new)

        score_new = float(c_new.max())

            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            imp_count += 1
        else:
            a = a_new  # still move (exploration)

                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

                a = a_pert
                no_imp = 0

def _lp_refine(a0, deadline, max_iter=20000, inflate_start=1.003, inflate_min=1.0003):
    """LP-based refinement."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(c.max())
    
    no_imp = 0
    inflate = inflate_start
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        
        # Solve LP: max sum(b) s.t. conv(a, b) <= M_lp, b >= 0
        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)
        
        try:
            from scipy.optimize import linprog
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')
        except:
            break
        
        if not res.success:
            break
        
        g = res.x
        sum_g = float(g.sum())
        
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate = min(0.02, inflate * 1.2)
            if no_imp > 200:
                break
            continue
        
        g = g / sum_g
        
        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            a_try = _project_simplex(a_try)
            c_try = _fft_convolve(a_try)
            sc_try = float(c_try.max())
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t

            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            inflate = max(inflate_min, inflate * 0.95)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.02, inflate * 1.1)
            if no_imp > max(100, n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0
    
    return best_a

    a = a / s

    c = _fft_convolve(a)
    best_score = float(c.max())

    beta = 0.08 * float(c.max())
    lr = base_lr
    momentum = 0.8
    v = np.zeros(n)
    no_imp = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        _, w = _smooth_max_weights(c, beta)
        grad = _grad_from_weights(w, a)

        # Momentum step
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = _project_simplex(a_new)

        c_new = _fft_convolve(a_new)
        score_new = float(c_new.max())

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            a = a_new
            no_imp += 1
            if no_imp % 150 == 0:
                lr = max(0.003, lr * 0.93)
            if no_imp > max(3000, 4 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
                beta = max(1e-7, 0.08 * float(_fft_convolve(a).max()))

        if it % 300 == 299:
            beta = max(1e-8, beta * 0.82)

        # Random perturbation every 400 iterations
        if it % 400 == 399:
            n_pert = max(3, n // 30)
            idx = rng.integers(0, n, size=n_pert)
            a_pert = a.copy()
            a_pert[idx] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = _project_simplex(a_pert)
            c_pert = _fft_convolve(a_pert)
            if c_pert.max() < best_score:
                best_score = float(c_pert.max())
                best_a = a_pert.copy()
                a = a_pert
                no_imp = 0

    return best_a


def _lp_refine(a0, deadline, max_iter=20000, inflate_start=1.004, inflate_min=1.0004):
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(c.max())

    no_imp = 0
    inflate = inflate_start

    # Pre-build constraint matrix structure (Toeplitz-like)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    c_obj = -np.ones(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate
        b_ub = np.full(2 * n - 1, M_lp)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            break

        g = res.x
        sum_g = float(g.sum())

        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 8 == 0:
                inflate = min(0.03, inflate * 1.15)
            if no_imp > 150:
                break
            continue

        g = g / sum_g

        # Line search with more points
        best_t = 0.0
        best_sc = best_score
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            a_try = _project_simplex(a_try)
            c_try = _fft_convolve(a_try)
            sc_try = float(c_try.max())
            if sc_try < best_sc:
                best_sc = sc_try
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            inflate = max(inflate_min, inflate * 0.93)
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.03, inflate * 1.08)
            if no_imp > max(100, n // 2):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0

    return best_a


def _simulated_annealing(a0, deadline, max_iter=100000, rng=None):
    if rng is None:
        rng = np.random.default_rng(999)
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(c.max())

    T = 0.005
    T_min = 1e-7
    cooldown = 0.9997
    n_pert = max(2, n // 40)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_new = a.copy()
        idx = rng.integers(0, n, size=n_pert)
        a_new[idx] *= (1.0 + T * rng.standard_normal(n_pert))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        c_new = _fft_convolve(a_new)
        score_new = float(c_new.max())
        cur_score = float(_fft_convolve(a).max())

        delta = score_new - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()

        T = max(T_min, T * cooldown)

    return best_a


def _coord_descent(a0, deadline, max_iter=200000, rng=None):
    if rng is None:
        rng = np.random.default_rng(777)
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float(_fft_convolve(a).max())
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        i = rng.integers(0, n)
        base = a.copy()
        best_local = best_score
        best_val = a[i]

        # Try several perturbations of coordinate i
        deltas = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.001, -0.001]
        for delta in deltas:
            a_try = base.copy()
            a_try[i] = max(0.0, a_try[i] + delta)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            c_try = _fft_convolve(a_try)
            sc = float(c_try.max())
            if sc < best_local:
                best_local = sc
                best_val = a_try[i]

        if best_val < a[i] or best_val > a[i]:
            a[i] = best_val
            a = a / a.sum()
            if best_local < best_score - 1e-15:
                best_score = best_local
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 2000:
            break

    return best_a


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-8
    return a / a.sum()


def _make_ramp_rev(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** alpha + 1e-8
    return a / a.sum()


def _make_lin(n, slope=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_quad(n, c1=2.0, c2=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + c1 * t + c2 * t * t
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_exp_decay(n, rate=3.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t) + 1e-8
    return a / a.sum()


def _make_two_hump(n, center=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-20.0 * (t - center) ** 2) + 0.3 * np.exp(-10.0 * (t - 0.7) ** 2) + 1e-8
    return a / a.sum()


def _make_random_smooth(n, rng, sigma_frac=0.08):
    sig = max(1.0, sigma_frac * n)
    a_raw = rng.standard_normal(n)
    # Simple moving average smoothing
    k = max(1, int(3 * sig))
    kernel = np.ones(k) / k
    a_raw = np.convolve(a_raw, kernel, mode='same')
    a = np.maximum(a_raw, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 35.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Generate diverse initializations
    lengths = [512, 768, 1024, 1280, 1536]
    configs = []

    for n_t in lengths:
        # Parent interpolated
        if parent is not None:
            configs.append(('par', n_t, _interpolate(parent, n_t)))
        # Ramp profiles
        configs.append(('r1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('r2', n_t, _make_ramp(n_t, 2.0)))
        configs.append(('r3', n_t, _make_ramp(n_t, 3.0)))
        # Reverse ramps
        configs.append(('rr1', n_t, _make_ramp_rev(n_t, 1.5)))
        configs.append(('rr2', n_t, _make_ramp_rev(n_t, 2.5)))
        # Linear
        configs.append(('l2', n_t, _make_lin(n_t, 2.0)))
        configs.append(('l4', n_t, _make_lin(n_t, 4.0)))
        configs.append(('l8', n_t, _make_lin(n_t, 8.0)))
        # Quadratic
        configs.append(('q', n_t, _make_quad(n_t, 3.0, -1.0)))
        # Exponential decay
        configs.append(('e3', n_t, _make_exp_decay(n_t, 3.0)))
        configs.append(('e5', n_t, _make_exp_decay(n_t, 5.0)))
        # Two hump
        configs.append(('th', n_t, _make_two_hump(n_t, 0.25)))
        # Random smooth
        for rs in range(3):
            configs.append((f'rs{rs}', n_t, _make_random_smooth(n_t, rng, 0.05 + 0.03 * rs)))

    # Score all initializations
    scored = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    print(f"Generated {len(scored)} initializations, best initial: {scored[0][0]:.6f} ({scored[0][1]}, n={scored[0][2]})")

    # Phase 1: Quick subgradient on top initializations
    top_k = min(10, len(scored))
    phase1_deadline = t_start + 600.0

    for i in range(top_k):
        if time.monotonic() > phase1_deadline:
            break
        sc0, name, n_t, a0 = scored[i]
        if sc0 >= best_score + 0.15:
            continue

        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Allocate time: more for better candidates
        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break
        frac = max(0.3, 1.0 - 0.1 * i)
        t_alloc = min(80.0, remaining * frac)
        if t_alloc < 15:
            t_alloc = 15

        end = time.monotonic() + t_alloc
        a_opt = _subgradient_optimize(a0, end, max_iter=30000, base_lr=0.012)
        sc_opt = _score(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  Subgrad {name} n={n_t}: {sc_opt:.10f}")
        else:
            print(f"  Subgrad {name} n={n_t}: {sc_opt:.8f} (no imp)")

    # Phase 2: LP refinement on best 3-5
    phase2_deadline = t_start + 1100.0
    if best_seq is not None and time.monotonic() < phase2_deadline:
        # Take top candidates near the best
        lp_candidates = []
        if best_seq is not None:
            lp_candidates.append(best_seq.copy())
        for i in range(min(5, top_k)):
            sc0, name, n_t, a0 = scored[i]
            if sc0 < best_score + 0.05 and a0.size != best_seq.size:
                lp_candidates.append(a0.copy())

        for j, a_cand in enumerate(lp_candidates[:4]):
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 30:
                break
            t_alloc = min(120.0, remaining * 0.6)
            if t_alloc < 20:
                t_alloc = 20

            n_cand = len(a_cand)
            if n_cand <= 1536:
                end = time.monotonic() + t_alloc
                a_lp = _lp_refine(a_cand, end, max_iter=15000, inflate_start=1.003, inflate_min=1.0003)
                sc_lp = _score(a_lp)
                if np.isfinite(sc_lp) and sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                    print(f"  LP {j} n={n_cand}: {sc_lp:.10f}")
                else:
                    print(f"  LP {j} n={n_cand}: {sc_lp:.8f}")
            else:
                # For large n, use subgradient instead
                end = time.monotonic() + t_alloc
                a_sg = _subgradient_optimize(a_cand, end, max_iter=30000, base_lr=0.008)
                sc_sg = _score(a_sg)
                if np.isfinite(sc_sg) and sc_sg < best_score:
                    best_score = sc_sg
                    best_seq = a_sg.copy()
                    print(f"  Subgrad(large) {j} n={n_cand}: {sc_sg:.10f}")

    # Phase 3: Simulated annealing + multi-start on best
    phase3_deadline = t_start + 1450.0
    if best_seq is not None and time.monotonic() < phase3_deadline:
        n_best = len(best_seq)
        print(f"Phase 3: SA and multi-start, n={n_best}, score={best_score:.10f}")

        # Simulated annealing
        sa_time = min(200.0, phase3_deadline - time.monotonic() - 100)
        if sa_time > 30:
            end = time.monotonic() + sa_time
            a_sa = _simulated_annealing(best_seq, end, max_iter=200000, rng=rng)
            sc_sa = _score(a_sa)
            if np.isfinite(sc_sa) and sc_sa < best_score:
                best_score = sc_sa
                best_seq = a_sa.copy()
                print(f"  SA: {sc_sa:.10f}")
            else:
                print(f"  SA: {sc_sa:.8f}")

        # Multi-start: perturb best and re-optimize
        if time.monotonic() < phase3_deadline - 30:
            for ms in range(3):
                if time.monotonic() > phase3_deadline - 20:
                    break
                a_pert = best_seq.copy()
                n_pert = max(5, n_best // 20)
                idx = rng.integers(0, n_best, size=n_pert)
                a_pert[idx] *= (1.0 + 0.2 * rng.standard_normal(n_pert))
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = _project_simplex(a_pert)

                t_alloc = min(60.0, phase3_deadline - time.monotonic() - 20)
                if t_alloc < 10:
                    break
                end = time.monotonic() + t_alloc
                a_opt = _subgradient_optimize(a_pert, end, max_iter=20000, base_lr=0.01)
                sc_opt = _score(a_opt)
                if np.isfinite(sc_opt) and sc_opt < best_score:
                    best_score = sc_opt
                    best_seq = a_opt.copy()
                    print(f"  Multi-start {ms}: {sc_opt:.10f}")
                else:
                    print(f"  Multi-start {ms}: {sc_opt:.8f}")

    # Phase 4: Final LP + coordinate descent
    phase4_deadline = deadline
    if best_seq is not None and time.monotonic() < phase4_deadline - 10:
        n_best = len(best_seq)

        # Final LP if feasible
        if n_best <= 1280:
            remaining = phase4_deadline - time.monotonic() - 60
            if remaining > 20:
                end = time.monotonic() + min(100.0, remaining)
                a_lp = _lp_refine(best_seq, end, max_iter=10000, inflate_start=1.002, inflate_min=1.0002)
                sc_lp = _score(a_lp)
                if np.isfinite(sc_lp) and sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                    print(f"  Final LP: {sc_lp:.10f}")

        # Coordinate descent
        remaining = phase4_deadline - time.monotonic() - 10
        if remaining > 10:
            end = time.monotonic() + min(80.0, remaining)
            a_cd = _coord_descent(best_seq, end, max_iter=100000, rng=rng)
            sc_cd = _score(a_cd)
            if np.isfinite(sc_cd) and sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
                print(f"  Coord descent: {sc_cd:.10f}")

        # One more subgradient pass
        remaining = phase4_deadline - time.monotonic() - 5
        if remaining > 10:
            end = time.monotonic() + min(30.0, remaining)
            a_sg = _subgradient_optimize(best_seq, end, max_iter=10000, base_lr=0.005)
            sc_sg = _score(a_sg)
            if np.isfinite(sc_sg) and sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"  Final subgrad: {sc_sg:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(768, 1.0)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(256, 1.0)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={elapsed:.1f}s")
    return [float(x) for x in final_seq]
