"""
Hybrid optimization for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
1. LP-guided direction finding (solving a separation LP for peak-balancing directions)
2. Nesterov-accelerated mirror descent in logit space with adaptive subgradients
3. Coordinate descent polish
4. Multi-scale exploration with diverse initializations and structural restarts

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    """Full linear autocorrelation via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _subgradient(c, a, top_k=5, eps_frac=0.01):
    """
    Compute subgradient of max(conv(a,a)).
    Uses top-k peaks with weights proportional to their values.
    Returns gradient vector g of length n.
    """
    n = a.size
    max_c = c.max()
    if max_c < 1e-30:
        return np.zeros(n)

    # Find active peaks (within eps_frac of max)
    threshold = max_c * (1.0 - eps_frac)
    active = np.where(c >= threshold)[0]

    if len(active) > top_k:
        # Take the top_k largest
        sorted_idx = active[np.argsort(c[active])[::-1][:top_k]]
        active = sorted_idx

    # Weights: proportional to peak values (normalized)
    w_vals = c[active]
    w_sum = w_vals.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = np.zeros_like(c)
        w[active] = w_vals / w_sum

    # Compute g[m] = 2 * sum_k w[k] * a[k-m] for 0<=k-m<n
    # This is cross-correlation: g = 2 * (w * a[::-1])[n-1:2n-1]... 
    # Actually: g[m] = 2 * sum_{k} w[k] * a[k-m] where a[j]=0 outside [0,n-1]
    # = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    # This is the cross-correlation of w and a.
    # Using FFT: convolve w with reversed a, then extract
    N = _next_pow2(n + len(w) - 1)
    w_use = w[:2*n-1] if len(w) >= 2*n-1 else w
    W = len(w_use)
    N = _next_pow2(n + W - 1)
    fa = np.fft.rfft(a[:n], N)
    fw = np.fft.rfft(w_use[:W], N)
    # We need: g[m] = sum_{j=0}^{n-1} w_use[m+j] * a[j]
    # = (w_use * a)[n-1+m] in convolution terms? Let's verify.
    # conv(w,a)[k] = sum_i w[i]*a[k-i]
    # We want g[m] = sum_j w[m+j]*a[j] = sum_j w[j+m]*a[j]
    # Let i = j, then g[m] = sum_i w[i+m]*a[i]
    # If we reverse w: wr[i] = w[W-1-i], then
    # conv(wr, a)[k] = sum_i wr[i]*a[k-i] = sum_i w[W-1-i]*a[k-i]
    # Let j = W-1-i, i = W-1-j: = sum_j w[j]*a[k-W+1+j]
    # We want sum_j w[j+m]*a[j]. Set k-W+1+j = j => k = W-1.
    # So conv(wr,a)[W-1] = sum_j w[j]*a[j]. That's just the dot product.
    # 
    # Let me think differently. 
    # g[m] = sum_{j=0}^{n-1} w[m+j] * a[j]
    # This is (w ✶ a)[m] where ✶ is cross-correlation.
    # Cross-correlation: (w ✶ a)[m] = sum_j w[j+m] * a[j]
    # = conv(w, a[::-1])[W-1 + m]... 
    # 
    # Actually, cross-correlation via FFT:
    # (w ✶ a)[m] = sum_j w[j+m] * conj(a[j]) = sum_j w[j+m] * a[j] (real)
    # = IFFT(FFT(w) * conj(FFT(a))) ... but this is circular.
    # 
    # For linear cross-correlation:
    # Pad w and a to length N, compute IFFT(FFT(w) * conj(FFT(a)))
    # The linear result is in indices [0, N-W-n+2] or similar.
    # 
    # Let me just use a simpler approach:
    # g[m] = sum_{j=0}^{n-1} w[m+j] * a[j]
    # Create a padded version and use the fact that this is a "diagonal sum".
    # 
    # Alternative: g = 2 * h where h[m] = sum_j a[j]*w[m+j]
    # h is the output of a Toeplitz system. Let's just compute it directly for small n
    # or use the convolution trick.
    
    # h[m] = sum_{j=0}^{n-1} a[j] * w_use[m+j] for m = 0,...,n-1
    # This equals (a * w_use_reversed)[n-1+m] where w_use_reversed = w_use[::-1]
    # Check: (a * wr)[k] = sum_i a[i] * wr[k-i] = sum_i a[i] * w_use[W-1-(k-i)]
    # = sum_i a[i] * w_use[W-1-k+i]
    # Set W-1-k+i = m+j => we want the coefficient of w_use[m+j] to be a[j]
    # W-1-k+j = m+j => W-1-k = m => k = W-1-m
    # So (a * wr)[W-1-m] = sum_j a[j]*w_use[m+j] + (other terms where index out of range)
    # For proper zero-padding, the "other terms" vanish.
    
    wr = w_use[::-1]
    N2 = _next_pow2(n + W - 1)
    fa2 = np.fft.rfft(a[:n], N2)
    fwr = np.fft.rfft(wr[:W], N2)
    conv_result = np.fft.irfft(fa2 * fwr, N2)
    # h[m] = conv_result[W-1-m] for m=0,...,n-1
    # So h = conv_result[W-1:W-1-n:-1] (reversed slice)
    h = conv_result[W - 1: W - 1 - n: -1]
    # Verify length
    if h.size != n:
        # Fallback: direct computation (slow but correct)
        h = np.zeros(n)
        for m in range(n):
            h[m] = np.dot(a, w_use[m:m+n])
    g = 2.0 * h
    return g


def _soft_max_subgradient(c, a, beta):
    """Compute smoothed subgradient using soft-max weighting."""
    n = a.size
    max_c = c.max()
    if beta < 1e-14:
        # Use exact argmax
        idx = int(np.argmax(c))
        w = np.zeros(2*n-1)
        w[idx] = 1.0
    else:
        x = (c - max_c) / beta
        x = np.clip(x, -50.0, 0.0)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-300:
            w = np.zeros(2*n-1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s

    # Compute h[m] = sum_j a[j]*w[m+j] via FFT
    W = 2 * n - 1
    wr = w[:W][::-1]
    N2 = _next_pow2(n + W - 1)
    fa2 = np.fft.rfft(a[:n], N2)
    fwr = np.fft.rfft(wr[:W], N2)
    conv_result = np.fft.irfft(fa2 * fwr, N2)
    h = conv_result[W - 1: W - 1 - n: -1]
    if h.size != n:
        h = np.zeros(n)
        for m in range(n):
            h[m] = np.dot(a, w[m:m+n])
    return 2.0 * h


def _lp_direction(f, M, n_max_vars=2048):
    """
    Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Returns optimal g (length n) or None if infeasible/too large.
    """
    n = len(f)
    if n > n_max_vars:
        return None

    # Build constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n else 0
    # A is (2n-1) x n
    # For efficiency, build it using the structure
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k,j] = f[k-j] for k=j,...,j+n-1 (i.e., k-j in [0,n-1])
        A[j:j+n, j] = f

    # LP: max 1^T g s.t. A g <= M, g >= 0
    # linprog minimizes, so: min -1^T g s.t. A g <= M, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs')
    if res.status != 0:
        return None
    g = res.x
    return g


def _nesterov_mirror_descent(a0, deadline, rng, lr=0.05, momentum=0.9,
                              beta_init=0.05, beta_decay=0.95, beta_min=1e-8,
                              top_k=5, eps_frac=0.02, restart_period=500):
    """
    Nesterov-accelerated mirror descent in logit space.
    Uses adaptive subgradients (exact for well-separated peaks, soft for close peaks).
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = z - z.mean()

    c = _fft_convolve(a0)
    max_c = float(c.max())

    # Analyze peak structure to set initial beta
    sorted_c = np.sort(c)[::-1]
    if len(sorted_c) > 1:
        gap = sorted_c[0] - sorted_c[min(10, len(sorted_c)-1)]
        if gap < 0.01 * max_c:
            beta = max(beta_init * 0.5, beta_min)
        else:
            beta = max(beta_init * 0.1, beta_min)
    else:
        beta = beta_init

    # Nesterov momentum
    v = np.zeros(n)
    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr

    while time.monotonic() < deadline:
        c = _fft_convolve(a0)
        max_c = float(c.max())

        # Adaptive choice: exact vs soft subgradient
        sorted_c = np.sort(c)[::-1]
        if len(sorted_c) > 1:
            gap = (sorted_c[0] - sorted_c[min(10, len(sorted_c)-1)]) / max(max_c, 1e-30)
            use_soft = gap < 0.05
        else:
            use_soft = False

        if use_soft:
            g = _soft_max_subgradient(c, a0, beta)
        else:
            g = _subgradient(c, a0, top_k=top_k, eps_frac=eps_frac)

        # Mirror descent gradient in logit space
        # a = softmax(z), so da/dz = a (I - a a^T)
        # Gradient w.r.t. z: a * (g - mean(g * a) * ... )
        # Actually: dL/dz = a * (g - sum(a*g) * 1) ... 
        # For mirror descent with KL: z_new = z - LR * g / a ... 
        # Simpler: z_new = z - LR * g (since a = exp(z)/sum(exp(z)), 
        # and the natural gradient is g/a, but we use Euclidean in logit space)
        dot = float(np.dot(a0, g))
        dz = g - dot  # project gradient to tangent space of simplex in logit coords
        # Actually for logit space: the constrained gradient is a*(g - a·g * 1/a) = g - a·g * 1
        # But this isn't quite right. Let me use: dz = a * g - mean(a*g) * a
        # which is the projection of g onto the tangent space.
        dz = a0 * (g - dot)

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * dz
        step = v

        # Adaptive step size: normalize
        step_norm = np.linalg.norm(step)
        if step_norm > 1e-30:
            step = step * (current_lr / step_norm)

        z = z + step
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            v.fill(0)
            continue
        a0 = a0 / s

        c = _fft_convolve(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt beta
        if no_improve % 30 == 0:
            beta = max(beta_min, beta * beta_decay)

        # Adapt learning rate
        if no_improve > 200:
            current_lr = max(0.005, current_lr * 0.98)

        # Restart
        if no_improve > restart_period:
            noise = 0.15 * rng.normal(size=n)
            z = best_z + noise
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            v.fill(0)
            current_lr = lr
            beta = max(beta_init * 0.05, beta_min)
            no_improve = 0

    return best_a, best_score


def _coordinate_descent(a0, deadline, rng, n_iter_per_coord=3, batch_size=64):
    """
    Coordinate descent refinement. For each coordinate, try small perturbations
    and keep the best.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = a0 / a0.sum()
    c = _fft_convolve(a0)
    best_score = float(2.0 * n * c.max())
    best_a = a0.copy()

    # Precompute convolution for fast updates
    # When we change a[m] by delta and a[m'] by -delta (to maintain sum),
    # the change in (a*a)[k] is:
    # delta * (a[k-m'] + a'[k-m]) + delta^2 * delta(k-m-m')... 
    # This is complex. Let's just recompute for small batches.

    iter_count = 0
    while time.monotonic() < deadline:
        # Pick a batch of random coordinate pairs
        indices = rng.integers(0, n, size=batch_size)
        for idx in indices:
            if time.monotonic() > deadline:
                break
            # Try moving mass from idx to a random other position
            other = rng.integers(0, n)
            if other == idx:
                continue
            # Try several step sizes
            a_trial = a0.copy()
            for frac in [0.01, 0.02, 0.05, 0.01, 0.02]:
                delta = a0[idx] * frac
                if delta < 1e-15:
                    break
                a_trial = a0.copy()
                a_trial[idx] -= delta
                a_trial[other] += delta
                if a_trial[idx] < 0:
                    a_trial[idx] = 0
                    a_trial[other] += a0[idx]
                    delta = a0[idx]
                c_trial = _fft_convolve(a_trial)
                sc = float(c_trial.max())
                if sc < best_score * (1 - 1e-10):
                    a0 = a_trial.copy()
                    best_score = sc
                    best_a = a0.copy()
                    break

        iter_count += 1
        if iter_count % 100 == 0:
            print(f"    CD iter {iter_count}: score={best_score:.10f}")

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    """Resample profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def _diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Linear ramps
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Cosine modes
    for k in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # Superposition of cosines
    for _ in range(8):
        a = np.ones(n)
        for k in range(1, 5):
            a += rng.uniform(-0.5, 0.5) * np.cos(2.0 * np.pi * k * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Exponential decay / growth
    for rate in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(-rate * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        a = np.exp(rate * (t - 1.0))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Bump functions
    for width in [0.2, 0.3, 0.5, 0.7]:
        for center in [0.3, 0.5, 0.7]:
            a = np.exp(-((t - center) / width) ** 2)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())

    # U-shaped (high at edges)
    for p in [1.0, 2.0, 3.0]:
        a = (t - 0.5)**p + 0.01
        profiles.append(a / a.sum())

    # Random smooth (Poisson process smoothed)
    for _ in range(4):
        a = np.abs(rng.normal(size=n))
        # Smooth with moving average
        k = max(3, n // 20)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def _lp_iteration(f, n_iter=5, t_start=0.01, t_end=0.001):
    """
    Perform several LP-guided iterations.
    f is normalized so sum=1.
    Returns improved f.
    """
    f = f.copy()
    n = f.size
    c = _fft_convolve(f)
    M = float(c.max())
    best_score = M
    best_f = f.copy()

    for i in range(n_iter):
        g = _lp_direction(f, M)
        if g is None:
            break
        s_g = g.sum()
        if s_g < 1e-10:
            break
        g = g / s_g  # normalize to sum=1

        # Try several step sizes
        t = t_start * (t_end / t_start) ** (i / max(n_iter - 1, 1))
        f_new = (1 - t) * f + t * g
        c_new = _fft_convolve(f_new)
        M_new = float(c_new.max())

        if M_new < best_score:
            best_score = M_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            t_start = t * 0.8  # shrink for next
        else:
            # Try smaller t
            for t_try in [t * 0.5, t * 0.25, t * 0.1]:
                f_try = (1 - t_try) * f + t_try * g
                c_try = _fft_convolve(f_try)
                if float(c_try.max()) < best_score:
                    best_score = float(c_try.max())
                    best_f = f_try.copy()
                    f = f_try
                    M = best_score
                    break

    return best_f, best_score


def _structural_restart(a0, rng, mode):
    """Apply a structural transformation to escape local minima."""
    n = len(a0)
    t = np.linspace(0, 1, n)
    if mode == 0:
        # Multiply by cosine
        a = a0 * (1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(1, 5) * t))
    elif mode == 1:
        # Multiply by ramp
        p = rng.uniform(0.5, 3.0)
        a = a0 * ((1.0 - t) ** p + 0.1)
    elif mode == 2:
        # Flip and blend
        a = a0 * (0.7 + 0.3 * np.linspace(0, 1, n))
    elif mode == 3:
        # Gaussian perturbation
        a = a0 + 0.1 * rng.normal(size=n)
    elif mode == 4:
        # Scale different halves differently
        half = n // 2
        scale = rng.uniform(0.5, 2.0)
        a = a0.copy()
        a[:half] *= scale
    else:
        a = a0 + 0.05 * rng.normal(size=n)

    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid search: LP-guided + Nesterov mirror descent + coordinate descent.
    Explores multiple scales with diverse initializations and structural restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Try to load initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting hybrid search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = _compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Quick exploration at n=512 =====
    n1 = 512
    phase1_deadline = min(deadline, t_start + 200.0)
    print(f"\n--- Phase 1: n={n1} (exploration, {int(phase1_deadline - t_start)}s) ---")

    profiles_1 = _diverse_profiles(n1, rng)
    if init_seq is not None:
        profiles_1.insert(0, _interpolate_profile(init_seq, n1))

    for idx, p in enumerate(profiles_1[:12]):
        if time.monotonic() > phase1_deadline - 5:
            break
        dl = min(phase1_deadline, time.monotonic() + 15.0)
        result, sc = _nesterov_mirror_descent(p, dl, rng, lr=0.08,
                                               momentum=0.85, beta_init=0.1)
        sc_final = _compute_score(result)
        update(result, sc_final)
        if idx < 5 or sc_final < best_score + 0.01:
            print(f"  Profile {idx}: score={sc_final:.8f}")

    print(f"  Phase 1 best: {best_score:.10f}")

    # ===== Phase 2: LP-guided optimization at n=512 =====
    if best_seq is not None and best_seq.size <= 1024:
        p2 = best_seq if best_seq.size == n1 else _interpolate_profile(best_seq, n1)
        phase2_deadline = min(deadline, t_start + 350.0)
        if time.monotonic() < phase2_deadline - 10:
            print(f"\n--- Phase 2: LP-guided at n={n1} ---")
            p2 = p2 / p2.sum()
            p2_lp, sc_lp = _lp_iteration(p2, n_iter=8, t_start=0.1, t_end=0.01)
            sc_lp_full = _compute_score(p2_lp)
            update(p2_lp, sc_lp_full)
            print(f"  After LP: {sc_lp_full:.10f}")

            # Follow up with mirror descent
            dl2 = min(phase2_deadline, time.monotonic() + 100.0)
            if time.monotonic() < phase2_deadline - 5:
                result2, sc2 = _nesterov_mirror_descent(p2_lp, dl2, rng, lr=0.04,
                                                         momentum=0.9, beta_init=0.03)
                sc2_f = _compute_score(result2)
                update(result2, sc2_f)
                print(f"  After MD refinement: {sc2_f:.10f}")

    # ===== Phase 3: Scale up to n=1024 =====
    n3 = 1024
    phase3_deadline = min(deadline, t_start + 800.0)
    print(f"\n--- Phase 3: n={n3} (main optimization) ---")

    if best_seq is not None:
        p3 = _interpolate_profile(best_seq, n3)
    else:
        p3 = _diverse_profiles(n3, rng)[3]

    sc3_init = _compute_score(p3)
    update(p3, sc3_init)
    print(f"  Interpolated to n={n3}: {sc3_init:.10f}")

    # Main optimization round 1
    dl3a = min(phase3_deadline, time.monotonic() + 250.0)
    result3, sc3 = _nesterov_mirror_descent(p3, dl3a, rng, lr=0.04,
                                             momentum=0.9, beta_init=0.04,
                                             top_k=5, eps_frac=0.02)
    sc3_f = _compute_score(result3)
    update(result3, sc3_f)
    print(f"  After MD round 1: {sc3_f:.10f}")

    # LP-guided at n=1024
    if time.monotonic() < phase3_deadline - 60:
        print(f"  Running LP at n={n3}...")
        p3_lp, sc_lp3 = _lp_iteration(result3, n_iter=5, t_start=0.05, t_end=0.005)
        sc_lp3_f = _compute_score(p3_lp)
        update(p3_lp, sc_lp3_f)
        print(f"  After LP: {sc_lp3_f:.10f}")
        if sc_lp3_f < sc3_f:
            result3 = p3_lp

    # Main optimization round 2
    if time.monotonic() < phase3_deadline - 30:
        dl3b = min(phase3_deadline, time.monotonic() + 200.0)
        result3b, sc3b = _nesterov_mirror_descent(result3, dl3b, rng, lr=0.02,
                                                   momentum=0.92, beta_init=0.02,
                                                   top_k=3, eps_frac=0.01)
        sc3b_f = _compute_score(result3b)
        update(result3b, sc3b_f)
        print(f"  After MD round 2: {sc3b_f:.10f}")
        if sc3b_f < sc3_f:
            result3 = result3b

    print(f"  Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Scale to n=2048 =====
    n4 = 2048
    phase4_deadline = min(deadline, t_start + 1400.0)
    print(f"\n--- Phase 4: n={n4} (large scale) ---")

    if best_seq is not None:
        p4 = _interpolate_profile(best_seq, n4)
    else:
        p4 = _interpolate_profile(result3, n4)

    sc4_init = _compute_score(p4)
    update(p4, sc4_init)
    print(f"  Interpolated to n={n4}: {sc4_init:.10f}")

    dl4a = min(phase4_deadline, time.monotonic() + 300.0)
    result4, sc4 = _nesterov_mirror_descent(p4, dl4a, rng, lr=0.025,
                                             momentum=0.91, beta_init=0.03,
                                             top_k=5, eps_frac=0.015)
    sc4_f = _compute_score(result4)
    update(result4, sc4_f)
    print(f"  After MD: {sc4_f:.10f}")

    # LP at n=2048 (might be slow, try with limited iterations)
    if time.monotonic() < phase4_deadline - 120 and sc4_f > 1.503:
        print(f"  Running LP at n={n4}...")
        t_lp_start = time.monotonic()
        p4_lp, sc_lp4 = _lp_iteration(result4, n_iter=3, t_start=0.03, t_end=0.003)
        sc_lp4_f = _compute_score(p4_lp)
        update(p4_lp, sc_lp4_f)
        print(f"  After LP ({time.monotonic()-t_lp_start:.1f}s): {sc_lp4_f:.10f}")
        if sc_lp4_f < sc4_f:
            result4 = p4_lp

    # Another MD round
    if time.monotonic() < phase4_deadline - 30:
        dl4b = min(phase4_deadline, time.monotonic() + 150.0)
        result4b, sc4b = _nesterov_mirror_descent(result4, dl4b, rng, lr=0.015,
                                                   momentum=0.93, beta_init=0.015,
                                                   top_k=3, eps_frac=0.005)
        sc4b_f = _compute_score(result4b)
        update(result4b, sc4b_f)
        print(f"  After MD round 2: {sc4b_f:.10f}")

    print(f"  Phase 4 best: {best_score:.10f}")

    # ===== Phase 5: Try n=3072 or n=4096 =====
    if best_score > 1.503 and time.monotonic() < deadline - 200:
        n5 = 3072
        print(f"\n--- Phase 5: n={n5} ---")
        if best_seq is not None:
            p5 = _interpolate_profile(best_seq, n5)
        else:
            p5 = _interpolate_profile(result4, n5)
        sc5_init = _compute_score(p5)
        update(p5, sc5_init)
        print(f"  Interpolated: {sc5_init:.10f}")
        dl5 = min(deadline - 200, time.monotonic() + 250.0)
        result5, sc5 = _nesterov_mirror_descent(p5, dl5, rng, lr=0.02,
                                                 momentum=0.92, beta_init=0.025)
        sc5_f = _compute_score(result5)
        update(result5, sc5_f)
        print(f"  After MD: {sc5_f:.10f}")

    # ===== Phase 6: Coordinate descent polish =====
    print("\n--- Phase 6: Coordinate descent polish ---")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_cd = best_seq.size
        # Downsample for CD if too large (CD is slow)
        if n_cd > 2048:
            p_cd = _interpolate_profile(best_seq, 1024)
        else:
            p_cd = best_seq.copy()
        dl_cd = min(deadline - 10, time.monotonic() + 80.0)
        result_cd, sc_cd = _coordinate_descent(p_cd, dl_cd, rng)
        sc_cd_f = _compute_score(result_cd)
        update(result_cd, sc_cd_f)
        print(f"  After CD (n={result_cd.size}): {sc_cd_f:.10f}")

        # If CD helped at smaller n, try upscaling
        if sc_cd_f < best_score and result_cd.size < n_cd:
            p_up = _interpolate_profile(result_cd, n_cd)
            sc_up = _compute_score(p_up)
            update(p_up, sc_up)
            if time.monotonic() < deadline - 20:
                dl_up = min(deadline - 10, time.monotonic() + 60.0)
                r_up, _ = _nesterov_mirror_descent(p_up, dl_up, rng, lr=0.01,
                                                    momentum=0.93, beta_init=0.01)
                sc_up2 = _compute_score(r_up)
                update(r_up, sc_up2)
                print(f"  After upscale + MD: {sc_up2:.10f}")

    # ===== Phase 7: Structural restarts around best =====
    if best_seq is not None and time.monotonic() < deadline - 30:
        print("\n--- Phase 7: Structural restarts ---")
        for mode in range(5):
            if time.monotonic() > deadline - 15:
                break
            a_restart = _structural_restart(best_seq, rng, mode)
            dl_r = min(deadline - 10, time.monotonic() + 40.0)
            r, sc_r = _nesterov_mirror_descent(a_restart, dl_r, rng, lr=0.03,
                                                momentum=0.9, beta_init=0.05)
            sc_r_f = _compute_score(r)
            update(r, sc_r_f)
            if sc_r_f < best_score + 0.01:
                print(f"  Restart mode {mode}: {sc_r_f:.10f}")

    # ===== Phase 8: Final tiny perturbations =====
    if best_seq is not None and time.monotonic() < deadline - 5:
        print("\n--- Phase 8: Final perturbations ---")
        for _ in range(5):
            if time.monotonic() > deadline - 5:
                break
            a_pert = best_seq + 0.001 * rng.normal(size=best_seq.size)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            sc_pert = _compute_score(a_pert)
            update(a_pert, sc_pert)

    # ===== Final validation =====
    print(f"\n  Final best score: {best_score:.10f}")

    if best_seq is not None:
        final_seq = best_seq
    else:
        # Fallback
        n_fb = 512
        t = np.linspace(0, 1, n_fb)
        final_seq = 1.0 + 1.5 * (0.5 - t)
        final_seq = np.maximum(final_seq, 0.0)
        final_seq = final_seq / final_seq.sum()

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(256) / 256.0
    else:
        # Keep original scale (don't normalize) since the score is scale-invariant
        # But ensure sum > 0.01
        final_seq = final_seq / s

    final_score = _compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
