```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.argmax((u - (cv - 1) / np.arange(1, n + 1)) > 0)
    theta = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
                continue
            qat_flip = np.correlate(q, cand, mode='valid')[:n]
            obj_c = float(cand @ qat_flip)
            if obj_c <= obj + 1e-14:
                a_new = cand
                obj_new = obj_c
                break
            step *= 0.5
        a = a_new
        obj = obj_new
        qat = np.correlate(q, a, mode='valid')[:n]
        lr = max(step, 1e-12)
        if abs(prev_obj - obj) < 1e-15:
            break
        prev_obj = obj
    score = _score_norm(a)
    return a, score


def _minimax_iterate(a, deadline, max_rounds=200):
    best = a.copy()
    best_sc = _score_norm(a)
    no_imp = 0
    for rd in range(max_rounds):
        if time.monotonic() > deadline - 3:
            break
        a_new, sc_new = _qp_minimax_step(a, n_iter=200, lr_init=0.05)
        if sc_new < best_sc - 1e-12:
            best = a_new.copy()
            best_sc = sc_new
            no_imp = 0
        else:
            t = 0.3
            a_blend = (1.0 - t) * a + t * a_new
            a_blend = np.maximum(a_blend, 1e-15)
            a_blend = a_blend / a_blend.sum()
            sc_blend = _score_norm(a_blend)
            if sc_blend < best_sc - 1e-12:
                best = a_blend.copy()
                best_sc = sc_blend
                no_imp = 0
            else:
                no_imp += 1
            a = a_blend
        if no_imp > 50:
            break
    return best, best_sc


def _combined_lp_qp(a, deadline, max_rounds=100):
    best = a.copy()
    best_sc = _score_norm(a)
    no_imp = 0
    for rd in range(max_rounds):
        if time.monotonic() > deadline - 3:
            break
        t_opt, b_opt = _lp_solve(a)
        if b_opt is not None:
            a_try = 0.7 * a + 0.3 * b_opt
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc_try = _score_norm(a_try)
            if sc_try < best_sc - 1e-12:
                best = a_try.copy()
                best_sc = sc_try
                no_imp = 0
                a = a_try
            else:
                a = a_try
        a2, sc2 = _qp_minimax_step(a, n_iter=100, lr_init=0.03)
        if sc2 < best_sc - 1e-12:
            best = a2.copy()
            best_sc = sc2
            no_imp = 0
            a = a2
        else:
            no_imp += 1
        if no_imp > 40:
            break
    return best, best_sc


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def _smooth_restart(a, rng, sigma):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z -= z.mean()
    z = z + sigma * rng.normal(size=n)
    z = np.clip(z, -50, 50)
    z -= z.max()
    e = np.exp(z)
    p = e + 1e-12
    return p / p.sum()


def _adam_smooth_max(a0, deadline, rng, lr=0.04, no_imp_limit=200):
    n = len(a0)
    p = np.maximum(a0, 1e-12)
    p = p / p.sum()
    z = np.log(p)
    z -= z.mean()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, 0.08 * maxc)
    best_p = p.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    while time.monotonic() < deadline:
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros(n * 2 - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s
        gw = np.zeros(2 * n - 1)
        for i in range(n):
            for j in range(n):
                gw[i + j] += w[i + j] * p[j]
        vp = np.zeros(n)
        for j in range(n):
            vp[j] = float(np.dot(a0[:0].size > 0 and p or p, gw[:n]))
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= gz.mean()
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z -= z.mean()
        e = np.exp(z - z.max())
        p = (e + 1e-12) / (e + 1e-12).sum()
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.002, lr * 0.9)
        if t_adam % 40 == 0:
            beta = max(1e-4, beta * 0.93)
        if no_imp > no_imp_limit:
            z = best_z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z - z.max())
            p = (e + 1e-12) / (e + 1e-12).sum()
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, 0.08 * maxc)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = float('inf')

    def _update(p, sc, label):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  [t={time.monotonic()-t_start:.0f}s] BEST: {best_score:.10f} n={p.size} [{label}]")

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={parent.size}")

    targets = [1024, 2048]
    if parent is not None and 0 < parent.size <= 4096:
        targets.insert(0, parent.size)

    for n_target in targets:
        if time.monotonic() > deadline - 60:
            break
        if parent is not None:
            a = _interp_profile(parent, n_target)
        else:
            a = _make_profile_linear(n_target, 1.5)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        sc = _score_norm(a)
        print(f"Starting n={n_target}, initial score={sc:.10f}")
        _update(a, sc, f"init_n{n_target}")

        phase_time = min(300.0, (deadline - t_start) / max(1, len(targets) - targets.index(n_target)))
        p_end = time.monotonic() + min(phase_time, deadline - time.monotonic() - 10)

        a, sc = _lp_refine_linesearch(a, p_end, max_iters=200)
        _update(a, sc, f"lp_n{n_target}")

        if time.monotonic() < deadline - 30:
            p_end2 = min(time.monotonic() + 150, deadline - 15)
            a2, sc2 = _minimax_iterate(a, p_end2, max_rounds=100)
            _update(a2, sc2, f"qp_n{n_target}")

        if time.monotonic() < deadline - 30 and best_seq is not None:
            p_end3 = min(time.monotonic() + 200, deadline - 15)
            a3, sc3 = _combined_lp_qp(best_seq, p_end3, max_rounds=80)
            _update(a3, sc3, f"combined_n{best_seq.size}")

    if time.monotonic() < deadline - 30 and best_seq is not None:
        n_b = best_seq.size
        p_end4 = min(time.monotonic() + 200, deadline - 15)
        a4, sc4 = _lp_refine_linesearch(best_seq, p_end4, max_iters=200)
        _update(a4, sc4, f"lp2_n{n_b}")

    if time.monotonic() < deadline - 30 and best_seq is not None:
        n_b = best_seq.size
        p_end5 = min(time.monotonic() + 150, deadline - 10)
        a5, sc5 = _minimax_iterate(best_seq, p_end5, max_rounds=80)
        _update(a5, sc5, f"qp2_n{n_b}")

    if time.monotonic() < deadline - 40 and best_seq is not None:
        n_b = best_seq.size
        n_restarts = 3
        per_r = max(15.0, (deadline - time.monotonic() - 20) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            per = min(per_r, deadline - time.monotonic() - 5)
            if per < 10:
                break
            sigma = 0.08 + 0.04 * i
            a_s = _smooth_restart(best_seq, rng, sigma)
            a_r, sc_r = _lp_refine_linesearch(a_s, time.monotonic() + per, max_iters=100)
            _update(a_r, sc_r, f"restart{i}_n{n_b}")

    if time.monotonic() < deadline - 10 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 15:
            a_f, sc_f = _lp_refine_linesearch(best_seq, time.monotonic() + per, max_iters=100)
            _update(a_f, sc_f, "final_lp")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(1024, 1.5)
        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
        except Exception:
            pass
        return [float(x) for x in final]

    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
```
