3. Decrease beta over time

- More robust to changesin the objective

Let me also think about **the subgradient step** more carefully.

When beta is very small (approaching 0), the logsumexp approaches the max, and the gradient approaches the subgradient at the argmax. At this point, we can:
1. Find k* = argmax(c)
2. Compute the exact gradient of c_{k*} w.r.t. a: `g_j = a[k*-j] + a[j-k*]` (with out-of-range = 0)
3. Take a step in the negative gradient direction
4. Project onto the simplex (or in log-space, just update z)

This avoids the numerical issues of very small beta in logsumexp.

**Another idea: Mirror descent**

Since we're in log-space (softmax parameterization), we can use mirror descent, which is equivalent to multiplicative weight updates:
`z_j <- z_j - lr * h_j` (in log-space)
This corresponds to `a_j <- a_j * exp(-lr * h_j)` (then renormalize).

This is like a "relative" update: coordinates with high h (high contribution to the peak) are decreased more.

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses accurate gradient (both convolution terms), L-BFGS with continuation,
Nesterov-accelerated subgradient descent, coordinate descent polishing,
and multi-restart exploration. Optimizes in log-space (softmax) with
simplex projection for subgradient steps.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a)
    cm = float(c.max())
    c_shifted = np.clip((c - cm) / beta, -100, 100)
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _smooth_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a)
    cm = float(c.max())
    c_shifted = np.clip((c - cm) / beta, -100, 100)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    S1 = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
    S2 = _full_convolve(w, a)[0:n]
    h = S1 + S2
    W = float(np.dot(h, a))
    grad_z = 2.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _subgradient_of_peak(a):
    n = len(a)
    c = _full_convolve(a)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]
        g[j] = val
    return g, k_star


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = int(np.max(np.where(u * np.arange(1, n + 1) > (cssv - 1))))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _lbfgs_continuation(a_init, deadline, max_wall_time):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 3e-5]
    betas = [b * max_c0 for b in beta_starts]

    per_level = max_wall_time / len(betas)

    for beta in betas:
        if time.monotonic() > deadline:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        try:
            result = minimize(
                _smooth_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_smooth_gradient,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 30, 'maxfun': 1500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

def _coord_descent(a_init, deadline, top_k=100):

    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    cm = float(c.max())

    g1 = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        if 0 <= j - k_star < n:
            val += a[j - k_star]
        g1[j] = val
    W = float(np.dot(g1, a))
    grad = 2.0 * n * (g1 - W)

    for round_num in range(2000):
        if time.monotonic() > deadline:
            break

        top_idx = np.argsort(np.abs(grad))[-top_k:]
        top_idx = top_idx[np.argsort(np.abs(grad)[top_idx])[::-1]]

        improved = False
        for j in top_idx:
            if time.monotonic() > deadline:
                break

            best_local = best_score
            best_a_j = a[j]
            step = max(a[j] * 0.1, 1e-8)

            for direction in [-1, 1]:
                for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:

                    s = a.sum()
                    if s < 1e-15:

                        continue
                    a_norm = a / s

                    sc = float(2.0 * n * c_new.max())
                    if sc < best_local:
                        best_local = sc
                        best_a_j = new_val
                        improved = True
                        break
                a[j] = best_a_j

            a[j] = best_a_j

        s = a.sum()
        if s > 1e-15:
            a = a / s

        sc = _score_norm(a)

def _adam_logspace(a_init, deadline, lr=0.02, beta_frac=0.005, n_max_iter=50000):
