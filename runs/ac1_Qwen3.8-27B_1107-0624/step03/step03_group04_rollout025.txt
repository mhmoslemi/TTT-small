```python
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

    if s < 1e-10:
        return np.inf

        A[j:j + n, j] = a
    return A

    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s
```

```python
"""
KKT-based multiplicative optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Uses the KKT conditions of the primal problem to derive a fast iterative
method that equalizes the "pressure" on each coordinate. Combined with
periodic LP refinement and gradient polish for final convergence.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _kkt_multiplicative(a, deadline, max_iter=200000, n_top=20, alpha_init=0.3):
    """KKT-based multiplicative update: equalize pressure on coordinates."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_convolve(a).max())
    best_c = _fft_convolve(a)
    no_imp = 0
    alpha_cur = alpha_init
    n_top_cur = n_top

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        score = 2.0 * n * c_max

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_c = c.copy()
            no_imp = 0
            # Slowly increase alpha and n_top
            if no_imp % 100 == 0:
                alpha_cur = min(0.8, alpha_cur * 1.01)
                n_top_cur = min(50, n_top + it // 500)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                alpha_cur = max(0.05, alpha_cur * 0.97)
            if no_imp > max(10000, 15 * n):
                break

        # Find top peaks
        n_top_now = min(n_top_cur, len(c))
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:n_top_now]
        
        # Weights: soft emphasis on the top peaks
        # Use exponential weighting based on normalized values
        c_sorted = c[sorted_idx]
        c_min_top = c_sorted[min(n_top_now, len(c_sorted)-1)]
        if c_sorted[0] > c_min_top + 1e-15:
            # Exponential weights
            scale = 5.0 / (c_sorted[0] - c_min_top)
            raw_w = np.exp(scale * (c[top_idx] - c_sorted[0]))
        else:
            raw_w = np.ones(n_top_now)
        weights = raw_w / raw_w.sum()

        # Compute pressure via FFT
        N = _next_pow2(2 * n - 1)
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = weights
        fw = np.fft.rfft(w_full, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        p_full = np.fft.irfft(fw * fa_rev, N)[:2 * n - 1]
        p = 2.0 * p_full[n - 1:2 * n - 1]
        
        # Clip small negative values from FFT roundoff
        p = np.maximum(p, 1e-15)

        # KKT: λ = weighted average pressure
        lam = float(np.dot(a, p))
        
        # Multiplicative update
        ratio = lam / p
        a = a * np.power(ratio, alpha_cur)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

    return best_a, best_score


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.05)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = 2.0 * n * float(c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.95)
            if no_imp > max(100, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _gradient_polish(a, deadline, max_iter=50000):
    """Fine gradient polish with soft-max."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.02 * c_max)

    best_a = a.copy()
    best_score = 2.0 * n * c_max

    z = np.log(a)
    z -= np.mean(z)
    lr = 0.005
    no_imp = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        c_max_cur = float(c.max())
        # Soft-max weights
        x = (c - c_max_cur) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        w = ex / ex.sum()

        # Gradient via FFT
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v_arr / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.001, lr * 0.95)

        if it % 500 == 0:
            beta = max(1e-12, beta * 0.85)

        if no_imp > max(5000, 10 * n):
            break

    return best_a


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, width=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
```

```python
"""
KKT multiplicative optimization + LP refinement for minimizing 2n*max(conv(a,a))/sum(a)^2.

Core idea: The KKT conditions state that at the optimum, the 'pressure' on each
coordinate (weighted sum of contributing a values from active peaks) must be constant.
We use a fast multiplicative update to equalize this pressure, combined with LP-based
refinement (which solves the KKT system exactly for the current constraint matrix)
and gradient polish for final convergence.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _compute_pressure(a, top_idx, weights):
    """Compute p_j = 2 * sum_k weights[k] * a[k-j] for j=0..n-1."""
    n = len(a)
    p = np.zeros(n)
    for idx in range(len(top_idx)):
        k = int(top_idx[idx])
        w_k = 2.0 * weights[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            p[lo:hi + 1] += w_k * a[k - hi:k - lo + 1][::-1]
    return p


def _kkt_multiplicative(a, deadline, max_iter=200000, n_top=20, alpha_init=0.4):
    """KKT-based multiplicative update: equalize pressure on coordinates."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_convolve(a).max())
    no_imp = 0
    alpha_cur = alpha_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        score = 2.0 * n * c_max

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
            if it % 200 == 0:
                alpha_cur = min(0.9, alpha_cur * 1.008)
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                alpha_cur = max(0.05, alpha_cur * 0.96)
            if no_imp > max(15000, 20 * n):
                break

        # Find top peaks with exponential weighting
        n_top_now = min(n_top, 2 * n - 1)
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:n_top_now]
        c_top = c[top_idx]
        c_floor = c_sorted_floor(c, sorted_idx, n_top_now)

        if c_top[0] > c_floor + 1e-15:
            scale = 8.0 / (c_top[0] - c_floor)
            raw_w = np.exp(scale * (c_top - c_top[0]))
        else:
            raw_w = np.ones(n_top_now)
        weights = raw_w / raw_w.sum()

        # Compute pressure
        p = _compute_pressure(a, top_idx, weights)
        p = np.maximum(p, 1e-15)

        # KKT target: lambda = weighted average pressure
        lam = float(np.dot(a, p))

        # Multiplicative update
        ratio = lam / p
        a = a * np.power(ratio, alpha_cur)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

    return best_a, best_score


def c_sorted_floor(c, sorted_idx, n_top):
    """Get the value just below the top n_top peaks."""
    if n_top < len(c) - 1:
        return c[sorted_idx[n_top]]
    return 0.0


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.3):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = 2.0 * n * float(c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.06)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = 2.0 * n * float(c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = min(1.005, inflate_cur * 0.96)
            if no_imp > max(150, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _gradient_polish(a, deadline, max_iter=50000):
    """Fine gradient polish with soft-max in logit space."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-8, 0.015 * c_max)

    best_a = a.copy()
    best_score = 2.0 * n * c_max

    z = np.log(a)
    z -= np.mean(z)
    lr = 0.008
    no_imp = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        c_max_cur = float(c.max())
        x = (c - c_max_cur) / beta
        x = np.clip(x, -50.0, 50.0)
        ex = np.exp(x)
        w = ex / ex.sum()

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v_arr / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60.0, 60.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = 2.0 * n * float(c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.001, lr * 0.95)

        if it % 400 == 0:
            beta = max(1e-12, beta * 0.8)

        if no_imp > max(5000, 10 * n):
            break

    return best_a


def _make_flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_linear_profile(n, eta=1.5):

def _make_bell_profile(n, width=0.3):

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

        if time.monotonic() > deadline - 60:
            break

        # For later lengths, use less time
        priority = 1.0 if li == 0 else (0.5 if li == 1 else 0.3)
        time_left = deadline - time.monotonic()
        if time_left < 100:
            break

        print(f"\n--- Processing n={n_t} (priority={priority:.1f}, time_left={time_left:.0f}s) ---", flush=True)

        # Generate diverse initializations
        inits = []
        if parent is not None:
            inits.append(('parent', _interpolate_profile(parent, n_t)))
        inits.append(('flat_spec1', _make_flat_spectrum_init(n_t, rng)))
        inits.append(('flat_spec2', _make_flat_spectrum_init(n_t, rng)))
        inits.append(('linear', _make_linear_profile(n_t, 1.5)))
        inits.append(('bell', _make_bell_profile(n_t, 0.35)))
        inits.append(('sym_edge', _make_symmetric_edge(n_t, 0.3)))

        # Score and select best initialization
        best_init = None
        best_init_score = np.inf
        for name, a0 in inits:
            sc = _score(a0)
            if sc < best_init_score:
                best_init_score = sc
                best_init = a0.copy()
                best_init_name = name

        print(f"  Best init: '{best_init_name}' score={best_init_score:.8f}", flush=True)

        # If best init is already better than global best, update
        if best_init_score < best_score:
            best_score = best_init_score
            best_seq = best_init.copy()

        # Run full optimization pipeline
        a_opt, sc_opt = _optimize_single_length(best_init, n_t, deadline, priority)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc_opt:.10f} (n={n_t})", flush=True)

    # Final fine polish on the global best
    if best_seq is not None and time.monotonic() < deadline - 10:
        gp_deadline = deadline - 5
        if gp_deadline - time.monotonic() > 15:
            a_fine = _gradient_polish(best_seq, gp_deadline, max_iter=30000)
            sc_fine = _score(a_fine)
            if sc_fine < best_score:
                best_score = sc_fine
                best_seq = a_fine.copy()
            print(f"Final gradient polish: {sc_fine:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(512, 1.5)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
```
