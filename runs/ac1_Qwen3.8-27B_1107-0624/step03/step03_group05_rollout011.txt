def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_raw(a):
    """Score without normalization (assumes sum=1)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())

def _grad_smooth_max(a, beta):
    """Gradient of beta*log(sum(exp(c/beta))) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    # Gradient: 2 * cross_correlation(w, a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, c

def _grad_sum_sq(a):

    h = h_full[n - 1:2 * n - 1]
    grad = 4.0 * h

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    return A

def _thue_morse(n):
    """Generate Thue-Morse based non-negative sequence."""
    # Thue-Morse sequence: t(k) = parity of number of 1s in binary representation of k
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=np.float64)
    # Map to [0.5, 1.5] for non-negative values
    a = 1.0 + 0.8 * (2.0 * tm - 1.0)
    a = np.maximum(a, 0.1)
    return a / a.sum()

    a = np.convolve(a, kernel, mode='same')

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()

    s = p.sum()

        # Build constraint matrix
        A = _make_conv_matrix(a)
        b = np.full(2 * n - 1, M_lp)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 5 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.02)
            if no_imp > 200:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0003, inflate_cur * 0.95)
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                blend_cur = max(0.05, blend_cur * 0.9)
                inflate_cur = min(1.005, inflate_cur * 0.97)
            if no_imp > max(100, n // 2):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        
        a = a_new
    
    return best_a, best_score

def _adam_logit_polish(a_init, deadline, lr=0.03, beta_scale=0.03):
    """Adam optimization in logit parameterization."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    n = len(a)
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = _score_raw(a)
    beta = max(1e-8, beta_scale * float(_fft_convolve(a).max()))
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        mval = float(c.max())
        x = (c - mval) / beta
        ex = np.exp(x)
        s_exp = float(ex.sum())
        if not np.isfinite(s_exp) or s_exp <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_exp
        
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        
        # Natural gradient in logit space
        vp = 2.0 * h + 1e-8 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = np.log(best_a + 1e-10)
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue
        
        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v_arr / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -30.0, 30.0)
        z -= float(np.mean(z))
        
        e = np.exp(z)
        a = e / e.sum()
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.95)
        
        if t % 100 == 0:
            beta = max(1e-10, beta * 0.8)
        
        if no_imp > max(1500, 3 * n):
            a = best_a.copy()
            z = np.log(a + 1e-10)
            z -= float(np.mean(z))
            rng = np.random.default_rng(t)
            z = z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -30.0, 30.0)
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            lr = base_lr
            beta = max(1e-8, 0.05 * float(_fft_convolve(best_a).max()))
            no_imp = 0
    
    return best_a, best_score

def _cross_peak_reduction(a, deadline, num_peaks=5):
    """Targeted reduction of top peaks using coordinate descent."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_raw(a)
    rng = np.random.default_rng(42)
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        peak_idx = np.argsort(c)[::-1][:num_peaks]
        
        improved = False
        for pk in peak_idx:
            # Try to reduce c[pk] by adjusting a
            # c[pk] = sum_j a[j]*a[pk-j]
            # To reduce it, we can decrease a[j] or a[pk-j] for j where product is large
            for _trial in range(5):
                j = rng.integers(0, n)
                k = pk - j
                if k < 0 or k >= n:
                    # Try the other term
                    j2 = rng.integers(0, n)
                    k2 = pk - j2
                    if k2 < 0 or k2 >= n:
                        continue
                    j, k = j2, k2
                
                # Move mass from a[j] to a random position
                if a[j] < 1e-6:
                    continue
                delta = a[j] * rng.uniform(0.01, 0.1)
                dest = rng.integers(0, n)
                if dest == j:
                    continue
                
                a_try = a.copy()
                a_try[j] -= delta
                a_try[dest] += delta
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                
                sc = _score_raw(a_try)
                if sc < best_score - 1e-10:
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try a different approach: scramble parts
            block = rng.integers(max(4, n//32), max(8, n//16))
