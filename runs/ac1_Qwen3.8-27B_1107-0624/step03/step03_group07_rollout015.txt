    # Subgradient averaging
    avg_a = a0.copy()
    avg_weight = 1.0
    avg_best_score = np.inf

    iters = 0
    while time.monotonic() < deadline:
        _, w = soft_max_weights(c, beta)
        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        # Frank-Wolfe hybrid: occasionally move toward a vertex
        if fw_period > 0 and iters > 0 and iters % fw_period == 0 and rng.random() < 0.5:
            # Move toward the vertex that most reduces the gradient
            j = int(np.argmin(dz))
            # Blended step: mix Adam direction with vertex direction
            vertex_dir = np.zeros(n)
            vertex_dir[j] = 1.0
            vertex_dir = vertex_dir - vertex_dir.mean()
            dz = (1.0 - fw_step) * dz + fw_step * vertex_dir * a0.sum()

        # Adam update
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        # Enforce symmetry
        z = 0.5 * (z + z[::-1])

        # Stabilize
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        z = 0.5 * (z + z[::-1])

        # New a via softmax
        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        # Evaluate
        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Subgradient averaging
        if use_avg:
            avg_a = avg_a + a0 / (avg_weight + 1.0)
            avg_weight += 1.0
            if iters % 50 == 0:
                avg_score = compute_score(avg_a)
                if avg_score < avg_best_score:
                    avg_best_score = avg_score
                    if avg_score < best_score:
                        best_score = avg_score
                        best_a = avg_a.copy()

        iters += 1

        # Adapt beta
        if t_adam % 30 == 0:
            beta = max(beta_min, beta * beta_decay)

        # Adapt learning rate
        if no_improve > 200:
            current_lr = max(0.003, current_lr * 0.98)
        if no_improve > 500:
            current_lr = max(0.001, current_lr * 0.95)

        # Random restart
        if no_improve > restart_period:
            noise_scale = 0.15 + 0.1 * (no_improve / restart_period)
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = 0.5 * (z + z[::-1])
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(0.05 * max_c, beta_min)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = base_lr
            no_improve = 0
            avg_a = a0.copy()
            avg_weight = 1.0
            avg_best_score = np.inf

    # Check if averaged solution is better
    if use_avg and avg_best_score < best_score:
        best_score = avg_best_score
        best_a = avg_a.copy()

    return best_a, best_score


def peak_shave_refine(a0, deadline, rng, n_peaks=3, lr=0.02):
    """
    Refine by specifically targeting the top peaks in the convolution.
    Uses alternating between soft-max and hard-max weighting.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z = np.log(a0)
    z = 0.5 * (z + z[::-1])
    z = z - z.mean()

    best_a = a0.copy()
    c = full_convolve_fft(a0)
    best_score = float(2.0 * n * c.max())

    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    use_hard = False
    hard_period = 50

    while time.monotonic() < deadline:
        c = full_convolve_fft(a0)
        max_c = float(c.max())

        if use_hard:
            # Target only the single peak
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            # Target top peaks with uniform weight
            peak_indices = np.argsort(c)[-n_peaks:]
            w = np.zeros_like(c)
            for idx in peak_indices:
                w[idx] = 1.0 / n_peaks

        h = compute_h(w, a0)
        grad_a = 2.0 * h
        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = 0.5 * (z + z[::-1])
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if score < best_score:
            best_score = score
            best_a = a0.copy()

        # Alternate between hard and soft targeting
        if t_adam % hard_period == 0:
            use_hard = not use_hard

    return best_a, best_score


def lbfhs_refine(a0, deadline, max_iter=200):
    """
    Try L-BFGS-B optimization in logit space.
    Uses smooth approximation (log-sum-exp) of the max.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    z0 = np.log(a0)
    z0 = z0 - z0.mean()
    z0 = 0.5 * (z0 + z0[::-1])

    c0 = full_convolve_fft(a0)
    max_c0 = float(c0.max())
    best_score = float(2.0 * n * max_c0)
    best_z = z0.copy()

    beta = 0.02 * max_c0
    if beta < 1e-6:
        beta = 1e-5

    def objective(z):
        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-15:
            return 1e10
        a = a / s
        c = full_convolve_fft(a)
        # Log-sum-exp
        mval = c.max()
        if beta > 1e-14:
            x = np.clip((c - mval) / beta, -50, 0)
            val = mval + beta * np.log(np.sum(np.exp(x)))
        else:
            val = mval
        return 2.0 * n * val

    def grad(z):
        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-15:
            return np.zeros_like(z)
        a = a / s
        c = full_convolve_fft(a)
        # Soft-max weights
        if beta > 1e-14:
            mval = c.max()
            x = np.clip((c - mval) / beta, -50, 0)
            ex = np.exp(x)
            w = ex / ex.sum()
        else:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        h = compute_h(w, a)
        grad_a = 2.0 * h
        dot = float(np.dot(a, grad_a))
        return a * (grad_a - dot)

    bounds = [(-40.0, 40.0)] * n
    t_deadline = deadline

    res = scipy_minimize(objective, z0, method='L-BFGS-B',
                         jac=grad, bounds=bounds,
                         options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                  'ftol': 1e-15, 'gtol': 1e-12,
                                  'maxls': 20})

    if res.success or res.fun < best_score:
        z_new = res.x
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_z = z_new.copy()
            a0 = a_new

    return a0, best_score


def diverse_profiles(n, rng, count=10):
    """Generate diverse starting profiles."""
    profiles = []
    for _ in range(count):
        profiles.append(make_profile(n, rng))
    # Add some structured profiles
    t = np.linspace(0.0, 1.0, n)
    for eta in [1.0, 2.0]:
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: fine-tune initial sequence with Adam+FW hybrid,
    try L-BFGS-B refinement, peak-shaving, and multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting multi-strategy search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)
    else:
        init_seq = make_profile(1024, rng)
        sc_init = compute_score(init_seq)
        print(f"Generated initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Fine-tune initial sequence with Adam+FW =====
    n1 = init_seq.size
    phase1_deadline = min(deadline, t_start + 400.0)
    print(f"\n--- Phase 1: Adam+FW at n={n1} ---")

    result1, sc1 = logit_adam_fw_optimize(
        init_seq, phase1_deadline, rng,
        lr=0.025, beta_factor=0.05, beta_decay=0.94,
        fw_period=150, fw_step=0.03, restart_period=2000, use_avg=True
    )
    sc1_final = compute_score(result1)
    update(result1, sc1_final)
    print(f"  Phase 1 Adam+FW: {sc1_final:.10f}")

    # ===== Phase 2: Peak-shaving refinement =====
    if time.monotonic() < deadline - 60:
        phase2_deadline = min(deadline, time.monotonic() + 200.0)
        print(f"\n--- Phase 2: Peak shaving ---")
        result2, sc2 = peak_shave_refine(best_seq, phase2_deadline, rng,
                                         n_peaks=3, lr=0.015)
        sc2_final = compute_score(result2)
        update(result2, sc2_final)
        print(f"  Phase 2 peak shave (3 peaks): {sc2_final:.10f}")

        if time.monotonic() < deadline - 40:
            phase2b_deadline = min(deadline, time.monotonic() + 100.0)
            result2b, sc2b = peak_shave_refine(best_seq, phase2b_deadline, rng,
                                               n_peaks=1, lr=0.01)
            sc2b_final = compute_score(result2b)
            update(result2b, sc2b_final)
            print(f"  Phase 2b peak shave (1 peak): {sc2b_final:.10f}")

    # ===== Phase 3: L-BFGS-B refinement =====
    if time.monotonic() < deadline - 60:
        phase3_deadline = min(deadline, time.monotonic() + 300.0)
        print(f"\n--- Phase 3: L-BFGS-B refinement ---")
        try:
            result3, sc3 = lbfhs_refine(best_seq, phase3_deadline, max_iter=300)
            sc3_final = compute_score(result3)
            update(result3, sc3_final)
            print(f"  Phase 3 L-BFGS-B: {sc3_final:.10f}")
        except Exception as e:
            print(f"  L-BFGS-B failed: {e}")

    # ===== Phase 4: Try different n values =====
    print(f"\n--- Phase 4: Multi-scale exploration ---")
    current_n = best_seq.size if best_seq is not None else 1024
    n_candidates = []
    for n_try in [512, 768, 1024, 1536, 2048, 3072]:
        if n_try != current_n and n_try not in n_candidates:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        if best_score < 1.503 and n_try > 2048:
            continue  # Don't spend too much time on very large n if we're close

        p_try = interpolate_profile(best_seq, n_try)
        # Enforce symmetry
        p_try = 0.5 * (p_try + p_try[::-1])
        sc_try = compute_score(p_try)
        update(p_try, sc_try)
        print(f"  Interpolated to n={n_try}: {sc_try:.10f}")

        # Quick optimization
        alloc = min(80.0, (deadline - time.monotonic()) - 30)
        if alloc > 20:
            dl = time.monotonic() + alloc
            r, sc_r = logit_adam_fw_optimize(
                p_try, dl, rng,
                lr=0.02, beta_factor=0.04, beta_decay=0.94,
                fw_period=100, fw_step=0.02, restart_period=1000, use_avg=True
            )
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  After n={n_try} optimization: {sc_f:.10f}")

    # ===== Phase 5: Final polish on best sequence =====
    phase5_deadline = deadline
    print(f"\n--- Phase 5: Final polish ---")

    if best_seq is not None:
        n_polish = best_seq.size

        # Round 1: Very fine Adam
        if time.monotonic() < phase5_deadline - 30:
            dl = min(phase5_deadline, time.monotonic() + 150.0)
            p0 = best_seq.copy()
            p0 = p0 + 0.0003 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            r, sc_r = logit_adam_fw_optimize(
                p0, dl, rng,
                lr=0.01, beta_factor=0.02, beta_decay=0.95,
                fw_period=200, fw_step=0.01, restart_period=3000, use_avg=True
            )
            sc_f = compute_score(r)
            update(r, sc_f)
            print(f"  Polish round 1 (fine Adam): {sc_f:.10f}")

        # Round 2: Peak shaving
        if time.monotonic() < phase5_deadline - 20:
            dl = min(phase5_deadline, time.monotonic() + 80.0)
            r2, sc_r2 = peak_shave_refine(best_seq, dl, rng, n_peaks=5, lr=0.008)
            sc2_f = compute_score(r2)
            update(r2, sc2_f)
            print(f"  Polish round 2 (peak shave 5): {sc2_f:.10f}")

        # Round 3: Try L-BFGS again
        if time.monotonic() < phase5_deadline - 20:
            try:
                dl = min(phase5_deadline, time.monotonic() + 120.0)
                r3, sc3 = lbfhs_refine(best_seq, dl, max_iter=200)
                sc3_f = compute_score(r3)
                update(r3, sc3_f)
                print(f"  Polish round 3 (L-BFGS-B): {sc3_f:.10f}")
            except Exception as e:
                print(f"  L-BFGS-B polish failed: {e}")

    # ===== Phase 6: Quick try of a completely different approach =====
    if time.monotonic() < deadline - 30 and best_score > 1.504:
        print(f"\n--- Phase 6: Diverse restarts ---")
        for n_try in [best_seq.size if best_seq is not None else 1024]:
            for restart_i in range(3):
                if time.monotonic() > deadline - 15:
                    break
                p_new = make_profile(n_try, rng)
                # Mix with current best
                if best_seq is not None and best_seq.size == n_try:
                    p_new = 0.5 * p_new + 0.5 * best_seq
                    p_new = np.maximum(p_new, 1e-12)
                    p_new = p_new / p_new.sum()
                alloc = min(50.0, (deadline - time.monotonic()) - 15)
                if alloc > 15:
                    dl = time.monotonic() + alloc
                    r, sc_r = logit_adam_fw_optimize(
                        p_new, dl, rng,
                        lr=0.03, beta_factor=0.06, beta_decay=0.93,
                        fw_period=100, fw_step=0.03, restart_period=800, use_avg=True
                    )
                    sc_f = compute_score(r)
                    update(r, sc_f)
                    print(f"  Diverse restart {restart_i}: n={n_try}, score={sc_f:.10f}")

    print(f"\n  Final best score: {best_score:.10f}")

    # ===== Final validation and return =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_profile(512, rng)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_profile(256, rng)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
