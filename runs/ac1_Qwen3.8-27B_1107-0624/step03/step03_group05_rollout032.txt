Convex-aware optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key insight: The problem is non-convex due to quadratic (bilinear) constraints,
but has structure exploitable via: (1) parametric shape optimization for good

(3) LP-based spreading for local refinement, (4) simulated annealing for
escaping local minima, (5) multi-length search.

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _gradient_smooth_max(a, beta, top_k=0):
    n = len(a)
    c = _fft_convolve(a)
    if top_k > 0:
        sorted_idx = np.argsort(c)[::-1]
        w = np.zeros_like(c)
        w[sorted_idx[:top_k]] = 1.0 / top_k
    else:
        w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max()), c


def _parametric_linear(n, r):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + r * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _parametric_quadratic(n, r, s):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + r * t + s * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _parametric_cubic(n, r, s, u):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + r * t + s * t * t + u * t * t * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _parametric_exp(n, r):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(r * t)
    return a / a.sum()


def _parametric_power(n, r):
    t = np.linspace(0.0, 1.0, n)
    a = (t + 0.01) ** r
    return a / a.sum()


def _parametric_cosine(n, r):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + r * np.cos(np.pi * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _optimize_linear_shape(n):
    """Find best r for linear shape 1+r*t."""
    def objective(r):
        a = _parametric_linear(n, r)
        return _score(a)
    # Grid search first
    best_r = 0.0
    best_sc = np.inf
    for r in np.linspace(-2.0, 5.0, 71):
        sc = objective(r)
        if sc < best_sc:
            best_sc = sc
            best_r = r
    # Refine with golden section
    result = minimize_scalar(objective, bounds=(best_r - 0.5, best_r + 0.5),
                            method='bounded', options={'xatol': 1e-6})
    return _parametric_linear(n, result.x), result.fun


def _optimize_quadratic_shape(n):
    """Find best (r,s) for quadratic shape 1+r*t+s*t^2."""
    best_a = None
    best_sc = np.inf
    for r in np.linspace(-2.0, 4.0, 17):
        for s in np.linspace(-3.0, 3.0, 13):
            a = _parametric_quadratic(n, r, s)
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc


def _optimize_exp_shape(n):
    """Find best r for exponential shape exp(r*t)."""
    best_a = None
    best_sc = np.inf
    for r in np.linspace(-3.0, 5.0, 41):
        a = _parametric_exp(n, r)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=10000, inflate_init=1.002, blend_init=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend_init
    inflate_cur = inflate_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.03, inflate_cur * 1.06)
            if no_imp > 1000:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0005, inflate_cur * 0.97)
            blend_cur = min(0.4, blend_cur * 1.02)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.03, blend_cur * 0.93)
                inflate_cur = min(1.015, inflate_cur * 0.98)
            if no_imp > max(500, 4 * n):
                a = best_a.copy()
                blend_cur = blend_init
                inflate_cur = inflate_init
                no_imp = 0

        a = a_new

    return best_a


def _nesterov_gradient_polish(a, deadline, max_iter=30000, beta_init_scale=0.03):
    """Nesterov-accelerated gradient with backtracking line search."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    beta = max(1e-8, beta_init_scale * float(c.max()))
    lr = 0.05
    momentum = 0.85
    y = a.copy()  # Nesterov extrapolation point
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    top_k = 0  # Start with smooth max, switch to exact later

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        if it > 0:
            t_k = 1.5 / (it + 2.5)
            y = a + t_k * (a - y + v)
            y = _project_simplex(y)

        # Compute gradient
        if it > 2000:
            top_k = min(10, max(3, it // 1000))
        grad, maxc, c_arr = _gradient_smooth_max(y, beta, top_k=top_k)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 3.0)
            v = np.zeros(n)
            continue

        # Backtracking line search
        a_trial = _project_simplex(a - lr * grad + momentum * v)
        sc_trial = float(2.0 * n * _fft_convolve(a_trial).max())

        lr_eff = lr
        backtracks = 0
        while sc_trial >= best_score + 1e-14 and backtracks < 8:
            lr_eff *= 0.5
            a_trial = _project_simplex(a - lr_eff * grad + momentum * v)
            sc_trial = float(2.0 * n * _fft_convolve(a_trial).max())
            backtracks += 1

        v = a_trial - a
        a = a_trial

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
            lr = min(0.1, lr * 1.02)
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.9)
            if no_imp > max(3000, 5 * n):
                a = best_a.copy()
                v = np.zeros(n)
                y = a.copy()
                beta = max(1e-8, beta_init_scale * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 150 == 0:
            beta = max(1e-12, beta * 0.7)

    return best_a


def _adams_logit_polish(a, deadline, max_iter=25000, lr=0.04):
    """Adam optimizer in log-space with smooth-max objective."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.03 * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.003, lr * 0.95)

        if it % 200 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(3000, 5 * n):
            rng_r = np.random.default_rng((it * 7 + n) % (2**32))
            z = 0.6 * best_z + 0.4 * rng_r.normal(size=n) * 0.3
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.03 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a


def _simulated_annealing(a, deadline, max_iter=20000, T_init=0.01, T_min=1e-6):
    """SA to escape local minima. Perturbs random pairs of elements."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()
    rng = np.random.default_rng(999)

    best_a = a.copy()
    best_score = _score(a)
    current_score = best_score
    T = T_init

    no_imp = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        T = max(T_min, T_init * (T_min / T_init) ** (it / max(1, max_iter)))

        # Random perturbation: pick 2-5 random indices and redistribute
        n_pert = rng.integers(2, 6)
        idx = rng.choice(n, size=n_pert, replace=False)
        a_new = a.copy()
        total = a_new[idx].sum()
        if total < 1e-12:
            continue
        # Redistribute among chosen indices with noise
        new_vals = np.abs(rng.normal(total / n_pert, total / n_pert * 0.5, n_pert))
        new_vals = np.maximum(new_vals, 0.0)
        ns = new_vals.sum()
        if ns < 1e-12:
            continue
        new_vals = new_vals / ns * total
        a_new[idx] = new_vals
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc_new = _score(a_new)

        # Metropolis acceptance
        delta = sc_new - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_new
            current_score = sc_new
            if sc_new < best_score - 1e-13:
                best_score = sc_new
                best_a = a_new.copy()
                no_imp = 0
                T = max(T_min, T * 0.95)
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 5000:
            break

    return best_a


def _exact_subgradient_polish(a, deadline, max_iter=20000):
    """Use exact subgradient (average over active set) with Polyak step size."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    lower_bound = 1.0  # Theoretical lower bound

    no_imp = 0
    prev_score = best_score
    momentum_buf = np.zeros(n)
    mu = 0.7  # Momentum for subgradient

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        score = float(2.0 * n * M)

        # Find active set: indices within 1% of max
        tol = 0.01 * M
        active = np.where(c >= M - tol)[0]

        # Compute subgradient: average gradient over active constraints
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        if not np.all(np.isfinite(grad)):
            continue

        # Polyak step size
        gap = score - lower_bound
        gnorm2 = float(np.dot(grad, grad))
        if gnorm2 < 1e-20:
            break
        step = min(1.0, gap / (gnorm2 + 1e-12))
        step = max(step, 0.01)

        # Momentum
        momentum_buf = mu * momentum_buf + (1 - mu) * grad
        a_new = _project_simplex(a - step * momentum_buf)

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        if no_imp > 5000:
            a = best_a.copy()
            momentum_buf = np.zeros(n)
            no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s
