"""
Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: (1) Adam smooth-max in logit space with adaptive temperature,
(2) LP duality-based refinement with guaranteed improvement,
(3) coordinate descent polish, (4) structured multi-restart,
(5) multi-scale escalation.
Key improvements over previous: better LP scheduling, coordinate descent,
power-mean smooth max alternative, and more aggressive restart strategy.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 256:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    c = _full_convolve(a)
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * c.max()) / (s * s)


def _smooth_max_weights(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 1e-30:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-14):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-30:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def _adam_logit(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                no_imp_limit=300, noise_scale=None):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 40 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.2:
                    beta = max(1e-5, beta * 0.4)

        limit = max(80, no_imp_limit)
        if no_imp > limit:
            if noise_scale is not None:
                ns = noise_scale
            else:
                if best_score < 1.52:
                    ns = 0.06
                elif best_score < 1.55:
                    ns = 0.12
                else:
                    ns = 0.25
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, t_init=0.5, max_solves=40):
    n = len(a)
    a = _normalize(a)
    if a is None:
        return a, np.inf
    current_best = a.copy()
    current_score = float(2.0 * n * _full_convolve(a).max())
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0

    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new = float(2.0 * n * _full_convolve(a_new).max())
        if sc_new < current_score:
            current_score = sc_new
            current_best = a_new.copy()
        a = a_new
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, t * 0.85)

    return current_best, current_score


def _coord_descent(a, n_sweeps=3, rng=None):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_score = float(2.0 * n * _full_convolve(a).max())

    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            old_val = a[j]
            a[j] = 0.0
            c_base = _full_convolve(a)
            s_base = float(a.sum())
            if s_base < 1e-10:
                a[j] = old_val
                continue
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                idx = k - j
                if 0 <= idx < n:
                    d[k] = a[idx]
            best_x = old_val
            best_sc = float(2.0 * n * _full_convolve(a + old_val * np.zeros(2*n-1)[:0].reshape(-1)).max()) if False else None

            # Evaluate at a few candidate values
            candidates = [old_val * 0.5, old_val, old_val * 1.5, old_val * 2.0, 2.0 * s_base / n]
            if rng is not None:
                candidates.append(old_val * (1.0 + 0.5 * rng.standard_normal()))
            candidates.append(0.01 * s_base / n)

            for xv in candidates:
                if xv < 0:
                    xv = 0.0
                a[j] = xv
                s_new = s_base + xv
                if s_new < 1e-10:
                    continue
                c_new = c_base + xv * d
                max_c = float(c_new.max())
                sc = 2.0 * n * max_c / (s_new * s_new)
                if sc < best_score:
                    best_score = sc
                    best_x = xv
                    improved = True
            a[j] = best_x
            a = a / a.sum()

        if not improved:
            break

    return a, best_score


def _subgradient_step(a, n_steps=50, lr_init=0.1, lr_decay=0.95, rng=None):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best = a.copy()
    best_score = float(2.0 * n * _full_convolve(a).max())
    lr = lr_init

    for step in range(n_steps):
        c = _full_convolve(a)
        maxc = float(c.max())
        # Subgradient: pick a random argmax
        idxs = np.where(c >= maxc - 1e-12)[0]
        k = int(idxs[rng.integers(len(idxs))]) if len(idxs) > 0 else int(np.argmax(c))
        # Gradient of conv[k] w.r.t. a[j] is a[k-j]
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad = np.zeros(n)
        grad[valid] = a[idx[valid]]
        # Projected gradient descent in simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s
        sc = float(2.0 * n * _full_convolve(a_new).max())
        if sc < best_score:
            best_score = sc
            best = a_new.copy()
        a = a_new
        lr *= lr_decay

    return best, best_score


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))

        n_lp = best_seq.size
        print(f"  LP refine n={n_lp}")
        p_lp, sc_lp = _lp_refine(best_seq, phase3_end, t_init=0.5, max_solves=50)
        _update(p_lp, sc_lp, 'LP_refine')

        # Adam after LP
        if time.monotonic() < phase3_end - 20:
            per = min(80.0, phase3_end - time.monotonic() - 5)
            if per > 15:
                p_ap, sc_ap = _adam_logit(p_lp, time.monotonic() + per, rng,
                                           lr=0.03, beta_frac=0.07, beta_decay=0.94,
                                           no_imp_limit=200)
                _update(p_ap, sc_ap, 'post_LP')

    # --- Phase 4: Coordinate descent ---
    phase4_end = t_start + 1100.0
    if best_seq is not None and time.monotonic() < phase4_end:
        n_cd = min(best_seq.size, 2048)
        p_cd_in = _interp_profile(best_seq, n_cd) if best_seq.size != n_cd else best_seq.copy()
        p_cd, sc_cd = _coord_descent(p_cd_in, n_sweeps=5, rng=rng)
        _update(p_cd, sc_cd, 'coord_desc')

        # More coord descent sweeps
        remaining = phase4_end - time.monotonic()
        if remaining > 20:
            p_cd2, sc_cd2 = _coord_descent(p_cd, n_sweeps=3, rng=rng)
            _update(p_cd2, sc_cd2, 'coord_desc2')

    # --- Phase 5: Multi-scale escalation ---
    phase5_end = t_start + 1350.0
    if best_seq is not None and time.monotonic() < phase5_end:
        n_cur = best_seq.size
        if n_cur <= 1024:
            for tn in [2048, 4096]:
                if time.monotonic() > phase5_end - 40:
                    break
                p_int = _interp_profile(best_seq, tn)
                if p_int is None:
                    continue
                sc_int = _score(p_int)
                _update(p_int, sc_int, f'interp_n{tn}')

                per = min(50.0, phase5_end - time.monotonic() - 20)
                if per > 15:
                    p_e, sc_e = _adam_logit(p_int, time.monotonic() + per, rng,
                                             lr=0.04, beta_frac=0.08, beta_decay=0.93,
                                             no_imp_limit=200)
                    _update(p_e, sc_e, f'esc_n{tn}')

    # --- Phase 6: Multi-restart with structured perturbations ---
    phase6_end = deadline - 40.0
    if best_seq is not None and time.monotonic() < phase6_end:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(15.0, (phase6_end - time.monotonic()) / max(1, n_restarts + 2))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 10:
                break

            # Structured perturbation: mix with a random profile
            zp = np.log(np.maximum(best_seq, 1e-12))
            zp -= zp.mean()
            ns = 0.08 + 0.06 * i
            zp = zp + ns * rng.normal(size=n_best)
            zp = np.clip(zp, -60, 60)
            zp -= zp.mean()
            p_s = _softmax_from_z(zp)

            p_r, sc_r = _adam_logit(p_s, time.monotonic() + per, rng, lr=0.04,
                                     beta_frac=0.07, beta_decay=0.93,
                                     no_imp_limit=200)
            _update(p_r, sc_r, f'restart{i}')

        # Subgradient on best
        if time.monotonic() < phase6_end - 10:
            per = min(40.0, phase6_end - time.monotonic() - 5)
            if per > 10:
                p_sg, sc_sg = _subgradient_step(best_seq, n_steps=min(200, int(per * 50)),
                                                 lr_init=0.05, lr_decay=0.98, rng=rng)
                _update(p_sg, sc_sg, 'subgrad')

    # --- Phase 7: Final polish ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        per = deadline - time.monotonic() - 5
        if per > 10:
            p_f, sc_f = _adam_logit(best_seq, time.monotonic() + per, rng,
                                     lr=0.015, beta_frac=0.04, beta_decay=0.96,
                                     no_imp_limit=150)
            _update(p_f, sc_f, 'final_polish')

        # Quick coord descent
        if time.monotonic() < deadline - 3 and best_seq.size <= 2048:
            p_cd_f, sc_cd_f = _coord_descent(best_seq, n_sweeps=2, rng=rng)
            _update(p_cd_f, sc_cd_f, 'final_cd')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(512, 1.5)

        exact_sc = _score(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size} time={time.monotonic()-t_start:.1f}s")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
                if eval_sc < best_score:
                    print(f"  (eval differs from local score)")
        except Exception as e:
            print(f"  eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(512, 1.5)
    print(f"FALLBACK n=512")
    return [float(x) for x in fb]
