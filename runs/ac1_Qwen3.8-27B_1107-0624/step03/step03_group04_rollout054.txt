So `A[k, i] = a[k-i]` (when 0 <= k-i < n, else 0).

In the code: `A[j:j+n, j] = a` means `A[j+i, j] = a[i]` for i=0,...,n-1. So `A[k, j] = a[k-j]` when 0 <= k-j < n. Yes, this is correct.

So the LP constraint is `A @ g <= M` where A is the convolution matrix.

Fine-tuning optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses very fine LP refinement, subgradient coordinate descent,
and multi-length search to improve upon parent solution.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = c - c.min()
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _coordinate_descent(a, deadline, max_iter=100000, n_top=10):
    """
    Coordinate descent / mass shuffling to reduce peak convolution.
    For the top peaks, find mass moves that reduce the peak.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        c_max = float(c.max())
        
        # Find top peaks
        top_idx = np.argsort(c)[::-1][:n_top]
        
        # For each top peak, compute the gradient w.r.t. a
        # grad[i] = 2 * a[k-i] for peak at position k
        # We want to move mass from high-gradient positions to low-gradient positions
        
        # Compute "pressure" on each coordinate from all top peaks
        pressure = np.zeros(n)
        for k in top_idx:
            # c[k] = sum_i a[i] * a[k-i]
            # dc[k]/da[i] = 2 * a[k-i] (if k-i in range)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += 2.0 * a[j]
        
        # We want to reduce mass at positions with high pressure
        # and add mass at positions with low pressure
        # Use exponential tilting
        max_p = pressure.max()
        if max_p < 1e-15:
            break
        
        # Gentle tilt
        eps = 0.01
        a_new = a * np.exp(-eps * pressure / max_p)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            # Increase eps slightly
            eps = min(0.1, eps * 1.01)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                eps = max(0.001, eps * 0.9)
            if no_imp > 2000:
                break
        
        a = a_new
    
    return best_a, best_score

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _flat_spectrum_init(n, rng):
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s

But we need to be careful: changing a[i] also affects other convolution entries c[k] for k ≠ k*. So a naive gradient step might reduce c[k*] but increase c[k'] for some k'.

A safer approach: try small perturbations and check if the MAX peak decreases (not just c[k*]).

For efficiency, I can:
- Precompute the "impulse response" of the perturbation: if I add δ to a[i], the change in c[k] is 2*δ*a[k-i] (for all k).
- So the change in the peak is max_k 2*δ*a[k-i] (for the increase case) or min_k 2*δ*a[k-i] (for the decrease case).

Wait, that's not quite right either. If I decrease a[i] by δ and increase a[j] by δ:
- Δc[k] = 2δ(a[k-j] - a[k-i]) for all k
- The new max is max_k (c[k] + 2δ(a[k-j] - a[k-i]))
- We want this to be < current max

This gives us a condition on δ for the move to be beneficial. We can compute the maximum allowable δ and use a fraction of it.

```python
def _peak_reduction_move(a, c, k_peak, i, j, delta_frac=0.5):
    """
    Try moving mass from position i to position j to reduce peak at k_peak.
    Returns the new a if the move is beneficial, else None.
    """
    n = len(a)
    # Change in c[k] for all k: 2*delta*(a[k-j] - a[k-i])
    # We need: c[k] + 2*delta*(a[k-j] - a[k-i]) <= c_max for all k
    # => 2*delta*(a[k-j] - a[k-i]) <= c_max - c[k] for all k
    # => delta <= (c_max - c[k]) / (2*(a[k-j] - a[k-i])) for all k where a[k-j] > a[k-i]
    
    c_max = float(c.max())
    # Compute the constraint on delta
    # For k where a[k-j] > a[k-i]: delta <= (c_max - c[k]) / (2*(a[k-j] - a[k-i]))
    # For k where a[k-j] <= a[k-i]: no upper bound on delta (the change is <= 0)
    
    # We also need delta <= a[i] (can't make a[i] negative)
    max_delta = a[i]  # Upper bound from non-negativity
    
    for k in range(2 * n - 1):
        diff = 0.0
        # a[k-j] if 0 <= k-j < n
        kj = k - j
        if 0 <= kj < n:
            diff += a[kj]
        # a[k-i] if 0 <= k-i < n
        ki = k - i
        if 0 <= ki < n:
            diff -= a[ki]
        
        if diff > 1e-15:
            bound = (c_max - c[k]) / (2.0 * diff)
            if bound < max_delta:
                max_delta = bound
    
    if max_delta <= 0:
        return None
    
    delta = max_delta * delta_frac
    if delta < 1e-15:
        return None
    
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    if a_new.min() < 0:
        return None
    
    return a_new
```

"""
Fine-tuning optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Strategy: Very fine LP refinement with adaptive inflation/blending,
gradient smooth-max polish, perturbation restarts, and coordinate descent.
Builds on parent solution to achieve sub-1.503 scores.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0:
        c = c - c.min()
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _fine_lp_refine(a, deadline, max_iter=100000, rng=None):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)
    best_a = a.copy()

    inflate = 1.0005
    blend = 0.25
    no_imp = 0
    stall_total = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _lp_spread(a, M_lp)
        if g is None:
            inflate = min(1.01, inflate * 1.5)
            no_imp += 5
            stall_total += 1
            if stall_total > 100:
                a = best_a.copy()
                inflate = 1.0005
                blend = 0.25
                no_imp = 0
                stall_total = 0
            continue

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            stall_total += 1
            if no_imp % 15 == 0:
                inflate = min(1.005, inflate * 1.01)
                blend = max(0.05, blend * 0.97)
            if no_imp > 300:
                a = best_a.copy()
                inflate = 1.0005
                blend = 0.25
                no_imp = 0
                stall_total = 0
            continue

        g = g / sum_g
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
            stall_total = 0
            inflate = max(1.0 + 1e-6, inflate * 0.97)
            blend = max(0.03, blend * 0.995)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend = max(0.05, blend * 0.95)

        a = a_new

    return best_a, best_score


def _gradient_smooth_max(a, deadline, max_iter=50000, lr=0.015):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-7, 0.01 * c_max)

    z = np.log(a)
    z -= np.mean(z)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())

        x = (c - c_max) / beta
        x = np.clip(x, -50, 50)
        w = np.exp(x)
        w = w / w.sum()

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        grad = 2.0 * h_full[n - 1:2 * n - 1]

        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= np.mean(gz)

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -60, 60)
        z -= np.mean(z)
        e = np.exp(z)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

            best_a = a_new.copy()
            best_z = z.copy()
            no_imp = 0

def _coordinate_descent(a, deadline, max_iter=50000, n_top=5):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    no_imp = 0
    eps = 0.02

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_top]

        pressure = np.zeros(n)
        for k in top_idx:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += 2.0 * a[j]

        max_p = pressure.max()
        if max_p < 1e-15:
            break

        a_new = a * np.exp(-eps * pressure / max_p)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            eps = min(0.15, eps * 1.005)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                eps = max(0.003, eps * 0.92)
            if no_imp > 3000:
                break

        a = a_new

    return best_a, best_score


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _perturb_and_refine(a, deadline, rng, max_iter=3000):
    """Perturb current solution and do quick LP refinement."""
    n = len(a)
    noise_scales = [0.02, 0.05, 0.1, 0.2]
    best = a.copy()
    best_sc = _score(a)

    for ns in noise_scales:
        if time.monotonic() > deadline:
            break
        a_pert = a * (1.0 + ns * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        sc = _score(a_pert)
        if sc > best_sc * 1.1:
            continue

        sub_dl = min(deadline, time.monotonic() + 20.0)
        a_ref, sc_ref = _fine_lp_refine(a_pert, sub_dl, max_iter=max_iter)
        if sc_ref < best_sc:
            best_sc = sc_ref
            best = a_ref.copy()

    return best, best_sc
