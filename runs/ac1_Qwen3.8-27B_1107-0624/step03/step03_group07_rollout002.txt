        # But we need dst + src's partner != peak_idx
        # Actually, let's just try moving to a few candidate positions
        
        # Try moving a small fraction from src to several candidate positions
        frac = 0.01
        delta = a[src] * frac
        
        best_new_cmax = cmax
        best_dst = -1
        
        # Try a few random positions + the position with minimum a
        candidates = np.setdiff1d(np.arange(n), [src])
        # Add the minimum position
        min_pos = int(np.argmin(a))
        if min_pos != src:
            candidates = np.unique(np.append(candidates, min_pos))
        
        # Limit candidates for speed
        if len(candidates) > 50:
            rand_idx = np.random.choice(len(candidates), 50, replace=False)
            candidates = candidates[rand_idx]
        
        for dst in candidates:
            a_trial = a.copy()
            a_trial[src] -= delta
            a_trial[dst] += delta
            if a_trial[src] < 0:
                continue
            a_trial = a_trial / a_trial.sum()
            c_trial = full_convolve_fft(a_trial)
            cm = c_trial.max()
            if cm < best_new_cmax:
                best_new_cmax = cm
                best_dst = dst
        
        if best_dst >= 0 and best_new_cmax < cmax - 1e-15:
            a[src] -= delta
            a[best_dst] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_new_cmax < best_cmax:
                best_cmax = best_new_cmax
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            # Increase fraction slightly
            frac = min(0.05, frac * 1.05)
        else:
            # Try smaller fraction
            frac = max(0.001, frac * 0.7)
            no_improve += 1
            if no_improve > 50:
                break
        
        iters += 1
    
    return best_a


def exact_subgradient_optimize(a0, deadline, rng, initial_step=0.05,
                                peak_width_init=0.02, decay=0.98,
                                restart_period=300):
    """
    Optimization using exact subgradient (with small Gaussian smoothing)
    and mirror descent with adaptive step size.
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    c = full_convolve_fft(a)
    best_cmax = c.max()
    best_score = float(2.0 * n * best_cmax)
    
    step = initial_step
    min_step = 1e-7
    max_step = 0.5
    peak_width = peak_width_init
    no_improve = 0
    
    momentum = np.zeros(n)
    beta_mom = 0.8
    
    while time.monotonic() < deadline:
        # Compute peak weights
        w = peak_weights(c, peak_width)
        
        # Subgradient
        sg = compute_h(w, a)
        
        # Center (project onto tangent space)
        sg_centered = sg - sg.mean()
        
        # Normalize
        sg_norm = np.linalg.norm(sg_centered)
        if sg_norm < 1e-15:
            break
        sg_centered = sg_centered / sg_norm
        
        # Momentum
        momentum = beta_mom * momentum + (1 - beta_mom) * sg_centered
        momentum_norm = np.linalg.norm(momentum)
        if momentum_norm > 1e-15:
            momentum = momentum / momentum_norm
        else:
            momentum = sg_centered
        
        # Try the update
        a_new = mirror_descent_step(a, momentum, step)
        
        # Evaluate
        c_new = full_convolve_fft(a_new)
        cmax_new = c_new.max()
        score_new = float(2.0 * n * cmax_new)
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            c = c_new
            # Slightly increase step
            step = min(max_step, step * 1.02)
        else:
            no_improve += 1
            step = max(min_step, step * 0.92)
            # Revert momentum influence
            momentum *= 0.5
        
        # Adapt peak width
        if no_improve % 50 == 0:
            peak_width = max(1e-5, peak_width * decay)
        
        # Restart
        if no_improve > restart_period:
            # Smooth restart: blend with a smooth profile
            t = np.linspace(0, 1, n)
            smooth = 1.0 + 0.3 * np.cos(2 * np.pi * t)
            smooth = np.maximum(smooth, 1e-10)
            a = 0.7 * best_a + 0.3 * smooth / smooth.sum() * best_a.sum()
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            c = full_convolve_fft(a)
            momentum = np.zeros(n)
            step = initial_step
            peak_width = peak_width_init
            no_improve = 0
    
    return best_a, best_score


def softmax_adam_optimize(a0, deadline, rng, lr=0.03, beta_init=0.03,
                          beta_decay=0.95, restart_period=500):
    """
    Soft-max based Adam optimization (similar to previous but with tuned params).
    """
    n = len(a0)
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    
    z = np.log(a)
    z = z - z.mean()
    
    c = full_convolve_fft(a)
    beta = beta_init * c.max()
    if beta < 1e-10:
        beta = 1e-5
    
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())
    no_improve = 0
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    current_lr = lr
    
    while time.monotonic() < deadline:
        # Soft-max weights
        cmax = c.max()
        x = (c - cmax) / beta
        x = np.clip(x, -60.0, 0.0)
        ex = np.exp(x)
        s = ex.sum()
        if s < 1e-300:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s = 1.0
        w = ex / s
        
        # Gradient
        h = compute_h(w, a)
        grad_a = 2.0 * h
        dot = float(np.dot(a, grad_a))
        dz = a * (grad_a - dot)
        
        # Adam
        t_adam += 1
        m = b1 * m + (1 - b1) * dz
        v = b2 * v + (1 - b2) * (dz * dz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = np.log(best_a)
            z = z - z.mean()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        c = full_convolve_fft(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if t_adam % 30 == 0:
            beta = max(1e-8, beta * beta_decay)
        
        if no_improve > 200:
            current_lr = max(0.005, current_lr * 0.95)
        
        if no_improve > restart_period:
            z = np.log(best_a) + 0.1 * rng.normal(size=n)
            z = z - z.mean()
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = full_convolve_fft(a)
            beta = beta_init * c.max()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            current_lr = lr
            no_improve = 0
    
    return best_a, best_score


def interpolate_profile(p0, n_new):
    """Interpolate profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_raised_cosine(n, alpha=1.0):
    """Raised cosine profile with optional cosine modulation."""
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * (2 * t - 1)))
    a = a ** alpha + 1e-10
    return a / a.sum()


def make_optimized_profile(n, rng):
    """Generate a diverse set of starting profiles."""
    profiles = []
    # Raised cosine variants
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(make_raised_cosine(n, alpha))
    # Linear ramps
    t = np.linspace(0, 1, n)
    for eta in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 + eta * (0.5 - t) + 1e-10
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    # Cosine modulated
    for k in [1, 2, 3, 5]:
        for amp in [0.2, 0.4, 0.6]:
            a = 1.0 + amp * np.cos(2 * np.pi * k * t)
            a = np.maximum(a, 1e-10)
            profiles.append(a / a.sum())
    # Sinusoidal
    for k in [1, 2, 4]:
        a = 1.0 + 0.4 * np.sin(2 * np.pi * k * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    # Random smooth
    for _ in range(8):
        coeffs = rng.uniform(-0.5, 0.5, 5)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    best_seq = None

    def update(p, sc):
        nonlocal best_seq, best_score

            if s > 0.01:
