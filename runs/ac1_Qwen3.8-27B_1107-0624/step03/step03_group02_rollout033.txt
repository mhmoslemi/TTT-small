        # The subgradient at max_idx tells us which coordinates to adjust
        # d(c[max_idx])/d(a[i]) = 2*a[max_idx-i] if 0<=max_idx-i<n
        # To reduce c[max_idx], we should decrease a[i] where a[max_idx-i] is large
        # and increase a[j] where a[max_idx-j] is small
        sg = np.zeros(n)
        for i in range(n):
            k = max_idx - i
            if 0 <= k < n:
                sg[i] = 2 * a[k]

        # the "least contributing" coordinate j (smallest sg[j]*a[j])
        contrib = sg * a
        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))
        
        if i_src == i_dst:
            a = best_a.copy()
            break

        for frac in [0.01, 0.005, 0.001, 0.0005]:
            delta = frac * a[i_src]
            a_new = a.copy()

            a_new[i_dst] += delta

                a = a_new.copy()
                improved = True

Multi-scale step function optimization using Adam in logit-space with smooth-max,
peak-peeling subgradient, and coordinate descent polishing.
Explores n=1024 and n=2048 with diverse parametric initializations.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score1(a):
    """Score assuming a sums to 1."""
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _grad_smooth_max(a, w):
    n = len(a)
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w
    a_rev = np.zeros(N)
    a_rev[:n] = a[::-1]
    fw = np.fft.rfft(w_pad, N)
    fa = np.fft.rfft(a_rev, N)
    corr = np.fft.irfft(fw * fa, N)
    return 2.0 * corr[n - 1:2 * n - 1]

    if s1 < best_sc:
        best_a, best_sc = a1, s1
    print(f"  {label} Adam1: {best_sc:.10f}")

    # Phase 2: Multi-peak peel
    if time.monotonic() < deadline - 40:
        d2 = min(deadline - 40, time.monotonic() + 120)
        a2, s2 = _multi_peel(best_a, d2, max_iter=15000, top_k=3, base_lr=0.08)
        if s2 < best_sc:
            best_a, best_sc = a2, s2
        print(f"  {label} Peel: {best_sc:.10f}")

    # Phase 3: Single-peak peel (fine)
    if time.monotonic() < deadline - 30:
        d3 = min(deadline - 30, time.monotonic() + 80)
        a3, s3 = _peel_subgradient(best_a, d3, max_iter=20000, base_lr=0.05, momentum=0.5)
        if s3 < best_sc:
            best_a, best_sc = a3, s3
        print(f"  {label} Peel2: {best_sc:.10f}")

    # Phase 4: Coordinate polish
    if time.monotonic() < deadline - 10:
        d4 = min(deadline - 10, time.monotonic() + 60)
        a4, s4 = _coord_polish(best_a, d4, max_rounds=150)
        if s4 < best_sc:
            best_a, best_sc = a4, s4
        print(f"  {label} Coord: {best_sc:.10f}")

    # Phase 5: Fine Adam
    if time.monotonic() < deadline - 5:
        d5 = deadline - 5
        a5, s5 = _adam_logit(best_a, d5, max_iter=30000, lr=0.01, beta0=0.008, beta_decay=0.97, beta_min=0.001)
        if s5 < best_sc:
            best_a, best_sc = a5, s5
        print(f"  {label} Adam2: {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-length, multi-init optimization pipeline.
    Uses Adam + peak-peeling + coordinate descent to minimize 2n*max(conv)/sum^2.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = float(p.sum())
            if s > 0.01:
                parent = p / s
                sc = _score1(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define target lengths and time allocation
    # Priority: parent@1024, parent@2048, parametric@1024, parametric@2048
    n_list = [1024, 2048]

    for n in n_list:
        if time.monotonic() > deadline - 300:
            break

        # Parametric sweep for good init
        d_sweep = time.monotonic() + 20
        p_best, p_sc = _sweep_parametric(n, d_sweep)
        print(f"  n={n} parametric best: {p_sc:.10f}")

        # Build list of initializations
        inits = []
        init_labels = []
        if parent is not None:
            p_interp = _interpolate(parent, n)
            inits.append(p_interp)
            init_labels.append("parent")
            # Perturbed parent
            pert = p_interp * (1.0 + 0.1 * rng.normal(size=n))
            pert = np.maximum(pert, 0.0)
            inits.append(pert / pert.sum())
            init_labels.append("parent_pert")
        inits.append(p_best)
        init_labels.append("parametric")
        inits.append(_make_ramp(n, 1.5))
        init_labels.append("ramp1.5")
        inits.append(_make_exp_ramp(n, 2.0))
        init_labels.append("exp2")

        # Optimize each init
        for ii, (a0, lab) in enumerate(zip(inits, init_labels)):
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - 200 - time.monotonic()
            per_init = remaining / max(1, len(inits) - ii)
            if per_init < 60:
                break
            alloc = min(250.0, per_init)
            d = time.monotonic() + alloc

            a_opt, sc = _full_optimize(a0, d, f"n={n}/{lab}")
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()

    # Final focused polish on the best
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"=== Final polish (score={best_score:.10f}) ===")
        # Fine Adam
        d_f1 = min(deadline - 8, time.monotonic() + 80)
        a_f1, s_f1 = _adam_logit(best_seq, d_f1, max_iter=30000, lr=0.005, beta0=0.005, beta_decay=0.98, beta_min=0.001)
        if s_f1 < best_score:
            best_score = s_f1
            best_seq = a_f1
        print(f"  Final Adam: {best_score:.10f}")

        # Coordinate polish
        if time.monotonic() < deadline - 5:
            d_f2 = min(deadline - 5, time.monotonic() + 40)
            a_f2, s_f2 = _coord_polish(best_seq, d_f2, max_rounds=200)
            if s_f2 < best_score:
                best_score = s_f2
                best_seq = a_f2
            print(f"  Final Coord: {best_score:.10f}")

        # Single peak peel (very fine)
        if time.monotonic() < deadline - 3:
            d_f3 = deadline - 3
            a_f3, s_f3 = _peel_subgradient(best_seq, d_f3, max_iter=10000, base_lr=0.02, momentum=0.3)
            if s_f3 < best_score:
                best_score = s_f3
                best_seq = a_f3
            print(f"  Final Peel: {best_score:.10f}")

    if best_seq is None:
        best_seq = _make_ramp(512, 1.0)
        best_score = _score1(best_seq)

    # Normalize and validate
    final = np.asarray(best_seq, dtype=np.float64)
    final = np.maximum(final, 0.0)
    final = np.minimum(final, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_ramp(512, 1.0)
    else:
        final = final / s

    # Validate with evaluate_sequence
    if final.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
            else:
                print(f"evaluate_sequence returned inf, using internal score")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")

    print(f"FINAL: score={best_score:.10f}, n={final.size}")
    return [float(x) for x in final]
