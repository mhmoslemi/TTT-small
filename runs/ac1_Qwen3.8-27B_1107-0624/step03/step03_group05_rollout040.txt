def _score(a):
    """Compute the score efficiently."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    # Use FFT for large n, direct for small
    if n > 128:
        N = 1
        while N < 2*n-1:
            N <<= 1
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2*n-1]
        c = np.maximum(c, 0)  # numerical errors
    else:
        c = np.convolve(a, a)
    max_c = float(c.max())
    return float(2.0 * n * max_c / (s * s))


def _convolve(a):
    """Compute full convolution."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n > 128:
        N = 1
        while N < 2*n-1:
            N <<= 1
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2*n-1]
        c = np.maximum(c, 0)
    else:
        c = np.convolve(a, a)
    return c


def _coord_descent(a, deadline, max_sweeps=5000, verbose=False):
    """
    Exact coordinate descent: for each coordinate, solve the 1D subproblem
    to find the optimal value that minimizes the score.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s  # normalize to sum=1
    
    conv = _convolve(a)
    best_score = _score(a)
    best_a = a.copy()

    no_improve = 0

            #             = conv[k] + (x^2 - ai^2) (if k==2i)
            #             = conv[k] (otherwise)

            sp = 1.0 - ai  # sum of all except a[i]

            # B_k = 2*a[k-i] if 0<=k-i<n and k!=2i, else 0

            # For k where 0<=k-i<n: B[k] = 2*a[k-i]
            k_min = max(0, i)  # k-i >= 0 => k >= i

            if k_min <= k_max:
                idx = np.arange(k_min, k_max+1)
                a_idx = idx - i  # = k-i, ranges from 0 to n-1
                B[idx] = 2.0 * a[a_idx]
            
            # Set D for k=2i
            if 2*i < 2*n-1:
                B[2*i] = 0.0  # Override: at k=2i, the formula is different
                # Actually at k=2i: new_conv = conv - ai^2 + x^2
                # So B[2i] should be 0 and D[2i] = 1
                # And C[2i] = conv[2i] - ai^2
            
            C = conv.copy()
            # For k in [k_min, k_max] where k != 2i: C[k] = conv[k] - B[k]*ai
            if k_min <= k_max:
                idx = np.arange(k_min, k_max+1)
                mask = idx != 2*i
                C[idx[mask]] = conv[idx[mask]] - B[idx[mask]] * ai
            
            if 2*i < 2*n-1:
                C[2*i] = conv[2*i] - ai*ai
            
            # Now: new_conv[k] = C[k] + B[k]*x + D[k]*x^2
            # where D[k] = 1 if k==2i else 0
            # score(x) = 2n * max_k(C[k] + B[k]*x + D[k]*x^2) / (sp + x)^2
            
            # Golden section search
            lo = 0.0
            hi = max(3.0 * ai, 2.0 / n)  # upper bound
            
            # Check if expanding upper bound is needed
            # At x=hi, the score should be reasonable
            
            # Golden ratio
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # ≈ 0.618
            
            def obj1d(x):
                # Compute max_k(C[k] + B[k]*x + (1 if k==2i else 0)*x^2)
                new_max = float(C.max())  # base
                # Add B*x terms
                vals = C + B * x
                if 2*i < 2*n-1:
                    vals[2*i] = C[2*i] + x*x  # D[k]=1 at k=2i
                mx = float(vals.max())
                denom = (sp + x) * (sp + x)
                if denom < 1e-20:
                    return np.inf
                return mx / denom
            
            # Golden section search
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            f1 = obj1d(x1)
            f2 = obj1d(x2)
            
            for _ in range(50):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    f1 = obj1d(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    f2 = obj1d(x2)
            
            x_opt = (lo + hi) / 2.0
            f_opt = obj1d(x_opt)
            
            # Compare with current
            f_current = obj1d(ai)  # should be the current score (times 2n)
            
            if f_opt < f_current - 1e-15:

                # Update conv and score
                conv = _convolve(a)
                new_s = a.sum()
                new_score = 2.0 * n * float(conv.max()) / (new_s * new_s)
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a.copy()

                a = a / a.sum()
                conv = _convolve(a)
        
        if not improved:
            no_improve += 1
            if no_improve > 100:

        else:

        if verbose and sweep % 100 == 0:
            print(f"    Sweep {sweep}: score={best_score:.10f}")

def _lp_refine(a, deadline, max_iter=5000):
    """LP-based refinement."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    a = a / a.sum()
    
    best_a = a.copy()
    conv = _convolve(a)
    best_M = float(conv.max())
    best_score = 2.0 * n * best_M  # since sum=1
    
    # Build constraint matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    inflate = 1.001
    blend = 0.3
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        conv = _convolve(a)
        M = float(conv.max())
        M_lp = M * inflate
        
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M_lp),
                     bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(1.01, inflate * 1.03)
            if no_imp > 300:
                break
            continue
        
        g = g / sum_g
        a_new = (1 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        conv_new = _convolve(a_new)
        M_new = float(conv_new.max())
        score_new = 2.0 * n * M_new
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate = max(1.0005, inflate * 0.97)
        else:
            no_imp += 1
            if no_imp % 50 == 0:
                blend = max(0.05, blend * 0.9)
            if no_imp > 200:
                break
        
        a = a_new
    
    return best_a


def _make_profile(n, type='linear', **kwargs):
    """Create initial profiles."""
    t = np.linspace(0.0, 1.0, n)
    if type == 'linear':

        a = 1.0 + eta * (t - 0.5)
    elif type == 'u':

        a = 1.0 + kwargs.get('a1', 2.0)*t + kwargs.get('a3', -1.0)*t**3
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
