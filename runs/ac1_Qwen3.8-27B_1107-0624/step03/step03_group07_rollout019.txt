def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    n = a.size
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)

def compute_h_fft(w, a):
    n = len(a)
    af = a[::-1]
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def softmax_objective_and_grad(z, beta):
    """Objective and gradient in logit space."""
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    a = a / s
    n = a.size
    
    c = full_convolve_fft(a)
    c_max = c.max()
    x = (c - c_max) / beta
    x = np.clip(x, -50, 0)
    ex = np.exp(x)
    ex_sum = ex.sum()
    w = ex / ex_sum
    soft_max = c_max + beta * np.log(ex_sum)
    
    obj = 2.0 * n * soft_max  # Since sum(a) = 1
    
    h = compute_h_fft(w, a)
    grad_a = 2.0 * h
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z, a, soft_max, w, c

    """L-BFGS-B optimization in logit space."""
    def fun_and_grad(z):
