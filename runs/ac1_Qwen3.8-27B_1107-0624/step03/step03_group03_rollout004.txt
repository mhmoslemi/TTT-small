"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines LSE-Adam gradient, adaptive LP spreading, guided local search,
and multi-length exploration with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    a_n = a / s
    c = _fft_convolve(a_n)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / a.size
    return a / s

    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            A[row, j] = a[j]
    return A, b


def _lp_spread(a, M, top_k=80):
    n = len(a)
    c_obj = -np.ones(n)
    if n > 64:
        A, b = _make_conv_matrix_reduced(a, M, top_k)
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search_blend(a, g, max_iters=25):
    n = len(a)
    best_score = _score_normalized(a)
    gr_ratio = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr_ratio * (hi - lo)
    d = lo + gr_ratio * (hi - lo)
    fc = _score_normalized(np.maximum(a + c * (g - a), 0))
    fd = _score_normalized(np.maximum(a + d * (g - a), 0))
    for _ in range(max_iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr_ratio * (hi - lo)
            fc = _score_normalized(np.maximum(a + c * (g - a), 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr_ratio * (hi - lo)
            fd = _score_normalized(np.maximum(a + d * (g - a), 0))
    best_t = (lo + hi) / 2.0
    return best_t


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.2, use_line_search=True):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_normalized(a)
    no_imp = 0
    inflate_cur = inflate
    blend_cur = blend

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp, top_k=80)
        if g is None:
            inflate_cur = min(1.02, inflate_cur * 1.15)
            no_imp += 5
            if no_imp > 500:
                break
            continue
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.1)
            if no_imp > 600:
                break
            continue
        g = g / sum_g
        if use_line_search:
            t_opt = _line_search_blend(a, g)
            if t_opt < 0.005:
                t_opt = blend_cur * 0.3
        else:
            t_opt = blend_cur
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        score_new = _score_normalized(a_new)
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.02, blend_cur * 0.85)
                inflate_cur = max(inflate, inflate_cur * 0.92)
            if no_imp > max(300, 3 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        a = a_new
    return best_a


def _lse_adam_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -60, 60)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))
    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val
    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -50.0, 50.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    a_new = e / e.sum()
    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a, deadline, max_iter=100000, T_start=0.05, T_end=1e-6, lr=0.1):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    z = np.log(a)
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    best_a = a.copy()
    best_score = _score_normalized(a)
    T = T_start
    no_imp = 0
    base_lr = lr
    if max_iter > 1:
        T_ratio = (T_end / T_start) ** (1.0 / max(max_iter - 1, 1))
    else:
        T_ratio = 1.0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a, z, state, m_val = _lse_adam_step(a, z, state, T, lr)
        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue
        score = _score_normalized(a)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 800 == 0:
                lr = max(0.005, lr * 0.88)
            if no_imp > max(4000, 8 * n):
                rng_r = np.random.default_rng(12345 + it)
                a = best_a * (1.0 + 0.05 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0
        T *= T_ratio
        T = max(T, T_end)
    return best_a


def _subgradient_polish(a, deadline, max_iter=50000, lr=0.02):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_normalized(a)
    v = np.zeros(n)
    momentum = 0.7
    no_imp = 0
    base_lr = lr
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        max_idx = np.where(c >= m_val - 1e-12)[0]
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        score = _score_normalized(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 400 == 0:
                lr = max(0.001, lr * 0.88)
            if no_imp > max(4000, 6 * n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0
    return best_a


def _local_search(a, deadline, max_rounds=200, top_k=15):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_normalized(a)
    rng = np.random.default_rng(777)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        m_val = float(c.max())
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(c / c.sum(), N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        high_idx = np.argsort(grad_tan)[::-1][:top_k]
        low_idx = np.argsort(grad_tan)[:top_k]

        improved = False
        for i in high_idx:
            for j in low_idx:
                if i == j:
                    continue
                max_delta = a[i] * 0.9
                lo_d = 0.0
                hi_d = max_delta
                gr_ratio = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(15):
                    mid1 = hi_d - gr_ratio * (hi_d - lo_d)
                    mid2 = lo_d + gr_ratio * (hi_d - lo_d)
                    a_test1 = a.copy()
                    a_test1[i] -= mid1
                    a_test1[j] += mid1
                    s1 = a_test1.sum()
                    if s1 > 1e-10:
                        a_test1 = a_test1 / s1
                        sc1 = _score_normalized(a_test1)
                    else:
                        sc1 = np.inf
                    a_test2 = a.copy()
                    a_test2[i] -= mid2
                    a_test2[j] += mid2
                    s2 = a_test2.sum()
                    if s2 > 1e-10:
                        a_test2 = a_test2 / s2
                        sc2 = _score_normalized(a_test2)
                    else:
                        sc2 = np.inf
                    if sc1 < sc2:
                        hi_d = mid2
                    else:
                        lo_d = mid1
                best_delta = (lo_d + hi_d) / 2.0
                if best_delta > 1e-6:
                    a_new = a.copy()
                    a_new[i] -= best_delta
                    a_new[j] += best_delta
                    s_new = a_new.sum()
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                        sc_new = _score_normalized(a_new)
                        if sc_new < best_score - 1e-14:
                            best_score = sc_new
                            best_a = a_new.copy()
                            a = a_new
                            improved = True
        if not improved:
            # Random perturbation to escape
            if rnd % 20 == 19:
                a = best_a * (1.0 + 0.02 * rng.normal(size=n))
                a = _normalize(a)
    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    return _normalize(p)


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tent(n):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(2.0 * t - 1.0)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, center_sep=0.25, width=0.12):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.25) / width) ** 2) + np.exp(-0.5 * ((t - 0.75) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_edges(n, edge_weight=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + edge_weight * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gp_sample(n, seed_val, length_scale=0.2):
    rng = np.random.default_rng(seed_val)
    t = np.linspace(0.0, 1.0, n)
    # Simple GP-like smooth random function
    n_freq = 8
    coeffs = rng.normal(size=n_freq)
    phases = rng.uniform(0, 2 * np.pi, size=n_freq)
    a = np.zeros(n)
    for k in range(n_freq):
        freq = (k + 1) / length_scale
        a += coeffs[k] * np.cos(2 * np.pi * freq * t + phases[k]) * np.exp(-k * 0.4)
    a = a - a.min() + 0.1
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LSE-Adam, LP refinement, local search,
    subgradient polish. Explores multiple lengths and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = [1024]
    if parent is not None:
        if parent.size not in lengths:
            lengths.insert(0, parent.size)
    for ln in [768, 1536]:
        if ln not in lengths:
            lengths.append(ln)

    print(f"Lengths to try: {lengths}")

    # Generate candidates per length
    def get_candidates(n_t):
        cands = []
        labels = []
        if parent is not None:
            c = _interpolate_profile(parent, n_t)
            cands.append(c)
            labels.append("parent")
            for pi in range(3):
                p_pert = c * (1.0 + 0.06 * rng.normal(size=n_t))
                p_pert = _normalize(p_pert)
                cands.append(p_pert)
                labels.append(f"parent_pert_{pi}")
        cands.append(_make_gaussian(n_t, 0.3))
        labels.append("gaussian")
        cands.append(_make_gaussian(n_t, 0.2))
        labels.append("gaussian_narrow")
        cands.append(_make_raised_cosine(n_t))
        labels.append("raised_cos")
        cands.append(_make_tent(n_t))
        labels.append("tent")
        cands.append(_make_bimodal(n_t, 0.25, 0.15))
        labels.append("bimodal")
        cands.append(_make_flat_edges(n_t, 0.3))
        labels.append("flat_edges")
        cands.append(_make_flat_edges(n_t, 0.5))
        labels.append("flat_edges_hi")
        cands.append(_make_linear_profile(n_t, 2.0))
        labels.append("linear")
        for gi in range(3):
            cands.append(_make_gp_sample(n_t, 1000 + gi * 7, 0.15 + gi * 0.05))
            labels.append(f"gp_{gi}")
        return cands, labels

    # Phase 1: LSE-Adam on diverse candidates
    print("\n=== Phase 1: LSE-Adam optimization ===")
    best_per_length = {}
    lse_configs = [
        {'T_start': 0.04, 'T_end': 1e-6, 'lr': 0.12},
        {'T_start': 0.08, 'T_end': 1e-5, 'lr': 0.08},
        {'T_start': 0.02, 'T_end': 1e-7, 'lr': 0.15},
    ]

    for n_t in lengths:
        if time.monotonic() > deadline - 250:
            break
        cands, labels = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf

        for ci, (a0, lab) in enumerate(zip(cands, labels)):
            if time.monotonic() > deadline - 250:
                break
            sc0 = _score(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            print(f"  n={n_t}, {lab}: init={sc0:.8f}", end="")

            # Pick best LSE config based on initial score
            cfg_idx = 0 if sc0 < 1.8 else (1 if sc0 < 1.6 else 2)
            cfg = lse_configs[cfg_idx]
            lse_budget = min(120.0, deadline - time.monotonic() - 250.0)
            if lse_budget < 15:
                break
            lse_end = time.monotonic() + lse_budget
            a_lse = _lse_optimize(a0, lse_end, max_iter=60000,
                                  T_start=cfg['T_start'], T_end=cfg['T_end'], lr=cfg['lr'])
            sc_lse = _score(a_lse)
            if sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse
            print(f" -> LSE={sc_lse:.10f}")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best for n={n_t}: {best_here_score:.10f}")

    # Phase 2: LP refinement
    print("\n=== Phase 2: LP refinement ===")
    for n_t in sorted(best_per_length.keys(), key=lambda x: _score(best_per_length[x])):
        if time.monotonic() > deadline - 180:
            break
        a_in = best_per_length[n_t]
        sc_in = _score(a_in)
        print(f"  LP start n={n_t}: score={sc_in:.10f}")
        lp_budget = min(180.0, deadline - time.monotonic() - 180.0)
        if lp_budget < 15:
            continue
        lp_end = time.monotonic() + lp_budget
        a_lp = _lp_refine(a_in, lp_end, max_iter=4000,
                          inflate=1.0008, blend=0.15, use_line_search=True)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP result n={n_t}: score={sc_lp:.10f}")

    # Phase 3: Local search (guided pairwise mass transfer)
    print("\n=== Phase 3: Local search ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        ls_budget = min(150.0, deadline - time.monotonic() - 120.0)
        if ls_budget > 20:
            ls_end = time.monotonic() + ls_budget
            a_ls = _local_search(best_seq, ls_end, max_rounds=150, top_k=12)
            sc_ls = _score(a_ls)
            if sc_ls < best_score:
                best_score = sc_ls
                best_seq = a_ls.copy()
            print(f"  Local search: score={sc_ls:.10f}")

    # Phase 4: Subgradient polish
    print("\n=== Phase 4: Subgradient polish ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        sp_budget = min(100.0, deadline - time.monotonic() - 80.0)
        if sp_budget > 15:
            sp_end = time.monotonic() + sp_budget
            a_sp = _subgradient_polish(best_seq, sp_end, max_iter=60000, lr=0.012)
            sc_sp = _score(a_sp)
            if sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            print(f"  Subgradient: score={sc_sp:.10f}")

    # Phase 5: Final LP polish (very fine)
    print("\n=== Phase 5: Final LP polish ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        lp_budget = min(60.0, deadline - time.monotonic() - 40.0)
        if lp_budget > 10:
            lp_end = time.monotonic() + lp_budget
            a_final = _lp_refine(best_seq, lp_end, max_iter=2000,
                                 inflate=1.0003, blend=0.08, use_line_search=True)
            sc_final = _score(a_final)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"  Final LP: score={sc_final:.10f}")

    # Phase 6: One more LSE pass with fine parameters
    print("\n=== Phase 6: Final LSE ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        lse_budget = min(40.0, deadline - time.monotonic() - 15.0)
        if lse_budget > 5:
            lse_end = time.monotonic() + lse_budget
            a_last = _lse_optimize(best_seq, lse_end, max_iter=20000,
                                   T_start=0.01, T_end=1e-7, lr=0.04)
            sc_last = _score(a_last)
            if sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"  Final LSE: score={sc_last:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(1024) / 1024.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(1024) / 1024.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
