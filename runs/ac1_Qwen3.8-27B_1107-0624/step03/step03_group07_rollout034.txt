```python
def build_conv_matrix(f, n):
    """Build the convolution constraint matrix A where A[k, j] = f[k-j] if valid."""
    # A is (2n-1) x n
    # Row k: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            vals.append(f[i])
    A = scipy.sparse.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A.tocsr()
```

def soft_max_weights(c, beta):
    """Compute soft-max weights on convolution values."""
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    x = np.clip(x, -60.0, 0.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s < 1e-300:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return mval, w


def compute_h(w, a):
    """h[i] = sum_m a[m]*w[i+m] via FFT."""
    n = len(a)
    af = a[::-1]
    N = 1
    while N < n + len(w) - 1:
        N <<= 1
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    return conv[n - 1:2 * n - 1]

def make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (0.5 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_ramp_profile(n, p=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** p + 0.05
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1, amp=0.4):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_bump(n, width=0.5, offset=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - offset) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def interpolate_profile(p0, n_new):
    """Interpolate profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result
