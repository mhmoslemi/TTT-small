def _next_pow2(n):
    k = 1
    while k < n:

    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

    mval = float(c.max())
    x = (c - mval) / beta

    if not np.isfinite(s) or s <= 0:

        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s

    adam_t = 0

    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()
    best_z = z.copy()

    beta = max(beta_factor * float(c.max()), 1e-10)
    no_imp = 0
    base_lr = lr
    lr_cur = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        w = _smooth_max_weights(c_curr, beta)
        grad_a = _grad_smooth_max(a, w)

        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v_arr = b2 * v_arr + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)

        z_new = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))

        e = np.exp(z_new)
        a_new = e / e.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(3000, 5 * n):
                rng_r = np.random.default_rng(7777 + it * 31)
                z = best_z + 0.4 * rng_r.normal(size=n)
                z = np.clip(z, -50.0, 50.0)
                z -= float(np.mean(z))
                e = np.exp(z)
                a = e / e.sum()
                beta = max(1e-10, beta_factor * float(_fft_convolve(a).max()))
                m = np.zeros(n)
                v_arr = np.zeros(n)
                adam_t = 0
                lr_cur = base_lr
                no_imp = 0
                continue

        z = z_new
        a = a_new

        # Cosine learning rate schedule
        progress = it / max(1, max_iter)
        if lr_schedule == "cosine":
            lr_cur = base_lr * 0.5 * (1.0 + np.cos(np.pi * progress))
            lr_cur = max(lr_cur, base_lr * 0.05)
        else:
            if no_imp % 200 == 0:
                lr_cur = max(0.001, lr_cur * 0.95)

        if it % 100 == 0:
            beta = max(1e-12, beta * 0.82)

    return best_a, best_score


def _multi_subgradient(a, deadline, max_iter=12000, top_k=3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    base_lr = 0.25 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c_curr = _fft_convolve(a)
        k = min(top_k, len(c_curr))
        top_indices = np.argsort(c_curr)[::-1][:k]
        max_val = float(c_curr[top_indices[0]])

        weights = np.array([(1.0 + 0.5 * i) for i in range(k)])
        weights = weights / weights.sum()

        sg = np.zeros(n)
        for i, idx in enumerate(top_indices):
            sg += weights[i] * _subgradient_at_max(a, int(idx))

        sg_norm = float(np.linalg.norm(sg))
        if sg_norm < 1e-20:
            break

        lr = base_lr / np.sqrt(it + 1)
        a_new = a - lr * sg / sg_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _lp_spread_topk(a, M, top_k=300):
    n = len(a)
    c = _fft_convolve(a)
    # Select top-k constraints (most binding)
    sorted_idx = np.argsort(c)[::-1][:min(top_k, 2 * n - 1)]
    # Add the center constraint always
    center = n - 1
    if center not in sorted_idx:
        sorted_idx = np.append(sorted_idx, center)

    m = len(sorted_idx)
    A = np.zeros((m, n))
    for row, k in enumerate(sorted_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[row, j_start:j_end + 1] = a[n - 1 - (j_end - j_start):n - 1 - (j_start - j_start) + 1]
            # Actually: A[row, j] = a[j] if k-j is in [0, n-1], i.e., j in [k-n+1, k]
            # c[k] = sum_j a[j] * b[k-j] where b[k-j] is the coefficient
            # In the LP, we fix a and solve for b: constraint is sum_j a[j]*A[k,j]*b[j] <= M
            # where A[k,j] = 1 if k-j is in [0,n-1] else 0
            # So A[row, j] = a[j] for j in [max(0,k-n+1), min(k,n-1)]
            A[row, j_start:j_end + 1] = a[j_start:j_end + 1]

    b_ub = np.full(m, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=2000, inflate=1.002, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    top_k = min(400, 2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread_topk(a, M_lp, top_k)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-6:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.08)
            if no_imp > 100:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 40 == 0:
                blend_cur = max(0.05, blend_cur * 0.85)
            if no_imp > max(150, n // 2):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_cosine_profile(n, alpha=0.8):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, eta=1.2):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_symmetric_profile(n, shape="dome"):
    t = np.linspace(0.0, 1.0, n)
    if shape == "dome":
        a = 1.0 - 0.3 * np.cos(2.0 * np.pi * t)
    elif shape == "u":
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _coordinate_descent_peak(a, deadline, max_sweeps=20):
    """Coordinate descent: for each coordinate, find optimal value to reduce peak."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            # Try a few values around orig
            for scale in [0.7, 0.85, 1.0, 1.15, 1.3]:
                a[i] = orig * scale
                a[i] = max(a[i], 0.0)
                s_new = float(a.sum())
                if s_new < 1e-10:
                    a[i] = orig
                    continue
                a_norm = a / s_new
                c_check = _fft_convolve(a_norm)
                score_check = float(2.0 * n * c_check.max())
                if score_check < best_score - 1e-12:
                    best_score = score_check
                    best_a = a_norm.copy()
                    improved = True
                    a = a_norm.copy()
                    break
            else:
                a[i] = orig  # No improvement, revert
        if not improved:
            break

    return best_a


def _full_optimize(a0, deadline):
    """Run full optimization pipeline on a0."""
    n = len(a0)
    s0 = _score(a0)

    best_a = a0.copy()
    best_sc = s0

    # Phase A: Adam (main)
    if time.monotonic() < deadline - 30:
        dA = min(deadline - 30, time.monotonic() + 0.55 * (deadline - time.monotonic()))
        aA, sA = _adam_smooth_max(a0, dA, max_iter=40000, lr=0.025, beta_factor=0.04)
        if sA < best_sc:
            best_a, best_sc = aA, sA

    # Phase B: LP refinement
    if time.monotonic() < deadline - 20:
        dB = min(deadline - 20, time.monotonic() + 0.2 * (deadline - time.monotonic()))
        aB = _lp_refine(best_a, dB, max_iter=1500, inflate=1.002, blend=0.3)
        sB = _score(aB)
        if sB < best_sc:
            best_a, best_sc = aB, sB

    # Phase C: Adam fine-tune
    if time.monotonic() < deadline - 10:
        dC = min(deadline - 10, time.monotonic() + 0.15 * (deadline - time.monotonic()))
        aC, sC = _adam_smooth_max(best_a, dC, max_iter=20000, lr=0.01, beta_factor=0.015)
        if sC < best_sc:
            best_a, best_sc = aC, sC

    # Phase D: Multi-subgradient
    if time.monotonic() < deadline - 5:
        dD = deadline - 5
        aD, sD = _multi_subgradient(best_a, dD, max_iter=8000, top_k=4)
        if sD < best_sc:
            best_a, best_sc = aD, sD

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-length, multi-strategy optimization. Focuses on n=1024 and n=2048
    with adaptive Adam, LP refinement, and subgradient. Uses parent sequence
    as primary seed plus diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Build initializations for multiple lengths
    lengths = [1024, 2048]

    inits = {}
    for n in lengths:
        cands = []
        if parent is not None:
            p_interp = _interpolate_profile(parent, n)
            cands.append(p_interp)
            # Perturbed version
            p_pert = p_interp * (1.0 + 0.1 * rng.normal(size=n))
            p_pert = np.maximum(p_pert, 0.0)
            cands.append(p_pert / p_pert.sum())
        # Diverse profiles
        cands.append(_make_cosine_profile(n, 0.6))
        cands.append(_make_cosine_profile(n, 1.0))
        cands.append(_make_linear_profile(n, 1.0))
        cands.append(_make_symmetric_profile(n, "dome"))
        inits[n] = cands

    # Phase 1: Quick screening (find best initialization)
    print(f"Phase 1: Quick screening (budget: 200s)")
    screen_deadline = min(deadline, t_start + 200)
    for n in lengths:
        for i, a0 in enumerate(inits[n]):
            if time.monotonic() > screen_deadline:
                break
            quick_budget = min(screen_deadline, time.monotonic() + 15)
            a_q, s_q = _adam_smooth_max(a0, quick_budget, max_iter=2000, lr=0.03, beta_factor=0.05)
            if s_q < best_score:
                best_score = s_q
                best_seq = a_q.copy()
                print(f"  Screen n={n}, init {i}: {s_q:.10f}")

    # Phase 2: Main optimization on best initializations
    print(f"Phase 2: Main optimization (budget: ~900s)")

    # Determine which (n, init) combinations to optimize
    # Prioritize: parent@1024, parent@2048, best found in screening
    main_candidates = []

    # Always include parent interpolations
    if parent is not None:
        main_candidates.append(("parent_1024", _interpolate_profile(parent, 1024)))
        main_candidates.append(("parent_2048", _interpolate_profile(parent, 2048)))

    # Add the best found so far (resized if needed)
    if best_seq is not None:
        main_candidates.append(("best_screen", best_seq.copy()))

    # Add diverse profiles at 1024
    main_candidates.append(("cosine_1024", _make_cosine_profile(1024, 0.8)))
    main_candidates.append(("linear_1024", _make_linear_profile(1024, 1.5)))

    # Deduplicate by length and keep at most 4
    seen = {}
    unique_candidates = []
    for label, a in main_candidates:
        n = a.size
        if n not in seen:
            seen[n] = (label, a)
            unique_candidates.append((label, a))
        elif _score(a) < _score(seen[n][1]):
            seen[n] = (label, a)
            # Replace in unique_candidates
            for i, (l2, a2) in enumerate(unique_candidates):
                if a2.size == n:
                    unique_candidates[i] = (label, a)
                    break

    main_deadline = min(deadline, t_start + 1100)
    per_candidate_budget = (main_deadline - time.monotonic()) / max(1, len(unique_candidates))

    for label, a0 in unique_candidates:
        if time.monotonic() > main_deadline - 60:
            break
        per_budget = min(main_deadline - 60, time.monotonic() + max(60, per_candidate_budget))
        print(f"  Optimizing {label} (n={a0.size}), budget={per_budget - time.monotonic():.0f}s")
        a_opt, s_opt = _full_optimize(a0, per_budget)
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    -> {s_opt:.10f} (new best)")
        else:
            print(f"    -> {s_opt:.10f}")

    # Phase 3: Focused refinement on the best
    print(f"Phase 3: Focused refinement (budget: ~400s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        focus_deadline = deadline - 20
        best_n = best_seq.size
        print(f"  Best so far: score={best_score:.10f}, n={best_n}")

        # Try to improve at current length
        d3a = min(focus_deadline - 30, time.monotonic() + 150)
        a3a, s3a = _adam_smooth_max(best_seq, d3a, max_iter=30000, lr=0.008, beta_factor=0.01)
        if s3a < best_score:
            best_score = s3a
            best_seq = a3a.copy()
            print(f"  Adam fine: {best_score:.10f}")

        # LP refinement
        d3b = min(focus_deadline - 20, time.monotonic() + 100)
        a3b = _lp_refine(best_seq, d3b, max_iter=1000, inflate=1.001, blend=0.25)
        s3b = _score(a3b)
        if s3b < best_score:
            best_score = s3b
            best_seq = a3b.copy()
            print(f"  LP: {best_score:.10f}")

        # Multi-subgradient
        if time.monotonic() < focus_deadline - 10:
            d3c = min(focus_deadline - 10, time.monotonic() + 80)
            a3c, s3c = _multi_subgradient(best_seq, d3c, max_iter=10000, top_k=4)
            if s3c < best_score:
                best_score = s3c
                best_seq = a3c.copy()
                print(f"  MultiSG: {best_score:.10f}")

        # Try interpolation to 2048 and optimize
        if best_n <= 1024 and time.monotonic() < focus_deadline - 40:
            print("  Trying n=2048 interpolation...")
            a_2048 = _interpolate_profile(best_seq, 2048)
            s_2048 = _score(a_2048)
            print(f"  Interpolated to 2048: {s_2048:.10f}")
            d3d = min(focus_deadline - 30, time.monotonic() + 120)
            a3d, s3d = _adam_smooth_max(a_2048, d3d, max_iter=20000, lr=0.015, beta_factor=0.02)
            if s3d < best_score:
                best_score = s3d
                best_seq = a3d.copy()
                print(f"  Adam 2048: {best_score:.10f}")
            # LP on 2048
            if time.monotonic() < focus_deadline - 15:
                d3e = min(focus_deadline - 15, time.monotonic() + 60)
                a3e = _lp_refine(best_seq, d3e, max_iter=500, inflate=1.001, blend=0.2)
                s3e = _score(a3e)
                if s3e < best_score:
                    best_score = s3e
                    best_seq = a3e.copy()
                    print(f"  LP 2048: {best_score:.10f}")

    # Phase 4: Ultra-final polish
    print(f"Phase 4: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        d4 = deadline - 5
        a4, s4 = _adam_smooth_max(best_seq, d4, max_iter=10000, lr=0.003, beta_factor=0.005)
        if s4 < best_score:
            best_score = s4
            best_seq = a4.copy()
            print(f"  Ultra-final: {best_score:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.0)

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"Evaluate_sequence: {sc_eval:.10f}")
        else:
            print(f"Evaluate_sequence returned inf, using internal score: {best_score:.10f}")
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
