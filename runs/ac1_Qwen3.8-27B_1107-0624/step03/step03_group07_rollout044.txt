"""
LP-based iterative refinement combined with gradient polishing for minimizing
2n * max(conv(a,a)) / sum(a)^2. Uses scipy HiGHS LP solver to find optimal
directions, with subgradient polishing and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def full_convolve(a):
    """Full linear convolution, using FFT for large n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def build_conv_matrix(f):
    """Build matrix A where A @ g = conv(f, g). Shape: (2n-1) x n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None


def lp_iterate(f, deadline, verbose=True):
    """
    LP-based iterative optimization.
    f is normalized to sum=1. Minimizes max(conv(f,f)).
    """
    n = len(f)
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.maximum(f, 0.0)
    f = f / f.sum()

    c = full_convolve(f)
    best_f = f.copy()
    best_max = float(c.max())
    best_score = 2.0 * n * best_max

    iteration = 0
    consecutive_no_improve = 0

    while time.monotonic() < deadline:
        c = full_convolve(f)
        M = float(c.max())
        iteration += 1

        # Solve LP
        t0 = time.monotonic()
        g = solve_lp_direction(f, M)
        t_lp = time.monotonic() - t0

        if g is None:
            if verbose:
                print(f"  LP failed at iteration {iteration}")
            break

        g = np.maximum(g, 0.0)
        sum_g = float(g.sum())
        sum_f = float(f.sum())

        if sum_g < 1e-15:
            break

        # If LP solution is essentially the same as f, we've converged
        if sum_g <= sum_f * (1.0 + 1e-10):
            consecutive_no_improve += 1
            if consecutive_no_improve > 20:
                if verbose:
                    print(f"  LP converged at iteration {iteration}, T={t_lp:.3f}s")
                break
            continue

        consecutive_no_improve = 0

        # Scale g to match sum of f
        g = g * (sum_f / sum_g)

        # Line search for step size
        best_t = 0.0
        best_new_max = M
        for t_try in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            f_try = (1.0 - t_try) * f + t_try * g
            f_try = np.maximum(f_try, 0.0)
            s = f_try.sum()
            if s < 1e-15:
                continue
            f_try = f_try / s
            c_try = full_convolve(f_try)
            new_max = float(c_try.max())
            if new_max < best_new_max:
                best_new_max = new_max
                best_t = t_try

        if best_t == 0.0:
            consecutive_no_improve += 1
            if consecutive_no_improve > 30:
                break
            # Try a smaller perturbation
            f = f + 0.001 * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            continue

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-15:
            break
        f = f / s

        c = full_convolve(f)
        new_max = float(c.max())
        score = 2.0 * n * new_max

        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()
            best_max = new_max
            if verbose and iteration % 5 == 0:
                print(f"  Iter {iteration}: max={new_max:.10f}, score={score:.8f}, "
                      f"t={best_t:.3f}, lp_time={t_lp:.2f}s")
        else:
            f = best_f.copy()

    return best_f, best_score


def subgradient_polish(f, deadline, lr_init=0.02, beta_init=None, n_peaks=5):
    """
    Polishing via subgradient/soft-max with adaptive beta.
    f normalized to sum=1.
    """
    n = len(f)
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()

    z = np.log(f)
    z = z - z.mean()

    c = full_convolve(f)
    max_c = float(c.max())
    if beta_init is None:
        beta = max_c * 0.05
    else:
        beta = beta_init

    best_f = f.copy()
    best_score = 2.0 * n * max_c
    best_z = z.copy()
    no_improve = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    lr = lr_init

    while time.monotonic() < deadline:
        # Compute soft-max weights
        mval = float(c.max())
        x = (c - mval) / beta
        x = np.clip(x, -60.0, 0.0)
        ex = np.exp(x)
        s_ex = float(np.sum(ex))
        if s_ex < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ex / s_ex

        # Gradient computation: h[i] = sum_m a[m] * w[i+m]
        af = f[::-1]
        N = 1
        while N < n + len(w) - 1:
            N <<= 1
        faf = np.fft.rfft(af, N)
        fw = np.fft.rfft(w, N)
        conv = np.fft.irfft(faf * fw, N)
        h = conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        dot = float(np.dot(f, grad_a))
        dz = f * (grad_a - dot)

        # Adam update in logit space
        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        f = np.exp(z - z.max())
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
            z = best_z.copy()
            continue
        f = f / s

        c = full_convolve(f)
        max_c = float(c.max())
        score = 2.0 * n * max_c

        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt beta
        if no_improve % 30 == 0 and no_improve > 0:
            beta = max(beta * 0.92, 1e-10)
            lr = max(lr * 0.95, 0.003)

        if no_improve > 500:
            z = best_z + 0.1 * np.random.default_rng(int(time.monotonic() * 1e6) % 100000).normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = full_convolve(f)
            beta = float(c.max()) * 0.05
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = lr_init
            no_improve = 0

    return best_f, best_score


def make_initial_profile(n, profile_type="linear", rng=None):
    """Generate an initial profile."""
    if rng is None:
        rng = np.random.default_rng(42)
    t = np.linspace(0.0, 1.0, n)
    if profile_type == "linear":
        a = 1.0 + 1.5 * (0.5 - t)
    elif profile_type == "cosine":
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
    elif profile_type == "bump":
        a = np.exp(-((t - 0.5) / 0.4) ** 2)
    elif profile_type == "ramp":
        a = (1.0 - t) + 0.2
    elif profile_type == "random_smooth":
        coeffs = rng.uniform(-0.3, 0.3, 5)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2 * np.pi * i * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    result = np.maximum(result, 1e-12)
    s = result.sum()
    return result / s if s > 1e-10 else result
