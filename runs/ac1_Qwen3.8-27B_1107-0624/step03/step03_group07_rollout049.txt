    # grad_a[1] = d/d a1 of (...) = 2*w1*a0 + 2*w2*a1
    # Using formula: 2*sum_j a[j]*w[1+j] = 2*(a[0]*w[1] + a[1]*w[2]) = 2*a0*w1 + 2*a1*w2 ✓
    
    # Great! So grad_a[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # (where w[k] = 0 for k outside [0, 2n-2])
    
    # Now the gradient w.r.t. z (logit space):
    # a = softmax(z), so da/dz_i = a_i * (delta_{ik} - a_k) for each k
    # dL/dz_k = sum_i (dL/da_i) * (da_i/dz_k)
    #         = sum_i grad_a[i] * a_i * (delta_{ik} - a_k)
    #         = a_k * grad_a[k] - a_k * sum_i grad_a[i] * a_i
    #         = a_k * (grad_a[k] - sum_i a_i * grad_a[i])
    
    # So grad_z[k] = a[k] * (grad_a[k] - dot(a, grad_a))
    
    # Compute h[i] = sum_j a[j] * w[i+j] via FFT
    # This is a cross-correlation. Let's define:
    # h = cross_corr(a, w) where (cross_corr(a,w))[i] = sum_j a[j] * w[i+j]
    # = convolve(a_flipped, w)[i + n - 1]
    
    af = a[::-1]  # a flipped
    N = next_pow2(n + len(w) - 1)
    faf = np.fft.rfft(af, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(faf * fw, N)
    h = conv[n-1:2*n-1]  # h[i] = sum_j a[j] * w[i+j] for i=0,...,n-1
    
    grad_a = 2.0 * h
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)
    
    return float(val), grad_z

def lbfgs_optimize(a0, deadline, beta=0.05, maxiter=200):
    """L-BFGS-B optimization in logit space with fixed beta smoothing."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    z0 = np.log(a0)
    z0 = z0 - z0.mean()
    
    best_z = z0.copy()
    best_score = compute_score(a0)
    
    def objective(z):
        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-15:
            return 1e10
        a = a / s
        return compute_score(a)
    
    def objective_smooth(z):
        val, _ = softmax_objective_and_grad(z, beta)
        return val
    
    def gradient_smooth(z):
        _, g = softmax_objective_and_grad(z, beta)
        return g
    
    # Use L-BFGS-B with the smoothed objective
    result = minimize(
        objective_smooth, z0, jac=gradient_smooth,
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 500}
    )
    
    z1 = result.x
    a1 = np.exp(z1 - z1.max())
    a1 = a1 / a1.sum()
    score1 = compute_score(a1)
    
    if score1 < best_score:
        best_score = score1
        best_z = z1.copy()
    
    return a1, best_score

def stochastic_peak_targeting(a0, deadline, rng, n_steps=200):
    """
    Stochastic subgradient method: at each step, pick a random peak and
    take a step to reduce it.
    """
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    lr = 0.01
    step = 0
    while time.monotonic() < deadline and step < n_steps:
        c = full_convolve_fft(a)
        c_max = float(c.max())
        
        # Pick 1-3 random peak locations
        n_peaks = rng.integers(1, 4)
        # Get indices where c is in the top 10%
        threshold = np.percentile(c, 90)
        peak_candidates = np.where(c >= threshold)[0]
        selected = rng.choice(peak_candidates, size=min(n_peaks, len(peak_candidates)), replace=False)
        
        w = np.zeros_like(c)
        for idx in selected:
            w[idx] = 1.0 / len(selected)
        
        # Compute gradient
        af = a[::-1]
        N = next_pow2(n + len(w) - 1)
        faf = np.fft.rfft(af, N)
        fw = np.fft.rfft(w, N)
        conv = np.fft.irfft(faf * fw, N)
        h = conv[n-1:2*n-1]
        grad_a = 2.0 * h
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        
        # Step
        z = z - lr * grad_z
        z = np.clip(z, -40, 40)
        z = z - z.mean()
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        else:
            # Revert
            a = best_a.copy()
            z = np.log(a)
            z = z - z.mean()
            lr = max(0.001, lr * 0.95)
        
        step += 1
    
    return best_a, best_score

def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result

def diverse_profiles(n, rng):
    profiles = []
    t = np.linspace(0.0, 1.0, n)

        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * t)

        profiles.append(a / a.sum())

            a = a + cv * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles

"""
Adaptive L-BFGS-B + stochastic subgradient optimization for minimizing
2n*max(conv(a,a))/sum(a)^2. Uses multi-scale exploration, adaptive beta
smoothing schedule, and population-based restarts to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def softmax_obj_grad(z, beta):
    """
    Compute smoothed objective value and gradient in logit space.
    a = softmax(z/n) normalized to sum 1.
    Objective: 2n * soft_max(conv(a,a), beta)
    """
    n = len(z)
    z_c = z - z.max()
    a = np.exp(z_c)
    s = a.sum()
    if s < 1e-300:
        return 1e10, np.zeros(n)
    a = a / s

    c = full_convolve_fft(a)
    c_max = float(c.max())

    if beta < 1e-14:
        idx = int(np.argmax(c))
        val = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        x = (c - c_max) / beta
        x = np.clip(x, -50.0, 0.0)
        ex = np.exp(x)
        s_ex = float(ex.sum())
        if s_ex < 1e-300:

            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        val = 2.0 * n * (c_max + beta * np.log(s_ex))
        w = ex / s_ex

    # h[i] = sum_j a[j] * w[i+j]
    af = a[::-1]

    grad_a = 2.0 * h
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)

    return float(val), grad_z


def lbfgs_step(a0, beta, maxiter=150, ftol=1e-14, gtol=1e-10):
    """One L-BFGS-B optimization round with fixed beta."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    z0 = np.log(a0)
    z0 = z0 - z0.mean()

    def fun(z):
        v, _ = softmax_obj_grad(z, beta)
        return v

    def grad(z):
        _, g = softmax_obj_grad(z, beta)
        return g

    try:
        res = minimize(fun, z0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': gtol, 'maxfun': maxiter * 5})
        z1 = res.x
    except Exception:
        z1 = z0

    a1 = np.exp(z1 - z1.max())
    s = a1.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a1 = a1 / s
    score = compute_score(a1)
    if not np.isfinite(score):
        return a0.copy(), compute_score(a0)
    return a1, score

def stochastic_refine(a0, deadline, rng, lr_init=0.02, n_peaks_range=(1, 4)):
    """
    Stochastic subgradient method: randomly pick peaks and take steps to reduce them.
    Used to escape local minima.
    """
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    z = np.log(a)
    z = z - z.mean()

    best_a = a.copy()
    best_score = compute_score(a)
    lr = lr_init
    no_improve = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        c_max = float(c.max())

        # Pick random peaks from top 20%
        n_peaks = int(rng.integers(n_peaks_range[0], n_peaks_range[1] + 1))
        threshold = np.percentile(c, 80)
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([int(np.argmax(c))])
        selected = peak_idx[rng.choice(len(peak_idx), size=min(n_peaks, len(peak_idx)), replace=False)]

        w = np.zeros_like(c)
        for idx in selected:
            w[idx] = 1.0 / len(selected)

        # Gradient
        af = a[::-1]
        N = next_pow2(n + len(w) - 1)
        faf = np.fft.rfft(af, N)
        fw = np.fft.rfft(w, N)
        conv = np.fft.irfft(faf * fw, N)
        h = conv[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)

        # Step with momentum
        z = z - lr * grad_z
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        a = np.exp(z - z.max())
        a = a / a.sum()

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.02, lr_init * 2)
        else:
            no_improve += 1
            # Revert
            a = best_a.copy()
            z = np.log(a)
            z = z - z.mean()
            lr = max(0.001, lr * 0.93)
            if no_improve > 500:
                break

    return best_a, best_score


def coordinate_descent_refine(a0, deadline, rng, passes=10):
    """
    Coordinate descent: for each coordinate, try small perturbations and keep the best.
    """
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = compute_score(a)

    for p in range(passes):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            orig = a[i]
            # Try several multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[i] = orig * mult
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-10:
                    continue
                a = a / s
                score = compute_score(a)
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                else:
                    a = best_a.copy()
            a = best_a.copy()
        if not improved:
            break

    return best_a, best_score


def interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_diverse_profiles(n, rng):
    """Generate diverse starting profiles."""
    profiles = []
    t = np.linspace(0.0, 1.0, n)
    # Linear ramps
    for eta in [1.0, 2.0, 3.0, 5.0]:
        a = 1.0 + eta * (0.5 - t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Cosine
    for k in [1, 2, 3]:
        for amp in [0.3, 0.5]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * k * t)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    # Gaussian bump
    for width in [0.3, 0.5, 0.7]:
        for center in [0.3, 0.5, 0.7]:
            a = np.exp(-((t - center) / width) ** 2)
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    # Power law
    for p in [0.5, 1.0, 2.0]:
        a = (t + 0.1) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Random low-freq
    for _ in range(6):
        coeffs = rng.uniform(-0.4, 0.4, 5)
        a = np.ones(n)
        for i, cv in enumerate(coeffs, 1):
            a = a + cv * np.cos(2 * np.pi * i * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search: adaptive L-BFGS-B with beta annealing,
    stochastic subgradient escape, coordinate descent polish,
    and multi-scale exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"=== Starting search. Budget={budget_s}s ===")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial sequence: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Primary optimization on initial sequence =====
    if init_seq is not None:
        n_init = init_seq.size
        print(f"\n--- Phase 1: Adaptive L-BFGS on n={n_init} ---")
        dl1 = min(deadline, t_start + 500.0)
        a, sc = adaptive_lbfgs(init_seq, dl1, rng, n_betas=8)
        update(a, sc)
        print(f"  Adaptive L-BFGS: {sc:.10f}")

        # Stochastic refinement
        if time.monotonic() < deadline - 100:
            dl1b = min(deadline, time.monotonic() + 200.0)
            a_st, sc_st = stochastic_refine(a, dl1b, rng, lr_init=0.015)
            update(a_st, sc_st)
            print(f"  Stochastic refine: {sc_st:.10f}")
            if sc_st < sc:
                a = a_st

        # Coordinate descent
        if time.monotonic() < deadline - 50:
            dl1c = min(deadline, time.monotonic() + 60.0)
            a_cd, sc_cd = coordinate_descent_refine(a, dl1c, rng, passes=5)
            update(a_cd, sc_cd)
            print(f"  Coordinate descent: {sc_cd:.10f}")

    # ===== Phase 2: Explore other scales =====
    print(f"\n--- Phase 2: Multi-scale exploration ---")
    n_candidates = [256, 384, 512, 768, 1024, 1536, 2048, 3072]
    if init_seq is not None:
        n_init = init_seq.size
        # Sort by proximity to current best
        n_candidates = [x for x in n_candidates if x != n_init]
        n_candidates.sort(key=lambda x: abs(x - n_init))
        n_candidates = n_candidates[:5]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        # Get starting point
        if best_seq is not None:
            p = interpolate_profile(best_seq, n_try)
        elif init_seq is not None:
            p = interpolate_profile(init_seq, n_try)
        else:
            p = make_diverse_profiles(n_try, rng)[0]

        sc_p = compute_score(p)
        print(f"  n={n_try}: interp score={sc_p:.10f}", end="")

        if sc_p < best_score + 0.03:
            dl = min(deadline, time.monotonic() + min(150, remaining * 0.4))
            a, sc = adaptive_lbfgs(p, dl, rng, n_betas=6)
            update(a, sc)
            print(f" -> L-BFGS: {sc:.10f}")
        else:
            print(f" (skipped, too far from best)")

    # ===== Phase 3: Diverse restarts at best scale =====
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = best_seq.size
        print(f"\n--- Phase 3: Diverse restarts at n={n_best} ---")

        profiles = make_diverse_profiles(n_best, rng)
        # Also add perturbed versions of best
        for _ in range(3):
            p_pert = best_seq + 0.05 * rng.normal(size=n_best)
            p_pert = np.maximum(p_pert, 1e-10)
            p_pert = p_pert / p_pert.sum()
            profiles.append(p_pert)

        for idx, p in enumerate(profiles[:6]):
            if time.monotonic() > deadline - 60:
                break
            sc_p = compute_score(p)
            if sc_p > best_score + 0.02:
                continue
            dl = min(deadline, time.monotonic() + 80.0)
            a, sc = adaptive_lbfgs(p, dl, rng, n_betas=5)
            update(a, sc)
            if sc < best_score + 0.005:
                print(f"  Restart {idx} (sc_p={sc_p:.6f}): {sc:.10f}")

    # ===== Phase 4: Final intensive polish =====
    if best_seq is not None:
        n_final = best_seq.size
        print(f"\n--- Phase 4: Final polish at n={n_final} ---")

        a = best_seq.copy()

        # Very fine L-BFGS
        for beta_scale in [0.01, 0.005, 0.003, 0.001, 0.0005]:
            if time.monotonic() > deadline - 15:
                break
            c = full_convolve_fft(a)
            c_max = float(c.max())
            beta = beta_scale * c_max
            dl = min(deadline, time.monotonic() + 40.0)
            a_new, sc = lbfgs_step(a, beta, maxiter=100)
            if sc < best_score:
                best_score = sc
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  Fine L-BFGS (beta={beta:.6f}): {sc:.10f}")

        # Stochastic with very small steps
        if time.monotonic() < deadline - 20:
            dl_st = min(deadline, time.monotonic() + 30.0)
            a_st, sc_st = stochastic_refine(a, dl_st, rng, lr_init=0.005,
                                            n_peaks_range=(1, 2))
            update(a_st, sc_st)
            print(f"  Fine stochastic: {sc_st:.10f}")

        # Final coordinate descent
        if time.monotonic() < deadline - 10:
            dl_cd = min(deadline, time.monotonic() + 20.0)
            a_cd, sc_cd = coordinate_descent_refine(best_seq, dl_cd, rng, passes=3)
            update(a_cd, sc_cd)
            print(f"  Final coord descent: {sc_cd:.10f}")

    # ===== Phase 5: Try to find even better at nearby scales =====
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_cur = best_seq.size
        nearby = [n_cur // 2, n_cur * 2, n_cur - (n_cur // 16), n_cur + (n_cur // 16)]
        nearby = [x for x in nearby if 64 <= x <= 4096 and x != n_cur]
        for n_t in nearby[:2]:
            if time.monotonic() > deadline - 10:
                break
            p = interpolate_profile(best_seq, n_t)
            sc_p = compute_score(p)
            if sc_p < best_score:
                dl = min(deadline, time.monotonic() + 30.0)
                a, sc = adaptive_lbfgs(p, dl, rng, n_betas=4)
                update(a, sc)
                print(f"  Nearby n={n_t}: {sc:.10f}")

    print(f"\n=== Final best score: {best_score:.10f} ===")

    # Final validation
    if best_seq is not None:
        final = best_seq
    else:
        final = np.ones(512) / 512.0

    final = np.maximum(final, 0.0)
    final = np.minimum(final, 1000.0)
    s = float(final.sum())
    if s > 0.01:
        final = final / s
    else:
        final = np.ones(256) / 256.0

    final_score = compute_score(final)
    print(f"FINAL: n={final.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final]
