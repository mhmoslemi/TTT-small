import numpy as np

import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Assume sum(a)=1, return 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()

    return a / s

    """Build sparse matrix A where (A @ g) = f * g (full convolution)."""
    n = len(f)
    A = lil_matrix((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        row_start = j  # k >= j (since k-j >= 0)
        row_end = min(2 * n - 2, j + n - 1)  # k-j <= n-1
        A[row_start:row_end + 1, j] = f[j]
    return A.tocsr()


def _solve_lp(f, M, top_k_frac=1.0):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for selected k, g >= 0.
    top_k_frac: fraction of constraints to use (1.0 = all).
    Returns normalized g or None.
    """
    n = len(f)
    conv = _full_convolve(f, f)
    
    if top_k_frac < 1.0:
        # Only use top fraction of constraints
        num_constraints = max(n, int(len(conv) * top_k_frac))
        sorted_idx = np.argsort(conv)[::-1][:num_constraints]
        sorted_idx = np.sort(sorted_idx)
    else:
        sorted_idx = np.arange(2 * n - 1)
    
    # Build reduced constraint matrix
    n_rows = len(sorted_idx)
    A = lil_matrix((n_rows, n), dtype=np.float64)
    for i, k in enumerate(sorted_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[i, j] = f[j]
    A = A.tocsr()
    
    b_ub = np.full(n_rows, M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    
    if res.success and res.x is not None:
        g = res.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def _lbfgs_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    exp_vals = np.exp(c_shifted)
    log_sum = float(np.log(np.sum(exp_vals)))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros(len(c))
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(p_init, deadline, n_levels=15, beta_start=0.1, beta_end=0.00005):
    n = len(p_init)
    z = np.log(np.maximum(p_init, 1e-15))
    z = z - z.mean()
    best_a = p_init.copy()
    best_score = _score_norm_fn(p_init)
    c0 = _full_convolve(p_init, p_init)
    max_c0 = float(c0.max())
    if max_c0 < 1e-15:
        return best_a, best_score
    log_ratios = np.linspace(np.log(beta_start), np.log(beta_end), n_levels)
    betas = max_c0 * np.exp(log_ratios)
    remaining = deadline - time.monotonic()
    if remaining <= 0:
        return best_a, best_score
    per_level = max(remaining / n_levels, 1.0)
    for level, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        level_dl = time.monotonic() + per_level
        try:
            result = minimize(
                _lbfgs_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lbfgs_gradient,
                options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 25, 'maxfun': 1500}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _lp_refine(p, deadline, max_iters=200, top_frac=0.5):
    n = len(p)
    p = _normalize(p.copy())
    best_p = p.copy()
    best_score = _score_norm_fn(p)
    stagnant = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = float(_full_convolve(p, p).max())
        g = _solve_lp(p, M, top_frac=top_frac)
        if g is None:
            break
        if np.allclose(g, p, atol=1e-8):
            stagnant += 1
            if stagnant > 3:
                break
            continue
        best_t = 0.0
        best_sc = best_score
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            p_new = (1.0 - t) * p + t * g
            p_new = np.maximum(p_new, 0.0)
            p_new = _normalize(p_new)
            sc = _score_norm_fn(p_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        if best_t > 0:
            p = (1.0 - best_t) * p + best_t * g
            p = np.maximum(p, 0.0)
            p = _normalize(p)
            sc = _score_norm_fn(p)
            if sc < best_score:
                best_score = sc
                best_p = p.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
            if stagnant > 5:
                break
    return best_p, best_score


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_hann_profile(n):
    t = np.linspace(0.0, np.pi, n)
    a = 0.5 + 0.5 * np.cos(t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, sigma):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5) ** 2) / (2.0 * sigma * sigma))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_profile(n, k):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.3 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS log-sum-exp continuation + LP active-set refinement.
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        update_best(parent, sc_parent, "parent")
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
    else:
        n_parent = 2048
        parent = _make_linear_profile(n_parent, 1.5)
        sc_parent = _score_norm_fn(parent)
        update_best(parent, sc_parent, "default_linear")
        print(f"No parent, n={n_parent}, score={sc_parent:.10f}")

    n = n_parent

    # Phase 0: Diverse initializations
    phase0_end = min(deadline, t_start + 20.0)
    candidates = []
    profiles = [
        ("lin1.0", _make_linear_profile(n, 1.0)),
        ("lin1.5", _make_linear_profile(n, 1.5)),
        ("lin2.0", _make_linear_profile(n, 2.0)),
        ("lin2.5", _make_linear_profile(n, 2.5)),
        ("lin3.0", _make_linear_profile(n, 3.0)),
        ("hann", _make_hann_profile(n)),
        ("gauss0.3", _make_gaussian_profile(n, 0.3)),
        ("gauss0.25", _make_gaussian_profile(n, 0.25)),
        ("gauss0.2", _make_gaussian_profile(n, 0.2)),
        ("cos1", _make_cosine_profile(n, 1)),
        ("cos2", _make_cosine_profile(n, 2)),
        ("cos3", _make_cosine_profile(n, 3)),
        ("quad1_0.3", _make_quadratic_profile(n, 1.0, 0.3)),
        ("quad1.5_m0.2", _make_quadratic_profile(n, 1.5, -0.2)),
        ("cubic1_m0.5", _make_cubic_profile(n, 1.0, -0.5)),
        ("cubic2_m1", _make_cubic_profile(n, 2.0, -1.0)),
    ]
    for name, p in profiles:
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, name))
        update_best(p, sc, f"init_{name}")

    # Add parent and small perturbations
    for pname, noise_scale in [("parent", 0.0), ("perturb01", 0.05), ("perturb03", 0.1)]:
        z = np.log(np.maximum(parent, 1e-15))
        z = z - z.mean()
        if noise_scale > 0:
            z = z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
        p = _softmax_fn(z)
        sc = _score_norm_fn(p)
        candidates.append((sc, p, pname))
        update_best(p, sc, f"init_{pname}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # Phase 1: L-BFGS on top candidates
    phase1_end = min(deadline, t_start + 500.0)
    n_attempts = 0
    for idx, (sc0, p, name) in enumerate(candidates):
        if time.monotonic() > phase1_end:
            break
        if n_attempts >= 5:
            break
        if sc0 > best_score * 1.02:
            break
        per_time = min(90.0, (phase1_end - time.monotonic()) - 5.0)
        if per_time < 10:
            break
        dl = time.monotonic() + per_time
        p_best, sc_best = _run_lbfgs(p, dl, n_levels=12, beta_start=0.1, beta_end=0.0001)
        update_best(p_best, sc_best, f"lbfgs1_{name}")
        n_attempts += 1

    print(f"Phase 1 done: best={best_score:.10f}")

    # Phase 2: LP refinement on best
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        p_work = _normalize(best_seq.copy())
        per_dl = phase2_end - 5
        p_best, sc_best = _lp_refine(p_work, per_dl, max_iters=150, top_frac=0.5)
        update_best(p_best, sc_best, "lp_refine")
        print(f"Phase 2 done: best={best_score:.10f}")

        # Second LP pass with tighter constraints
        if time.monotonic() < phase2_end - 30 and best_seq is not None:
            p_work2 = _normalize(best_seq.copy())
            per_dl2 = phase2_end - 10
            p_best2, sc_best2 = _lp_refine(p_work2, per_dl2, max_iters=100, top_frac=0.3)
            update_best(p_best2, sc_best2, "lp_tight")

    # Phase 3: L-BFGS with very small beta after LP
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        p_work = _normalize(best_seq.copy())
        per_dl = min(phase3_end - 5, time.monotonic() + 150.0)
        p_best, sc_best = _run_lbfgs(p_work, per_dl, n_levels=10, beta_start=0.01, beta_end=0.00002)
        update_best(p_best, sc_best, "lbfgs3_fine")
        print(f"Phase 3 done: best={best_score:.10f}")

    # Phase 4: Multi-restart on best
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_restarts = 4
        n_best = len(best_seq)
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 30:
                break
            noise_scales = [0.05, 0.1, 0.15, 0.2]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_scales[i] * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            per_dl = min(phase4_end - 10, time.monotonic() + 50.0)
            p_best, sc_best = _run_lbfgs(p_start, per_dl, n_levels=8, beta_start=0.05, beta_end=0.0001)
            update_best(p_best, sc_best, f"restart{i}")
        print(f"Phase 4 done: best={best_score:.10f}")

    # Phase 5: Try different length
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        n_curr = len(best_seq)
        # Try n/2
        if n_curr >= 512:
            n_small = n_curr // 2
            p_interpol = _interp_profile(best_seq, n_small)
            sc_interpol = _score_norm_fn(p_interpol)
            update_best(p_interpol, sc_interpol, f"interp_n{n_small}")
            if time.monotonic() < phase5_end - 20:
                per_dl = min(phase5_end - 10, time.monotonic() + 40.0)
                p_best, sc_best = _run_lbfgs(p_interpol, per_dl, n_levels=8, beta_start=0.05, beta_end=0.0001)
                update_best(p_best, sc_best, f"lbfgs_n{n_small}")

        # Try 2n
        if n_curr <= 1024 and time.monotonic() < phase5_end - 60:
            n_large = min(4096, n_curr * 2)
            p_interpol = _interp_profile(best_seq, n_large)
            sc_interpol = _score_norm_fn(p_interpol)
            update_best(p_interpol, sc_interpol, f"interp_n{n_large}")
            if time.monotonic() < phase5_end - 20:
                per_dl = min(phase5_end - 10, time.monotonic() + 50.0)
                p_best, sc_best = _run_lbfgs(p_interpol, per_dl, n_levels=8, beta_start=0.05, beta_end=0.0001)
                update_best(p_best, sc_best, f"lbfgs_n{n_large}")

        print(f"Phase 5 done: best={best_score:.10f}")

    # Phase 6: Final LP + L-BFGS polish
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        p_work = _normalize(best_seq.copy())
        per_dl = min(phase6_end - 20, time.monotonic() + 40.0)
        p_best, sc_best = _lp_refine(p_work, per_dl, max_iters=50, top_frac=0.3)
        update_best(p_best, sc_best, "final_lp")

        if time.monotonic() < phase6_end - 5 and best_seq is not None:
            p_work = _normalize(best_seq.copy())
            per_dl = min(phase6_end - 2, time.monotonic() + 15.0)
            p_best, sc_best = _run_lbfgs(p_work, per_dl, n_levels=5, beta_start=0.005, beta_end=0.00001)
            update_best(p_best, sc_best, "final_lbfgs")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_linear_profile(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            if len(final_seq) <= 8192:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Evaluate failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear_profile(512, 1.5)
    print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
