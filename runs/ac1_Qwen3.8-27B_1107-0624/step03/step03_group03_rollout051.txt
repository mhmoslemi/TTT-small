"""

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_norm(a):
    """Score with sum(a)=1 assumed: 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    a = a / s
    return _score_norm(a)

    best_score = _score_norm(a)

    v = np.zeros(n)  # velocity
    a_prev = a.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        a_look = a - momentum * v if it > 0 else a.copy()
        a_look = np.maximum(a_look, 1e-12)
        a_look = a_look / a_look.sum()

        # Compute log-sum-exp gradient at lookahead point
        c = _fft_convolve(a_look)
        m_val = float(c.max())
        x = (c - m_val) / T
        x = np.clip(x, -50, 50)
        ex = np.exp(x)
        s = float(ex.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            s = 1.0
        else:
            w = ex / s

        N = _next_pow2(2 * n - 1)

        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]

        dot = float(np.dot(a, grad))

        state['v_mat'] = vv

        c_new = _fft_convolve(a_new)

def _lp_spread_reduced(a, M, top_k=80):
    """LP spread with reduced constraints."""
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[row, lo:hi] = a[lo:hi]
    
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

        hi = 1.0
        cc = hi - gr * (hi - lo)
        dd = lo + gr * (hi - lo)

                dd = cc
                fd = fc
                cc = hi - gr * (hi - lo)
                fc = _score_norm(np.maximum(a + cc * (g - a), 0))
            else:
                lo = cc
                cc = dd
                fc = fd
                dd = lo + gr * (hi - lo)
                fd = _score_norm(np.maximum(a + dd * (g - a), 0))
        t_opt = (lo + hi) / 2.0

        if t_opt < 0.01:
            t_opt = cur_blend * 0.5

        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        score_new = _score_norm(a_new)

        if score_new < best_score - 1e-14:

            best_a = a_new.copy()

        else:

                a = best_a.copy()
                cur_blend = blend
                cur_inflate = inflate
                no_imp = 0

        a = a_new

    return best_a, best_score


def _interpolate_to(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _spectral_equalize(a, deadline, max_iter=20000, lr=0.05):
    """Spectral phase adjustment to flatten the convolution spectrum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    no_imp = 0

    N = _next_pow2(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute current spectrum
        fa = np.fft.rfft(a, N)
        spectrum = fa * fa  # This is the "spectrum" of the convolution (complex)
        # The real part of IFFT(spectrum) is the convolution
        cov = np.fft.irfft(spectrum, N)[:2*n-1]
        
        # Compute error
        target = 1.0 / (2 * n - 1)
        error = cov - target
        
        # We want to modify a to reduce the error
        # Gradient approach: same as flat_conv_optimize but with spectral insight
        fe = np.fft.rfft(error, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fe * fa_rev, N)
        h = h_full[n-1:2*n-1]
        grad = 2.0 * h
        
        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a
        
        a_new = a - lr * grad_tan
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        score_new = _score_norm(a_new)
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
            lr = min(0.1, lr * 1.01)  # slightly increase lr on success
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.005, lr * 0.9)
            if no_imp > 3000:
                a = best_a.copy()
                lr = 0.05
                no_imp = 0

    return best_a, best_score
