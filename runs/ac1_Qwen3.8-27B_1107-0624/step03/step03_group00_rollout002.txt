import time

    while k < n:

    n = a.size

def _softmax_from_z(z, floor=1e-12):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:

def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-10:

def _make_symmetric(p):
    """Make a sequence symmetric: a[i] = a[n-1-i]"""
    n = len(p)
    half = p[: (n + 1) // 2].copy()
    if n % 2 == 0:
        p_sym = np.concatenate([half, half[::-1]])
    else:
        p_sym = np.concatenate([half, half[-2::-1]])
    return p_sym / p_sym.sum()

def _make_profile_smooth(n, alpha, beta):
    """Create a smooth profile: (1+t)^alpha * (1-t)^beta on [-1,1]"""
    t = np.linspace(-1.0, 1.0, n)
    a = np.power(1.0 + t, alpha) * np.power(1.0 - t, beta)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    a = np.exp(-0.5 * (t / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    t = np.linspace(-1.0, 1.0, n)
    a = np.ones(n)
    for k, c in coeffs:

    a = np.maximum(a, 1e-12)
    return a / a.sum()

    kr = np.outer(p1, p2)
    kr = kr.ravel()
    n_target = n1 * n2

    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def _conv_grad(w, a):
    """Gradient of smooth_max wrt a, given weights w on convolution values.
    d/d a[i] of sum_k w[k] * c[k] where c = a*a
    = sum_k w[k] * (a[k-i] + a[k-i]) ... 
    Actually: c[k] = sum_j a[j]*a[k-j]
    dc[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (when both indices valid)
    But more carefully: dc[k]/da[i] = a[k-i] (from the j=i term) + a[k-i] (from the k-j=i term, i.e., j=k-i)
    Wait: c[k] = sum_j a[j] * a[k-j]
    dc[k]/da[i] = a[k-i] (when j=i) + sum_j a[j] * da[k-j]/da[i] = a[k-i] + a[k-i] = 2*a[k-i]
    But only when 0 <= k-i < n.
    
    So the gradient is: grad[i] = sum_k w[k] * 2 * a[k-i] (where 0 <= k-i < n)
    = 2 * (w * a_reversed)[something]
    
    Actually, this is a cross-correlation: grad[i] = 2 * sum_k w[k] * a[k-i] (valid indices)
    = 2 * (w ⊛ a)[i] where ⊛ is cross-correlation
    = 2 * (ifft(fft(w) * conj(fft(a))[reversed]))[i]
    
    Simpler: grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
    Let j = k-i, so k = i+j:
    grad[i] = 2 * sum_{j: 0<=i+j<2n-1, 0<=j<n} w[i+j] * a[j]
    = 2 * cross_corr(w, a)[i]
    
    Where cross_corr(w, a)[i] = sum_j w[i+j] * a[j]
    This is the same as convolving w with reversed a, shifted.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    # h[m] = sum_j w[m-j] * a_rev[j] = sum_j w[m-j] * a[n-1-j]
    # We want: cross_corr[i] = sum_j w[i+j] * a[j]
    # Let's verify: h[m] = sum_j w[m-j] * a[n-1-j]
    # Let p = n-1-j, j = n-1-p:
    # h[m] = sum_p w[m-(n-1-p)] * a[p] = sum_p w[m-n+1+p] * a[p]
    # We want cross_corr[i] = sum_j w[i+j] * a[j]
    # Set i+j = m-n+1+p, j = p: i+p = m-n+1+p, so m = i+n-1
    # So cross_corr[i] = h[i+n-1]
    return 2.0 * h[n-1:2*n-1]

def _adam_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                   no_imp_limit=300, use_symmetry=False):
    """Adam optimizer in logit space with smooth-max objective"""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    
    z = np.log(p0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    
    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)
    
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    
    while time.monotonic() < deadline:
        _, w = _smooth_max_weights(c, beta)
        g = _conv_grad(w, p)
        
        # Project gradient to sum-zero (tangent to probability simplex)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        
        if use_symmetry:
            p = _make_symmetric(p)
        
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.003, lr * 0.95)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        
        if t_adam % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.5)
        
        if no_imp > no_imp_limit:
            # Restart with noise
            ns = 0.1 if best_score < 1.52 else 0.2
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            if use_symmetry:
                p = _make_symmetric(p)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0
    
    return best_p, best_score

def _lp_refine_sca(a, deadline, max_iters=50, solver='auto'):
    """SCA-based LP refinement using CVXPY.
    At each step, solve:
      minimize t
      subject to: 2*(a*b)[k] - (a*a)[k] <= t for all k
                   sum(b) = 1
                   b >= 0
    where a is the current iterate.
    """
    n = len(a)
    current = a.copy()
    c = _full_convolve(current)
    M = float(c.max())
    best = current.copy()
    best_score = 2.0 * n * M
    
    # Build the convolution matrix A where (A @ b)[k] = (a * b)[k]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        rows.extend([k] * len(idx))
        cols.extend(idx.tolist())
        vals.extend(a[k - idx].tolist())
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c_cur = _full_convolve(current)
        c_max = float(c_cur.max())
        
        try:
            b = cp.Variable(n)
            t_var = cp.Variable()
            constraints = [
                cp.sum(b) == 1,
                b >= 0,
                2.0 * (A @ b) - c_cur <= t_var
            ]
            obj = cp.Minimize(t_var)
            prob = cp.Problem(obj, constraints)
            
            # Try different solvers
            try:
                prob.solve(solver=cp.MOSEK)
            except Exception:
                try:
                    prob.solve(solver=cp.CLARABEL)
                except Exception:
                    prob.solve(solver=cp.ECOS)
            
            if b.value is not None and np.all(np.isfinite(b.value)):
                b_new = np.maximum(b.value, 0.0)
                s = float(b_new.sum())
                if s > 1e-10:
                    b_new = b_new / s
                    # Blend
                    t_blend = 0.5
                    current_new = (1 - t_blend) * current + t_blend * b_new
                    current_new = np.maximum(current_new, 1e-15)
                    current_new = current_new / current_new.sum()
                    
                    c_new = _full_convolve(current_new)
                    sc_new = 2.0 * n * float(c_new.max())
                    
                    if sc_new < best_score:
                        best_score = sc_new
                        best = current_new.copy()
                    
                    current = current_new
            else:
                break
        except Exception as e:
            print(f"  LP SCA error at iter {it}: {e}")
            break
    
    return best, best_score

def _lp_refine_orig(a, deadline, t_blend=0.3, max_solves=20):
    """Original LP refinement: max sum(b) s.t. (a*b)[k] <= M, b >= 0"""
    n = len(a)
    current = a.copy()
    c = _full_convolve(current)
    M = float(c.max())
    best = current.copy()
    best_score = 2.0 * n * M
    n_solves = 0
    t = t_blend
    
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        c = _full_convolve(current)
        M = float(c.max())
        
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(current[k - idx].tolist())
        
        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        except Exception:
            break
        
        if res.success:
            b = np.maximum(res.x, 0.0)
            s_b = float(b.sum())
            if s_b < 1e-10:
                break
            b_norm = b / s_b
            a_new = (1.0 - t) * current + t * b_norm
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            
            c_new = _full_convolve(a_new)
            sc_new = 2.0 * n * float(c_new.max())
            
            if sc_new < best_score:
                best_score = sc_new
                best = a_new.copy()
            
            current = a_new
            t = max(0.1, t * 0.9)
        else:
            break
    
    return best, best_score

def _peak_reduction(a, n_iters=100, deadline=None):
    """Reduce peak convolution by reducing coordinates that contribute to the peak"""
    n = len(a)
    current = a.copy()
    c = _full_convolve(current)
    best = current.copy()
    best_score = 2.0 * n * float(c.max())
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(current)
        peak_val = float(c.max())
        peak_pos = int(np.argmax(c))
        
        # Find coordinates that contribute most to the peak
        # c[peak_pos] = sum_i current[i] * current[peak_pos - i] (valid indices)
        contributions = np.zeros(n)
        for i in range(n):
            j = peak_pos - i
            if 0 <= j < n:
                contributions[i] = current[i] * current[j]
        
        # Reduce the top contributors
        top_k = max(5, n // 32)
        top_idx = np.argsort(contributions)[::-1][:top_k]
        
        # Reduce these coordinates
        reduction = 0.1
        for i in top_idx:
            current[i] = max(0.0, current[i] * (1.0 - reduction))
        
        # Renormalize
        s = float(current.sum())
        if s < 1e-10:
            break
        current = current / s
        
        c = _full_convolve(current)
        sc = 2.0 * n * float(c.max())
        if sc < best_score:
            best_score = sc
            best = current.copy()
    
    return best, best_score

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}]")
    
    # Get parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    print(f"Parent n={parent.size if parent is not None else 0}, budget={budget_s}s")
    
    # Phase 1: Generate diverse initial candidates
    candidates = []
    
    # Try different lengths
    lengths = [512, 768, 1024, 1536, 2048]
    
    for n in lengths:
        # Beta-like profiles: (1+t)^alpha * (1-t)^beta
        for alpha in [0.5, 1.0, 1.5]:
            for beta_p in [0.5, 1.0, 1.5]:
                if alpha == beta_p:
                    continue  # Skip symmetric (handled separately)
                p = _make_profile_smooth(n, alpha, beta_p)
                candidates.append((p, f'beta_{alpha}_{beta_p}_n{n}'))
        
        # Gaussian profiles
        for sigma in [0.3, 0.5, 0.7, 1.0]:
            p = _make_profile_gaussian(n, sigma)
            candidates.append((p, f'gauss_{sigma}_n{n}'))
        
        # Polynomial profiles
        for c2, c4 in [(-0.3, 0.1), (-0.5, 0.2), (-0.2, -0.1), (0.1, 0.3)]:
            p = _make_profile_polynomial(n, [(2, c2), (4, c4)])
            candidates.append((p, f'poly_{c2}_{c4}_n{n}'))
    
    # Parent-based candidates
    if parent is not None:
        pn = parent.size
        candidates.append((parent.copy(), f'parent_n{pn}'))
        for n in lengths:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    candidates.append((pi, f'parent_interp_n{n}'))
    
    # Evaluate all candidates
    scored = []
    for p, label in candidates:
        sc = _score_norm(p)
        scored.append((sc, p, label))
    
    scored.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(scored)} candidates, best={scored[0][0]:.10f} [{scored[0][2]}]")
    
    for sc, p, label in scored[:3]:
        _update(p, sc, f'init_{label}')
    
    # Phase 2: Adam optimization on top candidates
    phase2_end = t_start + 600.0
    polished = []
    count = 0
    for sc, p, label in scored:
        if time.monotonic() > phase2_end or count >= 12:
            break
        per = min(50.0, phase2_end - time.monotonic() - 5)
        if per < 10:
            break
        
        use_sym = (p.size <= 1024)
        p_b, sc_b = _adam_optimize(p, time.monotonic() + per, rng,
                                   lr=0.05, beta_frac=0.1, beta_decay=0.93,
                                   no_imp_limit=200, use_symmetry=use_sym)
        _update(p_b, sc_b, f'adam_{label}')
        polished.append((sc_b, p_b, label))
        count += 1
    
    polished.sort(key=lambda x: x[0])
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # Phase 3: LP refinement
    phase3_end = t_start + 1000.0
    if best_seq is not None:
        n_best = best_seq.size
        print(f"  LP refine n={n_best}, score={best_score:.10f}")
        
        # Try original LP
        p_lp, sc_lp = _lp_refine_orig(best_seq, min(phase3_end - 60, time.monotonic() + 200),
                                       t_blend=0.4, max_solves=30)
        _update(p_lp, sc_lp, 'LP_orig')
        
        # Try SCA LP (if CVXPY works)
        if time.monotonic() < phase3_end - 30:
            try:
                p_sca, sc_sca = _lp_refine_sca(best_seq, min(phase3_end - 10, time.monotonic() + 200),
                                                max_iters=30)
                _update(p_sca, sc_sca, 'LP_sca')
            except Exception as e:
                print(f"  SCA LP failed: {e}")
        
        # Peak reduction
        if time.monotonic() < phase3_end - 10:
            p_pr, sc_pr = _peak_reduction(best_seq, n_iters=200,
                                           deadline=min(phase3_end - 5, time.monotonic() + 100))
            _update(p_pr, sc_pr, 'peak_reduce')
    
    # Phase 4: Multi-restart
    phase4_end = deadline - 60.0
    if time.monotonic() < phase4_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(20.0, (phase4_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase4_end:
                break
            per = min(per_r, phase4_end - time.monotonic() - 5)
            if per < 10:
                break
            ns = 0.05 + 0.05 * i
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_optimize(p_s, time.monotonic() + per, rng, lr=0.04,
                                        beta_frac=0.08, beta_decay=0.93, no_imp_limit=150)
            _update(p_r, sc_r, f'restart{i}_ns{ns}')
    
    # Phase 5: Final polish
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
