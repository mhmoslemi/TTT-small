"""
Minimize 2n*max(conv(a,a))/sum(a)^2 using linearized minimax LP,
combined with the previous spread-LP and gradient methods.
Key innovation: direct minimax LP that linearizes the quadratic
convolution constraint, providing a principled descent direction
with built-in step-size control via line search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]

def _smooth_max_weight(c, beta):

    mval = float(c.max())

    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0

    return w

def _make_conv_matrix(a):
    """Build the (2n-1) x n matrix M such that M @ d = conv(a, d)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _minimax_lp_step(a, conv_a, M_current):
    """
    Solve the linearized minimax LP.
    Minimize t
    Subject to: 2*(M@d)[k] - t <= -conv_a[k]  for k = 0..2n-2
                sum(d) = 0
                -a_i <= d_i <= a_i
    Returns (d, t_opt) or (None, None) if infeasible.
    """
    n = len(a)
    M = _make_conv_matrix(a)
    
    # Build A_ub: (2n-1 + n) x (n+1)
    num_ineq = 2 * n - 1 + n  # convolution constraints + lower bound on d
    num_vars = n + 1  # d_0..d_{n-1}, t
    
    A_ub = np.zeros((num_ineq, num_vars))
    b_ub = np.zeros(num_ineq)
    
    # Convolution constraints: 2*(M@d)[k] - t <= -conv_a[k]
    A_ub[:2*n-1, :n] = 2.0 * M
    A_ub[:2*n-1, n] = -1.0
    b_ub[:2*n-1] = -conv_a
    
    # Lower bound on d: d_i >= -a_i  =>  -d_i <= a_i
    for i in range(n):
        A_ub[2*n-1+i, i] = -1.0
        b_ub[2*n-1+i] = a[i]
    
    # Upper bound on d: d_i <= a_i  =>  d_i <= a_i
    # We'll handle this via bounds parameter
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: d_i in [-a_i, a_i], t in [0, None]
    bounds = [(-a[i], a[i]) for i in range(n)] + [(0.0, None)]
    
    # Objective: minimize t
    c_obj = np.zeros(num_vars)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        t_opt = res.x[n]
        return d, t_opt
    return None, None


def _minimax_lp_optimize(a, deadline, max_iter=5000, line_search_fracs=None):
    """
    Iteratively apply the linearized minimax LP to optimize `a`.
    Uses line search to determine step size.
    """
    if line_search_fracs is None:
        line_search_fracs = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]
    
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        conv_a = _fft_convolve(a)
        M_curr = float(conv_a.max())
        curr_score = float(2.0 * n * M_curr)
        
        # Solve LP
        d, t_opt = _minimax_lp_step(a, conv_a, M_curr)
        if d is None:
            break
        
        # Check if d is negligible
        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-10 * n:
            break
        
        # Line search
        best_alpha = 0.0
        best_alpha_score = curr_score
        for alpha in line_search_fracs:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc = float(2.0 * n * _fft_convolve(a_try).max())
            if sc < best_alpha_score - 1e-12:
                best_alpha_score = sc
                best_alpha = alpha
        
        if best_alpha > 0:
            a = a + best_alpha * d
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            
            if best_alpha_score < best_score - 1e-12:
                best_score = best_alpha_score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            # No improvement with this direction; try smaller perturbations
            no_imp += 1
        
        if no_imp > 100:
            break
    
    return best_a, best_score


def _spread_lp_step(a, M_target):
    """The original spread LP: max sum(b) s.t. conv(a,b) <= M_target, b >= 0."""
    n = len(a)
    M = _make_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    res = linprog(c_obj, A_ub=M, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _spread_lp_optimize(a, deadline, max_iter=5000, inflate=1.002, blend=0.3):

                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

def _adam_smooth_max_polish(a, deadline, max_iter=10000, lr=0.04):
    """Adam optimization on the smooth-max of the convolution."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    z = np.log(np.maximum(a, 1e-15))
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0
    
    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    beta = max(1e-8, 0.05 * float(_fft_convolve(a).max()))
    no_imp = 0
    base_lr = lr

        if time.monotonic() > deadline:
            break
        
        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            continue
        
        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_a, 1e-15))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue
        
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        
        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(0.005, lr * 0.97)
        
        if it % 100 == 0:
            beta = max(1e-10, beta * 0.8)
        
        if no_imp > max(1500, 3 * n):
            z = np.log(np.maximum(best_a, 1e-15))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_a, best_score

def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

"""
Minimize 2n*max(conv(a,a))/sum(a)^2 using a combination of:
1. Linearized minimax LP (direct LP to minimize the peak of the linearized convolution)
2. Spread LP (maximize sum subject to convolution constraint)
3. Exact subgradient descent on the true max
4. Adam smooth-max gradient polish
Key innovation: the direct minimax LP provides a principled, parameter-free
descent direction that directly targets the minimax objective, with a
line-search-based step size for stability.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return w


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _minimax_lp_step(a):
    n = len(a)
    conv_a = _fft_convolve(a)
    M = _make_conv_matrix(a)

    num_ineq = 2 * n - 1
    num_vars = n + 1

    A_ub = np.zeros((num_ineq, num_vars))
    b_ub = -conv_a.copy()
    A_ub[:, :n] = 2.0 * M
    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = [(-a[i], a[i]) for i in range(n)] + [(0.0, None)]

    c_obj = np.zeros(num_vars)
    c_obj[n] = 1.0

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.success:
        d = res.x[:n]
        return d, float(res.x[n]), conv_a
    return None, None, conv_a


def _minimax_lp_optimize(a, deadline, max_iter=10000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0

    alphas = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        d, t_opt, conv_a = _minimax_lp_step(a)
        if d is None:
            break

        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-12 * n:
            break

        M_curr = float(conv_a.max())
        curr_score = float(2.0 * n * M_curr)

        best_alpha = 0.0
        best_alpha_score = curr_score
        for alpha in alphas:
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            c_try = _fft_convolve(a_try)
            sc = float(2.0 * n * c_try.max())
            if sc < best_alpha_score - 1e-13:
                best_alpha_score = sc
                best_alpha = alpha

        if best_alpha > 0:
            a = a + best_alpha * d
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_alpha_score < best_score - 1e-13:
                best_score = best_alpha_score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 80:
            break

    return best_a, best_score


def _spread_lp_step(a, M_target):
    n = len(a)
    M = _make_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    res = linprog(c_obj, A_ub=M, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _spread_lp_optimize(a, deadline, max_iter=5000, inflate=1.002, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        g = _spread_lp_step(a, M * inflate_cur)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 15 == 0:
                inflate_cur = min(1.008, inflate_cur * 1.04)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0003, inflate_cur * 0.96)
            blend_cur = max(0.08, blend_cur * 0.97)
        else:
            no_imp += 1
            if no_imp % 25 == 0:
                blend_cur = max(0.05, blend_cur * 0.95)
                inflate_cur = min(1.004, inflate_cur * 0.97)
            if no_imp > max(150, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a, best_score


def _exact_subgradient_descent(a, deadline, max_iter=20000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())

    lr = 0.5 / n
    momentum = 0.7
    v = np.zeros(n)
    no_imp = 0
    best_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        k_star = int(np.argmax(c))
        M_row = _make_conv_matrix(a)[k_star]
        grad = 2.0 * M_row

        a_nest = a + momentum * v
        c_nest = _fft_convolve(a_nest)
        k_nest = int(np.argmax(c_nest))
        M_row_nest = _make_conv_matrix(a_nest)[k_nest]
        grad_nest = 2.0 * M_row_nest

        v = momentum * v - lr * grad_nest
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.1 / n, lr * 0.95)
            if no_imp > max(500, n):
                a = best_a.copy()
                v = np.zeros(n)
                lr = best_lr
                no_imp = 0

    return best_a, best_score


def _adam_smooth_max_polish(a, deadline, max_iter=10000, lr=0.04):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)

    z = np.log(np.maximum(a, 1e-15))
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    adam_t = 0

    best_a = a.copy()
    best_score = float(2.0 * n * _fft_convolve(a).max())
    beta = max(1e-8, 0.03 * float(_fft_convolve(a).max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v_arr / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_a, 1e-15))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr = max(0.003, lr * 0.96)

        if it % 80 == 0:
            beta = max(1e-10, beta * 0.75)

        if no_imp > max(1000, 2 * n):
            z = np.log(np.maximum(best_a, 1e-15))
            z -= float(np.mean(z))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_a, best_score


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _run_pipeline(a_init, n, deadline):
    a = _interpolate_profile(a_init, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_linear_profile(n, 1.5)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    print(f"    Init: score={best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining < 30:
        return best_a, best_score

    # Phase 1: Minimax LP (most time - most principled)
    mm_budget = min(remaining * 0.45, 600.0)
    mm_deadline = time.monotonic() + mm_budget
    a_mm, sc_mm = _minimax_lp_optimize(best_a, mm_deadline, max_iter=10000)
    if sc_mm < best_score:
        best_score = sc_mm
        best_a = a_mm.copy()
    print(f"    MinimaxLP: {sc_mm:.10f}")

    # Phase 2: Spread LP
    remaining = deadline - time.monotonic()
    if remaining < 20:
        return best_a, best_score
    sp_budget = min(remaining * 0.25, 300.0)
    sp_deadline = time.monotonic() + sp_budget
    a_sp, sc_sp = _spread_lp_optimize(best_a, sp_deadline, max_iter=5000, inflate=1.0015, blend=0.25)
    if sc_sp < best_score:
        best_score = sc_sp
        best_a = a_sp.copy()
    print(f"    SpreadLP: {sc_sp:.10f}")

    # Phase 3: Adam polish
    remaining = deadline - time.monotonic()
    if remaining < 15:
        return best_a, best_score
    ad_budget = min(remaining * 0.2, 200.0)
    ad_deadline = time.monotonic() + ad_budget
    a_ad, sc_ad = _adam_smooth_max_polish(best_a, ad_deadline, max_iter=8000, lr=0.03)
    if sc_ad < best_score:
        best_score = sc_ad
        best_a = a_ad.copy()
    print(f"    Adam: {sc_ad:.10f}")

    # Phase 4: Final minimax LP round
    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best_a, best_score
    mm2_budget = min(remaining * 0.5, 150.0)
    mm2_deadline = time.monotonic() + mm2_budget
    a_mm2, sc_mm2 = _minimax_lp_optimize(best_a, mm2_deadline, max_iter=5000)
    if sc_mm2 < best_score:
        best_score = sc_mm2
        best_a = a_mm2.copy()
    print(f"    MinimaxLP2: {sc_mm2:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [768, 700, 832, 640, 896, 1024]:
        if ln not in lengths:
            lengths.append(ln)
    lengths = list(dict.fromkeys(lengths))
    # Prioritize parent length first
    if parent is not None:
        pl = parent.size
        lengths.remove(pl)
        lengths.insert(0, pl)

    print(f"Lengths to try: {lengths}")
    print(f"Budget: {budget_s}s, effective deadline in {deadline - time.monotonic():.0f}s")

    # Allocate time: 60% to primary length, rest split
    primary_budget_frac = 0.55
    primary_deadline = t_start + (deadline - t_start) * primary_budget_frac

    # Run primary length (parent length)
    n_primary = lengths[0]
    print(f"\n=== Primary length {n_primary} ===")
    if parent is not None:
        a_init = _interpolate_profile(parent, n_primary)
    else:
        a_init = _make_linear_profile(n_primary, 1.5)

    a_p, sc_p = _run_pipeline(a_init, n_primary, primary_deadline)
    if sc_p < best_score:
        best_score = sc_p
        best_seq = a_p.copy()
        print(f"  ** Best after primary: {best_score:.10f}")

    # Remaining lengths with less time each
    remaining_time = deadline - time.monotonic() - 60.0
    other_lengths = lengths[1:]
    if other_lengths and remaining_time > 60:
        per_other = remaining_time / len(other_lengths)
        for n_t in other_lengths:
            if time.monotonic() > deadline - 60:
                break
            alloc = min(per_other, deadline - time.monotonic() - 60.0)
            if alloc < 30:
                continue
            n_deadline = time.monotonic() + alloc
            print(f"\n=== Length {n_t} (budget={alloc:.0f}s) ===")

            if parent is not None:
                a_init = _interpolate_profile(parent, n_t)
            else:
                a_init = _make_linear_profile(n_t, 1.5)

            a_t, sc_t = _run_pipeline(a_init, n_t, n_deadline)
            if sc_t < best_score:
                best_score = sc_t
                best_seq = a_t.copy()
                print(f"  ** New best at n={n_t}: {best_score:.10f}")

    # Final intensive phase: diverse initializations at best length
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_best = best_seq.size
        print(f"\n=== Final phase: diverse inits at n={n_best} ===")

        inits = []
        # Perturbed best
        for scale in [0.1, 0.2, 0.3]:
            a_pert = best_seq * (1.0 + scale * rng.normal(size=n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            inits.append(a_pert)
        # U-shaped
        for alpha in [1.5, 3.0, 5.0]:
            inits.append(_make_u_profile(n_best, alpha))
        # Linear
        for eta in [1.0, 2.0, 3.5]:
            inits.append(_make_linear_profile(n_best, eta))
        # Random
        for _ in range(3):
            a_r = np.maximum(rng.normal(1.0, 0.4, n_best), 0.0)
            a_r = a_r / a_r.sum()
            inits.append(a_r)

        time_per_init = max(20.0, (deadline - time.monotonic() - 40.0) / len(inits))
        for a0 in inits:
            if time.monotonic() > deadline - 40:
                break
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick optimization: minimax LP + Adam
            sub_deadline = min(time.monotonic() + time_per_init, deadline - 30.0)
            a_r, sc_r = _minimax_lp_optimize(a0, sub_deadline, max_iter=3000)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  ** Perturbed best: {best_score:.10f}")

            remaining = deadline - time.monotonic() - 30.0
            if remaining > 15:
                ad_deadline = min(time.monotonic() + remaining * 0.5, deadline - 25.0)
                a_ad, sc_ad = _adam_smooth_max_polish(best_seq, ad_deadline, max_iter=5000, lr=0.03)
                if sc_ad < best_score:
                    best_score = sc_ad
                    best_seq = a_ad.copy()
                    print(f"  ** Adam improved: {best_score:.10f}")

    # Final polish on overall best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 10.0
        print(f"\n=== Final polish (budget={remaining:.0f}s) ===")
        a_final, sc_final = _minimax_lp_optimize(best_seq, time.monotonic() + remaining * 0.6, max_iter=5000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final MinimaxLP: {sc_final:.10f}")

        remaining2 = deadline - time.monotonic() - 10.0
        if remaining2 > 10:
            a_sp, sc_sp = _spread_lp_optimize(best_seq, time.monotonic() + remaining2,
                                              max_iter=3000, inflate=1.001, blend=0.15)
            if sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            print(f"  Final SpreadLP: {sc_sp:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(512, 1.5)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
