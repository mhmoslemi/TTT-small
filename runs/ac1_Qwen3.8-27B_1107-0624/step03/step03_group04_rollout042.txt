"""
Optimized step function coefficient search using:
1. Anti-palindromic (ramp/skewed) initializations exploiting the key insight
   that np.convolve(a,a)[n-1] = sum a[i]*a[n-1-i] (NOT sum a[i]^2),
   so non-palindromic sequences can achieve score < 2.
2. Improved Adam in log-space with adaptive smooth-max temperature.
3. Coordinate descent for local refinement.
4. LP-based spreading (improved from parent).
5. Spectral-domain smoothing.
6. Multi-length search (640, 768, 896) with diversification.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros_like(c)
    w = ex / s
    val = mval + beta * float(np.log(s))
    return val, w


def _grad_smooth_max(a, beta):
    n = len(a)
    c = _fft_conv(a)
    _, w = _smooth_max(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad


def _init_ramp(n, p=1.0, rng=None):
    """Ramp initialization: a[i] = (i+1)^p, anti-palindromic."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _init_skew_bell(n, mu=0.35, width=0.25):
    """Skewed (offset) bell - anti-palindromic."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - mu) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _init_parabola(n, shift=0.0):
    """Parabolic: a[i] = (i+1)*(n-i), peaks in middle but asymmetric with shift."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    if shift != 0.0:
        a = a * np.exp(shift * i / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _init_two_hump(n, mu1=0.25, mu2=0.75, w1=0.2, w2=0.15, ratio=0.6):
    """Two-hump asymmetric profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - mu1) / w1) ** 2) + ratio * np.exp(-((t - mu2) / w2) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _init_flat_spec(n, rng):
    """Flat power spectrum initialization."""
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _interp_to_len(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _adam_logit(a, deadline, max_iter=80000, lr=0.04):
    """Adam optimization in log-space with adaptive beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    z = np.log(a)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t_adam = 0

    c = _fft_conv(a)
    c_max = float(c.max())
    beta = max(1e-6, 0.03 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    best_z = z.copy()
    no_imp = 0
    base_lr = lr

    prev_z = z.copy()
    prev_grad = None
    bb_scale = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        p = a
        c = _fft_conv(p)
        _, w = _smooth_max(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            continue

        if prev_grad is not None and np.all(np.isfinite(prev_grad)):
            sz = z - prev_z
            sg = gz - prev_grad
            dot_ss = float(np.dot(sz, sz))
            dot_sg = float(np.dot(sz, sg))
            if dot_ss > 1e-20:
                bb_cand = dot_ss / max(abs(dot_sg), 1e-20)
                bb_cand = np.clip(bb_cand, 0.1, 10.0)
                bb_scale = 0.7 * bb_scale + 0.3 * bb_cand
        prev_z = z.copy()
        prev_grad = gz.copy()

        cur_lr = base_lr * bb_scale
        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v_arr / (1.0 - b2 ** t_adam)
        z_new = z - cur_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z_new)):
            continue

        z_new = np.clip(z_new, -50.0, 50.0)
        z_new -= float(np.mean(z_new))
        e = np.exp(z_new)
        a = e / e.sum()

        c = _fft_conv(a)
        c_max_new = float(c.max())
        score = float(2.0 * n * c_max_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_z = z_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                base_lr = max(0.002, base_lr * 0.9)

        if it % 200 == 0 and it > 200:
            top = np.sort(c)[::-1][:min(10, len(c))]
            if len(top) > 1:
                gap = top[0] - top[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.6)
                elif gap > beta * 8.0:
                    beta = min(0.3 * c_max_new, beta * 1.3)

        if no_imp > max(8000, 10 * n):
            rng_r = np.random.default_rng(it * 31 + 7)
            z_restart = best_z + 0.25 * rng_r.standard_normal(n)
            z_restart = np.clip(z_restart, -50.0, 50.0)
            z_restart -= np.mean(z_restart)
            e = np.exp(z_restart)
            a = e / e.sum()
            z = z_restart
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t_adam = 0
            base_lr = lr
            bb_scale = 1.0
            prev_grad = None
            beta = max(1e-6, 0.03 * float(_fft_conv(a).max()))
            no_imp = 0

        z = z_new

    return best_a


def _coord_descent(a, deadline, max_sweeps=200):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False

        for i in range(n):
            if time.monotonic() > deadline + 60:
                break

            old_val = a[i]
            # Compute current conv
            c = _fft_conv(a)
            c_max = float(c.max())

            # Try to find better a[i]
            # The change in C(k) when a[i] changes from old to new:
            # For k where 0 <= k-i < n (and k-i != i): C(k) += 2*(new-old)*a[k-i]
            # For k = 2i: C(k) += new^2 - old^2
            # For k where 0 <= i-(k-i) < n i.e. 0 <= 2i-k < n and k != 2i: same term
            # Actually: C(k) = sum_j a[j]*a[k-j]. a[i] appears when j=i (term a[i]*a[k-i])
            # or when k-j=i (term a[k-i]*a[i]). These are the same term counted once unless k-i=i.

            # Simpler: just try a few candidate values
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 1.7,
                          old_val + 0.01, old_val - 0.01, 0.0]
            # Also try values based on current max
            if c_max > 0:
                candidates.append(old_val * 0.3)
                candidates.append(old_val * 2.0)

            best_local = old_val
            best_local_score = c_max

            for cand in candidates:
                cand = max(0.0, cand)
                a[i] = cand
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                c_new = _fft_conv(a / s)
                sc = float(2.0 * n * c_new.max())
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = cand

            if best_local != old_val:
                a[i] = best_local
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    improved = True

            # Restore if no improvement
            if not improved:
                a[i] = old_val

        c = _fft_conv(a)
        sc = float(2.0 * n * c.max() / (a.sum() ** 2))
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 10:
                break

    return best_a


def _spectral_smooth(a, deadline, n_iters=200, alpha=0.3):
    """Spectral smoothing: in frequency domain, smooth |A|^2 to reduce time peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    N = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        # Smooth P in frequency domain (circularly)
        # Use a simple moving average on P
        P_smooth = np.convolve(P, np.ones(5) / 5, mode='same')
        # Blend
        P_new = (1 - alpha) * P + alpha * P_smooth
        # Ensure P_new >= 0
        P_new = np.maximum(P_new, 1e-20)
        # Recover A_new with same phase
        mag_new = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        # Recover a_new
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            no_imp += 1
            if no_imp > 30:
                break
            continue
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                alpha = max(0.05, alpha * 0.9)
            if no_imp > 50:
                break

    return best_a


def _lp_refine(a, deadline, max_iter=15000, inflate=1.0005, blend=0.2):
    """LP-based spreading refinement."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    c = _fft_conv(a)
    best_score = float(2.0 * n * c.max())
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_conv(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)
        if g is None:
            break
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 20:
                inflate_cur = min(1.01, inflate_cur * 1.03)
            if no_imp > 300:
                break
            continue
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        c_new = _fft_conv(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.93)
                inflate_cur = min(1.001, inflate_cur * 0.97)
            if no_imp > max(150, 2 * n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        a = a_new

    return best_a


def _fine_polish(a, deadline, max_iter=50000):
    """Very fine gradient polish with tiny beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    c = _fft_conv(a)
    c_max = float(c.max())
    beta = max(1e-9, 0.005 * c_max)
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    z = np.log(a)
    z -= np.mean(z)
    lr = 0.005
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        p = a
        c = _fft_conv(p)
        _, w = _smooth_max(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-9 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        if not np.all(np.isfinite(gz)):
            break
        z = z - lr * gz
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()
        c = _fft_conv(a)
        score = float(2.0 * n * c.max())
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.0005, lr * 0.9)
        if it % 500 == 0:
            beta = max(1e-12, beta * 0.7)
        if no_imp > max(5000, 8 * n):
            break
    return best_a


def _multi_perturb_refine(a, deadline, n_starts=6, rng=None):
    """Multiple perturbations + quick refinement."""
    if rng is None:
        rng = np.random.default_rng(999)
    a = np.asarray(a, dtype=np.float64).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)

    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        noise_scale = 0.03 * (i + 1)
        a_pert = a * (1.0 + noise_scale * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()
        sub_dl = min(deadline, time.monotonic() + 40.0)
        a_ref = _adam_logit(a_pert, sub_dl, max_iter=3000, lr=0.03)
        sc = _score(a_ref)
        if sc < best_score:
            best_score = sc
            best_a = a_ref.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses anti-palindromic initializations, Adam logit-space optimization,
    coordinate descent, spectral smoothing, and LP refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths
    lengths = [768]
    if parent is not None:
        lengths = [parent.size, 640, 896]
    else:
        lengths = [512, 768, 1024]

    # Generate initializations
    print(f"Generating initializations...", flush=True)
    init_pool = []

    for n_t in lengths:
        if parent is not None:
            init_pool.append(('parent', n_t, _interp_to_len(parent, n_t)))
            # Reversed parent (anti-palindromic variant)
            init_pool.append(('parent_rev', n_t, _interp_to_len(parent[::-1], n_t)))

        # Ramp inits
        for p_exp in [0.5, 0.8, 1.0, 1.5]:
            init_pool.append((f'ramp_{p_exp}', n_t, _init_ramp(n_t, p_exp)))

        # Skewed bells
        for mu in [0.25, 0.35, 0.65, 0.75]:
            init_pool.append((f'skew_{mu}', n_t, _init_skew_bell(n_t, mu=mu)))

        # Parabolic
        init_pool.append(('parab', n_t, _init_parabola(n_t)))
        init_pool.append(('parab_s', n_t, _init_parabola(n_t, shift=0.5)))

        # Two hump
        init_pool.append(('2hump', n_t, _init_two_hump(n_t)))
        init_pool.append(('2hump_r', n_t, _init_two_hump(n_t, mu1=0.7, mu2=0.3, ratio=0.6)))

        # Flat spectrum
        init_pool.append(('flat_spec', n_t, _init_flat_spec(n_t, rng)))

    # Score all inits
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print(f"  Top 10 inits:", flush=True)
    for sc, name, n_t, _ in init_scores[:10]:
        print(f"    {name} (n={n_t}): {sc:.8f}", flush=True)

    # Optimize top candidates
    top_k = 6
    for idx, (sc0, name, n_t, a0) in enumerate(init_scores[:top_k]):
        if time.monotonic() > deadline - 120:
            break
        # Skip if clearly worse and not parent
        if sc0 > best_score * 1.1 and 'parent' not in name:
            continue

        # Phase A: Adam logit optimization
        opt_dl = min(deadline - 80, time.monotonic() + 300.0)
        print(f"[{idx}] Adam on {name} (n={n_t}), start={sc0:.8f}, budget={opt_dl - time.monotonic():.0f}s", flush=True)
        a_opt = _adam_logit(a0, opt_dl, max_iter=60000, lr=0.04)
        sc_opt = _score(a_opt)
        print(f"  Adam: {sc_opt:.10f}", flush=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

        # Phase B: Spectral smoothing
        if time.monotonic() < deadline - 60:
            sp_dl = min(deadline - 50, time.monotonic() + 60.0)
            a_sp = _spectral_smooth(a_opt, sp_dl, n_iters=100, alpha=0.3)
            sc_sp = _score(a_sp)
            print(f"  Spectral: {sc_sp:.10f}", flush=True)
            if sc_sp < best_score:
                best_score = sc_sp
                best_seq = a_sp.copy()
            a_opt = a_sp

        # Phase C: LP refinement
        if time.monotonic() < deadline - 40:
            lp_dl = min(deadline - 30, time.monotonic() + 150.0)
            a_lp = _lp_refine(a_opt, lp_dl, max_iter=10000, inflate=1.0005, blend=0.2)
            sc_lp = _score(a_lp)
            print(f"  LP: {sc_lp:.10f}", flush=True)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            a_opt = a_lp

        # Phase D: Coordinate descent
        if time.monotonic() < deadline - 20:
            cd_dl = min(deadline - 15, time.monotonic() + 120.0)
            a_cd = _coord_descent(a_opt, cd_dl, max_sweeps=50)
            sc_cd = _score(a_cd)
            print(f"  CoordDesc: {sc_cd:.10f}", flush=True)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            a_opt = a_cd

    # Final polish on global best
    if best_seq is not None and time.monotonic() < deadline - 10:
        fp_dl = min(deadline - 5, time.monotonic() + 100.0)
        print(f"Fine polish, budget={fp_dl - time.monotonic():.0f}s", flush=True)
        a_fine = _fine_polish(best_seq, fp_dl, max_iter=40000)
        sc_fine = _score(a_fine)
        print(f"  FinePolish: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()

    # Multi-perturb on best
    if best_seq is not None and time.monotonic() < deadline - 5:
        mp_dl = deadline - 3
        a_mp = _multi_perturb_refine(best_seq, mp_dl, n_starts=4, rng=rng)
        sc_mp = _score(a_mp)
        print(f"  MultiPerturb: {sc_mp:.10f}", flush=True)
        if sc_mp < best_score:
            best_score = sc_mp
            best_seq = a_mp.copy()

    # Final validation
    if best_seq is None:
        best_seq = _init_ramp(768, 1.0)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _init_ramp(768, 1.0)

    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
