def _next_pow2(n):

def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    if mn < -1e-12:

    if s < 1e-10:

    cols = []

        best_t = 0.0
        best_s = best_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_c = (1.0 - t) * a + t * g_norm
            a_c = np.maximum(a_c, 0.0)
            sc = _score(a_c)
            if sc < best_s:
                best_s = sc
                best_t = t
        if best_t == 0.0:
            no_imp += 1
            if no_imp % 50 == 0:
                inflate_cur = min(1.01, inflate_cur * 1.01)
            if no_imp > 300:
                break
            continue
        a = (1.0 - best_t) * a + best_t * g_norm
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        if best_s < best_score - 1e-13:
            best_score = best_s
            best_a = a.copy()
            no_imp = 0
            A_sp = _make_conv_matrix_sparse(a)
            inflate_cur = max(1.0001, inflate_cur * 0.98)
        else:
            no_imp += 1
    return best_a


def _adam_opt(a, deadline, lr=0.04, max_iter=30000):
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    z = np.log(a)
    z -= z.mean()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    c = _fft_convolve(a)
    beta = max(1e-6, 0.05 * float(c.max()))
    best_a = a.copy()
    best_score = float(2.0 * n * float(c.max()))
    no_imp = 0
    base_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        grad, c_max, c = _grad_smooth_max(a, beta)
        dot = float(a @ grad)
        gz = a * (grad - dot)
        gz -= gz.mean()
        if not np.all(np.isfinite(gz)):
            z = np.log(best_a) - np.log(best_a).mean()
            m, v, t_adam = np.zeros(n), np.zeros(n), 0
            continue
        t_adam += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z_new = z - base_lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new -= z_new.mean()
        e = np.exp(z_new)
        a = e / e.sum()
        c = _fft_convolve(a)
        score = float(2.0 * n * float(c.max()))
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                base_lr = max(0.002, base_lr * 0.9)
        if it % 200 == 0 and it > 100:
            top = np.sort(c)[::-1][:5]
            if len(top) > 1:
                gap = top[0] - top[-1]
                if gap < beta * 0.3:
                    beta = max(1e-10, beta * 0.6)
                elif gap > beta * 5.0:
                    beta = min(0.3 * float(c.max()), beta * 1.3)
        if no_imp > max(4000, 6 * n):
            r = np.random.default_rng(it * 31 + 7)
            z = np.log(best_a) - np.log(best_a).mean()
            z += 0.2 * r.standard_normal(n)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            e = np.exp(z)
            a = e / e.sum()
            m, v, t_adam = np.zeros(n), np.zeros(n), 0
            base_lr = lr
            no_imp = 0
            beta = max(1e-6, 0.05 * float(_fft_convolve(a).max()))
        z = z_new
    return best_a


def _dual_guided_step(a, deadline, max_iters=5000, step_size=0.02):
    """Use LP dual variables to guide targeted gradient steps."""
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    A_sp = _make_conv_matrix_sparse(a)
    best_score = _score(a)
    best_a = a.copy()
    no_imp = 0
    ss = step_size

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a)
        M = float(c.max()) * 1.001
        g, dual = _lp_spread(a, M, A_sp)
        if dual is None or not np.all(np.isfinite(dual)):
            break
        active = dual > 1e-8
        n_active = int(active.sum())
        if n_active == 0:
            break
        y = np.where(active, dual, 0.0)
        y = y / (y.sum() + 1e-30)
        pressure = np.zeros(n)
        for i in range(n):
            lo = max(0, i - n + 1)
            hi = min(2 * n - 2, 2 * n - 2 - i)
            # pressure[i] = sum_k y[k] * a[k-i] (valid range)
            k_range = np.arange(lo, hi + 1)
            a_range = a[k_range - i]
            y_range = y[k_range]
            pressure[i] = float(np.dot(y_range, a_range))
        pressure = pressure / (pressure.max() + 1e-15)
        a_new = a * np.exp(-ss * pressure)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc = _score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            ss = min(0.05, ss * 1.01)
            A_sp = _make_conv_matrix_sparse(a)
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                ss = max(0.003, ss * 0.85)
            if no_imp > 500:
                a = best_a.copy()
                ss = step_size
                no_imp = 0
                A_sp = _make_conv_matrix_sparse(a)
    return best_a


def _gradient_polish(a, deadline, max_iter=20000):
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    c = _fft_convolve(a)
    beta = max(1e-10, 0.005 * float(c.max()))
    z = np.log(a) - np.log(a).mean()
    lr = 0.003
    best_a = a.copy()
    best_score = float(2.0 * n * float(c.max()))
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        grad, c_max, c = _grad_smooth_max(a, beta)
        dot = float(a @ grad)
        gz = a * (grad - dot)
        gz -= gz.mean()
        z -= lr * gz
        z = np.clip(z, -60, 60)
        z -= z.mean()
        e = np.exp(z)
        a = e / e.sum()
        score = _score(a)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 300 == 0:
                lr = max(1e-5, lr * 0.9)
        if it % 300 == 0:
            beta = max(1e-14, beta * 0.5)
        if no_imp > 3000:
            break
    return best_a


def _try_random_start(n, deadline, rng):
    """Try a random non-negative initialization and optimize briefly."""
    a = rng.exponential(1.0, n)
    a = a / a.sum()
    d = min(deadline, time.monotonic() + 30)
    a_opt = _lp_line_search(a, d, max_iters=500, inflate=1.005)
    return a_opt


def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                best_score = sc
                best_seq = parent.copy()
                print(f"Parent: n={parent.size}, score={sc:.10f}", flush=True)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, using uniform n=768: score={best_score:.10f}", flush=True)

    n_p = len(best_seq)

    # Phase 1: LP line search on parent (150s budget)
    print(f"\nPhase 1: LP line search n={n_p}", flush=True)
    d1 = min(deadline, t0 + 150)
    a1 = _lp_line_search(best_seq, d1, max_iters=5000, inflate=1.002)
    sc1 = _score(a1)
    print(f"  Score: {sc1:.10f} (from {best_score:.10f}) [{time.monotonic()-t0:.0f}s]", flush=True)
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()

    # Phase 2: Adam optimization (150s budget)
    print(f"\nPhase 2: Adam optimization n={n_p}", flush=True)
    d2 = min(deadline, t0 + 300)
    a2 = _adam_opt(best_seq, d2, lr=0.03, max_iter=20000)
    sc2 = _score(a2)
    print(f"  Score: {sc2:.10f} (from {best_score:.10f}) [{time.monotonic()-t0:.0f}s]", flush=True)
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()

    # Phase 3: Dual-guided steps (100s budget)
    print(f"\nPhase 3: Dual-guided n={n_p}", flush=True)
    d3 = min(deadline, t0 + 400)
    a3 = _dual_guided_step(best_seq, d3, max_iters=3000, step_size=0.015)
    sc3 = _score(a3)
    print(f"  Score: {sc3:.10f} (from {best_score:.10f}) [{time.monotonic()-t0:.0f}s]", flush=True)
    if sc3 < best_score:
        best_score = sc3
        best_seq = a3.copy()

    # Phase 4: More LP (tighter) (150s budget)
    print(f"\nPhase 4: Tight LP n={n_p}", flush=True)
    d4 = min(deadline, t0 + 550)
    a4 = _lp_line_search(best_seq, d4, max_iters=5000, inflate=1.0005)
    sc4 = _score(a4)
    print(f"  Score: {sc4:.10f} (from {best_score:.10f}) [{time.monotonic()-t0:.0f}s]", flush=True)
    if sc4 < best_score:
        best_score = sc4
        best_seq = a4.copy()

    # Phase 5: Try larger lengths
    large_lengths = [1024, 1536, 2048]
    for n_large in large_lengths:
        if time.monotonic() > deadline - 200:
            break
        if n_large < len(best_seq):
            continue
        print(f"\nPhase 5: Scale to n={n_large}", flush=True)
        a_int = _interpolate(best_seq, n_large)
        sc_int = _score(a_int)
        print(f"  Interpolated: {sc_int:.10f} [{time.monotonic()-t0:.0f}s]", flush=True)

        # LP at new length
        d_lp = min(deadline - 150, time.monotonic() + 200)
        a_l = _lp_line_search(a_int, d_lp, max_iters=3000, inflate=1.001)
        sc_l = _score(a_l)
        print(f"  LP: {sc_l:.10f} [{time.monotonic()-t0:.0f}s]", flush=True)

        # Adam at new length
        d_ad = min(deadline - 150, time.monotonic() + 120)
        a_ad = _adam_opt(a_l, d_ad, lr=0.03, max_iter=10000)
        sc_ad = _score(a_ad)
        print(f"  Adam: {sc_ad:.10f} [{time.monotonic()-t0:.0f}s]", flush=True)

        # More LP
        d_l2 = min(deadline - 150, time.monotonic() + 100)
        a_l2 = _lp_line_search(a_ad, d_l2, max_iters=2000, inflate=1.0005)
        sc_l2 = _score(a_l2)
        print(f"  LP2: {sc_l2:.10f} [{time.monotonic()-t0:.0f}s]", flush=True)

        local_best = min(sc_l, sc_ad, sc_l2)
        local_seq = a_l2.copy() if sc_l2 == local_best else (a_ad.copy() if sc_ad == local_best else a_l.copy())
        if local_best < best_score:
            best_score = local_best
            best_seq = local_seq
            print(f"  *** New best at n={n_large}: {best_score:.10f} ***", flush=True)

    # Phase 6: Go back to best and do more LP + dual-guided
    if time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        print(f"\nPhase 6: Final LP+dual n={n_best}", flush=True)
        d6 = min(deadline - 80, time.monotonic() + 200)
        a6 = _lp_line_search(best_seq, d6, max_iters=4000, inflate=1.0003)
        sc6 = _score(a6)
        if sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
        print(f"  LP: {sc6:.10f}", flush=True)

        if time.monotonic() < deadline - 50:
            d6b = min(deadline - 50, time.monotonic() + 80)
            a6b = _dual_guided_step(best_seq, d6b, max_iters=2000, step_size=0.01)
            sc6b = _score(a6b)
            if sc6b < best_score:
                best_score = sc6b
                best_seq = a6b.copy()
            print(f"  Dual: {sc6b:.10f}", flush=True)

    # Phase 7: Fine polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        d7 = deadline - 15
        print(f"\nPhase 7: Fine polish [{time.monotonic()-t0:.0f}s]", flush=True)
        a7 = _gradient_polish(best_seq, d7, max_iter=30000)
        sc7 = _score(a7)
        print(f"  Score: {sc7:.10f}", flush=True)
        if sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

    # Phase 8: One final LP
    if best_seq is not None and time.monotonic() < deadline - 5:
        d8 = deadline - 3
        a8 = _lp_line_search(best_seq, d8, max_iters=2000, inflate=1.0002)
        sc8 = _score(a8)
        if sc8 < best_score:
            best_score = sc8
            best_seq = a8.copy()
        print(f"\nFinal LP: {sc8:.10f}", flush=True)

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(768) / 768

    print(f"\nFINAL: score={best_score:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s", flush=True)
    return [float(x) for x in final]
