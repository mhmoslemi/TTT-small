# Helper functions (top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c

def _score_normalized(a):
    """Score with sum(a)=1: returns 2n*max(conv)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

    t = np.linspace(1, n, n)
    a = t**(-p)  # decreasing
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_hill(n, peak=0.3, width=0.2):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - peak) / width)**2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_two_hump(n, p1=0.2, p2=0.8, w=0.15):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5*((t-p1)/w)**2) + np.exp(-0.5*((t-p2)/w)**2)

def _make_linear(n, slope=2.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a/s if s > 1e-30 else np.ones(n_new)/n_new
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0)
    s = p.sum()
    return p/s if s > 1e-30 else np.ones(n_new)/n_new

This is related to the "anti-concentration" problem! We want the sum distribution to be as flat as possible.

"""
Multi-scale optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses cycle-annealing LSE gradient, LP refinement with dual guidance,
multi-scale upsampling, and diverse asymmetric initializations.
Key improvements: cycle annealing to escape local minima, better LP
constraint selection, multi-scale approach, and extended optimization budget.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    return _score_normalized(a / s)


def _make_exp_decay(n, rate=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_exp_rise(n, rate=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(rate * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_power_law(n, p=0.3):
    t = np.linspace(1.0, float(n), n)
    a = t ** (-p)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_hill(n, peak=0.3, width=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - peak) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_two_hump(n, p1=0.15, p2=0.85, w=0.12):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - p1) / w) ** 2) + np.exp(-0.5 * ((t - p2) / w) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_linear(n, slope=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_sin_mod(n, freq=1.0, amp=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tanh_ramp(n, sharpness=5.0, center=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.tanh(sharpness * (t - center)))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        return a / s if s > 1e-30 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 1e-30 else np.ones(n_new) / n_new


def _lse_step(a, z, state, T, lr):
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())
    x = (c - m_val) / T
    x = np.clip(x, -60.0, 60.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, z, state, m_val

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    z = z - lr * mh / (np.sqrt(vh) + 1e-8)
    z = np.clip(z, -50.0, 50.0)
    z -= float(np.mean(z))
    e = np.exp(z)
    emax = float(e.max())
    if emax > 0:
        e = e / emax
    a_new = e / e.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v
    return a_new, z, state, m_val


def _lse_optimize(a, deadline, max_iter=50000, T_start=0.05, T_end=1e-5, lr=0.1, restart_interval=5000):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    z = np.log(np.maximum(a, 1e-12))
    z -= float(np.mean(z))
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

    best_a = a.copy()
    best_score = _score_normalized(a)
    base_lr = lr
    no_imp = 0
    rng_re = np.random.default_rng(12345)

    # Cycle annealing: T goes down, then up, then down...
    T = T_start
    t_down = T_start
    t_up = max(T_start * 0.3, T_end * 10)
    T_ratio_down = (t_up / T_start) ** (1.0 / max(restart_interval, 1))
    T_ratio_up = (T_start * 0.7 / t_up) ** (1.0 / max(restart_interval // 2, 1))
    phase = 0  # 0 = decreasing, 1 = increasing

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a, z, state, m_val = _lse_step(a, z, state, T, lr)

        if not np.all(np.isfinite(a)):
            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-12))
            z -= float(np.mean(z))
            state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            continue

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 2000 == 0:
                lr = max(0.01, lr * 0.95)
            if no_imp > restart_interval:
                # Restart from best with perturbation
                a = best_a * (1.0 + 0.05 * rng_re.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                z = np.log(a)
                z -= float(np.mean(z))
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                lr = base_lr
                no_imp = 0
                # Reset temperature
                if phase == 0:
                    T = T_start * 0.5
                    phase = 1
                else:
                    T = T_start * 0.8
                    phase = 0

        # Cycle annealing
        if phase == 0:
            T *= T_ratio_down
            if T <= t_up:
                T = t_up
                phase = 1
        else:
            T *= T_ratio_up
            if T >= T_start * 0.7:
                T = T_start * 0.7
                phase = 0
        T = max(T, T_end)

    return best_a


def _subgradient_optimize(a, deadline, max_iter=50000, lr=0.02, momentum=0.85):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())

    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    rng_r = np.random.default_rng(999)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        m_val = float(c.max())
        # Use all lags within a small epsilon of max for smoother subgradient
        thresh = m_val * 0.999
        max_idx = np.where(c >= thresh)[0]
        if len(max_idx) == 0:
            max_idx = np.array([int(np.argmax(c))])
        w = np.zeros_like(c)
        w[max_idx] = 1.0 / len(max_idx)

        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h_full = np.fft.irfft(fw * fa_rev, N)
        h = h_full[n - 1:2 * n - 1]
        grad = 2.0 * h

        dot = float(np.dot(a, grad))
        grad_tan = grad - dot * a

        v = momentum * v - lr * grad_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        score = float(2.0 * n * c_new.max())

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.002, lr * 0.92)
            if no_imp > max(5000, n):
                a = best_a * (1.0 + 0.03 * rng_r.normal(size=n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                v = np.zeros(n)
                lr = base_lr
                no_imp = 0

    return best_a


def _make_conv_matrix_reduced(a, M, top_k=80):
    n = len(a)
    c_aa = _fft_convolve(a)
    idx = np.argsort(c_aa)[::-1]
    active = set()
    for i in idx[:top_k]:
        for offset in range(-3, 4):
            j = i + offset
            if 0 <= j < 2 * n - 1:
                active.add(j)
    active = sorted(active)
    A = np.zeros((len(active), n))
    b = np.full(len(active), M)
    for row, k in enumerate(active):
        jmin = max(0, k - n + 1)
        jmax = min(n, k + 1)
        for j in range(jmin, jmax):
            A[row, j] = a[j]
    return A, b


def _lp_spread(a, M, n_reduced=None):
    n = len(a)
    c_obj = -np.ones(n)
    if n_reduced is not None and n > 128:
        A, b = _make_conv_matrix_reduced(a, M, top_k=n_reduced)
    else:
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _line_search(a, g, max_iter=30):
    n = len(a)
    best_score = _score_normalized(a)
    best_t = 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _score_normalized(np.maximum(a + c * (g - a), 0))
    fd = _score_normalized(np.maximum(a + d * (g - a), 0))
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _score_normalized(np.maximum(a + c * (g - a), 0))
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _score_normalized(np.maximum(a + d * (g - a), 0))
        if abs(hi - lo) < 1e-6:
            break
    best_t = (lo + hi) / 2.0
    best_score = _score_normalized(np.maximum(a + best_t * (g - a), 0))
    return best_t

def _coordinate_descent(a, deadline, max_iter=20000):
    """Coordinate descent: adjust one element at a time to reduce max(conv)."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_normalized(a)
    rng = np.random.default_rng(777)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        i = rng.integers(0, n)
        # Try to redistribute mass from i to other coordinates
        # or adjust i directly
        delta = a[i] * 0.01 * rng.uniform(-1, 1)
        a_try = a.copy()
        a_try[i] = max(0, a_try[i] + delta)
        # Redistribute the difference to maintain sum=1
        diff = a.sum() - a_try.sum()
        if abs(diff) > 1e-15:
            # Distribute to a random other coordinate
            j = (i + 1 + rng.integers(0, n - 1)) % n
            a_try[j] = max(0, a_try[j] + diff)
            s = a_try.sum()
            if s > 1e-10:
                a_try = a_try / s

        sc = _score_normalized(a_try)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_try.copy()
            a = a_try
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max(2000, n * 2):
                break

    return best_a

    lengths = [256, 512, 1024]
    best_per_length = {}

    for n_t in lengths:
        if time.monotonic() > deadline - 400:
            break
        cands = get_candidates(n_t)
        best_here = None
        best_here_score = np.inf

        for ci, a0 in enumerate(cands):
            if time.monotonic() > deadline - 400:
                break
            sc0 = _score_normalized(a0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()

            # Quick LSE (limited budget)
            lse_budget = min(15.0, deadline - time.monotonic() - 400.0)
            if lse_budget < 3:
                break
            lse_end = time.monotonic() + lse_budget
            a_lse = _lse_optimize(a0, lse_end, max_iter=15000,
                                  T_start=0.04, T_end=5e-5, lr=0.08,
                                  restart_interval=3000)
            sc_lse = _score_normalized(a_lse)

            if np.isfinite(sc_lse) and sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
            if sc_lse < best_here_score:
                best_here_score = sc_lse
                best_here = a_lse

            if ci % 4 == 0:
                print(f"  n={n_t}, cand {ci}: init={sc0:.8f} -> LSE={sc_lse:.10f} (best: {best_score:.10f})")

        if best_here is not None:
            best_per_length[n_t] = best_here
            print(f"  Best for n={n_t}: {best_here_score:.10f}")

    # Phase 2: Multi-scale upsample and refine
    print("\n=== Phase 2: Multi-scale upsample ===")

    if 256 in best_per_length and 1024 in best_per_length:
        if time.monotonic() < deadline - 300:
            a_up = _interpolate_to_length(best_per_length[256], 1024)
            # Blend with the direct 1024 solution
            a_blend = 0.5 * a_up + 0.5 * best_per_length[1024]
            a_blend = a_blend / a_blend.sum()
            sc_blend = _score_normalized(a_blend)
            if sc_blend < best_per_length[1024] is not None and sc_blend < _score_normalized(best_per_length[1024]):
                print(f"  Upsample blend improved: {sc_blend:.10f}")
                best_per_length[1024] = a_blend
                if sc_blend < best_score:
                    best_score = sc_blend
                    best_seq = a_blend.copy()

    if 512 in best_per_length and 1024 in best_per_length:

            a_blend = a_blend / a_blend.sum()
            sc_blend = _score_normalized(a_blend)
            if sc_blend < _score_normalized(best_per_length[1024]):
                print(f"  512->1024 blend improved: {sc_blend:.10f}")
                best_per_length[1024] = a_blend
                if sc_blend < best_score:
                    best_score = sc_blend
                    best_seq = a_blend.copy()

    # Phase 3: Focused LSE optimization on best candidates
    print("\n=== Phase 3: Focused LSE optimization ===")
    for n_t in [1024, 512]:
        if n_t not in best_per_length:
            continue
        if time.monotonic() > deadline - 200:
            break

        a_in = best_per_length[n_t]
        sc_in = _score_normalized(a_in)
        print(f"  LSE n={n_t}: start score={sc_in:.10f}")

        lse_budget = min(180.0, deadline - time.monotonic() - 200.0)
        if lse_budget < 20:
            continue
        lse_end = time.monotonic() + lse_budget

        a_lse = _lse_optimize(a_in, lse_end, max_iter=200000,
                              T_start=0.03, T_end=2e-6, lr=0.06,
                              restart_interval=5000)
        sc_lse = _score_normalized(a_lse)

        if np.isfinite(sc_lse) and sc_lse < best_score:
            best_score = sc_lse
            best_seq = a_lse.copy()
        best_per_length[n_t] = a_lse
        print(f"  LSE n={n_t}: result={sc_lse:.10f}")

    # Phase 4: LP refinement
    print("\n=== Phase 4: LP refinement ===")
    for n_t in [1024, 512]:
        if n_t not in best_per_length:
            continue
        if time.monotonic() > deadline - 120:
            break

        a_in = best_per_length[n_t]
        sc_in = _score_normalized(a_in)
        print(f"  LP n={n_t}: start={sc_in:.10f}")

        lp_budget = min(100.0, deadline - time.monotonic() - 120.0)
        if lp_budget < 10:
            continue
        lp_end = time.monotonic() + lp_budget

        a_lp = _lp_refine(a_in, lp_end, max_iter=2000, inflate=1.0002, use_line_search=True)
        sc_lp = _score_normalized(a_lp)

        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
        best_per_length[n_t] = a_lp
        print(f"  LP n={n_t}: result={sc_lp:.10f}")

    print("\n=== Phase 5: Subgradient polish ===")
