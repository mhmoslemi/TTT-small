    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _hankel_matvec(s_ind, v):
    """Compute H_S @ v where H_S[i,j] = s_ind[i+j]. s_ind is bool of length 2n-1."""
    n = len(v)
    # result[i] = sum_j v[j] * s_ind[i+j]
    # Use FFT-based approach for efficiency
    # s_ind has length 2n-1, v has length n
    # We want: result[i] = sum_{j=0}^{n-1} v[j] * s_ind[i+j]
    # This is the "anti-convolution"
    # Pad v to length 2n-1
    v_pad = np.zeros(2 * n - 1)
    v_pad[:n] = v
    # s_ind as float
    s_float = s_ind.astype(np.float64)
    # Use FFT: result[i] = sum_j v_pad[j] * s_float[i+j]
    # This is the cross-correlation of v_pad and s_float at specific lags
    # np.correlate(s_float, v_pad, mode='full')[k] = sum_n s_float[n+k] * v_pad[n]
    # So np.correlate(s_float, v_pad, mode='full')[i] = sum_n s_float[n+i] * v_pad[n]
    # But we want sum_j v_pad[j] * s_float[i+j] = sum_j s_float[i+j] * v_pad[j]
    # Let k = i+j, j = k-i: sum_k s_float[k] * v_pad[k-i] (where k-i in [0,n-1])
    # This is np.correlate(v_pad, s_float, mode='full')[i] = sum_n v_pad[n] * s_float[n+i]
    # Yes! So result = np.correlate(v_pad, s_float, mode='full')[:n]
    # Wait, np.correlate(a, v, mode='full')[k] = sum_n a[n+k] * conj(v[n]) for k>=0
    # Actually in numpy: correlate(a, v) = sum_n a[n+k] * v[n]
    # So correlate(v_pad, s_float, 'full')[i] = sum_n v_pad[n+i] * s_float[n]
    # That's not what I want either.
    
    # Let me just use the direct approach vectorized
    # result[i] = dot(v, s_ind[i:i+n])
    # For efficiency, build the matrix
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    idx = i_idx + j_idx  # idx[i,j] = i+j, shape (n,n)
    # s_ind[idx] gives H[i,j] = s_ind[i+j]
    # But this creates an n*n array. For n=2048 that's 4M (32MB as bool, 32MB as float)
    # Let's do it in chunks for memory efficiency
    chunk = max(1, min(n, 20000000 // n))  # ~20M elements per chunk
    result = np.zeros(n)
    for start in range(0, n, chunk):
        end = min(start + chunk, n)
        H_chunk = s_ind[idx[start:end, :]].astype(np.float64)
        result[start:end] = H_chunk @ v
    return result


def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(np.minimum(x, 50.0))
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    return ex / s


def _gradient_smooth_max(a, beta):
    n = len(a)
    c = _fft_convolve(a)
    w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h, float(c.max()), c


def _active_set_qp(a, deadline, max_iter=50000, frac=0.95, lr_init=0.01, momentum=0.9):
    """Active-set QP: minimize sum of active convolution values.
    Uses projected gradient with Nesterov momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    best_score = _score(a)
    
    c = _fft_convolve(a)
    M = float(c.max())
    active = c >= frac * M
    s_shape = active.astype(np.float64)
    
    # Estimate Lipschitz constant
    # lambda_max of H_S is bounded by sum of active entries in one row
    # For row i: sum_j s_ind[i+j] = number of active k where k-i in [0,n-1]
    # Upper bound: number of active entries
    n_active = int(active.sum())
    # Conservative: lr = 1 / (2 * n_active)
    lr = min(lr_init, 1.0 / (2.0 * max(n_active, 1)))
    lr = max(lr, 1e-6)
    
    v = np.zeros(n)
    a_prev = a.copy()
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient: grad = 2 * H_S @ a
        grad = 2.0 * _hankel_matvec(s_shape, a)
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a_prev = a
        a = a_new
        
        # Periodically check improvement
        if it % 50 == 0:
            c = _fft_convolve(a)
            sc = float(2.0 * n * c.max())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > 20:
                    # Update active set
                    M_new = float(c.max())
                    active_new = c >= frac * M_new
                    if not np.array_equal(active_new, active):
                        active = active_new
                        s_shape = active.astype(np.float64)
                        n_active = int(active.sum())
                        lr = min(lr * 1.5, 1.0 / (2.0 * max(n_active, 1)))
                    # Decay lr slightly
                    lr = max(1e-7, lr * 0.99)
            if no_imp > 100:
                a = best_a.copy()
                v = np.zeros(n)
                no_imp = 0
                lr = min(lr_init, 1.0 / (2.0 * max(n_active, 1)))
        
        # Adaptively update active set every 200 iterations
        if it % 200 == 0:
            c = _fft_convolve(a)
            M_new = float(c.max())
            active_new = c >= frac * M_new
            if not np.array_equal(active_new, active):
                active = active_new
                s_shape = active.astype(np.float64)
                n_active = int(active.sum())
                lr = min(lr * 1.5, 1.0 / (2.0 * max(n_active, 1)))
    
    return best_a, best_score


def _gradient_polish(a, deadline, max_iter=30000, beta_scale=0.05):
    """Smooth-max gradient polish with momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _project_simplex(a)
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    beta = max(1e-8, beta_scale * float(c.max()))
    lr = 0.03
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad, maxc, c_arr = _gradient_smooth_max(a, beta)
        if not np.all(np.isfinite(grad)):
            beta = max(1e-8, beta * 2.0)
            v = np.zeros(n)
            continue
        
        v = momentum * v - lr * grad
        a = _project_simplex(a + v)
        
        if it % 30 == 0:
            c = _fft_convolve(a)
            score = float(2.0 * n * c.max())
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 10 == 0:
                    lr = max(0.003, lr * 0.95)
                if no_imp > max(50, n // 5):
                    a = best_a.copy()
                    v = np.zeros(n)
                    beta = max(1e-8, beta_scale * float(_fft_convolve(a).max()))
                    lr = base_lr
                    no_imp = 0
        
        if it % 100 == 0:
            beta = max(1e-10, beta * 0.85)
    
    return best_a, best_score


def _logit_adam_polish(a, deadline, max_iter=20000, lr=0.04):
    """Adam optimization in logit (log) parameterization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = len(a)
    
    z = np.log(a)
    z -= float(np.mean(z))
    
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_z = z.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    beta = max(1e-8, 0.05 * float(c.max()))
    no_imp = 0
    base_lr = lr
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        p = a
        c = _fft_convolve(p)
        w = _smooth_max_weight(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fp_rev = np.fft.rfft(p[::-1], N)
        h_full = np.fft.irfft(fw * fp_rev, N)
        h = h_full[n - 1:2 * n - 1]
        vp = 2.0 * h + 1e-8 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue
        
        t += 1
        m = b1 * m + (1.0 - b1) * gz
        v_arr = b2 * v_arr + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t)
        vh = v_arr / (1.0 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            continue
        
        z = np.clip(z, -40.0, 40.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()
        
        if it % 40 == 0:
            c = _fft_convolve(a)
            score = float(2.0 * n * c.max())
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 20 == 0:
                    lr = max(0.003, lr * 0.95)
        
        if it % 150 == 0:
            beta = max(1e-10, beta * 0.75)
        
        if no_imp > max(300, n):
            rng_r = np.random.default_rng(hash((it, n)) % (2**31))
            z = 0.8 * best_z + 0.2 * rng_r.normal(size=n) * 0.3
            z = np.clip(z, -40.0, 40.0)
            z -= float(np.mean(z))
            e = np.exp(z)
            a = e / e.sum()
            c = _fft_convolve(a)
            beta = max(1e-8, 0.05 * float(c.max()))
            m = np.zeros(n)
            v_arr = np.zeros(n)
            t = 0
            lr = base_lr
            no_imp = 0
    
    return best_a, best_score


def _make_linear_profile(n, eta=1.5):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, alpha=2.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic_profile(n, a1=1.5, a3=-1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quartic_profile(n, a1=1.0, a2=0.5, a4=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2 + a4 * u ** 4
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_mixture(n, centers, sigmas, weights):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c, s, w in zip(centers, sigmas, weights):
        a += w * np.exp(-0.5 * ((t - c) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _lp_spread(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    # Build constraint matrix: A[k, j] = a[k-j] if valid
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A[rows, j] = a
    b = np.full(2 * n - 1, M)
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a, deadline, max_iter=5000, inflate=1.001, blend=0.3):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, _score(a)
    a = a / s
    
    best_a = a.copy()
    c = _fft_convolve(a)
    best_score = float(2.0 * n * c.max())
    
    no_imp = 0
    blend_cur = blend
    inflate_cur = inflate
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        M = float(c.max())
        g = _lp_spread(a, M * inflate_cur)
        
        if g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.02, inflate_cur * 1.05)
            if no_imp > 300:
                break
            continue
        
        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            inflate_cur = max(1.0005, inflate_cur * 0.97)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.93)
                inflate_cur = min(1.01, inflate_cur * 0.97)
            if no_imp > max(200, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0
        
        a = a_new
    
    return best_a, best_score


def _random_perturb(a, sigma=0.1, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    a_p = a * (1.0 + sigma * rng.normal(size=len(a)))
    a_p = np.maximum(a_p, 0.0)
    s = a_p.sum()
    if s < 1e-10:
        return a.copy()
    return a_p / s


def _optimal_profile_search(n, deadline, rng):
    """Search over parametric profiles to find good initializations."""
    best_a = None
    best_score = np.inf
    
    # Try various U-shaped profiles
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 8.0, 12.0, 20.0]:
        a = _make_u_profile(n, alpha)
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = _make_linear_profile(n, eta)
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    # Cubic profiles
    for a1 in [0.5, 1.0, 1.5, 2.0]:
        for a3 in [-0.5, -1.0, -1.5, -2.0]:
            a = _make_cubic_profile(n, a1, a3)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    # Quartic profiles
    for a4 in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = _make_quartic_profile(n, 0.0, 0.0, a4)
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    # Gaussian mixtures (bimodal)
    for c in [0.05, 0.1, 0.15, 0.2]:
        for s_ in [0.05, 0.1, 0.15, 0.2]:
            a = _make_gaussian_mixture(n, [c, 1-c], [s_, s_], [1.0, 1.0])
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    # Random structured
    for _ in range(20):
        a = np.maximum(rng.normal(1.0, 0.5, n), 0.0)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def _full_pipeline_for_init(a_init, n, deadline):
    """Run full optimization pipeline for a single initialization."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = _make_linear_profile(n, 1.5)
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    t_start = time.monotonic()
    
    # Phase 1: Active-set QP
    qp_budget = (deadline - time.monotonic()) * 0.3
    if qp_budget > 5:
        qp_end = time.monotonic() + qp_budget
        a_qp, sc_qp = _active_set_qp(a, qp_end, max_iter=30000, frac=0.93)
        if sc_qp < best_score:
            best_score = sc_qp
            best_a = a_qp.copy()
            print(f"      QP: {sc_qp:.10f}")
    
    # Phase 2: Gradient polish
    gp_budget = (deadline - time.monotonic()) * 0.25
    if gp_budget > 5:
        gp_end = time.monotonic() + gp_budget
        a_gp, sc_gp = _gradient_polish(best_a, gp_end, max_iter=20000)
        if sc_gp < best_score:
            best_score = sc_gp
            best_a = a_gp.copy()
            print(f"      Grad: {sc_gp:.10f}")
    
    # Phase 3: Active-set QP again (with updated active set)
    qp2_budget = (deadline - time.monotonic()) * 0.2
    if qp2_budget > 5:
        qp2_end = time.monotonic() + qp2_budget
        a_qp2, sc_qp2 = _active_set_qp(best_a, qp2_end, max_iter=20000, frac=0.90)
        if sc_qp2 < best_score:
            best_score = sc_qp2
            best_a = a_qp2.copy()
            print(f"      QP2: {sc_qp2:.10f}")
    
    # Phase 4: LP refine
    lp_budget = (deadline - time.monotonic()) * 0.15
    if lp_budget > 10:
        lp_end = time.monotonic() + lp_budget
        a_lp, sc_lp = _lp_refine(best_a, lp_end, max_iter=3000, inflate=1.001, blend=0.25)
        if sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()
            print(f"      LP: {sc_lp:.10f}")
    
    # Phase 5: Adam polish
    ad_budget = (deadline - time.monotonic()) * 0.1
    if ad_budget > 5:
        ad_end = time.monotonic() + ad_budget
        a_ad, sc_ad = _logit_adam_polish(best_a, ad_end, max_iter=10000, lr=0.03)
        if sc_ad < best_score:
            best_score = sc_ad
            best_a = a_ad.copy()
            print(f"      Adam: {sc_ad:.10f}")
    
    # Phase 6: Final gradient
    fin_budget = (deadline - time.monotonic()) * 0.05
    if fin_budget > 3:
        fin_end = time.monotonic() + fin_budget
        a_fin, sc_fin = _gradient_polish(best_a, fin_end, max_iter=5000, beta_scale=0.03)
        if sc_fin < best_score:
            best_score = sc_fin
            best_a = a_fin.copy()
            print(f"      Final: {sc_fin:.10f}")
    
    return best_a, best_score


def _intensive_polish(a, deadline):
    """Intensive final polishing cycle."""
    best_a = np.asarray(a, dtype=np.float64).copy()
    best_score = _score(best_a)
    n = len(best_a)
    
    # Multiple rounds of optimization
    for round_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        # QP
        qp_b = min(remaining * 0.25, 120.0)
        if qp_b > 10:
            a_qp, sc = _active_set_qp(best_a, time.monotonic() + qp_b, max_iter=20000, frac=0.92)
            if sc < best_score:
                best_score = sc
                best_a = a_qp.copy()
        
        # Gradient
        gp_b = min(remaining * 0.2, 100.0)
        if gp_b > 10:
            a_gp, sc = _gradient_polish(best_a, time.monotonic() + gp_b, max_iter=15000)
            if sc < best_score:
                best_score = sc
                best_a = a_gp.copy()
        
        # Adam
        ad_b = min(remaining * 0.2, 80.0)
        if ad_b > 10:
            a_ad, sc = _logit_adam_polish(best_a, time.monotonic() + ad_b, max_iter=10000, lr=0.02)
            if sc < best_score:
                best_score = sc
                best_a = a_ad.copy()
        
        # LP
        lp_b = min(remaining * 0.15, 60.0)
        if lp_b > 10:
            a_lp, sc = _lp_refine(best_a, time.monotonic() + lp_b, max_iter=2000, inflate=1.0008, blend=0.2)
            if sc < best_score:
                best_score = sc
                best_a = a_lp.copy()
        
        # Small perturbation + gradient (to escape local min)
        rem2 = deadline - time.monotonic()
        if rem2 > 30 and round_idx < 4:
            rng_p = np.random.default_rng(1000 + round_idx)
            a_pert = _random_perturb(best_a, sigma=0.05, rng=rng_p)
            a_p, sc = _gradient_polish(a_pert, time.monotonic() + min(rem2 * 0.1, 30.0), max_iter=5000)
            if sc < best_score:
                best_score = sc
                best_a = a_p.copy()
                print(f"        Perturbed improvement: {sc:.10f}")
        
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        print(f"    Round {round_idx + 1}: score={best_score:.10f}, time_left={remaining:.0f}s")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining active-set QP, smooth-max gradient,
    LP spreading, and Adam in logit space. Uses multi-length search with
    diverse initializations and intensive final polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass
    
    # Determine target lengths
    lengths = [512, 768, 1024, 1536]
    if parent is not None:
        pn = parent.size
        if pn not in lengths:
            lengths.insert(0, pn)
    # Sort by potential (larger n can do better theoretically)
    # But also keep parent length first for warm start
    lengths = list(dict.fromkeys(lengths))
    
    print(f"Target lengths: {lengths}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")
    
    # Phase 1: Search over lengths
    # Allocate time: more time for parent length, less for others
    num_lengths = len(lengths)
    base_time = (deadline - time.monotonic() - 300.0) / num_lengths
    
    for idx, n_t in enumerate(lengths):
        if time.monotonic() > deadline - 200:
            break
        
        # Time allocation: parent gets more time
        if parent is not None and n_t == parent.size:
            phase_time = min(base_time * 1.5, deadline - time.monotonic() - 250.0)
        else:
            phase_time = min(base_time * 0.8, deadline - time.monotonic() - 250.0)
        
        phase_time = max(30.0, phase_time)
        
        if phase_time < 20:
            continue
        
        print(f"\n=== Length n={n_t}, time_budget={phase_time:.0f}s ===")
        phase_deadline = time.monotonic() + phase_time
        
        # Generate initializations
        inits = []
        if parent is not None:
            inits.append(_interpolate_profile(parent, n_t))
        
        # Parametric search
        a_opt, sc_opt = _optimal_profile_search(n_t, time.monotonic() + phase_time * 0.15, rng)
        if a_opt is not None:
            inits.append(a_opt)
            print(f"  Best parametric init: score={sc_opt:.10f}")
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
        
        # Add parent-based perturbations
        if parent is not None:
            for sigma in [0.05, 0.1, 0.2]:
                inits.append(_random_perturb(_interpolate_profile(parent, n_t), sigma, rng))
        
        # Limit number of inits based on time
        max_inits = max(2, int(phase_time / 60.0))
        inits = inits[:max_inits]
        
        # Run pipeline on each init
        per_init_time = phase_time / max(len(inits), 1)
        for i_init, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline - 10:
                break
            init_deadline = min(phase_deadline - 5, time.monotonic() + per_init_time)
            if init_deadline - time.monotonic() < 15:
                continue
            
            sc0 = _score(a0)
            print(f"  Init {i_init}: score={sc0:.8f}")
            
            a_res, sc_res = _full_pipeline_for_init(a0, n_t, init_deadline)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best at n={n_t}: {sc_res:.10f}")
        
    # Phase 2: Diverse inits at the best length
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = best_seq.size
        print(f"\n=== Phase 2: Diverse inits at n={n_best} ===")
        
        inits2 = []
        # Strong U-shapes
        for alpha in [3.0, 5.0, 8.0, 12.0, 20.0, 40.0]:
            inits2.append(_make_u_profile(n_best, alpha))
        # Bimodal
        for c in [0.05, 0.1, 0.15, 0.2, 0.3]:
            inits2.append(_make_gaussian_mixture(n_best, [c, 1-c], [0.08, 0.08], [1.0, 1.0]))
        # Perturbed best
        for sigma in [0.03, 0.07, 0.15, 0.3]:
            inits2.append(_random_perturb(best_seq, sigma, rng))
        # Random
        for _ in range(5):
            a_r = np.maximum(rng.normal(1.0, 0.7, n_best), 0.0)
            inits2.append(a_r / a_r.sum())
        
        # Limit by time
        rem = deadline - time.monotonic() - 80.0
        n_inits2 = max(2, min(len(inits2), int(rem / 45.0)))
        inits2 = inits2[:n_inits2]
        
        per_init = rem / n_inits2
        for a0 in inits2:
            if time.monotonic() > deadline - 60:
                break
            dead = min(deadline - 50, time.monotonic() + per_init)
            if dead - time.monotonic() < 10:
                continue
            sc0 = _score(a0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = a0.copy()
            a_res, sc_res = _full_pipeline_for_init(a0, n_best, dead)
            if sc_res < best_score:
                best_score = sc_res
                best_seq = a_res.copy()
                print(f"  ** New best: {sc_res:.10f}")
    
    # Phase 3: Intensive final polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        final_deadline = deadline - 20.0
        print(f"\n=== Phase 3: Intensive polish, time_left={final_deadline - time.monotonic():.0f}s ===")
        best_seq, best_score = _intensive_polish(best_seq, final_deadline)
    
    # Final validation
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)
    
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(512, 1.5)
    
    # Evaluate with authoritative function
    if final_seq.size <= 5000:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass
    
    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
