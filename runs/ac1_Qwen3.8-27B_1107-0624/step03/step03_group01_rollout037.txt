    # This is the cross-correlation of w and a at lag j: (w ⋆ a)(j) = sum_m w[m+j]*a[m]

    # Let m = j-k, so k = j-m. sum_m w[j-m] * a[m]
    # This is the cross-correlation of w and a at lag -j: (w ⋆ a)(-j) = sum_m w[j-m]*a[m]
    
    # So grad[j] = (1/beta) * ((w ⋆ a)(j) + (w ⋆ a)(-j))
    # = (1/beta) * ((w ⋆ a)(j) + (a ⋆ w)(j))  [by symmetry of cross-correlation]
    # = (2/beta) * (w ⋆ a)(j)  [if w is symmetric, which it's not in general]
    
    # Hmm, let me just compute it directly using convolution:
    # sum_k w[k] * a[k-j]: rewrite as sum_k w[k] * a[(k-j)]
    # = sum_k a[k-j] * w[k] = (a * w)[2n-2-j] ... no.
    
    # OK let me just do it the straightforward way with FFT:
    # Term 1: T1[j] = sum_{k=j}^{j+n-1} w[k] * a[k-j] = sum_{m=0}^{n-1} w[m+j] * a[m]
    # This is the correlation of w (length 2n-1) with a (length n) at lag j.
    # 
    # Using FFT: correlate(w, a) at lag j = sum_m w[m+j]*a[m]
    # = ifft(fft(w) * conj(fft(a))) at appropriate index
    # But w has length 2n-1 and a has length n, so we need to be careful.
    
    # Actually, the simplest approach: 
    # T1[j] = sum_{m=0}^{n-1} a[m] * w[m+j] for j = 0, ..., n-1
    # This is just the first n elements of the cross-correlation.
    # crosscorr = np.correlate(w, a, mode='full') ... but this might be slow for large n.
    
    # Better: use FFT
    # T1 = ifft(fft(w, N) * conj(fft(a, N))) -- but we need to handle lengths carefully.
    
    # Actually, let me just use the approach from the previous program which worked:
    # a_rev = a[::-1]
    # w_conv = _full_convolve(w, a_rev)  # this gives sum_k w[k]*a_rev[n-1-?] ... 
    
    # Hmm, let me think about what _full_convolve(w, a_rev) gives:
    # (w * a_rev)[j] = sum_k w[k] * a_rev[j-k] = sum_k w[k] * a[n-1-(j-k)] = sum_k w[k] * a[n-1-j+k]
    # Let m = n-1-j+k, so k = m - (n-1-j) = m + j - n + 1
    # = sum_m w[m+j-n+1] * a[m]
    # 
    # So (w * a_rev)[j] = sum_m a[m] * w[m + j - n + 1]
    # For j from n-1 to 2n-2:
    # (w * a_rev)[n-1+j'] = sum_m a[m] * w[m+j'] for j'=0,...,n-1
    # 
    # So T1[j'] = (w * a_rev)[n-1+j'] for j' = 0, ..., n-1
    # i.e., T1 = (w * a_rev)[n-1 : 2n-1]
    
    # Similarly, T2[j] = sum_k w[k] * a[j-k]
    # (w * a)[j] = sum_k w[k] * a[j-k]
    # So T2[j] = (w * a)[j] for j = 0, ..., n-1
    # But (w * a) has length 2n-1, and we want j from 0 to n-1.
    
    # Wait, but j-k needs to be in [0, n-1], so k is in [j-(n-1), j].
    # And k needs to be in [0, 2n-2].
    # So T2[j] = (w * a)[j] directly, for j = 0, ..., n-1.
    
    # So grad[j] = (1/beta) * (T1[j] + T2[j])
    # = (1/beta) * ((w*a_rev)[n-1+j] + (w*a)[j])
    
    N = _next_pow2(3 * n)  # need enough size for convolutions of length up to 3n-2
    wa = np.fft.irfft(np.fft.rfft(w, N) * np.fft.rfft(a, N), N)[:2*n-1]
    wa_rev = np.fft.irfft(np.fft.rfft(w, N) * np.fft.rfft(a[::-1], N), N)[:2*n-1]
    
    T2 = wa[:n]  # (w*a)[0:n]
    T1 = wa_rev[n-1:2*n-1]  # (w*a_rev)[n-1:2n-1]
    
    grad = (1.0 / beta) * (T1 + T2)
    
    # Total gradient of 2*n*logsumexp(c/beta) w.r.t. a:
    total_grad = 2.0 * n * grad
    
    # Now, we also need the gradient w.r.t. z (log-space), which involves the softmax Jacobian.
    # But for simplicity, let's return the gradient w.r.t. a and handle the transformation externally.
    
    return total_grad, w, c, cm


def _obj_and_grad_z(z, beta, n, top_k=None):
    """
    Objective and gradient in log-space z, where a = softmax(z).
    Objective: 2*n*(cm + beta*log(sum(exp((c-cm)/beta)))) / (sum(a))^2
    
    Actually, since a = softmax(z), sum(a) = 1 always. So objective is just:
    2*n*(cm + beta*log(sum(exp((c-cm)/beta))))
    """
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_work = np.full_like(c, -1e30)
        c_work[idx] = c[idx]
    else:
        c_work = c.copy()
    
    cm = float(c_work.max())
    c_shifted = (c_work - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * cm
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (cm + beta * np.log(s))
        w_valid = e / s
        w = np.zeros_like(c)
        if top_k is not None and top_k < len(c):
            w[idx] = w_valid
        else:
            w = w_valid.copy()
    
    # Gradient w.r.t. a
    wa = _full_convolve(w, a)
    wa_rev = _full_convolve(w, a[::-1])
    
    T2 = wa[:n]
    T1 = wa_rev[n-1:2*n-1]
    
    grad_a = (2.0 * n / beta) * (T1 + T2)
    
    # Chain rule: d/dz = d/da * da/dz = d/da * (a*I - a*a^T) for softmax
    # grad_z[j] = sum_i grad_a[i] * da[i]/dz[j]
    # da[i]/dz[j] = a[i]*(delta_ij - a[j])
    # grad_z[j] = sum_i grad_a[i]*a[i]*(delta_ij - a[j])
    # = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
    
    ga_dot_a = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - ga_dot_a)
    
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    
    return obj, grad_z


def _obj_fn_z(z, beta, n, top_k=None):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_work = np.full_like(c, -1e30)
        c_work[idx] = c[idx]
    else:
        c_work = c.copy()
    
    cm = float(c_work.max())
    c_shifted = (c_work - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _grad_fn_z(z, beta, n, top_k=None):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_work = np.full_like(c, -1e30)
        c_work[idx] = c[idx]
    else:
        c_work = c.copy()
        idx = None
    
    cm = float(c_work.max())
    c_shifted = (c_work - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = np.zeros_like(c)
        w_valid = e / s
        if idx is not None:
            w[idx] = w_valid
        else:
            w = w_valid.copy()
    
    wa = _full_convolve(w, a)
    wa_rev = _full_convolve(w, a[::-1])
    
    T2 = wa[:n]
    T1 = wa_rev[n-1:2*n-1]
    
    grad_a = (2.0 * n / beta) * (T1 + T2)
    
    ga_dot_a = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - ga_dot_a)
    
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    
    return grad_z


def _lbfgs_phase(z, beta, n, max_iter, top_k=None, time_limit=60.0):
    """Run L-BFGS for a fixed beta."""
    t0 = time.monotonic()
    try:
        result = minimize(
            _obj_fn_z, z, args=(beta, n, top_k),
            method='L-BFGS-B', jac=_grad_fn_z,
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10,
                     'maxls': 50, 'maxfun': max_iter * 3}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return result.x
    except Exception as ex:
        print(f"  L-BFGS error: {ex}")
    return z


def _adam_polish(z, n, deadline, lr=0.05, beta_start=None, top_k=None,
                 lr_decay=0.9995, beta_decay=0.99):
    """Adam optimization in log-space."""
    t0 = time.monotonic()
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    
    if beta_start is None:
        beta = 0.01 * cm
    else:
        beta = beta_start
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    best_z = z.copy()
    best_obj = _obj_fn_z(z, beta, n, top_k)
    
    max_iter = 100000
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        
        t += 1
        obj, grad = _obj_and_grad_z(z, beta, n, top_k)
        
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_z = z.copy()
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        lr *= lr_decay
        beta *= beta_decay
        beta = max(beta, 1e-15)
    
    return best_z


def _direct_peak_grad_step(a, n, step_size=0.01, max_iter=200, deadline=None):
    """
    Direct gradient descent on the actual max peak.
    a is the normalized sequence (sum=1).
    Works in z-space (log of a).
    """
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    best_z = z.copy()
    best_score = _score_norm_fn(_softmax_fn(z))
    
    lr = step_size
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        k_max = int(np.argmax(c))
        cm = c[k_max]
        
        # Gradient of c[k_max] w.r.t. a:
        # dc[k_max]/da[j] = a[k_max-j] + a[j-k_max] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            i1 = k_max - j
            if 0 <= i1 < len(c):  # k_max - j is valid index in c, but we need a[i1] valid
                if 0 <= i1 < n:
                    g += a[i1]
            i2 = j - k_max
            if 0 <= i2 < n:
                g += a[i2]
            grad[j] = g
        
        # Actually, let me redo this. c[k] = sum_i a[i]*a[k-i] where 0<=i<n, 0<=k-i<n
        # dc[k]/da[j]: the term a[j]*a[k-j] contributes a[k-j], and the term a[k-j]*a[j] contributes a[j]
        # But we need k-j to be in [0,n-1] for the first, and j to be in [0,n-1] for the second.
        # Wait, a has length n, indices 0 to n-1.
        # c[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
        # = sum_{i: max(0,k-n+1)<=i<=min(k,n-1)} a[i]*a[k-i]
        
        # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # But wait, the second term: if we differentiate a[k-i] w.r.t. a[j], we get it when k-i=j, i.e., i=k-j.
        # So the contribution is a[i] = a[k-j] (from the a[i] term) + a[i] = a[k-j] (from the a[k-i] term when i=k-j)
        # Hmm, this is the same term! Let me be more careful.
        
        # c[k] = sum_i a[i] * a[k-i]
        # dc[k]/da[j] = sum_i [da[i]/da[j] * a[k-i] + a[i] * da[k-i]/da[j]]
        # = sum_i [delta_{ij} * a[k-i] + a[i] * delta_{k-i,j}]
        # = a[k-j] + a[k-j]  (the two terms are the same when both conditions are met)
        # Wait no: delta_{k-i,j} means k-i=j, so i=k-j. Then a[i] = a[k-j].
        # So dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j]? That can't be right for asymmetric cases.
        
        # Let me redo: dc[k]/da[j] = a[k-j] (from first term, when i=j) + a[i] where i=k-j (from second term)
        # = a[k-j] + a[k-j] = 2*a[k-j]? 
        # But this only works if the convolution is "symmetric" in the sense that both a[i]*a[k-i] and a[k-i]*a[i] appear.
        # In the sum c[k] = sum_i a[i]*a[k-i], each pair (i, k-i) appears once.
        # If i != k-i (i.e., k is odd or i != k/2), then the pair (i, k-i) and (k-i, i) are different terms in the sum.
        # But in the sum, we sum over ALL valid i, so if both i and k-i are valid, both terms appear.
        # So dc[k]/da[j] = a[k-j] (from i=j) + a[k-j] (from i=k-j, if k-j != j)
        # = 2*a[k-j] if k-j != j and both valid
        # = a[k-j] if k-j == j (i.e., k = 2j) [only one term]
        # = a[k-j] if only one of the indices is valid
        
        # Hmm wait, I think I'm overcomplicating this. Let me just use:
        # c[k] = sum_i a[i]*a[k-i]
        # dc[k]/da[j] = a[k-j] + a[j-k] (with the convention that a[x] = 0 if x not in [0,n-1])
        # Wait no, a[j-k] doesn't make sense. Let me be very explicit.
        
        # c[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (with a[m] = 0 if m < 0 or m >= n)
        # dc[k]/da[j] = sum_{i=0}^{n-1} [delta(i,j) * a[k-i] + a[i] * delta(k-i, j)]
        # = a[k-j] + a[k-j]  [first term: i=j gives a[k-j]; second term: k-i=j gives i=k-j, and a[i]=a[k-j]]
        # = 2*a[k-j] (if 0 <= k-j < n)
        # But wait, this assumes both i=j and i=k-j give valid contributions.
        # If j = k-j (i.e., k=2j), then we're double-counting the same term.
        # In that case, dc[k]/da[j] = a[k-j] (only one term, since i=j=k-j).
        # Hmm, actually no. If k=2j, then i=j=k-j, and the term is a[j]*a[j] = a[j]^2.
        # dc[k]/da[j] for this term = 2*a[j].
        # And there's no other i that gives delta(k-i,j) unless k-i=j, which means i=k-j=j.
        # So it's just one term with derivative 2*a[j].
        
        # OK so the formula is:
        # dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] if k-j != j and 0<=k-j<n
        # dc[k]/da[j] = 2*a[k-j] if k-j == j (same thing, since a[j]*a[j] has derivative 2*a[j])
        # Actually, I realize the formula is always 2*a[k-j] (with a[m]=0 for invalid m).
        
        # Wait, that's only true if the convolution sums over all i from 0 to n-1.
        # Let me verify: c[k] = sum_{i=0}^{n-1} a[i] * a[k-i]
        # dc[k]/da[j] = a[k-j] + sum_{i=0}^{n-1} a[i] * delta(k-i,j)
        # = a[k-j] + a[k-j]  [the second sum picks out i=k-j, giving a[k-j]]
        # = 2*a[k-j]
        
        # But this is only if k-j is in [0,n-1]. If k-j < 0 or k-j >= n, then a[k-j] = 0 and the
        # second sum also gives 0.
        
        # Hmm wait, I think I'm making an error. Let me re-derive:
        # c[k] = sum_{i=0}^{n-1} a[i] * b[k-i] where b = a (same sequence)
        # dc[k]/da[j] = sum_{i=0}^{n-1} [delta(i,j)*a[k-i] + a[i]*delta(k-i,j)]
        # First sum: i=j, gives a[k-j] (valid if 0<=k-j<n)
        # Second sum: k-i=j => i=k-j, gives a[k-j] (valid if 0<=k-j<n)
        # Total: 2*a[k-j] if 0<=k-j<n, else 0.
        
        # So grad[j] = 2*a[k_max - j] if 0 <= k_max - j < n, else 0.
        # This is just 2 * reversed_and_shifted a!
        
        # Much simpler than I thought! Let me redo:
        grad = np.zeros(n)
        for j in range(n):
            idx = k_max - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        
        # Gradient w.r.t. z (log-space):
        ga_dot_a = float(np.dot(grad, a))
        grad_z = a * (grad - ga_dot_a)
        
        z = z - lr * grad_z
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        a_new = _softmax_fn(z)
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_z = z.copy()
        
        # Adaptive learning rate
        if sc > best_score * 1.001:
            lr *= 0.5
        elif it % 50 == 0:
            lr = min(lr * 0.98, step_size)
    
    return best_z, best_score


def _make_profile(n, profile_type="linear", param=1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if profile_type == "linear":
        a = 1.0 + param * u
    elif profile_type == "quadratic":
        a = 1.0 + param * u + 0.3 * u**2
    elif profile_type == "cubic":
        a = 1.0 + param * u - 0.5 * u**3
    elif profile_type == "alternating":
        a = np.ones(n)
        a[1::2] = 1.0 + param
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization: L-BFGS continuation + Adam + direct peak chasing.
    Works in log-space (softmax parameterization) on the probability simplex.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and sc < best_score and np.isfinite(sc):
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] ***")
    
    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 2048
        parent = _make_profile(n, "linear", 1.5)
        print(f"No parent, using linear n={n}, score={_score_fn(parent):.10f}")
    
    a0 = parent.copy()
    sc0 = _score_norm_fn(a0)
    update_best(a0, sc0, "parent")
    
    # Convert to log-space
    z = np.log(np.maximum(a0, 1e-15))
    z = z - z.mean()
    
    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    
    # === PHASE 1: L-BFGS with continuation (first ~40% of budget) ===
    phase1_end = t_start + 0.40 * (deadline - t_start)
    print(f"\nPhase 1: L-BFGS continuation, deadline={phase1_end - t_start:.0f}s")
    
    # Adaptive beta schedule
    # Start with beta relative to the peak
    beta_rel_schedule = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    n_betas = len(beta_rel_schedule)
    time_per_beta = (phase1_end - time.monotonic()) / n_betas
    time_per_beta = max(2.0, min(time_per_beta, 60.0))
    
    for i, beta_rel in enumerate(beta_rel_schedule):
        if time.monotonic() > phase1_end:
            break
        
        beta = beta_rel * max_c0
        # Update max_c0 based on current state
        a_curr = _softmax_fn(z)
        c_curr = _full_convolve(a_curr, a_curr)
        max_c0 = float(c_curr.max())
        beta = beta_rel * max_c0
        
        tl = min(time_per_beta, phase1_end - time.monotonic())
        if tl < 1.0:
            break
        
        max_iter = max(50, int(tl * 3))

"""
Multi-stage optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines L-BFGS continuation with adaptive logsumexp smoothing, top-K peak
selection, Adam with cosine annealing, and direct peak-chasing gradient steps.
All optimization performed in log-space (softmax parameterization) on simplex.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_weights(c, beta, top_k=None):
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_work = np.full_like(c, -1e30)
        c_work[idx] = c[idx]
    else:
        idx = None
        c_work = c
    
    cm = float(np.max(c_work))
    c_shifted = (c_work - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, cm
    w = np.zeros_like(c)
    w_valid = e / s
    if idx is not None:
        w[idx] = w_valid
    else:
        w = w_valid.copy()
    return w, cm


def _obj_z(z, beta, n, top_k=None):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta, top_k)
    return 2.0 * n * (cm + beta * np.log(float(np.sum(np.exp((c - cm) / beta)))))


def _obj_z_safe(z, beta, n, top_k=None):
    try:
        val = _obj_z(z, beta, n, top_k)
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return 1e10


def _grad_z(z, beta, n, top_k=None):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    w, cm = _compute_weights(c, beta, top_k)
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _grad_z_safe(z, beta, n, top_k=None):
    try:
        g = _grad_z(z, beta, n, top_k)
        if np.all(np.isfinite(g)):
            return g
    except Exception:
        pass
    return np.zeros(len(z))

    t0 = time.monotonic()

        result = minimize(
            _obj_z_safe, z, args=(beta, n, top_k),
            method='L-BFGS-B', jac=_grad_z_safe,
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10,
                     'maxls': 50, 'maxfun': max_iter * 3}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return result.x
    except Exception:
        pass
    return z


def _adam_run(z, n, deadline, lr, beta_init, top_k=None,
              lr_decay=0.9998, beta_decay=0.995, schedule='exp'):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm_init = float(c.max())
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    best_z = z.copy()
    best_obj = _obj_z_safe(z, beta_init, n, top_k)
    
    max_iter = 200000
    total_iter = max(1000, int((deadline - time.monotonic()) * 50))
    
    for t in range(total_iter):
        if time.monotonic() > deadline:
            break
        
        beta = max(1e-15, beta_init * (beta_decay ** t))
        
        if schedule == 'cosine':
            progress = min(1.0, (t + 1) / total_iter)
            cur_lr = lr * 0.5 * (1.0 + np.cos(np.pi * progress))
            cur_lr = max(cur_lr, lr * 0.01)
        else:
            cur_lr = lr * (lr_decay ** t)
        
        obj, grad = _obj_z_safe(z, beta, n, top_k), _grad_z_safe(z, beta, n, top_k)
        
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_z = z.copy()
        
        t_idx = t + 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t_idx)
        vh = v / (1 - b2 ** t_idx)
        
        z = z - cur_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
    
    # Check if final z is better
    final_obj = _obj_z_safe(z, max(1e-15, beta_init * (beta_decay ** t)), n, top_k)
    if np.isfinite(final_obj) and final_obj < best_obj:
        best_z = z.copy()
    
    return best_z


def _peak_chase(z, n, step_size, deadline, max_iter=10000):
    a = _softmax_fn(z)
    best_z = z.copy()
    best_score = _score_norm_fn(a)
    lr = step_size
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        k_max = int(np.argmax(c))
        cm = c[k_max]
        
        # Gradient of c[k_max] w.r.t. a: 2*a[k_max - j] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_max - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        
        ga_dot_a = float(np.dot(grad, a))
        grad_z = a * (grad - ga_dot_a)
        
        z = z - lr * grad_z
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        a_new = _softmax_fn(z)
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_z = z.copy()
            if it % 100 == 0:
                lr = min(lr * 1.01, step_size)
        else:
            lr *= 0.7
            # Revert
            z = best_z.copy()
        
        if lr < 1e-8:
            break
    
    return best_z, best_score


def _make_profile(n, ptype, param=1.5):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if ptype == "linear":
        a = 1.0 + param * u
    elif ptype == "quadratic":
        a = 1.0 + param * u + 0.3 * u ** 2
    elif ptype == "cubic":
        a = 1.0 + param * u - 0.5 * u ** 3
    elif ptype == "alternating":
        a = np.ones(n)
        a[1::2] = 1.0 + param
    elif ptype == "sinusoidal":
        a = 1.0 + param * np.sin(2 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS continuation + Adam (cosine) + peak chasing.
    Returns best non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Get parent sequence
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={_score_fn(parent):.10f}")
    else:
        n = 2048
        parent = _make_profile(n, "linear", 1.5)
        print(f"No parent, using linear n={n}")
    
    a0 = parent.copy()
    sc0 = _score_norm_fn(a0)
    if sc0 < best_score:
        best_score = sc0
        best_seq = a0.copy()
        print(f"Initial: score={best_score:.10f}")
    
    z = np.log(np.maximum(a0, 1e-15))
    z = z - z.mean()
    
    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    
    # === PHASE 1: L-BFGS continuation (~35% budget) ===
    phase1_end = t_start + 0.35 * (deadline - t_start)
    print(f"\nPhase 1: L-BFGS continuation ({phase1_end - t_start:.0f}s)")
    
    beta_schedule = [0.08, 0.05, 0.03, 0.018, 0.010, 0.006, 0.003, 0.0015, 0.0008, 0.0004]
    n_beta = len(beta_schedule)
    time_per = (phase1_end - time.monotonic()) / n_beta
    time_per = max(3.0, min(time_per, 80.0))
    
    # Also try top-K variants at later stages
    top_k_schedule = [None, None, None, None, 50, 30, 20, 10, 5, 3]
    
    for i, (beta_rel, tk) in enumerate(zip(beta_schedule, top_k_schedule)):
        if time.monotonic() > phase1_end - 2:
            break
        
        a_curr = _softmax_fn(z)
        c_curr = _full_convolve(a_curr, a_curr)
        max_c0 = float(c_curr.max())
        beta = beta_rel * max_c0
        
        tl = min(time_per, phase1_end - time.monotonic() - 2)
        if tl < 2:
            break
        
        max_iter = max(30, int(tl * 2.5))
        z = _lbfgs_run(z, beta, n, max_iter=max_iter, top_k=tk, time_limit=tl)
        
        a_curr = _softmax_fn(z)
        sc = _score_norm_fn(a_curr)
        if sc < best_score:
            best_score = sc
            best_seq = a_curr.copy()
            print(f"  beta={beta_rel}, top_k={tk}: score={sc:.10f} *** NEW BEST ***")
        else:
            print(f"  beta={beta_rel}, top_k={tk}: score={sc:.10f}")
    
    print(f"  Phase 1 done: best={best_score:.10f}")
    
    # === PHASE 2: Adam with cosine annealing (~25% budget) ===
    phase2_end = t_start + 0.60 * (deadline - t_start)
    if time.monotonic() < phase2_end:
        print(f"\nPhase 2: Adam polishing ({phase2_end - t_start:.0f}s)")
        
        a_curr = _softmax_fn(z)
        c_curr = _full_convolve(a_curr, a_curr)
        max_c0 = float(c_curr.max())
        
        # Multiple Adam runs with different configs
        configs = [
            (0.08, 0.015, None, 'cosine'),
            (0.05, 0.008, 30, 'cosine'),
            (0.03, 0.004, 10, 'cosine'),
            (0.02, 0.002, 5, 'exp'),
        ]
        
        n_cfg = len(configs)
        time_per_cfg = (phase2_end - time.monotonic()) / n_cfg
        time_per_cfg = max(10.0, time_per_cfg)
        
        for ci, (lr, beta_frac, tk, sched) in enumerate(configs):
            if time.monotonic() > phase2_end - 5:
                break
            
            beta_init = beta_frac * max_c0
            dl = min(time.monotonic() + time_per_cfg, phase2_end - 5)
            
            z = _adam_run(z, n, dl, lr=lr, beta_init=beta_init,
                         top_k=tk, schedule=sched)
            
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if sc < best_score:
                best_score = sc
                best_seq = a_curr.copy()
                print(f"  Adam cfg{ci}: score={sc:.10f} *** NEW BEST ***")
            else:
                print(f"  Adam cfg{ci}: score={sc:.10f}")
            
            # Update max_c0
            c_curr = _full_convolve(a_curr, a_curr)
            max_c0 = float(c_curr.max())
        
        print(f"  Phase 2 done: best={best_score:.10f}")
    
    # === PHASE 3: Direct peak chasing (~20% budget) ===
    phase3_end = t_start + 0.80 * (deadline - t_start)
    if time.monotonic() < phase3_end:
        print(f"\nPhase 3: Peak chasing ({phase3_end - t_start:.0f}s)")
        
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        n_steps = len(step_sizes)
        time_per_step = (phase3_end - time.monotonic()) / n_steps
        time_per_step = max(3.0, time_per_step)
        
        for si, ss in enumerate(step_sizes):
            if time.monotonic() > phase3_end - 3:
                break
            
            dl = min(time.monotonic() + time_per_step, phase3_end - 3)
            z, sc = _peak_chase(z, n, step_size=ss, deadline=dl, max_iter=20000)
            
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if sc < best_score:
                best_score = sc
                best_seq = a_curr.copy()
                print(f"  Peak ss={ss}: score={sc:.10f} *** NEW BEST ***")
            else:
                print(f"  Peak ss={ss}: score={sc:.10f}")
        
        print(f"  Phase 3 done: best={best_score:.10f}")
    
    # === PHASE 4: Fine-tuning with small perturbations + final polish (~15%) ===
    phase4_end = deadline
    if time.monotonic() < phase4_end and best_seq is not None:
        print(f"\nPhase 4: Final refinement ({phase4_end - time.monotonic():.0f}s)")
        
        remaining = phase4_end - time.monotonic()
        
        # Small random perturbations with brief optimization
        n_perturb = 4
        time_per_pert = remaining / (n_perturb + 3)
        
        for pi in range(n_perturb):
            if time.monotonic() > phase4_end - 20:
                break
            
            noise = 0.02 + 0.01 * pi
            z_p = z + noise * rng.normal(size=n)
            z_p = np.clip(z_p, -60, 60)
            z_p = z_p - z_p.mean()
            
            a_p = _softmax_fn(z_p)
            c_p = _full_convolve(a_p, a_p)
            mc_p = float(c_p.max())
            
            # Brief L-BFGS
            tl = min(time_per_pert * 0.6, phase4_end - 20)
            if tl > 3:
                beta = 0.003 * mc_p
                z_p = _lbfgs_run(z_p, beta, n, max_iter=80, top_k=5, time_limit=tl)
            
            # Brief Adam
            if time.monotonic() < phase4_end - 10:
                a_p = _softmax_fn(z_p)
                c_p = _full_convolve(a_p, a_p)
                mc_p = float(c_p.max())
                dl = min(time.monotonic() + time_per_pert * 0.4, phase4_end - 10)
                z_p = _adam_run(z_p, n, dl, lr=0.015, beta_init=0.002 * mc_p,
                               top_k=3, schedule='cosine')
            
            a_final = _softmax_fn(z_p)
            sc = _score_norm_fn(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                z = z_p.copy()
                print(f"  Perturb {pi}: score={sc:.10f} *** NEW BEST ***")
            else:
                print(f"  Perturb {pi}: score={sc:.10f}")
        
        # Final ultra-fine Adam
        if time.monotonic() < phase4_end - 10:
            a_best = _softmax_fn(z)
            c_best = _full_convolve(a_best, a_best)
            mc_best = float(c_best.max())
            dl = min(phase4_end - 5, time.monotonic() + 30)
            z = _adam_run(z, n, dl, lr=0.008, beta_init=0.0005 * mc_best,
                         top_k=3, schedule='cosine')
            a_final = _softmax_fn(z)
            sc = _score_norm_fn(a_final)
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"  Final Adam: score={sc:.10f} *** NEW BEST ***")
            
            # Final peak chase
            if time.monotonic() < phase4_end - 3:
                dl2 = min(phase4_end - 2, time.monotonic() + 8)
                z, sc = _peak_chase(z, n, step_size=0.0005, deadline=dl2, max_iter=5000)
                a_final = _softmax_fn(z)
                sc = _score_norm_fn(a_final)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_final.copy()
                    print(f"  Final peak: score={sc:.10f} *** NEW BEST ***")
        
        print(f"  Phase 4 done: best={best_score:.10f}")
    
    # === Final validation and return ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile(256, "linear", 1.5)
        
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"  Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"  Eval error: {ex}")
        
        return [float(x) for x in final_seq]
    
    fallback = _make_profile(256, "linear", 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
