from scipy.optimize import linprog
from scipy.optimize import minimize
import time
import sys

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_conv_matrix(a):
    """Build matrix A where (A @ b) = conv(a, b)."""
    n = len(a)
    # A is (2n-1) x n
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n - 1, n))
    for i in range(n):
        A[i:i+n, i] = a
    return A

def _solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = _build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    
    # Use HiGHS solver

                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        return res.x
    return None

        p = p / p.sum()
        s = _score_fn(p)
        if s < best_s:

"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal directions that reduce the autocorrelation peak,
combined with Adam polishing in logit-space for fine-tuning.
Multi-phase approach: LP iterations, Adam polish, multi-start diversity.
"""
import numpy as np
from scipy.optimize import linprog, minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    """Build matrix A where (A @ b) = conv(a, b). Shape: (2n-1) x n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _solve_lp_min_peak(a, time_limit=15.0):
    """
    Solve LP: minimize M subject to conv(a,b)[k] <= M for all k,
    sum(b) = sum(a), b >= 0, M >= 0.
    Returns (b, M_new) or (None, None) on failure.
    """
    n = len(a)
    sum_a = float(a.sum())
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    for i in range(n):
        A_ub[i:i + n, i] = a
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([sum_a])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success and res.x is not None and np.all(np.isfinite(res.x)):
        b = res.x[:n]
        M_new = float(res.x[n])
        return b, M_new
    return None, None


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _adam_polish(a_init, deadline, lr=0.04, beta_frac=0.008, n_max_iter=50000):
    """Adam-based polishing in logit space using logsumexp smooth max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    a = _softmax_fn(z).copy()

    best_a = a_init.copy()
    best_score = _score_fn(a_init)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta = max(1e-10, beta_frac * max_c)

    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 300 == 0:
            beta = max(1e-12, beta * 0.8)
        if t % 2000 == 0:
            lr = max(0.003, lr * 0.92)

    return best_a, best_score


def _lbfgs_polish(a_init, deadline, n):
    """L-BFGS polishing with sharp logsumexp."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta = max(1e-6, 0.002 * max_c0)

    def obj(z):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        log_sum = float(np.log(np.sum(np.exp(c_shifted))))
        return 2.0 * n * (cm + beta * log_sum)

    def grad_fn(z):
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        g = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(g)):
            return np.zeros_like(z)
        return g

    try:
        result = minimize(obj, z, jac=grad_fn, method='L-BFGS-B',
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10,
                                   'maxls': 30, 'maxfun': 1500})
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            sc = _score_fn(a_new)
            if np.isfinite(sc):
                return a_new, sc
    except Exception:
        pass
    return a_init, _score_fn(a_init)


def _make_profile(n, kind, rng=None):
    """Create various profile types."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == 'linear':
        a = 1.0 + 1.2 * t
    elif kind == 'quadratic':
        a = 1.0 + 0.5 * u * u
    elif kind == 'cubic':
        a = 1.0 + 0.8 * u - 0.3 * u ** 3
    elif kind == 'cosine':
        a = np.cos(np.pi * t / 2.0) ** 0.7 + 0.05
    elif kind == 'gaussian':
        a = np.exp(-2.0 * u * u) + 0.1
    elif kind == 'random':
        if rng is None:
            rng = np.random.default_rng(0)
        a = 1.0 + 0.5 * rng.standard_normal(n)
    else:
        a = 1.0 + 1.0 * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    Phase 2: Iterative LP-based peak reduction.
    Phase 3: Adam polishing.
    Phase 4: L-BFGS fine polish.
    Phase 5: Multi-start diversity + final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a0 = parent.copy()
        sc0 = _score_fn(a0)
        print(f"Parent: n={n}, score={sc0:.10f}")
        best_score = sc0
        best_seq = a0.copy()
    else:
        n = 1024
        a0 = _make_profile(n, 'linear')
        sc0 = _score_fn(a0)
        print(f"No parent, linear n={n}, score={sc0:.10f}")
        best_score = sc0
        best_seq = a0.copy()

    # --- Phase 1: Try several initial profiles and LP from each ---
    phase1_end = min(deadline, t_start + 60.0)
    candidates = []
    if parent is not None:
        candidates.append(('parent', parent.copy()))

    for kind in ['linear', 'quadratic', 'cubic', 'cosine', 'gaussian', 'random']:
        for pn in [512, 1024]:
            if time.monotonic() > phase1_end:
                break
            p = _make_profile(pn, kind, rng)
            sc = _score_fn(p)
            candidates.append((f"{kind}_n{pn}", p))
            if sc < best_score:
                best_score = sc
                best_seq = p.copy()
                print(f"  Phase1 NEW BEST: {best_score:.10f} [{kind}_n{pn}]")

    # Do a quick LP step on the parent to get an early improvement
    if parent is not None:
        a_try = parent.copy()
        sc_try = best_score

        A_mat = _build_conv_matrix(a_try)
        M_curr = float(_full_convolve(a_try, a_try).max())
        b_dir, M_new = _solve_lp_min_peak(a_try, time_limit=20.0)
        if b_dir is not None:
            sum_a = float(a_try.sum())
            b_dir = b_dir / float(b_dir.sum()) * sum_a
            a_try, sc_try, t_used = _line_search_step(a_try, b_dir, sum_a)
            if sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()
                print(f"  Phase1 LP NEW BEST: {best_score:.10f} [parent_lp1]")

    # Also try LP on the best non-parent candidate
    if len(candidates) > 1:
        for label, p in candidates[1:4]:
            if time.monotonic() > phase1_end:
                break
            a_try = p.copy()
            b_dir, M_new = _solve_lp_min_peak(a_try, time_limit=10.0)
            if b_dir is not None:
                sum_a = float(a_try.sum())
                b_dir = b_dir / float(b_dir.sum()) * sum_a
                a_try, sc_try, t_used = _line_search_step(a_try, b_dir, sum_a)
                if sc_try < best_score:
                    best_score = sc_try
                    best_seq = a_try.copy()
                    print(f"  Phase1 LP NEW BEST: {best_score:.10f} [{label}_lp]")

    print(f"Phase 1 done: best={best_score:.10f}, n={len(best_seq)}")

    # --- Phase 2: Iterative LP optimization ---
    phase2_end = min(deadline, t_start + 1000.0)
    a_curr = best_seq.copy()
    n_curr = len(a_curr)
    stagnant_count = 0

    iter_count = 0
    while time.monotonic() < phase2_end:
        iter_count += 1
        iter_start = time.monotonic()

        c_curr = _full_convolve(a_curr, a_curr)
        M_curr = float(c_curr.max())
        sc_curr = _score_fn(a_curr)

        # Solve LP
        b_dir, M_lp = _solve_lp_min_peak(a_curr, time_limit=20.0)
        if b_dir is None:
            break

        sum_a = float(a_curr.sum())
        b_sum = float(b_dir.sum())
        if b_sum < 1e-15:
            break
        b_dir = b_dir / b_sum * sum_a

        # Line search
        a_new, sc_new, t_best = _line_search_step(a_curr, b_dir, sum_a)

        if sc_new < sc_curr - 1e-12:
            a_curr = a_new
            stagnant_count = 0
            if sc_new < best_score:
                best_score = sc_new
                best_seq = a_curr.copy()
                if iter_count % 5 == 0 or sc_new < best_score + 1e-8:
                    print(f"  LP iter {iter_count}: score={sc_new:.10f}, t={t_best:.3f}, "
                          f"M_lp={M_lp:.6f}, M_curr={M_curr:.6f}")
        else:
            stagnant_count += 1
            # Try a smaller step or perturbation
            if stagnant_count >= 3:
                # Perturb and try again
                z = np.log(np.maximum(a_curr, 1e-15)) + 0.05 * rng.standard_normal(n_curr)
                z = z - z.mean()
                a_pert = _softmax_fn(z)
                b_dir2, _ = _solve_lp_min_peak(a_pert, time_limit=10.0)
                if b_dir2 is not None:
                    sum_p = float(a_pert.sum())
                    b_dir2 = b_dir2 / float(b_dir2.sum()) * sum_p
                    a_try, sc_try, _ = _line_search_step(a_pert, b_dir2, sum_p)
                    if sc_try < best_score:
                        a_curr = a_try
                        best_score = sc_try
                        best_seq = a_curr.copy()
                        stagnant_count = 0
                        print(f"  LP perturbed iter {iter_count}: score={sc_try:.10f}")
                    else:
                        break
                else:
                    break

        # Check time per iteration
        iter_time = time.monotonic() - iter_start
        if iter_time > 30:
            print(f"  LP iteration taking too long ({iter_time:.1f}s), stopping LP phase")
            break

    print(f"Phase 2 done: iter={iter_count}, best={best_score:.10f}, n={len(best_seq)}")

    # --- Phase 3: Adam polishing ---
    phase3_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        print(f"Phase 3: Adam polish, time remaining={remaining:.1f}s")
        a_pol, sc_pol = _adam_polish(best_seq, phase3_end, lr=0.04, beta_frac=0.005,
                                     n_max_iter=40000)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
            print(f"  Adam polish NEW BEST: {best_score:.10f}")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: L-BFGS fine polish ---
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        n_pol = len(best_seq)
        a_lb, sc_lb = _lbfgs_polish(best_seq, phase4_end, n_pol)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS NEW BEST: {best_score:.10f}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-start with perturbation + re-optimize ---
    phase5_end = deadline
    if time.monotonic() < phase5_end and best_seq is not None:
        n_best = len(best_seq)
        n_starts = 3
        per_start = min(40.0, (phase5_end - time.monotonic()) / max(1, n_starts))

        for i in range(n_starts):
            if time.monotonic() > phase5_end - 10:
                break
            noise_level = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z + noise_level * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax_fn(z)

            # Quick LP step
            b_dir, _ = _solve_lp_min_peak(a_pert, time_limit=10.0)
            if b_dir is None:
                continue
            sum_p = float(a_pert.sum())
            b_dir = b_dir / float(b_dir.sum()) * sum_p
            a_try, sc_try, _ = _line_search_step(a_pert, b_dir, sum_p)

            # Adam polish on the result
            ad_end = min(phase5_end - 5, time.monotonic() + per_start - 10)
            if time.monotonic() < ad_end:
                a_try, sc_try = _adam_polish(a_try, ad_end, lr=0.03, beta_frac=0.003,
                                             n_max_iter=10000)

            if np.isfinite(sc_try) and sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()
                print(f"  Multi-start {i} NEW BEST: {best_score:.10f}")

    # --- Final: one more Adam pass with very small beta ---
    if time.monotonic() < deadline - 3 and best_seq is not None:
        a_final, sc_final = _adam_polish(best_seq, deadline - 2, lr=0.01, beta_frac=0.001,
                                         n_max_iter=20000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"  Final polish NEW BEST: {best_score:.10f}")

    # --- Validation ---
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        fallback = _make_profile(512, 'linear')
        final_seq = fallback
        best_score = _score_fn(fallback)

    sc_final = _score_fn(final_seq)
    print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

    # Authoritative evaluation
    if len(final_seq) <= 8192:
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
                sc_final = sc_eval
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

    return [float(x) for x in final_seq]
