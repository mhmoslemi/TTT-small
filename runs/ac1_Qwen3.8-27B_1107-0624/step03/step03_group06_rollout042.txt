def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_ramp(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = t ** alpha + 1e-6
    return a / a.sum()


def _make_ramp_down(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - t) ** alpha + 1e-6
    return a / a.sum()


def _make_gaussian(n, sigma=0.3, center=0.7):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2) + 1e-8
    return a / a.sum()


def _make_asym_gaussian(n, sigma=0.35, center=0.6):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = a * (1.0 + 0.5 * t) + 1e-8
    return a / a.sum()


def _make_cosine(n, skew=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * (t - 0.5 + skew * (t - 0.5)) ** 2) ** 2 + 1e-8
    return a / a.sum()


def _make_power_asym(n, p=2.0, q=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = t ** p * (1.0 - t) ** q + 1e-8
    return a / a.sum()


def _solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _lp_refine(a0, deadline, max_iter=50000, inflate_start=1.003, inflate_min=1.0002):
    """LP-based refinement with adaptive inflation and line search."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    no_imp = 0
    inflate = inflate_start

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        g = _solve_lp(a, M_lp, n)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(0.01, inflate * 1.2)
            if no_imp > 400:
                break
            continue

        g = g / sum_g

        # Line search
        best_t = 0.0
        best_sc = best_score
        ts = np.linspace(0.02, 1.0, 12)
        for t in ts:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = _fft_convolve(a_new)
            sc = float(2.0 * n * c_new.max())
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t > 0 and best_sc < best_score - 1e-14:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            a_new = a_new / s_new
            best_score = best_sc
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 50 == 49:
                inflate = max(inflate_min, inflate * 0.94)
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                inflate = min(0.015, inflate * 1.1)
            if no_imp > max(200, 2 * n):
                a = best_a.copy()
                inflate = inflate_start
                no_imp = 0

    return best_a


def _coord_descent(a0, deadline, max_passes=300, rng=None, grid_size=25):
    """
    Coordinate descent: for each coordinate p, find delta that minimizes
    the score after updating a[p] += delta and renormalizing.
    Uses grid search over delta for robustness (handles non-convexity).
    """
    a = np.asarray(a0, dtype=np.float64).copy().ravel()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_cmax = float(c.max())

    if rng is None:
        rng = np.random.default_rng()

    no_imp = 0
    k_vals = np.arange(2 * n - 1)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        pass_improved = False
        old_cmax = float(c.max())

        for pi in order:
            pi = int(pi)
            a_p = a[pi]
            if a_p < 1e-14:
                continue

            # b[k] = a[k - pi] for valid k (shifted a)
            b = np.zeros(2 * n - 1)
            b[pi:pi + n] = a

            is_2p = (k_vals == 2 * pi).astype(np.float64)

            # Grid search over delta
            # delta range: [-a_p * 0.95, a_p * 2.0] or so
            d_lo = -a_p * 0.95
            d_hi = max(a_p * 2.0, 0.02)
            deltas = np.linspace(d_lo, d_hi, grid_size)

            # Evaluate score at each delta
            # After update: a_new = a with a[pi] += delta, then normalize by (1+delta)
            # c_new = (c + 2*delta*b + delta^2*is_2p) / (1+delta)^2
            best_delta = 0.0
            best_val = float(c.max())

            for d in deltas:
                if d <= -1.0 + 1e-10:
                    continue
                denom = (1.0 + d) ** 2
                c_new = (c + 2.0 * d * b + d * d * is_2p) / denom
                val = float(c_new.max())
                if val < best_val - 1e-15:
                    best_val = val
                    best_delta = d

            if best_delta != 0.0:
                # Apply update
                a[pi] = a_p + best_delta
                if a[pi] < 0:
                    a[pi] = 0.0
                    best_delta = -a_p
                s_new = 1.0 + best_delta
                if s_new > 1e-10:
                    a = a / s_new
                # Update c incrementally (then fix up)
                # Actually, just recompute if improvement is significant
                c_new = (c + 2.0 * best_delta * b + best_delta * best_delta * is_2p)
                c = c_new / (1.0 + best_delta) ** 2
                c = np.maximum(c, 0.0)
                pass_improved = True

        cmax = float(c.max())
        if pass_improved and cmax < old_cmax - 1e-14:
            if cmax < best_cmax - 1e-14:
                best_cmax = cmax
                best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 15:
                # Try a random kick
                idx = rng.integers(0, n)
                a[idx] *= (1.0 + 0.1 * rng.standard_normal())
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                c = _fft_convolve(a)
                no_imp = 0

    return best_a


def _smooth_max_grad(a, beta):
    """Compute subgradient approximation using smooth max (softmax)."""
    n = len(a)
    c = _fft_convolve(a)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    N_pad = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N_pad)
    fa_rev = np.fft.rfft(a[::-1], N_pad)
    corr = np.fft.irfft(fw * fa_rev, N_pad)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, mval


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _gradient_polish(a0, deadline, max_iter=50000):
    """Subgradient/gradient polish with momentum and adaptive beta."""
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_score = float(2.0 * n * c.max())

    beta = max(1e-6, 0.03 * float(c.max()))
    lr = 0.05
    base_lr = lr
    momentum = 0.85
    v = np.zeros(n)
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_nest = a + momentum * v
        a_nest = np.maximum(a_nest, 0.0)
        s_nest = a_nest.sum()
        if s_nest < 1e-10:
            continue
        a_nest = a_nest / s_nest

        grad, maxc = _smooth_max_grad(a_nest, beta)

        if not np.all(np.isfinite(grad)):
            beta = max(1e-6, beta * 3.0)
            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad
        a_new = _project_simplex(a + v)
        a = a_new

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                lr = max(0.005, lr * 0.93)
            if no_imp > max(2000, 3 * n):
                a = best_a.copy()
                v = np.zeros(n)
                beta = max(1e-7, 0.03 * float(_fft_convolve(a).max()))
                lr = base_lr
                no_imp = 0

        if it % 50 == 49:
            beta = max(1e-8, beta * 0.85)

    return best_a


def _subgradient_peak_equalize(a0, deadline, max_iter=30000):
    """
    Peak equalization: directly reduce the peak by adjusting coordinates
    that contribute most to it. Uses hard subgradient (no smoothing).
    """
    a = np.asarray(a0, dtype=np.float64).copy().ravel()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    c = _fft_convolve(a)
    best_a = a.copy()
    best_cmax = float(c.max())

    lr = 0.02
    no_imp = 0
    rng = np.random.default_rng(123)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        cmax = float(c.max())

        # Find peak positions
        threshold = cmax * 0.999
        peak_mask = c >= threshold
        n_peaks = int(peak_mask.sum())
        if n_peaks == 0:
            break

        # Compute subgradient: for each p, grad[p] = 2 * sum_{k in peaks} a[k-p] (if valid)
        grad = np.zeros(n)
        peak_indices = np.where(peak_mask)[0]
        for ki in peak_indices:
            ki = int(ki)
            # a[k-p] is valid for p in [ki - (n-1), ki] ∩ [0, n-1]
            p_lo = max(0, ki - (n - 1))
            p_hi = min(n - 1, ki)
            if p_lo <= p_hi:
                # a[ki - p] for p in [p_lo, p_hi]
                idx = ki - np.arange(p_lo, p_hi + 1)
                grad[p_lo:p_hi + 1] += 2.0 * a[idx]

        # Add the diagonal term: if 2p is a peak, grad[p] += a[p]
        for p in range(n):
            if 2 * p < 2 * n - 1 and peak_mask[2 * p]:
                grad[p] += a[p]

        # Normalized gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            no_imp += 1
            if no_imp > 500:
                break
            continue

        lr_adapt = lr / (1.0 + 0.01 * it * 0.01)
        a_new = a - lr_adapt * grad / n
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        cmax_new = float(c_new.max())

        if cmax_new < best_cmax - 1e-14:
            best_cmax = cmax_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr *= 0.9
            if no_imp > 2000:
                a = best_a.copy()
                lr = 0.02
                no_imp = 0
                # Random perturbation
                idx = rng.integers(0, n, size=max(1, n // 10))
                a[idx] *= 1.0 + 0.05 * rng.standard_normal(len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimization combining coordinate descent, LP refinement,
    subgradient peak equalization, and gradient polish.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Define configurations: (name, length, initialization)
    configs = []
    lengths = [256, 384, 512, 768, 1024]

    for n_t in lengths:
        if parent is not None:
            configs.append(('parent', n_t, _interpolate(parent, n_t)))
        configs.append(('ramp1', n_t, _make_ramp(n_t, 1.0)))
        configs.append(('ramp2', n_t, _make_ramp(n_t, 2.0)))
        configs.append(('rampd1', n_t, _make_ramp_down(n_t, 1.0)))
        configs.append(('rampd2', n_t, _make_ramp_down(n_t, 2.0)))
        configs.append(('gauss', n_t, _make_gaussian(n_t)))
        configs.append(('asym_gauss', n_t, _make_asym_gaussian(n_t)))
        configs.append(('cosine', n_t, _make_cosine(n_t)))
        configs.append(('power', n_t, _make_power_asym(n_t, 2.0, 0.5)))
        configs.append(('power2', n_t, _make_power_asym(n_t, 3.0, 1.0)))

    # Score and sort
    scored = []
    for name, n_t, a0 in configs:
        sc = _score(a0)
        scored.append((sc, name, n_t, a0))
    scored.sort(key=lambda x: x[0])

    print(f"\nTop 10 initializations:")
    for sc, name, n_t, _ in scored[:10]:
        print(f"  {name} n={n_t}: {sc:.8f}")

    # Update best from initializations
    for sc, name, n_t, a0 in scored:
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a0.copy()

    # Phase 1: Run various optimization methods on top configs
    # Allocate time to each config based on quality
    phase1_deadline = t_start + budget_s * 0.55

    for sc0, name, n_t, a0 in scored:
        if time.monotonic() > phase1_deadline:
            break
        if sc0 >= best_score + 0.15:
            continue

        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break

        # Allocate time based on initial quality
        if sc0 < best_score + 0.02:
            alloc = min(120.0, remaining * 0.3)
        elif sc0 < best_score + 0.05:
            alloc = min(80.0, remaining * 0.2)
        else:
            alloc = min(40.0, remaining * 0.1)

        if alloc < 10:
            continue

        print(f"\nPhase1: {name} n={n_t} init={sc0:.8f} alloc={alloc:.0f}s")
        end1 = time.monotonic() + alloc * 0.4
        end2 = time.monotonic() + alloc * 0.7
        end3 = time.monotonic() + alloc

        # Step 1: Coordinate descent
        a1 = _coord_descent(a0, end1, max_passes=100, rng=rng)
        sc1 = _score(a1)
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()
            print(f"  coord_desc: {sc1:.10f}")

        # Step 2: LP refine on the result
        a2 = _lp_refine(a1, end2, inflate_start=1.003)
        sc2 = _score(a2)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(f"  lp_refine: {sc2:.10f}")

        # Step 3: Coord descent again on LP result
        a3 = _coord_descent(a2, end3, max_passes=80, rng=rng)
        sc3 = _score(a3)
        if sc3 < best_score:
            best_score = sc3
            best_seq = a3.copy()
            print(f"  coord_desc2: {sc3:.10f}")

    # Phase 2: Focus on the best sequence with more intensive optimization
    if best_seq is not None:
        print(f"\nPhase2: intensive optimization on best (score={best_score:.10f}, n={best_seq.size})")
        phase2_deadline = t_start + budget_s * 0.80

        # Gradient polish
        if time.monotonic() < phase2_deadline - 100:
            gp_end = time.monotonic() + min(150, phase2_deadline - time.monotonic() - 50)
            a_gp = _gradient_polish(best_seq, gp_end)
            sc_gp = _score(a_gp)
            if sc_gp < best_score:
                best_score = sc_gp
                best_seq = a_gp.copy()
            print(f"  gradient: {sc_gp:.10f}")

        # Peak equalization
        if time.monotonic() < phase2_deadline - 60:
            pe_end = time.monotonic() + min(100, phase2_deadline - time.monotonic() - 30)
            a_pe = _subgradient_peak_equalize(best_seq, pe_end)
            sc_pe = _score(a_pe)
            if sc_pe < best_score:
                best_score = sc_pe
                best_seq = a_pe.copy()
            print(f"  peak_eq: {sc_pe:.10f}")

        # More coord descent
        if time.monotonic() < phase2_deadline - 30:
            cd_end = time.monotonic() + min(100, phase2_deadline - time.monotonic() - 10)
            a_cd = _coord_descent(best_seq, cd_end, max_passes=200, rng=rng, grid_size=30)
            sc_cd = _score(a_cd)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
            print(f"  coord_desc: {sc_cd:.10f}")

    # Phase 3: Try different lengths by interpolating the best
    if best_seq is not None:
        print(f"\nPhase3: length exploration (best score={best_score:.10f})")
        phase3_deadline = t_start + budget_s * 0.92

        for n_try in [512, 640, 768, 896, 1024, 1280]:
            if time.monotonic() > phase3_deadline:
                break
            if n_try == best_seq.size:
                continue

            a_interp = _interpolate(best_seq, n_try)
            sc_interp = _score(a_interp)
            print(f"  n={n_try} interp: {sc_interp:.8f}")

            if sc_interp < best_score + 0.05:
                # Try to improve
                alloc = min(60.0, phase3_deadline - time.monotonic() - 20)
                if alloc < 15:
                    continue

                a_c = _coord_descent(a_interp, time.monotonic() + alloc * 0.5, max_passes=80, rng=rng)
                sc_c = _score(a_c)
                if sc_c < best_score:
                    best_score = sc_c
                    best_seq = a_c.copy()
                    print(f"    coord: {sc_c:.10f}")

                a_l = _lp_refine(a_c, time.monotonic() + alloc * 0.9, inflate_start=1.002)
                sc_l = _score(a_l)
                if sc_l < best_score:
                    best_score = sc_l
                    best_seq = a_l.copy()
                    print(f"    lp: {sc_l:.10f}")

    # Phase 4: Final polish
    if best_seq is not None and time.monotonic() < deadline:
        print(f"\nPhase4: final polish")
        final_end = deadline

        # Short gradient
        if time.monotonic() < final_end - 30:
            g_end = min(time.monotonic() + 60, final_end - 20)
            a_g = _gradient_polish(best_seq, g_end, max_iter=20000)
            sc_g = _score(a_g)
            if sc_g < best_score:
                best_score = sc_g
                best_seq = a_g.copy()
            print(f"  final_grad: {sc_g:.10f}")

        # Final coord descent
        if time.monotonic() < final_end - 10:
            c_end = min(time.monotonic() + 40, final_end - 5)
            a_c = _coord_descent(best_seq, c_end, max_passes=50, rng=rng, grid_size=35)
            sc_c = _score(a_c)
            if sc_c < best_score:
                best_score = sc_c
                best_seq = a_c.copy()
            print(f"  final_coord: {sc_c:.10f}")

    # Fallback
    if best_seq is None:
        best_seq = _make_ramp(384, 1.5)
        best_score = _score(best_seq)

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_ramp(256, 1.0)

    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nevaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Eval error: {e}")

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
