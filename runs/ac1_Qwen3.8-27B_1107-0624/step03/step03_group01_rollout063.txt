def _full_convolve_fft(a, b):
    n, m = len(a), len(b)

    while N < out_n:

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lbfgs_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):

    return grad

"""
Coordinate descent with exact per-coordinate minimization for minimizing
2*n*max(conv(a,a))/sum(a)^2. Exploits the piecewise-quadratic structure when
all but one coordinate are fixed, using bisection on the subgradient.
Includes diverse initializations, multiple sequence lengths, randomized
coordinate ordering, and L-BFGS polishing with logsumexp smooth maximization.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _coord_minimize(a, c, j, n, max_iter=80):
    """Min h(t)=max_i(base_i+slope_i*t+quad_i*t^2) over t>=0 via bisection."""
    aj = a[j]
    c_len = 2 * n - 1
    i_lo = max(0, j)
    i_hi = min(c_len - 1, j + n - 1)

    # Build base, slopes, quads arrays (full length for simplicity)
    slopes = np.zeros(c_len)
    quads = np.zeros(c_len)

    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        slopes[idx] = 2.0 * a[idx - j]

    if 0 <= 2 * j <= c_len - 1:
        slopes[2 * j] = 0.0
        quads[2 * j] = 1.0

    base = c - slopes * aj - quads * aj * aj

    # Check if t=0 is optimal
    vals_0 = base
    max_0 = vals_0.max()
    active_0 = vals_0 >= max_0 - 1e-14
    if not active_0.any():
        active_0[int(np.argmax(vals_0))] = True
    derivs_0 = slopes[active_0]
    if derivs_0.min() >= 0:
        return 0.0

    # Bisection on right subgradient
    t_lo = 0.0
    t_hi = max(aj * 3.0, 1.0 / n) + 0.01

    for _ in range(max_iter):
        t_mid = (t_lo + t_hi) * 0.5
        vals = base + slopes * t_mid + quads * t_mid * t_mid
        max_val = vals.max()
        active = vals >= max_val - 1e-13
        if not active.any():
            active[int(np.argmax(vals))] = True
        derivs = slopes[active] + 2.0 * quads[active] * t_mid
        min_d = derivs.min()
        max_d = derivs.max()

        if max_d < 0:
            t_lo = t_mid
        elif min_d > 0:
            t_hi = t_mid
        else:
            t_lo = t_hi = t_mid
            break

    t_opt = (t_lo + t_hi) * 0.5

    # Safety: ensure t_opt doesn't worsen vs t=0
    h_opt = (base + slopes * t_opt + quads * t_opt * t_opt).max()
    h_zero = base.max()
    if h_zero < h_opt - 1e-15:
        t_opt = 0.0

    return max(0.0, t_opt)


def _cd_sweeps(a_init, deadline, max_sweeps=500, seed=0):
    """Coordinate descent with incremental convolution updates."""
    n = len(a_init)
    a = a_init.copy()
    c = _full_convolve(a, a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(seed)

    stall_count = 0
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            return best_a, best_score

        order = list(range(n))
        rng.shuffle(order)

        any_significant_change = False
        for j in order:
            if time.monotonic() > deadline:
                return best_a, best_score

            old_val = a[j]
            t_opt = _coord_minimize(a, c, j, n)
            delta = t_opt - old_val

            if abs(delta) < 1e-16:
                continue

            if abs(delta) > 1e-10:
                any_significant_change = True

            a[j] = t_opt

            # Incremental update of c
            i_lo = max(0, j)
            i_hi = min(2 * n - 2, j + n - 1)
            two_j = 2 * j

            if i_lo <= i_hi:
                if i_lo <= two_j <= i_hi:
                    if two_j > i_lo:
                        li = np.arange(i_lo, two_j)
                        c[li] += 2.0 * delta * a[li - j]
                    if two_j < i_hi:
                        ri = np.arange(two_j + 1, i_hi + 1)
                        c[ri] += 2.0 * delta * a[ri - j]
                    c[two_j] += 2.0 * old_val * delta + delta * delta
                else:
                    idx = np.arange(i_lo, i_hi + 1)
                    c[idx] += 2.0 * delta * a[idx - j]

        # Normalize
        s = a.sum()
        if s < 1e-15:
            return best_a, best_score
        a = a / s

        # Recompute c each sweep to prevent drift
        c = _full_convolve(a, a)

        sc = _score_norm(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            stall_count = 0
            if sweep < 20 or sweep % 20 == 0:
                print(f"      CD sweep {sweep}: score={best_score:.10f}")
        else:
            stall_count += 1
            if stall_count > 30:
                # Add small perturbation to escape plateau
                a += 0.001 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c = _full_convolve(a, a)
                stall_count = 0

        if not any_significant_change:
            if sweep > 5:
                break

    return best_a, best_score


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lbfgs_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    log_sum = float(np.log(np.sum(np.exp(c_shifted))))
    return 2.0 * n * (cm + beta * log_sum)


def _lbfgs_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _make_profile(n, kind, **kwargs):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == 'linear':
        slope = kwargs.get('slope', 1.0)
        a = 1.0 + slope * (t - 0.5)
    elif kind == 'quadratic':
        a2 = kwargs.get('a2', 0.3)
        a1 = kwargs.get('a1', 0.0)
        a = 1.0 + a1 * u + a2 * u ** 2
    elif kind == 'cosine':
        power = kwargs.get('power', 1.0)
        a = np.cos(np.pi * (t - 0.5)) ** power
    elif kind == 'bell':
        width = kwargs.get('width', 0.3)
        a = np.exp(-0.5 * (u / width) ** 2)
    elif kind == 'U_shape':
        depth = kwargs.get('depth', 0.5)
        a = 1.0 + depth * u ** 2
    elif kind == 'W_shape':
        k = kwargs.get('k', 2)
        depth = kwargs.get('depth', 0.3)
        a = 1.0 + depth * np.cos(2.0 * np.pi * k * t)
    elif kind == 'random':
        rng = np.random.default_rng(kwargs.get('seed', 0))
        a = rng.random(n) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        except Exception as ex:
            print(f"Eval failed: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, 'linear', slope=1.0)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
