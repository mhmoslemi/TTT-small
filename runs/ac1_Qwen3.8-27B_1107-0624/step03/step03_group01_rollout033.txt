import scipy.optimize as so

def _score_norm(a):
    """Score = 2*n*max(conv(a,a)) assuming a is normalized (sum=1)."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * float(c.max())

def _build_C(f):
    """Build matrix C where (C@b)[k] = conv(f,b)[k]. C is (2n-1) x n."""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(k + 1, n)
        if js < je:
            # C[k, j] = f[k-j] for j in [js, je)
            C[k, js:je] = f[k-je+1 : k-js+1][::-1]
    return C

    """Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0. Return b or None."""
    n = len(f)
    C = _build_C(f)

    b_ub = np.full(2*n-1, M)

                grad[m] = 2.0 * a[km]

        a_new = a - lr_curr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        sc = _score_norm(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            lr_curr *= 0.9

    return best_a, best_sc


def _make_profile(n, kind='linear', slope=1.0):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if kind == 'linear':
        a = 1.0 + slope * u
    elif kind == 'quad':
        a = 1.0 + slope * u + 0.3 * u ** 2
    elif kind == 'cubic':
        a = 1.0 + slope * u - 0.3 * u ** 3
    elif kind == 'cosine':
        a = 0.5 + 0.5 * np.cos(np.pi * u)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_to_length(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_a = None
    best_score = np.inf

    def update_best(a, sc, label=""):
        nonlocal best_a, best_score
        if a is not None and np.isfinite(sc) and sc < best_score:
            best_a = np.asarray(a, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_a)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score_norm(parent):.10f}")
        update_best(parent, _score_norm(parent), "parent")
    else:
        n_def = 2048
        parent = _make_profile(n_def, 'linear', 1.0)
        print(f"No parent, using linear profile n={n_def}, score={_score_norm(parent):.10f}")
        update_best(parent, _score_norm(parent), "default")

    # Phase 1: Try a few alternative starting points with quick LP
    phase1_end = min(deadline, t_start + 150.0)
    n_base = len(best_a)
    alternatives = []
    for kind, slope in [('linear', 0.5), ('linear', 1.5), ('quad', 1.0),
                        ('cosine', 0.0), ('cubic', 1.0)]:
        if time.monotonic() > phase1_end:
            break
        p = _make_profile(n_base, kind, slope)
        sc = _score_norm(p)
        alternatives.append((sc, p, f"{kind}_{slope}"))
        update_best(p, sc, f"init_{kind}_{slope}")

    # Quick LP on best alternative (if different from parent)
    alternatives.sort(key=lambda x: x[0])
    for sc_alt, p_alt, label in alternatives[:2]:
        if time.monotonic() > phase1_end:
            break
        if sc_alt > best_score * 1.02:
            break
        for _ in range(5):
            a_new, sc_new = _lp_step(p_alt, t_candidates=[0.3, 0.2, 0.1])
            if sc_new < sc_alt:
                p_alt = a_new
                sc_alt = sc_new
            else:
                break
        update_best(p_alt, sc_alt, f"quickLP_{label}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # Phase 2: Main LP optimization on the best sequence
    phase2_end = min(deadline, t_start + 1300.0)
    a_curr = best_a.copy()
    sc_curr = best_score
    n_lp = len(a_curr)
    iterCount = 0
    stall_count = 0
    t_adaptive = 0.5

    while time.monotonic() < phase2_end:
        iterCount += 1
        a_new, sc_new = _lp_step(a_curr, t_candidates=[t_adaptive, 0.3 * t_adaptive,
                                                       0.1 * t_adaptive, 0.05 * t_adaptive])
        if sc_new < sc_curr * (1 - 1e-12):
            a_curr = a_new
            sc_curr = sc_new
            stall_count = 0
            if iterCount % 20 == 1:
                print(f"  LP iter {iterCount}: score={sc_curr:.10f}")
        else:
            stall_count += 1
            if stall_count > 30:
                # Reduce step size and add small perturbation
                t_adaptive *= 0.7
                if t_adaptive < 0.01:
                    # Perturb and try again
                    z = np.log(np.maximum(a_curr, 1e-15))
                    z = z - z.mean()
                    z += 0.05 * rng.normal(size=n_lp)
                    z = np.clip(z, -50, 50)
                    z = z - z.mean()
                    a_curr = np.exp(z)
                    a_curr = a_curr / a_curr.sum()
                    sc_curr = _score_norm(a_curr)
                    t_adaptive = 0.3
                    stall_count = 0
                else:
                    a_new, sc_new = _lp_step(a_curr, t_candidates=[0.05, 0.02, 0.01])
                    if sc_new < sc_curr:
                        a_curr = a_new
                        sc_curr = sc_new
                        stall_count = 0

        # Periodically update best
        if sc_curr < best_score:
            update_best(a_curr, sc_curr, f"LP_{iterCount}")

    print(f"Phase 2 done: best={best_score:.10f}, LP iters={iterCount}")

    # Phase 3: Adam polish
    if time.monotonic() < deadline - 60 and best_a is not None:
        phase3_end = min(deadline - 60, t_start + 1600.0)
        a_new, sc_new = _adam_polish(best_a, phase3_end, lr=0.04, beta_frac=0.008)
        update_best(a_new, sc_new, "adam_polish")
        print(f"Phase 3 done: best={best_score:.10f}")

    # Phase 4: Subgradient polish on the best
    if time.monotonic() < deadline - 40 and best_a is not None:
        phase4_end = min(deadline - 40, t_start + 1700.0)
        a_new, sc_new = _subgrad_polish(best_a, phase4_end, lr=0.0005)
        update_best(a_new, sc_new, "subgrad")
        print(f"Phase 4 done: best={best_score:.10f}")

    # Phase 5: Try larger length via interpolation + LP
    if time.monotonic() < deadline - 30 and best_a is not None:
        n_curr = len(best_a)
        if n_curr <= 2048:
            n_new = n_curr * 2
            a_interp = _interp_to_length(best_a, n_new)
            sc_interp = _score_norm(a_interp)
            update_best(a_interp, sc_interp, f"interp_n{n_new}")

            if time.monotonic() < deadline - 30:
                for _ in range(30):
                    a_new, sc_new = _lp_step(a_interp, t_candidates=[0.3, 0.15, 0.05])
                    if sc_new < sc_interp:
                        a_interp = a_new
                        sc_interp = sc_new
                        update_best(a_interp, sc_interp, f"LP_n{n_new}")
                    else:
                        break
            print(f"Phase 5 done: best={best_score:.10f}")

    # Phase 6: Final Adam on the overall best
    if time.monotonic() < deadline - 5 and best_a is not None:
        a_new, sc_new = _adam_polish(best_a, deadline - 5, lr=0.02, beta_frac=0.003)
        update_best(a_new, sc_new, "final_adam")

    # Final validation
    if best_a is not None:
        final_seq = np.clip(best_a, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
            # Rescale so that sum is exactly something safe
            final_seq = final_seq * s  # keep sum=1
        else:
            final_seq = _make_profile(256, 'linear', 1.0)

        sc_final = _score_norm(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Evaluator check: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profile(256, 'linear', 1.0)
    print(f"FALLBACK: score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
