        return a / s
    return np.ones_like(a) / len(a)


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _smooth_max_weight(c, beta):
    m = float(c.max())
    x = (c - m) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return m, np.zeros_like(c)
    w = ex / s
    return m, w


def _subgradient_step(a, beta, lr):
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_weight(c, beta)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h
    return grad, float(c.max())


def _logit_subgradient_optimize(a, deadline, initial_beta=None, initial_lr=0.02):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    if initial_beta is not None:
        beta = initial_beta
    else:
        beta = max(1e-6, 0.02 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= float(np.mean(z))
    lr = initial_lr

    no_imp = 0
    patience = max(2000, 4 * n)

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        grad, c_max_cur = _subgradient_step(a, beta, lr)
        dot = float(np.dot(a, grad))
        gz = a * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.001, lr * 0.95)
                beta = max(1e-10, beta * 0.8)

        if it % 1000 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > patience:
            break

    return best_a


def _lp_peel(a, deadline, n_peel_iters=100000):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    inflate = 1.00001
    blend = 0.05
    no_imp = 0
    prev_x = None

    for it in range(n_peel_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        A_mat = _make_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)

        kwargs = dict(
            c=c_obj,
            A_ub=A_mat,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs'
        )
        if prev_x is not None:
            kwargs['x0'] = prev_x

        try:
            res = linprog(**kwargs, options={'presolve': True, 'time_limit': 5})
        except Exception:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')

        if not res.success:
            if no_imp > 50:
                inflate = min(1.01, inflate * 1.5)
            no_imp += 1
            if no_imp > 500:
                break
            continue

        b_sol = res.x
        sum_b = float(b_sol.sum())
        prev_x = b_sol.copy()

        if sum_b <= 1.0 + 1e-8:
            no_imp += 1
            if no_imp % 20 == 0:
                inflate = min(1.005, inflate * 1.2)
            if no_imp > 300:
                inflate = 1.00001
                blend = 0.05
                no_imp = 0
            if no_imp > 2000:
                break
            continue

        b_sol = np.maximum(b_sol, 0.0)
        b_sol = b_sol / b_sol.sum()

        a_new = (1.0 - blend) * a + blend * b_sol
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            if it % 500 == 0:
                inflate = max(1.000005, inflate * 0.85)
                blend = max(0.01, blend * 0.98)
            prev_x = _make_conv_matrix(a_new) @ np.ones(n) * 0 + prev_x
        else:
            no_imp += 1
            if no_imp % 10 == 0:
                blend = max(0.005, blend * 0.95)
            if no_imp % 50 == 0:
                inflate = min(1.001, inflate * 1.1)
            if no_imp > 500:
                a = best_a.copy()
                no_imp = 0
                blend = 0.05
                inflate = 1.00001

    return best_a


def _lp_peel_aggressive(a, deadline):
    """More aggressive LP peeling with adaptive parameters."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    c = _fft_convolve(a)
    best_M = float(c.max())
    best_score = float(2.0 * n * best_M)

    phase = 0
    phases_params = [
        {'inflate': 1.00001, 'blend': 0.1},
        {'inflate': 1.000005, 'blend': 0.05},
        {'inflate': 1.000001, 'blend': 0.02},
        {'inflate': 1.0000005, 'blend': 0.01},
    ]

    total_iters = 0
    while time.monotonic() < deadline:
        params = phases_params[min(phase, len(phases_params) - 1)]
        inflate = params['inflate']
        blend = params['blend']
        no_imp = 0
        phase_start = time.monotonic()

        while time.monotonic() < deadline and no_imp < 10000:
            c = _fft_convolve(a)
            M = float(c.max())
            M_lp = M * inflate

            A_mat = _make_conv_matrix(a)
            c_obj = -np.ones(n)
            b_ub = np.full(2 * n - 1, M_lp)

            try:
                res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                              bounds=[(0, None)] * n, method='highs',
                              options={'presolve': True, 'time_limit': 3})
            except Exception:
                break

            if not res.success:
                no_imp += 1
                if no_imp > 200:
                    break
                continue

            b_sol = np.maximum(res.x, 0.0)
            sum_b = float(b_sol.sum())

            if sum_b <= 1.0 + 1e-8:
                no_imp += 1
                if no_imp % 100 == 0:
                    inflate = min(1.001, inflate * 1.3)
                continue

            b_sol = b_sol / sum_b
            a_new = (1.0 - blend) * a + blend * b_sol
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

            c_new = _fft_convolve(a_new)
            M_new = float(c_new.max())
            score_new = float(2.0 * n * M_new)

            total_iters += 1

            if score_new < best_score - 1e-14:
                best_score = score_new
                best_M = M_new
                best_a = a_new.copy()
                a = a_new
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 50 == 0:
                    blend = max(0.003, blend * 0.9)

        # Advance phase if this phase made progress
        elapsed_phase = time.monotonic() - phase_start
        if elapsed_phase > 2.0:
            phase += 1
        elif no_imp >= 10000:
            phase += 1

    return best_a


def _peak_shave_refined(a, deadline, n_peaks=20):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    strength = 0.02

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        c_max = float(c.max())
        thresh = np.percentile(c, 100.0 - 100.0 * n_peaks / len(c))
        top_mask = c > max(thresh, c_max * 0.95)
        top_idx = np.where(top_mask)[0]

        if len(top_idx) == 0:
            break

        weights = c[top_idx] / c[top_idx].sum()

        pressure = np.zeros(n)
        N = _next_pow2(2 * n - 1)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                pressure[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]

        a_new = a * np.exp(-strength * pressure / (pressure.max() + 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        score_new = _score(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_imp = 0
            strength = min(0.1, strength * 1.005)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                strength = max(0.005, strength * 0.9)
            if no_imp > 2000:
                a = best_a.copy()
                strength = 0.02
                no_imp = 0

    return best_a


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0.copy() / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase LP-peeling optimization.
    Focuses on tight LP-based refinement with adaptive parameters,
    combined with subgradient polish and peak shaving.
    """
    t_start = time.monotonic()
    hard_deadline = t_start + budget_s - 10.0
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = np.maximum(p, 0.0)
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception as e:
        print(f"Parent load error: {e}", flush=True)

    if parent is None:
        n = 768
        t = np.linspace(0, 1, n)
        parent = 1.5 + t
        parent = np.maximum(parent, 0.1)
        parent = parent / parent.sum()
        best_score = _score(parent)
        best_seq = parent.copy()
        print(f"Generated fallback parent: score={best_score:.10f}, n={n}", flush=True)

    # Try multiple lengths
    n_parent = parent.size
    lengths_to_try = [n_parent]
    for alt_n in [n_parent + 128, n_parent - 128, 512, 1024]:
        if 256 <= alt_n <= 2048 and alt_n not in lengths_to_try:
            lengths_to_try.append(alt_n)

    # Phase 1: Quick evaluation of different lengths
    print(f"\nPhase 1: Multi-length evaluation", flush=True)
    candidates = []
    for n_t in lengths_to_try:
        a_t = _interpolate_profile(parent, n_t)
        sc_t = _score(a_t)
        candidates.append((sc_t, n_t, a_t))
        print(f"  n={n_t}: score={sc_t:.10f}", flush=True)

    candidates.sort(key=lambda x: x[0])
    best_sc_1, best_n_1, best_a_1 = candidates[0]
    if best_sc_1 < best_score:
        best_score = best_sc_1
        best_seq = best_a_1.copy()

    # Phase 2: LP peeling on best candidate
    print(f"\nPhase 2: LP peeling (n={best_n_1}), start={best_score:.10f}", flush=True)
    lp_deadline = t_start + budget_s * 0.55
    a_lp = _lp_peel_aggressive(best_a_1, lp_deadline)
    sc_lp = _score(a_lp)
    print(f"  LP result: {sc_lp:.10f} (improvement: {best_score - sc_lp:.6e})", flush=True)
    if sc_lp < best_score:
        best_score = sc_lp
        best_seq = a_lp.copy()

    # Phase 3: Subgradient polish
    print(f"\nPhase 3: Subgradient polish", flush=True)
    sub_deadline = t_start + budget_s * 0.75
    a_sub = _logit_subgradient_optimize(best_seq, sub_deadline,
                                         initial_beta=1e-5, initial_lr=0.01)
    sc_sub = _score(a_sub)
    print(f"  Subgradient result: {sc_sub:.10f}", flush=True)
    if sc_sub < best_score:
        best_score = sc_sub
        best_seq = a_sub.copy()

    # Phase 4: Peak shaving
    print(f"\nPhase 4: Peak shaving", flush=True)
    ps_deadline = t_start + budget_s * 0.85
    a_ps = _peak_shave_refined(best_seq, ps_deadline, n_peaks=15)
    sc_ps = _score(a_ps)
    print(f"  Peak shave result: {sc_ps:.10f}", flush=True)
    if sc_ps < best_score:
        best_score = sc_ps
        best_seq = a_ps.copy()

    # Phase 5: Final LP pass
    print(f"\nPhase 5: Final LP pass", flush=True)
    final_lp_deadline = t_start + budget_s * 0.95
    a_final = _lp_peel_aggressive(best_seq, final_lp_deadline)
    sc_final = _score(a_final)
    print(f"  Final LP result: {sc_final:.10f}", flush=True)
    if sc_final < best_score:
        best_score = sc_final
        best_seq = a_final.copy()

    # Phase 6: Try a different length with remaining time
    time_left = hard_deadline - time.monotonic()
    if time_left > 120:
        print(f"\nPhase 6: Try alternative length (time left: {time_left:.0f}s)", flush=True)
        alt_candidates = [c for c in candidates if c[1] != best_n_1]
        if alt_candidates:

            alt_deadline = time.monotonic() + min(100, time_left - 60)

            sc_alt = _score(a_alt)
            print(f"  Alt n={alt_n}: {sc_alt:.10f}", flush=True)
            if sc_alt < best_score:
                best_score = sc_alt
                best_seq = a_alt.copy()

    # Final: quick subgradient on the overall best
    time_left = hard_deadline - time.monotonic()
    if time_left > 15:
        a_fine = _logit_subgradient_optimize(best_seq, hard_deadline,
                                               initial_beta=1e-6, initial_lr=0.005)
        sc_fine = _score(a_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()
        print(f"  Fine polish: {sc_fine:.10f}", flush=True)

    # Ensure validity
    final_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = parent.copy()
        s = float(final_seq.sum())

    # Validate with evaluate_sequence
    if final_seq.size <= 4096:
        try:

            if np.isfinite(sc_eval):

                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
