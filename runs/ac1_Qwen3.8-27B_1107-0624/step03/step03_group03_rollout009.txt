import numpy as np

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-12:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = _fft_convolve(a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / len(a)
    return a / s


def _gradient_conv_max(a, k_target):
    """Gradient of conv(a,a)[k_target] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        if j + k_target < n:
            g[j] += a[j + k_target]
        if j - k_target >= 0:
            g[j] += a[j - k_target]
    return g


def _gradient_conv_max_fft(a, k_target):
    """Efficient gradient using FFT."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    # conv(a,a) is IfFT(|fa|^2), gradient w.r.t. a[j] involves shift
    # g[j] = a[j+k] + a[j-k] (boundary handled)
    # Use circular correlation approach
    g = np.zeros(n)
    # Forward shift
    for j in range(n - k_target):
        g[j] += a[j + k_target]
    # Backward shift
    for j in range(k_target, n):
        g[j] += a[j - k_target]
    return g


def _lse_step(a, T, lr, state, n_steps_total, step_idx):
    """One LSE-Adam step with piecewise-linear temperature schedule."""
    n = len(a)
    c = _fft_convolve(a)
    m_val = float(c.max())

    # Piecewise temperature: linear decay from T_start to T_end
    # (temperature is passed in externally)
    x = (c - m_val) / max(T, 1e-10)
    x = np.clip(x, -80, 80)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = ex / s

    # Gradient: 2 * conv(w, reverse(a)) centered
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h

    # Logit-space projection
    dot = float(np.dot(a, grad))
    gz = a * (grad - dot)
    gz -= float(np.mean(gz))

    if not np.all(np.isfinite(gz)):
        return a, state, m_val

    t = state['t'] + 1
    b1, b2 = 0.9, 0.999
    m = b1 * state['m'] + (1.0 - b1) * gz
    v = b2 * state['v'] + (1.0 - b2) * (gz * gz)
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)

    a_new = a - lr * mh / (np.sqrt(vh) + 1e-8)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()

    state['t'] = t
    state['m'] = m
    state['v'] = v

    return a_new, state, m_val

def _dynamic_lp_refine(a, deadline, max_iter=2000, initial_k=20, max_k=80):
    """LP refinement with dynamic constraint set that grows as needed."""
    a = np.asarray(a, dtype=np.float64).copy()

    a = _normalize(a)

    best_score = _score_norm(a)

    no_imp = 0

    for it in range(max_iter):

            break

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate

        # Top cur_k constraints plus their neighbors
        idx = np.argsort(c)[::-1][:cur_k]
        active = set()
        for i in idx:
            for offset in range(-3, 4):
                j = i + offset
                if 0 <= j < 2 * n - 1:
                    active.add(j)
        active = sorted(active)

        # Build LP: maximize sum(b) subject to conv(a, b)[k] <= M_lp, b >= 0
        # conv(a,b)[k] = sum_j a[j] * b[j+k] (linear in b)
        A = np.zeros((len(active), n))
        b_vec = np.full(len(active), M_lp)
        for row, k in enumerate(active):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                A[row, j] = a[j]

        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})

            inflate = min(1.01, inflate * 1.15)
            no_imp += 3
            if no_imp > 200:
                break
            continue

        g = res.x
        sum_g = float(g.sum())

        if sum_g <= 1.0 + 1e-9:
            no_imp += 1
            # Try growing constraint set
            if no_imp % 10 == 0:

                inflate = min(1.01, inflate * 1.05)
            if no_imp > 300:
                break
            continue

        g = g / sum_g

        # Line search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)

                d_pt = c_pt
                fd = fc
                c_pt = hi - gr * (hi - lo)
                fc = _score_norm(np.maximum(a + c_pt * (g - a), 0))
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + gr * (hi - lo)
                fd = _score_norm(np.maximum(a + d_pt * (g - a), 0))
        t_opt = (lo + hi) / 2.0

        a_new = np.maximum(a + t_opt * (g - a), 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc_new = _score_norm(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
            # Shrink inflate on success
            inflate = max(1.0001, inflate * 0.9)
        else:
            no_imp += 1
            if no_imp % 20 == 0:
                cur_k = min(max_k, cur_k + 3)
                inflate = min(1.01, inflate * 1.08)
            if no_imp > 400:
                break

        a = a_new

    return best_a


def _freq_perturb(a, rng, sigma=0.1):
    """Perturb in frequency domain then transform back, clip and normalize."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    # Perturb magnitude
    mag = np.abs(fa)
    phase = np.angle(fa)
    mag_new = mag * (1.0 + sigma * rng.normal(size=mag.shape))
    mag_new = np.maximum(mag_new, 0)
    fa_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(fa_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _subgradient_polish_v2(a, deadline, max_iter=50000, lr_start=0.04, lr_end=0.002):
    """Subgradient polish with geometric LR decay and momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = _normalize(a)

    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)
    momentum = 0.7
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Geometric LR decay
        frac = min(1.0, it / max(max_iter, 1))
        lr = lr_start * (lr_end / lr_start) ** frac

        c = _fft_convolve(a)
        m_val = float(c.max())
        # Uniform over near-max lags
        thresh = m_val * 0.999
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Gradient is average of gradients at active lags
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                if j + k < n:
                    grad[j] += a[j + k]
                if j - k >= 0:
                    grad[j] += a[j - k]
        grad /= len(active)
        grad *= 2.0  # factor of 2 from quadratic

        # Project onto sum=1 hyperplane
        dot = float(np.dot(a, grad))
        grad_proj = grad - dot * a / a  # actually just subtract mean weighted
        grad_proj = grad - np.mean(grad)

        v = momentum * v - lr * grad_proj
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new

        sc_new = _score_norm(a_new)
        if sc_new < best_score - 1e-13:
            best_score = sc_new
            best_a = a_new.copy()
            no_imp = 0
            a = a_new
        else:
            no_imp += 1
            if no_imp > max(5000, 8 * n):
                a = best_a.copy()
                v = np.zeros(n)
                no_imp = 0
            # Decay momentum when stuck
            momentum = max(0.3, momentum * 0.999)

    return best_a


def _make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - 0.5) / (sigma_frac * 0.5)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_beta(n, alpha=2.5):
    """Beta distribution profile."""
    t = np.linspace(0.01, 0.99, n)
    a = t ** (alpha - 1) * (1 - t) ** (alpha - 1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_sinc_like(n):
    """Sinc-like profile (Barker-like)."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    t = (j - center) / (n * 0.4)
    with np.errstate(divide='ignore', invalid='ignore'):
        a = np.sinc(t)
    a = np.where(np.isnan(a), 0, a)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:
            return p0 / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s

        print(f"  LP start n={n_t}: norm_score={sc_in:.10f}", flush=True)

        lp_deadline = min(p2_deadline, time.monotonic() + 100.0)
        a_lp = _dynamic_lp_refine(a_in, lp_deadline, max_iter=1500,
                                   initial_k=15, max_k=60)
        sc_lp = _score_norm(a_lp)
        full_sc_lp = _score(a_lp)

        if np.isfinite(full_sc_lp) and full_sc_lp < best_score:
            best_score = full_sc_lp
            best_seq = a_lp.copy()

        best_per_length[n_t] = a_lp
        print(f"  LP result n={n_t}: norm_score={sc_lp:.10f} "
              f"(full={full_sc_lp:.10f})", flush=True)

    # === Phase 3: Multi-start exploration on best ===
    print("\n=== Phase 3: Multi-start exploration ===", flush=True)
    p3_deadline = t_start + (budget_s - 30) * 0.82

    if best_seq is not None and time.monotonic() < p3_deadline:
        a_exp = _multi_start_explore(best_seq, len(best_seq), p3_deadline, rng,
                                      n_starts=4)
        sc_exp = _score(a_exp)
        if np.isfinite(sc_exp) and sc_exp < best_score:
            best_score = sc_exp
            best_seq = a_exp.copy()
            print(f"  Multi-start improved: {sc_exp:.10f}", flush=True)
        else:
            print(f"  Multi-start: no improvement (best still {best_score:.10f})", flush=True)

    # === Phase 4: Subgradient polish ===
    print("\n=== Phase 4: Subgradient polish ===", flush=True)
    p4_deadline = t_start + (budget_s - 30) * 0.92

    if best_seq is not None and time.monotonic() < p4_deadline:
        a_sp = _subgradient_polish_v2(best_seq, p4_deadline, max_iter=60000,
                                       lr_start=0.03, lr_end=0.001)
        sc_sp = _score(a_sp)
        if np.isfinite(sc_sp) and sc_sp < best_score:
            best_score = sc_sp
            best_seq = a_sp.copy()
        print(f"  Subgradient: {sc_sp:.10f}", flush=True)

    # === Phase 5: Final fine LSE ===
    print("\n=== Phase 5: Final fine LSE ===", flush=True)
    if best_seq is not None and time.monotonic() < deadline:
        a_final = _lse_optimize_v2(best_seq, deadline, max_iter=50000,
                                    T_start=0.01, T_end=1e-7, lr=0.04,
                                    restart_interval=10000)
        sc_final = _score(a_final)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
        print(f"  Final LSE: {sc_final:.10f}", flush=True)

    # === Phase 6: One more LP pass on best ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_lp2 = _dynamic_lp_refine(best_seq, deadline - 5, max_iter=500,
                                    initial_k=10, max_k=40)
        sc_lp2 = _score(a_lp2)
        if np.isfinite(sc_lp2) and sc_lp2 < best_score:
            best_score = sc_lp2
            best_seq = a_lp2.copy()
        print(f"  Final LP: {sc_lp2:.10f}", flush=True)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(256) / 256.0
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256.0

    # Evaluate with authoritative function
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"\nEvaluate_sequence score: {sc_eval:.10f}", flush=True)
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Evaluation error: {e}", flush=True)

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}", flush=True)
    return [float(x) for x in final_seq]
