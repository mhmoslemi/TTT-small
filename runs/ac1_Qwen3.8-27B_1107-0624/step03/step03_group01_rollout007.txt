def solve_lp_direction(a, M):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # Build the constraint matrix: (a * b)[k] for k = 0, ..., 2n-2
    # (a*b)[k] = sum_i a[i] * b[k-i] where b is zero outside [0, n-1]
    # We can express this as A @ b <= M * ones(2n-1)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0

    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve()
    return b.value

LP-guided direction finding + FISTA polishing for minimizing autocorrelation peak.
Uses the LP relaxation to find optimal descent directions, combined with
log-sum-exp smoothed FISTA for fine polishing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

# Helper functions (top-level)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    N = max(2, N)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score_normalized(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())

def score_full(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return e / s

    cols = []

"""
LP-guided direction finding + FISTA polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP relaxation to find optimal descent directions (Weisfeiler-Lehman style),
combined with Nesterov-accelerated gradient (FISTA) on log-sum-exp smoothed surrogate,
and multi-phase search with diverse initializations and multi-length exploration.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix, csc_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def score_normalized(a):
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())


def score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def build_conv_matrix(a):
    n = len(a)
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(2 * n - 1), n)
    # Column j: a[0..n-1] in rows j..j+n-1
    col_idx = []
    row_idx = []
    vals = []
    for j in range(n):
        row_idx.extend(range(j, j + n))
        col_idx.extend([j] * n)
        vals.extend(a.tolist())
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(a, M):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    A = build_conv_matrix(a)
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    solvers_to_try = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.GLPK, cp.CBC]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, verbose=False)
            if b.value is not None and np.all(np.isfinite(b.value)):
                return np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
        except Exception:
            continue
    return None


def lp_direction_step(a, deadline, max_iters=100):
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = full_convolve(a, a)
        M = float(c.max())

        b = solve_lp_direction(a, M)
        if b is None:
            break
        b_sum = float(b.sum())
        if b_sum < 1e-10:
            break

        b = b / b_sum

        # Line search
        best_t = 0.0
        best_sc = best_score
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:

        a = np.maximum(a, 1e-12)
    elif ptype == 'cosine':
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-12)
    elif ptype == 'bimodal':
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 2.0 * np.exp(-((t - 0.25) / 0.1) ** 2) + 2.0 * np.exp(-((t - 0.75) / 0.1) ** 2)
        a = np.maximum(a, 1e-12)
    elif ptype == 'flat_edge':
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 3.0 * (t * (1.0 - t)) ** 0.5
        a = np.maximum(a, 1e-12)
    elif ptype == 'inverse_edge':
        t = np.linspace(0.0, 1.0, n)
        a = 1.0 + 2.0 / (1.0 + 10.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
    else:
        a = np.ones(n)
    return a / a.sum()


def interp_to_length(a, n_new):
    n = len(a)
    if n == n_new:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    LP-guided direction finding + FISTA polishing for minimizing autocorrelation peak.
    Multi-phase: LP coarse search -> FISTA polish -> LP refine -> multi-length -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        print(f"Parent: n={n}, score={score_full(parent):.10f}")
    else:
        n = 1024
        parent = make_profile(n, 'linear', 1.5)
        print(f"No parent, using linear profile n={n}")

    # === Phase 0: Quick diverse initialization (30s) ===
    phase0_end = min(deadline, t_start + 25.0)

    if parent is not None:
        sc = score_normalized(parent)
        update_best(parent, sc, "parent")

    for ptype, params in [('linear', 0.5), ('linear', 1.0), ('linear', 2.0),
                          ('quadratic', [1.0, 0.5]), ('quadratic', [2.0, -0.3]),
                          ('cosine', None), ('bimodal', None), ('flat_edge', None)]:
        if time.monotonic() > phase0_end:
            break
        p = make_profile(n, ptype, params)
        sc = score_normalized(p)
        update_best(p, sc, f"{ptype}_n{n}")

    print(f"Phase 0 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 1: LP direction finding (coarse optimization, up to 500s) ===
    phase1_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase1_end and best_seq is not None:
        print("Phase 1: LP direction finding...")
        p_best, sc_best = lp_direction_step(best_seq, phase1_end, max_iters=150)
        update_best(p_best, sc_best, "lp_phase1")

    # === Phase 2: FISTA polishing (up to 850s) ===
    phase2_end = min(deadline, t_start + 850.0)
    if time.monotonic() < phase2_end and best_seq is not None:
        print(f"Phase 2: FISTA polishing... (starting score={best_score:.10f})")
        per_dl = min(phase2_end - 10, time.monotonic() + 350.0)
        p_best, sc_best = fista_polish(best_seq, per_dl, beta_init_frac=0.01, lr_init=0.3, max_iters=100000)
        update_best(p_best, sc_best, "fista_phase2")

    # === Phase 3: LP refinement on FISTA result (up to 1150s) ===
    phase3_end = min(deadline, t_start + 1150.0)
    if time.monotonic() < phase3_end and best_seq is not None:
        print(f"Phase 3: LP refinement... (starting score={best_score:.10f})")
        p_best, sc_best = lp_direction_step(best_seq, phase3_end, max_iters=80)
        update_best(p_best, sc_best, "lp_phase3")

        if time.monotonic() < phase3_end - 20:
            per_dl = min(phase3_end - 10, time.monotonic() + 100.0)
            p_b2, sc_b2 = fista_polish(best_seq, per_dl, beta_init_frac=0.005, lr_init=0.2, max_iters=50000)
            update_best(p_b2, sc_b2, "fista_after_lp3")

    # === Phase 4: Adam polishing with very small beta (up to 1400s) ===
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end and best_seq is not None:
        print(f"Phase 4: Adam fine polish... (starting score={best_score:.10f})")
        per_dl = min(phase4_end - 10, time.monotonic() + 200.0)
        p_best, sc_best = adam_polish(best_seq, per_dl, lr=0.01, beta_frac=0.002, max_iters=30000)
        update_best(p_best, sc_best, "adam_phase4")

    # === Phase 5: Multi-length exploration (up to 1650s) ===
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end and best_seq is not None:
        print(f"Phase 5: Multi-length exploration... (best n={len(best_seq)})")
        n_best = len(best_seq)

        length_candidates = []
        for factor in [0.5, 2.0, 0.75, 1.25, 1.5]:
            n_try = int(n_best * factor)
            n_try = max(128, min(8192, n_try))
            if n_try != n_best and n_try not in [x for x in length_candidates]:
                length_candidates.append(n_try)

        for n_try in length_candidates[:4]:
            if time.monotonic() > phase5_end - 20:
                break

            p_interp = interp_to_length(best_seq, n_try)
            sc_interp = score_normalized(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{n_try}")

            # Quick FISTA on new length
            if time.monotonic() < phase5_end - 30:
                per_dl = min(phase5_end - 20, time.monotonic() + 50.0)
                p_b, sc_b = fista_polish(p_interp, per_dl, beta_init_frac=0.01, lr_init=0.3, max_iters=20000)
                update_best(p_b, sc_b, f"fista_n{n_try}")

    # === Phase 6: Final intensive polish (remaining time) ===
    phase6_end = deadline
    if time.monotonic() < phase6_end and best_seq is not None:
        print(f"Phase 6: Final polish... (score={best_score:.10f}, remaining={phase6_end-time.monotonic():.1f}s)")
        remaining = phase6_end - time.monotonic()

        # LP
        if remaining > 80:
            p_b, sc_b = lp_direction_step(best_seq, min(phase6_end - 40, time.monotonic() + 150.0), max_iters=30)
            update_best(p_b, sc_b, "lp_final")

        # FISTA with tiny beta
        if time.monotonic() < phase6_end - 30:
            per_dl = min(phase6_end - 20, time.monotonic() + 120.0)
            p_b2, sc_b2 = fista_polish(best_seq, per_dl, beta_init_frac=0.001, lr_init=0.1, max_iters=50000)
            update_best(p_b2, sc_b2, "fista_tiny_beta")

        # One more LP
        if time.monotonic() < phase6_end - 15:
            p_b3, sc_b3 = lp_direction_step(best_seq, phase6_end - 5, max_iters=15)
            update_best(p_b3, sc_b3, "lp_abs_final")

        # Final Adam
        if time.monotonic() < phase6_end - 5:
            per_dl = min(phase6_end - 3, time.monotonic() + 40.0)
            p_b4, sc_b4 = adam_polish(best_seq, per_dl, lr=0.005, beta_frac=0.001, max_iters=10000)
            update_best(p_b4, sc_b4, "adam_abs_final")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = make_profile(256, 'linear', 1.5)

        sc_final = score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, total_time={time.monotonic()-t_start:.1f}s")

        # Validate with evaluator if short enough
        if len(final_seq) <= 8192:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = make_profile(256, 'linear', 1.5)
    print(f"FALLBACK: n=256, score={score_full(fallback):.10f}")
    return [float(x) for x in fallback]
