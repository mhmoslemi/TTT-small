    return best_a


def _multiplicative_weights(a, deadline, max_iter=20000, eta=0.1):
    """Multiplicative weights (no-regret) approach for the game."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    eta_cur = eta
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        M = float(c.max())

        # Solve LP to get best response g
        g = _lp_solve(a, M * 1.0001)
        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g < 1e-10:
            break
        g = g / sum_g

        # MW update: a_j <- a_j * exp(eta * (g_j - a_j)) approximately
        # Actually, in the game interpretation, the "payoff" for choosing a_j
        # is related to how well it matches g. Use: a_j <- a_j * (g_j/a_j)^eta
        ratio = g / (a + 1e-30)
        a_new = a * np.exp(eta_cur * np.log(np.maximum(ratio, 1e-30)))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                eta_cur = max(0.01, eta_cur * 0.9)
            if no_imp > 3000:
                break

        a = a_new

    return best_a


def _fourier_perturb(a, rng, strength=0.1):
    """Add a Fourier-based perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    # Add a random sinusoidal pattern
    freq = rng.integers(1, max(2, n // 4))
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.5, 1.5)
    t = np.arange(n)
    perturbation = strength * amp * np.sin(2 * np.pi * freq * t / n + phase)

    a_new = a + perturbation
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def _random_perturb(a, rng, strength=0.1):
    """Add random noise perturbation."""
    n = len(a)
    a_new = a + strength * rng.standard_normal(n)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def _block_perturb(a, rng, block_size=None):
    """Random block permutation."""
    n = len(a)
    if block_size is None:
        block_size = max(2, min(n // 8, 32))
    a_new = a.copy()
    i = rng.integers(0, n - block_size)
    j = rng.integers(0, n - block_size)
    if abs(i - j) < block_size:
        return a
    a_new[i:i + block_size], a_new[j:j + block_size] = \
        a_new[j:j + block_size].copy(), a_new[i:i + block_size].copy()
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def _interpolate_to_length(a, n_new):
    """Interpolate sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        s = a.sum()
        if s > 1e-30:
            return a / s
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_initial(n, type='linear', rng=None):
    """Create various initial profiles."""
    if rng is None:
        rng = np.random.default_rng(42)

    if type == 'uniform':
        return np.ones(n) / n
    elif type == 'linear':
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * (t - 0.5)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif type == 'u_shape':
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (2.0 * t - 1.0) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif type == 'random':
        a = np.maximum(rng.standard_normal(n), 0.0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    elif type == 'bimodal':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.25) / 0.15) ** 2) + np.exp(-0.5 * ((t - 0.75) / 0.15) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif type == 'peaked_center':
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - 0.5) / 0.1) ** 2) + 0.1
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _full_optimize(a_init, n, deadline, verbose=True):
    """Full optimization pipeline for a given initial sequence."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    if verbose:
        print(f"  Initial: score={best_score:.10f}, n={n}")

    # Phase 1: Aggressive LP
    t1 = time.monotonic()
    rem = deadline - t1
    lp_deadline = t1 + min(rem * 0.45, 600.0)
    a_lp = _lp_refine_tight(a, lp_deadline, max_iter=50000, inflate=1.00005, blend=0.08)
    sc_lp = _score(a_lp)
    if sc_lp < best_score:
        best_score = sc_lp
        best_a = a_lp.copy()
    if verbose:
        print(f"  After LP: score={best_score:.10f} ({time.monotonic()-t1:.1f}s)")

    # Phase 2: Subgradient polish
    t2 = time.monotonic()
    rem = deadline - t2
    sg_deadline = t2 + min(rem * 0.2, 200.0)
    a_sg = _subgradient_polish(best_a, sg_deadline, max_iter=20000, beta_init=0.005)
    sc_sg = _score(a_sg)
    if sc_sg < best_score:
        best_score = sc_sg
        best_a = a_sg.copy()
    if verbose:
        print(f"  After SG: score={best_score:.10f} ({time.monotonic()-t2:.1f}s)")

    # Phase 3: MW approach
    t3 = time.monotonic()
    rem = deadline - t3
    mw_deadline = t3 + min(rem * 0.15, 150.0)
    a_mw = _multiplicative_weights(best_a, mw_deadline, max_iter=10000, eta=0.05)
    sc_mw = _score(a_mw)
    if sc_mw < best_score:
        best_score = sc_mw
        best_a = a_mw.copy()
    if verbose:
        print(f"  After MW: score={best_score:.10f} ({time.monotonic()-t3:.1f}s)")

    # Phase 4: Final tight LP
    t4 = time.monotonic()
    rem = deadline - t4
    lp2_deadline = t4 + min(rem * 0.2, 300.0)
    a_lp2 = _lp_refine_tight(best_a, lp2_deadline, max_iter=30000, inflate=1.00002, blend=0.04)
    sc_lp2 = _score(a_lp2)
    if sc_lp2 < best_score:
        best_score = sc_lp2
        best_a = a_lp2.copy()
    if verbose:
        print(f"  After LP2: score={best_score:.10f} ({time.monotonic()-t4:.1f}s)")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining tight LP best-response, subgradient
    polish, multiplicative weights, Fourier perturbations, and multi-length search.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**32 - 1))

    best_seq = None
    best_score = np.inf

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}")
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for ln in [512, 640, 768, 1024, 1280, 1536]:
        if ln not in lengths:
            lengths.append(ln)
    # Sort: parent length first, then by size
    lengths = sorted(set(lengths), key=lambda x: 0 if x == (parent.size if parent is not None else -1) else 1)

    print(f"Lengths to try: {lengths}")
    print(f"Total budget: {budget_s}s, effective deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: Intensive optimization at parent/best length
    if parent is not None and time.monotonic() < deadline - 300:
        n_parent = parent.size
        print(f"\n=== Phase 1: Intensive optimization at n={n_parent} ===")
        phase1_deadline = min(deadline - 400.0, t_start + budget_s * 0.4)
        a_opt, sc_opt = _full_optimize(parent, n_parent, phase1_deadline, verbose=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  ** New best: {sc_opt:.10f}")

    # Phase 2: Try different lengths
    if time.monotonic() < deadline - 200:
        print(f"\n=== Phase 2: Multi-length search ===")
        tried = [parent.size] if parent is not None else []
        remaining_lengths = [ln for ln in lengths if ln not in tried]

        for n_t in remaining_lengths:
            if time.monotonic() > deadline - 150:
                break

            # Allocate time based on remaining budget and number of lengths left
            n_remaining = sum(1 for ln in lengths if ln > n_t and ln not in tried)
            per_length_time = min(200.0, (deadline - time.monotonic() - 100.0) / max(1, n_remaining + 1))
            if per_length_time < 30:
                continue

            print(f"\n  Trying n={n_t} (budget={per_length_time:.0f}s)")
            t_phase = time.monotonic()
            phase_deadline = t_phase + per_length_time

            # Initialize from best known (interpolated) or from scratch
            if best_seq is not None:
                a_init = _interpolate_to_length(best_seq, n_t)
            else:
                a_init = _make_initial(n_t, 'linear', rng)

            a_opt, sc_opt = _full_optimize(a_init, n_t, phase_deadline, verbose=False)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  ** New best at n={n_t}: {sc_opt:.10f}")
            else:
                print(f"  n={n_t}: score={sc_opt:.10f} (not better than {best_score:.10f})")

    # Phase 3: Escape from local minima via perturbations at best length
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = best_seq.size
        print(f"\n=== Phase 3: Perturbation escape at n={n_best} ===")

        n_perturbations = 6
        pert_time = min(120.0, (deadline - time.monotonic() - 60.0) / n_perturbations)

        for p_idx in range(n_perturbations):
            if time.monotonic() > deadline - 60:
                break

            # Different perturbation strategies
            if p_idx == 0:
                a_pert = _fourier_perturb(best_seq, rng, strength=0.15)
            elif p_idx == 1:
                a_pert = _fourier_perturb(best_seq, rng, strength=0.25)
            elif p_idx == 2:
                a_pert = _random_perturb(best_seq, rng, strength=0.1)
            elif p_idx == 3:
                a_pert = _random_perturb(best_seq, rng, strength=0.2)
            elif p_idx == 4:
                a_pert = _block_perturb(best_seq, rng, block_size=max(4, n_best // 16))
            else:
                # Mix: try a completely different initialization
                a_pert = _make_initial(n_best, 'bimodal', rng)

            sc_pert = _score(a_pert)
            print(f"  Pert {p_idx}: init_score={sc_pert:.10f}")

            if sc_pert < best_score:
                best_score = sc_pert
                best_seq = a_pert.copy()

            # Quick optimization from perturbed point
            pert_deadline = time.monotonic() + min(pert_time, deadline - 50.0)
            a_opt, sc_opt = _full_optimize(a_pert, n_best, pert_deadline, verbose=False)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"  ** New best after pert {p_idx}: {sc_opt:.10f}")

    # Phase 4: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        print(f"\n=== Phase 4: Final polish at n={n_best} ===")
        final_deadline = deadline - 10.0
        final_time = final_deadline - time.monotonic()
        print(f"  Final budget: {final_time:.0f}s")

        # Very tight LP
        a_lp = _lp_refine_tight(best_seq, final_deadline, max_iter=50000,
                                inflate=1.00001, blend=0.03)
        sc_lp = _score(a_lp)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: {sc_lp:.10f}")

        # Brief subgradient
        if time.monotonic() < final_deadline - 10:
            a_sg = _subgradient_polish(best_seq, final_deadline, max_iter=10000, beta_init=0.002)
            sc_sg = _score(a_sg)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"  Final SG: {sc_sg:.10f}")

    # Final validation
    if best_seq is None:
        best_seq = _make_initial(768, 'linear', rng)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        pass  # Keep as-is (scale-invariant objective)
    else:
        final_seq = _make_initial(768, 'linear', rng)

    # Validate with evaluate_sequence
    try:
        sc_eval = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(sc_eval):
            print(f"evaluate_sequence: {sc_eval:.10f}")
            if sc_eval < best_score:
                best_score = sc_eval
    except Exception as e:
        print(f"Validation error: {e}")

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
