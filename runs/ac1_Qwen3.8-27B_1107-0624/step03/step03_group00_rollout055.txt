"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) Nesterov-accelerated gradient in logit space with adaptive smooth-max,
(2) LP-based refinement with alternating solves,
(3) multi-scale escalation from smaller to larger n,
(4) diverse asymmetric initializations exploiting the a*reverse(a) structure,
(5) curriculum on smooth-max temperature, and (6) structured noise restarts.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _smooth_max_and_weights(c, beta):
    c = np.asarray(c, dtype=np.float64).ravel()
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def _cross_corr(w, p):
    n = len(p)
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def _softmax_from_z(z, floor=1e-15):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + floor
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def _score_norm(p):
    n = len(p)
    c = _fft_convolve(p)
    return float(2.0 * n * c.max())


def _score_and_conv(p):
    n = len(p)
    c = _fft_convolve(p)
    return float(2.0 * n * c.max()), c


def _lp_solve(a, M):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, t_init=0.35, max_solves=50):
    n = len(a)
    current_best = a.copy()
    current_score, c = _score_and_conv(a)
    M = float(c.max())
    n_solves = 0
    t = t_init
    no_improve = 0
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        a_new = (1.0 - t) * a + t * b_norm
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc_new, c_new = _score_and_conv(a_new)
        if sc_new < current_score - 1e-12:
            current_score = sc_new
            current_best = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break
        a = a_new
        M = float(c_new.max())
        t = max(0.08, t * 0.93)
    return current_best, current_score


def _nesterov_optimize(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.93,
                       no_imp_limit=500, momentum=0.9):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    p = p0.copy()
    p_ys = p.copy()
    c = _fft_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    t_count = 0
    v = np.zeros(n)
    base_lr = lr

    while time.monotonic() < deadline:
        t_count += 1
        y = 2 * p - p_ys
        c_y = _fft_convolve(y)
        _, w = _smooth_max_and_weights(c_y, beta)
        g_y = _cross_corr(w, y)
        grad = g_y

        dot = float(np.dot(y, grad))
        gz = y * (grad - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            p = best_p.copy()
            z = best_z.copy()
            p_ys = p.copy()
            v.fill(0.0)
            continue

        v = momentum * v - lr * gz
        p_ys = p.copy()
        p = p + v
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

        z = np.log(np.maximum(p, 1e-15))
        z -= float(np.mean(z))

        c = _fft_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.002, lr * 0.9)
        else:
            p = best_p.copy()
            z = best_z.copy()
            p_ys = p.copy()
            v.fill(0.0)
            continue

        if t_count % 30 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-5, beta * 0.5)

        if no_imp > no_imp_limit:
            ns = 0.05 + 0.05 * min(no_imp / 1000.0, 1.0)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            p_ys = p.copy()
            c = _fft_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            v.fill(0.0)
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _adam_optimize(p0, deadline, rng, lr=0.06, beta_frac=0.1, beta_decay=0.92,
                   no_imp_limit=400):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _fft_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = _smooth_max_and_weights(c, beta)
        g = _cross_corr(w, p)
        dot = float(np.dot(p, g))
        gz = p * (g - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -60, 60)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _fft_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.003, lr * 0.92)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        if t_adam % 50 == 0:
            beta = max(1e-5, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.1:
                    beta = max(1e-5, beta * 0.5)

        if no_imp > no_imp_limit:
            ns = 0.06 + 0.06 * min(no_imp / 2000.0, 1.0)
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _fft_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _make_profile_tilted(n, alpha, gamma):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(alpha * t) * (1.0 + gamma * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s1, s2, h1, h2):
    t = np.linspace(0.0, 1.0, n)
    a = h1 * np.exp(-0.5 * ((t - c1) / s1) ** 2) + h2 * np.exp(-0.5 * ((t - c2) / s2) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp, phase=0.0):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_ramped_gauss(n, center, width, ramp_slope):
    t = np.linspace(0.0, 1.0, n)
    gauss = np.exp(-0.5 * ((t - center) / width) ** 2)
    ramp = 1.0 + ramp_slope * t
    a = gauss * ramp
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == 0:
        return None
    if n0 == n_new:
        p0 = np.maximum(p0, 1e-15)
        s = float(p0.sum())
        if s <= 0:
            return None
        return p0 / s
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    s = float(p.sum())
    if s <= 0:
        return None
    return p / s


def _asymmetry_boost(a, strength):
    n = len(a)
    t = np.linspace(0.0, 1.0, n)
    ramp = 1.0 + strength * (2.0 * t - 1.0)
    a_new = a * ramp
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s, deadline in {deadline-t_start:.0f}s")

    # === Phase 1: Generate diverse candidates ===
    phase1_end = t_start + 90.0
    candidates = []

    n_start = 512
    for n in [n_start, 1024]:
        profs = [
            ('tilt2', _make_profile_tilted(n, 2.0, 1.0)),
            ('tilt3', _make_profile_tilted(n, 3.0, 1.5)),
            ('tilt4', _make_profile_tilted(n, 4.0, 2.0)),
            ('lin2', _make_profile_linear(n, 2.0)),
            ('lin4', _make_profile_linear(n, 4.0)),
            ('lin6', _make_profile_linear(n, 6.0)),
            ('quad1', _make_profile_quadratic(n, 1.5, 0.5)),
            ('quad2', _make_profile_quadratic(n, 2.5, -0.5)),
            ('cubic1', _make_profile_cubic(n, 2.0, -1.0)),
            ('cubic2', _make_profile_cubic(n, 3.0, -2.0)),
            ('tb1', _make_profile_two_bump(n, 0.3, 0.7, 0.15, 0.15, 1.0, 1.0)),
            ('tb2', _make_profile_two_bump(n, 0.2, 0.8, 0.2, 0.1, 1.0, 1.5)),
            ('tb3', _make_profile_two_bump(n, 0.25, 0.75, 0.12, 0.2, 1.5, 1.0)),
            ('cos1', _make_profile_cosine(n, 1, 0.7)),
            ('cos2', _make_profile_cosine(n, 2, 0.5)),
            ('cos3', _make_profile_cosine(n, 3, 0.4)),
            ('rg1', _make_profile_ramped_gauss(n, 0.6, 0.2, 3.0)),
            ('rg2', _make_profile_ramped_gauss(n, 0.5, 0.3, 4.0)),
            ('rg3', _make_profile_ramped_gauss(n, 0.4, 0.25, 5.0)),
        ]
        for label, p in profs:
            sc = _score_norm(p)
            candidates.append((sc, p, n, label))

    if parent is not None:
        for n in [512, 1024, 2048]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, 'parent'))
            else:
                sc = _score_norm(parent)
                candidates.append((sc, parent.copy(), n, 'parent'))
        # Asymmetric versions of parent
        for n in [512, 1024]:
            pi = _interp_profile(parent, n)
            if pi is not None:
                for amp in [1.0, 2.0, 3.0]:
                    pa = _asymmetry_boost(pi, amp)
                    sc = _score_norm(pa)
                    candidates.append((sc, pa, n, f'parent_asym{amp}'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f} [{candidates[0][3]}]")
    _update(candidates[0][1], candidates[0][0], 'init')

    # === Phase 2: Nesterov optimization on top candidates at n=512 ===
    phase2_end = t_start + 400.0
    polished_512 = []
    count = 0
    for sc, p, n, label in candidates:
        if time.monotonic() > phase2_end or count >= 12:
            break
        if n != 512:
            continue
        per_end = min(phase2_end, time.monotonic() + 25.0)
        p_b, sc_b = _nesterov_optimize(p, per_end, rng, lr=0.05, beta_frac=0.1,
                                        beta_decay=0.94, no_imp_limit=300)
        _update(p_b, sc_b, f'n2_{label}')
        polished_512.append((sc_b, p_b, n, label))
        count += 1

    # Also try Adam on top few
    polished_512.sort(key=lambda x: x[0])
    for sc, p, n, label in polished_512[:4]:
        if time.monotonic() > phase2_end:
            break
        per_end = min(phase2_end, time.monotonic() + 20.0)
        p_b, sc_b = _adam_optimize(p, per_end, rng, lr=0.06, beta_frac=0.08,
                                    beta_decay=0.93, no_imp_limit=200)
        _update(p_b, sc_b, f'n2a_{label}')

    print(f"Phase 2 done: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 3: LP refinement at n=512 ===
    phase3_end = t_start + 550.0
    best_512 = None
    best_512_sc = np.inf
    for sc, p, n, label in polished_512:
        if sc < best_512_sc:
            best_512_sc = sc
            best_512 = p
    if best_512 is not None and time.monotonic() < phase3_end:
        p_lp, sc_lp = _lp_refine(best_512, min(phase3_end, time.monotonic() + 80),
                                  t_init=0.3, max_solves=30)
        _update(p_lp, sc_lp, 'LP_512')
        best_512 = p_lp
        best_512_sc = sc_lp

    # === Phase 4: Scale up to n=1024 ===
    phase4_end = t_start + 1100.0
    if best_512 is not None:
        p_1024 = _interp_profile(best_512, 1024)
        if p_1024 is not None:
            sc_1024 = _score_norm(p_1024)
            print(f"  Interp to 1024: {sc_1024:.10f}")
            _update(p_1024, sc_1024, 'interp_1024')

        # Nesterov on 1024
        per = min(200.0, phase4_end - time.monotonic() - 30)
        if per > 20:
            p_b, sc_b = _nesterov_optimize(p_1024, time.monotonic() + per, rng,
                                            lr=0.04, beta_frac=0.08, beta_decay=0.94,
                                            no_imp_limit=500)
            _update(p_b, sc_b, 'nest_1024')
            p_1024 = p_b

        # LP on 1024
        if time.monotonic() < phase4_end - 30:
            p_lp, sc_lp = _lp_refine(p_1024, min(phase4_end - 10, time.monotonic() + 120),
                                      t_init=0.25, max_solves=40)
            _update(p_lp, sc_lp, 'LP_1024')
            p_1024 = p_lp

        # Adam polish on 1024
        if time.monotonic() < phase4_end - 15:
            per = min(60.0, phase4_end - time.monotonic() - 5)
            if per > 10:
                p_b, sc_b = _adam_optimize(p_1024, time.monotonic() + per, rng,
                                            lr=0.03, beta_frac=0.06, beta_decay=0.95,
                                            no_imp_limit=300)
                _update(p_b, sc_b, 'adam_1024')
                p_1024 = p_b

    # Also try direct optimization from best 512 candidate at 1024 via different inits
    if time.monotonic() < phase4_end - 60:
        for sc, p, n, label in candidates[:6]:
            if n == 512:
                p_int = _interp_profile(p, 1024)
                if p_int is not None:
                    per = min(30.0, phase4_end - time.monotonic() - 20)
                    if per < 10:
                        break
                    p_b, sc_b = _nesterov_optimize(p_int, time.monotonic() + per, rng,
                                                    lr=0.04, beta_frac=0.08,
                                                    beta_decay=0.94, no_imp_limit=200)
                    _update(p_b, sc_b, f'cross_{label}_1024')

    print(f"Phase 4 done: best={best_score:.10f} n={best_seq.size if best_seq is not None else 0}")

    # === Phase 5: Scale up to n=2048 ===
    phase5_end = t_start + 1500.0
    if best_seq is not None and best_seq.size == 1024 and time.monotonic() < phase5_end:
        p_2048 = _interp_profile(best_seq, 2048)
        if p_2048 is not None:
            sc_2048 = _score_norm(p_2048)
            print(f"  Interp to 2048: {sc_2048:.10f}")
            _update(p_2048, sc_2048, 'interp_2048')

            per = min(150.0, phase5_end - time.monotonic() - 30)
            if per > 20:
                p_b, sc_b = _nesterov_optimize(p_2048, time.monotonic() + per, rng,
                                                lr=0.03, beta_frac=0.07, beta_decay=0.95,
                                                no_imp_limit=400)
                _update(p_b, sc_b, 'nest_2048')
                p_2048 = p_b

            if time.monotonic() < phase5_end - 20:
                p_lp, sc_lp = _lp_refine(p_2048, min(phase5_end - 5, time.monotonic() + 80),
                                          t_init=0.2, max_solves=20)
                _update(p_lp, sc_lp, 'LP_2048')

    # === Phase 6: Multi-restart at best length ===
    phase6_end = deadline - 60.0
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 3
        per_r = max(15.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 10:
                break
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.08 + 0.08 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _nesterov_optimize(p_s, time.monotonic() + per, rng,
                                            lr=0.04, beta_frac=0.07, beta_decay=0.94,
                                            no_imp_limit=int(0.4 * n_best))
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

    # === Phase 7: Final polish ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 10:
            p_f, sc_f = _nesterov_optimize(best_seq, time.monotonic() + min(per, 60), rng,
                                            lr=0.015, beta_frac=0.04, beta_decay=0.96,
                                            no_imp_limit=300)
            _update(p_f, sc_f, 'final_nest')
            if time.monotonic() < deadline - 5:
                per2 = deadline - time.monotonic() - 5
                if per2 > 10:
                    p_f2, sc_f2 = _adam_optimize(p_f, time.monotonic() + min(per2, 40), rng,
                                                  lr=0.01, beta_frac=0.03, beta_decay=0.97,
                                                  no_imp_limit=200)
                    _update(p_f2, sc_f2, 'final_adam')

    # === Final validation and return ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(512, 2.0)

        exact_sc, _ = _score_and_conv(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size} t={time.monotonic()-t_start:.0f}s")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
            else:
                print(f"EVAL returned inf, using fallback")
                fb = _make_profile_linear(512, 2.0)
                return [float(x) for x in fb]
        except Exception:
            pass

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(512, 2.0)
    print(f"FALLBACK n=512, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
