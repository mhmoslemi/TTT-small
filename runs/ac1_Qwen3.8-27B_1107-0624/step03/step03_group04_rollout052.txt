import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def _smooth_max_topk(c, beta, topk=50):
    """Compute smooth max focusing on top-k values for aggressive peak reduction."""
    c = np.asarray(c, dtype=np.float64)
    m = len(c)
    if topk >= m:
        topk = m
    # Get indices of top-k
    if m > topk:
        part = np.argpartition(c, -topk)[-topk:]
    else:
        part = np.arange(m)
    c_top = c[part]
    mval = float(c_top.max())
    x = (c_top - mval) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        return mval, np.zeros(m)
    w_top = ex / s
    w = np.zeros(m)
    w[part] = w_top
    val = mval + beta * float(np.log(s))
    return val, w


def _smooth_max_full(c, beta):

def _conv_grad_smooth(a, w, n):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a, given weights w."""
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    return 2.0 * h


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        if s > 1e-30:

        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_raised_cosine(n):
    """Raised cosine: a_i = (1 + cos(2*pi*i/n))/2, normalized."""
    t = np.arange(n)
    a = (1.0 + np.cos(2.0 * np.pi * t / n)) / 2.0
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_center_gaussian(n, sigma_frac=0.3):
    """Gaussian centered in the middle."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    a = 1.0 + eta * (t - 0.5)

def _make_flat_spectrum_init(n, rng):
    """Flat power spectrum: IFFT of random phases."""
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    X[0] = 1.0
    for k in range(1, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    a = np.abs(a)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _make_stripped_flat_spectrum(n, rng):
    """Flat spectrum with DC removed and non-negativity enforced via soft thresholding."""
    N = _next_pow2(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    X = np.zeros(N, dtype=complex)
    for k in range(0, N // 2 + 1):
        X[k] = np.exp(1j * phases[k])
        X[N - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(X))[:n]
    # Soft thresholding to enforce non-negativity
    tau = 0.3 * (a.max() - a.min())
    a = np.maximum(a - tau, 0.0)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _spectrum_flatten(a, strength=0.5, rng=None):
    """
    Flatten the power spectrum of a in the Fourier domain.
    a_new = IFFT(sqrt(Phase(a) * target_mag)) where target_mag is a flat target.
    Then enforce non-negativity and renormalize.
    """
    n = len(a)
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    # Target: geometric mean of current magnitudes (flattening)
    log_mag = np.log(mag + 1e-20)
    target_log = np.full_like(log_mag, np.mean(log_mag))
    # Blend: current + strength * flat
    blended_log = (1.0 - strength) * log_mag + strength * target_log
    new_mag = np.exp(blended_log)
    # Ensure DC is real and positive
    new_mag[0] = mag[0]
    new_A = new_mag * np.exp(1j * phase)
    # Make conjugate symmetric
    full_A = np.zeros(N, dtype=complex)
    full_A[:N // 2 + 1] = new_A
    for k in range(1, N // 2):
        full_A[N - k] = np.conj(new_A[k])
    a_new = np.real(np.fft.ifft(full_A))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def _peak_pair_reduce(a, n_peaks=3, strength=0.05):
    """
    Reduce the top peaks by directly adjusting the pairs that contribute most.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = _fft_convolve(a)
    c_max = float(c.max())
    top_idx = np.argsort(c)[::-1][:n_peaks]

    # Compute pressure on each coordinate from top peaks
    pressure = np.zeros(n)
    for k in top_idx:
        # c[k] = sum_i a[i] * a[k-i] (valid indices)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                pressure[i] += a[j]
                pressure[j] += a[i]  # symmetric contribution

    # Reduce high-pressure coordinates
    p_max = pressure.max()
    if p_max < 1e-15:
        return a
    a_new = a * np.exp(-strength * pressure / p_max)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new

        c = _fft_convolve(a)
        M = float(c.max())
        M_lp = M * inflate_cur
        g = _lp_spread(a, M_lp)

        if g is None:
            break

        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp % 10 == 0:
                inflate_cur = min(1.003, inflate_cur * 1.03)
            if no_imp > 200:
                break
            continue

        g = g / sum_g
        a_new = (1.0 - blend_cur) * a + blend_cur * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        c_new = _fft_convolve(a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 30 == 0:
                blend_cur = max(0.05, blend_cur * 0.92)
                inflate_cur = max(1.0001, inflate_cur * 0.97)
            if no_imp > max(150, n):
                a = best_a.copy()
                blend_cur = blend
                inflate_cur = inflate
                no_imp = 0

        a = a_new

    return best_a


def _fine_polish(a, deadline, max_iter=50000):
    """Very fine gradient polish with tiny beta."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    c = _fft_convolve(a)
    c_max = float(c.max())
    beta = max(1e-10, 0.005 * c_max)

    best_a = a.copy()
    best_score = float(2.0 * n * c_max)

    z = np.log(a)
    z -= np.mean(z)
    lr = 0.005
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        _, w = _smooth_max_topk(c, beta, topk=10)
        grad = _conv_grad_smooth(a, w, n)
        vp = grad + 1e-9 * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            break

        z = z - lr * gz
        z = np.clip(z, -50.0, 50.0)
        z -= np.mean(z)
        e = np.exp(z)
        a = e / e.sum()

        c = _fft_convolve(a)
        score = float(2.0 * n * c.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr = max(0.0005, lr * 0.9)

        if it % 500 == 0:
            beta = max(1e-14, beta * 0.85)

        if no_imp > max(5000, 4 * n):
            break

    return best_a


def _peak_chase(a, deadline, max_iter=10000):
    """Iteratively reduce the single top peak using direct pair reduction + gradient step."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)
    no_imp = 0
    strength = 0.03
    blend = 0.5

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Peak pair reduction
        a_pp = _peak_pair_reduce(a, n_peaks=3, strength=strength)

        # Blend with original
        a_new = (1.0 - blend) * a + blend * a_pp
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        c_new = _fft_convolve(a_new)
        score_new = float(2.0 * n * c_new.max())

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_imp = 0
            strength = min(0.15, strength * 1.005)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                strength = max(0.005, strength * 0.9)
                blend = max(0.2, blend * 0.95)
            if no_imp > 2000:
                a = best_a.copy()
                strength = 0.03
                blend = 0.5
                no_imp = 0

        a = a_new

    return best_a


def _alternating_optimize(a, deadline):
    """
    Alternating optimization: alternate between gradient descent, LP spreading,
    peak chasing, and spectrum flattening.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = _score(a)

    phase_budgets = [0.30, 0.25, 0.20, 0.15, 0.10]  # fractions of remaining time
    methods = ['adam', 'lp', 'peak', 'spectrum', 'fine']

    t_remaining = deadline - time.monotonic()
    if t_remaining <= 0:
        return best_a

    idx = 0
    while time.monotonic() < deadline - 5:
        method = methods[idx % len(methods)]
        frac = phase_budgets[idx % len(phase_budgets)]
        sub_deadline = min(deadline - 5, time.monotonic() + frac * t_remaining * 0.7)

        if sub_deadline - time.monotonic() < 5:
            break

        if method == 'adam':
            a_new = _logit_adam_optimize(a, sub_deadline, max_iter=20000, lr=0.04,
                                         topk=max(20, n // 4), beta_frac=0.04,
                                         spectrum_period=0, spectrum_strength=0)
        elif method == 'lp':
            a_new = _lp_refine(a, sub_deadline, max_iter=10000, inflate=1.0004, blend=0.15)
        elif method == 'peak':
            a_new = _peak_chase(a, sub_deadline, max_iter=5000)
        elif method == 'spectrum':
            # Series of spectrum flattening steps with decreasing strength
            a_cur = a.copy()
            for s in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
                if time.monotonic() > sub_deadline:
                    break
                a_cur = _spectrum_flatten(a_cur, strength=s)
                # Quick gradient step after each
                a_cur = _logit_adam_optimize(a_cur, min(sub_deadline, time.monotonic() + 10),
                                             max_iter=2000, lr=0.03,
                                             topk=30, beta_frac=0.03,
                                             spectrum_period=0, spectrum_strength=0)
            a_new = a_cur
        else:  # fine
            a_new = _fine_polish(a, sub_deadline, max_iter=20000)

        sc_new = _score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Keep current but move on
            a = np.maximum(a_new, 1e-15)
            a = a / a.sum()

        idx += 1

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase, multi-length optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. Load parent, generate diverse inits across lengths
    2. Quick screening of inits
    3. Alternating optimization on top candidates
    4. Final polish on global best
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    best_seq = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            s = p.sum()
            if s > 0.01:
                parent = p / s
                sc = _score(parent)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent: score={sc:.10f}, n={parent.size}", flush=True)
    except Exception:
        pass

    # Determine lengths to try
    lengths = list(set([512, 768, 1024, 1536]))
    if parent is not None:
        lengths = [parent.size] + [l for l in lengths if l != parent.size]

    # Generate initializations
    print(f"Generating initializations, deadline in {deadline - time.monotonic():.0f}s", flush=True)
    init_pool = []

    for n_t in lengths:
        if parent is not None:
            init_pool.append(('parent', n_t, _interpolate_profile(parent, n_t)))

        init_pool.append(('raised_cos', n_t, _make_raised_cosine(n_t)))
        init_pool.append(('center_gauss', n_t, _make_center_gaussian(n_t, 0.3)))
        init_pool.append(('linear', n_t, _make_linear_profile(n_t, 1.0)))
        init_pool.append(('flat_spec', n_t, _make_flat_spectrum_init(n_t, rng)))
        init_pool.append(('strip_flat', n_t, _make_stripped_flat_spectrum(n_t, rng)))

    # Quick screening
    init_scores = []
    for name, n_t, a0 in init_pool:
        sc = _score(a0)
        init_scores.append((sc, name, n_t, a0))
    init_scores.sort(key=lambda x: x[0])

    print(f"Top 8 initializations:", flush=True)
    for sc, name, n_t, _ in init_scores[:8]:
        print(f"  {name} (n={n_t}): {sc:.8f}", flush=True)

    # Optimize top candidates with alternating method
    # Allocate time proportionally
    n_top = min(5, len(init_scores))
    top_inits = init_scores[:n_top]

    # Give more time to the parent (if it's in the top)
    time_weights = []
    for i, (sc, name, n_t, _) in enumerate(top_inits):
        if 'parent' in name:
            time_weights.append(3.0)
        elif i == 0:
            time_weights.append(2.5)
        else:
            time_weights.append(1.0)
    total_w = sum(time_weights)

    elapsed_for_search = deadline - t_start - 120  # reserve 120s for final polish
    search_deadline = t_start + elapsed_for_search

    for i, (sc0, name, n_t, a0) in enumerate(top_inits):
        if time.monotonic() > search_deadline - 60:
            break

        w_frac = time_weights[i] / total_w
        alloc_time = 0.9 * (time.monotonic() - t_start)  # not used, just for logic
        sub_deadline = min(search_deadline, time.monotonic() + w_frac * (search_deadline - t_start - 60))

        if sub_deadline - time.monotonic() < 30:
            sub_deadline = min(search_deadline, time.monotonic() + max(30, (search_deadline - time.monotonic()) * 0.3))

        print(f"\nOptimizing {name} (n={n_t}), start={sc0:.8f}, budget={sub_deadline - time.monotonic():.0f}s", flush=True)

        # Run alternating optimization
        a_opt = _alternating_optimize(a0, sub_deadline)
        sc_opt = _score(a_opt)
        print(f"  After alternating: {sc_opt:.10f}", flush=True)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.10f} ***", flush=True)

        # Additional LP refinement
        if time.monotonic() < search_deadline - 30:
            lp_dl = min(search_deadline, time.monotonic() + 60)
            a_lp = _lp_refine(a_opt, lp_dl, max_iter=5000, inflate=1.0003, blend=0.1)
            sc_lp = _score(a_lp)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"  LP improved: {best_score:.10f}", flush=True)
            a_opt = a_lp

        # Peak chase
        if time.monotonic() < search_deadline - 20:
            pc_dl = min(search_deadline, time.monotonic() + 40)
            a_pc = _peak_chase(a_opt, pc_dl, max_iter=5000)
            sc_pc = _score(a_pc)
            if sc_pc < best_score:
                best_score = sc_pc
                best_seq = a_pc.copy()
                print(f"  Peak chase improved: {best_score:.10f}", flush=True)

    # Final phase: intensive polish on the global best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\nFinal polish phase, budget={deadline - time.monotonic() - 10:.0f}s", flush=True)

        # Fine gradient polish
        fp_dl = min(deadline - 15, time.monotonic() + 80)
        a_fine = _fine_polish(best_seq, fp_dl, max_iter=40000)
        sc_fine = _score(a_fine)
        print(f"  Fine polish: {sc_fine:.10f}", flush=True)
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()

        # Final LP
        if time.monotonic() < deadline - 15:
            flp_dl = min(deadline - 10, time.monotonic() + 40)
            a_flp = _lp_refine(best_seq, flp_dl, max_iter=3000, inflate=1.0002, blend=0.08)
            sc_flp = _score(a_flp)
            if sc_flp < best_score:
                best_score = sc_flp
                best_seq = a_flp.copy()
                print(f"  Final LP: {best_score:.10f}", flush=True)

        # One more fine polish
        if time.monotonic() < deadline - 10:
            fp2_dl = min(deadline - 5, time.monotonic() + 30)
            a_fp2 = _fine_polish(best_seq, fp2_dl, max_iter=20000)
            sc_fp2 = _score(a_fp2)
            if sc_fp2 < best_score:
                best_score = sc_fp2
                best_seq = a_fp2.copy()
                print(f"  Second fine polish: {best_score:.10f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = _make_linear_profile(512, 1.5)
        best_score = _score(best_seq)

    # Final validation and cleanup
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_linear_profile(512, 1.5)

    # Evaluate with authoritative function if sequence is small enough
    if final_seq.size <= 5000:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}", flush=True)
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception as e:
            print(f"Eval error: {e}", flush=True)

    print(f"\nFINAL: score={best_score:.10f}, n={final_seq.size}, time={time.monotonic()-t_start:.1f}s", flush=True)
    return [float(x) for x in final_seq]
