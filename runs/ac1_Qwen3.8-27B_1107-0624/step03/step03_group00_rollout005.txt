        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            rows.extend([row_idx] * len(idx))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
        row_idx += 1
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(row_idx, n))
    c_obj = -np.ones(n)
    b_ub_arr = M * np.ones(row_idx)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_arr, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_refine_enhanced(a, deadline, t_init=0.4, max_solves=40):
    """Enhanced LP refinement with line search and adaptive step size."""
    n = len(a)
    current_best = a.copy()
    current_score = _score_norm(a)
    c = _full_convolve(a)
    M = float(c.max())
    t = t_init
    n_solves = 0
    
    while time.monotonic() < deadline and n_solves < max_solves:
        n_solves += 1
        b = _lp_solve_dense(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b < 1e-10:
            break
        b_norm = b / s_b
        
        # Line search for optimal step size
        best_t = t
        best_sc = current_score
        for trial_t in [t * 0.5, t, t * 1.5, t * 2.0]:
            a_new = (1.0 - trial_t) * a + trial_t * b_norm
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc_new = _score_norm(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_t = trial_t
        
        a = (1.0 - best_t) * a + best_t * b_norm
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        sc = _score_norm(a)
        
        if sc < current_score:
            current_score = sc
            current_best = a.copy()
        
        c = _full_convolve(a)
        M = float(c.max())
        t = max(0.05, best_t * 0.85)
    
    return current_best, current_score


def _peak_flatten(a, deadline, n_iters=50):
    """Direct peak-flattening: reduce weight at positions contributing to peaks."""
    n = len(a)
    best = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        # Find top peaks
        top_k = np.argsort(c)[::-1][:max(1, n // 10)]
        peak_val = c[top_k[0]]
        
        # Compute gradient contribution from top peaks
        grad = np.zeros(n)
        for k in top_k:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += a[k - idx]
        
        # Normalize gradient
        g_norm = grad / (np.linalg.norm(grad) + 1e-15)
        
        # Move opposite to gradient (reduce contributions to peaks)
        lr = 0.01 * (best_score / 2.0)
        a_new = a - lr * g_norm
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        sc = _score_norm(a_new)
        
        if sc < best_score:
            best_score = sc
            best = a_new.copy()
            a = a_new
            c = _full_convolve(a)
        else:
            lr *= 0.5
    
    return best, best_score


def _adam_logit_v2(p0, deadline, rng, lr=0.05, beta_frac=0.1, beta_decay=0.92,
                   no_imp_limit=300, noise_scale=None):
    """Improved Adam in logit space with multi-scale smooth-max."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    p = p0.copy()
    c = _full_convolve(p)
    maxc = float(c.max())
    beta = max(1e-3, beta_frac * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        # Multi-scale smooth max: combine sharp and smooth weights
        _, w_sharp = _smooth_max_and_weights(c, beta)
        beta_soft = beta * 5.0
        _, w_soft = _smooth_max_and_weights(c, beta_soft)
        w = 0.7 * w_sharp + 0.3 * w_soft

        g = _conv_cross_corr(w, p)

        vp = g
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = _softmax_from_z(z)
        c = _full_convolve(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.003, lr * 0.94)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        # Adaptive beta
        if t_adam % 30 == 0:
            beta = max(1e-4, beta * beta_decay)
            if c.size >= 2:
                sidx = np.argsort(c)[::-1]
                gap = c[sidx[0]] - c[sidx[1]]
                if gap < beta * 0.05:
                    beta = max(1e-4, beta * 0.4)

        limit = max(80, no_imp_limit)
        if no_imp > limit:
            if noise_scale is None:
                if best_score < 1.51:
                    ns = 0.06
                elif best_score < 1.53:
                    ns = 0.12
                else:
                    ns = 0.25
            else:
                ns = noise_scale
            z = best_z + ns * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = _softmax_from_z(z)
            c = _full_convolve(p)
            maxc = float(c.max())
            beta = max(1e-3, beta_frac * maxc)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def _make_profile_exponential(n, lam):
    t = np.arange(n)
    a = np.exp(-lam * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_two_bump(n, c1, c2, s):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - c1) / s) ** 2) + np.exp(-0.5 * ((t - c2) / s) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cubic(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_cosine(n, freq, amp):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_edge(n, gamma):
    """Profile that emphasizes edges."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-gamma * (t - 0.5) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_u_shape(n, gamma):
    """U-shaped profile (high at edges, low in middle)."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + gamma * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} n={p.size} [{label}] t={time.monotonic()-t_start:.0f}s")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    pn = parent.size if parent is not None else 0
    print(f"Parent n={pn}, budget={budget_s}s, seed={seed}")

    # --- Phase 1: Initial candidates at multiple scales ---
    print("Phase 1: Initialization")
    candidates = []

    # Try multiple sizes and profiles
    sizes = [512, 1024, 2048]
    if parent is not None and 256 <= pn <= 4096:
        sizes.append(pn)

    profile_configs = [
        ('lin1.0', lambda n: _make_profile_linear(n, 1.0)),
        ('lin1.5', lambda n: _make_profile_linear(n, 1.5)),
        ('lin2.5', lambda n: _make_profile_linear(n, 2.5)),
        ('cubic1', lambda n: _make_profile_cubic(n, 1.0, -0.5)),
        ('cubic2', lambda n: _make_profile_cubic(n, 2.0, -1.0)),
        ('exp0.5', lambda n: _make_profile_exponential(n, 0.5)),
        ('exp1.0', lambda n: _make_profile_exponential(n, 1.0)),
        ('exp2.0', lambda n: _make_profile_exponential(n, 2.0)),
        ('twobump1', lambda n: _make_profile_two_bump(n, 0.3, 0.7, 0.15)),
        ('twobump2', lambda n: _make_profile_two_bump(n, 0.2, 0.8, 0.25)),
        ('cos1', lambda n: _make_profile_cosine(n, 1, 0.7)),
        ('cos2', lambda n: _make_profile_cosine(n, 2, 0.5)),
        ('edge2', lambda n: _make_profile_edge(n, 2.0)),
        ('edge4', lambda n: _make_profile_edge(n, 4.0)),
        ('u1', lambda n: _make_profile_u_shape(n, 1.0)),
        ('u2', lambda n: _make_profile_u_shape(n, 2.0)),
    ]

    for n in sizes:
        for label, func in profile_configs:
            try:
                p = func(n)
                sc = _score_norm(p)
                candidates.append((sc, p, n, label))
            except Exception:
                pass

    # Add parent at multiple scales
    if parent is not None:
        for n in [1024, 2048, 4096]:
            if pn != n:
                pi = _interp_profile(parent, n)
                if pi is not None:
                    sc = _score_norm(pi)
                    candidates.append((sc, pi, n, 'parent_interp'))
        if 0 < pn <= 4096:
            sc = _score_norm(parent)
            candidates.append((sc, parent.copy(), pn, 'parent'))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.10f}")
    _update(candidates[0][1], candidates[0][0], 'init')

    # --- Phase 2: Polish top candidates with Adam ---
    print("Phase 2: Adam polish")
    phase2_end = t_start + 400.0
    polished = []
    count = 0
    top_k = 6  # polish top 6 candidates
    
    for sc, p, n, label in candidates:
        if time.monotonic() > phase2_end or count >= top_k:
            break
        per_end = min(phase2_end, time.monotonic() + 60.0)
        if per_end - time.monotonic() < 15:
            break
        p_b, sc_b = _adam_logit_v2(p, per_end, rng, lr=0.05, beta_frac=0.1,
                                    beta_decay=0.91, no_imp_limit=250)
        _update(p_b, sc_b, f'polish_{label}_n{n}')
        polished.append((sc_b, p_b, n, label + '_pol'))
        count += 1

    polished.sort(key=lambda x: x[0])
    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: LP refinement on best candidates ---
    print("Phase 3: LP refinement")
    phase3_end = t_start + 800.0
    for sc, p, n, label in polished[:3]:
        if time.monotonic() > phase3_end:
            break
        if n > 2048:
            continue
        per = min(120.0, phase3_end - time.monotonic() - 10)
        if per < 30:
            break
        p_lp, sc_lp = _lp_refine_enhanced(p, time.monotonic() + per, t_init=0.4, max_solves=25)
        _update(p_lp, sc_lp, f'LP_{label}')
        
        # Quick Adam after LP
        if time.monotonic() < phase3_end - 20:
            per2 = min(40.0, phase3_end - time.monotonic() - 10)
            if per2 > 15:
                p_a2, sc_a2 = _adam_logit_v2(p_lp, time.monotonic() + per2, rng,
                                              lr=0.03, beta_frac=0.06,
                                              beta_decay=0.93, no_imp_limit=200)
                _update(p_a2, sc_a2, f'postLP_{label}')

    # --- Phase 4: Peak flattening ---
    print("Phase 4: Peak flattening")
    phase4_end = t_start + 1000.0
    if best_seq is not None:
        per = min(80.0, phase4_end - time.monotonic() - 10)
        if per > 20:
            p_pf, sc_pf = _peak_flatten(best_seq, time.monotonic() + per, n_iters=100)
            _update(p_pf, sc_pf, 'peak_flat')
            
            # Adam after peak flattening
            if time.monotonic() < phase4_end - 20:
                per2 = min(40.0, phase4_end - time.monotonic() - 10)
                if per2 > 15:
                    p_a3, sc_a3 = _adam_logit_v2(p_pf, time.monotonic() + per2, rng,
                                                  lr=0.03, beta_frac=0.05,
                                                  beta_decay=0.94, no_imp_limit=150)
                    _update(p_a3, sc_a3, 'post_peak_flat')

    # --- Phase 5: Multi-scale escalation ---
    print("Phase 5: Multi-scale escalation")
    phase5_end = t_start + 1400.0
    if time.monotonic() < phase5_end and best_seq is not None:
        n_cur = best_seq.size
        target_ns = []
        if n_cur <= 1024:
            target_ns = [2048, 4096]
        elif n_cur <= 2048:
            target_ns = [4096]
        
        for tn in target_ns:
            if time.monotonic() > phase5_end - 60:
                break
            p_int = _interp_profile(best_seq, tn)
            if p_int is None:
                continue
            sc_int = _score_norm(p_int)
            print(f"  Escalate to n={tn}: {sc_int:.10f}")
            _update(p_int, sc_int, f'interp_n{tn}')

            # Adam at new scale
            per = min(80.0, phase5_end - time.monotonic() - 30)
            if per > 20:
                p_e, sc_e = _adam_logit_v2(p_int, time.monotonic() + per, rng,
                                            lr=0.04, beta_frac=0.08,
                                            beta_decay=0.92, no_imp_limit=250)
                _update(p_e, sc_e, f'escalate_n{tn}')

            # LP at new scale (if feasible)
            if best_seq is not None and best_seq.size <= 2048 and time.monotonic() < phase5_end - 30:
                per_lp = min(60.0, phase5_end - time.monotonic() - 20)
                if per_lp > 20:
                    p_lp2, sc_lp2 = _lp_refine_enhanced(best_seq, time.monotonic() + per_lp,
                                                         t_init=0.3, max_solves=15)
                    _update(p_lp2, sc_lp2, f'LP_esc_n{best_seq.size}')

    # --- Phase 6: Multi-restart ---
    print("Phase 6: Multi-restart")
    phase6_end = deadline - 30.0
    if time.monotonic() < phase6_end and best_seq is not None:
        n_best = best_seq.size
        n_restarts = 5
        per_r = max(15.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end:
                break
            per = min(per_r, phase6_end - time.monotonic() - 5)
            if per < 10:
                break
            z = np.log(np.maximum(best_seq, 1e-12))
            z -= z.mean()
            ns = 0.05 + 0.05 * i
            z = z + ns * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            p_s = _softmax_from_z(z)
            p_r, sc_r = _adam_logit_v2(p_s, time.monotonic() + per, rng, lr=0.04,
                                        beta_frac=0.06, beta_decay=0.93,
                                        no_imp_limit=int(0.25 * n_best))
            _update(p_r, sc_r, f'restart{i}_n{n_best}')

    # --- Phase 7: Final polish ---
    print("Phase 7: Final polish")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = deadline - time.monotonic() - 5
        if per > 15:
            # Try both Adam and LP
            half_per = per // 2
            p_f1, sc_f1 = _adam_logit_v2(best_seq, time.monotonic() + half_per, rng,
                                          lr=0.02, beta_frac=0.04, beta_decay=0.95,
                                          no_imp_limit=150)
            _update(p_f1, sc_f1, 'final_adam')
            
            if best_seq is not None and best_seq.size <= 2048:
                remaining = deadline - time.monotonic() - 5
                if remaining > 20:
                    p_f2, sc_f2 = _lp_refine_enhanced(best_seq, time.monotonic() + remaining,
                                                       t_init=0.2, max_solves=10)
                    _update(p_f2, sc_f2, 'final_lp')

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_profile_linear(512, 1.5)

        exact_sc = _score_norm(final)
        print(f"FINAL: score={exact_sc:.10f} n={final.size}")

        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(eval_sc):
                print(f"EVAL: {eval_sc:.10f}")
            else:

                # Try with sum = 1
                final2 = final / final.sum()
                eval_sc2 = float(evaluate_sequence([float(x) for x in final2]))
                print(f"EVAL2: {eval_sc2:.10f}")
        except Exception as e:
            print(f"Eval error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = _make_profile_linear(1024, 1.5)
    print(f"FALLBACK n=1024")
    return [float(x) for x in fb]
