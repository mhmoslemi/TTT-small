import numpy as np

from scipy.optimize import linprog

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = max(1, (2*n-1).bit_length())
    N = 1 << N  # next power of 2 (or just use 2n)
    N = 2 * n  # simple: use 2n as FFT size
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    c = np.maximum(c, 0)  # numerical safety
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_min_peak_cross(a):
    """Solve LP: min t s.t. conv(a,b)_k <= t for all k, sum(b) = sum(a), b >= 0."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    S = float(a.sum())

    # A_ub has shape (n_constr, n+1) where last column is for t

        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, n] = -1.0  # coefficient of t
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

"""
Optimized step function coefficient search using LP-based peak minimization
and smooth-max gradient descent.

Key innovation: A new LP formulation that directly minimizes the peak of
the cross-correlation conv(a, b) subject to sum(b) = sum(a), b >= 0.
This provides a more direct descent direction than the previous "max sum(b)
under cap" LP.

Algorithm:
1. Initialize from parent sequence (or other profiles).
2. Alternating LP refinement and smooth-max gradient descent.
3. Multi-length exploration.
4. Fine final polish.

import numpy as np
import time
from scipy.optimize import linprog


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = max(2, 1 << (2 * n - 1).bit_length())
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    # Remove tiny negative values from numerical error
    c = np.maximum(c, 0.0)
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where (A @ b)_k = conv(a, b)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_min_peak_cross(a):
    """
    Solve LP: min t s.t. conv(a,b)_k <= t for all k, sum(b) = sum(a), b >= 0.
    Returns (b_optimal, t_optimal) or (None, None) on failure.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    S = float(a.sum())
    if S < 1e-15:
        return None, None
    
    n_constr = 2 * n - 1
    A_conv = _build_conv_matrix(a)
    
    # A_ub: (n_constr, n+1), last column is for t
    A_ub = np.hstack([A_conv, -np.ones((n_constr, 1))])
    b_ub = np.zeros(n_constr)
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality: sum(b) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None

def _smooth_max_and_grad(a, beta):
    """Compute smooth max value and gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    c_max = float(c.max())
    x = (c - c_max) / beta
    x = np.clip(x, -50.0, 50.0)
    ex = np.exp(x)
    s = float(ex.sum())
    if s <= 0 or not np.isfinite(s):
        return c_max, np.zeros_like(a), c_max
    w = ex / s
    val = c_max + beta * float(np.log(s))
    
    # Gradient: d/d a_i [sum_k w_k * c_k] where c_k = sum_j a_j a_{k-j}
    # = sum_k w_k * (a_{k-i} + a_{i-(k-n+1)}) (with bounds)
    # Using FFT: grad_i = 2 * (w * a)_i (cross-correlation of w with a)
    N = max(2, 1 << (2 * n - 1).bit_length())
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fa_rev, N)
    h = h_full[n - 1:2 * n - 1]
    grad = 2.0 * h + 1e-10 * a  # small regularization
    
    return val, grad, c_max

def _interpolate(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()

    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)

    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return p / s
