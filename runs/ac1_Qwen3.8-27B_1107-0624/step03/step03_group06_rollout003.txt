"""
Optimized step function search using alternative LP formulation,
larger sequence lengths, symmetry enforcement, and adaptive restarts.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 300:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _symmetrize(a):
    n = len(a)
    return (a + a[::-1]) / 2.0

def _interpolate_to_symmetric(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return _symmetrize(a)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    return _symmetrize(p)

    a = 1.0 + alpha * (1.0 - np.abs(2.0 * t - 1.0))

def _solve_lp_minmax(a, S, inflate=1.0):
    """Solve: min t s.t. conv(a,g) <= t, sum(g) = S, g >= 0"""
    n = len(a)
    # conv(a, g)_k = sum_j a_j * g_{k-j} for k = 0, ..., 2n-2
    # We want: for each k, sum_j a_{k-j} * g_j <= t (where a_{k-j} = 0 if out of bounds)
    # Variables: g_0, ..., g_{n-1}, t
    # Objective: minimize t (so c = [0, ..., 0, 1])
    
    # Constraint: sum(g) = S -> this is an equality constraint
    # scipy's linprog uses A_ub * x <= b_ub and A_eq * x = b_eq
    
    n_vars = n + 1  # g_0..g_{n-1} and t
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0  # minimize t
    
    # Equality: sum(g) = S
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Inequality: conv(a,g)_k <= t for k = 0, ..., 2n-2
    # conv(a,g)_k - t <= 0
    # sum_j a_{k-j} * g_j - t <= 0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n_vars))
    b_ub = np.zeros(n_constraints)
    
    for k in range(n_constraints):
        # conv(a,g)_k = sum_j a_j * g_{k-j}

        for j in range(j_min, j_max + 1):
            A_ub[k, j] = a[k - j]  # coefficient of g_j is a_{k-j}
        A_ub[k, n] = -1.0  # coefficient of t
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def _line_search_golden(a, g, n_eval=20):

    best_sc = _score(a)
    best_t = 0.0

    d = lo + gr * (hi - lo)

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)

        if s < 1e-10:
            return np.inf
        return _score(a_new)
    
    fc = eval_at(c)
    fd = eval_at(d)

            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_at(d)

    best_sc = eval_at(best_t)
    
    # Also check t=0 (no change)
