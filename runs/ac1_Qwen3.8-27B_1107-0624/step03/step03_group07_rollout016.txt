**Cutting plane / Frank-Wolfe approach:**

The Frank-Wolfe (conditional gradient) method for minimizing a convex function `f` on a polytope `P`:
1. Start at a vertex of `P`
2. At each iteration, find the gradient `g` of `f` at current point
3. Find the vertex `v` of `P` that minimizes `g^T v`
4. Move towards `v`

For the probability simplex, the vertex that minimizes `g^T v` is simply `e_i` where `i = argmin(g)`. So the FW step is:
`a_new = (1-t) * a + t * e_{argmin(g)}`

This is equivalent to moving all the "mass" towards the coordinate with the smallest gradient component.

The issue with FW is slow convergence (O(1/t)), but it's simple and doesn't require projection.

**A better FW variant: Pairwise FW or Away-step FW**

The pairwise FW method moves along edges of the simplex:
1. Find `i = argmax(g)` (most positive gradient - want to decrease)
2. Find `j = argmin(g)` (most negative gradient - want to increase)
3. Move mass from `i` to `j`

The away-step FW also allows moving away from the current vertex:
1. If `g[i] > 0` for some active vertex `i`, move away from it.

        s_idx = int(np.argmin(g))

        # Line search
        t = fw_line_search(a, s_idx, max_evals=8)

        # Update
        a = a * (1.0 - t)
        a[s_idx] += t

        # Ensure validity
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s

        iter_count += 1
        if iter_count > max_iters_per_phase:
            break

    return best_a, best_val


def subgradient_adam_optimize(a0, deadline, rng, lr=0.03, beta_init=0.04,
                              beta_decay=0.93, restart_period=800):
    """Logit-space Adam optimization (similar to previous approach but refined)."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()

    c = full_convolve_fft(a0)
    max_c = float(c.max())

    z = np.log(a0)
    z = z - z.mean()

    beta = beta_init * max_c
    if beta < 1e-12:
        beta = 1e-6

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    current_lr = lr

    while time.monotonic() < deadline:
        _, w = softmax_weights(c, beta), None
        w = softmax_weights(c, beta)
        h = compute_grad_from_weights(w, a0)
        grad_a = h  # already 2*h

        dot = float(np.dot(a0, grad_a))
        dz = a0 * (grad_a - dot)

        t_adam += 1
        m = b1 * m + (1.0 - b1) * dz
        v = b2 * v + (1.0 - b2) * (dz * dz)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        a0 = np.exp(z - z.max())
        s = a0.sum()
        if s < 1e-15:
            z = best_z.copy()
            a0 = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        a0 = a0 / s

        c = full_convolve_fft(a0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)

        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a0.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t_adam % 30 == 0:
            beta = max(1e-12, beta * beta_decay)

        if no_improve > 200:
            current_lr = max(0.003, current_lr * 0.97)

        if no_improve > restart_period:
            z = best_z + 0.12 * rng.normal(size=n)
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            a0 = np.exp(z - z.max())
            a0 = a0 / a0.sum()
            c = full_convolve_fft(a0)
            max_c = float(c.max())
            beta = max(1e-12, 0.06 * max_c)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            current_lr = lr
            no_improve = 0

    return best_a, best_score


def polyak_subgradient_optimize(a0, deadline, rng, top_k=10):
    """
    Projected subgradient method with Polyak step sizes and iterate averaging.
    Minimizes max(conv(a,a)) on the probability simplex.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    f_lower = 1.0 / (2 * n - 1)  # theoretical lower bound

    c = full_convolve_fft(a)
    f_a = float(c.max())

    best_a = a.copy()
    best_f = f_a

    avg_a = a.copy()
    count = 0
    beta = 0.03 * f_a

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        f_a = float(c.max())

        # Compute softmax weights on top peaks
        w = softmax_weights(c, beta)
        g = compute_grad_from_weights(w, a)

        # Polyak step size
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-20:
            break
        lr = (f_a - f_lower) / g_norm_sq
        lr = min(lr, 1.0)  # cap to avoid huge steps

        # Gradient step
        a_candidate = a - lr * g

        # Project onto simplex
        a_new = proj_simplex(a_candidate)

        # Ensure sum is 1
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Update average
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        a = a_new

        # Check both iterate and average
        c_new = full_convolve_fft(a)
        f_new = float(c_new.max())
        if f_new < best_f:
            best_f = f_new
            best_a = a.copy()

        c_avg = full_convolve_fft(avg_a)
        f_avg = float(c_avg.max())
        if f_avg < best_f:
            best_f = f_avg
            best_a = avg_a.copy()

        # Adapt beta
        if count % 50 == 0:
            beta = max(1e-10, beta * 0.95)

        # Periodic restart from average
        if count % 200 == 0:
            a = avg_a.copy()
            c = full_convolve_fft(a)
            f_a = float(c.max())

    return best_a, best_f


def peak_shaving_refine(a0, deadline, rng, n_peaks=5):
    """
    Directly target the top peaks in the convolution and reduce them
    via coordinate descent on the simplex.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    c = full_convolve_fft(a)
    best_f = float(c.max())

    step = 0.01
    stale = 0

    while time.monotonic() < deadline:
        c = full_convolve_fft(a)
        f_a = float(c.max())
        if f_a < best_f - 1e-14:
            best_f = f_a
            best_a = a.copy()
            stale = 0
        else:
            stale += 1

        if stale > 50:
            break

        # Find top peak index
        peak_idx = int(np.argmax(c))

        # The peak at c[peak_idx] = sum_j a[j]*a[peak_idx-j]
        # To reduce it, we want to reduce a[j] for j that contribute most
        # to this peak, and increase a[j'] elsewhere.
        
        # Contribution of a[j] to c[peak_idx] is a[peak_idx - j] (if valid)
        # So the "most contributing" indices are those where a[peak_idx-j] is large.
        
        # Find which a[j] values contribute most to the peak
        contributions = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contributions[j] = a[k]
        
        # We want to move mass from high-contribution positions to low-contribution ones
        # Find the position with highest contribution (to decrease)
        if contributions.max() < 1e-15:
            break
        
        i_donate = int(np.argmax(contributions))
        
        # Find position with lowest contribution (to receive)
        # But we also need to consider the global effect
        # Simple heuristic: add mass where a is smallest
        i_receive = int(np.argmin(a))
        
        if i_donate == i_receive:
            i_receive = (i_donate + 1) % n
        
        # Deterministic step: move small amount from donate to receive
        amount = min(step * a[i_donate], 0.01)
        if amount < 1e-12:
            break
            
        a[i_donate] -= amount
        a[i_receive] += amount
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            break

        # Reduce step if not improving
        if stale > 10:
            step *= 0.9

    return best_a, best_f


def interpolate_profile(p0, n_new):
    """Interpolate profile to new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    result = np.interp(x1, x0, p0)
    s = result.sum()
    return result / s if s > 1e-10 else result


def make_random_smooth(n, rng, n_harmonics=6, max_amp=0.3):
    """Generate a random smooth positive profile."""
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harmonics + 1):
        amp = max_amp / k
        a = a + amp * rng.uniform(-1, 1) * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-method convex optimization search.
    Phase 1: Explore with FW at multiple sizes.
    Phase 2: Polish with subgradient + Polyak steps.
    Phase 3: Local refinement with logit-Adam.
    Phase 4: Peak shaving on the best solution.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get initial sequence if available
    init_seq = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            raw = np.maximum(raw, 0.0)
            s = raw.sum()
            if s > 0.01:
                init_seq = raw / s
    except Exception:
        pass

    print(f"Starting search. Budget={budget_s}s")
    if init_seq is not None:
        sc_init = compute_score(init_seq)
        print(f"Initial: n={init_seq.size}, score={sc_init:.10f}")
        update(init_seq, sc_init)

    # ===== Phase 1: Frank-Wolfe exploration at multiple sizes =====
    n_candidates = [512, 1024, 2048]
    if init_seq is not None:
        n_candidates.append(init_seq.size)
        # Remove duplicates
        n_candidates = list(dict.fromkeys(n_candidates))

    phase1_deadline = min(deadline, t_start + 500.0)
    time_per_n = 120.0

    for n_val in n_candidates:
        if time.monotonic() > phase1_deadline - 5:
            break
        print(f"\n--- Phase 1: FW at n={n_val} ---")

        # Generate diverse starting points
        starts = []
        if init_seq is not None:
            starts.append(interpolate_profile(init_seq, n_val))
        # Flat
        starts.append(np.ones(n_val) / n_val)
        # Cosine
        t = np.linspace(0, 1, n_val)
        starts.append(np.maximum(1 + 0.5 * np.cos(2 * np.pi * t), 1e-10))
        starts[-1] = starts[-1] / starts[-1].sum()
        # Random smooth
        for _ in range(3):
            starts.append(make_random_smooth(n_val, rng, n_harmonics=5, max_amp=0.25))

        best_fw_a = None
        best_fw_f = np.inf

        for idx, start in enumerate(starts):
            if time.monotonic() > phase1_deadline - 10:
                break
            dl = min(phase1_deadline, time.monotonic() + time_per_n / len(starts))
            fw_a, fw_f = frank_wolfe_optimize(start, dl, rng, max_iters_per_phase=100)
            sc = compute_score(fw_a)
            if sc < best_fw_f:
                best_fw_f = sc
                best_fw_a = fw_a.copy()
            update(fw_a, sc)
            if idx < 3:
                print(f"  FW start {idx}: score={sc:.10f}")

        if best_fw_a is not None:
            print(f"  Best FW at n={n_val}: {best_fw_f:.10f}")

    print(f"\nPhase 1 best: {best_score:.10f}")

    # ===== Phase 2: Polyak subgradient polishing =====
    phase2_deadline = min(deadline, t_start + 1000.0)
    print(f"\n--- Phase 2: Polyak subgradient ---")

    if best_seq is not None:
        n_polish = best_seq.size
    else:
        n_polish = 1024

    p2_start = interpolate_profile(best_seq if best_seq is not None else np.ones(256)/256, n_polish)
    sc_p2_start = compute_score(p2_start)
    update(p2_start, sc_p2_start)

    dl2 = min(phase2_deadline, time.monotonic() + 300.0)
    psg_a, psg_f = polyak_subgradient_optimize(p2_start, dl2, rng)
    sc_psg = compute_score(psg_a)
    update(psg_a, sc_psg)
    print(f"  After Polyak subgradient (n={n_polish}): {sc_psg:.10f}")

    # Also try at a larger size
    if best_seq is not None and best_seq.size < 2048:
        n_large = 2048
        p2_large = interpolate_profile(best_seq, n_large)
        sc_p2l = compute_score(p2_large)
        update(p2_large, sc_p2l)
        if time.monotonic() < phase2_deadline - 20:
            dl2l = min(phase2_deadline, time.monotonic() + 200.0)
            psg_l, _ = polyak_subgradient_optimize(p2_large, dl2l, rng)
            sc_psl = compute_score(psg_l)
            update(psg_l, sc_psl)
            print(f"  After Polyak at n={n_large}: {sc_psl:.10f}")

    print(f"  Phase 2 best: {best_score:.10f}")

    # ===== Phase 3: Logit-Adam local refinement =====
    phase3_deadline = min(deadline, t_start + 1400.0)
    print(f"\n--- Phase 3: Logit-Adam refinement ---")

    if best_seq is not None:
        n_ref = best_seq.size
    else:
        n_ref = 1024

    # Multiple refinement rounds with different hyperparameters
    hyperparams = [
        (0.03, 0.04, 0.93),
        (0.02, 0.03, 0.94),
        (0.015, 0.02, 0.95),
        (0.04, 0.05, 0.92),
    ]

    for lr_v, beta_f, beta_d in hyperparams:
        if time.monotonic() > phase3_deadline - 10:
            break
        p0 = best_seq.copy() if best_seq is not None else np.ones(n_ref) / n_ref
        p0 = p0 + 0.001 * rng.normal(size=n_ref)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()

        dl3 = min(phase3_deadline, time.monotonic() + 80.0)
        ref_a, ref_f = subgradient_adam_optimize(p0, dl3, rng, lr=lr_v,
                                                   beta_init=beta_f, beta_decay=beta_d)
        sc_ref = compute_score(ref_a)
        update(ref_a, sc_ref)
        print(f"  Adam lr={lr_v}, beta={beta_f}: {sc_ref:.10f}")

    print(f"  Phase 3 best: {best_score:.10f}")

    # ===== Phase 4: Peak shaving =====
    phase4_deadline = deadline
    print(f"\n--- Phase 4: Peak shaving ---")

    if best_seq is not None:
        dl4 = min(phase4_deadline, time.monotonic() + 100.0)
        ps_a, ps_f = peak_shaving_refine(best_seq, dl4, rng, n_peaks=5)
        sc_ps = compute_score(ps_a)
        update(ps_a, sc_ps)
        print(f"  After peak shaving: {sc_ps:.10f}")

        # Final micro-refinement
        if time.monotonic() < phase4_deadline - 10:
            p_final = best_seq.copy()
            p_final = p_final + 0.0003 * rng.normal(size=len(p_final))
            p_final = np.maximum(p_final, 1e-12)
            p_final = p_final / p_final.sum()
            dl4b = min(phase4_deadline, time.monotonic() + 50.0)
            fin_a, _ = subgradient_adam_optimize(p_final, dl4b, rng, lr=0.01,
                                                   beta_init=0.015, beta_decay=0.95)
            sc_fin = compute_score(fin_a)
            update(fin_a, sc_fin)
            print(f"  Final micro-refinement: {sc_fin:.10f}")

    # ===== Try a few more n values if time allows =====
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n--- Trying additional sizes ---")
        for n_try in [768, 1536, 3072, 4096]:
            if time.monotonic() > deadline - 15:
                break
            if n_try == best_seq.size:
                continue
            p_try = interpolate_profile(best_seq, n_try)
            sc_try = compute_score(p_try)
            update(p_try, sc_try)
            print(f"  n={n_try}: {sc_try:.10f}")
            if sc_try < best_score + 0.01 and time.monotonic() < deadline - 20:
                dl_t = min(deadline, time.monotonic() + 30.0)
                tt_a, _ = polyak_subgradient_optimize(p_try, dl_t, rng)
                sc_tt = compute_score(tt_a)
                update(tt_a, sc_tt)
                print(f"  After Polyak n={n_try}: {sc_tt:.10f}")

    print(f"\nFinal best score: {best_score:.10f}")

    # ===== Final validation and return =====
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = np.ones(512) / 512

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(256) / 256

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Evaluation check: {e}")

    return [float(x) for x in final_seq]
