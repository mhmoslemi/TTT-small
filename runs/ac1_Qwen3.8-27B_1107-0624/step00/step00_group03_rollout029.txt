def compute_conv_and_grad(a, t):
    """Compute smoothed max and its gradient for the autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    # Stable log-sum-exp
    tc = t * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    sum_exp = exp_tc.sum()
    w = exp_tc / sum_exp  # softmax weights
    m = (tc_max + np.log(sum_exp)) / t  # smooth max
    # Gradient: 2 * correlate(w, a, 'valid')
    grad_c = 2.0 * np.correlate(w, a, mode='valid')
    return m, grad_c, c

def adam_update(params, grads, m, v, beta1, beta2, eps, t):
    """Adam optimizer update. params and grads are numpy arrays."""
    m[:] = beta1 * m + (1 - beta1) * grads
    v[:] = beta2 * v + (1 - beta2) * (grads ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    return m_hat / (np.sqrt(v_hat) + eps)

def gradient_descent_init(a_init, t_start, t_end, n_steps, lr, seed=0):

    rng = np.random.default_rng(seed)
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        a = a + 0.01 / n
        s = a.sum()
    
    best_a = a.copy()
    best_score = np.inf
    best_m = np.inf

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)

def compute_smooth_max_and_grad(a, t):
    n = len(a)
    c = np.convolve(a, a)
    tc = t * c
    tc_shift = tc - tc.max()
    exp_tc = np.exp(tc_shift)
    w = exp_tc / exp_tc.sum()
    m = (tc.max() + np.log(exp_tc.sum())) / t
    grad_c = 2.0 * np.correlate(w, a, mode='valid')
    return m, grad_c, c
