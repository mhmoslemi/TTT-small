- 'valid': output length is max(len(a), len(v)) - min(len(a), len(v)) + 1
- 'same': output length is len(a)

In 'valid' mode, we only compute n where the overlap is complete, i.e., n = 0 to len(f) - len(v) = (2n-1) - n = n-1.

def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def fast_score(a):
    """Fast computation of the objective (without Python overhead of evaluate_sequence)."""
    n = len(a)
    b = fft_convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))

    """Single multiplicative update step."""
    n = len(a)
    b = fft_convolve(a, a)
    T = b.max() * peak_frac

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using FFT-accelerated multiplicative
updates with adaptive parameters, multiple structured starting points,
and simulated annealing local search.

Key insight: The self-convolution peak can be reduced below the trivial bound of 2
by iteratively pushing down elements that contribute most to convolution peaks.
The multiplicative update a_j <- a_j * exp(-eta * grad_j) naturally preserves
non-negativity and the simplex constraint.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n_out]


def fft_cross_correlate(e, a):
    """Compute (e cross a)[k] = sum_m e[m+k]*a[m] for k=0..len(a)-1 using FFT."""
    n = len(e)
    m = len(a)
    # (e * a_flip)[j] = sum_i e[i]*a_flip[j-i]
    # We want corr[k] = sum_i e[i]*a[i-k] for k=0..m-1
    # a_flip[t] = a[m-1-t]
    # (e * a_flip)[j] = sum_i e[i]*a[m-1-(j-i)] = sum_i e[i]*a[m-1-j+i]
    # Set m-1-j+i = i-k => j = m-1+k
    # So corr[k] = (e * a_flip)[m-1+k]
    a_flip = a[::-1]
    conv = fft_convolve(e, a_flip)
    return conv[m - 1: 2 * m - 1]


def fast_score(a):
    """Fast score computation without Python overhead."""
    n = len(a)
    b = fft_convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def multiplicative_step(a, eta, peak_frac):
    """One multiplicative update step to flatten self-convolution."""
    n = len(a)
    b = fft_convolve(a, a)
    bmax = b.max()
    T = bmax * peak_frac
    e = np.maximum(0.0, b - T)
    if e.max() < 1e-14:
        return a, bmax

    # grad[k] = sum_m e[m+k] * a[m] for k = 0..n-1
    grad = fft_cross_correlate(e, a)
    grad = np.maximum(0.0, grad)
    max_grad = grad.max()
    if max_grad < 1e-14:
        return a, bmax

    a_new = a * np.exp(-eta * grad / max_grad)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new, bmax


def run_multiplicative(a, eta, peak_frac, max_iter, deadline):
    """Run multiplicative updates until improvement stalls or time runs out."""
    n = len(a)
    best = a.copy()
    best_score = fast_score(best)
    a = a.copy()
    stall = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a, _ = multiplicative_step(a, eta, peak_frac)
        score = fast_score(a)
        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                break
    return best, best_score


def local_search_sa(a, deadline, rng, max_iters=200000):
    """Simulated annealing local search with block perturbations."""
    n = len(a)
    best = a.copy()
    best_score = fast_score(best)
    current = a.copy()
    current_score = best_score
    temp = 0.005

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        candidate = current.copy()
        n_blocks = int(rng.integers(1, 4))
        for _ in range(n_blocks):
            width = int(rng.integers(max(2, n // 300), max(3, n // 15)))
            start = int(rng.integers(0, max(1, n - width)))
            scale = np.exp(rng.normal(0.0, 0.12, size=width))
            candidate[start:start + width] *= scale

        candidate = np.clip(candidate, 0.0, None)
        s = candidate.sum()
        if s < 1e-10:
            continue
        candidate = candidate / s

        score = fast_score(candidate)
        delta = score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            current = candidate
            current_score = score
            if score < best_score:
                best = candidate.copy()
                best_score = score
                temp = max(0.0005, temp * 0.97)

        if it % 10000 == 0 and it > 0:
            print(f"    SA iter {it}: best={best_score:.6f}, cur={current_score:.6f}, T={temp:.6f}")

    return best, best_score
