def fast_score(f):
    """Compute 2*n*max(conv(f,f))/sum(f)^2 directly."""
    n = len(f)

    if s < 1e-9:
        return np.inf
    c = np.convolve(f, f)
    return 2.0 * n * c.max() / (s * s)

def gradient_step(f, temperature):
    """One step of smoothed gradient descent."""
    n = len(f)
    c = np.convolve(f, f)
    # Softmax with temperature
    sc = temperature * c
    sc -= sc.max()  # numerical stability
    p = np.exp(sc)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones_like(c) / len(c)
    else:
        p /= p_sum
    # Gradient via cross-correlation
    corr = np.correlate(p, f, mode='full')
    g = 2.0 * corr[n-1:2*n-1]
    return g, c

    s = f.sum()
    if s < 1e-9:
