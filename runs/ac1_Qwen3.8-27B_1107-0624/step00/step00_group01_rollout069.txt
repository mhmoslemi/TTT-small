"""
LP-driven search for non-negative step-function heights minimizing
    score = 2*n*max(convolve(a,a)) / (sum a)^2
which, up to the harmless global scale, equals 2*n*max(conv) when sum(a)=1.

Primary operator (from the SOTA idea): solve
    max sum(b)  s.t.  b >= 0,  (a * b)[k] <= M  for all k
with M = m_factor * max(a*a).  The optimal b is a maximal "dual" partner whose
cross-convolution with a stays under M; blending a toward b (with a line search)
reduces the peak of a*a.  We wrap this with adaptive m_factors to escape the
fixed point b ~= a, cheap projected-subgradient moves on the convolution peak,
random-structure restarts, and periodic verification against evaluate_sequence.

The working length is bounded (<= 3000) so the banded LP matrix stays fast and
memory-safe; the objective is scale- and length-invariant in "quality", so the
final returned sequence is valid at its own length.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# --------------------------------------------------------------------------- #
# Core numeric helpers
# --------------------------------------------------------------------------- #
def auto_conv(a):
    """Full linear auto-convolution a*a via FFT (clipped to >=0)."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << max(1, (size - 1).bit_length())
    fa = np.fft.rfft(a, L)
    r = np.fft.irfft(fa * fa, L)[:size]
    return np.maximum(r, 0.0)


def score_func(a):
    """Fast objective matching evaluate_sequence (no clamping in our range)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    ac = auto_conv(a)
    return float(2.0 * n * float(np.max(ac)) / (s * s))


def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def resize_to(a, n):
    """Resample a 1-D array to exactly n points (linear interp)."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.clip(np.interp(x_new, x_old, a), 0.0, None)


# --------------------------------------------------------------------------- #
# LP operator
# --------------------------------------------------------------------------- #
def build_lp_matrix(a):
    """Sparse band matrix A with A[k, j] = a[j] for max(0,k-n+1) <= j <= min(k,n-1)."""
    n = len(a)
    row_chunks = []
    col_chunks = []
    data_chunks = []
    for j in range(n):
        r0 = j
        r1 = min(j + n, 2 * n - 1)
        if r0 >= r1:
            continue
        cnt = r1 - r0
        row_chunks.append(np.arange(r0, r1, dtype=np.int32))
        col_chunks.append(np.full(cnt, j, dtype=np.int32))
        data_chunks.append(np.full(cnt, a[j], dtype=np.float64))
    rows = np.concatenate(row_chunks)
    cols = np.concatenate(col_chunks)
    data = np.concatenate(data_chunks)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_partner(a, m_factor=1.0, timeout=90):
    """Solve the LP; return normalized partner b or None on failure."""
    n = len(a)
    ac = auto_conv(a)
    M = float(np.max(ac)) * m_factor
    if M <= 0.0 or not np.isfinite(M):
        return None
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res.x is None:
        return None
    b = np.clip(res.x, 0.0, None)
    s = float(np.sum(b))
    if s <= 1e-10 or not res.success and res.status not in (2, 9):
        # accept only if we have a usable feasible-ish point
        if s <= 1e-10:
            return None
    return b / s


def lp_blend_step(a, m_factor, timeout):
    """One LP-based blended move; returns (new_a, improved, partner_close_to_a)."""
    b = solve_lp_partner(a, m_factor=m_factor, timeout=timeout)
    if b is None:
        return a, False, False
    close = bool(np.max(np.abs(b - a)) < 1e-8)
    best_a = a
    best_sc = score_func(a)
    for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0):
        cand = (1.0 - t) * a + t * b
        cand = np.clip(cand, 0.0, None)
        cand = normalize(cand)
        sc = score_func(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
    improved = best_sc < score_func(a) - 1e-12
    return best_a, improved, close


# --------------------------------------------------------------------------- #
# Cheap subgradient move on the convolution peak
# --------------------------------------------------------------------------- #
def crosscorr_w_a(w, a):
    """c[i] = sum_j a[j] * w[j+i]  (w has length 2n-1), returned for i=0..n-1."""
    n = len(a)
    need = n + len(w)
    L = 1 << max(1, (need - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    return np.fft.irfft(fw * np.conj(fa), L)[:n]


def subgradient_move(a, lr, tau, rng):
    """Projected (scaled) subgradient step on max(conv) with softmax smoothing."""
    ac = auto_conv(a)
    mx = float(np.max(ac))
    if mx <= 0 or not np.isfinite(mx):
        return a
    w = np.exp(tau * ac / mx)
    wsum = float(np.sum(w))
    if wsum <= 0 or not np.isfinite(wsum):
        return a
    w = w / wsum
    # gradient of approx max(conv) w.r.t a  ~  2 * crosscorr(w, a)
    grad = 2.0 * crosscorr_w_a(w, a)
    # keep non-negative: only move down on positive gradient, up on negative
    step = np.where(grad > 0.0, grad, 0.0)          # pure "reduce peak" descent
    a2 = a - lr * step
    if rng.random() < 0.15:                          # occasional refill of valleys
        a2 = a2 + 0.02 * lr * float(np.max(step)) * rng.random(a.size)
    a2 = np.clip(a2, 0.0, None)
    a2 = normalize(a2)
    return a2

    strat = int(rng.integers(0, 5))
    if strat == 0:
        a = rng.uniform(0.1, 1.0, n)
    elif strat == 1:
        nb = int(rng.integers(4, 20))
        bl = max(1, n // nb)
        a = np.repeat(rng.uniform(0.4, 2.0, nb), bl)
        if a.size < n:
            a = np.concatenate([a, rng.uniform(0.4, 2.0, n - a.size)])
        a = a[:n]
    elif strat == 2:
        a = rng.uniform(0.0, 1.0, n)
        k = np.ones(max(3, n // 200))
        k /= k.sum()
        a = np.convolve(a, k, mode='same')
    elif strat == 3:
        x = np.arange(n)
        a = np.ones(n)
        for _ in range(int(rng.integers(2, 5))):
            freq = float(rng.uniform(0.5, 5.0))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            a += 0.4 * np.sin(2.0 * np.pi * freq * x / n + phase)
        a = np.abs(a)
    else:
        a = rng.exponential(1.0, n) + 0.05
    a = np.clip(a, 0.0, None)
    return normalize(a)

        if imp and score_func(na) < score_func(best):
            best = na.copy()

    final_score = score_func(best)
    print("DONE it=%d n=%d score=%.6f eval=%.6f"
          % (it, best.size, final_score, evaluate_sequence(best.tolist())), flush=True)

    out = [float(x) for x in best]
    # final safety: ensure every entry is a finite non-negative python float
    clean = []
    for x in out:
        if not np.isfinite(x):
            clean.append(0.0)
        else:
            clean.append(max(0.0, x))
    s = sum(clean)
    if s < 0.01:
        clean = [1.0 / len(clean)] * len(clean)
    return clean
