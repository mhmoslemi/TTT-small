import time

def compute_objective(a):
    """Compute the objective and the location of the max."""
    n = len(a)
    c = np.convolve(a, a)
    S = np.sum(a)
    max_idx = np.argmax(c)
    max_val = c[max_idx]
    score = 2 * n * max_val / S**2
    return score, c, max_idx, S

def compute_gradient(a, c, max_idx, S):
    """Compute the subgradient of the objective."""
    n = len(a)
    max_val = c[max_idx]
    # d(score)/d(a[j]) = 2n * (dc[max_idx]/da[j]) / S^2 - 4n * max_val / S^3
    # dc[max_idx]/da[j] = a[max_idx - j] (if valid) + a[j - max_idx] (if valid)
    grad = np.zeros(n)
    for j in range(n):
        dc = 0.0
        if 0 <= max_idx - j < n:
            dc += a[max_idx - j]
        if 0 <= j - max_idx < n:
            dc += a[j - max_idx]
        grad[j] = 2 * n * dc / S**2 - 4 * n * max_val / S**3
    return grad
