"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for a non-negative step sequence.

Strategy:
  * Initialize with the binomial / (1-x)^(-1/2) construction whose
    self-convolution is (nearly) flat and whose score is already ~ pi/2.
  * Improve it with a biconvex alternating (LP) scheme: given the current
    sequence a, solve  max sum(b)  s.t.  conv(a,b)[k] <= M,  b >= 0
    (a linear program), rescale b to match a's energy, and blend.
  * Sweep several sequence lengths and seeds, keep the best by the exact
    evaluator, and time-box the whole search.
  * Finish with a light coordinate/stretch local polish.

No closures, no lambdas; all helpers are top level.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _init_binom(n):
    """a[i] = C(2i, i) / 4^i  (coefficients of (1-x)^(-1/2), truncated)."""
    i = np.arange(n, dtype=np.float64)
    # Stable recurrence: a[0]=1, a[i] = a[i-1] * (2i-1)/(2i)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def _conv_full(a):
    return np.convolve(a, a, mode="full")


def _score_internal(a):
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    m = float(np.max(_conv_full(a)))
    return 2.0 * len(a) * m / (s * s)


def _build_conv_matrix(a):
    """Sparse matrix A with A[k, j] = a[k-j] when 0 <= k-j < n."""
    n = len(a)
    nrows = 2 * n - 1
    cols = np.repeat(np.arange(n, dtype=np.int64), n)
    rows = (np.arange(n, dtype=np.int64)[:, None]
            + np.arange(n, dtype=np.int64)[None, :]).ravel()
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(nrows, n))


def _lp_next(a, M):
    """Solve max 1'b s.t. A a_obj <= M, b >= 0.  Returns b or None."""
    n = len(a)
    A = _build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method="highs",
                      options={"presolve": True,
                               "time_limit": 60.0})
    except Exception:
        return None
    if res is None or res.status not in (0, 3):  # 3 = time limit w/ iterate
        return None
    x = np.asarray(res.x, dtype=np.float64)
    if not np.all(np.isfinite(x)) or np.sum(x) <= 0.0:
        return None
    x = np.clip(x, 0.0, None)
    return x


def _alternating_polish(a, deadline, max_iters=400, blend=0.5):
    """Run the biconvex LP alternating scheme, tracking best score."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, None)
    s0 = float(np.sum(a))
    if s0 <= 0:
        return a, float("inf")
    a = a / s0
    best = a.copy()
    best_score = _score_internal(best)
    n = len(a)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        conv = _conv_full(a)
        M = float(np.max(conv))
        b = _lp_next(a, M)
        if b is None:
            break
        sb = float(np.sum(b))
        if sb <= 0:
            break
        b = b / sb
        cand = (1.0 - blend) * a + blend * b
        cand = np.clip(cand, 0.0, None)
        sc = _score_internal(cand)
        if sc < best_score:
            best = cand
            best_score = sc
        a = cand
        if it % 25 == 0:
            print("  alt it=%d score=%.6f" % (it, best_score))
    return best, best_score

    """Search for a low-score non-negative step sequence (time-boxed)."""
    t_end = time.monotonic() + max(0.0, float(budget_s))
