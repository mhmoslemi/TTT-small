import time

def propose_candidate(seed=42, budget_s=1800, **kwargs):

    Uses log-sum-exp smoothing of the max for smooth gradients.
    """

    f = np.asarray(height_sequence_1, dtype=np.float64).copy()
    f = np.clip(f, 0, 1000)

    n = len(f)
    
    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Initial score: {best_score:.6f}, n={n}")
    
    # FFT setup
    N = 1
    while N < 4 * n:
        N *= 2
    
    tau = 0.005  # temperature for log-sum-exp
    eta = 0.5  # step size
    
    iteration = 0
    eval_count = 0

        # Compute autocorrelation via FFT
        c = np.fft.irfft(np.fft.rfft(f, N) * np.fft.rfft(f, N), N)[:2*n-1]
        c = np.maximum(c, 0)  # numerical safety

        c_max = c.max()
        exp_c = np.exp((c - c_max) / tau)
        w = exp_c / exp_c.sum()
        
        # Gradient via FFT cross-correlation
        W = np.fft.rfft(w, N)
        F = np.fft.rfft(f, N)
        grad = np.fft.irfft(W * np.conj(F), N)[:n] / tau

        grad = np.clip(grad, -100, 100)
        
        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -10, 10)  # prevent overflow
        f = f * np.exp(log_update)
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()  # reset if degenerate

            f = f / s
        
        iteration += 1

            if np.isfinite(score) and score < best_score:

            # Adapt step size and temperature
            if iteration % 500 == 0:

    # Final evaluation

    if np.isfinite(score) and score < best_score:
