    return np.convolve(a, a)

def _water_fill_step(a, top_k_list, amount=0.01):
    """Move mass away from positions that contribute to the peak."""
    n = len(a)
    # Compute sensitivity to peak
    sens = np.zeros(n)
    for k in top_k_list:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                sens[i] += a[j]
    
    # Positions that contribute most to the peak
    # We want to decrease a_i where sens[i] is large
    # And increase a_i where sens[i] is small (or zero)
    
    # Scale to move 'amount' total mass
    total_sens = np.sum(sens)
    if total_sens < 1e-15:
        return a
    
    # Decrease proportionally to sensitivity
    decrease = amount * sens / total_sens
    # But don't go below 0
    decrease = np.minimum(decrease, a * 0.5)  # Don't remove more than half
    
    # Increase at positions with low sensitivity
    increase_mask = sens < np.median(sens)
    if not np.any(increase_mask):
        increase_mask = np.ones(n, dtype=bool)
    
    total_decrease = np.sum(decrease)
    if total_decrease < 1e-15:
        return a
    
    # Distribute the freed mass
    low_sens = sens[increase_mask]
    if np.sum(low_sens) < 1e-15:
        # Distribute uniformly to low-sensitivity positions
        increase = np.zeros(n)
        increase[increase_mask] = total_decrease / np.sum(increase_mask)
    else:
        # Distribute inversely proportional to sensitivity
        inv_sens = np.zeros(n)
        inv_sens[increase_mask] = 1.0 / (low_sens + 1e-10)
        increase = total_decrease * inv_sens / np.sum(inv_sens)
    
    a_new = a - decrease + increase
    a_new = np.maximum(a_new, 0)
    a_new /= np.sum(a_new)
    return a_new

"""
Multi-phase subgradient search minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses soft-max gradient with momentum, asymmetric initializations to break
symmetry, adaptive step size, and periodic reheating.
"""

import numpy as np
import time


def _conv(a):
    """Self-convolution."""
    return np.convolve(a, a)


def _obj_val(a):
    """Compute objective (assumes sum=1 normalization)."""
    n = len(a)
    c = _conv(a)
    return 2.0 * n * np.max(c)


def _soft_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*c_k))) which approximates
    max(c_k) for large beta. Returns (gradient, convolution, max_c).
    """
    n = len(a)
    c = _conv(a)
    mc = np.max(c)
    exp_c = np.exp(beta * (c - mc))
    total = np.sum(exp_c)
    weights = exp_c / total
    # grad_i = sum_k weights_k * 2*a_{k-i} (for valid indices)
    grad = np.zeros(n)
    for k in range(len(c)):
        w = 2.0 * weights[k]
        if w < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += w * a[k - hi:k - lo + 1]
    return grad, c, mc


def _hard_sens(a, top_idx):
    """Hard sensitivity for top convolution positions."""
    n = len(a)
    s = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            s[lo:hi + 1] += a[k - hi:k - lo + 1]
    return s


def _search_softmax(a, timeout, rng, beta=50.0, eta0=0.1, decay=0.999, mom=0.3):
    """Soft-max gradient descent with momentum."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return a, np.inf
    a /= s
    best = a.copy()
    c = _conv(a)
    best_mc = np.max(c)
    best_obj = 2.0 * n * best_mc
    eta = eta0
    vel = np.zeros(n)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < timeout:
        grad, c, mc = _soft_grad(a, beta)
        if mc < best_mc - 1e-10:
            best_mc = mc
            best_obj = 2.0 * n * mc
            best = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            eta = max(eta, eta0 * 0.3)
            beta = min(beta * 1.5, 200.0)
            a = a * np.exp(rng.normal(0, 0.08, n))
            a = np.maximum(a, 0.0)
            ss = np.sum(a)
            if ss < 1e-12:
                break
            a /= ss
            stall = 0

        gm = np.max(grad)
        if gm < 1e-15:
            break
        grad /= gm
        vel = mom * vel + (1.0 - mom) * grad

def _make_inits(n, rng):

    out = []

    r = x + 1.0
    out.append(r / r.sum())

    # Linear ramp reversed
    r2 = n - x
    out.append(r2 / r2.sum())

    # Exponential decay (asymmetric)
    for rho in [0.95, 0.98, 0.85]:
        e = rho ** x
        out.append(e / e.sum())

    # Asymmetric Gaussian (centered off-center)
    for frac in [0.3, 0.7]:
        g = np.exp(-0.5 * ((x - n * frac) / (n * 0.25)) ** 2)
        out.append(g / g.sum())

    # Cosine raised (one-sided emphasis)
    c1 = np.cos(np.pi * x / (2.0 * n)) ** 2
    out.append(c1 / c1.sum())

    # Power law asymmetric
    p = 1.0 / (x + 1) ** 0.6
    out.append(p / p.sum())

    # Bump + tail
    b = np.exp(-((x - n * 0.3) / (n * 0.2)) ** 2) + 0.1 * np.exp(-((x - n * 0.8) / (n * 0.1)) ** 2)
    out.append(b / b.sum())

    # Random
    r = rng.random(n)
    out.append(r / r.sum())

    # Uniform (as baseline)
    out.append(np.ones(n) / n)

    # Triangular (asymmetric)
    t = np.minimum(x + 1, n - x) * 0.5 + 0.5
    out.append(t / t.sum())

    return out


def _perturb(a, rng, strength=0.2):

    ss = np.sum(ap)
    if ss < 1e-12:

    return ap / ss
