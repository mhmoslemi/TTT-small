    b = np.convolve(a, a)
    M = b.max()

    threshold = M * (1 - top_frac * 0.1)  # k values where b_k > threshold
    active_k = np.where(b > threshold)[0]
    # Compute subgradient
    g = np.zeros(n)
    for k in active_k:
        # (a*a)_k = sum_i a_i * a_{k-i}
        # d/d a_i of (a*a)_k = a_{k-i} (with bounds)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            g[i] += a[k - i]
    g /= len(active_k)
    # Update
    a_new = np.maximum(0, a - eta * g)
    a_new /= a_new.sum()
    return a_new
