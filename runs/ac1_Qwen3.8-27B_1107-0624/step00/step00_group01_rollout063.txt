def solve_lp_for_g(f, M):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    n = len(f)
    # Build constraint matrix A (2n-1 x n)
    # A[k,j] = f[j] if 0 <= k-j <= n-1, else 0
    # The constraints are A @ g <= M (scalar)
    
    # For efficiency, build A row by row
    # Row k: A[k,:] has f[j] at position j where 0 <= k-j <= n-1, i.e., max(0,k-n+1) <= j <= min(n-1, k)
    
    # Actually, let's just build it directly
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n-1, k)
        A[k, j_min:j_max+1] = f[j_min:j_max+1]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    # Solve
    result = sco.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    return result.x

Minimizes the peak-to-average ratio of the autocorrelation of a non-negative sequence.
Uses FFT-based smooth-max gradient descent with periodic random restarts and
coordinate descent polishing.
"""

import numpy as np
import time


def compute_smooth_max_grad(f, alpha, L):
    """Compute gradient of smooth max of f*f."""
    n = len(f)
    c = np.convolve(f, f)
    # Softmax weights centered at max
    c_centered = c - c.max()
    w = np.exp(alpha * c_centered)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.ones(len(c)) / len(c)
    # Gradient via FFT correlation: grad[j] = 2 * sum_k w[k] * f[k-j]
    w_pad = np.zeros(L)
    w_pad[:2*n-1] = w
    f_pad = np.zeros(L)
    f_pad[:n] = f
    W = np.fft.rfft(w_pad)
    F = np.fft.rfft(f_pad)
    corr = np.fft.irfft(np.conj(W) * F, n=L)
    grad = 2.0 * corr[:n]
    return grad, c.max()


def coordinate_descent_pass(f, S, n_trials=3, delta=0.01):
    """One pass of coordinate descent."""

    best_score = evaluate_sequence(f.tolist())
    improved = False
    for j in range(n):
        best_val = f[j]
        best_s = best_score
        for t in range(n_trials):
            d = delta * (1 + t * 0.5)
            for sign in [1, -1]:
                trial = f[j] + sign * d
                if trial < 0:
                    continue
                f[j] = trial
                f = f * (S / f.sum())
                s = evaluate_sequence(f.tolist())
                if s < best_s:
                    best_s = s
                    best_val = trial
        f[j] = best_val
        if best_val != f[j]:  # This check is redundant but safe
            pass
        if best_s < best_score:

            improved = True

"""
Minimizes the peak-to-average ratio of autocorrelation for non-negative sequences.
Strategy: Smooth-max gradient descent (FFT-accelerated) with multiple initializations,
adaptive annealing, periodic reheats, and coordinate descent polishing.
"""

import numpy as np
import time


def compute_grad_fft(f, alpha, L):
    """Compute gradient of smooth-max of autocorrelation using FFT."""
    n = len(f)
    c = np.convolve(f, f)
    c_centered = c - c.max()
    w = np.exp(alpha * c_centered)
    ws = w.sum()
    if ws > 1e-300:
        w /= ws
    else:
        w[:] = 1.0 / len(w)
    # grad[j] = 2 * sum_k w[k] * f[k-j]  (cross-correlation)
    wp = np.zeros(L)
    wp[:2 * n - 1] = w
    fp = np.zeros(L)
    fp[:n] = f
    corr = np.fft.irfft(np.conj(np.fft.rfft(wp)) * np.fft.rfft(fp), n=L)
    return 2.0 * corr[:n], c.max()


def cd_polish(f, n_trials, delta, deadline):
    """Coordinate descent polishing. Returns (f, score)."""
    n = len(f)
    S = f.sum()
    score = evaluate_sequence(f.tolist())
    for _pass in range(30):
        if time.monotonic() > deadline:
            break
        any_change = False
        for j in range(n):
            if time.monotonic() > deadline:
                return f, score
            best_v = f[j]
            best_s = score
            for t in range(n_trials):
                d = delta * (1.0 + 0.5 * t)
                for sgn in (1.0, -1.0):
                    v = f[j] + sgn * d
                    if v < 0:
                        continue
                    f[j] = v
                    f *= (S / f.sum())
                    sv = evaluate_sequence(f.tolist())
                    if sv < best_s:
                        best_s = sv
                        best_v = v
            f[j] = best_v
            f *= (S / f.sum())
            if best_v != f[j]:
                any_change = True
            if best_s < score:
                score = best_s
                any_change = True
        if not any_change:
            break
    return f, score
