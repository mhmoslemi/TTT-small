```python
def optimize_sequence(n, max_time):
    rng = np.random.default_rng(42)
    # Start from uniform
    a = np.ones(n) / n
    best_a = a.copy()
    best_obj = 2.0  # uniform always gives 2
    
    for iteration in range(100000):
        # Compute convolution
        c = np.convolve(a, a, 'full')
        max_b = np.max(c)
        obj = 2 * n * max_b  # since sum(a) = 1
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Find peak position(s)
        # Use top few peaks for stability
        k_star = np.argmax(c)
        
        # Compute subgradient
        g = np.zeros(n)
        for j in range(n):
            k_minus_j = k_star - j
            if 0 <= k_minus_j < n:
                g[j] = a[k_minus_j] + a[j]
        
        # Adaptive step size
        # We want to reduce the peak by a fraction
        step = 0.01 * max_b / (np.sum(g**2) + 1e-15)
        
        # Update
        a = np.maximum(0, a - step * g)
        
        # Renormalize
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        
        # Occasional random perturbation
        if iteration % 500 == 499:
            noise = rng.normal(0, 0.01, n)
            a = np.maximum(0, a + noise)
            a = a / np.sum(a)
    
    return best_a, best_obj
```

```python
        kmj = k_star - j
        if 0 <= kmj < n:
            reduction = f * a[j] * a[kmj]
            a_new[j] = max(0.0, a[j] - reduction)
    
    # Renormalize
    s = np.sum(a_new)
    if s < 1e-15:
        return a
    return a_new / s
```

This is O(n) per step (after the O(n²) convolution). Very fast!

But it only targets one peak. I can extend it to target the top-K peaks:

```python
def multi_peak_reduction(a, c, K=5, f=0.05):
    """Reduce elements contributing to top-K peaks."""
    n = len(a)
    sort_idx = np.argsort(c)[::-1][:K]
    
    a_new = a.copy()
    for k in sort_idx:
        for j in range(n):
            kmj = int(k) - j
            if 0 <= kmj < n:
                reduction = f * a_new[j] * a_new[kmj]
                a_new[j] = max(0.0, a_new[j] - reduction)
    
    s = np.sum(a_new)
    if s < 1e-15:
        return a
    return a_new / s
```
