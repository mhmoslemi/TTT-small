```python
import time

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Algorithm: Mirror descent (exponential weights) with adaptive step size,
    multiple restarts, and periodic exploration of different sequence structures.
    
    The objective 2n*max_k(a*a)[k]/(sum a)^2 is scale-invariant, so we fix
    sum(a)=1 and minimize 2n*max_k(a*a)[k].
    """
    deadline = time.monotonic() + budget_s - 10
    rng = np.random.default_rng(seed)
    
    # Start from given sequence
    a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    n = len(a0)
    a = a0 / a0.sum()
    
    best_a = a.copy()
    best_score = 2 * n * np.convolve(a, a).max()
    eval_best = evaluate_sequence(best_a.tolist())
    print(f"Initial: n={n}, score={eval_best:.6f}")
    
    def compute_grad(a, k):
        """Compute gradient of (a*a)[k] w.r.t. a."""
        n = len(a)
        g = np.zeros(n)
        lo1 = max(0, k - n + 1)
        hi1 = min(n, k + 1)
        if lo1 < hi1:
            g[lo1:hi1] += a[k - np.arange(lo1, hi1)]
        lo2 = max(0, k)
        hi2 = min(n, k + n)
        if lo2 < hi2:
            g[lo2:hi2] += a[np.arange(lo2, hi2) - k]
        return g
    
    def mirror_step(a, g, lr):
        """Multiplicative weights update."""
        # Clip to avoid overflow
        exponents = -np.clip(lr * g, -20, 20)
        update = np.exp(exponents)
        a_new = a * update
        s = a_new.sum()
        if s > 1e-300:
            a_new /= s
        else:
            a_new = np.ones(len(a)) / len(a)
        return a_new
    
    t = 0
    lr = 1.0
    stuck_count = 0
    best_streak = 0
    
    while time.monotonic() < deadline:
        corr = np.convolve(a, a)
        peak = corr.max()
        current_score = 2 * n * peak
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        k_star = int(np.argmax(corr))
        g = compute_grad(a, k_star)
        
        # Also use second peak if close
        corr_sorted = np.sort(corr)[::-1]
        if corr_sorted[1] > 0.95 * corr_sorted[0]:
            k_star2 = int(np.argsort(corr)[::-1][1])
            g2 = compute_grad(a, k_star2)
            g = 0.5 * (g + g2)
        
        # Mirror descent step
        a = mirror_step(a, g, lr)
        
        # Adaptive learning rate
        if stuck_count % 500 == 0 and stuck_count > 0:
            lr *= 0.95
            if lr < 0.001:
                # Restart
                a = best_a + rng.normal(0, 0.02, n)
                a = np.clip(a, 1e-10, None)
                a /= a.sum()
                lr = 1.0
                stuck_count = 0
        
        t += 1
        if t % 1000 == 0:
            s = evaluate_sequence(a.tolist())
            print(f"t={t}, score={s:.6f}, best_eval={eval_best:.6f}, lr={lr:.4f}")
            if s < eval_best:
                eval_best = s
                best_a = a.copy()
        
        # Every 5000 steps, do a "giant step" exploration
        if t % 5000 == 0 and time.monotonic() < deadline - 30:
            # Try a few different perturbations of best_a
            for _ in range(3):
                perturb = best_a.copy()
                # Random block scaling
                w = rng.integers(10, max(20, n // 100))
                s = rng.integers(0, n - w)
                perturb[s:s+w] *= np.exp(rng.normal(0, 0.3))
                perturb = np.clip(perturb, 1e-10, None)
                perturb /= perturb.sum()
                s = evaluate_sequence(perturb.tolist())
                if s < eval_best:
                    eval_best = s
                    best_a = perturb.copy()
                    a = perturb.copy()
                    print(f"  Exploration improved to {eval_best:.6f}")
    
    # Final
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Final best score: {final_score:.6f}")
    return [float(x) for x in best_a]
```

- To reduce the product, increase |d|
- New values: a[i]' = (s + s*η)/2, a[k-i]' = (s - s*η)/2 for some η ∈ [0,1)
  (This keeps the sum = s and increases the difference)

But we need to be careful: changing a[i] also affects other autocorrelation lags. So we can't just do this greedily.

However, as a heuristic, it might work well.

**Revised algorithm:**

1. Compute autocorrelation, find peak lag k*
2. For each pair (i, k-i) at the peak lag, "unbalance" the pair (increase the difference)
3. This reduces (a*a)[k*] but might increase other lags
4. Repeat

This is similar to a "peeling" or "iterative peak reduction" algorithm.

Let me implement this:

```python
def unbalance_step(a, k, alpha):
    """For lag k, unbalance each pair (i, k-i) to reduce the product."""
    n = len(a)
    a_new = a.copy()
    # Pairs: (i, k-i) for i where 0 <= i < n and 0 <= k-i < n and i < k-i
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    for i in range(i_min, i_max + 1):
        j = k - i
        if j < 0 or j >= n or j <= i:
            continue
        s = a_new[i] + a_new[j]
        if s == 0:
            continue
        # Current difference
        d = a_new[i] - a_new[j]
        # New difference: increase by factor (1 + alpha)
        d_new = d * (1 + alpha)
        # But ensure non-negativity: s/2 > |d_new|
        max_d = s
        d_new = np.clip(d_new, -max_d + 1e-15, max_d - 1e-15)
        a_new[i] = (s + d_new) / 2
        a_new[j] = (s - d_new) / 2
    return a_new
```

Wait, but this doesn't change the sum of a, so we don't need to renormalize. Good.

But there's a problem: we're modifying a[i] and a[j] for many pairs, and these modifications interact. If i is part of multiple pairs (at different lags), we're double-counting.

Actually, for a fixed lag k, each index i is in exactly one pair (i, k-i). So for a single lag, the pairs are disjoint (except for the middle element when k is even). So the unbalancing is well-defined for a single lag.

Let me refine: for lag k, the pairs are:
- (max(0, k-n+1), k - max(0, k-n+1)) through (min(n-1, k), k - min(n-1, k))
- But we only want each pair once, so i < k-i, i.e., i < k/2.

So for i from max(0, k-n+1) to floor((k-1)/2):
- j = k - i
- Unbalance the pair (i, j)

And if k is even, the middle element a[k/2] is "self-paired" (contributes a[k/2]^2). We can't unbalance this; we can only reduce a[k/2] (and compensate by increasing others).

This is a nice structured move! Let me incorporate it into the main algorithm.

**Combined algorithm:**

1. Compute autocorrelation, find peak lag k*
2. Apply unbalancing at lag k* with some strength α
3. Check if the score improved; if not, try a smaller α or a different approach
4. Also periodically apply mirror descent steps
5. Track best solution

Let me now write the complete implementation:

```python
    Optimizes non-negative sequence to minimize 2n*max_k(a*a)[k]/(sum a)^2.
    
    Uses a combination of:
    1. Peak unbalancing: for the dominant autocorrelation lag, unbalance
       paired elements to reduce their product (by AM-GM).
    2. Mirror descent: multiplicative weights update along the autocorrelation
       gradient.
    3. Periodic restarts with perturbation to escape local optima.
    """

    a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    n = len(a0)

    print(f"Initial: n={n}, score={best_score:.6f}")
    
    def autocorr(a):
        return np.convolve(a, a)
    
    def score_from_corr(corr, n):
        return 2.0 * n * corr.max()  # since sum(a)=1
    
    def unbalance(a, k, alpha):
        """Unbalance pairs at lag k to reduce autocorrelation at that lag."""
        n = len(a)
        a_new = a.copy()
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            if j >= n or j <= i:
                continue
            s = a_new[i] + a_new[j]
            if s < 1e-15:
                continue
            d = a_new[i] - a_new[j]
            # Increase |d| by factor (1+alpha), capped at s (full unbalance)
            d_new = d * (1.0 + alpha)
            eps = 1e-15
            d_new = np.clip(d_new, -s + eps, s - eps)
            a_new[i] = (s + d_new) / 2.0
            a_new[j] = (s - d_new) / 2.0
        # Handle middle element if k is even
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                # a[mid]^2 contributes; reduce a[mid] slightly and distribute
                pass  # Skip for now
        return a_new
    
    def mirror_step(a, k, lr):
        """Mirror descent step for the quadratic at lag k."""
        n = len(a)
        g = np.zeros(n)
        lo1 = max(0, k - n + 1)
        hi1 = min(n, k + 1)
        if lo1 < hi1:
            g[lo1:hi1] += a[k - np.arange(lo1, hi1)]
        lo2 = max(0, k)
        hi2 = min(n, k + n)
        if lo2 < hi2:
            g[lo2:hi2] += a[np.arange(lo2, hi2) - k]
        
        exponents = -np.clip(lr * g, -20, 20)
        a_new = a * np.exp(exponents)
        s = a_new.sum()
        if s > 1e-300:
            a_new /= s
        else:
            a_new = np.ones(n) / n
        return a_new
    
    t = 0
    lr = 1.0
    alpha = 0.1  # Unbalancing strength
    mode = 'mirror'  # Alternate between mirror and unbalance
    stuck = 0
    
    while time.monotonic() < deadline:
        corr = autocorr(a)
        peak = corr.max()
        current_score = score_from_corr(corr, n)
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        k_star = int(np.argmax(corr))
        
        # Choose operation
        if mode == 'mirror':
            a = mirror_step(a, k_star, lr)
            mode = 'unbalance' if stuck > 100 else 'mirror'
        else:
            a = unbalance(a, k_star, alpha)
            # Renormalize (unbalancing preserves sum, but just in case)
            a = np.maximum(a, 0)
            a /= a.sum()
            mode = 'mirror'
        
        # Adaptive parameters
        if stuck % 200 == 0 and stuck > 0:
            lr *= 0.95
            alpha *= 0.9
            if lr < 0.01:
                # Restart
                a = best_a + rng.normal(0, 0.01, n)
                a = np.clip(a, 1e-10, None)
                a /= a.sum()
                lr = 1.0
                alpha = 0.1
                stuck = 0
        
        t += 1
        if t % 2000 == 0:
            s = evaluate_sequence(a.tolist())
            if s < best_score:
                best_score = s
                best_a = a.copy()
            print(f"t={t}, score={s:.6f}, best={best_score:.6f}, lr={lr:.4f}, alpha={alpha:.4f}, stuck={stuck}")
        
        # Periodic exploration
        if t % 10000 == 0 and time.monotonic() < deadline - 30:
            for trial in range(5):
                cand = best_a.copy()

                w = rng.integers(5, max(10, n // 50))
                s = rng.integers(0, n - w)
                cand[s:s+w] *= np.exp(rng.normal(0, 0.5, w))
                cand = np.clip(cand, 1e-10, None)
                cand /= cand.sum()
                cs = score_from_corr(autocorr(cand), n)
                if cs < best_score:
                    best_score = cs
                    best_a = cand.copy()
                    a = cand.copy()
                    print(f"  Exploration: improved to {best_score:.6f}")
    
    final = evaluate_sequence(best_a.tolist())
    print(f"Final: {final:.6f}")
    return [float(x) for x in best_a]
```

For a pair (i, j) where i + j = k, the contribution is a[i] * a[j].

If we fix s = a[i] + a[j] and change the difference d = a[i] - a[j], then:

Also, I need to be careful: when I modify a[i] for the pair (i, k-i), this also affects the autocorrelation at OTHER lags. For example, a[i] also appears in (a*a)[i+j'] for all j'. So the unbalancing might increase peaks at other lags.

Despite this, the unbalancing should be a good local move because:
- It directly reduces the peak at lag k
- The effect on other lags is usually smaller (since a[i] appears in many other pairs, but the change is small)

```python
"""
Optimizes non-negative step function sequence to minimize 2n*max_k(a*a)[k]/(sum a)^2.

Algorithm: Combined mirror descent (exponential weights) with peak unbalancing,
adaptive single-precision autocorrelation for speed, multiple restarts, and
periodic exploration. The key insight is that to reduce autocorrelation peak
at lag k, we should "unbalance" pairs (i, k-i) making them more unequal
(reducing their product by AM-GM), while mirror descent provides smooth
global progress.
"""

import numpy as np
import time


def _autocorr_sp(a):
    """Single-precision autocorrelation via FFT for speed."""
    n = a.size
    N = 2 * n - 1
    Nfft = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float32), Nfft)
    return np.fft.irfft(A * A, Nfft)[:N].astype(np.float64)


def _compute_grad(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a (vectorized)."""
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    lo1 = max(0, k - n + 1)
    hi1 = min(n, k + 1)
    if lo1 < hi1:
        g[lo1:hi1] += a[k - np.arange(lo1, hi1)]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if lo2 < hi2:
        g[lo2:hi2] += a[np.arange(lo2, hi2) - k]
    return g


def _mirror_step(a, g, lr):
    """Multiplicative weights update with floor."""
    exponents = -np.clip(lr * g, -30.0, 30.0)
    a_new = a * np.exp(exponents)
    s = a_new.sum()
    if s > 1e-300:
        a_new /= s
    else:
        a_new = np.ones(a.size, dtype=np.float64) / a.size
    return a_new


def _unbalance(a, k, alpha):
    """Unbalance pairs (i, k-i) at lag k to reduce their product."""
    n = a.size
    a_new = a.copy()
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    for i in range(i_min, i_max + 1):
        j = k - i
        if j >= n or j <= i:
            continue
        s = a_new[i] + a_new[j]
        if s < 1e-18:
            continue
        d = a_new[i] - a_new[j]
        d_new = d * (1.0 + alpha)
        eps = 1e-15
        if d_new > s - eps:
            d_new = s - eps
        elif d_new < -(s - eps):
            d_new = -(s - eps)
        a_new[i] = (s + d_new) * 0.5
        a_new[j] = (s - d_new) * 0.5
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function: finds sequence minimizing 2n*max_b/(sum_a)^2."""
    deadline = time.monotonic() + budget_s - 8.0
    rng = np.random.default_rng(seed)

    # Initialize from provided sequence
    a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = a0.size
    a = a0 / a0.sum()

    best_a = a.copy()
    best_inner = 2.0 * n * _autocorr_sp(a).max()
    best_eval = float(evaluate_sequence(best_a.tolist()))
    print(f"Init: n={n}, inner_score={best_inner:.6f}, eval={best_eval:.6f}")

    # Tracking
    stuck = 0
    lr = 1.0
    alpha = 0.15
    t = 0
    use_double_fft = False  # switch to double precision for final polishing

    def full_eval(x):
        sc = float(evaluate_sequence(x.tolist()))
        return sc

    while time.monotonic() < deadline:
        if use_double_fft:
            corr = np.convolve(a, a)
        else:
            corr = _autocorr_sp(a)

        peak = float(corr.max())
        current_inner = 2.0 * n * peak

        if current_inner < best_inner - 1e-10:
            best_inner = current_inner
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        k_star = int(np.argmax(corr))
        g = _compute_grad(a, k_star)

        # Check if second peak is close -> use average gradient
        corr_sorted = np.sort(corr)[::-1]
        if corr_sorted[1] > 0.97 * corr_sorted[0]:
            k2 = int(np.argsort(corr)[::-1][1])
            g2 = _compute_grad(a, k2)
            w = float(corr[k2] / (corr[k_star] + corr[k2]))
            g = (1.0 - w) * g + w * g2

        # Choose operation based on stuck counter
        if stuck < 200:
            # Mirror descent phase
            a = _mirror_step(a, g, lr)
        elif stuck < 600:
            # Try unbalancing
            a_trial = _unbalance(a, k_star, alpha)
            a_trial = np.maximum(a_trial, 0.0)
            a_trial /= a_trial.sum()
            if use_double_fft:
                trial_corr = np.convolve(a_trial, a_trial)
            else:
                trial_corr = _autocorr_sp(a_trial)
            if trial_corr.max() < peak:
                a = a_trial
            else:
                a = _mirror_step(a, g, lr)
        else:
            # Stuck: do unbalance + mirror combo
            a = _unbalance(a, k_star, alpha)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            g2 = _compute_grad(a, k_star)
            a = _mirror_step(a, g2, lr * 0.5)

        # Adaptive learning rate
        if stuck > 0 and stuck % 300 == 0:
            lr *= 0.92
            alpha *= 0.92
            if lr < 0.005:

                a = np.clip(a, 1e-10, None)
                a /= a.sum()
                lr = 0.8
                alpha = 0.15
                stuck = 0

        t += 1

        # Periodic full evaluation (expensive)
        if t % 5000 == 0 or (t % 500 == 0 and stuck == 0):
            sc = full_eval(a)
            if sc < best_eval:
                best_eval = sc
                best_a = a.copy()
                print(f"  t={t}, eval={sc:.6f} (improved)")
            elif t % 20000 == 0:
                print(f"  t={t}, eval={sc:.6f}, best={best_eval:.6f}")

        # Periodic exploration: try random perturbations of best
        if t % 15000 == 0 and time.monotonic() < deadline - 20:
            for trial in range(8):
                cand = best_a.copy()
                strategy = trial % 4
                if strategy == 0:
                    # Block scaling
                    w = int(rng.integers(5, max(10, n // 40)))
                    s = int(rng.integers(0, n - w))
                    cand[s:s + w] *= float(np.exp(rng.normal(0, 0.6)))
                elif strategy == 1:
                    # Global multiplicative noise
                    cand *= np.exp(rng.normal(0, 0.1, n))
                elif strategy == 2:
                    # Symmetrization
                    cand = 0.5 * (cand + cand[::-1])
                else:
                    # Sparse modification
                    idx = rng.choice(n, size=max(1, n // 500), replace=False)
                    for i in idx:
                        cand[i] *= float(np.exp(rng.normal(0, 0.4)))

                cand = np.clip(cand, 1e-10, None)
                cand /= cand.sum()
                if use_double_fft:
                    c_corr = np.convolve(cand, cand)
                else:
                    c_corr = _autocorr_sp(cand)
                c_inner = 2.0 * n * c_corr.max()
                if c_inner < best_inner * 0.999:
                    sc = full_eval(cand)
                    if sc < best_eval:
                        best_eval = sc
                        best_a = cand.copy()
                        a = cand.copy()
                        stuck = 0
                        print(f"  Exploration t={t}: eval={sc:.6f}")

        # Switch to double precision near the end for accuracy
        if time.monotonic() > deadline - 120:
            use_double_fft = True
            # Reset with best
            a = best_a.copy()
            stuck = 0
            lr = 0.1
            alpha = 0.05

    # Final polish with double precision
    if time.monotonic() < deadline - 2:
        for _ in range(2000):
            if time.monotonic() > deadline - 2:
                break
            corr = np.convolve(a, a)
            peak = corr.max()
            current_inner = 2.0 * n * peak
            if current_inner < best_inner - 1e-10:
                best_inner = current_inner
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
            k_star = int(np.argmax(corr))
            g = _compute_grad(a, k_star)
            a = _mirror_step(a, g, 0.05)
            if stuck > 100:
                a = _unbalance(a, k_star, 0.05)
                a = np.maximum(a, 0.0)
                a /= a.sum()

    # Return best solution
    final_score = full_eval(best_a)
    print(f"Final: score={final_score:.6f}, n={n}")
    return [float(x) for x in best_a]
```
