"""
Search for a non-negative sequence minimizing

    score(a) = 2 * len(a) * max(convolve(a, a)) / (sum(a) ** 2).

The algorithm normalizes sequences to sum 1, then minimizes the exact
score using FFT-based monitoring of max(conv(a,a)), while taking projected
gradient steps on a log-sum-exp smoothed surrogate of the maximum.
Several initializations are tried, followed by a local multiplicative
polish.  The final sequence is trimmed, clamped, and returned.
"""

import numpy as np
import time


def _as_1d_float(x):
    """Convert input to a flat float64 array, replacing non-finite entries."""
    arr = np.asarray(x, dtype=np.float64).ravel()
    if arr.size == 0:
        arr = np.ones(64, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    return arr


def _normalize_simplex(x):
    """Project/normalize a non-negative vector to sum 1 in an O(n) way."""
    x = _as_1d_float(x)
    x = np.clip(x, 0.0, None)
    s = float(x.sum())
    if s < 1e-15:
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    return x / s


def _proj_simplex_sum1(x):
    """
    Euclidean projection onto the probability simplex {z >= 0, sum(z)=1}.

    Uses the standard Duchi et al. algorithm with sorting.
    """
    x = _as_1d_float(x)
    n = x.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)

    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho_candidates = np.nonzero(u * idx > (cssv - 1.0))[0]

    if rho_candidates.size == 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    rho = int(rho_candidates[-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    w = np.maximum(x - theta, 0.0)
    s = float(w.sum())
    if s < 1e-15:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return w / s


def _next_pow2(m):
    """Smallest power of two >= m."""
    m = max(1, int(m))
    p = 1
    while p < m:
        p *= 2
    return p


def _square_convolution_fft(x, N, n):
    """
    Return (FA, c), where FA = rfft(x, N) and
    c = linear convolution x*x of length 2n-1.
    """
    fa = np.fft.rfft(x, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return fa, c


def _gradient_from_softmax_fft(w, fa, N, n):
    """
    Gradient of log-sum-exp(beta * conv(a,a)) with a fixed softmax weight w.

    Formula:
        grad[j] = 2 * sum_i a[i] * w[j+i]
    computed by an FFT cross-correlation.
    """
    fw = np.fft.rfft(w, N)
    grad = np.fft.irfft(fw * np.conjugate(fa), N)[:n]
    grad *= 2.0
    return grad


def _stable_softmax(c, beta):
    """Stable softmax of c with scale beta."""
    c = _as_1d_float(c)
    if c.size == 0:
        return np.ones(1, dtype=np.float64)
    if beta is None or not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0
    cmax = float(c.max())
    z = beta * (c - cmax)
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
    return e / s


def _fast_score_N(x, N, n=None):
    """Fast score using FFT instead of direct O(n^2) convolution."""
    x = _as_1d_float(x)
    if n is None:
        n = x.size
    if n == 0 or x.sum() < 1e-12:
        return float("inf")
    _, c = _square_convolution_fft(x, N, n)
    maxc = float(np.max(c)) if c.size else 0.0
    if maxc < 0.0 or not np.isfinite(maxc):
        maxc = 0.0
    s = float(x.sum())
    return float(2.0 * n * maxc / (s * s))


def _fast_score_auto(x):
    """Convenience wrapper for _fast_score_N."""
    x = _as_1d_float(x)
    n = x.size
    N = _next_pow2(2 * max(1, n) - 1)
    return _fast_score_N(x, N, n)


def _gaussian_length(n, center=0.5, width=0.15):
    """A simple non-negative Gaussian-shaped initialization."""
    n = max(16, int(n))
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return np.exp(-((t - center) / width) ** 2)


def _bimodal_length(n):
    """Two separated bumps, often a useful non-symmetric starting point."""
    n = max(16, int(n))
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.exp(-((t - 0.25) / 0.09) ** 2)
    a += np.exp(-((t - 0.75) / 0.09) ** 2)
    return a


def _optimize_init(init, duration_s, beta_start, beta_end, lr_base, momentum, tag):
    """
    Run projected smoothed-gradient optimization from one initialization.

    The exact score is monitored every iteration at zero extra convolution
    cost because the smoothed-gradient step already computes conv(a,a).
    """
    x = _normalize_simplex(init)
    n = x.size
    if n < 8:
        x = np.ones(32, dtype=np.float64)
        n = x.size

    N = _next_pow2(3 * n - 2)
    p = np.zeros(n, dtype=np.float64)
    lr_cur = float(max(1e-6, lr_base))
    momentum = float(np.clip(momentum, 0.0, 0.98))

    start = time.monotonic()
    duration_s = max(1.0, float(duration_s))
    deadline = start + duration_s - 1.0

    best_x = x.copy()
    _, c0 = _square_convolution_fft(x, N, n)
    best_s = _fast_score_N(x, N, n)

    it = 0
    no_improve = 0
    last_print = start
    print_every = max(0.5, min(5.0, duration_s / 40.0))

    while time.monotonic() < deadline:
        t_elapsed = time.monotonic() - start
        progress = min(1.0, t_elapsed / max(1e-9, deadline - start))
        beta = float(beta_start) * (float(beta_end) / max(1e-12, float(beta_start))) ** progress

        fa, c = _square_convolution_fft(x, N, n)

        s = float(x.sum())
        maxc = float(np.max(c)) if c.size else 0.0
        if maxc < 0.0 or not np.isfinite(maxc):
            maxc = 0.0
        if s < 1e-15:
            x = np.full(n, 1.0 / float(n), dtype=np.float64)
            p = np.zeros(n, dtype=np.float64)
            continue

        cur_score = float(2.0 * n * maxc / (s * s))
        if cur_score < best_s:
            best_s = cur_score
            best_x = x.copy()
            no_improve = 0
            if lr_cur < 0.10:
                lr_cur = min(0.10, lr_cur * 1.02)
        else:
            no_improve += 1
            if no_improve >= 6000:
                lr_cur *= 0.78
                if no_improve >= 22000:
                    x = best_x.copy()
                    p = np.zeros(n, dtype=np.float64)
                    lr_cur = max(1e-5, min(0.12, lr_cur * 1.7))
                    no_improve = 0

        w = _stable_softmax(c, beta)
        grad = _gradient_from_softmax_fft(w, fa, N, n)

        if not np.all(np.isfinite(grad)):
            p = np.zeros(n, dtype=np.float64)
            x = best_x.copy()
            continue

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm <= 1e-14:
            step = 0.0
        else:
            alpha = lr_cur / gnorm
            alpha = min(100.0, max(1e-8, alpha))
            step = alpha * grad

        p = momentum * p - step
        if not np.all(np.isfinite(p)):
            p = -step

        x_new = x + p
        if not np.all(np.isfinite(x_new)):
            x_new = x.copy()
        else:
            x_new = _proj_simplex_sum1(x_new)

        x = x_new
        it += 1

        if time.monotonic() - last_print >= print_every:
            last_print = time.monotonic()
            print(f"[{tag}] it={it} beta={beta:.3g} best={best_s:.7f} lr={lr_cur:.5g}", flush=True)

    # Final exact-ish score with the same FFT pipeline.
    final_score = _fast_score_N(best_x, _next_pow2(2 * best_x.size - 1), best_x.size)
    return best_x, final_score


def _polish_sequence(a, deadline):
    """
    Local multiplicative polish.  Uses fast FFT scoring only.
    """
    x = _normalize_simplex(a)
    n = x.size
    N = _next_pow2(2 * n - 1)
    best_x = x.copy()
    best_s = _fast_score_N(best_x, N, n)
    rng = np.random.default_rng(7357)

    max_iter = 2000000
    accept_count = 0
    last_print = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        y = best_x.copy()
        r = rng.random()

        if r < 0.55:
            width = max(1, int(rng.integers(1, max(2, n // 70))))
            start = int(rng.integers(0, max(1, n - width)))
            end = min(n, start + width)
            sigma = float(np.exp(rng.normal(np.log(0.08), 0.65)))
            sigma = float(np.clip(sigma, 0.01, 0.8))
            y[start:end] *= np.exp(rng.normal(0.0, sigma, end - start))
        elif r < 0.75:
            width = max(1, int(rng.integers(1, max(2, n // 200))))
            start = int(rng.integers(0, max(1, n - width)))
            end = min(n, start + width)
            y[start:end] += rng.normal(0.0, 0.03 * (1.0 / n + 1.0), end - start)
        elif r < 0.90:
            y = 0.5 * (y + y[::-1])
            y += rng.normal(0.0, 0.003 / float(n), y.size)
        else:
            # larger structural mutation: random intervals with strong scaling
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 25))))
                start = int(rng.integers(0, max(1, n - width)))
                end = min(n, start + width)
                y[start:end] *= float(np.exp(rng.normal(0.0, 0.25)))

        y = np.maximum(y, 0.0)
        s = float(y.sum())
        if s < 1e-15:
            continue
        y /= s

        sc = _fast_score_N(y, N, n)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_x = y.copy()
            accept_count += 1
            if time.monotonic() - last_print > 2.0:
                last_print = time.monotonic()
                print(f"[polish] it={it} best={best_s:.7f}", flush=True)

    return best_x, best_s


def _trim_final_sequence(a):
    """
    Remove leading/trailing near-zero entries and normalize.
    Trimming can reduce len(a), which directly helps the objective.
    """
    x = _as_1d_float(a)
    x = np.clip(x, 0.0, None)
    s = float(x.sum())
    if s < 1e-15:
        return []
    x = x / s

    # Zero only extremely tiny boundary mass; do not zero interior structure.
    x[x < 1e-14] = 0.0

    nz = np.nonzero(x > 0.0)[0]
    if nz.size == 0:
        return []

    x = x[int(nz[0]):int(nz[-1]) + 1]
    s = float(x.sum())
    if s < 1e-15:
        return []
    x = x / s

    # Re-check after trim.
    nz = np.nonzero(x > 0.0)[0]
    if nz.size == 0:
        return []
    return x[int(nz[0]):int(nz[-1]) + 1]
