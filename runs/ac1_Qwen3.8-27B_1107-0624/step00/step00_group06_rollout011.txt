```python
def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Find tau such that sum(max(0, v - tau)) = 1
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    # Find the largest rho such that sum_{i=1}^{rho} (u[i-1] - tau) = 1
    # where tau = (sum_{i=1}^{rho} u[i-1] - 1) / rho
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    rho_max = np.max(rho[cond])
    tau = (cssv[rho_max - 1] - 1) / rho_max
    return np.maximum(v - tau, 0)
```

import numpy as np
import time

def compute_gradient(a, p, L):
    """Compute grad[j] = 2 * sum_m a[m] * p[m+j] using FFT."""
    n = len(a)
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    S = np.fft.irfft(np.conj(A) * P, L)[:n]
    return 2.0 * S

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_max = np.max(rho[cond])
    tau = (cssv[rho_max - 1] - 1) / rho_max
    return np.maximum(v - tau, 0)

def fft_convolve_same(a):
    """Full autocorrelation using FFT (a is real, length n)."""
    n = len(a)
    L = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:2 * n - 1]

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho_max = np.max(rho[cond])
    tau = (cssv[rho_max - 1] - 1.0) / rho_max
    return np.maximum(v - tau, 0.0)
