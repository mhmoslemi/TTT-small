def make_z_init(n, kind, rng):
    if n < 2: return zeros
    i = np.arange(n, dtype=np.float64)
    z = np.zeros(n)
    if kind == 0:
       z = rng.normal(0.0, 0.3, n)
    elif kind == 1:
       fmax = max(3, min(32, n//8))
       amps = rng.normal(0.0, 1.0, fmax)
       phases = rng.uniform(0.0, 2*np.pi, fmax)
       for f in range(1, fmax+1):
           z += amps[f-1]*np.cos(2*np.pi*f*i/float(n)+phases[f-1])
       z += rng.normal(0.0,0.1,n)
    elif kind == 2:
       x = 2.0*i/float(max(1,n-1))-1.0
       z = 1.5*(x*x - 1.0) + rng.normal(0.0,0.1,n)
    else:
       z = rng.normal(0.0, 0.3, n)
    z -= z.mean()
    return np.clip(z, -30.0, 30.0)

def resample_sequence(a, n):

    m=a.size
    if m == n:
        out=a.copy()
    elif m > n:
        out=np.zeros(n)
        idx = np.floor(np.arange(n) * m / float(n)).astype(int)
        # need average blocks. Vectorized with np.add.at maybe:
        # simpler loop
        for i in range(n):
            start = int(i * m // n)
            end = int((i + 1) * m // n)
            if end <= start:
                end = start + 1
            if end > m:
                end = m
            out[i] = float(np.mean(a[start:end])) if end > start else 0.0
    else:
        idx = np.floor(np.arange(n) * m / float(n)).astype(int)
        idx = np.clip(idx, 0, m-1)
        out = a[idx]
    s=float(np.sum(out))

         out=np.ones(n)/n
    else:
         out=out/s
    return out

"""
FFT-accelerated randomized non-linear local search for non-negative step
functions.  The search optimizes the smooth log-sum-exp surrogate

    J_beta(a) = (2 / beta) * logsumexp(beta * n * conv(a, a))

on the simplex sum(a) = 1, using a softmax parameterization a = softmax(z)
and Adam.  It is run from several lengths and initializations, including the
provided parent construction, two-level random patterns, low-dimensional
Fourier patterns, and edge/U-shaped patterns.  A small exact constructor of
length 4 with value around 1.817 is used as a safe fallback.
"""

import numpy as np
import time


TARGET_SCORE = 1.5027


def next_pow2(x):
    p = 1
    if x <= 1:
        return 1
    while p < x:
        p <<= 1
    return p


def sanitize_array(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x
    return np.where(np.isfinite(x), x, 0.0)


def softmax_positive(z):
    z = sanitize_array(np.asarray(z, dtype=np.float64))
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def conv_D(a, fft_len):
    """Return n * full linear convolution of a with itself."""
    n = a.size
    fa = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(fa * fa, fft_len)
    c = c[:2 * n - 1]
    return np.maximum(c, 0.0) * float(n)


def internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size < 2:
        return np.inf
    s = float(np.sum(a))
    if s <= 0.0:
        return np.inf
    a = a / s
    fft_len = next_pow2(2 * a.size - 1)
    D = conv_D(a, fft_len)
    return 2.0 * float(np.max(D))


def make_weights(D, beta):
    Dmax = float(np.max(D))
    X = beta * (D - Dmax)
    W = np.exp(X)
    s = float(np.sum(W))
    if not np.isfinite(s) or s <= 0.0:
        W = np.zeros_like(D)
        W[int(np.argmax(D))] = 1.0
    else:
        W = W / s
    return W


def corr_W_a(W, a, M):
    n = a.size
    a_rev = a[::-1]
    FW = np.fft.rfft(W, M)
    FA = np.fft.rfft(a_rev, M)
    H = np.fft.irfft(FW * FA, M)
    return H[n - 1:2 * n - 1]


def load_parent_array():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        arr = np.ones(1024, dtype=np.float64)

    arr = arr.reshape(-1)
    arr = sanitize_array(arr)
    arr = np.clip(arr, 0.0, 1000.0)

    if arr.size == 0:
        arr = np.ones(1024, dtype=np.float64)

    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones(max(2, arr.size), dtype=np.float64)
    else:
        arr = arr / s
    return arr


def resample_sequence(a, n):
    a = sanitize_array(np.asarray(a, dtype=np.float64))
    m = a.size
    if m == n:
        out = a.copy()
    elif m > n:
        out = np.zeros(n, dtype=np.float64)
        for i in range(n):
            start = int(i * m // n)
            end = int((i + 1) * m // n)
            if end <= start:
                end = start + 1
            if end > m:
                end = m
            if end > start:
                out[i] = float(np.mean(a[start:end]))
    else:
        idx = np.floor(np.arange(n) * m / float(m)).astype(int)
        idx = np.clip(idx, 0, m - 1)
        out = a[idx]

    s = float(np.sum(out))
    if not np.isfinite(s) or s <= 0.0:
        out = np.ones(n, dtype=np.float64) / float(n)
    else:
        out = out / s
    return out


def z_from_sequence(a):
    a = sanitize_array(np.asarray(a, dtype=np.float64))
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if a.size == 0 or s <= 0.0:
        return np.zeros(max(2, a.size), dtype=np.float64)
    a = a / s
    a = np.maximum(a, 1e-12)
    a = a / float(np.sum(a))
    z = np.log(a)
    z -= float(np.mean(z))
    return np.clip(z, -50.0, 50.0)


def make_z_init(n, kind, rng):
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)

    if kind == 4:
        # two-level randomized pattern, positive by construction
        alpha = float(rng.uniform(0.25, 1.0))
        signs = np.where(rng.random(n) < 0.5, 1.0, -1.0)
        a = 1.0 + alpha * signs
        a = np.maximum(a, 1e-8)
        a = a / float(np.sum(a))
        z = np.log(a) + rng.normal(0.0, 0.05, n)
    elif kind == 2:
        # U-shaped pattern, relatively higher at the endpoints
        if n == 1:
            z = np.zeros(1, dtype=np.float64)
        else:
            x = 2.0 * i / float(n - 1) - 1.0
            z = 1.5 * (x * x - 1.0) + rng.normal(0.0, 0.1, n)
    elif kind == 1:
        # low-frequency Fourier pattern
        fmax = max(3, min(64, n // 8))
        z = np.zeros(n, dtype=np.float64)
        amps = rng.normal(0.0, 1.0, fmax)
        phases = rng.uniform(0.0, 2.0 * np.pi, fmax)
        for f in range(1, fmax + 1):
            z += amps[f - 1] * np.cos(2.0 * np.pi * f * i / float(n) + phases[f - 1])
        z += rng.normal(0.0, 0.1, n)
    else:
        z = rng.normal(0.0, 0.3, n)

    z -= float(np.mean(z))
    return np.clip(z, -30.0, 30.0)


def run_adam_opt(n, z0, time_limit, rng, name="run"):
    if n < 2 or time_limit <= 0.0:
        return None, np.inf

    z = sanitize_array(np.asarray(z0, dtype=np.float64).copy())
    if z.size != n:
        z = make_z_init(n, 4, rng)
    z = np.clip(z, -30.0, 30.0)
    z -= float(np.mean(z))

    a = softmax_positive(z)
    fft_len = next_pow2(2 * n - 1)
    M = next_pow2(3 * n - 2)

    D = conv_D(a, fft_len)
    cur_score = 2.0 * float(np.max(D))

    best_score = cur_score
    best_a = a.copy()

    m1 = np.zeros(n, dtype=np.float64)
    m2 = np.zeros(n, dtype=np.float64)
    t = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-9

    lr = max(0.01, min(0.20, 0.10 * float(np.sqrt(1024.0 / max(1, n)))))
    beta = 20.0
    beta_end = 500.0

    no_improve = 0
    stall_bad = 0
    it = 0
    deadline = time.monotonic() + max(0.0, float(time_limit))
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        t += 1

        W = make_weights(D, beta)
        h = corr_W_a(W, a, M)
        grad_a = 4.0 * float(n) * h
        dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - dot)
        grad_z = sanitize_array(grad_z)

        m1 = b1 * m1 + (1.0 - b1) * grad_z
        m2 = b2 * m2 + (1.0 - b2) * grad_z * grad_z
        bias1 = 1.0 - b1 ** t
        bias2 = 1.0 - b2 ** t
        step = (m1 / bias1) / (np.sqrt(m2 / bias2) + eps)
        step = np.clip(step, -5.0, 5.0)

        z_try = z - lr * step
        z_try -= float(np.mean(z_try))
        z_try = np.clip(z_try, -40.0, 40.0)

        a_try = softmax_positive(z_try)
        D_try = conv_D(a_try, fft_len)
        score_try = 2.0 * float(np.max(D_try))

        old_score = cur_score
        if np.isfinite(score_try) and score_try <= old_score * 1.05 + 1e-10:
            z = z_try
            a = a_try
            D = D_try
            cur_score = score_try

            if score_try < best_score - 1e-12:
                best_score = score_try
                best_a = a.copy()
                no_improve = 0
                beta = min(beta_end, beta * 1.05)
                lr = min(0.30, lr * 1.01)
            else:
                no_improve += 1

            if score_try > old_score * 1.02:
                lr *= 0.85
        else:
            no_improve += 1
            stall_bad += 1
            lr *= 0.85
            beta = max(5.0, beta * 0.9)
            if score_try > old_score * 1.5:
                stall_bad += 1

        if no_improve > 900:
            if beta < 120.0:
                beta = min(beta_end, beta * 1.4)
            else:
                beta = max(20.0, beta * 0.85)
                z = z + rng.normal(0.0, 0.05, n)
                z -= float(np.mean(z))
                z = np.clip(z, -40.0, 40.0)
                a = softmax_positive(z)
                D = conv_D(a, fft_len)
                cur_score = 2.0 * float(np.max(D))
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
            no_improve = 0
            stall_bad = 0

        if stall_bad > 30 or lr < 5e-6:
            break

        if it % 5000 == 0:
            print(f"{name} n={n} it={it} wall={time.monotonic()-t0:.1f}s "
                  f"cur={cur_score:.6f} best={best_score:.6f} beta={beta:.1f} lr={lr:.4f}")

    return best_a, best_score


def local_block_refine(a, time_limit, rng):
    a = sanitize_array(np.asarray(a, dtype=np.float64).copy())
    n = a.size
    if n < 2 or time_limit <= 0.0:
        return a, internal_score(a)

    s = float(np.sum(a))
    if s <= 0.0:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / s

    fft_len = next_pow2(2 * n - 1)
    D = conv_D(a, fft_len)
    cur_score = 2.0 * float(np.max(D))
    best_score = cur_score
    best = a.copy()

    deadline = time.monotonic() + max(0.0, float(time_limit))
    no_improve = 0

    while time.monotonic() < deadline:
        cand = best.copy()

        low = max(1, n // 1000)
        high = max(low + 1, n // 5)
        if high > n:
            high = n
        width = int(rng.integers(low, high))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.15)))

        if rng.random() < 0.30:
            cand = 0.5 * (cand + cand[::-1])

        s = float(np.sum(cand))
        if s <= 1e-12:
            continue
        cand = cand / s

        Dc = conv_D(cand, fft_len)
        sc = 2.0 * float(np.max(Dc))

        if np.isfinite(sc) and sc < cur_score - 1e-12:
            best = cand
            cur_score = sc
            if sc < best_score:
                best_score = sc
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 8000:
            break

    return best, best_score


def fallback_sequence():
    # Optimized one-edge pattern of length 4: [x, y, y, y]
    # x = (1 + 3*sqrt(3)) / 13, y = (4 - sqrt(3)) / 13.

    x = (1.0 + 3.0 * np.sqrt(3.0)) / 13.0
    y = (4.0 - np.sqrt(3.0)) / 13.0
    return np.array([x, y, y, y], dtype=np.float64)
