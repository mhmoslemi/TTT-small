import numpy as np

from scipy.fft import fft, ifft


def _next_pow2(n):
    """Return smallest power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _convolve_self(a, fs):
    """Compute linear self-convolution of a using precomputed fft size fs."""
    n = len(a)
    size = 2 * n - 1
    A = fft(a, fs)
    C = ifft(A * A)
    return C[:size].real


def _project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    s = v.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, 1.0 / n)
    rho = idx[0]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_subgrad(a, active, fs):
    """Compute subgradient: 2*convolve(a, h)[:n] where h is uniform on active."""

    h[active] = 1.0 / len(active)
    A = fft(a, fs)
    H = fft(h, fs)
    G = ifft(A * H)
