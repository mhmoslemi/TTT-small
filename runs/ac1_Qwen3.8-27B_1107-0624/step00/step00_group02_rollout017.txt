    """Optimize using soft-max gradient descent."""
    n = len(f)
    a = f.copy()
    a = a / a.sum()  # normalize to sum=1
    
    best_a = a.copy()
    best_m = np.max(np.convolve(a, a))
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < max_time:
        # Compute convolution
        b = np.convolve(a, a)
        m = b.max()
        
        if m < best_m:
            best_m = m
            best_a = a.copy()
        
        # Compute soft-max weights

    return None

"""
LP-based and gradient-based optimization for minimizing the auto-convolution peak.
Uses alternating LP steps (for big moves) and soft-max gradient descent (for refinement).
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def build_conv_matrix(f):
    """Build sparse convolution constraint matrix."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        j_range = range(j_start, j_end + 1)
        rows.extend([k] * len(j_range))
        cols.extend(j_range)
        vals.extend(f[j_start:j_end + 1])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_step(f, M):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.success:
        return res.x
    return None


def lp_refine(a, max_time=100.0, verbose=True):
    """LP-based refinement loop. Returns best sequence and its max-conv."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = np.max(np.convolve(a, a))
    start = time.monotonic()
    t = 0.15
    it = 0
    stagnant = 0
    
    while time.monotonic() - start < max_time:
        M = np.max(np.convolve(a, a))
        
        g = solve_lp_step(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        
        # Rescale g to same sum as a
        a_sum = a.sum()
        g = g * (a_sum / g_sum)
        
        # Convex combination
        a_new = (1 - t) * a + t * g
        
        m_new = np.max(np.convolve(a_new, a_new))
        
        if m_new < best_m * (1 - 1e-10):
            best_m = m_new
            best_a = a_new.copy()
            a = a_new.copy()
            stagnant = 0
            if verbose:
                print(f"    LP iter {it}: m={m_new:.8f}, score={2*n*m_new:.6f}")
        else:
            a = a_new.copy()
            stagnant += 1
            if stagnant > 20:
                # Perturb and reset
                a = a + 0.01 * np.random.default_rng(it).standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stagnant = 0
                t = 0.05
        
        # Adaptive t
        if m_new <= M:
            t = min(0.4, t * 1.05)
        else:
            t = max(0.02, t * 0.7)
        
        it += 1
    
    return best_a, best_m


def softmax_grad_refine(a, max_time=100.0, beta=30.0, lr_init=0.3, verbose=True):
    """Soft-max gradient descent refinement."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = np.max(np.convolve(a, a))
    start = time.monotonic()
    lr = lr_init
    it = 0
    
    while time.monotonic() - start < max_time:
        b = np.convolve(a, a)
        m = float(b.max())
        
        if m < best_m:
            best_m = m
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print(f"    SG iter {it}: m={m:.8f}, score={2*n*m:.6f}")
        
        # Soft-max weights
        max_b = float(b.max())
        # Use shifted exp for numerical stability
        exp_b = np.exp(beta * (b - max_b))
        w = exp_b / exp_b.sum()
        
        # Gradient computation via FFT
        # grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
        # = 2 * cross_correlate(w, a)[j]
        # cross_correlate(w, a)[j] = convolve(w, a[::-1])[n-1+j]
        conv_w_a = np.fft.irfft(
            np.fft.rfft(w, 2 * (2 * n - 1)) * np.fft.rfft(a[::-1], 2 * (2 * n - 1))
        )[:2 * (2 * n - 1) - 1]
        # Hmm, this is getting complicated with FFT sizes. Let me use direct convolve.
        
        # For n up to ~5000, direct convolve is fine (O(n^2) but fast in numpy)
        if n <= 5000:
            conv_w_a = np.convolve(w, a[::-1])
            cross = conv_w_a[n - 1: 2 * n - 1]
        else:
            # Use FFT
            N_fft = 1 << (2 * (2 * n - 1) - 1).bit_length()
            W = np.fft.rfft(w, N_fft)
            A_rev = np.fft.rfft(a[::-1], N_fft)
            conv_w_a = np.fft.irfft(W * A_rev)[:2 * (2 * n - 1) - 1]
            cross = conv_w_a[n - 1: 2 * n - 1]
        
        grad = 2.0 * cross
        
        # Update with adaptive learning rate
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new_sum
        
        # Adaptive learning rate
        m_new = float(np.convolve(a_new, a_new).max())
        if m_new > m * 1.5:
            lr *= 0.5
        elif m_new < m * 0.95:
            lr = min(1.0, lr * 1.1)
        
        a = a_new
        it += 1
    
    return best_a, best_m
