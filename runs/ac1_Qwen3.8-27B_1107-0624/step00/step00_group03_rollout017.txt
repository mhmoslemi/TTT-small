        for _ in range(500):
            if time.monotonic() > deadline:
                break
            # Small random perturbation
            f_cand = f + 0.01 * rng.standard_normal(n)
            f_cand = np.maximum(f_cand, 0.0)
            f_cand = f_cand / f_cand.sum()
            if eval_local(f_cand) < best_score:
                f = f_cand
                best_f = f_cand.copy()
                best_score = eval_local(f)

    # Verify with global evaluator
    final_score = float(evaluate_sequence(best_f.tolist()))
    print(f"\nFinal best: local={best_score:.6f}, global={final_score:.6f}")
    print(f"Total LP calls: {total_lp_calls}")
    print(f"Total time: {time.monotonic()-start_time:.1f}s")

    return [float(x) for x in best_f]
