"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Power-norm gradient descent (smooth approximation of max)
2. LP-based dual updates for global moves
3. Multiple initializations and local refinement
"""

import numpy as np
import time

def compute_obj_normalized(a):
    """Compute objective for sequence with sum=1: 2*n*max_b."""
    n = len(a)
    b = fftconvolve(a, a)
    return 2.0 * n * b.max()

def compute_gradient_power(a, b, p=4):
    """
    Compute gradient of (sum b_k^p)^(1/p) with respect to a.
    b = convolve(a, a)
    """
    # w[k] = b[k]^(p-1)
    w = b ** (p - 1)
    # part1[j] = sum_k w[k] * a[j-k] = (w * a)[j]
    part1 = np.convolve(w, a, 'full')[:len(a)]
    # part2[j] = sum_k w[k] * a[k-j] = correlate(w, a)[j + n - 1]
    part2 = np.correlate(w, a, 'full')[len(a)-1:2*len(a)-1]
    return part1 + part2

def lp_dual_update(a, M, max_time=30):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns rescaled g or None if failed.
    """
    from scipy.optimize import linprog
    n = len(a)
    
    # Build constraint matrix A: (2n-1) x n
    # A[k, m] = a[k-m] if 0 <= k-m < n and 0 <= m < n
    # This is a banded matrix.
    # For memory efficiency, build it row by row or use a sparse representation.
    
    # For n up to ~2000, dense is fine (2n-1 x n ≈ 4M entries)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        m_start = max(0, k - n + 1)
        m_end = min(n, k + 1)
        # A[k, m] = a[k-m] for m in [m_start, m_end)
        # k-m ranges from k-m_start to k-(m_end-1)
        # = k - max(0,k-n+1) to k - (min(n,k+1)-1)
        for m in range(m_start, m_end):
            A[k, m] = a[k - m]

    """Compute full linear convolution a*a using FFT."""
    from scipy.signal import fftconvolve
    return fftconvolve(a, a)

def _compute_gradient(a, b, p=4):
    """
    Gradient of (sum b^p)^(1/p) w.r.t. a.
    Assumes b = convolve(a, a).
    """
    if p == 1:
        w = np.ones_like(b)
    else:
        w = b ** (p - 1)
    n = len(a)
    # part1[j] = sum_k w[k] * a[j-k] for valid j,k
    # = (w * a)[j]
    part1 = np.convolve(w, a, 'full')[:n]
    # part2[j] = sum_k w[k] * a[k-j] for valid j,k  
    # = correlate(w, a)[j + n - 1]
    part2 = np.correlate(w, a, 'full')[n-1:2*n-1]
    return part1 + part2

def _lp_update(a, M, timeout=30):
    """
    LP: max sum(g) s.t. convolve(a,g)[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    from scipy.optimize import linprog
    n = len(a)
    
    # Build A matrix: A[k,m] = a[k-m] if 0<=k-m<n else 0
    # Shape: (2n-1) x n
    A = np.zeros((2*n-1, n))

        A[m:m+n, m] = a  # column m is a shifted and zero-padded version

    c = -np.ones(n)
    
    # Constraints: A@g <= M (upper bound on convolution)
    #              g >= 0 (non-negativity)

    b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])
