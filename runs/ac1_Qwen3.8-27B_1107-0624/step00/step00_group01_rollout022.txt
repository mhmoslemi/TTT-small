    inits.append(normalize_nonneg(make_u_shape(n, 3.0)))
    inits.append(normalize_nonneg(make_block_pattern(n)))
    inits.append(normalize_nonneg(make_cosine_envelope(n)))
    inits.append(normalize_nonneg(1.0 + 0.5 * np.sin(2 * np.pi * np.arange(n) / n)))
    inits.append(normalize_nonneg(np.abs(np.sin(2 * np.pi * np.arange(n) / (2 * n)))))

    # Find the best initial
    best_a = inits[0].copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Initial best score: {best_score:.6f}")

    for init in inits[1:]:
        score = float(evaluate_sequence(init.tolist()))
        if score < best_score:
            best_a = init.copy()
            best_score = score
            print(f"Better initialization: score={best_score:.6f}")

    # Gradient-based optimization
    a = best_a.copy()
    tau = 1.0
    base_lr = 0.02
    iteration = 0
    stall_counter = 0
    max_stall = 300
    tau_schedule = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]
    tau_idx = 0

    last_exact_check = 0
    consecutive_no_improve = 0

    while time.monotonic() < deadline:
        # Compute gradient
        dL_da, L, c, p = logsumexp_and_gradient(a, tau)
        n_curr = len(a)
        grad = 2.0 * n_curr * dL_da

        # Project gradient to maintain sum=1 constraint
        grad = grad - np.mean(grad)

        # Adaptive learning rate with decay
        lr = base_lr / (1.0 + 0.005 * iteration)

        # Line search: try multiple step sizes
        f_curr = 2.0 * n_curr * L
        best_trial = a
        best_trial_f = f_curr

        for scale in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            a_trial = a - scale * grad
            a_trial = np.maximum(a_trial, 0.0)
            s = np.sum(a_trial)
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            c_trial = np.convolve(a_trial, a_trial, 'full')
            tc_trial = tau * c_trial
            tc_max_trial = np.max(tc_trial)
            L_trial = (tc_max_trial + np.log(np.sum(np.exp(tc_trial - tc_max_trial)))) / tau
            f_trial = 2.0 * n_curr * L_trial
            if f_trial < best_trial_f:
                best_trial_f = f_trial
                best_trial = a_trial

        a = best_trial

        # Periodic exact evaluation
        if iteration - last_exact_check >= 200:
            exact_score = float(evaluate_sequence(a.tolist()))
            last_exact_check = iteration
            if exact_score < best_score:
                best_a = a.copy()
                best_score = exact_score
                consecutive_no_improve = 0
                stall_counter = 0
                print(f"Iter {iteration}: exact score improved to {best_score:.6f}")
            else:
                consecutive_no_improve += 1

        # Adapt tau: increase for tighter max approximation
        if iteration % 500 == 0 and tau_idx < len(tau_schedule) - 1:
            tau_idx += 1
            tau = tau_schedule[tau_idx]
            print(f"Iter {iteration}: increased tau to {tau}")

        # If stalled, restart with perturbation
        if consecutive_no_improve >= max_stall:
            print(f"Iter {iteration}: stalled, restarting. Best score: {best_score:.6f}")
            strategy = rng.integers(0, 4)
            if strategy == 0:
                # Add noise
                a = best_a + 0.1 * rng.standard_normal(n)
            elif strategy == 1:
                # Try a new structured init
                a = inits[rng.integers(0, len(inits))].copy()
            elif strategy == 2:
                # Scale and add directional perturbation
                d = rng.standard_normal(n)
                d -= np.mean(d)
                a = best_a + 0.2 * d
            else:
                # Random weighted combination
                other = inits[rng.integers(0, len(inits))].copy()
                w = rng.uniform(0.3, 0.7)
                a = w * best_a + (1 - w) * other

            a = normalize_nonneg(a)
            tau = 0.5
            tau_idx = 0
            base_lr = 0.01 + 0.03 * rng.random()
            consecutive_no_improve = 0
            last_exact_check = iteration

        iteration += 1

    # Final evaluation
    final_score = float(evaluate_sequence(best_a.tolist()))
    if final_score < best_score:
        best_score = final_score
        print(f"Final evaluation improved: {best_score:.6f}")

    print(f"Final best score: {best_score:.6f} at iteration {iteration}")

    # Ensure output is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    return result
