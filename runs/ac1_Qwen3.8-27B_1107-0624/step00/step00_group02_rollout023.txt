```python
import numpy as np
import time

def simplex_projection(v, s=1.0):
    """Project v onto the simplex {x : sum(x) = s, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Find the threshold
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```
