def compute_subgradient(b, active_lags, fft_size):

    n = len(b)
    W = np.zeros(2 * n - 1)
    W[active_lags] = 1.0
    b_rev = b[::-1]
    
    # Full convolution of b_rev and W
    N = fft_size  # should be >= n + (2n-1) - 1 = 3n - 2
    fb = np.fft.rfft(b_rev, N)
    fw = np.fft.rfft(W, N)
    conv = np.fft.irfft(fb * fw, N)
    
    # r[j] = conv[n-1 + j] for j = 0, ..., n-1
    grad = 2.0 * conv[n-1 : 2*n-1]
    return grad
