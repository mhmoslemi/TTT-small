    L = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, L)
    FA = np.fft.rfft(a[::-1], L)
    conv = np.fft.irfft(P * FA, L)
    return 2.0 * conv[n - 1:2 * n - 1]


def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_candidates = np.where(u * idx > (cssv - S))[0]
    rho = rho_candidates[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _topk_grad(a, c, top_k):
    """Subgradient using top-K peaks of autocorrelation."""
    n = len(a)
    k = min(top_k, len(c))
    part = np.argpartition(c, -k)[-k:]
    order = np.argsort(c[part])[::-1]
    top_idx = part[order]
    weights = c[top_idx].astype(np.float64)
    wsum = weights.sum()
    if wsum <= 1e-30:
        weights = np.ones(k) / k
    else:
        weights /= wsum
    g = np.zeros(n)
    for m, w in zip(top_idx, weights):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            g[lo:hi + 1] += w * 2.0 * a[m - np.arange(lo, hi + 1)]
    return g


def _block_swap(a, rng, n):
    """Swap two random contiguous blocks."""
    out = a.copy()
    w1 = rng.integers(max(2, n // 100), max(3, n // 10))
    w2 = rng.integers(max(2, n // 100), max(3, n // 10))
    s1 = rng.integers(0, max(1, n - w1))
    s2 = rng.integers(0, max(1, n - w2))
    # Ensure non-overlapping
    if s1 + w1 > s2 and s2 + w2 > s1:
        return a
    tmp = out[s1:s1 + w1].copy()
    out[s1:s1 + w1] = out[s2:s2 + w2]
    out[s2:s2 + w2] = tmp
    return out


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Soft-max gradient descent with FFT cross-correlation,
    adaptive step size, temperature annealing, and diverse restarts
    to minimize peak autocorrelation of non-negative sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    a_init = np.asarray(height_sequence_1, dtype=np.float64)
    a_init = np.clip(a_init, 0.0, 1000.0)
    n = len(a_init)
    S = np.sqrt(2.0 * n)

    cur_sum = a_init.sum()
    if cur_sum < 1e-10:
        a_init = np.ones(n)
        cur_sum = n
    a_init = a_init * S / cur_sum

    best = a_init.copy()
    c_init = _fft_autocorr(best)
    best_score = c_init.max()
    print(f"n={n}, S={S:.4f}, initial max_autocorr={best_score:.8f}")

    a = best.copy()
    eta = 0.05
    T = 1.0  # temperature multiplier (relative to c range)
    stall = 0
    it = 0
    no_best_improve = 0
    max_no_improve = 8000

    while time.monotonic() < deadline:
        it += 1
        c = _fft_autocorr(a)
        max_c = c.max()
        c_range = c.max() - c.min()
        if c_range < 1e-10:
            c_range = max_c if max_c > 1e-10 else 1.0

        # Choose gradient method
        use_soft = (c_range > 1e-8 and it % 3 != 0)
        if use_soft:
            T_eff = T * c_range
            c_norm = (c - c.min()) / (T_eff + 1e-30)
            c_norm -= c_norm.max()
            p = np.exp(c_norm)
            p_sum = p.sum()
            if p_sum > 0:
                p /= p_sum
            else:
                p = np.full(len(c), 1.0 / len(c))
            g = _soft_grad(a, p)
        else:
            top_k = 1 if stall < 20 else rng.integers(1, 6)
            g = _topk_grad(a, c, top_k)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            stall += 1
        else:
            # Backtracking line search
            found = False
            scale = eta * S / g_norm  # normalize step relative to sequence scale
            for ls in range(10):
                step = scale / (10.0 ** ls)
                a_new = a - step * g
                a_new = _project_simplex(a_new, S)
                c_new = _fft_autocorr(a_new)
                nc = c_new.max()
                if nc < max_c * (1.0 - 1e-9):
                    a = a_new
                    found = True
                    if nc < best_score:
                        best = a.copy()
                        best_score = nc
                        no_best_improve = 0
                    else:
                        no_best_improve += 1
                    break
            if not found:
                stall += 1
            else:
                if stall > 0:
                    eta = min(eta * 1.03, 5.0)
                stall = 0

        # Restart logic
        if stall > 60 or no_best_improve > max_no_improve:
            strategy = rng.integers(0, 5)
            if strategy == 0:
                a = a + rng.normal(0, 0.03 * S / n, size=n)
            elif strategy == 1:
                a = a + rng.normal(0, 0.15 * S / n, size=n)
            elif strategy == 2:
                a = _block_swap(a, rng, n)
            elif strategy == 3:
                a = best.copy() + rng.normal(0, 0.08 * S / n, size=n)
            else:
                # Exponential random
                a = rng.exponential(1.0, size=n)
            a = _project_simplex(a, S)
            stall = 0
            no_best_improve = 0
            eta = 0.05
            T = max(T * 0.85, 0.1)

        # Anneal temperature
        T = max(T * (1.0 - 0.00005), 0.1)

        if it % 3000 == 0:
            c_chk = _fft_autocorr(a)
            print(f"Iter {it}: cur={c_chk.max():.8f}, best={best_score:.8f}, "
                  f"eta={eta:.5f}, T={T:.4f}, stall={stall}, no_impr={no_best_improve}")

    # Final polish: try a few top-1 subgradient steps
    for _ in range(200):
        if time.monotonic() > deadline:
            break
        c = _fft_autocorr(best)
        max_c = c.max()
        g = _topk_grad(best, c, 1)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        scale = 0.02 * S / g_norm
        for ls in range(8):
            step = scale / (10.0 ** ls)
            a_new = best - step * g
            a_new = _project_simplex(a_new, S)
            c_new = _fft_autocorr(a_new)
            if c_new.max() < max_c * (1.0 - 1e-9):
                best = a_new
                max_c = c_new.max()
                break

    final = np.clip(best, 0.0, 1000.0)
    result = [float(x) for x in final]
    print(f"Total iterations: {it}")
    print(f"Final max_autocorr: {best_score:.8f}")
    eval_score = evaluate_sequence(result)
    print(f"Verified score via evaluate_sequence: {eval_score:.8f}")
    return result
