```python
def subgradient_step(a, lr=0.1):

    B = np.convolve(a, a)
    B_max = B.max()
    S = a.sum()
    
    # Find peak positions
    tol = B_max * 1e-6  # tolerance for "equal" peak
    P = np.where(B >= B_max - tol)[0]
    
    # Compute g(j) = sum_{k in P} a(k-j) for valid k-j
    g = np.zeros(n)
    for k in P:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                g[j] += a[idx]
    
    # Update
    c = B_max / S
    a_new = a - lr * (g - c)
    a_new = np.maximum(a_new, 0)
    
    # Renormalize
    if a_new.sum() > 1e-10:
        a_new = a_new * (S / a_new.sum())
    
    return a_new
```

def compute_subgradient(a, peak_fraction=1.0):
    """Compute the subgradient of the peak autocorrelation objective."""
    n = len(a)
    B = np.convolve(a, a)
    B_max = B.max()
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n), B_max, S
    
    threshold = B_max * peak_fraction
    P = np.where(B >= threshold)[0]
    
    # Limit to top 100 if too many
    if len(P) > 100:
        B_P = B[P]
        top_idx = np.argsort(B_P)[-100:]
        P = P[top_idx]
    
    g = np.zeros(n)
    for k in P:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            g[j_min:j_max+1] += a[k - j_max : k - j_min + 1 : -1]
    
    return g, B_max, S
