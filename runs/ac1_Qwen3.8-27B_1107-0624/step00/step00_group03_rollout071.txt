def build_constraint_matrix(f, n):
    """Build A where (A @ g)[k] = (f*g)[k] for full convolution. A[k,j] = f[k-j] if valid."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:

    return A


def max_conv(a):
    """Max of full self-convolution."""
    return float(np.max(np.convolve(a, a)))


def lp_solve_direction(f, M, n):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = build_constraint_matrix(f, n)

        candidates.append(x ** p)

    # Reverse ramps (decreasing)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        x = np.arange(1, n + 1, dtype=np.float64)
        candidates.append((n + 1 - x) ** p)

    # Exponential ramps
    for rate in [0.5, 1.0, 1.5, 2.0, 3.0]:
        x = np.arange(n, dtype=np.float64)
        candidates.append(np.exp(rate * x / n))

    # Log ramps
    for c in [1.0, 2.0, 3.0]:
        x = np.arange(1, n + 1, dtype=np.float64)
        candidates.append(np.log(x + c))

    # Piecewise: ramp up then flat, or flat then ramp up
    for split in [0.3, 0.5, 0.7]:
        x = np.arange(n, dtype=np.float64)
        s = int(n * split)
        a = np.ones(n)
        a[:s] = np.linspace(0.1, 1.0, s)
        candidates.append(a)
        a2 = np.ones(n)
        a2[s:] = np.linspace(1.0, 3.0, n - s)
        candidates.append(a2)

    # Evaluate all
    scored = []
    for a in candidates:
        if np.sum(a) < 0.01:
            continue
        score = float(evaluate_sequence(a.tolist()))
        scored.append((score, a))
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    scored.sort(key=lambda x: x[0])
    print(f"  Best parametric: {best_score:.6f}")
    for i in range(min(5, len(scored))):
        print(f"    [{i}] score={scored[i][0]:.6f}")

    # === Phase 2: LP refinement on top candidates ===
    print(f"Phase 2: LP refinement")
    top_k = min(5, len(scored))
    for i in range(top_k):
        if time.monotonic() > deadline:
            break
        score, a = scored[i]
        print(f"  Refining candidate {i + 1} (score={score:.6f})...")
        refined, ref_score = lp_refine(a, n, max_iters=400, deadline=deadline)
        if ref_score < best_score:
            best_score = ref_score
            best_seq = refined.copy()
            print(f"    -> {ref_score:.6f}")

    # === Phase 3: Coordinate descent ===
    print(f"Phase 3: Coordinate descent polish (from score={best_score:.6f})")
    polished, pol_score = coord_descent(best_seq, n, rng, max_iters=50000, deadline=deadline)
    if pol_score < best_score:
        best_score = pol_score
        best_seq = polished.copy()
        print(f"  -> {best_score:.6f}")

    # === Phase 4: Block perturbations + re-refine ===
    if time.monotonic() < deadline - 60:
        print(f"Phase 4: Diversification search")
        for trial in range(20):
            if time.monotonic() > deadline:
                break
            cand = block_perturbation(best_seq, rng, n)
            s = float(evaluate_sequence(cand.tolist()))
            if s < best_score:
                best_score = s
                best_seq = cand.copy()
                print(f"  Block {trial}: {best_score:.6f}")

        # Re-refine best
        if time.monotonic() < deadline - 30:
            refined, ref_score = lp_refine(best_seq, n, max_iters=500, deadline=deadline)
            if ref_score < best_score:
                best_score = ref_score
                best_seq = refined.copy()
                print(f"  Re-refined: {best_score:.6f}")

            polished, pol_score = coord_descent(best_seq, n, rng, max_iters=30000, deadline=deadline)
            if pol_score < best_score:
                best_score = pol_score
                best_seq = polished.copy()
                print(f"  Final polish: {best_score:.6f}")

    # === Phase 5: Final LP if time remains ===
    if time.monotonic() < deadline - 5:
        refined, ref_score = lp_refine(best_seq, n, max_iters=1000, deadline=deadline)
        if ref_score < best_score:
            best_score = ref_score
            best_seq = refined.copy()

    print(f"\nFINAL: score={best_score:.6f}, length={n}, sum={np.sum(best_seq):.4f}")
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
