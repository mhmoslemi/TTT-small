"""
Optimize a non-negative step-function sequence to minimize
    J = 2*n*max(convolve(f,f)) / (sum(f))^2
using per-iteration p-norm gradient (mirror/multiplicative descent) with an
adaptive decaying step, several diverse initializations (given parent, U-shape,
dome, random), and a final block-perturbation polish.  FFT is used internally
for fast convolution; the authoritative evaluate_sequence (direct convolve) is
called periodically to track progress and only finite improvements are kept.
"""

import numpy as np
import time


# ----------------------------- top-level helpers -----------------------------

def _fft_convolve(a, b):
    """Full linear convolution of two 1-D arrays via FFT."""
    need = len(a) + len(b) - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:need]


def _project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S} by bisection on the threshold."""
    if v.max() <= 0.0:
        return np.full(len(v), S / len(v))
    lo = 0.0
    hi = float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.maximum(v - t, 0.0)


def _pnorm_gradient(f, p):
    """Gradient of sum_k c_k^p where c = convolve(f,f) (length n vector)."""
    n = len(f)
    c = _fft_convolve(f, f)                       # length 2n-1
    c_pow = np.power(np.maximum(c, 0.0), p - 1)   # length 2n-1
    R = np.zeros(2 * n - 1)
    R[:n] = f[::-1]
    g_full = _fft_convolve(c_pow, R)
    return 2.0 * p * g_full[n - 1:2 * n - 1]


def _optimize_loop(f_init, S, n, deadline, p, alpha_init, decay, eval_every, stall_limit):
    """Mirror/multiplicative p-norm gradient descent from a single start."""
    n = int(n)
    S = float(S)

    best_local_score = float(evaluate_sequence(best_local.tolist()))

    alpha = float(alpha_init)
    count = 0
    stall = 0
    last_improvement = best_local_score

    while time.monotonic() < deadline:
        count += 1
        g = _pnorm_gradient(f, p)
        gmax = float(g.max())
        if not np.isfinite(gmax) or gmax < 1e-15:
            break

        # multiplicative (mirror) update: reduce high-pressure coordinates
        f = f * np.exp(-alpha * g / gmax)
        f = np.maximum(f, 0.0)
        fs = float(f.sum())
        if fs < 1e-300:
            f = np.full(n, S / n)
        else:
            f = f * (S / fs)
        f = np.clip(f, 0.0, 1000.0)

        alpha *= decay

        if count % eval_every == 0:
            score = float(evaluate_sequence(f.tolist()))
            if np.isfinite(score) and score < best_local_score:
                best_local_score = score
                best_local = f.copy()
                if score < last_improvement * (1.0 - 1e-5):
                    last_improvement = score
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            if stall >= stall_limit:
                break

    return best_local, best_local_score


def _block_perturb(f, rng):
    """Randomized multiplicative rescale of one contiguous block."""
    n = len(f)
    c = f.copy()
    wmin = max(2, int(0.005 * n))
    wmax = max(wmin + 1, int(0.15 * n))
    width = int(rng.integers(wmin, wmax + 1))
    start = int(rng.integers(0, n - width + 1))
    c[start:start + width] *= np.exp(rng.normal(0.0, 0.20, width))
    return np.clip(c, 0.0, 1000.0)


def _u_shape(n):
    x = np.linspace(0.0, 1.0, n)
    v = 0.2 + 0.8 * (1.0 - 2.5 * 4.0 * x * (1.0 - x))
    return np.clip(v, 0.01, None)


def _dome(n):
    x = np.linspace(0.0, 1.0, n)
    return 0.2 + 0.8 * (4.0 * x * (1.0 - x))
