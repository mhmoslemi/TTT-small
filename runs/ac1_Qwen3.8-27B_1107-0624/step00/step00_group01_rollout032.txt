def compute_grad_fft(weights, a):
    """Compute 2 * cross_correlation(weights, a)[:len(a)] using FFT."""
    n = len(a)
    N = 1
    while N < 3*n - 2:
        N *= 2
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]

def softmax_grad_descent(a, max_iters, base_tau, min_tau, lr, budget_fn):
    """Run soft-max gradient descent on a (already normalized)."""
    n = len(a)
    a = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

        if budget_fn():

        b_max = np.max(b)

    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * out[:n]

def _run_optimization(a, time_budget, lr=0.005):
    """Run soft-max gradient descent with Adam for the given time budget."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    target = np.sqrt(2.0 * n)
    s = np.sum(a)
    if s < 1e-15:
        a = np.full(n, target / n)
        return a
    a = a * target / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps_adam = 0.9, 0.999, 1e-8
    
    deadline = time.monotonic() + time_budget
    best_local = a.copy()
    best_local_max = np.max(np.convolve(a, a))
    
    while time.monotonic() < deadline:
        t += 1
        b = np.convolve(a, a)
        b_max = np.max(b)
        
        if b_max < best_local_max:
            best_local_max = b_max
            best_local = a.copy()
        
        b_range = b_max - np.median(b)
        if b_range < 1e-12:
            b_range = max(b_max * 0.1, 1e-10)
        
        # Temperature: geometric annealing
        frac = min(1.0, t / 30000.0)
        tau = b_range * 0.25 * (0.02 / 0.25) ** frac
        tau = max(tau, 1e-12 * b_max)
        
        # Soft-max weights
        exponents = (b - b_max) / tau
        exponents = np.clip(exponents, -60.0, 0.0)
        weights = np.exp(exponents)
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            idx = int(np.argmax(b))
            weights = np.zeros_like(b)
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient via FFT
        grad = _fft_grad(weights, a)
        
        # Adam update
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.full(n, target / n)
        else:
            a = a * target / s
    
    a = a * target / np.sum(a)
    return a

def propose_candidate(budget_s=1800):
    """
    Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
    1. Soft-max gradient descent with Adam optimizer
    2. Basin hopping with geometrically decreasing perturbation
    3. Multiple random initializations with various sequence lengths
    4. Final refinement of the best solution found
    """
    total_deadline = time.monotonic() + max(1.0, budget_s - 8.0)
    
    # Get initial sequence
    a0 = np.asarray(height_sequence_1, dtype=np.float64)
    a0 = np.clip(a0, 0.0, 1000.0)
    n0 = len(a0)
    
    rng = np.random.default_rng(2024)
    
    best = a0.copy()
    best_score = float(evaluate_sequence([float(x) for x in best]))
    print(f"Start: score={best_score:.6f}, n={n0}", flush=True)
    
    n = n0
    target = np.sqrt(2.0 * n)
    s0 = np.sum(best)
    a_star = best * target / s0 if s0 > 1e-10 else np.full(n, target / n)
    
    # --- Phase 1: Optimize initial sequence (35% of time) ---
    t1 = (total_deadline - time.monotonic()) * 0.35
    print(f"Phase 1: t={t1:.0f}s", flush=True)
    a = _run_optimization(a_star, time_budget=t1, lr=0.005)
    sc = float(evaluate_sequence([float(x) for x in a]))
    if sc < best_score:
        best_score = sc
        best = a.copy()
        a_star = a.copy()
        print(f"  Phase 1: score={best_score:.6f}", flush=True)
    
    # --- Phase 2: Basin hopping (45% of time) ---
    t2 = (total_deadline - time.monotonic()) * 0.45
    print(f"Phase 2: t={t2:.0f}s", flush=True)
    restart_budget = max(2.0, t2 / 80.0)  # budget per restart
    sigmas = [0.8, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    n_per_sigma = max(1, int(t2 / (len(sigmas) * restart_budget)))
    
    for si, sigma in enumerate(sigmas):
        for ri in range(n_per_sigma):
            if time.monotonic() >= total_deadline:
                break
            if time.monotonic() > total_deadline - 5.0:
                break
            
            # Perturb
            a_pert = a_star * np.exp(rng.normal(0.0, sigma, size=n))
            a_pert = np.maximum(a_pert, 0.0)
            sp = np.sum(a_pert)
            if sp < 1e-10:
                continue
            a_pert = a_pert * target / sp
            
            # Optimize
            a_opt = _run_optimization(a_pert, time_budget=restart_budget, lr=0.005)
            sc = float(evaluate_sequence([float(x) for x in a_opt]))
            if sc < best_score:
                best_score = sc
                best = a_opt.copy()
                a_star = a_opt.copy()
                print(f"  S{si}R{ri}: score={best_score:.6f}", flush=True)
    
    # --- Phase 3: Different lengths (15% of time) ---
    t3 = (total_deadline - time.monotonic()) * 0.15
    if t3 > 20.0:
        print(f"Phase 3: t={t3:.0f}s", flush=True)
        for n_try in [n, int(n*2/3), int(n*4/3), 4096, 2048]:
            if time.monotonic() > total_deadline - 5.0:
                break
            if n_try <= 10 or n_try > 20000:
                continue
            ts = np.sqrt(2.0 * n_try)
            for attempt in range(2):
                if time.monotonic() > total_deadline - 5.0:
                    break
                a_r = rng.uniform(0.05, 1.0, size=n_try)
                a_r = a_r * ts / np.sum(a_r)
                tb = max(5.0, (total_deadline - time.monotonic()) * 0.3)
                a_opt = _run_optimization(a_r, time_budget=tb, lr=0.005)
                sc = float(evaluate_sequence([float(x) for x in a_opt]))
                if sc < best_score:
                    best_score = sc
                    best = a_opt.copy()
                    print(f"  n={n_try} a{attempt}: score={best_score:.6f}", flush=True)
    
    # --- Final refinement ---
    tf = total_deadline - time.monotonic() - 3.0
    if tf > 10.0:
        print(f"Final refinement: t={tf:.0f}s", flush=True)
        s = np.sum(best)
        a_ref = best * np.sqrt(2.0 * len(best)) / s if s > 1e-10 else best.copy()
        a_final = _run_optimization(a_ref, time_budget=tf, lr=0.002)
        sc = float(evaluate_sequence([float(x) for x in a_final]))
        if sc < best_score:
            best_score = sc
            best = a_final.copy()
            print(f"  Final: score={best_score:.6f}", flush=True)
    
    print(f"RESULT: score={best_score:.6f}, n={len(best)}", flush=True)
    return [float(x) for x in best]
