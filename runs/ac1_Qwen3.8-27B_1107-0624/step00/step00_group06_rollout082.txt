```
def fft_convolve(a,b):
    a=np.asarray(a,float); b=np.asarray(b,float)
    if a.size==0 or b.size==0: return np.zeros(0)
    need=a.size+b.size-1
    size=1
    while size<need: size <<=1
    fa=np.fft.rfft(a, size)
    fb=np.fft.rfft(b, size)
    out=np.fft.irfft(fa*fb, size)[:need]
    out=np.asarray(out, dtype=np.float64)
    out[out<0.0]=0.0
    return out
```

```python
import numpy as np
import time

try:
    from scipy.optimize import minimize as _MINIMIZE
except Exception:
    _MINIMIZE = None

_THETA_BOUND = 2.0
_TANH_EPS = 0.99


def normalize_pos(a):
    """Return a non-negative 1D array normalized to sum one."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if (not np.isfinite(s)) or s <= 1e-300:
        n = max(1, a.size)
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def fft_convolve(a, b):
    """Real linear convolution via FFT, with small negatives clipped to zero."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    if a.size == 0 or b.size == 0:
        return np.zeros(0, dtype=np.float64)
    need = a.size + b.size - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa * fb, size)[:need]
    out = np.asarray(out, dtype=np.float64)
    out[out < 0.0] = 0.0
    return out


def cross_gradient(a, weights):
    """
    H[m] = sum_j weights[m+j] * a[j], for m=0..n-1.
    This is the unnormalized gradient kernel for c = conv(a,a).
    """
    n = a.size
    weights = np.asarray(weights, dtype=np.float64).reshape(-1)
    if weights.size != 2 * n - 1:
        weights = np.resize(weights, 2 * n - 1)
    conv = fft_convolve(a[::-1], weights)
    g = conv[n - 1:2 * n - 1]
    g = np.asarray(g, dtype=np.float64)
    g[g < 0.0] = 0.0
    return g


def softmax_weights(u, T):
    """Stable softmax over u with temperature T."""
    u = np.asarray(u, dtype=np.float64).reshape(-1)
    if u.size == 0:
        return u.copy()
    umax = float(u.max())
    if (not np.isfinite(umax)) or T <= 1e-9:
        q = np.zeros_like(u)
        q[int(np.argmax(u))] = 1.0
        return q
    sh = (u - umax) / T
    e = np.exp(sh)
    s = e.sum()
    if (not np.isfinite(s)) or s <= 0.0:
        q = np.zeros_like(u)
        q[int(np.argmax(u))] = 1.0
        return q
    return e / s


def surrogate_value(u, T):
    """Soft peak proxy: u_max + T*logsumexp((u-u_max)/T)."""
    u = np.asarray(u, dtype=np.float64).reshape(-1)
    umax = float(u.max())
    if not np.isfinite(umax):
        return np.inf
    if T <= 1e-9:
        return umax
    sh = (u - umax) / T
    e = np.exp(sh)
    s = e.sum()
    if (not np.isfinite(s)) or s <= 0.0:
        return umax
    return float(umax + T * np.log(s))


def fast_score(a):
    """FFT-based score approximation for sum(a)=1: 2*n*max(conv(a,a))."""
    a = normalize_pos(a)
    n = a.size
    c = fft_convolve(a, a)
    u = n * c
    return float(2.0 * max(0.0, float(u.max())))


def exact_score_list(a):
    """Authoritative score via the provided evaluator, when available."""
    try:
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return np.inf


def build_features(n, fourier_modes=10, poly_deg=5):
    """Zero-mean normalized smooth features: polynomial and low-order Fourier."""
    x = np.arange(n, dtype=np.float64)
    feats = []

    if n > 1:
        u = 2.0 * x / (n - 1) - 1.0
    else:
        u = np.zeros(n, dtype=np.float64)

    for d in range(1, poly_deg + 1):
        f = u ** d
        f = f - f.mean()
        std = f.std()
        if std > 1e-12:
            f = f / std
        else:
            f = np.zeros(n, dtype=np.float64)
        feats.append(f)

    if fourier_modes > 0:
        for r in range(1, fourier_modes + 1):
            phase = 2.0 * np.pi * r * x / max(1, n)

            f = np.cos(phase)
            f = f - f.mean()
            std = f.std()
            if std > 1e-12:
                f = f / std
            else:
                f = np.zeros(n, dtype=np.float64)
            feats.append(f)

            f = np.sin(phase)
            f = f - f.mean()
            std = f.std()
            if std > 1e-12:
                f = f / std
            else:
                f = np.zeros(n, dtype=np.float64)
            feats.append(f)

    if len(feats) == 0:
        feats.append(np.zeros(n, dtype=np.float64))

    F = np.column_stack(feats).astype(np.float64, copy=False)
    return F


def theta_to_a(F, theta, eps=_TANH_EPS):
    """Positive sequence from bounded tanh deformation of uniform."""
    theta = np.clip(np.asarray(theta, dtype=np.float64), -_THETA_BOUND, _THETA_BOUND)
    z = F.dot(theta)
    z = np.where(np.isfinite(z), z, 0.0)
    p = np.tanh(z)
    b = 1.0 + eps * p
    b = np.maximum(b, 1e-12)
    a = b / b.sum()
    return a


def theta_val_only(theta, F, T, eps=_TANH_EPS):
    """Value for L-BFGS-B."""
    a = theta_to_a(F, theta, eps)
    n = a.size
    c = fft_convolve(a, a)
    u = n * c
    return surrogate_value(u, T)


def theta_grad_only(theta, F, T, eps=_TANH_EPS):
    """Gradient for L-BFGS-B."""
    theta = np.clip(np.asarray(theta, dtype=np.float64), -_THETA_BOUND, _THETA_BOUND)
    a = theta_to_a(F, theta, eps)
    n = a.size

    c = fft_convolve(a, a)
    u = n * c

    umax = float(u.max())
    if (not np.isfinite(umax)) or T <= 1e-9:
        q = np.zeros_like(u)
        if u.size:
            q[int(np.argmax(u))] = 1.0
    else:
        sh = (u - umax) / T
        e = np.exp(sh)
        s = e.sum()
        if (not np.isfinite(s)) or s <= 0.0:
            q = np.zeros_like(u)
            q[int(np.argmax(u))] = 1.0
        else:
            q = e / s

    H = cross_gradient(a, q)
    grad_a = 2.0 * n * H

    z = F.dot(theta)
    z = np.where(np.isfinite(z), z, 0.0)
    p = np.tanh(z)
    b = 1.0 + eps * p
    b = np.maximum(b, 1e-12)
    B = b.sum()
    h = 1.0 - p * p

    term1 = eps * (F.T.dot(grad_a * h))
    term2 = eps * (F.T.dot(h)) * float(np.dot(grad_a, a))
    grad = (term1 - term2) / B
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def theta_both(F, theta, T, eps=_TANH_EPS):
    """Value and gradient in one path for custom optimizer."""
    theta = np.clip(np.asarray(theta, dtype=np.float64), -_THETA_BOUND, _THETA_BOUND)
    a = theta_to_a(F, theta, eps)
    n = a.size

    c = fft_convolve(a, a)
    u = n * c
    val = surrogate_value(u, T)

    umax = float(u.max())
    if (not np.isfinite(umax)) or T <= 1e-9:
        q = np.zeros_like(u)
        if u.size:
            q[int(np.argmax(u))] = 1.0
    else:
        sh = (u - umax) / T
        e = np.exp(sh)
        s = e.sum()
        if (not np.isfinite(s)) or s <= 0.0:
            q = np.zeros_like(u)
            q[int(np.argmax(u))] = 1.0
        else:
            q = e / s

    H = cross_gradient(a, q)
    grad_a = 2.0 * n * H

    z = F.dot(theta)
    z = np.where(np.isfinite(z), z, 0.0)
    p = np.tanh(z)
    b = 1.0 + eps * p
    b = np.maximum(b, 1e-12)
    B = b.sum()
    h = 1.0 - p * p

    term1 = eps * (F.T.dot(grad_a * h))
    term2 = eps * (F.T.dot(h)) * float(np.dot(grad_a, a))
    grad = (term1 - term2) / B
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return val, grad


def a_to_theta(F, a, eps=_TANH_EPS):
    """Approximate inverse of theta_to_a by least squares on arctanh shape."""
    a = normalize_pos(a)
    mean_a = float(a.mean())
    if mean_a <= 1e-300:
        return np.zeros(F.shape[1], dtype=np.float64)
    b = a / mean_a
    p = (b - 1.0) / eps
    p = np.clip(p, -0.999999999, 0.999999999)
    z = np.arctanh(p)
    try:
        theta = np.linalg.lstsq(F, z, rcond=None)[0]
    except Exception:
        theta = np.zeros(F.shape[1], dtype=np.float64)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    return np.clip(theta, -_THETA_BOUND, _THETA_BOUND)


def custom_theta_optimize(F, theta0, T, budget_s, seed):
    """Fallback Adam optimizer for the low-dimensional Fourier/tanh family."""
    rng = np.random.default_rng(seed)
    theta = np.clip(np.asarray(theta0, dtype=np.float64), -_THETA_BOUND, _THETA_BOUND).copy()
    d = theta.size
    m = np.zeros(d, dtype=np.float64)
    v = np.zeros(d, dtype=np.float64)
    lr = 0.08
    best_val = np.inf
    best_theta = theta.copy()
    deadline = time.monotonic() + max(0.0, float(budget_s))
    it = 0

    while time.monotonic() < deadline:
        val, grad = theta_both(F, theta, T)
        if np.isfinite(val) and val < best_val:
            best_val = val
            best_theta = theta.copy()

        if not np.all(np.isfinite(grad)):
            grad = np.zeros(d, dtype=np.float64)

        it += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        theta = np.clip(theta - lr * m / (np.sqrt(v) + 1e-8), -_THETA_BOUND, _THETA_BOUND)
        if it % 100 == 0:
            lr *= 0.985

    return best_theta


def run_theta_task(F, theta0, T, maxfun, budget_s):
    """Run one bounded L-BFGS-B/CMA-less low-dimensional task."""
    theta0 = np.clip(np.asarray(theta0, dtype=np.float64), -_THETA_BOUND, _THETA_BOUND)

    if _MINIMIZE is None:
        return custom_theta_optimize(F, theta0, T, min(30.0, float(budget_s)), 12345)

    try:
        bounds = [(-_THETA_BOUND, _THETA_BOUND)] * theta0.size
        res = _MINIMIZE(
            theta_val_only,
            theta0,
            args=(F, T),
            jac=theta_grad_only,
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": max(10, min(300, maxfun)),
                "maxfun": max(10, min(700, 2 * maxfun)),
                "ftol": 1e-9,
                "gtol": 1e-6,
                "maxls": 20,
                "disp": False,
            },
        )
        x = np.asarray(res.x, dtype=np.float64)
        if np.all(np.isfinite(x)):
            return np.clip(x, -_THETA_BOUND, _THETA_BOUND)
        return theta0
    except Exception:
        return custom_theta_optimize(F, theta0, T, min(30.0, float(budget_s)), 54321)


def full_mirror_optimize(a0, budget_s, seed, T_start=0.35, T_end=0.02, initial_score=None):
    """Full-vector natural-gradient/mirror descent on the simplex."""
    a = normalize_pos(a0)
    n = a.size
    if n == 0:
        return a, np.inf

    deadline = time.monotonic() + max(0.0, float(budget_s))
    best = a.copy()
    if initial_score is None or not np.isfinite(initial_score):
        best_score = exact_score_list(best)
    else:
        best_score = float(initial_score)

    if not np.isfinite(best_score):
        best_score = fast_score(best)

    rng = np.random.default_rng(seed)
    T = float(T_start)
    last_soft = None
    stuck = 0
    it = 0
    exact_interval = 5.0 if n <= 5000 else 10.0
    last_exact = 0.0

    while time.monotonic() < deadline:
        c = fft_convolve(a, a)
        u = n * c
        umax = float(u.max())
        fast = 2.0 * umax

        if fast < best_score - 1e-7 and time.monotonic() - last_exact > exact_interval:
            sc = exact_score_list(a)
            last_exact = time.monotonic()
            if np.isfinite(sc) and sc < best_score:
                best = a.copy()
                best_score = sc

        last_soft = surrogate_value(u, T)

        q = softmax_weights(u, T)
        H = cross_gradient(a, q)
        G = 2.0 * n * H
        Gmax = float(G.max())

        accepted = False
        if Gmax > 1e-300:
            for lam in (1.0, 0.5, 0.25, 0.125):
                a_new = a * np.exp(-lam * G / Gmax)
                if not np.all(np.isfinite(a_new)):
                    break
                a_new = normalize_pos(a_new)
                c_new = fft_convolve(a_new, a_new)
                u_new = n * c_new
                soft_new = surrogate_value(u_new, T)
                if soft_new < last_soft - 1e-12:
                    a = a_new
                    last_soft = soft_new
                    accepted = True
                    stuck = 0
                    break

        if accepted:
            T = max(float(T_end), T * 0.9995)
        else:
            stuck += 1
            if stuck == 1:
                T = min(0.6, T * 1.2)
            elif stuck % 60 == 0:
                noise = np.exp(rng.normal(0.0, 0.08, n))
                a = normalize_pos(a * noise)
                c = fft_convolve(a, a)
                u = n * c
                last_soft = surrogate_value(u, T)

        it += 1
        if it % 3000 == 0:
            print(f"  full mirror iter {it}, T={T:.4f}, fast={fast:.6f}, best={best_score:.6f}")

    sc = exact_score_list(best)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return best, best_score


def block_local_search(a0, budget_s, seed):
    """Fast pre-filtered exact local search with random block rescaling."""
    a = normalize_pos(a0)
    n = a.size
    if n == 0:
        return a, np.inf

    deadline = time.monotonic() + max(0.0, float(budget_s))
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_score = exact_score_list(best)
    if not np.isfinite(best_score):
        best_score = fast_score(best)
    best_fast = fast_score(best)
    last_exact = 0.0

    while time.monotonic() < deadline:
        cand = best.copy()
        wlo = max(1, n // 800)
        whi = max(2, n // 25)
        width = int(rng.integers(wlo, whi + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.16)))

        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        cand = normalize_pos(cand)
        f = fast_score(cand)

        if f < best_fast - 1e-7 and time.monotonic() - last_exact > 0.15:
            sc = exact_score_list(cand)
            last_exact = time.monotonic()
            if np.isfinite(sc) and sc < best_score:
                best = cand.copy()
                best_score = sc
                best_fast = f
                print(f"  block exact improve {best_score:.6f}")

    sc = exact_score_list(best)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimizer:
    1. cheap smooth sine candidates,
    2. low-dimensional tanh/Fourier L-BFGS-B optimization,
    3. full-vector mirror descent / natural gradient refinement,
    4. optional fast-local block search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(90.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(seed)

    # Choose length from the supplied construction if reasonable.
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        raw = np.ones(2048, dtype=np.float64)

    raw = np.where(np.isfinite(raw), raw, 0.0)
    if 512 <= raw.size <= 8192:
        n = raw.size
        init = normalize_pos(raw)
    else:
        n = 2048
        if raw.size >= n:
            init = normalize_pos(raw[:n])
        else:
            init = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = init.copy()
    best_score = exact_score_list(best_a)
    if not np.isfinite(best_score):
        best_score = fast_score(best_a)
    print(f"initial length={n}, best_score={best_score:.6f}")

    # Very cheap smooth starting candidates.
    if n > 100:
        js = np.arange(n, dtype=np.float64)
        ph = 2.0 * np.pi * js / n
        for amp in (0.10, 0.18, 0.27, 0.36, 0.48, 0.62):
            cand = 1.0 + amp * np.sin(ph)
            cand = np.maximum(cand, 0.01)
            cand = normalize_pos(cand)
            f = fast_score(cand)
            if f < best_score - 1e-7:
                sc = exact_score_list(cand)
                if np.isfinite(sc) and sc < best_score:
                    best_a = cand.copy()
                    best_score = sc
                    print(f"sine candidate improved score={best_score:.6f}")

    # Build smooth features.
    fourier_modes = 10 if n <= 6000 else 8
    poly_deg = 5
    F = build_features(n, fourier_modes=fourier_modes, poly_deg=poly_deg)
    d = F.shape[1]

    # Low-dimensional theta optimization.
    theta_best = np.zeros(d, dtype=np.float64)

    starts = [theta_best.copy()]
    for sigma in (0.2, 0.5, 0.9):
        starts.append(rng.normal(0.0, sigma, d))

    det = np.zeros(d, dtype=np.float64)
    if d > 0:
        det[0] = 1.0
        starts.append(det)
    det = np.zeros(d, dtype=np.float64)
    if d > 0:
        det[0] = -1.0
        starts.append(det)
    if d > 6:
        det = np.zeros(d, dtype=np.float64)
        det[6] = 1.0
        starts.append(det)

    T_list = (0.5, 0.25, 0.12, 0.06, 0.03)
    task_starts = starts[:4]

    for T in T_list:
        if time.monotonic() > deadline - 240.0:
            break

        chosen = []
        for th in task_starts:
            if len(chosen) >= 2:
                break
            th = np.clip(np.asarray(th, dtype=np.float64), -_THETA_BOUND, _THETA_BOUND)
            duplicate = False
            for cth in chosen:
                if float(np.abs(th - cth).sum()) < 1e-7:
                    duplicate = True
                    break
            if not duplicate:
                chosen.append(th)

        maxfun = 80 if n > 6000 else 150

        for th0 in chosen:
            if time.monotonic() > deadline - 240.0:
                break
            th_res = run_theta_task(F, th0, T, maxfun, min(120.0, max(1.0, deadline - time.monotonic())))
            a = theta_to_a(F, th_res)
            sc = exact_score_list(a)
            if np.isfinite(sc) and sc < best_score:
                best_a = a.copy()
                best_score = sc
                print(f"theta T={T:.3f} exact improved score={best_score:.6f}")

            a_best = theta_to_a(F, theta_best)
            if fast_score(a) < fast_score(a_best):
                theta_best = np.clip(th_res, -_THETA_BOUND, _THETA_BOUND)

        task_starts = [theta_best.copy(), rng.normal(0.0, 0.45, d)] + starts[:1]

    # Full-vector refinement starts.
    x = np.arange(n, dtype=np.float64)
    if n > 1:
        u = 2.0 * x / (n - 1) - 1.0
    else:
        u = np.zeros(n, dtype=np.float64)

    full_starts = []
    full_starts.append(best_a.copy())

    ramp = 1.0 + 0.85 * np.clip(u, -1.0, 1.0)
    full_starts.append(normalize_pos(ramp))

    sine = 1.0 + 0.38 * np.sin(2.0 * np.pi * x / n)
    full_starts.append(normalize_pos(np.maximum(sine, 0.05)))

    pert = best_a * np.exp(rng.normal(0.0, 0.10, n))
    full_starts.append(normalize_pos(pert))

    noisy_init = init * np.exp(rng.normal(0.0, 0.15, n))
    full_starts.append(normalize_pos(noisy_init))

    num_full = 5
    for i in range(num_full):
        if time.monotonic() > deadline - 60.0:
            break
        remaining_starts = max(1, num_full - i)
        slice_s = max(30.0, (deadline - time.monotonic() - 60.0) / float(remaining_starts))
        slice_s = min(slice_s, 360.0)

        start_a = full_starts[i % len(full_starts)]
        a, sc = full_mirror_optimize(start_a, slice_s, seed=1000 + i, T_start=0.32, T_end=0.018)
        if np.isfinite(sc) and sc < best_score:
            best_a = a.copy()
            best_score = sc
            print(f"full run {i} improved score={best_score:.6f}")

        pert = best_a * np.exp(rng.normal(0.0, 0.09, n))
        full_starts.append(normalize_pos(pert))

    # Optional local exact polish.
    if n <= 6500 and time.monotonic() < deadline - 75.0:
        block_budget = min(120.0, deadline - time.monotonic() - 35.0)
        if block_budget > 20.0:
            a, sc = block_local_search(best_a, block_budget, seed=987654)
            if np.isfinite(sc) and sc < best_score:
                best_a = a.copy()
                best_score = sc
                print(f"block search improved score={best_score:.6f}")

    final_a = normalize_pos(best_a)
    final_a = np.clip(final_a, 0.0, 1000.0)

    # Final exact check if time clearly allows.
    if time.monotonic() < deadline - 20.0:
        sc = exact_score_list(final_a)
        if np.isfinite(sc):
            best_score = sc

    return [float(v) for v in final_a]
```
