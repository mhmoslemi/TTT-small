   lr=base_lr; tau_factor=4.0; no_imp=0; step=0
   inv? no.
   while time.monotonic() < deadline:
      fa=np.fft.rfft(a, fft_size)
      c=np.fft.irfft(fa*fa, fft_size)[:2*n-1]
      maxc=float(c.max())
      if not np.isfinite(maxc) or maxc <= 1e-300: break

"""
Search for nonnegative step-function coefficients minimizing
evaluate_sequence using:
1) a pool of structured initializations,
2) a smoothed log-sum-exp projected gradient method,
3) exact one-dimensional coordinate-descent polish,
4) occasional random multiplicative perturbations.
"""

import time
import numpy as np


docstring_summary = "FFT-accelerated smoothed gradient + exact 1D coordinate descent."


def next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def fft_convolve(a):
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    size = next_pow2(L)
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)
    return out[:L]


def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    a = np.clip(a, 0.0, 1000.0)
    if not np.all(np.isfinite(a)):
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return np.ones(a.size, dtype=np.float64)
    return a / s


def fast_score(a, c=None):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")
    if c is None:
        a = normalize_nonnegative(a)
        c = fft_convolve(a)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float("inf")
    return float(2.0 * n * mx)


def make_power(n, p, alpha, kind):
    idx = np.arange(n, dtype=np.float64)
    if kind == 3:
        v = (idx + alpha + 1e-8) ** (-p)
        v = v + 0.5 * (n - 1.0 - idx + alpha + 1e-8) ** (-p)
        return normalize_nonnegative(v)
    if kind == 0:
        base = idx
    elif kind == 1:
        base = n - 1.0 - idx
    elif kind == 2:
        base = np.minimum(idx, n - 1.0 - idx)
    else:
        base = idx
    v = (base + alpha + 1e-8) ** (-p)
    return normalize_nonnegative(v)


def candidate_key(item):
    return item[0]


def perturb_block(a, rng, sigma):
    n = int(a.size)
    a2 = a.copy()
    for _ in range(2):
        max_w = max(2, min(n, n // 8))
        width = int(rng.integers(1, max_w + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        a2[start:start + width] *= float(np.exp(rng.normal(0.0, sigma)))
    a2 = np.clip(a2, 0.0, 1000.0)
    a2 = np.maximum(a2, 1e-14 / max(1, n))
    s = float(np.sum(a2))
    if s <= 1e-12:
        return a
    return a2 / s


def random_perturb(a, rng, sigma):
    n = int(a.size)
    if rng.random() < 0.5:
        a2 = a * np.exp(rng.normal(0.0, sigma, n))
        a2 = np.clip(a2, 0.0, 1000.0)
        a2 = np.maximum(a2, 1e-14 / max(1, n))
        s = float(np.sum(a2))
        if s <= 1e-12:
            return a
        return a2 / s
    return perturb_block(a, rng, sigma)


def build_initial_pool(deadline, seed):
    rng = np.random.default_rng(seed)
    candidates = []

    parent = globals().get("height_sequence_1", None)
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=np.float64)
            if p.size >= 128 and p.size <= 20000:
                p = np.clip(p, 0.0, 1000.0)
                if np.all(np.isfinite(p)):
                    s = float(np.sum(p))
                    if s > 1e-12:
                        p = p / s
                        c = fft_convolve(p)
                        sc = 2.0 * p.size * float(c.max())
                        candidates.append((sc, p.size, p.copy()))
        except Exception:
            pass

    n_list = [1024, 2048, 4096, 6144, 8192]

    for n in n_list:
        if time.monotonic() > deadline:
            break

        if n <= 2048:
            ps = [0.45, 0.50, 0.55, 0.60]
            alphas = [0.1, 0.5, 2.0, 10.0]
            kinds = (0, 1, 2, 3)
        else:
            ps = [0.48, 0.50, 0.52, 0.55]
            alphas = [0.1, 0.5, 2.0, 10.0]
            kinds = (0, 1, 2, 3)

        for p in ps:
            if time.monotonic() > deadline:
                break
            for alpha in alphas:
                if time.monotonic() > deadline:
                    break
                for kind in kinds:
                    if time.monotonic() > deadline:
                        break
                    a = make_power(n, p, alpha, kind)
                    c = fft_convolve(a)
                    sc = 2.0 * n * float(c.max())
                    if np.isfinite(sc):
                        candidates.append((sc, n, a.copy()))

        for _ in range(2):
            if time.monotonic() > deadline:
                break
            a = rng.exponential(1.0, size=n)
            a = normalize_nonnegative(a)
            c = fft_convolve(a)
            sc = 2.0 * n * float(c.max())
            if np.isfinite(sc):
                candidates.append((sc, n, a.copy()))

    if not candidates:
        a = make_power(2048, 0.5, 0.1, 0)
        c = fft_convolve(a)
        sc = 2.0 * 2048 * float(c.max())
        candidates.append((sc, 2048, a.copy()))

    candidates.sort(key=candidate_key)
    return candidates


def active_eval(y, C0, U, D, S):
    if y < 0.0:
        y = 0.0
    if y > 1.0:
        y = 1.0
    q = (1.0 - y) / S
    vals = (q * q) * C0 + (2.0 * y * q) * U + (y * y) * D
    return float(np.max(vals))


def active_search(K, a, c, n, L, i, old, S, grid_points, local_iter, target):
    K = np.asarray(K, dtype=int)
    K = K[(K >= 0) & (K < L)]
    if K.size == 0:
        return old, float(c.max())

    if K.size > 128:
        cK = c[K]
        ordidx = np.argsort(cK)[-128:]
        K = K[ordidx]

    diag_val = 2 * i
    D = (K == diag_val).astype(np.float64)
    idx = K - i
    valid = (idx >= 0) & (idx < n)
    U = np.zeros(K.size, dtype=np.float64)
    U[valid] = a[idx[valid]]
    dmask = K == diag_val
    if np.any(dmask):
        U[dmask] = 0.0

    C0 = c[K] - (2.0 * old) * U - (old * old) * D

    eps = 1e-10
    ys = np.linspace(0.0, 1.0 - eps, grid_points)
    oldc = old
    if oldc < 0.0:
        oldc = 0.0
    if oldc > 1.0 - eps:
        oldc = 1.0 - eps
    if not np.any(np.isclose(ys, oldc, rtol=0.0, atol=1e-8)):
        ys = np.append(ys, oldc)
    ys = np.unique(ys)

    qg = (1.0 - ys) / S
    vals = ((qg * qg)[:, None] * C0[None, :]
            + (2.0 * ys * qg)[:, None] * U[None, :]
            + (ys * ys)[:, None] * D[None, :])
    obj = vals.max(axis=1)

    pos = int(np.argmin(obj))
    best_y = float(ys[pos])
    best_f = float(obj[pos])

    if target is not None and best_f >= target - 1e-10:
        return best_y, best_f

    if ys.size < 3:
        return best_y, best_f

    if pos == 0:
        yl = 0.0
        yr = float(ys[1])
    elif pos == int(ys.size) - 1:
        yl = float(ys[-2])
        yr = 1.0 - eps
    else:
        yl = float(ys[pos - 1])
        yr = float(ys[pos + 1])

    if not (yr > yl):
        yl = 0.0
        yr = 1.0 - eps

    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = yl + (1.0 - invphi) * (yr - yl)
    x2 = yl + invphi * (yr - yl)
    f1 = active_eval(x1, C0, U, D, S)
    f2 = active_eval(x2, C0, U, D, S)

    for _ in range(local_iter):
        if f1 < f2:
            yr = x2
            x2 = x1
            f2 = f1
            x1 = yl + (1.0 - invphi) * (yr - yl)
            f1 = active_eval(x1, C0, U, D, S)
        else:
            yl = x1
            x1 = x2
            f1 = f2
            x2 = yl + invphi * (yr - yl)
            f2 = active_eval(x2, C0, U, D, S)
        if (yr - yl) < 1e-8:
            break

    for yy in (x1, x2, yl, yr):
        ff = active_eval(yy, C0, U, D, S)
        if ff < best_f:
            best_f = ff
            best_y = float(yy)

    return best_y, best_f


def compute_full_new(a, c, i, n, old, S, y, viol_count):
    q = (1.0 - y) / S
    new_c = (q * q) * c

    seg = a.copy()
    seg[i] = 0.0

    base = c[i:i + n] - (2.0 * old) * seg
    base[i] -= old * old

    newslice = (q * q) * base + (2.0 * y * q) * seg
    newslice[i] += y * y

    new_c[i:i + n] = newslice

    mx = float(np.max(new_c))
    mn = float(np.min(new_c))
    if mn < -1e-10:
        new_c = np.maximum(new_c, 0.0)

    if viol_count <= 0 or new_c.size <= viol_count:
        viol = np.arange(new_c.size)
    else:
        viol = np.argpartition(new_c, -viol_count)[-viol_count:]

    return new_c, mx, viol


def lse_gradient_refine(a, deadline, seed, base_lr=0.08):
    rng = np.random.default_rng(seed)
    n = int(a.size)
    a = normalize_nonnegative(a)
    L = 2 * n - 1
    factor = 2.0 * float(n)

    c = fft_convolve(a)
    best_score = factor * float(c.max())
    best = a.copy()

    fft_size = next_pow2(L)
    corr_size = next_pow2(L + n - 1)

    lr = base_lr
    tau_factor = 4.0
    no_imp = 0
    step = 0

    while time.monotonic() < deadline:
        fa = np.fft.rfft(a, fft_size)
        c = np.fft.irfft(fa * fa, fft_size)[:L]
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 1e-300:
            break

        tau = tau_factor / maxc
        tc = tau * c
        mtc = float(tc.max())
        ex = np.exp(tc - mtc)
        ws = float(ex.sum())
        if not np.isfinite(ws) or ws <= 0.0:
            break
        w = ex / ws

        fw = np.fft.rfft(w, corr_size)
        fa2 = np.fft.rfft(a, corr_size)
        h = np.fft.irfft(fw * np.conj(fa2), corr_size)[:n]

        grad = 2.0 * h
        diag_w = w[:2 * n:2]
        if diag_w.size < n:
            tmp = np.zeros(n, dtype=np.float64)
            tmp[:diag_w.size] = diag_w
            diag_w = tmp
        grad = grad - diag_w * a
        grad = np.maximum(grad, 0.0)

        gmax = float(np.max(grad))
        if not np.isfinite(gmax) or gmax <= 1e-30:
            break

        update = np.exp(-lr * grad / gmax)
        a2 = a * update
        a2 = np.maximum(a2, 1e-14 / max(1, n))
        s2 = float(np.sum(a2))
        if not np.isfinite(s2) or s2 <= 1e-300:
            break
        a = a2 / s2
        step += 1

        if step % 5 == 0:
            fa2 = np.fft.rfft(a, fft_size)
            c2 = np.fft.irfft(fa2 * fa2, fft_size)[:L]
            sc = factor * float(c2.max())
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best = a.copy()
                best_score = sc
                no_imp = 0
            else:
                no_imp += 1

            if no_imp > 35:
                a = random_perturb(a, rng, 0.05)
                no_imp = 0
                lr = min(0.6, lr * 1.5)
                tau_factor = min(25.0, tau_factor + 2.0)
            elif no_imp > 15:
                lr = min(0.6, lr * 1.15)

            if step % 1000 == 0:
                print(f"lse step={step} best_score={best_score:.6f}")

    c = fft_convolve(best)
    sc = factor * float(c.max())
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return best, best_score


def coord_refine(a, deadline, seed, top_m=64, grid_points=33, local_iter=38, viol_count=12):
    rng = np.random.default_rng(seed)
    n = int(a.size)
    L = 2 * n - 1
    factor = 2.0 * float(n)

    a = normalize_nonnegative(a)
    c = fft_convolve(a)
    maxc = float(c.max())
    score = factor * maxc

    best_a = a.copy()
    best_score = score

    top_m = max(16, int(top_m))
    top_m = min(top_m, L)

    pass_no = 0
    no_imp = 0

    while time.monotonic() < deadline:
        pass_no += 1
        c = fft_convolve(a)
        maxc = float(c.max())
        score = factor * maxc

        if score < best_score - 1e-12:
            best_a = a.copy()
            best_score = score
            no_imp = 0
        else:
            no_imp += 1

        if L <= top_m:
            top_global = np.arange(L)
        else:
            top_global = np.argpartition(c, -top_m)[-top_m:]

        peak_idx = int(np.argmax(c))
        order = rng.permutation(n)
        pass_improved = 0
        coord_idx = 0

        for i in order:
            if time.monotonic() > deadline:
                break

            coord_idx += 1

            if coord_idx % 64 == 0:
                peak_idx = int(np.argmax(c))
            if coord_idx % 256 == 0:
                if L <= top_m:
                    top_global = np.arange(L)
                else:
                    top_global = np.argpartition(c, -top_m)[-top_m:]
            if coord_idx % 1024 == 0:
                a = normalize_nonnegative(a)
                c = fft_convolve(a)
                maxc = float(c.max())
                score = factor * maxc
                if score < best_score - 1e-12:
                    best_a = a.copy()
                    best_score = score
                peak_idx = int(np.argmax(c))
                if L <= top_m:
                    top_global = np.arange(L)
                else:
                    top_global = np.argpartition(c, -top_m)[-top_m:]

            old = float(a[i])
            if old < 1e-14:
                continue

            S = 1.0 - old
            if S <= 1e-13:
                continue

            extras = np.array([2 * i, peak_idx, i, i + n - 1, 0, L - 1], dtype=int)
            extras = extras[(extras >= 0) & (extras < L)]
            K = np.unique(np.concatenate((top_global, extras)))

            y, f_active = active_search(
                K, a, c, n, L, i, old, S,
                grid_points, local_iter, maxc
            )

            if f_active >= maxc - 1e-10:
                continue

            y_best = old
            maxc_best = maxc
            new_c_best = None

            for outer in range(2):
                new_c, max_new, viol = compute_full_new(
                    a, c, i, n, old, S, y, viol_count
                )

                if max_new < maxc_best - 1e-12:
                    y_best = y
                    maxc_best = max_new
                    new_c_best = new_c

                if outer == 0:
                    viol_keep = viol[(viol >= 0) & (viol < L)]
                    K = np.unique(np.concatenate((K, viol_keep)))
                    if K.size > 3 * top_m:
                        cK = c[K]
                        K = K[np.argsort(cK)[-3 * top_m:]]
                    y2, _ = active_search(
                        K, a, c, n, L, i, old, S,
                        grid_points, max(20, local_iter // 2), maxc
                    )
                    if abs(y2 - y) > 1e-7:
                        y = y2
                        continue
                    break

            if new_c_best is not None and maxc_best < maxc - 1e-12:
                q = (1.0 - y_best) / S
                a *= q
                a[i] = y_best
                c = new_c_best
                maxc = maxc_best
                pass_improved += 1

        c = fft_convolve(a)
        maxc = float(c.max())
        score = factor * maxc

        if score < best_score - 1e-9:
            best_a = a.copy()
            best_score = score
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= 3 and time.monotonic() < deadline - 5.0:
            for sigma in (0.02, 0.05, 0.10, 0.20):
                a2 = random_perturb(a, rng, sigma)
                c2 = fft_convolve(a2)
                sc2 = factor * float(c2.max())
                if np.isfinite(sc2) and sc2 < best_score - 1e-9:
                    a = a2
                    c = c2
                    best_a = a.copy()
                    best_score = sc2
                    no_imp = 0
                    break
                if np.isfinite(sc2) and sc2 < score:
                    a = a2
                    c = c2
                    score = sc2

            if no_imp >= 6:
                no_imp = 3

        if pass_no % 5 == 0 or score < best_score + 1e-12:
            print(f"coord pass={pass_no} score={score:.6f} best={best_score:.6f} improved={pass_improved}")

    c = fft_convolve(best_a)
    sc = factor * float(c.max())
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return best_a, best_score


def improve_candidate(a, limit, seed):
    now = time.monotonic()
    a = normalize_nonnegative(a)
    c = fft_convolve(a)
    sc = 2.0 * float(a.size) * float(c.max())

    if now + 5.0 < limit:
        lse_budget = min(30.0, 0.25 * (limit - now))
        lse_deadline = now + lse_budget
        if lse_deadline > now + 5.0:
            a, sc = lse_gradient_refine(a, limit - 10.0, seed + 17, base_lr=0.08)

    if now + 10.0 < limit:
        a, sc = coord_refine(a, limit - 3.0, seed + 101)

    return a, sc


def fallback_sequence():
    return [1.0 / 256.0] * 256


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point. Returns the best numerical sequence found."""
    start = time.monotonic()
    budget_s = max(60.0, float(budget_s))
    hard_deadline = start + max(30.0, budget_s - 20.0)

    rng = np.random.default_rng(int(seed) if np.isfinite(seed) else 42)

    pool_budget = min(90.0, max(20.0, 0.07 * budget_s))
    pool_deadline = min(start + pool_budget, hard_deadline - 120.0)
    if pool_deadline < start + 5.0:
        pool_deadline = start + 5.0

    pool = build_initial_pool(pool_deadline, int(seed) + 7)
    candidates = pool[:3]
    del pool

    if not candidates:
        return fallback_sequence()

    global_best = None
    global_best_score = float("inf")

    for sc, n, a in candidates:
        a = normalize_nonnegative(a)
        if not np.all(np.isfinite(a)):
            continue
        c = fft_convolve(a)
        sc = 2.0 * float(n) * float(c.max())
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best = a.copy()

    if global_best is None:
        return fallback_sequence()

    remaining = hard_deadline - time.monotonic()
    if remaining < 20.0:
        return [float(x) for x in global_best]

    fractions = [0.45, 0.35, 0.20][:len(candidates)]
    total_frac = float(sum(fractions))

    for idx, (sc, n, a) in enumerate(candidates):
        now = time.monotonic()
        if now >= hard_deadline - 20.0:
            break

        left_frac = float(sum(fractions[idx:]))
        frac = fractions[idx] / total_frac
        limit = min(remaining, max(15.0, frac * remaining))
        limit = min(hard_deadline - 10.0, now + limit)

        a = normalize_nonnegative(a)
        a2, sc2 = improve_candidate(a, limit, int(seed) + 1000 * (idx + 1))

        a2 = normalize_nonnegative(a2)
        if np.all(np.isfinite(a2)):
            c2 = fft_convolve(a2)
            sc2 = 2.0 * float(a2.size) * float(c2.max())
            if np.isfinite(sc2) and sc2 < global_best_score:
                global_best_score = sc2
                global_best = a2.copy()

        remaining = hard_deadline - time.monotonic()
        print(f"candidate {idx + 1}/{len(candidates)} done, fast_best={global_best_score:.6f}, remaining={remaining:.1f}s")

    global_best = normalize_nonnegative(global_best)

    try:
        final_exact = float(evaluate_sequence([float(x) for x in global_best]))
    except Exception:
        final_exact = float("inf")

    if (not np.isfinite(final_exact)) or final_exact > global_best_score + 0.01:
        try:
            alt = fallback_sequence()
            final_exact = float(evaluate_sequence(alt))
            if np.isfinite(final_exact):
                return alt
        except Exception:
            pass

        final_exact = float("inf")
        seq = [float(x) for x in global_best]
        try:
            ex = float(evaluate_sequence(seq))
        except Exception:
            ex = float("inf")
        if np.isfinite(ex):
            return seq

    return [float(x) for x in global_best]
