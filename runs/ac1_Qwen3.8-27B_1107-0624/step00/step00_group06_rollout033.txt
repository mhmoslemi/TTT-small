So if we want $\max_k (a*a)[k] = 1$, the only option (that achieves equality for all $k < n$) is the central binomial sequence. Any other sequence with $\max_k (a*a)[k] = 1$ must have $(a*a)[k] < 1$ for some $k < n$, which means the convolution is "sub-flat" somewhere in the first half.

```python
"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 for non-negative sequences.

Strategy:
1. Initialize with central binomial coefficients a_k = C(2k,k)/4^k, which give
   a perfectly flat convolution (=1) for k < n, achieving objective ~ pi/2.
2. After the LP-based refinement (which finds the unique flat-convolution solution
   is the central binomial sequence), use block perturbation local search to
   either reduce the peak or increase the sum, targeting the optimal trade-off.
3. The key insight: the objective 2nM/S^2 can be reduced below pi/2 by allowing
   a slightly higher peak M in exchange for a proportionally larger sum S.
"""

import numpy as np
import time


def central_binomial(n):
    """Generate a_k = C(2k,k)/4^k for k=0,...,n-1 using recurrence."""
    a = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def fast_convolve_peak(a):
    """Compute max of full convolution efficiently."""
    b = np.convolve(a, a)
    return float(np.max(b))


def score_sequence(a):
    """Compute objective 2n*max(conv)/sum^2 without calling evaluate_sequence."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    m = float(np.max(np.convolve(a, a)))
    return 2.0 * n * m / (s * s)


def make_block_perturbation(base, rng, perturbation_scale=0.15):
    """Return a perturbed copy of base with block-level multiplicative noise."""
    candidate = base.copy()
    n = candidate.size
    # Random block width
    width = int(rng.integers(max(2, n // 100), max(3, n // 8)))
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    candidate[start:stop] *= np.exp(rng.normal(0.0, perturbation_scale, size=stop - start))
    # Optionally symmetrize
    if rng.random() < 0.2:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def make_targeted_perturbation(base, conv, rng, M, perturbation_scale=0.1):
    """Perturb elements that are not at the peak of the convolution."""
    candidate = base.copy()
    n = candidate.size
    # Find which elements contribute to the peak convolution positions
    peak_idx = np.argwhere(conv == M).flatten()
    # Elements at the edges of the sequence contribute to fewer peak positions
    # Increase elements near the end (they contribute less to early peak positions)
    tail_start = n // 2
    tail_len = n - tail_start
    candidate[tail_start:] *= np.exp(rng.normal(0.0, perturbation_scale, size=tail_len))
    # Decrease elements near the beginning slightly (they contribute more to peaks)
    head_len = n // 10
    candidate[:head_len] *= np.exp(rng.normal(0.0, perturbation_scale * 0.5, size=head_len))
    return np.clip(candidate, 0.0, 1000.0)


def single_coordinate_improve(a, M, rng, n_steps=50):
    """Try to increase coordinates that have the most slack."""
    candidate = a.copy()
    n = candidate.size
    conv = np.convolve(candidate, candidate)
    # Find coordinates with most "slack" - those that can be increased the most
    # without violating the constraint conv[k] <= M * (1 + small_delta)
    delta = 0.001
    improved = False
    for _ in range(n_steps):
        # For each coordinate i, compute max increase:
        # Increasing a[i] by d increases conv[k] by d * a[k-i] for valid k
        # max d such that d * max_i a[k-i] <= M + M*delta - conv[k] for all k
        best_i = -1
        best_gain = 0
        # Check a random subset of coordinates for efficiency
        indices = rng.integers(0, n, size=min(50, n))
        for i in indices:
            # Max increase for coordinate i
            # conv[k] += d * a[k-i] for k = i, i+1, ..., i+n-1
            # Need d * a[j] <= M*(1+delta) - conv[i+j] for j = 0, ..., n-1
            k_vals = i + np.arange(n)
            valid = k_vals < (2 * n - 1)
            k_vals = k_vals[valid]
            a_vals = a[:n - (n - i) if i > 0 else n]  # a[k-i] for valid k
            # More carefully: for k = i+j, a[k-i] = a[j] for j=0..min(i, n-1-i+1)
            # Actually a[k-i] where k-i = j, so we need a[j] for j=0..n-1
            # but k = i+j must be in [0, 2n-2], so j in [max(0,-i), min(2n-2-i, n-1)]
            j_start = max(0, 0 - i)  # = 0
            j_end = min(2 * n - 2 - i, n - 1)
            if j_start > j_end:
                continue
            a_slice = a[j_start:j_end + 1]
            conv_slice = conv[i + j_start:i + j_end + 1]
            positive = a_slice > 0
            if not np.any(positive):
                # Can increase without bound (within 1000 limit)
                max_d = min(1000.0 - candidate[i], 0.1)
            else:
                slack = M * (1 + delta) - conv_slice
                ratios = slack[positive] / a_slice[positive]
                max_d = min(ratios)
            max_d = min(max_d, 1000.0 - candidate[i])
            if max_d > best_gain:
                best_gain = max_d
                best_i = i
        if best_i >= 0 and best_gain > 1e-8:
            candidate[best_i] += best_gain * 0.5  # take half the max to be safe
            improved = True
        else:
            break
    return candidate, improved


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(convolve(a,a))/(sum(a))^2.
    Uses central binomial coefficients as initialization, then applies
    multi-scale local search with block perturbations and coordinate improvements.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    # Try multiple lengths; the objective is ~ pi/2 for central binomial regardless of n
    # but optimization potential may vary
    best_score = np.inf
    best_seq = None

    lengths_to_try = [2000, 3000, 4000, 2500]

    # Sort by expected time (smaller first for quick wins)
    lengths_to_try = sorted(set(lengths_to_try))

    time_per_n = 1800.0 / len(lengths_to_try)  # rough budget per length

    for n in lengths_to_try:
        if time.monotonic() > deadline - 30:  # need 30s remaining
            break

        n_deadline = min(deadline - 10, time.monotonic() + time_per_n)

        # Initialize
        a = central_binomial(n)
        cur_score = score_sequence(a)
        if not np.isfinite(cur_score):
            continue

        # Verify with evaluate_sequence
        eval_score = float(evaluate_sequence([float(x) for x in a]))
        if np.isfinite(eval_score):
            cur_score = eval_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        print(f"  n={n}: initial score={cur_score:.6f}")

        # Phase 1: Block perturbation local search
        step = 0
        no_improve_count = 0
        pert_scale = 0.2
        while time.monotonic() < n_deadline - 30:
            step += 1
            # Adaptive perturbation scale
            if no_improve_count > 20:
                pert_scale = max(0.05, pert_scale * 0.95)
            elif no_improve_count < 5:
                pert_scale = min(0.3, pert_scale * 1.02)

            candidate = make_block_perturbation(a, rng, pert_scale)
            cand_score = score_sequence(candidate)

            if np.isfinite(cand_score) and cand_score < cur_score:
                a = candidate
                cur_score = cand_score
                no_improve_count = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    if step % 100 == 0:
                        print(f"  n={n} step={step}: new best={best_score:.6f}")
            elif step % 200 == 0 and rng.random() < 0.3:
                # Try targeted perturbation
                conv = np.convolve(a, a)
                M = float(np.max(conv))
                candidate = make_targeted_perturbation(a, conv, rng, M, 0.08)
                cand_score = score_sequence(candidate)
                if np.isfinite(cand_score) and cand_score < cur_score:
                    a = candidate
                    cur_score = cand_score
                    no_improve_count = 0
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
            else:
                no_improve_count += 1

            # Periodic coordinate improvement
            if step % 100 == 0 and time.monotonic() < n_deadline - 60:
                conv = np.convolve(a, a)
                M = float(np.max(conv))
                cand, improved = single_coordinate_improve(a, M, rng, n_steps=30)
                if improved:
                    cand_score = score_sequence(cand)
                    if np.isfinite(cand_score) and cand_score < cur_score:
                        a = cand
                        cur_score = cand_score
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = a.copy()

        print(f"  n={n} final: score={cur_score:.6f} (best overall: {best_score:.6f})")

    # Final verification with evaluate_sequence
    if best_seq is not None:
        final_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"Final verified score: {final_score:.6f}")
        if not np.isfinite(final_score):
            # Fallback: use central binomial of a safe length
            n_fb = 2000
            best_seq = central_binomial(n_fb)
            final_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"Fallback score: {final_score:.6f}")

    return [float(x) for x in best_seq]
```
