```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a)) / (sum(a))**2.

The program uses several complementary ideas:
1. Fast internal scoring with zero-padded FFT convolution.
2. Low-dimensional smooth profiles optimized by coordinate descent.
3. Full-length log-domain active-set local search.
4. Block perturbations and restarts.

The final sequence is validated with the provided exact evaluate_sequence().
"""

import time
import math
import numpy as np

G_N = 0
G_M = 1


def norm_seq(seq):
    x = np.asarray(seq, dtype=np.float64).ravel()
    x = np.clip(x, 0.0, 1000.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    x /= s
    return x


def fast_conv(seq):
    X = np.fft.rfft(seq, G_M)
    C = np.fft.irfft(X * X, G_M)
    return C[:2 * G_N - 1]


def fast_score(seq):
    s = float(np.sum(seq))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    C = fast_conv(seq)
    mb = float(np.max(C))
    if not np.isfinite(mb) or mb < 0.0:
        return np.inf
    return float(2.0 * G_N * mb / (s * s))


def make_seq_from_theta(theta, basis):
    base = basis.dot(theta)
    if not np.all(np.isfinite(base)):
        return None
    base = base - float(np.max(base))
    base = np.clip(base, -50.0, 50.0)
    seq = np.exp(base)
    s = float(np.sum(seq))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    seq /= s
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def score_theta(theta, basis):
    seq = make_seq_from_theta(theta, basis)
    if seq is None:
        return 1e12
    sc = fast_score(seq)
    return 1e12 if not np.isfinite(sc) else float(sc)


def add_candidate(pool, seq, fscore, max_size=40):
    if seq is None:
        return
    if not np.isfinite(fscore):
        return
    seq_copy = np.asarray(seq, dtype=np.float64).copy()
    if len(pool) < max_size:
        pool.append((float(fscore), seq_copy))
    else:
        worst_i = 0
        worst_v = pool[0][0]
        for i in range(1, len(pool)):
            if pool[i][0] > worst_v:
                worst_v = pool[i][0]
                worst_i = i
        if fscore < worst_v:
            pool[worst_i] = (float(fscore), seq_copy)


def update_best_exact(best_seq, best_score, seq):
    if seq is None:
        return best_seq, best_score
    try:
        sc = evaluate_sequence(seq.tolist())
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        return np.asarray(seq, dtype=np.float64).copy(), float(sc)
    return best_seq, best_score


def build_trig_basis(N, M):
    t = np.linspace(0.0, 1.0, N)
    cols = []
    for k in range(1, M + 1):
        cols.append(np.cos(np.pi * k * t))
        cols.append(np.sin(np.pi * k * t))
    if not cols:
        return np.zeros((N, 0), dtype=np.float64)
    return np.column_stack(cols)


def build_poly_basis(N, K):
    t = np.linspace(0.0, 1.0, N)
    cols = []
    power = np.ones_like(t)
    for _ in range(K):
        power = power * t
        cols.append(power.copy())
    if not cols:
        return np.zeros((N, 0), dtype=np.float64)
    return np.column_stack(cols)


def build_beta_basis(N):
    t = np.linspace(0.0, 1.0, N)
    eps = 1e-8
    a = np.log(t + eps)
    b = np.log((1.0 - t) + eps)
    return np.column_stack((a, b))


def eval_theta_i(theta, idx, val, basis):
    arr = theta.copy()
    arr[idx] = val
    return score_theta(arr, basis)


def golden_min_score(theta, idx, lo, hi, basis, max_eval=12):
    if lo == hi:
        return float(lo), score_theta(theta, basis)
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = eval_theta_i(theta, idx, x1, basis)
    f2 = eval_theta_i(theta, idx, x2, basis)
    for _ in range(max_eval):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = eval_theta_i(theta, idx, x1, basis)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = eval_theta_i(theta, idx, x2, basis)
    mid = 0.5 * (lo + hi)
    fm = eval_theta_i(theta, idx, mid, basis)
    best_val = mid
    best_f = fm
    if f1 < best_f:
        best_val = x1
        best_f = f1
    if f2 < best_f:
        best_val = x2
        best_f = f2
    return float(best_val), float(best_f)


def coordinate_optimize(theta0, basis, bounds, max_cycles=5, max_eval=12, time_limit=0.0):
    theta = np.asarray(theta0, dtype=np.float64).copy()
    cur = score_theta(theta, basis)
    if not np.isfinite(cur):
        return theta, cur
    best_theta = theta.copy()
    best_score = cur

    for _ in range(max_cycles):
        if time.monotonic() > time_limit:
            break
        improved = False
        for i in range(len(theta)):
            if time.monotonic() > time_limit:
                break
            lo, hi = bounds[i]
            if theta[i] < lo:
                theta[i] = lo
                cur = score_theta(theta, basis)
            if theta[i] > hi:
                theta[i] = hi
                cur = score_theta(theta, basis)
            nv, nf = golden_min_score(theta, i, lo, hi, bounds, max_eval)
            if nf < cur - 1e-9:
                theta[i] = nv
                cur = nf
                improved = True
                if cur < best_score:
                    best_score = cur
                    best_theta = theta.copy()
        if not improved:
            break
    return best_theta, best_score


def initial_theta_list(K, bounds, seed, n_random=None):
    if K <= 0:
        return []
    if n_random is None:
        n_random = max(8, min(40, 320 // max(1, K)))
    rng = np.random.default_rng(seed)
    lo = []
    hi = []
    for b in bounds:
        lo.append(b[0])
        hi.append(b[1])
    lo = np.asarray(lo, dtype=np.float64)
    hi = np.asarray(hi, dtype=np.float64)

    starts = []
    starts.append(np.zeros(K, dtype=np.float64))

    # Low-dimensional structured starts.
    for scale in [0.25, 0.5, 1.0, 1.5, 2.0, 3.0]:
        for sign in [1.0, -1.0]:
            th = np.zeros(K, dtype=np.float64)
            th[0] = sign * scale
            if K > 2:
                th[2] = 0.5 * sign * scale
            if K > 4:
                th[4] = 0.25 * sign * scale
            if K > 1:
                th[1] = 0.2 * sign * scale
            th = np.clip(th, lo, hi)
            starts.append(th)

    # Random smooth starts.
    for _ in range(n_random):
        th = rng.uniform(lo, hi)
        starts.append(th)

    # Scaled normal starts.
    for scale in [0.5, 1.2, 2.5]:
        for _ in range(max(5, n_random // 3)):
            th = scale * rng.standard_normal(K)
            th = np.clip(th, lo, hi)
            starts.append(th)

    if len(starts) > 70:
        starts = starts[:70]
    return starts


def optimize_basis(basis, bounds, initial, pool, time_limit, seed):
    if time.monotonic() > time_limit:
        return
    K = len(bounds)
    if K <= 3:
        max_cycles = 12
        max_eval = 18
    else:
        max_cycles = 6
        max_eval = 12

    for i, th0 in enumerate(initial):
        if time.monotonic() > time_limit:
            break
        # Give each start a bounded slot, but allow more if time remains.
        remaining = time_limit - time.monotonic()
        per_start = min(45.0, max(5.0, remaining / max(1, len(initial) - i)))
        theta, fs = coordinate_optimize(
            th0,
            basis,
            bounds,
            max_cycles=max_cycles,
            max_eval=max_eval,
            time_limit=time.monotonic() + per_start
        )
        seq = make_seq_from_theta(theta, basis)
        if seq is not None:
            if not np.isfinite(fs):
                fs = fast_score(seq)
            add_candidate(pool, seq, fs)
        if (i + 1) % 5 == 0:
            print("optimize_basis start", i + 1, "fscore", float(fs))


def compute_pressure(x, c, A):
    N = G_N
    m = min(A, c.size)
    if m <= 0:
        return None
    if m >= c.size:
        idx = np.arange(c.size)
    else:
        idx = np.argpartition(c, -m)[-m:]
    P = np.zeros(N, dtype=np.float64)
    for kk in idx:
        k = int(kk)
        a = 0 if k < N else k - N + 1
        b = N if k >= N - 1 else k + 1
        left = k - b + 1
        right = k - a + 1
        if left < 0:
            left = 0
        if right > N:
            right = N
        if left >= right:
            continue
        P[a:b] += 2.0 * x[left:right][::-1]
    pm = float(np.mean(P))
    P -= pm
    mx = float(np.max(np.abs(P)))
    if not np.isfinite(mx) or mx <= 1e-15:
        return None
    P /= mx
    return P


def gradient_step(x, c, P, alpha):
    logx = np.log(np.maximum(x, 1e-300))
    logx = logx - alpha * P
    logx = np.clip(logx, -50.0, 50.0)
    y = np.exp(logx)
    s = float(np.sum(y))
    if not np.isfinite(s) or s <= 1e-300:
        return x.copy()
    y /= s
    y = np.clip(y, 1e-12, 1000.0)
    s2 = float(np.sum(y))
    if s2 <= 0:
        return x.copy()
    y /= s2
    return y


def block_perturb(x, rng):
    n = G_N
    y = np.asarray(x, dtype=np.float64).copy()
    if n < 4:
        return y
    lo_w = max(2, n // 300)
    hi_w = max(3, n // 20)
    if hi_w <= lo_w:
        hi_w = lo_w + 1
    width = int(rng.integers(lo_w, hi_w))
    width = min(width, n)
    start = int(rng.integers(0, n - width + 1))
    factor = math.exp(float(rng.normal(0.0, 0.25)))
    y[start:start + width] *= factor
    y = np.clip(y, 1e-12, 1000.0)
    s = float(np.sum(y))
    if s <= 0:
        return x.copy()
    y /= s
    return y


def smooth_seq(x):
    kernel = np.array([0.2, 0.6, 0.2], dtype=np.float64)
    y = np.convolve(x, kernel, mode="same")
    y = np.clip(y, 1e-12, 1000.0)
    s = float(np.sum(y))
    if s <= 0:
        return x.copy()
    y /= s
    return y


def log_noise_restart(x, rng, sigma=0.08):
    logx = np.log(np.maximum(x, 1e-300))
    logx += sigma * rng.standard_normal(G_N)
    logx = np.clip(logx, -50.0, 50.0)
    y = np.exp(logx)
    s = float(np.sum(y))
    if s <= 0:
        return x.copy()
    y /= s
    y = np.clip(y, 1e-12, 1000.0)
    s2 = float(np.sum(y))
    if s2 <= 0:
        return x.copy()
    y /= s2
    return y


def local_improve(seq, time_limit, rng):
    x = norm_seq(seq)
    if x is None:
        return seq, np.inf
    best_x = x.copy()
    best_score = fast_score(x)
    stall = 0
    it = 0
    alphas = [0.05, 0.12, 0.25, 0.5, 0.9, 1.6]
    active_sizes = [32, 64, 96, 128, 192, 256]

    while time.monotonic() < time_limit:
        c = fast_conv(x)
        if not np.all(np.isfinite(c)):
            break
        A = active_sizes[it % len(active_sizes)]
        P = compute_pressure(x, c, A)
        try_best = best_score
        try_x = x

        if P is not None:
            for alpha in alphas:
                y = gradient_step(x, c, P, alpha)
                sc = fast_score(y)
                if np.isfinite(sc) and sc < try_best - 1e-12:
                    try_best = sc
                    try_x = y

        # Block perturbation.
        if it % 3 == 0:
            y = block_perturb(x, rng)
            sc = fast_score(y)
            if np.isfinite(sc) and sc < try_best - 1e-12:
                try_best = sc
                try_x = y

        # Occasional smoothing.
        if it % 25 == 5:
            y = smooth_seq(x)
            sc = fast_score(y)
            if np.isfinite(sc) and sc < try_best - 1e-12:
                try_best = sc
                try_x = y

        if np.isfinite(try_best) and try_best < best_score - 1e-12:
            x = np.asarray(try_x, dtype=np.float64).copy()
            best_x = x.copy()
            best_score = try_best
            stall = 0
        else:
            stall += 1
            if stall >= 80:
                x = log_noise_restart(x, rng, sigma=0.05 + 0.1 * (stall % 5) / 5.0)
                stall = 0
        it += 1

    final_score = best_score
    if not np.isfinite(final_score):
        final_score = fast_score(best_x)
    return best_x, final_score

        basis_poly = build_poly_basis(N, 8)
        bounds = [(-4.0, 4.0)] * basis_poly.shape[1]
        init = initial_theta_list(basis_poly.shape[1], bounds, seed + 21, n_random=24)
        optimize_basis(basis_poly, bounds, init, pool, time_limit, seed + 21)
        print("after poly8, pool best fast", min((z[0] for z in pool), default=np.inf) if pool else np.inf)

    # Family 4: more trigonometric modes if time remains.
    local_time_limit = min(hard_deadline - 30.0, time_limit + 300.0)
    if time.monotonic() < local_time_limit - 120.0:
        basis_trig2 = build_trig_basis(N, 6)
        bounds = [(-3.5, 3.5)] * basis_trig2.shape[1]
        init = initial_theta_list(basis_trig2.shape[1], bounds, seed + 31, n_random=18)
        optimize_basis(basis_trig2, bounds, init, pool, local_time_limit - 120.0, seed + 31)
        print("after trig6, pool best fast", min((z[0] for z in pool), default=np.inf) if pool else np.inf)

    # Local full-length improvements from the best few candidates.
    rng = np.random.default_rng(seed + 99)
    local_deadline = min(hard_deadline - 25.0, local_time_limit)
    top_seqs = pick_top_pool_seqs(pool, 6)
    if not top_seqs:
        top_seqs = [uniform]

    for i, seq0 in enumerate(top_seqs):
        if time.monotonic() >= local_deadline - 30.0:
            break
        per_local = min(180.0, max(30.0, (local_deadline - time.monotonic()) / max(1.0, float(len(top_seqs) - i))))
        improved_seq, improved_score = local_improve(seq0, time.monotonic() + per_local, rng)
        add_candidate(pool, improved_seq, improved_score)
        final_q, final_s = update_best_exact(best_seq, best_score, improved_seq)
        best_seq, best_score = final_q, final_s
        print("local", i, "fast", improved_score, "best exact", best_score)

    # Very last local polish if time remains.
    if time.monotonic() < hard_deadline - 35.0:
        top_seqs = pick_top_pool_seqs(pool, 2)
        for seq0 in top_seqs:
            if time.monotonic() >= hard_deadline - 35.0:
                break
            per_local = min(90.0, hard_deadline - 35.0 - time.monotonic())
            improved_seq, improved_score = local_improve(seq0, time.monotonic() + per_local, rng)
            add_candidate(pool, improved_seq, improved_score)
            final_q, final_s = update_best_exact(best_seq, best_score, improved_seq)
            best_seq, best_score = final_q, final_s
            print("polish fast", improved_score, "best exact", best_score)

    # Validate the fast-best pool with exact evaluator.
    final_deadline = min(hard_deadline, start_time + budget - 2.0)
    best_seq, best_score = evaluate_pool_final(pool, best_seq, best_score, final_deadline)

    # Final safeguard: check starting sequences one more time.
    best_seq, best_score = update_best_exact(best_seq, best_score, base)
    best_seq, best_score = update_best_exact(best_seq, best_score, uniform)

    if best_seq is None:
        best_seq = uniform
        final_q, final_s = update_best_exact(best_seq, np.inf, best_seq)
        best_seq, best_score = final_q, final_s
    else:
        final_q, final_s = update_best_exact(best_seq, best_score, best_seq)
        if final_s < best_score:
            best_seq = final_q
            best_score = final_s

    print("final best exact", best_score)

    out = np.asarray(best_seq, dtype=np.float64).ravel()
    out = np.clip(out, 0.0, 1000.0)
    s = float(np.sum(out))
    if not np.isfinite(s) or s <= 0.01:
        out = np.ones(G_N, dtype=np.float64) / float(G_N)
    return [float(v) for v in out]
```
