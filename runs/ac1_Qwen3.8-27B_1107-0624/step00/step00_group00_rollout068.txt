def _proj_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)

def _smoothed_objective_and_grad(a, tau):
    """Compute smoothed log-sum-exp objective and its gradient w.r.t. a."""
    n = len(a)
    r = np.convolve(a, a, mode='full')
    r_max = r.max()
    exponents = tau * (r - r_max)
    exponents = np.clip(exponents, -500.0, 500.0)
    weights = np.exp(exponents)
    w_sum = weights.sum()
    obj = r_max + np.log(w_sum) / tau
    p = weights / w_sum
    L = 1
    while L < 3 * n:

    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    corr = np.fft.irfft(np.conj(A) * P, L)
    g = 2.0 * corr[:n]
    return obj, g


def _fista(a0, tau, lr, max_iter, deadline):
