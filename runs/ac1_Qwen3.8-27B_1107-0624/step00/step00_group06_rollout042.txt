    best_score = float('inf')

    # Build list of initializations
    inits = []
    
    # 1. Given sequence (normalized)
    p0 = np.clip(base, 0, 1000)
    s0 = p0.sum()
    if s0 > 0.01:
        inits.append(p0 / s0)
    
    # 2. Uniform
    inits.append(np.ones(n) / n)
    
    # 3. Random Gaussian (normalized)
    rng = np.random.default_rng(seed)
    for i in range(4):
        p_r = np.abs(rng.standard_normal(n)) + 0.1
        inits.append(p_r / p_r.sum())
    
    # 4. Two-bump (edges)
    p_edge = np.zeros(n)
    p_edge[:n // 3] = 1.0
    p_edge[2 * n // 3:] = 1.0
    inits.append(p_edge / p_edge.sum())
    
    # 5. Linear ramp
    p_ramp = np.linspace(0.1, 3.0, n)
    inits.append(p_ramp / p_ramp.sum())
    
    # 6. Cosine bump
    x = np.linspace(0, np.pi, n)
    p_cos = np.cos(x) ** 2 + 0.01
    inits.append(p_cos / p_cos.sum())
    
    # Optimization routine for one initialization
    def run_optimization(p_init, time_budget_frac, label=""):
        nonlocal best_score, best_p, deadline
        
        run_deadline = min(deadline, time.monotonic() + time_budget_frac * (budget_s - 10))
        p = np.maximum(p_init, 1e-15).copy()
        p = p / p.sum()
        
        local_best = p.copy()
        local_best_F = 2.0 * n * np.max(_fft_convolve(p, p, nfft_conv)[:conv_len])
        
        # Hyperparameters
        T_frac_start = 0.1   # initial temperature as fraction of M
        T_frac_end = 0.005   # final temperature
        eta_start = 1.5      # initial step size
        eta_min = 0.01       # minimum step size
        lp_every = 200       # do LP-based update every this many iterations
        patience = 500       # restart if no improvement for this many iters
        
        # Precompute p_rev FFT (updated each iteration)
        p_rev = p[::-1]
        fp_rev = fft(p_rev, nfft_large)
        
        start_time = time.monotonic()
        last_improve = 0
        last_improve_F = local_best_F
        iter_count = 0
        
        while time.monotonic() < run_deadline:
            iter_count += 1
            t = iter_count
            
            # Compute convolution p*p
            fp = fft(p, nfft_conv)
            conv = np.real(ifft(fp * fp))[:conv_len]
            M = conv.max()
            current_F = 2.0 * n * M
            
            # Track local best
            if current_F < local_best_F:
                local_best_F = current_F
                local_best = p.copy()
                last_improve = t
                last_improve_F = current_F
                if current_F < best_score:
                    best_score = current_F
                    best_p = p.copy()
                    if t % 200 == 0 or best_score < 1.55:
                        print(f"[{label}] iter={t}, F={best_score:.6f}")
            
            # Plateau escape: if stuck, increase temperature temporarily
            progress = t - last_improve
            if progress > patience:
                T_frac = max(T_frac_start, T_frac_end * 3)
                eta = max(eta_start * 0.5, eta_min)
                # Perturb p slightly
                p = p * (1 + 0.1 * rng.standard_normal(n))
                p = np.maximum(p, 1e-15)
                p = p / p.sum()
                last_improve = t
            else:
                # Schedule: decrease temperature and step size
                frac = min(1.0, t / 50000.0)
                T_frac = T_frac_start + (T_frac_end - T_frac_start) * frac
                eta = max(eta_start * (1.0 / np.sqrt(t + 1)), eta_min)
            
            # Temperature
            T = max(T_frac * M, 1e-20)
            
            # Soft subgradient
            w = np.exp((conv - M) / T)
            w_sum = w.sum()
            if w_sum > 0:
                w = w / w_sum
            else:
                w = np.ones(conv_len) / conv_len
            
            # Compute g via FFT cross-correlation
            fw = fft(w, nfft_large)
            # Update p_rev FFT (p may have changed)
            p_rev = p[::-1]
            fp_rev = fft(p_rev, nfft_large)
            g = 2.0 * np.real(ifft(fp_rev * fw))[n-1:2*n-1]
            
            # Multiplicative weights update
            g_max = g.max()
            if g_max > 1e-15:
                log_update = -eta * g / g_max
                # Numerical stability
                log_update -= log_update.max()
                p = p * np.exp(log_update)
            else:
                p = p.copy()
            
            p = np.maximum(p, 1e-15)
            p_sum = p.sum()
            if p_sum > 0:
                p = p / p_sum
            else:
                p = np.ones(n) / n
            
            # Periodic LP-based refinement
            if t % lp_every == 0 and t > 100:
                p_lp, F_lp = _lp_based_update(p, n, nfft_conv)
                if F_lp < current_F * 1.01:  # accept if not much worse
                    p = p_lp
                    if F_lp < local_best_F:
                        local_best_F = F_lp
                        local_best = p.copy()
                        last_improve = t
                        if F_lp < best_score:
                            best_score = F_lp
                            best_p = p.copy()
                            print(f"[{label}] LP iter={t}, F={best_score:.6f}")
        
        return local_best, local_best_F
    
    # Run optimizations with time allocation
    n_inits = len(inits)
    # Allocate more time to promising inits
    time_allocs = []
    # First init (given sequence) gets 30%
    # Uniform gets 20%
    # Rest split among remaining
    remaining = 0.50
    per_init = remaining / max(1, n_inits - 2)
    time_allocs.append(0.30)
    time_allocs.append(0.20)
    for i in range(n_inits - 2):
        time_allocs.append(per_init)
    
    # Normalize to sum to 1
    total_alloc = sum(time_allocs)
    time_allocs = [a / total_alloc for a in time_allocs]
    
    for i, (init, alloc) in enumerate(zip(inits, time_allocs)):
        if time.monotonic() > deadline:
            break
        label = f"init{i}"
        t_start = time.monotonic()
        local_best, local_F = run_optimization(init, alloc, label)
        elapsed = time.monotonic() - t_start
        print(f"  [{label}] done in {elapsed:.1f}s, local_F={local_F:.6f}, global_best={best_score:.6f}")
    
    # Final refinement: run one more optimization from the best found
    if time.monotonic() < deadline:
        remaining_time_frac = max(0, (deadline - time.monotonic()) / max(1, budget_s - 10))
        if remaining_time_frac > 0.05:
            label = "refine"
            local_best, local_F = run_optimization(best_p, remaining_time_frac * 0.9, label)
            if local_F < best_score:
                best_score = local_F
                best_p = local_best
            print(f"  [{label}] F={best_score:.6f}")
    
    # Verify with actual evaluator
    final_score = float(evaluate_sequence(best_p.tolist()))
    print(f"Final validated score: {final_score:.6f}")
    
    if not np.isfinite(final_score) or final_score > best_score:
        # Fallback
        best_p = np.ones(n) / n
        final_score = float(evaluate_sequence(best_p.tolist()))
        print(f"Fallback score: {final_score:.6f}")
    
    return [float(x) for x in best_p]
