from scipy.optimize import linprog

def compute_convolution(a):
    """Compute full convolution using FFT for speed."""
    n = len(a)
    # Use FFT for O(n log n) convolution
    size = 2 * n  # next power of 2 >= 2n-1
    size = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fa2 = fa * fa
    result = np.fft.irfft(fa2, size)[:2*n-1]
    return result

    n = len(a)

            cols.append(j)

from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix

def compute_score(a):
    """Compute the score 2*n*max(a*a)/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    max_b = np.max(b)
    return float(2 * n * max_b / (S ** 2))

def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    # Vectorized construction
    j = np.arange(n)
    j_rep = np.repeat(j, n)
    offset = np.tile(np.arange(n), n)
    k = j_rep + offset
    valid = k < 2 * n - 1
    j_rep = j_rep[valid]
    k = k[valid]
    data = a[offset[valid]]
    return csr_matrix((data, (k.astype(np.int32), j_rep.astype(np.int32))), shape=(2*n-1, n))

def lp_iteration(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)_k <= M for all k."""
    n = len(a)

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return None
