```python
import numpy as np
import time

def compute_convolution(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2*n-1]

def compute_score(a):
    """Compute the score for a non-negative sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = compute_convolution(a)
    return 2.0 * n * np.max(c) / (s * s)
```

    a = a.copy()
    a = np.clip(a, 0, 1000)
    n = len(a)
    
    # Normalize
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Precompute FFT size
    size = 1
    while size < 2*n - 1:
        size *= 2
    
    start = time.monotonic()
    lr = 0.01
    
    while time.monotonic() - start < budget:
        c = np.fft.irfft(np.fft.rfft(a, size)**2, size)[:2*n-1]
        max_c = np.max(c)
        score = 2.0 * n * max_c / 1.0  # since sum(a) = 1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find active set (peaks)
        threshold = max_c * 0.95
        active = np.where(c >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            # c[k] = sum_i a[i] * a[k-i]
            # d c[k] / d a[j] = 2 * a[k-j] if 0 <= k-j < n
            # So grad += a[k-j] for valid k-j
            lo = max(0, k - (n-1))
            hi = min(k, n-1)
            # a[k-j] for j in [lo, hi] means a[k-hi] to a[k-lo]
            # a[k-j] where j goes from lo to hi: k-j goes from k-lo to k-hi
            idx = k - np.arange(lo, hi+1)  # these are the indices into a
            grad[lo:hi+1] += a[idx]
        
        grad /= len(active)  # average over active set
        
        # Also account for the normalization constraint (sum(a) = 1)
        # Using Lagrange multiplier: subtract mean of grad
        grad -= np.mean(grad)
        
        # Gradient step with non-negativity
        a = a - lr * grad
        a = np.clip(a, 0, None)
        a = a / np.sum(a)  # re-normalize
        
        # Adaptive learning rate
        new_score = 2.0 * n * np.max(np.fft.irfft(np.fft.rfft(a, size)**2, size)[:2*n-1])
        if new_score > score * 1.01:
            lr *= 0.9
        else:
            lr = min(lr * 1.05, 0.1)
    
    return best_a

and projected gradient descent with adaptive step size.

def _next_power_of_2(x):
    """Return the next power of 2 >= x."""
    p = 1
    while p < x:
        p <<= 1
    return p

def _opt_sequence(a, n, deadline, rng):
    """Optimize a non-negative sequence to minimize the score."""
    a = np.clip(a.astype(np.float64).copy(), 0.0, None)
    s = np.sum(a)

        s = n
    a = a / s  # normalize to sum = 1

    N = _next_power_of_2(3 * n - 2)
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.fft.irfft(np.fft.rfft(a, N)**2, N)[:2*n-1])

    lr = 0.005   # learning rate

        c = np.fft.irfft(fa * fa, N)[:2*n-1]
        c_max = np.max(c)
        score = 2.0 * n * c_max
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            patience = 0

Looking at this problem, I need to minimize `2n * max(a*a) / (sum(a))^2` where `a*a` is the full linear convolution. 

Key mathematical insight: `sum(a*a) = (sum(a))^2`, and the convolution has `2n-1` elements. So the average element is `(sum(a))^2/(2n-1)`, giving a theoretical lower bound of `2n/(2n-1) ≈ 1.0001` for n=4806. The constant sequence gives exactly 2.0.

The strategy is to use **projected gradient descent** with a **log-sum-exp smoothing** of the max, and **FFT-based gradient computation** (O(n log n) per iteration). The gradient of the soft-max of the convolution can be computed as a cross-correlation, which is also FFT-accelerated.

import numpy as np
import time

"""
Minimizes 2n*max(a*a)/(sum(a))^2 via projected gradient descent with
log-sum-exp smoothing and FFT-based gradients. Tries multiple
initializations (structured, random, windowed) to avoid local minima.
"""


def _next_pow2(x):
    p = 1
    while p < x:
        p <<= 1
    return p


def _optimize_run(a_init, n, deadline, rng, alpha0=20.0, lr0=0.003):
    """Core optimization loop for one initialization."""
    a = np.clip(a_init.astype(np.float64).copy(), 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n, dtype=np.float64)
        s = n
    a = a / s

    N = _next_pow2(3 * n - 2)
    n_lin = 2 * n - 1

    best_a = a.copy()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:n_lin]
    best_score = 2.0 * n * np.max(c)

    alpha = alpha0
    lr = lr0
    momentum = 0.9
    v = np.zeros(n, dtype=np.float64)
    stagnant = 0

    it = 0
    while time.monotonic() < deadline:
        # Self-convolution
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:n_lin]
        c_max = np.max(c)
        score = 2.0 * n * c_max

        if score < best_score * (1.0 - 1e-9):
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Softmax weights on c
        exp_c = np.exp(alpha * (c - c_max))
        w = exp_c / exp_c.sum()

        # Gradient: grad[j] = 2 * sum_m a[m] * w[m+j]
        # = 2 * cross_corr(a, w)[j] via FFT
        fw = np.fft.rfft(w, N)
        R = np.fft.irfft(fa.conj() * fw, N)
        grad = 2.0 * R[:n]

        # Center for sum=1 constraint
        grad = grad - grad.mean()

        # Nesterov-like momentum
        v = momentum * v + (1.0 - momentum) * grad
        a = a - lr * v
        a = np.clip(a, 0.0, None)
        s2 = np.sum(a)
        if s2 < 1e-10:
            a = np.ones(n, dtype=np.float64) * (1.0 / n)
        else:
            a = a / s2

        it += 1

        # Adaptive parameter updates
        if it % 100 == 0:
            # Increase alpha slowly (sharper max approximation)
            alpha = min(alpha * 1.003, 300.0)

            # Adapt learning rate
            fa_check = np.fft.rfft(a, N)
            c_check = np.fft.irfft(fa_check * fa_check, N)[:n_lin]
            curr_score = 2.0 * n * np.max(c_check)

            if curr_score > best_score * 1.05:
                lr = max(lr * 0.8, 1e-5)
            elif curr_score < best_score * 0.999:
                lr = min(lr * 1.15, 0.02)

            # Perturbation to escape local minima
            if stagnant > 3000:
                a += rng.normal(0, 0.002, size=n)
                a = np.clip(a, 0.0, None)
                a = a / np.sum(a)
                v *= 0.5
                stagnant = 0
            if stagnant > 8000:
                # Bigger escape
                a += rng.normal(0, 0.01, size=n)
                a = np.clip(a, 0.0, None)
                a = a / np.sum(a)
                v = np.zeros(n, dtype=np.float64)
                lr = lr0
                alpha = alpha0
                stagnant = 0

    return best_a, best_score
