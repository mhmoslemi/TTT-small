"""
Minimize 2*n*max(conv)/sum^2 for non-negative sequences using:
1. FFT-accelerated convolution for speed
2. Gradient descent with multi-peak tracking and momentum
3. Annealed restarts to escape local minima
4. LP-based optimization for fine refinement
5. Block coordinate descent for additional improvement
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def _compute_score(f):
    """Compute the score directly (faster than calling evaluate_sequence)."""
    n = len(f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    conv = _fft_convolve(f, f)
    max_b = conv.max()
    return float(2.0 * n * max_b / (S * S))


def _gradient_step(f, k_star, b_max, S, alpha, momentum_buf, momentum_decay):
    """One gradient step with momentum. Returns updated f and momentum buffer."""
    n = len(f)
    # Compute f_{k*-j} for j = 0, ..., n-1
    # k*-j ranges from k*-n+1 to k*
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    # shifted[j] = f[k*-j] if 0 <= k*-j < n, else 0
    # j = k*-idx, so idx = k*-j, j ranges over [k*-hi, k*-lo]
    shifted = np.zeros(n)
    # idx goes from lo to hi, j = k_star - idx
    js = k_star - np.arange(lo, hi + 1)
    shifted[js] = f[lo:hi + 1]

    grad = shifted - b_max / S
    grad *= 4.0 * n / (S * S)

    # Momentum
    momentum_buf[:] = momentum_decay * momentum_buf - alpha * grad

    f_new = f + momentum_buf
    f_new = np.maximum(f_new, 0.0)
    return f_new, momentum_buf


def _lp_refine(f, M, tol=0.02):
    """LP-based refinement: find direction g that maximizes sum subject to f*g <= M."""
    from scipy.optimize import linprog
    n = len(f)
    conv_len = 2 * n - 1

    # Build constraint matrix A such that A @ g <= M * ones
    # A is (2n-1) x n, where A[k, j] = f[k-j] if 0<=k-j<n else 0
    # Use sparse for memory efficiency
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        js = k - np.arange(lo, hi + 1)
        rows.extend([k] * (hi - lo + 1))
        cols.extend(js.tolist())
        vals.extend(f[lo:hi + 1].tolist())

    A = csr_matrix((vals, (rows, cols)), shape=(conv_len, n))

    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.status == 0:
        g = res.x
        sum_g = g.sum()
        if sum_g > tol:
            # Blend f toward g (normalized)
            g_norm = g * (f.sum() / sum_g)
            return (f + g_norm) / 2.0
    return f


def _multi_peak_gradient(f, num_peaks=3):
    """Compute gradient considering multiple peak positions."""
    n = len(f)
    S = f.sum()
    conv = _fft_convolve(f, f)
    # Find top peaks
    sorted_idx = np.argsort(conv)[::-1]
    peaks = sorted_idx[:num_peaks]
    peak_vals = conv[peaks]

    # Weight peaks by their values (highest peak gets most weight)
    weights = peak_vals / peak_vals.sum()

    grad = np.zeros(n)
    for w, k in zip(weights, peaks):
        b_max = conv[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        js = k - np.arange(lo, hi + 1)
        shifted = np.zeros(n)
        shifted[js] = f[lo:hi + 1]
        grad += w * (shifted - b_max / S)

    grad *= 4.0 * n / (S * S)
    return grad

            # Blend
            f_blend = 0.7 * f + 0.3 * f_lp
            f_blend = np.maximum(f_blend, 0.0)
            score_lp = _compute_score(f_blend)
            if score_lp < best_score:
                best_score = score_lp
                best_seq = f_blend.copy()
                print(f"  LP refinement: {best_score:.6f}")
        except Exception as e:
            print(f"  LP failed: {e}")

    # === Final verification ===
    final_score = evaluate_sequence([float(x) for x in best_seq])
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"Final score: {best_score:.6f} (n={n})")
    return [float(x) for x in best_seq]
