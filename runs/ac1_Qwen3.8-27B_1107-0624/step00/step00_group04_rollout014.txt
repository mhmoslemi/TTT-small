def compute_convolution(f):
    """Compute the full self-convolution of f."""

    return 2.0 * n * np.max(conv) / (s * s)

def compute_score_fast(f):
    """Compute 2n * max(conv) / sum^2 efficiently."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * np.max(conv) / (s * s)

def lp_direction(f, M, verbose=False):
    """Solve LP to find direction that maximizes sum while keeping (f*b)_k <= M."""
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build sparse constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    row_indices = []
    col_indices = []
    values = []
    
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row_indices.append(k)
            col_indices.append(j)
            values.append(f[k - j])
    
    A = csr_matrix((values, (row_indices, col_indices)), shape=(num_constraints, n))
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(num_constraints, M)
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

    k_star = np.argmax(conv)
    M = conv[k_star]

    # ∂c_k/∂f_j = 2 * f[k-j] if 0 <= k-j < n, else 0
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
