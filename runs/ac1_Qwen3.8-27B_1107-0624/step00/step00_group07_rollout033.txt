import numpy as np
from numpy.fft import fft, ifft
import time

def fast_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = fft(a, size)
    fb = fft(b, size)
    result = ifft(fa * fb)
    return result[:n].real

def obj_value(seq):
    """Compute the objective: 2*n*max(conv)/sum^2. Returns (score, max_b, sum_a)."""
    n = len(seq)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf, 0, 0
    conv = fast_convolve(seq, seq)
    max_b = np.max(conv)

def fast_convolve(a):
    """Compute full autocorrelation using FFT. Returns real array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2*n:
        size *= 2
    fa = fft(a, size)
    result = ifft(fa * fa)
    return result[:2*n-1].real

    n = len(seq)

        return np.inf
    conv = fast_convolve(seq)
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

def projected_simplex(v):
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)

"""
Minimizes the step function evaluation score using FFT-accelerated
projected subgradient descent with simulated annealing on the probability simplex.
Exploits scale invariance of the objective. Uses multiple active-set subgradients
and restart strategies to escape local minima.
"""

import numpy as np
from numpy.fft import rfft, irfft
import time


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    result = irfft(fa * fb, size)
    return result[:n]


def _fast_autocorr(p):
    """Full linear autocorrelation (convolution with itself) via FFT."""
    n = len(p)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = rfft(p, size)
    ac = irfft(fp * fp, size)
    return ac[:2 * n - 1]


def _simplex_project(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_normalized(p):
    """Score for p with sum=1: 2n * max(autocorr)."""
    n = len(p)
    ac = _fast_autocorr(p)
    return 2.0 * n * np.max(ac)


def _subgradient_for_lag(p, k):
    """Compute subgradient of conv(p,p)[k] with respect to p."""
    n = len(p)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return np.zeros(n)
    j = np.arange(lo, hi + 1)
    sg = np.zeros(n)
    # For each j in [lo, hi]: d(conv[k])/dp[j] = 2*p[k-j] (or p[j] if 2j==k)
    sg[j] = 2.0 * p[k - j]
    if k % 2 == 0:
        mid = k // 2
        if lo <= mid <= hi:
            sg[mid] = p[mid]
    return sg


def _generate_initializations(n, rng):
    """Generate diverse initial points on the simplex."""
    inits = []
    # 1. Uniform
    inits.append(np.ones(n) / n)
    # 2. Random
    r = rng.exponential(1.0, size=n)
    inits.append(r / r.sum())
    # 3. Triangular (peak in middle)
    t = np.arange(n)
    mid = n // 2
    tri = np.minimum(t + 1, n - t)
    tri = tri / tri.sum()
    inits.append(tri)
    # 4. Cosine window
    c = np.cos(np.pi * t / n) ** 2
    c = np.maximum(c, 0)
    if c.sum() > 0:
        inits.append(c / c.sum())
    # 5. Two-bump
    b1 = np.exp(-((t - n * 0.25) / (n * 0.1)) ** 2)
    b2 = np.exp(-((t - n * 0.75) / (n * 0.1)) ** 2)
    two_bump = b1 + b2
    inits.append(two_bump / two_bump.sum())
    # 6. Random block structure
    blocks = rng.integers(3, 10)
    block_sizes = rng.integers(n // blocks - 1, n // blocks + 2, size=blocks)
    block_sizes[-1] = n - block_sizes[:-1].sum()
    block_heights = rng.uniform(0.5, 2.0, size=blocks)
    seq = np.zeros(n)
    offset = 0
    for bs, bh in zip(block_sizes, block_heights):
        if bs > 0:
            seq[offset:offset + bs] = bh
        offset += bs
    inits.append(seq / seq.sum())
    return inits


def _local_search(p, rng, deadline, verbose=True):

    n = len(p)
    best = p.copy()
    best_score = _score_normalized(best)
    if verbose:
        print(f"  Local search start: score={best_score:.8f}")

    T = 0.003
    lr = 0.5 / np.max(_fast_autocorr(best))
    no_improve = 0
    patience = 50000
    iter_count = 0
    n_better = 0

    while time.monotonic() < deadline:
        ac = _fast_autocorr(best)
        max_val = np.max(ac)

        # Find active lags (within tolerance of max)
        tol = 0.98
        active = np.where(ac >= tol * max_val)[0]
        # Limit to most active ones

            top_idx = np.argsort(ac[active])[::-1][:20]

        sg_total /= len(active)

        sg_norm = np.linalg.norm(sg_total)
        if sg_norm < 1e-15:

        sg_total /= sg_norm

def _polish(p, rng, deadline):
    """Coordinate-wise polishing: try adjusting individual elements."""
    n = len(p)
    best = p.copy()
    best_score = _score_normalized(best)
    if time.monotonic() > deadline:
        return best, best_score

    # For each coordinate, try a few values
    coords = rng.permutation(n)
    steps_tried = 0
    for idx in coords:
        if time.monotonic() > deadline:
            break
        old_val = best[idx]
        if old_val < 1e-10:
            continue
        # Try relative perturbations
        for frac in [0.5, 0.7, 1.3, 2.0, 0.1]:
            candidate = best.copy()
            candidate[idx] = old_val * frac
            # Renormalize
            s = candidate.sum()
            if s < 1e-10:
                continue
            candidate /= s
            sc = _score_normalized(candidate)
            if sc < best_score:
                best = candidate
                best_score = sc
            steps_tried += 1

    return best, best_score, steps_tried

    # Phase 1: Brief search from multiple inits (20% of time)
    phase1_deadline = time.monotonic() + 0.2 * float(budget_s)

    for n_try in candidate_ns[:2]:  # Try at most 2 sizes
        if time.monotonic() > phase1_deadline:
            break

        inits = _generate_initializations(n_try, rng)
        # Also try the provided sequence if it matches this size
        try:
            base_norm = np.array(height_sequence_1, dtype=np.float64)
            base_norm = np.clip(base_norm, 0, 1000)
            if len(base_norm) == n_try and base_norm.sum() > 0.01:
                inits.append(base_norm / base_norm.sum())
        except:
            pass

        for init in inits:
            if time.monotonic() > phase1_deadline:
                break
            p, score, iters = _local_search(init, rng, phase1_deadline, verbose=False)
            if score < overall_best_score:
                overall_best_score = score
                overall_best = p
                print(f"Phase 1: n={n_try}, score={score:.8f} (iters={iters})")

    # Phase 2: Extended local search from best (60% of time)
    phase2_deadline = time.monotonic() + 0.6 * float(budget_s)
    if overall_best is not None:
        p, score, iters = _local_search(overall_best, rng, phase2_deadline, verbose=True)
        if score < overall_best_score:
            overall_best_score = score
            overall_best = p
            print(f"Phase 2: score={score:.8f} (iters={iters})")
    else:
        # Fallback
        overall_best = np.ones(n) / n
        overall_best_score = _score_normalized(overall_best)

    # Phase 3: Polishing (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 5:
        # Multiple polish passes with randomization
        n_passes = min(5, int(remaining / 30) + 1)
        for pass_idx in range(n_passes):
            polish_deadline = time.monotonic() + remaining / (n_passes - pass_idx)
            p, score, steps = _polish(overall_best, rng, polish_deadline)
            if score < overall_best_score:
                overall_best_score = score
                overall_best = p
                print(f"Polish pass {pass_idx + 1}: score={score:.8f} (steps={steps})")
            else:
                break

    # Final: verify with evaluate_sequence and return
    result = overall_best
    # Scale so the sequence is "nice" - sum to sqrt(2n) as in the reference
    target_sum = np.sqrt(2 * n)
    result = result * target_sum
    result = np.clip(result, 0, 1000)

    final_score = evaluate_sequence([float(x) for x in result])
    print(f"Final: n={n}, score={final_score:.8f}")

    return [float(x) for x in result]
