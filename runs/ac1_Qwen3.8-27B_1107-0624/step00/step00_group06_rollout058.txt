```python
from scipy.optimize import linprog

def lp_step(a, M):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # conv(a, g)[k] = sum_j a[j]*g[k-j] for valid j
    # This is a linear function of g.
    # The constraint matrix A has rows corresponding to each k (0 to 2n-2)
    # and columns corresponding to each j (0 to n-1).
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    
    # Build A (size (2n-1) x n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            A[k, j] = a[k-j]
    
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: A @ g <= M, g >= 0
    # In linprog form: A_ub @ g <= b_ub, bounds = (0, None)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```
