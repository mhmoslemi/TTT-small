import numpy as np
import time

def project_onto_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_score_normalized(a):

    n = len(a)
    c = np.convolve(a, a)

def lse_gradient(c, t, n):
    """Compute the LSE-approximated gradient of 2n*max(c) w.r.t. a.
    
    Uses FFT for efficiency.
    Returns gradient array of length n.

    c_max = np.max(c)
    exp_c = np.exp(t * (c - c_max))

    L = 1
    while L < 3 * n:  # need L >= n + (2n-1) - 1 = 3n-2
        L *= 2
    
    # We'll pass a separately since we need it for the gradient
    return weights, L

def _project_simplex(v):
