def build_constraint_matrix(a):
    """Build the sparse constraint matrix A for the LP.
    A @ b gives the linear convolution a * b (length 2n-1).
    A[k, j] = a[j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    data = np.repeat(a, n)
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.concatenate([np.full(n, j) for j in range(n)])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp(a, M):
    """Solve the LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None
