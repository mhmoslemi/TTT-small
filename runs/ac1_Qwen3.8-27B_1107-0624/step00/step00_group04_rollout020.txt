import time
from scipy.optimize import linprog

def solve_lp_direction(f, M, time_budget=30):
    """Solve the LP to find direction g that maximizes sum(g) subject to conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1} (all >= 0)
    # Maximize: sum(g) = -min(-sum(g))
    # Constraints: conv(f, g)[k] <= M for k = 0, ..., 2n-2
    #   conv(f, g)[k] = sum_i f[i] * g[k-i] (valid indices)
    #   For k = 0: f[0]*g[0] <= M
    #   For k = 1: f[0]*g[1] + f[1]*g[0] <= M
    #   For k = j < n: sum_{i=0}^{j} f[i]*g[j-i] <= M
    #   For k = n + j: sum_{i=j+1}^{n-1} f[i]*g[n-1-(i-j-1)] ... 
    #   Actually: conv(f,g)[k] = sum_{i: 0<=i<n, 0<=k-i<n} f[i]*g[k-i]
    
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix A such that A @ g <= M (vector of length 2n-1)
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1)
    
    # This is a (2n-1) x n matrix. For n=3809, it's 7617 x 3809 ≈ 29M entries.
    # Most are zero (it's banded). We can use sparse.
    
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n))
    for row in range(2*n-1):
        for j in range(max(0, row - n + 1), min(row, n-1) + 1):
            A[row, j] = f[j]
    A = A.tocsr()
    
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

2. Set `A(0) = 1` (DC component, which must be positive and equal to `sum a = 1`).

4. `a[n] = ifft(A)`.
5. Project to non-negativity: `a[n] = max(0, Re(a[n]))`.
6. Renormalize.
7. Update phases based on the projected `a`: `A = fft(a_projected)`, update `phi_k = angle(A(k))` for k≠0.
8. Repeat.

This is essentially the **Gerchberg-Saxton** (GS) or **error reduction** algorithm! It alternates between:
- Frequency domain: enforce flat magnitude
- Time domain: enforce non-negativity

This algorithm is known to converge (at least to a stationary point) for many problems.

```python
def gs_algorithm(n, max_iters=100000):
    """Gerchberg-Saxton-like algorithm for flat spectral magnitude with non-negativity."""
    # Target: |A(k)| proportional to 1 for all k
    # But A(0) = sum(a) = 1 (fixed)
    
    # Initialize with random phases
    rng = np.random.default_rng(42)
    L = 2 * n - 1  # Use DFT of length 2n-1 (or larger for better resolution)
    # Actually, for linear convolution, we should use DFT of length >= 2n-1.
    # Let's use L = 2n-1 or the next power of 2.
    
    # Hmm, the issue is that the linear convolution corresponds to the circular
    # convolution of the zero-padded sequences. So we zero-pad to length L >= 2n-1.
    
    # Actually, let's use L = 2*(2n-1) or the next power of 2 >= 2n-1.
    L = 1
    while L < 2 * (2 * n - 1):
        L *= 2
    
    # Target magnitude spectrum
    target_mag = np.ones(L // 2 + 1)
    target_mag[0] = 1.0  # DC: sum(a) = 1
    
    # Initialize with random phases
    phases = rng.uniform(0, 2*np.pi, L // 2 + 1)
    phases[0] = 0  # DC phase is 0 (for real non-negative a)
    
    # Make it real: A(k) = conj(A(L-k)) for real signal
    # So we need phases[k] = -phases[L-k]
    # For the half-spectrum, we set phases[1:L//2] freely, and the rest is determined.
    
    # Let's work with the full complex spectrum for clarity.
    A = target_mag * np.exp(1j * phases)
    # Enforce Hermitian symmetry: A[k] = conj(A[(L-k) % L])
    for k in range(1, L//2):
        A[L-k] = np.conj(A[k])
    A[0] = 1.0
    if L % 2 == 0:
        A[L//2] = np.abs(A[L//2])  # Nyquist is real
    
    # Iterate
    a = np.zeros(n)
    for it in range(max_iters):
        # Frequency -> Time
        a_full = np.fft.ifft(A).real[:n]  # Take first n elements (the rest should be ~0 if L >= 2n-1)
        
        # Enforce non-negativity
        a = np.maximum(a_full, 0)
        
        # Renormalize
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        
        # Time -> Frequency
        a_padded = np.zeros(L)
        a_padded[:n] = a
        A_new = np.fft.fft(a_padded)
        
        # Update: keep target magnitude, use new phase
        # A(k) = target_mag[k] * exp(j * angle(A_new(k)))
        for k in range(1, L):
            if A_new[k] != 0:
                A[k] = target_mag[k] * A_new[k] / np.abs(A_new[k])
            else:
                A[k] = target_mag[k] * np.exp(1j * rng.uniform(0, 2*np.pi))
        
        # Re-enforce Hermitian symmetry
        for k in range(1, L//2):
            A[L-k] = np.conj(A[k])
        A[0] = 1.0
        if L % 2 == 0:
            A[L//2] = np.abs(A[L//2])
    
    return a
```

We want `max_k b_k` to be minimized, where `b = ifft(FFT(a_padded)^2)` (zero-padded a, length L).

We want `b = ifft(B)` to be as flat as possible (all `b_k ≈ 1/(2n-1)`).

This means `B(k) ≈ (1/(2n-1)) * L * delta(k)` in the frequency domain... no, that's not right.

If `b` is constant (all `b_k = c` for k=0..2n-2), then in the frequency domain:
`B(k) = c * L * delta(k) = c*L` for k=0, 0 for k≠0.

But `B(k) = A(k)^2`, so `A(0)^2 = c*L` and `A(k)^2 = 0` for k≠0.
That means `A(k) = 0` for k≠0, which means `a` is a delta function! That's the opposite of what we want.

I was confusing myself. The "flat in time" corresponds to "concentrated in frequency", not the other way around.

What we actually want is: `b_k = ifft(A^2)[k]` to have small maximum. This is NOT the same as making `b` flat (which would require all `b_k` equal, which is impossible for a non-negative `a` unless `a` is uniform, giving objective = 2).

Wait, can `b` actually be flat? If `b_k = c` for all `k = 0, ..., 2n-2`, then `sum b_k = (2n-1)*c = (sum a)^2 = 1`, so `c = 1/(2n-1)`.

In the frequency domain: `B(0) = sum b_k = 1`, `B(k) = sum_k b_k exp(-2πik/(2n-1)) = c * sum_k exp(-2πik/(2n-1)) = 0` for k ≠ 0 (mod 2n-1).

So `A(k)^2 = B(k)` would give `A(k) = 0` for k ≠ 0 (in a DFT of length 2n-1). This means `a` is constant (uniform), which we already know gives objective = 2.

So the "flat b" target is only achievable by the uniform sequence, which gives the worst case (objective = 2) for non-uniform sequences!

This means my GS approach targeting flat `b` is wrong. We don't want `b` to be flat; we want `max b` to be small, which is a different (and harder) requirement.

Let me reconsider. The ideal would be `b_k = 1/(2n-1)` for all k, but as we showed, this forces `a` to be uniform. So the best we can do is make `b` "as flat as possible" subject to `a >= 0` and `sum a = 1`.

The GS approach should target the "best achievable flatness". But what's the target?

The key question is what parameters to use. Let me think about the learning rate.

The gradient `grad_j = p * sum_i a[i] * (w[i+j] + w[j-i])` where `w = b^(p-1)`.

The typical magnitude of `grad_j` is about `p * n * mean(b)^(p-1) * mean(a)`. With `sum a = 1` and `n` elements, `mean(a) ≈ 1/n`. And `mean(b) ≈ 1/(2n-1)` (since `sum b = 1`). So `grad_j ≈ p * n * (1/(2n))^(p-1) * (1/n) = p * (1/(2n))^(p-1)`.

For p=4, n=4000: `grad_j ≈ 4 * (1/8000)^3 ≈ 4 * 1.95e-12 ≈ 7.8e-12`.

That's very small! The learning rate needs to be very large to make progress. This is because the `b^p` gradient is tiny when `b` values are small (which they are, being ~1/n).

A better normalization: divide the gradient by `sum(b^p)` to get a "normalized gradient" of order 1. Or use the gradient of `log(sum(b^p))` which is `(1/sum(b^p)) * grad`.

Let me use the gradient of `log(sum(b^p))`:
`d/da_j log(sum b_k^p) = p * (sum_k b_k^(p-1) * (a_{k-j} + a_{j-k})) / sum_k b_k^p`

This is a "softmax-weighted" average of the local gradients. It should be of order 1.

The update: `a <- a - lr * normalized_grad`, where `lr` is of order `1/n` or `1/sqrt(n)`.

Let me code this up properly.

Actually, let me think about this more carefully using a different formulation.

We want to minimize `F(a) = log(sum_k b_k^p)` subject to `sum a = 1, a >= 0`.

The "unconstrained" gradient (ignoring the sum constraint) is:
`g_j = p * sum_k b_k^(p-1) * (a_{k-j} + a_{j-k}) / sum_k b_k^p`

To satisfy `sum a = 1`, we subtract the mean: `g_j <- g_j - mean(g)`.

Then: `a <- a - lr * (g_j - mean(g))`, project, renormalize.

This is like a "natural gradient" or "mirror descent" in the simplex.

For the learning rate, a safe choice is `lr = 1 / (2 * max|g_j - mean(g)|)` which ensures the update doesn't make any `a_j` negative (approximately).

Actually, let me just use `lr * max|g| < 0.5 * min_positive_a` as a rule, but that's hard to implement cleanly. Let me just use a fixed `lr` and rely on the projection + renormalization to keep things stable.

def gradient_optimize(a, rng, time_budget=60):
    """Optimize sequence a using gradient descent on L_p norm of convolution."""
    n = len(a)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1
    a = a / s  # normalize to sum=1
    
    deadline = time.monotonic() + time_budget
    best_local = np.inf
    best_local_a = a.copy()
    
    # FFT size
    L = 1
    while L < 4 * n:
        L *= 2
    
    step = 0
    while time.monotonic() < deadline:
        # Compute convolution
        b = np.convolve(a, a)
        max_b = np.max(b)
        
        if max_b < best_local:
            best_local = max_b
            best_local_a = a.copy()
        
        step += 1
        
        # Adaptive p: start moderate, increase over time
        p = min(2 + 0.001 * step, 50)
        
        # Compute gradient of log(sum(b^p))
        w = b ** (p - 1)
        w_sum_p = np.sum(b ** p)
        if w_sum_p < 1e-300:
            break
        
        # grad_j = p * (S1[j] + S2[j]) / sum(b^p)
        # S1[j] = sum_i a[i] * w[i+j]
        # S2[j] = sum_i a[i] * w[j-i]
        
        # Use FFT for S1 and S2
        A_f = np.fft.rfft(a, L)
        W_f = np.fft.rfft(w, L)
        
        # S2[j] = (a * w)[j] for j=0..n-1
        conv_aw = np.fft.irfft(A_f * W_f, L)
        S2 = conv_aw[:n]
        
        # S1[j] = (a_rev * w)[j+n-1] for j=0..n-1
        a_rev = a[::-1]
        AR_f = np.fft.rfft(a_rev, L)
        conv_arw = np.fft.irfft(AR_f * W_f, L)
        S1 = conv_arw[n-1:2*n-1]
        
        grad = p * (S1 + S2) / w_sum_p
        
        # Center the gradient (to preserve sum)
        grad_centered = grad - np.mean(grad)
        
        # Learning rate: adaptive
        lr = 0.5 / (np.max(np.abs(grad_centered)) + 1e-10)
        lr = min(lr, 1.0)  # Cap it
        
        # Update
        a = a - lr * grad_centered
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = best_local_a.copy()
        else:
            a = a / s
        
        # Periodic random perturbation to escape local minima
        if step % 5000 == 0 and rng.random() < 0.3:
            idx = rng.integers(0, n)
            a[idx] += rng.normal(0, 0.01)
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_local_a

import numpy as np
import time

def _convolve_fft(a, b):
    """Linear convolution using FFT."""
    n = len(a)
    m = len(b)
    L = 1
    while L < n + m - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n+m-1]

def _grad_logsumexp_p(a, b, p):
    """
    Compute gradient of log(sum(b_k^p)) w.r.t. a.
    b = convolve(a, a), length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    w = b ** (p - 1)  # length 2n-1
    w_sum_p = np.sum(b ** p)
    if w_sum_p < 1e-300:
        return np.zeros(n)
    
    L = 1
    while L < 4 * n:
        L *= 2
    
    A_f = np.fft.rfft(a, L)
    W_f = np.fft.rfft(w, L)
    
    # S2[j] = sum_i a[i] * w[j-i] = (a * w)[j] for j=0..n-1
    conv_aw = np.fft.irfft(A_f * W_f, L)
    S2 = conv_aw[:n]
    
    # S1[j] = sum_i a[i] * w[i+j] = (a_rev * w)[j+n-1] for j=0..n-1
    a_rev = a[::-1]
    AR_f = np.fft.rfft(a_rev, L)
    conv_arw = np.fft.irfft(AR_f * W_f, L)
    S1 = conv_arw[n-1:2*n-1]
    
    grad = p * (S1 + S2) / w_sum_p
    return grad

def _lp_step(a, rng, time_budget=30):

    Try an LP-based direction step.
    Returns the blended sequence if it improves, otherwise None.

    from scipy.optimize import linprog

    b = np.convolve(a, a)
    M = float(np.max(b))

    A = lil_matrix((2*n-1, n))

        j_hi = min(k, n - 1)
