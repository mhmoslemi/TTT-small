import numpy as np
import time
from scipy.optimize import linprog

def compute_objective(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. g>=0, conv(f,g)[k] <= M for all k."""
    n = len(f)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) → minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: conv(f, g)[k] <= M for k = 0, ..., 2n-2
    # conv(f,g)[k] = sum_i f[i] * g[k-i] (valid i: max(0,k-n+1) <= i <= min(n-1, k))
    # This is a linear constraint in g.
    
    A_ub = []
    b_ub = []
    
    for k in range(2*n - 1):
        # conv(f,g)[k] = sum_i f[i] * g[k-i]
        # For a given k, the valid pairs (i, j=k-i) with 0<=i<n and 0<=j<n:
        # i ranges from max(0, k-(n-1)) to min(n-1, k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        row = np.zeros(n)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]  # coefficient of g[j]
        A_ub.append(row)
        b_ub.append(M)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: g >= 0 (already handled by linprog's default lower bound of 0)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    deadline = time.monotonic() + budget_s - 10
    
    best = None
    best_score = np.inf
    
    # Try multiple initializations
    for n in [256, 512, 1024]:
        for init_type in ['ramp', 'random', 'two_block']:
            # Create initial sequence
            if init_type == 'ramp':
                a = np.linspace(0.1, 2.0, n)
            elif init_type == 'random':
                rng = np.random.default_rng(seed + n + hash(init_type) % 1000)
                a = rng.uniform(0.1, 1.0, n)
            elif init_type == 'two_block':
                a = np.concatenate([np.ones(n//3) * 2.0, np.ones(n - n//3) * 0.5])
            
            a = a / a.sum()  # normalize to sum=1
            
            # Run LP iterations
            t_blend = 0.1
            for it in range(100):
                if time.monotonic() > deadline:
                    break
                
                b = np.convolve(a, a)
                M = np.max(b)
                score = compute_objective(a)
                
                if score < best_score:
                    best_score = score
                    best = a.copy()
                
                # Solve LP
                g = solve_lp(a, M)
                if g is None or np.sum(g) < 1e-10:
                    break
                
                # Rescale g to sum=1
                g = g / g.sum()
                
                # Blend
                a_new = (1 - t_blend) * a + t_blend * g
                a_new = np.clip(a_new, 0, None)
                
                if np.sum(a_new) < 1e-10:
                    break
                
                a_new = a_new / a_new.sum()
                a = a_new
    
    # Final fallback
    if best is None:
        best = np.ones(256)
        best_score = 2.0
    
    return [float(x) for x in best]
