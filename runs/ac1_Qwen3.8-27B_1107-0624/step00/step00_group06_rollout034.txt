For example, if a is concentrated in the first half: a[j] > 0 for j < n/2, a[j] = 0 for j >= n/2. Then (a*a)[n-1] = sum_{j=0}^{n/2-1} a[j] a[n-1-j]. But n-1-j >= n-1-(n/2-1) = n/2, so a[n-1-j] = 0. Thus (a*a)[n-1] = 0!

But then (a*a)[m] for small m (where both a[j] and a[m-j] are in the first half) will be large. Specifically, (a*a)[0] = a[0]^2, and (a*a)[m] for m < n/2 involves products of elements in the first half.

So the tradeoff is: concentrating a on one side makes the center lag small but the edge lags large.

The optimal solution is somewhere in between - a "bump" that's wide enough to keep the edge lags small but asymmetric enough to keep the center lag small.

This is reminiscent of the "optimal window" or "optimal pulse shape" problem in signal processing.

def compute_convolution(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    L = 2 * n - 1

    L_fft = 1

        L_fft *= 2
    A = np.fft.rfft(a, L_fft)
    conv = np.fft.irfft(A * A, L_fft)[:L]

import time

def _fft_correlate(w, a):
    """Compute correlate(w, a, 'valid') using FFT. w length >= a length."""
    n_a = len(a)
    n_w = len(w)
    a_flipped = a[::-1]
    L = n_w + n_a - 1
    # Next power of 2
    L_fft = 1 << (L - 1).bit_length()
    W = np.fft.rfft(w, L_fft)
    Af = np.fft.rfft(a_flipped, L_fft)
    conv = np.fft.irfft(W * Af, L_fft)
    return conv[n_a - 1: n_a - 1 + n_a]

def _convolve(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    L = 2 * n - 1
    L_fft = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, L_fft)
    return np.fft.irfft(A * A, L_fft)[:L]

def _softmax_weights(c, beta):
    """Compute softmax(beta * c) with numerical stability."""
    c_shifted = beta * (c - np.max(c))
    exp_c = np.exp(c_shifted)
    return exp_c / np.sum(exp_c)

    """Compute F(a) = 2n * max(a*a) / (sum a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = _convolve(a)
    return 2.0 * n * np.max(conv) / (s * s)

def _gradient_step(a, beta, lr, v, momentum):
    """One gradient step with momentum."""
    c = _convolve(a)
    w = _softmax_weights(c, beta)
    grad = 2.0 * _fft_correlate(w, a)
    v = momentum * v + (1.0 - momentum) * grad
    a_new = a - lr * v
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-10:
        a_new = np.ones_like(a) / len(a)
    else:
        a_new = a_new / s
    return a_new, v
