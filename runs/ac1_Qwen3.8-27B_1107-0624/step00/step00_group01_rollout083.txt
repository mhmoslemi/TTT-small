```
def conv_self(a):

    n = a.size
    if n == 0: return np.array([], dtype=np.float64)

    fa = np.fft.rfft(a, n=size)
    c = np.fft.irfft(fa * fa, n=size)
```

```python
"""
Hybrid search for low normalized self-convolution step functions.

The program combines:
1. Asymmetric structured starts, especially antisymmetric sine perturbations,
   which break the symmetry barrier that tends to give score 2.
2. Smoothed projected gradient / multiplicative updates.
3. Approximate Frank-Wolfe linear-program directions with line search.
4. Multiple restarts and several sequence lengths.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog as _LINPROG
except Exception:
    _LINPROG = None

_INF = float('inf')


def normalize_nonnegative(a, cap=1000.0):
    if a is None:
        return None
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    arr = np.minimum(arr, float(cap))
    s = float(np.sum(arr))
    if s <= 0.0 or not np.isfinite(s):
        return None
    return arr / s


def add_tiny_width(a, eps=1e-8):
    a = normalize_nonnegative(a)
    if a is None:
        return None
    n = max(1, int(a.size))
    a = a + (float(eps) / float(n))
    return normalize_nonnegative(a)


def conv_self(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, n=size)
    c = np.fft.irfft(fa * fa, n=size)
    return c[:need].astype(np.float64, copy=False)


def score_from_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return _INF
    s = float(np.sum(a))
    if s < 0.01 or not np.isfinite(s):
        return _INF
    c = conv_self(a)
    if c.size == 0:
        return _INF
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return _INF
    if mx < 0.0:
        mx = 0.0
    return float(2.0 * n * mx / (s * s))


def downsample_sequence(a, target):
    a = np.asarray(a, dtype=np.float64).ravel()
    target = int(target)
    if a.size == 0 or target < 2:
        return None
    if a.size <= target:
        return normalize_nonnegative(a)
    blocks = np.array_split(a, target)
    out = []
    for b in blocks:
        if b.size == 0:
            out.append(0.0)
        else:
            out.append(float(np.sum(b)))
    return normalize_nonnegative(np.array(out, dtype=np.float64))


def get_valid_global():
    val = globals().get('height_sequence_1', None)
    if val is None:
        return None
    try:
        a = np.asarray(val, dtype=np.float64).ravel()
    except Exception:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    if a.size == 0:
        return None
    if a.size > 16384:
        a = downsample_sequence(a, 8192)
        if a is None:
            return None
    if a.size < 2:
        return None
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return None
    return a / s


def perturb_sequence(a, sigma, rng):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = a * np.exp(rng.normal(0.0, float(sigma), size=a.size))
    return normalize_nonnegative(a)


def antisym_perturb(a, amp=0.4):
    a = normalize_nonnegative(a)
    if a is None:
        return None
    n = int(a.size)
    x = np.linspace(0.0, 1.0, n)
    b = a + float(amp) * float(np.mean(a)) * np.sin(2.0 * np.pi * x)
    return normalize_nonnegative(np.maximum(b, 0.0))


def make_antisym_start(n, rng):
    n = int(n)
    if n < 2:
        n = 2
    x = np.linspace(0.0, 1.0, n)
    f1 = int(rng.integers(1, 4))
    f2 = int(rng.integers(f1 + 1, f1 + 4))
    a1 = float(rng.uniform(0.35, 0.85))
    a2 = float(rng.uniform(0.05, 0.35))
    a = 1.0
    a = a + a1 * np.sin(2.0 * np.pi * f1 * x)
    a = a + a2 * np.sin(2.0 * np.pi * f2 * x)
    a = np.maximum(a, 1e-12)
    return normalize_nonnegative(a)


def make_random_start(n, rng, kind=0):
    n = int(n)
    if n < 2:
        n = 2
    if kind == 0:
        base = rng.exponential(scale=1.0, size=n)
        trend = float(rng.uniform(-0.4, 0.4))
        x = np.linspace(0.0, 1.0, n)
        a = base * np.exp(trend * x) + 1e-12
    elif kind == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(rng.normal(0.0, 0.5, size=n))
        a = a * (1.0 + 0.35 * np.cos(2.0 * np.pi * float(rng.uniform(0, 3)) * x
                                      + float(rng.uniform(0.0, 2.0 * np.pi))))
        a = np.maximum(a, 1e-12)
    else:
        left = np.exp(-float(rng.uniform(0.05, 1.5)) * np.arange(n, dtype=np.float64))
        right = np.exp(-float(rng.uniform(0.05, 1.5)) * (float(n - 1) - np.arange(n, dtype=np.float64)))
        p = float(rng.uniform(0.2, 0.8))
        a = p * left + (1.0 - p) * right + 1e-12
    return normalize_nonnegative(a)


def make_binary_start(n, rng, p=0.5):
    n = int(n)
    if n < 2:
        n = 2
    p = float(p)
    a = (rng.random(n) < p).astype(np.float64)
    if float(np.sum(a)) == 0.0:
        a[int(rng.integers(n))] = 1.0
    return a


def smooth_sequence(a, width=7):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    width = int(width)
    if width <= 1 or n < max(3, width):
        return a
    if width % 2 == 0:
        width += 1
    k = np.ones(width, dtype=np.float64) / float(width)
    b = np.convolve(a, k, mode='same')
    return normalize_nonnegative(b)


def gradient_from_weights(a, w):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    L = int(w.size)
    if n <= 0 or L <= 0:
        return np.zeros(n, dtype=np.float64)
    need = n + L - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w[::-1], n=size)
    c = np.fft.irfft(fa * fw, n=size)
    idx = (L - 1) - np.arange(n, dtype=np.int64)
    return 2.0 * c[idx]


def optimize_continuous(a, rng, deadline, current_best=_INF):
    a = add_tiny_width(a, 1e-8)
    if a is None:
        return None, _INF
    n = int(a.size)
    best = a.copy()
    best_score = score_from_a(best)
    if not np.isfinite(best_score):
        return best, best_score

    lr = 0.35
    no_improve = 0
    it = 0
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        conv = conv_self(a)
        if conv.size == 0:
            break
        maxc = float(np.max(conv))
        if not np.isfinite(maxc) or maxc <= 0.0:
            break
        s = float(np.sum(a))
        if s <= 0.0 or not np.isfinite(s):
            break
        cur_score = float(2.0 * n * maxc / (s * s))

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        elif cur_score < best_score - 1e-12:
            no_improve = max(0, no_improve - 1)
        else:
            no_improve += 1

        if no_improve > 700:
            a = best.copy()
            sigma = 0.02 + 0.08 * float(rng.random())
            a = a * np.exp(rng.normal(0.0, sigma, size=n))
            cnt = max(1, n // 50)
            idx = rng.integers(0, n, size=cnt)
            a[idx] += rng.random(size=cnt) * (1.0 / float(n))
            a = add_tiny_width(a, 1e-8)
            if a is None:
                break
            lr = 0.25
            no_improve = 0
            continue

        if it % 5 == 0:
            top_num = 16
            if top_num < conv.size:
                idxs = np.argpartition(conv, -top_num)[-top_num:]
            else:
                idxs = np.arange(conv.size, dtype=np.int64)
            w = np.zeros_like(conv)
            w[idxs] = 1.0 / float(max(1, idxs.size))
        else:
            T = 12.0 / maxc
            shifted = T * (conv - maxc)
            shifted = np.minimum(shifted, 0.0)
            w = np.exp(shifted)
            ws = float(np.sum(w))
            if not np.isfinite(ws) or ws <= 0.0:
                w = np.zeros_like(conv)
                w[int(np.argmax(conv))] = 1.0
            else:
                w = w / ws

        grad = gradient_from_weights(a, w)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        grad = np.maximum(grad, 0.0)
        gmax = float(np.max(grad))
        if gmax <= 1e-300:
            break

        rel = grad / gmax
        accepted = False
        floor = 1e-15 / float(n)

        new = a * np.exp(-lr * rel)
        new = np.maximum(new, floor)
        ns = float(np.sum(new))
        if ns > 0.0 and np.isfinite(ns):
            new = new / ns
            new_score = score_from_a(new)
            if np.isfinite(new_score) and new_score < cur_score:
                a = new
                lr = min(2.0, lr * 1.07)
                accepted = True
                no_improve = 0

        if not accepted:
            for trial in (0.4, 0.15, 0.04):
                new = a * np.exp(-trial * rel)
                new = np.maximum(new, floor)
                ns = float(np.sum(new))
                if ns <= 0.0 or not np.isfinite(ns):
                    continue
                new = new / ns
                new_score = score_from_a(new)
                if np.isfinite(new_score) and new_score < cur_score:
                    a = new
                    lr = min(2.0, trial * 1.2)
                    accepted = True
                    no_improve = 0
                    break

        if not accepted:
            lr = max(0.02, lr * 0.65)
            no_improve += 1
            if lr < 0.02:
                a = best.copy()
                a *= (1.0 + 0.02 * rng.standard_normal(n))
                a = add_tiny_width(a, 1e-8)
                if a is None:
                    break
                lr = 0.3

        it += 1
        now = time.monotonic()
        if now - last_print > 5.0:
            print("continuous n=%d best=%.10f" % (n, best_score), flush=True)
            last_print = now

    return best, best_score


def build_dense_convolution_matrix(f):
    n = int(f.size)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def build_active_convolution_matrix(f, rows):
    n = int(f.size)
    A = np.zeros((rows.size, n), dtype=np.float64)
    for r in range(rows.size):
        k = int(rows[r])
        start = 0 if k < n else k - (n - 1)
        end = n - 1 if k >= n - 1 else k
        if start <= end:
            A[r, start:end + 1] = f[k - end:k - start + 1][::-1]
    return A


def lp_direction(f, margin, deadline, ub, conv_in=None, dense_limit=1024):
    if _LINPROG is None:
        return None
    f = np.asarray(f, dtype=np.float64).ravel()
    n = int(f.size)
    if n < 2:
        return None
    if conv_in is None:
        conv = conv_self(f)
    else:
        conv = np.asarray(conv_in, dtype=np.float64).ravel()
    L = 2 * n - 1
    if conv.size < L or not np.all(np.isfinite(conv)):
        conv = conv_self(f)
        L = conv.size
    maxc = float(np.max(conv))
    if maxc <= 0.0 or not np.isfinite(maxc):
        return None

    rem = max(0.2, float(deadline - time.monotonic()))
    tlimit = min(10.0, max(0.5, rem / 12.0))
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, float(ub))] * n

    if n <= dense_limit:
        A = build_dense_convolution_matrix(f)
        b = np.full(L, maxc * (1.0 + float(margin)) + 1e-15, dtype=np.float64)
    else:
        ktop = min(400, L)
        kextra = min(128, L)
        if ktop < L:
            top = np.argpartition(conv, -ktop)[-ktop:].astype(np.int64, copy=False)
        else:
            top = np.arange(L, dtype=np.int64)
        if kextra > 0:
            extra = np.linspace(0, L - 1, kextra).astype(np.int64)
        else:
            extra = np.array([], dtype=np.int64)
        special = np.array([0, n - 1, L - 1], dtype=np.int64)
        rows = np.unique(np.concatenate((top, extra, special)))
        A = build_active_convolution_matrix(f, rows)
        b = np.full(rows.size, maxc * (1.0 + float(margin)) + 1e-15, dtype=np.float64)

    b = np.where(np.isfinite(b), b, maxc * (1.0 + float(margin)) + 1e-15)

    try:
        res = _LINPROG(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method='highs',
            options={'time_limit': float(tlimit), 'presolve': True}
        )
    except Exception:
        if n <= 512:
            try:
                res = _LINPROG(
                    c,
                    A_ub=A,
                    b_ub=b,
                    bounds=bounds,
                    method='highs',
                    options={'presolve': True}
                )
            except Exception:
                return None
        else:
            return None

    if res is None:
        return None
    x = getattr(res, 'x', None)
    if x is None:
        return None
    x = np.asarray(x, dtype=np.float64).ravel()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 0.0 or not np.isfinite(s):
        return None
    return x


def optimize_lp(a, rng, deadline, current_best=_INF, dense_limit=1024):
    a = normalize_nonnegative(a)
    if a is None:
        return None, _INF
    n = int(a.size)
    if _LINPROG is None or n < 8 or n > 8192:
        return a, score_from_a(a)

    best = a.copy()
    best_score = score_from_a(best)
    if not np.isfinite(best_score):
        return best, best_score

    margin = 0.25
    t = 0.12
    no_improve = 0
    solves = 0
    if n <= 512:
        max_solves = 400
    elif n <= 1024:
        max_solves = 180
    elif n <= 2048:
        max_solves = 80
    else:
        max_solves = 30
    last_print = time.monotonic()

    while time.monotonic() < deadline and solves < max_solves:
        if no_improve > 25:
            a = best.copy()
            a = a * np.exp(rng.normal(0.0, 0.05, size=n))
            a = add_tiny_width(a, 1e-8)
            if a is None:
                break
            margin = 0.30
            t = 0.12
            no_improve = 0

        conv = conv_self(a)
        if conv.size == 0:
            break
        maxc = float(np.max(conv))
        if not np.isfinite(maxc) or maxc <= 0.0:
            break
        s = float(np.sum(a))
        if s <= 0.0 or not np.isfinite(s):
            break
        cur_score = float(2.0 * n * maxc / (s * s))
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0

        ub = max(30.0 / float(n), 2.0 * float(np.max(a)), 1e-12)
        g = lp_direction(a, margin, deadline, ub, conv_in=conv, dense_limit=dense_limit)
        solves += 1

        if g is None:
            margin = min(0.8, margin * 1.5)
            continue

        sum_g = float(np.sum(g))
        if sum_g <= s * (1.0 + 1e-6):
            margin = min(0.8, margin * 1.3)
            continue

        directions = []
        g_norm = normalize_nonnegative(g * (s / sum_g))
        if g_norm is not None:
            directions.append(g_norm)
        if n >= 32 and g_norm is not None:
            g_sm = smooth_sequence(g_norm, 7)
            if g_sm is not None:
                directions.append(g_sm)
        if g_norm is not None:
            g_mix = 0.5 * g_norm + 0.5 * np.full(n, 1.0 / float(n), dtype=np.float64)
            g_mix = normalize_nonnegative(g_mix)
            if g_mix is not None:
                directions.append(g_mix)

        tlist = [t, 0.2 * t, 0.5 * t, min(0.9, 2.0 * t), 0.05, 0.10, 0.20, 0.30, 0.50]
        best_dir_score = best_score
        best_dir_cand = None
        best_tt = t

        for gdir in directions:
            if gdir is None:
                continue
            for tt in tlist:
                tt = float(tt)
                if tt <= 0.0 or tt > 0.9:
                    continue
                cand = (1.0 - tt) * a + tt * gdir
                cand = normalize_nonnegative(cand)
                if cand is None:
                    continue
                sc = score_from_a(cand)
                if np.isfinite(sc) and sc < best_dir_score:
                    best_dir_score = sc
                    best_dir_cand = cand
                    best_tt = tt

        if best_dir_cand is not None and best_dir_score < best_score:
            a = best_dir_cand
            best = a.copy()
            best_score = best_dir_score
            t = min(0.9, max(0.05, best_tt * 1.15))
            margin = max(0.03, margin * 0.75)
            no_improve = 0
        else:
            no_improve += 1
            t = max(0.02, t * 0.60)
            margin = min(0.8, margin * 1.20)

        now = time.monotonic()
        if now - last_print > 5.0:
            print("lp n=%d solves=%d best=%.10f margin=%.3f" % (n, solves, best_score, margin), flush=True)
            last_print = now

    return best, best_score


def merge_candidate(a, score, goods, best_by_n, seq_id, best_a, best_score):
    if a is None or not np.isfinite(score):
        return seq_id, best_a, best_score
    a = normalize_nonnegative(a)
    if a is None:
        return seq_id, best_a, best_score
    n = int(a.size)
    if score < best_by_n.get(n, _INF):
        best_by_n[n] = float(score)
        goods.append((float(score), n, seq_id, a.copy()))
        seq_id += 1
        while len(goods) > 12:
            worst = 0
            for i in range(len(goods)):
                if goods[i][0] > goods[worst][0]:
                    worst = i
            goods.pop(worst)
    if score < best_score:
        best_score = float(score)
        best_a = a.copy()
    return seq_id, best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if not np.isfinite(budget) or budget < 30.0:
        budget = 1800.0

    start_time = time.monotonic()
    deadline = start_time + max(10.0, budget - 5.0)

    goods = []
    best_by_n = {}
    seq_id = 0
    best_a = None
    best_score = _INF

    global_seq = get_valid_global()
    starts = []

    if global_seq is not None:
        ga = antisym_perturb(global_seq, 0.45)
        if ga is not None:
            starts.append(('global_antisym', ga))

        gp = perturb_sequence(global_seq, 0.10, rng)
        if gp is not None:
            starts.append(('global_perturb', gp))

        gs = global_seq.copy()
        gm = float(np.mean(global_seq))
        if gm > 0.0:
            gs[0] += 0.10 * gm
            gs[min(1, gs.size - 1)] += 0.05 * gm
        gs = normalize_nonnegative(gs)
        if gs is not None:
            starts.append(('global_shift', gs))

        try:
            symmetric = bool(np.allclose(global_seq, global_seq[::-1], rtol=1e-13, atol=1e-13))
        except Exception:
            symmetric = False
        if not symmetric:
            starts.append(('global', global_seq.copy()))

    for n in (512, 1024, 2048, 4096):
        aa = make_antisym_start(n, rng)
        if aa is not None:
            starts.append(('antisym%d' % n, aa))
        rr = make_random_start(n, rng, 0)
        if rr is not None:
            starts.append(('rand%d' % n, rr))

    b1024 = make_binary_start(1024, rng, 0.5)
    if b1024 is not None:
        starts.append(('binary1024', b1024))

    if budget > 1400.0:
        a8 = make_antisym_start(8192, rng)
        if a8 is not None:
            starts.append(('antisym8192', a8))
        r8 = make_random_start(8192, rng, 2)
        if r8 is not None:
            starts.append(('rand8192', r8))

    for name, arr in starts:
        if arr is None:
            continue
        sc = score_from_a(arr)
        seq_id, best_a, best_score = merge_candidate(arr, sc, goods, best_by_n, seq_id, best_a, best_score)
        print("start %s n=%d local=%.10f" % (name, int(arr.size), sc), flush=True)

    cont_reserve = 500.0 if budget > 1400.0 else 300.0
    cont_deadline = deadline - cont_reserve

    for name, arr in starts:
        if time.monotonic() >= cont_deadline:
            break
        if arr is None:
            continue
        arrc = add_tiny_width(arr, 1e-8)
        if arrc is None:
            continue
        n = int(arrc.size)
        if name.startswith('global'):
            if 'antisym' in name:
                slice_s = 200.0
            elif 'perturb' in name:
                slice_s = 120.0
            else:
                slice_s = 80.0
        elif n <= 512:
            slice_s = 70.0
        elif n <= 1024:
            slice_s = 90.0
        elif n <= 2048:
            slice_s = 100.0
        else:
            slice_s = 120.0
        if n > 6000:
            slice_s = 220.0
        slice_deadline = min(cont_deadline, time.monotonic() + slice_s)

        a1, s1 = optimize_continuous(arrc, rng, slice_deadline, best_score)
        seq_id, best_a, best_score = merge_candidate(a1, s1, goods, best_by_n, seq_id, best_a, best_score)

        if np.isfinite(s1) and s1 <= best_score + 1e-12 and s1 < 1.75:
            if time.monotonic() < cont_deadline - 30.0:
                a2, s2 = optimize_continuous(a1, rng, min(cont_deadline, time.monotonic() + 90.0), s1)
                seq_id, best_a, best_score = merge_candidate(a2, s2, goods, best_by_n, seq_id, best_a, best_score)

    print("after continuous local=%.10f" % best_score, flush=True)

    lp_reserve = 180.0
    lp_deadline = deadline - lp_reserve
    goods.sort()

    for item in list(goods):
        if time.monotonic() >= lp_deadline:
            break
        sc, n, idx, arr = item
        if n < 8 or n > 6500:
            continue
        slice_s = min(240.0, lp_deadline - time.monotonic())
        if slice_s < 30.0:
            break
        a1, s1 = optimize_lp(arr, rng, time.monotonic() + slice_s, best_score)
        seq_id, best_a, best_score = merge_candidate(a1, s1, goods, best_by_n, seq_id, best_a, best_score)
        print("after LP n=%d local=%.10f" % (n, best_score), flush=True)
        if best_score < 1.2:
            break

    final_deadline = deadline - 5.0
    if best_a is not None and time.monotonic() < final_deadline:
        a1, s1 = optimize_continuous(best_a, rng, final_deadline, best_score)
        seq_id, best_a, best_score = merge_candidate(a1, s1, goods, best_by_n, seq_id, best_a, best_score)

    if best_a is None:
        best_a = np.ones(256, dtype=np.float64)

    best_a = normalize_nonnegative(best_a)
    if best_a is None:
        best_a = np.ones(256, dtype=np.float64)

    print("final local score=%.10f n=%d" % (score_from_a(best_a), int(best_a.size)), flush=True)
    return [float(x) for x in best_a]
```
