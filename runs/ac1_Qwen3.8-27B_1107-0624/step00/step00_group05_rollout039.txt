import numpy as np
import time

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    # Initialize
    a = np.asarray(height_sequence_1, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    S = a.sum()
    if S < 0.01:
        a = np.ones(n)
        S = n
    a = a / S  # Normalize to sum = 1
    
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    
    # FFT size
    N = 1
    while N < 3 * n:
        N *= 2
    
    # Parameters
    eps = 0.01  # Smoothing parameter
    lr = 0.01  # Learning rate
    patience = 50
    no_improve = 0
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        c = np.convolve(a, a)  # length 2n-1
        M = c.max()
        
        # Smoothing weights
        c_shifted = c - M  # For numerical stability
        weights = np.exp(c_shifted / eps)
        weights /= weights.sum()
        
        # Gradient via cross-correlation
        P = np.fft.fft(weights, N)
        A = np.fft.fft(a, N)
        cross_corr = np.fft.ifft(P * np.conj(A)).real
        grad = 2 * cross_corr[:n]
        
        # Projected gradient step
        # We want to minimize M, so move in -grad direction
        # But maintain sum(a) = 1 and a >= 0
        a_new = a - lr * (grad - grad.mean())  # Subtract mean to roughly maintain sum
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = a_new.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
            # Revert if no improvement
            a_new = a  # Keep old a
        
        a = a_new
        
        # Adapt learning rate and smoothing
        if no_improve > patience:
            lr *= 0.5
            eps = max(eps * 0.5, 1e-6)
            no_improve = 0
            # Also try a random perturbation
            if rng.random() < 0.5:
                perturbation = rng.standard_normal(n) * 0.01
                a = np.maximum(a + perturbation, 0)
                a = a / a.sum()
        
        # Periodic restart from best
        if it % 100 == 99:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Compute linear convolution of a with a using FFT."""
    A = np.fft.rfft(a, N)
    C = A * A
    c = np.fft.irfft(C, N)[:len(a)*2-1]
    return c

import numpy as np
import time

def project_simplex(v, s=1.0):
    """Project v onto the probability simplex {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, s / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def convolve_autof(a, N):
    """Compute a * a (linear convolution) using FFT. a has length n, N >= 2n-1."""
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:len(a) * 2 - 1]

def cross_corr_fft(w, a, N):
    """Compute (w cross-correlate a)[j] = sum_m w[m+j]*a[m] for j=0..len(a)-1.

    W = np.fft.fft(w, N)
