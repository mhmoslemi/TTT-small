    t0 = time.monotonic()
    a = a_init.copy()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(fft_autoconv(a))  # since sum=1
    eta_scale = 5.0
    m_top = 5
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_budget_s:
        b = fft_autoconv(a)
        max_b = float(np.max(b))
        score = 2.0 * n * max_b
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve == 1:
            if verbose and it % 10000 == 0:
                print(f"    it={it}, score={score:.6f}, eta_scale={eta_scale:.4f}")
        
        if no_improve > patience:
            eta_scale *= 0.5
            m_top = max(1, m_top // 2)
            no_improve = 0
            if eta_scale < 0.05:
                break
        
        if max_b <= 1e-300:
            break
        
        g = top_m_subgradient(a, b, n, m_top)
        eta = eta_scale / np.sqrt(float(it + 1))
        
        a = mirror_descent_update(a, g, eta)
        a = np.maximum(a, 0.0)
        a = a / np.sum(a)
        
        it += 1
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"  MD done: {it} iters in {elapsed:.1f}s, best={best_score:.6f}")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Searches for the best non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses mirror descent on the simplex with multiple initializations (ramp, given, antisymmetric).
    """
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    remaining = deadline - time.monotonic()
    
    n_primary = 6121
    
    # Build list of (initialization, name) pairs
    initializations = []
    
    # 1. Ramp (linearly increasing)
    ramp = np.arange(1, n_primary + 1, dtype=np.float64)
    ramp = ramp / np.sum(ramp)
    initializations.append((ramp, "ramp"))
    
    # 2. Inverted ramp
    inv_ramp = ramp[::-1].copy()
    initializations.append((inv_ramp, "inv_ramp"))
    
    # 3. Given sequence if available
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if given_raw.ndim == 1 and given_raw.size == n_primary and np.isfinite(given_raw).all():
            given = np.clip(given_raw, 0, 1000.0)
            if np.sum(given) > 0.01:
                given_n = given / np.sum(given)
                initializations.append((given_n, "given"))
    except Exception:
        pass
    
    # 4. Ramp with quadratic (faster growth at end)
    quad = np.arange(1, n_primary + 1, dtype=np.float64) ** 1.5
    quad = quad / np.sum(quad)
    initializations.append((quad, "quad_ramp"))
    
    # 5. Sine-modulated ramp
    idx = np.arange(n_primary, dtype=np.float64)
    sine_mod = (idx + 1) * (0.8 + 0.4 * np.sin(2 * np.pi * idx / n_primary))
    sine_mod = np.maximum(sine_mod, 1e-10)
    sine_mod = sine_mod / np.sum(sine_mod)
    initializations.append((sine_mod, "sine_mod"))
    
    best_score = np.inf
    best_seq = None
    
    # Allocate time: split among initializations, give more to earlier ones
    n_trials = len(initializations)
    # Weights: first trial gets more time
    weights = [3.0, 2.0] + [1.0] * (n_trials - 2)
    total_w = sum(weights)
    
    for trial_idx, (a_init, name) in enumerate(initializations):
        time_left = deadline - time.monotonic()
        if time_left < 10.0:
            break
        
        if trial_idx < n_trials - 1:
            t_alloc = time_left * (weights[trial_idx] / sum(weights[trial_idx:])) * 0.9
        else:
            t_alloc = time_left * 0.95
        
        t_alloc = max(t_alloc, 5.0)
        
        print(f"Trial {trial_idx} ({name}): allocating {t_alloc:.1f}s")
        
        best_a, best_s = run_mirror_descent(a_init, n_primary, t_alloc, verbose=True)
        
        # Final exact evaluation
        final_s = float(evaluate_sequence(best_a.tolist()))
        if final_s < best_score:
            best_score = final_s
            best_seq = best_a.copy()
            print(f"  Updated global best: {best_score:.6f}")
    
    # If we still have time, do a refinement from the best found
    time_left = deadline - time.monotonic()
    if time_left > 20.0 and best_seq is not None:
        print(f"Refinement phase: {time_left:.1f}s remaining")
        # Refine with smaller patience and more focused subgradient
        refined_a, refined_s = run_mirror_descent(
            best_seq, n_primary, time_left * 0.9,
            patience=2000, verbose=True
        )
        final_s = float(evaluate_sequence(refined_a.tolist()))
        if final_s < best_score:
            best_score = final_s
            best_seq = refined_a
    
    # Final sanity check
    if best_seq is None:
        best_seq = np.ones(n_primary) / n_primary
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if final_score < best_score:
        best_score = final_score
    
    print(f"\nFINAL: score = {best_score:.6f}, length = {len(best_seq)}")
    return [float(x) for x in best_seq]
