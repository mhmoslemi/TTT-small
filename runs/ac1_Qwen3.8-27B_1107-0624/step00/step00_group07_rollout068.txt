            a_best = a_new.copy()
            s_best = s_new
            a = a_new.copy()
            stalled = 0
            lr = min(lr * 1.003, 0.5)
        else:
            stalled += 1
            if stalled % 5 == 0:
                lr *= 0.94
            if stalled >= restart_thresh:
                # Perturb the best solution
                a = a_best.copy()
                scale = max(0.02, s_best / (4.0 * n))
                noise = rng.standard_normal(n) * scale
                a = np.maximum(a + noise, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a /= s
                else:
                    a = a_best.copy()
                lr = lr_init
                stalled = 0
                restart_thresh = max(40, restart_thresh // 2)

        it += 1
        if it % 10000 == 0:
            rem = t_end - time.monotonic()
            print(f"    [{label}] iter={it}, score={s_best:.6f}, lr={lr:.5f}, rem={rem:.0f}s")

    return a_best, s_best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase subgradient search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phase 1: Gradient descent from initial sequence.
    Phase 2: Multi-start with structured and random initializations.
    Phase 3: Intensive refinement of the best solution found.
    Uses fast direct convolution for evaluation; calls evaluate_sequence
    only for verification. All sequences normalized to sum=1, non-negative.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Setup
    n = len(height_sequence_1)
    a_raw = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
    if a_raw.sum() < 1e-10:
        a_raw = np.ones(n)
    a_init = a_raw / a_raw.sum()

    best = a_init.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"n={n}, initial score={best_score:.6f}, budget={budget_s}s")

    # Phase 1: Gradient search from initial (500s)
    print("Phase 1: Gradient search from initial...")
    t1 = time.monotonic() + 500
    a1, s1 = local_search(a_init, 0.2, t1, n, rng, top_k=20, label="P1")
    if s1 < best_score:
        best_score = s1
        best = a1.copy()
        verify = float(evaluate_sequence(best.tolist()))
        best_score = verify
        print(f"  Phase 1 result: {best_score:.6f}")

    # Phase 2: Multi-start (600s)
    print("Phase 2: Multi-start exploration...")
    t2 = time.monotonic() + 600

    # Build diverse initializations
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / max(1, n - 1) - 1.0  # normalized position in [-1, 1]
    inits = []

    # Noisy versions of best
    for nl in [0.03, 0.08, 0.15, 0.3]:
        noisy = best + rng.standard_normal(n) * nl * np.mean(best)
        noisy = np.maximum(noisy, 1e-8)
        inits.append(noisy / noisy.sum())

    # Reversed best
    rev = best[::-1].copy()
    inits.append(rev / rev.sum())

    # Cosine window
    cw = 0.5 * (1.0 + np.cos(np.pi * x))
    cw = np.maximum(cw, 0.005)
    inits.append(cw / cw.sum())

    # Raised cosine (broader)
    rc = 0.5 * (1.0 + np.cos(0.5 * np.pi * x))
    rc = np.maximum(rc, 0.005)
    inits.append(rc / rc.sum())

    # Central Gaussian
    gc = np.exp(-3.0 * x * x)
    inits.append(gc / gc.sum())

    # Two-hump
    for cpos in [-0.4, -0.3, 0.0, 0.3, 0.4]:
        w = rng.uniform(0.08, 0.2)
        h = 1.0 + 0.7 * np.exp(-((x - cpos) ** 2) / (2.0 * w * w))
        h = np.maximum(h, 0.005)
        inits.append(h / h.sum())

    # Sinusoidal perturbations of uniform
    for freq in [2, 3, 5, 7, 11]:
        amp = rng.uniform(0.1, 0.35)
        ph = rng.uniform(0, 2 * np.pi)
        sig = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + ph)
        sig = np.maximum(sig, 0.005)
        inits.append(sig / sig.sum())

    # Multi-frequency random smooth
    for _ in range(4):
        nf = rng.integers(2, 7)
        sig = np.ones(n)
        for _ in range(nf):
            f = rng.uniform(1, 20)
            p = rng.uniform(0, 2 * np.pi)
            a_m = rng.uniform(0.05, 0.3)
            sig += a_m * np.cos(2.0 * np.pi * f * x + p)
        sig = np.maximum(sig, 0.005)
        inits.append(sig / sig.sum())

    # Pure uniform (escape if stuck in bad basin)
    inits.append(np.ones(n) / n)

    # Shuffle for diversity
    rng.shuffle(inits)

    # Time allocation
    n_inits = len(inits)
    time_per = max(3.0, (t2 - time.monotonic()) / max(1, n_inits))

    for idx, init in enumerate(inits):
        rem_total = t2 - time.monotonic()
        if rem_total < 5:
            break
        t_alloc = min(time_per + 5.0, rem_total - 2.0)
        if t_alloc < 3:
            break
        lr_try = rng.uniform(0.04, 0.15)
        tk = rng.integers(10, 40)
        a_try, s_try = local_search(init, lr_try, time.monotonic() + t_alloc,
                                     n, rng, top_k=tk, label=f"P2-{idx}")
        if s_try < best_score:
            best_score = s_try
            best = a_try.copy()
            verify = float(evaluate_sequence(best.tolist()))
            best_score = verify
            print(f"  Init {idx} improved: {best_score:.6f}")

    # Phase 3: Final intensive search
    remaining = deadline - time.monotonic()
    print(f"Phase 3: Final refinement, best={best_score:.6f}, remaining={remaining:.0f}s")
    if remaining > 15:
        # Fine-tune from best with small lr
        a3, s3 = local_search(best, 0.02, deadline, n, rng, top_k=30, label="P3")
        if s3 < best_score:
            best_score = s3
            best = a3.copy()
            verify = float(evaluate_sequence(best.tolist()))
            best_score = verify
            print(f"  Phase 3: {best_score:.6f}")

    # One more pass with moderate perturbation if time remains
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_pert = best + rng.standard_normal(n) * 0.05 * np.mean(best)
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert /= a_pert.sum()
        a4, s4 = local_search(a_pert, 0.05, deadline, n, rng, top_k=25, label="P4")
        if s4 < best_score:
            best_score = s4
            best = a4.copy()
            verify = float(evaluate_sequence(best.tolist()))
            best_score = verify
            print(f"  Phase 4: {best_score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"FINAL SCORE: {final_score:.6f}")
    return [float(x) for x in best]
