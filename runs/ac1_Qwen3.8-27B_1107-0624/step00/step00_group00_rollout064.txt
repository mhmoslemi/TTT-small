Approach: fix sum(a)=1, then minimize 2*n*max_k (a*a)[k].  The objective is

gradient descent with an adaptive temperature schedule.  The gradient of the LSE
objective,  grad[j] = 4*n * sum_k w[k] * a[k-j]  with
w[k] = exp(t*c[k]) / sum(exp(t*c[k])), is computed in O(n log n) via FFT as
grad = 4*n * conv(a[::-1], w)[n-1 : n-1+n].  We project onto the simplex each

def _project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _fft_autocorr(a, L):
    """Full linear self-convolution of a via FFT (valid part length 2n-1)."""
    A = np.fft.rfft(a, L)
    out = np.fft.irfft(A * A, L)
    return out[: 2 * len(a) - 1]


def _next_pow2(m):
    L = 1
    while L < m:
        L *= 2
    return L
