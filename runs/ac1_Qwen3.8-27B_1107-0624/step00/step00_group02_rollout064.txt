"""
LP-based Frank-Wolfe optimization for minimizing 2*n*max(conv(b,b))/sum(b)^2.
Uses scipy HiGHS solver for the LP subproblem, with adaptive step sizes and
multiple initialization strategies. Falls back to vertex-based Frank-Wolfe
if LP solve fails or is too slow.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len
from scipy import sparse
from scipy.optimize import linprog


def compute_autocorr(b):
    """Compute linear autocorrelation of b using FFT."""
    n = len(b)
    size = next_fast_len(2 * n - 1)
    B = fft(b, size)
    c = np.real(ifft(B * B, size))[:2 * n - 1]
    return c


def build_conv_matrix(b):
    """Build sparse convolution matrix A where A[k,m] = b[k-m] if 0<=k-m<n."""
    n = len(b)
    # Column m has nonzeros at rows m:m+n (clipped), with values b[0:n]
    rows = np.repeat(np.arange(n), n)
    for m in range(n):
        start = m
        end = min(2 * n - 1, m + n)
        length = end - start
        if length < n:
            # Adjust: column m has nonzeros at rows m to m+n-1, but clipped to 2n-2
            pass
    # More efficient: build column by column
    row_parts = []
    col_parts = []
    val_parts = []
    for m in range(n):
        k_start = m
        k_end = min(2 * n - 1, m + n)
        length = k_end - k_start
        if length <= 0:
            continue
        row_parts.append(np.arange(k_start, k_end))
        col_parts.append(np.full(length, m))
        val_parts.append(b[:length])
    rows_all = np.concatenate(row_parts)
    cols_all = np.concatenate(col_parts)
    vals_all = np.concatenate(val_parts)
    A = sparse.csc_matrix((vals_all, (rows_all, cols_all)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(b, M):
    """Solve LP: max sum(g) s.t. A@b*g <= M, g >= 0. Returns g or None."""
    n = len(b)
    A = build_conv_matrix(b)
    # scipy linprog: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2 * n - 1, M)
    # Add non-negativity via bounds
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None:
        g = np.maximum(0, res.x)
        if np.sum(g) > 0:
            return g
    return None


def fw_vertex_step(b):
    """Frank-Wolfe vertex step: find best single coordinate to move mass to."""
    n = len(b)
    c = compute_autocorr(b)
    max_c = c.max()
    # Subgradient: g[m] = (2/|S|) * sum_{k in S} b[k-m]
    # Active set: positions where c is close to max
    S = np.where(c >= max_c - 1e-12)[0]
    if len(S) == 0:
        S = np.array([np.argmax(c)])
    # Compute h[m] = sum_{k in S} b[k-m] for valid k-m
    h = np.zeros(n)
    for k in S:
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        h[m_min:m_max + 1] += b[k - np.arange(m_min, m_max + 1)]
    g = h / len(S)
    # Move mass to coordinate with minimum subgradient
    j_star = np.argmin(g)
    return j_star, max_c


def blend_sequences(b, g, t):
    """Blend b toward g with step t, maintain non-negativity and unit sum."""
    b_new = (1.0 - t) * b + t * g
    b_new = np.maximum(0, b_new)
    s = np.sum(b_new)
    if s > 0:
        b_new /= s
    else:
        b_new = np.ones(len(b)) / len(b)
    return b_new


def evaluate_and_clamp(b):
    """Evaluate objective directly (unit sum assumed)."""
    n = len(b)
    c = compute_autocorr(b)
    return 2.0 * n * c.max()
