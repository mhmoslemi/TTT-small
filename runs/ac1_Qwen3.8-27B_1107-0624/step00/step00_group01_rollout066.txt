"""
Minimize 2n*max(autocorr)/(sum)^2 for non-negative step functions.
Strategy: Projected subgradient descent with FFT-based convolution,
momentum, adaptive learning rates, and multiple diverse initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fft_convolve(a, b=None):
    """Full linear convolution using FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def proj_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_obj_conv(a):
    """Return (objective, full_convolution, max_val) for nonneg a."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-12:
        return np.inf, np.zeros(2 * n - 1), 0.0
    c = fft_convolve(a)
    M = np.max(c)
    obj = 2.0 * n * M / (S * S)
    return obj, c, M


def sgd_step(a, c, M, n, lr, momentum_buf):
    """One projected subgradient step with momentum. Returns (new_a, new_buf)."""
    tol = 1e-10 * max(M, 1e-15)
    K_star = np.where(c >= M - tol)[0]
    g = np.zeros(n)
    for k in K_star:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:

    g -= np.mean(g)
    momentum_buf = 0.9 * momentum_buf + 0.1 * g
    a_new = proj_simplex(a - lr * momentum_buf)
