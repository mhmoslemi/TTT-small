```python
import numpy as np
import time

def frank_wolfe_step(f, t_iter):
    """One Frank-Wolfe step."""
    n = len(f)
    # Compute convolution
    conv = np.convolve(f, f, mode='full')  # length 2n-1
    max_val = np.max(conv)
    # Find all indices achieving the max
    max_indices = np.where(np.isclose(conv, max_val, atol=1e-12))[0]
    # Compute subgradient: average of gradients over max-achieving k
    # Gradient of (f*f)[k] w.r.t. f[m] = 2*f[k-m] if 0 <= k-m < n, else 0
    subgrad = np.zeros(n)
    for k in max_indices:
        # f[k-m] for m = 0,...,n-1 where 0 <= k-m < n
        # i.e., m ranges from max(0, k-n+1) to min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        subgrad[lo:hi+1] += 2 * f[k - hi : k - lo : -1]  # f[k-(hi)], f[k-(hi-1)], ..., f[k-lo]
    subgrad /= len(max_indices)
    # Frank-Wolfe: find v that minimizes subgrad^T v over simplex
    # This is just v = e_{argmin(subgrad)}
    v_idx = np.argmin(subgrad)
    # Step size
    step = 2.0 / (t_iter + 2)
    # Update
    new_f = (1 - step) * f + step * np.zeros(n)
    new_f[v_idx] = step
    return new_f, max_val
```

```python
import time

def fast_autoconvolve(a):
    """Compute full auto-convolution of a using FFT."""
    n_a = len(a)
    n = 2 * n_a - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:n]

def simplex_project(y):
    """Project y onto the simplex {x >= 0, sum x = 1}."""
    n = len(y)
    if np.all(y <= 0):
        return np.ones(n) / n
    u = np.sort(y)[::-1]
    cv = np.cumsum(u) - 1
    positive = u - cv / np.arange(1, n + 1) > 0
    if not np.any(positive):
        return np.ones(n) / n
    rho = np.max(np.where(positive)[0])
    theta = cv[rho] / (rho + 1)
    return np.maximum(y - theta, 0)

def compute_subgradient(f, conv, tol=1e-10):
    """Compute subgradient of max(conv) w.r.t. f."""
    n = len(f)
    max_val = np.max(conv)
    max_indices = np.where(conv >= max_val - tol)[0]
    subgrad = np.zeros(n)
    for k in max_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        subgrad[lo:hi + 1] += 2.0 * f[hi:lo - 1:-1]
    subgrad /= len(max_indices)
    return subgrad
```
