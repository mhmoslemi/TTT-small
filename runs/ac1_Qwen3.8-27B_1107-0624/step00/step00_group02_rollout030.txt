def compute_convolution_fft(a):
    """Compute convolution of a with itself using FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)
    return b[:2*n-1]

def compute_gradient(w, a):
