import numpy as np
import time


"""
Minimize 2*n*max(a*a)/sum(a)^2 over non-negative step functions a of fixed length.

Approach: mirror descent (multiplicative-weight update) on the simplex
{a >= 0, sum(a) = 1}, since the objective is scale-invariant. The "peak"
term max(a*a) is replaced by a smoothing (log-sum-exp) to obtain a smooth
gradient; the gradient is a softmax-weighted cross-correlation of a with the
convolution a*a, computed in O(n log n) via FFT. Several structured initial
shapes are tried, step size is annealed, and periodic relative perturbations
escape plateaus. If a good global starting sequence is available it is also
polished. The best-scoring candidate found is returned as a list of floats.
"""


def _next_pow2(x):
    return 1 << (int(x) - 1).bit_length() if int(x) > 1 else 2


def _fast_convolve(a):
    """Linear convolution a*a computed with a power-of-two FFT."""
    n = a.size
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _compute_grad(a, beta):
    """
    Gradient of  (1/beta) * log( sum_k exp(beta * b[k]) )  where b = a*a.
    For large beta this approaches the subgradient of max(a*a).
    Returns (grad, max_b).
    """
    n = a.size
    b = _fast_convolve(a)
    max_b = float(np.max(b))

    # numerically stable softmax weights on b
    logits = beta * (b - max_b)          # all <= 0
    logits = np.maximum(logits, -50.0)
    exp_b = np.exp(logits)
    total = float(exp_b.sum())
    if total < 1e-300:
        weights = np.zeros_like(b)
        weights[int(np.argmax(b))] = 1.0
    else:
        weights = exp_b / total

    # grad[i] = 2 * sum_k weights[k] * a[k-i]   (cross-correlation a ⋆ weights)
    corr_size = n + (2 * n - 1) - 1       # = 3n - 2
    fft_size = _next_pow2(corr_size)
    A_f = np.fft.rfft(a, fft_size)
    W_f = np.fft.rfft(weights, fft_size)
    corr = np.fft.irfft(A_f * np.conj(W_f), fft_size)[:n]

    grad = 2.0 * corr
    return grad, max_b


def _mirror_step(a, grad, eta):
    """One multiplicative-weight (mirror-descent) step, stays on the simplex."""
    log_factor = -eta * grad
    log_factor = np.clip(log_factor, -100.0, 100.0)
    a_new = a * np.exp(log_factor)
    s = float(a_new.sum())
    if s < 1e-300:
        return a
    a_new /= s
    return a_new


def _score(a):
    """Objective assuming sum(a) == 1: 2*n*max(a*a)."""
    n = a.size
    return float(2.0 * n * np.max(_fast_convolve(a)))


def _run_mirror(a, beta, eta0, max_steps, deadline, best_box):
    """Run the mirror-descent loop in place, tracking best via best_box dict."""
    n = a.size
    local_best = a.copy()
    local_score = _score(a)
    eta = eta0
    stagnation = 0
    rng = np.random.default_rng(12345 + n)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        grad, max_b = _compute_grad(a, beta)
        cur = float(2.0 * n * max_b)

        if cur < local_score - 1e-12:
            local_score = cur
            local_best = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if cur < best_box["score"]:
            best_box["score"] = cur
            best_box["arr"] = a.copy()

        a = _mirror_step(a, grad, eta)

        if step % 200 == 199:
            eta *= 0.9
            if eta < 1e-4:
                eta = 1e-4

        # do not let mass collapse to zero at any coordinate

        if amin < 1e-12:
            a += 1e-12
            a /= a.sum()

        # periodic relative perturbation to leave plateaus

            a = a * (1.0 + rel)
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            stagnation = 0

    return local_best, local_score
