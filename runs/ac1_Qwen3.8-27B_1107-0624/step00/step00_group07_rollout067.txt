"""
Minimize  2*n*max(conv(a,a)) / (sum(a))^2  over non-negative sequences a.

Insight: the objective is scale invariant, so we work in the log-domain
x = log(a) (which guarantees positivity and is scale invariant).  A
projected / mean-removed sub-gradient descent on x that *fills* the
top autocorrelation peaks drives max(conv(a,a)) down from the uniform
value 2.0 toward ~1.5.  Each peak tap k contributes g_j += a_j*a[k-j];
averaging g over several near-max taps produces a water-filling step.
Convolutions are computed with a zero-padded FFT (O(n log n)) so we can
use long sequences (thousands of taps) and many iterations within budget.

We seed from several deterministic families (two-block, ramp, sine,
random) and run multiple restarts, always keeping the best sequence
confirmed by the authoritative evaluate_sequence().
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Fast helpers
# ----------------------------------------------------------------------

def _fft_conv(a):
    """Linear (full) convolution a*a of length 2n-1 via zero-padded FFT."""
    n = a.size
    size = 1
    need = 2 * n - 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    S = A * A
    c = np.fft.irfft(S, size)
    return c[:need]


def fast_obj(a):
    """Fast proxy for the official objective (same formula)."""
    n = a.size
    s = a.sum()
    if s <= 0:
        return float("inf")
    c = _fft_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def make_seed(n, kind, rng):
    """Deterministic / random non-negative seed of length n."""
    a = np.ones(n, dtype=np.float64)
    if kind == "two_block":
        m = n // 2
        a[n - m:] = 2.0
    elif kind == "three_block":
        m1 = n // 3
        m2 = n // 3
        a[:m1] = 1.0
        a[m1:m1 + m2] = 3.0
        a[m1 + m2:] = 1.5
    elif kind == "ramp":
        a = 1.0 + np.linspace(0.0, 1.0, n)
    elif kind == "sine":
        f = rng.integers(1, max(2, n // 400))
        phi = rng.uniform(0.0, 2.0 * math.pi)
        a = 1.0 + 0.8 * np.sin(2.0 * math.pi * f * np.arange(n) / n + phi)
        a = np.abs(a) + 0.2
    elif kind == "steps":
        k = int(rng.integers(2, 7))
        a = 1.0 + rng.uniform(0.2, 1.2, size=k)
        idx = np.linspace(0, n, k + 1).astype(int)
        for j in range(k):
            a[idx[j]:idx[j + 1]] = 1.0 + 0.5 * j + rng.uniform(-0.1, 0.1)
    else:  # random
        a = 1.0 + rng.uniform(0.0, 1.0, size=n)
    a = np.clip(a, 1e-6, 1000.0)
    return a


def polish(a0, t_end, max_iters, rng, active_frac=0.05, max_active=300,
           eta0=0.7, noise_every=60, noise_scale=0.02):
    """Log-domain peak-filling sub-gradient descent.  Returns (best_a, best_val)."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.clip(a, 1e-6, 1000.0)
    n = a.size
    x = np.log(a)
    best = a.copy()
    best_val = fast_obj(a)
    lo = math.log(1e-6)
    hi = math.log(1000.0)
    it = 0
    while it < max_iters and time.monotonic() < t_end:
        c = _fft_conv(a)
        mx = c.max()
        if not np.isfinite(mx) or mx <= 0:
            break
        thr = mx * (1.0 - active_frac)
        act = np.where(c >= thr)[0]
        if act.size > max_active:
            act = act[np.argsort(-c[act])[:max_active]]
        g = np.zeros(n, dtype=np.float64)
        for k in act:
            j = np.arange(n)
            pj = k - j
            valid = (pj >= 0) & (pj < n)
            if not valid.any():
                continue
            jv = j[valid]
            pv = pj[valid]
            g[jv] += a[jv] * a[pv]
        g = g - g.mean()            # remove the scale mode (objective is scale invariant)
        scale = np.linalg.norm(g)
        if scale < 1e-14:
            eta = 0.0
        else:
            eta = eta0 / scale
        x = x - eta * g
        # occasional gentle noise to escape local minima
        if it % max(1, noise_every) == max(1, noise_every) - 1:
            x = x + noise_scale * rng.standard_normal(n)
        x = np.clip(x, lo, hi)
        a = np.exp(x)
        it += 1
        val = fast_obj(a)
        if np.isfinite(val) and val < best_val:
            best_val = val
            best = a.copy()
    return best, best_val


def propose_candidate(seed=1234, budget_s=1800, **kwargs):
    """Multi-restart log-domain peak-filling search; returns best list[float]."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # Authoritative sanity-checked best (fallback valid sequence).
    fallback = make_seed(2048, "two_block", rng)
    best = fallback.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print("initial score = %.6f (n=%d)" % (best_score, best.size), flush=True)

    # Lengths to try; FFT makes long sequences cheap.
    n_list = [1024, 2048, 3072, 4096, 6144, 8192]
    seed_kinds = ["two_block", "three_block", "ramp", "sine", "steps", "random"]

    attempt = 0
    # Assign a per-restart wall-clock slice out of the remaining budget.
    while time.monotonic() < deadline - 5.0:
        attempt += 1
        n = int(n_list[attempt % len(n_list)])
        kind = seed_kinds[rng.integers(0, len(seed_kinds))]
        a0 = make_seed(n, kind, rng)
        # carve out this restart's slice, leaving room for final checks
        remaining = deadline - time.monotonic()
        slice_end = time.monotonic() + max(8.0, min(remaining - 4.0, 90.0))
        max_iters = int(max(50, remaining * 300.0))
        cand, _val = polish(a0, slice_end, max_iters, rng)
        cand = np.clip(cand, 0.0, 1000.0)
        if time.monotonic() > deadline - 4.0:
            break
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = cand.copy()
            print("attempt %d n=%d seed=%s  score=%.6f" % (attempt, n, kind, s), flush=True)

    # Final authoritative confirmation.
    final = [float(v) for v in np.clip(best, 0.0, 1000.0)]
    fs = float(evaluate_sequence(final))
    if not np.isfinite(fs):
        # should not happen, but be safe with a known-valid sequence
        final = [float(v) for v in make_seed(2048, "two_block", rng)]
        fs = float(evaluate_sequence(final))
    print("FINAL score = %.6f  n=%d" % (fs, len(final)), flush=True)
    return final
