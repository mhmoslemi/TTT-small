import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp


def build_conv_structure(n):
    """Precompute row/col indices for the convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        rows.extend([k] * (j_hi - j_lo + 1))
        cols.extend(range(j_lo, j_hi + 1))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_conv_matrix(s, row_idx, col_idx):
    """Create the sparse convolution matrix with current s values."""
    n = len(s)
    L = 2 * n - 1
    s_idx = row_idx.astype(np.int64) - col_idx.astype(np.int64)
    data = s[s_idx]
    return sp.csr_matrix((data, (row_idx, col_idx)), shape=(L, n))

def stochastic_rejected(s, s_try):
    """Simple check if s_try is very different from s (for reset decision)."""
    diff = np.mean(np.abs(s_try - s)) / (np.mean(s) + 1e-10)
