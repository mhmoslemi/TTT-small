```
def make_power_p(n, a, b):
    n = int(n)
    if n <= 0: return np.array([], dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    y = np.power(x, float(a))
    y *= np.power(1.0 - x, float(b))
    y = np.where(np.isfinite(y) & (y > 0.0), y, 0.0)
    s = y.sum()
    if s <= 0 or not np.isfinite(s):
        y = np.ones(n, dtype=np.float64)
    else:
        y /= s
    y = (1.0 - 1e-12) * y + (1e-12 / n)
    return y
```

```python
    if base is None:
        logp = np.zeros(n, dtype=np.float64)
    else:
        logp = np.log(np.maximum(np.asarray(base, dtype=np.float64), 1e-300))
        if logp.size != n:
            logp = np.resize(logp, n)
    y = rng.normal(size=n)
    yf = fft(y)
    freq = np.arange(n)
    dist = np.minimum(freq, n - freq)
    att = 1.0 / (1.0 + np.power(np.maximum(dist, 1.0), 1.2))
    yf = yf * att
    y = ifft(yf).real
    y -= y.mean()
    sd = y.std()
    if sd > 0.0 and np.isfinite(sd):
        y = y / sd
    logp = logp + float(tau) * y
    return exp_normalize(logp, 1e-12)


def resize_p(p, n):
    p = np.asarray(p, dtype=np.float64)
    n = int(max(1, n))
    if p.size == 0:
        return np.ones(n, dtype=np.float64)
    if p.size == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    if p.size == n:
        return normalize_p(p, 1e-12)
    x_old = np.linspace(0.0, 1.0, p.size)
    x_new = np.linspace(0.0, 1.0, n)
    q = np.interp(x_new, x_old, p)
    return normalize_p(q, 1e-12)


def update_best(best_p, best_score, best_n, p, score):
    if p is None or not np.isfinite(score) or score >= best_score - 1e-12:
        return best_p, best_score, best_n
    return np.asarray(p, dtype=np.float64).copy(), float(score), int(len(p))


def local_optimize(p_start, rng, time_limit):
    p = normalize_p(p_start, 1e-12)
    _, best_score = conv_score(p)
    if not np.isfinite(best_score):
        best_score = np.inf
    best_p = p.copy()

    if time_limit <= 0.0:
        return best_p, best_score

    deadline = time.monotonic() + float(time_limit)
    move = 0.35
    stuck = 0
    step = 0

    exp_params = (2.0, 4.0, 6.0, 10.0, 16.0, 25.0, 40.0, 70.0)
    pow_params = (1.0, 1.8, 2.5, 3.5, 5.0, 7.0, 10.0)
    moves = (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 2.0)

    while time.monotonic() < deadline:
        step += 1
        c, score = conv_score(p)
        if not np.isfinite(score):
            score = np.inf
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = float(score)
            best_p = p.copy()

        if np.isfinite(best_score) and best_score < 1.001:
            break

        if c.size == 0 or not np.isfinite(score):
            p = best_p.copy()
            continue

        if (step % 2) == 0:
            A = exp_params[(step // 2) % len(exp_params)]
            w = weights_exp(c, A)
        else:
            g = pow_params[((step - 1) // 2) % len(pow_params)]
            w = weights_power(c, g)

        gc = 2.0 * cross_corr_p_weights(p, w)
        gc = np.where(np.isfinite(gc), gc, 0.0)
        gc -= gc.mean()
        sdev = float(gc.std())

        if sdev <= 1e-15 or not np.isfinite(sdev):
            p = random_log_p(p.size, rng, base=p, tau=0.1)
            stuck += 1
            if stuck > 10:
                p = best_p.copy()
                stuck = 0
            continue

        trial_score = score
        trial_p = p
        for mult in moves:
            if not np.isfinite(mult) or mult <= 0.0:
                continue
            eta = float(move) * float(mult) / sdev
            if not np.isfinite(eta) or eta <= 0.0:
                continue
            pt = multiplicative_update(p, gc, eta)
            sc = fast_score(pt)
            if np.isfinite(sc) and sc < trial_score - 1e-13:
                trial_score = float(sc)
                trial_p = pt

        if np.isfinite(trial_score) and trial_score < score - 1e-12:
            p = trial_p
            stuck = 0
            move = min(1.2, move * 1.15)
        else:
            stuck += 1
            move = max(0.03, move * 0.82)
            bad = (
                stuck > 35
                or (not np.isfinite(score))
                or (np.isfinite(best_score) and score > best_score * 1.25)
            )
            if bad:
                tau = 0.05 + 0.35 * float(rng.random())
                pe = random_log_p(p.size, rng, base=p, tau=tau)
                pb = random_log_p(p.size, rng, base=best_p, tau=0.5 * tau)
                score_e = fast_score(pe)
                score_b = fast_score(pb)

                chosen = best_p.copy()
                chosen_score = best_score
                if np.isfinite(score) and score < chosen_score:
                    chosen = p.copy()
                    chosen_score = score
                if np.isfinite(score_e) and score_e < chosen_score:
                    chosen = pe
                    chosen_score = score_e
                if np.isfinite(score_b) and score_b < chosen_score:
                    chosen = pb
                    chosen_score = score_b

                p = chosen
                stuck = 0
                move = 0.3

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    budget = max(0.0, float(budget_s))
    safety = min(5.0, max(1.0, 0.02 * budget))
    deadline = start + max(0.0, budget - safety)
    if deadline <= start:
        deadline = start + min(1.0, budget)

    best_p = make_power_p(1024, 0.0, 0.0)
    best_score = fast_score(best_p)
    best_n = best_p.size
    power_a = 0.0
    power_b = 0.0
    have_power = True

    print("initial score {:.12f}".format(best_score), flush=True)

    def try_candidate(p, use_power=False, aa=None, bb=None):
        nonlocal best_p, best_score, best_n, power_a, power_b, have_power
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0:
            return
        sc = fast_score(p)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = float(sc)
            best_p = p.copy()
            best_n = int(p.size)
            if use_power:
                power_a = float(aa) if aa is not None else power_a
                power_b = float(bb) if bb is not None else power_b
                have_power = True

    grid_deadline = min(deadline, start + min(35.0, max(5.0, 0.02 * budget)))

    a_coarse = np.arange(-0.95, 0.55, 0.02)
    for n in (256, 512, 1024, 2048):
        if time.monotonic() >= grid_deadline:
            break
        for a in a_coarse:
            if time.monotonic() >= grid_deadline:
                break
            p = make_power_p(n, a, a)
            try_candidate(p, True, a, a)

        if time.monotonic() >= grid_deadline:
            break

        da_x = np.arange(-0.25, 0.30, 0.05)
        for da in da_x:
            if time.monotonic() >= grid_deadline:
                break
            for db in da_x:
                if time.monotonic() >= grid_deadline:
                    break
                aa = power_a + float(da)
                bb = power_b + float(db)
                if aa < -1.2 or aa > 1.0 or bb < -1.2 or bb > 1.0:
                    continue
                p = make_power_p(n, aa, bb)
                try_candidate(p, True, aa, bb)

    print("grid best {:.12f} n={}".format(best_score, best_n), flush=True)

    refine_deadline = min(deadline, start + min(150.0, max(15.0, 0.08 * budget)))

    if have_power:
        for n in (1024, 2048, 4096, 8192):
            if time.monotonic() >= refine_deadline:
                break
            p = make_power_p(n, power_a, power_b)
            try_candidate(p, True, power_a, power_b)

        for n in (2048, 4096):
            if time.monotonic() >= refine_deadline:
                break
            center = float(power_a)
            aa_vals = np.arange(center - 0.12, center + 0.13, 0.01)
            for aa in aa_vals:
                if time.monotonic() >= refine_deadline:
                    break
                p = make_power_p(n, aa, aa)
                try_candidate(p, True, aa, aa)

            off = np.arange(-0.08, 0.09, 0.02)
            for da in off:
                if time.monotonic() >= refine_deadline:
                    break
                for db in off:
                    if time.monotonic() >= refine_deadline:
                        break
                    aa = power_a + float(da)
                    bb = power_b + float(db)
                    if aa < -1.2 or aa > 1.0 or bb < -1.2 or bb > 1.0:
                        continue
                    p = make_power_p(n, aa, bb)
                    try_candidate(p, True, aa, bb)

        for n in (2048, 4096):
            if time.monotonic() >= refine_deadline:
                break
            for k in (1, 2, 3, 4):
                for amp in (0.05, 0.1, 0.2, 0.35, 0.5):
                    for phase in (0.0, 0.5 * np.pi, np.pi):
                        if time.monotonic() >= refine_deadline:
                            break
                        p = make_cos_mod_p(n, power_a, power_b, amp, k, phase)
                        try_candidate(p, True, power_a, power_b)

    parent = globals().get("height_sequence_1")
    if parent is not None:
        try:
            par = normalize_p(parent)
        except Exception:
            par = None
        if par is not None and par.size > 0:
            for n in (512, 1024, 2048, 4096, min(par.size, 8192)):
                if time.monotonic() >= deadline:
                    break
                p = resize_p(par, n)
                try_candidate(p)

    for n in (1024, 2048):
        if time.monotonic() >= deadline:
            break
        for _ in range(4):
            p = random_log_p(n, rng, tau=0.7)
            try_candidate(p)

    print("pre-local best {:.12f} n={}".format(best_score, best_n), flush=True)

    local_n = max(1024, int(best_n))
    p_local = best_p.copy()

    if have_power:
        for n_try in (2048, 4096, 8192):
            if time.monotonic() >= deadline:
                break
            p_try = make_power_p(n_try, power_a, power_b)
            try_candidate(p_try, True, power_a, power_b)

        p_pow4096 = make_power_p(4096, power_a, power_b)
        sc_pow4096 = fast_score(p_pow4096)
        if np.isfinite(sc_pow4096) and sc_pow4096 < 1.70:
            p_local = p_pow4096
            local_n = 4096
        elif np.isfinite(sc_pow4096) and sc_pow4096 < best_score:
            p_local = p_pow4096
            local_n = 4096
            best_p = p_pow4096.copy()
            best_score = float(sc_pow4096)
            best_n = 4096

    if best_n < 4096 and time.monotonic() < deadline:
        p_res = resize_p(best_p, 4096)
        sc_res = fast_score(p_res)
        if np.isfinite(sc_res) and sc_res < best_score:
            best_p = p_res.copy()
            best_score = float(sc_res)
            best_n = 4096
            p_local = p_res.copy()
            local_n = 4096

    if not (np.isfinite(fast_score(p_local))):
        p_local = best_p.copy()
        local_n = best_n

    if time.monotonic() < deadline - 20.0:
        remaining = deadline - time.monotonic()
        local_budget = min(remaining - 15.0, 1200.0)
        if local_budget > 20.0:
            p_path, sc_path = local_optimize(p_local, rng, local_budget)
            best_p, best_score, best_n = update_best(best_p, best_score, best_n, p_path, sc_path)
            print("local1 best {:.12f} n={}".format(best_score, best_n), flush=True)

    if time.monotonic() < deadline - 20.0:
        base = p_local if p_local is not None else best_p
        p_start2 = random_log_p(base.size, rng, base=base, tau=0.25)
        remaining = deadline - time.monotonic()
        local_budget2 = min(remaining - 10.0, 600.0)
        if local_budget2 > 20.0:
            p_path2, sc_path2 = local_optimize(p_start2, rng, local_budget2)
            best_p, best_score, best_n = update_best(best_p, best_score, best_n, p_path2, sc_path2)
            print("local2 best {:.12f} n={}".format(best_score, best_n), flush=True)

    if time.monotonic() < deadline - 5.0 and have_power:
        for n_try in (4096, 8192):
            p_try = make_power_p(n_try, power_a, power_b)
            best_p, best_score, best_n = update_best(best_p, best_score, best_n, p_try, fast_score(p_try))

    if best_p is None or best_p.size == 0 or not np.isfinite(fast_score(best_p)):
        best_p = normalize_p(np.ones(1024))
        best_score = fast_score(best_p)
        best_n = best_p.size

    print("final fast score {:.12f} n={}".format(best_score, best_n), flush=True)

    return [float(x) for x in best_p]
```
