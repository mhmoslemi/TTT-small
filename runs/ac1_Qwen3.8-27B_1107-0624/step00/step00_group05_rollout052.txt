import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix(a):
    n = len(a)
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.intp)
    cols = np.empty(total_nnz, dtype=np.intp)
    data = np.empty(total_nnz, dtype=np.float64)
    for j in range(n):
        start = j
        end = min(j + n, 2*n - 1)
        length = end - start
        idx = j * n
        rows[idx:idx+length] = np.arange(start, end, dtype=np.intp)
        cols[idx:idx+length] = j
        data[idx:idx+length] = a[:length]
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    if not res.success:
        return None
    return res.x
