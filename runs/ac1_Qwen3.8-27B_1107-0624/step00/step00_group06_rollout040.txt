        size <<= 1
    fa = np.fft.rfft(a, size)

def _subgrad_direction(a, b, eps=0.0):
    """Compute the (averaged) subgradient direction array (owing to the peak of b)."""
    n = len(a)
    max_b = np.max(b)
    # Active set: positions where b[k] >= max_b * (1 - eps)
    if eps > 0:
        active = np.where(b >= max_b * (1.0 - eps))[0]
    else:
        active = np.array([int(np.argmax(b))])
    
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            indices = np.arange(j_min, j_max + 1)
            a_idx = k - indices
            grad += 2.0 * a[a_idx]

        # Asymmetric: ramp-like
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.random.default_rng(0).uniform(-1, 1, n)
        a = np.maximum(a, 0.01)
    elif trial == 3:
        # Cosine window (tapered edges)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
        a = np.maximum(a, 1e-6)
    else:
        # Slightly perturbed uniform
        a = np.ones(n)
        a += rng.normal(0, 0.1, n)
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses FFT-accelerated subgradient descent with multiple diverse starting points.
    """
    rng = np.random.default_rng(seed)
    total_time = float(budget_s)
    deadline = time.monotonic() + max(0, total_time) - 5.0

    # Load reference sequence
    reference_seq = None
    reference_len = 0
    try:
        hs1 = np.asarray(height_sequence_1, dtype=np.float64)
        if hs1.size > 0 and np.all(np.isfinite(hs1)):
            reference_seq = hs1
            reference_len = hs1.size
    except:
        pass

    if reference_seq is None:
        reference_seq = np.ones(2000)
        reference_len = 2000

    # Choose n values to try
    n_values = sorted(set([
        reference_len,
        1000, 2000, 3000, 4000, 5000,
        6000, 7000, 8000, 10000, 12000, 15000
    ]))
    n_values = [n for n in n_values if 500 <= n <= 15000]

    best_a = None
    best_score = np.inf
    best_n = 0

    # Phase 1: Brief subgradient from many starts (25% of budget)
    phase1_time = 0.25 * (deadline - time.monotonic())
    per_start = max(2.0, phase1_time / (len(n_values) * 3))
    print(f"Phase 1: {len(n_values)} lengths x 3 trials, {per_start:.1f}s each")

    for n in n_values:
        for trial in range(3):
            if time.monotonic() > deadline:
                break
            a = _try_initialization(n, trial, rng, reference_seq, reference_len)
            a, score = _subgrad_descent(a, per_start, rng, eta0=0.3)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_n = n
                print(f"  New best: n={n}, trial={trial}, score={score:.6f}")

    print(f"After Phase 1: best_score={best_score:.6f}, n={best_n}")

    # Phase 2: Longer refinement from best (50% of budget)
    remaining = deadline - time.monotonic()
    phase2_time = 0.55 * remaining
    if phase2_time > 10:
        print(f"Phase 2: refining best (n={best_n}) for {phase2_time:.0f}s")
        a, score = _subgrad_descent(best_a, phase2_time, rng, eta0=0.1, verbose=True)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        print(f"After Phase 2: best_score={best_score:.6f}")

    # Phase 3: Final long run with smaller step (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print(f"Phase 3: final polish for {remaining:.0f}s")
        a, score = _subgrad_descent(best_a, remaining, rng, eta0=0.05, verbose=True)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        print(f"After Phase 3: best_score={best_score:.6f}")

    # Final verification with the actual evaluator
    final_list = [float(x) for x in best_a]
    final_score = evaluate_sequence(final_list)
    print(f"Final verified score: {final_score:.6f}")

    return final_list
