Hybrid optimizer for minimizing auto-convolution peak ratio 2n*max(a*a)/(sum(a))^2.
Uses L2-based FFT gradient descent followed by peak-subgradient refinement,
with adaptive step sizes and periodic perturbations to escape local minima.
"""

import numpy as np
import time


def fft_autocorr(a):
    """Compute linear auto-convolution a*a via FFT (zero-padded)."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def peak_subgradient(a, c, top_k=30):
    """Compute subgradient of max(c) w.r.t. a, using top_k peak positions."""
    n = len(a)
    top_k = min(top_k, len(c))
    idx = np.argpartition(c, -top_k)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def l2_gradient(a, c):
    """
    Compute gradient of ||c||_2^2 w.r.t. a where c = a*a.
    Result: g_i = 4 * sum_k c_k * a_{k-i} (for valid indices).
    Computed via FFT for efficiency.
    """
    n = len(a)
    # We need: g_i = 4 * sum_{k: 0<=k<2n-1, 0<=k-i<n} c_k * a_{k-i}
    # This is 4 * (c_conv_with_reversed_a)[i]
    # Let's compute via cross-correlation using FFT
    # g_i = 4 * sum_k c_k * a_{k-i}
    # Let j = k-i, so k = j+i: g_i = 4 * sum_j c_{j+i} * a_j
    # This is the cross-correlation of c and a at lag -i
    
    # Use FFT: cross-corr(c, a)[lag] = IFFT(FFT(c) * conj(FFT(a)))
    # We need a window of the cross-correlation
    N = 1
    while N < 3 * n:
        N *= 2
    fc = np.fft.rfft(c, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fc * np.conj(fa), N)
    # corr[m] = sum_j c[(j+m) mod N] * a[j] (circular correlation)
    # For linear: we need corr[i] for i = 0..n-1 but with the right indexing
    # Actually corr[m] = sum_j c[j+m] * a[j] for the linear part
    # g_i = 4 * sum_j c[j+i] * a[j] = 4 * corr[i] for i = 0..n-1
    # But we need to be careful about the circular vs linear part
    # Since N >= 3n > (2n-1) + n - 1, the circular wrap-around doesn't affect
    # the first (2n-1) entries of c convolved with n entries of a
    g = 4.0 * corr[:n]
    return g

    x = np.arange(n, dtype=np.float64)
    a = np.exp(-(x - (n - 1) / 2.0) ** 2 / (2.0 * sigma * sigma))
    return a


def make_bump_init(n, power=2.0):
    """Create a super-Gaussian bump initialization."""
    x = np.arange(n, dtype=np.float64)
    t = np.abs(x - (n - 1) / 2.0) / (n / 2.0)
    a = np.maximum(0.0, 1.0 - t) ** power
    return a


def make_sinc_like_init(n):
    """Create a windowed-sinc-like initialization (has some oscillation)."""
    x = np.arange(n, dtype=np.float64)

    window = np.exp(-2.0 * t * t)
    a = np.abs(sinc) * window  # Make non-negative
    return a
