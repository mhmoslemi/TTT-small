import numpy as np
import time

# ---------------------------------------------------------------------------
# Internal global state for fast FFT helper routines.
# ---------------------------------------------------------------------------
_CUR_N = 0
_CUR_L1 = 0
_CUR_L2 = 0
_EPS = 1e-300


def _set_fft_lens(n):
    global _CUR_N, _CUR_L1, _CUR_L2
    _CUR_N = int(n)
    L1 = 1
    while L1 < 2 * _CUR_N - 1:
        L1 *= 2
    L2 = 1
    while L2 < 3 * _CUR_N - 2:
        L2 *= 2
    _CUR_L1 = L1
    _CUR_L2 = L2


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= _EPS:
        return a
    return a / s


def _fast_self_convolve(a):
    n = len(a)
    if n <= 256:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    L = _CUR_L1
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)
    out_len = 2 * n - 1
    c = c[:out_len]
    return np.maximum(c, 0.0)


def _fast_score(a):
    n = len(a)
    s = float(a.sum())
    if s <= _EPS:
        return float("inf")
    c = _fast_self_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _fast_corr_v_a(v, a):
    """
    Return H[j] = sum_{r=0}^{n-1} a[r] * v[j+r], j = 0..n-1.
    """
    n = len(a)
    if n <= 256:
        c = np.convolve(v, a[::-1])
        return c[n - 1:n - 1 + n]
    L = _CUR_L2
    f = np.fft.rfft(v, L)
    g = np.fft.rfft(a[::-1], L)
    h = np.fft.irfft(f * g, L)
    return h[n - 1:n - 1 + n]


def _pool_score(item):
    return item[1]


def _keep_pool(pool, cap=16):
    if len(pool) <= cap:
        return
    pool.sort(key=_pool_score)
    del pool[cap:]


def _resample_shape(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return _normalize(a)
    x_old = (np.arange(n_old) + 0.5) / float(n_old)
    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    a_safe = np.maximum(a, _EPS)
    loga = np.log(a_safe)
    lognew = np.interp(x_new, x_old, loga)
    lognew = lognew - float(lognew.max())
    new = np.exp(lognew)
    return _normalize(new)


def _fallback_sequence(n):
    a = np.ones(int(n), dtype=np.float64)
    return a / a.sum()


def _flat_root_sequence(n):
    p = np.zeros(int(n), dtype=np.float64)
    if n <= 0:
        return None
    p[0] = 1.0
    for k in range(1, n):
        if k > 1:
            s = np.dot(p[1:k], p[k - 1:0:-1])
        else:
            s = 0.0
        val = (1.0 - s) / 2.0
        if val < -1e-8:
            return None
        p[k] = max(0.0, val)
    return p


def _add_existing_to_shapes(shapes, n):
    try:
        hs = globals().get("height_sequence_1", None)
        if hs is None:
            return
        arr = np.asarray(hs, dtype=np.float64)
        if arr.ndim != 1 or arr.size < 2:
            return
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if float(arr.sum()) <= _EPS:
            return
        a = _normalize(arr)
        if a.size != n:
            a = _resample_shape(a, n)
        shapes.append(a)
        shapes.append(a[::-1].copy())
    except Exception:
        pass


def _generate_initials(n, rng, include_recurrence=True):
    shapes = []
    n = int(n)
    if n <= 0:
        return shapes

    i = np.arange(1, n + 1, dtype=np.float64)
    x = (np.arange(1, n + 1, dtype=np.float64)) / float(n)
    xm = (np.arange(n) + 0.5) / float(n)

    # One-sided singular / flat-first-half family.
    for p in (0.35, 0.42, 0.48, 0.52, 0.58, 0.65):
        a = np.power(i, -p)
        a = _normalize(a)
        shapes.append(a)
        shapes.append(a[::-1].copy())

    # Finite-exponential-like families.
    for c in (0.2, 0.5, 1.0, 2.0):
        a = 1.0 / (1.0 + c * x)
        shapes.append(_normalize(a))

    for c in (0.1, 0.3, 0.7, 1.2):
        a = np.exp(-c * x)
        shapes.append(_normalize(a))

    # Smooth sine window.
    shapes.append(_normalize(np.sin(0.5 * np.pi * xm)))
    shapes.append(_normalize(np.sin(0.5 * np.pi * xm) ** 2))

    # Mild taper perturbations of the strong one-sided singular shape.
    base = np.power(i, -0.5)
    for q in (-0.25, -0.1, 0.0, 0.1, 0.25):
        a = base * (1.0 + q * x)
        a = _normalize(np.maximum(a, 0.0))
        shapes.append(a)

    # Slightly two-sided or asymmetric shapes.
    a = np.power(i, -0.25) * np.power(n + 1.0 - i, -0.08)
    shapes.append(_normalize(a))

    a = np.power(i, -0.6) * np.power(n + 1.0 - i, 0.2)
    shapes.append(_normalize(np.maximum(a, 0.0)))

    # Known construction, if present.
    _add_existing_to_shapes(shapes, n)

    # Exact flat first-half polynomial construction is cheap for small n.
    if include_recurrence and n <= 600:
        p = _flat_root_sequence(n)
        if p is not None:
            q = _normalize(p)
            shapes.append(q)
            shapes.append(q[::-1].copy())

    # Remove obviously invalid sequences.
    good = []
    for a in shapes:
        a = _normalize(a)
        if a.size == n and np.isfinite(a).all() and float(a.sum()) > _EPS:
            good.append(a)
    return good


def _evaluate_initials(n, rng, start_pool=None, include_recurrence=True):
    _set_fft_lens(n)
    pool = []
    if start_pool is not None:
        for a in start_pool:
            a = _normalize(a)
            if len(a) != n:
                a = _resample_shape(a, n)
            sc = _fast_score(a)
            if np.isfinite(sc):
                pool.append([a, sc])

    if start_pool is None or len(pool) < 3:
        shapes = _generate_initials(n, rng, include_recurrence=include_recurrence)
        for a in shapes:
            sc = _fast_score(a)
            if np.isfinite(sc):
                pool.append([a, sc])

    if not pool:
        a = _fallback_sequence(n)
        pool.append([a, _fast_score(a)])

    pool.sort(key=_pool_score)
    if len(pool) > 16:
        del pool[16:]
    return pool


def _mul_update(a, b, eta):
    # Dividing out the minimum is numerically harmless after normalization and
    # makes the update scale-invariant with respect to the gradient offset.
    b = np.asarray(b, dtype=np.float64)
    b = np.maximum(b, 0.0)
    bmin = float(b.min())
    b = b - bmin
    bmax = float(b.max())
    if bmax <= _EPS:
        return _normalize(a)
    b = b / bmax
    new = a * np.exp(-eta * b)
    return _normalize(new)


def _soft_step(a, rng):
    n = len(a)
    C = _fast_self_convolve(a)
    cmax = float(C.max())
    s = float(a.sum())
    if cmax <= _EPS or s <= _EPS:
        return a, _fast_score(a), 1

    cur = float(2.0 * n * cmax / (s * s))
    m = int(rng.choice((1, 2, 3, 4, 6, 8, 12)))

    # Either use all lags or focus on a small number of largest lags.
    if rng.random() < 0.45:
        v = np.power(C / cmax, m)
    else:
        tcnt = int(rng.choice((8, 16, 32, 64))] if True else 0)
        # The above line is intentionally simple; avoid special cases.
        tcnt = int(rng.choice((8, 16, 32, 64]))
        v = np.zeros(C.shape, dtype=np.float64)
        if tcnt >= C.size:
            idx = np.arange(C.size)
        else:
            idx = np.argpartition(C, -tcnt)[-tcnt:]
        vals = C[idx] / cmax
        v[idx] = np.power(np.maximum(vals, 0.0), m)

    ps = np.empty(2 * n, dtype=np.float64)
    ps[0] = 0.0
    ps[1:] = np.cumsum(v)
    V = ps[n:2 * n] - ps[:n]
    H = _fast_corr_v_a(v, a)
    b = V * a + H
    b = np.maximum(b, 0.0)
    bmax = float(b.max())
    if bmax <= _EPS:
        return a, cur, 1

    b = b - b.min()
    bspan = float(b.max())
    if bspan <= _EPS:
        return a, cur, 1
    b = b / bspan

    best_a = a
    best_sc = cur
    attempts = 1
    for eta in (0.01, 0.03, 0.08, 0.15, 0.25):
        new = _mul_update(a, b, eta)
        sc = _fast_score(new)
        attempts += 1
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = new
            if best_sc < cur * 0.97:
                break
    return best_a, best_sc, attempts


def _onehot_step(a, rng):
    n = len(a)
    C = _fast_self_convolve(a)
    cmax = float(C.max())
    s = float(a.sum())
    if cmax <= _EPS or s <= _EPS:
        return a, _fast_score(a), 1

    cur = float(2.0 * n * cmax / (s * s))
    r = rng.random()
    if r < 0.72:
        k = int(np.argmax(C))
    elif r < 0.90:
        # Often use the largest, but not always; this keeps the search from
        # freezing on a single lag.
        k = int(np.argmax(np.where(np.arange(C.size) == (k if False else 0), 0.0, C)))
        # Simpler and robust randomized top-lag sampling:
        if rng.random() < 0.7:
            k = int(np.argmax(C))
        else:
            k = int(rng.integers(0, C.size))
    else:
        k = int(rng.integers(0, C.size))

    L = max(0, k - n + 1)
    U = min(n - 1, k)
    if L > U:
        return a, cur, 1
    idx = np.arange(L, U + 1)
    partner = a[k - idx]
    alpha = float(rng.choice((0.0, 0.5, 1.0, 2.0)))
    b = np.zeros(n, dtype=np.float64)
    b[idx] = partner + alpha * a[idx]
    bmax = float(b.max())
    if bmax <= _EPS:
        return a, cur, 1

    b = b - b.min()
    bspan = float(b.max())
    if bspan <= _EPS:
        return a, cur, 1
    b = b / bspan

    best_a = a
    best_sc = cur
    attempts = 1
    for eta in (0.01, 0.03, 0.08, 0.15, 0.25):
        new = _mul_update(a, b, eta)
        sc = _fast_score(new)
        attempts += 1
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = new
            if best_sc < cur * 0.97:
                break
    return best_a, best_sc, attempts


def _modifier_step(a, rng):
    n = len(a)
    cur = _fast_score(a)
    best_a = a
    best_sc = cur
    attempts = 0
    base = np.maximum(a, _EPS)
    x = (np.arange(1, n + 1, dtype=np.float64)) / float(n)
    y = 1.0 - 0.9 * x
    kind = int(rng.integers(0, 5))

    if kind == 0:
        vals = (-0.20, -0.10, -0.05, 0.05, 0.10, 0.20)
        for c in vals:
            new = _normalize(base * np.power(x, c))
            sc = _fast_score(new)
            attempts += 1
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = new
    elif kind == 1:
        vals = (-0.20, -0.10, -0.05, 0.05, 0.10, 0.20)
        for c in vals:
            new = _normalize(base * np.power(max(float(y.min()), 1e-12) + y, c))
            sc = _fast_score(new)
            attempts += 1
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = new
    elif kind == 2:
        vals = (-0.50, -0.20, -0.10, 0.10, 0.20, 0.50)
        for c in vals:
            new = _normalize(base * np.exp(c * x))
            sc = _fast_score(new)
            attempts += 1
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = new
    elif kind == 3:
        vals = (-0.30, -0.15, 0.0, 0.15, 0.30)
        for c in vals:
            new = _normalize(base * (1.0 + c * np.sin(np.pi * x)))
            sc = _fast_score(new)
            attempts += 1
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = new
    else:
        # Low-frequency smooth random modulation.
        for _ in range(3):
            x0 = np.arange(n, dtype=np.float64) / float(n)
            amp = float(rng.uniform(0.05, 0.25))
            freq = int(rng.integers(1, 8))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            mod = amp * np.sin(2.0 * np.pi * freq * x0 + phase)
            new = _normalize(base * np.exp(mod))
            sc = _fast_score(new)
            attempts += 1
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = new

    return best_a, best_sc, max(1, attempts)


def _random_step(a, rng):
    n = len(a)
    cur = _fast_score(a)
    best_a = a
    best_sc = cur
    attempts = 0

    for _ in range(3):
        a2 = a.copy()
        mode = int(rng.integers(0, 4))

        if mode == 0:
            width = int(rng.integers(max(3, n // 30), max(4, n // 3) + 1))
            st = int(rng.integers(0, n - width + 1))
            a2[st:st + width] *= float(np.exp(rng.normal(0.0, 0.20)))
        elif mode == 1:
            cnt = int(rng.integers(max(3, n // 50), max(4, n // 5) + 1))
            idx = rng.choice(n, cnt, replace=False)
            a2[idx] *= np.exp(rng.normal(0.0, 0.25, cnt))
        elif mode == 2:
            x0 = np.arange(n, dtype=np.float64) / float(n)
            mod = 0.0
            for ff in range(1, 6):
                mod += float(rng.normal(0.0, 0.08)) * np.sin(
                    2.0 * np.pi * ff * x0 + float(rng.uniform(0.0, 2.0 * np.pi))
                )
            a2 = a2 * np.exp(mod)
        else:
            x = (np.arange(1, n + 1, dtype=np.float64)) / float(n)
            pa = float(rng.uniform(-0.20, 0.20))
            pb = float(rng.uniform(-0.20, 0.20))
            a2 = a2 * np.power(x, pa) * np.power(1.0 - x + 1e-6, pb)

        if rng.random() < 0.08:
            a2 = 0.5 * (a2 + a2[::-1])

        a2 = _normalize(a2)
        sc = _fast_score(a2)
        attempts += 1
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a2

    return best_a, best_sc, max(1, attempts)


def _random_perturb(a, rng, sigma=0.10):
    new = a * np.exp(rng.normal(0.0, sigma, size=a.size))
    return _normalize(new)


def _optimize_size(n, time_budget, seed, start_pool=None):
    """
    Local multi-basin search at a fixed sequence length.
    Returns (best_sequence, best_internal_score, pool).
    """
    n = int(n)
    _set_fft_lens(n)
    rng = np.random.default_rng(seed)
    pool = _evaluate_initials(n, rng, start_pool=start_pool, include_recurrence=(n <= 600))

    best_a = pool[0][0]
    best_sc = pool[0][1]

    t0 = time.monotonic()
    end = t0 + max(0.0, float(time_budget) - 1.0)
    stall = 0
    total = 0
    print(f"[size {n}] initial best internal score = {best_sc:.10f}", flush=True)

    while time.monotonic() < end:
        if not pool:
            break

        # Mostly exploit the best current pool entry, sometimes explore basins.
        if rng.random() < 0.72:
            idx = 0
        else:
            idx = int(rng.integers(0, len(pool)))

        a, sc = pool[idx]
        op = rng.random()

        if op < 0.46:
            new, nsc, tries = _soft_step(a, rng)
        elif op < 0.68:
            new, nsc, tries = _onehot_step(a, rng)
        elif op < 0.86:
            new, nsc, tries = _modifier_step(a, rng)
        else:
            new, nsc, tries = _random_step(a, rng)

        total += tries
        if np.isfinite(nsc) and nsc < sc:
            pool[idx] = [new, nsc]
            if nsc < best_sc:
                best_sc = nsc
                best_a = new
                stall = 0
            else:
                stall += tries
        else:
            stall += tries

        if stall > 30000:
            pert = _random_perturb(best_a, rng, sigma=0.12)
            psc = _fast_score(pert)
            repl = int(rng.integers(0, len(pool)))
            pool[repl] = [pert, psc]
            stall = 0

        if total and total % 5000 < max(1, tries):
            print(
                f"[size {n}] elapsed {time.monotonic() - t0:.1f}s "
                f"best internal score = {best_sc:.10f}",
                flush=True,
            )

        _keep_pool(pool, 16)

    pool.sort(key=_pool_score)
    best_a = pool[0][0]
    best_sc = pool[0][1]
    return best_a, best_sc, pool


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.

    Strategy:
      1. Build strong one-sided singular / flat-first-half starts.
      2. Multi-resolution local search: 256 and 512, then final length.
      3. At final length, run a pool-based local search mixing:
         - soft gradient/peak-weighted convex-conjugate updates,
         - exact max-lag subgradient pushes,
         - smooth shape modifiers,
         - random block/block-log perturbations.
      4. Validate top candidates with the authoritative evaluate_sequence.
    """
    hard_start = time.monotonic()
    hard_deadline = hard_start + max(0.0, float(budget_s) - 5.0)

    # Final length.  The previous construction length is a good pragmatic choice,
    # but we clamp to a range that is fast for FFT-based local search.
    N = 2535
    try:
        hs = globals().get("height_sequence_1", None)
        if hs is not None:
            L = int(np.asarray(hs).size)
            if 1024 <= L <= 6144:
                N = L
    except Exception:
        pass

    N = int(max(1024, min(6144, N)))

    start_pool = None

    # Small-resolution passes only when enough time is available.
    if hard_deadline - time.monotonic() > 520.0:
        t1 = min(90.0, (hard_deadline - time.monotonic()) * 0.12)
        if t1 > 25.0:
            b1, s1, _ = _optimize_size(256, t1, int(seed) + 101, None)
            start_pool = [_resample_shape(b1, N)]
            print(f"[small 256] best internal score = {s1:.10f}", flush=True)

    if hard_deadline - time.monotonic() > 760.0:
        t2 = min(180.0, (hard_deadline - time.monotonic()) * 0.18)
        if t2 > 35.0:
            b2, s2, _ = _optimize_size(512, t2, int(seed) + 202, start_pool)
            start_pool = [_resample_shape(b2, N)]
            print(f"[small 512] best internal score = {s2:.10f}", flush=True)

    final_budget = hard_deadline - time.monotonic() - 12.0
    if final_budget < 30.0:
        # Not enough time for a final pass; fall back to the best resampled one.
        if start_pool is not None:
            best = _normalize(start_pool[0])
        else:
            _set_fft_lens(N)
            best = _fallback_sequence(N)
        return [float(x) for x in best]

    best_a, best_sc, pool = _optimize_size(N, final_budget, int(seed) + 1000, start_pool)
    print(f"[final] best internal score = {best_sc:.10f}", flush=True)

    # Validate with the authoritative evaluator.  Prefer the best exact score.
    _set_fft_lens(N)
    evaluated = []
    best_exact = float("inf")
    best_exact_seq = _normalize(best_a)

    # Evaluate the best few internal candidates.
    for item in pool[:12]:
        a = _normalize(item[0])
        if len(a) != N:
            a = _resample_shape(a, N)
        # Skip duplicates by a quick fingerprint of sum/max/index max.
        fp = (
            float(a.sum()),
            float(a.max()),
            int(np.argmax(a)),
            float(a[:: max(1, a.size // 64)].sum()),
        )
        if fp in evaluated:
            continue
        evaluated.append(fp)
        try:
            val = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            val = float("inf")
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_exact_seq = a

    # Also validate the overall internal best explicitly.
    a = _normalize(best_a)
    try:
        val = float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        val = float("inf")
    if np.isfinite(val) and val < best_exact:
        best_exact = val
        best_exact_seq = a

    if not np.isfinite(best_exact):
        best_exact_seq = _fallback_sequence(N)

    print(f"[final] best exact score = {best_exact:.10f}", flush=True)
    return [float(x) for x in best_exact_seq]
