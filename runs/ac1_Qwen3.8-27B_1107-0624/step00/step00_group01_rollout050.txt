```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

"""Combined LP-based direction finding with subgradient refinement to minimize
the autocorrelation peak ratio 2n*max(a*a)/sum(a)^2 for non-negative sequences."""

def build_convolution_matrix(n, a):
    """Build the (2n-1) x n matrix A such that A@x = a*x (full convolution)."""
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp_direction(a, S):
    """Solve LP: min t s.t. sum(g)=S, g>=0, (a*g)_k <= t for all k.
    Returns (g*, t*)."""
    n = len(a)
    # Variables: g_0,...,g_{n-1}, t (total n+1)
    # Objective: minimize t => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: (a*g)_k <= t for all k
    # Rearranged: (a*g)_k - t <= 0
    # A_ub has shape (2n-1, n+1)
    A_conv = build_convolution_matrix(n, a)
    # A_ub = [A_conv | -I_column] where I_column is a column of ones
    # Actually: (a*g)_k - t <= 0 means row k of A_conv dot g - 1*t <= 0
    from scipy.sparse import eye as sp_eye
    neg_t_col = -np.ones((2*n-1, 1))
    A_ub = np.hstack([A_conv.toarray(), neg_t_col])  # This will be dense, might be slow for large n
    
    b_ub = np.zeros(2*n - 1)
    
    # Equality constraint: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: g_j >= 0, t free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
```

```python
def build_convolution_matrix(n, a):
    """Efficiently build (2n-1) x n sparse matrix for convolution."""
    # For each column j, the non-zero entries are in rows j to j+n-1
    # A[k, j] = a[k-j] for k in [j, j+n-1] (capped at 2n-2)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(2*n - 1, j + n)
        length = k_end - k_start
        rows.extend(range(k_start, k_end))
        cols.extend([j] * length)
        vals.extend(a[:length])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

```python
    n = len(a)
    size = 2 * n - 1
    # Pad to next power of 2 for efficiency

    A = fft(a, fsize)
    conv = np.real(ifft(A * A, fsize))[:size]
    return conv

def subgradient_step(a, eta):

    n = len(a)

    M = np.max(conv)
```

The objective $f(a) = 2n \max_k (a*a)_k$ (with $S=1$) is the maximum of $2n-1$ quadratic functions. We can think of this as a "max-of-quadratics" problem.

One approach is to use a "barrier" or "log-sum-exp" approximation:
$f_\epsilon(a) = 2\epsilon \log \sum_k \exp((a*a)_k / \epsilon)$

This is a smooth approximation of $f(a)$ that converges to it as $\epsilon \to 0$. We can minimize $f_\epsilon$ using gradient-based methods and then decrease $\epsilon$.

The gradient of $f_\epsilon$ is:
