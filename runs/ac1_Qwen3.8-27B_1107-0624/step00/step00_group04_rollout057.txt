def build_conv_matrix_sparse(f):
    """Build sparse (2n-1) x n matrix A where A @ g = convolve(f, g) full."""
    n = len(f)
    diags = [np.full(n, f[d], dtype=np.float64) for d in range(n)]

    A = scipy.sparse.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')

    """Compute the objective 2n*max(conv)/(sum)^2 locally."""
    s = float(np.sum(f))
    if s < 0.01:
        return np.inf

    mb = float(np.max(conv))
    n = len(f)
    return float(2.0 * n * mb / (s * s))


def solve_lp_direction(f, M, active_frac=1.0):
    """
    Solve LP: maximize sum(g) subject to (f*g)[k] <= M for active k, g >= 0.
    Returns g array or None on failure.
    """
    n = len(f)
    conv = np.convolve(f, f)

    if active_frac < 1.0:
        threshold = M * (1.0 - active_frac)
        active_mask = conv >= threshold
        active_mask[np.argmax(conv)] = True
        # Also always include the global max and a few neighbors
        max_idx = np.argmax(conv)
        for di in range(-2, 3):
            idx = max_idx + di
            if 0 <= idx < len(conv):
                active_mask[idx] = True
        active_indices = np.where(active_mask)[0]
    else:
        active_indices = np.arange(2 * n - 1)

    n_acts = len(active_indices)

    # Build the full constraint matrix and extract active rows
    full_A = build_conv_matrix_sparse(f)
    A = full_A[active_indices, :].tocsr()

    b = np.full(n_acts, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)

    try:
        res = opt.linprog(
            c, A_ub=A, b_ub=b,
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': 60}

    except Exception as e:
        return None

    if res.success:
        return res.x
    elif res.status == 9:  # Time limit reached, use best available

            return res.x
    return None

            overall_best_f = f.copy()

    # Phase 1: Quick screening of various (n, init) combinations
    print("Phase 1: Screening...")
    screening_results = []

    # Candidate lengths
    lengths = [1500, 3000, 5000, 7000, 10000]
    if ref_seq is not None and 0 < len(ref_seq) <= 15000:
        rl = len(ref_seq)
        if rl not in lengths:
            lengths.append(rl)
        lengths.sort()

    screening_scores = 20  # seconds per screening trial
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break

        inits = []
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0

        # Gaussian (wide)
        sigma = n / 3.0
        inits.append(np.exp(-((x - center) ** 2) / (2 * sigma ** 2)))

        # Gaussian (medium)
        sigma = n / 5.0
        inits.append(np.exp(-((x - center) ** 2) / (2 * sigma ** 2)))

        # Perturbed uniform
        pu = np.ones(n) + 0.02 * rng.standard_normal(n)
        inits.append(np.maximum(0.1, pu))

        # Sine bump
        inits.append(np.maximum(0.0, np.sin(np.pi * x / (n - 1)**2 * n)))

        # Triangular (peaked center)
        inits.append(1.0 - 2.0 * np.abs(x - center) / n)

        # Reference (if matching length)
        if ref_seq is not None and len(ref_seq) == n:
            inits.append(ref_seq.copy())

        for fi, f_init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            f_init = np.maximum(0.0, f_init)
            if np.sum(f_init) < 0.01:
                continue

            t0 = time.monotonic()
            f_best, sc = run_lp_search(f_init, time_budget=screening_scores, verbose=False)
            elapsed = time.monotonic() - t0

            sc_verified = compute_score_local(f_best)
            screening_results.append((n, fi, f_best, sc_verified, elapsed))
            if sc_verified < overall_best_score:
                overall_best_score = sc_verified
                overall_best_f = f_best.copy()
                print(f"  n={n}, init={fi}: score={sc_verified:.6f} ({elapsed:.1f}s)")

    if overall_best_f is None:
        # Fallback
        f = np.ones(256)
        return f.tolist()

    print(f"Phase 1 best: score={overall_best_score:.6f}, n={len(overall_best_f)}")

    # Phase 2: Extended refinement of the best
    print("Phase 2: Extended refinement...")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        f_refined, sc_refined = run_lp_search(
            overall_best_f.copy(),
            time_budget=remaining * 0.8,
            active_frac=0.5,
            verbose=True
        )
        if sc_refined < overall_best_score:
            overall_best_score = sc_refined
            overall_best_f = f_refined.copy()
            print(f"  Refined: score={sc_refined:.6f}")

    # Phase 3: Try to improve with different active_frac settings
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for af in [0.3, 0.7, 1.0]:
            if remaining < 20:
                break
            f_try = overall_best_f.copy()
            # Small perturbation to escape potential local min
            f_try += 0.001 * rng.standard_normal(len(f_try))
            f_try = np.maximum(0.0, f_try)
            t_budget = min(remaining * 0.3, 120)
            f_r, sc_r = run_lp_search(f_try, time_budget=t_budget, active_frac=af, verbose=False)
            if sc_r < overall_best_score:
                overall_best_score = sc_r
                overall_best_f = f_r.copy()
                print(f"  Phase 3 (af={af}): score={sc_r:.6f}")
            remaining = deadline - time.monotonic()

    # Phase 4: Final local refinement with full active set
    remaining = deadline - time.monotonic()
    if remaining > 10:
        f_final, sc_final = run_lp_search(
            overall_best_f.copy(),
            time_budget=remaining * 0.9,
            active_frac=1.0,
            verbose=True
        )
        if sc_final < overall_best_score:
            overall_best_score = sc_final
            overall_best_f = f_final.copy()

    # Final verification with the actual evaluator
    result_list = [float(max(0.0, x)) for x in overall_best_f]
    final_score = float(evaluate_sequence(result_list))
    print(f"Final verified score: {final_score:.8f}")

    if not np.isfinite(final_score):
        # Shouldn't happen, but fallback
        if ref_seq is not None:
            return [float(max(0.0, x)) for x in ref_seq]
        return [1.0] * 256

    return result_list
