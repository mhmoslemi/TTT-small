"""
Minimize 2*n*max(f*f)/(sum f)^2 over non-negative sequences f.

Insight: the score is scale-invariant and equals 2 for any *constant*
sequence (whose self-convolution is a triangle peaking at S^2/n).  To beat 2 we
need a *flatter* self-convolution: by the identity sum(f*f) = (sum f)^2 the
average level is S^2/(2n-1), so a flatter-than-triangular profile lowers the
peak and hence the score.  We therefore run a simulated-annealing style local
search (contiguous-block multiplicative & additive perturbations, projected to
the non-negative orthant) from several seeds of several lengths, and polish the
global best.  Exact np.convolve is used for speed; the authoritative
evaluate_sequence is used to validate every reported improvement.
"""

import time
import numpy as np


# ----------------------------------------------------------------------------
# Fast internal scorer (matches evaluate_sequence for non-negative values).
# ----------------------------------------------------------------------------
def _score_vec(f):
    """Return the objective for a 1-D non-negative numpy array (fast path)."""
    f = np.ascontiguousarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.inf
    s = float(np.sum(f))
    if s < 0.01:
        return np.inf
    c = np.convolve(f, f, mode="full")
    peak = float(np.max(c))
    return 2.0 * n * peak / (s * s)


def _clip_nonneg(f):
    return np.clip(f, 0.0, 1000.0)


def _seed_hamming(n, rng):
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1))
    return _clip_nonneg(w + 1e-6)


def _seed_edge(n, rng):
    # high at the ends, low in the middle (tent pointing down)
    i = np.arange(n, dtype=np.float64)
    d = np.abs(i - (n - 1.0)) / n
    f = 1.0 - 0.8 * (1.0 - d)          # large near edges
    f = f + rng.normal(0.0, 0.02, n)
    return _clip_nonneg(f)


def _seed_gauss_center(n, rng):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1.0) / 2.0
    f = np.exp(-0.5 * ((i - c) / (0.18 * n)) ** 2)
    f = f + rng.normal(0.0, 0.02, n)
    return _clip_nonneg(f)


def _seed_smooth_rand(n, rng):
    base = rng.random(n)
    # simple low-pass smoothing to start reasonably flat
    k = max(1, n // 64)
    ker = np.ones(k) / k
    f = np.convolve(base, ker, mode="same")
    f = f + rng.normal(0.0, 0.05, n)
    return _clip_nonneg(f + 1e-4)


def _seed_from_parent(n, rng):
    """Resample the provided parent construction to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        p = np.clip(p, 0.0, 1000.0)
        if p.size != n:
            p = np.interp(np.linspace(0, p.size - 1, n),
                          np.arange(p.size), p)
        return _clip_nonneg(p + 1e-6)
    except Exception:
        return _clip_nonneg(1.0 + rng.normal(0.0, 1e-3, n))


# ----------------------------------------------------------------------------
# Perturbation operators (each O(n), non-closing helpers at top level).
# ----------------------------------------------------------------------------
def _perturb(f, rng, temp):
    n = f.size
    g = f.copy()
    mode = rng.random()
    if mode < 0.5:
        # multiplicative on a contiguous block
        width = int(rng.integers(1, max(2, n // 8)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        log_amp = rng.normal(0.0, 0.25 + 0.4 * min(1.0, temp))
        g[start:stop] *= float(np.exp(log_amp))
    elif mode < 0.8:
        # additive gaussian on a contiguous block
        width = int(rng.integers(1, max(2, n // 6)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        amt = (f[start:stop].mean() + 1e-6) * (0.1 + 0.5 * min(1.0, temp))
        g[start:stop] += rng.normal(0.0, amt, stop - start)
    else:
        # whole-array gentle wobble
        amt = (f.mean() + 1e-6) * (0.05 + 0.2 * min(1.0, temp))
        g = g + rng.normal(0.0, amt, n)
    return _clip_nonneg(g)


def _anneal_optimize(seed_vec, rng, deadline):
    """Hill-climb + occasional simulated-annealing acceptance from one seed."""
    n = seed_vec.size
    best = seed_vec.copy()
    best_score = _score_vec(best)
    cur = seed_vec.copy()
    cur_score = best_score
    temp = 0.9

    while time.monotonic() < deadline:
        cand = _perturb(cur, rng, temp)
        cand_score = _score_vec(cand)
        if not np.isfinite(cand_score):
            temp = max(0.05, temp * 0.985)
            continue
        if cand_score <= cur_score or rng.random() < np.exp(
                (cur_score - cand_score) / (0.01 + temp * (cur_score + 1e-9))):
            # accept (improvement or annealed move)
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        temp = max(0.02, temp * 0.9995)
    return best, best_score


def _polish(vec, rng, deadline):
    """Fine local search with shrinking temperature to tighten the optimum."""
    n = vec.size
    best = vec.copy()
    best_score = _score_vec(best)
    cur = vec.copy()
    cur_score = best_score
    temp = 0.15

    while time.monotonic() < deadline:
        cand = _perturb(cur, rng, temp)
        cand_score = _score_vec(cand)
        if not np.isfinite(cand_score):
            continue
        if cand_score <= cur_score or rng.random() < np.exp(
                (cur_score - cand_score) / (0.005 + temp * 0.05)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        temp = max(0.01, temp * 0.999)
    return best, best_score


# ----------------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1800, **kwargs):
    """Search for the best non-negative coefficient sequence.

    Returns a plain list of floats.  Uses the fast internal scorer for the
    inner loop and validates every improvement candidate against the
    authoritative evaluate_sequence.  Falls back to a constant sequence
    (provably score = 2) if nothing else is found before the deadline.
    """
    rng = np.random.default_rng(seed)
    overall_deadline = time.monotonic() + max(60.0, float(budget_s) - 8.0)

    # A safe baseline: constant sequence gives exactly score 2.
    baseline = np.ones(256, dtype=np.float64)
    try:
        baseline_score = float(evaluate_sequence(baseline.tolist()))
    except Exception:
        baseline_score = _score_vec(baseline)
    global_best = baseline.copy()
    global_best_score = baseline_score

    # Lengths to explore; longer sequences allow flatter self-convolutions.
    lengths = [256, 512, 1024, 2048, 4096]
    seeds = [
        _seed_hamming,
        _seed_edge,
        _seed_gauss_center,
        _seed_smooth_rand,
        _seed_from_parent,
    ]

    # Allocate the time across (length, seed) jobs, skip work as time runs out.
    n_jobs = len(lengths) * len(seeds)
    per_job = max(3.0, (overall_deadline - time.monotonic()) / max(1, n_jobs))

    for n in lengths:
        for s in seeds:
            if time.monotonic() > overall_deadline:
                break
            try:
                start_vec = s(n, rng)
            except Exception:
                continue
            if start_vec.size != n or not np.isfinite(start_vec).any():
                continue
            job_deadline = min(overall_deadline, time.monotonic() + per_job)
            cand, sc = _anneal_optimize(start_vec, rng, job_deadline)
            if np.isfinite(sc) and sc < global_best_score:
                # validate against the authoritative evaluator
                try:
                    vs = float(evaluate_sequence([float(x) for x in cand]))
                except Exception:
                    vs = np.inf
                if np.isfinite(vs) and vs < global_best_score:
                    global_best = cand.copy()
                    global_best_score = vs
                    print("length=%d new best=%f" % (n, global_best_score),
                          flush=True)

    # Final polish phase on the global best with remaining time.
    if time.monotonic() < overall_deadline - 3.0:
        cand, sc = _polish(global_best, rng, overall_deadline)
        if np.isfinite(sc) and sc < global_best_score:
            try:
                vs = float(evaluate_sequence([float(x) for x in cand]))
            except Exception:
                vs = np.inf
            if np.isfinite(vs) and vs < global_best_score:
                global_best = cand.copy()
                global_best_score = vs
                print("polished best=%f" % global_best_score, flush=True)

    # Sanitize the return: non-negative finite floats as a plain list.
    out = _clip_nonneg(np.asarray(global_best, dtype=np.float64))
    out = out[np.isfinite(out)]
    if out.size == 0:
        out = np.ones(256, dtype=np.float64)
    result = [float(x) for x in out]

    # Final authoritative check; fall back to baseline if it misbehaves.
    try:
        final_score = float(evaluate_sequence(result))
    except Exception:
        final_score = np.inf
    if not np.isfinite(final_score):
        result = [float(x) for x in baseline]
        final_score = baseline_score
    print("FINAL score=%f length=%d" % (final_score, len(result)), flush=True)
    return result
