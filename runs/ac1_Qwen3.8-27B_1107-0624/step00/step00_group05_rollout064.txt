def _fft_convolve_self(a, n_fft):
    """Full linear autocorrelation conv(a, a) using FFT. Returns length 2n-1."""
    n = len(a)

def _fft_cross_correlate(p, a, n_fft):
    """Cross-correlation: out[m] = sum_k p[k+m]*a[k] for m=0,...,n-1."""
