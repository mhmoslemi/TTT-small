```python
    return score, grad
```

```python
"""
Optimizes non-negative sequences to minimize 2*n*max(autocorr)/sum^2
using gradient-based methods with smoothed max (log-sum-exp) annealing.
"""

import numpy as np
from numpy.lib.stride_tricks import as_strided
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time


def compute_full_convolution(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    N = 2 * n - 1
    fa = fft(a, N)
    conv = ifft(fa * fa).real
    np.maximum(conv, 0.0, out=conv)
    return conv


def compute_smoothed_max(conv, alpha):
    """Log-sum-exp smoothed maximum and softmax weights."""
    cmax = conv.max()
    shifted = alpha * (conv - cmax)
    # shifted is always <= 0, so no overflow
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    S = cmax + np.log(sum_exp) / alpha
    p = exp_s / sum_exp
    return S, p


def obj_grad(x, alpha):
    """Return (objective_value, gradient) for L-BFGS-B."""
    a = np.maximum(x, 1e-8)
    n = a.size
    conv = compute_full_convolution(a)
    S, p = compute_smoothed_max(conv, alpha)
    sa = a.sum()
    val = 2.0 * n * S / (sa * sa)
    # Gradient: g[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    P = as_strided(p, shape=(n, n), strides=(p.strides[0], p.strides[0]))
    g = 2.0 * (P @ a)
    grad = (2.0 * n / (sa ** 3)) * (g * sa - 2.0 * S)
    return val, grad


def run_minimize(x0, alpha, max_iter, bounds):
    """Run L-BFGS-B with given parameters."""
    result = minimize(
        fun=lambda x: obj_grad(x, alpha)[0],  # NO LAMBDAS!
        ...
    )
```

def compute_full_convolution(a):
    """Full linear convolution of a with itself via FFT."""
    n = a.size
    N = 2 * n - 1

    conv = ifft(fa * fa).real
    np.maximum(conv, 0.0, out=conv)
    return conv


def compute_smoothed_max(conv, alpha):
    """Log-sum-exp smoothed maximum and softmax probability weights."""
    cmax = conv.max()
    shifted = alpha * (conv - cmax)  # always <= 0
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    S = cmax + np.log(sum_exp) / alpha
    p = exp_s / sum_exp
    return S, p


def obj_and_grad(x, alpha):
    """Compute (objective, gradient) for the smoothed problem."""
    a = np.maximum(x, 1e-8)
    n = a.size
