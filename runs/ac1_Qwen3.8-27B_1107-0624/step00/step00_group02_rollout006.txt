```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_lp(a, M):
    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0."""
    n = len(a)
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: conv(a,b)[k] <= M for k = 0, ..., 2n-2
    # conv(a,b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[k-j] * b[j]
    # That is: sum_j A[k,j] * b[j] <= M
    # where A[k,j] = a[k-j] if 0<=k-j<n else 0
    
    num_constraints = 2*n - 1
    A = np.zeros((num_constraints, n))
    for k in range(2*n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    b_ub = np.full(num_constraints, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None
```

```python
    n = len(a)

    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _lp_step(f, M):
    """Solve the LP: max sum(b) s.t. conv(f,b)[k] <= M, b >= 0."""
    n = len(f)
    c_obj = -np.ones(n)

    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    num_rows = 2 * n - 1

    for j in range(n):

                  bounds=[(0, None)] * n, method='highs')

    return None
```
