def internal_score(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return float('inf')
    if not np.all(np.isfinite(a)):
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    mb = float(np.max(c))
    return float(2.0 * a.size * mb / (s * s))

"""
Search for a nonnegative coefficient vector minimizing

    2 * n * max(conv(a, a)) / sum(a)^2

by first constructing strong probability-distribution initializations from the
continuous "square root of uniform under convolution" density, then polishing
with a mirror-descent / active-lag local search.

The continuous density is obtained numerically from the Laplace transform

    F(s) = sqrt((1 - exp(-2s)) / (2s)),

whose square is the Laplace transform of Uniform(0, 2).  Discretizing this
density gives sequences whose full convolution is close to flat, often giving
scores well below 2 without any iterative search.
"""

import numpy as np
import time


def internal_score(a):
    """Same objective as the external evaluator, but without list conversion."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return float("inf")
    if not np.all(np.isfinite(a)):
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    mb = float(np.max(c))
    return float(2.0 * a.size * mb / (s * s))


def gradient_from_weights(w, u):
    """
    Gradient of sum_z w_z (u*u)_z with respect to u.

    For each lag z,
        (u*u)_z = sum_i u_i u_{z-i}
    and derivative with respect to u_i is 2 * u_{z-i}.
    """
    n = len(u)
    off = len(w) - 1
    corr = np.convolve(w[::-1], u)
    return 2.0 * corr[off:off + n]


def make_geometric_sequence(n, alpha=1.0):
    """Simple asymmetric nonnegative fallback."""
    if n <= 0:
        return None
    idx = np.arange(n, dtype=np.float64)
    u = np.exp(-alpha * idx / float(n))
    s = float(np.sum(u))
    if s <= 0.0 or not np.isfinite(s):
        return None
    return u / s


def make_random_sequence(n, seed):
    """Random positive fallback / diversification."""
    rng = np.random.default_rng(seed)
    u = rng.gamma(2.0, size=n)
    s = float(np.sum(u))
    if s <= 0.0 or not np.isfinite(s):
        return None
    return u / s


def make_density_sequence(n, c=1.0, L=2.0, Q=32):
    """
    Discretize the continuous density whose self-convolution is uniform on [0,2].

    Uses Bromwich inversion:
        f(x) = (1 / 2pi) int_{-inf}^{inf} exp((c+iw)x) F(c+iw) dw
    approximated by an FFT over a finite frequency window.
    """
    if n <= 1:
        return None

    minN = int(Q * L * float(n))
    N = 1
    while N < minN:
        N <<= 1

    dx = L / float(N)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)

    # For exact powers used here, this lands exactly at midpoints.
    j = np.rint(x / dx).astype(np.int64)
    j = np.clip(j, 0, N - 1)

    freq = np.fft.fftfreq(N, d=dx)
    omega = 2.0 * np.pi * freq
    s = c + 1j * omega

    with np.errstate(all="ignore"):
        z = (1.0 - np.exp(-2.0 * s)) / (2.0 * s)
        F = np.sqrt(z)
        base = np.fft.ifft(F)
        vals = (N / L) * np.exp(c * x) * np.real(base[j])

    vals = np.asarray(vals, dtype=np.float64)
    if not np.all(np.isfinite(vals)):
        return None

    pos_sum = float(np.sum(vals[vals > 0.0]))
    if pos_sum <= 0.0:
        return None

    neg_sum = float(np.sum(-vals[vals < 0.0]))
    if neg_sum > 2.0 * pos_sum:
        # Numerical inversion produced too much negativity.
        return None

    vals = np.maximum(vals, 0.0)
    s = float(np.sum(vals))
    if s <= 0.0:
        return None
    return vals / s


def get_base_sequence():
    """Use the externally provided height_sequence_1 if available and finite."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    if not np.all(np.isfinite(arr)):
        return None
    s = float(np.sum(arr))
    if s < 0.01:
        return None
    return arr / s


def collect_candidates(gen_deadline):
    """Generate and score a diverse set of starting sequences."""
    cands = []
    idx = 0

    # Main construction: continuous-square-root-of-uniform density.
    for n in (256, 512, 1024, 2048, 4096):
        if time.monotonic() > gen_deadline - 30.0:
            break
        for c in (1.0, 1.5, 2.0):
            if time.monotonic() > gen_deadline - 15.0:
                break
            u = make_density_sequence(n, c=c, L=2.0, Q=32)
            if u is not None:
                sc = internal_score(u)
                if np.isfinite(sc):
                    tag = "density n=%d c=%.1f" % (n, c)
                    cands.append((float(sc), idx, tag, int(n), u.copy()))
                    idx += 1

    # Geometric fallbacks. These are not expected to be best, but can rescue
    # the search if the Laplace inversion is numerically poor.
    for n in (512, 1024, 2048):
        if time.monotonic() > gen_deadline - 20.0:
            break
        for alpha in (0.5, 1.0, 1.5, 2.0, 3.0):
            u = make_geometric_sequence(n, alpha)
            if u is not None:
                sc = internal_score(u)
                if np.isfinite(sc):
                    tag = "geom n=%d alpha=%.1f" % (n, alpha)
                    cands.append((float(sc), idx, tag, int(n), u.copy()))
                    idx += 1

    # A few random starts.
    if time.monotonic() < gen_deadline - 10.0:
        for seed in (1, 2, 3):
            n = 1024
            u = make_random_sequence(n, seed)
            if u is not None:
                sc = internal_score(u)
                if np.isfinite(sc):
                    tag = "rand n=1024 seed=%d" % seed
                    cands.append((float(sc), idx, tag, int(n), u.copy()))
                    idx += 1

    # Include the previous construction only if small enough to polish quickly.
    base = get_base_sequence()
    if base is not None and 64 <= base.size <= 2048:
        sc = internal_score(base)
        if np.isfinite(sc):
            tag = "base n=%d" % int(base.size)
            cands.append((float(sc), idx, tag, int(base.size), base.copy()))
            idx += 1

    if not cands:
        u = make_geometric_sequence(512, 1.0)
        if u is not None:
            sc = internal_score(u)
            if np.isfinite(sc):
                cands.append((float(sc), 0, "fallback geom", 512, u.copy()))

    # Sort by score; unique integer id prevents comparison of numpy arrays.
    cands.sort()

    # Keep the search set small.
    if len(cands) > 12:
        del cands[12:]

    return cands


def polish_sequence(start_u, stop_time, tag="polish"):
    """
    Local mirror-descent polish on the probability simplex.

    For current u, choose nonnegative lag weights w that concentrate on large
    convolution coefficients, compute the gradient of the weighted maximum,
    and perform multiplicative steps u <- u * exp(-eta * grad), renormalizing.
    """
    u = np.asarray(start_u, dtype=np.float64).reshape(-1).copy()
    u = np.maximum(u, 0.0)
    s = float(np.sum(u))
    if s <= 0.0 or not np.isfinite(s):
        return np.asarray(start_u, dtype=np.float64).reshape(-1), float("inf")
    u /= s

    best = u.copy()
    best_score = internal_score(best)
    n = best.size
    base_u = best.copy()

    print("polish start %s n=%d score=%.6f" % (tag, n, best_score))
    if not np.isfinite(best_score):
        return best, best_score

    alpha_cycle = (2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0)
    eta_cycle = (0.001, 0.003, 0.01, 0.03, 0.1)

    alpha_idx = 0
    no_improve = 0
    rng = np.random.default_rng(987654)

    while time.monotonic() < stop_time:
        c = np.convolve(best, best)
        maxc = float(np.max(c))
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        alpha = alpha_cycle[alpha_idx % len(alpha_cycle)]
        beta = alpha / maxc

        with np.errstate(over="ignore", under="ignore", invalid="ignore"):
            w = np.exp(beta * (c - maxc))

        wsum = float(np.sum(w))
        if not np.isfinite(wsum) or wsum <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= wsum

        grad = gradient_from_weights(w, best)
        gmax = float(np.max(grad))
        if not np.isfinite(gmax) or gmax <= 0.0:
            gmax = 1.0

        improved = False

        for eta in eta_cycle:
            if time.monotonic() > stop_time:
                break

            if no_improve > 20:
                for step in (eta * 0.3, eta, eta * 3.0):
                    cand = best * np.exp(-step * grad / gmax)
                    cs = float(np.sum(cand))
                    if cs <= 0.0 or not np.isfinite(cs):
                        continue
                    cand /= cs
                    sc = internal_score(cand)
                    if sc < best_score - 1e-12:
                        best = cand
                        best_score = sc
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            else:
                cand = best * np.exp(-eta * grad / gmax)
                cs = float(np.sum(cand))
                if cs > 0.0 and np.isfinite(cs):
                    cand /= cs
                    sc = internal_score(cand)
                    if sc < best_score - 1e-12:
                        best = cand
                        best_score = sc
                        improved = True
                        no_improve = 0
                        break

        if not improved:
            no_improve += 1
            alpha_idx += 1

            # Small multiplicative noise to escape very flat plateaus.
            for _ in range(8):
                if time.monotonic() > stop_time:
                    break
                if no_improve < 30:
                    sigma = 1e-3
                elif no_improve < 100:
                    sigma = 1e-2
                else:
                    sigma = 5e-2
                cand = best * (1.0 + sigma * rng.standard_normal(n))
                cand = np.maximum(cand, 0.0)
                cs = float(np.sum(cand))
                if cs <= 0.0 or not np.isfinite(cs):
                    continue
                cand /= cs
                sc = internal_score(cand)
                if sc < best_score - 1e-12:
                    best = cand
                    best_score = sc
                    improved = True
                    no_improve = 0
                    break

            # Occasionally blend back toward the original good initialization.
            if not improved and no_improve > 10:
                for t in (0.1, 0.3, 0.5):
                    cand = (1.0 - t) * best + t * base_u
                    cs = float(np.sum(cand))
                    if cs > 0.0 and np.isfinite(cs):
                        cand /= cs
                        sc = internal_score(cand)
                        if sc < best_score - 1e-12:
                            best = cand
                            best_score = sc
                            improved = True
                            no_improve = 0
                            break

            if not improved:
                if no_improve % 100 == 0:
                    print("polish progress %s n=%d score=%.6f no_improve=%d" % (
                        tag, n, best_score, no_improve))
                if no_improve > 350:
                    break
        else:
            # Slowly advance alpha after a few successful steps.
            if alpha_idx % 3 == 2:
                alpha_idx += 1

    print("polish end %s n=%d score=%.6f" % (tag, n, best_score))
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point required by the evaluator."""
    start = time.monotonic()
    total = max(1.0, float(budget_s) - 10.0)
    deadline = start + total

    # Generate diverse initializations, but always leave substantial polishing time.
    gen_time = min(360.0, total * 0.30)
    gen_deadline = start + gen_time

    cands = collect_candidates(gen_deadline)

    best_seq = None
    best_score = float("inf")

    if cands:
        for item in cands:
            sc = item[0]
            seq = item[4]
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

    if best_seq is None:
        best_seq = make_geometric_sequence(512, 1.0)
        best_score = internal_score(best_seq)

    # Polish the best candidate with a generous share of the time.
    if time.monotonic() < deadline - 60.0:
        if total > 600.0:
            stop1 = start + 0.68 * total
        else:
            stop1 = deadline - 60.0
        seq, score = polish_sequence(best_seq, stop1, tag="primary")
        if score < best_score:
            best_seq = seq
            best_score = score

    # Polish one promising alternate candidate if it is materially different
    # and we still have meaningful time left.
    if len(cands) > 1 and time.monotonic() < deadline - 180.0:
        second = None
        for item in cands:
            sc = item[0]
            n = item[3]
            seq = item[4]
            if np.array_equal(seq, best_seq):
                continue
            if sc <= max(1.60, best_score * 1.10) and n <= 2048:
                second = item
                break

        if second is not None:
            alloc = min(360.0, deadline - time.monotonic() - 120.0)
            if alloc > 90.0:
                stop2 = time.monotonic() + alloc
                seq2, score2 = polish_sequence(second[4], stop2, tag="secondary")
                if score2 < best_score:
                    best_seq = seq2
                    best_score = score2

    # Final polish on the current champion.
    if time.monotonic() < deadline - 30.0:
        stop3 = deadline - 10.0
        seq3, score3 = polish_sequence(best_seq, stop3, tag="final")
        if score3 < best_score:
            best_seq = seq3
            best_score = score3

    out = np.asarray(best_seq, dtype=np.float64).reshape(-1)
    out = np.clip(out, 0.0, 1000.0)
    out_sum = float(np.sum(out))

    if out.size == 0 or out_sum < 0.01 or not np.all(np.isfinite(out)):
        out = np.ones(256, dtype=np.float64)
        out_sum = float(np.sum(out))

    # Scale invariance means normalizing to probability form is safe.
    out = out / out_sum

    # Optional authoritative check for logging.
    try:
        final_list = [float(x) for x in out]
        eval_score = evaluate_sequence(final_list)
    except Exception:
        final_list = [float(x) for x in out]
        eval_score = best_score

    print("final internal=%.6f external=%.6f len=%d" % (best_score, eval_score, len(out)))
    return final_list
