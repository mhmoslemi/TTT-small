"""
Algorithm: LP-based iterative optimization for minimizing peak autocorrelation
of a non-negative sequence. Uses the cross-correlation LP to find descent
directions, combined with random restarts and adaptive blending.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_lp_matrix(f, n):
    """Build sparse constraint matrix for the LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, n - 1 - k)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = range(j_min, j_max + 1)
            ks = [k] * (j_max - j_min + 1)
            cols.extend(js)
            rows.extend(ks)
            vals.extend(f[k - j] for j in js)
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(f, M, n, time_limit=5.0):
    """Solve the cross-correlation LP."""
    A = build_lp_matrix(f, n)
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, 
                     method='highs', options={'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception as e:
        print(f"LP error: {e}")
    return None
