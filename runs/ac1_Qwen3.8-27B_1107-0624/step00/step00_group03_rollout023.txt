"""
Algorithm: Mirror descent (exponentiated gradient) with KL divergence
for minimizing the maximum auto-convolution of a non-negative sequence.

The key insight is that the objective max(conv(a,a)) is convex over the
simplex {a >= 0, sum(a) = 1}, so mirror descent with KL divergence is
the natural choice. The subgradient is computed efficiently via FFT-based
convolution.

Multiple initializations and adaptive learning rates are used to escape
local minima and converge to the global optimum.
"""

import numpy as np
import time


def compute_subgradient(a, w):
    """
    Compute subgradient of sum(w[k]) where w[k] = max(conv[k] - target, 0).
    Uses FFT-based convolution for O(n log n) complexity.
    """
    n = len(a)
    # g[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m] (with bounds)
    # = 2 * (a * w_rev)[2n-2-j] where w_rev = w reversed
    w_rev = w[::-1]
    conv_aw = np.convolve(a, w_rev)
    # conv_aw has length 3n-2
    # We want indices 2n-2, 2n-3, ..., n-1 (for j=0, 1, ..., n-1)
    h = conv_aw[n-1:2*n-1][::-1]
    return 2.0 * h


def mirror_descent(a0, deadline, eta_init=0.5, eta_decay=0.99995, 
                   patience=5000, verbose=False):
    """
    Mirror descent with KL divergence for the simplex.
    Minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a))
        s = len(a)
    a = a / s  # normalize to sum=1
    
    n = len(a)
    target = 1.0 / (2.0 * n - 1.0)
    eta = eta_init
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    
    iter_no_improve = 0
    min_improve = 1e-10
    
    for it in range(10000000):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        current_score = 2.0 * n * max_conv
        
        if current_score < best_score - min_improve:
            best_score = current_score
            best_a = a.copy()
            iter_no_improve = 0
            if verbose and it % 10000 == 0:
                print(f"  iter {it}: score={current_score:.8f}")
        else:
            iter_no_improve += 1
        
        if iter_no_improve > patience:
            break
        
        # Compute excess
        w = np.maximum(conv - target, 0.0)
        max_w = np.max(w)
        if max_w < 1e-14:
            break
        
        # Compute subgradient
        g = compute_subgradient(a, w)
        g_max = np.max(g)
        if g_max < 1e-15:
            break
        
        # Scale gradient
        g = g / g_max
        
        # Mirror descent update: a * exp(-eta * g), then renormalize
        log_update = -eta * g
        # Clip for numerical stability

def make_block_init(n, rng):
    """Create a random block-structured initial sequence."""
    a = np.ones(n)
    num_blocks = max(2, n // 64)
    for _ in range(num_blocks):
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 8))
        end = min(n, start + width)
        a[start:end] *= float(rng.uniform(0.2, 3.0))
    return a

        # Use log-space to avoid overflow
        log_w = (p-1) * np.log(np.maximum(conv, 1e-30))
        log_w -= np.max(log_w)  # for numerical stability
        w = np.exp(log_w)
        w /= np.sum(w)  # normalize weights
        
        # Compute weighted gradient
        # g[j] = 2 * sum_m a[m] * (w * shift)[j+m] ... 
        # Actually, it's the same formula as before but with w instead of the binary mask
        g = compute_subgradient(a, w * np.sum(conv**p) / p)  # scale doesn't matter for direction

        g = compute_subgradient(a, w)
