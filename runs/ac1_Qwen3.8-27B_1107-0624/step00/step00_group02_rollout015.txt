Let me now think about the "soft max" approach. Instead of using just the argmax, we use a weighted average over the top few values:

`M_soft = sum_k w(k) * c[k]` where `w(k) = exp(beta * c[k]) / sum exp(beta * c[j])`

For large beta, this approaches the hard max. For smaller beta, it's smoother.

`d M_soft / da[i] = sum_k w(k) * 2*a[k-i]` (for valid k-i)

This is a weighted average of the gradients.

This "soft max" approach should give a smoother optimization landscape.

```python
def compute_grad(a, c, beta, n):
    """Compute gradient of soft-max objective."""
    S = np.sum(a)

    c_max = np.max(c)
    # For numerical stability, subtract max
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / np.sum(exp_c)

    M = np.dot(w, c)

    # = 2 * sum_k w[k] * a[k-i] * I(0 <= k-i < n)

    # dM/da[i] = 2 * sum_{j: 0<=j<n, 0<=i+j<2n-1} w[i+j] * a[j]
    # = 2 * dot(w[i : i+n], a)  (with appropriate bounds)

    # Valid j: 0 <= j < n. So k = i+j ranges from i to i+n-1.
```

```python
import numpy as np
import time

def soft_max_gradient(a, beta):
    """
    Compute the gradient of f(a) = 2n * softmax_beta(conv(a,a)) / (sum(a))^2.
    Uses soft max for smoothness.
    Returns (gradient, objective_value, active_weights).
    """
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return None, np.inf, None
    
    c = np.convolve(a, a)  # length 2n-1
    c_max = np.max(c)
    
    # Soft max weights (numerically stable)
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / np.sum(exp_c)
    M = np.dot(w, c)  # soft max value
    
    # Compute grad_M[i] = 2 * sum_j a[j] * w[i+j] (valid j)
    # Using the FFT trick: g[i] = conv(a_rev, w)[i+n-1]
    a_rev = a[::-1]
    conv_result = np.convolve(a_rev, w, mode='full')  # length 3n-2
    g = conv_result[n-1:2*n-1]  # extract the right part
    
    grad_M = 2.0 * g
    
    # Full gradient
    obj = 2.0 * n * M / (S ** 2)
    grad = 2.0 * n / (S ** 2) * grad_M - 4.0 * n * M * a / (S ** 3)
    
    return grad, obj, w

def normalize(a):
    """Normalize a to have sum 1, clip negatives."""
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / S
```
