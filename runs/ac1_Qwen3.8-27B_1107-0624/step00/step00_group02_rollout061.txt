def compute_obj(a):
    """Compute objective via FFT-based convolution."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    N = 2 * n
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return 2.0 * n * float(c.max()) / (S * S)


def get_convolution(a):
    """Return the full linear convolution of a with itself."""
    n = len(a)
    N = 2 * n
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2*n-1]


def make_subgradient_step(a, lr):
    """Apply one subgradient step to reduce the objective."""
    n = len(a)
    S = a.sum()
    c = get_convolution(a)
    k_star = int(c.argmax())
    c_max = c[k_star]
    
    # Compute d(c[k*])/d(a[i])
    dc = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            dc[i] = 2.0 * a[j] if j != i else 2.0 * a[i]
    
    # Direction to decrease J: -(dJ/d a)
    # dJ/d(a[i]) ≈ (2n/S²)(dc[i] - 2*c_max/S)
    # So we want to move in the direction of -(dc[i] - 2*c_max/S)
    direction = 2.0 * c_max / S - dc
    
    # Take a step
    new_a = np.maximum(0.0, a + lr * direction)
    
    return new_a


def make_block_perturbation(a, rng, sigma=0.1):
    """Apply a block scaling perturbation."""
    n = len(a)
    new_a = a.copy()
    width = int(rng.integers(1, max(2, n // 30)))
    start = int(rng.integers(0, n - width + 1))
    factor = np.exp(rng.normal(0, sigma))
    new_a[start:start+width] *= factor
    return np.clip(new_a, 0, 1000)


def make_element_perturbation(a, rng, sigma=0.1):
    """Perturb a few random elements."""
    n = len(a)
    new_a = a.copy()
    n_elems = int(rng.integers(1, max(2, n // 50)))
    idx = rng.choice(n, size=n_elems, replace=False)
    for i in idx:
        new_a[i] = max(0.0, a[i] + rng.normal(0, sigma * a.mean() + 1e-10))
    return np.clip(new_a, 0, 1000)


def make_swap(a, rng):
    """Swap two elements."""
    n = len(a)
    if n < 2:
        return a.copy()
    new_a = a.copy()
    i, j = rng.integers(0, n, size=2)
    if i != j:
        new_a[i], new_a[j] = a[j], a[i]
    return new_a


def make_broad_perturbation(a, rng, sigma=0.2):
    """Apply noise to the entire array (like a low-pass filtered random perturbation)."""
    n = len(a)
    noise = rng.normal(0, sigma, n)
    # Apply a simple low-pass filter (moving average)
    k = max(1, n // 100)
    noise = np.convolve(noise, np.ones(k) / k, mode='same')
    new_a = a + noise * a.std()
    return np.clip(new_a, 0, 1000)

This is equivalent to: minimize `max_k c_k` subject to `c = a*a`, `a >= 0`.

The objective is `2n * max_c / S²`, and since `sum(c) = S²`, we have `max_c / S² = max_c / sum(c)`. So the objective is `2n * max(c) / sum(c)`.

This is a **shape-dependent** quantity (independent of the scale of `a`). So I can normalize `a` to have `sum(a) = 1` without loss of generality.

With `sum(a) = 1`, the objective is `2n * max(c)`. And `sum(c) = 1`, so `max(c) >= 1/(2n-1)`, giving the lower bound `2n/(2n-1)`.

Now, the problem is: find a non-negative sequence `a` of length `n` with `sum(a) = 1` that minimizes `max(a*a)`.

This is a specific optimization problem. Let me think about what algorithms would work well.

**Approach 1: Direct optimization with scipy**

I can use `scipy.optimize.minimize` with a smooth approximation of the max (e.g., soft-max or the log-sum-exp trick).

`soft_max(c, t) = (1/t) * log(sum(exp(t*c)))` approximates `max(c)` as `t → ∞`.

The objective becomes: `2n * (1/t) * log(sum(exp(t * c(a))))` where `c(a) = a*a`.

This is smooth in `a`, and I can use gradient-based methods.

The gradient: `d/d(a[i]) of c(a)[k] = 2*a[k-i]` (as computed before). So:
`d/d(a[i]) of soft_max = (1/t) * sum_k (t*c[k]*exp(t*c[k]))/sum(exp(t*c[j])) * 2*a[k-i]`
`= sum_k (c[k]*exp(t*c[k]))/sum(exp(t*c[j])) * 2*a[k-i]`

This is like a "soft" version of the subgradient, where all `c[k]` contribute (weighted by their softmax weights) instead of just the max.

For large `t`, this approaches the subgradient. For small `t`, it's smoother.

**Approach 2: Iterative reweighting**

Another idea: start with a uniform `a`, and iteratively:
1. Compute `c = a*a`.
2. Compute the "deficit" `d[k] = max(c) - c[k]` (how much each `c[k]` is below the max).
3. Adjust `a` to increase the `c[k]` values that are below the max.

This is like a "water-filling" or "iterative proportional fitting" approach.

from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. g >= 0, f*g <= M (element-wise)."""
    n = len(f)
    
    # Variables: g[0], g[1], ..., g[n-1]
    # Objective: maximize sum(g) → minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_j f[j] * g[k-j] (with valid indices)
    # For each k, the constraint is: sum_j A[k,j] * g[j] <= M
    # where A[k,j] = f[j] if 0 <= k-j < n, else 0
    
    # Build the constraint matrix A (shape: (2n-1, n))
    # A[k, j] = f[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k and 0 <= j < n
    # A[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    # This is a (2n-1) x n matrix, but it's very sparse (each row has at most n non-zeros)
    # Actually, each row has at most min(n, k+1, 2n-1-k) non-zeros
    
    # For efficiency, let's use sparse format
    from scipy.sparse import lil_matrix, csr_matrix
    
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[j]
    
    A = A.tocsr()
    b = np.full(2*n-1, M)
    
    # g >= 0 (bounds)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.zeros(n)
