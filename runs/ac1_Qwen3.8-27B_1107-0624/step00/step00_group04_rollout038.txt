import numpy as np
import time

def compute_convolution(a):
    """Compute full convolution of a with itself using FFT for speed."""
    n = len(a)
    # Need FFT size at least 2n-1
    fft_size = 1
    while fft_size < 2*n - 1:
        fft_size *= 2
    fa = np.fft.rfft(a, n=fft_size)
    ca = np.fft.irfft(fa * fa, n=fft_size)
    return ca[:2*n-1]

def evaluate_local(a):
    """Fast local evaluation matching evaluate_sequence."""
    n = len(a)
    c = compute_convolution(a)
    max_c = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_c / (sum_a ** 2))

def _fft_convolve(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def _soft_subgradient(c, a, n_top=20, temperature=1.0):
    """Compute soft subgradient of max(c) using top-K lags."""
    n = len(a)
    # Get top-K values and their indices
    idx = np.argpartition(c, -n_top)[-n_top:]
    vals = c[idx]
    # Soft weighting (Gumbel-softmax-like)
    weights = np.exp(vals / temperature - np.max(vals) / temperature)
    weights /= weights.sum()
    
    # Compute sg[i] = 2 * sum_k weights[k] * a[k-i]
    sg = np.zeros(n)
    for k, w in zip(idx, weights):
        if w < 1e-12:

        # Valid i: 0 <= k-i < n => k-n+1 <= i <= k, clipped to [0, n-1]

        j = k - np.arange(lo, hi + 1)  # = k-i for i in [lo, hi]
        sg[lo:hi+1] += w * 2 * a[j]

def _fft_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

    """Fast objective: 2n*max(conv(a,a))/(sum a)^2 using FFT."""
    s = a.sum()
    if s < 1e-10:

    return 2.0 * len(a) * float(_fft_conv(a).max()) / (s * s)
