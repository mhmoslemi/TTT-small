"""
Optimization of step function sequences to minimize 2n*max(a*a)/(sum(a))^2.
Uses truncated 1/sqrt(1-x) initialization combined with subgradient descent
and LP-based refinement (cutting plane method).
"""

import numpy as np
from scipy.optimize import linprog
import time


def init_1_over_sqrt(n):
    """Initialize with truncated coefficients of 1/sqrt(1-x)."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    return a


def compute_subgradient(a, top_k=50):
    """Compute subgradient of max(a*a) at a, using top-k tight constraints."""
    n = len(a)
    b = np.convolve(a, a)
    M = b.max()
    
    # Get top-k positions
    if top_k >= len(b):
        tight_mask = b >= M * (1 - 1e-10)
    else:
        # Use a threshold to include all near-max positions
        threshold = M * (1 - 0.01)
        tight_mask = b >= threshold
    
    # Compute subgradient
    U = tight_mask.astype(np.float64)[::-1]
    conv_full = np.convolve(a, U)
    c = conv_full[n-1::-1][:n]
    grad = 2.0 * c
    
    return grad, M, tight_mask


def lp_step(a, M, max_constraints=150):
    """Perform one LP-based refinement step."""
    n = len(a)
    b = np.convolve(a, a)
    
    # Get tight constraints
    threshold = M * 0.95
    tight_positions = np.where(b >= threshold)[0]
    
    if len(tight_positions) > max_constraints:
        # Use the most binding ones
        tight_b = b[tight_positions]
        order = np.argsort(tight_b)[::-1]
        tight_positions = tight_positions[order[:max_constraints]]
    
    if len(tight_positions) == 0:
        return a
    
    # Build constraint matrix
    # (a*g)_k = sum_{j} a[k-j] * g_j <= M
    A_rows = []
    for k in tight_positions:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        row = np.zeros(n)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            row[j_range] = a[k - j_range]
        A_rows.append(row)
    
    A = np.array(A_rows)
    
    # Solve: max sum(g) s.t. A @ g <= M, g >= 0
    # linprog: min c @ x s.t. A_ub @ x <= b_ub, bounds
    c = -np.ones(n)
    b_ub = np.full(len(A_rows), M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.status != 0 or res.x is None:
        return a
    
    g = np.maximum(res.x, 0)
    if g.sum() <= 1e-15:
        return a
    
    return g


def optimize_sequence(a, rng, deadline, use_lp=True, lp_interval=50):
    """Optimize sequence a to minimize max(a*a) / (sum(a))^2."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n)
        S = n
    a = a / S  # Normalize to sum = 1
    
    best_a = a.copy()
    best_score = 2 * n * np.max(np.convolve(a, a))  # With sum=1, score = 2n*max(a*a)
    
    lr = 0.05
    lr_decay = 0.9999
    momentum = np.zeros(n)
    mu = 0.9  # Momentum coefficient
    
    it = 0
    while time.monotonic() < deadline:
        # Subgradient step
        grad, M, tight_mask = compute_subgradient(a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Momentum update
        momentum = mu * momentum + (1 - mu) * grad
        a = a - lr * momentum
        a = np.maximum(a, 0)
        S = a.sum()
        if S > 1e-15:
            a = a / S
        
        lr *= lr_decay
        
        it += 1
        
        # Check improvement
        b = np.convolve(a, a)
        score = 2 * n * b.max()  # sum(a) = 1
        if score < best_score * (1 - 1e-6):
            best_score = score
            best_a = a.copy()
        
        # Periodic LP refinement
        if use_lp and it % lp_interval == 0 and time.monotonic() < deadline - 10:
            b_now = np.convolve(a, a)
            M_now = b_now.max()
            g = lp_step(a, M_now)
            if g is not None and g.sum() > a.sum() * (1 + 1e-6):
                # Blend
                g_norm = g / g.sum()
                t = 0.3
                a = (1 - t) * a + t * g_norm
                a = np.maximum(a, 0)
                S = a.sum()
                if S > 1e-15:
                    a = a / S
                
                # Reset momentum after LP step
                momentum = np.zeros(n)
                lr = max(lr, 0.02)
    
    return best_a

def optimize_sequence(a, rng, deadline):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = np.max(np.convolve(a, a))
    
    # Phase 1: LP-based optimization (most impactful)
    phase1_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.5)
    while time.monotonic() < phase1_deadline:
        b = np.convolve(a, a)
        M = b.max()
        
        # Find tight constraints
        tight_threshold = M * 0.99
        tight_positions = np.where(b >= tight_threshold)[0]
        
        if len(tight_positions) == 0:
            break
        
        # Build LP
        max_cons = min(200, len(tight_positions))
        if len(tight_positions) > max_cons:
            # Keep most binding
            tb = b[tight_positions]
            idx = np.argsort(tb)[::-1][:max_cons]
            tight_positions = tight_positions[idx]
        
        A = np.zeros((len(tight_positions), n))
        for i, k in enumerate(tight_positions):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                js = np.arange(lo, hi + 1)
                A[i, js] = a[k - js]
        
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(len(tight_positions), M),
                     bounds=[(0, None)]*n, method='highs',
                     options={'presolve': True, 'time_limit': 20})
        
        if res.status != 0:
            break
        
        g = np.maximum(res.x, 0)
        S_g = g.sum()
        if S_g <= 1 + 1e-10:
            break  # No improvement
        
        # Update
        g = g / S_g
        t = 0.5  # Blending parameter
        a = (1 - t) * a + t * g
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        # Check
        M_new = np.max(np.convolve(a, a))
        if M_new < best_M * (1 - 1e-8):
            best_M = M_new
            best_a = a.copy()
    
    # Phase 2: Subgradient refinement
    lr = 0.01
    momentum = np.zeros(n)
    mu = 0.8
    while time.monotonic() < deadline:
        grad, M = compute_subgradient(a)
        grad_norm = np.linalg.norm(grad)

        a = np.maximum(a, 0)

    x = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for f, p, a_ in zip(freqs, phases, amps):
        a += a_ * np.cos(2 * np.pi * f * x + p)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def compute_subgradient(a):
    """Compute subgradient of max_k (a*a)_k w.r.t. a.
    
    Returns (grad, M) where grad is the subgradient and M is the max value.
    """
    n = len(a)
    b = np.convolve(a, a)
    M = b.max()
    
    # Tight mask: positions where b is close to M
    threshold = M * (1.0 - 1e-9)
    T = (b >= threshold).astype(np.float64)
    
    # Compute c[i] = sum_j a[j] * T[i+j] using convolution
    a_rev = a[::-1]
    conv_full = np.convolve(a_rev, T)
    # c[i] = conv_full[n-1+i] for i=0,...,n-1
    c = conv_full[n - 1:2 * n - 1]
    
    # Gradient of max is 2 * c (subgradient using all tight constraints equally)
    grad = 2.0 * c
    return grad, M


def lp_step(a, M, max_constraints=200, time_limit=15):
    """Perform one LP-based refinement step.
    
    Solves: max sum(g) s.t. (a*g)_k <= M for tight k, g >= 0.
    Returns the optimal g, or None if LP fails.
    """
    n = len(a)
    b = np.convolve(a, a)
    
    # Find tight constraints (within 1% of M)
    tight_threshold = M * 0.99
    tight_positions = np.where(b >= tight_threshold)[0]
    
    if len(tight_positions) == 0:
        return None
    
    # Limit number of constraints
    if len(tight_positions) > max_constraints:
        tb = b[tight_positions]
        idx = np.argsort(tb)[::-1][:max_constraints]
        tight_positions = tight_positions[idx]
    
    # Build constraint matrix: row i corresponds to constraint (a*g)_k <= M
    # (a*g)_k = sum_j a[k-j] * g_j for j in [max(0,k-n+1), min(k,n-1)]
    A = np.zeros((len(tight_positions), n), dtype=np.float64)
    for i, k in enumerate(tight_positions):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            A[i, js] = a[k - js]
    
    # Solve: min -sum(g) s.t. A@g <= M, g >= 0
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(len(tight_positions), M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.maximum(res.x, 0.0)
    if g.sum() < 1e-15:
        return None
    
    return g


def optimize_sequence(a, deadline, use_lp=True, lp_interval=20):
    """Optimize a sequence to minimize max(a*a) / (sum(a))^2.
    
    Uses alternating LP refinement and subgradient descent with momentum.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n)
        S = n
    a = a / S  # Normalize to sum = 1
    
    b_init = np.convolve(a, a)
    best_a = a.copy()
    best_M = b_init.max()
    
    # Phase 1: LP-based optimization
    if use_lp:
        phase1_deadline = min(deadline, time.monotonic() + 0.4 * (deadline - time.monotonic()))
        lp_count = 0
        while time.monotonic() < phase1_deadline:
            b = np.convolve(a, a)
            M = b.max()
            
            g = lp_step(a, M, max_constraints=200, time_limit=10)
            if g is None:
                break
            
            S_g = g.sum()
            if S_g <= 1.0 + 1e-8:
                # No improvement from LP
                break
            
            # Blend update
            g_norm = g / S_g
            t = 0.4
            a = (1.0 - t) * a + t * g_norm
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S > 1e-15:
                a = a / S
            
            lp_count += 1
            
            # Check improvement
            M_new = np.max(np.convolve(a, a))
            if M_new < best_M * (1.0 - 1e-7):
                best_M = M_new
                best_a = a.copy()
            
            # Reset if no progress after several LPs
            if lp_count > 3 and M_new >= best_M * (1.0 - 1e-6):
                # Try with fewer constraints for diversity
                pass
    
    # Phase 2: Subgradient refinement with momentum
    lr = 0.03
    lr_min = 1e-4
    momentum = np.zeros(n)
    mu = 0.85
    stuck_count = 0
    
    while time.monotonic() < deadline:
        grad, M = compute_subgradient(a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        else:
            break
        
        # Momentum update
        momentum = mu * momentum + (1.0 - mu) * grad
        a = a - lr * momentum
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S > 1e-15:
            a = a / S
        
        lr = max(lr * 0.9995, lr_min)
        
        # Periodically try LP in phase 2
        if use_lp and stuck_count > 50 and time.monotonic() < deadline - 5:
            b_now = np.convolve(a, a)
            M_now = b_now.max()
            g = lp_step(a, M_now, max_constraints=100, time_limit=8)
            if g is not None and g.sum() > 1.0 + 1e-6:
                g_norm = g / g.sum()
                t = 0.3
                a = (1.0 - t) * a + t * g_norm
                a = np.maximum(a, 0.0)
                S = a.sum()
                if S > 1e-15:
                    a = a / S
                momentum = np.zeros(n)
                lr = max(lr, 0.02)
                stuck_count = 0
        
        M_new = np.max(np.convolve(a, a))
        if M_new < best_M * (1.0 - 1e-7):
            best_M = M_new
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Reduce momentum if stuck
        if stuck_count > 200:
            mu = max(mu * 0.99, 0.5)
            lr = max(lr * 1.01, lr_min)
    
    return best_a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for the best sequence of non-negative heights minimizing the evaluation."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 10.0)
    
    best_score = np.inf
    best_seq = None
    
    # Determine lengths to try
    lengths_to_try = set()
    
    # Use height_sequence_1 length if available
    try:
        n1 = len(height_sequence_1)
        if 50 <= n1 <= 30000:
            lengths_to_try.add(n1)
    except (NameError, TypeError):
        pass
    
    # Add standard lengths
    for n in [3000, 5000, 6000, 8000, 10000]:
        lengths_to_try.add(n)
    
    lengths_to_try = sorted(lengths_to_try)
    
    # Allocate time: give more time to lengths near 6000-8000
    n_lengths = len(lengths_to_try)
    total_budget = max(60.0, float(budget_s) - 30.0)
    
    # Weight lengths by preference
    weights = []
    for n in lengths_to_try:
        w = 1.0 / (1.0 + abs(n - 7000) / 5000.0)
        weights.append(w)
    total_w = sum(weights)
    weights = [w / total_w for w in weights]
    
    time_allocs = []
    remaining_time = total_budget
    for i, n in enumerate(lengths_to_try):
        t_alloc = weights[i] * total_budget
        if i == n_lengths - 1:
            t_alloc = remaining_time
        time_allocs.append(max(30.0, t_alloc))
        remaining_time -= t_alloc
    
    # Try each length with different initializations
    for idx, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline:
            break
        
        length_deadline = min(deadline, time.monotonic() + time_allocs[idx])
        
        # Initialization 1: 1/sqrt(1-x)
        a = init_1_over_sqrt(n)
        a_opt = optimize_sequence(a, length_deadline, use_lp=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"n={n}, init=1/sqrt, score={score:.6f}, time={time.monotonic()-start_time:.1f}s")
        
        # Initialization 2: symmetric 1/sqrt(1-x)
        if time.monotonic() < length_deadline - 30:
            a2 = init_symmetric_1_over_sqrt(n)
            sym_deadline = min(length_deadline, time.monotonic() + (length_deadline - time.monotonic()) * 0.6)
            a2_opt = optimize_sequence(a2, sym_deadline, use_lp=True)
            score2 = float(evaluate_sequence(a2_opt.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_seq = a2_opt.copy()
                print(f"n={n}, init=sym, score={score2:.6f}, time={time.monotonic()-start_time:.1f}s")
        
        # Initialization 3: perturbed 1/sqrt(1-x)
        if time.monotonic() < length_deadline - 30:
            a3 = init_1_over_sqrt(n)
            a3 *= np.exp(rng.normal(0, 0.1, size=n))
            a3 = np.maximum(a3, 0.0)
            a3_deadline = min(length_deadline, time.monotonic() + (length_deadline - time.monotonic()) * 0.5)
            a3_opt = optimize_sequence(a3, a3_deadline, use_lp=True)
            score3 = float(evaluate_sequence(a3_opt.tolist()))
            if np.isfinite(score3) and score3 < best_score:
                best_score = score3
                best_seq = a3_opt.copy()
                print(f"n={n}, init=perturbed, score={score3:.6f}, time={time.monotonic()-start_time:.1f}s")
    
    # Try optimizing height_sequence_1 directly if time remains
    if time.monotonic() < deadline - 60:
        try:
            h1 = np.asarray(height_sequence_1, dtype=np.float64)
            h1 = np.maximum(h1, 0.0)
            if h1.size > 50:
                h1_opt = optimize_sequence(h1, deadline, use_lp=True)
                score = float(evaluate_sequence(h1_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = h1_opt.copy()
                    print(f"h1 optimized, score={score:.6f}, time={time.monotonic()-start_time:.1f}s")
        except (NameError, TypeError):
            pass
    
    # Fallback
    if best_seq is None:
        n_fb = 6000
        best_seq = init_1_over_sqrt(n_fb)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback: n={n_fb}, score={best_score:.6f}")
    
    print(f"FINAL: best_score={best_score:.6f}, length={len(best_seq)}, "
          f"total_time={time.monotonic()-start_time:.1f}s")
    
    return [float(x) for x in best_seq]
