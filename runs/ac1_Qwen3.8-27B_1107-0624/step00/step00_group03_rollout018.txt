Gradient descent with soft-max smoothing for minimizing the peak autoconvolution
score of a non-negative sequence. Uses FFT-based gradient computation, momentum,
adaptive learning rate, and multiple restarts with diverse initializations.
"""

import numpy as np
import time


def compute_score_grad_fft(s, T):
    """Compute soft score and gradient using FFT."""
    n = len(s)
    b = np.convolve(s, s)
    sum_a = np.sum(s)
    if sum_a < 1e-10:
        return np.inf, np.zeros(n)
    
    b_max = np.max(b)
    scaled = T * (b - b_max)
    # Clip to avoid underflow/overflow
    scaled = np.clip(scaled, -500, 500)
    weights = np.exp(scaled)
    w_sum = np.sum(weights)
    if w_sum == 0 or not np.isfinite(w_sum):
        # Fallback to hard max
        weights = np.zeros(len(b))
        weights[np.argmax(b)] = 1.0
        soft_max_val = b_max
    else:
        weights /= w_sum
        soft_max_val = b_max + np.log(w_sum) / T
    
    score = 2.0 * n * soft_max_val / (sum_a ** 2)
    
    if n == 1:
        return score, np.array([2.0 * n * b[0] / (sum_a ** 2) - 4.0 * n * b[0] / (sum_a ** 3)])
    
    # FFT-based gradient
    L = 1
    while L < 3 * n - 2:
        L *= 2
    
    s_rev = s[::-1]
    W = np.fft.rfft(weights, L)
    Sr = np.fft.rfft(s_rev, L)
    C_full = np.fft.irfft(W * Sr, L)
    grad_b = 2.0 * C_full[n-1 : 2*n-1]
    
    grad = (2.0 * n * grad_b) / (sum_a ** 2) - (4.0 * n * soft_max_val) / (sum_a ** 3)
    
    return score, grad
