Asymmetric multiplicative weight descent with simulated annealing for minimizing
2n * max(convolve(a,a)) / sum(a)^2. Key insight: symmetric sequences are stuck
at objective >= 2 by Cauchy-Schwarz; asymmetric sequences can approach ~1.5.
Uses FFT-based convolution and cross-correlation for fast pressure computation.

import numpy as np
import time

def next_pow2(x):
    """Next power of 2 >= x."""
    p = 1
    while p < x:
        p *= 2
    return p


def fast_objective(seq):
    """Compute 2n * max(convolve(seq,seq)) / sum(seq)^2 using FFT."""
    n = len(seq)
    if n == 0:
        return np.inf
    total = np.sum(seq)
    if total < 0.01:
        return np.inf
    size = next_pow2(2 * n - 1)
    F = np.fft.rfft(seq, size=size)
    conv = np.fft.irfft(F * F, n=size)[:2 * n - 1]
    max_conv = np.max(conv)
    if max_conv < 0:
        max_conv = 0.0
    return float(2 * n * max_conv / (total ** 2))


def compute_convolution(seq):
    """Compute full convolution via FFT."""
    n = len(seq)
    size = next_pow2(2 * n - 1)
    F = np.fft.rfft(seq, size=size)
    return np.fft.irfft(F * F, n=size)[:2 * n - 1]


def compute_pressure(f, peaks, n):
    """
    Compute pressure[i] = sum_{k in peaks} f[k-i] for valid indices.
    Uses cross-correlation via FFT.
    """
    peak_ind = np.zeros(2 * n - 1, dtype=np.float64)
    peak_ind[peaks] = 1.0
    size = next_pow2(3 * n)
    F = np.fft.rfft(f, size=size)
    G = np.fft.rfft(peak_ind, size=size)
    # cross-correlation: ifft(conj(FFT(f)) * FFT(g))
    corr = np.fft.irfft(np.conj(F) * G, n=size)
    return corr[:n]
