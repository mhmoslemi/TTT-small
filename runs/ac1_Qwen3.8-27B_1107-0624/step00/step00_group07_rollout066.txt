"""
Minimize  2*n*max_k (p*p)[k] / (sum p)^2  for a non-negative step height sequence
by treating it (after normalizing sum p = 1) as: minimize the peak of the linear
self-convolution c[k] = sum_i p[i]*p[k-i].

Method (gradient / mirror descent on the active peak):
  * For the active lag k*,  d c[k*]/d p[j] = 2*p[k*-j]  when 0<=k*-j<n else 0.
    This is the reflection of p across k*/2 (zero outside the support).
  * Center that gradient on its support to project onto the simplex tangent.
  * Multiplicative (mirror) update  p <- p * exp(-alpha * grad_centred),
    then renormalize to sum 1  ->  stays non-negative and on the simplex.
  * Adaptive step size, periodic switching among top peaks to equalize them,
    and perturbation-based restarts to escape local minima.
  * Several structured initial conditions (uniform, U-shape, two-bump,
    Gaussian, symmetric-noise) across a few FFT-friendly lengths are tried;
    the global best is returned. Fast convolutions use the FFT.

Fallback guarantees a valid non-negative sequence with sum >= 0.01.
"""

import time
import numpy as np


def _convolve_max_fft(p):
    """Return (max_k c[k], argmax) for the linear self-convolution of p."""
    n = len(p)
    N = 2 * n - 1
    P = np.fft.fft(p, N)
    C = np.fft.ifft(P * P).real
    return float(C.max()), int(np.argmax(C))


def _top_peaks(p, m):
    """Indices (in c-space) of the m largest convolution peaks."""
    n = len(p)
    N = 2 * n - 1
    P = np.fft.fft(p, N)
    C = np.fft.ifft(P * P).real
    return np.argsort(C)[::-1][:m]


def _peak_peel(p, deadline, rng, alpha0=40.0):
    """
    Mirror-descent peak reduction.  Returns (best_p_normalized, best_score).
    p is assumed non-negative; it is normalized internally to sum 1.
    """
    n = len(p)
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.clip(p, 0.0, None)
    s = p.sum()
    if s <= 1e-12:
        p = np.ones(n, dtype=np.float64) / n
    else:
        p = p / s

    peak, k_star = _convolve_max_fft(p)
    best_peak = peak
    best_p = p.copy()

    alpha = float(alpha0)
    patience = 0
    max_patience = 4000
    step = 0
    n_valid = n  # fallback window length

    while time.monotonic() < deadline:
        step += 1

        # Active lag support
        valid_lo = max(0, k_star - n + 1)
        valid_hi = min(k_star, n - 1)
        if valid_lo > valid_hi:
            valid_lo, valid_hi = 0, n - 1

        j_range = np.arange(valid_lo, valid_hi + 1)
        ref = k_star - j_range
        ref = np.clip(ref, 0, n - 1)

        grad = np.zeros(n, dtype=np.float64)
        grad[j_range] = 2.0 * p[ref]

        # Center on the support  ->  tangent of the simplex
        g_mean = grad[j_range].mean()
        grad[j_range] -= g_mean

        # Mirror (multiplicative) descent update
        exp_arg = -alpha * grad
        exp_arg -= exp_arg.max()
        w = p * np.exp(exp_arg)
        tw = w.sum()
        if tw <= 1e-300:
            break
        p = w / tw
        p = np.clip(p, 0.0, None)
        p = p / p.sum()

        new_peak, new_k = _convolve_max_fft(p)

        if new_peak < best_peak:
            best_peak = new_peak
            best_p = p.copy()
            patience = 0
            alpha = min(alpha * 1.03, 500.0)
            if step % 2000 == 0:
                print("  step %d  best_peak=%.8f  score=%.6f" %
                      (step, best_peak, 2.0 * n * best_peak))
        else:
            patience += 1
            if patience > 150:
                alpha = max(alpha * 0.9, 1.0)
                patience = 0

        peak = new_peak
        k_star = new_k

        # Periodically retarget a different (top) peak to equalize the plateau
        if step % 1200 == 600:
            tops = _top_peaks(p, 16)
            k_star = int(tops[rng.integers(len(tops))])
            patience = min(patience, 200)

        # Escape: perturb from the current best
        if patience > max_patience:
            noise = rng.normal(0.0, 3.0 / n, n)
            p = best_p + noise
            p = np.clip(p, 0.0, None)
            ps = p.sum()
            if ps <= 1e-12:
                p = best_p.copy()
                ps = 1.0
            p = p / ps
            peak, k_star = _convolve_max_fft(p)
            patience = 0
            alpha = float(alpha0)

        # Occasionally a stronger global shake
        if step % 6000 == 3000:
            m = max(1, n // 40)
            idx = rng.choice(n, size=m, replace=False)
            p[idx] = rng.random(m)
            p = np.clip(p, 0.0, None)
            ps = p.sum()
            if ps <= 1e-12:
                p = best_p.copy()
            else:
                p = p / ps
            peak, k_star = _convolve_max_fft(p)
            patience = 0

    return best_p, 2.0 * n * best_peak


def _make_init(n, kind, rng):
    i = np.arange(n, dtype=np.float64)
    c = 0.5 * (n - 1)
    if kind == 0:            # uniform
        return np.ones(n, dtype=np.float64)
    if kind == 1:            # U-shape (anti-central)
        return 1.0 + 0.5 * np.cos(2.0 * np.pi * i / n)
    if kind == 2:            # two bumps at the quarter points
        v = np.zeros(n, dtype=np.float64)
        v[i < 0.25 * n] = 1.0
        v[i >= 0.75 * n] = 1.0
        v += 0.05
        return v
    if kind == 3:            # central Gaussian
        return np.exp(-0.5 * ((i - c) / (0.18 * n)) ** 2) + 1e-3
    if kind == 4:            # symmetric noise
        r = rng.random(n)
        return 0.5 * (r + r[::-1]) + 0.05
    if kind == 5:            # faint comb / structured spokes
        k = max(1, n // 512)
        return (1.0 - 0.4 * np.cos(2.0 * np.pi * i / k)) + 0.05
    return np.ones(n, dtype=np.float64)
