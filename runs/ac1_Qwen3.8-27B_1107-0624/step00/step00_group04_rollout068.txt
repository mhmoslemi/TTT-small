import numpy as np
import time

def fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

    """Compute objective given convolution, length, and sum."""
    M = conv.max()
    return 2.0 * n * M / (S * S)

def refine(f, budget_s, rng, verbose=True):
    """Iteratively refine f to minimize the objective."""
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f = f / S  # normalize to sum = 1
    
    best_f = f.copy()
    conv = fft_convolve(f, f)

    eta = 0.05
    patience = 500
    imp_count = 0

        # For each peak position k, d(conv[k])/df[i] = f[k-i] (if valid) + f[i-k] (if valid, i≠k-i)
        # Simplified: the "pressure" on f[i] is sum over peaks k of f[|k-i|] (approximately)

        peak_threshold = 0.95 * M
        peaks = np.where(conv >= peak_threshold)[0]

            break

        grad = np.zeros(n)
        for k in peaks:

            # f[k-i] for i in [lo, hi]

        obj = objective_from_conv(conv, n, 1.0)
