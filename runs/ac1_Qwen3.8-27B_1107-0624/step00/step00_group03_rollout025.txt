def compute_gradient(a, w):
    """Compute grad[i] = 2 * sum_j a[j] * w[j+i] for i=0..n-1"""
    n = len(a)
    # Pad to size >= n + (2n-1) - 1 = 3n-2
    size = 1
    while size < 3*n:
        size *= 2
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    # cross-corr(a, w)[i] = IFFT(A * conj(W))[i]
    result = np.fft.irfft(A * np.conj(W), size)
