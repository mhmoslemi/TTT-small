    return int(2 ** np.ceil(np.log2(n)))

    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)

    """Compute g[i] = sum_k w[k]*a[k-i] using FFT."""
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    return np.fft.irfft(np.conj(A) * W, N)

def _smoothed_gradient_step(a, t, lr):
    """
    One step of smoothed gradient descent.
    a: current sequence (sum=1, non-negative)
    t: temperature for softmax
    lr: learning rate
    Returns: new a, current objective value
    """
    n = len(a)
    N_conv = _next_pow2(2 * n - 1)
    N_corr = _next_pow2(3 * n - 2)
    
    # Compute convolution
    A = np.fft.rfft(a, N_conv)
    b = np.fft.irfft(A * A, N_conv)[:2*n-1]
    
    # Compute softmax weights
    b_max = np.max(b)
    exp_b = np.exp(t * (b - b_max))
    w = exp_b / np.sum(exp_b)
    
    # Compute gradient
    A2 = np.fft.rfft(a, N_corr)
    W = np.fft.rfft(w, N_corr)

    g_proj = g - np.mean(g)

    else:

minimize f(a) = max_k b[k] (smoothed as log-sum-exp)
subject to: sum(a) = 1, a >= 0

This is a convex problem (f is convex as a max of linear functions, or as a log-sum-exp).

FISTA would give O(1/k^2) convergence rate, which is better than O(1/k) for vanilla gradient descent.

But the constraint set {a: sum(a) = 1, a >= 0} is a simplex, and the projection onto the simplex is O(n) (sort the components). So FISTA is feasible.

Actually, the projection onto {a: sum(a) = 1, a >= 0} (the probability simplex) is:
1. Compute a - lr * g
2. Project onto simplex: find threshold θ such that max(0, a_i - lr*g_i - θ) sums to 1
3. This requires sorting, which is O(n log n)

But this is the same as what I was doing (normalize after clipping). The difference is that FISTA uses an extrapolation step:
y_k = x_k + (k-1)/(k+2) * (x_k - x_{k-1})
x_{k+1} = proj(y_k - lr * grad(y_k))

This acceleration can significantly speed up convergence.
