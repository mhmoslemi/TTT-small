"""
Optimizes non-negative step function sequences to minimize 2n*max(f*f)/(sum(f))^2.
Uses chirp-based initializations with projected gradient descent, multiple
lengths, and adaptive step sizes to beat the constant-sequence score of 2.0.
"""

import numpy as np
import time


def internal_score(f):
    """Fast score computation."""
    n = len(f)
    conv = np.convolve(f, f)
    M = np.max(conv)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    return float(2.0 * n * M / (S * S))


def make_chirp(n, alpha, beta):
    """Generate chirp-like non-negative sequence."""
    i = np.arange(n, dtype=np.float64)
    f = 1.0 + alpha * np.cos(2.0 * np.pi * beta * i * i / n)
    return np.clip(f, 0.01, 1000.0)


def make_modulated(n, alpha, freq, phase):
    """Generate modulated cosine sequence."""
    i = np.arange(n, dtype=np.float64)
    f = 1.0 + alpha * np.cos(2.0 * np.pi * freq * i / n + phase)
    return np.clip(f, 0.01, 1000.0)


def make_squared_cos(n, freq):
    """Generate cos^2 based sequence (always non-negative)."""
    i = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * freq * i / n) ** 2
    return np.clip(f, 0.01, 1000.0)


def make_random_nonneg(n, seed):
    """Random non-negative initialization."""
    rng = np.random.default_rng(seed)
    f = rng.random(n) * 0.5 + 0.5
    return f


def compute_gradient(f, top_k=5):
    """Compute smoothed subgradient of the objective."""
    n = len(f)
    conv = np.convolve(f, f)
    M = np.max(conv)
    S = np.sum(f)
    if S < 0.01:
        return None, None, None

    k = min(top_k, len(conv))
    K_star = np.argpartition(conv, -k)[-k:]

    grad = np.zeros(n, dtype=np.float64)
    for ki in K_star:
        ki = int(ki)
        i_s = max(0, ki - n + 1)
        i_e = min(n - 1, ki)
        if i_s <= i_e:
            grad[i_s:i_e + 1] += f[ki - i_e:ki - i_s + 1][::-1]

    grad *= 4.0 * n / (S * S * len(K_star))
    grad -= 4.0 * n * M / (S * S * S)
    return grad, M, S


def gradient_descent(f, budget_s, seed=0, top_k=5, verbose=False):
    """
    Projected gradient descent on the sequence.
    Returns (best_f, best_score).
    """
    rng = np.random.default_rng(seed)
    n = len(f)
    f = f.copy().astype(np.float64)
    np.clip(f, 0.0, 1000.0, out=f)

    best_f = f.copy()
    best_score = internal_score(f)
    base_eta = 0.5
    eta = base_eta
    deadline = time.monotonic() + budget_s
    iters = 0
    stuck = 0

    while time.monotonic() < deadline:
        grad, M, S = compute_gradient(f, top_k)
        if grad is None:
            f = np.ones(n) * 0.1
            continue

        f_new = f - eta * grad
        np.clip(f_new, 0.0, 1000.0, out=f_new)

        sc = internal_score(f_new)
        if sc < best_score:
            best_f = f_new.copy()
            best_score = sc
            f = f_new
            stuck = 0
            eta = max(0.01, eta * 0.9995)
        else:
            stuck += 1
            eta *= 0.85
            if stuck > 30:
                # Random perturbation to escape local minimum
                scale = 0.1 * np.mean(f)
                f = f + rng.normal(0, scale, n)
                np.clip(f, 0, 1000, out=f)
                eta = base_eta
                stuck = 0
                if verbose:
                    print(f"  Perturbed at iter {iters}, score={best_score:.6f}")

        iters += 1
        if verbose and iters % 5000 == 0:
            print(f"  iter {iters}, score={best_score:.6f}, eta={eta:.4f}")

    return best_f, best_score


def block_perturb_local_search(f, budget_s, seed=0):
    """
    Local search with block multiplicative perturbations.
    Useful for polishing.
    """
    rng = np.random.default_rng(seed)
    n = len(f)
    f = f.copy().astype(np.float64)
    np.clip(f, 0.0, 1000.0, out=f)
    best_f = f.copy()
    best_score = internal_score(f)
    deadline = time.monotonic() + budget_s

    while time.monotonic() < deadline:
        candidate = f.copy()
        n_blocks = rng.integers(1, 5)
        for _ in range(n_blocks):
            width = int(rng.integers(max(2, n // 100), max(3, n // 10)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate[start:stop] *= np.exp(rng.normal(0, 0.15))
        np.clip(candidate, 0, 1000, out=candidate)
        sc = internal_score(candidate)
        if sc < best_score:
            best_f = candidate.copy()
            best_score = sc
            f = candidate

    return best_f, best_score


def try_sequence(n, init_type, params, budget_s, seed, verbose=False):
    """Try a specific initialization and optimize."""
    if init_type == "chirp":
        f = make_chirp(n, params["alpha"], params["beta"])
    elif init_type == "modulated":
        f = make_modulated(n, params["alpha"], params["freq"], params["phase"])
    elif init_type == "squared_cos":
        f = make_squared_cos(n, params["freq"])
    elif init_type == "random":
        f = make_random_nonneg(n, seed)
    elif init_type == "ones":
        f = np.ones(n)
    else:
        f = np.ones(n)

    sc0 = internal_score(f)
    if verbose:
        print(f"  n={n}, type={init_type}, params={params}, initial score={sc0:.6f}")

    if sc0 < 1.8:
        f, sc = gradient_descent(f, budget_s, seed=seed, top_k=7, verbose=verbose)
    else:
        f, sc = gradient_descent(f, budget_s * 0.7, seed=seed, top_k=5, verbose=verbose)
        # Polish with local search
        f, sc2 = block_perturb_local_search(f, budget_s * 0.3, seed=seed + 1)
        if sc2 < sc:
            sc = sc2

    return f, sc
