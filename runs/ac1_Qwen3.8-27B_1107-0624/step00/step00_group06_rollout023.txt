    max_c = c.max()
    threshold = max_c * (1 - eps_frac)
    T = (c >= threshold).astype(np.float64)
    count = T.sum()
    if count == 0:
        T = np.zeros(2*n-1)
        T[np.argmax(c)] = 1.0
