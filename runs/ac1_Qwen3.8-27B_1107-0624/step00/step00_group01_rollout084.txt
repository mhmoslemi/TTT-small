def fft_linear_convolution(a):
    n = a.size
    if n < 256:
       return np.convolve(a,a)
    size = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(a, size)

```python
"""
Search for a non-negative coefficient sequence minimizing
    2*n*max(conv(a,a)) / (sum(a))^2.

The program uses:
  - FFT-based proxy scoring for many candidates,
  - piecewise-constant profile parameterization with mirror-style updates,
  - several non-symmetric periodic/step initializations,
  - full sequence multiplicative mirror descent with top-lag and power weights,
  - a final exact evaluation of a few shortlisted candidates.

The returned sequence is normalized so that its sum is 1, keeping values small
and safe for the evaluator.
"""

import numpy as np
import time


def fft_linear_convolution(a):
    """Linear convolution a*a using FFT for fairly large arrays."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n <= 256:
        return np.convolve(a, a)
    need = 2 * n - 1
    size = 1 << int(need.bit_length())
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)
    return b[:need]


def fast_score(a):
    """Fast scale-invariant proxy score: 2*n*max(conv)/(sum^2)."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")
    if not np.all(np.isfinite(a)):
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return float("inf")
    b = fft_linear_convolution(a)
    if b.size == 0 or not np.all(np.isfinite(b)):
        return float("inf")
    mb = float(np.max(b))
    if mb < 0.0 and mb > -1e-12:
        mb = 0.0
    if not np.isfinite(mb) or mb < 0.0:
        return float("inf")
    return float(2.0 * n * mb / (s * s))


def positive_log_to_array(x, max_exp=10.0):
    """Convert a log-height vector to a positive probability-like array."""
    x = np.asarray(x, dtype=np.float64)
    if x.ndim != 1 or x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = x - float(np.mean(x))
    x = np.clip(x, -float(max_exp), float(max_exp))
    y = np.exp(x)
    y = np.where(np.isfinite(y), y, 0.0)
    s = float(np.sum(y))
    if not np.isfinite(s) or s <= 0.0:
        return None
    return y / s


def center_clip_log(x):
    """Center and clip a log vector for numerical stability."""
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = x - float(np.mean(x))
    return np.clip(x, -15.0, 15.0)


def log_from_positive_h(h):
    """Convert positive heights to centered log heights."""
    h = np.asarray(h, dtype=np.float64)
    h = np.where(np.isfinite(h), h, 1.0)
    h = np.maximum(h, 1e-3)
    s = float(np.sum(h))
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros_like(h, dtype=np.float64)
    h = h / s
    return np.log(h)


def sequence_from_log_heights(x, seg_len, max_exp=10.0):
    """Expand segment log-heights into one full sequence."""
    h = positive_log_to_array(x, max_exp=max_exp)
    if h is None:
        return None
    rep = int(seg_len)
    if rep <= 0:
        return None
    a = np.repeat(h, rep)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a / s


def segment_objective(x, M, seg_len, max_exp=10.0):
    """Objective for a segment-height log vector."""
    a = sequence_from_log_heights(x, seg_len, max_exp=max_exp)
    if a is None:
        return float("inf")
    m = int(M)
    rep = int(seg_len)
    if a.size != m * rep:
        return float("inf")
    sc = fast_score(a)
    if not np.isfinite(sc):
        return float("inf")
    return float(sc)


def gradient_hankel(a, w):
    """
    For F = sum_k w_k * (a*a)_k, return unnormalized gradient vector wrt a.
    This is g_i = sum_j a_j * w_{i+j}.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size != 2 * n - 1:
        return None
    if n <= 256:
        g = np.empty(n, dtype=np.float64)
        for i in range(n):
            g[i] = float(np.dot(w[i:i + n], a))
        g = np.maximum(g, 0.0)
        return g
    need = 2 * n - 1
    size = 1 << int(need.bit_length())
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w[::-1], size)
    h = np.fft.irfft(fa * fw, size)[:need]
    g = h[need - 1 :: -1][:n]
    g = np.maximum(g, 0.0)
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def power_weights(b, power):
    """Weights proportional to (b / max(b))^power."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return None
    mb = float(np.max(b))
    if not np.isfinite(mb) or mb <= 0.0:
        return None
    x = b / mb
    x = np.clip(x, 0.0, 1.0)
    w = np.power(x, float(power))
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 0.0:
        return None
    return w / s


def top_weights(b, frac):
    """Smooth ramp weights on the top frac fraction of convolution lags."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return None
    mb = float(np.max(b))
    if not np.isfinite(mb) or mb <= 0.0:
        return None
    frac = float(np.clip(frac, 1e-6, 1.0))
    q = float(np.quantile(b, 1.0 - frac))
    rngv = mb - q
    if not np.isfinite(rngv) or rngv <= 1e-300:
        w = np.where(b >= mb, 1.0, 0.0)
    else:
        w = np.clip((b - q) / rngv, 0.0, 1.0)
    w = np.where(np.isfinite(w), w, 0.0)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        s = float(b.size)
    return w / s


def random_log_block_candidate(a, rng):
    """Random multiplicative perturbation of a positive sequence."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n <= 2:
        return None
    if n <= 64:
        width = int(rng.integers(1, n + 1))
    else:
        min_w = max(1, n // 2000)
        max_w = max(min_w + 1, n // 50)
        width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    sigma = float(rng.uniform(0.03, 0.25))
    loga = np.log(a + 1e-300)
    loga[start:start + width] += rng.normal(0.0, sigma, size=width)
    m = np.max(loga)
    if not np.isfinite(m):
        return None
    loga = loga - m
    loga = np.clip(loga, -50.0, 0.0)
    cand = np.exp(loga)
    s = float(np.sum(cand))
    if not np.isfinite(s) or s <= 0.0:
        return None
    return cand / s


def bin_to_log_heights(a, M):
    """Approximate a sequence by M segment log-heights."""
    a = np.asarray(a, dtype=np.float64)
    M = int(max(2, M))
    if a.size == 0:
        return np.zeros(M, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    if a.size <= M:
        if a.size == 1:
            return np.zeros(M, dtype=np.float64)
        old = np.arange(a.size, dtype=np.float64)
        new = np.linspace(0.0, float(a.size - 1), M, dtype=np.float64)
        vals = np.interp(new, old, a)
        vals = np.maximum(vals, 1e-15)
        return log_from_positive_h(vals)
    chunks = np.array_split(a, M)
    vals = np.empty(M, dtype=np.float64)
    for i in range(M):
        vals[i] = max(float(np.mean(chunks[i])), 1e-15)
    return log_from_positive_h(vals)


def initial_log_heights(M, kind, rng):
    """Create a diverse set of non-symmetric piecewise-constant initial profiles."""
    M = int(max(4, M))
    pos = np.linspace(0.0, 1.0, M, endpoint=False)
    h = np.ones(M, dtype=np.float64)

    if kind == 0:
        return np.zeros(M, dtype=np.float64)

    if kind == 1:
        h[:M // 2] = 2.0 / 3.0
        h[M // 2:] = 4.0 / 3.0
        return log_from_positive_h(h)

    if kind == 2:
        h[:M // 2] = 4.0 / 3.0
        h[M // 2:] = 2.0 / 3.0
        return log_from_positive_h(h)

    if kind == 3:
        h[:M // 2] = 0.5
        h[M // 2:] = 1.5
        return log_from_positive_h(h)

    if kind == 4:
        h[:M // 2] = 1.5
        h[M // 2:] = 0.5
        return log_from_positive_h(h)

    if kind == 5:
        z = 1.0 * np.cos(2.0 * np.pi * pos + 2.0943951023931953)
        z -= float(np.mean(z))
        h = np.exp(np.clip(z, -6.0, 6.0))
        return log_from_positive_h(h)

    if kind == 6:
        z = 1.8 * np.cos(2.0 * np.pi * pos + 2.0943951023931953)
        z -= float(np.mean(z))
        h = np.exp(np.clip(z, -6.0, 6.0))
        return log_from_positive_h(h)

    if kind == 7:
        z = 1.2 * np.cos(2.0 * np.pi * pos + 1.5707963267948966)
        z -= float(np.mean(z))
        h = np.exp(np.clip(z, -6.0, 6.0))
        return log_from_positive_h(h)

    if kind == 8:
        z = 0.8 * np.cos(2.0 * np.pi * pos + 2.5)
        z -= float(np.mean(z))
        h = np.exp(np.clip(z, -6.0, 6.0))
        return log_from_positive_h(h)

    if kind == 9:
        h = 1.0 + 0.7 * np.sin(2.0 * np.pi * pos + 1.0)
        h = np.maximum(h, 0.05)
        return log_from_positive_h(h)

    if kind == 10:
        k1 = M // 3
        k2 = 2 * M // 3
        h[:k1] = 0.8
        h[k1:k2] = 1.6
        h[k2:] = 1.0
        return log_from_positive_h(h)

    if kind == 11:
        k1 = M // 3
        k2 = 2 * M // 3
        h[:k1] = 1.2
        h[k1:k2] = 0.8
        h[k2:] = 1.5
        return log_from_positive_h(h)

    if kind == 12 or kind == 13:
        sig = 0.45 if kind == 12 else 0.90
        raw = rng.normal(0.0, sig, size=M)
        kernel = np.array([1.0, 2.0, 3.0, 2.0, 1.0], dtype=np.float64)
        kernel /= float(kernel.sum())
        raw = np.convolve(raw, kernel, mode="same")
        z = raw - float(np.mean(raw))
        h = np.exp(np.clip(z, -4.0, 4.0))
        return log_from_positive_h(h)

    # Default: random asymmetric smooth-ish profile.
    z = rng.normal(0.0, 1.0, size=M)
    kernel = np.array([1.0, 2.0, 3.0, 2.0, 1.0], dtype=np.float64)
    kernel /= float(kernel.sum())
    z = np.convolve(z, kernel, mode="same")
    z -= float(np.mean(z))
    h = np.exp(np.clip(z, -5.0, 5.0))
    return log_from_positive_h(h)


def optimize_segments_log(x0, M, seg_len, end_time, rng):
    """
    Local optimization over M segment log-heights using multiplicative
    mirror-style updates and occasional random restarts.
    """
    M = int(max(2, M))
    seg_len = int(max(1, seg_len))
    x0 = np.asarray(x0, dtype=np.float64)
    if x0.size != M:
        x0 = np.resize(x0, M)
    best_x = center_clip_log(x0)
    best_score = segment_objective(best_x, M, seg_len)
    if not np.isfinite(best_score):
        best_x = np.zeros(M, dtype=np.float64)
        best_score = segment_objective(best_x, M, seg_len)

    if not np.isfinite(best_score):
        return best_x, float("inf")

    etas = (0.03, 0.1, 0.3, 1.0, 3.0)
    powers = (1, 2, 4, 8, 6, 3, 5)
    top_fracs = (0.05, 0.1, 0.2, 0.35, 0.5, 0.15, 0.25)

    stuck = 0
    mode_counter = 0

    while time.monotonic() < end_time:
        a_seq = sequence_from_log_heights(best_x, seg_len)
        if a_seq is None:
            break

        b = fft_linear_convolution(a_seq)
        if b.size == 0 or not np.all(np.isfinite(b)):
            break
        bmax = float(np.max(b))
        if not np.isfinite(bmax) or bmax <= 0.0:
            break

        improved = False

        for mode_idx in range(3):
            if time.monotonic() >= end_time:
                break

            total_mode = mode_counter + mode_idx

            # Rotate through power weights and top-lag weights.
            if total_mode % 3 == 0:
                w = power_weights(b, powers[total_mode % len(powers)])
            else:
                frac = top_fracs[total_mode % len(top_fracs)]
                w = top_weights(b, frac)

            if w is None:
                continue

            g = gradient_hankel(a_seq, w)
            if g is None or g.size != a_seq.size:
                continue

            G = g.reshape(M, seg_len).sum(axis=1)
            Gscale = float(np.max(G)) if G.size else 0.0
            if not np.isfinite(Gscale) or Gscale <= 1e-300:
                continue

            for eta in etas:
                if time.monotonic() >= end_time:
                    break
                nx = best_x - eta * (G / Gscale)
                nx = center_clip_log(nx)
                sc = segment_objective(nx, M, seg_len)
                if np.isfinite(sc) and sc < best_score - 1e-13:
                    best_x = nx
                    best_score = float(sc)
                    improved = True
                    break

            if improved:
                break

        mode_counter += 1

        if improved:
            stuck = 0

        else:
            stuck += 1
            for sigma in (0.15, 0.40, 1.00):
                if time.monotonic() >= end_time:
                    break
                nx = best_x + rng.normal(0.0, sigma, size=M)
                nx = center_clip_log(nx)
                sc = segment_objective(nx, M, seg_len)
                if np.isfinite(sc) and sc < best_score - 1e-13:
                    best_x = nx
                    best_score = float(sc)
                    improved = True
                    break

            if not improved and stuck > 800:
                break

    return best_x, float(best_score) if np.isfinite(best_score) else float("inf")


def full_mirror_refine(a0, end_time, rng):
    """
    Full-sequence multiplicative mirror-style refinement.
    Uses weights derived from the current convolution.
    """
    a = np.asarray(a0, dtype=np.float64)
    if a.size == 0:
        return None, float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-15)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return None, float("inf")
    a = a / s

    best = a.copy()
    best_score = fast_score(best)
    if not np.isfinite(best_score):
        best = np.ones(a.size, dtype=np.float64)
        best /= float(best.size)
        best_score = fast_score(best)

    etas = (0.03, 0.1, 0.3, 1.0, 3.0)
    powers = (1, 2, 4, 8, 6)
    top_fracs = (0.05, 0.1, 0.2, 0.35)

    stuck = 0
    mode_counter = 0

    while time.monotonic() < end_time:
        b = fft_linear_convolution(best)
        if b.size == 0 or not np.all(np.isfinite(b)):
            break
        bmax = float(np.max(b))
        if not np.isfinite(bmax) or bmax <= 0.0:
            break

        improved = False

        for mode_idx in range(3):
            if time.monotonic() >= end_time:
                break

            total_mode = mode_counter + mode_idx
            if total_mode % 3 == 0:
                w = power_weights(b, powers[total_mode % len(powers)])
            else:
                frac = top_fracs[total_mode % len(top_fracs)]
                w = top_weights(b, frac)

            if w is None:
                continue

            g = gradient_hankel(best, w)
            if g is None or g.size != best.size:
                continue

            Gscale = float(np.max(g)) if g.size else 0.0
            if not np.isfinite(Gscale) or Gscale <= 1e-300:
                continue

            for eta in etas:
                if time.monotonic() >= end_time:
                    break

                loga = np.log(best + 1e-300)
                loga = loga - eta * (g / Gscale)
                m = np.max(loga)
                if not np.isfinite(m):
                    continue
                loga = loga - m
                loga = np.clip(loga, -50.0, 0.0)
                cand = np.exp(loga)
                cs = float(np.sum(cand))
                if not np.isfinite(cs) or cs <= 0.0:
                    continue
                cand = cand / cs
                sc = fast_score(cand)
                if np.isfinite(sc) and sc < best_score - 1e-13:
                    best = cand
                    best_score = float(sc)
                    improved = True
                    break

            if improved:
                break

        mode_counter += 1

        if improved:
            stuck = 0
        else:
            stuck += 1
            for _ in range(2):
                if time.monotonic() >= end_time:
                    break
                cand = random_log_block_candidate(best, rng)
                if cand is None:
                    continue
                sc = fast_score(cand)
                if np.isfinite(sc) and sc < best_score - 1e-13:
                    best = cand
                    best_score = float(sc)
                    improved = True
                    break
            if not improved and stuck > 800:
                break

    return best, float(best_score) if np.isfinite(best_score) else float("inf")


def consider_candidate(a, state):
    """
    Update state = [best_array, best_fast_score] if a is better.
    This is top-level to avoid closures.
    """
    if a is None:
        return
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-300:
        return
    arr = arr / s
    sc = fast_score(arr)
    if np.isfinite(sc) and sc < float(state[1]) - 1e-14:
        state[0] = arr.copy()
        state[1] = float(sc)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for the best sequence within the time budget.

    Returns a Python list of non-negative floating point coefficients.
    """
    rng = np.random.default_rng(int(seed))
    t_start = time.monotonic()
    total = max(60.0, float(budget_s))
    end = t_start + max(20.0, total - 70.0)

    # Final output length chosen to be long enough to approximate continuous
    # profiles but not so long that exact final evaluation becomes dangerous.
    FINAL_N = 8192
    SEG_M = 32
    SEG_LEN = max(1, FINAL_N // SEG_M)

    state = [None, float("inf")]

    # Parent / fallback starting point.
    parent_arr = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent_arr = None

    if parent_arr is not None:
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
        ps = float(np.sum(parent_arr))
        if np.isfinite(ps) and ps > 1e-300:
            parent_norm = parent_arr / ps
            consider_candidate(parent_norm, state)
        else:
            parent_norm = None
    else:
        parent_norm = None

    if state[0] is None:
        consider_candidate(np.ones(FINAL_N, dtype=np.float64), state)

    print(f"initial fastest internal score: {state[1]:.10f}")

    # Build a pool of initial segment profiles.
    starts = []
    for kind in range(16):
        xk = initial_log_heights(SEG_M, kind, rng)
        starts.append(xk)

    # If we already have a good full-length sequence, add its binned profile.
    if state[0] is not None and state[0].size >= SEG_M:
        xb = bin_to_log_heights(state[0], SEG_M)
        starts.append(xb)

    # If parent has a different length, also add its binned profile.
    if parent_norm is not None and parent_norm.size >= SEG_M:
        xp = bin_to_log_heights(parent_norm, SEG_M)
        starts.append(xp)

    # Short multi-start profile optimization.
    short_time = min(240.0, max(60.0, total * 0.20))
    short_end = min(end, t_start + short_time)

    best_x = None
    best_x_score = float("inf")
    profile_seq_best = None
    profile_seq_score = float("inf")

    n_starts = max(1, len(starts))
    per_start = max(4.0, (short_end - t_start) / float(n_starts))

    print(f"running {n_starts} segment-profile starts, short deadline {short_end - t_start:.1f}s")

    for idx, xk in enumerate(starts):
        now = time.monotonic()
        if now >= short_end:
            break
        remaining_starts = max(1, n_starts - idx)
        slice_end = min(short_end, now + max(3.0, per_start * 0.9))
        slice_end = min(slice_end, now + 45.0)

        xo, so = optimize_segments_log(xk, SEG_M, SEG_LEN, slice_end, rng)
        if np.isfinite(so):
            if so < best_x_score:
                best_x = xo
                best_x_score = so
                a_prof = sequence_from_log_heights(xo, SEG_LEN)
                if a_prof is not None:
                    a_prof = np.clip(a_prof, 0.0, 1000.0)
                    ss = float(np.sum(a_prof))
                    if np.isfinite(ss) and ss > 1e-300:
                        a_prof = a_prof / ss
                        consider_candidate(a_prof, state)
                        if state[1] < profile_seq_score:
                            profile_seq_best = a_prof.copy()
                            profile_seq_score = state[1]
                        print(f"  short start {idx}: profile score={so:.10f}, best={state[1]:.10f}")

    # Longer optimization from the best coarse/binned profile.
    long_time = min(600.0, max(120.0, total * 0.35))
    long_end = min(end, short_end + long_time)

    if best_x is None:
        best_x = initial_log_heights(SEG_M, 1, rng)
        best_x_score = segment_objective(best_x, SEG_M, SEG_LEN)

    if best_x_score < profile_seq_score:
        a_prof = sequence_from_log_heights(best_x, SEG_LEN)
        if a_prof is not None:
            a_prof = np.clip(a_prof, 0.0, 1000.0)
            ss = float(np.sum(a_prof))
            if np.isfinite(ss) and ss > 1e-300:
                profile_seq_best = (a_prof / ss).copy()
                profile_seq_score = fast_score(profile_seq_best)

    if time.monotonic() < long_end:
        print(f"long segment refinement from best profile, deadline {long_end - time.monotonic():.1f}s")
        x_long, s_long = optimize_segments_log(best_x, SEG_M, SEG_LEN, long_end, rng)
        if np.isfinite(s_long) and s_long < best_x_score:
            best_x = x_long
            best_x_score = s_long
            a_prof = sequence_from_log_heights(best_x, SEG_LEN)
            if a_prof is not None:
                a_prof = np.clip(a_prof, 0.0, 1000.0)
                ss = float(np.sum(a_prof))
                if np.isfinite(ss) and ss > 1e-300:
                    profile_seq_best = (a_prof / ss).copy()
                    profile_seq_score = fast_score(profile_seq_best)
                    consider_candidate(profile_seq_best, state)
                    print(f"  long profile score={s_long:.10f}, best={state[1]:.10f}")

    # Choose a full-sequence initializer. Prefer the current internal best if
    # it has the selected final length; otherwise use the optimized profile.
    if state[0] is not None and shape_equivalent_len(state[0], FINAL_N):
        full_init = state[0].copy()
    elif profile_seq_best is not None and profile_seq_best.size == FINAL_N:
        full_init = profile_seq_best.copy()
    elif best_x is not None:
        full_init = sequence_from_log_heights(best_x, SEG_LEN)
        if full_init is None:
            full_init = np.ones(FINAL_N, dtype=np.float64)
    else:
        full_init = np.ones(FINAL_N, dtype=np.float64)

    if full_init is None or full_init.size == 0:
        full_init = np.ones(FINAL_N, dtype=np.float64)

    full_init = np.clip(full_init, 0.0, 1000.0)
    fs = float(np.sum(full_init))
    if np.isfinite(fs) and fs > 1e-300:
        full_init = full_init / fs
    else:
        full_init = np.ones(FINAL_N, dtype=np.float64) / float(FINAL_N)

    # Full-sequence refinement.
    full_end = min(end, t_start + total - 100.0)
    if time.monotonic() < full_end:
        print(f"full mirror refinement deadline {full_end - time.monotonic():.1f}s")
        full_arr, full_score = full_mirror_refine(full_init, full_end, rng)
        if full_arr is not None and np.isfinite(full_score):
            consider_candidate(full_arr, state)
            print(f"  full refined score={full_score:.10f}, best={state[1]:.10f}")

    # If there is still time, try one more short full refinement from another
    # good candidate (either parent-ish or profile) to avoid too local a path.
    if time.monotonic() < end - 60.0:
        alt_init = None
        if profile_seq_best is not None and profile_seq_best.size == FINAL_N:
            alt_init = profile_seq_best
        elif parent_norm is not None and parent_norm.size == FINAL_N:
            alt_init = parent_norm
        if alt_init is not None and state[0] is not None:
            # Simple random-jump start: add block noise to current best.
            alt_init = random_log_block_candidate(full_init, rng)
            if alt_init is not None:
                budget2 = min(end - 30.0, time.monotonic() + 120.0)
                full_arr2, full_score2 = full_mirror_refine(alt_init, budget2, rng)
                if full_arr2 is not None and np.isfinite(full_score2):
                    consider_candidate(full_arr2, state)
                    print(f"  alternate full refined score={full_score2:.10f}, best={state[1]:.10f}")

    best_internal = state[0]
    if best_internal is None:
        best_internal = np.ones(FINAL_N, dtype=np.float64)
        sc = fast_score(best_internal)
        state[0] = best_internal
        state[1] = sc

    # Shortlist for exact evaluation.
    shortlist = []
    if state[0] is not None:
        shortlist.append(state[0])
    if profile_seq_best is not None:
        shortlist.append(profile_seq_best)
    if parent_norm is not None:
        shortlist.append(parent_norm)
    shortlist.append(np.ones(FINAL_N, dtype=np.float64))

    # Deduplicate by exact length and a very coarse checksum to avoid repeated
    # expensive exact convolutions.
    unique_list = []
    seen = set()
    for seq in shortlist:
        arr = np.asarray(seq, dtype=np.float64)
        if arr.size == 0 or arr.size > 16000:
            continue
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(np.sum(arr))
        if not np.isfinite(s) or s <= 1e-300:
            continue
        arr = arr / s
        # A cheap fingerprint; exact equality is not required.
        fp = (int(arr.size), tuple(np.round(arr[::max(1, arr.size // 17)], 10)))
        if fp in seen:
            continue
        seen.add(fp)
        unique_list.append(arr)

    final_best_seq = None
    final_best_score = float("inf")

    # Exact final scoring. Keep the time margin but allow a few exact calls.
    for arr in unique_list[:3]:
        if time.monotonic() > end - 10.0:
            break
        lst = [float(x) for x in arr]
        try:
            sc = evaluate_sequence(lst)
        except Exception:
            sc = float("inf")
        sc = float(sc)
        print(f"exact score candidate n={arr.size}: {sc:.10f}")
        if np.isfinite(sc) and sc < final_best_score:
            final_best_score = sc
            final_best_seq = lst

    if final_best_seq is None:
        fallback = state[0]
        if fallback is None:
            fallback = np.ones(FINAL_N, dtype=np.float64)
        arr = np.asarray(fallback, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(np.sum(arr))
        if not np.isfinite(s) or s <= 1e-300:
            arr = np.ones(FINAL_N, dtype=np.float64)
            s = float(FINAL_N)
        arr = arr / s
        final_best_seq = [float(x) for x in arr]
        final_best_score = float("inf")

    print(f"final reported best internal/exact heuristic score: {state[1]:.10f}")
    return final_best_seq
```
