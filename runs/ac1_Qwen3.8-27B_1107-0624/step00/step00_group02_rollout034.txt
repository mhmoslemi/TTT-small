```python
def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum x = s}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.full(n, s/n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import time

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - s)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, s / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a, N_fft)
    power = A.real**2 + A.imag**2
    ac = np.fft.irfft(power, N_fft)[:N]
    return ac

def subgradient(a, k):
    """Compute subgradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
```

Instead of using the gradient at a single point, I can use the "moreau Envelope" or proximal variant:

This is the proximal gradient step. For the max of quadratics, the proximal step doesn't have a closed form, but we can approximate it.

Alternatively, I can use the "accelerated" Proximal Gradient Method (FISTA) which gives O(1/k^2) convergence vs O(1/k) for vanilla gradient descent.

```python
    n = len(a0)
    f = a0.copy()  # FISTA iterate
    y = a0.copy()  # FISTA extrapolation
    t_k = 1.0
    eta = 1.0 / np.linalg.norm(compute_autocorr(a0))  # initial step size
    
    for k in range(100000):
        # Gradient at y
        ac = compute_autocorr(y)
        max_idx = np.argmax(ac)
        g = np.zeros(n)
        lo = max(0, max_idx - n + 1)
        hi = min(n, max_idx + 1)
        g[lo:hi] += 2.0 * a[max_idx - lo : max_idx - hi : -1]
        
        # Gradient step
        f_new = y - eta * g
        f_new = project_simplex(f_new, 1.0)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        y = f_new + ((t_k - 1) / t_new) * (f_new - f)
        y = project_simplex(y, 1.0)  # keep y in feasible set
        f = f_new
        t_k = t_new
    
    return f
```

Actually, FISTA is designed for smooth convex functions. Our function `max_k ac[k]` is convex but not smooth (it's the max of smooth functions). FISTA can still be applied to the "smoothed" version, or we can use it with a smooth approximation.

Let me use the log-sum-exp smoothing:
`F(a) = (1/t) * log(sum_k exp(t * ac[k]))`

This is smooth (twice differentiable) and convex. Its gradient is:
`grad F(a)[m] = (1/t) * sum_k exp(t*ac[k]) * 2*a[k-m] / sum_j exp(t*ac[j])`
= `sum_k p_k * 2*a[k-m]` where `p_k = exp(t*(ac[k] - max_ac)) / sum_j exp(t*(ac[j] - max_ac))`

For large t, this concentrates on the max. For moderate t, it's a weighted average.

import numpy as np
import time

def project_simplex(v, s=1.0):
    """Project v onto the probability simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, s / n, dtype=np.float64)
    rho = rho_arr[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_autocorr(a):
    """Compute linear autocorrelation of a using FFT."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a, N_fft)
    power = A.real ** 2 + A.imag ** 2
    ac = np.fft.irfft(power, N_fft)[:N]
    return ac

def compute_smoothed_gradient(a, t=20.0):
    """
    Compute the gradient of the smoothed objective (1/t)*log(sum exp(t*ac[k])).
    Returns (smoothed_obj, gradient).

    ac = fast_autocorr(a)

    shifted = t * (ac - ac_max)
