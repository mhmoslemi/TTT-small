Now, let me think about the projection onto the simplex. The problem is:
minimize sum (x_i - v_i)^2 subject to x_i >= 0, sum x_i = 1.

The solution is x_i = max(v_i - tau, 0) where tau is chosen so that sum x_i = 1.

We can find tau by:
- Sort v in descending order: v_(1) >= v_(2) >= ... >= v_(n)
- For each k from 1 to n, compute tau_k = (sum_{i=1}^k v_(i) - 1) / k
- The correct k is the largest one where v_(k) - tau_k > 0.

This is O(n log n) due to sorting.

Alternatively, we can use a simpler (but iterative) approach:
- Clip to non-negative
- If sum > 1, subtract a constant from all positive elements
- If sum < 1, add to zero elements
- Repeat

```python
def project_simplex(v, target_sum=1.0):
    """Project v onto the probability simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1] + 1  # 1-indexed
    tau = (cssv[rho-1] - target_sum) / rho
    return np.maximum(v - tau, 0)
```

import numpy as np

def fast_self_convolve(a):
    """Compute full self-convolution using FFT for efficiency."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    result = np.fft.irfft(A * A, fsize)[:size]
    # Clip tiny negative values from numerical error
    np.maximum(result, 0, out=result)
    return result


def project_simplex(v, target_sum=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1)
    cond = v_sorted * idx > (cssv - target_sum)
    rho_indices = np.where(cond)[0]
    if len(rho_indices) == 0:
        return np.full(n, target_sum / n)
    rho = rho_indices[-1] + 1
    tau = (cssv[rho - 1] - target_sum) / rho
    return np.maximum(v - tau, 0.0)


def compute_objective(a):
    """Compute the objective function value for a non-negative sequence."""
    a = np.clip(a, 0.0, 1000.0)

def compute_peak_gradient(a, k_star):
    """Compute gradient of (a*a)[k_star] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k_star - (n - 1))
    j_end = min(n - 1, k_star)
    if j_start <= j_end:
        idx = np.arange(j_start, j_end + 1)
        grad[idx] = 2.0 * a[k_star - idx]
    return grad

        if time.monotonic() > scan_deadline:
            break
        
        inits = generate_initializations(n, rng, parent if (parent is not None and len(parent) == n) else None)
        
        # Time per init: limited
        time_left = scan_deadline - time.monotonic()
        time_per = min(3.0, time_left / max(1, len(inits)))
        
        for init in inits:
            if time.monotonic() > scan_deadline - 1:
                break
            if time_per < 0.2:
                break
            
            t_max = min(time_per, scan_deadline - time.monotonic() - 0.5)
            if t_max < 0.1:
                continue
            
            # Quick evaluation of init
            init_score = compute_objective(init)
            
            if init_score < best_score:
                best_score = init_score
                best_seq = init.copy()
                print(f"  Init n={n}: score={init_score:.6f}")
            
            # Brief optimization
            seq, score = optimize_sequence(init, rng, max_time=t_max)
            
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Optimized n={n}: score={score:.6f}")
            
            # Keep top candidates for deep optimization
            candidates.append((score, seq.copy(), n))
    
    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    
    # Phase 2: Deep optimization of top candidates
    deep_deadline = deadline
    top_k = min(8, len(candidates))
    
    print(f"\nPhase 2: Deep optimization of top {top_k} candidates")
    
    for i in range(top_k):
        if time.monotonic() > deep_deadline - 10:
            break
        
        score, seq, n = candidates[i]
        time_left = deep_deadline - time.monotonic()
        t_max = min(120.0, time_left * 0.8)  # Cap at 2 min per candidate
        
        if t_max < 5:
            break
        
        print(f"  Deep optimizing candidate {i+1}/{top_k} (n={n}, current={score:.6f})")
        
        seq2, score2 = deep_optimize(seq, rng, max_time=t_max)
        
        if score2 < best_score:
            best_score = score2
            best_seq = seq2.copy()
            print(f"    Improved to: {score2:.6f}")
        
        # Also try a fresh deep optimization from the best
        if time.monotonic() < deep_deadline - 10:
            t_max2 = min(60.0, deep_deadline - time.monotonic() - 5)
            if t_max2 > 5:
                seq3, score3 = deep_optimize(best_seq.copy(), rng, max_time=t_max2)
                if score3 < best_score:
                    best_score = score3
                    best_seq = seq3.copy()
                    print(f"    Best improved to: {score3:.6f}")
    
    # Final verification
    if best_seq is None:
        best_seq = np.ones(64, dtype=np.float64) / 64
    
    # Scale the sequence so it's within [0, 1000] and sum >= 0.01
    # (it already is since we work with sum=1, but be safe)
    best_seq = np.clip(best_seq, 0, 1000)
    if np.sum(best_seq) < 0.01:
        best_seq = best_seq / np.sum(best_seq) * 1.0
    
    final_score = compute_objective(best_seq)
    print(f"\nFinal score: {final_score:.6f}")
    
    return [float(x) for x in best_seq]
