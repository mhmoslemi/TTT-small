def _next_pow2(m):
    """Smallest power of two >= m (m >= 1)."""
    N = 1
    while N < m:
        N <<= 1
    return N


def _fft_grad(weights, a, N):
    """grad of soft-max = 2 * cross_correlation(weights, a)[:len(a)] via FFT."""
    n = len(a)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(weights, N)
    out = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * out[:n]


def _run_optimization(a, time_budget, lr=0.005):
    """Soft-max (smooth max of the autocorrelation) gradient descent with Adam.

    Input a is rescaled so that its sum is sqrt(2*len(a)); under this
    normalization the evaluator's score equals max(convolve(a,a)) exactly,
    so progress is tracked with that free quantity. The element-wise peak of
    the (renormalized) sequence is returned.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    if n <= 1:
        return np.full(n, 1.0)
    target = np.sqrt(2.0 * n)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.full(n, target / n)
    a = a * (target / s)

    N = _next_pow2(3 * n - 2)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

        if b_max < best_maxb:

            best_a = a.copy()

        sc2 = float(evaluate_sequence([float(x) for x in r2]))
        if sc2 < best_score:
            best_score, best = sc2, r2.copy()
        print("FINAL score=%.6f n=%d" % (best_score, best.size), flush=True)

    # Guard: ensure a valid nonempty list is always returned.
    if best is None or len(best) == 0:
        return [1.0]
    return [float(x) for x in best]
