import numpy as np

def _next_power_of_two(m):
    p=1
    while p < m: p <<=1
    return p

def _full_convolution_fft(a, b=None):
    if b is None: b=a
    a=np.asarray(a,dtype=np.float64)
    b=np.asarray(b,dtype=np.float64)
    la=a.size; lb=b.size
    L=la+lb-1
    N=_next_power_of_two(L)
    fa=np.fft.rfft(a,N)
    fb=np.fft.rfft(b,N)
    out=np.fft.irfft(fa*fb,N)[:L]
    return out

def _local_search_binary(arr, deadline, max_passes=100000, seed=1):
    a=np.asarray(arr,dtype=np.float64).copy()
    n=a.size
    if n==0: return a
    a = (a > 0.5).astype(np.float64)
    if a.sum()==0: a[0]=1.0
    c=np.convolve(a,a)
    t=2.0*a
    S=int(round(float(a.sum())))
    cur_score=(2.0*n*float(np.max(c)))/(float(S)*float(S))
    rng=np.random.default_rng(seed)
    passes=0
    stuck=0
    while time.monotonic() < deadline and passes < max_passes:
       order=rng.permutation(n)
       improved=False
       for i in order:
          if time.monotonic() > deadline: break
          d = 1.0 if a[i] == 0.0 else -1.0
          if d > 0:
             c[i:i+n] += t
          else:
             c[i:i+n] -= t
          c[2*i] += 1.0
          newS = S + (1 if d>0 else -1)
          if newS > 0:
             maxc = float(np.max(c))
             new_score = (2.0*n*maxc)/(float(newS)*float(newS))
             if new_score < cur_score - 1e-14:
                a[i] = 1.0 - a[i]
                t[i] = 2.0 * a[i]
                S = newS
                cur_score = new_score
                improved = True
             else:
                if d > 0:
                   c[i:i+n] -= t
                else:
                   c[i:i+n] += t
                c[2*i] -= 1.0
          else:
             if d > 0:
                c[i:i+n] -= t
             else:
                c[i:i+n] += t

       passes += 1
       if not improved:
          stuck += 1
          if stuck >= 2: break
       else:
          stuck=0
    return a

```python
"""
Search method for low 2*n*max(conv(a,a))/(sum(a)**2).

Key idea:
  Random balanced 0/1 sequences already have nearly flat sum-convolution
  coefficients and typically give scores close to 1, far below 2. We evaluate
  several fixed-weight random binary sequences of moderate length, keep the best
  small one, and refine it with a greedy bit-flip local search using exact O(n)
  convolution updates. A final tiny positive floor is used so all entries are
  strictly positive while changing the objective negligibly.
"""

import numpy as np
import time


def _next_pow2(m: int) -> int:
    """Smallest power of two >= m."""
    p = 1
    while p < m:
        p <<= 1
    return p


def _full_convolution_fft(a: np.ndarray, b: np.ndarray = None) -> np.ndarray:
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)

    la = int(a.size)
    lb = int(b.size)
    if la == 0 or lb == 0:
        return np.zeros(0, dtype=np.float64)

    out_len = la + lb - 1
    nfft = _next_pow2(out_len)

    fa = np.fft.rfft(a, nfft)
    fb = np.fft.rfft(b, nfft)
    out = np.fft.irfft(fa * fb, nfft)[:out_len]
    return out


def _score_fft(arr: np.ndarray) -> float:
    """
    Fast internal score using FFT. Avoids calling the exact evaluator in tight
    loops. This is only for search; final candidates are checked externally.
    """
    a = np.asarray(arr, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return float("inf")

    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 0.01:
        return float("inf")

    conv = _full_convolution_fft(a)
    if conv.size == 0:
        return float("inf")

    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    if m < 0.0:
        m = 0.0

    return float(2.0 * n * m / (s * s))


def _random_binary_fixed_k(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Random nonnegative 0/1 vector of length n with exactly k ones."""
    a = np.zeros(n, dtype=np.float64)
    if n <= 0:
        return a
    k = max(0, min(int(k), n))
    if k == 0:
        a[0] = 1.0
        return a
    if k == n:
        a[:] = 1.0
        return a

    idx = rng.permutation(n)[:k]
    a[idx] = 1.0
    return a


def _is_binary(arr: np.ndarray, tol: float = 1e-8) -> bool:
    """True if arr is essentially a 0/1 vector."""
    a = np.asarray(arr, dtype=np.float64)
    r = np.rint(a)
    return bool(np.all(np.abs(a - r) <= tol) and np.all((r <= tol) | (r >= 1.0 - tol)))


def _local_search_binary(
    arr: np.ndarray,
    deadline: float,
    max_passes: int = 1000000,
    seed: int = 12345,
) -> np.ndarray:
    """
    Greedy bit-flip local search for binary sequences.

    Maintains the full convolution c = a*a exactly (up to floating-point
    representability of integers) and updates it in O(n) per tested flip.
    Only improvements in the actual objective are accepted.
    """
    a = np.asarray(arr, dtype=np.float64).copy()
    n = int(a.size)
    if n == 0:
        return a

    # Binarize defensively.
    a = np.where(a >= 0.5, 1.0, 0.0)

    if float(a.sum()) <= 0.0:
        a[0] = 1.0

    # Exact-ish convolution initialisation. Binary values are small enough that
    # floating point stays exact for all relevant integer counts.
    c = np.convolve(a, a)
    t = 2.0 * a
    s = int(round(float(a.sum())))
    if s <= 0:
        return a

    cur_score = (2.0 * float(n) * float(np.max(c))) / float(s * s)
    rng = np.random.default_rng(seed)

    passes = 0
    stuck_passes = 0

    while time.monotonic() < deadline and passes < max_passes:
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            # d = +1 for 0 -> 1, d = -1 for 1 -> 0.
            if a[i] == 0.0:
                d = 1.0
            else:
                d = -1.0

            # Apply candidate flip:
            # (P + d*x^i)^2 = P^2 + 2*d*x^i*P + d^2*x^{2i}
            if d > 0.0:
                c[i:i + n] += t
            else:
                c[i:i + n] -= t
            c[2 * i] += 1.0

            new_s = s + (1 if d > 0.0 else -1)

            if new_s > 0:
                new_score = (
                    2.0 * float(n) * float(np.max(c)) / float(new_s * new_s)
                )

                if new_score < cur_score - 1e-14:
                    # Accept.
                    a[i] = 1.0 - a[i]
                    t[i] = 2.0 * a[i]
                    s = new_s
                    cur_score = new_score
                    improved = True
                else:
                    # Revert.
                    if d > 0.0:
                        c[i:i + n] -= t
                    else:
                        c[i:i + n] += t
                    c[2 * i] -= 1.0
            else:
                # Revert invalid empty sum.
                if d > 0.0:
                    c[i:i + n] -= t
                else:
                    c[i:i + n] += t
                c[2 * i] -= 1.0

        passes += 1

        if improved:
            stuck_passes = 0
        else:
            stuck_passes += 1
            if stuck_passes >= 2:
                break

    return a

        official = float(evaluate_sequence(candidate))
    except Exception:
        official = _score_fft(final_arr)

    # If something went unexpectedly wrong or we are above the stated target,
    # try a couple of fresh fallbacks.
    if (not np.isfinite(official)) or official > 1.5027:
        for attempt in range(4):
            fb_n = 4096 if attempt % 2 == 0 else 8192
            fb = _random_binary_fixed_k(fb_n, fb_n // 2, rng)
            fb = np.clip(fb, 0.0, 1000.0)
            fb = np.where(fb > 1e-12, fb, 1e-12)
            fb_candidate = [float(x) for x in fb]

            fb_official = float("inf")
            try:
                fb_official = float(evaluate_sequence(fb_candidate))
            except Exception:
                fb_official = _score_fft(fb)

            if np.isfinite(fb_official) and fb_official < official:
                candidate = fb_candidate
                official = float(fb_official)

            if np.isfinite(official) and official <= 1.5027:
                break

    return candidate
```
