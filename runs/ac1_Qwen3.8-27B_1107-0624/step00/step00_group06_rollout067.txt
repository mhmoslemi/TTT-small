"""
Smoothed subgradient optimization with mirror descent to minimize the
peak of the self-convolution relative to the sum squared.
Uses log-sum-exp smoothing of max for smooth gradients, multiple
diverse initializations, adaptive step sizes, and periodic restarts.

import numpy as np
import time


def compute_score(a, n):
    """Compute the evaluation score for a non-negative sequence (sum normalized to 1)."""
    c = np.convolve(a, a)
    M = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * M / (S * S))


def make_gaussian_init(n, center_frac=0.5, sigma_frac=0.25):
    """Create a Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - n * center_frac) / (n * sigma_frac)) ** 2)
    s = g.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return g / s


def make_bimodal_init(n):
    """Create a bimodal (two Gaussian bumps) initialization."""
    x = np.arange(n, dtype=np.float64)
    b = (np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) +
         np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2))
    s = b.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return b / s


def make_triangular_init(n):
    """Create a triangular initialization."""
    x = np.arange(n, dtype=np.float64)
    t = np.maximum(0, 1 - 2 * np.abs(x - n * 0.5) / n)
    s = t.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return t / s


def make_block_init(n, nblocks=10, low_frac=0.2):
    """Create an alternating block initialization."""
    blk = np.ones(n, dtype=np.float64)
    blen = n // nblocks
    for i in range(0, nblocks, 2):
        start = i * blen
        end = min(n, (i + 1) * blen)
        blk[start:end] = low_frac
    s = blk.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return blk / s


def make_cosine_init(n):
    """Create a cosine window initialization."""
    x = np.arange(n, dtype=np.float64)
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    s = c.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return c / s


def make_dip_init(n, dip_center_frac=0.5, dip_width_frac=0.1, dip_frac=0.3):
    """Create a uniform sequence with a dip in the middle."""
    a = np.ones(n, dtype=np.float64)
    dc = int(n * dip_center_frac)
    dw = int(n * dip_width_frac)
    a[dc - dw:dc + dw] *= dip_frac
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def make_random_init(n, rng):
    """Create a random initialization."""
    r = rng.random(n)
    s = r.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return r / s


def make_symmetric_init(a, n):
    """Symmetrize a sequence."""
    return 0.5 * (a + a[::-1])

def run_lp_step(f, n, M):
    """
    Try one LP-based direction step. Returns the new sequence or None if LP fails.
    Maximize sum(g) subject to (f*g)[k] <= M, g >= 0.
    """
    try:
        import cvxpy as cp

        g = cp.Variable(n, nonneg=True)
        # Build convolution constraints: (f * g)[k] <= M for k = 0, ..., 2n-2
        # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
        # Use the Toeplitz structure
        # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
        # So A is a convolution matrix

        # Build A efficiently using numpy
        # A[k, j] = f[k - j] for max(0, k-n+1) <= j <= min(k, n-1)
        A_rows = []
        A_cols = []
        A_vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                val = f[k - j]
                if val > 1e-15:
                    A_rows.append(k)
                    A_cols.append(j)
                    A_vals.append(val)
        # This is too slow for n=5582. Use scipy sparse.
        from scipy import sparse

        # Build sparse matrix
        row_idx = np.array(A_rows)
        col_idx = np.array(A_cols)
        data = np.array(A_vals)
        A_mat = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        constraints = [A_mat @ g <= M]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)

        # Try ECOS first (faster for this size)
        try:
            prob.solve(solver=cp.ECOS, max_iter=5000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                return None

        if g.value is None or not np.all(np.isfinite(g.value)):
            return None

        g_val = np.clip(g.value, 0, None)
        s = g_val.sum()
        if s < 1e-10:
            return None
        return g_val / s

    except Exception:
        return None

    t_mix = 0.3  # mixing parameter
    iter_count = 0

    while time.monotonic() < deadline:

        M = c.max()

            break

        a_new = np.clip(a_new, 0, None)

        else:
            break

    if verbose:
        print(f"  {tag} LP done: {iter_count} iters, best score={best_local_score:.6f}")

    return best_local, best_local_score
