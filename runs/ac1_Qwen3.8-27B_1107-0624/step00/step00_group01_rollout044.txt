import time
from scipy.fft import fft, ifft

def fast_self_convolve(a):
    """Compute a * a (full linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = fft(a, fsize)
    return ifft(fa * fa).real[:size]

def score_from_conv(c, n):
    """Compute score from convolution c and sequence length n."""
    M = np.max(c)
    S = np.sum(c) ** 0.5  # S = sum of a, since sum(c) = (sum a)^2
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def subgradient_single_lag(a, k_star):
    """Compute subgradient of (a*a)_{k_star} w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        # grad[j] = 2 * a[k_star - j] for j in [lo, hi]
        # k_star - j ranges from k_star - lo to k_star - hi (decreasing)
        grad[lo:hi+1] = 2.0 * a[k_star - lo : k_star - hi : -1]
    return grad

def subgradient_weighted(a, w):
    """Compute weighted subgradient: sum_k w_k * d(a*a)_k / da.
    w has length 2n-1 (one weight per lag)."""
    n = len(a)
    # g_j = 2 * sum_k w_k * a_{k-j} for valid k-j
    # This is a cross-correlation
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    g = ifft(np.conj(W) * A).real[:n]
    return 2.0 * g

**Approach: "Chase and Balance"**

1. Compute $c = a * a$
2. Find the peak: $M = \max c$, $k^* = \arg\max c$
3. Compute the "excess": $e_k = \max(0, c_k - M_{\text{target}})$ where $M_{\text{target}}$ is some target (e.g., the 90th percentile of $c$)
4. Compute the "pressure" on each $a_j$:
   $p_j = \sum_k e_k \cdot a_{k-j}$ (weighted by how much each lag exceeds the target)
5. Update: $a_j \leftarrow a_j \cdot (1 - \eta \cdot p_j / \bar{p})$ (reduces high-pressure elements)
6. Also add a small "entropic" term to prevent elements from going to zero:
   $a_j \leftarrow a_j + \alpha / n$ (small constant)
7. Normalize: $a \leftarrow a / \sum a$

The idea is: elements that contribute a lot to the "excess" lags get reduced, while others stay the same or get slightly boosted.

Let me also add a "balancing" step: if some lag is below the target, we want to increase the contributions to that lag. But this is trickier because we can't directly "increase" a specific lag without potentially increasing others.
