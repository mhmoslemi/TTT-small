def make_random_seq(n, seed, dist):
    """Random non-negative sequence."""
    rng = np.random.default_rng(seed)
    if dist == 'uniform':
        a = rng.random(n)

        a = rng.beta(0.5, 0.5, n)

    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:

"""
Multi-strategy optimization minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP-based peak reduction, soft-max gradient descent, spectral shaping,
and perturbation search across multiple sequence lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v, target_sum):
    """Project v onto {x>=0, sum(x)=target_sum} via bisection on threshold."""
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    n = len(v)
    if n == 0:
        return v
    s = v.sum()
    if s < 1e-15:
        return np.full(n, target_sum / n)
    if abs(s - target_sum) < 1e-14:
        return v
    lo = 0.0
    hi = v.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(v - mid, 0.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    w = np.maximum(v - theta, 0.0)
    s2 = w.sum()
    if s2 > 1e-15:
        w *= target_sum / s2
    return w


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if n == 0 or s < 1e-10:
        return np.inf
    c = np.convolve(a, a, 'full')
    return float(2.0 * n * c.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A@b)[k] = (a*b)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        if jlo <= jhi:
            A[k, jlo:jhi + 1] = a[k - jhi:k - jlo + 1][::-1]
    return A


def solve_lp(a, M):
    """Solve max sum(b) s.t. (a*b)_k <= M, b >= 0. Return b or None."""
    n = len(a)
    A = build_conv_matrix(a)
    cvec = -np.ones(n)
    bub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(cvec, A_ub=A, b_ub=bub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def soft_max_gradient(a, soft_T):
    """Compute gradient of soft-max(conv(a,a)) w.r.t. a, assuming sum(a)=1."""
    n = len(a)
    c = np.convolve(a, a, 'full')
    cmax = c.max()
    csh = c - cmax
    w = np.exp(csh / soft_T)
    sw = w.sum()
    if sw < 1e-30:
        return np.zeros(n), cmax
    w = w / sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        if jlo <= jhi:
            segment = a[k - jhi:k - jlo + 1][::-1]
            grad[jlo:jhi + 1] += 2.0 * wk * segment
    return grad, cmax

def spectral_shaping(a_init, n_iter):
    """Flatten convolution via iterative spectral magnitude equalization."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    L = 2 * n
    alpha = 0.3
    for _ in range(n_iter):
        A = np.fft.rfft(a, n=L)
        mag = np.abs(A)
        phase = np.angle(A)
        target_mag = mag.mean()
        new_mag = (1.0 - alpha) * mag + alpha * target_mag
        new_mag[0] = mag[0]
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n=L)[:n]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / sn
        obj = compute_obj(a_new)
        if np.isfinite(obj) and obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            alpha = min(0.6, alpha * 1.1)
        else:
            alpha = max(0.02, alpha * 0.7)
    return best_a

def make_u_shaped(n, beta):
    if n <= 1:
        return np.ones(max(1, n))
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / (n - 1.0) - 1.0
    a = np.abs(x) ** beta + 0.05
    return a / a.sum()


def make_inverted_cosine(n):
    if n <= 1:
        return np.ones(max(1, n))
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / n))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_random_seq(n, seed, dist):
    rng = np.random.default_rng(seed)
    if dist == 'beta':
        a = rng.beta(0.5, 0.5, n)
    elif dist == 'bimodal':
        a = rng.exponential(1.0, n)
        w = np.ones(n)
        w[:max(1, n // 4)] = 3.0
        w[-max(1, n // 4):] = 3.0
        a = a * w
    elif dist == 'exponential':
        a = rng.exponential(1.0, n)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    return a
