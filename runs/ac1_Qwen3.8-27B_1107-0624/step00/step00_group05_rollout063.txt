Subgradient descent with FFT-based convolution to minimize the peak-to-sum ratio
of the autocorrelation. Uses multiple initializations (random, cosine-modulated,
power-law, Gaussian) and top-P peak targeting for robust convergence.
"""

import numpy as np
import time
import math


def _fft_convolve(a, n_fft):
    """Fast linear convolution of a with itself via FFT."""
    A = np.fft.rfft(a, n_fft)
    return np.fft.irfft(A * A, n_fft)


def _compute_gradient(c, a, n, top_P):
    """Compute subgradient of max(c) using top-P peaks."""
    n_fft = len(c)
    top_idx = np.argpartition(c, -top_P)[-top_P:]
    # Only use lags with significant values
    threshold = c.max() * 0.5
    top_idx = top_idx[c[top_idx] > threshold]
    
    grad = np.zeros(n)
    for k in top_idx:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            grad[j_range] += 2.0 * a[k - j_range]
    return grad


def _evaluate_fast(a, n, n_fft):
    """Fast score evaluation using FFT (approximates evaluate_sequence)."""
    c = _fft_convolve(a, n_fft)
    peak = c.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * peak / (s * s)


def _subgradient_descent(a, n, n_fft, deadline, rng, P=50, verbose=False):
    """Run subgradient descent to minimize the score."""
    best_a = a.copy()
    best_score = _evaluate_fast(a, n, n_fft)
    a_work = a.copy()
    no_improve = 0
    it = 0
    momentum = np.zeros(n)
    beta_mom = 0.9
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a_work, n_fft)
        peak = c.max()
        
        if peak < 1e-15:
            break
        
        grad = _compute_gradient(c, a_work, n, P)
        
        # Adaptive learning rate
        max_grad = np.abs(grad).max()
        if max_grad > 1e-15:
            eta = 0.3 / (max_grad * (1.0 + 0.1 * math.log1p(it / 100.0)))
        else:
            eta = 0.01
        
        # Momentum update
        momentum = beta_mom * momentum + grad
        step = eta * momentum
        
        # Multiple trial steps (line search)
        best_trial = None
        best_trial_score = best_score
        for scale in [1.0, 0.5, 0.25, 0.1]:
            a_trial = a_work - scale * step
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            trial_score = _evaluate_fast(a_trial, n, n_fft)
            if trial_score < best_trial_score:
                best_trial_score = trial_score
                best_trial = a_trial
        
        if best_trial is not None:
            a_work = best_trial
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_a = best_trial.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Periodic restart if stuck
        if no_improve > 200:
            # Perturb from best
            scale = 0.05 + 0.1 * rng.random()
            a_work = best_a * np.exp(rng.normal(0, scale, n))
            a_work = np.maximum(a_work, 0)
            s = a_work.sum()
            if s > 1e-10:
                a_work = a_work / s
            no_improve = 0
            momentum = np.zeros(n)
        
        it += 1
        if verbose and it % 2000 == 0:
            print(f"  SGD iter {it}, score={best_score:.6f}")
    
    return best_a, best_score
