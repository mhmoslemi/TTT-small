def run_stage(n, init_seq, time_budget, seed):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(time_budget))
    rng = np.random.default_rng(seed)
    candidates = candidate_sequences(n, seed, init_seq)
    a = None; best_val = np.inf; best_seq = None
    for c in candidates:
        c = make_sum1_nonneg(c)
        if c.size != n: continue
        val = internal_score(c)
        if np.isfinite(val) and val < best_val:
            best_val = val; best_seq = c; a = c.copy()
    if a is None:
        a = central_binomial(n); best_val = internal_score(a); best_seq = a.copy()
    no_improve = 0; fail_count = 0; it = 0
    while time.monotonic() < deadline - 1.0:
        M = max_conv(a)
        if not np.isfinite(M) or M <= 0: break
        improved = False
        lp_failed = False
        res = solve_min_cross(a)
        if res is None:
            lp_failed = True
        else:
            g, t_opt = res
            cand = line_search(a, g, M)
            if cand is not None:
                val = internal_score(cand)
                if np.isfinite(val) and val < best_val - 1e-12:
                    best_val = val; best_seq = cand.copy(); a = cand
                    improved = True
        if not improved and time.monotonic() < deadline - 2.0:
            res2 = solve_max_sum(a, M)
            if res2 is None:
                lp_failed = True
            else:
                cand = line_search(a, res2, M)
                if cand is not None:
                    val = internal_score(cand)
                    if np.isfinite(val) and val < best_val - 1e-12:
                        best_val = val; best_seq = cand.copy(); a = cand
                        improved = True
        if not improved:
            # cheap deterministic directions
            mirror = make_sum1_nonneg(a[::-1])
            if np.max(np.abs(a - mirror)) > 1e-12:
                cand = line_search(a, mirror, M)
                if cand is not None:
                    val = internal_score(cand)
                    if val < best_val - 1e-12:
                        best_val = val; best_seq = cand.copy(); a = cand; improved = True
            if not improved:
                for sigma in (0.03, 0.08, 0.15):
                    cand = random_perturb(a, rng, sigma)
                    val = internal_score(cand)
                    if np.isfinite(val) and val < best_val - 1e-12:
                        best_val = val; best_seq = cand.copy(); a = cand; improved = True; break
        if improved:
            no_improve = 0; fail_count = 0
        else:
            no_improve += 1
            if lp_failed:
                fail_count += 1
            else:
                fail_count = 0
            if no_improve > 12:
                # working point drift
                drift = random_perturb(best_seq, rng, 0.2)
                drift_val = internal_score(drift)
                if np.isfinite(drift_val) and drift_val < best_val * 1.1 + 1e-6:
                    a = drift
                else:
                    a = best_seq.copy()
                no_improve = 6
        if fail_count >= 6:
            break
        it += 1
        if it % 5 == 0:
            print("n=%d iter=%d internal=%.8f time=%.1f" % (n, it, best_val, time.monotonic()-t0))
    final = make_sum1_nonneg(best_seq)
    score = official_score(final)
    if not np.isfinite(score):
        score = internal_score(final)
    if score < best_val:
        # not update best_val necessary
        pass
    return final, score

```python
"""
Search for a non-negative step-function sequence by alternating:
1) an LP that minimizes the worst cross-correlation with the current vector,
2) a line search along the segment between the current vector and the LP solution.
The method starts from the central-binomial seed, which has almost flat
self-convolution, and then refines it.  Multiple lengths are warm-started.
"""

import time
import warnings
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csc_matrix

warnings.filterwarnings("ignore")


def make_sum1_nonneg(x):
    x = np.asarray(x, dtype=np.float64).copy()
    if x.size == 0:
        return np.ones(0, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return np.ones(x.size, dtype=np.float64)
    x /= s
    return x


def central_binomial(n):
    a = np.ones(int(n), dtype=np.float64)
    for k in range(1, int(n)):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    return make_sum1_nonneg(a)


def max_conv(x):
    n = int(len(x))
    if n == 0:
        return 0.0
    if n == 1:
        return float(x[0] * x[0])
    size = 2 * n - 1
    fft_len = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(x, fft_len)
    c = np.fft.irfft(fa * fa, fft_len)[:size]
    return float(np.max(c))


def internal_score(x):
    # Valid only when x is normalized to sum 1.
    return float(2.0 * len(x) * max_conv(x))


def official_score(seq):
    try:
        return float(evaluate_sequence([float(v) for v in seq]))
    except Exception:
        return np.inf


def resample_keep_sum(x, n):
    x = make_sum1_nonneg(x)
    m = int(len(x))
    n = int(n)
    if m == n:
        return x.copy()
    if m == 0:
        return np.ones(n, dtype=np.float64)
    cdf_old = np.empty(m + 1, dtype=np.float64)
    cdf_old[0] = 0.0
    cdf_old[1:] = np.cumsum(x)
    cdf_old[-1] = 1.0
    t_old = np.linspace(0.0, 1.0, m + 1, dtype=np.float64)
    t_new = np.linspace(0.0, 1.0, n + 1, dtype=np.float64)
    cdf_new = np.interp(t_new, t_old, cdf_old)
    y = np.diff(cdf_new)
    return make_sum1_nonneg(y)


def candidate_sequences(n, seed=0, extra=None):
    n = int(n)
    out = []
    base = central_binomial(n)
    out.append(base.copy())

    if extra is not None:
        try:
            out.append(resample_keep_sum(extra, n))
        except Exception:
            pass

    k = np.arange(n, dtype=np.float64)

    for alpha in (0.45, 0.50, 0.55):
        out.append(make_sum1_nonneg((k + 0.5) ** (-alpha)))

    for beta in (0.05, 0.10, 0.15):
        rise = (1.0 - k / float(n)) ** (-beta)
        fall = (1.0 - k / float(n)) ** beta
        out.append(make_sum1_nonneg(base * rise))
        out.append(make_sum1_nonneg(base * fall))

    rng = np.random.default_rng(seed)
    for sigma in (0.03, 0.08):
        out.append(make_sum1_nonneg(base * np.exp(rng.normal(0.0, sigma, n))))

    return out


def build_conv_matrix(n, a):
    n = int(n)
    m = 2 * n - 1
    nnz = n * n

    data = np.empty(nnz, dtype=np.float64)
    indices = np.empty(nnz, dtype=np.int32)
    col = np.arange(n, dtype=np.int32)

    for j in range(n):
        sl = slice(j * n, (j + 1) * n)
        data[sl] = a
        indices[sl] = j + col

    indptr = np.arange(n + 1, dtype=np.int64) * n
    return csc_matrix((data, indices, indptr), shape=(m, n))


def build_min_cross_A(n, a):
    n = int(n)
    m = 2 * n - 1
    total = n * n + m

    data = np.empty(total, dtype=np.float64)
    indices = np.empty(total, dtype=np.int32)
    col = np.arange(n, dtype=np.int32)

    for j in range(n):
        sl = slice(j * n, (j + 1) * n)
        data[sl] = a
        indices[sl] = j + col

    data[n * n:] = -1.0
    indices[n * n:] = np.arange(m, dtype=np.int32)

    indptr = np.empty(n + 2, dtype=np.int64)
    indptr[:n + 1] = np.arange(n + 1, dtype=np.int64) * n
    indptr[n + 1] = total

    return csc_matrix((data, indices, indptr), shape=(m, n + 1))


def solve_min_cross(a):
    n = int(len(a))
    m = 2 * n - 1
    try:
        A = build_min_cross_A(n, a)
        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0
        b_ub = np.zeros(m, dtype=np.float64)

        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        bounds = [(0.0, None)] * n + [(0.0, None)]
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs"
        )
        if not res.success:
            return None
        g = np.asarray(res.x[:-1], dtype=np.float64)
        t_opt = float(res.x[-1])
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s <= 1e-300:
            return None
        g /= s
        return g, t_opt
    except Exception:
        return None


def solve_max_sum(a, M):
    n = int(len(a))
    m = 2 * n - 1
    try:
        A = build_conv_matrix(n, a)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(m, float(M), dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs"
        )
        if not res.success:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s <= 1e-300:
            return None
        g /= s
        return g
    except Exception:
        return None


def normalized_mix(a, g, t):
    h = (1.0 - t) * a + t * g
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    s = float(h.sum())
    if s <= 1e-300:
        return None
    return h / s


def line_search(a, g, current_M):
    ts = (0.0, 0.01, 0.02, 0.05, 0.10, 0.20, 0.30, 0.40,
          0.50, 0.60, 0.70, 0.80, 0.90, 1.00)

    best_t = 0.0
    best_val = float(current_M)

    for t in ts[1:]:
        h = normalized_mix(a, g, t)
        if h is None:
            continue
        val = max_conv(h)
        if val < best_val:
            best_val = val
            best_t = t

    if best_val >= current_M * (1.0 - 1e-12):
        return None

    if best_t <= 0.02:
        lo, hi = 0.0, 0.20
    elif best_t >= 0.90:
        lo, hi = 0.80, 1.0
    else:
        lo, hi = 0.0, 1.0
        for i in range(1, len(ts)):
            if ts[i] > best_t:
                lo = ts[i - 1]
                hi = ts[i]
                break

    if hi - lo < 1e-8:
        return normalized_mix(a, g, best_t)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    hc = normalized_mix(a, g, c)
    fc = max_conv(hc) if hc is not None else float(current_M)
    hd = normalized_mix(a, g, d)
    fd = max_conv(hd) if hd is not None else float(current_M)

    if fc < best_val:
        best_val = fc
        best_t = c
    if fd < best_val:
        best_val = fd
        best_t = d

    for _ in range(24):
        if hi - lo < 1e-5:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            hc = normalized_mix(a, g, c)
            fc = max_conv(hc) if hc is not None else float(current_M)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            hd = normalized_mix(a, g, d)
            fd = max_conv(hd) if hd is not None else float(current_M)

        if fc < best_val:
            best_val = fc
            best_t = c
        if fd < best_val:
            best_val = fd
            best_t = d

    if best_val < current_M:
        return normalized_mix(a, g, best_t)
    return None


def random_perturb(seq, rng, sigma=0.08):
    seq = make_sum1_nonneg(seq)
    n = int(len(seq))
    if n == 0:
        return seq
    h = seq.copy()
    h *= np.exp(rng.normal(0.0, float(sigma), n))
    if n > 32:
        width = max(1, n // 120)
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        h[start:stop] *= np.exp(rng.normal(0.0, 2.0 * float(sigma), stop - start))
    return make_sum1_nonneg(h)


def run_stage(n, init_seq, time_budget, seed):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(time_budget))
    rng = np.random.default_rng(seed)

    candidates = candidate_sequences(int(n), seed, init_seq)
    a = None
    best_seq = None
    best_val = np.inf

    for c in candidates:
        c = make_sum1_nonneg(c)
        if len(c) != int(n):
            continue
        val = internal_score(c)
        if np.isfinite(val) and val < best_val:
            best_val = val
            best_seq = c.copy()
            a = c.copy()

    if a is None:
        a = central_binomial(int(n))
        best_val = internal_score(a)
        best_seq = a.copy()

    no_improve = 0
    fail_count = 0
    it = 0

    while time.monotonic() < deadline - 1.0:
        M = max_conv(a)
        if not np.isfinite(M) or M <= 0.0:
            break

        improved = False
        lp_failed = False

        out = solve_min_cross(a)
        if out is None:
            lp_failed = True
        else:
            g, _ = out
            cand = line_search(a, g, M)
            if cand is not None:
                val = internal_score(cand)
                if np.isfinite(val) and val < best_val - 1e-12:
                    best_val = val
                    best_seq = cand.copy()
                    a = cand
                    improved = True

        if not improved and time.monotonic() < deadline - 2.0:
            out2 = solve_max_sum(a, M)
            if out2 is None:
                lp_failed = True
            else:
                cand = line_search(a, out2, M)
                if cand is not None:
                    val = internal_score(cand)
                    if np.isfinite(val) and val < best_val - 1e-12:
                        best_val = val
                        best_seq = cand.copy()
                        a = cand
                        improved = True

        if not improved and time.monotonic() < deadline - 1.5:
            mirror = make_sum1_nonneg(a[::-1])
            if float(np.max(np.abs(a - mirror))) > 1e-12:
                cand = line_search(a, mirror, M)
                if cand is not None:
                    val = internal_score(cand)
                    if np.isfinite(val) and val < best_val - 1e-12:
                        best_val = val
                        best_seq = cand.copy()
                        a = cand
                        improved = True

            if not improved:
                for sigma in (0.03, 0.08, 0.15):
                    cand = random_perturb(a, rng, sigma)
                    val = internal_score(cand)
                    if np.isfinite(val) and val < best_val - 1e-12:
                        best_val = val
                        best_seq = cand.copy()
                        a = cand
                        improved = True
                        break

        if improved:
            no_improve = 0
            fail_count = 0
        else:
            no_improve += 1
            if lp_failed:
                fail_count += 1
            else:
                fail_count = 0

            if no_improve > 12:
                drift = random_perturb(best_seq, rng, 0.20)
                drift_val = internal_score(drift)
                if np.isfinite(drift_val) and drift_val < best_val * 1.10 + 1e-6:
                    a = drift
                else:
                    a = best_seq.copy()
                no_improve = 6

        if fail_count >= 6:
            break

        it += 1
        if it % 5 == 0:
            print("stage n=%d iter=%d internal=%.8f time=%.1fs" % (
                int(n), it, best_val, time.monotonic() - t0))

    final = make_sum1_nonneg(best_seq)
    score = official_score(final)
    if not np.isfinite(score):
        score = internal_score(final)
    return final, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    budget = max(60.0, float(budget_s))
    deadline = t_start + max(30.0, budget - 10.0)

    best_seq = None
    best_score = np.inf

    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size > 0:
            hs = make_sum1_nonneg(hs)
            sc = official_score(hs)
            if np.isfinite(sc):
                best_seq = hs
                best_score = sc
    except Exception:
        pass

    if budget >= 1500:
        plans = [(1024, 150.0), (2048, 900.0), (3072, 720.0)]
    elif budget >= 800:
        plans = [(1024, 120.0), (2048, max(90.0, budget - 130.0))]
    else:
        plans = [(512, max(30.0, 0.4 * budget)), (1024, max(30.0, 0.6 * budget))]

    for n, alloc in plans:
        if time.monotonic() > deadline - 45.0:
            break
        remaining = deadline - time.monotonic()
        alloc = min(float(alloc), remaining - 6.0)
        if alloc < 30.0:
            break
        try:
            seq, sc = run_stage(int(n), best_seq, alloc, seed + int(n))
            if np.isfinite(sc) and sc < best_score:
                best_seq = seq
                best_score = sc
                print("new best after n=%d score=%.8f" % (int(n), sc))
        except Exception:
            pass

    if best_seq is None:
        best_seq = central_binomial(1024)
        best_score = official_score(best_seq)
        if not np.isfinite(best_score):
            best_score = internal_score(best_seq)

    remaining = deadline - time.monotonic()
    if best_seq is not None and remaining > 120.0:
        try:
            seq, sc = run_stage(int(len(best_seq)), best_seq, remaining - 6.0, seed + 987654)
            if np.isfinite(sc) and sc < best_score:
                best_seq = seq
                best_score = sc
                print("new best after final polish score=%.8f" % sc)
        except Exception:
            pass

    final = make_sum1_nonneg(best_seq)
    return [float(v) for v in final]
```
