- For each `j in S_k`, the term `a_j * a_{k-j}` contributes to `dc_k/da_i` only if `j = i` or `k-j = i`.
- If `j = i` (and `i in S_k`): contribution is `a_{k-i}`.
- If `k-j = i` (i.e., `j = k-i` and `k-i in S_k`): contribution is `a_{k-i}`.
- If both `j = i` and `j = k-i` (i.e., `i = k-i`, so `2i = k`): we have the same term counted once, but it's `a_i * a_i`, and its derivative is `2*a_i`.

- If `2i = k` and `i in S_k`: `dc_k/da_i = 2 * a_i`
- If `2i != k`: `dc_k/da_i = ([i in S_k] * a_{k-i}) + ([k-i in S_k] * a_{k-i})`

Wait, both conditions `i in S_k` and `k-i in S_k` reduce to `0 <= i < n and 0 <= k-i < n`. And since `i` is always in `[0, n)`, the condition is just `0 <= k-i < n`.

- If `2i != k`: `dc_k/da_i = a_{k-i} + a_{k-i} = 2 * a_{k-i}` (when `0 <= k-i < n`)

Wait, that simplifies to: `dc_k/da_i = 2 * a_{k-i}` when `0 <= k-i < n`, and 0 otherwise!

"""
Minimizes 2*n*max(convolvE(a,a))/sum(a)^2 using projected subgradient descent
on the active (peak) lags of the self-convolution. Uses FFT for fast convolution
and adaptive step sizes. Explores multiple sequence lengths.
"""

import numpy as np
import time


def fast_self_convolve(a):
    """Compute full self-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return result[:size]


def compute_active_gradient(a, active_lags):
    """Compute the summed gradient of c_k w.r.t. a for k in active_lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        # grad[i] += 2 * a[k-i] for i in [i_start, i_end]
        idx = np.arange(i_start, i_end + 1)
        grad[idx] += 2.0 * a[k - idx]
    return grad


def optimize_sequence(a, deadline, rng, verbose=True):

    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s  # normalize to sum = 1
    
    best = a.copy()
    best_c = fast_self_convolve(best)
    best_obj = 2.0 * n * best_c.max()
    
    eta = 0.005
    no_improve = 0
    max_eta = 0.05
    min_eta = 0.0001
    
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        
        # Compute convolution and find active lags
        c = fast_self_convolve(a)
        max_c = c.max()
        
        # Active lags: those above 90% of max
        threshold = 0.90 * max_c
        active_mask = c > threshold
        active_lags = np.where(active_mask)[0]
        
        # Limit to top 30 lags for efficiency

def fast_self_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]

def compute_gradient(a, lags):
    """Sum of gradients dc_k/da for k in lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in lags:
        i_s = max(0, k - n + 1)
        i_e = min(n - 1, k)
        if i_s <= i_e:
            idx = np.arange(i_s, i_e + 1)
            grad[idx] += 2.0 * a[k - idx]
    return grad
