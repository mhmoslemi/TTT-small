import numpy as np
from scipy.optimize import linprog
import time

def make_ramp(n):
    a = np.linspace(0.01, 1.0, n)
    return a / np.sum(a)

def make_exp_decay(n):
    a = np.exp(-3.0 * np.arange(n) / n)
    return a / np.sum(a)

def local_score(a):
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv)  # assuming sum(a)=1

def subgradient_direction(a, k):
    """Return the subgradient of conv[k] at point a (for sum-constrained problem)."""
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        grad[j_lo:j_hi+1] = 2.0 * a[k - np.arange(j_lo, j_hi+1)]
    return grad

def optimize_lp(a, time_limit):
    """LP-based optimization (GJK method)."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        return a, local_score(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = local_score(a)
    start = time.monotonic()
    
    for it in range(5000):
        if time.monotonic() - start > time_limit:
            break
        
        conv_aa = np.convolve(a, a)
        M = np.max(conv_aa)
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Build T matrix and solve LP
        # T[k,i] = a[k-i] if max(0,k-n+1) <= i <= min(k,n-1)
        # This is a (2n-1) x n matrix
        T = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:

        M = np.max(conv)
        score = 2.0 * n * M

            best_a = a.copy()

from scipy.optimize import linprog
import time

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using asymmetric ramp initialization
combined with LP-based (GJK) direction finding and subgradient refinement.
The ramp breaks symmetry, achieving ~4/3 score; LP refines further.
"""

def make_ramp(n):
    a = np.linspace(0.0, 1.0, n)
    a = np.maximum(a, 1e-8)
    return a / np.sum(a)

def make_ramp_pow(n, p=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = x ** p
    a = np.maximum(a, 1e-8)
    return a / np.sum(a)

def make_exp_decay(n, rate=3.0):
    a = np.exp(-rate * np.arange(n) / n)
    return a / np.sum(a)

def local_score(a):
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv)

def build_T(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            T[k, i_lo:i_hi + 1] = a[k - np.arange(i_lo, i_hi + 1)]
    return T

                     options={'time_limit': max(1.0, (time.monotonic() - _lp_start)) if '_lp_start' in dir() else 30.0})

        elif itype == 'ramp2':
            a = make_ramp_pow(n, 2.0)
        elif itype == 'exp':
            a = make_exp_decay(n)
        else:
            continue

        s0 = local_score(a)
        print(f"\nn={n}, init={itype}: initial score={s0:.6f}")
        if s0 < best_score:
            best_score = s0
            best_seq = a.copy()

        lp_limit = min(100.0, max(10.0, (deadline - time.monotonic()) * 0.3))
        if n <= 1024:
            a_opt, s_lp = optimize_lp(a, time_limit=lp_limit)
            if s_lp < best_score:
                best_score = s_lp
                best_seq = a_opt.copy()
                print(f"  After LP: {s_lp:.6f}")
        else:
            a_opt = a

        if time.monotonic() < deadline - 20.0:
            sg_limit = min(60.0, max(5.0, (deadline - time.monotonic()) * 0.2))
            a_opt2, s_sg = optimize_subgrad(a_opt, time_limit=sg_limit)
            if s_sg < best_score:
                best_score = s_sg
                best_seq = a_opt2.copy()
                print(f"  After subgrad: {s_sg:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 10.0:
        pl_limit = min(30.0, max(5.0, (deadline - time.monotonic()) * 0.3))
        a_final, s_final = polish(best_seq, time_limit=pl_limit)
        if s_final < best_score:
            best_score = s_final
            best_seq = a_final.copy()
            print(f"\nAfter polish: {s_final:.6f}")

    if best_seq is not None:
        final = evaluate_sequence(best_seq.tolist())
        if np.isfinite(final):
            print(f"\nFinal verified score: {final:.6f} (len={len(best_seq)})")
        else:
            print("Final evaluate returned inf, using local score:", best_score)
        return [float(x) for x in best_seq]

    a = make_ramp(256)
    return [float(x) for x in a]
