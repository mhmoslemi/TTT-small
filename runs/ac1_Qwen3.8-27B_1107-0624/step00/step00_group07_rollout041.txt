Hybrid LP + local search optimizer for step function coefficients.
Uses the ideal sequence (1-z)^(-1/2) coefficients as initialization,
applies LP-based improvement, then hill-climbing with diverse perturbations.
"""

import numpy as np
from scipy import sparse
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def make_ideal_sequence(n):
    """Generate a_k = binom(2k,k)/4^k = Taylor coeff of (1-z)^(-1/2)."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(n - 1):
        a[k + 1] = a[k] * (2 * k + 1) / (2 * k + 2)
    return a


def fft_convolve(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    result = ifft(fft(a, N) * fft(b, N))[:n].real
    return result


def build_conv_matrix(f):
    """Build (2n-1) x n sparse matrix A where (A g)_k = (f*g)_k."""
    n = len(f)
    if n == 1:
        return sparse.csr_matrix(np.array([[f[0]]]))
    # Diagonals: offset d >= 0 has value f[d], length n
    #             offset -k has value f[k], length n-k
    diags = []
    offsets = []
    for d in range(n):
        diags.append(np.full(n, f[d]))
        offsets.append(d)
    for k in range(1, n):
        diags.append(np.full(n - k, f[k]))
        offsets.append(-k)
    return sparse.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')

    """Solve max 1^T g s.t. A_g <= M, g >= 0. Returns g or None."""
    n = len(f)

    if f[0] < 1e-10:
        return None
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)

                      method='highs',

        pass
    return None

    end = min(n, start + width)
    noise = rng.normal(0.0, strength, size=end - start)
    candidate[start:end] *= np.exp(noise)
    # Occasionally mirror

        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

"""
Hybrid optimizer for step function coefficient sequence.
Strategy: ideal sequence (1-z)^(-1/2) init + LP improvement + diverse local search.
Targets minimum of 2n*max(conv(a,a))/(sum(a))^2 for non-negative a.

import numpy as np
from scipy import sparse
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def make_ideal_sequence(n):
    """a_k = C(2k,k)/4^k, Taylor coefficients of (1-z)^(-1/2)."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(n - 1):
        a[k + 1] = a[k] * (2 * k + 1) / (2 * k + 2)
    return a


def fft_convolve(a, b=None):
    """Full linear convolution using FFT (returns real part)."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    return ifft(fft(a, N) * fft(b, N))[:n].real


def build_conv_matrix(f):
    """Sparse (2n-1) x n matrix A with (A@g)_k = (f*g)_k."""
    n = len(f)
    diags = []
    offsets = []
    for d in range(n):
        diags.append(np.full(n, f[d]))
        offsets.append(d)
    for k in range(1, n):
        diags.append(np.full(n - k, f[k]))
        offsets.append(-k)
    return sparse.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')


def solve_lp_direction(f, M, time_limit=45):
    """LP: max sum(g) s.t. conv(f,g) <= M, 0 <= g <= 1000."""
    n = len(f)
    if f[0] < 1e-12:
        return None
    try:
        A = build_conv_matrix(f)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, 1000.0)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        pass
    return None


def perturb_block(a, rng, sigma=0.12):
    """Multiply a random block by exp(N(0, sigma))."""
    n = len(a)

    out[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    if rng.random() < 0.25:
        out = 0.5 * (out + out[::-1])
    return np.clip(out, 0.0, 1000.0)
