```python
from scipy.optimize import linprog

def lp_update(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    # A[k, j] = f(k-j) if 0 <= k-j <= n-1, else 0
    # This is a (2n-1) x n matrix
    # c = -1 (we want to maximize sum(g), so minimize -sum(g))
    # A_ub @ g <= M * ones(2n-1)
    # g >= 0
    
    c = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_ub[k, j] = f[k-j]
    b_ub = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return -res.fun, res.x
    return None, None
```
