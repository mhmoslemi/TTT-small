        return float(2*n_dim/math.sqrt(4*best_score))

    try:

        best_score = _initialization_phase(1.5027)  # Target-defined structure

        conv_arr = np.convolve(best_seq, best_seq, mode="full")
        conv_max = np.max(conv_arr)
        # Select top 3 active peaks for dimensionality reduction
        active_indices = []
        for idx, val in enumerate(conv_arr):
            if val > conv_max * 0.85 and spent_time < last_update_time:
                active_indices.append(idx)
        active_constraints = active_indices[:min(len(active_indices), 180)]  # Vertex-restricted LP
                
    except KeyboardInterrupt:
        # Fast-abort handling
        return best_seq[:100] + [0]*(n_val-100)  # Return valid partial candidate
    
    # --- Adaptive Thermal Budget ---
    while spent_time < budget:
        # Calculate time available
        remaining = budget - spent_time
        if remaining < last_update_time:
            # Final deterministic cleanup
            break
        
        # --- Lightweight Verifier ---
        def _verify_boundaries(sequence_to_check, candidate):
            # Boundary enforcement
            arr = np.asarray(sequence_to_check)
            fixed_arr = np.clip(arr, 0, 1000.0)
            # State consistency check
            total_sum = np.sum(arr)
            if total_sum < 0.01:  # Pure repair
                return fixed_arr * (0.01/total_sum) if total_sum > 0 else np.clip(arr*0.5, 0, 1000)
            # Restoration for gradient effects
            log_vals = np.log(arr + 1e-10)
            adjustment = np.exp(log_vals - (np.mean(log_vals) if len(log_vals) > 1 else log_vals[0]))
            return fixed_arr.tolist()
        
        # --- Local Structure Refinement ---
        trial_seq = best_seq.copy()
        # Dynamic perturbation: uniform vs. gradient-guided
        scope = int(37 * math.sqrt(remaining)) # Adaptive active range
        
        for i in range(0, scope, scope//50):
            if i >= n_val:  # Fail-safe boundary
                break
                
            # Gradient-distorted updates based on active constraints
            if i in active_constraints and i > active_constraints[0]:
                # Normalize gradient input by active constraint weights
                constraint_factor = next((x for j, ignored_constraint in enumerate(active_constraints) 
                                         if ignored_constraint == i), 0.5)
                trial_seq[i] += constraint_factor * (np.random.rand() - 0.5) * 0.05

                trial_seq[i] += 0.03 * (np.random.rand() - 0.5)

        spent_time += 0.1
        remaining_time = budget - spent_time

        try:

                return best_seq
            
            trial_score = float(evaluate_sequence(trial_seq))

                conv_results = np.convolve(trial_seq, trial_seq, mode="full")
                new_peaks = [(idx, val) for idx, val in enumerate(conv_results)

    # Materially improve if possible beyond the initial 2.0 score

    final_seq = _verify_boundaries(best_seq, "production")

    # --- Return low-dimensional representation as fallback if needed ---
    return final_seq[:min(len(final_seq), 100)] + [10]*max(0, len(final_seq)-100)
