```python
"""
Search for a non-negative finite step-function coefficient sequence minimizing

    2*n*max(convolve(a,a)) / sum(a)^2.

The method combines:
  1. A fast parametric profile search in a two-exponential-mixture family.
  2. Smooth-max exponentiated-gradient optimization using FFT convolutions.
  3. Occasional Frank-Wolfe-like dual updates.
  4. Random smooth perturbations and final block local search.

All optimization is done in a scale-invariant probability-vector representation
(sum = 1), where the objective reduces to 2*n*max(conv(p,p)).
"""

import numpy as np
import time
import math


def npt_next_pow2(x):
    """Smallest power of two >= x."""
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize_vector(v):
    """Clip to non-negative and normalize to sum 1. Stable fallback to uniform."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or v.size == 0:
        return np.ones((0,), dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.clip(v, 0.0, None)
    s = float(v.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.ones_like(v, dtype=np.float64) / float(v.size)
    return v / s


def softmax_log(logv):
    """Stable softmax over log-vector."""
    logv = np.asarray(logv, dtype=np.float64)
    logv = np.where(np.isfinite(logv), logv, -745.0)
    m = float(logv.max())
    e = np.exp(logv - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return np.ones_like(e, dtype=np.float64) / float(e.size)
    return e / s


def convolution_eval(p, N=None):
    """Convolution of p with itself, length 2n-1."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    L = 2 * n - 1
    if N is None:
        N = npt_next_pow2(L)
    F = np.fft.rfft(p, N)
    c = np.fft.irfft(F * F, N)[:L]
    c = np.asarray(c, dtype=np.float64)
    c[~np.isfinite(c)] = 0.0
    c = np.maximum(c, 0.0)
    return c


def convolution_and_fft(p, N):
    """Return convolution p*p and the FFT of p, reusing the transform."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    L = 2 * n - 1
    F = np.fft.rfft(p, N)
    c = np.fft.irfft(F * F, N)[:L]
    c = np.asarray(c, dtype=np.float64)
    c[~np.isfinite(c)] = 0.0
    c = np.maximum(c, 0.0)
    return c, F


def exact_score(p, N=None):
    """Exact internal score 2*n*max(conv(p,p)) for sum(p)=1."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    c = convolution_eval(p, N)
    return float(2.0 * n * np.max(c))


def soft_max_weights(c, tau):
    """
    Approximate max(c) by log-sum-exp and return positive weights w over c.
    """
    c = np.asarray(c, dtype=np.float64)
    tc = tau * c
    m = float(tc.max())
    e = np.exp(tc - m)
    s = float(e.sum())
    if not math.isfinite(s) or s <= 1e-300:
        idx = int(np.argmax(c))
        w = np.zeros_like(c, dtype=np.float64)
        w[idx] = 1.0
        smax = float(c[idx])
    else:
        w = e / s
        smax = float(m / tau + math.log(s) / tau)
    return smax, w


def gradient_from_weights(p, w, Fp, N, n):
    """
    Compute grad_p[j] = 2 * sum_k w[k] p[k-j] for valid k-j.

    Uses circular correlation ifft(FFT(w)*conj(FFT(p))). With N >= 2n,
    the first n entries are exactly the linear cross-correlation needed.
    """
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(W * np.conjugate(Fp), N)[:n]
    corr = np.asarray(corr, dtype=np.float64)
    corr[~np.isfinite(corr)] = 0.0
    corr = np.maximum(corr, 0.0)
    return 2.0 * corr


def blur_1d(v, sigma):
    """Small Gaussian blur. Used only for smoothing random noise."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n <= 3:
        return v.copy()
    sigma = float(sigma)
    if sigma <= 0:
        return v.copy()
    half = int(max(1, min(1000, sigma * 2.5)))
    half = half | 1
    x = np.arange(-half, half + 1, dtype=np.float64)
    k = np.exp(-(x * x) / (2.0 * sigma * sigma))
    ks = float(k.sum())
    if ks <= 0 or not math.isfinite(ks):
        return v.copy()
    k /= ks
    out = np.convolve(v, k, mode="same")
    out[~np.isfinite(out)] = 0.0
    return out


def add_smooth_noise(p, rng, sigma):
    """Perturb log(p) with heavily smoothed Gaussian noise."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    noise = rng.normal(0.0, 1.0, n).astype(np.float64)
    noise = blur_1d(noise, max(1.0, n / 40.0))
    ns = float(np.linalg.norm(noise))
    if ns > 1e-300:
        noise /= ns
    logp = np.log(np.maximum(p, 1e-300)) + float(sigma) * noise
    return softmax_log(logp)


def make_mixture_exp(n, a, b, q):
    """
    Positive density:
        q * Exponential(a) on [0,1] left-anchored
      + (1-q) * Exponential(b) on [0,1] right-anchored.
    Each component is normalized separately.
    """
    a = float(a)
    b = float(b)
    q = float(q)
    if not math.isfinite(a) or a < 0.0:
        a = 1.0
    if not math.isfinite(b) or b < 0.0:
        b = 1.0
    if not math.isfinite(q) or q <= 0.0:
        q = 1.0
    if q >= 1.0:
        q = 1.0
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    left = np.exp(-a * x)
    ls = float(left.sum())
    if ls <= 0 or not math.isfinite(ls):
        left = np.ones_like(x)
        ls = float(left.sum())
    left /= ls

    right = np.exp(-b * (1.0 - x))
    rs = float(right.sum())
    if rs <= 0 or not math.isfinite(rs):
        right = np.ones_like(x)
        rs = float(right.sum())
    right /= rs

    p = q * left + (1.0 - q) * right
    return normalize_vector(p)


def resample_sequence(p_old, n_new):
    """Linearly interpolate an old probability vector to length n_new."""
    p_old = np.asarray(p_old, dtype=np.float64)
    p_old = normalize_vector(p_old)
    n_old = p_old.size
    if n_old == 0:
        return np.ones(n_new, dtype=np.float64)
    if n_old == n_new:
        return p_old.copy()
    x_old = np.linspace(0.0, 1.0, n_old, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    y = np.interp(x_new, x_old, p_old)
    y[~np.isfinite(y)] = 0.0
    return normalize_vector(y)


def score_fft(p, N=None):
    """Convenience exact score."""
    return exact_score(p, N)


def profile_search(n, deadline, rng):
    """
    Search a two-exponential mixture family:
        a in a_values, b in a_values, q in q_values.

    For each (a,b), precompute LL, RR, LR convolutions and sweep q cheaply.
    Returns list of (score, p) sorted ascending.
    """
    results = []
    if time.monotonic() >= deadline:
        return results

    N = npt_next_pow2(2 * n)
    L = 2 * n - 1
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    a_values = [
        0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0,
        4.5, 7.0, 11.0, 18.0, 30.0, 50.0
    ]
    q_values = np.linspace(0.02, 0.98, 33, dtype=np.float64)

    # Precompute left components for all a.
    left_data = []
    for a in a_values:
        left = np.exp(-float(a) * x)
        left = normalize_vector(left)
        F = np.fft.rfft(left, N)
        c = np.fft.irfft(F * F, N)[:L]
        c = np.maximum(c, 0.0)
        left_data.append((a, left, F, c))

    for a_idx, (a, left, FL, cLL) in enumerate(left_data):
        if time.monotonic() >= deadline:
            break
        for b_idx, (b, right, FR, cRR) in enumerate(left_data):
            if time.monotonic() >= deadline:
                break
            cLR = np.fft.irfft(FL * FR, N)[:L]
            cLR = np.maximum(cLR, 0.0)

            qs = q_values
            q2 = qs * qs
            qq1 = 2.0 * qs * (1.0 - qs)
            r2 = (1.0 - qs) * (1.0 - qs)

            # Vectorized: shape (len(q), L).
            c_all = (q2[:, None] * cLL[None, :] +
                     qq1[:, None] * cLR[None, :] +
                     r2[:, None] * cRR[None, :])
            cm = c_all.max(axis=1)
            scores = 2.0 * n * cm
            finite = np.isfinite(scores)
            if not np.any(finite):
                continue
            for qi in np.where(finite)[0]:
                score = float(scores[qi])
                q = float(qs[qi])
                p = make_mixture_exp(n, a, b, q)
                results.append((score, p))

    if not results:
        for a in [1.0, 1.5, 2.0]:
            p = make_mixture_exp(n, a, a, 1.0)
            s = exact_score(p, N)
            results.append((s, p))
        return results

    # Keep distinct best params by rounding, so refinements do not all duplicate.
    results.sort(key=lambda z: z[0])
    # Refine top entries by local coordinate searches in (a,b,q).
    top = []
    seen = set()
    for score, p in results:
        key = (round(score, 10))
        if key in seen:
            continue
        seen.add(key)
        top.append((score, p))
        if len(top) >= 24:
            break

    refined = []
    for i, (score, p) in enumerate(top[:8]):
        # Invert approximate parameters? Instead refine by treating p as candidate
        # and doing local parameter random moves around a nearby pure exponential
        # start. This is simpler and still useful.
        base_a = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0][i % 7]
        base_b = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0][(i + 1) % 7]
        base_q = float(np.clip(0.2 + 0.5 * (i % 7) / 6.0, 0.05, 0.95))

        best_a, best_b, best_q, best_s = base_a, base_b, base_q, score
        for _ in range(70):
            if time.monotonic() >= deadline:
                break
            typ = int(rng.integers(0, 3))
            a2, b2, q2 = best_a, best_b, best_q
            if typ == 0:
                a2 = max(0.0, best_a + rng.normal(0.0, 0.12 * max(best_a, 1.0) + 0.05))
            elif typ == 1:
                b2 = max(0.0, best_b + rng.normal(0.0, 0.12 * max(best_b, 1.0) + 0.05))
            else:
                q2 = float(np.clip(best_q + rng.normal(0.0, 0.035), 0.02, 0.98))
            a2 = float(min(50.0, max(0.0, a2)))
            b2 = float(min(50.0, max(0.0, b2)))
            cand = make_mixture_exp(n, a2, b2, q2)
            cs = exact_score(cand, N)
            if cs < best_s:
                best_a, best_b, best_q, best_s = a2, b2, q2, cs
            if time.monotonic() >= deadline:
                break
        refined.append((best_s, make_mixture_exp(n, best_a, best_b, best_q)))

    refined.sort(key=lambda z: z[0])
    out = refined[:5]
    for item in top:
        if len(out) >= 5:
            break
        if item not in out:
            out.append(item)

    out.sort(key=lambda z: z[0])
    return out[:5]


def dual_frank_wolfe_like(p, n, N, target_frac=0.95, iters=25, mix=0.25):
    """
    Frank-Wolfe-like step: linearize self-convolution at p, solve approximately
    for a non-negative direction g such that conv(p,g) <= M_target, then mix.
    """
    p = normalize_vector(p)
    start_score = exact_score(p, N)
    c_p, Fp = convolution_and_fft(p, N)
    Mc = float(c_p.max())
    avg = 1.0 / (2.0 * n - 1.0)
    Mtarget = float(target_frac) * Mc
    if Mtarget < avg * 1.0001:
        Mtarget = max(avg * 1.001, 0.999 * Mc)
    if Mtarget <= 0.0:
        Mtarget = 0.5 * Mc

    L = 2 * n - 1
    lam = np.zeros(L, dtype=np.float64)
    g = np.ones(n, dtype=np.float64) / float(n)

    best_p = p.copy()
    best_s = start_score
    check_every = 4

    for it in range(max(1, int(iters))):
        Fg = np.fft.rfft(g, N)
        h = np.fft.irfft(Fp * Fg, N)[:L]
        h = np.asarray(h, dtype=np.float64)
        h[~np.isfinite(h)] = 0.0
        h = np.maximum(h, 0.0)

        viol = h - Mtarget
        viol = np.maximum(viol, 0.0)
        if float(viol.sum()) > 0.0:
            lam += (0.5 * viol) / max(Mtarget, 1e-300)
            lm = float(lam.max())
            if lm > 100.0 and math.isfinite(lm):
                lam /= lm

        W = np.fft.rfft(lam, N)
        corr = np.fft.irfft(W * np.conjugate(Fp), N)[:n]
        corr = np.asarray(corr, dtype=np.float64)
        corr[~np.isfinite(corr)] = 0.0
        corr = np.maximum(corr, 0.0)
        cm = float(corr.max())
        if cm > 1e-300:
            g = softmax_log(-corr / cm)
        else:
            g = np.ones(n, dtype=np.float64) / float(n)

        if it % check_every == 0 or it == iters - 1:
            cand = (1.0 - float(mix)) * p + float(mix) * g
            cand = normalize_vector(cand)
            cs = exact_score(cand, N)
            if cs < best_s:
                best_s = cs
                best_p = cand.copy()

    return best_p, best_s


def local_search(p, n, deadline, current_best_score):
    """Simple random block multiplicative local search with exact FFT scoring."""
    p = normalize_vector(p)
    best = p.copy()
    best_score = exact_score(p)
    current_best_score = min(current_best_score, best_score)
    N = npt_next_pow2(2 * n)

    # Width ranges from very fine to broad.
    w_ticks = [
        int(max(2, n // 4096)),
        int(max(3, n // 1024)),
        int(max(4, n // 256)),
        int(max(5, n // 64)),
        int(max(6, n // 16)),
    ]
    w_ticks = sorted(set(w_ticks))
    if len(w_ticks) < 2:
        w_ticks = [2, max(3, n // 100)]
    ranges = []
    for i in range(len(w_ticks) - 1):
        ranges.append((w_ticks[i], w_ticks[i + 1], 0.15))
    ranges.append((w_ticks[-1], max(2 * w_ticks[-1], n // 3), 0.45))
    ranges.append((max(2, n // 8), max(3, n // 2), 0.9))

    tries = 0
    while time.monotonic() < deadline:
        if time.monotonic() >= deadline:
            break
        lo, hi, sigma = ranges[int(rng := (__import__('numpy').random.default_rng(0))) ] if False else ranges[tries % len(ranges)]
        # The line above is intentionally avoided below; keep clean simple selection.
        lo, hi, sigma = ranges[tries % len(ranges)]
        w = int(np.random.randint(lo, max(lo + 2, hi)))
        s = int(np.random.randint(0, max(1, n - w + 1)))
        cand = best.copy()
        mult = np.exp(float(np.random.normal(0.0, sigma)))
        cand[s:s + w] *= mult
        cand = normalize_vector(cand)
        cs = exact_score(cand, N)
        if cs < best_score - 1e-14:
            best = cand
            best_score = cs
            current_best_score = min(current_best_score, best_score)
        tries += 1
        if (tries % 5000) == 0:
            # occasionally try a smoothed tail of best to avoid tiny ripples
            try:
                sm = blur_1d(best, max(2.0, n / 200.0))
                sm = normalize_vector(sm)
                ss = exact_score(sm, N)
                if ss < best_score:
                    best = sm
                    best_score = ss
            except Exception:
                pass

    return best, current_best_score


def optimize_sequence(p, n, time_limit, rng):
    """
    Main smooth-max exponentiated gradient optimizer.
    """
    p = normalize_vector(p)
    N = npt_next_pow2(2 * n)
    L = 2 * n - 1

    start = time.monotonic()
    deadline = start + max(0.0, float(time_limit))

    best = p.copy()
    best_score = exact_score(p, N)
    if rng is None:
        rng = np.random.default_rng(0)

    # Phases: tau_factor * n.  Early phases are smoother, later phases sharper.
    tau_factors = [2.0, 5.0, 10.0, 20.0, 40.0]
    fracs = [0.12, 0.18, 0.22, 0.24, 0.24]

    it_total = 0
    for phase_idx, tf in enumerate(tau_factors):
        if phase_idx == len(tau_factors) - 1:
            phase_end = deadline
        else:
            phase_end = start + fracs[phase_idx] * float(time_limit)

        tau = float(tf) * float(n)
        step = 0.7
        no_improve = 0
        stuck_noise_sigma = 0.18

        while time.monotonic() < phase_end:
            if time.monotonic() >= deadline:
                break

            c, Fp = convolution_and_fft(p, N)
            smax, w = soft_max_weights(c, tau)
            score = float(2.0 * n * np.max(c))

            if score < best_score - 1e-13:
                best_score = score
                best = p.copy()
                no_improve = 0
                if it_total % 5000 == 0:
                    print(f"  optimize np={n} score={best_score:.7f} tau_factor={tf:g}")
            else:
                no_improve += 1

            if no_improve > 4500:
                # Smoothed random restart from current best.
                cand = add_smooth_noise(best, rng, stuck_noise_sigma)
                cs = exact_score(cand, N)
                if cs < best_score * 1.04:
                    p = cand
                else:
                    p = best.copy()
                no_improve = 0
                step = max(0.25, 0.6 * step)
                stuck_noise_sigma = min(0.9, stuck_noise_sigma * 1.15)

            grad = gradient_from_weights(p, w, Fp, N, n)
            gmax = float(grad.max())
            if not math.isfinite(gmax) or gmax <= 1e-300:
                p = best.copy()
                break

            accepted = False
            cur_step = step
            for _ in range(4):
                alpha = float(cur_step) / gmax
                # Cap to avoid pathological collapse under very small gmax differences.
                alpha = min(alpha, 100.0 * float(n))
                logp = np.log(np.maximum(p, 1e-300)) - (alpha * grad)
                p_new = softmax_log(logp)

                # Reject obviously collapsed vectors.
                if p_new.max() > 0.45:
                    cur_step *= 0.5
                    continue

                c_new = convolution_eval(p_new, N)
                score_new = float(2.0 * n * np.max(c_new))
                smax_new, _ = soft_max_weights(c_new, tau)

                if (smax_new < smax - 1e-14) or (score_new < score - 1e-13):
                    p = p_new
                    if score_new < best_score - 1e-13:
                        best_score = score_new
                        best = p_new.copy()
                        no_improve = 0
                    accepted = True
                    step = min(1.8, cur_step * 1.06)
                    break

                cur_step *= 0.5

            if not accepted:
                step = max(0.08, step * 0.65)
                # If current p is far worse than best, step back.
                if score > best_score * 1.35:
                    p = best.copy()

            it_total += 1

            # Occasionally try Frank-Wolfe-like dual update inside later phases
            # if the main steps are slow to improve.
            if phase_idx >= 2 and no_improve % 4000 == 4000:
                try:
                    p_dual, s_dual = dual_frank_wolfe_like(
                        p, n, N, target_frac=0.94, iters=18, mix=0.25
                    )
                    if s_dual < best_score:
                        best = p_dual.copy()
                        best_score = s_dual
                        p = p_dual.copy()
                        no_improve = 0
                except Exception:
                    pass

        # Between phases: if current p drifted poorly, reset to best.
        try:
            cp = convolution_eval(p, N)
            sp = float(2.0 * n * np.max(cp))
            if sp > best_score * 1.15:
                p = best.copy()
        except Exception:
            p = best.copy()

    # Final exact cleanup.
    last = exact_score(best, N)
    if last < best_score:
        best_score = last
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Entry point. Searches for the best coefficient sequence under the given
    wall-clock budget and returns a Python list of non-negative floats.
    """
    start_time = time.monotonic()
    total_budget = float(budget_s)
    hard_deadline = start_time + max(1.0, total_budget) - 12.0

    # Choose length. 8192 is large enough for very small finite-size error,
    # but still fast for FFT-based local optimization.
    if total_budget < 500:
        n = 4096
    else:
        n = 8192

    rng = np.random.default_rng(seed)

    print(f"propose_candidate start: budget={total_budget:.1f}s n={n}")

    # Parent construction, if available.
    parent = None
    try:
        parent_raw = globals().get("height_sequence_1", None)
        if parent_raw is not None:
            arr = np.asarray(parent_raw, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                arr = np.where(np.isfinite(arr), arr, 0.0)
                arr = np.clip(arr, 0.0, 1000.0)
                if float(arr.sum()) > 1e-12:
                    parent = arr
    except Exception:
        parent = None

    # Budget profile search.
    profile_budget = min(170.0, max(20.0, total_budget * 0.10))
    profile_deadline = min(start_time + profile_budget, hard_deadline - 300.0)
    profiles = []
    try:
        profiles = profile_search(n, profile_deadline, rng)
        for s, p in profiles[:5]:
            print(f"profile candidate score={s:.7f}")
    except Exception as e:
        print("profile_search failed:", repr(e))

    # Build initial starts.
    starts = []

    for _, p in profiles[:3]:
        starts.append(normalize_vector(p))

    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Guaranteed useful fallback profiles.
    starts.append(make_mixture_exp(n, 1.0, 1.0, 1.0))
    starts.append(make_mixture_exp(n, 1.5, 1.0, 1.0))
    starts.append(make_mixture_exp(n, 2.0, 2.0, 1.0))

    # If same-length parent exists, also try a mirror/blend with best profile.
    if parent is not None and len(starts) >= 2:
        try:
            blend = 0.6 * normalize_vector(starts[0]) + 0.4 * resample_sequence(parent, n)
            blend = normalize_vector(blend)
            starts.append(blend)
        except Exception:
            pass

    # Remove trivially empty/bad starts.
    clean_starts = []
    seen_scores = []
    for p in starts:
        try:
            p = normalize_vector(p)
            if p.size != n:
                continue
            s = exact_score(p)
            if not math.isfinite(s):
                continue
            # Avoid too many starts with nearly identical score.
            if any(abs(s - old) < 1e-8 for old in seen_scores):
                continue
            seen_scores.append(s)
            clean_starts.append(p)
            print(f"start score={s:.7f}")
        except Exception:
            pass
    if not clean_starts:
        clean_starts = [make_mixture_exp(n, 1.0, 1.0, 1.0)]

    best = clean_starts[0].copy()
    best_score = exact_score(best)
    print(f"initial best score={best_score:.7f}")

    # Allocate time for main optimization.
    local_time = min(180.0, max(30.0, total_budget * 0.10))
    reserve = 18.0
    remaining = max(0.0, hard_deadline - time.monotonic() - local_time - reserve)

    # Use up to 3 starts. If time is short, use only 1.
    if remaining < 80.0:
        n_starts = 1
    elif remaining < 400.0:
        n_starts = min(2, len(clean_starts))
    else:
        n_starts = min(3, len(clean_starts))

    per_start = remaining / max(1, n_starts)
    for idx in range(n_starts):
        if time.monotonic() >= hard_deadline - local_time - reserve - 8:
            break
        limit = per_start - 3.0
        if limit < 10.0:
            break
        p0 = clean_starts[idx]
        try:
            p_opt, s_opt = optimize_sequence(p0, n, limit, rng)
            if s_opt < best_score:
                best = p_opt.copy()
                best_score = s_opt
                print(f"optimizer start {idx}: best={best_score:.7f}")
            else:
                print(f"optimizer start {idx}: no improvement, best remains {best_score:.7f}")
        except Exception as e:
            print("optimizer failed:", repr(e))

        # Polish with a Frank-Wolfe-like step if time remains.
        try:
            if time.monotonic() < hard_deadline - local_time - reserve - 25:
                N = npt_next_pow2(2 * n)
                p_dw, s_dw = dual_frank_wolfe_like(
                    best, n, N, target_frac=0.93, iters=30, mix=0.30
                )
                if s_dw < best_score:
                    best = p_dw.copy()
                    best_score = s_dw
                    print(f"dual polish score={best_score:.7f}")
        except Exception as e:
            print("dual polish failed:", repr(e))

    # If significant extra time remains, continue polishing the current best.
    extra = max(0.0, hard_deadline - local_time - reserve - time.monotonic())
    if best is not None and extra > 25.0:
        try:
            p_extra, s_extra = optimize_sequence(best, n, extra, rng)
            if s_extra < best_score:
                best = p_extra.copy()
                best_score = s_extra
                print(f"extra polish score={best_score:.7f}")
        except Exception as e:
            print("extra polish failed:", repr(e))

    # Final local search.
    local_deadline = min(hard_deadline - 2.0, time.monotonic() + local_time)
    try:
        if time.monotonic() < local_deadline:
            best_l, best_score = local_search(best, n, local_deadline, best_score)
            if best_score > exact_score(best_l):
                best_score = exact_score(best_l)
            print(f"after local search score={best_score:.7f}")
    except Exception as e:
        print("local search failed:", repr(e))

    # Final exact internal score.
    try:
        s_internal = exact_score(best)
        if s_internal < best_score:
            best_score = s_internal
    except Exception:
        pass

    # Prepare output.
    out = np.asarray(best, dtype=np.float64)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    out = normalize_vector(out)
    out_list = [float(x) for x in out]

    # Verify with the official evaluator if available.
    try:
        official = float(evaluate_sequence(out_list))
        print(f"internal final={best_score:.7f} official={official:.7f}")
    except Exception:
        print(f"internal final={best_score:.7f}")

    return out_list
```
