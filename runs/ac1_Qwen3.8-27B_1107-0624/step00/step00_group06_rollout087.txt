```python
"""
Search for low-score nonnegative step functions.

Key construction:
  * Start from the finite prefix of the infinite sequence whose coefficients
    are those of (1 - z)^(-1/2), i.e. x_k = binom(2k,k)/4^k.
    This prefix has perfectly flat convolution on the first n lags, giving an
    asymptotic score near pi/2, far better than the all-ones value 2.
  * Then optimize in log-space using a smoothed log-score subgradient:
        grad_l = (sum_k w_k x_{k-l}) / M - alpha / S
    where M is the current peak convolution value, S is the sum, and w_k
    is a softmax/peak-emphasis weight over lags.
  * Use adaptive step size, beta annealing, random jumps/restarts, and exact
    calls to evaluate_sequence for safety and best-sequence selection.
"""

import numpy as np
import time


def flat_prefix(n: int) -> np.ndarray:
    """Finite prefix of the flat-convolution infinite sequence."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    ratios = np.ones(n, dtype=np.float64)
    if n > 1:
        i = np.arange(1, n, dtype=np.float64)
        ratios[1:] = (2.0 * i - 1.0) / (2.0 * i)
    return np.cumprod(ratios)


def fft_conv_square(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x * x using FFT."""
    n = int(x.size)
    m = 2 * n - 1
    N = 1 << max(0, (m - 1).bit_length())
    f = np.fft.rfft(x, N)
    c = np.fft.irfft(f * f, N)
    return c[:m]


def fast_score(x: np.ndarray) -> float:
    """FFT-based approximation of evaluate_sequence, used for fast search."""
    n = int(x.size)
    if n <= 0:
        return float("inf")
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 0.0:
        return float("inf")
    b = fft_conv_square(x)
    M = float(np.max(b))
    if not np.isfinite(M) or M <= 0.0:
        return float("inf")
    return float((2.0 * n * M) / (s * s))


def full_correlate(x: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    h_l = sum_i x_i * w_{l+i}, for l=0..n-1.

    This is the subgradient-like contribution vector from lag weights w.
    """
    n = int(x.size)
    if n == 1:
        return np.array([float(x[0] * w[0])], dtype=np.float64)
    m = 3 * n - 2
    N = 1 << max(0, (m - 1).bit_length())
    w_rev = w[::-1].copy()
    fx = np.fft.rfft(x, N)
    fw = np.fft.rfft(w_rev, N)
    c = np.fft.irfft(fx * fw, N)
    return c[2 * n - 2 : n - 2 : -1].copy()


def active_weights(b: np.ndarray, M: float, beta: float) -> np.ndarray:
    """Smoothed weights over lags, emphasizing values near the peak M."""
    if M <= 0.0 or not np.isfinite(M):
        return np.full(b.size, 1.0 / max(1, b.size), dtype=np.float64)
    t = b / M
    np.clip(t, 0.0, 1.0, out=t)
    w = np.exp(float(beta) * (t - 1.0))
    s = float(np.sum(w))
    if (not np.isfinite(s)) or s <= 0.0:
        w = (t > 0.999).astype(np.float64)
        s = float(np.sum(w))
        if (not np.isfinite(s)) or s <= 0.0:
            w = np.full(b.size, 1.0 / max(1, b.size), dtype=np.float64)
            return w
    w /= s
    return w


def optimize_direct_n(n, initial_z, budget, rng):
    """
    Run the log-subgradient optimizer for a fixed length n.

    Returns:
      exact_score: np.inf if final exact verification was not done/succeeded
      seq: list of floats (best fast-search sequence)
      fast_score: best FFT-based score seen
      best_z: best log-state
    """
    n = int(n)
    if n < 256:
        n = 256
    budget = max(10.0, float(budget))
    t_start = time.time()
    deadline = t_start + budget - 4.0

    if initial_z is None:
        x = flat_prefix(n)
        z = np.log(x)
    else:
        z = np.asarray(initial_z, dtype=np.float64).copy()
        if z.size != n:
            x = flat_prefix(n)
            z = np.log(x)
        else:
            x = flat_prefix(n)  # only for safety if z is bad
            np.log(x, out=x)
            z += 0.0
            if not np.isfinite(z).all():
                z = np.log(x)

    z -= float(np.mean(z))
    np.clip(z, -40.0, 40.0, out=z)
    x = np.exp(z - float(np.max(z)))
    score = fast_score(x)

    best_z = z.copy()
    best_x = x.copy()
    best_fast = score

    beta = 8.0
    step = 0.32
    alpha = 1.0
    stall = 0
    it = 0
    last_exact_time = t_start
    exact_interval = 90.0 if n <= 4096 else 180.0

    while time.time() < deadline:
        S = float(np.sum(x))
        if not np.isfinite(S) or S <= 0.0:
            break

        b = fft_conv_square(x)
        M = float(np.max(b))
        if not np.isfinite(M) or M <= 0.0:
            break

        w = active_weights(b, M, beta)
        h = full_correlate(x, w)
        np.maximum(h, 0.0, out=h)

        # Gradient of smoothed log-score:
        # log(score) ~ log(max b) - 2 log(sum), up to scale factors.
        grad = h / M - alpha / S
        np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0, copy=False)

        lo = float(np.percentile(grad, 1.0))
        hi = float(np.percentile(grad, 99.0))
        span = hi - lo
        if not np.isfinite(span) or span <= 1e-14:
            break

        mid = 0.5 * (lo + hi)
        gn = (grad - mid) / (0.5 * span)
        np.clip(gn, -3.0, 3.0, out=gn)

        improved = False
        for factor in (1.0, 0.5, 0.25):
            zc = z - step * factor * gn
            zc -= float(np.mean(zc))
            np.clip(zc, -40.0, 40.0, out=zc)
            xc = np.exp(zc - float(np.max(zc)))
            sc = fast_score(xc)
            if np.isfinite(sc) and sc < score:
                z = zc
                x = xc
                score = sc
                improved = True
                break

        if improved:
            stall = 0
            if score < best_fast:
                best_fast = score
                best_x = x.copy()
                best_z = z.copy()
            step = min(1.2, step * 1.04)
            beta = max(5.0, beta * 0.99)
            alpha = min(1.5, alpha * 1.03)
        else:
            stall += 1
            sig = 0.05 + min(0.45, 0.005 * float(stall))
            zc = z + rng.normal(0.0, sig, n)
            zc -= float(np.mean(zc))
            np.clip(zc, -40.0, 40.0, out=zc)
            xc = np.exp(zc - float(np.max(zc)))
            sc = fast_score(xc)

            if np.isfinite(sc) and sc < score:
                z = zc
                x = xc
                score = sc
                improved = True
                stall = 0
                if score < best_fast:
                    best_fast = score
                    best_x = x.copy()
                    best_z = z.copy()
            else:
                if stall % 40 == 0:
                    step *= 0.85
                    beta = min(120.0, beta * 1.4)
                    alpha = max(0.25, alpha * 0.8)

                if stall > 250:
                    z = best_z + rng.normal(0.0, 0.18, n)
                    z -= float(np.mean(z))
                    np.clip(z, -40.0, 40.0, out=z)
                    x = np.exp(z - float(np.max(z)))
                    score = fast_score(x)
                    beta = min(120.0, beta * 1.5)
                    step = 0.35
                    alpha = 1.0
                    stall = 0

        it += 1
        if it % 50 == 0 and (time.time() - last_exact_time) > exact_interval:
            if time.time() < deadline - 5.0:
                try:
                    evaluate_sequence([float(v) for v in best_x])
                except Exception:
                    pass
            last_exact_time = time.time()

    seq = [float(v) for v in best_x]

    # Final exact verification only if enough time remains.
    exact_score = float("inf")
    if time.time() < deadline:
        try:
            exact_score = float(evaluate_sequence(seq))
        except Exception:
            exact_score = float("inf")

    if not np.isfinite(exact_score):
        # Fallback: a certainly valid flat prefix.
        try:
            fb = flat_prefix(n).tolist()
            es = float(evaluate_sequence(fb))
            if np.isfinite(es):
                return es, fb, best_fast, best_z.copy()
        except Exception:
            pass

    return exact_score, seq, best_fast, best_z.copy()

            if fz is not None and len(fz) == n:
                best_z = fz.copy()
        print(f"phase n={n} exact={es:.6f} fast={fs:.6f} global={best_score:.6f} t={time.time()-t0:.1f}")

    # Final phase: continue polishing the currently best length.
    rem = total_deadline - time.time()
    if rem > 90.0 and best_n > 0 and best_z is not None:
        es, seq, fs, fz = optimize_direct_n(best_n, best_z, min(rem - 20.0, total_budget), rng)
        if np.isfinite(es) and es < best_score:
            best_score = es
            best_seq = seq
            if fz is not None and len(fz) == best_n:
                best_z = fz.copy()
        print(f"final polish n={best_n} exact={es:.6f} fast={fs:.6f} global={best_score:.6f} t={time.time()-t0:.1f}")

    # Ensure a valid list is returned.
    if not best_seq or not np.all(np.isfinite(best_seq)):
        x = flat_prefix(1024)
        best_seq = [float(v) for v in x]

    # Final safeguard: the evaluator accepts nonnegative floats; clip if needed.
    return [float(max(0.0, min(1000.0, y))) for y in best_seq]
```
