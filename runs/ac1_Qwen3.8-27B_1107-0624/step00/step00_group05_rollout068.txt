import numpy as np
import time


def fast_self_conv(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = int(2 ** np.ceil(np.log2(max(size, 4))))
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    conv[conv < 0] = 0
    return conv


def compute_objective(a):
    """Compute the objective function value (fast, for optimization)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fast_self_conv(a)
    max_c = np.max(conv)
    return 2.0 * n * max_c / (S * S)


def gradient_direction(a, conv, top_k=5):
    """Compute the normalized descent direction for the objective."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return None

    if len(conv) <= top_k:
        peak_lags = np.arange(len(conv))
    else:
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]

    C_vals = conv[peak_lags]
    C_avg = np.mean(C_vals)

    rev_contrib = np.zeros(n)
    for k in peak_lags:
        k = int(k)
        valid_start = max(0, k - n + 1)
        valid_end = min(k, n - 1)
        if valid_start <= valid_end:
            src_start = valid_end
            src_end = valid_start
            rev_contrib[valid_start:valid_end + 1] += a[src_start:src_end - 1:-1]

    rev_contrib /= len(peak_lags)

    descent = C_avg - rev_contrib * S
    desc_norm = np.linalg.norm(descent)
    if desc_norm < 1e-15:
        return None
    return descent / desc_norm


def optimize_sequence(a, deadline, rng, top_k=5, max_iters=100000):
    """Gradient-based optimization of a single sequence with adaptive steps and perturbations."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return a, compute_objective(a)
    a = a / S

    current_score = compute_objective(a)
    best = a.copy()
    best_score = current_score

    step_size = 0.5
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_self_conv(a)
        descent = gradient_direction(a, conv, top_k)
        if descent is None:
            break

        step = (step_size / n) * descent
        new_a = a + step
        new_a = np.maximum(new_a, 0)

        new_S = np.sum(new_a)
        if new_S < 1e-10:
            a = best.copy()
            step_size *= 0.5
            stall += 1
            if stall > 200:
                break
            continue
        new_a = new_a / new_S

        new_score = compute_objective(new_a)

        if np.isfinite(new_score) and new_score < current_score:
            a = new_a
            current_score = new_score
            stall = 0
            step_size = min(step_size * 1.05, 1.5)
            if current_score < best_score:
                best = a.copy()
                best_score = current_score
        else:
            step_size *= 0.8
            stall += 1
            if stall > 30:
                pert_scale = 0.05 + 0.005 * min(stall, 50)
                pert = rng.standard_normal(n) * pert_scale
                a = np.maximum(best + pert, 0)
                S2 = np.sum(a)
                if S2 > 1e-10:
                    a = a / S2
                    current_score = compute_objective(a)

                    a = best.copy()
                    current_score = best_score
