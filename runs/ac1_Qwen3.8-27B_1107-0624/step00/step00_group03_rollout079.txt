"""
Hybrid log-sum-exp gradient descent with momentum for minimizing the peak
of the self-convolution of a non-negative sequence.

Key ideas:
- Normalize sum(s)=1, minimize 2n*max(conv(s,s))
- Use log-sum-exp as smooth approximation of max, with annealed sharpness
- FFT-based gradient computation for speed
- Momentum to escape oscillatory behavior at the constant-sequence fixed point
- Multiple diverse initializations + periodic diversification
- Simulated annealing acceptance for exploration
"""
import numpy as np
import time


def normalize_nonneg(s):
    """Clip to non-negative and normalize to sum=1."""
    s = np.clip(s, 0.0, None)
    total = s.sum()
    if total < 1e-15:
        return np.ones(len(s)) / len(s)
    return s / total


def compute_objective(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    s_sum = s.sum()
    if s_sum < 0.01:
        return float('inf')
    c = np.convolve(s, s)
    n = len(s)
    return 2.0 * n * c.max() / (s_sum * s_sum)


def compute_soft_max_and_weights(c, beta):
    """Compute soft max value and softmax weights from convolution c."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted -= shifted.max()
    exps = np.exp(shifted)
    log_sum = np.log(exps.sum())
    weights = exps / exps.sum()
    soft_val = c_max + log_sum / beta
    return soft_val, weights


def compute_gradient_fft(s, weights):
    """
    Compute grad[j] = 2 * sum_k weights[k] * s[k-j] for valid k-j.
    Uses FFT-based cross-correlation for O(n log n) speed.
    """
    n = len(s)
    N = 1
    while N < 3 * n:
        N <<= 1
    W_ext = np.zeros(N)
    W_ext[:len(weights)] = weights
    S = np.zeros(N)
    S[:n] = s
    T = np.fft.ifft(np.fft.fft(W_ext) * np.conj(np.fft.fft(S))).real[:n]
    return 2.0 * T


def make_initializations(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    # Linear ramp up
    s = np.arange(1, n + 1, dtype=np.float64)
    inits.append(normalize_nonneg(s))
    # Linear ramp down
    s = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(normalize_nonneg(s))
    # Half sine
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    inits.append(normalize_nonneg(s))
    # Gaussian center
    x = np.arange(n) - n / 2
    s = np.exp(-(x / (n / 3.0)) ** 2)
    inits.append(normalize_nonneg(s))
    # Two bumps
    x = np.arange(n)
    s = np.exp(-((x - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-((x - n * 0.75) / (n * 0.1)) ** 2)
    inits.append(normalize_nonneg(s))
    # Random log-normal
    for _ in range(3):
        s = np.exp(rng.normal(0, 0.7, n))
        inits.append(normalize_nonneg(s))
    # Random blocks
    for _ in range(3):
        n_blocks = rng.integers(3, 12)
        boundaries = np.sort(rng.choice(np.arange(1, n), n_blocks - 1, replace=False))
        s = np.zeros(n)
        for i in range(n_blocks):
            start = 0 if i == 0 else boundaries[i - 1]
            end = boundaries[i] if i < n_blocks - 1 else n
            s[start:end] = rng.exponential(1.0)
        inits.append(normalize_nonneg(s))
    # Anti-symmetric perturbation of constant
    base = np.ones(n) / n
    for _ in range(3):
        eps = rng.normal(0, 0.3, n)
        eps -= eps.mean()
        # Make it have an anti-symmetric component
        anti = eps - eps[::-1]
        s = base + 0.05 * anti / max(1.0, np.abs(anti).max())
        inits.append(normalize_nonneg(s))
    return inits


def run_optimization(s_init, n, rng, deadline, beta_start=1.0, beta_end=150.0,
                     lr_start=0.02, lr_end=0.001, momentum_decay=0.9):
    """
    Run log-sum-exp gradient descent with momentum on a single sequence.
    Returns best_s and best_obj found.
    """
    s = s_init.copy()
    best_s = s.copy()
    best_obj = compute_objective(s)
    momentum = np.zeros(n)
    it = 0
    T = 0.003
    T_min = 1e-10
    no_improve = 0
    total_budget_iters = 30000

    while time.monotonic() < deadline and it < total_budget_iters:
        it += 1
        progress = min(1.0, it / total_budget_iters)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # Compute convolution and soft gradient
        c = np.convolve(s, s)
        soft_val, weights = compute_soft_max_and_weights(c, beta)
        grad = compute_gradient_fft(s, weights)

        # Project gradient to tangent space (remove mean component)
        grad = grad - grad.mean()

        # Momentum update
        momentum = momentum_decay * momentum + (1.0 - momentum_decay) * grad
        s_new = s - lr * momentum
        s_new = np.clip(s_new, 0.0, None)
        s_new = normalize_nonneg(s_new)

        # Evaluate true objective
        trial_obj = compute_objective(s_new)

        # Accept/reject with simulated annealing
        if trial_obj < best_obj:
            s = s_new
            best_obj = trial_obj
            best_s = s_new.copy()
            no_improve = 0
        elif trial_obj < compute_objective(s) or (T > T_min and rng.random() < np.exp(-(trial_obj - compute_objective(s)) / T)):
            s = s_new
            no_improve += 1
        else:
            no_improve += 1

        # Cool
        T = max(T * 0.99995, T_min)

        # Diversification: if stuck, perturb
        if no_improve > 200:
            scale = 0.03 + 0.08 * rng.random()
            s = s * np.exp(rng.normal(0, scale, n))
            s = normalize_nonneg(s)
            cur_obj = compute_objective(s)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_s = s.copy()
            no_improve = 0
            T = max(T * 5.0, 0.005)

    return best_s, best_obj
