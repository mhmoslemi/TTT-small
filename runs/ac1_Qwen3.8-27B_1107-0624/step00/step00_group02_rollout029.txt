from scipy.optimize import linprog

def compute_grad_a(a, alpha):
    """Gradient of soft_max(convolve(a,a), alpha) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    w = exp_c / exp_c.sum()
    conv_result = np.convolve(w, a[::-1], mode='full')
    grad = 2.0 * conv_result[n-1:2*n-1]
    return grad, c.max()

def softmax_from_b(b):
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    return exp_b / exp_b.sum()

def gradient_optimize(n, seed, time_budget):
    """Optimize using softmax-parameterized gradient descent."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + time_budget
    
    b = 0.05 * rng.standard_normal(n)
    momentum = np.zeros(n)
    beta = 0.9
    alpha = 5.0
    lr_scale = 1.0
    
    best_val = np.inf
    best_b = b.copy()
    best_a = None
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        
        a = softmax_from_b(b)
        grad_a, max_c = compute_grad_a(a, alpha)
        
        # Convert to gradient w.r.t. b
        grad_b = grad_a * a - a * np.dot(grad_a, a)
        
        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad_b
        lr = lr_scale / alpha
        b = b - lr * momentum
        
        val = 2 * n * max_c
        if val < best_val:
            best_val = val
            best_b = b.copy()
            best_a = a.copy()
        
        # Adaptive alpha
        if it % 100 == 99:
            alpha = min(alpha * 1.15, 500.0)
            lr_scale = max(lr_scale * 0.95, 0.1)
    
    if best_a is None:
        best_a = softmax_from_b(best_b)
        best_val = 2 * n * np.convolve(best_a, best_a).max()
    
    return best_a, best_val

def lp_polish(a, max_iters=50, time_budget=30):
    """Polish using LP-based iteration."""
    deadline = time.monotonic() + time_budget
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_val = 2 * n * np.convolve(a, a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        # Build constraint matrix: A[k, i] = a[k-i] if 0 <= k-i < n
        A = np.zeros((2*n-1, n))
        for i in range(n):
            end = min(n, 2*n-1-i)
            A[i:i+end, i] = a[:end]
        
        # Solve: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Update with step size
        for t in [0.5, 0.3, 0.2, 0.1]:
            a_new = (1-t)*a + t*g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            new_val = 2 * n * np.convolve(a_new, a_new).max()
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
            if new_val < 2*n*(np.convolve(a,a)).max():
                a = a_new
                break
        else:
            # No improvement with any step size, stop
            break
    
    return best_a, best_val
