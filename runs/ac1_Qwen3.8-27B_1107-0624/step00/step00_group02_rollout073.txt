import numpy as np
import time


def fast_self_convolve(a):
    """Compute full self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return result[:size]


def compute_gradient(a, lags):
    """Compute sum of gradients dc_k/da for k in lags."""
    n = len(a)
    grad = np.zeros(n)
    for k in lags:
        i_s = max(0, k - n + 1)
        i_e = min(n - 1, k)
        if i_s <= i_e:
            idx = np.arange(i_s, i_e + 1)
            grad[idx] += 2.0 * a[k - idx]
    return grad


def objective_from_conv(c, n):
    """Compute objective given convolution array c and sequence length n."""
    return 2.0 * n * c.max()

    inits.append(np.exp(-0.5 * t ** 2) + 0.01)

    # 5. Cosine
    t = np.linspace(0, np.pi, n)
    inits.append(np.abs(np.cos(t)) + 0.01)

    # 6. Alternating high-low
    inits.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.3))
