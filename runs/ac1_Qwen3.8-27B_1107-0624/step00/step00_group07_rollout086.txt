```
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def _full_convolution(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2*n - 1
    N = 1 << (m - 1).bit_length()
    X = rfft(x, N)
    y = irfft(X * X, N)[:m]
    np.maximum(y, 0.0, out=y)
    return y

def _normalize(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
       return x.copy()
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s <= 1e-300:
       return np.ones_like(x) / x.size
    return x / s

def _score_simple(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
       return np.inf
    s = x.sum()
    if s <= 1e-300:
       return np.inf
    c = _full_convolution(x)
    if c.size == 0:
       return np.inf
    return float(2.0 * n * c.max() / (s*s))
```

```
def _smooth_noise(rng, n, passes=2):
   x = rng.normal(size=n)
   kernel = np.array([1.0, 2.0, 1.0]) / 4.0
   for _ in range(passes):
      x = np.convolve(x, kernel, mode='same')
```

```
def _lp_direction(f, M, rng, A=None):
   n=f.size
   if n==0: return None,A
   if not np.isfinite(M) or M <= 1e-300:
      return None,A
   h = f / M
   if not np.all(np.isfinite(h)):
      return None,A
   shape=(2*n-1,n)
   if A is None or A.shape != shape:
      A=np.zeros(shape, dtype=np.float64)
   else:
      A.fill(0.0)
   for i in range(n):
      A[i:i+n, i] = h
   b = np.ones(2*n-1, dtype=np.float64) * (1.0 + 1e-9)
   if rng is not None:
      w = 1.0 + 1e-5 * rng.random(n)
   else:
      w = np.ones(n, dtype=np.float64)
   c = -w
   bounds = [(0.0, None)] * n
   try:

   if res is None or not res.success:

   g = res.x
   if g is None:

   g = np.asarray(g, dtype=np.float64)
   np.clip(g, 0.0, None, out=g)
   s = g.sum()
   if s <= 1e-12:
      return None, A
   g /= s
```

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def _full_convolution(x):
    conv = np.asarray(x, dtype=np.float64)
    n = conv.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    fft_len = 1 << (m - 1).bit_length()
    f = rfft(conv, fft_len)
    y = irfft(f * f, fft_len)[:m]
    np.maximum(y, 0.0, out=y)
    return y


def _normalize(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x.copy()
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s <= 1e-300:
        return np.ones_like(x, dtype=np.float64) / float(x.size)
    return x / s


def _score_fast(x):
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return np.inf
    s = x.sum()
    if s <= 1e-300:
        return np.inf
    c = _full_convolution(x)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _central_sequence(n):
    n = int(n)
    b = np.empty(max(0, n), dtype=np.float64)
    if n <= 0:
        return b
    b[0] = 1.0
    for k in range(n - 1):
        b[k + 1] = b[k] * (2.0 * k + 1.0) / (2.0 * (k + 1))
    return b


def _smooth_noise(rng, n, passes=2):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    x = rng.normal(size=n)
    kernel = np.array([1.0, 2.0, 1.0], dtype=np.float64) / 4.0
    for _ in range(max(0, int(passes))):
        x = np.convolve(x, kernel, mode="same")
    return x


def _interpolate_to_length(ref, n):
    arr = np.asarray(ref, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s <= 0:
        return None
    arr = arr / s
    if arr.size == int(n):
        return arr
    xp = np.linspace(0.0, 1.0, arr.size)
    xnew = np.linspace(0.0, 1.0, int(n))
    return np.interp(xnew, xp, arr)


def _make_starts(n, rng, ref1=None, ref2=None):
    n = int(n)
    if n <= 0:
        return []

    starts = []

    base_un = _central_sequence(n)
    base = _normalize(base_un)
    starts.append(base)

    idx = np.arange(n, dtype=np.float64)

    for lam in (-1.5, -0.75, 0.75, 1.5):
        x = base_un * np.exp(lam * idx / float(n))
        x = np.clip(x, 1e-300, None)
        starts.append(_normalize(x))

    for eta in (0.10, 0.20, 0.35):
        x = base_un * np.exp(eta * _smooth_noise(rng, n, 3))
        x = np.clip(x, 1e-300, None)

def _lp_direction(f, M, rng, A=None):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return None, A
    if not np.isfinite(M) or M <= 1e-300:
        return None, A

    h = f / M
    if not np.all(np.isfinite(h)):
        return None, A

    shape = (2 * n - 1, n)
    if A is None or A.shape != shape:
        A = np.zeros(shape, dtype=np.float64)
    else:
        A.fill(0.0)

    for i in range(n):
        A[i:i + n, i] = h

    b = np.ones(2 * n - 1, dtype=np.float64) * (1.0 + 1e-9)
    if rng is not None:
        w = 1.0 + 1e-5 * rng.random(n)
    else:
        w = np.ones(n, dtype=np.float64)
    c = -w

    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True}
        )
    except Exception:
        return None, A

    if res is None or not res.success:
        return None, A

    g = res.x
    if g is None:
        return None, A
    g = np.asarray(g, dtype=np.float64)
    np.clip(g, 0.0, None, out=g)
    s = g.sum()
    if s <= 1e-12:
        return None, A
    g /= s
    return g, A


def _add_candidate_if_better(d, n, current_best_score):

        C = _full_convolution(f)
        M = C.max()
        S = f.sum()
        if not np.isfinite(M) or M <= 0 or S <= 0:
            break

        g, A = _lp_direction(f, M, rng, A=A)

        candidates = []
        if g is not None and np.linalg.norm(g - f) > 1e-8:
            for alpha in (0.05, 0.10, 0.20, 0.35, 0.50, 0.75, 1.00):
                d = (1.0 - alpha) * f + alpha * g
                candidates.append(d)

        noise = _smooth_noise(rng, n, 2)
        for eta in (0.02, 0.05, 0.10, 0.15):
            d = f * np.exp(eta * noise)
            candidates.append(d)

        improved = False
        count_eval = 0
        for d in candidates:
            cand, score = _add_candidate_if_better(d, n, best_score)
            count_eval += 1
            if cand is not None:
                f = cand
                best_f = cand.copy()
                best_score = score
                improved = True
                no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve == 20:
                kick = f * np.exp(0.30 * _smooth_noise(rng, n, 1))
                cand, score = _add_candidate_if_better(kick, n, best_score)
                if cand is not None:
                    f = cand
                    best_f = cand.copy()
                    best_score = score
                    improved = True
                    no_improve = 0

            if no_improve > 70:
                break

        if iteration % 25 == 0:
            print(f"n={n} iter={iteration} best_score={best_score:.10f}")

    return best_f, best_score


def _optimize_n(n, allocation, rng, ref1, ref2, best_seq, best_score):
    allocation = float(allocation)
    if allocation <= 0:
        return best_seq, best_score

    starts = _make_starts(int(n), rng, ref1, ref2)
    if not starts:
        return best_seq, best_score

    t0 = time.monotonic()
    deadline = t0 + allocation
    primary_fraction = 0.75
    primary_deadline = t0 + max(30.0, primary_fraction * allocation)

    for idx, f0 in enumerate(starts):
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break

        if idx == 0:
            run_deadline = min(deadline, primary_deadline)
        else:
            share = (1.0 - primary_fraction) * allocation / max(1, len(starts) - 1)
            run_deadline = min(deadline, time.monotonic() + max(20.0, share))

        seq, score = _improve_sequence(f0, run_deadline, rng)
        if seq is not None and np.isfinite(score) and score < best_score:
            best_seq = seq
            best_score = score
            print(f"updated best n={int(n)} start={idx} score={best_score:.10f}")

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    budget_s = float(budget_s)
    t_start = time.monotonic()
    total_deadline = t_start + max(0.0, budget_s - 12.0)
    rng = np.random.default_rng(int(seed))

    previous = None
    try:
        previous = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        previous = None

    best_seq = None
    best_score = np.inf

    def _try_block(n, allocation, ref1, ref2):
        nonlocal best_seq, best_score
        if allocation <= 0:
            return
        try:
            best_seq, best_score = _optimize_n(n, allocation, rng, ref1, ref2, best_seq, best_score)
        except Exception:
            pass

    rem = total_deadline - time.monotonic()
    if rem > 250.0:
        _try_block(512, min(35.0, 0.04 * rem), previous, None)

    rem = total_deadline - time.monotonic()
    if rem > 650.0:
        ref_mid = previous
        if best_seq is not None and len(best_seq) > 0:
            ref_mid = best_seq
        _try_block(1024, min(300.0, 0.20 * rem), previous, ref_mid)

    rem = total_deadline - time.monotonic()
    if rem > 250.0:
        ref_main = previous
        if best_seq is not None and len(best_seq) > 0:
            ref_main = best_seq
        try:
            best_seq, best_score = _optimize_n(2048, rem, rng, previous, ref_main, best_seq, best_score)
        except Exception:
            rem = total_deadline - time.monotonic()
            if rem > 150.0:
                try:
                    best_seq, best_score = _optimize_n(1024, rem, rng, previous, ref_main, best_seq, best_score)
                except Exception:
                    pass

    if best_seq is None or not np.isfinite(best_score):
        best_seq = _normalize(_central_sequence(512))
        best_score = _score_fast(best_seq)

    try:
        exact = evaluate_sequence([float(x) for x in best_seq])
        if not np.isfinite(exact):
            fallback = _normalize(_central_sequence(512))
            best_seq = fallback
    except Exception:
        pass

    return [float(x) for x in best_seq]
```
