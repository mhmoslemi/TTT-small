"""

- Outer: update p as soft distribution over autocorrelation peaks

import numpy as np
import time


def simplex_project(v, total=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = total}."""
    n = len(v)
    if n == 0:
        return v
    # Find the threshold using the algorithm from Duchi et al.
    # We want max(v - tau, 0) to sum to total
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - total))[0]
    if len(rho) == 0:
        return np.full(n, total / n)
    rho = rho[-1]
    theta = (cssv[rho] - total) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a, fft_size):
    """Compute Q(p) @ a where Q[i,j] = p[i+j], using FFT.
    p has length 2n-1, a has length n.
    Result has length n: result[i] = sum_j p[i+j]*a[j] for j=0..n-1.
    """
    n = len(a)
    # result[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    # This is the cross-correlation: result = correlation(p, a) restricted to lags 0..n-1
    # correlation(p, a)[k] = sum_j p[j+k] * a[j] (for valid j)
    # We can compute this via FFT: it's the convolution of a (reversed) with p
    a_rev = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(a_rev, fft_size) * np.fft.rfft(p, fft_size), fft_size)
    # conv[k] = sum_j a_rev[j] * p[k-j] = sum_j a[n-1-j] * p[k-j]
    # Let m = n-1-j, so j = n-1-m:
    # conv[k] = sum_m a[m] * p[k-(n-1-m)] = sum_m a[m] * p[k-n+1+m]
    # We want result[i] = sum_j p[i+j]*a[j] = sum_m a[m] * p[i+m]
    # So result[i] = conv[k-n+1+m] where i+m = k-n+1+m, so k = i+n-1
    # result[i] = conv[i+n-1]
    return conv[n-1:n-1+n]


def compute_autocorr(a, fft_size):
    """Compute linear autocorrelation conv(a,a) of length 2n-1."""
    A = np.fft.rfft(a, fft_size)
    r = np.fft.irfft(A * A, fft_size)
    return r[:2*len(a)-1]

import numpy as np
import time


def simplex_project(v, total=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = total}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - total))[0]
    if len(rho) == 0:
        return np.full(n, total / n)
    rho = rho[-1]
    theta = (cssv[rho] - total) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a, fft_size):
    """Compute Q(p) @ a where Q[i,j] = p[i+j]. Returns vector of length len(a)."""
    n = len(a)
    a_rev = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(a_rev, fft_size) * np.fft.rfft(p, fft_size), fft_size)
    return conv[n - 1:n - 1 + n]


def compute_autocorr(a, fft_size):
    """Compute linear autocorrelation conv(a,a), length 2n-1."""
    A = np.fft.rfft(a, fft_size)
    r = np.fft.irfft(A * A, fft_size)
    return r[:2 * len(a) - 1]


def fast_objective_normalized(a, fft_size):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    r = compute_autocorr(a, fft_size)
    return 2.0 * len(a) * np.max(r)

    a = np.full(n, 1.0 / n)
    y = a.copy()
    L = 2.0
    alpha = 1.0 / L

        a_prev = a.copy()

        momentum = t / (t + 3.0)
        y = a_new + momentum * (a_new - a_prev)

            a = a_new
            break
        a = a_new

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a

    best_val = fast_objective_normalized(a, fft_size)

    eta = 0.3
    for it in range(50000):
        if time.monotonic() > deadline:
            break

        r = compute_autocorr(a, fft_size)
        current_val = 2.0 * n * np.max(r)

        if current_val < best_val:
            best_val = current_val
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print(f"    Refine iter {it}: obj={2*n*np.max(r):.6f}")

        # Use all peaks above 99% threshold for sharper gradient
        peak_thresh = 0.99 * np.max(r)
        peaks = np.where(r >= peak_thresh)[0]

        grad = np.zeros(n)
        for k in peaks:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                grad[j_start:j_end + 1] += 2.0 * a[k - j_start:k - j_end + 1][::-1]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = eta / (grad_norm + 1e-10)
        a = a - step * grad
        a = simplex_project(a, 1.0)

        eta *= (1 - 5e-7)

    best_a = np.maximum(best_a, 0)
    s = np.sum(best_a)
    if s > 0:
        best_a = best_a / s
    return best_a, best_val


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for the best non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses direct subgradient descent, alternating minimax, and refinement.
    """
    total_deadline = time.monotonic() + budget_s - 15
    t_start = time.monotonic()

    # Get initial sequence
    try:
        a_raw = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        n = len(a_raw)
    except Exception:
        n = 6733
        a_raw = np.ones(n)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Normalize initial
    a_norm = np.maximum(a_raw, 0)
    s = np.sum(a_norm)
    if s < 1e-15:
        a_norm = np.full(n, 1.0 / n)
    else:
        a_norm = a_norm / s

    # Track best
    best_a = a_norm.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Initial score: {best_score:.6f}")

    rng = np.random.default_rng(seed)

    # Build list of initializations
    inits = []
    inits.append(("provided", a_raw.copy()))
    inits.append(("uniform", np.ones(n)))

    # U-shape (high at edges, low in middle)
    x = np.linspace(0, 1, n)
    inits.append(("u_shape", 0.1 + 4.0 * x * (1 - x)))

    # Bump (high in middle)
    inits.append(("bump", 0.1 + np.exp(-8 * (x - 0.5) ** 2)))

    # Two bumps
    inits.append(("two_bumps", 0.1 + np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)))

    # Linear ramp
    inits.append(("ramp", 0.1 + x))

    # Cosine (non-negative)
    inits.append(("cosine", 0.5 + 0.5 * np.cos(2 * np.pi * x)))

    for i, (name, a_init_i) in enumerate(inits):
        remaining = total_deadline - time.monotonic()
        if remaining < 90:
            break
        # Allocate time proportionally, giving more to first initializations
        frac = 0.35 if i == 0 else (0.2 if i == 1 else 0.12)
        time_alloc = min(remaining * frac, max(60, remaining / (len(inits) - i)))

        print(f"\n=== Init {i+1}/{len(inits)}: {name} | budget={time_alloc:.0f}s | remaining={remaining:.0f}s ===")

        # Phase A: Direct subgradient descent (fast, good for initial shaping)
        time_a = time_alloc * 0.4
        a_dsd, val_dsd = direct_subgradient_descent(a_init_i, n, fft_size, time_a, verbose=(i < 2))

        # Phase B: Alternating minimax (more structured)
        time_b = time_alloc * 0.5
        a_am, val_am = alternating_minimax(a_init_i, n, fft_size, time_b, seed=seed + i*100, verbose=(i < 2))

        # Phase C: Refinement from the best of A and B
        time_c = time_alloc * 0.1
        if time_c > 10:
            if val_dsd <= val_am:
                a_ref, val_ref = refine_with_direct(a_dsd, n, fft_size, time_c, verbose=False)
            else:
                a_ref, val_ref = refine_with_direct(a_am, n, fft_size, time_c, verbose=False)
        else:
            a_ref = a_dsd if val_dsd <= val_am else a_am
            val_ref = min(val_dsd, val_am)

        # Evaluate the best from this initialization
        candidates = [(a_dsd, val_dsd), (a_am, val_am), (a_ref, val_ref)]
        for a_cand, val_cand in candidates:
            a_cand = np.maximum(a_cand, 0)
            s_c = np.sum(a_cand)
            if s_c > 1e-15:
                a_cand = a_cand / s_c
                exact = float(evaluate_sequence(a_cand.tolist()))
                if exact < best_score:
                    best_score = exact
                    best_a = a_cand.copy()
                    print(f"  *** NEW BEST: {best_score:.6f} (from {name}) ***")

        elapsed = time.monotonic() - t_start
        print(f"  [{elapsed:.0f}s elapsed] Best so far: {best_score:.6f}")

    # Final global refinement
    remaining = total_deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Final refinement: {remaining:.0f}s ===")
        a_final, val_final = refine_with_direct(best_a, n, fft_size, remaining * 0.6, verbose=True)
        exact = float(evaluate_sequence(a_final.tolist()))
        if exact < best_score:
            best_score = exact
            best_a = a_final.copy()
            print(f"  Final refinement improved: {best_score:.6f}")

        # One more round of alternating if time permits
        remaining2 = total_deadline - time.monotonic()
        if remaining2 > 30:
            a_am2, val_am2 = alternating_minimax(best_a, n, fft_size, remaining2, seed=seed + 9999, verbose=True)
            exact2 = float(evaluate_sequence(a_am2.tolist()))
            if exact2 < best_score:
                best_score = exact2
                best_a = a_am2.copy()
                print(f"  Second AM improved: {best_score:.6f}")

    print(f"\n=== FINAL SCORE: {best_score:.6f} ===")
    return best_a.tolist()
