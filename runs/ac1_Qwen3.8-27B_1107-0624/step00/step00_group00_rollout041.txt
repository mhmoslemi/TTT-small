    """Compute the subgradient direction for minimizing max(conv)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()

        return np.ones(n) / n, max_b, conv

    S = np.where(conv >= threshold)[0]
    if len(S) == 0:

    # Compute subgradient: sg[j] = sum_{k in S} (a[k-j] + a[j-k])
    # Use a weighting: weight_k = (conv[k] - threshold) / (max_b - threshold + 1e-10)
    weights = (conv[S] - threshold) / (max_b - threshold + 1e-15)
    weights = weights / weights.sum()  # normalize

Looking at this problem, I need to minimize `2*n*max(convolve(a,a))/(sum(a))²` over non-negative sequences.

**Key mathematical insight**: For any non-negative sequence `a` with `sum=S`, the center of the convolution `(a*a)[n-1] = Σ a[i]·a[n-1-i] = |a · rev(a)|`. By Cauchy-Schwarz: `|a · rev(a)| ≥ (Σa)²/n = S²/n`. Therefore `max_b ≥ S²/n` and `score ≥ 2`. 

The **uniform sequence** (constant values) achieves exactly `score = 2`: the convolution is triangular with peak `S²/n` at the center, and Cauchy-Schwarz is tight (vector proportional to its own reverse).

So **2 is the theoretical minimum** for non-negative sequences. I'll implement a mirror-descent search to verify this numerically, with the uniform sequence as guaranteed-optimal fallback.

"""
Optimization for minimizing score = 2*n*max(convolve(a,a))/(sum(a))^2
over non-negative sequences a of length n.

Key finding: For non-negative a with sum S, the center of the convolution
(a*a)[n-1] = <a, rev(a)> >= S^2/n by Cauchy-Schwarz (where rev reverses a).
Thus max_b >= S^2/n and score >= 2. The uniform sequence achieves score = 2
exactly (Cauchy-Schwarz tight: a = c*1, so a and rev(a) are parallel).
Therefore 2 is the provable minimum for non-negative sequences.

This implementation runs mirror descent from multiple initializations to
numerically confirm no score below 2 is found, and returns the best result
(the uniform sequence, which achieves the optimal score of 2).

import numpy as np
import time

def mirrored_convolve(a):
    """Full via FFT convolution a*a with result length 2*n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa)[:2 * n - 1]

def compute_subgradient(conv, a, eps):
    """Compute weighted subgradient direction from near-peak positions."""
    n = len(a)
    peak = conv.max()
    if peak <= 1e-15:
        return np.ones(n)
    mask = np.zeros(2 * n - 1)
    nm = conv >= (1.0 - eps) * peak
    if not nm.any():
        nm[np.argmax(conv)] = True
    mask[nm] = 1.0
    # subgrad via convolution
    size = 1
    while size < 2 * n:
        size *= 2
    fm = np.fft.rfft(mask, size)
    fa = np.fft.rfft(a, size)
    full = np.fft.irfft(fm * fa)[:2 * n - 1]
    sgrad = full[n - 1:n - 1 + n]
    # Also add the symmetric part (a * mask_rev) contribution
    mask_rev = mask[::-1]
    fm2 = np.fft.rfft(mask_rev, size)
    full2 = np.fft.irfft(fm2 * fa)[:2 * n - 1]
    sgrad += full2[n - 1:n - 1 + n]
    return sgrad

    eta = 15.0
    eps = 0.02
    start_time = time.monotonic()
    no_improve = 0
    max_no_improve = 80
    roles = 1.0
    last_eval = 0

    while time.monotonic() - start_time < max_time:
        it = int((time.monotonic() - start_time) / 0.005)
        if it % 400 == 0:
            eta *= 0.72
        sgrad = compute_subgradient(conv, a, eps)
        sgrad_c = sgrad - sgrad.mean()
        norm_sg = np.max(np.abs(sgrad_c)) + 1e-15
        sgrad_c = sgrad_c / norm_sg
        sgrad_c = np.clip(sgrad_c, -roles * 4.0, roles * 4.0)
        weights = np.exp(-eta * sgrad_c)
        a_new = a * weights
        total = a_new.sum()
        if total < 1e-12:

            total = 1.0

        a = a_new

                        a = np.clip(a, 1e-12, 1000.0)
