"""
Search for a non-negative step function minimizing  2*n*max(conv(a,a))/sum(a)^2.

Approach:
  * Homogeneous objective (scale invariant) -> always keep sum(a)=1.
  * Fast objective via zero-padded FFT: linear self-convolution has DFT = A^2.
  * Build a small pool of diverse seeds (provided sequence + structured/random
    lengths), keep the best per length.
  * Spectral shaping (Gerchberg-Saxton style): repeatedly flatten |A| with
    mag <- mag**alpha and re-impose non-negativity + normalization.
  * Projected subgradient descent on the *actual* linear-convolution peak with
    an O(n) subgradient (slicing) and an exact FFT-based line search that only
    accepts improvements (monotone, so it can never hurt).
  * Light random local moves (block scaling / mass redistribution) again guarded
    by line search.
  * Track the global best; return it (validated with evaluate_sequence).
"""
import numpy as np
import time

_EPS = 1e-12


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fast_self_conv(a):
    """Linear self-convolution of a real 1-D array via FFT (DFT = A^2)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[: 2 * n - 1]
    np.maximum(b, 0.0, out=b)
    return b


def _fast_score(a):
    s = a.sum()
    if s < 1e-6:
        return np.inf
    b = _fast_self_conv(a)
    return float(2.0 * a.size * b.max() / (s * s))


def _norm(a):
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _subgradient(a, topk):
    """A valid subgradient of max_k b[k] for b = conv(a,a) (topk peaks)."""
    n = a.size
    b = _fast_self_conv(a)
    mb = b.max()
    if topk >= b.size:
        idx = np.arange(b.size)
    else:
        idx = np.argpartition(-b, topk - 1)[:topk]
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        L1 = max(0, k - (n - 1))
        R1 = min(k, n - 1)
        if L1 <= R1:
            g[L1:R1 + 1] += a[k - R1: k - L1 + 1][::-1]
        L2 = max(0, k)
        R2 = min(n - 1, k + n - 1)
        if L2 <= R2:
            g[L2:R2 + 1] += a[0: R2 - L2 + 1]
    g /= max(1, idx.size)
    return g


def _line_search_subgrad(a, g, rho, n_try):
    """Monotone projected subgradient step (only keeps improvements)."""
    n = a.size
    s0 = _fast_score(a)
    gn = np.linalg.norm(g)
    if gn < _EPS:
        return a, s0, False
    t0 = rho / gn
    best_a, best_s, improved = a, s0, False
    for m in n_try:
        t = t0 / m
        cand = a - t * g
        np.maximum(cand, 0.0, out=cand)
        nrm = _norm(cand)
        if nrm is None:
            continue
        sc = _fast_score(nrm)
        if sc < best_s - 1e-12:
            best_a, best_s, improved = nrm, sc, True
    return best_a, best_s, improved


def _spectral_shaping(a, iters, alpha, rng, noise):
    n = a.size
    N = _next_pow2(2 * n - 1)
    best = a.copy()
    best_s = _fast_score(best)
    cur = a.copy()
    L = _norm(cur)
    if L is None:
        return best, best_s
    cur = L
    for _ in range(iters):
        A = np.fft.rfft(cur, N)
        mag = np.abs(A)
        ph = np.exp(1j * np.angle(A))
        mag2 = mag ** alpha
        A2 = mag2 * ph
        nxt = np.fft.irfft(A2, N)[:n].copy()
        np.maximum(nxt, 0.0, out=nxt)
        if noise > 0.0:
            nxt += noise * np.linalg.norm(nxt) * rng.standard_normal(n)
            np.maximum(nxt, 0.0, out=nxt)
        nrm = _norm(nxt)
        if nrm is None:
            break
        sc = _fast_score(nrm)
        if sc < best_s - 1e-12:
            best, best_s = nrm, sc
        # keep iterating from the improved (or current) point
        cur = nrm
    return best, best_s


def _block_move(a, rng, wlo, whi):
    n = a.size
    cand = a.copy()
    w = int(rng.integers(max(1, n * wlo), max(2, n * whi)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    seg = rng.normal(1.0, 0.25, size=w)
    cand[st:st + w] *= seg
    np.maximum(cand, 0.0, out=cand)
    nrm = _norm(cand)
    return nrm


def _mass_move(a, rng):
    """Move a small amount of mass from a random donor to a random set."""
    n = a.size
    cand = a.copy()
    donor = int(rng.integers(0, n))
    amt = 0.01 * cand[donor]
    cand[donor] -= amt
    k = int(rng.integers(1, max(2, n // 50)))
    tgt = rng.integers(0, n, size=k)
    cand[tgt] += amt / k
    np.maximum(cand, 0.0, out=cand)
    return _norm(cand)


def _accept(a, s, cand):
    if cand is None:
        return a, s, False
    sc = _fast_score(cand)
    if sc < s - 1e-12:
        return cand, sc, True
    return a, s, False

            a, s, ok = _accept(a, s, cand)
        if improved or ok:
            best_a, best_s = a, s
            rho *= 0.95
        else:
            break
    _try_update(best_a, best_s)

    # ---- Validate with the authoritative evaluator and return ----
    res = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
    try:
        real = float(evaluate_sequence(res))
    except Exception:
        real = np.inf
    print("[final] len=%d fast=%.6f real=%.6f" % (len(res), _fast_score(best_a), real), flush=True)

    if not np.isfinite(real) or real < 0.01:
        # guaranteed valid fallback (non-negative, positive sum)
        fb = [0.5] * 256
        return fb
    return res
