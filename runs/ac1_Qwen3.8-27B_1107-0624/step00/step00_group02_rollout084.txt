def solve_lp_direction(p, M):
    n=len(p)
    # build C
    offsets=[]; diags=[]
    for d in range(1,n):
       diags.append(p[d]*np.ones(n-d))
       offsets.append(d)
    diags.append(p[0]*np.ones(n))
    offsets.append(0)
    for d in range(1,n):
       diags.append(p[d]*np.ones(n-d))
       offsets.append(-d)
    C=scipy.sparse.diags(diags, offsets=offsets, shape=(2*n-1,n), format='csc')

    if res.success: return res.x

       x = 1.0 + 0.4*np.abs(rng.normal(0,1,n))

def _anti_gradient_direction(p, power=4.0):
    p = _normalize(p); n=p.size
    if n==0: return np.ones(1)
    b = np.convolve(p,p)
    w = b ** power
    s = float(w.sum())
    if s <= 1e-300:
        return p.copy()
    w /= s
    # gradient of smooth max
    conv = np.convolve(p, w[::-1])
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    gmax = float(np.max(grad))
    if gmax <= 1e-300:
        return p.copy()
    v = np.exp(-4.0 * (grad / gmax))
    v += 1e-12
    v /= v.sum()
    return v

```python
"""
LP-guided search for nonnegative sequences with low normalized peak self-convolution.

The score is 2*n*max(conv(a,a))/(sum(a)^2) and is scale invariant, so all working
sequences are normalized to sum 1.  For a normalized nonnegative sequence p, an
auxiliary LP direction is solved:

    max 1^T y
    s.t. T(p) y <= 1, y >= 0,

where T(p) is the convolution operator.  The LP direction is normalized to a
probability vector q and then mixed with p by a line search.  Random restarts,
smooth-gradient fallback directions, and length-adaptive search include both
length 1024 and length 2048 sequences.
"""

import numpy as np
import time
import scipy.sparse
from scipy.optimize import linprog


_LP_CACHE = {}


def _get_lp_structure(n: int):
    """Precompute sparse convolution-matrix structure for length n."""
    if n in _LP_CACHE:
        return _LP_CACHE[n]

    # Column i has rows i, i+1, ..., i+n-1, with values p[0], p[1], ..., p[n-1].
    idx2d = (
        np.arange(0, n * n, dtype=np.int32).reshape(n, n)
        + np.arange(n, dtype=np.int32).reshape(1, n)
    )
    indices = idx2d.ravel()
    indptr = np.arange(0, n * n + 1, n, dtype=np.int32)
    ub = np.ones(2 * n - 1, dtype=np.float64)

    _LP_CACHE[n] = (indices, indptr, ub)
    return _LP_CACHE[n]


def _normalize(p):
    """Project to the nonnegative simplex."""
    p = np.asarray(p, dtype=np.float64)
    if p.ndim != 1 or p.size == 0:
        return np.ones(1, dtype=np.float64)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-15:
        return np.ones(p.size, dtype=np.float64) / float(p.size)
    return p / s


def _score(p):
    """Exact objective for nonnegative sequences, using normalization invariance."""
    p = _normalize(p)
    n = p.size
    if n == 0:
        return np.inf
    b = np.convolve(p, p)
    return float(2.0 * n * np.max(b))


def _adapt_length(seq, n: int):
    """Interpolate a sequence to a new length while preserving nonnegativity."""
    seq = np.asarray(seq, dtype=np.float64)
    m = seq.size
    if m == n:
        return _normalize(seq)
    if m <= 1:
        return np.ones(n, dtype=np.float64) / float(n)

    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    y = np.interp(x_new, x_old, seq)
    return _normalize(y)


def _random_init(n: int, kind: int, rng: np.random.Generator):
    """Create diverse initial nonnegative sequences."""
    k = int(kind) % 8
    t = np.linspace(0.0, 1.0, n)

    if k == 0:
        x = 1.0 + 0.6 * t
    elif k == 1:
        x = 1.0 + 0.6 * (1.0 - t)
    elif k == 2:
        x = 1.0 + 0.3 * np.abs(rng.normal(0.0, 1.0, n))
    elif k == 3:
        x = np.exp(0.5 * t)
    elif k == 4:
        x = np.exp(0.5 * (1.0 - t))
    elif k == 5:
        x = rng.exponential(1.0, n) + 1e-6
    elif k == 6:
        x = np.zeros(n, dtype=np.float64)
        m = max(8, min(n, int(3.0 * np.sqrt(n))))
        idx = rng.choice(n, size=m, replace=False)
        x[idx] = rng.gamma(1.0, 1.0, m)
        x += 1e-6
    else:
        x = 1.0 + 0.25 * rng.random(n)

    return _normalize(x)


def _perturb(p, rng: np.random.Generator, strength: float):
    """Finite random perturbation used to escape local stagnation."""
    p = np.asarray(p, dtype=np.float64)
    q = p * np.exp(rng.normal(0.0, float(strength), p.size))
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-15:
        return p.copy()
    q /= s

    if rng.random() < 0.35:
        r = _random_init(p.size, int(rng.integers(0, 8)), rng)
        alpha = min(0.6, 0.2 + 0.5 * float(strength))
        q = (1.0 - alpha) * p + alpha * r
        q = _normalize(q)

    return q


def _make_t_list(t: float):
    """Line-search step sizes for p -> p + beta*(q-p)."""
    base = float(t)
    if not np.isfinite(base) or base <= 0.0:
        base = 0.1
    base = max(0.005, min(0.6, base))

    vals = []
    for v in (base, base * 0.5, base * 0.25, 0.1, 0.05, 0.02, 0.01, 0.005):
        if np.isfinite(v) and v > 1e-6:
            dup = False
            for u in vals:
                if abs(v - u) <= 1e-12:
                    dup = True
                    break
            if not dup:
                vals.append(v)
    return vals


def _anti_gradient_direction(p, power: float = 4.0):
    """
    Fallback probability vector.  It places mass where the smoothed-maximum
    subgradient is small, so a move toward it is more likely to reduce peaks.
    """
    p = _normalize(p)
    n = p.size
    if n == 0:
        return np.ones(1, dtype=np.float64)

    b = np.convolve(p, p)
    w = b ** power
    s = float(w.sum())
    if s <= 1e-300:
        return p.copy()
    w /= s

    # grad_i = 2 * sum_k w[k] p[k-i] (valid indices)
    conv = np.convolve(p, w[::-1])
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]

    gmax = float(np.max(grad))
    if gmax <= 1e-300:
        return p.copy()

    v = np.exp(-4.0 * (grad / gmax))
    v += 1e-12
    v /= v.sum()
    return v


def _solve_lp(p, deadline: float, rng: np.random.Generator):
    """
    Solve the LP direction:
        max sum(y)
        s.t. T(p) y <= 1, 0 <= y <= 100*n.
    A tiny secondary objective removes degeneracy.
    """
    rem = deadline - time.monotonic()
    if rem < 1.0:
        return None

    n = p.size
    lp_time = 8.0 if n <= 1024 else 18.0
    lp_time = min(lp_time, max(1.0, rem / 3.0))

    pf = np.maximum(p, 1e-6 / n)
    pf /= pf.sum()

    indices, indptr, ub = _get_lp_structure(n)
    data = np.tile(pf, n)
    A = scipy.sparse.csc_matrix((data, indices, indptr), shape=(2 * n - 1, n))

    c = -np.ones(n, dtype=np.float64)
    c += 1e-4 * pf

    if rng is not None:
        r = rng.standard_normal(n)
        r -= r.mean()
        rn = float(np.linalg.norm(r))
        if rn > 1e-15:
            c -= 1e-5 * (r / rn)

    upper = 100.0 * float(n)

    base_opts = {"presolve": True, "output": False}

    try:
        opts = dict(base_opts)
        opts["time_limit"] = float(lp_time)
        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=(0.0, upper),
            method="highs",
            options=opts,
        )
    except Exception:
        # Fallback if time-limit option name differs or HiGHS option is rejected.
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=ub,
                bounds=(0.0, upper),
                method="highs",
                options=base_opts,
            )
        except Exception:
            return None

    if res is not None and res.success and res.x is not None:
        y = np.asarray(res.x, dtype=np.float64)
        if np.isfinite(y).all():
            y = np.maximum(y, 0.0)
            if float(y.sum()) > 0.0:
                return y

    return None


def _refine_lp(p, deadline: float, rng: np.random.Generator):
    """Iteratively improve p using LP directions and line searches."""
    p = _normalize(p)
    n = p.size
    score = _score(p)
    best = p.copy()
    best_score = score

    t = 0.2
    stall = 0

    while time.monotonic() < deadline:
        # If stuck, make a finite perturbation to break symmetry/local traps.
        if stall >= 8:
            strength = min(0.8, 0.15 + 0.05 * float(stall))
            p = _perturb(p, rng, strength)
            score = _score(p)
            stall = 0
            t = 0.1

        y = _solve_lp(p, deadline, rng)

        if y is None:
            q = _anti_gradient_direction(p)
        else:
            sy = float(y.sum())
            if sy <= 1e-12:
                q = p.copy()
            else:
                q = y / sy

                # In near-degenerate cases, LP can return a far simplex vertex.
                # Mix in a smoother descent bias.
                weak = sy < n * 1.02
                cap = 0.5 if n <= 32 else 0.2
                if float(np.max(q)) > cap:
                    q = np.minimum(q, cap)
                    qs = float(q.sum())
                    if qs <= 1e-15:
                        q = p.copy()
                    else:
                        q /= qs

                if weak:
                    qd = _anti_gradient_direction(p)
                    q = 0.5 * q + 0.5 * qd
                    q = np.maximum(q, 0.0)
                    qs = float(q.sum())
                    if qs <= 1e-15:
                        q = p.copy()
                    else:
                        q /= qs

        if np.linalg.norm(q - p) < 1e-10:
            stall += 1
            t = max(0.005, t * 0.7)
            continue

        accepted = False
        for tt in _make_t_list(t):
            cand = _normalize((1.0 - tt) * p + tt * q)
            sc = _score(cand)
            if sc < score - 1e-12:
                p = cand
                score = sc
                t = min(0.6, tt * 1.3)
                accepted = True
                if sc < best_score - 1e-12:
                    best = p.copy()
                    best_score = sc
                    stall = 0
                else:
                    stall += 1
                break

        if not accepted:
            tt = max(1e-5, t * 0.05)
            cand = _normalize((1.0 - tt) * p + tt * q)
            sc = _score(cand)
            if sc < score - 1e-12:
                p = cand
                score = sc
                t = min(0.6, tt * 1.3)
                if sc < best_score - 1e-12:
                    best = p.copy()
                    best_score = sc
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
                t = max(0.005, t * 0.7)

    return best, best_score


def _optimize_n(
    n: int,
    budget_s: float,
    best_score: float,
    best_seq,
    initial_seq,
    rng: np.random.Generator,
    verbose: bool = True,
):
    """Search at a fixed length n within a time budget."""
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s))

    kind = 0
    current = None
    if initial_seq is not None:
        current = _adapt_length(initial_seq, n)

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem <= 0.0:
            break

        if current is None:
            current = _random_init(n, kind, rng)
            kind += 1

        if n <= 512:
            rt = 20.0
        elif n <= 1024:
            rt = 35.0
        else:
            rt = 60.0

        rt = min(rem, max(10.0, rt))
        local_best, local_score = _refine_lp(current, time.monotonic() + rt, rng)

        if local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            if verbose:
                print(f"  n={n} new best {best_score:.6f} elapsed {time.monotonic() - start:.1f}s")

        # Continue from a promising local solution; otherwise restart randomly.
        if local_score < best_score + 1e-9:
            current = _perturb(local_best, rng, 0.25)
        else:
            current = None

    return best_score, best_seq


def propose_candidate(seed: int = 42, budget_s: float = 1800.0, **kwargs):
    """
    Main entry point.  Returns the best nonnegative sequence found before deadline.
    """
    rng = np.random.default_rng(int(seed))
    total = max(0.0, float(budget_s) - 12.0)

    if total <= 1.0:
        n0 = 256
        seq = np.ones(n0, dtype=np.float64) / float(n0)
        try:
            sc = float(evaluate_sequence([float(x) for x in seq]))
        except Exception:
            sc = 2.0
        print(f"final n={n0} score={sc:.6f}")
        return [float(x) for x in seq]

    # Known valid fallback: uniform sequences have score exactly 2.
    n0 = 1024
    best_seq = np.ones(n0, dtype=np.float64) / float(n0)
    best_score = _score(best_seq)
    print(f"initial n={n0} score={best_score:.6f}")

    # Phase 1: length 1024.  Fast enough for many LP iterations.
    phase1 = total
    phase2 = 0.0
    if total > 900.0:
        phase1 = min(1000.0, total * 0.55)
        phase2 = min(700.0, total * 0.30)
        if phase1 + phase2 > total - 60.0:
            phase1 = total * 0.55
            phase2 = total * 0.25

    best_score, best_seq = _optimize_n(
        1024, phase1, best_score, best_seq, None, rng, verbose=True
    )

    # Phase 2: length 2048, seeded by the best shorter sequence if available.
    rem = time.monotonic() + float(budget_s) - 12.0 - time.monotonic()
    if rem > 150.0 and phase2 > 0.0:
        p2 = min(phase2, rem * 0.75)
        if p2 > 60.0:
            best_score, best_seq = _optimize_n(
                2048, p2, best_score, best_seq, best_seq, rng, verbose=True
            )

    # Phase 3: continue length 1024 if substantial time remains.
    rem = time.monotonic() + float(budget_s) - 12.0 - time.monotonic()
    if rem > 60.0:
        best_score, best_seq = _optimize_n(
            1024, rem, best_score, best_seq, best_seq, rng, verbose=True
        )

    # Final cleanup and validation.
    best_seq = np.asarray(best_seq, dtype=np.float64)
    if best_seq.ndim != 1 or best_seq.size == 0:
        best_seq = np.ones(n0, dtype=np.float64) / float(n0)
    best_seq = np.maximum(best_seq, 0.0)
    s = float(best_seq.sum())
    if s <= 1e-15:
        best_seq = np.ones(n0, dtype=np.float64) / float(n0)
    else:
        best_seq /= s

    try:
        exact = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(exact):
            best_score = exact
    except Exception:
        pass

    print(f"final n={best_seq.size} score={best_score:.6f}")
    return [float(x) for x in best_seq]
```
