"""
Minimize score = 2*n*max(conv(f,f))/(sum f)^2 over non-negative f of length n.

Approach:
  * The score is scale invariant, so we fix sum(f)=1 and minimize 2*n*max(conv).
  * The full linear convolution is computed with an FFT (O(n log n)).
  * The (sub)gradient of the peaked convolution term is computed as a
    cross-correlation, also via FFT, so each iteration is fast (O(n log n)).
  * We use a smoothed-max (softmax-weighted) objective so the gradient is
    dense and stable, with adaptive learning rate + temperature annealing.
  * Multiple asymmetric starting points are tried (asymmetry is essential to
    beat the symmetric/constant lower barrier of 2.0), each given a time slice.
  * Between/after gradient phases we run a block-perturbation local search.
  * Always returns the best finite sequence found within the budget.
"""

import numpy as np
import time
import math


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _make_convolver(n):
    N = _next_pow2(3 * n - 2)
    return N


def _full_convolution(f, N):
    F = np.fft.rfft(f, N)
    c = np.fft.irfft(F * F, N)
    return c[:2 * len(f) - 1]


def _objective(conv, n):
    return 2.0 * n * float(np.max(conv))


def _gradient_for_smoothed_max(conv, f, t, N):
    # stable softmax weights over the convolution entries
    cx = conv - np.max(conv)
    e = np.exp(cx / t)
    wsum = e.sum()
    weights = e / wsum if wsum > 0 else np.zeros_like(conv)
    # d(smooth_max)/df[m] = 2 * sum_j weights[m+j] * f[j]  (linear cross-corr)
    Wr = np.fft.rfft(weights, N)
    Fr = np.fft.rfft(f, N)
    cross = np.fft.irfft(Wr * np.conj(Fr), N)
    grad = 2.0 * cross[:len(f)]
    smooth_val = float(np.max(conv) + t * math.log(wsum)) if wsum > 0 else float(np.max(conv))
    return grad, smooth_val


def _normalize_nonneg(f):
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s <= 1e-12:
        f = np.ones_like(f)
        s = f.sum()
    return f / s


def _optimize_from(f_init, N, n, deadline, rng, seed_label=""):
    f = _normalize_nonneg(f_init.copy())
    conv = _full_convolution(f, N)
    score = _objective(conv, n)
    best_f = f.copy()
    best_score = score

    lr = 0.1
    t = 1e-3
    stall = 0

    while time.monotonic() < deadline:
        conv = _full_convolution(f, N)
        grad, _ = _gradient_for_smoothed_max(conv, f, t, N)
        gnorm = float(np.max(np.abs(grad)))
        if not np.isfinite(gnorm) or gnorm == 0.0:
            break

        # adaptive learning rate scaled to the gradient magnitude
        step = lr * grad / (gnorm + 1e-30)

        f_try = f - step
        f_try = np.clip(f_try, 0.0, None)
        st = f_try.sum()
        if st <= 1e-12:
            lr *= 0.5
            stall += 1
            continue
        f_try = f_try / st

        conv_try = _full_convolution(f_try, N)
        score_try = _objective(conv_try, n)

        if score_try < score - 1e-12:
            f = f_try
            score = score_try
            stall = 0
            lr = min(lr * 1.03, 2.0)
            t = max(t * 0.9985, 5e-5)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                if len(seed_label):
                    print("  [grad %s] %f" % (seed_label, best_score))
        else:
            lr *= 0.5
            t = min(t * 1.03, 5e-2)
            stall += 1
            if stall > 400:
                # escape: add a small asymmetric random bump to best so far
                w = int(max(1, rng.integers(n // 60, n // 4)))
                s = int(rng.integers(0, n - w + 1))
                f_try = best_f.copy()
                f_try[s:s + w] *= float(np.exp(rng.normal(0.0, 0.2)))
                f_try = _normalize_nonneg(f_try)
                f = f_try
                conv = _full_convolution(f, N)
                score = _objective(conv, n)
                lr = 0.1
                t = 1e-3
                stall = 0

    return best_f, best_score


def _block_local_search(f, N, n, deadline, rng):
    f = _normalize_nonneg(f.copy())
    conv = _full_convolution(f, N)
    score = _objective(conv, n)
    best_f = f.copy()
    best_score = score
    sigma = 0.08

    while time.monotonic() < deadline:
        w = int(max(1, rng.integers(max(2, n // 200), max(3, n // 6))))
        s = int(rng.integers(0, n - w + 1))
        f[:w] = 0
        f_try = best_f.copy()
        f_try[s:s + w] *= float(np.exp(rng.normal(0.0, sigma)))
        f_try = _normalize_nonneg(f_try)
        conv_try = _full_convolution(f_try, N)
        score_try = _objective(conv_try, n)
        if score_try < best_score - 1e-12:
            best_f = f_try
            best_score = score_try
            sigma = min(sigma * 1.05, 0.5)
            print("  [block] %f" % best_score)
        else:
            sigma = max(sigma * 0.97, 0.01)

    return best_f, best_score

            fb = np.exp(-((x - c) / s) ** 2)
            inits.append((_normalize_nonneg(fb), "bump@%.2f" % c))

    for qq in (1.0, 2.0, 4.0, 8.0):
        fq = 1.0 + qq * (x - 0.35)
        inits.append((_normalize_nonneg(fq), "quad%.0f" % qq))

    # Evaluate all initial points, keep the best global answer.
    best_f = inits[0][0].copy()
    best_score = _objective(_full_convolution(best_f, N), n)
    print("Initial best score: %f" % best_score)

    # Time-sliced gradient optimization over the pool (skip clearly bad starts).
    pool = list(inits)
    # quick screen
    screened = []
    for fp, name in pool:
        s = _objective(_full_convolution(fp, N), n)
        screened.append((fp, name, s))
    screened.sort(key=lambda z: z[2])

    idx = 0
    while idx < len(screened) and time.monotonic() < deadline:
        fp, name, s = screened[idx]
        remaining = deadline - time.monotonic()
        if remaining < 15.0:
            break
        n_left = len(screened) - idx
        slice_t = min(remaining - 10.0, max(40.0, remaining / max(1, n_left) * 1.3))
        t_end = time.monotonic() + slice_t
        # only worth trying if not already terrible
        if s < best_score + 1.2:
            cand_f, cand_s = _optimize_from(fp, N, n, t_end, rng, name)
            if cand_s < best_score:
                best_score = cand_s
                best_f = cand_f
                print("[accept] %s -> %f" % (name, best_score))
        idx += 1

    # Final phase: block-perturbation local search + occasional gradient polish.
    if time.monotonic() < deadline - 10.0:
        t_end = min(deadline - 20.0, time.monotonic() + 450.0)
        best_f, best_score = _block_local_search(best_f, N, n, t_end, rng)

    if time.monotonic() < deadline - 10.0:
        t_end = deadline - 5.0
        best_f, best_score = _optimize_from(best_f, N, n, t_end, rng, "polish")

    best_f = np.clip(best_f, 0.0, 1000.0)
    final = _objective(_full_convolution(best_f, N), n)
    print("FINAL score: %f" % min(final, best_score))

    return [float(v) for v in best_f]
