def compute_score(f):
    """Compute the objective value."""
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    c = np.convolve(f, f)
    return 2 * len(f) * np.max(c) / s**2

def project_onto_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # Use the standard simplex projection algorithm
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) > (cssv - target_sum))
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_step(f, f_new, learning_rate):
    """Take a subgradient step to minimize max_k c[k]."""
    c = np.convolve(f_new, f_new)
    k_star = np.argmax(c)
    # Subgradient of c[k_star] w.r.t. f is: grad[i] = f[k_star - i] (for valid indices)
    n = len(f)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = f_new[j]
    # Scale gradient
    grad = 2 * grad / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else grad
    # Take step
    f_candidate = f_new - learning_rate * grad
    # Project onto simplex (f >= 0, sum = 1)
    f_candidate = project_onto_simplex(f_candidate, 1.0)
    return f_candidate
