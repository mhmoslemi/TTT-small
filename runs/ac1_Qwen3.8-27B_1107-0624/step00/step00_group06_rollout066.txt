Search for a non-negative step-function coefficient sequence minimizing
    F(a) = 2*n*max(conv(a,a)) / (sum(a))^2
using FFT-accelerated smoothed (log-sum-exp) gradient descent with an Adam
updater, multiple diverse initializations, an adaptive smoothing parameter,
and a final block-perturbation hill-climb that is verified against the
authoritative evaluate_sequence(). Scale invariance lets us normalize to
sum==1 inside the loop (then renormalize), so the optimizer minimizes
2*n*max(conv(a,a)) on the simplex.
"""

import numpy as np

# ----------------------------------------------------------------------
# Top-level helpers (no closures, no lambdas)
# ----------------------------------------------------------------------

def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _self_convolution(a):
    """Full linear convolution of a with itself via real FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = _next_pow2(size)
    A = fft(a, fsize)
    return np.real(ifft(A * A, fsize))[:size]


def _cross_corr(a, p):
    """
    g[i] = sum_{j=0}^{n-1} a[j] * p[i + j]  for i = 0..n-1,
    where len(p) = 2n-1. Computed as a linear cross-correlation via FFT.
    """
    n = len(a)
    N = _next_pow2(3 * n - 1)
    A = fft(a, N)
    P = fft(p, N)
    return np.real(ifft(np.conj(A) * P, N))[:n]


def _objective(a):
    """Direct value of 2*n*max(conv(a,a))/(sum a)^2 (matches evaluator)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = _self_convolution(a)
    return float(2.0 * n * np.max(c) / (S * S))
