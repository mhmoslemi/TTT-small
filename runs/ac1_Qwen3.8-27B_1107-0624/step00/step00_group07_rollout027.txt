Multi-phase optimization minimizing 2*n*max(auto-conv) for non-negative sequences.
Uses softmax-weighted gradient descent (PAPR-inspired), parametric initialization,
and local search refinement.
"""
import numpy as np
import time

def fast_conv_auto(p):
    n = len(p)
    size = 2 * n
    F = np.fft.rfft(p, size=size)
    return np.fft.irfft(F * F, n=size)[:2*n-1]

def compute_obj(p):
    n = len(p)
    conv = fast_conv_auto(p)
    return 2.0 * n * conv.max()

def softmax_step(p, T, lr):
    n = len(p)
    conv = fast_conv_auto(p)
    Tc = T * conv
    Tc -= Tc.max()
    w = np.exp(Tc)
    ws = w.sum()
    if ws < 1e-30:
        return p
    w /= ws
    N = 1 << (3 * n).bit_length()
    W = np.fft.rfft(w, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(W * np.conj(P), N)
    grad = 2.0 * corr[:n]
    p_new = p * np.exp(-lr * grad)
    p_new = np.maximum(p_new, 1e-30)
    p_new /= p_new.sum()
    return p_new

def try_parametric(n, beta_range):
    """Try several parametric families and return the best."""
    candidates = []
    
    # Exponential decay from center
    for alpha in beta_range:
        j = np.arange(n)
        p = np.exp(-alpha * (j - n/2.0)**2 / n)
        p /= p.sum()
        candidates.append(p)
    
    # U-shaped (cosine)
    for alpha in beta_range:
        j = np.arange(n)
        p = 1.0 + alpha * np.cos(2*np.pi*j/(n-1))
        if p.min() < 0:
            continue
        p /= p.sum()
        candidates.append(p)
    
    # Linear ramp
    for alpha in beta_range:
        j = np.arange(n)
        p = (j + 1) ** alpha
        p /= p.sum()
        candidates.append(p)
    
    # Bimodal
    for alpha in beta_range:
        j = np.arange(n)
        p = np.exp(-alpha*(j-n/4.0)**2/n) + np.exp(-alpha*(j-3.0*n/4.0)**2/n)
        p /= p.sum()
        candidates.append(p)
    
    best = None
    best_obj = np.inf
    for p in candidates:
        obj = compute_obj(p)
        if obj < best_obj:
            best_obj = obj
            best = p
    return best, best_obj

    param_p, param_obj = try_parametric(n, [0.5, 1, 2, 3, 5, 10])
    if param_obj < best_obj:
        best_p = param_p
        best_obj = param_obj
        print(f"After parametric: obj={best_obj:.6f}")
    
    # Phase 1: Softmax gradient descent
    for T in [1, 2, 5, 10, 20, 50, 100, 200, 500]:
        if time.monotonic() > deadline:
            break
        p = best_p.copy()
        conv = fast_conv_auto(p)
        Tc = T * conv
        Tc -= Tc.max()
        w = np.exp(Tc)
        obj_before = best_obj
        
        for lr in [10, 5, 2, 1, 0.5, 0.2, 0.1]:
            if time.monotonic() > deadline:
                break
            improved = True
            while improved and time.monotonic() < deadline:
                improved = False
                for it in range(100):
                    if time.monotonic() > deadline:
                        break
                    p = softmax_step(p, T, lr)
                    obj = compute_obj(p)
                    if obj < best_obj - 1e-10:
                        best_obj = obj
                        best_p = p.copy()
                        improved = True
                if not improved:
                    break
        if best_obj < obj_before:
            print(f"After T={T}: obj={best_obj:.6f}")
    
    # Phase 2: Local search with coordinate updates
    if time.monotonic() < deadline:
        p = best_p.copy()
        rng = np.random.default_rng(seed)
        conv = fast_conv_auto(p)
        
        while time.monotonic() < deadline:
            # Pick a random index to perturb
            j = rng.integers(0, n)
            # Try moving mass from j to a random other index k
            k = rng.integers(0, n)
            if j == k:
                continue
            # Try several delta values
            best_delta = 0
            best_local_obj = compute_obj(p)
            p_save = p.copy()
            for delta_frac in [0.01, 0.005, 0.002, -0.005, -0.01, 0.02]:
                delta = delta_frac * p[j]
                if p[j] - delta < 0:
                    continue
                p[j] -= delta
                p[k] += delta
                p /= p.sum()
                obj = compute_obj(p)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_delta = delta
                p = p_save.copy()

                p[j] -= best_delta
                p[k] += best_delta
                p /= p.sum()
                best_obj = best_local_obj
                best_p = p.copy()
    
    return [float(x) for x in best_p]
