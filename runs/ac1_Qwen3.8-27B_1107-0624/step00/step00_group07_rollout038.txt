def compute_score(seq):
    """Compute the score using FFT for speed."""
    n = len(seq)
    # Compute linear convolution using FFT

import numpy as np
import time

def compute_score_fast(a):
    """Compute score using FFT. a should be 1D array, sum(a) > 0."""
    n = len(a)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    F = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(F * F, fft_size)[:2*n-1]
    max_c = np.max(C)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (sum_a * sum_a))

def optimize_sequence(a_init, budget_s):
    """Optimize a sequence using multiplicative subgradient method."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n)
        S = 1.0
    a = a / S  # normalize to sum = 1
    
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    
    best_a = a.copy()
    best_score = compute_score_fast(a)
    deadline = time.monotonic() + budget_s
    
    t = 0
    eta = 0.1  # initial step size
    
    while time.monotonic() < deadline:
        # Compute convolution
        F = np.fft.rfft(a, fft_size)
        C = np.fft.irfft(F * F, fft_size)[:2*n-1]
        
        # Find max and subgradient
        k_star = int(np.argmax(C))
        
        # Subgradient: g[i] = 2*a[k_star - i] for valid i
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        g = np.zeros(n)
        indices = np.arange(lo, hi + 1)
        g[indices] = 2.0 * a[k_star - indices]
        
        # Adaptive step size
        g_norm = np.sqrt(np.sum(g * g))
        if g_norm < 1e-15:
            break
        eta = 0.5 / (g_norm * (1.0 + 0.01 * t))
        
        # Multiplicative update
        a = a * np.exp(-eta * g)
        
        # Ensure positivity and rescale
        a = np.maximum(a, 1e-15)
        S = np.sum(a)
        a = a / S
        
        # Check score periodically
        if t % 10 == 0:
            score = compute_score_fast(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        t += 1
        
        if t % 10000 == 0:
            print(f"  t={t}, score={best_score:.6f}, current_max_c={np.max(C):.8f}")
    
    return best_a, best_score

    n = len(f)
    b = cp.Variable(n, nonneg=True)
    
    # conv(f, b)[k] = sum_j f[j] * b[k-j]
    # We need to express this as a matrix multiplication
    # Let T be the Toeplitz-like matrix where T[k, j] = f[j] if 0 <= k-j < n, else 0
    # Then conv(f,b) = T @ b

    # conv(f,b)[k] = sum_{j} f[j] * b[k-j] for k = 0, ..., 2n-2
