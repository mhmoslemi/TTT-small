def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    rho = np.arange(1, n+1)
    # Find largest k such that sorted_v[k-1] > (cumsum[k-1] - S) / k
    condition = sorted_v > (cumsum - S) / rho
    k = np.sum(condition)
    if k == 0:
        return np.full(n, S / n)
    tau = (cumsum[k-1] - S) / k
    return np.maximum(0, v - tau)

    """Compute the evaluation score for sequence s."""
    n = len(s)
    c = np.convolve(s, s)
    M = np.max(c)
    S = np.sum(s)
    return 2 * n * M / (S * S)

import numpy as np
import time


def simplex_project(v, S):
    """Project vector v onto the non-negative simplex {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    rho = np.arange(1, n + 1, dtype=np.float64)
    condition = sorted_v > (cumsum - S) / rho
    k = int(np.sum(condition))
    if k == 0:
        return np.full(n, S / n, dtype=np.float64)
    tau = (cumsum[k - 1] - S) / k
    return np.maximum(0.0, v - tau)


def linear_ramp(n):
    """Generate normalized linear ramp [1/n, 2/n, ..., 1]."""
    return np.arange(1, n + 1, dtype=np.float64) / n


def compute_subgradient(s, m_star):
    """Compute gradient of c[m_star] = sum_k s[k]*s[m_star-k] w.r.t. s."""
    n = len(s)
    g = s.copy()
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    if lo <= hi:
        valid_k = np.arange(lo, hi + 1)
        g[valid_k] += s[m_star - valid_k]
    return g


def fast_score(s):
    """Compute 2n*max(conv)/sum^2 quickly."""
    n = len(s)
    c = np.convolve(s, s)
    M = float(np.max(c))
    S = float(np.sum(s))
    return 2.0 * n * M / (S * S), M
