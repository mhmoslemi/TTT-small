import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import math

# ---------------------------------------------------------------------------
# Minimize 2*n*max_k (a*a)[k] / (sum a)^2 over non-negative a of fixed length.
#
# The objective is scale invariant, so we normalise sum(a) = sqrt(2n) and then
# the target is simply to push down  max_k (a*a)[k].  The engine is a blend
# of:
#   * a linear-programming "fill-the-valleys" direction (the proof-backed
#     state-of-the-art update: fix a, maximise sum(g) subject to g>=0 and
#     (a*g)_k <= M),
#   * fast sub-block multiplicative perturbations,
#   * coordinate moves,
# run over several good initial profiles and restarted to escape local minima.
# A cheap closed-form evaluator (identical to the security-checked one for
# valid in-window inputs) is used for the inner loop.
# ---------------------------------------------------------------------------


def _obj(a, n):
    """Fast objective identical to evaluate_sequence for in-window inputs."""
    s = float(a.sum())
    if s < 0.01 or not math.isfinite(s):
        return np.inf
    c = np.convolve(a, a)
    mb = float(np.max(c))
    if not math.isfinite(mb):
        return np.inf
    return float(2.0 * n * mb / (s * s))


def _renorm(a, n):
    """Rescale to the canonical normalisation sum = sqrt(2n)."""
    target = math.sqrt(2.0 * n)
    s = float(a.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    return np.clip(a * (target / s), 0.0, 1000.0)


def _lp_fill_direction(a, n, M):
    """LP: maximise sum(g) s.t. g>=0 and (a*g)_k <= M for all k.

    Returns the (unnormalised) direction g, or None on failure.
    """
    # A[k, j] = a[k-j] for 0 <= k-j <= n-1, else 0  -> (a*g)_k
    row_parts = []
    col_parts = []
    dat_parts = []
    for d in range(n):
        m = n - d
        if m <= 0:
            break
        row_parts.append(np.arange(d, d + m, dtype=np.int32))
        col_parts.append(np.arange(0, m, dtype=np.int32))
        dat_parts.append(np.full(m, a[d], dtype=np.float64))
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    data = np.concatenate(dat_parts)
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n, dtype=np.float64)
    b = M * np.ones(2 * n - 1, dtype=np.float64)
    try:
        sol = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 120.0},
        )
    except Exception:
        return None
    if sol is None or not getattr(sol, "success", False) or sol.x is None:
        return None
    g = np.asarray(sol.x, dtype=np.float64)
    g = np.clip(g, 0.0, None)
    if not np.isfinite(g).all() or g.sum() <= 0.0:
        return None
    return g


def _subblock(a, rng, n):
    """Multiplicative random-width block perturbation."""
    b = a.copy()
    w = int(rng.integers(2, max(3, n // 8)))
    w = min(w, n)
    st = int(rng.integers(0, n - w + 1))

    if rng.random() < 0.15:
        b[st:st + w] *= float(np.exp(rng.normal(0.0, 0.35)))
    return np.clip(b, 0.0, 1000.0)


def _coord_move(a, rng, n):
    """Perturb a small random handful of single coordinates."""
    b = a.copy()
    for _ in range(int(rng.integers(1, 4))):
        i = int(rng.integers(n))
        b[i] *= float(np.exp(rng.normal(0.0, 0.4)))

def _anti_peak(a, rng, n):
    """Shift mass out of an index sitting in the dominant product of the
    maximum lag, into a random low-mass index."""
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    best_i = lo
    best_v = -1.0
    for i in range(lo, hi + 1):
        j = k - i
        v = a[i] * a[j]
        if v > best_v:
            best_v = v
            best_i = i
    donor = best_i
    recv = int(rng.integers(n))
    while recv == donor:
        recv = int(rng.integers(n))
    b = a.copy()
    frac = float(rng.uniform(0.1, 0.5))
    amt = b[donor] * frac
    b[donor] -= amt
    b[recv] += amt
    return np.clip(b, 0.0, 1000.0)

    # raised cosine window
    i = np.arange(n, dtype=np.float64)
    profiles.append(0.5 * (1.0 + np.cos(2.0 * math.pi * i / max(1, n - 1))))

    # triangular / tent
    mid = (n - 1) / 2.0
    profiles.append(np.clip(1.0 - np.abs(i - mid) / max(1.0, mid + 1.0), 0.0, 1.0))

    # linear glide (increasing)
    profiles.append(i + 1.0)

    # geometric (gently increasing, capped)
    r = 1000.0 ** (1.0 / max(1, n - 1))
    gseq = np.power(r, i)
    profiles.append(np.clip(gseq, 0.0, 1000.0))

    # ones
    profiles.append(np.ones(n, dtype=np.float64))

    # two-edge block
    e = max(2, n // 4)
    edge = np.zeros(n, dtype=np.float64)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    profiles.append(edge)

    returns = []
    for p in profiles:
        p = np.asarray(p, dtype=np.float64)
        if p.size != n or not np.isfinite(p).all():
            continue
        p = np.clip(p, 0.0, 1000.0)
        if p.sum() <= 0:
            continue
        returns.append(_renorm(p, n))
    # random sparse fill
    rng0 = np.random.default_rng(12345)
    rnd = rng0.random(n)
    rnd[rnd < 0.5] = 0.0
    if rnd.sum() > 0:
        returns.append(_renorm(rnd, n))
    if not returns:
        returns.append(np.ones(n, dtype=np.float64) * (math.sqrt(2.0 * n) / n))
    return returns


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a low-objective non-negative coefficient sequence.

    Combines an LP fill-the-valleys direction with fast local perturbations,
    run over several initial profiles with periodic restarts, all under a
    wall-clock budget.  Returns the best sequence found as a list of floats.
    """
    # ---- choose length -------------------------------------------------
    n = 2535
    try:
        _p = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        if _p.ndim == 1 and 200 <= _p.size <= 5000 and _p.max() > 0:
            n = int(_p.size)
    except Exception:
        pass

    deadline = time.monotonic() + max(0.0, float(budget_s) - 6.0)
    rng = np.random.default_rng(seed)

    # ---- pick best initial profile ------------------------------------
    inits = _init_profiles(n)
    best = None
    best_score = np.inf
    for prof in inits:
        prof = np.clip(prof, 0.0, 1000.0)
        sc = _obj(prof, n)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best = prof.copy()
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = _obj(best, n)
    print(f"[init] n={n} best_score={best_score:.6f}", flush=True)

    cur = best.copy()
    cur_score = best_score
    stuck = 0
    last_lp = 0.0

    def consider(cand):
        nonlocal cur, cur_score, best, best_score, stuck
        cand = np.clip(cand, 0.0, 1000.0)
        if not np.isfinite(cand).all():
            return
        sc = _obj(cand, n)
        if math.isfinite(sc):
            if sc < cur_score - 1e-12:
                cur = cand.copy()
                cur_score = sc
                stuck = 0
            else:
                stuck += 1
            if sc < best_score - 1e-12:
                best = cand.copy()
                best_score = sc
                stuck = 0
                print(f"[best] score={best_score:.6f} t_left="
                      f"{deadline - time.monotonic():.1f}s", flush=True)

    # ---- main loop -----------------------------------------------------
    while time.monotonic() < deadline:
        if stuck >= 60:
            # restart around the global best with a hot start, or fresh
            if rng.random() < 0.5:
                base = best.copy()
                warm = base * np.exp(rng.normal(0.0, 0.4, n))
                cur = _renorm(np.clip(warm, 0.0, None), n)
            else:
                cur = _renorm(
                    _init_profiles(n)[int(rng.integers(len(_init_profiles(n))))], n)
            cur_score = _obj(cur, n)
            stuck = 0
            continue

        now = time.monotonic()
        # Periodic LP fill step (gated so it cannot monopolise the budget).
        if now - last_lp > 0.5:
            M = float(np.max(np.convolve(cur, cur)))
            if math.isfinite(M) and M > 0:
                g = _lp_fill_direction(cur, n, M)
                if g is not None:
                    # full flip
                    consider(_renorm(g, n))
                    # blends
                    if time.monotonic() < deadline:
                        for t in (0.5, 0.25, 0.75):
                            if time.monotonic() >= deadline:
                                break
                            consider(_renorm((1.0 - t) * cur + t * g, n))
                last_lp = time.monotonic()

        if time.monotonic() >= deadline:
            break

        # fast local moves
        move = int(rng.integers(4))
        if move == 0:
            consider(_renorm(_subblock(cur, rng, n), n))
        elif move == 1:
            consider(_renorm(_coord_move(cur, rng, n), n))
        elif move == 2:
            consider(_renorm(_anti_peak(cur, rng, n), n))
        else:
            consider(_renorm(_subblock(cur, rng, n), n))

    out = np.clip(best, 0.0, 1000.0)
    out = [_obj(out, n) if False else float(x) for x in out]
    final_sc = _obj(out, n)
    print(f"[final] n={n} returned_score={final_sc:.6f}", flush=True)
    return [float(x) for x in out]
