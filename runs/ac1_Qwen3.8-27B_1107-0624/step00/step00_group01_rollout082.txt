def next_pow2(m):
    if m <= 1: return 1
    p=1
    while p < m: p <<= 1
    return p

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a / s

def convolution_fft(a, fft_len=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([float(a[0]*a[0])])
    if fft_len is None:
        fft_len = next_pow2(2*n - 1)
    else:
        fft_len = max(fft_len, next_pow2(2*n - 1))
    fa = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(fa * fa, fft_len)
    c = c[:2*n-1]
    c = np.asarray(c, dtype=np.float64)
    # remove negative numerical
    if c.size:
       c[c < 0] = 0.0
    return c

    a=np.asarray(a,float); h=np.asarray(h,float)

import numpy as np
import time

try:
    from scipy.optimize import linprog
    HAS_LINPROG = True
except Exception:
    HAS_LINPROG = False

try:
    np.set_num_threads(1)
except Exception:
    pass


def next_pow2(m):
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.clip(a, 0.0, None)
    if not np.all(np.isfinite(a)):
        return None
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a / s


def convolution_fft(a, fft_len=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)

    need = 2 * n - 1
    if fft_len is None:
        fft_len = next_pow2(need)
    else:
        fft_len = max(next_pow2(fft_len), next_pow2(need))

    fa = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(fa * fa, fft_len)
    c = c[:need]
    c = np.asarray(c, dtype=np.float64)
    if c.size:
        c[c < 0.0] = 0.0
    return c


def cross_corr_fft(a, h, fft_len):
    a = np.asarray(a, dtype=np.float64)
    h = np.asarray(h, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * h[0])], dtype=np.float64)

    if fft_len is None:
        fft_len = next_pow2(3 * n - 2)
    else:
        fft_len = max(next_pow2(fft_len), next_pow2(3 * n - 2))

    hrev = h[::-1]
    fa = np.fft.rfft(a, fft_len)
    fh = np.fft.rfft(hrev, fft_len)
    conv = np.fft.irfft(fa * fh, fft_len)

    # Want s_i = sum_j a[j] * h[i+j] for i=0..n-1.
    s = conv[2 * n - 2 : n - 2 : -1]
    s = np.asarray(s, dtype=np.float64)
    if s.size:
        s[s < 0.0] = 0.0
    return s


def make_central(n):
    if n <= 0:
        return np.array([], dtype=np.float64)

    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * i + 2.0)

    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def make_update(a, r, eta):
    logf = -float(eta) * np.asarray(r, dtype=np.float64)
    logf = np.clip(logf, -50.0, 0.0)
    out = np.clip(a * np.exp(logf), 0.0, None)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 0.0:
        return a.copy()
    return out / s


def exact_score(seq):
    try:
        return float(evaluate_sequence(seq))
    except Exception:
        return float("inf")


def generate_starts(n, rng, count=4):
    base = make_central(n)
    arrs = [base.copy()]

    idx = np.arange(n, dtype=np.float64)

    # Slight exponent perturbations of the central-binomial profile.
    for q in (-0.35, -0.20, -0.10, 0.10, 0.25):
        a = base * np.power(idx + 1.0, q)
        nz = normalize_nonneg(a)
        if nz is not None:
            arrs.append(nz)

    # Linear endpoint weighting.
    if n > 1:
        for gamma in (-0.5, 0.5, 1.5):
            w = 1.0 + gamma * (idx / float(n - 1))
            a = base * w
            nz = normalize_nonneg(a)
            if nz is not None:
                arrs.append(nz)

    # Random multiplicative noise around the central profile.
    for sigma in (0.4, 1.0, 2.0):
        a = base * np.exp(rng.normal(0.0, sigma, n))
        nz = normalize_nonneg(a)
        if nz is not None:
            arrs.append(nz)

    # A generic exponential-random start.
    nz = normalize_nonneg(rng.exponential(1.0, n))
    if nz is not None:
        arrs.append(nz)

    scored = []
    for i, a in enumerate(arrs):
        if a is None:
            continue
        c = convolution_fft(a)
        m = float(c.max())
        sc = 2.0 * n * m
        if np.isfinite(sc):
            scored.append((float(sc), i))

    scored.sort()
    out = []
    for _sc, i in scored[:max(1, int(count))]:
        out.append(arrs[i])

    if not out:
        out.append(base.copy())
    return out


def lp_direction(a, c, M, K, max_seconds=10.0):
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = a.size
    L = 2 * n - 1

    if not HAS_LINPROG or n <= 0 or K <= 0 or M <= 0.0 or not np.isfinite(M):
        return None
    if c.size != L:
        c = convolution_fft(a)

    if K >= L:
        ks = np.arange(L, dtype=np.int64)
    else:
        thresh = 0.98 * M
        active = np.where(c >= thresh)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c))], dtype=np.int64)

        if active.size > K:
            pos = np.linspace(0, active.size - 1, max(1, K)).astype(np.int64)
            ks = active[pos]
        else:
            ks = active

        amax = int(np.argmax(c))
        if amax not in ks:
            ks = np.concatenate((ks, np.array([amax], dtype=np.int64)))

    A = np.zeros((ks.size, n), dtype=np.float64)
    for rr, k in enumerate(ks):
        i0 = int(k - (n - 1))
        if i0 < 0:
            i0 = 0
        i1 = int(k)
        if i1 > n - 1:
            i1 = n - 1
        if i1 >= i0:
            idx = np.arange(i0, i1 + 1, dtype=np.int64)
            A[rr, idx] = a[k - idx]

    obj = -np.ones(n, dtype=np.float64)
    b = np.full(ks.size, float(M), dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    options = {}
    if max_seconds is not None and max_seconds > 0.0:
        options["time_limit"] = float(max_seconds)

    try:
        res = linprog(obj, A_ub=A, b_ub=b, bounds=bounds, method="highs", options=options)
    except Exception:
        try:
            res = linprog(obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        except Exception:
            return None

    if res is None or not getattr(res, "success", False):
        return None

    g = getattr(res, "x", None)
    if g is None:
        return None

    g = np.asarray(g, dtype=np.float64)
    if g.size != n or not np.all(np.isfinite(g)):
        return None

    g = np.clip(g, 0.0, None)
    if float(g.sum()) <= 1.0:
        return None
    return g


def lp_blend(a, current_M, g, fft_len, delta):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    n = a.size

    sg = float(g.sum())
    if not np.isfinite(sg) or sg <= 1.0:
        return None

    bestM = None
    bestA = None
    bestC = None

    dvals = (delta, 0.5 * delta, 2.0 * delta)

    for d in dvals:
        d = float(np.clip(d, 0.02, 2.0))
        target = 1.0 + d

        if sg >= target:
            gt = g * (target / sg)
        else:
            gt = g

        gst = float(gt.sum())
        if not np.isfinite(gst) or gst <= 0.0:
            continue

        gn = gt / gst

        for t in (1.0, 0.5, 0.25, 0.1):
            at = (1.0 - t) * a + t * gn
            at = normalize_nonneg(at)
            if at is None:
                continue

            ct = convolution_fft(at, fft_len)
            Mt = float(ct.max())

            if not np.isfinite(Mt) or not (Mt < current_M - 1e-14):
                continue

            if bestM is None or Mt < bestM:
                bestM = Mt
                bestA = at
                bestC = ct

    if bestM is None:
        return None
    return bestM, bestA, bestC

    best_c = c.copy()

    it = 0
    random_sigma = 0.35

    if n <= 512:
        lp_interval = 12
        K = 2 * n - 1
    elif n <= 1024:
        lp_interval = 20
        K = 2 * n - 1
    elif n <= 2500:
        lp_interval = 40
        K = max(384, min(768, n // 4))
    elif n <= 5000:
        lp_interval = 90
        K = max(512, min(1024, n // 4))
    else:
        lp_interval = 10**9
        K = 0

    last_lp = -10**9
    lp_delta = 0.2
    lp_fails = 0

    while time.monotonic() < deadline:
        if not np.isfinite(M) or M <= 0.0:
            a = best_a.copy()
            c = best_c.copy()
            M = best_M

        accepted = False

        if M > 0.0:
            z = c / M
            z = np.clip(z, 0.0, 1.0)

            p_use = p * (0.9 + 0.2 * rng.random())
            if p_use < 1e-6:
                p_use = 1e-6

            h = np.power(z, p_use)
            hsum = float(h.sum())
            if not np.isfinite(hsum) or hsum <= 0.0:
                h = np.zeros_like(z)
                h[int(np.argmax(z))] = 1.0
                hsum = 1.0
            else:
                h = h / hsum

            s = cross_corr_fft(a, h, ftl)
            smax = float(s.max())

            if np.isfinite(smax) and smax > 1e-300:
                r = s / smax
                trial = make_update(a, r, eta)
                ct = convolution_fft(trial, ftl)
                Mt = float(ct.max())

                if np.isfinite(Mt) and Mt < M - 1e-14:
                    a = trial
                    c = ct
                    M = Mt
                    accepted = True
                    stall = 0
                    eta = min(eta * 1.08, 5.0)
                    p = min(p * 1.05, 80.0)
                else:
                    for mult in (0.5, 0.25, 1.5, 2.0):
                        et = eta * mult
                        if et <= 0.0 or et > 10.0:
                            continue
                        trial2 = make_update(a, r, et)
                        ct2 = convolution_fft(trial2, ftl)
                        Mt2 = float(ct2.max())
                        if np.isfinite(Mt2) and Mt2 < M - 1e-14:
                            a = trial2
                            c = ct2
                            M = Mt2
                            accepted = True
                            stall = 0
                            eta = min(et * 1.08, 5.0)
                            p = min(p * 1.02, 80.0)
                            break

        if accepted:
            if M < best_M:
                best_M = M
                best_a = a.copy()
                best_c = c.copy()
        else:
            stall += 1
            if stall <= 5:
                eta = max(eta * 0.7, 0.05)
            elif stall <= 12:
                p = max(p * 0.75, 0.2)
            else:
                eta = 0.7
                p = 1.0
                stall = 0

                a = best_a * np.exp(rng.normal(0.0, random_sigma, n))
                a = normalize_nonneg(a)
                if a is None:
                    a = best_a.copy()
                c = convolution_fft(a, ftl)
                M = float(c.max())

                random_sigma = min(random_sigma * 1.25, 2.0)

        use_lp_now = (
            use_lp
            and HAS_LINPROG
            and K > 0
            and (it - last_lp) >= lp_interval
            and time.monotonic() < deadline - 5.0
            and M <= best_M * 1.5
            and lp_fails < 6
        )

        if use_lp_now:
            left = float(deadline - time.monotonic())
            max_sec = max(2.0, min(25.0, left / 8.0))
            g = lp_direction(a, c, M, K, max_sec)

            if g is not None:
                lpres = lp_blend(a, M, g, ftl, lp_delta)
                if lpres is not None:
                    M2, a2, c2 = lpres
                    if M2 < M:
                        a = a2
                        c = c2
                        M = M2

                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                            best_c = c.copy()

                        lp_delta = min(lp_delta * 1.4, 1.5)
                        lp_fails = 0
                    else:
                        lp_delta = max(lp_delta * 0.7, 0.02)
                        lp_fails += 1
                else:
                    lp_delta = max(lp_delta * 0.7, 0.02)
                    lp_fails += 1
            else:
                lp_fails += 1

            last_lp = it

        it += 1
        if it > 200000:
            break

    if best_a is None or best_a.size == 0 or not np.all(np.isfinite(best_a)):
        return None, float("inf")

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()

    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1800.0

    if not np.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 1800.0

    deadline = start + max(10.0, budget_s - 20.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # Consider the supplied earlier construction as a valid fallback/seed.
    init = globals().get("height_sequence_1", None)
    if init is not None:
        try:
            arr = np.asarray(init, dtype=np.float64)
            if arr.ndim != 1:
                arr = arr.reshape(-1)
            if arr.size > 0 and np.all(np.isfinite(arr)):
                arr = np.clip(arr, 0.0, 1000.0)
                sc = exact_score(arr.tolist())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = [float(x) for x in arr.tolist()]
                    print(f"initial height_sequence_1 exact_score={best_score:.9f}")
        except Exception:
            pass

    total_window = max(30.0, deadline - start)

    # Allocate time to a few sequence lengths.  Central-binomial profiles are
    # strong starts because their finite convolution is exactly flat on the
    # first n lags; local search then uses tail slack to improve the score.
    plan = (
        (512, 0.14),
        (1024, 0.24),
        (2048, 0.30),
        (4096, 0.32),
    )

    for n, frac in plan:
        if time.monotonic() >= deadline:
            break

        sub_seconds = total_window * float(frac)
        sub_deadline = min(deadline, time.monotonic() + sub_seconds)
        if sub_deadline - time.monotonic() < 15.0:
            continue

        starts = generate_starts(n, rng, 5)
        a = starts[0]

        c0 = convolution_fft(a)
        fast0 = 2.0 * float(n) * float(c0.max())
        print(f"n={n} initial_fast_score={fast0:.9f}")

        best_a, fast_M = local_search(a, sub_deadline, rng, use_lp=True)

        if best_a is not None:
            seq = [float(x) for x in best_a.tolist()]
            sc = exact_score(seq)
            free = 2.0 * float(n) * float(fast_M) if np.isfinite(fast_M) else float("inf")
            print(f"n={n} local_fast={free:.9f} exact={sc:.9f} elapsed={time.monotonic() - start:.1f}s")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq

    # Use any remaining time to polish the current best sequence directly.
    if time.monotonic() < deadline - 20.0:
        if best_seq is None:
            a = make_central(2048)
        else:
            a = normalize_nonneg(np.asarray(best_seq, dtype=np.float64))
            if a is None:
                a = make_central(2048)

        best_a, fast_M = local_search(a, deadline, rng, use_lp=True)

        if best_a is not None:
            seq = [float(x) for x in best_a.tolist()]
            sc = exact_score(seq)
            n = seq.__len__()
            free = 2.0 * float(n) * float(fast_M) if np.isfinite(fast_M) else float("inf")
            print(f"final_polish n={n} fast={free:.9f} exact={sc:.9f} elapsed={time.monotonic() - start:.1f}s")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq

    if best_seq is None or len(best_seq) == 0:
        a = make_central(256)
        best_seq = [float(x) for x in a.tolist()]
        best_score = exact_score(best_seq)

    # Final safety check: guarantee a finite, valid list.
    try:
        out = [float(x) for x in best_seq]
        if len(out) == 0 or not all(np.isfinite(x) and x >= 0.0 for x in out):
            a = make_central(256)
            out = [float(x) for x in a.tolist()]
    except Exception:
        a = make_central(256)
        out = [float(x) for x in a.tolist()]

    return out
