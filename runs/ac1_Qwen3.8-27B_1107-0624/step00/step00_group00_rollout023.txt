import cvxpy as cp

def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    g = cp.Variable(n)
    
    # Build the convolution constraint matrix
    # A[k, j] = f[j] if k-j is in [0, n-1], else 0
    # i.e., A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Wait, let me re-derive.

    # This is valid when 0 <= k-j' < n, i.e., k-n < j' <= k.
    
    # Let me just build it directly.
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC)  # or cp.GLPK, cp.PDLP, etc.
    
    return g.value

    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
