import numpy as np
import time
from scipy.optimize import linprog

def fft_full_convolve(x, y=None):
    """Full linear convolution of x with y (or x with itself if y is None) using FFT."""
    if y is None:
        y = x
    n = len(x)
    m = len(y)
    size = n + m - 1
    # Next power of 2
    fsize = 1
    while fsize < size:
        fsize *= 2
    X = np.fft.rfft(x, fsize)
    Y = np.fft.rfft(y, fsize)
    result = np.fft.irfft(X * Y, fsize)[:size]
    return result

def subgradient_step(a, eta, active_frac=0.1):
    """One subgradient descent step to minimize max of full self-convolution."""
    n = len(a)
    # Compute full self-convolution
    c = fft_full_convolve(a)
    max_c = np.max(c)
    
    # Find active set
    threshold = max_c * (1.0 - active_frac)
    S = np.where(c >= threshold)[0]
    
    # Compute subgradient: g[j] = sum_{k in S} a[k-j] for valid k-j
    # g[j] = sum_m a[m] * 1[m+j in S]
    # Using FFT: g = (a_rev * b)[n-1 : 2n-1] where b = indicator of S
    b = np.zeros(2*n - 1)
    b[S] = 1.0
    a_rev = a[::-1]
    conv = fft_full_convolve(a_rev, b)
    g = conv[n-1:2*n-1]  # This gives g[0], ..., g[n-1]
    
    # Multiplicative update
    max_g = np.max(g)
    if max_g > 0:
        a_new = a * (1.0 - eta * g / max_g)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = a.copy()
    else:
        a_new = a.copy()
    
    return a_new, max_c

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_full_convolve(x, y=None):
    """Full linear convolution of x with y (or x with itself) using FFT."""
    if y is None:
        y = x
    nx = len(x)
    ny = len(y)
    size = nx + ny - 1
    fsize = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fsize)
    Y = np.fft.rfft(y, fsize)
    return np.fft.irfft(X * Y, fsize)[:size]


def compute_subgradient(a, c, active_frac=0.05):
    """Compute subgradient of max(c) w.r.t. a. Returns (g, max_c)."""
    n = len(a)
    max_c = np.max(c)
    threshold = max_c * (1.0 - active_frac)
    S = np.where(c >= threshold)[0]
    if len(S) == 0:
        S = np.array([np.argmax(c)])
    b = np.zeros(2 * n - 1)
    b[S] = 1.0
    a_rev = a[::-1]
    conv = fft_full_convolve(a_rev, b)
    g = conv[n - 1:2 * n - 1].copy()
    return g, max_c


def subgradient_update(a, g, eta):
    """Multiplicative weight update: a * (1 - eta*g/max(g)), projected & normalized."""
    max_g = np.max(g)
    if max_g < 1e-15:
        return a.copy()
    a_new = a * (1.0 - eta * g / max_g)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def lp_polish_reduced(a, active_indices, M):
    """LP polish with reduced active constraints only."""
    n = len(a)
    a_safe = np.maximum(a, 1e-10)
    k_list = active_indices
    n_constr = len(k_list)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_list):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            cols.extend(range(i_start, i_end + 1))
            vals.extend(a_safe[k - i_start:k - i_start + (i_end - i_start + 1)].tolist())
            rows.extend([idx] * (i_end - i_start + 1))
    if len(vals) == 0:
        return a.copy()
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_constr, n))
    b_ub = M * np.ones(n_constr)
    c_obj = -np.ones(n)
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0, 1.0)] * n, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            b = result.x
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return a.copy()

    """Run subgradient descent for a given time budget. Returns best (a, score)."""
    best_a = a.copy()

    eta = eta0
