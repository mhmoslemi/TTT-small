import numpy as np
import time

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 1e-15, 1000.0)
    s = a.sum()
    if not np.isfinite(s) or s <=0:
        a[:] = 1.0
        s = a.size
    return a/s

def _fftconvolve(a,b):
    a=np.asarray(a,float); b=np.asarray(b,float)
    L=len(a)+len(b)-1
    N=1
    while N < L: N <<=1
    fa=np.fft.rfft(a,N)
    fb=np.fft.rfft(b,N)
    c=np.fft.irfft(fa*fb,N)
    return c[:L]

def _score_fft(a):
    s = np.sum(a)
    if s < 1e-12: return np.inf
    p = a/s
    C = _fftconvolve(p,p)
    m = np.max(C)
    if not np.isfinite(m) or m < 0: m=0.0
    return float(2.0 * len(p) * m)

def _sliding_dot(p,w):

def _step_log(p, A, eta, floor=1e-15):
    gmax=float(np.max(A))
    if not np.isfinite(gmax) or gmax <= 1e-300: return p
    logp=np.log(np.maximum(p, floor))

import numpy as np
import time


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 1e-15, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        a[:] = 1.0 / max(1, a.size)
        s = float(a.size)
    return a / s


def _fft_self(a):
    n = int(a.size)
    n = max(1, n)
    L = 2 * n - 1
    N = 1 << max(1, (L - 1).bit_length())
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:L]


def _score_norm(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    p = a / s
    c = _fft_self(p)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    if m < 0.0:
        m = 0.0
    return float(2.0 * p.size * m)


def _sliding_dot(p, w):
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(p.size)
    L = n + w.size - 1
    N = 1 << max(1, (L - 1).bit_length())
    fa = np.fft.rfft(p[::-1], N)
    fb = np.fft.rfft(w, N)
    c = np.fft.irfft(fa * fb, N)
    return c[n - 1:n + n - 1]


def _make_weights(p, target):
    c = _fft_self(p)
    m = float(np.max(c))
    if not np.isfinite(m) or m <= 1e-300:
        w = np.zeros_like(c)
        w[0] = 1.0
        return c, 1.0, w
    beta = max(1e-12, float(target) / m)
    val = beta * (c - m)
    val = np.clip(val, -700.0, 0.0)
    w = np.exp(val)
    z = float(w.sum())
    if not np.isfinite(z) or z <= 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        z = 1.0
    w /= z
    return c, m, w


def _step_log(p, A, eta, floor=1e-15):
    gmax = float(np.max(A))
    if not np.isfinite(gmax) or gmax <= 1e-300:
        return np.asarray(p, dtype=np.float64).copy()
    logp = np.log(np.maximum(p, floor))
    logp -= (eta / gmax) * A
    mx = float(np.max(logp))
    logp -= mx
    q = np.exp(logp)
    q = np.maximum(q, floor)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.asarray(p, dtype=np.float64).copy()
    return q / s


def _perturb(p, rng, alpha=0.25):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    q = p.copy()
    if n <= 1:
        return q
    lo = max(2, n // 1000)
    hi = max(lo + 1, n // 10)
    if hi >= n:
        hi = n
    width = int(rng.integers(lo, hi + 1))
    if width <= 0:
        width = 1
    if width > n:
        width = n
    start = int(rng.integers(0, n - width + 1))
    q[start:start + width] *= np.exp(rng.normal(0.0, alpha, width))
    if rng.random() < 0.5:
        lo2 = max(1, n // 1000)
        hi2 = max(lo2 + 1, n // 100)
        if hi2 >= n:
            hi2 = n
        width2 = int(rng.integers(lo2, hi2 + 1))
        if width2 <= 0:
            width2 = 1
        if width2 > n:
            width2 = n
        start2 = int(rng.integers(0, n - width2 + 1))
        q[start2:start2 + width2] *= np.exp(rng.normal(0.0, alpha * 0.5, width2))
    q = np.maximum(q, 1e-15)
    return _normalize(q)


def _contrast(p, lam, margin=1e-6):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    f = p * float(n)
    fs = float(f.sum())
    if not np.isfinite(fs) or fs <= 0.0:
        return _normalize(np.ones(n, dtype=np.float64))
    f = f / fs * float(n)
    v = f - 1.0
    minv = float(np.min(v))
    if minv < 0.0:
        max_scale = (1.0 - margin) / (-minv)
        scale = min(float(lam), max_scale)
    else:
        scale = float(lam)
    if not np.isfinite(scale) or scale < 0.0:
        scale = 0.0
    fnew = 1.0 + scale * v
    fnew = np.maximum(fnew, margin)
    return _normalize(fnew)


def _try_transforms(p, rng, n_shifts=3, use_contrast=True):
    p = np.asarray(p, dtype=np.float64)
    best = p.copy()
    best_s = _score_norm(best)
    n = int(best.size)
    if n > 1:
        shifts = [0]
        for _ in range(max(0, int(n_shifts))):
            shifts.append(int(rng.integers(1, n)))
        seen = set()
        for sh in shifts:
            if sh in seen:
                continue
            seen.add(sh)
            if sh == 0:
                continue
            q = np.roll(best, sh)
            s = _score_norm(q)
            if np.isfinite(s) and s < best_s:
                best = q
                best_s = s
    if use_contrast:
        for lam in (0.65, 0.9, 1.25, 1.7, 2.3):
            q = _contrast(best, lam)
            s = _score_norm(q)
            if np.isfinite(s) and s < best_s:
                best = q
                best_s = s
    return best, float(best_s)


def _seq_from_v(v, amp=0.9):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return _normalize(np.ones(8, dtype=np.float64))
    vmax = float(np.max(np.abs(v)))
    if not np.isfinite(vmax) or vmax <= 1e-300:
        return _normalize(np.ones(v.size, dtype=np.float64))
    v = v / vmax
    f = 1.0 + float(amp) * v
    minf = float(np.min(f))
    if minf < 1e-8:
        minv = float(np.min(v))
        if minv < 0.0:
            safe = (1.0 - 1e-8) / (-minv)
            amp = min(float(amp), safe)
        f = 1.0 + float(amp) * v
    f = np.maximum(f, 1e-12)
    return _normalize(f)


def _run_smooth(p0, rng, time_limit, target0, target1, tag=""):
    if time_limit <= 0.0:
        p = _normalize(p0)
        return p, _score_norm(p)
    start = time.monotonic()
    deadline = start + float(time_limit)
    p, loc_s = _try_transforms(_normalize(p0), rng, n_shifts=3, use_contrast=True)
    best = p.copy()
    best_s = float(loc_s)
    no_best = 0
    it = 0
    while time.monotonic() < deadline:
        frac = (time.monotonic() - start) / max(1e-9, float(time_limit))
        if frac < 0.0:
            frac = 0.0
        if frac > 1.0:
            frac = 1.0
        target = float(target0) + (float(target1) - float(target0)) * frac
        if rng.random() < 0.02:
            target *= float(rng.uniform(0.3, 3.0))
        if target < 0.05:
            target = 0.05
        _, _, w = _make_weights(p, target)
        A = _sliding_dot(p, w)
        A = np.maximum(A, 0.0)
        improved = False
        if A.size:
            gmax = float(np.max(A))
            if np.isfinite(gmax) and gmax > 1e-300:
                best_q = None
                best_local = float(loc_s)
                for eta in (0.05, 0.1, 0.2, 0.35, 0.6):
                    q = _step_log(p, A, eta)
                    if q is None:
                        continue
                    s = _score_norm(q)
                    if np.isfinite(s) and s < best_local:
                        best_local = float(s)
                        best_q = q
                if best_q is not None:
                    p = best_q
                    loc_s = best_local
                    improved = True
                    if loc_s < best_s or it % 50 == 0:
                        p2, t2 = _try_transforms(
                            p,
                            rng,
                            n_shifts=2,
                            use_contrast=(loc_s < best_s or it % 120 == 0)
                        )
                        if t2 < loc_s:
                            p = p2
                            loc_s = t2
                        if loc_s < best_s:
                            best = p.copy()
                            best_s = float(loc_s)
                            no_best = 0
                        else:
                            no_best += 1
                    else:
                        no_best += 1
        if not improved:
            p = _perturb(p, rng, 0.35)
            loc_s = _score_norm(p)
            if loc_s < best_s:
                p, loc_s = _try_transforms(p, rng, n_shifts=2, use_contrast=True)
                best = p.copy()
                best_s = float(loc_s)
                no_best = 0
            else:
                no_best += 1
                if no_best > 100 or (loc_s > best_s * 1.25 and no_best > 25):
                    p = best.copy()
                    loc_s = best_s
                    no_best = 0
        it += 1
        if it % 800 == 0:
            print(tag, "iter", it, "local", float(loc_s), "best", float(best_s), flush=True)
    p2, t2 = _try_transforms(best, rng, n_shifts=4, use_contrast=True)
    if t2 < best_s:
        best = p2
        best_s = t2
    return best, float(best_s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Full smooth peak-reduction search with multiple starts, contrast moves, and exact final verification."""
    t0 = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    rng = np.random.default_rng(seed)

    base = None
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base.ravel(), 0.0, 1000.0)
        if base.size < 256 or base.size > 10000:
            base = None
        elif not np.isfinite(base).all() or float(np.sum(base)) < 0.01:
            base = None
    except Exception:
        base = None

    if base is not None:
        n = int(base.size)
        base_seq = _normalize(base)
    else:
        n = 5582
        base_seq = None

    starts = []
    if base_seq is not None:
        starts.append(base_seq)
        starts.append(_perturb(base_seq, rng, 0.2))
        starts.append(_contrast(base_seq, 1.5))

    x = (np.arange(n) + 0.5) / float(n)
    u = x - 0.5

    for amp in (0.5, 0.65, 0.75, 0.85, 0.95):
        starts.append(_seq_from_v(u, amp))

    ab_pairs = (
        (1.0, 0.0),
        (1.0, 0.5),
        (1.0, -0.5),
        (0.5, 1.0),
        (0.5, -1.0),
        (0.0, 1.0),
        (0.0, -1.0),
    )
    for a, b in ab_pairs:
        starts.append(_seq_from_v(a * u + b * (u * u * u), 0.95))

    for alpha in (0.6, 0.9, 1.2):
        for sval in (-0.8, -0.4, 0.0, 0.4, 0.8):
            for m in (1, 2, 3, 5):
                v = alpha * u + sval * np.sin(2.0 * np.pi * float(m) * x)
                starts.append(_seq_from_v(v, 0.95))

    for _ in range(18):
        M = int(rng.integers(3, 9))
        coeffs = rng.normal(0.0, 0.7, M)
        v = np.zeros(n, dtype=np.float64)
        for m in range(1, M + 1):
            v += float(coeffs[m - 1]) * np.sin(2.0 * np.pi * float(m) * x)
        amp = float(rng.uniform(0.6, 0.95))
        starts.append(_seq_from_v(v, amp))

    for m in (1, 2, 3, 4):
        starts.append(_seq_from_v(np.sin(2.0 * np.pi * float(m) * x), 0.95))

    scored = []
    idx = 0
    for arr in starts:
        try:
            a = _normalize(arr)
            sc = _score_norm(a)
            if np.isfinite(sc):
                scored.append((float(sc), idx, a))
                idx += 1
        except Exception:
            pass

    scored.sort()
    top = scored[:8]
    pool = []
    pool_idx = 0

    global_best = None
    global_score = np.inf
    for sc, _, seq in top:
        if global_best is None or sc < global_score:
            global_best = seq.copy()
            global_score = float(sc)
        pool.append((float(sc), pool_idx, seq.copy()))
        pool_idx += 1

    if global_best is None:
        global_best = _normalize(np.ones(n, dtype=np.float64))
        global_score = _score_norm(global_best)
        pool.append((float(global_score), pool_idx, global_best.copy()))
        pool_idx += 1

    print("initial best", float(global_score), "n", n, flush=True)

    end_time = t0 + total_budget
    final_len = min(360.0, max(90.0, 0.22 * total_budget))
    main_deadline = end_time - final_len
    if main_deadline < t0 + 90.0:
        main_deadline = t0 + 90.0

    for i, (sc, _, seq) in enumerate(top):
        now = time.monotonic()
        if now >= main_deadline - 30.0:
            break
        remaining = main_deadline - now
        if i == 0:
            limit = min(480.0, remaining)
            target1 = 80.0
        else:
            limit = min(260.0, remaining)
            target1 = 55.0
        if limit < 25.0:
            break
        p_best, p_score = _run_smooth(seq, rng, limit, 1.0, target1, tag="run%d" % i)
        if p_score < global_score:
            global_best = p_best.copy()
            global_score = p_score
        pool.append((float(p_score), pool_idx, p_best.copy()))
        pool_idx += 1
        print("run", i, "score", float(p_score), "global", float(global_score), flush=True)

    final_deadline = end_time - 8.0
    if global_best is not None and time.monotonic() < final_deadline - 60.0:
        rem = final_deadline - time.monotonic() - 4.0
        if rem > 60.0:
            p_best, p_score = _run_smooth(global_best, rng, rem, 4.0, 250.0, tag="final")
            if p_score < global_score:
                global_best = p_best.copy()
                global_score = p_score
            pool.append((float(p_score), pool_idx, p_best.copy()))
            pool_idx += 1
        pt, st = _try_transforms(global_best, rng, n_shifts=5, use_contrast=True)
        if st < global_score:
            global_best = pt.copy()
            global_score = st
        pool.append((float(global_score), pool_idx, global_best.copy()))
        pool_idx += 1

    pool.sort()
    best_exact_seq = None
    best_exact_score = np.inf
    tried = 0
    for sc, idx2, seq in pool:
        tried += 1
        if tried > 14:
            break
        try:
            vals = [float(z) for z in seq]
            es = float(evaluate_sequence(vals))
        except Exception:
            continue
        print("exact candidate", idx2, "fft", float(sc), "exact", float(es), flush=True)
        if np.isfinite(es) and es < best_exact_score:
            best_exact_score = es
            best_exact_seq = vals

    if best_exact_seq is None:
        if global_best is not None:
            best_exact_seq = [float(z) for z in global_best]
        else:
            best_exact_seq = [1.0]
    return best_exact_seq
