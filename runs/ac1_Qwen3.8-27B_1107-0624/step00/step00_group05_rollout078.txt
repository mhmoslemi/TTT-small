```python
"""
Optimization of step function heights to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses Frank-Wolfe (LP-based) method with subgradient acceleration.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def fast_full_conv(a, b=None):
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def obj_value(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * fast_full_conv(a, a).max() / (S * S)

def build_conv_matrix(a):
    n = len(a)
    row_idx = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    col_idx = np.repeat(np.arange(n), n)
    vals = np.repeat(a, n)

def frank_wolfe_step(a, M):

    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None
```

```python
import numpy as np
import time

def fft_convolve(a, b=None):
    """Full convolution using FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * fft_convolve(a, a).max() / (S * S)

def subgrad_at(a, k):
    """Compute subgradient of c_k at a. Returns grad (not projected)."""
    n = len(a)
    p = np.arange(n)
    j = k - p
    mask = (j >= 0) & (j < n)
    grad = np.zeros(n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad

def optimize_sequence(a0, deadline, rng, max_iters=1000000):
    """Subgradient descent with noise and momentum."""
    n = len(a0)
    a = a0 / a0.sum()
    best = a.copy()
    best_val = 2.0 * n * fft_convolve(a, a).max()
    
    # Momentum
    v = np.zeros(n)
    beta = 0.9
    
    # Step size and noise
    t = 1.0 / np.sqrt(n)
    noise = 0.05 / np.sqrt(n)
    
    it = 0
    time_per_iter_budget = 0.005  # 5ms per iteration target
    
    while it < max_iters:
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        c = fft_convolve(a, a)
        k_star = int(np.argmax(c))
        cur_val = 2.0 * n * c[k_star]
        
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best = a.copy()
        
        # Subgradient
        grad = subgrad_at(a, k_star)
        grad -= grad.mean()  # Project to sum=0
        
        # Add noise
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad = grad / gnorm  # Normalize to unit norm
            step = t * (1.0 + noise * rng.standard_normal(n))
        else:
            step = t * noise * rng.standard_normal(n)
            step -= step.mean()
        
        # Momentum
        v = beta * v + (1.0 - beta) * step
        a = a - v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best.copy()
        else:
            a = a / s
        
        # Decay
        t *= 0.99999
        noise *= 0.9995
        
        it += 1
    
    return best, best_val
```
