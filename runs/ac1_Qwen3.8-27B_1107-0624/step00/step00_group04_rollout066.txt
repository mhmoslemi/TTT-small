def _cross_corr_w_a(w, a, N):
    """Compute grad_i = sum_j w[i+j]*a[j] via FFT cross-correlation. w has len 2n-1, a has len n."""
    W = np.fft.fft(w, n=N)
    A = np.fft.fft(a, n=N)
    xcorr = np.fft.ifft(W * np.conj(A))
    return xcorr[:len(a)].real


def _sa_phase(a_init, n, budget, rng, label=""):
    """Run one phase of soft-subgradient simulated annealing."""
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a /= S

    best_a = a.copy()
    best_score = 2.0 * n * _fft_convolve(best_a).max()

    alpha = 1e6
    T = 0.008
    step = 0.012
    it = 0
    no_improve = 0
    deadline = time.monotonic() + budget
    t0 = time.monotonic()

    conv_len = 2 * n - 1
    N_xcorr = 1 << (conv_len + n - 1).bit_length()

    while time.monotonic() < deadline:
        it += 1
        b = _fft_convolve(a)
        b_max = b.max()
        score = 2.0 * n * b_max

        shifted = alpha * (b - b_max)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()

        grad = _cross_corr_w_a(w, a, N_xcorr)
        grad -= grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            direction = rng.standard_normal(n)
            direction -= direction.mean()
            dn = np.linalg.norm(direction)
            if dn < 1e-15:
                continue
            direction /= dn
        else:
            direction = -grad / gnorm

        a_new = a + step * direction
        a_new += rng.normal(0.0, 8e-6, n)
        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-15:
            continue
        a_new /= S_new

        b_new = _fft_convolve(a_new)
        new_score = 2.0 * n * b_new.max()

        delta = new_score - score
        accept = False
        if delta < 0:
            accept = True
        elif T > 1e-12:
            ratio = -delta / T
            if ratio < 700.0:
                accept = rng.random() < np.exp(ratio)

        if accept:
            a = a_new
            if new_score < best_score:
                best_a = a_new.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1
            step = min(0.06, step * 1.008)
        else:
            no_improve += 1
            step = max(0.002, step * 0.96)

        T = max(1e-12, T * 0.9997)

        if no_improve > 4000:
            frac = rng.uniform(0.15, 0.55)
            n_pert = max(30, int(frac * n))
            idx = rng.choice(n, size=n_pert, replace=False)
            a[idx] *= np.exp(rng.normal(0.0, 0.4, n_pert))
            a = np.clip(a, 0.0, None)
            if a.sum() > 1e-15:
                a /= a.sum()
            no_improve = 0
            T = max(T, 0.004)

        if it % 10000 == 0:
            el = time.monotonic() - t0
            print(f"  [{label}] it={it}, el={el:.0f}s, best={best_score:.6f}, curr={score:.6f}, T={T:.1e}, step={step:.4f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences
    using FFT-accelerated soft-subgradient simulated annealing with
    multiple initializations and adaptive step size.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + budget_s - 4.0

    try:
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy().flatten()
    except Exception:
        a0 = np.ones(3809, dtype=np.float64)

    n = len(a0)
    a0 = np.clip(a0, 0.0, 1000.0)
    if a0.sum() < 1e-10:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / a0.sum()

    inits = [a0.copy()]

    r = np.abs(rng.normal(0.5, 1.0, n)) + 0.001
    inits.append(r / r.sum())

    x = np.arange(n, dtype=np.float64)
    c = 1.0 + 0.7 * np.cos(2.0 * np.pi * x / n)
    c = np.clip(c, 0.01, None)
    inits.append(c / c.sum())

    t = np.ones(n) * 0.3
    t[n // 4: 3 * n // 4] = 1.0
    t += np.abs(rng.normal(0.0, 0.1, n))
    inits.append(t / t.sum())

    lin = np.linspace(0.3, 2.0, n)
    inits.append(lin / lin.sum())

    n_phases = len(inits)
    time_per = max(10.0, (budget_s - 20.0) / n_phases)

    global_best_a = inits[0].copy()
    global_best_score = 2.0 * n * _fft_convolve(global_best_a).max()
    print(f"n={n}, initial best={global_best_score:.6f}")

    for i, init in enumerate(inits):
        rem = total_deadline - time.monotonic()
        if rem < 12.0:
            break
        alloc = min(time_per, rem - 5.0)
        if alloc < 8.0:
            alloc = max(5.0, rem - 3.0)

        print(f"Phase {i + 1}/{n_phases}: alloc={alloc:.0f}s")
        best_a, best_score = _sa_phase(init, n, alloc, rng, label=f"p{i}")

        if best_score < global_best_score:
            global_best_score = best_score
            global_best_a = best_a
            print(f"  -> global best updated: {global_best_score:.6f}")

    result = [float(v) for v in global_best_a]
    fs = evaluate_sequence(result)
    print(f"Final evaluate_sequence: {fs:.6f}")
    return result
