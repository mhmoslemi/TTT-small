import numpy as np
import time

"""
Top-k/softmax sharpening optimizer for nonnegative step functions.

The objective is homogeneous, so all internal candidates are normalized to
sum 1.  The score is then 2*n*max(f*f).  The search uses:
  - low-dimensional cosine ancestor searches for strong asymmetric starts,
  - random smooth nonnegative starts,
  - mirror-descent / Adam updates on log-f,
  - subgradients of the average top-k entries of f*f, with k annealed down.
"""


def next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s


def normalize_dist(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(f.size, 1.0 / max(1, f.size), dtype=np.float64)
    return f / s


def convolve_with_size(f, size):
    n = f.size
    F = np.fft.rfft(f, size)
    q = np.fft.irfft(F * F, size)
    return q[:2 * n - 1]


def fast_score_size(f, size):
    q = convolve_with_size(f, size)
    return float(2.0 * f.size * np.max(q))


def fast_score(f):
    n = f.size
    size = next_pow2(2 * n - 1)
    return fast_score_size(f, size)


def exact_score_list(f):
    return float(evaluate_sequence([float(x) for x in f]))


def exp_normalize(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.size, 1.0 / z.size, dtype=np.float64)
    return e / s


def shape_affine_t(t, a, b=0.0):
    f = 1.0 + float(a) * t + float(b) * t * t
    f = np.maximum(f, 1e-4)
    return normalize_dist(f)


def shape_cos_t(t, coeffs):
    f = np.ones(t.size, dtype=np.float64)
    for m, c in enumerate(coeffs, start=1):
        c = float(c)
        if c != 0.0:
            f = f + c * np.cos(float(m) * np.pi * t)
    f = np.maximum(f, 1e-4)
    return normalize_dist(f)


def shape_beta_t(t, a, b, delta=0.08):
    a = float(a)
    b = float(b)
    delta = float(delta)
    x = (t + delta) / (1.0 + 2.0 * delta)
    f = np.power(x, a) * np.power(1.0 - x, b)
    f = np.maximum(f, 1e-6)
    return normalize_dist(f)


def shape_gauss_t(t, c, s, amp, floor=0.03):
    c = float(c)
    s = float(max(1e-3, s))
    amp = float(amp)
    floor = float(floor)
    f = np.exp(-amp * (t - c) * (t - c) / (2.0 * s * s)) + floor
    return normalize_dist(f)


def random_shape_t(n, rng, t):
    kind = int(rng.integers(0, 6))
    if kind == 0:
        k = int(rng.integers(4, 18))
        coeffs = rng.normal(0.0, 0.45, size=k)
        for i in range(k):
            if (i + 1) % 2 == 0:
                coeffs[i] *= 0.25
        if rng.random() < 0.35:
            coeffs = -coeffs
        return shape_cos_t(t, coeffs)

    if kind == 1:
        c = float(rng.uniform(0.12, 0.88))
        s = float(rng.uniform(0.08, 0.38))
        amp = float(rng.uniform(1.0, 8.0))
        return shape_gauss_t(t, c, s, amp, float(rng.uniform(0.02, 0.12)))

    if kind == 2:
        a = float(rng.uniform(0.2, 6.0))
        b = float(rng.uniform(0.2, 6.0))
        if rng.random() < 0.5:
            a, b = b, a
        return shape_beta_t(t, a, b, float(rng.uniform(0.03, 0.20)))

    if kind == 3:
        a = float(rng.uniform(-1.0, 1.0))
        b = float(rng.uniform(-0.8, 0.8))
        c = float(rng.uniform(-0.6, 0.6))
        return shape_affine_t(t, a, b)

    if kind == 4:
        c1 = float(rng.uniform(0.10, 0.35))
        c2 = float(rng.uniform(0.65, 0.90))
        s1 = float(rng.uniform(0.05, 0.20))
        s2 = s1 * float(rng.uniform(0.6, 1.8))
        a1 = float(rng.uniform(1.0, 6.0))
        a2 = a1 * float(rng.uniform(0.25, 0.95))
        f = (
            np.exp(-a1 * (t - c1) * (t - c1) / (2.0 * s1 * s1))
            + a2 * np.exp(-a2 * (t - c2) * (t - c2) / (2.0 * s2 * s2))
            + 0.03
        )
        return normalize_dist(f)

    k = int(rng.integers(5, 22))
    coeffs = rng.normal(0.0, 0.55, size=k)
    coeffs *= np.exp(-0.10 * np.arange(k, dtype=np.float64))
    return shape_cos_t(t, coeffs)


def cand_key(item):
    return item[0]


def sort_cands(cands):
    cands.sort(key=cand_key)


def add_top(cands, f, sc, maxc):
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return
    if not np.all(np.isfinite(f)):
        return
    if not np.isfinite(sc):
        return
    if len(cands) < maxc:
        cands.append((sc, f.copy()))
        sort_cands(cands)
    elif sc < cands[-1][0]:
        cands[-1] = (sc, f.copy())
        sort_cands(cands)


def optimize_cos_coeffs(t, rng, coeffs, max_evals, size):
    coeffs = np.asarray(coeffs, dtype=np.float64).copy()
    f = shape_cos_t(t, coeffs)
    curr = fast_score_size(f, size)
    best = curr
    best_f = f.copy()
    scale = 0.18
    for _ in range(max(0, int(max_evals))):
        jump = 0.12 if rng.random() < 0.12 else 1.0
        cand = coeffs + rng.normal(0.0, scale * jump, size=coeffs.size)
        f2 = shape_cos_t(t, cand)
        s2 = fast_score_size(f2, size)
        if s2 < curr:
            coeffs = cand
            curr = s2
            scale = min(0.60, scale * 1.08)
        else:
            scale = max(0.02, scale * 0.97)
        if s2 < best:
            best = s2
            best_f = f2.copy()
    return best_f, best


def quick_candidates_with_t(n, rng, t, budget_s, size, cands, maxc=12):
    deadline = time.monotonic() + max(0.0, float(budget_s))
    if time.monotonic() >= deadline:
        sort_cands(cands)
        return cands

    if budget_s > 25.0:
        cos_starts = [
            (0.80,),
            (0.90, -0.20),
            (0.90, -0.40),
            (0.70, 0.20, -0.20),
            (0.80, -0.10, 0.25),
            (-0.80, 0.20),
            (0.60, -0.30, 0.30),
            (0.50, 0.40, -0.40, 0.20),
            (0.90, -0.45, 0.25, -0.20, 0.10),
            (0.40, 0.50, -0.30, 0.25, -0.20, 0.15),
        ]
        ev = 180 if budget_s > 80.0 else 80
        for cs in cos_starts:

                break
            f, s = optimize_cos_coeffs(t, rng, np.asarray(cs, dtype=np.float64), ev, size)
            add_top(cands, f, s, maxc)
    else:
        for cs in [(0.80,), (0.90, -0.30), (-0.80,)]:
            f = shape_cos_t(t, cs)
            add_top(cands, f, fast_score_size(f, size), maxc)

    for a in np.linspace(-0.9, 0.9, 9):
        if time.monotonic() >= deadline:
            break
        f = shape_affine_t(t, a, 0.0)
        add_top(cands, f, fast_score_size(f, size), maxc)

    beta_vals = (0.35, 0.8, 1.6, 3.5)
    for a in beta_vals:
        for b in beta_vals:
            if time.monotonic() >= deadline:
                break
            f = shape_beta_t(t, a, b, 0.10)
            add_top(cands, f, fast_score_size(f, size), maxc)

    count = 0
    max_rand = 20000 if budget_s > 80.0 else 3000
    while time.monotonic() < deadline and count < max_rand:
        f = random_shape_t(n, rng, t)
        sc = fast_score_size(f, size)
        add_top(cands, f, sc, maxc)
        count += 1

    sort_cands(cands)
    return cands


def refine_topk(f0, budget_s, deadline, size, k_start, k_end, alpha0, rng):
    f0 = np.asarray(f0, dtype=np.float64)
    n = f0.size
    L = 2 * n - 1
    budget_s = max(0.0, float(budget_s))
    k_start = int(max(4, min(L - 1, int(k_start))))
    k_end = int(max(1, min(k_start, int(k_end))))

    z = np.log(np.maximum(f0, 1e-300))
    z -= float(np.mean(z))
    z = np.clip(z, -12.0, 12.0)
    f = exp_normalize(z)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    alpha = float(alpha0)

    q0 = convolve_with_size(f, size)
    best_score = float(2.0 * n * np.max(q0))
    best_f = f.copy()

    no_impr = 0
    stagnant = 0
    start = time.monotonic()
    stop_min = min(deadline, start + budget_s)

    beta1 = 0.90
    beta2 = 0.999
    schedule_iters = 35000

    while time.monotonic() < stop_min:
        t += 1
        F = np.fft.rfft(f, size)
        q = np.fft.irfft(F * F, size)[:L]
        qmax = float(np.max(q))
        cur_score = 2.0 * float(n) * qmax

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_f = f.copy()
            no_impr = 0
            stagnant = 0
            alpha = min(0.28, alpha * 1.04)
        else:
            no_impr += 1
            if no_impr > 700:
                no_impr = 0
                alpha *= 0.86
                stagnant += 1
                if stagnant >= 4 or alpha < 0.025:
                    z += 0.20 * rng.normal(size=n)
                    z -= float(np.mean(z))
                    z = np.clip(z, -15.0, 15.0)
                    f = exp_normalize(z)
                    alpha = 0.14
                    stagnant = 0

        frac = min(1.0, float(t) / float(schedule_iters))
        logk = np.log(float(k_start)) + frac * (np.log(float(k_end)) - np.log(float(k_start)))
        k = int(np.exp(logk))
        if k < 1:
            k = 1
        if k > L:
            k = L
        if k == L:
            k = L - 1

        if k <= 1:
            w = np.zeros(L, dtype=np.float64)
            w[int(np.argmax(q))] = 1.0
        else:
            idx = np.argpartition(q, -k)[-k:]
            w = np.zeros(L, dtype=np.float64)
            w[idx] = 1.0 / float(k)

        W = np.fft.rfft(w, size)
        c = np.fft.irfft(W * np.conj(F), size)
        gf = 2.0 * c[:n]
        dot = float(np.dot(gf, f))
        gz = f * (gf - dot)

        m = beta1 * m + (1.0 - beta1) * gz
        v = beta2 * v + (1.0 - beta2) * (gz * gz)
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        step = alpha * (m / bc1) / (np.sqrt(v / bc2) + 1e-8)

        z -= step
        z -= float(np.mean(z))
        z = np.clip(z, -15.0, 15.0)
        f = exp_normalize(z)

    q = convolve_with_size(f, size)
    cur = float(2.0 * n * np.max(q))
    if cur < best_score:
        best_score = cur
        best_f = f.copy()

    return best_f, best_score


def resample_log(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = f.size
    if old_n <= 0:
        return np.full(new_n, 1.0 / max(1, new_n), dtype=np.float64)
    x0 = np.linspace(0.0, 1.0, old_n)
    x1 = np.linspace(0.0, 1.0, int(new_n))
    z = np.log(np.maximum(f, 1e-300))
    z1 = np.interp(x1, x0, z)
    f1 = exp_normalize(z1)
    return f1
