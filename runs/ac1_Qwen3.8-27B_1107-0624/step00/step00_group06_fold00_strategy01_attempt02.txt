def evaluate_sequence(sequence):
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if type(x) is bool or not isinstance(x, (int, float)):
            return np.inf

            return np.inf
