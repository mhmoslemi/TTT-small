Uses LP-based direction finding with active set reduction,

import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix

def compute_score_fast(f):
    """Fast score computation using FFT for convolution."""
    n = len(f)
    # Use FFT for efficient convolution
    f_padded = np.zeros(2 * n - 1)
    f_padded[:n] = f
    conv = np.real(np.fft.irfft(np.fft.rfft(f, 2*n-1) ** 2, 2*n-1))

    M = np.max(conv)

    fft_f = np.fft.rfft(f, size)
    conv = np.fft.irfft(fft_f * fft_f, size)
    return conv[:size]  # should be exactly size

Uses active-set LP direction finding combined with multiplicative updates.

from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import math


def compute_conv(f):
    """Compute f*f using FFT. Returns array of length 2n-1."""
    n = len(f)
    if n <= 0:
        return np.array([])
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    F = np.fft.rfft(f, fft_size)
    conv = np.fft.irfft(F * F, fft_size)
    # FFT convolution might give tiny negative values due to numerical error
    conv = np.maximum(conv, 0.0)
    return conv[:size]


def get_score(f):
    """Compute score for sequence f."""
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return float('inf')
    conv = compute_conv(f)
    M = np.max(conv)

def mul_update(f, top_frac=0.05, beta=1.0):
    """
    Multiplicative update step.
    Reduces f_j proportionally to its contribution to peak convolution entries.
    """
    n = len(f)

    M = np.max(conv)
    S = np.sum(f)

        return f, conv

"""
Step function optimization: minimize 2n*max(f*f)/(sum f)^2.
Uses active-set LP direction finding combined with multiplicative updates.
Tries the parent sequence and alternative lengths/initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import math


def compute_conv(f):
    """Compute f*f using FFT. Returns array of length 2n-1."""
    n = len(f)
    if n <= 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    F = np.fft.rfft(f, fft_size)
    conv = np.fft.irfft(F * F, fft_size)
    conv = np.maximum(conv, 0.0)
    return conv[:size]


def mul_update(f, top_frac=0.05, beta=1.0):
    """Multiplicative update: reduce f where it contributes to peak convolution."""
    n = len(f)
    conv = compute_conv(f)
    M = np.max(conv)
    S = np.sum(f)

    if S < 0.01 or M <= 0:
        return f

    k_count = max(5, int(top_frac * (2 * n - 1)))
    active = np.argsort(conv)[-k_count:]

    h = np.zeros(n)
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            h[js] += f[k - js]

    max_h = np.max(h)
    if max_h > 0:
        g = f * np.exp(-beta * h / max_h)
    else:
        g = f.copy()

    g_sum = np.sum(g)
    if g_sum > 0:
        g *= S / g_sum

    return g


def solve_lp(f, M, active, f_sum, timeout=30.0):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for k in active."""
    n = len(f)
    m = len(active)
    if m == 0 or m > 2000:
        return None

    row_parts = []
    col_parts = []
    val_parts = []

    for idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            row_parts.append(np.full(len(js), idx, dtype=np.int32))
            col_parts.append(js.astype(np.int32))
            val_parts.append(f[k - js].astype(np.float64))

    if not row_parts:
        return None

    A = csr_matrix(
        (np.concatenate(val_parts), (np.concatenate(row_parts), np.concatenate(col_parts))),
        shape=(m, n)
    )

    c = -np.ones(n)
    b_ub = np.full(m, M + 1e-10)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': timeout})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            g_sum = np.sum(g)
            if g_sum > 0:
                g *= f_sum / g_sum
            return g
    except Exception:
        pass

    return None


def optimize_sequence(f_init, budget_s, use_lp=True, lp_every=4, lp_active_frac=0.025, mul_beta=0.7):
    """Optimize sequence within time budget. Returns (best_f, best_internal_score)."""
    f = np.array(f_init, dtype=np.float64).copy()
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return f, float('inf')

    # Normalize so sum = sqrt(2n), making score = max(conv)
    target_sum = math.sqrt(2.0 * n)
    f = f * (target_sum / S)

    conv = compute_conv(f)
    best_score = float(np.max(conv))
    best_f = f.copy()

    deadline = time.monotonic() + budget_s
    iter_count = 0
    no_improve = 0
    best_beta = mul_beta

    print(f"    Start: n={n}, score={best_score:.6f}")

    while time.monotonic() < deadline - 3:
        # Multiplicative update
        g = mul_update(f, top_frac=0.05, beta=best_beta)
        f = g
        conv = compute_conv(f)
        score = float(np.max(conv))

        if score < best_score - 1e-9:
            best_score = score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # LP direction finding
        if use_lp and iter_count % lp_every == lp_every - 1 and time.monotonic() < deadline - 10:
            M = float(np.max(conv))
            k_count = max(10, int(lp_active_frac * (2 * n - 1)))
            active = np.argsort(conv)[-k_count:]
            f_sum = float(np.sum(f))

            g_lp = solve_lp(f, M, active, f_sum, timeout=20.0)

            if g_lp is not None:
                for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                    f_try = (1.0 - t) * f + t * g_lp
                    f_try = np.clip(f_try, 0.0, 1000.0)

                    if s_try < 0.01:

                    # Score is scale-invariant, compute directly

                    score_try = 2.0 * n * float(np.max(conv_try)) / (s_try * s_try)
                    if score_try < best_score:
                        f = f_try * (target_sum / s_try)
                        best_score = score_try

                        no_improve = 0

        iter_count += 1

            best_beta = min(2.0, best_beta * 1.1)
            no_improve = 0

        # Early stop if very stuck
        if no_improve > 500:
            break

    return best_f, best_score
