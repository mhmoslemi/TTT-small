def build_conv_matrix(f, n):
    """Build matrix A where (A@g)[k] = (f*g)[k] for all k."""
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            js = k - np.arange(m_start, m_end + 1)
            A[k, m_start:m_end+1] = f[js]
    return A

    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k=0,...,2n-2.
