def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, s / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    return result

    """Linear convolution of a with itself via FFT. Returns array of length 2*len(a)-1."""
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:len(a) * 2 - 1]

    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
