import numpy as np
import time

def _fast_conv(a, b=None):
    """Fast full linear convolution via FFT."""
    if b is None:
        b = a
    m, p = len(a), len(b)
    n = m + p - 1
    size = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

def _soft_grad(a, alpha):
    """Return (gradient, current_max_R) for soft-max of conv peak."""
    R = _fast_conv(a)
    mk = R.max()
    sh = alpha * (R - mk)
    w = np.exp(sh)
    s = w.sum()
    if s == 0:
        w = np.zeros_like(R); w[R.argmax()] = 1.0
    else:
        w /= s
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, mk
