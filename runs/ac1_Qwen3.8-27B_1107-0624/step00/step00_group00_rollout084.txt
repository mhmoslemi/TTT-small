import numpy as np
import time

np.seterr(over='ignore', under='ignore', invalid='ignore')


def _next_pow2(m):
    """Next power of two >= m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _softmax(z):
    """Stable softmax, returning a probability vector."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z.astype(np.float64)
    zm = np.max(z)
    if not np.isfinite(zm):
        return np.full(z.shape, 1.0 / float(z.size), dtype=np.float64)
    e = np.exp(z - zm)
    s = np.sum(e)
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.shape, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def _z_from_a(a):
    """Log-space representation of a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 1e-100, 1000.0)
    z = np.log(a)
    return z - np.max(z)


def _fft_conv_full(a, size):
    """Full auto-convolution using FFT zero-padded to size."""
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)


def _score_a(a, n, size):
    """Internal scale-free objective for sum(a)=1: 2*n*max(conv(a,a))."""
    c = _fft_conv_full(a, size)
    c = c[:2 * n - 1]
    cm = np.max(c)
    if not np.isfinite(cm) or cm < 0.0:
        cm = 0.0
    return float(2.0 * n * cm)


def _score_candidate(z, n, size):
    """Score a log-domain candidate and return its probability vector."""
    a = _softmax(z)
    s = _score_a(a, n, size)
    return s, a


def _blend_probs(a1, a2, w1):
    """Geometric blend of two probability vectors in log space."""
    w1 = float(w1)
    l1 = np.log(np.clip(a1, 1e-100, 1000.0))
    l2 = np.log(np.clip(a2, 1e-100, 1000.0))
    l = w1 * l1 + (1.0 - w1) * l2
    lm = np.max(l)
    if not np.isfinite(lm):
        return np.full(a1.shape, 1.0 / float(a1.size), dtype=np.float64)
    e = np.exp(l - lm)
    s = np.sum(e)
    if not np.isfinite(s) or s <= 0.0:
        return np.full(a1.shape, 1.0 / float(a1.size), dtype=np.float64)
    return e / s


def _make_phase_rev(n, size):
    """Phase factors allowing FFT of reversed a from FFT of a."""
    k = np.arange(size // 2 + 1, dtype=np.float64)
    return np.exp(-2j * np.pi * k * float(n - 1) / float(size))


def _fallback_geometric(n):
    """Safe geometric sequence with known good scale behavior."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(1.6 * (t - 1.0))
    s = np.sum(a)
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _sanitize_sequence(a):
    """Convert to a valid non-negative list and keep sum at least 0.01."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        a = np.ones(max(1, int(a.size)), dtype=np.float64)
    elif s > 0.0:
        a = a / s
    seq = []
    for x in a:
        seq.append(float(x))
    return seq

def _refine_one_start(z, a, best_a, best_score, phase_seconds, deadline, rng,
                      n, size, phase_rev, len_c, name):
    """Adam refinement of a softmax-parameterized probability vector."""
    z = np.asarray(z, dtype=np.float64).copy()
    a = np.asarray(a, dtype=np.float64).copy()
    best_a = np.asarray(best_a, dtype=np.float64).copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    b1_corr = 1.0
    b2_corr = 1.0

    start_t = time.monotonic()
    last_improve = start_t
    restarts = 0
    iter_count = 0
    next_print = start_t + 20.0
    copy_threshold = 1e-7

    while time.monotonic() < deadline:
        iter_count += 1
        a = _softmax(z)

        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fa * fa, size)
        c = c[:len_c]

        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax < 0.0:
            cmax = 0.0
        score = 2.0 * n * cmax
        if not np.isfinite(score):
            score = float('inf')

        if score < best_score - copy_threshold:
            best_score = score
            best_a = a.copy()
            last_improve = time.monotonic()
        elif score < best_score:
            best_score = score

        elapsed = time.monotonic() - start_t
        progress = 0.0
        if phase_seconds > 0.0:
            progress = elapsed / float(phase_seconds)
            if progress > 1.0:
                progress = 1.0

        base_T = 25.0 + (5.0 - 25.0) * progress
        base_lr = 0.08 + (0.005 - 0.08) * progress
        no_imp = time.monotonic() - last_improve
        if no_imp > 90.0:
            if base_T < 60.0:
                base_T = 60.0
            if base_lr < 0.03:
                base_lr = 0.03

        T = base_T
        lr = base_lr

        tau = cmax / T
        if not np.isfinite(tau) or tau <= 0.0:
            tau = 1e-12

        # Convert c in-place into softmax weights p = softmax(c/tau).
        c *= (1.0 / tau)
        pm = float(np.max(c))
        c -= pm
        np.exp(c, out=c)
        ps = float(np.sum(c))
        if not np.isfinite(ps) or ps <= 0.0:
            c = np.ones(len_c, dtype=np.float64) / float(len_c)
        else:
            c /= ps

        # Reuse fa to form FFT of reversed a.
        np.conjugate(fa, out=fa)
        fa *= phase_rev

        fp = np.fft.rfft(c, size)
        h = np.fft.irfft(fp * fa, size)
        G = h[n - 1:len_c]

        dot = float(np.dot(a, G))
        grad = a * (G - dot)
        grad *= 2.0 * n

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        b1_corr *= beta1
        b2_corr *= beta2

        mt = m / (1.0 - b1_corr)
        vt = v / (1.0 - b2_corr)
        update = lr * mt / (np.sqrt(vt) + eps)

        umax = float(np.max(np.abs(update)))
        if not np.isfinite(umax):
            z = _z_from_a(best_a)
            m.fill(0.0)
            v.fill(0.0)
            b1_corr = 1.0
            b2_corr = 1.0
            last_improve = time.monotonic()
            restarts += 1
            continue

        max_step = 0.20
        if umax > max_step:
            update *= max_step / umax
        z = z - update

        if iter_count % 1000 == 0:
            zm = float(np.mean(z))
            if np.isfinite(zm) and abs(zm) > 1.0:
                z -= zm

        now = time.monotonic()
        if now > next_print:
            print("phase " + str(name) +
                  " iter " + str(iter_count) +
                  " score " + str(score) +
                  " best " + str(best_score) +
                  " T " + str(T) +
                  " lr " + str(lr) +
                  " restarts " + str(restarts), flush=True)
            next_print = now + 30.0

        if no_imp > 150.0 and restarts < 12:
            if np.isfinite(score) and score < best_score * 1.03:
                base_z = z.copy()
                scale = 0.03 + 0.08 * float(rng.random())
            else:
                base_z = _z_from_a(best_a)
                scale = 0.08 + 0.18 * float(rng.random())
            z = base_z + rng.standard_normal(n) * scale
            z -= float(np.mean(z))
            m.fill(0.0)
            v.fill(0.0)
            b1_corr = 1.0
            b2_corr = 1.0
            restarts += 1
            last_improve = time.monotonic()
            print("restart in phase " + str(name) + " best " + str(best_score), flush=True)

    return z, a, best_a, best_score


def propose_candidate(seed=42, budget_s=1800):

    Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative a.

    Uses a softmax simplex parametrization and Adam on log-probabilities,
    with FFT auto-convolution, log-sum-exp smoothed max objective,
    multi-start initialization, blending, and restarts.
    """
    if budget_s is None:
        budget_s = 1800.0
    budget_s = max(60.0, float(budget_s))
    rng = np.random.default_rng(int(seed))

    start_all = time.monotonic()
    hard = start_all + budget_s
    reserve_final = 45.0 if budget_s > 200.0 else 10.0
    opt_deadline = hard - reserve_final
    if opt_deadline < start_all + 20.0:
        opt_deadline = hard - 2.0

    n = 4096
    try:
        if budget_s >= 1500.0:
            n = 10000
        elif budget_s >= 800.0:
            n = 8000
        elif budget_s >= 300.0:
            n = 5000
        else:
            n = 2048

        size = _next_pow2(3 * n - 2)
        len_c = 2 * n - 1
        phase_rev = _make_phase_rev(n, size)
        direct_list = []

        t = np.linspace(0.0, 1.0, n, dtype=np.float64)
        u = t - 0.5
        uu = u * u

        initial = []
        cidx = 0

        # Uniform baseline.
        z = np.zeros(n, dtype=np.float64)
        sc, a = _score_candidate(z, n, size)
        if np.isfinite(sc):
            initial.append((sc, cidx, a))
            cidx += 1

        # Monotone exponentials / power-like ramps.
        div = 1
        while div <= 50:

            z = y * t
            z -= float(np.max(z))
            sc, a = _score_candidate(z, n, size)
            if np.isfinite(sc):
                initial.append((sc, cidx, a))
                cidx += 1
            div += 1

        # Power-law ramps.

        for alpha in powers:
            z = np.log1p(alpha * t)
            z -= float(np.max(z))
            sc, a = _score_candidate(z, n, size)
            if np.isfinite(sc):
                initial.append((sc, cidx, a))
                cidx += 1

        # U-shaped and inverted U-shaped candidates.
        amps = (1.0, 2.0, 4.0, 8.0, 15.0)

            if np.isfinite(sc):

            z_cur, a_cur, best_a, best_score = _refine_one_start(
                z_cur, a_cur, best_a, best_score, phase_seconds,

            )

            direct_list.append((best_score, len(direct_list), np.array(best_a, dtype=np.float64)))
            print("after phase", phase_name, "best", best_score, flush=True)

        direct_list.sort()

            if np.isfinite(final_score) and sc > final_score + 0.05:

            a_fb = _fallback_geometric(4096)

        print("exception fallback len", len(seq), flush=True)
