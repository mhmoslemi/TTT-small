Minimize 2*n*max(convolve(a,a))/sum(a)^2 using FFT-accelerated multiplicative
updates with adaptive parameters, multiple structured starting points,
and simulated annealing local search.

The multiplicative update a_j <- a_j * exp(-eta * grad_j / max_grad) where
grad_j measures each element's contribution to convolution peaks, naturally
preserves non-negativity and flattens the autocorrelation.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """FFT-based linear convolution of two 1D arrays."""
    n_out = len(a) + len(b) - 1
    size = 1 << max(0, (n_out - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n_out]


def fast_score(a):
    """Fast score: 2*n*max(conv)/sum^2."""
    n = len(a)
    b = fft_convolve(a, a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * b.max() / (sum_a * sum_a))


def multiplicative_step(a, eta, peak_frac):
    """
    One multiplicative update step.
    Returns (new_a, old_score_before_update).
    """
    n = len(a)
    b = fft_convolve(a, a)
    bmax = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return a, np.inf
    old_score = 2.0 * n * bmax / (sum_a * sum_a)

    T = bmax * peak_frac
    e = np.maximum(0.0, b - T)
    if e.max() < 1e-14:
        return a, old_score

    # grad[k] = sum_m e[m+k] * a[m] for k=0..n-1
    # Computed as (e * a_flip)[n-1+k] where a_flip = a[::-1]
    a_flip = a[::-1]
    conv = fft_convolve(e, a_flip)
    grad = conv[n - 1: 2 * n - 1]
    grad = np.maximum(0.0, grad)
    max_grad = grad.max()
    if max_grad < 1e-14:
        return a, old_score

    a_new = a * np.exp(-eta * grad / max_grad)
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new, old_score


def run_multiplicative(a, eta, peak_frac, max_iter, deadline):

            best_score = score
            stall = 0

    return best, best_score

    n = len(a)
