import numpy as np
from scipy.optimize import linprog
import time

def build_conv_matrix(n):
    """Build the convolution matrix F where (F*b)[k] = sum_i F[k,i]*b[i] = (a*a)[k] with a replaced by b."""
    # F[k, i] = 1 if 0 <= k-i <= n-1, else 0
    # i.e., F[k, i] = 1 if max(0, k-n+1) <= i <= min(n-1, k)
    F = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        F[k, lo:hi+1] = 1.0
    return F

def compute_max_conv(a):
    """Compute max of full convolution of a with itself."""
    c = np.convolve(a, a)
    return np.max(c)

def solve_lp(a, M, F):
    """Solve: max sum(b) s.t. F*b <= M, b >= 0."""
    c_obj = -np.ones(len(a))  # minimize -sum(b)
    A_ub = F
    b_ub = np.full(F.shape[0], M)
    bounds = [(0, None)] * len(a)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
