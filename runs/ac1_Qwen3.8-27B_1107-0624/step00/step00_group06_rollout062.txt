import time

def _next_pow2(m):
    """Smallest power of 2 >= m."""
    if m <= 1:
        return 1
    p = 1
    while p < m:
        p *= 2
    return p

def _autocorr(a):
    """Linear autocorrelation (convolution with self) via FFT."""
    n = len(a)
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _grad_smooth_max(b, a, T):
    """
    Gradient of (1/T)log(sum exp(T*b_k)) w.r.t. a.
    = sum_k w_k * 2*a[k-j] (valid) = 2 * correlate(w, a)
