import numpy as np
from scipy.optimize import linprog
import time


"""
LP + local-search minimiser for  2*n*max(conv(a,a)) / (sum(a))**2 .

Strategy:
  * Normalise the working vector to unit mass (objective is scale invariant).
  * The uniform vector has self-convolution peak 1/n  ->  objective 2.0.
  * We need a "flat-convolution" structure, so we (a) run a fast local search
    (block scaling, low-frequency ripples, peak shaving) to escape the uniform
    basin, and (b) periodically solve an LP that finds a non-negative direction
    g with min peak of the cross-convolution f*g, then take a short line search
    mixing f into g.  Random restarts from several templates avoid local minima.
  * The exact evaluator 'evaluate_sequence' (already imported) is the authority
    whenever we promote a candidate to the global best; a cheap vectorised
    surrogate is used inside the loops.
"""


def _compute_obj(f, n):
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    c = np.convolve(f, f)
    return 2.0 * n * float(np.max(c)) / (s * s)


def _normalise(f):
    s = np.sum(f)
    if s < 0.01:
        return np.full_like(f, 1.0 / f.size)
    return f / s


def _templates(n, rng):
    """A handful of diverse non-negative unit-mass starting points."""
    out = []
    out.append(np.ones(n, dtype=np.float64) / n)
    try:
        g = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if g.ndim == 1 and g.size == n and np.isfinite(g).all():
            g = np.clip(g, 0.0, 1000.0)
            if np.sum(g) >= 0.01:
                out.append(_normalise(g))
    except Exception:
        pass
    out.append(_normalise(np.abs(rng.normal(0.0, 1.0, size=n)) + 1e-6))
    x = np.linspace(-1.0, 1.0, n)
    out.append(_normalise(np.abs(1.0 - x * x) + 1e-6))              # two-hump
    out.append(_normalise(0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 1e-6))  # ripple
    k = max(2, n // 257)
    w = _normalise(0.5 * (1.0 + np.cos(np.arange(n) * 2.0 * np.pi / n)) + 1e-6)
    m = int(n // k)
    for i in range(0, n, m):
        lo = max(0, i - 1)
        hi = min(n, i + 2)
        w[lo:hi] += 4.0
    out.append(_normalise(w))                                        # comb
    return out


def _block_perturb(f, rng):
    c = f.copy()
    n = c.size
    width = int(rng.integers(max(1, n // 400), max(2, n // 16)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c[start:stop] *= float(np.exp(rng.normal(0.0, 0.25)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return np.clip(c, 0.0, None)


def _ripple(f, rng):
    n = f.size
    freq = int(rng.integers(1, max(2, n // 100)))
    phase = rng.uniform(0.0, 2.0 * np.pi)
    amp = rng.uniform(0.05, 0.6)
    mod = 1.0 + amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return np.clip(f * mod, 0.0, None)


def _peak_shave(f, rng):
    n = f.size
    c = np.convolve(f, f)
    k = int(np.argmax(c))
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    grad[lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    lr = rng.uniform(0.02, 0.12)
    return np.clip(f - lr * grad, 0.0, None)


def _local_search(f, rng, deadline, best, best_score, step_budget):
    f = _normalise(np.clip(f, 0.0, None)).copy()
    cur = _compute_obj(f, f.size)
    n = f.size
    stall = 0
    while time.monotonic() < deadline and (step_budget is None or cur >= 0):
        r = rng.random()
        if r < 0.40:
            cand = _block_perturb(f, rng)
        elif r < 0.70:
            cand = _ripple(f, rng)
        else:
            cand = _peak_shave(f, rng)
        cand = _normalise(cand)
        sc = _compute_obj(cand, n)
        if sc < cur:
            f = cand
            cur = sc
            stall = 0
            if cur < best_score:
                best_score = float(evaluate_sequence(f.tolist()))
                if best_score < best[0]:
                    best[0] = best_score
                    best[1] = f.copy()
        else:
            stall += 1
            if stall > 300:
                break
    return f, best, best_score


def _lp_direction(f, n, deadline):
    A = np.empty((2 * n - 1, n + 1))
    A[:, n] = -1.0
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res is None or not getattr(res, "success", False):
        return None
    g = np.clip(res.x[:n], 0.0, None)
    s = np.sum(g)
    if s < 0.01:
        return None
    return g / s


def _lp_step(f, deadline, best, best_score):
    n = f.size
    if time.monotonic() > deadline - 6.0:
        return f, best, best_score
    g = _lp_direction(f, n, deadline)
    if g is None:
        return f, best, best_score
    cur = _compute_obj(f, n)
    best_t, best_sc = 0.0, cur
    for t in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8):
        hn = (1.0 - t) * f + t * g
        sc = _compute_obj(hn, n)
        if sc < best_sc:
            best_sc, best_t = sc, t
    if best_t > 0.0:
        f = _normalise((1.0 - best_t) * f + best_t * g)
        if best_sc < best_score - 1e-12:
            best_score = float(evaluate_sequence(f.tolist()))
            if best_score < best[0]:
                best[0] = best_score
                best[1] = f.copy()
    return f, best, best_score

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 8.0)

    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 4806
    if n < 2 or n > 20000:
        n = 4806

    best = [float("inf"), np.ones(n, dtype=np.float64) / n]

    tpls = _templates(n, rng)

    global_step = 0
    best_seen = best_score

        # local search until a short window is used, then an LP jump

            f, best, best_score = _local_search(f, rng, win_end, best, best_score, None)
            global_step += 1

        if time.monotonic() >= deadline:
            break

            print("rep %d  best = %.6f" % (rep, best_score), flush=True)
            best_seen = best_score
        else:

            best[1] = _normalise(best[1] + 1e-2 * np.abs(rng.normal(0, 1, size=n)))
            best_score = float(evaluate_sequence(best[1].tolist()))

    # final polished refinement from the global best
    f = best[1].copy()
    while time.monotonic() < deadline - 4.0:
        f, best, best_score = _local_search(f, rng, deadline, best, best_score, None)
        if time.monotonic() < deadline - 6.0:
            f, best, best_score = _lp_step(f, deadline, best, best_score)

    out = np.clip(best[1], 0.0, 1000.0).tolist()
    final = float(evaluate_sequence(out))
    print("FINAL  n =", n, "  score =", final, flush=True)
    return out
