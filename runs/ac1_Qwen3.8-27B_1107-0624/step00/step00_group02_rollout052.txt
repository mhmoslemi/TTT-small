   c. Adjust a to reduce the hot spots: for each hot spot k, the contribution to c[k] from a[i] is a[i]*a[k-i]. To reduce c[k], we should decrease a[i] * a[k-i] for the pairs (i, k-i) that contribute most.
   d. Normalize a to have sum 1.
   e. Project to non-negativity.

This is essentially a projected subgradient method.

def fast_convolution(a):
    """Compute full self-convolution using FFT for speed."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[:size]
    return c


def compute_subgradient(a, c, num_active=3):
    """
    Compute subgradient of max(c) w.r.t. a using top-k active indices.
    grad[i] = (2/num_active) * sum_{k in active} a[k-i] (if 0<=k-i<n)
    """
    n = len(a)
    # Find top-k values and their indices
    if num_active >= len(c):
        active = np.arange(len(c))
    else:
        # Use argpartition for efficiency
        part_idx = np.argpartition(c, -num_active)[-num_active:]
        active = part_idx

    grad = np.zeros(n)
    for k in active:
        k = int(k)
        # j = k - i must be in [0, n-1], so i in [max(0,k-n+1), min(k, n-1)]
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo:-1] if i_hi > i_lo else 2.0 * a[k - i_lo]
            # More carefully: for i in [i_lo, i_hi], j = k-i, so a[j] = a[k-i]
            # When i goes from i_lo to i_hi, j goes from k-i_lo down to k-i_hi
    grad /= num_active
    return grad


def optimize_fixed_length(n, time_budget, rng, init=None, use_topk=True):
    """
    Optimize a sequence of fixed length n using projected subgradient descent.
    Returns (best_sequence, best_objective).

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:

    else:
        a = a / s

    c = fast_convolution(a)
    best_c_max = c.max()
    best_a = a.copy()
    best_obj = 2.0 * n * best_c_max

    v = np.zeros(n)  # momentum velocity
    momentum = 0.9
    step_0 = 0.15
    topk = min(5, 2 * n - 1)

    t_start = time.monotonic()
    iteration = 0

    patience = 3000

    while time.monotonic() - t_start < time_budget:
        c = fast_convolution(a)
        c_max = c.max()
        obj = 2.0 * n * c_max

        if obj < best_obj:

            best_c_max = c_max
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > patience:
            break

        # Compute subgradient
        if use_topk:
            part_idx = np.argpartition(c, -topk)[-topk:]
            grad = np.zeros(n)
            for k in part_idx:

                i_lo = max(0, k - n + 1)

        else:

            v = np.zeros(n)
