Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses gradient descent on convolution flatness, LP-guided direction finding,
and multi-start diversity to escape local minima.
"""

import numpy as np
import time
from numpy.lib.stride_tricks import as_strided
from scipy.optimize import linear_program


def _compute_score(a):
    """Compute the objective 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * b.max() / (s * s)


def _gradient_flatness(a):
    """
    Compute the negative gradient of sum((b_k - mean(b))^2) w.r.t. a,
    projected to the sum-preserving subspace. Returns the descent direction.
    """
    n = len(a)
    a_norm = a / a.sum()
    b = np.convolve(a_norm, a_norm)  # length 2n-1
    c = b - b.mean()
    # grad[j] = sum_m c[j+m] * a_norm[m] for j=0..n-1
    # Use as_strided to create the (n,n) Hankel-like view
    window = as_strided(c, shape=(n, n), strides=(c.strides[0], c.strides[0]))
    grad = window @ a_norm
    # Project to sum-zero (preserve normalization)
    grad = grad - grad.mean()
    return -grad  # descent direction


def _cosine_init(n):
    """Raised cosine window."""
    x = np.linspace(0, np.pi, n)
    return np.cos(x) ** 2 + 1e-6


def _exp_init(n):
    """Exponential decay from center."""
    mid = (n - 1) / 2.0
    x = np.abs(np.arange(n) - mid) / mid
    return np.exp(-2.0 * x) + 1e-6


def _gaussian_init(n):
    """Gaussian window."""
    mid = (n - 1) / 2.0
    x = np.abs(np.arange(n) - mid) / mid
    return np.exp(-4.0 * x * x) + 1e-6


def _random_init(n, rng):
    """Random positive initialization."""
    return 0.1 + rng.random(n)


def _lp_direction(a, M, threshold=0.85, max_constraints=2000):
    """
    Solve the LP to find a direction g that maximizes sum(g)
    subject to ||a * g||_inf <= M and g >= 0.
    Returns the normalized direction g, or None if LP fails.
    """
    n = len(a)
    b = np.convolve(a, a)
    # Select active constraints
    active_thresh = M * threshold
    active_mask = b >= active_thresh
    active_indices = np.where(active_mask)[0]
    
    # Cap the number of constraints for speed
    if len(active_indices) > max_constraints:
        # Keep the most binding ones
        active_b = b[active_indices]
        order = np.argsort(active_b)[::-1][:max_constraints]
        active_indices = active_indices[order]
    
    n_active = len(active_indices)
    if n_active == 0:
        return None
    
    # Build constraint matrix
    C = np.zeros((n_active, n))
    for i, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        C[i, idx] = a[k - idx]
    
    # LP: max sum(g) s.t. C@g <= M, g >= 0
    # scipy form: min c@x s.t. A_ub@x <= b_ub, bounds
    c_obj = -np.ones(n)
    try:
        res = linear_program(
            c_obj,
            A_ub=C,
            b_ub=np.full(n_active, M),
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': 30}
        )
        if res.success and res.x is not None:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient descent on convolution flatness, LP-guided direction
finding, and multi-start diversity to escape local minima.
"""

import numpy as np

def _fft_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1
    while N < size:
        N *= 2
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def _compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _fft_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def _gradient_flatness(a):
    """
    Negative gradient of sum((b_k - mean(b))^2) w.r.t. a (with sum(a) fixed),
    computed via FFT cross-correlation. Returns descent direction.
    """
    n = len(a)
    a_norm = a / a.sum()
    b = _fft_convolve(a_norm)
    c = b - b.mean()
    # R[j] = sum_{m=0}^{n-1} c[m+j] * a_norm[m]

    Y = np.fft.rfft(a_norm[::-1], N)
    conv = np.fft.irfft(X * Y, N)
    R = conv[2 * n - 2 : n - 2 : -1].copy()
    R = R - R.mean()
    return -R

def _lp_direction(a, M, threshold, max_constraints):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    b = _fft_convolve(a)
    active_thresh = M * threshold
    active_mask = b >= active_thresh
    active_indices = np.where(active_mask)[0]

    if len(active_indices) > max_constraints:
        active_b = b[active_indices]
        order = np.argsort(active_b)[::-1][:max_constraints]
        active_indices = active_indices[order]

    n_active = len(active_indices)
    if n_active == 0:
        return None

    C = np.zeros((n_active, n))
    for i in range(n_active):
        k = active_indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        C[i, idx] = a[k - idx]

    c_obj = -np.ones(n)
    try:
        res = linear_program(
            c_obj, A_ub=C, b_ub=np.full(n_active, M),
            bounds=[(0.0, None)] * n,
            method='highs', options={'presolve': True, 'time_limit': 20}
        )
        if res.success and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

                        break

        if local_score < best_score:
            best_score = local_score
            best_a = local_best.copy()
            print(f"  Diversity [{init_idx}] improved: {best_score:.6f}")
        else:
            print(f"  Diversity [{init_idx}]: best={local_score:.4f} (no improvement)")

    # Final: verify with the official evaluator
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = evaluate_sequence(final_list)
    print(f"Final evaluator score: {final_score:.6f}")
    print(f"Sequence length: {len(final_list)}")

    return final_list
