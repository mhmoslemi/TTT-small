We normalize to sum(a) = 1 and minimize 2*n*max(a*a).

from scipy.fft import fft, ifft


def auto_convolve(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A, fsize)[:size])


def cross_correlation(w, a):
    """Compute cross-correlation: result[i] = sum_j w[i+j] * a[j]."""
    n = len(a)
    m = len(w)
    size = 1 << (m + n - 1 - 1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    return np.real(ifft(W * np.conj(A), size))[:n]


def simplex_project(v, target_sum):
    """Project v onto {x : x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    # Scale if needed
    if v.sum() == 0:
        return np.full(n, target_sum / n)
    
    # Binary search method
    lo = v.min()
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.maximum(v - mid, 0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.maximum(v - theta, 0)

    Compute gradient of logsumexp(beta * c) / beta w.r.t. a.
    c = a * a (auto-convolution).

    # = 2 * cross_correlation(weights, a)

"""
Hybrid optimization for minimizing f(a) = 2*n*max(a*a)/(sum(a))^2 over non-negative sequences.
Uses: (1) mirror descent subgradient method, (2) LP-based direction finding, (3) simulated annealing.
The objective is scale-invariant, so we work on the simplex sum(a)=1.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def auto_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A, fsize)[:size])


def cross_corr(w, a):
    """result[i] = sum_j w[i+j]*a[j], computed via FFT."""
    n = len(a)
    m = len(w)
    size = 1 << (m + n - 1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    return np.real(ifft(W * np.conj(A), size)[:n])


def simplex_project(v, s):
    """Project v onto {x >= 0, sum(x) = s} via bisection."""
    n = len(v)
    if s <= 0:
        return np.zeros(n)
    v_min, v_max = v.min(), v.max()
    if v_min >= 0 and v.sum() <= s:
        # Already in the simplex (if we just need sum <= s)
        pass
    lo, hi = v_min, v_max
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.maximum(v - mid, 0.0).sum() > s:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, c, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*c_k)) w.r.t. a.
    c = a * a.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Softmax weights
    c_shift = c - c.max()
    w = np.exp(beta * c_shift)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # Gradient = 2 * cross_correlation(w, a)
    return 2.0 * cross_corr(w, a)


def mirror_step(a, grad, eta):
    """Mirror descent step on the simplex (KL geometry)."""
    log_a = np.log(a + 1e-300)
    new_log = log_a - eta * grad
    new_log -= new_log.max()
    new_a = np.exp(new_log)
    s = new_a.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.full(len(a), 1.0 / len(a))
    return new_a / s


def smoothed_max_value(c, beta):
    """Compute (1/beta)*log(sum exp(beta*c))."""
    c_max = c.max()
    if beta < 1e-10:
        return c.max()
    return c_max + np.log(np.exp(beta * (c - c_max)).sum()) / beta


def solve_lp_direction(a, c, M, top_k=50):
    """
    Solve LP: max sum(b) s.t. (a*b)_k <= M for top-k active constraints, b >= 0.
    Returns the LP solution or None.
    """
    n = len(a)
    # Select top-k constraints
    if top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
    else:
        idx = np.arange(len(c))
    
    K = len(idx)
    A_ub = np.zeros((K, n))
    for i, k in enumerate(idx):
        # A_ub[i, j] = a[k-j] if 0 <= k-j < n
        j_range = np.arange(max(0, k - n + 1), min(n, k + 1))
        a_idx = k - j_range
        A_ub[i, j_range] = a[a_idx]
    
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, np.inf)] * n, method='highs',
                      options={'time_limit': 5})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            return res.x
    except Exception:
        pass
    return None
