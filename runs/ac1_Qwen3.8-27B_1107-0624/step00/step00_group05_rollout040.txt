"""
Smooth gradient descent with FFT-based convolution and multiple initializations
to minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences.
Uses softmax-smoothed max objective with adaptive parameters.
"""
import numpy as np
import time


def _fft_conv(a):
    """Compute full linear convolution a*a using real FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    A = np.fft.rfft(a, fft_size)
    C = A * A
    return np.fft.irfft(C, fft_size)[:size]


def _gradient_from_weights(a, p):
    """
    Compute grad[i] = 2 * sum_j p[i+j] * a[j] for i=0..n-1.
    p has length 2n-1, a has length n. Uses np.correlate.
    """
    return 2.0 * np.correlate(p, a, mode='valid')


def _smooth_descent(a, time_budget, verbose=True):

    Run smooth gradient descent to minimize max(a*a) subject to sum(a)=1, a>=0.
    Returns the best a found.

            print(f"    it={it}, elapsed={elapsed:.1f}s, max_c={max_c:.10f}, obj={obj:.6f}, mu={mu:.3e}")

    return best_a


def _make_initializations(n, seed):
    """Generate several initial candidates with different structures."""
    rng = np.random.default_rng(seed)
    inits = []

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Raised cosine
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = 1.0 + np.cos(x)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    inits.append(a.copy())

    # 3. Two-lobe (mass at ends)
    a = np.zeros(n)
    a[:n // 4] = 4.0 / n
    a[3 * n // 4:] = 4.0 / n
    a /= a.sum()
    inits.append(a.copy())

    # 4. Two-lobe wider
    a = np.zeros(n)
    a[:n // 3] = 3.0 / n
    a[2 * n // 3:] = 3.0 / n
    a /= a.sum()
    inits.append(a.copy())

    # 5. Three-lobe
    a = np.zeros(n)
    seg = n // 6
    a[:seg] = 6.0 / (2 * seg)
    a[n // 2 - seg // 2:n // 2 + seg // 2] = 6.0 / seg
    a[-seg:] = 6.0 / (2 * seg)
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a.copy())

    # 6. Random smooth
    a = np.abs(rng.standard_normal(n))
    # Smooth with a box filter
    k = max(3, n // 200)
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.0)
    a /= a.sum()
    inits.append(a.copy())

    # 7. "Sinc-like" (Gaussian bump in frequency -> spread in time)
    x = np.linspace(-1, 1, n)
    a = np.exp(-2.0 * x * x)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    inits.append(a.copy())

    # 8. Alternating (comb)
    a = np.zeros(n)
    a[::2] = 1.0
    a /= a.sum()
    inits.append(a.copy())

    # 9. "Edge-weighted" (high at boundaries, low in middle)
    x = np.linspace(0, 1, n)
    a = x ** 2 + (1 - x) ** 2
    a /= a.sum()
    inits.append(a.copy())

    return inits


def _try_different_lengths(n, budget, seed):
    """Try a few nearby lengths to see if they help."""
    results = {}
    candidates = [n, n + 1, n - 1, n // 2, 2 * n]
    candidates = [c for c in candidates if 10 <= c <= 20000]
    rng = np.random.default_rng(seed + 1000)

    for length in candidates[:3]:  # Limit to 3 to save time
        if length == n:
            continue
        a = np.abs(rng.standard_normal(length))
        k = max(3, length // 100)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.0)
        a /= a.sum()
        score = evaluate_sequence(a.tolist())
        results[length] = (score, a)
        if score < np.inf:
            print(f"    Length {length}: initial score = {score:.6f}")

    return results


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(a*a)/(sum(a))^2.
    Uses smooth gradient descent with multiple initializations and
    FFT-accelerated convolution.
    """
    total_start = time.monotonic()
    deadline = total_start + budget_s - 15.0  # 15s buffer for safety

    # Get base sequence
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        base = np.ones(1024, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(1024, dtype=np.float64)

    n = base.size
    print(f"Starting with n = {n}, budget = {budget_s}s")

    # Generate initializations
    inits = _make_initializations(n, seed)

    # Include the base sequence (normalized)
    base_norm = base.copy()
    s = base_norm.sum()
    if s > 1e-10:
        base_norm /= s
    else:
        base_norm = np.ones(n) / n
    inits.insert(0, base_norm)

    # Evaluate all initializations
    best_a = None
    best_score = np.inf

    print(f"Evaluating {len(inits)} initial candidates...")
    for i, init in enumerate(inits):
        score = evaluate_sequence(init.tolist())
        if score < best_score:
            best_score = score
            best_a = init.copy()
        if score < np.inf:
            print(f"  Init {i}: score = {score:.6f}")

    print(f"Best initial: {best_score:.6f}")

    # Phase 1: Gradient descent from best initialization (use 60% of time)
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        t1 = remaining * 0.5
        print(f"Phase 1: Gradient descent ({t1:.0f}s budget) from best init...")
        improved = _smooth_descent(best_a, t1, verbose=True)
        score = evaluate_sequence(improved.tolist())
        print(f"  Phase 1 result: score = {score:.6f}")
        if score < best_score:
            best_score = score
            best_a = improved

    # Phase 2: Try gradient descent from other good inits (use 25% of time)
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        # Sort inits by score (best first), skip the one we already used
        scored_inits = []
        for i, init in enumerate(inits):
            sc = evaluate_sequence(init.tolist())
            scored_inits.append((sc, i, init))
        scored_inits.sort(key=lambda x: x[0])

        t2_each = min(remaining * 0.25 / max(1, len(scored_inits) - 1), remaining * 0.1)
        if t2_each > 5.0:
            print(f"Phase 2: Trying {len(scored_inits)-1} other inits ({t2_each:.0f}s each)...")
            for sc, i, init in scored_inits[1:]:
                if time.monotonic() > deadline - 5.0:
                    break
                improved = _smooth_descent(init, t2_each, verbose=False)
                score = evaluate_sequence(improved.tolist())
                if score < best_score:
                    best_score = score
                    best_a = improved
                    print(f"  Phase 2 init {i}: NEW BEST score = {score:.6f}")
                else:
                    print(f"  Phase 2 init {i}: score = {score:.6f} (not better)")

    # Phase 3: Fine-tune from the best with a fresh descent (use remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        print(f"Phase 3: Fine-tuning best ({remaining:.0f}s budget)...")
        # Add a small random perturbation to escape potential stuck points
        rng = np.random.default_rng(seed + 7777)
        a_perturbed = best_a.copy()
        a_perturbed += rng.normal(0, 0.01 / n, n)
        a_perturbed = np.maximum(a_perturbed, 0.0)
        a_perturbed /= a_perturbed.sum()

        improved = _smooth_descent(a_perturbed, remaining * 0.6, verbose=True)
        score = evaluate_sequence(improved.tolist())
        if score < best_score:
            best_score = score
            best_a = improved
            print(f"  Phase 3 result: NEW BEST score = {score:.6f}")
        else:
            print(f"  Phase 3 result: score = {score:.6f} (not better)")

        # Also try without perturbation
        remaining = deadline - time.monotonic()
        if remaining > 10.0:
            improved2 = _smooth_descent(best_a, remaining * 0.4, verbose=False)
            score = evaluate_sequence(improved2.tolist())
            if score < best_score:
                best_score = score
                best_a = improved2
                print(f"  Phase 3b result: NEW BEST score = {score:.6f}")

    print(f"\nFinal best score: {best_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")
    return [float(x) for x in best_a]
