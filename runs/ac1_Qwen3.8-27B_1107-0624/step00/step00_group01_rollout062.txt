```python
import numpy as np
import time

def fft_autocorr(a):
    """Compute full autocorrelation (convolution of a with itself) using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, n=size)
    ac = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return ac

def subgradient_step(a, M, K_star):
    """Compute subgradient of max(c_k) for k in K_star."""
    n = len(a)
    g = np.zeros(n)
    for k in K_star:
        # dc_k/da_i = 2*a_{k-i} if 0 <= k-i <= n-1
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        if lo <= hi:
            # For i in [lo, hi], k-i is in [k-hi, k-lo], which is in [0, n-1]
            # g[i] += 2 * a[k-i]
            idx = k - np.arange(lo, hi+1)  # these are the k-i values
            g[lo:hi+1] += 2 * a[idx]
    return g / len(K_star)

def optimize_sequence(a_init, budget):
    """Run projected subgradient descent."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S = np.sum(a)
    if S <= 0:
        return a_init
    a = a / S  # normalize to sum = 1
    
    best_a = a.copy()
    best_obj = 2 * n * np.max(fft_autocorr(a))
    
    lr = 0.01
    deadline = time.monotonic() + budget
    
    while time.monotonic() < deadline:
        c = fft_autocorr(a)
        M = np.max(c)
        obj = 2 * n * M
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Find max indices (with tolerance)
        tol = M * 1e-8 + 1e-15
        K_star = np.where(c >= M - tol)[0]
        
        # Compute subgradient
        g = subgradient_step(a, M, K_star)
        
        # Take step: minimize 2n * max(c), direction is -g
        # But we need to maintain sum(a) = 1, so center the gradient
        g_centered = g - np.mean(g)
        a = a - lr * g_centered
        a = np.maximum(a, 0)
        S = np.sum(a)
        if S > 0:
            a = a / S
        
        # Adapt learning rate
        c_new = fft_autocorr(a)
        M_new = np.max(c_new)
        if 2 * n * M_new > obj * (1 + 0.1):
            lr *= 0.8
        elif 2 * n * M_new < obj * 0.99:
            lr = min(lr * 1.1, 1.0)
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b=None):
    """Full linear convolution using FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na+nb-1]

def proj_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_objective_and_conv(a):
    """Return (objective, convolution, max_val, sum)."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-12:
        return np.inf, None, 0, 0
    c = fft_convolve(a)
    M = np.max(c)
    obj = 2.0 * n * M / (S * S)
    return obj, c, M, S

def run_subgradient(a_init, max_time):
    """Projected subgradient descent on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    S = np.sum(a)
    if S < 1e-12:
        return a_init
    a = proj_simplex(a)
    
    obj, c, M, _ = compute_objective_and_conv(a)
    best_a = a.copy()
    best_obj = obj
    
    # Adaptive step size
    lr = 0.5 / np.sqrt(n)
    deadline = time.monotonic() + max_time
    stuck = 0

        # Find active set (indices near max)
        threshold = M - 1e-10
        K_star = np.array(np.where(c >= threshold)[0])

        for k in K_star:

        g -= np.mean(g)

        obj_new, c_new, M_new, _ = compute_objective_and_conv(a_new)

            best_a = a_new.copy()
            stuck = 0
        else:
```

```python
Uses projected subgradient descent with FFT-based convolution,

from scipy.sparse import csr_matrix, coo_matrix


def fft_convolve(a, b=None):
    """Full linear convolution using FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def proj_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_obj_and_conv(a):
    """Compute objective and convolution for a non-negative sequence."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-12:
        return np.inf, np.zeros(2 * n - 1), 0.0
    c = fft_convolve(a)
    M = np.max(c)
    obj = 2.0 * n * M / (S * S)
    return obj, c, M


def compute_subgradient(a, c, M, n):
    """Compute subgradient of max_k c_k with respect to a."""
    tol = 1e-10 * M + 1e-15
    K_star = np.where(c >= M - tol)[0]

    for k in K_star:

def lp_solve_single(f, M, n, time_limit=20):
    """Solve one LP: max sum(b) s.t. b>=0, (f*b)_k <= M for all k."""
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])

    A_ub = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
        else:
            return None
    except Exception:
        return None
```

    return best_a


def build_lp_matrix(f, n):
    """Build sparse A_ub matrix for LP: (f*b)_k <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_refine(a_init, max_time, verbose=False):
    """LP-based refinement: iteratively solve max sum(b) s.t. constraints."""
    n = len(a_init)
    f = a_init.astype(np.float64).copy()
    S = np.sum(f)
    if S < 1e-12:
        return a_init.copy()

    target_sum = np.sqrt(2.0 * n)
    f = f * target_sum / S

    obj, c, M = compute_obj_and_conv(f)
    best_f = f.copy()
    best_obj = obj

    t = 0.2
    deadline = time.monotonic() + max_time
    A_ub = build_lp_matrix(f, n)

    while time.monotonic() < deadline:
        # Solve LP
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        time_left = max(1.0, (deadline - time.monotonic()) * 0.3)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': time_left})
        except Exception:
            break

        if res.status != 0:
            break

        b = np.maximum(res.x, 0)
        sum_b = np.sum(b)
        if sum_b < 1e-12:
            break

        g = b * target_sum / sum_b
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        sf = np.sum(f_new)
        if sf < 1e-12:
            break
        f_new = f_new * target_sum / sf

        obj_new, c_new, M_new = compute_obj_and_conv(f_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            if verbose:
                print(f"    LP: obj={best_obj:.6f}")

        f, c, M = f_new, c_new, M_new

        # Rebuild constraint matrix (f changed)
        A_ub = build_lp_matrix(f, n)

        t = max(0.05, t * 0.9)

    return best_f


def make_random_init(n, rng):
    """Random smooth initialization."""
    a = rng.random(n)
    kernel = np.ones(7) / 7.0
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.05)
    return a


def make_cosine_init(n, rng):
    """Cosine-based initialization."""
    x = np.arange(n, dtype=np.float64)
    freq = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.3, 0.8)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    return np.maximum(a, 0.0)


def make_dual_cosine_init(n, rng):
    """Dual-frequency cosine initialization."""
    x = np.arange(n, dtype=np.float64)
    f1 = rng.uniform(0.5, 2.0)
    f2 = rng.uniform(2.0, 6.0)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    a = (1.0 + 0.5 * np.cos(2.0 * np.pi * f1 * x / n + p1)
         + 0.3 * np.cos(2.0 * np.pi * f2 * x / n + p2))
    return np.maximum(a, 0.0)


def make_block_init(n, rng):
    """Block-structured initialization."""
    nblocks = int(rng.integers(10, 50))
    block_size = max(1, n // nblocks)
    a = np.ones(n)
    for ib in range(nblocks):
        start = ib * block_size
        end = min((ib + 1) * block_size, n)
        scale = rng.uniform(0.2, 1.5)
        a[start:end] *= scale
    return np.maximum(a, 0.0)


def make_sawtooth_init(n, rng):
    """Sawtooth-like initialization with smooth transitions."""
    x = np.arange(n, dtype=np.float64)
    period = int(n / rng.integers(3, 10))
    a = 1.0 + 0.6 * np.sin(2.0 * np.pi * x / period)
    return np.maximum(a, 0.0)
