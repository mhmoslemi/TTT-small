def lpd_iteration(f, M):
    """One LPD iteration. f is normalized to sum=1."""
    n = len(f)
    # Build constraint matrix (sparse)
    # A[k, j] = f[j] if 0 <= k-j <= n-1
    # This is the convolution matrix of f
    
    # For efficiency, use scipy.sparse
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n-1)
        for j in range(j_lo, j_hi+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max sum(b) s.t. A @ b <= M, b >= 0
    # scipy linprog minimizes, so we negate the objective
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=M*np.ones(2*n-1), bounds=(0, None), method='highs')
    
    if res.status != 0:
        return None
