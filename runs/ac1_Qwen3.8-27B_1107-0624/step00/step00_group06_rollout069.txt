"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 via FISTA (accelerated projected gradient)
on the smooth log-sum-exp approximation of the max, with adaptive beta scheduling,
multiple initializations, and a local polishing phase.
"""

import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = int(np.max(rho[cond]))
    theta = (cssv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0.0)


def self_convolve(a, size):
    """Full self-convolution of a via FFT, length 2*len(a)-1."""
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    n = len(a)
    return c[:2 * n - 1]


def compute_gradient(a, c, beta, n, size):
    """
    Compute gradient of (1/beta)*logsumexp(beta*c) where c = self_conv(a).
    Result: 2 * cross_corr where cross_corr[j] = sum_m a[m]*p[j+m].
    """
    # Softmax with shift for stability
    c_shifted = beta * c - beta * c.max()
    p = np.exp(c_shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = p / p_sum

    # Cross-correlation: corr[j] = sum_m a[m] * p[j+m] for j=0,...,n-1
    # = conv(a[::-1], p)[n-1 + j]
    af = a[::-1]
    fa = np.fft.rfft(af, size)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fa * fp, size)
    return 2.0 * conv[n - 1:2 * n - 1]


def run_fista(n, init, time_budget, verbose=False):
    """
    Run FISTA to minimize 2*n*max(self_conv(a)) subject to sum(a)=1, a>=0.
    Uses smooth log-sum-exp approximation with ramping beta.
    """
    a = np.asarray(init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = a[:n].copy()
    if len(a) < n:
        pad = np.zeros(n - len(a))
        a = np.concatenate([a, pad])

    size = int(2 ** np.ceil(np.log2(4 * n + 1)))

    x = a.copy()
    y = a.copy()
    t = 1.0

    # Compute initial objective
    c0 = self_convolve(x, size)
    best_score = 2.0 * n * c0.max()
    best_a = x.copy()

    deadline = time.monotonic() + time_budget
    it = 0
    lr_div = 1.0  # Multiplier for adaptive lr
    prev_score = best_score

    beta_start = 5.0
    beta_end = 500.0
    ramp_iters = 20000  # Over how many iterations to ramp beta

    while time.monotonic() < deadline:
        # Adaptive beta
        frac = min(1.0, it / max(ramp_iters, 1))
        beta = beta_start + (beta_end - beta_start) * frac * frac

        # Compute gradient at y
        c_y = self_convolve(y, size)
        grad = compute_gradient(y, c_y, beta, n, size)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

def local_polish(a, n, time_budget, rng, verbose=False):
    """Local search: random block perturbations and small coordinate moves."""
    best_a = a.copy()
    size = int(2 ** np.ceil(np.log2(4 * n + 1)))
    c = self_convolve(best_a, size)
    best_score = 2.0 * n * c.max()
    deadline = time.monotonic() + time_budget

    while time.monotonic() < deadline:
        # Random block multiplicative perturbation
        candidate = best_a.copy()
        n_pert = rng.integers(1, 5)
        for _ in range(n_pert):
            width = int(rng.integers(3, max(5, n // 50)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            scale = np.exp(rng.normal(0.0, 0.08, size=stop - start))
            candidate[start:stop] *= scale
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 1e-10:
            continue
        candidate = candidate / s

        c_cand = self_convolve(candidate, size)
        score = 2.0 * n * c_cand.max()
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            if verbose:
                print(f"    polish improved: {best_score:.6f}")

            print(f"  -> {label}: {score:.6f} ({elapsed:.0f}s)")

    # Extended run on best result so far
    if best_global_result is not None and time.monotonic() < total_deadline - 30:
        best_n = len(best_global_result)
        best_init = np.array(best_global_result)
        ext_budget = min(extended_budget, total_deadline - time.monotonic() - 30)
        if ext_budget > 10:
            print(f"Extended FISTA on best (n={best_n}), budget={ext_budget:.0f}s")
            t0 = time.monotonic()
            best_a, score = run_fista(best_n, best_init, ext_budget, verbose=True)
            elapsed = time.monotonic() - t0
            if score < best_global_score:
                best_global_score = score
                best_global_result = [float(x) for x in best_a]
                print(f"  Extended improved: {best_global_score:.6f}")

    # Also try extended FISTA with different initializations on the best n
    best_n = len(best_global_result) if best_global_result is not None else 3000
    ext_inits = [
        np.ones(best_n) / best_n,
        np.sin(np.pi * np.arange(best_n) / best_n) ** 2,
    ]
    ext_inits = [x / x.sum() for x in ext_inits if x.sum() > 0]
    for ei in ext_inits:
        if time.monotonic() > total_deadline - 30:
            break
        ext_budget = min(60.0, total_deadline - time.monotonic() - 20)
        if ext_budget < 10:
            break
        print(f"Extended FISTA alt init (n={best_n}), budget={ext_budget:.0f}s")
        t0 = time.monotonic()
        best_a, score = run_fista(best_n, ei, ext_budget, verbose=False)
        elapsed = time.monotonic() - t0
        if score < best_global_score:
            best_global_score = score
            best_global_result = [float(x) for x in best_a]
            print(f"  Alt init improved: {best_global_score:.6f} ({elapsed:.0f}s)")

    # Phase 2: Local polishing
    if best_global_result is not None and time.monotonic() < total_deadline - 10:
        polish_budget = min(300.0, total_deadline - time.monotonic() - 5)
        if polish_budget > 5:
            best_n = len(best_global_result)
            best_arr = np.array(best_global_result)
            print(f"Polishing phase: n={best_n}, score={best_global_score:.6f}, budget={polish_budget:.0f}s")
            polished_a, polished_score = local_polish(
                best_arr, best_n, polish_budget, rng, verbose=True)
            if polished_score < best_global_score:
                best_global_score = polished_score
                best_global_result = [float(x) for x in polished_a]
                print(f"  Polished: {best_global_score:.6f}")

    # Final: verify with evaluate_sequence
    if best_global_result is None:
        # Fallback
        n_fb = 1000
        best_global_result = [1.0] * n_fb
        best_global_score = float(evaluate_sequence(best_global_result))
    else:
        # Verify
        final_score = float(evaluate_sequence(best_global_result))
        if not np.isfinite(final_score) or final_score > best_global_score * 1.01:
            print(f"WARNING: Final eval {final_score} differs from tracked {best_global_score}")
            best_global_score = final_score

    print(f"FINAL: score={best_global_score:.6f}, length={len(best_global_result)}")
    return best_global_result
