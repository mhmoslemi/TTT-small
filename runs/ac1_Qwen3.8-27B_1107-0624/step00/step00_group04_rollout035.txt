Key insight: the LP relaxation (fix f, optimize b subject to f*b bounded) provides
a direction to reduce the peak autocorrelation. Combined with randomization to escape
symmetric fixed points, this should improve upon the baseline of 2.0.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def compute_score(f):
    """Compute the objective value."""
    n = len(f)
    c = np.convolve(f, f)
    max_c = c.max()
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_c / (S * S)


def solve_lp_direction(a, M):
    """
    Solve LP: max w^T b subject to (a*b)[k] <= M for all k, b >= 0.
    Returns the optimal b.
    """
    n = len(a)
    
    # Use a slight asymmetry in weights to break ties
    w = np.linspace(1.0, 1.1, n)  # slight preference for later elements
    
    # Objective: minimize -w^T b
    c_obj = -w
    
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    
    for d in range(n):
        # Diagonal d: A[j+d, j] = a[d] for j = 0, ..., n-1
        # But we need j+d <= 2n-2, which is always true since j <= n-1 and d <= n-1
        rows = np.arange(d, d + n)
        cols = np.arange(n)
        A[rows, cols] = a[d]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status == 0:
        return result.x
    return None

        return f
    g = b * S / b_sum
    f_new = (1 - t_param) * f + t_param * g
    return np.clip(f_new, 0, None)
