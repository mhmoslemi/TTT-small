import time

def _fft_convolve(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fa2 = fa * fa
    result = np.fft.irfft(fa2, fft_size)[:size]
    return result

def _softmax(w):

    w = w - w.max()
    e = np.exp(w)
    return e / e.sum()

import numpy as np
import time

def _fft_convolve(a):
    """Compute full convolution using FFT for O(n log n) complexity."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    # Clamp small negative values from numerical error
    np.maximum(result, 0, out=result)
    return result

def _softmax(w):
    """Numerically stable softmax."""
    w = w - w.max()
    e = np.exp(w)
    return e / e.sum()

def _subgradient_descent_step(w, lr, rng, use_perturb=False):
    """One step of subgradient descent in log-space (softmax parameterization)."""
    n = len(w)
    a = _softmax(w)
    conv = _fft_convolve(a)
    k_star = np.argmax(conv)
    M = conv[k_star]
