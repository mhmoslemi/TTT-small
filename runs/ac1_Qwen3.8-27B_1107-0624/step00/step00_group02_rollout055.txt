import numpy as np
import time
from scipy.optimize import linprog

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = int(np.argmax(cond))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))

def lp_direction(a, M, method='highs'):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    
    # Build constraint matrix: (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            A[k, k-i] += a[i]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
    if res.success and res.x is not None:
        return res.x
    return None

    M = np.max(c)

    a_new = np.maximum(a_new, 0)
    a_new = a_new / np.sum(a_new)  # re-normalize
