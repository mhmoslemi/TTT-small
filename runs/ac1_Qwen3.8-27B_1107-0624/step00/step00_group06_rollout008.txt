import numpy as np

def _fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def _self_convolve(a):
    """Full self-convolution a * a using FFT."""
    return _fft_convolve(a, a)

def _conv_grad(a, w):
    """
    Compute gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a.
    Result: grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices only)
             = 2 * (a cross-correlate w)[j]
    """
    n = len(a)
    # Need: (a ⊛ w)[j] = sum_m a[m] * w[j+m] for j = 0,...,n-1
    # Via FFT: ifft(conj(fft(a)) * fft(w)) gives the cross-correlation
    # Need length at least n + (2n-1) - 1 = 3n-2 to avoid wrapping
    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(np.conj(A) * W, fft_size)
    return 2.0 * corr[:n]
