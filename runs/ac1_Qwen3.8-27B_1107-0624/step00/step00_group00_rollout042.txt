```
def optimize_sequence(n, z0, max_iters, focus_start, focus_end, eta0, deadline, best_score, label):

    best_z = z.copy(); best_sc = np.inf

       if time.monotonic() > deadline: break
       a,b,maxb = eval_z(z,n)
       score = 2*n*maxb
       if score < best_sc:
           best_sc=score; best_z=z.copy(); stalled=0
       else:

           # perturb

           if best_sc <= target: break?

       x = focus*(b - maxb)/maxb  # dimensionless exponent range focus

       grad = grad - mean

       gscale = np.std(grad) + 1e-12
```

```python
import numpy as np
import time
import math


"""
Strategy summary
================
Minimize:
    score(a) = 2 * n * max(a * a) / (sum(a)^2)
for a non-negative step sequence.  The score is scale invariant, so the
optimizer works with sum(a)=1 and minimizes max(a*a) using FFT-based
convolutions.  It uses several randomized/restarted exponentiated-gradient
refinements plus low-dimensional Fourier profile searches.  The final answer
is always verified with the provided evaluate_sequence.
"""


def _fft_len(m):
    m = max(1, int(m))
    n = 1
    while n < m:
        n <<= 1
    return n


def _norm_a(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if (not math.isfinite(s)) or s <= 1e-300:
        return np.ones_like(a, dtype=np.float64) / max(1, a.size)
    return a / s


def _score_a_fft(a):
    """FFT approximation of the official score for sum-normalized a."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return math.inf
    a = np.maximum(a, 0.0)
    a = _norm_a(a)
    n = int(a.size)
    L = 2 * n - 1
    N = _fft_len(L)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:L]
    b = np.maximum(b, 0.0)
    maxb = float(np.max(b))
    return 2.0 * float(n) * maxb


def _softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - float(np.mean(z))
    z = np.clip(z, -35.0, 35.0)
    a = np.exp(z)
    s = float(np.sum(a))
    if (not math.isfinite(s)) or s <= 1e-300:
        a = np.ones_like(z, dtype=np.float64)
    else:
        a = a / s
    return a


def _score_z_fft(z):
    return _score_a_fft(_softmax_from_z(z))


def _smooth_max_grad(z, focus):
    """
    Return gradient for a softmax approximation of max(convolution).

    The gradient is centered (mean removed), which corresponds to enforcing
    the normalization constraint in the log-parameterization.
    """
    a = _softmax_from_z(z)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64), math.inf, a

    L = 2 * n - 1
    N = _fft_len(L)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:L]
    b = np.maximum(b, 0.0)

    maxb = float(np.max(b))
    score = 2.0 * float(n) * maxb
    if maxb <= 1e-300:
        return np.zeros(n, dtype=np.float64), score, a

    x = focus * (b - maxb) / maxb
    w = np.exp(x)
    ws = float(np.sum(w))
    if (not math.isfinite(ws)) or ws <= 0.0:
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = w / ws

    # grad_i = sum_j w_{i+j} a_j, a length-n cross-correlation slice.
    N2 = _fft_len(3 * n - 2)
    fw = np.fft.rfft(w, N2)
    fr = np.fft.rfft(a[::-1], N2)
    corr = np.fft.irfft(fw * fr, N2)
    grad = corr[n - 1:n + n - 1].copy()
    grad = np.maximum(grad, 0.0)
    grad = grad - float(np.mean(grad))
    return grad, score, a

    z0,

    if n == 0:

    z = np.clip(z, -30.0, 30.0)

    best_z = z.copy()
    best_score = _score_z_fft(z)
    if not math.isfinite(best_score):
        best_score = math.inf

    eta = max(0.01, float(eta))
    max_eta = 12.0
    stalled = 0
    duration = max(1e-6, float(deadline) - float(phase_start))

    it = 0
    while it < int(max_iters):
        now = time.monotonic()
        if now > deadline:

        it += 1

        progress = min(1.0, max(0.0, (now - phase_start) / duration))
        focus = float(focus_start) + (float(focus_end) - float(focus_start)) * progress

        grad, score_cur, _ = _smooth_max_grad(z, focus)
        if not math.isfinite(score_cur):

        ref = float(np.max(np.abs(grad)))
        if ref < 1e-14:
            break

        alpha = eta / ref
        accepted = False
        for _ in range(8):
            z_try = z - alpha * grad
            z_try = np.clip(z_try, -30.0, 30.0)
            z_try = z_try - float(np.mean(z_try))
            score_try = _score_z_fft(z_try)
            if math.isfinite(score_try) and score_try < score_cur:
                z = z_try
                score_cur = score_try
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            eta = min(max_eta, eta * 1.07)
        else:
            eta = max(0.01, eta * 0.65)
            if eta < 0.02 and stalled > 40:
                break

    return best_z, best_score


def _random_profile_search(n, count, max_harm, deadline, rng):
    """Randomized low-dimensional Fourier profile search."""
    max_harm = max(1, int(max_harm))
    best_a = np.ones(int(n), dtype=np.float64) / float(n)
    best_score = math.inf

    idx = np.arange(int(n), dtype=np.float64)
    twopi = 2.0 * np.pi / float(n)

    harmonics = []
    for m in range(1, max_harm + 1):
        arg = twopi * float(m) * idx
        harmonics.append(np.sin(arg))
        harmonics.append(np.cos(arg))

    for k in range(int(count)):
        if time.monotonic() > deadline:
            break

        params = np.asarray(rng.normal(0.0, 1.0, 2 * max_harm), dtype=np.float64)
        denom = float(np.sum(np.abs(params)))
        scale = (0.9 / denom) if denom > 1e-12 else 0.0

        h = np.zeros(int(n), dtype=np.float64)
        for m in range(max_harm):
            h += params[m] * harmonics[m]
            h += params[max_harm + m] * harmonics[max_harm + m]
        h *= scale

        if k % 5 == 4:
            a = np.exp(np.clip(2.0 * h, -3.0, 3.0))
        else:
            a = 1.0 + h
            if float(np.min(a)) <= 0.0:
                a = np.clip(a, 1e-12, None)

        a = _norm_a(a)
        sc = _score_a_fft(a)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _direct_score(seq):
    try:
        return float(evaluate_sequence(seq))
    except Exception:
        return math.inf


def _finalize_z_scales(z, best_seq, best_score):
    """Convert a log-parameterized vector back to a list and check direct score."""
    a = _softmax_from_z(z)
    if a.size == 0:
        return best_seq, best_score

    scales = (0.1, 1.0, 100.0)
    for scale in scales:
        arr = a * float(scale)
        arr = np.clip(arr, 0.0, 1000.0)
        if float(np.sum(arr)) < 0.01:
            continue
        seq = [float(x) for x in arr]
        sc = _direct_score(seq)
        if math.isfinite(sc) and sc < best_score:
            best_seq = seq
            best_score = sc

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a low-score nonnegative step sequence.

    Uses FFT-based exponentiated gradient optimization with randomized restarts
    and low-dimensional Fourier profile initializations.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    budget = max(0.0, float(budget_s))
    hard_deadline = start + max(0.0, budget - 5.0)

    target = 1.5027

    best_seq = [1.0]
    best_score = _direct_score(best_seq)
    if not math.isfinite(best_score):
        best_score = 2.0

    # If a reference sequence is present, a uniform sequence of the same length
    # is a cheap valid baseline.
    try:
        ref_len = len(height_sequence_1)
        if 1 <= int(ref_len) <= 5000:
            seq = [1.0] * int(ref_len)
            sc = _direct_score(seq)
            if math.isfinite(sc) and sc < best_score:
                best_seq = seq
                best_score = sc
    except Exception:
        pass

    print(f"initial direct score = {best_score:.12f}")

    # Try several sequence lengths.  Powers of two are efficient for FFT.
    plans = [
        (512, 0.05),
        (1024, 0.45),
        (2048, 0.35),
        (4096, 0.15),
    ]

    cumulative = 0.0
    for n, frac in plans:
        cumulative += max(0.0, float(frac))
        phase_deadline = min(hard_deadline, start + cumulative * budget - 3.0)
        if phase_deadline <= time.monotonic() + 10.0:
            continue

        # A short profile search gives better restarts than pure uniform starts.
        if n <= 512:
            profile_count = 20000
        elif n <= 1024:
            profile_count = 12000
        elif n <= 2048:
            profile_count = 6000
        else:
            profile_count = 2500

        profile_deadline = min(
            phase_deadline,
            time.monotonic() + 0.20 * max(1.0, phase_deadline - time.monotonic())
        )
        profile_a, profile_score = _random_profile_search(n, profile_count, 6, profile_deadline, rng)
        if math.isfinite(profile_score):
            print(f"n={n}, best FFT profile score={profile_score:.12f}")

        if math.isfinite(profile_score) and profile_score < best_score + 0.05:
            z_profile = np.log(np.maximum(profile_a, 1e-300)).astype(np.float64)
            best_seq, best_score = _finalize_z_scales(z_profile, best_seq, best_score)

        inits = []
        inits.append(np.zeros(n, dtype=np.float64))
        inits.append(np.log(np.maximum(profile_a, 1e-300)).astype(np.float64))

        for std in (0.1, 0.3, 0.7):
            inits.append(np.asarray(rng.normal(0.0, float(std), n), dtype=np.float64))

        remaining = max(0.0, phase_deadline - time.monotonic())
        per_init = max(5.0, remaining / max(1, len(inits)))

        for init_idx, z0 in enumerate(inits):
            init_start = time.monotonic()
            init_deadline = min(phase_deadline, init_start + per_init)
            if init_deadline <= init_start + 1.0:
                break

            z_best, fft_best = _optimize_z(
                z0,
                max_iters=1000000,
                eta=0.7,
                focus_start=3.0,
                focus_end=70.0,
                phase_start=init_start,
                deadline=init_deadline,
                rng=rng,
                stop_score=target - 0.01
            )

            if math.isfinite(fft_best):
                print(f"n={n}, init={init_idx}, best FFT score={fft_best:.12f}")

            best_seq, best_score = _finalize_z_scales(z_best, best_seq, best_score)
            if math.isfinite(best_score):
                print(f"n={n}, init={init_idx}, direct score={best_score:.12f}")

            if best_score < target - 0.005:
                # A valid improvement exists.  Continue only if there is time;
                # this tries to squeeze out a better value.
                if time.monotonic() > hard_deadline - 30.0:
                    break

    if not math.isfinite(best_score):
        best_seq = [1.0]
        best_score = 2.0

    print(f"final direct score = {best_score:.12f}, length = {len(best_seq)}")
    return [float(x) for x in best_seq]
```
