```python
import numpy as np
import time

def compute_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    # Use FFT for efficient convolution
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    return conv

def simplex_project(a):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a

# Better simplex projection (Euclidean):
def euclidean_project_simplex(a, target_sum=1.0):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    n = len(a)
    # Scale to target sum
    a = a.copy()
    # Use the standard simplex projection algorithm
    if target_sum <= 0:
        return np.zeros(n)
    a = a / target_sum  # normalize to sum=1 first
    # Now project onto {x >= 0, sum = 1}
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    result = np.maximum(a - theta, 0)
    return result * target_sum
```

```python
def lp_update(a, M, critical_lags, solver='scipy'):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for k in critical_lags, g >= 0."""
    from scipy.optimize import linprog
    n = len(a)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: for each critical lag k, sum_i a[i] * g[k-i] <= M
    # where the sum is over valid i
    A_ub = []
    for k in critical_lags:
        row = np.zeros(n)
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            # g index is k - i
            gj = k - i
            if 0 <= gj < n:
                row[gj] = a[i]
        A_ub.append(row)
    A_ub = np.array(A_ub)
    b_ub = np.full(len(critical_lags), M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
```

```python
def gradient_step(a, temperature=0.1):
    """One gradient descent step using softmax approximation to max."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax gradient
    exp_conv = np.exp(conv / temperature - np.max(conv) / temperature)
    weights = exp_conv / np.sum(exp_conv)
    # Gradient of sum(weights * conv) = sum(weights * grad(conv))
    # grad(conv)_k with respect to a is: 2 * shift(a)
    # So the full gradient is: sum_k weights[k] * 2 * shifted_a(k)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] > 1e-10:
            # Contribution to grad from lag k
            for j in range(max(0, k-n+1), min(n, k+1)):
                grad[j] += 2 * weights[k] * a[k-j]
    return grad
```
