def multiplicative_weights_descent(a, eta=0.1, lambda_smoothing=10, n_iter=1000):
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(0, a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score_direct(a)
    
    for it in range(n_iter):
        b = np.convolve(a, a)
        max_b = b.max()
        
        # Compute importance weights (smooth max approximation)
        # Use a "tilted" version: p_k proportional to exp(lambda * (b_k - max_b))
        # This focuses more on the peak
        p = np.exp(lambda_smoothing * (b - max_b) / max_b)
        p = p / p.sum()
        
        # Compute cost
        c = compute_cost(a, p)
        
        # Multiplicative update
        a = a * np.exp(-eta * c / c.max())
        a = np.maximum(0, a)
        a_sum = a.sum()
        if a_sum < 1e-10:
            break
        a = a / a_sum
        
        score = compute_score_direct(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
