def project_simplex(z, target=1.0):
    # find nu such that max(z_i - nu, 0) sum = target
    v = np.sort(z)[::-1]
    cdf = np.cumsum(v)
    idx = np.where(cdf - v * np.arange(1, len(v)+1) >= target)[0][0]
    nu = (cdf[idx-1] - target) / idx

def _positive_normalize(v):
    arr = np.asarray(v, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:

def _logsum_gradient(f, cv, M, F, L, focus, size):
    n=f.size
    if M <= 1e-15 or not np.isfinite(M):
        return np.zeros(n, dtype=np.float64)
    c = cv[:size]
    z = focus * (c / M - 1.0)
    z = np.clip(z, -700.0, 0.0)
    w = np.exp(z)
    s = float(np.sum(w))
    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros(size, dtype=np.float64)
        near = int(np.argmax(c))
        w[near] = 1.0
        s = 1.0
    w = w / s
    Fw = np.fft.rfft(w, L)
    h = np.fft.irfft(Fw * np.conj(F), L)

def _lp_direction(f, cv, M, top_k, time_limit):
    if not _HAVE_LINPROG:
        return None
    size = cv.size if cv.size else 2*f.size-1
    # cv may be full L; size = 2*n-1
    size = 2*int(f.size)-1
    if size <=0 or M <=0 or not np.isfinite(M):
       return None
    c = cv[:size]
    n=f.size
    if top_k >= size:
       active=np.arange(size)
    else:
       active=np.argpartition(c, -top_k)[-top_k:]
    A=np.zeros((len(active), n), dtype=np.float64)
    for r, kk in enumerate(active):
       k=int(kk)
       lo=max(0, k-n+1)
       hi=min(n, k+1)
       if hi > lo:
          fs=k-(hi-1)
          fe=k-lo+1
          if fs<0: fs=0
          if fe>n: fe=n
          if fe>fs:
             A[r, lo:hi] = f[fs:fe][::-1]
    b=np.full(len(active), M*(1.0+1e-5), dtype=np.float64)
    cobj=-np.ones(n, dtype=np.float64)
    # if n > 1, tie breaker
    if n > 1:
       cobj = cobj - 1e-7 * (np.linspace(0.0, 1.0, n, dtype=np.float64))
    bounds=np.zeros((n,2), dtype=np.float64)
    bounds[:,1] = 1.0
    try:
       res=linprog(cobj, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'time_limit': float(time_limit), 'presolve': True})
    except Exception:
       try:
          res=linprog(cobj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
       except Exception:
          return None
    if res is not None and getattr(res, 'success', False):
       g=np.asarray(res.x, dtype=np.float64)
       if g is None or g.size != n: return None
       g=np.where(np.isfinite(g), g, 0.0)
       g=np.clip(g, 0.0, 1.0)
       s=float(np.sum(g))
       if s > 1e-10:
          g=g/s
          return g
    return None

def _by_score(item):

def _positive_normalize(v):
    arr = np.asarray(v, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s < 1e-12:
        arr = np.ones(n, dtype=np.float64)
        return arr * (1.0 / n)

    arr = arr * (1.0 / s)
    return arr


def _next_fft_len(size):
    if size <= 0:
        return 1
    size = int(size)
    return 1 << (size - 1).bit_length()


def _convolve_and_fft(f, L):
    F = np.fft.rfft(f, L)
    c = np.fft.irfft(F * F, L)
    return c, F


def _score_from_cv(cv, f, n, size):
    c = cv[:size]
    M = float(np.max(c))
    S = float(np.sum(f))
    if not np.isfinite(M) or not np.isfinite(S) or S < 1e-10:
        return float("inf")
    return float((2.0 * n * M) / (S * S))


def _logsum_gradient(f, cv, M, F, L, focus, size):
    n = int(f.size)
    if M <= 1e-15 or not np.isfinite(M):
        return np.zeros(n, dtype=np.float64)

    c = cv[:size]
    z = focus * (c / M - 1.0)
    z = np.clip(z, -700.0, 0.0)
    w = np.exp(z)
    s = float(np.sum(w))

    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros(size, dtype=np.float64)
        near = int(np.argmax(c))
        w[near] = 1.0
        s = 1.0

    w = w / s
    Fw = np.fft.rfft(w, L)
    h = np.fft.irfft(Fw * np.conj(F), L)
    return 2.0 * h[:n]

    if not _HAVE_LINPROG:
        return None

    n = int(f.size)
    size = 2 * n - 1
    if size <= 0 or M <= 0.0 or not np.isfinite(M):
        return None

    c = cv[:size]
    if top_k >= size:
        active = np.arange(size)
    else:
        active = np.argpartition(c, -top_k)[-top_k:]
    active = np.asarray(active, dtype=np.int64)

    A = np.zeros((len(active), n), dtype=np.float64)
    for r, kk in enumerate(active):
        k = int(kk)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            fs = k - (hi - 1)
            fe = k - lo + 1
            if fs < 0:
                fs = 0
            if fe > n:
                fe = n
            if fe > fs:
                A[r, lo:hi] = f[fs:fe][::-1]

    b = np.full(len(active), M * (1.0 + 1e-5), dtype=np.float64)
    cobj = -np.ones(n, dtype=np.float64)
    if n > 1:

        if not np.isfinite(fs):
            out = np.ones(n, dtype=np.float64)
            if n > 0:
                out = out / float(n)
    except Exception:
        out = np.ones(n, dtype=np.float64)
        if n > 0:
            out = out / float(n)

    return [float(x) for x in out]
