    weights /= weights.sum()

import numpy as np
import time
from scipy.optimize import linprog

def compute_convolution(a):
    """Compute full self-convolution."""
    return np.convolve(a, a)

def compute_score(a):
    """Compute the objective function value."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def compute_gradient_weights(a, c, T):
    """Compute softmax weights for the smooth max approximation."""
    c_max = c.max()
    weights = np.exp((c - c_max) / T)
    weights /= weights.sum()
    return weights, c_max

def compute_gradient(a, weights, n):
    """Compute gradient of weighted sum of c[k] w.r.t. a.
    grad[j] = 2 * sum_k weights[k] * a[k-j]
    """
    a_flip = a[::-1]
    # Use FFT for efficient convolution
    size = 1
    while size < 3 * n:
        size *= 2
    W = np.fft.rfft(weights, size)
    AF = np.fft.rfft(a_flip, size)
    conv = np.fft.irfft(W * AF, size)
    grad = 2.0 * conv[n-1:2*n-1]
    return grad

def smooth_gradient_descent(a, n, deadline, T_init, lr_init, max_steps):
    """Run smooth gradient descent to minimize the objective."""
    T = T_init
    lr = lr_init
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_smooth = np.inf
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        c_max = c.max()
        
        # Track best (using exact max)
        exact_score = c_max  # since sum(a) = 1, score = 2n * c_max
        if exact_score < best_smooth:
            best_smooth = exact_score
            best_a = a.copy()
        
        # Compute weights and gradient
        weights, _ = compute_gradient_weights(a, c, T)
        grad = compute_gradient(a, weights, n)
        
        # Gradient step with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Adaptive learning rate
        c_new = np.convolve(a_new, a_new)
        if c_new.max() < c_max:
            lr = min(lr * 1.1, 1.0)
        else:
            lr *= 0.8
            a_new = a  # reject step
            if lr < 1e-12:
                break
        
        a = a_new
        T = max(T * 0.9995, 1.0 / (4 * n))

        threshold_idx = np.argsort(c)[-n_active:]

        A_eq[0, -1] = 0  # M' doesn't appear in equality

            break
        
        if it % 10 == 0:
            print(f"    LP iter {it}: score={2*n*best_score:.6f}")
    
    return best_a
