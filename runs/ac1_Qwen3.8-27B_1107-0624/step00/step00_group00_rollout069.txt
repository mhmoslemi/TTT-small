Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

def project_simplex(v, S):
    """Project v onto {x : x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_max_fft(a):
    """Fast max of linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)
    return float(np.max(conv[:2 * n - 1]))


def obj_fft(a):
    """Compute objective using FFT convolution."""
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = conv_max_fft(a)
    return 2.0 * n * M / (S * S)


def obj_exact(a):
    """Compute objective using exact convolution."""
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)

def build_reduced_A(f, top_k_indices, n):
    """Build sparse constraint matrix for reduced LP."""
    rows = []
    cols = []
    vals = []
    row_map = {}
    for idx, k in enumerate(top_k_indices):
        row_map[k] = idx
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        count = j_hi - j_lo + 1
        if count > 0:
            j_range = np.arange(j_lo, j_hi + 1)
            rows.extend([idx] * count)
            cols.extend(j_range.tolist())
            vals.extend(f[k - j_range].tolist())
    A = sparse.coo_matrix((vals, (rows, cols)), shape=(len(top_k_indices), n)).tocsr()
    return A


def lp_direction(f, n, top_k_count, deadline_lp):
    """Solve reduced LP to find a promising direction g."""
    conv_f = np.convolve(f, f)
    # Select top positions
    top_k_idx = np.argsort(conv_f)[::-1][:top_k_count]
    M = float(conv_f[top_k_idx[0]])

    A = build_reduced_A(f, top_k_idx, n)
    b = np.full(len(top_k_idx), M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': max(1, deadline_lp)})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None


def subgradient_step(f, n, eta):
    """One subgradient step for minimizing max(conv(f,f)) with sum(f)=n constraint."""
    conv_f = np.convolve(f, f)
    M = float(np.max(conv_f))
    # Active set: positions where conv_f is within 5% of max
    threshold = M * 0.95
    active = np.where(conv_f >= threshold)[0]

    if len(active) == 0:
        return f.copy()

    # Subgradient: g_j = (1/|S*|) * sum_{k in S*} 2 * a_{k-j} (valid indices)
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        g[j_lo:j_hi + 1] += 2.0 * f[k - np.arange(j_lo, j_hi + 1)]
    g /= len(active)

    # Update and project
    f_new = f - eta * g
    f_new = project_simplex(f_new, n)
    return f_new
