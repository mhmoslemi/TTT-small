```python
import numpy as np
from scipy.optimize import linprog
import time

def solve_lp(f, M):
    """Solve the LP: max sum(b) s.t. b >= 0, conv(f,b)_k <= M for all k."""
    n = len(f)
    # Variables: b_0, ..., b_{n-1}
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: conv(f, b)_k <= M for k = 0, ..., 2n-2
    # conv(f, b)_k = sum_{i} f[i] * b[k-i] where 0 <= i < n and 0 <= k-i < n
    # For a given k, the constraint is: sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * b[k-i] <= M
    
    # Build the constraint matrix A (shape: (2n-1, n))
    # A[k, j] = f[i] where i = k - j, if 0 <= i < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    
    b_constraints = np.full(2*n-1, M)
    
    # Solve
    result = linprog(c, A_ub=A, b_ub=b_constraints, bounds=[(0, None)]*n, method='highs')
    if result.success:
        return result.x
    else:
        return None
```

```python
        rows.append((i + j_vals))  # rows: i, i+1, ..., i+n-1
```

```python
"""
LP-based iterative optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses the central binomial coefficient sequence as initial point and
LP-guided direction finding for iterative improvement.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def build_conv_matrix(f):
    """Build sparse convolution constraint matrix.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0. Shape: (2n-1, n).
    """
    n = len(f)
    rows = np.concatenate([np.arange(i, i + n - i) for i in range(n)])
    cols = np.concatenate([np.arange(n - i) for i in range(n)])
    data = np.concatenate([np.full(n - i, f[i]) for i in range(n)])
    A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsr()

def flat_sequence(n):
    """Generate the flat-autocorrelation sequence: a[k] = C(2k,k)/4^k."""
    a = np.zeros(n)
    for k in range(n):
        # C(2k, k) / 4^k, computed iteratively to avoid overflow
        if k == 0:
            a[k] = 1.0
        else:
            a[k] = a[k-1] * (2*k - 1) / (2 * k)
    return a

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    # Try multiple starting points
    start_sequences = []
    
    # 1. Flat autocorrelation sequence
    for n in [2048, 4096, 8192, 4498]:
        start_sequences.append((n, flat_sequence(n)))
    
    # 2. The given base sequence
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0, 1000)
        if base.size > 0 and np.sum(base) > 0.01:
            start_sequences.append((len(base), base.copy()))
    
    best_overall = None
    best_score_overall = np.inf
    
    for n, seq in start_sequences:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        
        # Normalize
        S = np.sum(seq)
        if S < 0.01:
            continue
        target_sum = np.sqrt(2.0 * n)
        f = seq * (target_sum / S)
        f = np.clip(f, 0.0, 1000.0)
        
        # Evaluate
        score = float(evaluate_sequence(f.tolist()))
        if not np.isfinite(score):
            continue
        
        best = f.copy()
        best_score = score
        print(f"n={n}, initial score: {best_score:.6f}")
        
        # LP-based optimization
        iteration = 0
        while time.monotonic() < min(deadline, time.monotonic() + remaining * 0.7):
            iteration += 1
            M = np.max(np.convolve(f, f))
            
            # Solve LP
            A = build_conv_matrix(f)
            c = -np.ones(n)
            b_ub = np.full(2 * n - 1, M)
            result = linprog(c, A_ub=A, b_ub=b_ub, 
                           bounds=[(0, None)] * n, method='highs')
            if not result.success:
                break
            
            b = np.clip(result.x, 0, None)
            b_sum = np.sum(b)
            if b_sum < 1e-10:
                break
            
            # Scale b to target sum
            b_scaled = b * (target_sum / b_sum)
            b_scaled = np.clip(b_scaled, 0.0, 1000.0)
            
            # Line search
            improved = False
            for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
                f_new = (1 - t) * f + t * b_scaled
                f_new = np.clip(f_new, 0.0, 1000.0)

                if np.isfinite(s) and s < best_score:
                    best = f_new.copy()
                    best_score = s
                    f = f_new
                    improved = True
                    break
            
            if not improved:
                # Accept the direction anyway (small step) to explore
                f = 0.95 * f + 0.05 * b_scaled
                f = np.clip(f, 0.0, 1000.0)
            
            if iteration % 5 == 0:
                elapsed = time.monotonic() - (deadline - remaining / 0.7)  # approximate
                print(f"  iter {iteration}, best: {best_score:.6f}")
        
        if best_score < best_score_overall:
            best_score_overall = best_score
            best_overall = best.copy()
            print(f"New overall best: {best_score_overall:.6f} (n={n})")
    
    if best_overall is None:
        # Fallback
        best_overall = flat_sequence(2048)
        best_score_overall = float(evaluate_sequence(best_overall.tolist()))
    
    print(f"FINAL: score = {best_score_overall:.6f}")
    return [float(x) for x in best_overall]
```
