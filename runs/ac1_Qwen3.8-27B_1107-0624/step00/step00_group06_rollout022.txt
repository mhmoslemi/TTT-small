import numpy as np
from scipy.sparse import csc_matrix
from scipy.optimize import linprog
import time

def build_constraint_matrix(f):
    """Build the constraint matrix A where A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def propose_candidate(budget_s=1800):
    # Start with geometric sequence
    n = 5582  # or some other length
    epsilon = 0.0002
    r = 1 + epsilon
    a = np.array([r**j for j in range(n)])
    a /= a.sum()  # normalize to sum=1
    
    best_a = a.copy()
    best_obj = 2*n*np.max(np.convolve(a,a))/np.sum(a)**2
    
    deadline = time.monotonic() + budget_s - 10
    
    t = 0.5  # mixing parameter
    
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = np.max(conv)
        
        # Build constraint matrix
        A = build_constraint_matrix(a)
        
        # Solve LP: maximize sum(g) subject to A@g <= M, g >= 0
        c = -np.ones(n)  # minimize -sum(g)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        if g.sum() < 1e-10:
            break
        
        g /= g.sum()  # normalize to sum=1
        
        # Update
        a = (1-t)*a + t*g
        
        # Evaluate
        conv_new = np.convolve(a, a)
        obj_new = 2*n*np.max(conv_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
    
    # Scale best_a to have sum = sqrt(2n) for the evaluator
    # Actually, the evaluator is homogeneous, so we just return best_a as-is
    # (the evaluator will handle the normalization through the formula)
    return best_a.tolist()
