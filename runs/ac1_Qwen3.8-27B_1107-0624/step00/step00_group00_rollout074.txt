def compute_convolution(f):
    """Compute full convolution using FFT for speed."""
    n = len(f)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = np.fft.rfft(f, n=fft_size)
    conv = np.fft.irfft(F * F, n=fft_size)[:size]
    # Small numerical errors possible; clip
    conv = np.maximum(conv, 0)
    return conv

def obj_value(f):

def get_subgradient(f, conv, peak_thresh=0.95):

    Compute a subgradient of max(conv) w.r.t. f.
    Uses peaks above peak_thresh * max.
    Returns subgradient array of same length as f.
    """
    n = len(f)

    if num_peaks == 0:
        return np.zeros(n)

    # 2*f(i) if k == 2*i
    # 2*f(k-i) if 0 <= k-i < n and k != 2*i

Multi-strategy optimizer for minimizing 2*n*max(conv)/sum^2 over non-negative sequences.
Uses diverse initialization, convolution-flattening gradient steps, mass transfer,

def compute_convolution(f):
    """Compute full convolution using FFT for speed."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    F = np.fft.rfft(f, n=fft_size)
    conv = np.fft.irfft(F * F, n=fft_size)[:size]
    return np.maximum(conv, 0.0)


def get_gradient_of_max_conv(f, conv):
    """Compute subgradient of max(conv) with respect to f, using top peaks."""
    n = len(f)
    max_c = conv.max()
    if max_c <= 0:
        return np.zeros(n)
    # Use peaks in top 5%
    thresh = 0.95 * max_c
    peak_mask = conv >= thresh
    peak_indices = np.where(peak_mask)[0]
    num_peaks = len(peak_indices)
    if num_peaks == 0:
        peak_indices = np.array([int(np.argmax(conv))])
        num_peaks = 1
    subgrad = np.zeros(n)
    for k in peak_indices:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        i_range = np.arange(i_lo, i_hi + 1)
        k_minus_i = k - i_range
        subgrad[i_range] += 2.0 * f[k_minus_i] / num_peaks
    return subgrad


def flatten_convolution_step(f, lr):
    """One step of convolution flattening: reduce peaks, boost valleys."""
    n = len(f)
    conv = compute_convolution(f)
    S = f.sum()
    if S < 1e-10:
        return f
    mean_c = S * S / (2 * n - 1)
    # Excess: positive where peaks, negative where valleys
    excess = conv - mean_c
    # Gradient of sum of squared excesses (proxy for max)
    # d(sum_k excess(k)^2)/d f(i) = 2 * sum_k excess(k) * dc(k)/df(i)
    grad = np.zeros(n)
    for k in range(len(conv)):
        e = excess[k]
        if abs(e) < 1e-12:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        i_range = np.arange(i_lo, i_hi + 1)
        k_minus_i = k - i_range
        grad[i_range] += 2.0 * e * f[k_minus_i]
    # Update

    f_new = np.maximum(f_new, 0.0)
    # Renormalize
    S_new = f_new.sum()
    if S_new > 1e-10:
        f_new = f_new * (S / S_new)
    return f_new

def mass_transfer_optimize(f, budget_s, rng):
    """Optimize via targeted mass transfer: reduce peak contributors, boost random positions."""
    n = len(f)
    f = f.copy().astype(np.float64)
    S = f.sum()
    if S < 1e-10:
        return f, float('inf')
    
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    deadline = time.monotonic() + budget_s
    
    while time.monotonic() < deadline:
        conv = compute_convolution(f)
        peak_k = int(np.argmax(conv))
        
        # Find the pair with max contribution to the peak
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        if i_lo > i_hi:
            break
        
        i_range = np.arange(i_lo, i_hi + 1)
        contributions = f[i_range] * f[peak_k - i_range]
        max_local = int(np.argmax(contributions))
        src = i_lo + max_local
        dst = peak_k - src
        
        # Transfer mass from the peak-contributing pair
        if src != dst:
            transfer = 0.1 * min(f[src], f[dst])
        else:
            transfer = 0.1 * f[src]
        
        if transfer < 1e-12:
            break
        
        f[src] = max(0.0, f[src] - transfer)
        if src != dst:
            f[dst] = max(0.0, f[dst] - transfer)
        
        # Redistribute to random positions (avoiding peak area)
        num_targets = 2
        for _ in range(num_targets):
            t = int(rng.integers(0, n))
            f[t] += transfer / num_targets
        
        f = np.maximum(f, 0)
        # Renormalize
        S_new = f.sum()
        if S_new < 1e-10:
            f = best.copy()
            break

        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f.copy()
            best_score = score

def block_perturb_optimize(f, budget_s, rng):

    n = len(f)

            end = min(n, start + block_size)

            if strategy < 0.3:

            elif strategy < 0.7:

                ramp_min = rng.uniform(0.1, 0.5)
                candidate[start:end] *= np.linspace(ramp_min, 1.0, end - start)
            elif strategy < 0.85:

            else:

        candidate = np.maximum(candidate, 0)

def generate_diverse_starts(n, rng):

    starts.append(u)
    
    # 4. Inverted U (bell)
    bell = np.exp(-4.0 * (np.arange(n) / n - 0.5) ** 2)
    starts.append(0.3 + bell)

    # 6. Two bumps

    # 7. Smoothed random
    r = np.abs(rng.normal(1.0, 0.5, n))
    r = uniform_filter1d(r, size=max(2, n // 100))

    # 8. High-frequency oscillation (non-negative)

    pl = (1.0 - np.arange(n) / n) ** 2 + 0.1

    # 10. Asymmetric: high at very edges, low in middle
    edges = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float) / (n / 2)
