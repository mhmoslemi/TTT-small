"""
Algorithm: Multiplicative weight (mirror descent) optimization to minimize
2n*max(conv(a,a))/sum(a)^2 for non-negative sequences a.
Uses subgradient of the peak convolution value with multiplicative updates,
combined with local search and multiple restarts from different initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _fast_eval(a):
    """Fast objective evaluation for a non-negative numpy array."""
    n = len(a)
    if n == 0:
        return np.inf
    if a.sum() < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_c = c.max()
    s = a.sum()
    return 2.0 * n * max_c / (s * s)


def _compute_gradient(a, top_indices):
    """Compute subgradient of max convolution at specified lags."""
    n = len(a)
    g = np.zeros(n)
    for k in top_indices:
        # Part 1: d(conv(a,a)[k])/da[i] from first factor = a[k-i] if 0<=k-i<n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += a[k - lo: k - hi + 1][::-1]
        # Part 2: from second factor = a[i-k] if 0<=i-k<n
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] += a[0: hi2 - lo2 + 1]
    return g


def _multiplicative_step(a, eta, top_k):
    """One multiplicative weight update step. Returns updated a."""
    n = len(a)
    c = np.convolve(a, a)
    if top_k < 2 * n - 1:
        top_idx = np.argpartition(-c, min(top_k, 2*n-1)-1)[:top_k]
    else:
        top_idx = np.arange(2 * n - 1)
    g = _compute_gradient(a, top_idx)
    g_max = g.max()
    if g_max <= 0:
        return a, c.max()
    a_new = a * np.exp(-eta * g / g_max)
    a_new = np.maximum(a_new, 1e-15)
    target_sum = np.sqrt(2.0 * n)
    a_new = a_new * (target_sum / a_new.sum())
    return a_new, c.max()


def _lp_step(a, top_k=20, max_iter_lp=1):
    """Solve LP to find direction g maximizing sum subject to peak constraints."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    target_sum = np.sqrt(2.0 * n)
    
    # Get top K lags
    K = min(top_k, 2 * n - 1)
    top_idx = np.argpartition(-c, K - 1)[:K]
    
    # Build constraint matrix for these K lags
    # For lag k: sum_i a[i]*g[k-i] <= M (where k-i is valid)
    # Row for lag k: B[k, i] = a[i] if 0 <= k-i < n (i.e., a[k-i] is the coefficient of g[i])
    # Wait: conv(a, g)[k] = sum_i a[i] * g[k-i]
    # So coefficient of g[j] in row k is a[k-j] if 0 <= k-j < n, i.e., a[k-j]
    
    A_ub = np.zeros((K, n))
    for row, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A_ub[row, lo:hi+1] = a[k - lo: k - hi + 1][::-1]
    
    # LP: max sum(g) s.t. A_ub @ g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = -res.x
            g_sum = g.sum()
            if g_sum > 0:
                g = g * (target_sum / g_sum)
                return g
    except Exception:
        pass
    return None

        budgets = [0.6 * total_budget, 0.2 * total_budget, 0.2 * total_budget]
    
    best_overall_score = np.inf
    best_overall_seq = np.ones(n_parent)
    
    for i, n in enumerate(lengths_to_try):
        remaining = total_budget - (time.monotonic() - start_time)
        if remaining < 30:
            break
        alloc = min(budgets[i], remaining - 10)
        if alloc < 15:
            break
        
        print(f"Trying length {n}, budget {alloc:.1f}s")
        a, score = _try_length(n, alloc, rng)
        # Verify with actual evaluator
        actual_score = float(evaluate_sequence(a.tolist()))
        print(f"  Length {n}: internal={score:.6f}, actual={actual_score:.6f}")
        
        if np.isfinite(actual_score) and actual_score < best_overall_score:
            best_overall_score = actual_score
            best_overall_seq = a.copy()
    
    # Final local search on best
    remaining = total_budget - (time.monotonic() - start_time)
    if remaining > 10:
        print(f"Final local search with {remaining:.1f}s remaining")
        a_fin, s_fin = _local_search(best_overall_seq, n_iter=min(2000, int(remaining * 50)), rng=rng)
        actual_fin = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(actual_fin) and actual_fin < best_overall_score:
            best_overall_score = actual_fin
            best_overall_seq = a_fin
            print(f"  Improved to {actual_fin:.6f}")
    
    # One final verification
    final_score = float(evaluate_sequence(best_overall_seq.tolist()))
    print(f"Final score: {final_score:.6f}")
    
    return [float(x) for x in best_overall_seq]
