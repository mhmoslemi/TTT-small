```
def _active_gradient(f, c, maxc, k_top=24):
    n=f.size
    order=np.argsort(c)[::-1][:k_top]
    wk=c[order]
    # weights softmax with temp 0.05*maxc
    temp = max(1e-300, 0.05*maxc)
    ew=np.exp((wk - wk.max()) / temp)
    w=ew/ew.sum()
    g=np.zeros(n)
    for k, wgt in zip(order, w):
       if wgt ==0: continue
       i0=max(0, k - n + 1)
       i1=min(k, n - 1)
       j0=k - i1
       j1=k - i0
       # f[j0:j1+1][::-1]
       g[i0:i1+1] += wgt * f[j0:j1+1][::-1]
    return g
```

```python
"""
Search for nonnegative step-function coefficients with very low normalized
autocorrelation energy.

Algorithm:
1. Use a smooth positive parametrization p = softmax(F theta), where F contains
   low-order polynomial, trigonometric, and endpoint features.
2. Optimize theta with L-BFGS under multiple temperatures of a log-sum-exp
   smoothing of max(conv(p,p)). This finds globally useful asymmetric shapes.
3. Improve the best long sequence with top-lag projected gradient polishing and
   occasional random block perturbations.
4. Return the best exact sequence found.
"""

import numpy as np
import time
import scipy.optimize as opt


def _next_pow2(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _fft_square(x: np.ndarray) -> np.ndarray:
    n = x.size
    if n == 0:
        return np.array([], dtype=np.float64)
    x = np.asarray(x, dtype=np.float64)
    L = _next_pow2(2 * n - 1)
    X = np.fft.rfft(x, L)
    c = np.fft.irfft(X * X, L)[:2 * n - 1]
    c = np.where(c < 0.0, 0.0, c)
    c = np.where(np.isfinite(c), c, 0.0)
    return c


def _prob_grad(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of sum_k w_k sum_i p_i p_{k-i} with respect to p.
    p has length n, w has length 2n-1.
    """
    n = p.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = _next_pow2(3 * n - 2)
    P = np.fft.rfft(p, L)
    Wr = np.fft.rfft(w[::-1], L)
    conv = np.fft.irfft(P * Wr, L)
    g = conv[n - 1:2 * n - 1][::-1]
    g = np.where(np.isfinite(g), g, 0.0)
    return 2.0 * g


def _density_from_theta(F: np.ndarray, theta: np.ndarray) -> np.ndarray:
    theta = np.asarray(theta, dtype=np.float64)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    eta = F @ theta
    eta -= np.max(eta)
    p = np.exp(eta)
    s = p.sum()
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones(F.shape[0], dtype=np.float64)
        s = float(p.size)
    p /= s
    p = np.where(np.isfinite(p), p, 0.0)
    s = p.sum()
    if s <= 0.0:
        p = np.ones(F.shape[0], dtype=np.float64)
        s = float(p.size)
    p /= s
    return p


def _theta_objective(theta, F: np.ndarray, tau: float):
    theta = np.asarray(theta, dtype=np.float64)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    n = F.shape[0]
    p = _density_from_theta(F, theta)
    c = _fft_square(p)
    m = float(np.max(c))
    if not np.isfinite(m) or m <= 0.0:
        return 1.0e300, np.zeros_like(theta, dtype=np.float64)

    if not np.isfinite(tau) or tau <= 0.0:
        tau = 1.0 / max(1, n)

    u = c - m
    e = np.exp(u / tau)
    se = float(np.sum(e))
    if not np.isfinite(se) or se <= 0.0:
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        val = 2.0 * n * m
    else:
        w = e / se
        val = 2.0 * n * (m / tau + np.log(se))

    gp = _prob_grad(p, w)
    gphi = gp * p
    fxi = np.dot(F, p)
    xgi = float(np.dot(p, gphi))
    grad = np.dot(F.T, gphi) - fxi * xgi
    grad = np.where(np.isfinite(grad), grad, 0.0)

    if not np.isfinite(val):
        return 1.0e300, np.zeros_like(theta, dtype=np.float64)
    return float(val), grad


def _build_features(n: int, n_cheb: int = 6, n_trig: int = 12) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    u = 2.0 * x - 1.0
    cols = []

    if n_cheb > 0:
        Tprev = np.ones(n, dtype=np.float64)
        Tcurr = u.copy()
        cols.append(Tcurr)
        for d in range(2, n_cheb + 1):
            Tnext = 2.0 * u * Tcurr - Tprev
            Tprev, Tcurr = Tcurr, Tnext
            cols.append(Tcurr)

    for mm in range(1, n_trig + 1):
        cols.append(np.sin(2.0 * np.pi * mm * x))
        cols.append(np.cos(2.0 * np.pi * mm * x))

    sx = np.sqrt(np.clip(x, 0.0, 1.0))
    sofr = np.sqrt(np.clip(1.0 - x, 0.0, 1.0))
    cols.append(sx)
    cols.append(sofr)
    cols.append(x * sx)
    cols.append((1.0 - x) * sofr)

    F = np.column_stack(cols).astype(np.float64, copy=False)
    F -= F.mean(axis=0, keepdims=True)
    std = F.std(axis=0, keepdims=True)
    std = np.where(std < 1.0e-12, 1.0, std)
    F /= std
    F = np.where(np.isfinite(F), F, 0.0)
    return F


def _local_score(p: np.ndarray) -> float:
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return np.inf
    p = np.where(p < 0.0, 0.0, p)
    s = p.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.inf
    p = p / s
    c = _fft_square(p)
    m = float(np.max(c))
    sc = 2.0 * p.size * m
    if not np.isfinite(sc):
        return np.inf
    return float(sc)


def _make_target_p(n: int, kind: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    if kind == 0:
        p = np.ones(n, dtype=np.float64)
    elif kind == 1:
        p = 1.0 + 0.8 * x
    elif kind == 2:
        p = 1.0 - 0.8 * x
    elif kind == 3:
        p = np.where(x < 0.5, 0.7, 1.3)
    elif kind == 4:
        p = np.where(x < 0.5, 1.4, 0.6)
    elif kind == 5:
        p = np.exp(1.2 * x)
    elif kind == 6:
        p = np.exp(-1.2 * x)
    elif kind == 7:
        p = np.maximum(1.0 + 0.55 * np.sin(2.0 * np.pi * x), 1.0e-6)
    elif kind == 8:
        p = np.maximum(1.0 + 0.55 * np.sin(4.0 * np.pi * x), 1.0e-6)
    elif kind == 9:
        p = np.maximum(1.0 + 0.40 * np.cos(2.0 * np.pi * x)
                        - 0.35 * np.sin(2.0 * np.pi * x), 1.0e-6)
    elif kind == 10:
        p = np.maximum(1.0 - 0.40 * np.cos(4.0 * np.pi * x)
                        + 0.35 * np.sin(4.0 * np.pi * x), 1.0e-6)
    elif kind == 11:
        p = np.maximum(1.0 + 0.60 * (2.0 * x - 1.0) ** 3, 1.0e-6)
    elif kind == 12:
        p = np.maximum(1.0 + 0.45 * np.cos(3.0 * np.pi * x)
                        + 0.25 * np.sin(5.0 * np.pi * x), 1.0e-6)
    elif kind == 13:
        p = np.maximum(np.exp(0.8 * x) * (1.0 + 0.35 * np.sin(6.0 * np.pi * x)), 1.0e-6)
    else:
        p = np.ones(n, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 1.0e-6)
    p = np.maximum(p, 1.0e-6)
    s = p.sum()
    if s <= 0.0:
        p = np.ones(n, dtype=np.float64)
        s = float(n)
    p = p / s
    return p


def _fit_theta(F: np.ndarray, p: np.ndarray) -> np.ndarray:
    eta = np.log(np.maximum(p, 1.0e-12))
    eta -= eta.mean()
    try:
        coeff = np.linalg.lstsq(F, eta, rcond=None)[0]
    except Exception:
        coeff = np.zeros(F.shape[1], dtype=np.float64)
    coeff = np.where(np.isfinite(coeff), coeff, 0.0)
    norm = float(np.linalg.norm(coeff))
    if norm > 4.0:
        coeff *= 4.0 / norm
    return coeff


def _make_starts(F: np.ndarray, n: int, rng) -> list:
    K = F.shape[1]
    starts = []
    starts.append(np.zeros(K, dtype=np.float64))

    for sigma in (0.3, 0.6, 1.0, 1.5, 2.0, 2.6, 3.2):
        th = rng.standard_normal(K) * sigma
        th = np.clip(th, -5.0, 5.0)
        starts.append(th)

    for kind in range(14):
        p = _make_target_p(n, kind)
        th = _fit_theta(F, p)
        th = np.clip(th, -4.5, 4.5)
        starts.append(th)

    # A few random shaped targets encoded directly in theta space.
    for sigma in (1.0, 1.8, 2.6):
        th = rng.standard_normal(K) * sigma
        th = np.clip(th, -5.0, 5.0)
        starts.append(th)

    return starts[:28]


def _optimize_theta(theta0, F: np.ndarray, tau: float, maxiter: int, maxfun: int) -> np.ndarray:
    theta0 = np.asarray(theta0, dtype=np.float64).copy()
    theta0 = np.where(np.isfinite(theta0), theta0, 0.0)
    theta0 = np.clip(theta0, -6.0, 6.0)
    K = F.shape[1]
    bounds = [(-6.0, 6.0)] * K

    try:
        res = opt.minimize(
            _theta_objective,
            theta0,
            args=(F, tau),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxiter": maxiter,
                "maxfun": maxfun,
                "ftol": 1.0e-12,
                "gtol": 1.0e-7,
            },
        )
        th = np.asarray(res.x, dtype=np.float64)
    except Exception:
        th = theta0

    th = np.where(np.isfinite(th), th, 0.0)
    return np.clip(th, -6.0, 6.0)


def _process_theta(theta, F: np.ndarray, n: int, state) -> None:
    p = _density_from_theta(F, theta)
    sc = _local_score(p)
    if not np.isfinite(sc):
        return

    state["idx"] = int(state.get("idx", 0)) + 1
    idx = int(state["idx"])

    if sc < state["best_local"]:
        state["best_local"] = float(sc)
        state["best_particle"] = p.copy()
        state["best_theta"] = theta.copy()

    state["candidates"].append((float(sc), idx, p.copy(), theta.copy()))
    state["candidates"].sort()
    if len(state["candidates"]) > 36:
        del state["candidates"][24:]

    need_exact = (idx % 20 == 0)
    if np.isfinite(state["best_exact"]) and sc < state["best_exact"] - 1.0e-7:
        need_exact = True

    if need_exact:
        try:
            ex = evaluate_sequence(p.tolist())
        except Exception:
            ex = np.inf
        if np.isfinite(ex) and ex < state["best_exact"]:
            state["best_exact"] = float(ex)
            state["best_exact_particle"] = p.copy()


def _mix_candidates(cands, max_cands: int = 6, alphas=(0.0, 0.25, 0.5, 0.75, 1.0)):
    if len(cands) == 0:
        return None, np.inf
    plist = []
    for item in cands[:max_cands]:
        p = np.asarray(item[2], dtype=np.float64)
        s = p.sum()
        if np.isfinite(s) and s > 0.0:
            plist.append(p / s)
    if len(plist) == 0:
        return None, np.inf

    best = None
    best_score = np.inf
    k = len(plist)
    for i in range(k):
        for j in range(i, k):
            p1 = plist[i]
            p2 = plist[j]
            for a in alphas:
                p = a * p1 + (1.0 - a) * p2
                p = np.where(p < 0.0, 0.0, p)
                s = p.sum()
                if not np.isfinite(s) or s <= 0.0:
                    continue
                p = p / s
                sc = _local_score(p)
                if np.isfinite(sc) and sc < best_score:
                    best_score = float(sc)
                    best = p.copy()
    return best, best_score


def _polish_sequence(f: np.ndarray, budget_s: float, best_local: float, best_exact: float):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return None, np.inf, None, np.inf

    f = np.where(f < 0.0, 0.0, f)
    f = np.where(np.isfinite(f), f, 0.0)
    s = f.sum()
    if not np.isfinite(s) or s <= 0.0:
        f = np.ones(n, dtype=np.float64) / float(n)
    else:
        f /= s

    best_local_seq = f.copy()
    best_local_score = float(best_local) if np.isfinite(best_local) else _local_score(f)
    if not np.isfinite(best_local_score):
        best_local_score = 1.0e300

    best_exact_seq = None
    best_exact_score = float(best_exact) if np.isfinite(best_exact) else np.inf

    budget_s = max(0.0, float(budget_s))
    if budget_s < 5.0:
        return best_local_seq, best_local_score, best_exact_seq, best_exact_score

    deadline = time.monotonic() + max(0.0, budget_s - 2.0)
    rng = np.random.default_rng(20240618)

    it = 0
    stagnation = 0
    last_exact_update = 0

    while time.monotonic() < deadline - 1.0:
        c = _fft_square(f)
        maxc = float(np.max(c))
        if not np.isfinite(maxc) or maxc <= 0.0:
            break
        cur = 2.0 * n * maxc

        if cur < best_local_score:
            best_local_score = float(cur)
            best_local_seq = f.copy()
            stagnation = 0
        else:
            stagnation += 1

        if it % 160 == 0 or (np.isfinite(best_exact_score) and cur < best_exact_score - 1.0e-7):
            try:
                ex = evaluate_sequence(f.tolist())
            except Exception:
                ex = np.inf
            if np.isfinite(ex) and ex < best_exact_score:
                best_exact_score = float(ex)
                best_exact_seq = f.copy()
                last_exact_update = it

        if stagnation > 180:
            f = best_local_seq * (1.0 + 0.07 * rng.standard_normal(n))
            f = np.where(f < 0.0, 0.0, f)
            s = f.sum()
            if np.isfinite(s) and s > 0.0:
                f /= s
            else:
                f = best_local_seq.copy()
            stagnation = 0

        topk = min(28, max(10, n // 200))
        order = np.argsort(c)[::-1][:topk]
        wk = c[order]
        temp = max(1.0e-300, 0.05 * maxc)
        ew = np.exp((wk - np.max(wk)) / temp)
        se = float(np.sum(ew))
        if np.isfinite(se) and se > 0.0:
            wvals = ew / se
        else:
            wvals = np.ones(topk, dtype=np.float64) / float(topk)

        g = np.zeros(n, dtype=np.float64)
        for ii in range(topk):
            k = int(order[ii])
            wgt = float(wvals[ii])
            if wgt <= 0.0:
                continue
            i0 = max(0, k - n + 1)
            i1 = min(k, n - 1)
            if i1 < i0:
                continue
            j0 = k - i1
            j1 = k - i0
            g[i0:i1 + 1] += wgt * f[j0:j1 + 1][::-1]

        d = -g
        d -= d.mean()
        dn = float(np.linalg.norm(d))
        if not np.isfinite(dn) or dn <= 1.0e-20:
            f_new = f * (1.0 + 0.04 * rng.standard_normal(n))
            f_new = np.where(f_new < 0.0, 0.0, f_new)
            s = f_new.sum()
            if np.isfinite(s) and s > 0.0:
                f = f_new / s
            it += 1
            continue

        max_t = 1.0
        neg = d < 0.0
        if np.any(neg):
            ratios = f[neg] / (-d[neg])
            ratios = ratios[np.isfinite(ratios)]
            if ratios.size > 0:
                mt = float(np.min(ratios))
                if mt >= 0.0:
                    max_t = min(max_t, mt)

        base = 0.25 * max_t
        accepted = False
        for frac in (1.0, 0.5, 0.25, 0.125):
            t = base * frac
            if t <= 1.0e-16:
                continue
            f_new = f + t * d
            f_new = np.where(f_new < 0.0, 0.0, f_new)
            s = f_new.sum()
            if not np.isfinite(s) or s <= 0.0:
                continue
            f_new = f_new / s
            c_new = _fft_square(f_new)
            sc_new = 2.0 * n * float(np.max(c_new))
            if np.isfinite(sc_new) and sc_new < cur - 1.0e-12:
                f = f_new
                accepted = True
                break

        if not accepted:
            width = max(1, n // 256)
            start = int(rng.integers(0, max(1, n - width)))
            f_new = f.copy()
            gain = float(np.exp(rng.normal(0.0, 0.22)))
            f_new[start:start + width] *= gain
            f_new = np.where(f_new < 0.0, 0.0, f_new)
            s = f_new.sum()
            if np.isfinite(s) and s > 0.0:
                f_new = f_new / s
                c_new = _fft_square(f_new)
                sc_new = 2.0 * n * float(np.max(c_new))
                if np.isfinite(sc_new) and sc_new < cur:
                    f = f_new

        it += 1

    # Final exact checks on current and local best.
    for seq in (best_local_seq, f):
        try:
            ex = evaluate_sequence(np.asarray(seq, dtype=np.float64).tolist())
        except Exception:
            ex = np.inf
        if np.isfinite(ex) and ex < best_exact_score:
            best_exact_score = float(ex)
            best_exact_seq = np.asarray(seq, dtype=np.float64).copy()

    if best_exact_seq is not None:
        return best_exact_seq, best_local_score, best_exact_seq, best_exact_score
    return best_local_seq, best_local_score, None, best_exact_score


def propose_candidate(seed: int = 42, budget_s: float = 1800, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    deadline = t0 + max(1.0, budget - 4.0)
    rng = np.random.default_rng(seed)

    # Choose a length close to the previous construction if it is reasonable.
    n = 4096
    try:
        hs = height_sequence_1
        if hs is not None:
            arr = np.asarray(hs, dtype=np.float64)
            arr = arr[np.isfinite(arr)]
            if arr.size >= 1024:
                n = int(arr.size)
    except Exception:
        pass

    n = int(max(1024, min(8192, n)))
    print("chosen length:", n, flush=True)

    F = _build_features(n, n_cheb=6, n_trig=12)
    uniform = np.ones(n, dtype=np.float64) / float(n)

    state = {
        "best_local": np.inf,
        "best_particle": uniform.copy(),
        "best_theta": np.zeros(F.shape[1], dtype=np.float64),
        "best_exact": np.inf,
        "best_exact_particle": uniform.copy(),
        "candidates": [],
        "idx": 0,
    }

    polish_budget = max(90.0, min(650.0, 0.28 * budget))
    search_deadline = max(t0 + 30.0, deadline - polish_budget)
    search_duration = max(1.0, search_deadline - t0)
    K = F.shape[1]

    start_list = _make_starts(F, n, rng)
    phases = [(1.00, 0.46), (0.35, 0.32), (0.15, 0.22)]
    cumulative = 0.0
    first_call = True

    for tau_frac, frac in phases:
        phase_end = t0 + (cumulative + frac) * search_duration
        cumulative += frac
        if time.monotonic() > phase_end:
            continue

        seqs = []
        if first_call:
            seqs = [x.copy() for x in start_list]
        else:
            top_thetas = [item[3].copy() for item in state["candidates"][:6]]
            seqs = top_thetas
            for sigma in (1.0, 1.8, 2.6):
                for _ in range(3):
                    th = rng.standard_normal(K) * sigma
                    seqs.append(np.clip(th, -5.0, 5.0))

        if n > 5000:
            maxiter = 110
            maxfun = 360
        else:
            maxiter = 170
            maxfun = 650

        tau = tau_frac / float(n)
        for th0 in seqs:
            if time.monotonic() > phase_end:
                break
            try:
                th = _optimize_theta(th0, F, tau, maxiter, maxfun)
                _process_theta(th, F, n, state)
            except Exception:
                pass
            if time.monotonic() > search_deadline:
                break

        first_call = False
        print(
            "phase tau={:.3f} local_best={:.6f} exact_best={:.6f}".format(
                tau_frac, state["best_local"], state["best_exact"]
            ),
            flush=True,
        )
        if time.monotonic() > search_deadline:
            break

    # If there is extra search time, use low-temperature random restarts.
    while time.monotonic() < search_deadline:
        th0 = rng.standard_normal(K) * 2.0
        th0 = np.clip(th0, -5.0, 5.0)
        try:
            th = _optimize_theta(th0, F, 0.12 / float(n), 90, 260)
            _process_theta(th, F, n, state)
        except Exception:
            break

    print(
        "after smooth search local_best={:.6f} exact_best={:.6f}".format(
            state["best_local"], state["best_exact"]
        ),
        flush=True,
    )

    # Convex mixing of the best shapes can expose a slightly lower peak.
    try:
        mixed, mixed_score = _mix_candidates(state["candidates"], max_cands=6)
        if mixed is not None and np.isfinite(mixed_score) and mixed_score < state["best_local"]:
            state["best_local"] = float(mixed_score)
            state["best_particle"] = mixed
            try:
                ex = evaluate_sequence(mixed.tolist())
                if np.isfinite(ex) and ex < state["best_exact"]:
                    state["best_exact"] = float(ex)
                    state["best_exact_particle"] = mixed.copy()
            except Exception:
                pass
    except Exception:
        pass

    best_p = np.asarray(state["best_particle"], dtype=np.float64)
    if best_p is None or best_p.size != n:
        best_p = uniform
    else:
        best_p = np.where(best_p < 0.0, 0.0, best_p)
        best_p = np.where(np.isfinite(best_p), best_p, 0.0)
        s = best_p.sum()
        if np.isfinite(s) and s > 0.0:
            best_p /= s
        else:
            best_p = uniform.copy()

    remaining = deadline - time.monotonic() - 3.0
    polish_local_seq = None
    polish_local_score = state["best_local"]
    polish_exact_seq = None
    polish_exact_score = state["best_exact"]

    if remaining > 25.0:
        try:
            polish_local_seq, polish_local_score, polish_exact_seq, polish_exact_score = (
                _polish_sequence(best_p, remaining, state["best_local"], state["best_exact"])
            )
        except Exception:
            polish_local_seq = None
            polish_local_score = state["best_local"]
            polish_exact_seq = None
            polish_exact_score = state["best_exact"]

    candidates = [
        np.asarray(best_p, dtype=np.float64),
        np.asarray(state["best_particle"], dtype=np.float64),
        np.asarray(state["best_exact_particle"], dtype=np.float64),
        uniform,
    ]
    if polish_local_seq is not None:
        candidates.append(np.asarray(polish_local_seq, dtype=np.float64))
    if polish_exact_seq is not None:
        candidates.append(np.asarray(polish_exact_seq, dtype=np.float64))

    final_seq = uniform
    final_score = np.inf
    for seq in candidates:
        seq = np.asarray(seq, dtype=np.float64)
        if seq.size != n:
            continue
        seq = np.where(seq < 0.0, 0.0, seq)
        seq = np.where(np.isfinite(seq), seq, 0.0)
        s = seq.sum()
        if not np.isfinite(s) or s <= 0.0:
            continue
        seq = seq / s
        try:
            ex = evaluate_sequence(seq.tolist())
        except Exception:
            ex = np.inf
        if np.isfinite(ex) and ex < final_score:
            final_score = float(ex)
            final_seq = seq.copy()

    if not np.isfinite(final_score):
        final_seq = uniform
        try:
            final_score = evaluate_sequence(final_seq.tolist())
        except Exception:
            final_score = np.inf

    print("final local/selected exact score:", final_score, flush=True)
    return [float(x) for x in final_seq]
```
