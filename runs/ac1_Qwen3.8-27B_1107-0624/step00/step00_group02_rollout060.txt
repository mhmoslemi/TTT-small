```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp(a, M):
    """Solve the LP: max sum(g) s.t. (a*g)_k <= M, g >= 0"""
    n = len(a)
    # Build constraint matrix A such that (A @ g)_k = (a*g)_k
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    
    # Sparse matrix construction
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(k, n-1) + 1):
            idx = k - j  # index into a
            if 0 <= idx <= n-1:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None
```

```python
"""
Minimizes 2*n*max(self-convolution)/sum^2 using LP-guided iterations
with adaptive step size, gradient refinement, and multi-start.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def solve_lp_direction(a, M):
    """Find g that maximizes sum(g) subject to (a*g)_k <= M, g >= 0."""
    n = len(a)
    n_cons = 2 * n - 1
    
    # Build sparse matrix: A[k, j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(n_cons):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(n_cons, n))
    c = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(a, max_step=0.5):
    """One LP-based iteration. Returns improved sequence or None."""
    n = len(a)
    b = np.convolve(a, a)
    M = b.max()
    S = a.sum()
    
    g = solve_lp_direction(a, M)
    if g is None or g.sum() <= 0:
        return None
    
    g = g / g.sum()
    
    # Line search on step size
    best_a = a.copy()
    best_obj = 2 * n * M / (S ** 2)
    
    for t in [max_step, max_step * 0.5, max_step * 0.25, max_step * 0.1]:
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 0.01:
            continue
        a_new = a_new / S_new
        b_new = np.convolve(a_new, a_new)
        M_new = b_new.max()
        obj_new = 2 * n * M_new / (S_new ** 2)  # S_new is 1 after normalization
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new
    
    if best_obj < 2 * n * M / (S ** 2) * (1 - 1e-10):
        return best_a
    return None


def gradient_step(a, lr):
    """One projected gradient descent step on J."""
    n = len(a)
    S = a.sum()
    b = np.convolve(a, a)
    M = b.max()
    K = np.where(np.isclose(b, M, rtol=1e-12))[0]
    
    # Subgradient of max(b)
    grad_M = np.zeros(n)
    for k in K:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            grad_M[j] += 2.0 * a[k - j]
    grad_M /= len(K)
    
    # Gradient of J = 2n*M/S^2 (S=1 after normalization)
    # dJ/da_j = 2n * (grad_M_j / S - 2M/S^2)
    grad_J = 2.0 * n * grad_M / S - 4.0 * n * M / (S * S)
    
    a_new = a - lr * grad_J
    a_new = np.clip(a_new, 0, None)
    S_new = a_new.sum()
    if S_new > 0:
        a_new /= S_new
    return a_new
```

```python
def _compute_subgradient(a):
    """Compute subgradient of 2n*max(conv(a,a)) on the simplex (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    M = b.max()
    # Use tolerance for argmax
    K = np.where(b >= M * (1 - 1e-10))[0]
    grad = np.zeros(n)
    for k in K:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= len(K)
    return 2.0 * n * grad, M
```

LP-guided optimization with gradient refinement for minimizing
2*n*max(self-convolution)/sum(a)^2. Uses LP to find descent directions,
projected gradient descent for polishing, and multi-start for escaping
local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    n_cons = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_cons):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(n_cons, n))
    c = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _compute_subgrad(b, a, n):
    """Compute subgradient of max(b) w.r.t. a given current b and n."""
    M = b.max()
    K = np.where(b >= M * (1 - 1e-10))[0]
    grad = np.zeros(n)
    for k in K:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= float(len(K))
    return grad, M
