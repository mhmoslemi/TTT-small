import numpy as np
import time

"""
Minimize  2*n*max(conv(a,a))/sum(a)^2  for a non-negative sequence a.

Key facts used:
  * The objective is scale invariant, so we optimise on the simplex sum(a)=1, a>=0
    and track score = 2*n*max(linear_convolution(a,a)).
  * For the uniform sequence the score is exactly 2.0.  Non-symmetric (e.g. ramp-like)
    non-negative sequences push the convolution peak below that, so good local search
    starting from ramps / random shapes beats the 2.0 state of the art and moves toward
    the theoretical ~1 floor (flat convolution).
  * We use FFT-based linear convolution (O(n log n)) so large n (~24k) is cheap, and a
    softmax-smoothed "top-K lag" gradient with projected line search on the simplex to
    attack the dominant convolution peaks simultaneously (escapes the cat-and-mouse trap
    of pure subgradient).  Multiple lengths + up-sampling of the best sequence give extra
    attack surface, and periodic random perturbations diversify the search.
"""


_K_TOP = 16          # number of dominant lags used for the smoothed gradient
_ETA_STEPS = (0.05, 0.10, 0.20, 0.35, 0.60, 0.90, 1.30)


def _linconv(a):
    """Linear (full) convolution of a with itself via FFT, returned real and trimmed."""
    n = a.size
    N = 1
    need = 2 * n - 1
    while N < need:
        N <<= 1
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)
    return C[:need]


def _maxconv(a):
    return float(np.max(_linconv(a)))


def _score1(a, n):
    # sum(a) is kept == 1 by construction
    return 2.0 * n * _maxconv(a)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if not np.isfinite(s) or s <= 1e-9:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _init_seq(n, kind, rng):
    """Return a non-negative, sum-normalised starting sequence of length n."""
    if kind == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif kind == "ramp":
        slope = rng.uniform(0.3, 1.6) * (1.0 if rng.random() < 0.5 else -1.0)
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + slope * x
        a = np.clip(a, 1e-3, None)
    elif kind == "bump":
        x = np.linspace(0.0, 1.0, n)
        c = rng.uniform(0.15, 0.85)
        w = rng.uniform(0.08, 0.45)
        slope = rng.uniform(-1.0, 1.0)
        a = (1.0 + slope * x) + rng.uniform(0.5, 2.0) * np.exp(-((x - c) ** 2) / (2 * w * w))
        a = np.clip(a, 1e-3, None)
    else:  # noise
        a = np.abs(rng.standard_normal(n)) + 0.2
    return _normalize(a)


def _upsample(a, new_n):
    n = a.size
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    return _normalize(b)


def _topk_grad(a, n, conv):
    """Smoothed gradient of the top-K convolution lags (pointing to increase them)."""
    K = min(_K_TOP, conv.size)
    part = np.argpartition(conv, -K)[-K:]
    top = part[np.argsort(conv[part])[::-1]]
    T = max(float(conv.max()) * 0.7, 1e-12)
    v = conv[top] / T
    w = np.exp(v - v.max())
    w /= w.sum()
    grad = np.zeros(n, dtype=np.float64)
    for j, wj in zip(top, w):
        j = int(j)
        lo = j - n + 1
        if lo < 0:
            lo = 0
        hi = j
        if hi >= n:
            hi = n - 1
        if hi < lo:
            continue
        grad[lo:hi + 1] += (2.0 * wj) * a[lo:hi + 1][::-1]
    return grad


def _local_search(a, n, rng, t_end, tag):
    """Projected line-search local search on the simplex.  Returns (best_a, best_score)."""
    best = _normalize(a)
    best_s = _score1(best, n)
    print(f"[{tag}] n={n} start score={best_s:.6f}")
    last_eta = 0.20
    stall = 0
    it = 0
    while time.monotonic() < t_end:
        conv = _linconv(best)
        grad = _topk_grad(best, n, conv)
        gn = np.max(np.abs(grad))
        if gn <= 1e-15:
            break
        grad = grad / gn            # unit-scale; raw grad was ~O(1/n), re-scaling lets
        base = (2.0 / n) * grad * n  # == raw-ish magnitude; keep step ~O(1/n) per entry
        # line search over step sizes (in units of 'base')
        cur = best
        cur_s = best_s
        improved = False
        for eta in _ETA_STEPS:
            cand = cur - eta * base
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s <= 1e-9:
                continue
            cand = cand / s
            sc = _score1(cand, n)
            if sc + 1e-12 < cur_s:
                cur = cand
                cur_s = sc
                last_eta = eta
                improved = True
        if improved:
            best = cur
            best_s = cur_s
            stall = 0
            if it % 200 == 0:
                print(f"[{tag}] n={n} score={best_s:.6f} iter={it}")
        else:
            # escape: block multiplicative perturbation + small noise
            stall += 1
            width = max(2, n // 12)
            start = int(rng.integers(0, n - width + 1))
            cand = best.copy()
            cand[start:start + width] *= np.exp(rng.normal(0.0, 0.25))
            if rng.random() < 0.3:
                cand += 0.05 * (1.0 / n) * rng.standard_normal(n)
            cand = _normalize(cand)
            sc = _score1(cand, n)
            if sc + 1e-12 < best_s:
                best = cand
                best_s = sc
                stall = 0
            elif stall > 40:
                # harder diversification, keep best as anchor
                cand = 0.7 * best + 0.3 * _init_seq(n, "noise", rng)
                cand = _normalize(cand)
                sc = _score1(cand, n)
                if sc + 1e-12 < best_s:
                    best = cand
                    best_s = sc
                    stall = 0
                else:
                    stall = 0
        it += 1
    print(f"[{tag}] n={n} DONE score={best_s:.6f} iters={it}")
    return best, best_s

        print(f"verify evaluate_sequence(best) = {ev:.6f}  self={_score1(best, best.size):.6f}")
        if ev > global_score and ev == ev and global_score == global_score:
            # evaluator disagrees (should not), trust evaluator's value but keep sequence
            print("note: evaluator value used as reported score")
    except Exception as e:
        print("verify eval raise:", e)

    print(f"FINAL returned length={best.size} score~{_score1(best, best.size):.6f}")
    return out_list
