"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using analytical initializations
(linear ramps, power laws) refined by gradient descent with log-sum-exp
smoothing and FFT-based gradient computation.
"""

import numpy as np
import time


def _compute_convolution(a):
    """Compute full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = np.fft.rfft(a, n=size)
    return np.fft.irfft(A * A, n=size)[:2 * n - 1]


def _softmax_weights(c, tau):
    """Numerically stable softmax over convolution values."""
    c_shift = c - c.max()
    e = np.exp(tau * c_shift)
    return e / e.sum()


def _gradient(a, w):
    """
    Gradient of (1/tau)*log(sum exp(tau*c_k)) w.r.t. a[j].
    = sum_k w[k] * (a[k-j] + a[j-k])
    Computed via FFT.
    """
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1

    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)

    # term2[j] = (w * a)[j]
    wa = np.fft.irfft(W * A, n=size)
    term2 = wa[:n]

    # term1[j] = (a_rev * w)[j + n - 1]
    a_rev = a[::-1].copy()
    Ar = np.fft.rfft(a_rev, n=size)
    aw = np.fft.irfft(Ar * W, n=size)
    term1 = aw[n - 1:n - 1 + n]

    return term1 + term2


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} via clip-and-rescale."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones_like(v) / len(v)
    return v / s


def _refine(a, deadline, max_iter=100000):
    """
    Gradient descent on smoothed objective. a must have sum 1, a >= 0.
    Uses adaptive tau and learning rate. Returns (best_a, best_obj_value).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_obj = 2.0 * n * _compute_convolution(a).max()

    # Schedule parameters
    tau_base = 20.0
    lr_base = 0.5 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Adaptive tau: start moderate, increase over time
        progress = it / max(1, max_iter)
        tau = tau_base * (1.0 + 10.0 * progress)

        # Adaptive learning rate: decay over time
        lr = lr_base * (1.0 + 20.0 * progress) ** (-1.0)

        c = _compute_convolution(a)
        cur_obj = 2.0 * n * c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        w = _softmax_weights(c, tau)
        grad = _gradient(a, w)

        # Gradient step
        a = a - lr * grad
        a = _project_simplex(a)

    return best_a, best_obj
