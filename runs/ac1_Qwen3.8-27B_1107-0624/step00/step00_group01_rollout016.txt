**Alternative: Use the "multiplicative weights" / "mirror descent" approach**

Instead of additive updates, use multiplicative updates:
f_new[j] = f[j] * exp(-lr * gradient[j])

Actually, for problems with f >= 0 and sum(f) = const, the mirror map is the negative entropy, and the mirror descent update is:
f_new[j] = f[j] * exp(-lr * g[j]) / Z

    best_score = evaluate_sequence(best_f.tolist())
    
    # Main optimization loop
    lr = 1.0 / n  # initial learning rate
    k_peaks = 5  # number of peaks to consider

        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        conv = np.convolve(f, f)
        
        # Find top k peaks

        top_indices = top_indices[np.argsort(conv[top_indices])[::-1]]

        grad = np.zeros(n)
        for k in top_indices:
            # gradient of (f*f)[k] is 2*f[k::-1] (reversed and shifted)
            # More precisely: grad[j] += 2*f[k-j] for valid j
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2 * f[idx]
        
        grad /= k_peaks  # average
        
        # Mirror descent update
        f_new = f * np.exp(-lr * grad)
        f_new = np.clip(f_new, 0, None)
        S_new = np.sum(f_new)
        if S_new < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / S_new
        
        # Evaluate
        score = evaluate_sequence(f_new.tolist())
        
        if score < best_score:
            best_f = f_new.copy()
            best_score = score
            f = f_new
        else:
            # Reduce learning rate
            lr *= 0.95
        
        # Periodically add random perturbation
        if iteration % 100 == 0:
            noise = np.random.randn(n) * 0.01 / n
            f = f + noise
            f = np.clip(f, 0, None)
            f = f / np.sum(f)
    
    return [float(x) for x in best_f]
