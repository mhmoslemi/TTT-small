```python
def flat_spectrum_iteration(a, n_iter=10000):
    a = a.copy()
    n = len(a)
    # Zero-pad to get linear convolution
    N = 1  # next power of 2 >= 2n-1
    while N < 2*n-1:
        N *= 2
    for it in range(n_iter):
        # Forward FFT
        A = np.fft.fft(a, N)
        # Compute current spectrum
        power = np.abs(A)**2
        # Target: flat spectrum (average power)
        target_power = np.mean(power)
        # Modify: keep phase, set magnitude to sqrt(target_power)
        if target_power > 0:
            scale = np.sqrt(target_power) / np.abs(A)
            A_flat = A * scale
        # IFFT
        a_new = np.fft.ifft(A_flat).real[:n]
        # Project to non-negative
        a = np.maximum(a_new, 0)
    return a
```
