"""
Minimize 2*n*max(conv(a,a))/(sum(a)^2) over non-negative real sequences a.

Key insight: because the objective is scale-invariant, normalize sum(a)=1 and
minimize 2*n*max_k (a*a)[k].  A one-sided power "ramp" a_j ∝ (j+1)^p with p≈1
has its self-convolution peak at the centre equal to ≈ 2/(3n), giving score
≈ 4/3 (already below the 1.5027 target and far below the previous 2.0 uniform
result).  We grid over the exponent p and length n to find the best analytic
seed, then refine with a projected sub-gradient descent on the active (near-
max) convolution constraints, always validated against the authoritative
evaluate_sequence().
"""

import time
import math
import numpy as np


# ---------------------------------------------------------------------------
# Fast full linear convolution via real FFT (O(n log n)).
# ---------------------------------------------------------------------------
def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = 1
    while size < m:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.irfft(A * A, size)[:m]
    # tiny numerical negatives
    B[B < 0] = 0.0
    return B


def score_from_conv(conv, n):
    return float(2.0 * n * conv.max())


def make_power(seed_n, p):
    j = np.arange(1, seed_n + 1, dtype=np.float64)
    a = j ** p
    a /= a.sum()
    return a


def make_triangle(seed_n):
    j = np.arange(seed_n, dtype=np.float64)
    mid = 0.5 * (seed_n - 1)
    a = (seed_n - 1 - np.abs(j - mid)) + 1.0
    a = np.clip(a, 0, None)
    s = a.sum()
    if s <= 0:
        a = np.ones(seed_n)
        s = a.sum()
    return a / s


def make_exp(seed_n, q):
    j = np.arange(seed_n, dtype=np.float64)
    a = np.exp(-q * j)
    a /= a.sum()
    return a


# ---------------------------------------------------------------------------
# Projected sub-gradient step for minimizing max_k conv[k] under a>=0, sum a=1.
# Uses the top `topk` active constraints, averaged, for stability.
# ---------------------------------------------------------------------------
def subgrad_step(a, eta, topk):
    n = a.size
    conv = fast_conv(a)
    m = conv.mean()
    # active = indices where conv is within a small fraction of the max
    thr = 0.5 * (conv.max() + m)
    idx = np.where(conv >= thr)[0]
    if topk and idx.size > topk:
        # keep the largest topk
        order = np.argsort(conv[idx])[::-1][:topk]
        idx = idx[order]
    grad = np.zeros(n, dtype=np.float64)
    for k in idx:
        # d conv[k]/d a[j] = 2 * a[k-j] for 0<=k-j<n
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi >= lo:
            partners = k - np.arange(lo, hi + 1)
            grad[lo:hi + 1] += 2.0 * a[partners]
    # average over active constraints
    if idx.size:
        grad /= idx.size
    a_new = a - eta * grad
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s <= 0:
        return a.copy()
    a_new /= s
    return a_new


def local_refine(a0, deadline, rnd):
    best = a0.copy()
    conv0 = fast_conv(best)
    best_score = score_from_conv(conv0, best.size)
    n = best.size

    # a few initial step sizes, geometrically decaying
    eta0 = 0.05
    step = eta0
    stuck = 0
    while time.monotonic() < deadline - 8.0:
        a = subgrad_step(best, step, topk=6)
        conv = fast_conv(a)
        sc = score_from_conv(conv, n)
        if sc < best_score - 1e-9:
            best = a
            best_score = sc
            stuck = 0
        else:
            stuck += 1
            # occasional small random kick to escape plateaus
            if stuck > 40 and rnd.random() < 0.5:
                w = int(rnd.integers(max(1, n // 100), max(2, n // 12)))
                st = int(rnd.integers(0, max(1, n - w + 1)))
                best = best.copy()
                best[st:st + w] *= float(np.exp(rnd.normal(0, 0.10)))
                best = np.clip(best, 0, None)
                s = best.sum()
                if s > 0:
                    best /= s
            if stuck > 200:
                step *= 0.7
                stuck = 0
        if step > eta0 * 1e-3:
            step *= 0.97
    return best, best_score
