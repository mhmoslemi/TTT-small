Uses projected gradient descent with log-sum-exp smooth max and FFT-based
gradient computation. Multiple diverse initializations are explored in parallel
(time-multiplexed). Adaptive step size and periodic restarts help escape
local minima.

import numpy as np
import time


def _softmax_weights(c, beta):
    """Compute softmax(c * beta) as a probability vector."""
    bc = beta * c
    bc_max = np.max(bc)
    exp_bc = np.exp(bc - bc_max)
    s = np.sum(exp_bc)
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = exp_bc / s
    return w


def _gradient_smooth_max(a, w, fft_size):
    """Compute gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a.
    
    Result is 2 * cross_correlation(a, w)[:n]
    """
    Ga = np.fft.rfft(a, n=fft_size)
    Gw = np.fft.rfft(w, n=fft_size)
    g = np.fft.irfft(np.conj(Ga) * Gw, n=fft_size)
    return 2.0 * g[:len(a)]


def _evaluate_normalized(a):
    """Compute 2*n*max(convolve(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c)

    a_new = a_new / s

            if s_new < 1e-15:

                continue

            c_new = np.convolve(a_new, a_new)
            obj_new = float(np.max(c_new))
            if obj_new < obj_current:
                a = a_new
                accepted = True
                break
            lr *= 0.5

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                stall_count = 0
                lr = min(lr * 1.1, 0.5)
            else:
                stall_count += 1
        else:
            stall_count += 1
            lr = max(lr * 0.5, 1e-6)

            pert = rng.standard_normal(n) * (0.05 + 0.05 * rng.random())
            a = best_a + pert
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            lr = 0.02
            beta = rng.choice([10, 30, 50, 100, 200, 500])
            stall_count = 0
            max_stall = max(50, max_stall // 2)

        if it % 500 == 0 and it > 0:
            beta *= 1.2  # increase sharpness
            beta = min(beta, 1000.0)
