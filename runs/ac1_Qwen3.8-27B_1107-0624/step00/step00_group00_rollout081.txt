```
def normalize_p(a, floor=1e-300):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)

    if not np.isfinite(s) or s <= 1e-300:

        return np.ones(n, dtype=np.float64)/n
```

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))**2.

The objective is scale-invariant, so the internal search works on a
probability simplex p = a / sum(a).  The score is then 2*n*max(convolve(p,p)).

The search combines:
  - many low-dimensional/random initializations,
  - active-set / soft-max subgradient multiplicative updates,
  - occasional small linear-program directions on active lags,
  - occasional pair-unbalancing moves for the most critical lags,
  - time-boxed restarts/polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def normalize_p(a):
    """Clip to non-negative finite values and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a = a / s
    # A very small floor keeps LP rows strictly positive without affecting scale.
    a = a + 1e-15
    a = a / float(np.sum(a))
    return a


def full_auto_conv(a):
    """Full auto-convolution p*p, using FFT for long arrays."""
    n = int(a.size)
    if n <= 0:
        return np.array([0.0])
    m = 2 * n - 1
    if n < 800:
        c = np.convolve(a, a, mode="full")
    else:
        size = 1 << (m - 1).bit_length()
        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fa * fa, size, n=size)[:m]
    c = np.asarray(c, dtype=np.float64)
    # FFT may introduce tiny negative artifacts near zero.
    c = np.maximum(c, 0.0)
    return c


def score_p(p):
    """Scale-invariant score for normalized probability vector p."""
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return float("inf")
    c = full_auto_conv(p)
    if c.size == 0:
        return float("inf")
    return float(2.0 * p.size * float(np.max(c)))


def to_eval_sequence(p):
    """Convert normalized p to a valid input sequence for evaluate_sequence."""
    p = normalize_p(p)
    n = p.size
    maxp = float(np.max(p)) if n > 0 else 0.0
    if maxp > 0.0:
        scale = min(float(n), 1000.0 / maxp)
    else:
        scale = 1.0
    if scale <= 0.0 or not np.isfinite(scale):
        scale = 1.0
    seq = p * scale
    total = float(np.sum(seq))
    if total < 0.01:
        scale2 = 10.0 / total if total > 0 else 10.0
        seq = p * scale2
        total = float(np.sum(seq))
    if total < 0.01:
        seq = np.ones(n, dtype=np.float64)
        total = float(n)
        scale = 1.0
        seq = p * scale
        if float(np.sum(seq)) < 0.01:
            seq = np.ones(n, dtype=np.float64)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.maximum(seq, 0.0)
    seq = np.minimum(seq, 1000.0)
    if float(np.sum(seq)) < 0.01:
        seq = np.ones(max(1, n), dtype=np.float64)
    return [float(x) for x in seq]


def reshape_sequence(old, new_n):
    """Linearly resample a sequence to a new length."""
    old = np.asarray(old, dtype=np.float64)
    if old.ndim != 1:
        old = old.ravel()
    old = np.where(np.isfinite(old), old, 0.0)
    old = np.maximum(old, 0.0)
    if old.size == 0:
        return np.ones(max(1, new_n), dtype=np.float64)
    if old.size == new_n:
        return old
    if old.size == 1:
        return np.full(max(1, new_n), old[0], dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, max(1, new_n))
    return np.interp(x_new, x_old, old)


def top_active_indices(c, count):
    """Indices of the largest count coefficients in convolution c."""
    n = int(c.size)
    count = int(count)
    if count <= 0:
        return np.empty(0, dtype=np.int64)
    if count >= n:
        return np.arange(n, dtype=np.int64)
    idx = np.argpartition(c, -count)[-count:]
    order = np.argsort(c[idx])[::-1]
    return idx[order]


def active_gradient(p, active, weights):
    """
    Weighted average gradient of the selected convolution coefficients.

    For full polynomial convolution c[k] = sum_{i+j=k} p_i p_j,
    d c[k] / d p[j] = 2 p[k-j] if 0 <= k-j < n, else 0.
    The factor 2 is immaterial here.
    """
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    grad = np.zeros(n, dtype=np.float64)
    active = np.asarray(active, dtype=np.int64)
    if active.size == 0:
        return grad
    if weights is None:
        weights = np.ones(active.size, dtype=np.float64)
    else:
        weights = np.asarray(weights, dtype=np.float64)
        if weights.size != active.size:
            weights = np.ones(active.size, dtype=np.float64)
        weights = np.where(np.isfinite(weights), weights, 0.0)
        wsum = float(np.sum(weights))
        if wsum <= 0.0:
            weights = np.ones(active.size, dtype=np.float64)
        else:
            weights = weights / wsum

    for k, w in zip(active, weights):
        if not np.isfinite(w) or w == 0.0:
            continue
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        if jlo <= jhi:
            grad[jlo:jhi + 1] += w * p[k - jhi:k - jlo + 1][::-1]
    return grad


def normalize_weight(v):
    """Normalize a non-negative vector to sum 1; uniform fallback."""
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    s = float(np.sum(v))
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / max(1, v.size)
    return v / s


def multiplicative_candidate(p, w, eta, noise, rng):
    """
    Multiplicative projected step on the simplex:
      p_new ~ p * exp(-eta*(w - mean(w)) + noise)
    High-gradient coordinates are decreased.
    """
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return p.copy()
    w = normalize_weight(w)
    mean_w = 1.0 / float(n)
    logf = -float(eta) * (w - mean_w)
    if noise > 0.0:
        logf = logf + float(noise) * rng.standard_normal(n)
    logf = np.where(np.isfinite(logf), logf, 0.0)
    logf -= float(np.max(logf))
    q = p * np.exp(logf)
    return normalize_p(q)


def mix_candidate(p, q, t):
    """Convex mix of two normalized probability vectors."""
    p = np.asarray(p, dtype=np.float64)
    q = normalize_p(q)
    t = float(np.clip(t, 0.0, 1.0))
    return normalize_p((1.0 - t) * p + t * q)


def pair_unbalance_candidate(p, active, frac=0.40):
    """
    Reduce a selected lag by unbalancing paired coordinates p[j], p[k-j].
    For each pair, move some of the smaller piece into the larger piece.
    """
    p = normalize_p(p)
    n = p.size
    q = p.copy()
    frac = float(np.clip(frac, 0.0, 1.0))
    if frac <= 0.0:
        return q

    active = np.asarray(active, dtype=np.int64)
    for k in active:
        k = int(k)
        if k < 0 or k >= 2 * n - 1:
            continue
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        if jlo > jhi:
            continue

        a = np.arange(jlo, jhi + 1, dtype=np.int64)
        b = k - a
        mask = a < b
        if np.any(mask):
            ia = a[mask]
            ib = b[mask]
            va = q[ia]
            vb = q[ib]
            small_idx = np.where(va <= vb, ia, ib)
            large_idx = np.where(va <= vb, ib, ia)
            moved = np.minimum(va, vb) * frac
            q[small_idx] -= moved
            q[large_idx] += moved

        # Self-paired coordinate, if present: move toward an outside coordinate.
        if k % 2 == 0:
            m = k // 2
            if 0 <= m < n:
                out = -1
                if jlo > 0:
                    out = max(0, jlo - 1)
                elif jhi + 1 < n:
                    out = min(n - 1, jhi + 1)
                if out >= 0:
                    move = q[m] * frac * 0.5
                    q[m] -= move
                    q[out] += move

    q = np.maximum(q, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        return p
    return q / s


def lp_direction(p, active, M, obj_w, rng):
    """
    Small LP direction:
      minimize obj_w · b
      subject to sum(b)=1, b>=0, and convolve(p,b)[k] <= M for active lags.

    This finds a probability b with low current-gradient inner product that
    does not increase active cross-convolution at first order.
    """
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    active = np.asarray(active, dtype=np.int64)
    m = active.size
    if m == 0 or n == 0 or M <= 0.0:
        return None

    A = np.zeros((m, n), dtype=np.float64)
    for r, k in enumerate(active):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        if jlo <= jhi:
            A[r, jlo:jhi + 1] = p[k - jhi:k - jlo + 1][::-1]

    b_ub = M * (1.0 + 1e-8) + 1e-12
    w = normalize_weight(obj_w)
    c_obj = w
    if rng is not None:
        c_obj = 0.999 * w + 0.001 * (rng.random(n) / float(n))
        cs = float(np.sum(c_obj))
        if cs > 0.0 and np.isfinite(cs):
            c_obj = c_obj / cs

    A_eq = np.ones(n, dtype=np.float64)
    b_eq = 1.0
    res = None
    last_err = None
    try:
        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": 1.2, "disp": False}
            )
        except Exception:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=(0.0, None),
                method="highs"
            )
    except Exception as e:
        last_err = e

    x = None
    if res is not None:
        x = getattr(res, "x", None)
    if x is None:
        return None

    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != n or not np.all(np.isfinite(x)):
        return None

    q = normalize_p(x)
    # Validate feasibility before accepting, in case linprog returned a failed model.
    try:
        Aq = A @ q
        if np.all(Aq <= b_ub + 1e-5):
            return q
    except Exception:
        return None
    return None


def optimize_start(p0, n, deadline, rng, verbose_id=0):
    """Time-boxed local optimization from a normalized probability vector."""
    p = normalize_p(p0)
    n = int(p.size)
    best_p = p.copy()
    best_score = score_p(p)
    current_score = best_score

    eta = 4.0
    base_noise = 0.03
    no_imp = 0
    iteration = 0
    last_lp = time.monotonic()

    print(f"[optimize {verbose_id}] n={n} start_score={best_score:.8f}")

    while time.monotonic() < deadline:
        c = full_auto_conv(p)
        M = float(np.max(c)) if c.size else 0.0
        current_score = float(2.0 * n * M)

        if current_score < best_score:
            best_score = current_score
            best_p = p.copy()

        count_g = int(min(24, max(6, n // 400)))
        active = top_active_indices(c, count_g)
        if active.size == 0:
            active = np.array([n - 1], dtype=np.int64)

        vals = c[active]
        temp = max(1e-12, M * 0.08)
        weights_lag = np.exp((vals - float(np.max(vals))) / temp)
        ws = float(np.sum(weights_lag))
        if ws > 0.0 and np.isfinite(ws):
            weights_lag = weights_lag / ws
        else:
            weights_lag = np.ones(active.size, dtype=np.float64) / float(active.size)

        grad = active_gradient(p, active, weights_lag)
        w = normalize_weight(grad)

        improved = False

        # 1) Smooth multiplicative active-subgradient step.
        noise = base_noise * (1.0 + min(4.0, float(no_imp) / 25.0))
        etas = [eta, max(0.05, eta * 0.4)]
        if no_imp > 0 and no_imp % 20 == 0:
            etas = [eta * 1.6, eta, max(0.05, eta * 0.4)]

        for e in etas:
            q = multiplicative_candidate(p, w, e, noise if e == etas[0] else 0.5 * noise, rng)
            s = score_p(q)
            if np.isfinite(s) and s < current_score * (1.0 - 1e-12):
                p = q
                current_score = s
                improved = True
                break

        if improved:
            if current_score < best_score:
                best_score = current_score
                best_p = p.copy()
            no_imp = 0
            iteration += 1
            continue

        no_imp += 1

        # 2) Structural pair-unbalancing for the top few lags.
        if no_imp >= 3 and (iteration % 3 == 0):
            act3 = top_active_indices(c, min(3, active.size))
            if act3.size > 0:
                q = pair_unbalance_candidate(p, act3, frac=0.40)
                for t in (0.35, 0.15, 0.05):
                    qm = mix_candidate(p, q, t)
                    s = score_p(qm)
                    if np.isfinite(s) and s < current_score * (1.0 - 1e-12):
                        p = qm
                        current_score = s
                        improved = True
                        break

        if improved:
            if current_score < best_score:
                best_score = current_score
                best_p = p.copy()
            no_imp = 0
            iteration += 1
            continue

        # 3) Occasional active-LP direction.
        now = time.monotonic()
        if now - last_lp > 2.0 and no_imp >= 4:
            last_lp = now
            actlp = top_active_indices(c, min(8, count_g))
            q = lp_direction(p, actlp, M, w, rng)
            if q is not None:
                for t in (0.35, 0.18, 0.08, 0.03):
                    qm = mix_candidate(p, q, t)
                    s = score_p(qm)
                    if np.isfinite(s) and s < current_score * (1.0 - 1e-12):
                        p = qm
                        current_score = s
                        improved = True
                        break

        if improved:
            if current_score < best_score:
                best_score = current_score
                best_p = p.copy()
            no_imp = 0
            iteration += 1
            continue

        # 4) Adaptation / escape moves.
        if no_imp >= 10 and no_imp % 10 == 0:
            eta = max(0.1, min(30.0, eta * 0.85))

        if no_imp >= 60:
            # Try a modest randomization around the current point.
            q = multiplicative_candidate(p, np.ones(n) / float(n), 1.0, 0.60, rng)
            s = score_p(q)
            if np.isfinite(s) and s < best_score:
                p = q
                current_score = s
                best_score = s
                best_p = p.copy()
                no_imp = 0
                iteration += 1
                continue

        if no_imp >= 120:
            # Reset to the best point (with light noise) to avoid drifting too far.
            p = best_p.copy()
            p = multiplicative_candidate(p, np.ones(n) / float(n), 1.0, 0.03, rng)
            current_score = score_p(p)
            no_imp = 0
            iteration += 1
            continue

        iteration += 1

        if iteration % 80 == 0:
            print(f"[optimize {verbose_id}] iter={iteration} current={current_score:.8f} best={best_score:.8f}")

    return best_p, best_score


def candidate_arrays(n, base, rng):
    """Generate a diverse list of non-negative array shapes for given length."""
    arrs = []
    n = int(n)
    if n <= 0:
        return arrs

    x = np.arange(n, dtype=np.float64) / float(max(1, n - 1))
    y = np.arange(1, n + 1, dtype=np.float64) / float(n + 1)
    inc = np.arange(1, n + 1, dtype=np.float64)

    arrs.append(np.ones(n, dtype=np.float64))
    arrs.append(inc.copy())
    arrs.append(inc[::-1].copy())

    for alpha in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0, 9.0):
        arrs.append(np.exp(-alpha * x))
        arrs.append(np.exp(-alpha * (1.0 - x)))

    for beta in (0.10, 0.20, 0.30, 0.40, 0.50, 0.60):
        arrs.append(inc ** (-beta))
        arrs.append(inc[::-1] ** (-beta))

    arrs.append(np.sin(np.pi * y))
    arrs.append(0.5 + 0.5 * np.cos(2.0 * np.pi * x))
    arrs.append(np.exp(-0.5 * (((x - 0.5) / 0.10) ** 2)))

    for center in (0.15, 0.30, 0.50, 0.70, 0.85):
        for width in (0.05, 0.10, 0.18):
            arrs.append(np.exp(-0.5 * (((x - center) / width) ** 2)))

    # Mixture of left and right dimples.
    for a1 in (1.0, 2.0, 3.0):
        for a2 in (1.0, 2.0, 3.0):
            for w in (0.20, 0.50, 1.0, 2.0):
                arrs.append(np.exp(-a1 * x) + w * np.exp(-a2 * (1.0 - x)))

    if base is not None:
        base_arr = np.asarray(base, dtype=np.float64)
        if base_arr.size > 0:
            arrs.append(reshape_sequence(base_arr, n))

    if rng is not None:
        for _ in range(10):
            arrs.append(rng.random(n))
        for _ in range(6):
            a = float(rng.uniform(0.5, 8.0))
            b = float(rng.uniform(0.5, 8.0))
            w = float(rng.uniform(0.1, 4.0))
            arrs.append(np.exp(-a * x) + w * np.exp(-b * (1.0 - x)))

    return arrs


def candidate_item_score(item):
    return float(item[0])


def collect_initial_candidates(base, seed=1):
    """Evaluate many inexpensive initial candidates and return the best scored ones."""
    rng = np.random.default_rng(seed)
    base = np.asarray(base, dtype=np.float64) if base is not None else np.ones(1024, dtype=np.float64)
    base = np.where(np.isfinite(base), base, 0.0)
    base = np.maximum(base, 0.0)
    if base.size == 0 or float(np.sum(base)) <= 0.0:
        base = np.ones(1024, dtype=np.float64)

    base_n = int(base.size)
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    if 64 <= base_n <= 10000:
        lengths.append(base_n)
    if base_n >= 128 and 64 <= base_n // 2 <= 10000:
        lengths.append(base_n // 2)

    lengths = []
    seen = set()
    for L in lengths:
        L = int(L)
        if 64 <= L <= 10000 and L not in seen:
            seen.add(L)
            lengths.append(L)
    lengths.sort()

    all_res = []
    for n in lengths:
        arrays = candidate_arrays(n, base, rng)
        local_res = []
        for arr in arrays:
            try:
                p = normalize_p(arr)
                if p.size != n:
                    continue
                s = score_p(p)
                if np.isfinite(s):
                    local_res.append((s, n, p.copy()))
            except Exception:
                continue
        local_res.sort(key=candidate_item_score)
        all_res.extend(local_res[:12])

    all_res.sort(key=candidate_item_score)
    # Keep some diversity in length.
    top = []
    length_count = {}
    for item in all_res:
        n = int(item[1])
        if length_count.get(n, 0) >= 3:
            continue
        top.append(item)
        length_count[n] = length_count.get(n, 0) + 1
        if len(top) >= 8:
            break
    if not top:
        p = np.ones(1024, dtype=np.float64) / 1024.0
        top.append((score_p(p), 1024, p.copy()))
    return top


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Time-boxed constructor invoked by the harness.

    Returns a list of non-negative floats minimizing the given objective.
    """
    start_time = time.monotonic()
    budget_s = float(max(60.0, float(budget_s)))
    deadline = start_time + budget_s - 30.0

    rng = np.random.default_rng(seed)

    # Load prior construction if available.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = np.ones(2048, dtype=np.float64)
    if base.ndim != 1:
        base = base.ravel()
    if base.size == 0 or not np.all(np.isfinite(base)) or float(np.sum(base)) <= 0.0:
        base = np.ones(2048, dtype=np.float64)

    print(f"budget={budget_s:.1f}s base_len={base.size}")

    inits = collect_initial_candidates(base, seed=int(seed) + 7)
    if inits:
        global_best = inits[0][2].copy()
        global_best_score = float(inits[0][0])
    else:
        p = np.ones(1024, dtype=np.float64) / 1024.0
        global_best = p.copy()
        global_best_score = score_p(p)

    print(f"initial global best (FFT score) = {global_best_score:.8f}, len={global_best.size}")

    # Time-boxed optimization of several diverse starts, followed by a final polish.
    top = inits[:5]
    if not top:
        top = [(global_best_score, int(global_best.size), global_best.copy())]

    parts = len(top) + 1
    for idx, item in enumerate(top):
        if time.monotonic() >= deadline - 20.0:
            break
        remaining_parts = max(1, parts - idx)
        part_time = max(30.0, (deadline - time.monotonic()) / float(remaining_parts) - 5.0)
        part_time = min(420.0, part_time)
        cand_deadline = min(deadline - 20.0, time.monotonic() + part_time)

        n = int(item[1])
        p0 = normalize_p(item[2])
        print(f"start {idx}: n={n}, init_score={float(item[0]):.8f}, slice={part_time:.1f}s")
        try:
            p_best, s_best = optimize_start(
                p0, n, cand_deadline,
                np.random.default_rng(int(seed) + 100 + 11 * idx),
                verbose_id=idx
            )
        except Exception as e:
            print("optimize_start failed:", e)
            p_best = p0
            s_best = score_p(p0)

        if np.isfinite(s_best) and s_best < global_best_score:
            global_best_score = float(s_best)
            global_best = p_best.copy()
            print(f"  new global best = {global_best_score:.8f} at len={global_best.size}")

    # Final polishing phase from the current global best.
    if time.monotonic() < deadline - 15.0:
        n_best = int(global_best.size)
        p_noise = multiplicative_candidate(
            global_best,
            np.ones(n_best) / float(n_best),
            1.0,
            0.05,
            rng
        )
        try:
            p_best, s_best = optimize_start(
                p_noise, n_best, deadline,
                np.random.default_rng(int(seed) + 900),
                verbose_id=99
            )
            if np.isfinite(s_best) and s_best < global_best_score:
                global_best_score = float(s_best)
                global_best = p_best.copy()
        except Exception as e:
            print("final polish failed:", e)

    # Ensure a valid final returned sequence.
    seq = to_eval_sequence(global_best)
    try:
        ev = float(evaluate_sequence(seq))
        if not np.isfinite(ev):
            raise ValueError("eval failed")
    except Exception:
        # Conservative fallback: uniform sequence gives score 2.
        seq = [1.0] * 1024

    print(f"final sequence length={len(seq)} final_eval={ev:.8f}")
    return seq
```
