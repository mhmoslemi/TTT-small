            print("  [%s] n=%d beta=%.0f score=%.6f" % (verbose_tag, n, b, best_score))
    return best_a, best_score


# ------------------------------ driver ---------------------------------------
def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Anneated, multi-start, multi-length search for the best non-negative
    step function.  Always returns the best sequence (by evaluate_sequence)
    found before the time budget expires."""
    try:
        _ = _gradient_ok()
    except Exception:
        pass

    best_a = None
    best_score = np.inf

    # use the provided known-good construction as an initial candidate
    try:
        a_init = np.asarray(height_sequence_1, dtype=np.float64)
        if a_init.ndim == 1 and a_init.size >= 4 and np.isfinite(a_init).all():
            a_init = np.clip(a_init, 0.0, 1000.0)
            if a_init.sum() > 0:
                a_init = a_init / a_init.sum()
                best_a = a_init.copy()
                best_score = float(evaluate_sequence(a_init.tolist()))
                print("seed height_sequence_1 n=%d score=%.6f" %
                      (a_init.size, best_score))
    except Exception:
        pass
    if best_a is None:
        best_a = _binomial_a(64)
        best_score = float(evaluate_sequence(best_a.tolist()))

    deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)

    def try_block(a, n, alloc):
        nonlocal best_a, best_score
        if time.monotonic() > deadline or alloc <= 0.5:
            return
        a, sc = _refine_from_a(a, n, alloc, "opt")
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print("NEW GLOBAL BEST n=%d score=%.8f (left %.0fs)" %
                  (n, best_score, deadline - time.monotonic()))

    # ascending sweep; the largest length we reach sets the final answer
    n_list = [32, 48, 64, 96, 128, 192, 256, 384, 512, 768, 1024, 1536, 2048, 3072, 4096]
    for n in n_list:
        if time.monotonic() > deadline:
            break
        left = deadline - time.monotonic()
        alloc_n = min(left, max(15.0, 0.05 * n))
        if alloc_n < 4.0 and left > 0:
            alloc_n = left
        b0 = _binomial_a(n)
        starts = [b0, _tailboost_a(b0, 0.5), _tailboost_a(b0, 1.5), np.full(n, 1.0 / n)]
        sub = alloc_n / len(starts)
        for st in starts:
            if time.monotonic() > deadline:
                break
            try_block(st, n, sub)

    # final polish on the global best with extra budget
    if best_a is not None and time.monotonic() < deadline - 4.0:
        a, sc = _refine_from_a(best_a, best_a.size,
                               min(180.0, deadline - time.monotonic() - 2.0), "polish")
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print("POLISH -> %.8f" % best_score)

    print("FINAL score=%.8f n=%d" % (best_score, best_a.size))
    return [float(x) for x in best_a]
