    c = np.convolve(a, a, 'full')
    k_star = np.argmax(c)

    S = a.sum()

    # a_{k*-i} means: for position i in the gradient, the value is a[k_star - i] (if in range, else 0)
    reflected = np.zeros(n)
    for i in range(n):

        if 0 <= j < n:

    if a_new.sum() > 0:

    return a_new
