import time
from scipy.optimize import linprog
from scipy import sparse

def project_simplex(v, S):
    """Project v onto {x : x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute convolution and score."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, None
    conv = np.convolve(a, a)
    M = np.max(conv)
    score = 2.0 * n * M / (S * S)
    return score, conv

    """Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    rows = []
    cols = []
    vals = []

        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

    n = len(f)
    A = build_conv_matrix(f, n)

    b_ub = np.concatenate([f_peak * np.ones(n_constraints), np.zeros(n)])
    c = -np.ones(n)

def project_simplex(v, S):
    """Project v onto {x : x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def conv_max_fft(a):
    """Fast computation of max of linear convolution a*a using FFT."""
    n = len(a)

    conv = np.fft.irfft(A * A, size)
