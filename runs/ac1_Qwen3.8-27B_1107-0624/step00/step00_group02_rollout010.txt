def fft_flatten(a, n_fft, n_iter=100):

    a = np.asarray(a, dtype=np.float64)
    n = len(a)

        A = np.fft.rfft(a, n=n_fft)

For n = 7000:

- Subgradient step: O(n * top_m) = O(n) for fixed top_m
- Per iteration: O(n log n) for convolution + O(n) for update = O(n log n)

"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using:
1. Multiple initializations (provided, uniform, random, structured)
2. DFT flattening for initial refinement
3. Subgradient descent with top-m subgradient and diminishing step sizes
4. Local random search for final polish
"""

import numpy as np
import time

def fast_self_convolve(a):
    """Compute a*a using FFT."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a, N_fft)
    return np.fft.irfft(A * A, N_fft)[:N]

def dft_flatten(a, n_iter=50):
    """Flatten DFT magnitude to reduce autocorrelation peak."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / np.sum(a)
    N_fft = 1 << (2*n - 1).bit_length() if False else 1 << (2*n).bit_length()
    for _ in range(n_iter):
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        # Target: flatten to mean magnitude
        target = np.mean(mag)
        phase = np.angle(A)
        A_new = target * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new /= s
        else:
            a_new = np.ones(n) / n
        a = a_new
    return a

def subgradient_descent(a, time_budget=60, top_m=10, base_alpha=0.005, rng=None):
    """Subgradient descent to minimize max(self-convolution) on the simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0)
    a = a / np.sum(a)
    
    best = a.copy()
    best_max = np.max(fast_self_convolve(a))
    
    if rng is None:
        rng = np.random.default_rng()
    
    start = time.monotonic()
    t = 0
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        alpha = base_alpha / (1.0 + t / 500.0)
        
        b = fast_self_convolve(a)
        # Find top-m indices
        if top_m >= len(b):
            top_indices = np.arange(len(b))
        else:
            top_indices = np.argpartition(b, -top_m)[-top_m:]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in top_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # g[lo:hi+1] += 2 * a[k - hi : k - lo + 1][::-1]
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(top_indices)
        
        # Update
        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new /= s
        else:
            a_new = best.copy()
        
        b_new = fast_self_convolve(a_new)
        current_max = np.max(b_new)
        
        if current_max < best_max:
            best = a_new.copy()
            best_max = current_max
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        t += 1
        
        # Random restart from best if stalled
        if stall_count > 200 and rng.random() < 0.1:
            a = best + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a /= np.sum(a)
            stall_count = 0
    
    return best, best_max

def local_refine(a, time_budget=10, rng=None):
    """Local random search refinement."""
    if rng is None:
        rng = np.random.default_rng()
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    a = a / np.sum(a)
    best = a.copy()
    best_max = np.max(fast_self_convolve(a))
    
    start = time.monotonic()
    scale = 0.01
    while time.monotonic() - start < time_budget:
        # Random perturbation
        a_try = best + scale * rng.standard_normal(n)
        a_try = np.maximum(a_try, 0)
        a_try /= np.sum(a_try)
        
        max_val = np.max(fast_self_convolve(a_try))
        if max_val < best_max:
            best = a_try
            best_max = max_val
            scale = max(scale * 1.05, 0.001)
        else:
            scale = max(scale * 0.95, 0.0005)
    
    return best, best_max
