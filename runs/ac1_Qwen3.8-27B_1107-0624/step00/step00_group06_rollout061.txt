**Algorithm: LP-based iterative peak reduction**

1. Choose n (e.g., 256, 512, 1024).
2. Initialize f with some non-negative values (e.g., random, or based on a good pattern).
3. Repeat:

   b. Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
   c. If sum(g) <= sum(f) * (1 + epsilon), stop (converged).
   d. Rescale g to have sum = sum(f).
   e. Update f = (1-t)*f + t*g for appropriate t (line search).
4. Return f.

The LP in step (b) has:
- n variables (g_0, ..., g_{n-1})

```python
def optimize_lp(f, deadline):
    n = len(f)
    t = 0.5  # initial step size
    
    while time.monotonic() < deadline - 10:
        ff = np.convolve(f, f)
        M = np.max(ff)
        
        # Build constraint matrix A: A[k,j] = f[k-j] if 0 <= k-j < n
        # This is a (2n-1) x n matrix
        # For efficiency, build it row by row
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            # A[k,j] = f[k-j] for j in range(max(0, k-n+1), min(n, k+1))
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                A[k, j] = f[k - j]
        
        # LP: minimize -sum(g) subject to A @ g <= M, g >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        sum_g = np.sum(g)
        sum_f = np.sum(f)
        
        if sum_g <= sum_f * (1 + 1e-6):
            break  # converged
        
        # Rescale g to have same sum as f
        g = g * sum_f / sum_g
        
        # Line search for step size
        best_t = t
        best_obj = np.max(np.convolve(f, f))
        for trial_t in [t, 0.5*t, 0.25*t, 0.1*t]:
            f_trial = (1 - trial_t) * f + trial_t * g
            obj = np.max(np.convolve(f_trial, f_trial))
            if obj < best_obj:
                best_obj = obj
                best_t = trial_t
        
        f = (1 - best_t) * f + best_t * g
        t = best_t
    
    return f
```

def build_conv_matrix(f):
    """Build the convolution matrix A where (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        # A[k, j_start:j_end] = f[k - j_start : k - j_start + (j_end - j_start)][::-1]
        # Actually: A[k,j] = f[k-j] for j in [j_start, j_end)
        indices = np.arange(j_start, j_end)
        A[k, indices] = f[k - indices]
    return A

    c = -np.ones(n)
    b = np.full(2*n-1, M)

                  options={'disp': False, 'dual_feasibility_tolerance': 1e-9,

        return f, True

    sum_g = g.sum()
    sum_f = f.sum()

        return f, True
