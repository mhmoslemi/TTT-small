"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated convolution,
LP-inspired directional updates targeting top convolution lags, adaptive
step sizes, and periodic restarts with diverse initializations.
"""

import numpy as np
import time


def fft_self_convolve(a):
    """Compute full self-convolution np.convolve(a,a) using FFT."""
    n = len(a)
    result_len = 2 * n - 1
    size = 1
    while size < result_len:
        size *= 2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:result_len]


def compute_direction(a, top_k_indices):
    """
    Compute LP-inspired direction that reduces convolution at the given lags.
    Targets the top-K lags by averaging their masks.
    """
    n = len(a)
    # Average the masks from top-K lags
    c = np.zeros(n)
    for k_idx in top_k_indices:
        lo = max(0, k_idx - (n - 1))
        hi = min(n, k_idx + 1)
        for i in range(lo, hi):
            c[i] += a[k_idx - i]
    c /= len(top_k_indices)

    pos_mask = a > 1e-12
    c_pos = c[pos_mask]
    sum_c = np.sum(c_pos)

    if sum_c < 1e-15:
        return np.where(pos_mask, 1.0, 0.0)

    alpha = np.sum(c_pos * c_pos) / sum_c
    d = np.zeros(n)
    d[pos_mask] = alpha - c[pos_mask]
    return d


def compute_direction_single(a, max_idx):
    """LP-inspired direction for a single lag (faster, no loop)."""
    n = len(a)
    c = np.zeros(n)
    lo = max(0, max_idx - (n - 1))
    hi = min(n, max_idx + 1)
    # Vectorized: c[i] = a[max_idx - i] for i in [lo, hi)
    idx = np.arange(lo, hi)
    c[idx] = a[max_idx - idx]

    pos_mask = a > 1e-12
    c_pos = c[pos_mask]
    sum_c = np.sum(c_pos)

    alpha = np.sum(c_pos * c_pos) / sum_c
    d = np.zeros(n)
    d[pos_mask] = alpha - c[pos_mask]
    return d


def euclidian_norm(a):
    return np.sqrt(np.sum(a * a))

def random_init(n, rng, init_type):
    """Generate a random non-negative initialization of length n."""
    if init_type == 0:
        # Sinusoidal modulation
        x = np.arange(n)
        freq = rng.uniform(1, 8)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
    elif init_type == 1:
        # Triangle wave

        period = int(rng.uniform(n // 4, n))

    elif init_type == 2:
        # Smoothed random

        a = a - np.min(a)
        a = a / np.max(a) * 0.8 + 0.2
    elif init_type == 3:
        # Two-tone

        f1 = rng.uniform(1, 5)
        f2 = rng.uniform(5, 20)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * f1 * x / n) + 0.2 * np.sin(2 * np.pi * f2 * x / n)
    elif init_type == 4:
        # Exponential decay from center
        center = n // 2
        x = np.abs(np.arange(n) - center)
        a = np.exp(-x / (n * rng.uniform(0.1, 0.5)))
        a = a / np.max(a)
    else:
        # Random smooth (low-pass)

        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a)), a)
        a = a + 0.1

    a = np.maximum(a, 0.01)
    return a
