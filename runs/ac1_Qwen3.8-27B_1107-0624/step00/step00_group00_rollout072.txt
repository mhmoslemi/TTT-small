def soft_gradient(a, T):
    """Compute soft-max smoothed gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a, 'full')
    # Soft-max weights (stabilized)
    max_c = conv.max()
    w = np.exp(T * (conv - max_c) / conv.sum())
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(2*n-1)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # Gradient: grad[i] = 2 * sum_m w[m+i] * a[m]
