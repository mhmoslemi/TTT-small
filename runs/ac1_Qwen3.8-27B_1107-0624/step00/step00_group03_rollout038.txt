def project_sum1_nonneg(x):
    """Project x onto {a >= 0, sum(a) = 1}."""
    # Sort in decreasing order
    x_sorted = np.sort(x)[::-1]
    # Find the largest k such that x_sorted[k] - (sum(x_sorted[:k]) - 1)/k > 0
    # This is the "water-filling" algorithm
    cumsum = np.cumsum(x_sorted)
    rho = np.nonzero(x_sorted * np.arange(1, len(x)+1) > (cumsum - 1))[0]
    if len(rho) == 0:
        return np.ones_like(x) / len(x)
    k = rho[-1] + 1
    theta = (cumsum[k-1] - 1) / k
    return np.maximum(x - theta, 0)
```

This is the standard Euclidean projection onto the probability simplex.

import numpy as np
import time
from scipy.optimize import linprog

def project_simplex(x):
    """Euclidean projection onto probability simplex {a>=0, sum(a)=1}."""
    x_sorted = np.sort(x)[::-1]
    n = len(x)
    cumsum = np.cumsum(x_sorted)
    # Find rho: largest k such that x_sorted[k-1] > (cumsum[k-1]-1)/k
    idx = np.arange(1, n+1)
    mask = x_sorted > (cumsum - 1) / idx
    if not mask.any():
        return np.ones(n) / n
    k = np.nonzero(mask)[0][-1] + 1
    theta = (cumsum[k-1] - 1.0) / k
    return np.maximum(x - theta, 0.0)

def subgradient_step(a, eta, top_k=50):
    """One subgradient step to minimize max(convolve(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    # Find top-k positions
    if top_k >= len(c):
        top_idx = np.arange(len(c))
        weights = np.ones(len(c)) / len(c)
    else:
        top_idx = np.argsort(c)[::-1][:top_k]
        # Weights proportional to (c[k] - c.min())^2 or just uniform
        weights = np.ones(top_k) / top_k
    
    # Compute smoothed subgradient
    g = np.zeros(n)
    for k in top_idx:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            g[j_min:j_max+1] += weights[list(top_idx).index(k)] * 2 * a[k-j_max:k-j_min+1][::-1]
    
    x = a - eta * g
    return project_simplex(x)
```

        j_max = min(n - 1, k)
        if j_min <= j_max:
            g[j_min:j_max+1] += weights[idx] * 2 * a[k - j_max : k - j_min + 1][::-1]

    logits = -eta * g
    logits -= logits.max()  # for numerical stability
    exp_logits = np.exp(logits)
    a_new = a * exp_logits
    a_new = a_new / a_new.sum()
    return a_new

    """Euclidean projection onto {a >= 0, sum(a) = 1}."""
    n = len(x)
    x_sorted = np.sort(x)[::-1]

    idx = np.arange(1, n + 1)

    k = np.nonzero(mask)[0][-1] + 1

    return np.maximum(x - theta, 0.0)

    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-15:

    return 2.0 * n * c.max() / (s * s)

    n = len(a)

import numpy as np

def project_simplex(x):
    """Euclidean projection onto probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    idx = np.arange(1, n + 1)
    mask = x_sorted > (cumsum - 1.0) / idx
    if not np.any(mask):
        return np.ones(n) / n
    k = np.nonzero(mask)[0][-1] + 1
    theta = (cumsum[k - 1] - 1.0) / k
    return np.maximum(x - theta, 0.0)


def compute_score(a, n):
    """Compute 2n * max(convolve(a,a)) / sum(a)^2."""
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def get_subgradient(a, top_k=100):
    """Compute smoothed subgradient of max(convolve(a,a)) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    k_max = len(c)
    if top_k >= k_max:
        top_idx = np.arange(k_max)
        w = np.ones(k_max) / k_max
    else:
        top_idx = np.argsort(c)[::-1][:top_k]
        cvals = c[top_idx]
        w = (cvals - cvals.min() + 1e-10)
        w = w / w.sum()
    g = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            g[j_min:j_max + 1] += w[idx] * 2.0 * a[k - j_max:k - j_min + 1][::-1]
    return g

    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""

            s = b.sum()
            if s > 1e-10:

        pass
    return None
