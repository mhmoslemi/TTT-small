    p = exp_b / sum_exp

**Alternative: Spectral method / iterative projection**

The problem is to minimize $\|f*f\|_\infty / (\sum f)^2$ over $f \geq 0$.

Another idea: **Water-filling / iterative reweighting**

Start with $f^{(0)}$. At each iteration:
1. Compute $b = f*f$.
2. Identify "hot spots" where $b_k$ is large.
3. Reduce the weights $f_j$ that contribute most to these hot spots.
4. Increase the weights $f_j$ that don't contribute to hot spots.
5. Normalize.

This is similar to the " multiplicative weights" update:
$f_j^{(new)} = f_j^{(old)} \cdot \exp(-\eta \cdot c_j)$

where $c_j$ measures how much $f_j$ contributes to the max.

Let me formalize:
- $c_j = \sum_k w_k \frac{\partial b_k}{\partial f_j}$ where $w_k$ is a weight that emphasizes large $b_k$ values.
- If $w_k = e^{b_k/\tau}$, this is the same as the gradient of the soft-max!

So the multiplicative weights update is:
$f_j^{(new)} = f_j^{(old)} \cdot \exp(-\eta \cdot 2 \sum_k p_k f_{k-j} \mathbf{1}[0 \leq k-j < n])$

where $p_k \propto e^{b_k/\tau}$.

This is essentially gradient descent in the "exponential" parameterization, which is equivalent to Frank-Wolfe (conditional gradient) or mirror descent.

The advantage of this update is that it automatically maintains non-negativity (exponential of anything is positive). And it can be computed efficiently using the same FFT-based cross-correlation.

import numpy as np
import time
import scipy.optimize as so
import scipy.sparse as sp

    return np.convolve(f, f)


def compute_soft_stats(b, tau):

    exp_b = np.exp((b - m) / tau)
    sum_exp = np.sum(exp_b)
    soft_max = m + tau * np.log(sum_exp)
    p = exp_b / sum_exp

def cross_corr(f, p):

    f_rev = f[::-1]
    L = 1
    while L < n + len(p) - 1:
        L *= 2
    F = np.fft.rfft(f_rev, n=L)
    P = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(F * P, n=L)

    n = x.size
    b = compute_conv(x)
    S = x.sum()
    soft_max, _ = compute_soft_stats(b, tau)
    return 2.0 * n * soft_max / (S * S)


def smooth_gradient(x, tau):
    n = x.size
    b = compute_conv(x)
    S = x.sum()
    soft_max, p = compute_soft_stats(b, tau)
    R = cross_corr(x, p)
    g = 2.0 * R
    grad = 2.0 * n * (g / (S * S) - 2.0 * soft_max / (S ** 3))
    return grad

    n = f.size
    b = compute_conv(f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * np.max(b) / (S * S)


def multiplicative_step(f, tau, eta):
    n = f.size
    f_norm = f / f.sum()
    b = compute_conv(f_norm)
    _, p = compute_soft_stats(b, tau)
    R = cross_corr(f_norm, p)
    c = 2.0 * R
    c_max = c.max()
    if c_max < 1e-15:
        return f_norm.copy()
    f_new = f_norm * np.exp(-eta * c / c_max)
    f_new = np.maximum(f_new, 0.0)
    if f_new.sum() < 1e-15:
        return f_norm.copy()
    return f_new / f_new.sum()

    """Solve the LP to find a good direction g."""
    n = f.size
    # A_ub: (2n-1) x n matrix where A_ub[k, j] = f[k-j] if 0 <= k-j < n
    # Build as sparse
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])  # Wait, A_ub[k,j] should be f[k-j]

    # So A_ub[k, j] = f[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # Wait: (f*g)_k = sum_{j} f[j] g[k-j] where both j and k-j are in [0, n-1]

    rows = []
    cols = []
    vals = []

            cols.append(j)

"""
Minimizes the step-function self-convolution score using smoothed gradient
descent (L-BFGS-B with soft-max), multiplicative updates, and LP-based
direction finding. Starts from height_sequence_1.
"""

import numpy as np
import time
import scipy.optimize as so
import scipy.sparse as sp


def fft_convolve(f):
    """Compute full linear convolution f*f using FFT."""
    n = f.size
    L = 1
    while L < 2 * n - 1:
        L *= 2
    F = np.fft.rfft(f, n=L)
    return np.fft.irfft(F * F, n=L)[:2 * n - 1]


def soft_max_and_p(b, tau):
    """Numerically stable soft-max and probability vector."""
    m = b.max()
    shifted = (b - m) / tau
    exp_b = np.exp(shifted)
    s = exp_b.sum()
    soft = m + tau * np.log(s)
    p = exp_b / s
    return soft, p


def xcorr(f, p):
    """Compute R[j] = sum_{m=0}^{n-1} f[m]*p[m+j] for j=0,...,n-1 via FFT."""
    n = f.size
    f_rev = f[::-1]
    L = 1
    while L < n + p.size - 1:
        L *= 2
    F = np.fft.rfft(f_rev, n=L)
    P = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(F * P, n=L)
    return conv[n - 1:n - 1 + n]


def obj_smooth(x, tau):
    n = x.size
    b = fft_convolve(x)
    S = x.sum()
    sm, _ = soft_max_and_p(b, tau)
    return 2.0 * n * sm / (S * S)


def grad_smooth(x, tau):
    n = x.size
    b = fft_convolve(x)
    S = x.sum()
    sm, p = soft_max_and_p(b, tau)
    R = xcorr(x, p)
    g = 2.0 * R
    return 2.0 * n * (g / (S * S) - 2.0 * sm / (S ** 3))


def exact_score(f):
    n = f.size
    b = fft_convolve(f)
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * np.max(b) / (S * S)


def mult_step(f, tau, eta):
    """One multiplicative (exponential weights) gradient step."""
    f_n = f / f.sum()
    b = fft_convolve(f_n)
    _, p = soft_max_and_p(b, tau)
    R = xcorr(f_n, p)
    c = 2.0 * R
    cmax = c.max()
    if cmax < 1e-15:
        return f_n.copy()
    f_new = f_n * np.exp(-eta * c / cmax)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f_n.copy()
    return f_new / s


def build_conv_mat(f):
    """Build sparse convolution matrix A where (A@g)_k = (f*g)_k."""
    n = f.size
    m = 2 * n - 1
    diags = []
    offsets = []
    for d in range(n):
        length = n - d
        diags.append(np.full(length, f[d]))
        offsets.append(d)
    return sp.diags(diags, offsets, shape=(m, n), format='csr')


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = f.size
    A = build_conv_mat(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    res = so.linprog(c, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True})
    if res.success:
        return res.x
    return None
