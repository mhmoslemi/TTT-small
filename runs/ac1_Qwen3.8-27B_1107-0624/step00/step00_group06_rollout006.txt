def compute_score_local(a):
    """Fast local score computation (same as evaluate_sequence for valid inputs)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return float('inf')
    sum_a = np.sum(a)
    if sum_a < 0.01:

    conv = np.convolve(a, a)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a ** 2))


def try_perturbations(base, n_perturb=200, seed=42):
    """Try various perturbations to see if any beat the base score."""
