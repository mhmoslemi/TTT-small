    for k in range(1, 15):
        for eps in np.arange(0.1, 0.9, 0.05):
            a_try = 1.0 + eps * np.sin(2.0 * np.pi * k * np.arange(n) / n)
            a_try = np.clip(a_try, 0, None)
            sc = compute_score(a_try)
            if sc < best_score:
                best_score = sc
                best = a_try.copy()

    # Sum of multiple antisymmetric sinusoids
    for trial in range(40):
        n_sines = rng.integers(2, 6)
        freqs = rng.integers(1, 12, size=n_sines)
        amps = rng.uniform(0.2, 0.8, size=n_sines)
        phases = rng.uniform(0, np.pi, size=n_sines)
        a_try = np.ones(n)
        for f, amp, ph in zip(freqs, amps, phases):
            a_try += amp * np.sin(2.0 * np.pi * f * np.arange(n) / n + ph)
        a_try = np.clip(a_try, 0, None)
        sc = compute_score(a_try)
        if sc < best_score:
            best_score = sc
            best = a_try.copy()

    # Multi-level antisymmetric step
    for n_levels in [2, 3, 4, 5]:
        for total_amp in np.arange(0.2, 0.9, 0.1):
            a_try = np.ones(n)
            for level in range(n_levels):
                cut = m * (level + 1) / n_levels
                amp = total_amp / n_levels
                a_try[:int(cut)] -= amp
                a_try[int(n - cut):] += amp
            a_try = np.clip(a_try, 0, None)
            sc = compute_score(a_try)
            if sc < best_score:
                best_score = sc
                best = a_try.copy()

    print(f"Phase 1 done. Best score: {best_score:.6f}", flush=True)

    # === Phase 2: Gradient descent on smoothed max ===
    a = best.copy()
    S = a.sum()
    a = a / S  # normalize sum to 1, so score = 2n*max(c)

    # Precompute FFT size for convolution
    N_fft = 1
    while N_fft < 3 * n - 1:
        N_fft *= 2

    w_rev_buf = np.empty(2 * n - 1)

    def compute_gradient(c, a_curr, beta):
        """Compute gradient of smoothed max using FFT cross-correlation."""
        c_max = c.max()
        weights = np.exp(beta * (c - c_max) / (S * S))
        w_sum = weights.sum()
        if w_sum > 0:
            weights /= w_sum
        else:
            weights = np.zeros(2 * n - 1)
            weights[c.argmax()] = 1.0

        # grad[i] = 2 * sum_j a[j] * w[i+j] for j=0..n-1
        # = 2 * convolve(a, w_rev)[n-1 : 2n-1][::-1]
        w_rev_buf[:] = weights[:2*n-1][::-1]
        A = np.fft.rfft(a_curr, N_fft)
        Wr = np.fft.rfft(w_rev_buf, N_fft)
        conv_result = np.fft.irfft(A * Wr, N_fft)[:3*n-2]
        return 2.0 * conv_result[n-1:2*n-1][::-1]

    best_p2 = best_score
    best_p2_a = best.copy()

    beta = 30.0
    lr = 0.003
    momentum = np.zeros(n)
    mu = 0.85

    for it in range(500000):
        if time.monotonic() > deadline - 400:
            break
        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_p2:
                best_p2 = sc
                best_p2_a = a.copy()
                best_score = sc
                best = a.copy()
                print(f"  P2 it={it}: score={sc:.6f}", flush=True)
            if it % 20000 == 0:
                # Adapt beta and lr
                c_now = np.convolve(a, a)
                spread = c_now.max() - c_now[c_now > 0].min()
                if spread > 0:
                    beta = max(10.0, min(100.0, beta * (1.1 if c_now.std() / c_now.max() < 0.1 else 0.9)))

        c = np.convolve(a, a)
        S = a.sum()
        if S < 0.01:
            break

        grad = compute_gradient(c, a, beta)

        # Momentum update
        momentum = mu * momentum + (1 - mu) * grad
        a_new = a - lr * momentum

        # Project: non-negative + normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 0.01:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        # Accept/reject
        c_new = np.convolve(a_new, a_new)
        if c_new.max() <= c.max() * 1.005:
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - lr * 0.3 * momentum
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 0.01:
                a_new2 = a_new2 / s2
                c_new2 = np.convolve(a_new2, a_new2)
                if c_new2.max() <= c.max():
                    a = a_new2

    sc_final2 = compute_score(a)
    if sc_final2 < best_score:
        best_score = sc_final2
        best = a.copy()

    print(f"Phase 2 done. Best score: {best_score:.6f}", flush=True)

    # === Phase 3: Random local search with structured moves ===
    a = best.copy()
    S = a.sum()
    a = a / S

    no_improve = 0
    stall_limit = 5000

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        if it % 5000 == 0:
            print(f"  P3 it={it}: best={best_score:.6f} no_improve={no_improve}", flush=True)
            if no_improve > stall_limit:
                # Diversify: add random antisymmetric perturbation
                amp = 0.1
                a[:] = a + amp * rng.normal(0, 1, n)
                a[n-1 - np.arange(n)] -= 2 * amp * rng.normal(0, 1, n)[np.arange(n)]
                # Enforce antisymmetry more carefully
                a_half = a[:m].copy()
                a[m+1:] = 2.0 * a[:m] * 0 + (a.sum() / n)  # placeholder
                # Simpler: just random restart around best
                a = best.copy() / best.sum()
                a += 0.05 * rng.normal(0, 1, n)
                a = np.clip(a, 0, None)
                a = a / a.sum()
                no_improve = 0

        # Generate perturbation: mix of random and antisymmetric
        mode = rng.random()
        if mode < 0.4:
            # Antisymmetric pair move
            n_pairs = rng.integers(5, 50)
            for _ in range(n_pairs):
                i = rng.integers(0, m)
                delta = rng.uniform(-0.2, 0.2)
                j = n - 1 - i
                a[i] += delta
                a[j] -= delta
        elif mode < 0.7:
            # Block multiplicative perturbation
            width = rng.integers(20, 200)
            start = rng.integers(0, n - width)
            factor = np.exp(rng.normal(0, 0.1))
            a[start:start+width] *= factor
        else:
            # Random antisymmetric sinusoidal
            f = rng.integers(1, 20)
            amp = rng.uniform(0.02, 0.15)
            a += amp * np.sin(2.0 * np.pi * f * np.arange(n) / n)

        a = np.clip(a, 0, None)
        s_a = a.sum()
        if s_a > 0.01:
            a = a / s_a

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # With some probability, keep the perturbation (random walk)
            if rng.random() < 0.2:
                pass  # keep a as is
            else:
                # revert (a is already modified, so we don't revert for random walk)
                pass

    # Final: try a few more structured improvements
    print(f"Phase 3 done. Best score: {best_score:.6f}", flush=True)

    # Final polish: try small antisymmetric adjustments on best
    a = best.copy()
    S = a.sum()
    a = a / S
    for trial in range(200):
        if time.monotonic() > deadline:
            break
        # Small random antisymmetric nudge
        i = rng.integers(0, m)
        delta = rng.uniform(-0.05, 0.05)
        a[i] += delta
        a[n-1-i] -= delta
        a = np.clip(a, 0, None)
        s_a = a.sum()
        if s_a > 0.01:
            a = a / s_a
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()

    print(f"Final best score: {best_score:.6f}", flush=True)

    # Verify with actual evaluator
    final = [float(x) for x in np.clip(best, 0, 1000)]
    actual = evaluate_sequence(final)
    print(f"Actual evaluated score: {actual:.6f}", flush=True)

    if not np.isfinite(actual) or actual > 2.0:
        # Fallback
        final = [1.0] * n

    return final
