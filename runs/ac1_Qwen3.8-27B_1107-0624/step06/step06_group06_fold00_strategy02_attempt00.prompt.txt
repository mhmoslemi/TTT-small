def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        # b_k = sum_i a[i]*a[k-i], gradient: 2*(P_k a)_j where (P_k a)_j = a[k-j]
        # Hessian: 2 * sum_{k in active} P_k (symmetric part)
        # Actually Hessian of b_k is 2*P_k where (P_k)_{i,j} = 1 if i+j=k
        # So H = 2 * sum_{k in active} P_k

        # (Hv)_i = 2 * sum_{k in active} sum_j P_k(i,j) * v_j
        # = 2 * sum_{k in active} v[k-i] (if 0<=k-i<=n-1)
        # = 2 * (conv(v, active))[n-1-i]... let me think more carefully.

        # (P_k v)_i = v[k-i] if 0 <= k-i <= n-1, else 0
        # sum_k (P_k v)_i = sum_k v[k-i] where 0<=k<=2n-2 and 0<=k-i<=n-1
        # = sum_{j: 0<=j<=n-1} v[j] * (number of k such that k=j+i and k in active)
        # = sum_j v[j] * active[j+i] (where active is defined on 0..2n-2)

        # So (Hv)_i = 2 * sum_j v[j] * active[i+j]
        # This is 2 * (conv_flip(v, active))[i] where conv_flip reverses one input.
        # Actually: sum_j v[j] * active[i+j] is a cross-correlation.
        # Let's compute it via FFT: it's the "correlation" of v and active.
        # corr(i) = sum_j v[j] * active[i+j]

        # For the Newton step, we need to solve (H + mu*I) dx = -grad + lambda*1
        # where lambda enforces sum(dx) = 0 (since we're on the sum=1 simplex).

def _shrinkage(a, deadline, n_try=50):

    z = z - z.mean()

    a_sorted_idx = np.argsort(a)

    # Try zeroing bottom 1%, 2%, 5%, 10%, 20%
    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20]

            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            # Re-optimize from this point
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = a + 0.1
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)

def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):

        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_blocks=4):
    """Swap random blocks of the sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = n // (2 * n_blocks)
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i+block_size].copy()
        z[i:i+block_size] = z[j:j+block_size]
        z[j:j+block_size] = tmp
    return _softmax(z)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Adaptive beta schedule: start large, decrease based on gap
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-11,
                    'maxcor': 40,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a
