```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a, b):
    n, m = len(a), len(b)

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
```

4. Gerchberg-Saxton spectral iteration can find different local minima than pure gradient methods

My strategy combines:
- GS spectral iteration (to escape local minima the gradient methods got stuck in)
- Log-sum-exp Adam with aggressive beta scheduling
- Peak-directed coordinate updates (targeted reduction of elements contributing to peaks)
- Multi-restart with diverse structures

```python
"""
Hybrid spectral-flattening optimizer: combines Gerchberg-Saxton spectral iteration,
log-sum-exp Adam with aggressive beta scheduling, peak-directed coordinate updates,
and multi-restart diversification to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _self_conv(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _self_conv(a)
    return float(2.0 * n * c.max())


def _grad_conv(a, w):
    n = len(a)
    a_rev = a[::-1]
    out_n = len(w) + n - 1
    N = _next_pow2(out_n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a_rev, N)
    result = np.fft.irfft(fw * fa, N)[:out_n]
    return 2.0 * result[n - 1:2 * n - 1]


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gs_spectral(a, deadline, n_iters=50000):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    N = _next_pow2(2 * n - 1)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        fa = np.fft.rfft(a, N)
        power = fa.real**2 + fa.imag**2

        target = np.ones_like(power)
        p_max = power.max()
        p_min = power.min()
        if p_max < 1e-30:
            break

        # Soft flattening: compress dynamic range
        relaxation = 0.5
        # Reduce peaks, boost valleys
        ratio = np.sqrt(target / (power + 1e-30))
        fa_new = fa * ratio

        a_new = np.fft.irfft(fa_new, N)[:n].copy()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s

        if it % 100 == 0:
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        a = a_new

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_a = a.copy()
        best_score = sc_final
    return best_a, best_score


def _gs_relaxed(a, deadline, alpha_start=0.3, alpha_end=0.05):
    """GS with adaptive relaxation: a_new = (1-alpha)*a + alpha*a_projected."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    N = _next_pow2(2 * n - 1)
    it_count = 0

    while time.monotonic() < deadline:
        it_count += 1
        progress = min(1.0, it_count / max(1, int(deadline - time.monotonic() + time.monotonic() - time.monotonic())))
        alpha = alpha_start * (alpha_end / alpha_start) ** min(1.0, it_count / 2000.0)

        fa = np.fft.rfft(a, N)
        power = fa.real**2 + fa.imag**2
        p_mean = power.mean()

        # Target: flatten towards mean
        # Use log-domain update for stability
        log_power = np.log(power + 1e-30)
        log_target = np.full_like(log_power, np.log(p_mean + 1e-30))
        # Blend
        log_power_new = (1 - alpha) * log_power + alpha * log_target
        power_new = np.exp(log_power_new)

        # Update magnitude, keep phase
        phase = fa / (np.abs(fa) + 1e-30)
        fa_new = phase * np.sqrt(power_new)

        a_new = np.fft.irfft(fa_new, N)[:n].copy()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s

        if it_count % 50 == 0:
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def _adam_smoothed(a_init, deadline, beta_steps, lr=0.008):
    """Adam on log-sum-exp smoothed objective with scheduled beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    for beta_idx, beta in enumerate(beta_steps):
        if time.monotonic() > deadline:
            break
        phase_deadline = deadline - (len(beta_steps) - beta_idx - 1) * 5.0
        # Estimate iterations per beta phase
        while time.monotonic() < min(deadline, phase_deadline):
            if time.monotonic() > deadline:
                break
            t += 1
            a = _softmax(z)
            c = _self_conv(a)
            cm = float(c.max())
            c_sh = (c - cm) / beta
            e = np.exp(np.minimum(c_sh, 50.0))
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e

            grad = _grad_conv(a, w)
            W = float(np.dot(w, c))
            # Gradient of 2n * logsumexp(beta * c) / beta = 2n * (w @ c + log(sum exp)) ...
            # Actually grad of 2n*log(sum exp(beta*c)) w.r.t. a:
            # = 2n * (1/sum exp) * sum exp * beta * dc/da / beta = 2n * w^T * dc/da
            # which is 2n * grad_conv(a, w)
            # But we also have the /sum(a)^2 term. For normalized a (sum=1 on simplex),
            # we work in z-space where a = softmax(z), so:
            # grad_z = a * (grad_a - sum(a * grad_a))
            grad_a = 2.0 * n * _grad_conv(a, w)
            correction = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - correction)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1**t)
            vh = v_buf / (1.0 - b2**t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 5000 == 0:
                current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _peak_directed(a, deadline, n_rounds=100):
    """Directly reduce elements that contribute most to the convolution peak."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        # Find elements contributing to the peak
        # c[k_star] = sum_i a[i]*a[k_star - i] (valid indices)
        # Contribution of a[i] to peak: a[i] * a[k_star - i] (if valid)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
            # Also if i and k_star-i are the same element
            if i == k_star - i and 0 <= i < n:
                contrib[i] = a[i] * a[i]

        # Also consider: reducing a[i] affects many c[k] values
        # Focus on the top contributors
        top_k = min(n // 4, 256)
        top_indices = np.argsort(contrib)[::-1][:top_k]

        # Reduce top contributors slightly, increase others to maintain sum
        for _ in range(3):  # inner iterations
            if time.monotonic() > deadline:
                break
            # Reduce top contributors by small amount
            reduction = 0.01 * a[top_indices].mean()
            a[top_indices] -= reduction
            a[top_indices] = np.maximum(a[top_indices], 0.0)
            # Redistribute to others
            deficit = reduction * len(top_indices)
            others = np.array([i for i in range(n) if i not in set(top_indices.tolist())])
            if len(others) > 0:
                a[others] += deficit / len(others)

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s

            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            else:
                # Revert if worse
                break

    return best_a, best_score


def _fista_smoothed(a_init, beta, deadline):
    """FISTA on the log-sum-exp smoothed objective."""
    n = len(a_init)
    a_curr = a_init / a_init.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 8.0 * n / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val**2)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _self_conv(y)
        c_max = float(c_y.max())
        c_sh = (c_y - c_max) / beta
        e = np.exp(np.minimum(c_sh, 50.0))
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / s_e

        grad = 2.0 * n * _grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _lbfgs_refine(a_init, deadline, beta):
    """L-BFGS-B refinement in log-space."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax(z)
        c = _self_conv(a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(np.minimum(c_sh, 50.0))
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax(z)
        c = _self_conv(a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(np.minimum(c_sh, 50.0))
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad_a = 2.0 * n * _grad_conv(a, w)
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 0.5:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 20)), 'gtol': 1e-14, 'maxls': 10}
        )
        z_opt = result.x
        a_opt = _softmax(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    return _softmax(best_z), best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)**0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5)**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "double_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0*(t-0.3)**2) + np.exp(-8.0*(t-0.7)**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _low_freq_perturb(a, rng, n_modes=8):
    """Perturb in the low-frequency domain."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    # Add noise to low frequencies
    nfreq = len(fa)
    noise_scale = 0.01
    for mode in range(1, min(n_modes + 1, nfreq)):
        fa[mode] += noise_scale * (rng.standard_normal() + 1j * rng.standard_normal())
        if mode < nfreq:
            fa[nfreq - mode] = np.conj(fa[mode])
    a_new = np.fft.irfft(fa, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

                print(f"  Restart {i} error: {ex}")

    # === Phase 6: L-BFGS Final Polish ===
    print("Phase 6: L-BFGS Polish")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _self_conv(a_pol)
        max_cp = float(c_pol.max())
        b_final = 0.00005 * max_cp

        a_lb, _ = _lbfgs_refine(a_pol, deadline - 10.0, b_final)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f}")

        # One more Adam pass with tiny beta
        if time.monotonic() < deadline - 10:
            c_f = _self_conv(best_seq)
            b_tiny = 0.000005 * float(c_f.max())
            a_fin, _ = _adam_smoothed(best_seq, deadline - 5.0, [b_tiny], lr=0.002)
            sc_fin = _score(a_fin)
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
                print(f"  Final Adam: score={sc_fin:.10f}")

    # === Final check and return ===
    print(f"\nDone: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
