"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA, Adam in log-space, LP-based refinement (greedy water-filling),
and coordinate descent. Inspired by the state-of-the-art LP approach.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, deadline, beta, max_iter, top_k):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

        m_buf *= 0.5
        v_buf *= 0.5

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _greedy_waterfill(a, M, max_passes=20, alpha=0.1):
    """Greedy water-filling: increase coordinates of g to maximize sum(g)
    subject to conv(a, g) <= M and g >= 0.
    Returns g with sum(g) >= 1 (since a is feasible with sum=1).
    """
    n = len(a)
    g = a.copy()
    ag = _full_convolve(a, g)  # = conv(a, g)
    
    for p in range(max_passes):
        # For each coordinate j, compute max increase delta_j
        # delta_j = min_{k: a_k > 0} (M - ag[j+k]) / a_k  for k=0..n-1, j+k < 2n-1
        # Vectorized: for each j, we need min over k of (M - ag[j+k])/a_k
        # This is equivalent to: for each k, the constraint is ag[j+k] + a_k * delta <= M
        # => delta <= (M - ag[j+k]) / a_k
        
        # Compute delta_max for all j simultaneously
        # delta_max[j] = min_k (M - ag[j+k]) / a_k
        # = 1 / max_k a_k / (M - ag[j+k])  ... no that's wrong
        
        # Let's think differently. For each j:
        # delta_max[j] = min over valid (k, l) pairs where l = j+k:
        #   (M - ag[l]) / a[k]
        # = min over k=0..min(n-1, 2n-2-j): (M - ag[j+k]) / a[k] (where a[k] > 0)
        
        # This is a "min of ratios" which doesn't vectorize easily.
        # Alternative: use the fact that if we set delta_j = alpha * (M - ag[j]) / a[0] (roughly),
        # we can just do a simple heuristic.
        
        # Simpler approach: for each j, the binding constraint is the one with
        # the smallest (M - ag[j+k]) / a[k]. 
        # Approximate: use k=0 (the most restrictive since a[0] is typically large)
        # delta_max[j] ≈ (M - ag[j]) / a[0] if a[0] > 0
        
        # Better: compute for a few key k values
        # Actually, let's just do it properly but efficiently.
        
        # For each k, the constraint on delta_j is: delta_j <= (M - ag[j+k]) / a[k]
        # So delta_max[j] = min_k (M - ag[j+k]) / a[k]
        
        # Vectorized approach:
        # Create a matrix D where D[j, k] = (M - ag[j+k]) / a[k] for valid (j,k)
        # Then delta_max[j] = min_k D[j, k]
        
        # But this matrix is n x n which is 16M entries for n=4096. That's fine memory-wise.
        # Actually let's just do it in chunks or use a simpler heuristic.
        
        # Simplest effective approach: 
        # delta_j = alpha * min(1, (M - ag[j]) / max(a, 1e-10)[0])
        # This is a rough but effective heuristic.
        
        # More precise: for each j, find the most restrictive constraint
        # The most restrictive is: min_k (M - ag[j+k]) / a[k]
        # For efficiency, only check k in 0..K for some K
        
        K_check = min(n, 200)  # Only check first K_check values of k
        a_k = a[:K_check]
        mask = a_k > 1e-15
        
        if not np.any(mask):
            break
            
        # For each j, compute min over k (where mask) of (M - ag[j+k]) / a[k]
        # ag has length 2n-1
        # For j in 0..n-1, j+k ranges from j to j+K_check-1
        
        # Vectorized: create ag_ext and compute
        js = np.arange(n)
        ks = np.arange(K_check)[mask]
        
        # For each valid k, the constraint is: delta <= (M - ag[j + k]) / a[k]
        # Take the min over k
        
        # Using broadcasting would be n x len(ks) which could be large
        # Instead, iterate over k (but limit to, say, 50)
        K_eff = min(len(ks), 50)
        ks_eff = ks[:K_eff]
        
        delta_max = np.full(n, np.inf)
        for k in ks_eff:
            # ag[j+k] for j=0..n-1
            idx = js + k
            valid = idx < len(ag)
            if not np.all(valid):
                idx = idx[valid]
                dm = (M - ag[idx]) / a[k]
                # Need to handle the case where some j are invalid
                dm_full = np.full(n, np.inf)
                dm_full[:len(idx)] = dm
                delta_max = np.minimum(delta_max, dm_full)
            else:
                dm = (M - ag[idx]) / a[k]
                delta_max = np.minimum(delta_max, dm)
        
        delta_max = np.maximum(delta_max, 0.0)
        
        # Update g
        g = g + alpha * delta_max
        
        # Update ag = conv(a, g) efficiently: ag += alpha * conv(a, delta_max)
        # Actually, just recompute
        ag = _full_convolve(a, g)
        
        # Check if we can still improve
        if delta_max.max() < 1e-15:
            break
    
    return g


def _lp_refinement_greedy(a_init, deadline, max_iters=30):
    """LP-based refinement using greedy water-filling for direction finding."""
    n = len(a_init)
    a = a_init / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        
        # Find greedy direction
        g = _greedy_waterfill(a, M, max_passes=5, alpha=0.1)
        sum_g = float(g.sum())
        
        if sum_g <= 1.0 + 1e-10:
            break
        
        # Line search
        best_t = 0.0
        best_s = best_score
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            a_new = ((1.0 - t) * a + t * g) / ((1.0 - t) + t * sum_g)
            s = _score_norm(a_new)
            if s < best_s:
                best_s = s
                best_t = t
        
        if best_t == 0.0:
            break
        
        a = ((1.0 - best_t) * a + best_t * g) / ((1.0 - best_t) + best_t * sum_g)
        a = a / a.sum()
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    LP-Greedy iter {it}: t={best_t:.3f}, sum_g={sum_g:.4f}, score={sc:.8f}")
    
    return best_a, best_score


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Cyclic coordinate descent polishing."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            a_i_old = a[i]
            
            if a_i_old < 1e-15:
                continue
            
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]
            
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old
            
            idx_top = np.argsort(c)[-top_k_w:]
            
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5, a_i_old * 2.0,
                           a_i_old * 0.7, a_i_old * 0.3, a_i_old * 1.2]
            
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 10.0 * max(a_i_old, 0.1):
                        t_candidates.append(t_star)
                        t_candidates.append(t_star * 0.8)
                        t_candidates.append(t_star * 1.2)
            
            t_candidates = [max(0.0, min(10.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 10) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            a[i] = best_t
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA, Adam in log-space, LP-based greedy water-filling refinement,
and coordinate descent polishing. Key innovation: greedy water-filling for
direction finding inspired by the state-of-the-art LP approach, plus adaptive
top-K smoothing and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)

    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])

    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())

    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, deadline, beta, max_iter, top_k):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 300 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 80 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 4000 == 0:
                current_lr = max(0.0002, current_lr * 0.93)

        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _greedy_waterfill(a, M, max_passes=10, alpha=0.15):
    """Greedy water-filling: find g >= 0 with conv(a,g) <= M, maximizing sum(g).
    Uses approximate computation for speed.
    """
    n = len(a)
    g = a.copy()
    ag = _full_convolve(a, g)

    for p in range(max_passes):
        # Compute delta_max[j] = min_k (M - ag[j+k]) / a[k] for valid k
        # Only check first K_check values of k for speed
        K_check = min(n, 100)
        a_k = a[:K_check]
        mask = a_k > 1e-15
        ks_valid = np.where(mask)[0]

        if len(ks_valid) == 0:
            break

        # Limit to most restrictive k values (largest a_k)
        K_eff = min(len(ks_valid), 30)
        ks_eff = ks_valid[:K_eff]

        delta_max = np.full(n, np.inf)
        len_ag = len(ag)

        for k in ks_eff:
            # For j = 0..n-1: ag[j+k] must be valid (j+k < len_ag)
            max_j = min(n, len_ag - k)
            if max_j <= 0:
                continue
            dm = (M - ag[:max_j + k]) / a[k]
            # dm[j] = (M - ag[j+k]) / a[k] for j = 0..max_j-1
            # Actually ag[j+k] for j in range(max_j)
            dm_vals = (M - ag[k:k + max_j]) / a[k]
            if max_j < n:
                dm_full = np.full(n, np.inf)
                dm_full[:max_j] = dm_vals
                delta_max = np.minimum(delta_max, dm_full)
            else:
                delta_max = np.minimum(delta_max, dm_vals)

        delta_max = np.maximum(delta_max, 0.0)

        if delta_max.max() < 1e-18:
            break

        # Update with step size
        g_new = g + alpha * delta_max
        # Check feasibility (approximately)
        ag_new = _full_convolve(a, g_new)
        if ag_new.max() > M * 1.01:
            # Reduce step
            g_new = g + alpha * 0.5 * delta_max
            ag_new = _full_convolve(a, g_new)
            if ag_new.max() > M * 1.02:
                break

        g = g_new
        ag = ag_new

    return g


def _lp_direction(a, M, time_limit=10.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns g or None if LP fails or times out.
    """
    n = len(a)
    if n > 8192:
        return None

    # Build constraint matrix efficiently
    rows_list = []
    cols_list = []
    data_list = []

    for l in range(2 * n - 1):
        j_min = max(0, l - n + 1)
        j_max = min(n - 1, l)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            ks = l - js
            rows_list.extend([l] * len(js))
            cols_list.extend(js.tolist())
            data_list.extend(a[ks].tolist())

    A_mat = coo_matrix((data_list, (rows_list, cols_list)),
                       shape=(2 * n - 1, n)).tocsr()

    try:
        res = linprog(c=-np.ones(n), A_ub=A_mat, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def _lp_refinement(a_init, deadline, max_iters=25, use_lp=False):
    """LP or greedy-based refinement using the state-of-the-art approach."""
    n = len(a_init)
    a = a_init / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iters):

            break

        M = float(c.max())

        g = None
        if use_lp and n <= 4096:
            time_left = deadline - time.monotonic() - 5
            if time_left > 3:
                g = _lp_direction(a, M, time_limit=min(15.0, time_left))

        if g is None or not np.all(np.isfinite(g)):
            g = _greedy_waterfill(a, M, max_passes=5, alpha=0.12)

        sum_g = float(g.sum())

        best_t = 0.0
        best_s = best_score
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            denom = (1.0 - t) + t * sum_g
            if denom < 1e-10:
                continue
            a_new = ((1.0 - t) * a + t * g) / denom
            s = _score_norm(a_new)
            if s < best_s:
                best_s = s
                best_t = t

        if best_t == 0.0:
            break

        a = ((1.0 - best_t) * a + best_t * g) / ((1.0 - best_t) + best_t * sum_g)
        a = a / a.sum()

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if it % 5 == 0:
                print(f"    LP-refine iter {it}: t={best_t:.3f}, sum_g={sum_g:.4f}, score={sc:.8f}")

    return best_a, best_score


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Cyclic coordinate descent polishing with sparse convolution updates."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    rng = np.random.default_rng(98765)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            a_i_old = a[i]

            if a_i_old < 1e-15:
                continue

            c = _full_convolve(a, a)

            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]

            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old

            idx_top = np.argsort(c)[-top_k_w:]

            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5,
                            a_i_old * 2.0, a_i_old * 0.7, a_i_old * 0.3,
                            a_i_old * 1.2, a_i_old * 0.1]

            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 10.0 * max(a_i_old, 0.1):
                        t_candidates.append(t_star)
                        t_candidates.append(t_star * 0.8)
                        t_candidates.append(t_star * 1.2)
                        t_candidates.append(t_star * 0.5)

            t_candidates = [max(0.0, min(5.0, tc)) for tc in t_candidates]
            # Deduplicate
            t_candidates = sorted(set([round(tc, 8) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = 2.0 * n * c.max()
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            a[i] = best_t
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.02, 0.98, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-25 * (t - 0.3) ** 2) + np.exp(-25 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "ripple":
        t = np.arange(n) / n
        a = 1.0 + 0.15 * np.cos(2 * np.pi * 7 * t) + 0.1 * np.cos(2 * np.pi * 13 * t)
        a = np.maximum(a, 0.05)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Determine lengths to try
    n_parent = len(parent)
    if n_parent == 4096:
        lengths_to_try = [4096, 8192, 2048]
    elif n_parent == 8192:
        lengths_to_try = [8192, 4096]
    else:
        lengths_to_try = [4096, 2048]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 180:
            break

        remaining = deadline - time.monotonic()
        print(f"\n=== n={n_try} (remaining: {remaining:.0f}s) ===")

        if n_try == n_parent:
            a0 = parent.copy()
        else:
            a0 = _interp_profile(parent, n_try)

        sc0 = _score(a0)
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()
        print(f"  Start: {sc0:.10f}")

        # Phase A: FISTA coarse (15% of remaining)
        rem = deadline - time.monotonic()
        phase_a_end = time.monotonic() + min(rem * 0.15, 150.0)

        c_curr = _full_convolve(a0, a0)
        max_cc = float(c_curr.max())

        betas_fista = []
        beta = 0.015 * max_cc
        while beta > 0.00001 * max_cc:
            betas_fista.append(beta)
            beta *= 0.5
        if len(betas_fista) > 5:
            betas_fista = betas_fista[:5]

        a_curr = a0.copy()
        for bi, beta in enumerate(betas_fista):
            if time.monotonic() > phase_a_end - 2:
                break
            per_t = phase_a_end - time.monotonic() - 1
            max_it = max(500, int(per_t * 1500))
            top_k = max(20, min(120, int(20 + bi * 25)))
            a_curr, sc = _fista_smoothed(a_curr, time.monotonic() + min(25, per_t),
                                         beta, max_it, top_k)
            a_curr = a_curr / a_curr.sum()
            sc_full = _score(a_curr)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_curr.copy()
                print(f"  FISTA b={beta:.6f} k={top_k}: {sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

        # Phase B: Adam log-space fine (30% of remaining)
        rem = deadline - time.monotonic()
        phase_b_end = time.monotonic() + min(rem * 0.30, 350.0)

        c_curr = _full_convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())

        betas_fine = []
        beta = 0.0015 * max_cc
        while beta > 0.000001 * max_cc:
            betas_fine.append(beta)
            beta *= 0.45
        if len(betas_fine) > 12:
            betas_fine = betas_fine[:12]

        a_opt, sc_opt = _adam_logspace(a_curr, phase_b_end, betas_fine,
                                       lr_init=0.01, top_k=60,
                                       max_iter=200000)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam fine: {sc_opt_full:.10f} t={time.monotonic()-t_start:.0f}s")

        a_work = a_opt / a_opt.sum()

        # Phase C: LP-Greedy refinement (20% of remaining)
        rem = deadline - time.monotonic()
        phase_c_end = time.monotonic() + min(rem * 0.20, 200.0)
        if phase_c_end - time.monotonic() > 15:
            a_lp, sc_lp = _lp_refinement(a_work, phase_c_end, max_iters=20, use_lp=False)
            sc_lp_full = _score(a_lp)
            if sc_lp_full < best_score:
                best_score = sc_lp_full
                best_seq = a_lp.copy()
                print(f"  LP-Greedy: {sc_lp_full:.10f} t={time.monotonic()-t_start:.0f}s")
                a_work = a_lp

        # Phase D: Coordinate descent (10% of remaining)
        rem = deadline - time.monotonic()
        phase_d_end = time.monotonic() + min(rem * 0.10, 100.0)
        if phase_d_end - time.monotonic() > 10:
            a_cd, sc_cd = _coord_descend_polish(a_work, n_passes=3,
                                                top_k_w=40, deadline=phase_d_end)
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
                print(f"  CoordCD: {sc_cd_full:.10f} t={time.monotonic()-t_start:.0f}s")
                a_work = a_cd

        # Phase E: Ultra-fine Adam (8% of remaining)
        rem = deadline - time.monotonic()
        if rem > 20:
            phase_e_end = time.monotonic() + min(rem * 0.08, 60.0)
            c_work = _full_convolve(a_work, a_work)
            max_cw = float(c_work.max())
            betas_ult = [0.0002 * max_cw, 0.00008 * max_cw, 0.00003 * max_cw]
            a_ult, sc_ult = _adam_logspace(a_work, phase_e_end, betas_ult,
                                           lr_init=0.003, top_k=20,
                                           max_iter=30000)
            sc_ult_full = _score(a_ult)
            if sc_ult_full < best_score:
                best_score = sc_ult_full
                best_seq = a_ult.copy()
                print(f"  UltraAdam: {sc_ult_full:.10f} t={time.monotonic()-t_start:.0f}s")

    # Phase F: Diversity restarts from best
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Restarts (rem: {deadline - time.monotonic():.0f}s) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()

        n_restarts = 3
        per_restart = max(5.0, (deadline - time.monotonic() - 20) / n_restarts)

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if ri == 0:
                p_start = a_base + 0.008 * rng.standard_normal(n_best)
            elif ri == 1:
                shift = rng.integers(1, max(2, n_best // 1024))
                p_start = np.roll(a_base, shift).copy()
            else:
                freq = rng.uniform(2.0, 8.0)
                pert = 0.015 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.0008 * max_cs, 0.0003 * max_cs, 0.0001 * max_cs, 0.00004 * max_cs]

            p_best, _ = _adam_logspace(p_start, time.monotonic() + per_restart,
                                       betas_r, lr_init=0.006, top_k=40,
                                       max_iter=20000)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {ri}: {sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    # Final validation and output
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"RETURN: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
