    grad = np.zeros(n)
    for j in top_idx:
        lo = max(0, j - n + 1)
        hi = min(n - 1, j)
        if lo > hi:
            continue
        m_range = np.arange(lo, hi + 1)
        idx = j - m_range
        grad[m_range] += a[idx]
        if j % 2 == 0:
            mid = j // 2
            if 0 <= mid < n and lo <= mid <= hi:
                grad[mid] += a[mid]
    grad /= float(K)
    return grad, c, cmax


def _fista_topK(a_init, K_schedule, deadline, max_iter, step_scale=1.0):
    """FISTA with exact top-K subgradient and Nesterov acceleration."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    # Estimate Lipschitz constant
    g0, c0, cm0 = _topK_subgrad(a_curr, K_schedule[0])
    L = max(2.0 * np.linalg.norm(g0), 1e-10)
    step = step_scale / L

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Determine K for this iteration
        progress = it / max(1, max_iter)
        K_idx = min(int(progress * len(K_schedule)), len(K_schedule) - 1)
        K = K_schedule[K_idx]

        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        # Compute subgradient at y
        grad, c_y, cmax_y = _topK_subgrad(y, K)

        # Gradient step
        a_next = _proj_simplex(y - step * grad)

        # Adaptive step: if progress is slow, reduce step
        sc_next = _score_norm(a_next)
        if sc_next > best_sc * 1.05 and it > 100:
            step *= 0.7
        elif it % 200 == 0 and sc_next < best_sc:
            step *= 1.1
            if sc_next < best_sc:
                best_sc = sc_next
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

        # Nesterov restart if function value increases
        if it > 50 and sc_next > _score_norm(a_prev) * 1.01:
            t_val = 1.0
            a_prev = a_curr.copy()

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _kl_mirror_descent(a_init, deadline, max_iter, K_init=50, K_final=10, lr_init=0.02):
    """KL-based mirror descent in log-space with top-K subgradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        progress = it / max(1, max_iter)
        K = max(K_final, int(K_init * (1 - progress) + K_final * progress))
        lr = lr_init / (1 + 0.05 * progress)

        a = np.exp(z - z.max())
        a = a / a.sum()

        grad, c, cmax = _topK_subgrad(a, K)

        # Adam update
        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        # Clip update for stability
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)

        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if it % 30 == 0:
            a_chk = np.exp(z - z.max())
            a_chk = a_chk / a_chk.sum()
            sc = _score_norm(a_chk)
            if sc < best_sc:
                best_sc = sc
                best_a = a_chk.copy()

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coord_descent_focused(a_init, deadline, max_rounds=300, top_frac=0.05):
    """Coordinate descent focusing on coordinates that affect top convolution positions."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(999)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _convolve(a, a)
        cmax = float(c.max())
        # Find coordinates most affected by top convolution positions
        K = max(10, int(0.1 * len(c)))
        top_pos = np.argsort(c)[::-1][:K]

        # Determine which a_j values are involved
        affected = set()
        for j in top_pos:
            lo = max(0, j - n + 1)
            hi = min(n - 1, j)
            affected.update(range(lo, hi + 1))
        affected = np.array(list(affected))

        if len(affected) > n // 2:
            affected = np.arange(n)

        # Random subset for large n
        if len(affected) > 1024:
            selected = rng.choice(affected, size=512, replace=False)
        else:
            selected = affected.copy()
        rng.shuffle(selected)

        improved = False
        for j in selected:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.95, 0.98, 1.02, 1.05]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved and r > 30:
            break
    return best_a, best_sc


def _peak_reduction(a_init, deadline, n_peaks=20):
    """Explicitly try to reduce the top peaks of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    for it in range(5000):
        if time.monotonic() > deadline:
            break
        c = _convolve(a, a)
        cmax = float(c.max())
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # For each top peak position j, try reducing it by adjusting a_j
        for j in top_idx:
            if time.monotonic() > deadline:
                break
            lo = max(0, j - n + 1)
            hi = min(n - 1, j)
            # Pick a random coordinate in the affected range
            m = rng.integers(lo, hi + 1)
            old_val = a[m]
            # Try reducing
            for factor in [0.9, 0.95, 0.99]:
                a[m] = old_val * factor
                s = a.sum()
                if s < 1e-30:
                    a[m] = old_val
                    continue
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    break
                else:
                    a[m] = old_val

        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()


def _quick_optimize(a_init, deadline, K_schedule=None, max_iter=10000):
    """Quick optimization combining FISTA and KL descent."""
    n = len(a_init)
    if K_schedule is None:
        c0 = _convolve(a_init, a_init)
        cm0 = float(c0.max())
        K_schedule = [min(100, n), 50, 20, 10, 5]

    a, sc = _fista_topK(a_init, K_schedule, deadline, max_iter, step_scale=1.0)
    a = a / a.sum()
    return a, sc
