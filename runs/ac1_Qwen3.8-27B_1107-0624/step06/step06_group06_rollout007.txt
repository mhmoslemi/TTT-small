import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return np.maximum(r, 0.0)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_subgrad(z, a, c, n, tol_frac=0.002)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.9
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _conjugate_gradient(a_init, deadline, max_iter=5000):
    """Fletcher-Reeves CG on the simplex with log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    c_cur = _full_convolve(a_init, a_init)
    beta = max(float(c_cur.max()) * 0.01, 1e-6)
    beta_decay = 0.85
    beta_min = 1e-5

    g_old = _smograd(z, beta, n)
    d = -g_old.copy()

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        if i % 50 == 0:
            beta = max(beta * beta_decay, beta_min)

        a = _softmax(z)
        g_new = _smograd(z, beta, n)
        gnorm = np.linalg.norm(g_new)
        if gnorm < 1e-12:
            break

        # Wolfe line search
        f_k = _smoobj(z, beta, n)
        c1 = 1e-4
        c2 = 0.9
        alpha = 1.0
        dg = np.dot(g_new, d)
        if dg > 0:
            d = -g_new.copy()
            dg = -gnorm * gnorm

        for ls in range(40):
            z_try = z + alpha * d
            z_try = np.clip(z_try, -60.0, 60.0)
            z_try = z_try - z_try.mean()
            f_try = _smoobj(z_try, beta, n)
            g_try = _smograd(z_try, beta, n)
            dg_try = np.dot(g_try, d)
            if f_try <= f_k + c1 * alpha * dg and dg_try >= c2 * dg:
                break
            alpha *= 0.5

        z = z + alpha * d
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()

        # Fletcher-Reeves update
        g_new2 = _smograd(z, beta, n)
        gamma = np.dot(g_new2, g_new2) / max(np.dot(g_old, g_old), 1e-30)
        d = -g_new2 + gamma * d
        g_old = g_new2

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered3":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * t**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _spectral_init(n, n_peaks=5):
    """Initialize by designing a flat-magnitude spectrum in frequency domain."""
    N = _next_pow2(2 * n - 1)
    # Target: flat spectrum |A(k)|^2 across all frequencies
    target = np.ones(N // 2 + 1)
    # Force the signal to be zero after position n-1
    # This means A(k) must satisfy certain constraints
    # Instead, just create a random phase and inverse FFT
    phases = np.random.uniform(0, 2 * np.pi, N // 2 + 1)
    A = target * np.exp(1j * phases)
    A[0] = np.abs(A[0])  # make DC real and positive
    a_full = np.fft.irfft(A, N)
    a = a_full[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return _make_profile(n, "tapered")
    return a / s


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.5, 2.5)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_blocks=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _pinch_peaks(a, deadline, n_iter=200):
    """Actively try to reduce the top few convolution peaks."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find top 5 peaks
        top_k = 5
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:top_k]
        # Compute gradient pointing away from these peaks
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / top_k
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Step in negative gradient direction
        lr = 0.02 / (1.0 + 0.01 * i)
        z = z - lr * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 10 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with aggressive adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init.copy()

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    # More aggressive and longer beta schedule
    betas = base_beta * np.array([
        1.0, 0.6, 0.35, 0.2, 0.11, 0.06, 0.035,
        0.02, 0.011, 0.006, 0.0035, 0.002, 0.0011, 0.0006, 0.0003
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        # Estimate iterations affordable
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 8))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # Try CG from current point
                if remaining > 10:
                    a_cg, sc_cg = _conjugate_gradient(a_new,
                                                       min(deadline - min_time_remaining,
                                                           time.monotonic() + remaining * 0.3),
                                                       max_iter=min(2000, max(100, int(remaining / 0.01))))
                    if sc_cg < current_sc - 1e-14:
                        current_a = a_cg
                        current_sc = sc_cg
                        z = np.log(np.maximum(a_cg, 1e-30))
                        z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage(a, deadline):
    """Try zeroing out the smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.05, 0.10, 0.15]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: L-BFGS + CG + spectral init + multi-n (up to 16384)
    + pinching + diverse restarts + shrinkage.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Main optimization from parent ===
    print(f"\n[Phase 1] Main L-BFGS+CG from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=500.0)
    update_best(a_main, _score(a_main), "main_lbfgs")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 400:
        sg_deadline = min(deadline - 400, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=3000, lr_init=0.02)
        update_best(a_sg, sc_sg, "sg_polish_1")

    # Pinch peaks
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_pinch, sc_pinch = _pinch_peaks(best_seq, min(deadline - 350, time.monotonic() + 80))
        update_best(a_pinch, sc_pinch, "pinch_1")

    # === Phase 2: Try different n values ===
    print(f"\n[Phase 2] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    # Priority: try larger n first (more degrees of freedom)
    n_candidates = [8192, 16384, 4096, 2048]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = list(dict.fromkeys(n_candidates))  # dedupe preserving order

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic() - 100
        if remaining < 30:
            break

        # Initialize
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent, left=0, right=0)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init < 1e-10:
                a_init = _make_profile(n_try, "tapered")
            else:
                a_init = a_init / s_init
        else:
            a_init = parent.copy()

        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.10f}, budget={remaining:.1f}s")

        # Run L-BFGS
        alloc = min(remaining * 0.6, 300.0)
        a_opt = _run_lbfgs(a_init, t_start + (time.monotonic() - t_start) + alloc, min_time_remaining=5.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        # CG polish
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 150:
            a_cg, sc_cg = _conjugate_gradient(a_opt, min(deadline - 150, time.monotonic() + 60), max_iter=2000)
            update_best(a_cg, sc_cg, f"n_{n_try}_cg")

        # Subgradient
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 100:
            a_sg2, sc_sg2 = _subgrad_polish(a_opt, min(deadline - 100, time.monotonic() + 40), n_max_steps=1000, lr_init=0.015)
            update_best(a_sg2, sc_sg2, f"n_{n_try}_sg")

        # Pinch
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 80:
            a_p2, sc_p2 = _pinch_peaks(a_opt, min(deadline - 80, time.monotonic() + 30))
            update_best(a_p2, sc_p2, f"n_{n_try}_pinch")

    # === Phase 3: Spectral and diverse initializations ===
    print(f"\n[Phase 3] Spectral and diverse inits (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        # Try spectral init at best n
        n_best = len(best_seq) if best_seq is not None else 4096
        for init_kind in ["spectral", "tapered2", "raised_cosine", "gaussian"]:
            if time.monotonic() > deadline - 80:
                break
            if init_kind == "spectral":
                a_init = _spectral_init(n_best, n_peaks=3)
            else:
                a_init = _make_profile(n_best, init_kind)
            sc_init = _score(a_init)
            alloc = min(deadline - time.monotonic() - 60, 120.0)
            if alloc < 15:
                continue
            a_opt = _run_lbfgs(a_init, t_start + (time.monotonic() - t_start) + alloc, min_time_remaining=5.0)
            update_best(a_opt, _score(a_opt), f"init_{init_kind}")

    # === Phase 4: Diverse restarts from best ===
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restarts_configs = [
            (0.04, "gauss"),
            (0.06, "fourier"),
            (0, "block"),
            (0.08, "gauss"),
            (0.03, "fourier"),
        ]
        for i, (sigma, ptype) in enumerate(restarts_configs):
            if time.monotonic() > deadline - 30:
                break
            if ptype == "gauss":
                p_start = _perturb(best_seq, sigma, rng)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            else:
                p_start = _perturb_block_swap(best_seq, rng, n_blocks=4)

            alloc = min(deadline - time.monotonic() - 20, 80.0)
            if alloc < 10:
                continue
            a_r = _run_lbfgs(p_start, t_start + (time.monotonic() - t_start) + alloc, min_time_remaining=5.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}_{ptype}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _subgrad_polish(a_r, min(deadline - 10, time.monotonic() + 25), n_max_steps=500, lr_init=0.01)
                update_best(a_r2, sc_r2, f"restart_{i}_sg")

    # === Phase 5: Shrinkage ===
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_shr, sc_shr = _shrinkage(best_seq, min(deadline - 25, time.monotonic() + 30))
        update_best(a_shr, sc_shr, "shrinkage")

    # === Phase 6: Final fine polish ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very fine L-BFGS with tiny beta
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0002
        remaining = deadline - 3 - time.monotonic()
        if remaining > 3:
            max_iter = min(3000, max(100, int(remaining / 0.03)))
            try:
                result = minimize(
                    lambda z_local: _smoobj(z_local, beta_final, n),
                    z,
                    jac=lambda z_local: _smograd(z_local, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2, n_max_steps=500, lr_init=0.006)
            update_best(a_fin2, sc_fin2, "final_sg")

        # Final pinch
        if time.monotonic() < deadline - 1:
            a_fin3, sc_fin3 = _pinch_peaks(best_seq, deadline - 1, n_iter=100)
            update_best(a_fin3, sc_fin3, "final_pinch")

    # --- Final validation and return ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
