"""
Structural optimizer for minimum peak self-convolution sequences.
Uses L-BFGS-B in log-softmax space with log-sum-exp smoothing,
structured simulated annealing, and multi-scale exploration.
Targets sub-1.5027 score.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _logsumexp_smooth(c, tau):
    c_shifted = (c - c.max()) / tau
    np.clip(c_shifted, -500, 500, out=c_shifted)
    return c.max() + tau * np.log(np.exp(c_shifted).sum())

def _objective_and_grad(z, tau):
    n = len(z)
    a = _softmax(z)
    c = _convolve(a, a)
    val = 2.0 * n * _logsumexp_smooth(c, tau)
    c_shifted = (c - c.max()) / tau
    np.clip(c_shifted, -500, 500, out=c_shifted)
    w = np.exp(c_shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / ws
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    grad_a = 2.0 * wc[n - 1:2 * n - 1]
    grad_z = 2.0 * n * a * (grad_a - np.dot(grad_a, a))
    return val, grad_z

def _lbfgs_optimize(a_init, tau, max_iter, max_fun):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_z = z.copy()
    best_val = _objective_and_grad(z, tau)[0]

    def fun(gz):
        val, _ = _objective_and_grad(gz, tau)
        return val

    def fun_and_grad(gz):
        return _objective_and_grad(gz, tau)

    res = minimize(fun_and_grad, z, method='L-BFGS-B',
                   jac=True, options={'maxiter': max_iter, 'maxfun': max_fun,
                                      'gtol': 1e-10, 'ftol': 1e-14})
    if res.fun < best_val:
        best_z = res.x.copy()
        best_val = res.fun
    return _softmax(best_z), best_val

def _structured_candidates(n, rng):
    cands = []
    # Power law (symmetric)
    i = np.arange(n, dtype=np.float64)
    a1 = 1.0 / np.sqrt(np.maximum(i + 1, 1) * np.maximum(n - i, 1))
    a1 = np.maximum(a1, 1e-10)
    cands.append(a1 / a1.sum())
    # Sine window with various powers
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.sin(np.pi * (i + 0.5) / n) ** p
        a = np.maximum(a, 1e-10)
        cands.append(a / a.sum())
    # Hann window
    a = 0.5 * (1 + np.cos(2 * np.pi * i / n))
    a = np.maximum(a, 1e-10)
    cands.append(a / a.sum())
    # Raised cosine variant
    a = np.cos(np.pi * (i - (n-1)/2) / n) ** 2
    a = np.maximum(a, 1e-10)
    cands.append(a / a.sum())
    # Flat-top approximation
    cutoff = 0.3
    a = np.ones(n)
    edge = int(cutoff * n)
    for j in range(edge):
        frac = j / edge
        a[j] = 1.0 - 0.5 * (1 - np.cos(np.pi * frac))
        a[n-1-j] = a[j]
    cands.append(a / a.sum())
    # Two-sided decay from center
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / (n * 0.2)) ** 2)
    cands.append(a / a.sum())
    # Broad Gaussian
    a = np.exp(-0.5 * ((i - center) / (n * 0.4)) ** 2)
    cands.append(a / a.sum())
    # Uniform
    cands.append(np.ones(n) / n)
    return cands

def _sa_step(a, rng, move_type):
    n = len(a)
    work = a.copy()
    if move_type == 0:
        # Random block replacement with smooth profile
        block_len = rng.integers(8, max(9, n // 16))
        start = rng.integers(0, n - block_len)
        profile = rng.choice(['linear', 'quadratic', 'gaussian', 'sine', 'constant'])
        block = work[start:start + block_len].copy()
        scale = block.mean()
        t = np.linspace(0, 1, block_len)
        if profile == 'linear':
            new_block = (1 - t) + 0.5 * t
        elif profile == 'quadratic':
            new_block = (1 - t) ** 2 + 0.1
        elif profile == 'gaussian':
            new_block = np.exp(-2 * ((t - 0.5) * 3) ** 2) + 0.05
        elif profile == 'sine':
            new_block = np.sin(np.pi * t) ** 0.5 + 0.1
        else:
            new_block = np.ones(block_len) * (1 + 0.3 * rng.standard_normal())
        new_block = np.maximum(new_block, 0.0)
        if new_block.sum() > 0:
            new_block = scale * new_block / new_block.sum()
        work[start:start + block_len] = new_block
    elif move_type == 1:
        # Gaussian perturbation in log-space on a random block
        block_len = rng.integers(4, max(5, n // 32))
        start = rng.integers(0, n - block_len)
        sigma = 0.15
        log_block = np.log(np.maximum(work[start:start + block_len], 1e-15))
        log_block += sigma * rng.standard_normal(block_len)
        log_block = np.clip(log_block, -30, 30)
        exp_block = np.exp(log_block - log_block.max())
        work[start:start + block_len] = np.maximum(exp_block, 0.0)
    elif move_type == 2:
        # Shift a sub-block
        block_len = rng.integers(16, max(17, n // 8))
        start = rng.integers(0, n - block_len)
        shift = rng.integers(1, block_len // 2)
        work[start:start + block_len] = np.roll(work[start:start + block_len], shift)
    elif move_type == 3:
        # Reflect a sub-block
        block_len = rng.integers(16, max(17, n // 8))
        start = rng.integers(0, n - block_len)
        work[start:start + block_len] = work[start:start + block_len][::-1]
    elif move_type == 4:
        # Interpolate between two random existing blocks
        block_len = rng.integers(8, max(9, n // 16))
        start = rng.integers(0, n - block_len)
        src_start = rng.integers(0, n - block_len)
        if src_start == start:
            src_start = (src_start + 1) % max(1, n - block_len)
        alpha = rng.uniform(0.2, 0.8)
        work[start:start + block_len] = (1 - alpha) * work[start:start + block_len] + alpha * work[src_start:src_start + block_len]
    else:
        # Single element multiplicative perturbation
        idx = rng.integers(0, n)
        work[idx] *= rng.uniform(0.7, 1.3)
    work = np.maximum(work, 0.0)
    ws = work.sum()
    if ws < 1e-15:
        return a.copy()
    return work / ws

def _simulated_annealing(a_init, deadline, rng):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    curr_score = best_score

    t0 = time.monotonic()
    total_time = max(1.0, deadline - t0)
    n_iterations = int(total_time * 800)  # ~800 iterations per second estimate
    t_init = 0.002
    t_final = 0.0001
    cool_rate = (t_final / t_init) ** (1.0 / max(1, n_iterations - 1))

    temp = t_init
    no_improve = 0
    patience = 5000

    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break
        move_type = rng.integers(0, 6)
        a_new = _sa_step(a, rng, int(move_type))
        sc_new = _score_norm(a_new)

        if sc_new < curr_score:
            a = a_new
            curr_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            delta = sc_new - curr_score
            accept_prob = np.exp(-delta / temp)
            if rng.random() < accept_prob:
                a = a_new
                curr_score = sc_new
                no_improve += 1

        temp *= cool_rate

        if no_improve > patience and temp < 0.0005:
            # Restart from best with more aggressive moves
            a = _sa_step(best_a, rng, int(rng.integers(0, 5)))
            curr_score = _score_norm(a)
            no_improve = 0
            patience = 3000

    return best_a, _score_norm(best_a)

def _coord_descent_polish(a_init, deadline, rng):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    factors = [0.95, 0.98, 0.99, 1.01, 1.02, 1.05]

    for rnd in range(2000):
        if time.monotonic() > deadline:
            break
        improved = False
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for f in factors:
                new_val = old_val * f
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-15:
                    a[j] = old_val
                    continue
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    old_val = new_val
                else:
                    a[j] = old_val
                    break
        a = a / a.sum()
        if not improved and rnd > 20:
            # Try larger perturbations
            for _ in range(10):
                j = rng.integers(0, n)
                f = rng.uniform(0.5, 2.0)
                a[j] *= f
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    sc = _score_norm(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
            if not improved:
                break

    return best_a, _score_norm(best_a)

def _interp_to_n(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0, endpoint=False)
    x1 = np.linspace(0, 1, n_new, endpoint=False)
    a_scaled = a * np.sqrt(n0)
    a_new = np.interp(x1, x0, a_scaled)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()

    print(f"Phase 3: Simulated annealing (best={best_score:.10f})")
    phase3_deadline = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_deadline - 60:
        sa_deadline = min(phase3_deadline, time.monotonic() + 600)
        a_sa, sc_sa = _simulated_annealing(a_work, sa_deadline, rng)
        sc_sa_full = _score(a_sa)
        print(f"  SA: score={sc_sa_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_sa_full < best_score:
            best_score = sc_sa_full
            best_seq = a_sa.copy()
            a_work = a_sa.copy()

    # Phase 4: L-BFGS-B polish from SA result
    if time.monotonic() < deadline - 200:
        print("Phase 4: L-BFGS-B polish")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        c_work = _convolve(a_work, a_work)
        max_c = float(c_work.max())
        tau_levels2 = [0.01 * max_c, 0.001 * max_c, 0.0001 * max_c, 0.00001 * max_c]
        for tau in tau_levels2:
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - 200 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(300, int(remaining * 100))
            max_fun = min(1000, int(remaining * 300))
            if max_iter < 10:
                break
            a_opt, val = _lbfgs_optimize(a_work, tau, max_iter, max_fun)
            sc = _score(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                a_work = a_opt.copy()
            print(f"  Polish tau={tau:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Multi-restart SA from different perturbations
    print("Phase 5: Multi-restart SA")
    phase5_deadline = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_deadline - 120:
        n_restarts = 4
        per_restart = (phase5_deadline - time.monotonic() - 60) / n_restarts
        for ri in range(n_restarts):
            if time.monotonic() > phase5_deadline - 30:
                break
            if per_restart < 20:
                break
            # Different perturbation strategies
            if ri == 0:
                a_r = _sa_step(a_work, rng, 0)
                a_r = _sa_step(a_r, rng, 1)
            elif ri == 1:
                a_r = a_work.copy()
                z = np.log(np.maximum(a_r, 1e-15))
                z += 0.05 * rng.standard_normal(len(z))
                z = z - z.mean()
                a_r = _softmax(z)
            elif ri == 2:
                a_r = a_work.copy()
                shift = rng.integers(1, max(2, n_main // 64))
                a_r = np.roll(a_r, shift).copy()
            else:
                a_r = a_work.copy()
                block = n_main // 4
                a_r[:block] = a_r[:block][::-1]
                a_r[block:2*block] = np.roll(a_r[block:2*block], rng.integers(1, block))

            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            sc_r = _score(a_r)

            # Quick SA on this restart
            sa_d = time.monotonic() + max(5.0, per_restart * 0.7)
            a_sr, sc_sr = _simulated_annealing(a_r, sa_d, rng)
            sc_sr_full = _score(a_sr)
            # Quick L-BFGS polish
            if sc_sr_full < best_score + 0.01 and time.monotonic() < phase5_deadline - 10:
                c_sr = _convolve(a_sr, a_sr)
                max_cr = float(c_sr.max())
                a_sr, _ = _lbfgs_optimize(a_sr, 0.001 * max_cr, 200, 500)
                sc_sr_full = _score(a_sr)

            if sc_sr_full < best_score:
                best_score = sc_sr_full
                best_seq = a_sr.copy()
                print(f"  Restart {ri}: score={sc_sr_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 6: Try n*2
    print("Phase 6: Larger n exploration")
    phase6_deadline = min(deadline, t_start + 1680.0)
    if time.monotonic() < phase6_deadline - 80 and best_seq is not None:
        n_big = min(n_main * 2, 16384)
        a_big = _interp_to_n(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  n={n_big} interpolated: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < phase6_deadline - 40:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            # Quick L-BFGS on big
            tau_big = 0.01 * max_cb
            a_big_opt, _ = _lbfgs_optimize(a_big, tau_big, 300, 1000)
            sc_big_opt = _score(a_big_opt)
            if sc_big_opt < best_score:
                best_score = sc_big_opt
                best_seq = a_big_opt.copy()
            print(f"  n={n_big} optimized: score={sc_big_opt:.10f} t={time.monotonic()-t_start:.1f}s")

            tau_big2 = 0.001 * max_cb
            a_big_opt, _ = _lbfgs_optimize(a_big_opt, tau_big2, 300, 800)
            sc_big_opt = _score(a_big_opt)
            if sc_big_opt < best_score:
                best_score = sc_big_opt
                best_seq = a_big_opt.copy()
            print(f"  n={n_big} optimized2: score={sc_big_opt:.10f}")

    # Phase 7: Final coordinate descent polish
    print("Phase 7: Final coordinate descent")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 10, time.monotonic() + 120)
        a_cd, sc_cd = _coord_descent_polish(a_cd, cd_deadline, rng)
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # Phase 8: Ultra-fine final L-BFGS
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_fin = best_seq.copy()
        a_fin = a_fin / a_fin.sum()
        c_fin = _convolve(a_fin, a_fin)
        max_cf = float(c_fin.max())
        fin_deadline = deadline - 8
        per_tau = max(3.0, (fin_deadline - time.monotonic()) / 3 - 1)
        for frac in [0.0001, 0.00001, 0.000001]:
            tau_f = frac * max_cf
            if time.monotonic() > fin_deadline - 2:
                break
            a_fin, _ = _lbfgs_optimize(a_fin, tau_f, 200, 500)
            sc_f = _score(a_fin)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_fin.copy()
        print(f"  Final L-BFGS: score={best_score:.10f}")

    # Final validation
    print(f"\nAll phases done: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
