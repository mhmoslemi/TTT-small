"""
Self-Consistent Field (SCF) optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses alternation between:
  - Computing dual variables q = softmax(conv(a,a)/beta)
  - Solving the inner quadratic minimization a^T Q(q) a over the simplex
via conjugate gradient with FFT-based Hankel matrix-vector products.
Followed by Adam polishing and multi-restart exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.ones(n) * (target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _hankel_matvec(q, x):
    """Compute Q @ x where Q_{i,j} = q[i+j] (Hankel matrix, q has length 2n-1)."""
    n = len(x)
    # (Q x)_i = sum_j q[i+j] * x_j
    # = conv(q, x[::-1])[n-1 + i]
    N = _next_pow2(len(q) + n - 1)
    fq = np.fft.rfft(q, N)
    fx = np.fft.rfft(x[::-1], N)
    r = np.fft.irfft(fq * fx, N)
    return r[n-1:2*n-1].copy()

def _hankel_matvec_transpose(q, x):
    """Same as _hankel_matvec since Q is symmetric."""
    return _hankel_matvec(q, x)

def _cg_solve_hankel(q, b, x0=None, max_iter=200, tol=1e-10):
    """Solve Q x = b where Q is Hankel with anti-diagonal q, using CG."""
    n = len(b)
    if x0 is None:
        x = np.ones(n) / n
    else:
        x = x0.copy()
    
    r = b - _hankel_matvec(q, x)
    p = r.copy()
    rs = np.dot(r, r)
    b_norm = np.linalg.norm(b)
    if b_norm < 1e-15:
        return b
    
    for it in range(max_iter):
        if np.sqrt(rs) < tol * b_norm:
            break
        Ap = _hankel_matvec(q, p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        new_rs = np.dot(r, r)
        if new_rs >= rs:
            # CG breakdown, restart
            p = r.copy()
            rs = new_rs
        else:
            p = r + (new_rs / rs) * p
            rs = new_rs
    
    return x

def _scf_optimize(a_init, deadline, beta_schedule):
    """SCF optimization with given beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        c_curr = _full_convolve(a, a)
        max_c = float(c_curr.max())
        effective_beta = beta * max_c
        
        if effective_beta < 1e-12:
            effective_beta = 1e-12
        
        # Run SCF iterations
        max_scf_iter = 500
        last_improve = 0
        last_best = best_score
        
        for scf_it in range(max_scf_iter):
            if time.monotonic() > deadline - 2:
                break
            
            c = _full_convolve(a, a)
            c_max = float(c.max())
            
            # Compute dual variables
            c_shift = (c - c_max) / effective_beta
            e = np.exp(c_shift)
            e_sum = float(e.sum())
            if e_sum < 1e-300:
                q = np.zeros(len(c))
                q[int(np.argmax(c))] = 1.0
            else:
                q = e / e_sum
            
            # Solve inner problem: min a^T Q a s.t. sum(a)=1, a>=0
            ones = np.ones(n)
            x = _cg_solve_hankel(q, ones, x0=a, max_iter=100, tol=1e-8)
            
            # Project onto simplex
            x = _proj_simplex(x, 1.0)
            
            # Damping
            t_damp = 0.5
            a_new = (1 - t_damp) * a + t_damp * x
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                last_improve = scf_it
            
            a = a_new
        
        print(f"  SCF beta={effective_beta:.2e}: best_score={best_score:.10f}")
    
    return best_a, best_score
