```python
import cvxpy as cp

def lp_direction(f, M):
    """Find g >= 0 maximizing sum(g) s.t. conv(f, g) <= M."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # conv(f, g) has 2n-1 entries
    # We need all of them <= M
    # conv(f, g)[k] = sum_i f[i]*g[k-i]
    
    # Build the constraint matrix
    # For each k from 0 to 2n-2:
    #   sum_i f[i]*g[k-i] <= M
    #   where i ranges from max(0, k-n+1) to min(n-1, k)
    
    constraints = []
    for k in range(2*n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        # conv(f,g)[k] = sum_{i=i_min}^{i_max} f[i] * g[k-i]
        # Let j = k-i, so j ranges from k-i_max to k-i_min
        # j_min = k - i_max, j_max = k - i_min
        j_min = k - i_max
        j_max = k - i_min
        # The coefficient of g[j] in conv(f,g)[k] is f[k-j]
        coef = np.zeros(n)
        for j in range(j_min, j_max + 1):
            coef[j] = f[k - j]
        constraints.append(coef @ g <= M)
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP)  # or another solver
    return g.value
```

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding, Adam with adaptive softmax smoothing,
L-BFGS refinement, and multi-restart exploration across sequence lengths.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Compute full objective."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.ones(n) * (target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_max(a, w):
    """Gradient of dot(w, conv(a,a)) w.r.t. a, where w has same length as conv."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _smoothed_max_obj(a, beta):
    """Smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_sh = (c - cm) / beta
    e = np.exp(c_sh)
    s = float(e.sum())
    if s < 1e-300:
        return cm, 0.0
    return cm + beta * np.log(s), s


def _smoothed_max_gradient(a, beta):
    """Gradient of smoothed max of conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_sh = (c - cm) / beta
    e = np.exp(c_sh)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return _compute_grad_max(a, w), w


def _adam_opt(a_init, deadline, beta_seq, lr_init=0.005, max_iter_per_beta=50000):
    """Adam optimizer in log-space with multiple beta values."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta in beta_seq:
        if time.monotonic() > deadline - 2:
            break
        t = 0
        for it in range(max_iter_per_beta):
            if time.monotonic() > deadline - 2:
                break
            t += 1
            a = _softmax_compute(z)
            grad, _ = _smoothed_max_gradient(a, beta)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = lr_init * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.2:
                update = update * (0.2 / update_norm)
            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
        a_final = _softmax_compute(z)
        sc = _score_norm(a_final)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    """L-BFGS-B on log-space with smoothed max objective."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 2.0 * h * 2.0 * n
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = max(2.0, deadline - time.monotonic() - 2.0)
    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(20, int(max_time * 50)), 'gtol': 1e-10, 'maxls': 20}
        )
        a_opt = _softmax_compute(result.x)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = result.x.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _lp_direction(a, M, top_k=50, timeout=5.0):
    """
    LP: maximize sum(g) subject to conv(a, g) <= M, g >= 0.
    Uses top_k most violated constraints for efficiency.
    Returns normalized g or None if infeasible/slow.
    """
    n = len(a)
    c_current = _full_convolve(a, a)
    # Find top_k positions in the convolution
    top_idx = np.argsort(c_current)[::-1][:top_k]
    M_val = M

    # Build sparse constraint matrix for selected rows
    # A[k, j] = a[k-j] for valid indices
    rows = []
    cols = []
    vals = []
    for k in top_idx:
        i_min = max(0, int(k) - n + 1)
        i_max = min(n - 1, int(k))
        for j in range(i_min, i_max + 1):
            rows.append(int(k))
            cols.append(j)
            vals.append(a[int(k) - j])

    A_sparse = sp.csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    b_vec = np.full(len(top_idx), M_val)
    c_vec = -np.ones(n)  # minimize -sum(g)

    try:
        res = linprog(c_vec, A_ub=A_sparse, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def _lp_phase(a_init, deadline, n_lp=20):
    """Multiple LP-based direction steps."""
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()

    for i in range(n_lp):
        if time.monotonic() > deadline - 5:
            break
        c = _full_convolve(a, a)
        M = float(c.max())
        g = _lp_direction(a, M, top_k=100, timeout=3.0)
        if g is None:
            break
        # Blend
        t = 0.3 / (1 + i * 0.1)
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            break
        a_new = a_new / a_new_sum
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            t2 = t * 0.3
            a_new2 = (1 - t2) * a + t2 * g
            a_new2 = np.maximum(a_new2, 0)
            a_new2_sum = a_new2.sum()
            if a_new2_sum > 1e-10:
                a_new2 = a_new2 / a_new2_sum
                sc2 = _score_norm(a_new2)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a_new2.copy()
                    a = a_new2
                else:
                    break
            else:
                break
    return best_a, best_score


def _coordinate_descent(a_init, deadline, n_iter=5000):
    """
    Coordinate descent: at each step, find the max lag in conv(a,a),
    then try to reduce it by moving mass from high-contribution positions.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for it in range(n_iter):
        if time.monotonic() > deadline - 2:
            break
        c = _full_convolve(a, a)
        k_max = int(np.argmax(c))
        c_max = float(c[k_max])

        # Find which pairs (i, j) with i+j = k_max contribute most
        i_min = max(0, k_max - n + 1)
        i_max = min(n - 1, k_max)
        # Contribution of pair (i, k_max-i) is a[i]*a[k_max-i]
        contributions = np.zeros(n)
        for i in range(i_min, i_max + 1):
            j = k_max - i
            contributions[i] = a[i] * a[j]

        # Try to reduce by moving mass from a position with high contribution
        # to a position with low contribution at this lag
        hot_idx = np.argmax(contributions)
        # Find a "cold" position: one that doesn't participate in this lag much
        # or participates with a small partner
        cold_candidates = np.where(contributions < 0.1 * contributions[hot_idx])[0]
        if len(cold_candidates) == 0:
            cold_candidates = np.array([hot_idx])

        cold_idx = cold_candidates[rng.integers(0, len(cold_candidates))]

        # Move a small amount of mass
        eps = 1e-4 * a.sum() / n
        a_new = a.copy()
        a_new[hot_idx] -= eps
        a_new[cold_idx] += eps
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def _fista_opt(a_init, beta, deadline, max_iter=50000):
    """FISTA with projection onto simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 6.0 * n
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline - 1:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        grad, _ = _smoothed_max_gradient(y, beta)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.2) ** 2) + np.exp(-20.0 * (t - 0.8) ** 2)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score_full(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score_full(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = a0.copy() / a0.sum()
    n = len(a_work)

    # === Phase 1: FISTA warmup with moderate beta ===
    print("Phase 1: FISTA warmup")
    phase1_end = min(deadline, t_start + 200.0)
    if time.monotonic() < phase1_end - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_c0 = float(c_curr.max())
        beta = 0.05 * max_c0
        for sub in range(5):
            if time.monotonic() > phase1_end - 5:
                break
            per_t = min(phase1_end - time.monotonic() - 3, 40.0)
            if per_t < 3:
                break
            a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_t, max_iter=20000)
            a_work = a_work / a_work.sum()
            sc = _score_norm(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
            c_curr = _full_convolve(a_work, a_work)
            beta = 0.3 * float(c_curr.max())
            if beta < 1e-12:
                beta = 1e-12
        print(f"  End Phase 1: score={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam with aggressive beta sweep ===
    print("Phase 2: Adam beta sweep")
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end - 20:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [
            0.01 * max_cc, 0.003 * max_cc, 0.001 * max_cc,
            0.0003 * max_cc, 0.0001 * max_cc, 0.00003 * max_cc,
            0.00001 * max_cc
        ]
        # Determine iterations per beta based on time
        remaining = phase2_end - time.monotonic() - 15
        per_beta_time = remaining / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase2_end - 15:
                break
            per_iter = max(2000, int(per_beta_time * 80))
            a_work, sc = _adam_opt(a_work, time.monotonic() + per_beta_time,
                                   [beta], lr_init=0.004, max_iter_per_beta=per_iter)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        print(f"  End Phase 2: score={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: LP-based direction finding ===
    print("Phase 3: LP direction finding")
    phase3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase3_end - 20 and n <= 8192:
        a_lp, sc_lp = _lp_phase(a_work, phase3_end, n_lp=15)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            a_work = a_lp
            print(f"  LP: score={sc_lp:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            a_work = a_lp
        print(f"  End Phase 3: score={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Fine Adam polish ===
    print("Phase 4: Fine Adam polish")
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end - 20:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas_fine = [0.000005 * max_cc, 0.000001 * max_cc, 0.0000003 * max_cc]
        remaining = phase4_end - time.monotonic() - 15
        per_beta_time = remaining / len(betas_fine)
        for bi, beta in enumerate(betas_fine):
            if time.monotonic() > phase4_end - 15:
                break
            per_iter = max(3000, int(per_beta_time * 100))
            a_work, sc = _adam_opt(a_work, time.monotonic() + per_beta_time,
                                   [beta], lr_init=0.0015, max_iter_per_beta=per_iter)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Fine Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        print(f"  End Phase 4: score={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: L-BFGS refinement ===
    print("Phase 5: L-BFGS refinement")
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end - 20:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        beta_lb = 0.000001 * max_cc
        a_lb, sc_lb = _lbfgs_opt(a_work, phase5_end, beta_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            a_work = a_lb
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            a_work = a_lb
        print(f"  End Phase 5: score={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Coordinate descent on hot spots ===
    print("Phase 6: Coordinate descent")
    phase6_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase6_end - 10:
        a_cd, sc_cd = _coordinate_descent(a_work, phase6_end, n_iter=10000)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            a_work = a_cd
            print(f"  CoordDesc: score={sc_cd:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            a_work = a_cd
        print(f"  End Phase 6: score={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-restart exploration ===
    print("Phase 7: Multi-restart")
    phase7_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase7_end - 30:
        n_restarts = 5
        per_r = (phase7_end - time.monotonic() - 20) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 20:
                break
            if i == 0:
                p = _perturb(a_work, 0.01, rng)
            elif i == 1:
                p = _make_profile(n, "raised_cosine")
                p = 0.7 * a_work + 0.3 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 2:
                # Frequency-domain perturbation
                F = np.fft.rfft(a_work)
                F_mag = np.abs(F)
                F_ang = np.angle(F)
                noise = rng.standard_normal(len(F_mag)) * 0.005
                F_mag_new = F_mag * (1 + noise)
                F_new = F_mag_new * np.exp(1j * F_ang)
                p = np.fft.irfft(F_new, n * 2)[:n]
                p = np.maximum(p, 0)
                s_p = p.sum()
                if s_p > 1e-10:
                    p = p / s_p
                else:
                    p = a_work.copy()
            elif i == 3:
                # Shift + blend
                shift = rng.integers(1, max(2, n // 1024))
                p_shifted = np.roll(a_work, shift)
                p = 0.6 * a_work + 0.4 * p_shifted
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            else:
                p = _perturb(a_work, 0.025, rng)

            # Quick Adam refinement
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.00005 * max_cp
            p, _ = _adam_opt(p, time.monotonic() + per_r * 0.6,
                             [b_r], lr_init=0.003, max_iter_per_beta=20000)
            p, _ = _lbfgs_opt(p, time.monotonic() + per_r * 0.4, b_r * 0.1)
            sc_r = _score_norm(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")
        print(f"  End Phase 7: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Try different sequence length ===
    print("Phase 8: Length exploration")
    phase8_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase8_end - 20:
        # Try n=2048 (downsample parent)
        n_alt = n // 2 if n >= 2048 else n * 2
        if n_alt != n and n_alt >= 512:
            # Resample a_work to new length
            t_old = np.linspace(0, 1, n)
            t_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(t_new, t_old, a_work)
            a_alt = np.maximum(a_alt, 0)
            a_alt = a_alt / a_alt.sum()
            c_alt = _full_convolve(a_alt, a_alt)
            max_ca = float(c_alt.max())
            per_t = min(phase8_end - time.monotonic() - 10, 60.0)
            if per_t > 5:
                a_alt, _ = _adam_opt(a_alt, time.monotonic() + per_t * 0.6,
                                     [0.0001 * max_ca], lr_init=0.003, max_iter_per_beta=10000)
                a_alt, sc_alt = _lbfgs_opt(a_alt, time.monotonic() + per_t * 0.4, 0.00001 * max_ca)
                if sc_alt < best_score:
                    best_score = sc_alt
                    best_seq = a_alt.copy()
                    print(f"  Alt length n={n_alt}: score={sc_alt:.10f} t={time.monotonic()-t_start:.1f}s")
        print(f"  End Phase 8: best={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 9: Final ultra-polish ===
    print("Phase 9: Final ultra-polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 8
        if per_t > 10:
            # Very fine Adam
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + per_t * 0.5,
                                 [0.0000005 * max_cp, 0.0000001 * max_cp],
                                 lr_init=0.001, max_iter_per_beta=30000)
            sc_p = _score_norm(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
            # L-BFGS on remaining
            if time.monotonic() < deadline - 8:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.0000003 * float(c_f.max())
                a_pol, sc_f = _lbfgs_opt(a_pol, deadline - 3.0, b_f)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score_full(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
```
