"""
Multi-phase optimizer combining FISTA, coordinate descent, L-BFGS, and multi-restart
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: fast coordinate descent exploiting the 1D structure of each update,
combined with adaptive FISTA and L-BFGS polishing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_conv(a, w):
    """Compute gradient of sum_k w_k * c_k where c = conv(a,a).
    Returns 2 * conv(w, a_rev)[n-1:2n-1] which equals the full gradient."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = float(2.0 * n * c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 60 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _coord_descent_sweep(a, c, rng, n_sweeps=1):
    """Coordinate descent on the normalized sequence (sum=1)."""
    n = len(a)
    S = 1.0  # normalized
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for j in order:
            # Compute b_k = a_{k-j} + a_{j-k} for all k
            b = np.zeros(2 * n - 1)
            # a_{k-j} is non-zero when 0 <= k-j < n, i.e., j <= k < j+n
            lo1 = max(0, j)
            hi1 = min(2 * n - 1, j + n)
            idx1 = np.arange(lo1, hi1)
            b[lo1:hi1] += a[idx1 - j]
            # a_{j-k} is non-zero when 0 <= j-k < n, i.e., j-n+1 <= k <= j
            lo2 = max(0, j - n + 1)
            hi2 = min(2 * n - 1, j + 1)
            idx2 = np.arange(lo2, hi2)
            b[lo2:hi2] += a[j - idx2]

            # Now minimize f(delta) = 2n * max(c + delta*b) / (1 + delta)^2
            # where delta = a_j' - a_j, constraint: a_j + delta >= 0 => delta >= -a_j
            aj = a[j]
            d_min = -aj + 1e-15  # can reduce a_j by at most a_j
            d_max = 10.0  # upper bound on increase

            # Quick evaluations
            def eval_f(d):
                new_max = float(np.max(c + d * b))
                new_S = 1.0 + d
                if new_S < 1e-10:
                    return np.inf
                return 2.0 * n * new_max / (new_S * new_S)

            # Try a few candidate points
            candidates = [0.0, d_min * 0.5, d_min, -aj * 0.1, aj * 0.5, aj, 2.0 * aj]
            candidates = [max(d, d_min) for d in candidates]
            best_d = 0.0
            best_val = eval_f(0.0)
            for d in candidates:
                v = eval_f(d)
                if v < best_val:
                    best_val = v
                    best_d = d

            # Golden section search in a refined interval
            # Find brackets
            if best_d > d_min:
                lo = max(d_min, best_d - 2.0 * aj - 1.0)
                hi = best_d + 2.0 * aj + 1.0
            else:
                lo = d_min
                hi = 2.0 * aj + 1.0

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = eval_f(c1)
            f2 = eval_f(c2)
            for _ in range(30):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_f(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_f(c2)
            d_opt = (lo + hi) / 2.0
            f_opt = eval_f(d_opt)
            if f_opt < best_val:
                best_d = d_opt
                best_val = f_opt

            if abs(best_d) > 1e-12:
                a[j] += best_d
                c += best_d * b
                S = 1.0 + best_d

        # Renormalize
        S = float(a.sum())
        if S > 1e-15:
            a = a / S
            c = c / (S * S)
        sc = float(2.0 * n * c.max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score, c


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 4.0 * n * a * (h - W)
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 50)), 'gtol': 1e-14,
                     'maxls': 20, 'factr': 1e7}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _subgradient_nesterov(a_init, deadline, max_iter):
    """Nesterov-accelerated subgradient with Polyak step."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    c = _full_convolve(a, a)
    best_score = float(2.0 * n * c.max())
    best_a = a.copy()

    # Lower bound: 2n/(2n-1)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    # Nesterov momentum
    a_nest = a.copy()
    t_val = 1.0
    a_prev = a.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        a_y = a_nest + ((t_val - 1.0) / t_next) * (a_nest - a_prev)
        a_y = np.maximum(a_y, 0.0)
        sy = a_y.sum()
        if sy > 1e-30:
            a_y = a_y / sy
        else:
            a_y = a.copy()

        c_y = _full_convolve(a_y, a_y)
        c_max = float(c_y.max())
        # Active set: elements within small tolerance of max
        tol = 1e-10 * c_max + 1e-15
        active = np.where(c_y >= c_max - tol)[0]
        w = np.zeros_like(c_y)
        w[active] = 1.0 / len(active)

        grad = _compute_grad_conv(a_y, w)

        # Polyak step size
        gap = c_max - lower_bound  # approximate duality gap (normalized)
        grad_sq = float(np.dot(grad, grad))
        if grad_sq > 1e-30 and gap > 0:
            step = 2.0 * gap / grad_sq
            step = min(step, 1.0 / (2.0 * np.sqrt(it + 1.0)))
        else:
            step = 1.0 / np.sqrt(it + 2.0)

        a_new = a_y - step * grad
        a_new = _proj_simplex(a_new)

        # Check improvement
        if it % 50 == 0:
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        a_prev = a_nest.copy()
        a_nest = a_new
        t_val = t_next

    sc_final = _score_norm(a_nest)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_nest.copy()
    return best_a, best_score

    beta = 0.05 * float(c_curr.max())
    n_phases = 20

    for phase in range(n_phases):
        if time.monotonic() > phase1_end - 3:
            break
        remaining = phase1_end - time.monotonic() - 2
        if remaining < 1:
            break
        per_time = min(remaining / max(1, n_phases - phase), 20.0)
        max_it = max(2000, int(per_time * 5000))
        a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_time, max_it)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            if phase % 4 == 0:
                print(f"  FISTA p{phase} beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_work, a_work)
        beta = 0.3 * float(c_curr.max())
        if beta < 1e-12:
            beta = 1e-12

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Coordinate descent (250-600s) ===
    print("Phase 2: Coordinate descent")
    phase2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2_end - 5:
        a_cd = a_work.copy()
        a_cd = a_cd / a_cd.sum()
        c_cd = _full_convolve(a_cd, a_cd)
        iters = 0
        while time.monotonic() < phase2_end - 5 and iters < 50:
            a_cd, sc_cd, c_cd = _coord_descent_sweep(a_cd, c_cd, rng, n_sweeps=1)
            iters += 1
            a_cd = a_cd / a_cd.sum()
            sc_norm = float(2.0 * n_main * c_cd.max())
            sc_full = _score(a_cd)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_cd.copy()
                if iters % 5 == 0:
                    print(f"  CD iter {iters}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_cd
        print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: Adam fine optimization (600-900s) ===
    print("Phase 3: Adam fine optimization")
    phase3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase3_end - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.0005 * max_cc, 0.0001 * max_cc, 0.00002 * max_cc, 0.000005 * max_cc]
        per_beta = (phase3_end - time.monotonic() - 8) / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase3_end - 8:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + max(5.0, per_beta),
                                  beta, lr_init=0.005, max_iter=300000)
            a_work = a_work / a_work.sum()
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS refinement (900-1200s) ===
    print("Phase 4: L-BFGS refinement")
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        beta_lb = 0.00003 * max_cc
        a_lb, _ = _lbfgs_opt(a_work, phase4_end, beta_lb)
        a_lb = a_lb / a_lb.sum()
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lb

    # === Phase 5: More coordinate descent + subgradient (1200-1450s) ===
    print("Phase 5: CD + Subgradient alternating")
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end - 10:
        a_ws = a_work.copy()
        a_ws = a_ws / a_ws.sum()
        c_ws = _full_convolve(a_ws, a_ws)
        while time.monotonic() < phase5_end - 10:
            # CD sweep
            if time.monotonic() < phase5_end - 15:
                a_ws, _, c_ws = _coord_descent_sweep(a_ws, c_ws, rng, n_sweeps=1)
                a_ws = a_ws / a_ws.sum()
                c_ws = c_ws / (a_ws.sum() ** 2)
                sc_ws = _score(a_ws)
                if sc_ws < best_score:
                    best_score = sc_ws
                    best_seq = a_ws.copy()
                    print(f"  Phase5 CD: score={sc_ws:.10f} t={time.monotonic()-t_start:.1f}s")
            # Subgradient
            if time.monotonic() < phase5_end - 15:
                a_ws, _ = _subgradient_nesterov(a_ws, time.monotonic() + 10, 5000)
                a_ws = a_ws / a_ws.sum()
                sc_ws = _score(a_ws)
                if sc_ws < best_score:
                    best_score = sc_ws
                    best_seq = a_ws.copy()
                    print(f"  Phase5 SG: score={sc_ws:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_ws

    # === Phase 6: Multi-restart exploration (1450-1650s) ===
    print("Phase 6: Multi-restart")
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end - 20:
        n_restarts = 3
        per_r = (phase6_end - time.monotonic() - 15) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 15:
                break
            if i == 0:
                p = _perturb(a_work, 0.01, rng)
            elif i == 1:
                p = _perturb(a_work, 0.03, rng)
            else:
                # Structural perturbation: random shift + blend
                shift = rng.integers(1, max(2, n_main // 1024))
                p_shifted = np.roll(a_work, int(shift)).copy()
                p = 0.7 * a_work + 0.3 * p_shifted
                p = np.maximum(p, 1e-12)
                p = p / p.sum()

            # Quick FISTA + CD
            c_p = _full_convolve(p, p)
            b_r = 0.0002 * float(c_p.max())
            p, _ = _fista_opt(p, b_r, time.monotonic() + per_r * 0.4, 30000)
            p = p / p.sum()
            c_p = _full_convolve(p, p)
            p, _, _ = _coord_descent_sweep(p, c_p, rng, n_sweeps=2)
            p = p / p.sum()
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-polish (1650-1780s) ===
    print("Phase 7: Final ultra-polish")
    if best_seq is not None and time.monotonic() < deadline - 8:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            # Final coordinate descent sweeps
            a_pol = a_pol / a_pol.sum()
            c_pol = _full_convolve(a_pol, a_pol)
            while time.monotonic() < deadline - 15:
                a_pol, _, c_pol = _coord_descent_sweep(a_pol, c_pol, rng, n_sweeps=1)
                a_pol = a_pol / a_pol.sum()
                sc_p = _score(a_pol)
                if sc_p < best_score:
                    best_score = sc_p
                    best_seq = a_pol.copy()
                    print(f"  Final CD: score={sc_p:.10f} t={time.monotonic()-t_start:.1f}s")

            # L-BFGS on remaining
            if time.monotonic() < deadline - 10:
                c_f = _full_convolve(best_seq, best_seq)
                b_f = 0.000002 * float(c_f.max())
                a_pol2, _ = _lbfgs_opt(best_seq, deadline - 5.0, b_f)
                a_pol2 = a_pol2 / a_pol2.sum()
                sc_f = _score(a_pol2)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol2.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
