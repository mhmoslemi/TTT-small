import numpy as np
import time

def _next_pow2(n):
    return 1 << (n - 1).bit_length() if n > 1 else 2

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score_norm(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))"""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return _score_norm(a / s)

def _proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho_idx = np.where(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)

def _topk_weights(c, beta, k):
    """Softmax weights over top-k elements."""
    n_c = len(c)
    k = min(k, n_c)
    if k < n_c:
        thresh = np.partition(c, -k)[-k]
        mask = c >= thresh
    else:
        mask = np.ones(n_c, dtype=bool)
    idx = np.where(mask)[0]
    c_top = c[idx]
    c_max = c_top.max()
    exp_v = np.exp((c_top - c_max) / beta)
    s = exp_v.sum()
    if s < 1e-300:
        w = np.zeros(n_c)
        w[idx[0]] = 1.0
        return w
    w = np.zeros(n_c)
    w[idx] = exp_v / s
    return w

def _grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a. = 2 * cross_corr(a, w)"""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Improvements over previous: better coordinate descent with exact 1D subproblem solving,
adaptive beta schedules, multi-length exploration (4096, 8192), and enhanced Adam tuning.
"""
import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 2
    return 1 << (n - 1).bit_length()


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max()) / (s * s)


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _topk_weights(c, beta, k):
    n_c = len(c)
    k = min(k, n_c)
    if k < n_c:
        thresh = np.partition(c, -k)[-k]
        mask = c >= thresh
    else:
        mask = np.ones(n_c, dtype=bool)
    idx = np.where(mask)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(c))])
    c_top = c[idx]
    c_max = float(c_top.max())
    exp_v = np.exp((c_top - c_max) / beta)
    s = float(exp_v.sum())
    w = np.zeros(n_c)
    if s < 1e-300:
        w[idx[0]] = 1.0
    else:
        w[idx] = exp_v / s
    return w


def _grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _fista(a_init, deadline, beta, max_iter, top_k=100):
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    t_val = 1.0
    best_a = a.copy()
    best_sc = _score_norm(a)
    c_init = _convolve(a, a)
    L = 8.0 * n / max(1e-10, float(c_init.max()))
    step = 1.0 / L
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a + ((t_val - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _convolve(y, y)
        w = _topk_weights(c_y, beta, top_k)
        grad = _grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()
        a_prev = a
        a = a_next
        t_val = t_next
    sc = _score_norm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, betas, lr_init, top_k, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z -= z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        while it_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1
            a = np.exp(z - z.max())
            a /= a.sum()
            c = _convolve(a, a)
            w = _topk_weights(c, beta, top_k)
            grad_a = _grad_conv(a, w)
            mean_g = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - mean_g)
            if not np.all(np.isfinite(grad_z)):
                break
            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
            mh = m / (1.0 - b1 ** t)
            vh = v / (1.0 - b2 ** t)
            update = lr * mh / (np.sqrt(vh) + eps)
            unorm = np.linalg.norm(update)
            if unorm > 2.0:
                update *= 2.0 / unorm
            z -= update
            z = np.clip(z, -40.0, 40.0)
            z -= z.mean()
            if t % 200 == 0:
                a_chk = np.exp(z - z.max())
                a_chk /= a_chk.sum()
                sc = _score_norm(a_chk)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_chk.copy()
            if it_phase % 2000 == 0:
                lr = max(0.0003, lr * 0.96)
        m *= 0.3
        v *= 0.3
    a_final = np.exp(z - z.max())
    a_final /= a_final.sum()
    sc = _score_norm(a_final)
    if sc < best_sc:
        best_sc = sc
        best_a = a_final.copy()
    return best_a, best_sc
