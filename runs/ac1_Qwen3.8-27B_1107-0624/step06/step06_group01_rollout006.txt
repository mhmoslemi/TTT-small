"""
Optimized sequence generator using Adam in log-space with cosinusoidal annealing,
spectral jumps, and multi-length exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time

def _next_pow2(n):
    """Next power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    """Full convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    """Compute the full score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    """Compute softmax of z."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    - n: length of a

    The gradient w.r.t. a is: g[i] = 4n * sum_k w[k]*a[k-i] (zero-padded)

def _adam_logspace(a_init, deadline, betas, lr_init, max_iter_total, use_nesterov=True):

    z = z - z.mean()
    
    best_a = a_init.copy()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)

    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    # Nesterov momentum
    if use_nesterov:
        k_buf = np.zeros(n)
    
    for beta_idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        it_in_phase = 0
        phase_max_iter = max_iter_total // len(betas)
        
        while it_in_phase < phase_max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax(z)
            c = _fft_convolve(a, a)
            
            # Compute weights

            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            # Compute gradient
            grad = _compute_log_grad(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break
            
            # Adam update
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            if use_nesterov:
                k_buf = b1 * k_buf + update
                z = z - k_buf
            else:
                z = z - update

            z = z - z.mean()

        best_a = a_final.copy()

def _spectral_jump(a, rng, strength=0.1):
    """
    Apply a random spectral phase perturbation to a, then project back to non-negative.
    """
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    # Add random phase perturbation
    phase = rng.standard_normal(len(A)) * strength
    A = A * np.exp(1j * phase)
    # Inverse FFT
    a_new = np.fft.irfft(A, N)[:n]
    # Project to non-negative
    a_new = np.maximum(a_new, 1e-10)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new

def _perturb(a, sigma, rng):

    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(42)
        a = rng.random(n) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n

def _coord_descent_polish(a, deadline, n_steps=500):
    """Coordinate descent polish - try individual coordinate modifications."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(123)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        i = rng.integers(0, n)
        improved = False
        for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 0.5, 1.5]:
            a_try = best_a.copy()
            a_try[i] *= factor
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    improved = True
                    break
        if not improved:
            # Try pair modification
            if n > 1:
                j = rng.integers(0, n)
                if j != i:
                    a_try = best_a.copy()
                    a_try[i] *= 0.8
                    a_try[j] *= 1.25  # conserve approximately
                    a_try = np.maximum(a_try, 0.0)
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try = a_try / s
                        sc = _score_norm(a_try)
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a_try.copy()
    
    return best_a, best_score

            if s > 0.01:

    except Exception:

    else:

        lengths_to_try.append(8192)
    if n_base >= 4096:
        lengths_to_try.append(4096)
    # Remove duplicates
    lengths_to_try = list(dict.fromkeys(lengths_to_try))

    # Allocate time budget
    reserve = 60.0  # seconds reserved for final polish
    total_time = deadline - t_start
    time_per_length = (total_time - reserve) / len(lengths_to_try)
    
    for n_target in lengths_to_try:
        if time.monotonic() > deadline - reserve:
            break
        
        t_len_start = time.monotonic()

            a_main = np.interp(x1, x0, parent)
            a_main = np.maximum(a_main, 1e-10)

            n_beta_steps = 8
            beta_start = 0.05 * max_c
            beta_end = 0.0001 * max_c
            # Geometric schedule
            ratio = (beta_end / beta_start) ** (1.0 / n_beta_steps)
            betas = [beta_start * (ratio ** i) for i in range(n_beta_steps)]
            
            # Run Adam
            adam_deadline = t_init_start + time_per_this
            max_iter = int(time_per_this * 2000)  # rough estimate
            a_opt, sc_opt = _adam_logspace(
                a_init, adam_deadline, betas,
                lr_init=0.01, max_iter_total=max_iter,
                use_nesterov=True
            )
            
            sc_opt_full = _score_full(a_opt)
            if sc_opt_full < best_score:
                best_score = sc_opt_full
                best_seq = a_opt.copy()
                print(f"  New best: score={sc_opt_full:.10f} (init {init_i}, n={n_target})")

                    use_nesterov=True
                )

                    print(f"  Jump improved: score={sc_jumped_full:.10f}")

        if best_seq is not None and len(best_seq) == n_target:
