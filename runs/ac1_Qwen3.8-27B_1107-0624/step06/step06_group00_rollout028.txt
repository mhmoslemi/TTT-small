"""
Optimized step-function search using L-BFGS with log-sum-exp smoothing,
coordinate descent mass-transfer, and multi-restart perturbation.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative a of length n.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _lsq_func_and_grad(z, T):
    """Compute log-sum-exp smoothed objective and gradient in z-space."""
    n = len(z)
    a = _softmax_stable(z)
    c = _fft_convolve(a, a)
    # Stable log-sum-exp: (1/T)*log(sum exp(T*c_k)) = max(c) + (1/T)*log(sum exp(T*(c_k - max(c))))
    c_shifted = T * (c - c.max())
    m2 = c_shifted.max()
    lse = c.max() + (m2 + np.log(np.sum(np.exp(c_shifted - m2)))) / T
    # Softmax weights p_k = exp(T*(c_k - max(c))) / sum(...)
    e = np.exp(c_shifted - m2)
    p = e / e.sum()
    # Gradient: 2*a_j*(q_j - r)
    # q_j = sum_k p_k * a_{k-j} = cross-correlation of p and a
    p_flipped = p[::-1]
    conv_result = _fft_convolve(a, p_flipped)
    q = conv_result[n - 1:2 * n - 1][::-1]
    r = float(np.dot(p, c))
    grad = 2.0 * a * (q - r)
    return lse, grad


def _lbfgs_run(z_init, T, deadline=None, max_iter=500, ftol=1e-12, gtol=1e-10):
    """Run L-BFGS-B optimization in z-space with given T."""
    n = len(z_init)

    def objective(z):
        if deadline is not None and time.monotonic() > deadline:
            return np.inf, np.zeros(n)
        z = z - z.mean()
        val, grad = _lsq_func_and_grad(z, T)
        if not np.isfinite(val):
            return np.inf, np.zeros(n)
        return val, grad

    # Center z
    z = z_init - z_init.mean()
    res = minimize(objective, z, method='L-BFGS-B',
                   options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol,
                            'maxfun': max_iter * 3})
    z_opt = res.x - res.x.mean()
    return z_opt


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=200000, eval_interval=200, clip_norm=0.5):
    """Adam optimizer in z-space with temperature annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        # Soft gradient
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
        # Compute gradient
        p_flipped = p[::-1]
        conv_result = _fft_convolve(a, p_flipped)
        q = conv_result[n - 1:2 * n - 1][::-1]
        r = float(np.dot(p, c))
        grad_a = 2.0 * a * (q - r)
        # Chain rule through softmax
        W = float(np.dot(a, grad_a))
        z_grad = a * (grad_a - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.1:
                lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent(a, deadline, max_passes=50, step_frac=0.001):
    """Coordinate descent: move mass from high-contribution to low-contribution positions."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(123)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        max_c = float(c.max())
        # Compute per-position contribution to the max
        # Positions that contribute to the peak should lose mass
        active = c >= max_c * 0.999
        active_indices = np.where(active)[0]
        # For each i, compute its "congestion" = sum of a[j] for j in active lag positions
        # congestion_i = sum_{k in active} a[k-i] (where valid)
        congestion = np.zeros(n)
        for i in range(n):
            # c_k contains a[i]*a[k-i], so a[i] contributes to c_k for k where a[k-i] > 0
            # The "damage" of a[i] is: for each active k, a[i] * a[k-i] contributes to c_k
            # We want to reduce this, so move mass from high a[i]*a[active_lag - i]
            valid_k = np.arange(0, 2 * n - 1)
            a_shifted = np.zeros(2 * n - 1)
            a_shifted[i:i + n] = a  # a[k-i] for k in [i, i+n-1]
            # Actually a[k-i] is nonzero for k in [i, i+n-1]
            congestion[i] = float(np.sum(c[active] * a_shifted[active]))
        # Find best transfer: from high congestion to low congestion
        # Sort by congestion
        order = np.argsort(congestion)
        # Move mass from top to bottom
        n_transfer = max(1, n // 100)
        for _ in range(n_transfer):
            if time.monotonic() > deadline:
                break
            src = order[-1]  # highest congestion
            dst = order[0]  # lowest congestion
            if src == dst:
                break
            # Try moving a small amount
            eps = step_frac * a[src]
            if eps < 1e-12:
                break
            a_trial = a.copy()
            a_trial[src] -= eps
            a_trial[dst] += eps
            if a_trial[src] < 0:
                eps = a[src] * 0.5
                a_trial = a.copy()
                a_trial[src] -= eps
                a_trial[dst] += eps
            sc = _score_norm(a_trial)
            if sc < best_sc - 1e-12:
                a = a_trial
                best_sc = sc
                best_a = a.copy()
                # Recompute congestion
                c = _fft_convolve(a, a)
                max_c = float(c.max())
                active = c >= max_c * 0.999
                congestion = np.zeros(n)
                for i in range(n):
                    a_shifted = np.zeros(2 * n - 1)
                    a_shifted[i:i + n] = a
                    congestion[i] = float(np.sum(c[active] * a_shifted[active]))
                order = np.argsort(congestion)
            else:
                # Try a smaller step
                eps = eps * 0.3
                a_trial = a.copy()
                a_trial[src] -= eps
                a_trial[dst] += eps
                sc = _score_norm(a_trial)
                if sc < best_sc - 1e-12:
                    a = a_trial
                    best_sc = sc
                    best_a = a.copy()
                else:
                    break

    return best_a, best_sc


def _active_set_lbfgs(a, deadline, top_k=10, max_iter=200):
    """LBFGS focusing only on top-k convolution elements."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    def obj(z):
        if time.monotonic() > deadline:
            return np.inf, np.zeros(n)
        z_c = z - z.mean()
        a_c = _softmax_stable(z_c)
        c = _fft_convolve(a_c, a_c)
        # Get top-k
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        # Log-sum-exp on just top-k
        T = 50.0
        c_s = T * (c_top - c_top.max())
        lse = c_top.max() + (1.0 / T) * (c_s.max() + np.log(np.sum(np.exp(c_s - c_s.max()))))
        # Gradient w.r.t. z
        p_top = np.exp(c_s) / np.exp(c_s).sum()
        p = np.zeros(2 * n - 1)
        p[top_idx] = p_top
        p_flipped = p[::-1]
        conv_result = _fft_convolve(a_c, p_flipped)
        q = conv_result[n - 1:2 * n - 1][::-1]
        r = float(np.dot(p_top, c_top))
        grad = 2.0 * a_c * (q - r)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
        return lse, grad

    res = minimize(obj, z, method='L-BFGS-B',
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
    z_opt = res.x - res.x.mean()
    a_opt = _softmax_stable(z_opt)
    return a_opt, _score_norm(a_opt)


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
