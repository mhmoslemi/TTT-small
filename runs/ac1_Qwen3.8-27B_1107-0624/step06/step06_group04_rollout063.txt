import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _thue_morse_centered(n):
    tm = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return (2.0 * tm - 1.0).astype(np.float64)


def _softmax_weights(c, beta):
    cm = c.max()
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _compute_subgrad_topk(a, c, top_k, beta):

    n = len(a)
    n_c = len(c)

            grad[i_start:i_end + 1] += wt * segment[::-1]
    return 2.0 * grad

    A = np.fft.rfft(a_rev, N_fft)
    corr = np.fft.irfft(W * A, N_fft)

    # Actually, let's compute directly using the fact that:

    # That's not quite what we want either.

    # But we want sum_k w[k] * a[k-i], which is r'[i] = sum_k w[k] * a[k-i]
    # Let j = k-i, so k = j+i: r'[i] = sum_j w[j+i] * a[j]
    # This is the cross-correlation of w and a: (w ⊛ a)[i] = sum_j w[j+i]*a[j]

def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on simplex with softmax-smoothed max of convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 1e-30:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -30.0, 30.0)
            z = z - z.mean()

            if t % 50 == 0:
                z_max2 = z.max()
                a_chk = np.exp(z - z_max2)
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0002, current_lr * lr_decay_factor)
            if no_improve > 80 and t > 500:
                current_lr = max(0.0002, current_lr * 0.9)

    # Final
    z_max = z.max()
    a_final = np.exp(z - z_max)
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_tm(a, sigma, rng, n):
    """Perturb with Thue-Morse modulation."""
    tm = _thue_morse_centered(n)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * tm + sigma * 0.5 * rng.standard_normal(n)
    z = z - z.mean()
    z_max = z.max()
    a_new = np.exp(z - z_max)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def _perturb_random(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    z_max = z.max()
    a_new = np.exp(z - z_max)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def _antisymmetric_init(n, rng):
    """Create an anti-symmetric-like initialization that reduces center peak."""
    # a[i] * a[n-1-i] should be small on average
    # Use: a[i] = (1 + d * phase(i)) where phase creates anti-correlation
    d = 0.3 + 0.3 * rng.random()
    # Use a smooth anti-symmetric modifier
    t = np.linspace(0, 1, n)
    # Make first half larger, second half smaller (or vice versa)
    if rng.random() < 0.5:
        modifier = 1.0 - d * (2.0 * t - 1.0)
    else:
        modifier = 1.0 + d * (2.0 * t - 1.0)
    # Add some Thue-Morse texture
    tm = _thue_morse_centered(n)
    modifier = modifier + 0.1 * d * tm
    modifier = np.maximum(modifier, 0.01)
    return modifier / modifier.sum()


def _multi_peak_shave(a_init, deadline, phases=5):
    """Multi-scale peak shaving with varying top_k."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-30:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)

    for phase in range(phases):
        phase_deadline = deadline - (phases - 1 - phase) * (deadline - time.monotonic()) / phases
        top_k = [20, 50, 100, 200, 500][phase % 5]
        a, sc = _peak_shave_general(a, min(deadline, time.monotonic() + (deadline - time.monotonic()) / (phases - phase)),
                                     max_iter=50000, top_k=top_k, init_step=1.0 / (phase + 1))
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # === Phase 1: Aggressive peak shaving on parent ===
    print(f"\nPhase 1: Peak shaving on n={n_main}")
    phase1_end = min(deadline, t_start + 300.0)
    a_work = a0.copy()
    a_work, sc_ps = _peak_shave_general(a_work, phase1_end, max_iter=200000, top_k=30, init_step=1.0)
    sc_ps_full = _score(a_work)
    if sc_ps_full < best_score:
        best_score = sc_ps_full
        best_seq = a_work.copy()
        print(f"  PeakShave: score={sc_ps_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: FISTA refinement ===
    print(f"\nPhase 2: FISTA refinement on n={n_main}")
    phase2_end = min(deadline, t_start + 500.0)
    a_work = best_seq.copy()
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())

    betas = []
    b = 0.01 * max_cw
    while b > 0.00001 * max_cw:
        betas.append(b)
        b *= 0.5
    betas = betas[:10]

    for beta in betas:
        if time.monotonic() > phase2_end - 10:
            break
        a_work, sc = _fista_smoothed(a_work, beta, time.monotonic() + min(30.0, phase2_end - time.monotonic() - 10), 20000)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  FISTA beta={beta:.8f}: score={sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Adam log-space ===
    print(f"\nPhase 3: Adam log-space on n={n_main}")
    phase3_end = min(deadline, t_start + 800.0)
    a_work = best_seq.copy()
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())
    betas_fine = [0.001 * max_cw, 0.0005 * max_cw, 0.0002 * max_cw, 0.0001 * max_cw, 0.00005 * max_cw]
    a_opt, sc_opt = _adam_logspace(a_work, phase3_end, betas_fine, lr_init=0.008, max_iter=500000)
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Adam: score={sc_opt_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart with diverse initializations ===
    print(f"\nPhase 4: Multi-restart with diverse initializations")
    phase4_end = min(deadline, t_start + 1200.0)
    n_restarts = 6
    per_restart = max(30.0, (phase4_end - time.monotonic() - 20) / n_restarts)
    n_best = len(best_seq)

    for i in range(n_restarts):
        if time.monotonic() > phase4_end - 15:
            break

        # Diverse initialization
        r = rng.random()
        if r < 0.25:
            # Thue-Morse perturbation
            p_start = _perturb_tm(best_seq, 0.05 + 0.02 * i, rng, n_best)
        elif r < 0.5:
            # Random perturbation
            p_start = _perturb_random(best_seq, 0.03 + 0.02 * i, rng)
        elif r < 0.7:
            # Anti-symmetric init
            p_start = _antisymmetric_init(n_best, rng)
            # Blend with best
            p_start = 0.5 * p_start + 0.5 * best_seq
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        elif r < 0.85:
            # Frequency-domain perturbation
            p_start = best_seq.copy()
            A = np.fft.rfft(p_start, 2 * n_best)
            A += 0.01 * rng.standard_normal(len(A)) + 0.01j * rng.standard_normal(len(A))
            p_start = np.fft.irfft(A, 2 * n_best)[:n_best]
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        else:
            # Blend with structured profile
            t = np.linspace(0, 1, n_best)
            profiles = [
                1.0 + 0.3 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 0.5 + 0.1,
                1.0 + 0.2 * np.cos(4 * np.pi * t) + 0.1 * np.cos(8 * np.pi * t)
            ]
            p_struct = np.maximum(profiles[i % 3], 0.01)
            p_struct = p_struct / p_struct.sum()
            mix = 0.2 + 0.1 * i
            p_start = mix * p_struct + (1 - mix) * best_seq
            p_start = p_start / p_start.sum()

        # Optimize this restart
        restart_deadline = time.monotonic() + per_restart - 5

        # Quick peak shave
        p_opt, _ = _peak_shave_general(p_start, min(restart_deadline, time.monotonic() + per_restart * 0.4),
                                        max_iter=20000, top_k=20, init_step=0.5)
        p_opt = p_opt / p_opt.sum()

        # Then Adam
        if time.monotonic() < restart_deadline - 5:
            c_r = _full_convolve(p_opt, p_opt)
            max_cr = float(c_r.max())
            betas_r = [0.0005 * max_cr, 0.0002 * max_cr, 0.0001 * max_cr]
            p_opt, _ = _adam_logspace(p_opt, restart_deadline, betas_r, lr_init=0.005, max_iter=50000)

        sc_r = _score(p_opt)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = p_opt.copy()
            n_best = len(best_seq)
            print(f"  Restart {i}: score={sc_r:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Try different n ===
    print(f"\nPhase 5: Explore different sequence lengths")
    phase5_end = min(deadline, t_start + 1500.0)
    for n_try in [n_best * 2, n_best // 2]:
        if time.monotonic() > phase5_end - 60:
            break
        if n_try < 256 or n_try > 32768:
            continue
        print(f"  Trying n={n_try}")
        a_interp = _interp_profile(best_seq, n_try)
        sc_interp = _score(a_interp)
        print(f"  Interpolated score: {sc_interp:.10f}")

        # Optimize at this n
        opt_deadline = time.monotonic() + min(60.0, phase5_end - time.monotonic() - 30)
        a_opt_n, _ = _peak_shave_general(a_interp, opt_deadline, max_iter=50000, top_k=30, init_step=0.5)
        a_opt_n = a_opt_n / a_opt_n.sum()
        sc_opt_n = _score(a_opt_n)
        if sc_opt_n < best_score:
            best_score = sc_opt_n
            best_seq = a_opt_n.copy()
            print(f"  n={n_try} optimized: score={sc_opt_n:.10f}, t={time.monotonic()-t_start:.1f}s")

        # Quick FISTA
        if time.monotonic() < phase5_end - 30:
            c_n = _full_convolve(a_opt_n, a_opt_n)
            max_cn = float(c_n.max())
            a_fista, _ = _fista_smoothed(a_opt_n, 0.001 * max_cn,
                                          min(phase5_end - 30, time.monotonic() + 30), 10000)
            a_fista = a_fista / a_fista.sum()
            sc_fista = _score(a_fista)
            if sc_fista < best_score:
                best_score = sc_fista
                best_seq = a_fista.copy()
                print(f"  n={n_try} FISTA: score={sc_fista:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polish ===
    print(f"\nPhase 6: Ultra-fine polish")
    phase6_end = min(deadline, t_start + 1720.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # Peak shave with very small step
    a_ps, sc_ps = _peak_shave_general(a_work, phase6_end, max_iter=100000, top_k=100, init_step=0.1)
    sc_ps = _score(a_ps)
    if sc_ps < best_score:
        best_score = sc_ps
        best_seq = a_ps.copy()
        print(f"  Fine peak shave: score={sc_ps:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Adam with very small beta
    if time.monotonic() < phase6_end - 20:
        a_work = best_seq.copy() / best_seq.sum()
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())
        betas_ultra = [0.00002 * max_cw, 0.00001 * max_cw, 0.000005 * max_cw]
        a_ultra, _ = _adam_logspace(a_work, phase6_end, betas_ultra, lr_init=0.002, max_iter=200000,
                                     lr_decay_every=10000, lr_decay_factor=0.93)
        sc_ultra = _score(a_ultra)
        if sc_ultra < best_score:
            best_score = sc_ultra
            best_seq = a_ultra.copy()
            print(f"  Ultra Adam: score={sc_ultra:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = final.sum()
            final = final / s

        sc_final = _score(final)
        # Use evaluate_sequence for final check
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"FINAL: internal={sc_final:.10f}, evaluator={sc_eval:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        except Exception as ex:
            print(f"FINAL: internal={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s, eval_error={ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(512)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
