Multi-phase optimizer: LP-guided + FISTA + precise coordinate descent.
Improves upon previous 1.505776 by using better coordinate descent and LP direction finding.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:

    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]

    else:

    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300:
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _topk_weights(c, beta, k):
    n_c = len(c)
    k = min(k, n_c)
    topk_thresh = np.partition(c, -k)[-k] if k < n_c else c.min()
    mask = c >= topk_thresh
    idx = np.where(mask)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(c))])
    c_top = c[idx]
    c_max = float(c_top.max())
    exp_v = np.exp((c_top - c_max) / beta)
    w_sum = float(exp_v.sum())
    w = np.zeros(n_c)
    w[idx] = exp_v / w_sum if w_sum > 1e-300 else (1.0 / len(idx))
    return w

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:

    x0 = np.linspace(0, 1, n0)

    pn = np.maximum(pn, 1e-12)
    return pn / pn.sum()


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    if kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    if kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n


def _lp_direction(a, M, n_con=300):
    n = len(a)
    conv_a = _full_convolve(a, a)
    top_idx = np.argsort(conv_a)[-n_con:]
    A = np.zeros((n_con, n))
    for r, k in enumerate(top_idx):
        lo_j = max(0, k - n + 1)
        hi_j = min(n - 1, k)
        if hi_j >= lo_j:
            A[r, lo_j:hi_j + 1] = a[k - hi_j:k - lo_j + 1]
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(n_con, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.status == 0 and res.x is not None:
            b = np.maximum(res.x, 0.0)
            sb = float(b.sum())
            if sb > 1e-10:
                return b / sb
    except Exception:
        pass
    return None

        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score, best_seq = sc_full, a_work.copy()
        print(f"  FISTA beta={beta:.8f} k={tk}: {sc_full:.10f}  t={time.monotonic()-t_start:.0f}s")

    # ---------- PHASE 2: LP-guided + Adam ----------
    print(f"\n=== Phase 2: LP-guided Adam (n={n_primary}) ===")
    phase2_end = time.monotonic() + min(600.0, (deadline - time.monotonic()) * 0.35)
    lp_success = 0
    while time.monotonic() < phase2_end - 30:
        c_w = _full_convolve(a_work, a_work)
        M = float(c_w.max())
        b_dir = _lp_direction(a_work, M, n_con=400)
        if b_dir is not None:
            p = rng_master.uniform(0.3, 0.7)
            a_mix = (1 - p) * a_work + p * b_dir
            a_mix = np.maximum(a_mix, 0.0)
            s = a_mix.sum()
            if s > 1e-10:
                a_mix /= s
            sc_mix = _score(a_mix)
            if sc_mix < _score(a_work):
                a_work = a_mix
                lp_success += 1
                print(f"  LP iter {lp_success}: {sc_mix:.10f}  t={time.monotonic()-t_start:.0f}s")
        # Short Adam burst to refine
        c_w2 = _full_convolve(a_work, a_work)
        mc2 = float(c_w2.max())
        betas_ad = [mc2 * 0.005, mc2 * 0.002, mc2 * 0.001]
        a_work, sc_ad = _adam_logspace(
            a_work, min(phase2_end, time.monotonic() + 120),
            betas_ad, lr_init=0.015, top_k=150, max_iter=80000)
        sc_ad_full = _score(a_work)
        if sc_ad_full < best_score:
            best_score, best_seq = sc_ad_full, a_work.copy()
        print(f"  Adam burst: {sc_ad_full:.10f}  t={time.monotonic()-t_start:.0f}s")
        if time.monotonic() > phase2_end - 60:
            break

    a_work = a_work / a_work.sum()

    # ---------- PHASE 3: Coordinate Descent ----------
    print(f"\n=== Phase 3: Coordinate Descent (n={n_primary}) ===")
    cd_deadline = min(deadline - 60, time.monotonic() + 600.0)
    a_cd, sc_cd = _coord_descend(a_work, cd_deadline, rng_master,
                                 n_coords=400, max_passes=500)
    sc_cd_full = _score(a_cd)
    if sc_cd_full < best_score:
        best_score, best_seq = sc_cd_full, a_cd.copy()
    print(f"  CD done: {sc_cd_full:.10f}  t={time.monotonic()-t_start:.0f}s")
    a_work = a_cd

    # ---------- PHASE 4: Second lap ----------
    print(f"\n=== Phase 4: Second lap (n={n_primary}) ===")
    if time.monotonic() < deadline - 120:
        c_w = _full_convolve(a_work, a_work)
        M = float(c_w.max())
        b_dir = _lp_direction(a_work, M, n_con=400)
        if b_dir is not None:
            a_mix = 0.5 * a_work + 0.5 * b_dir
            a_mix = np.maximum(a_mix, 0.0)
            s = a_mix.sum()
            if s > 1e-10:
                a_mix /= s
            sc_mix = _score(a_mix)
            if sc_mix < _score(a_work):
                a_work = a_mix
                print(f"  LP2: {sc_mix:.10f}")
        c_w2 = _full_convolve(a_work, a_work)
        mc2 = float(c_w2.max())
        betas_ad2 = [mc2 * 0.003, mc2 * 0.001, mc2 * 0.0005]
        a_work, sc_ad2 = _adam_logspace(
            a_work, time.monotonic() + 120,
            betas_ad2, lr_init=0.008, top_k=200, max_iter=100000)
        sc_ad2_full = _score(a_work)
        if sc_ad2_full < best_score:
            best_score, best_seq = sc_ad2_full, a_work.copy()
        a_work = a_work / a_work.sum()
        cd_dl2 = min(deadline - 30, time.monotonic() + 200.0)
        a_cd2, sc_cd2 = _coord_descend(a_work, cd_dl2, rng_master,
                                       n_coords=500, max_passes=300)
        sc_cd2_full = _score(a_cd2)
        if sc_cd2_full < best_score:
            best_score, best_seq = sc_cd2_full, a_cd2.copy()
        print(f"  2nd lap CD: {sc_cd2_full:.10f}  t={time.monotonic()-t_start:.0f}s")

    # ---------- PHASE 5: Try other lengths ----------
    print(f"\n=== Phase 5: Other lengths ===")
    if time.monotonic() < deadline - 180:
        for alt_n in [2048, 8192]:
            if time.monotonic() > deadline - 60:
                break
            if alt_n == n_primary:
                continue
            print(f"\n  Trying n={alt_n}")
            a_alt = _interp_profile(best_seq, alt_n) if best_seq is not None else \
                _structured_init(alt_n, "cos_sum")
            sc_alt0 = _score(a_alt)
            if sc_alt0 < best_score:
                best_score, best_seq = sc_alt0, a_alt.copy()
            c_alt = _full_convolve(a_alt, a_alt)
            mc_alt = float(c_alt.max())
            betas_alt = [mc_alt * 0.02, mc_alt * 0.005, mc_alt * 0.002]
            a_alt, sc_alt = _fista(
                a_alt, time.monotonic() + 80, betas_alt[-1], 30000, 200)
            a_alt = a_alt / a_alt.sum()
            sc_alt = _score(a_alt)
            if sc_alt < best_score:
                best_score, best_seq = sc_alt, a_alt.copy()
            print(f"    n={alt_n} FISTA: {sc_alt:.10f}")
            cd_dn = min(deadline - 30, time.monotonic() + 60.0)
            if cd_dn - time.monotonic() > 10:
                a_alt, sc_alt2 = _coord_descend(a_alt, cd_dn, rng_master,
                                                n_coords=300, max_passes=100)
                sc_alt2 = _score(a_alt)
                if sc_alt2 < best_score:
                    best_score, best_seq = sc_alt2, a_alt.copy()
                print(f"    n={alt_n} CD: {sc_alt2:.10f}")

    # ---------- PHASE 6: Diversity restarts ----------
    if time.monotonic() < deadline - 40 and best_seq is not None:
        print(f"\n=== Phase 6: Diversity restarts ===")
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()
        remaining = deadline - time.monotonic() - 20
        n_rest = max(1, int(remaining // 25))
        per_rest = remaining / max(1, n_rest)

        for ri in range(n_rest):
            if time.monotonic() > deadline - 10:
                break
            if ri == 0:
                p0 = a_base + 0.01 * rng_master.standard_normal(n_best)
            elif ri == 1:
                sh = max(1, int(rng_master.integers(1, n_best // 256)))
                p0 = np.roll(a_base, sh).copy()
            elif ri == 2:
                fr = rng_master.uniform(0.5, 3.0)
                p0 = a_base + 0.02 * np.cos(2 * np.pi * fr * np.arange(n_best) / n_best)
            else:
                p0 = a_base.copy()
                blk = n_best // 16
                st = int(rng_master.integers(0, n_best - blk))
                p0[st:st + blk] *= rng_master.uniform(0.8, 1.2)
            p0 = np.maximum(p0, 1e-10)
            p0 = p0 / p0.sum()
            c_r = _full_convolve(p0, p0)
            mc_r = float(c_r.max())
            betas_r = [mc_r * 0.003, mc_r * 0.001]
            p_r, sc_r = _adam_logspace(p0, time.monotonic() + max(3, per_rest - 5),
                                       betas_r, lr_init=0.01, top_k=150,
                                       max_iter=50000)
            sc_r = _score(p_r)
            if sc_r < best_score:
                best_score, best_seq = sc_r, p_r.copy()
                print(f"  Restart {ri}: {sc_r:.10f}  t={time.monotonic()-t_start:.0f}s")

    # ---------- Final ----------
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
