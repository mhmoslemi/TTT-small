def _coordinate_descent_step(a, c, i):

    Minimize max(conv(a,a)) w.r.t. a[i], keeping other coordinates fixed.
    Returns new value of a[i].
    """
    n = len(a)
    ai = a[i]
    
    # Compute base values: base[k] = c[k] - contribution of a[i]
    # Contribution of a[i] to c[k]:
    #   if 0 <= k-i < n and k != 2i: 2*a[i]*a[k-i]
    #   if k == 2i: a[i]^2
    #   else: 0

    valid_ks = np.arange(i, min(i + n, 2*n-1))  # k values where k-i is in [0, n-1]
    for k in valid_ks:
        j = k - i
        if k == 2*i:
            base[k] = c[k] - ai*ai
        else:
            base[k] = c[k] - 2*ai*a[j]
    
    # Now minimize max(base[2i] + x^2, max_{k in valid_ks, k!=2i}(base[k] + 2*a[k-i]*x), max_{k not in valid_ks}(base[k]))
    # for x >= 0
    
    # The "constant" part (k not in valid_ks):
    const_max = base[~np.isin(np.arange(2*n-1), valid_ks)].max() if len(valid_ks) < 2*n-1 else 0.0
    # Actually, base[k] for k not in valid_ks is just c[k] (no contribution from a[i])
    # Let me just compute it as:
    all_ks = np.arange(2*n-1)
    const_mask = ~np.isin(all_ks, valid_ks)
    if np.any(const_mask):
        const_max = float(c[const_mask].max())
    else:
        const_max = 0.0
    
    # The quadratic: x^2 + base[2i] (if 2i is a valid index, i.e., 0 <= 2i < 2n-1, which is always true for i < n)
    quad_base = base[2*i] if 2*i < 2*n-1 else 0.0
    
    # The linear terms: for k in valid_ks, k != 2i: slope = 2*a[k-i], intercept = base[k]
    linear_ks = valid_ks[valid_ks != 2*i]
    if len(linear_ks) > 0:
        slopes = 2.0 * a[linear_ks - i]
        intercepts = base[linear_ks]
    else:
        slopes = np.array([])
        intercepts = np.array([])
    
    # Minimize f(x) = max(x^2 + quad_base, max(slopes*x + intercepts), const_max) for x >= 0
    # This is convex. Use bisection on x.
    
    # Upper bound on optimal x: if all slopes are 0, the quadratic dominates, and x* = 0.
    # A reasonable upper bound: current value * 2, or based on the slopes.
    x_lo = 0.0
    x_hi = max(ai * 3, np.abs(const_max - quad_base) + 1.0, 1.0)
    # Make sure x_hi is large enough
    for _ in range(20):
        f_hi = max(x_hi**2 + quad_base, 
                   np.max(slopes * x_hi + intercepts) if len(slopes) > 0 else -np.inf,
                   const_max)
        f_lo = max(x_lo**2 + quad_base,
                   np.max(slopes * x_lo + intercepts) if len(slopes) > 0 else -np.inf,
                   const_max)
        if f_hi > f_lo + 1e-15:
            x_hi = x_lo + (x_hi - x_lo) * 2
            continue
        break
    
    # Bisection: find where subgradient changes sign
    # Subgradient of f(x) at x: max of active terms' derivatives
    # Active terms: those achieving the max
    # Derivative of x^2 + quad_base = 2x
    # Derivative of slope*x + intercept = slope
    # Derivative of const_max = 0
    # Subgradient = max of active derivatives (for convex max, it's the set of convex combos,
    # but for finding zero crossing, we can use the max derivative)
    
    def f_and_subgrad(x):
        quad_val = x*x + quad_base
        vals = [quad_val, const_max]
        derivs = [2.0*x, 0.0]
        if len(slopes) > 0:
            lin_vals = slopes * x + intercepts
            vals.extend(lin_vals)
            derivs.extend(slopes)
        idx = np.argmax(vals)
        return vals[idx], derivs[idx]
    
    # Bisection on subgradient
    for _ in range(60):
        x_mid = (x_lo + x_hi) / 2
        f_mid, d_mid = f_and_subgrad(x_mid)
        if d_mid > 1e-10:
            x_hi = x_mid
        elif d_mid < -1e-10:
            x_lo = x_mid
        else:
            break
    
    x_opt = (x_lo + x_hi) / 2
    # Check x=0
    f_0, _ = f_and_subgrad(0.0)
    f_opt, _ = f_and_subgrad(x_opt)
    if f_0 <= f_opt:
        x_opt = 0.0
    
    return x_opt

Optimized step function search combining coordinate descent, Adam in log-space,
and active-set polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)
    c = _fft_convolve(a, a)

                   max_iter=200000, top_k=None, eval_interval=200):
    """Adam optimizer in log-space for a >= 0, sum(a) = 1."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)

    lr_mult = 1.0
    t = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        else:

        z_grad = a * (g - W)

def _coord_step_vectorized(a, c, i, n):
    """Find optimal value of a[i] given other coordinates fixed."""
    ai = a[i]
    
    # Valid k range: k from i to i+n-1 (clamped to [0, 2n-2])
    k_lo = i
    k_hi = min(i + n - 1, 2*n - 2)
    
    # For each valid k, the contribution of a[i] to c[k] is:
    #   if k == 2*i: a[i]^2
    #   else: 2 * a[i] * a[k-i]
    
    ks = np.arange(k_lo, k_hi + 1)
    js = ks - i  # these are in [0, n-1]
    
    # Linear coefficients (slope of c[k] as function of a[i]):
    # For k != 2i: slope = 2 * a[j], intercept = c[k] - 2*ai*a[j]
    # For k == 2i: slope is 0 for the linear part, but there's a quadratic term a[i]^2
    
    slopes = 2.0 * a[js]
    intercepts = c[ks] - 2.0 * ai * a[js]
    
    # Handle k = 2i specially:
    # c[2i] = a[i]^2 + (other terms)
    # So: c[2i] = ai^2 + base_2i, where base_2i = c[2i] - ai^2
    # As function of x: c[2i] = x^2 + base_2i
    # The "slope" interpretation doesn't apply; it's quadratic.
    
    mask_quad = (ks == 2*i)
    if np.any(mask_quad):
        # For k = 2i: c[k] = x^2 + (c[k] - ai^2)
        pass  # Handle separately
    
    # For non-quadratic terms: c[k] = slope_k * x + intercept_k
    mask_lin = ~mask_quad
    lin_slopes = slopes[mask_lin]
    lin_intercepts = intercepts[mask_lin]
    quad_intercept = c[2*i] - ai*ai if mask_quad.any() else 0.0
    
    # Also, for k NOT in [k_lo, k_hi], c[k] is constant (doesn't depend on a[i]):
    const_max = c[~np.isin(np.arange(2*n-1), ks)].max() if k_hi - k_lo < 2*n-2 else 0.0
    
    # Minimize f(x) = max(x^2 + quad_intercept, max(lin_slopes*x + lin_intercepts), const_max) for x >= 0
    
    # Use bisection on the subgradient
    x_lo, x_hi = 0.0, max(ai * 2, 1.0)
    
    # Expand x_hi if needed
    for _ in range(30):
        f_mid = max(x_hi*x_hi + quad_intercept,
                    np.max(lin_slopes * x_hi + lin_intercepts) if len(lin_slopes) > 0 else -np.inf,
                    const_max)
        f_lo = max(x_lo*x_lo + quad_intercept,
                   np.max(lin_slopes * x_lo + lin_intercepts) if len(lin_slopes) > 0 else -np.inf,
                   const_max)
        if f_mid < f_lo:
            break  # We're decreasing, x_hi is fine
        x_hi *= 2
        if x_hi > 100:
            break
    
    # Subgradient of f at x:
    # - From quadratic: 2x
    # - From each linear: slope_k
    # - From constant: 0
    # The subgradient (for convex max) is the set of convex combinations of active derivatives.
    # For bisection, we use: if max active derivative > 0, increase x; if < 0, decrease.
    
    def subgrad(x):
        quad_val = x*x + quad_intercept
        lit_vals = lin_slopes * x + lin_intercepts if len(lin_slopes) > 0 else np.array([])
        all_vals = np.concatenate([[quad_val, const_max], lit_vals]) if len(lit_vals) > 0 else np.array([quad_val, const_max])
        all_derivs = np.concatenate([[2*x, 0.0], lin_slopes]) if len(lin_slopes) > 0 else np.array([2*x, 0.0])
        max_idx = np.argmax(all_vals)
        return all_derivs[max_idx]
    
    for _ in range(100):
        x_mid = (x_lo + x_hi) / 2
        sg = subgrad(x_mid)
        if sg > 1e-12:
            x_lo = x_mid
        elif sg < -1e-12:
            x_hi = x_mid
        else:
            break
    
    x_opt = (x_lo + x_hi) / 2
    
    # Check boundary x=0
    f_opt = max(x_opt**2 + quad_intercept,
                np.max(lin_slopes * x_opt + lin_intercepts) if len(lin_slopes) > 0 else -np.inf,
                const_max)
    f_0 = max(quad_intercept,
              np.max(lin_intercepts) if len(lin_intercepts) > 0 else -np.inf,
              const_max)
    if f_0 <= f_opt:
        x_opt = 0.0
    
    return x_opt

"""
Multi-phase optimizer combining coordinate descent (exact 1D subproblem solving),
Adam in log-space with adaptive temperature, and multi-length exploration for
minimizing 2*n*max(conv(a,a))/sum(a)^2. Key innovation: vectorized coordinate
descent with exact 1D convex minimization per coordinate, alternating with
FFT-based gradient methods for global progress.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_from_p(a, p):
    n = len(a)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    return 2.0 * conv_full[n - 1:2 * n - 1][::-1]


def _soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    return _gradient_from_p(a, p)


def _topk_gradient(a, c, k=5):
    n = len(a)
    k_idx = np.argsort(c)[::-1][:k]
    p = np.zeros_like(c)
    p[k_idx] = 1.0 / k
    return _gradient_from_p(a, p)


def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end,

    z = z - z.mean()
    m_buf = np.zeros(n)

    t = 0

    while t < max_iter:
        if time.monotonic() > deadline:

        T = T_start * (T_end / T_start) ** frac

        a = _softmax_stable(z)

        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)

        if unorm > clip_norm:

        z = np.clip(z, -100.0, 100.0)

                patience = 0
            else:

                patience = 0

                z = z - z.mean()

                v_buf[:] = 0

    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_sc:

        best_a = a_final.copy()

def _coord_update(a, c, i, n):
    """Exact 1D minimization for coordinate i. Returns new value."""
    ai = a[i]
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)

    ks = np.arange(k_lo, k_hi + 1)
    js = ks - i

    slopes = 2.0 * a[js]
    intercepts = c[ks] - 2.0 * ai * a[js]

    mask_quad = (ks == 2 * i)
    if np.any(mask_quad):
        quad_base = c[2 * i] - ai * ai
    else:
        quad_base = 0.0

    mask_lin = ~mask_quad
    lin_sl = slopes[mask_lin]
    lin_ic = intercepts[mask_lin]

    all_ks = np.arange(2 * n - 1)
    const_mask = ~np.isin(all_ks, ks)
    if np.any(const_mask):
        const_max = float(c[const_mask].max())
    else:
        const_max = 0.0

    x_lo = 0.0
    x_hi = max(ai * 3.0, 0.5)

    for _ in range(25):
        f_hi = max(x_hi * x_hi + quad_base,
                   float(np.max(lin_sl * x_hi + lin_ic)) if len(lin_sl) > 0 else -1e18,
                   const_max)
        f_lo = max(x_lo * x_lo + quad_base,
                   float(np.max(lin_sl * x_lo + lin_ic)) if len(lin_sl) > 0 else -1e18,
                   const_max)
        if f_hi < f_lo + 1e-14:
            break
        x_hi *= 2.0
        if x_hi > 50.0:
            break

    for _ in range(80):
        x_mid = 0.5 * (x_lo + x_hi)
        qv = x_mid * x_mid + quad_base
        if len(lin_sl) > 0:
            lv = np.max(lin_sl * x_mid + lin_ic)
        else:
            lv = -1e18
        if qv >= lv and qv >= const_max:
            sg = 2.0 * x_mid
        elif lv >= qv and lv >= const_max:
            sg = float(lin_sl[np.argmax(lin_sl * x_mid + lin_ic)]) if len(lin_sl) > 0 else 0.0
        else:
            sg = 0.0
        if sg > 1e-10:
            x_lo = x_mid
        elif sg < -1e-10:
            x_hi = x_mid
        else:
            break

    x_opt = 0.5 * (x_lo + x_hi)

    f_opt = max(x_opt * x_opt + quad_base,
                float(np.max(lin_sl * x_opt + lin_ic)) if len(lin_sl) > 0 else -1e18,
                const_max)
    f_0 = max(quad_base,
              float(np.max(lin_ic)) if len(lin_ic) > 0 else -1e18,
              const_max)
    if f_0 <= f_opt:
        x_opt = 0.0

def _coordinate_descent(a, deadline, max_sweeps=200, rng=None):
    """Coordinate descent with exact 1D minimization. Returns (a, best_score_norm)."""
    n = len(a)
    a = a.copy()
    c = _fft_convolve(a, a)
    best_sc = float(c.max())
    sweep = 0
    stall_count = 0

    while sweep < max_sweeps:
        if time.monotonic() > deadline:
            break

        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        prev_max = float(c.max())
        changed = 0

        for i in order:
            x_new = _coord_update(a, c, i, n)
            if abs(x_new - a[i]) > 1e-12:
                delta = x_new - a[i]
                old_ai = a[i]
                a[i] = x_new
                # Incrementally update c
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                ks = np.arange(k_lo, k_hi + 1)
                js = ks - i
                mask_quad = (ks == 2 * i)
                # For k != 2i: c[k] += 2 * delta * a[k-i]
                mask_lin = ~mask_quad
                c[ks[mask_lin]] += 2.0 * delta * a[js[mask_lin]]
                # For k == 2i: c[k] changes by (x_new^2 - old_ai^2)
                if np.any(mask_quad):
                    kq = int(ks[mask_quad][0])
                    c[kq] = c[kq] - old_ai * old_ai + x_new * x_new
                changed += 1

        cur_max = float(c.max())
        if cur_max < best_sc - 1e-14:
            best_sc = cur_max
            stall_count = 0
        else:
            stall_count += 1

        if changed == 0 or (stall_count > 15 and sweep > 10):
            break
        sweep += 1

    a = a / a.sum()
    c = _fft_convolve(a, a)
    best_sc = float(c.max())
    return a, best_sc


def _subgradient_polish(a, deadline, max_iter=5000, top_k=5):
    """Simple subgradient method with averaging for final polish."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    alpha = 1.0
    z_avg = np.zeros(n)
    c = _fft_convolve(a, a)
    max_c = float(c.max())

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        max_c = float(c.max())
        g = _topk_gradient(a, c, top_k)
        W = float(np.dot(a, g))
        a_new = a - alpha * (g - W)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s
        z_avg += a_new
        a = a_new
        alpha = 1.0 / (np.sqrt(t + 1) + 1.0)
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    # Also check the averaged iterate
    a_avg = z_avg / (t + 1)
    a_avg = np.maximum(a_avg, 0)
    if a_avg.sum() > 0:
        a_avg = a_avg / a_avg.sum()
        sc_avg = _score_norm(a_avg)
        if sc_avg < best_sc:
            best_sc = sc_avg
            best_a = a_avg.copy()

    return best_a, best_sc


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
