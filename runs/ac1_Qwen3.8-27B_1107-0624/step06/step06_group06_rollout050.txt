        # where w_k = 0 for k < 0 or k >= 2n-1

        # Part 2: h2_i = sum_j a_j w_{i-j} = (a conv w)_i  [convolution of a with w]

        # h2_i for i=0..n-1: sum_j a_j w_{i-j}, this is the standard convolution evaluated at i
        # h1_i for i=0..n-1: sum_j a_j w_{j+i}, this is cross-correlation

        # h2 = _full_convolve(a, w) -> gives all 2n-1 values, we take first n
        # h1 = _full_convolve(a, w[::-1]) -> this gives a * rev(w) which is... 
        # Actually conv(a, rev(w))[i] = sum_j a_j * rev(w)[i-j] = sum_j a_j * w[L-1-(i-j)] where L=len(w)=2n-1
        # = sum_j a_j * w[2n-2-i+j]
        # Let k = j: h1_i = sum_j a_j w_{2n-2-i+j}... that's not quite what I want.
        
        # Let me just use a different approach. Since n can be up to 8192, direct computation
        # of g would be O(n^2) which is too slow. Let me use FFT.
        
        # The key identity: g = conv(w, a_rev)[n-1 : 2n-1] + conv(w, a)[0 : n]
        # Wait, let me verify:
        # conv(w, a_rev)[k] = sum_j w_j a_rev[k-j] = sum_j w_j a[n-1-(k-j)] = sum_j w_j a[n-1-k+j]

        # conv(w, a_rev)[n-1+i] = sum_j w_j a[i+j]
        # This is exactly h1_i = sum_j a_j w_{j+i} (by symmetry of the sum, renaming j->i... 
        # actually sum_j w_j a_{i+j} = sum_j a_j w_{i+j} ... yes by commutativity of addition)
        
        # conv(w, a)[k] = sum_j w_j a[k-j]
        # For k in 0..n-1: conv(w, a)[i] = sum_j w_j a[i-j] = sum_j a_j w_{i-j} = h2_i
        
        # So: g[i] = conv(w, a_rev)[n-1+i] + conv(w, a)[i] for i = 0..n-1

        wa_rev = _full_convolve(w, a_rev)
        wa = _full_convolve(w, a)
        g = wa_rev[n-1:2*n-1] + wa[0:n]

        ag = float(a @ g)
        grad = coef * a * (g - ag)
    
    val = 2.0 * n * fm
    return val, grad

Let me think about spectral equalization more carefully:

If `A = FFT(a)`, then `|A|^2` is (up to a constant) the periodogram. The autocorrelation is the inverse FFT of `|A|^2`. So making the autocorrelation flat is equivalent to making `|A|^2` flat (which is already the case for the "flat" part) - wait no, that's circular.

Actually: `c = ifft(fft(a) * conj(fft(a)))` (for circular correlation). For linear correlation, it's slightly different but similar.

So `c[k] = ifft(|A|^2)[k]`. We want `c[k]` to be approximately constant for all k. By the inversion, this means `|A|^2 = fft(c)` should be approximately constant, which means `|A[f]|` should be approximately constant for all f.

So the goal is: find a non-negative sequence `a` with `sum a = 1` such that `|FFT(a)[f]|` is as constant as possible.

This is a non-trivial constraint because `a >= 0` restricts the possible Fourier transforms significantly.

**Spectral equalization approach:**
1. Start with current best `a`
2. Compute `A = FFT(a)`
3. Compute target magnitude: `|A_target| = sqrt(sum|A|^2 / n)` (constant magnitude)
4. Update: `A_new = |A_target| * exp(i*angle(A))` (keep phase, change magnitude)
5. `a_new = ifft(A_new).real`
6. Project to non-negative: `a_new = max(a_new, 0)` and renormalize
7. Repeat

This is essentially the "Gerchberg-Saxton" or "error reduction" algorithm used in optics for phase retrieval! It alternates between the frequency domain (enforcing flat magnitude) and the time domain (enforcing non-negativity).

This is a fundamentally different approach from gradient-based optimization and might find different local minima!

Let me combine this with the gradient-based approach:
- Alternate between spectral equalization steps and gradient descent steps
- Use the gradient method to fine-tune within a basin
- Use spectral equalization to jump between basins

Wait, but this might not work well because the constant magnitude constraint in frequency doesn't directly correspond to flat autocorrelation in time (for linear vs circular correlation). Let me think more carefully.

For circular correlation of length n: `r[k] = sum_i a[i] a[(i+k) mod n]`, and `r = ifft(|FFT(a)|^2)`.

For linear correlation: `c[k] = sum_i a[i] a[i+k]` (with zero-padding).

If we zero-pad to length N >= 2n-1, then the circular correlation of the zero-padded sequence equals the linear correlation. So:

`c[k] = ifft_N(|FFT_N(a)|^2)[k]` for k = 0, ..., N-1

where N is the zero-padded length and `a` is zero-padded to length N.

So if we want `c[k]` to be constant (for the meaningful range k=0..2n-2), we need `FFT_N(c)` to be constant, which means `|FFT_N(a)|^2` should be constant.

So the spectral equalization approach IS valid for linear correlation! We just need to zero-pad properly.

"""
Spectral-equalized hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines Gerchberg-Saxton spectral equalization (fundamentally different from
gradient methods) with L-BFGS log-sum-exp smoothing and subgradient polishing.
Key novelty: error-reduction iteration in frequency domain to flatten autocorrelation,
combined with adaptive active-set gradient descent for fine-tuning.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _spectral_equalize(a, n_iter=500, overrelax=1.0, deadline=None,
                       soft_alpha=1.0, phase_jitter=0.0, rng=None):
    """
    Gerchberg-Saxton style spectral equalization.
    Alternates between:
      - Frequency domain: flatten |FFT(a)| (toward constant magnitude)
      - Time domain: project to non-negative simplex
    """
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)

    for i in range(n_iter):
        if deadline is not None and time.monotonic() > deadline - 1:
            break

        # Frequency domain step
        A = np.fft.rfft(a, N)
        mag = np.sqrt(A.real ** 2 + A.imag ** 2)
        # Target: constant magnitude
        mean_mag_sq = (A.real ** 2 + A.imag ** 2).sum() / (N // 2 + 1)
        target_mag = np.sqrt(mean_mag_sq)

        # Soft blending of current and target magnitude
        mag_new = mag ** (1.0 - soft_alpha) * target_mag ** soft_alpha

        # Optional phase jitter for exploration
        if phase_jitter > 0 and rng is not None:
            phase = np.angle(A) + phase_jitter * rng.standard_normal(len(A))
        else:
            phase = np.angle(A)

        A_new = mag_new * np.exp(1j * phase)

        # Over-relaxation in frequency domain
        A_new = (1.0 - overrelax) * A + overrelax * A_new

        # Time domain step: inverse FFT and project
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        # Check improvement periodically
        if i % 20 == 0 or i == n_iter - 1:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _spectral_equalize_band(a, n_iter=300, band=None, deadline=None, rng=None):
    """
    Spectral equalization with band-limited target.
    Instead of constant magnitude everywhere, target a raised-cosine shape
    to handle edge effects of linear correlation.
    """
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score(a)

    # Design band target: raised cosine
    freqs = np.linspace(0, 1, N // 2 + 1)
    if band is None:
        band = 0.85
    target_shape = 0.5 + 0.5 * np.cos(np.pi * freqs / band)
    target_shape[freqs > band] = 0.0
    target_shape /= target_shape[0] if target_shape[0] > 0 else 1.0

    for i in range(n_iter):
        if deadline is not None and time.monotonic() > deadline - 1:
            break
        A = np.fft.rfft(a, N)
        mag = np.sqrt(A.real ** 2 + A.imag ** 2)
        phase = np.angle(A)
        # Blend toward target shape
        mean_mag_sq = (A.real ** 2 + A.imag ** 2).sum() / (N // 2 + 1)
        base_mag = np.sqrt(mean_mag_sq)
        target_mag = base_mag * target_shape
        # Soft blend
        alpha = 0.5
        mag_new = mag ** (1 - alpha) * target_mag ** alpha
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        if i % 20 == 0 or i == n_iter - 1:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient descent with adaptive learning rate on log-parameterized simplex."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.002)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with adaptive beta continuation on log-parameterized simplex."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015,
                                  0.0075, 0.00375, 0.001875, 0.00094, 0.00047, 0.00024])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage(a, deadline, fractions=None):
    """Zero out smallest elements and re-evaluate."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    if fractions is None:
        fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20]
    a_sorted_idx = np.argsort(a)

    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "m_seq_like":
        # Mimic m-sequence structure projected to non-negative
        t = np.arange(n)
        a = 1.0 + 0.8 * np.sign(np.sin(2 * np.pi * 1.0 * t / n) *
                                 np.sin(2 * np.pi * 3.0 * t / n) *
                                 np.sin(2 * np.pi * 7.0 * t / n))
        a = np.maximum(a, 0.1)
        return a / a.sum()
    elif kind == "zadoff_like":
        # Zadoff-Chu inspired: constant amplitude in frequency
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t * t / n)
        a = np.maximum(a, 0.1)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_spectral(a, rng, n_modes=5):
    """Perturb in the spectral domain (add random sinusoidal components to phase)."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    mag = np.sqrt(A.real ** 2 + A.imag ** 2)
    phase = np.angle(A)
    for _ in range(n_modes):
        freq_idx = rng.integers(1, max(2, N // 4))
        amp = 0.1 * rng.uniform(0.5, 2.0)
        phase = phase + amp / np.maximum(mag, 1e-10) * np.cos(2 * np.pi * freq_idx * np.arange(N // 2 + 1) / N)
    A_new = mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def _fista_power_mean(a_init, deadline, p=16, n_max_iter=3000):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) style optimization
    using power-mean smoothing. Works in log-space with softmax.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    z_old = z.copy()
    t_k = 1.0
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.05

    for i in range(n_max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_pos = np.maximum(c, 1e-30)

        # Power mean objective gradient
        cp = c_pos ** (p - 1)
        S_cp = (c_pos ** p).sum()
        if S_cp < 1e-300:
            break
        coef = 2.0 * n * S_cp ** (1.0 / p - 1.0)

        # Compute g_i = sum_k cp_k * dc_k/da_i
        a_rev = a[::-1]
        g1 = _full_convolve(cp, a_rev)[n - 1:2 * n - 1]
        g2 = _full_convolve(cp, a)
        g = g1 + g2[:n]

        ag = float(a @ g)
        grad_z = coef * a * (g - ag)

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        z_corr = z + (t_k / t_new - 1.0) * (z - z_old)
        z_old = z.copy()
        t_k = t_new

        z = z_corr - lr * grad_z
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.02, 0.1)
            else:
                # Try larger p for sharper max approximation
                lr *= 0.9

    return best_a, best_sc


def _active_set_newton(a_init, deadline, max_iter=200):
    """Projected gradient with Barzilai-Borwein step size on active set."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    alpha = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        grad = _gradient_active(a, c, n, tol_frac=0.002)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-13:
            break

        # BB step size
        if it > 0 and hasattr(_active_set_newton, '_prev_grad_t'):
            prev_grad = _active_set_newton._prev_grad_t
            prev_a = _active_set_newton._prev_a
            s_vec = a - prev_a
            y_vec = grad_t - prev_grad
            sy = float(s_vec @ y_vec)
            yy = float(y_vec @ y_vec)
            ss = float(s_vec @ s_vec)
            if yy > 1e-16:
                alpha_bb = sy / yy if sy > 1e-16 else 1.0
                alpha = np.clip(alpha_bb, 0.01, 10.0)

        _active_set_newton._prev_grad_t = grad_t.copy()
        _active_set_newton._prev_a = a.copy()

        # Backtracking line search
        a_new = a.copy()
        f_old = c_max
        for ls in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid spectral-gradient search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses Gerchberg-Saxton spectral equalization as the primary exploration tool,
    combined with L-BFGS smoothing, FISTA power-mean, and active-set Newton methods.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === PHASE 1: Spectral Equalization (primary novel method) ===
    print(f"\n[Phase 1] Spectral Equalization (t={time.monotonic() - t_start:.1f}s)")

    # 1a: Hard spectral equalization from parent
    if time.monotonic() < deadline - 300:
        a_se, sc_se = _spectral_equalize(parent, n_iter=1000, overrelax=1.0,
                                         deadline=deadline - 400, soft_alpha=1.0,
                                         phase_jitter=0.0, rng=rng)
        update_best(a_se, sc_se, "spectral_eq_hard")

    # 1b: Soft spectral equalization (gentler blending)
    if time.monotonic() < deadline - 250:
        a_soft, sc_soft = _spectral_equalize(parent, n_iter=1500, overrelax=1.0,
                                             deadline=deadline - 350, soft_alpha=0.5,
                                             phase_jitter=0.0, rng=rng)
        update_best(a_soft, sc_soft, "spectral_eq_soft")

    # 1c: Spectral equalization with phase jitter (exploration)
    if time.monotonic() < deadline - 200:
        a_jit, sc_jit = _spectral_equalize(parent, n_iter=800, overrelax=1.0,
                                           deadline=deadline - 300, soft_alpha=0.8,
                                           phase_jitter=0.05, rng=rng)
        update_best(a_jit, sc_jit, "spectral_eq_jitter")

    # 1d: Band-limited spectral equalization
    if time.monotonic() < deadline - 150:
        a_band, sc_band = _spectral_equalize_band(parent, n_iter=800, band=0.8,
                                                   deadline=deadline - 250, rng=rng)
        update_best(a_band, sc_band, "spectral_band")

    # === PHASE 2: L-BFGS Polish on best spectral result ===
    print(f"\n[Phase 2] L-BFGS Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_lb = _run_lbfgs(best_seq, deadline, min_time_remaining=120.0)
        update_best(a_lb, _score(a_lb), "lbfgs_polish")

    # === PHASE 3: FISTA Power-Mean ===
    print(f"\n[Phase 3] FISTA Power-Mean (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        for p_val in [16, 32, 64]:
            if time.monotonic() > deadline - 60:
                break
            a_fista, sc_fista = _fista_power_mean(best_seq, min(deadline - 50,
                                                                time.monotonic() + 60),
                                                  p=p_val, n_max_iter=3000)
            update_best(a_fista, sc_fista, f"fista_p{p_val}")
            if sc_fista < best_score:
                best_seq = a_fista.copy()

    # === PHASE 4: Active Set Newton ===
    print(f"\n[Phase 4] Active Set Newton (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_newton, sc_newton = _active_set_newton(best_seq, min(deadline - 30,
                                                               time.monotonic() + 60),
                                                 max_iter=500)
        update_best(a_newton, sc_newton, "newton")

    # === PHASE 5: Subgradient Polish ===
    print(f"\n[Phase 5] Subgradient Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 20, time.monotonic() + 60),
                                      n_max_steps=5000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_polish")

    # === PHASE 6: Multi-n Exploration ===
    print(f"\n[Phase 6] Multi-n Exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 6144, 8192]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        # Resample or create
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        elif n_try == n_parent:
            a_init = parent.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        # Spectral equalize
        if time.monotonic() < deadline - 40:
            a_se_n, sc_se_n = _spectral_equalize(a_init, n_iter=500, overrelax=1.0,
                                                 deadline=deadline - 40, soft_alpha=0.7,
                                                 phase_jitter=0.0, rng=rng)
            update_best(a_se_n, sc_se_n, f"n{n_try}_spec")

            # Quick L-BFGS
            if sc_se_n < best_score + 0.01 and time.monotonic() < deadline - 30:
                a_lb_n = _run_lbfgs(a_se_n, deadline, min_time_remaining=20.0)
                update_best(a_lb_n, _score(a_lb_n), f"n{n_try}_lbfgs")

    # === PHASE 7: Spectral Perturbation + Re-optimize ===
    print(f"\n[Phase 7] Spectral Perturbation (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for i in range(3):
            if time.monotonic() > deadline - 20:
                break
            a_pert = _perturb_spectral(best_seq, rng, n_modes=5)
            sc_pert = _score(a_pert)
            # Quick spectral equalize from perturbed
            if time.monotonic() < deadline - 15:
                a_se_p, sc_se_p = _spectral_equalize(a_pert, n_iter=200, overrelax=1.0,
                                                     deadline=deadline - 15, soft_alpha=0.8,
                                                     phase_jitter=0.0, rng=rng)
                update_best(a_se_p, sc_se_p, f"perturb_spec_{i}")

    # === PHASE 8: Final Fine Polish ===
    print(f"\n[Phase 8] Final Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very fine L-BFGS
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0001
        remaining = deadline - 5 - time.monotonic()
        if remaining > 3:
            max_iter = min(3000, max(100, int(remaining / 0.03)))
            def final_fun(z_local):
                return _smoobj(z_local, beta_final, n)

            def final_jac(z_local):
                return _smograd(z_local, beta_final, n)

            try:
                result = minimize(
                    final_fun, z, jac=final_jac, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                             'maxcor': 50, 'maxls': 100}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2,
                                              n_max_steps=500, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_sg")

        # Final shrinkage
        if time.monotonic() < deadline - 2:
            a_shrink, sc_shrink = _shrinkage(best_seq, deadline - 2,
                                             fractions=[0.005, 0.01, 0.02, 0.03])
            update_best(a_shrink, sc_shrink, "final_shrink")

    # --- Final validation and return ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
