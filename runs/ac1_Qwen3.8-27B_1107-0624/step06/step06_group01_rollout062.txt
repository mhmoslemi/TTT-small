```python
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _io_iteration(a, nu=1.0):
    """One IO iteration: flatten the spectrum magnitude."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    # Target: flat magnitude
    mag = np.abs(A)
    # Use the mean magnitude as target (or could use a specific value)
    target_mag = np.mean(mag)
    # New spectrum with flat magnitude, same phase
    phase = A / (mag + 1e-300)
    A_new = target_mag * phase
    # Inverse FFT
    a_new = np.fft.irfft(A_new, N)[:n]
    # Enforce non-negativity and sum=1
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new

def _hio_iteration(a, gamma=0.85, beta=1.2):
    """One HIO (EOS) iteration."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    target_mag = np.mean(mag)
    phase = A / (mag + 1e-300)
    A_ge = target_mag * phase  # constraint-set projection in freq domain
    a_ge = np.fft.irfft(A_ge, N)[:n]  # inverse FFT of freq projection
    
    # Error

    # For simplicity, let's use a simple relaxation
    a_proj = np.maximum(a, 0.0)
    s = a_proj.sum()
    if s > 1e-15:
        a_proj = a_proj / s
    
    # HIO update
    a_new = a + gamma * (a_ge - a)  # move towards freq solution
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    
    return a_new

def _io_optimize(a_init, n_iter, nu=1.0, deadline=None):
```

```python
    n = len(a)
    N = _next_pow2(3*n - 2)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    C = np.fft.irfft(np.conj(A) * W, N)
    return (2.0 / beta) * C[:n]
```

```python
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _compute_conv_and_grad(a, beta):
    """Compute convolution and gradient of smooth max."""
    n = len(a)
    c = _fft_convolve(a, a)
    # Softmax weights
    c_shift = (c - c.max()) / beta
    e = np.exp(c_shift)
    w = e / e.sum()
    # Gradient via cross-correlation
    N = _next_pow2(3 * n - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    C = np.fft.irfft(np.conj(A) * W, N)
    grad = (2.0 / beta) * C[:n]
    return c, w, grad

def _proj_simplex(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    best_score = _score_norm(a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

        if time.monotonic() > deadline:
```

```python
Optimized minimizer for 2n*max(conv(a,a))/sum(a)^2 using:
- FFT-based gradient of smooth max via cross-correlation (3 FFTs/iter)
- Adam with adaptive beta schedule and buffer resets
- IO (Gerchberg-Saxton) spectral flattening for diversification

- Multi-length exploration (n=4096, 8192)

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_c_and_grad(a, beta):
    n = len(a)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    c_shift = (c - c_max) / beta
    e = np.exp(c_shift)
    e_sum = float(e.sum())
    if e_sum < 1e-300 or not np.isfinite(e_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / e_sum
    N = _next_pow2(3 * n - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    C = np.fft.irfft(np.conj(A) * W, N)
    grad = (2.0 / beta) * C[:n]
    return c, w, grad, c_max


def _proj_simplex(v, target=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _io_iteration(a):
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    target_mag = np.mean(mag)
    phase = A / (mag + 1e-300)
    A_new = target_mag * phase
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_final.copy()
    return best_a, best_score


def _make_window(n, kind="uniform"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "raised_cosine":
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    elif kind == "blackman":
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)
    elif kind == "blackman_harris":
        a = (0.35875 - 0.48829 * np.cos(2.0 * np.pi * t) +
             0.14128 * np.cos(4.0 * np.pi * t) - 0.01168 * np.cos(6.0 * np.pi * t))
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
    elif kind == "asymmetric":
        a = np.exp(-3.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-3.0 * (t - 0.7) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_sequence(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - np.max(z))
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return e / s


def _coordinate_polish(a, deadline, n_trials=5000):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(99)
    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break
        i = rng.integers(0, n)
        for f in [0.7, 0.85, 1.15, 1.3]:
            a_try = best_a.copy()
            a_try[i] *= f
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                c = _fft_convolve(a_try, a_try)
                sc = float(2.0 * n * c.max())
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    break
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _make_window(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, using uniform n={n_parent}, score={best_score:.10f}")

    # ===== Phase 1: IO pre-optimization for diversification (~60s) =====
    print(f"\n=== Phase 1: IO pre-optimization ===")
    io_deadline = t_start + 60.0
    a_io, sc_io = _io_optimize(parent, 5000, io_deadline, use_hio=False)
    sc_io_full = _score(a_io)
    if sc_io_full < best_score:
        best_score = sc_io_full
        best_seq = a_io.copy()
        print(f"  IO improved: score={sc_io_full:.10f}")
    a_hio, sc_hio = _io_optimize(parent, 5000, min(io_deadline + 30, deadline), use_hio=True, gamma=0.8)
    sc_hio_full = _score(a_hio)
    if sc_hio_full < best_score:
        best_score = sc_hio_full
        best_seq = a_hio.copy()
        print(f"  HIO improved: score={sc_hio_full:.10f}")

    # ===== Phase 2: Main Adam optimization n=4096 (~500s) =====
    print(f"\n=== Phase 2: Main optimization n={n_parent} ===")
    if n_parent == 4096:
        phase2_deadline = t_start + 560.0
        c_init = _fft_convolve(parent, parent)
        max_c = float(c_init.max())
        betas = [0.1 * max_c, 0.05 * max_c, 0.02 * max_c, 0.01 * max_c,
                 0.005 * max_c, 0.002 * max_c, 0.001 * max_c, 0.0005 * max_c, 0.0002 * max_c]
        lrs = [0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01]

        # Main Adam in simplex space
        a_opt, sc_opt = _adam_optimize(parent, phase2_deadline, betas, lrs, 80000)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam simplex: score={sc_opt_full:.10f}")

        # Adam in log-space for fine tuning
        if time.monotonic() < phase2_deadline - 120:
            betas_fine = [0.002 * max_c, 0.001 * max_c, 0.0005 * max_c, 0.0002 * max_c, 0.0001 * max_c, 0.00005 * max_c]
            a_opt2, sc_opt2 = _adam_logspace(a_opt, min(phase2_deadline - 60, time.monotonic() + 300),
                                             betas_fine, lr_init=0.02, max_iter_per_phase=50000)
            sc_opt2_full = _score(a_opt2)
            if sc_opt2_full < best_score:
                best_score = sc_opt2_full
                best_seq = a_opt2.copy()
                print(f"  Adam log-space: score={sc_opt2_full:.10f}")

    # ===== Phase 3: Try n=8192 (~400s) =====
    print(f"\n=== Phase 3: n=8192 exploration ===")
    n8 = 8192
    phase3_deadline = t_start + 960.0
    if time.monotonic() < phase3_deadline - 100:
        a_8192 = _interp_sequence(best_seq, n8)
        c_8192 = _fft_convolve(a_8192, a_8192)
        max_c_8 = float(c_8192.max())
        betas_8 = [0.05 * max_c_8, 0.02 * max_c_8, 0.01 * max_c_8,
                   0.005 * max_c_8, 0.002 * max_c_8, 0.001 * max_c_8]
        lrs_8 = [0.08, 0.06, 0.05, 0.04, 0.03, 0.02]

        a_opt8, sc_opt8 = _adam_optimize(a_8192, phase3_deadline, betas_8, lrs_8, 60000)
        sc_opt8_full = _score(a_opt8)
        if sc_opt8_full < best_score:
            best_score = sc_opt8_full
            best_seq = a_opt8.copy()
            print(f"  n=8192 Adam: score={sc_opt8_full:.10f}")
        else:
            print(f"  n=8192 best: score={sc_opt8_full:.10f} (not better)")

    # ===== Phase 4: Multi-restart diversification (~400s) =====
    print(f"\n=== Phase 4: Multi-restart ===")
    phase4_deadline = deadline - 60
    if time.monotonic() < phase4_deadline:
        n_best = len(best_seq)
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        restarts = [
            ("perturb_small", _perturb(a_work, 0.02, rng)),
            ("perturb_med", _perturb(a_work, 0.05, rng)),
            ("shift", np.roll(a_work, rng.integers(2, max(3, n_best // 2048))).copy()),
            ("blend_window", 0.8 * a_work + 0.2 * _make_window(n_best, "raised_cosine")),
            ("io_diverse", _io_optimize(_perturb(a_work, 0.1, rng), 2000, time.monotonic() + 20)[0]),
        ]
        for r in restarts:
            r[1] = np.maximum(r[1], 0.0)
            s = r[1].sum()
            if s > 1e-15:
                r[1] = r[1] / s

        per_restart = max(10.0, (phase4_deadline - time.monotonic()) / len(restarts))
        for name, a_start in restarts:
            if time.monotonic() > phase4_deadline:
                break
            c_s = _fft_convolve(a_start, a_start)
            max_cs = float(c_s.max())
            betas_r = [0.005 * max_cs, 0.002 * max_cs, 0.001 * max_cs, 0.0005 * max_cs, 0.0002 * max_cs]
            a_r, sc_r = _adam_logspace(a_start, time.monotonic() + per_restart,
                                       betas_r, lr_init=0.015, max_iter_per_phase=40000)
            sc_r_full = _score(a_r)
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_seq = a_r.copy()
                print(f"  Restart '{name}': score={sc_r_full:.10f}")
            else:
                print(f"  Restart '{name}': score={sc_r_full:.10f}")

    # ===== Phase 5: Final ultra-fine polish (~50s) =====
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.maximum(a_pol, 0.0)
        a_pol = a_pol / a_pol.sum()
        c_pol = _fft_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.0001 * max_cp, 0.00005 * max_cp, 0.00002 * max_cp]
        a_pol2, sc_pol2 = _adam_logspace(a_pol, deadline - 5,
                                         betas_ult, lr_init=0.005, max_iter_per_phase=30000)
        sc_pol2_full = _score(a_pol2)
        if sc_pol2_full < best_score:
            best_score = sc_pol2_full
            best_seq = a_pol2.copy()
            print(f"  Final log-space: score={sc_pol2_full:.10f}")

        # Coordinate descent
        if time.monotonic() < deadline - 5:
            a_cd, sc_cd = _coordinate_polish(best_seq, deadline - 2, n_trials=2000)
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
                print(f"  Coord descent: score={sc_cd_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        final_sum = float(final.sum())
        if final_sum < 0.01:
            final = _make_window(256, "uniform")
            final = final * 1.0  # keep sum=1
        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_window(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
