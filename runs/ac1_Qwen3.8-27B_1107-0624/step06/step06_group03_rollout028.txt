This is a "sliding dot product" or "cross-correlation" that can be computed via FFT.

We want h_i = C for all i, i.e., (lambda * a) = C * 1.

In the frequency domain: FFT(lambda) * FFT(a) = C * FFT(1) = C * n * delta_0

This means: the product of the FFTs should be concentrated at frequency 0 only. This is a very strong constraint and generally can't be satisfied exactly.

In practice, we want to minimize ||(lambda * a) - C * 1||^2 subject to sum(a) = 1, a >= 0.

This is a quadratic optimization problem with a quadratic objective (since (lambda*a) is linear in a, the squared norm is quadratic).

minimize ||H @ a - C * 1||^2 + (sum(a) - 1)^2
subject to a >= 0

where H is the matrix with H[i,j] = lambda[i+j].

This is a non-negative least squares problem (with an extra constraint). We can solve it with projected gradient, active set, or other methods.

Actually, let me simplify. If we fix C (or incorporate it), the problem becomes:

subject to sum(a) = 1, a >= 0

But C is also unknown, so really:
minimize_{a, C} ||H @ a - C * 1||^2
subject to sum(a) = 1, a >= 0

For fixed a, the optimal C is the projection: C = (1^T H a) / (1^T 1) = mean(H a).

So the problem reduces to:
minimize_a ||H @ a - mean(H a) * 1||^2
subject to sum(a) = 1, a >= 0

This is minimizing the variance of the cross-correlation H@a, i.e., making it as flat as possible!

This is a nice reformulation. And the gradient with respect to a is:
2 * H^T * (H @ a - mean(H a) * 1)

where H^T is the transpose of H (which is also a convolution-like operation).

This is essentially what the gradient-based methods are already doing, but with a more principled formulation.

    return conv[n - 1: n - 1 + n]


def _hankel_solve(lam, rhs, max_iter=None, tol=1e-10):
    """Solve M @ x = rhs where M[i,j] = lam[i+j] using CG."""
    n = len(rhs)
    if max_iter is None:
        max_iter = 3 * n
    x = np.zeros(n)
    r = rhs - _hankel_matvec(lam, x)
    p = r.copy()
    rs = float(r @ r)
    rhs_norm = float(np.linalg.norm(rhs))
    if rhs_norm < 1e-15:
        return x
    for _ in range(max_iter):
        Ap = _hankel_matvec(lam, p)
        pAp = float(p @ Ap)
        if abs(pAp) < 1e-30:
            break
        alpha = rs / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = float(r_new @ r_new)
        if np.sqrt(rs_new) < tol * rhs_norm:
            break
        beta = rs_new / rs
        p = r_new + beta * p
        r = r_new
        rs = rs_new
    return x


def _kkt_solve(a, active_frac=0.05, temperature=None, max_cg_iter=None):
    """Solve KKT conditions: find a' on simplex minimizing max conv given active set.
    Returns (a_new, success_flag)."""
    n = len(a)
    c = _full_convolve(a, a)
    max_c = float(c.max())
    
    if temperature is None:
        temperature = 0.005 * max_c
    
    # Soft weighting on active conv positions
    lam = np.exp((c - max_c) / temperature)
    lam = lam / lam.sum()
    
    # Ensure lam has length 2n-1 (it does, c has length 2n-1)
    # Solve M @ x = 1 where M[i,j] = lam[i+j]
    rhs = np.ones(n)
    x = _hankel_solve(lam, rhs, max_iter=max_cg_iter)
    
    # a = x / sum(x), clip to non-negative
    x = np.maximum(x, 0)
    s = float(x.sum())
    if s < 1e-15:
        return a, False
    a_new = x / s
    
    # Verify improvement
    sc_old = _score_norm(a)
    sc_new = _score_norm(a_new)
    if sc_new < sc_old:
        return a_new, True
    return a, False


def _exact_coord_opt(a, i, c, top_k=200):
    """For coordinate i, find optimal value by 1D search.
    c is the current convolution. Returns optimal value for a[i]."""
    n = len(a)
    a_i = a[i]
    
    # d[j] = a[j-i] for valid j (where 0 <= j-i < n)
    d = np.zeros(2 * n - 1)
    start = i
    end = min(i + n, 2 * n - 1)
    d[start:end] = a[start - i: end - i]
    
    # c_base[j] = c[j] - a_i * d[j]
    c_base = c - a_i * d
    
    # S0 = sum(a) - a_i = 1 - a_i (assuming sum=1)
    S0 = 1.0 - a_i
    
    # We want to minimize max_j (c_base[j] + t*d[j]) / (S0+t)^2
    # = max_j (c_base[j] + t*d[j]) / (S0+t)^2
    # Critical points: for each j with d[j] > 0, the function (c_base[j] + t*d[j])/(S0+t)^2
    # has maximum at t = (c_base[j]/d[j] - S0) / ... 
    # Actually f(t) = (c_base_j + t*d_j) / (S0+t)^2
    # f'(t) = d_j*(S0+t)^2 - (c_base_j + t*d_j)*2*(S0+t) = (S0+t)[d_j*(S0+t) - 2*(c_base_j + t*d_j)]
    # = (S0+t)[d_j*S0 - 2*c_base_j - t*d_j]
    # Setting to 0: t = d_j*S0/d_j - 2*c_base_j/d_j = S0 - 2*c_base_j/d_j
    
    # Candidate t values
    candidates = [0.0, a_i * 0.5, a_i, a_i * 1.5, a_i * 2.0, a_i * 0.1]
    
    # Add critical points from top-k positions
    idx_top = np.argsort(c)[-top_k:]
    for j in idx_top:
        if d[j] > 1e-14:
            t_star = S0 - 2.0 * c_base[j] / d[j]
            if 0 <= t_star <= 5.0:
                candidates.append(t_star)
                candidates.append(max(0, t_star * 0.9))
                candidates.append(t_star * 1.1)
    
    candidates = [max(0.0, min(5.0, float(tc))) for tc in candidates]
    # Deduplicate
    candidates = sorted(set(round(tc, 12) for tc in candidates))
    if not candidates:
        candidates = [a_i]
    
    best_t = a_i
    best_f = np.inf
    for tc in candidates:
        max_c_trial = float(np.max(c_base + tc * d))
        denom = (S0 + tc) ** 2
        if denom < 1e-30:
            continue
        f_val = 2.0 * n * max_c_trial / denom
        if f_val < best_f:
            best_f = f_val
            best_t = tc
    
    return best_t


def _coord_optimize(a, n_passes=5, top_k=100, deadline=None):
    """Cyclic coordinate descent with exact 1D optimization."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(999)
    
    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = 0
        for i in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            new_val = _exact_coord_opt(a, i, c, top_k=top_k)
            if new_val != a[i]:
                a[i] = new_val
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                improved += 1
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
    
    return a, best_score


def _adam_logspace_v2(a_init, deadline, betas, lr, top_k, max_iter, kkt_interval=5000):
    """Improved Adam in log-space with periodic KKT solves."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        while it_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1
            
            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            
            # Adaptive top-K: use threshold-based selection
            max_c = float(c.max())
            threshold = max_c - beta * max_c  # positions within beta*max_c of the max
            active_mask = c >= threshold
            n_active = int(active_mask.sum())
            
            # Use top-k softmax weights on active positions
            if n_active > top_k:
                # Use top_k among active
                topk_thresh = np.partition(c, -top_k)[-top_k]
                final_mask = c >= topk_thresh
            else:
                final_mask = active_mask
            
            if not np.any(final_mask):
                final_mask = c >= max_c - 0.01 * max_c
            if not np.any(final_mask):
                final_mask[int(np.argmax(c))] = True
            
            idx = np.where(final_mask)[0]
            c_sub = c[idx]
            c_max_sub = float(c_sub.max())
            # Use a tighter temperature for the softmax within active set
            temp = beta * max_c * 0.5
            exp_vals = np.exp((c_sub - c_max_sub) / max(temp, 1e-15))
            w_sub = exp_vals / exp_vals.sum()
            
            w = np.zeros(2 * n - 1)
            w[idx] = w_sub
            
            # Gradient
            grad_a = _compute_grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad_z = 2.0 * n * a * (grad_a - mean_grad)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            m_buf = b1 * m_buf + (1 - b1) * grad_z
            v_buf = b2 * v_buf + (1 - b2) * (grad_z * grad_z)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            upd_norm = np.linalg.norm(update)
            if upd_norm > 1.0:
                update = update / upd_norm
            
            z = z - update
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            
            # Periodic KKT solve
            if it_phase % kkt_interval == 0:
                a_curr = _softmax_normalize(z)
                a_kkt, ok = _kkt_solve(a_curr, temperature=beta * max_c)
                if ok:
                    sc = _score_norm(a_kkt)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_kkt.copy()
                        # Update z to match new best
                        z = np.log(np.maximum(a_kkt, 1e-15))
                        z = z - z.mean()
                        m_buf *= 0.1
                        v_buf *= 0.1
            
            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)
    
    a_final = _softmax_normalize(z)
    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score


def _compute_grad_conv(a, w):
    """Gradient of sum_j w_j * c_j with respect to a. Result has same length as a."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_v2(a_init, deadline, betas, top_k_schedule):
    """FISTA with adaptive top-K."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for beta, top_k in zip(betas, top_k_schedule):
        if time.monotonic() > deadline:
            break
        c_curr = _full_convolve(a_curr, a_curr)
        max_c = float(c_curr.max())
        L = 8.0 * n / max(1e-10, max_c)
        step = 1.0 / L
        it = 0
        while it < 100000:
            if time.monotonic() > deadline:
                break
            it += 1
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            c_y = _full_convolve(y, y)
            max_yc = float(c_y.max())
            threshold = max_yc - beta * max_yc
            mask = c_y >= threshold
            if mask.sum() > top_k:
                tk = np.partition(c_y, -top_k)[-top_k]
                mask = c_y >= tk
            if not np.any(mask):
                mask[int(np.argmax(c_y))] = True
            
            idx = np.where(mask)[0]
            c_sub = c_y[idx]
            c_mx = c_sub.max()
            temp = max(beta * max_yc * 0.5, 1e-15)
            ev = np.exp((c_sub - c_mx) / temp)
            w_sub = ev / ev.sum()
            w = np.zeros(2 * n - 1)
            w[idx] = w_sub
            
            grad = _compute_grad_conv(y, w)
            a_next = _proj_simplex(y - step * grad)
            
            if it % 500 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
            
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
    
    sc = _score_norm(a_curr)
    if sc < best_score:
        best_score = sc
        best_a = a_curr.copy()
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()


def _spectral_init(n, rng):
    """Initialize using frequency domain: flat spectrum in mid-frequencies."""
    N = _next_pow2(2 * n)
    freqs = np.fft.rfftfreq(N, d=1.0 / N)
    # Create a spectrum that is somewhat flat in the mid-range
    spec = np.ones(N // 2 + 1)
    # Taper at very low and very high frequencies
    low_mask = freqs < n * 0.01
    high_mask = freqs > n * 0.3
    spec[low_mask] = 0.5
    spec[high_mask] = 0.3
    # Add some random phase
    phases = rng.standard_normal(N // 2 + 1)
    spec = spec * np.exp(1j * phases)
    spec[0] = abs(spec[0])  # Keep DC real
    a = np.fft.irfft(spec * N, N)[:n]
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a
