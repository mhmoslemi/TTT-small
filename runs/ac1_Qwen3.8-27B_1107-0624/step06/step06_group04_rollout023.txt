def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_simplex(y, w):
    n = len(y)
    a_rev = y[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_logspace(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _fista_smoothed(a_init, beta, deadline, max_iter, use_backtrack=False):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    eval_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)

        if use_backtrack:
            step = 1.0
            L_est = 4.0 * n / beta
            step = 1.0 / L_est
            a_next = _proj_simplex(y - step * grad)
        else:
            step = 1.0 / (4.0 * n / beta)
            a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 5:
            sc = _score_norm(a_next)
            eval_count += 1
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score

def _subgradient_polyak(a_init, n_steps, deadline, f_star=None):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)

    if f_star is None:
        f_star = best_score * 0.9  # optimistic lower bound

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        f_curr = n * cm  # since sum(a)=1, score = n * max_c * 2 / 1 = 2n*max_c... 
        # Actually score_norm = 2n * max_c
        f_curr = _score_norm(a)

        # Subgradient at the max: pick the argmax lag
        peak_idx = int(np.argmax(c))
        # Subgradient of max(c) w.r.t. a is the subgradient of c[peak_idx]
        a_rev = a[::-1]
        e_peak = np.zeros(len(c))
        e_peak[peak_idx] = 1.0
        grad = _compute_grad_simplex(a, e_peak)

        # Polyak step size
        gap = f_curr - f_star
        grad_sq = float(np.dot(grad, grad))
        if grad_sq < 1e-30:
            break
        step = gap / grad_sq
        step = max(step, 1e-10)
        step = min(step, 0.5)  # cap the step

        a_new = _proj_simplex(a - step * grad)
        sc = _score_norm(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            f_star = min(f_star, best_score * 0.99)
        elif it % 100 == 0:
            f_star = min(f_star, best_score * 0.995)

    return best_a, best_score


def _peak_reduction(a_init, n_steps, deadline):
    """Directly target the peak lag by moving mass away from contributing elements."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    step_size = 0.001

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        peak_idx = int(np.argmax(c))

        # Find which elements contribute most to c[peak_idx]
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i] (valid indices)
        contributions = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]

        # Find elements with high contribution
        top_contributors = np.argsort(contributions)[::-1][:10]

        # Find elements with low contribution that we can increase
        low_contributors = np.argsort(contributions)[:10]

        # Transfer: decrease top contributors, increase low contributors
        transferred = 0.0
        for i in top_contributors:
            if a[i] > 1e-8:
                amt = min(a[i] * 0.1, step_size / max(1, len(top_contributors)))
                a[i] -= amt
                transferred += amt

        for j in low_contributors:
            a[j] += transferred / max(1, len(low_contributors))

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            step_size = min(step_size * 1.05, 0.01)
        else:
            step_size *= 0.9
            # Revert
            a = best_a.copy()

    return best_a, best_score


def _make_raised_cosine(n):
    t = np.arange(n)
    a = np.sin(np.pi * (t + 1) / (n + 1))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_modulated(n, freq=1, phase=0.0, amp=0.3):
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_anti_symmetric(n, amp=0.3):
    """Create a sequence that is anti-symmetric around the center."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    a = 1.0 - amp * np.sign(t - center) * np.exp(-((t - center) / (n * 0.1))**2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered_uniform(n, taper_frac=0.2):
    """Uniform in the middle, tapered at edges."""
    a = np.ones(n)
    taper_len = int(n * taper_frac / 2)
    for i in range(taper_len):
        frac = (i + 1) / (taper_len + 1)
        a[i] = frac
        a[n - 1 - i] = frac
    return a / a.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

            break
        # Allocate more time to smaller betas
        time_frac = 0.3 + 0.7 * (i / max(1, len(betas_list) - 1))
        phase_timeout = time.monotonic() + max(2.0, remaining * time_frac / (len(betas_list) - i))
        phase_timeout = min(phase_timeout, time.monotonic() + max(5.0, remaining * 0.15))

        a_curr, sc = _fista_smoothed(a_curr, b, phase_timeout, max_iter=100000)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            n = len(best_seq)
            print(f"  FISTA beta={b/max_cc:.8f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: L-BFGS in log-space ===
    print(f"Phase 3: L-BFGS optimization, n={n}")
    phase3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase3_end - 30:
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())

        # Try a few beta values with L-BFGS
        lbfgs_betas = [0.001 * max_cw, 0.0005 * max_cw, 0.0002 * max_cw]
        for b in lbfgs_betas:
            if time.monotonic() > phase3_end - 15:
                break
            remaining = phase3_end - time.monotonic() - 10
            if remaining < 5:
                break
            max_fun = int(remaining * 200)  # rough estimate of evals per second
            a_l, sc_l = _lbfgs_logspace(a_work, b, time.monotonic() + remaining, max_fun=max(100, max_fun))
            sc_l_full = _score(a_l)
            if sc_l_full < best_score:
                best_score = sc_l_full
                best_seq = a_l.copy()
                n = len(best_seq)
                a_work = a_l.copy()
                print(f"  L-BFGS beta={b/max_cw:.6f}: score={sc_l_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    print(f"End Phase 3: best={best_score:.10f}")

    # === Phase 4: Subgradient with Polyak step ===
    print(f"Phase 4: Subgradient polishing, n={n}")
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end - 10:
        f_star = best_score * 0.99  # optimistic lower bound
        a_sg, sc_sg = _subgradient_polyak(a_work, n_steps=20000, deadline=phase4_end, f_star=f_star)
        sc_sg_full = _score(a_sg)
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg.copy()
            n = len(best_seq)
            print(f"  Subgradient: score={sc_sg_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    print(f"End Phase 4: best={best_score:.10f}")

    # === Phase 5: Peak reduction ===
    print(f"Phase 5: Peak reduction, n={n}")
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end - 10:
        a_pr, sc_pr = _peak_reduction(a_work, n_steps=10000, deadline=phase5_end)
        sc_pr_full = _score(a_pr)
        if sc_pr_full < best_score:
            best_score = sc_pr_full
            best_seq = a_pr.copy()
            n = len(best_seq)
            print(f"  Peak reduction: score={sc_pr_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    print(f"End Phase 5: best={best_score:.10f}")

    # === Phase 6: Final FISTA polish with very small beta ===
    print(f"Phase 6: Final FISTA polish, n={n}")
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end - 10:
        c_fin = _full_convolve(a_work, a_work)
        max_cf = float(c_fin.max())
        # Very small beta for near-exact max approximation
        final_betas = [0.00005 * max_cf, 0.00002 * max_cf, 0.00001 * max_cf]
        a_fin = a_work.copy()
        for b in final_betas:
            if time.monotonic() > phase6_end - 5:
                break
            remaining = phase6_end - time.monotonic() - 5
            if remaining < 2:
                break
            a_fin, sc_fin = _fista_smoothed(a_fin, b, time.monotonic() + remaining, max_iter=200000)
            a_fin = a_fin / a_fin.sum()
            sc_fin_full = _score(a_fin)
            if sc_fin_full < best_score:
                best_score = sc_fin_full
                best_seq = a_fin.copy()
                n = len(best_seq)
                print(f"  Final FISTA beta={b/max_cf:.8f}: score={sc_fin_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-restart with best structure ===
    print(f"Phase 7: Multi-restart, n={n}")
    phase7_end = deadline - 30
    if time.monotonic() < phase7_end - 20:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())

        n_restarts = 5
        per_restart = (phase7_end - time.monotonic() - 10) / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            if per_restart < 3:
                break

            # Diverse perturbation
            strategies = [
                lambda: _perturb_log(a_work, 0.005 + 0.002*r, rng),
                lambda: _perturb_log(a_work, 0.01 + 0.005*r, rng),
                lambda: (lambda: (_make_modulated(n, 1, rng.uniform(0, 2*np.pi), 0.1 + 0.05*r), _perturb_log(_, 0.01, rng))[1])(),
            ]
            try:
                p_start = strategies[r % len(strategies)]()
            except Exception:
                p_start = _perturb_log(a_work, 0.01, rng)

            p_start = np.maximum(p_start, 1e-15)
            p_start = p_start / p_start.sum()

            # Quick FISTA
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            beta_r = 0.0005 * max_cs
            t_r_end = time.monotonic() + max(2.0, per_restart * 0.4)
            p_opt, _ = _fista_smoothed(p_start, beta_r, t_r_end, max_iter=50000)
            p_opt = p_opt / p_opt.sum()

            # L-BFGS follow-up
            if time.monotonic() < phase7_end - 15:
                t_lb_end = time.monotonic() + max(2.0, per_restart * 0.5)
                p_opt, _ = _lbfgs_logspace(p_opt, beta_r, t_lb_end, max_fun=2000)
                sc_r = _score(p_opt)
                if sc_r < best_score:
                    best_score = sc_r
                    best_seq = p_opt.copy()
                    n = len(best_seq)
                    print(f"  Restart {r}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"\n=== FINAL RESULTS ===")
    print(f"Best score: {best_score:.10f}")
    print(f"Sequence length: {len(best_seq)}")
    print(f"Total time: {time.monotonic()-t_start:.1f}s")

    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(4096) / 4096
        s = final.sum()

    sc_final = _score(final)
    print(f"Final score (internal): {sc_final:.10f}")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Final score (evaluator): {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
