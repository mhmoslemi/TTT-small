Uses aggressive L-BFGS with log-sum-exp smoothing, Nesterov-accelerated
subgradient with adaptive active sets, coordinate descent polishing,
and multi-length exploration to beat the previous best of 1.505784.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _self_convolve(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_n]
    return r


def _score(a):
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_convolve(a)
    return float(2.0 * len(a) * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _self_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _self_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # gradient of conv_k w.r.t. a_j is 2*a_{k-j} (if valid)
    # sum_k w_k * 2*a_{k-j} = 2 * (w * a_reversed)_j (circular, but we use linear)
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smoothed(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _beta_schedule(a_start, deadline, min_remaining=60.0):
    n = len(a_start)
    c = _self_convolve(a_start)
    max_c = float(c.max())
    if max_c < 1e-30:
        return a_start

    # More aggressive schedule with more betas at low end
    fracs = np.array([0.2, 0.1, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002,
                      0.001, 0.0005, 0.00025, 0.0001, 0.00005, 0.00002,
                      0.00001, 0.000005, 0.000002, 0.000001])
    betas = max_c * fracs

    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 15:
            break

        est_time_per_iter = 0.002 * (n / 2048.0) ** 1.5
        max_iter = min(10000, max(200, int(remaining * 0.8 / (est_time_per_iter * 20))))

        a_new = _lbfgs_smoothed(current_a, beta, n, max_iter)
        sc_new = _score(a_new)

        if sc_new < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc_new

    return current_a


def _subgrad_nesterov(a_init, deadline, tol_frac=1e-4, max_steps=20000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z_prev = z.copy()
    y = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    c_history = 1.0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a_y = _softmax(y)
        c = _self_convolve(a_y)
        c_max = c.max()
        tol = max(tol_frac * c_max, 1e-10)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a_y[::-1]
        wc = _convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a_y * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (i + 1) ** 2)) / 2.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * i ** 2)) / 2.0 if i > 0 else 1.0
        momentum = (t_prev - 1.0) / t_k

        step = lr / (1.0 + i * 0.0001)
        z_new = y - step * grad
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        y = z_new + momentum * (z_new - z_prev)
        y = y - y.mean()
        z_prev = z_new.copy()

        if i % 100 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
                c_history = 1.0
            else:
                c_history += 1.0
                if c_history > 5:
                    lr *= 0.75
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.95

    a_final = _softmax(z_prev)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _subgrad_direct_nesterov(a_init, deadline, tol_frac=1e-5, max_steps=30000):
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    a_prev = a.copy()
    y = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _self_convolve(y)
        c_max = c.max()
        tol = max(tol_frac * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True

        # Compute gradient at y
        a_rev = y[::-1]
        wc = _convolve(active.astype(float), a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(active.astype(float), c) / active.sum())
        w = active.astype(float) / active.sum()
        wc2 = _convolve(w, y_rev) if False else _convolve(w, a_rev)
        h2 = wc2[n - 1:2 * n - 1]
        grad = 2.0 * n * (h2 - W)
        grad = grad - grad.mean()

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (i + 1) ** 2)) / 2.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * i ** 2)) / 2.0 if i > 0 else 1.0
        momentum = (t_prev - 1.0) / t_k

        step = step_size / (1.0 + i * 0.00005)
        a_new = y - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            step_size *= 0.5
            continue
        a_new = a_new / s

        y = a_new + momentum * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        a_prev = a_new.copy()

        if i % 50 == 0:
            sc = _score(a_new)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_new.copy()
                stall = 0
                step_size = min(step_size * 1.01, 0.5)
            else:
                stall += 1
                if stall > 10:
                    step_size *= 0.7
                    if step_size < 1e-10:
                        break

    sc_final = _score(a_prev)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_prev.copy()

    return best_a, best_sc


def _coord_descent_polish(a_init, deadline, max_rounds=500):
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Pick a few random coordinates per round
        n_coords = min(n, 20)
        indices = rng.choice(n, size=n_coords, replace=False)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue

            # Compute the effect vector: how conv changes when a_j changes
            # delta_conv_k = 2 * delta * a_{k-j} (for k-j in [0,n-1])
            # plus delta^2 at k = 2j
            # So new_conv_k = old_conv_k + 2*delta*a_{k-j} + (delta^2 if k==2j)
            # We want to find delta that minimizes max(new_conv)

            c_old = _self_convolve(a)
            c_max_old = c_old.max()

            # Try a few candidate values
            for frac in [0.3, 0.5, 0.7, 1.0]:
                delta = old_val * (frac - 1.0)  # new_val = old_val * frac
                new_val = old_val * frac

                # Quick check: only positions affected are k where k-j in [0,n-1]
                # i.e., k in [j, j+n-1]
                k_start = max(0, j)
                k_end = min(2 * n - 2, j + n - 1)

                # Affected conv values
                affected_k = range(k_start, k_end + 1)
                max_affected = 0.0
                for k in affected_k:
                    offset = k - j
                    if 0 <= offset < n:
                        contrib = 2.0 * delta * a[offset]
                        if k == 2 * j:
                            contrib += delta * delta
                        new_val_k = c_old[k] + contrib
                        if new_val_k > max_affected:
                            max_affected = new_val_k

                # Unaffected max
                unaffected_mask = np.ones(2 * n - 1, dtype=bool)
                unaffected_mask[k_start:k_end + 1] = False
                if unaffected_mask.any():
                    max_unaffected = c_old[unaffected_mask].max()
                else:
                    max_unaffected = 0.0

                new_max = max(max_affected, max_unaffected)
                if new_max < c_max_old - 1e-12:
                    a[j] = new_val
                    a = a / a.sum()
                    improved = True
                    break

            if not improved:
                # Try increasing
                for frac in [1.3, 1.5, 2.0]:
                    new_val = old_val * frac
                    delta = new_val - old_val
                    k_start = max(0, j)
                    k_end = min(2 * n - 2, j + n - 1)
                    max_affected = 0.0
                    for k in range(k_start, k_end + 1):
                        offset = k - j
                        if 0 <= offset < n:
                            contrib = 2.0 * delta * a[offset]
                            if k == 2 * j:
                                contrib += delta * delta
                            new_val_k = c_old[k] + contrib
                            if new_val_k > max_affected:
                                max_affected = new_val_k
                    unaffected_mask = np.ones(2 * n - 1, dtype=bool)
                    unaffected_mask[k_start:k_end + 1] = False
                    max_unaffected = c_old[unaffected_mask].max() if unaffected_mask.any() else 0.0
                    new_max = max(max_affected, max_unaffected)
                    if new_max < c_max_old - 1e-12:
                        a[j] = new_val
                        a = a / a.sum()
                        improved = True
                        break

        if improved:
            sc_now = _score(a)
            if sc_now < best_sc - 1e-13:
                best_sc = sc_now
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_sc


def _perturb_diverse(a, rng, method="auto"):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    if method == "gauss_small":
        z = z + 0.01 * rng.standard_normal(n)
    elif method == "gauss_med":
        z = z + 0.03 * rng.standard_normal(n)
    elif method == "gauss_large":
        z = z + 0.08 * rng.standard_normal(n)
    elif method == "fourier_low":
        t = np.arange(n)
        for _ in range(3):
            freq = int(rng.integers(1, max(2, n // 16)))
            amp = 0.005 * rng.uniform(1.0, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif method == "fourier_high":
        t = np.arange(n)
        for _ in range(5):
            freq = int(rng.integers(n // 16, max(n // 16 + 2, n // 4)))
            amp = 0.003 * rng.uniform(1.0, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif method == "random_warp":
        t = np.linspace(0, 1, n)
        n_knots = int(rng.integers(8, 32))
        knot_t = np.sort(rng.uniform(0, 1, n_knots))
        knot_vals = rng.uniform(-0.02, 0.02, n_knots)
        z = z + np.interp(t, knot_t, knot_vals)
    elif method == "block":
        n_blocks = int(rng.integers(3, 10))
        block_size = n // n_blocks
        for _ in range(n_blocks):
            start = int(rng.integers(0, n - block_size))
            z[start:start + block_size] += rng.uniform(-0.01, 0.01)
    else:
        z = z + 0.02 * rng.standard_normal(n)

    z = z - z.mean()
    z = np.clip(z, -80, 80)
    return _softmax(z)


def _make_initial(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_bump":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS with log-sum-exp smoothing,
    Nesterov-accelerated subgradient, coordinate descent, and multi-length
    exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """

    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):

        if p is None:
            return

        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"t={elapsed:.1f}s")

    parent = None

            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_initial(4096, "tapered")
        try_update(a0, "init")

    if best_seq is None:
        best_seq = _make_initial(2048, "uniform")
        best_score_val = _score(best_seq)
        print(f"Using fallback init, score={best_score_val:.10f}")

    # === Phase 1: Main L-BFGS on parent length ===
    print(f"\n=== Phase 1: L-BFGS main (t={time.monotonic()-t_start:.1f}s) ===")
    a_p1 = _beta_schedule(best_seq, deadline, min_remaining=400.0)
    try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: Nesterov subgradient (z-space) ===
    print(f"\n=== Phase 2: Nesterov z-subgrad (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_sg, sc_sg = _subgrad_nesterov(best_seq, deadline - 250,
                                        tol_frac=1e-4, max_steps=15000, lr_init=0.03)
        try_update(a_sg, "phase2_nesterov_z")

    # === Phase 3: Direct Nesterov subgradient ===
    print(f"\n=== Phase 3: Nesterov direct (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg2, sc_sg2 = _subgrad_direct_nesterov(best_seq, deadline - 150,
                                                  tol_frac=1e-5, max_steps=20000)
        try_update(a_sg2, "phase3_nesterov_direct")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coord descent (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_cd, sc_cd = _coord_descent_polish(best_seq, deadline - 100, max_rounds=300)
        try_update(a_cd, "phase4_coord")

    # === Phase 5: Final L-BFGS at very small beta ===
    print(f"\n=== Phase 5: Final L-BFGS fine (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n = len(best_seq)
        c_b = _self_convolve(best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 45:
                break
            remaining = deadline - 45 - time.monotonic()
            if remaining < 10:

            max_iter = min(8000, max(500, int(remaining * 0.9 / 0.003)))
            a_new = _lbfgs_smoothed(best_seq, beta, n, max_iter)

    # === Phase 6: Multi-restart ===
    print(f"\n=== Phase 6: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        restart_methods = [
            "gauss_small", "gauss_med", "fourier_low",
            "fourier_high", "random_warp", "block"

        for mname in restart_methods:

            print(f"  Restart [{mname}]: start={sc_start:.10f}")

            n = len(p_start)

            a_r = p_start

                remaining = deadline - 55 - time.monotonic()
                if remaining < 10:
                    break
                max_iter = min(5000, max(200, int(remaining * 0.7 / 0.003)))
                a_r = _lbfgs_smoothed(a_r, beta, n, max_iter)
            try_update(a_r, f"restart_{mname}")

            # Quick Nesterov
            if time.monotonic() < deadline - 50:
                a_r2, _ = _subgrad_nesterov(a_r, deadline - 50,
                                             tol_frac=1e-4, max_steps=3000, lr_init=0.02)
                try_update(a_r2, f"restart_{mname}_ng")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Different lengths (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 50:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)

            # Optimize at this length
            a_opt = _beta_schedule(a_interp, deadline, min_remaining=30.0)
            try_update(a_opt, f"n_{target_n}")

            if time.monotonic() < deadline - 35:
                a_opt2, _ = _subgrad_nesterov(a_opt, deadline - 35,
                                               tol_frac=1e-4, max_steps=5000, lr_init=0.02)
                try_update(a_opt2, f"n_{target_n}_ng")

    # === Phase 8: Final aggressive cycle ===
    print(f"\n=== Phase 8: Final cycle (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None:
        for rnd in range(6):
            if time.monotonic() > deadline - 3:
                break

            # Very tight L-BFGS
            n = len(best_seq)
            c_b = _self_convolve(best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-7
            a_f = _lbfgs_smoothed(best_seq, beta, n, 3000)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight Nesterov
            if time.monotonic() < deadline - 3:
                a_s, _ = _subgrad_nesterov(best_seq, deadline - 3,
                                           tol_frac=5e-5, max_steps=5000, lr_init=0.01)
                try_update(a_s, f"final_ng_{rnd}")

            # Coord descent
            if time.monotonic() < deadline - 3:
                a_cd, _ = _coord_descent_polish(best_seq, deadline - 3, max_rounds=100)
                try_update(a_cd, f"final_cd_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initial(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_initial(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
