**Algorithm: Subgradient Descent with Exact Line Search and Active Set Tracking**

For a fixed active set `A` (positions where convolution is at max), the objective is:
`f(a) = ∑_{k∈A} w_k * (a*a)_k`

This is a quadratic function of `a`, and its gradient is linear in `a`. On the simplex, we can do:
- Compute gradient `g(a)`
- Find the direction of steepest descent on the simplex
- Do an exact line search in that direction

The steepest descent direction on the simplex is: `d_i = -g_i + mean(g_active)` where `active` are the indices with `a_i > 0`.

Then the line search: minimize `f(a + t*d)` for `t ∈ [0, 1]` (staying on the simplex).

Since `f` is quadratic (for fixed active set), this is a 1D quadratic minimization, which has a closed-form solution!

This is essentially **cyclic coordinate descent** generalized to the "face" of the simplex, which should converge very fast.

```python
def _face_descent_step(a, active_set_weights, beta):
    """One step of face descent: move along steepest descent direction on current face."""
    n = len(a)
    # Compute gradient
    c = _full_convolve(a, a)
    # Use soft max for active set
    c_max = c.max()
    w = np.exp((c - c_max) / beta)
    w = w / w.sum()
    
    # Gradient of sum_k w_k * (a*a)_k
    # = 2 * (w * a_reversed) [aligned properly]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n-1:2*n-1]
    
    # Steepest descent direction on the face
    support = np.where(a > 1e-15)[0]
    if len(support) == 0:
        return a
    grad_s = grad[support]
    d_full = np.zeros(n)
    d_full[support] = -(grad_s - grad_s.mean())
    
    # Line search: minimize f(a + t*d) for t in [0, t_max]
    # t_max is limited by non-negativity
    t_max = np.inf
    for i in support:
        if d_full[i] < -1e-15:
            t_max = min(t_max, a[i] / (-d_full[i]))
    t_max = min(t_max, 1.0)
    
    # Since f is quadratic in t (for fixed w), solve analytically
    # f(a + t*d) = sum_k w_k * ((a+td)*(a+td))_k
    # = sum_k w_k * [(a*a)_k + 2t * sum_j a_j * d_{k-j} + t^2 * (d*d)_k]
    # = f(a) + 2t * <grad/2, d> + t^2 * sum_k w_k * (d*d)_k
    
    # Actually, let me just do a golden section search for simplicity
    # Or use the quadratic formula
    
    # Coefficient of t^2: sum_k w_k * (d*d)_k
    dd = _full_convolve(d_full, d_full)
    quad_coeff = float(np.dot(w, dd))
    
    # Coefficient of t: 2 * sum_k w_k * sum_j a_j * d_{k-j} = 2 * <grad_half, d>
    # where grad_half = sum_k w_k * (a * e_{k-·}) 
    # Actually, linear_coeff = sum_k w_k * 2 * sum_j a_j * d_{k-j}
    ad = _full_convolve(a, d_full)
    lin_coeff = 2.0 * float(np.dot(w, ad))
    
    if quad_coeff > 1e-15:
        t_opt = -lin_coeff / (2.0 * quad_coeff)
    else:
        t_opt = t_max
    
    t_opt = np.clip(t_opt, 0.0, t_max)
    a_new = a + t_opt * d_full
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    
    return a_new
```

```python
def _block_qp_descent(a_init, deadline, n_inner=50, beta_init=0.05):
    """
    Block coordinate descent:
    - Outer loop: update weights w based on current convolution
    - Inner loop: minimize sum_k w_k * (a*a)_k s.t. a in simplex
    """
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    beta = beta_init
    
    while time.monotonic() < deadline:
        # Update weights
        c = _full_convolve(a, a)
        c_max = float(c.max())
        w = np.exp((c - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Inner loop: minimize sum_k w_k * (a*a)_k on simplex
        for _ in range(n_inner):
            if time.monotonic() > deadline:
                break
            # Gradient of the QP objective
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            grad = 2.0 * wc[n-1:2*n-1]
            
            # Face descent with exact line search
            support = np.where(a > 1e-12)[0]
            if len(support) < 2:
                break
            grad_s = grad[support]
            mean_g = grad_s.mean()
            d = np.zeros(n)
            d[support] = -(grad_s - mean_g)
            
            # Check t_max
            t_max = 1.0
            neg_mask = d[support] < -1e-15
            if np.any(neg_mask):
                t_candidates = a[support][neg_mask] / (-d[support][neg_mask])
                t_max = min(t_max, float(t_candidates.min()))
            
            # Quadratic coefficients
            dd = _full_convolve(d, d)
            quad_coeff = float(np.dot(w, dd))
            ad = _full_convolve(a, d)
            lin_coeff = 2.0 * float(np.dot(w, ad))
            
            if quad_coeff > 1e-15:
                t_opt = -lin_coeff / (2.0 * quad_coeff)
            else:
                t_opt = t_max if lin_coeff < 0 else 0.0
            
            t_opt = np.clip(t_opt, 0.0, t_max)
            a = a + t_opt * d
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            a = a / a_sum
        
        # Check improvement
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Decrease beta
        beta *= 0.5
        if beta < 1e-10:
            beta = 1e-10
    
    return best_a, best_score
```

Wait, I realize there's an issue. The "face descent" with exact line search minimizes the QP `∑ w_k (a*a)_k` on the current face. But the QP might not be convex (as we discussed earlier, the Hessian might not be PSD). In that case, the "exact line search" might find a local minimum of the quadratic along the line, which could be a maximum.

Let me check: `quad_coeff = w^T (d*d) = ∑_k w_k * (d*d)_k`. Since `w_k >= 0` and `(d*d)_k >= 0` (it's a sum of products of `d` values, but wait, `d` can be negative).

Actually, `(d*d)_k = ∑_j d_j * d_{k-j}`. This can be negative if `d` has mixed signs. So `quad_coeff` can be negative, meaning the QP is not convex along this direction.

In that case, the "minimum" of the quadratic along the line would be at `t = ±∞` (parabola opening downward). We should then use the endpoint `t_max`.

So the line search should be:
- If `quad_coeff > 0`: parabola opens up, minimum at `t = -lin/(2*quad)`, clip to [0, t_max]
- If `quad_coeff <= 0`: parabola opens down or is linear, minimum at one of the endpoints {0, t_max}

Now, let me think about another important aspect: **the non-smoothness at the max**.

When the max of the convolution is achieved at multiple positions (which is common for well-optimized sequences), the subgradient is the convex hull of the gradients at those positions. The softmax smoothing handles this, but it introduces a bias.

An alternative is to use the **exact subgradient** (uniform distribution over active positions) and do a "subgradient projection" method. This is what the mirror descent in the previous code does.

Let me combine ideas:

**Final Algorithm:**

1. **Phase A: FISTA with adaptive beta** (300s)
   - Start from parent
   - Use FISTA with softmax-smoothed max
   - Adaptively decrease beta from large to small

2. **Phase B: Block QP descent** (400s)
   - Alternate between updating weights and doing face descent
   - Decrease beta geometrically
   - This should converge to a local minimum of the smoothed problem

3. **Phase C: Exact subgradient with restart** (300s)
   - Use exact active set (all positions within epsilon of max)
   - Do subgradient descent with Nesterov acceleration
   - Periodically restart from best found

4. **Phase D: L-BFGS on log-parametrization** (400s)
   - Use scipy's L-BFGS with the smoothed objective
   - Very small beta for near-exact optimization

5. **Phase E: Coordinate descent polishing** (200s)
   - For each coordinate i, compute the optimal value given all other coordinates fixed
   - This is a 1D problem: minimize max_k (a*a)_k where a_i is the variable
   - Can be solved by bisection

Actually, for Phase E, fixing all coordinates except `a_i`, the convolution becomes:
`(a*a)_k = a_i * a_{k-i} + a_{k-i} * a_i + (a*a)_k - a_i*a_{k-i} - a_{k-i}*a_i`

Hmm, this is getting complicated. Let me think of a simpler way.

If we fix all `a_j` for `j ≠ i`, let `S = sum_{j≠i} a_j = 1 - a_i` (on the simplex). Then:
`(a*a)_k = a_i * a_{k-i} + a_{k-i} * a_i + (a_{-i} * a_{-i})_k`

where `a_{-i}` is the sequence with the i-th element removed (and sum = S).

Wait, more carefully:
`(a*a)_k = ∑_j a_j * a_{k-j}`

If we split the sum into `j=i` and `j≠i`:
`(a*a)_k = a_i * a_{k-i} + ∑_{j≠i} a_j * a_{k-j}`

The second term still depends on `a_i` when `k-j = i`, i.e., `j = k-i`. So:
`(a*a)_k = a_i * a_{k-i} + a_{k-i} * a_i + ∑_{j≠i, j≠k-i} a_j * a_{k-j}`

Wait, that's double-counting. Let me be more careful:

`(a*a)_k = ∑_j a_j * a_{k-j}`

Case 1: `j = i`: term is `a_i * a_{k-i}`
Case 2: `k-j = i`, i.e., `j = k-i`: term is `a_{k-i} * a_i`
Case 3: `j ≠ i` and `k-j ≠ i`: term is `a_j * a_{k-j}` (doesn't involve `a_i`)

If `i = k-i` (i.e., `k = 2i`), then cases 1 and 2 give the same term `a_i^2`.

So:
- If `2i ≠ k`: `(a*a)_k = 2*a_i*a_{k-i} + (a_{-i}*a_{-i})_k` where `a_{-i}` has `a_i` set to 0
- If `2i = k`: `(a*a)_k = a_i^2 + (a_{-i}*a_{-i})_k`

In either case, `(a*a)_k` is a quadratic (or linear) function of `a_i`.

The objective `max_k (a*a)_k` is the max of these quadratics, which is a convex function of `a_i` (max of convex functions is convex, and quadratics with positive leading coefficient are convex).

Wait, is `2*a_i*a_{k-i}` convex in `a_i`? It's linear in `a_i` (if `a_{k-i} > 0`), so it's both convex and concave. And `a_i^2` is convex. So yes, each `(a*a)_k` is convex in `a_i` (it's either linear or quadratic with positive leading coefficient).

Therefore, `max_k (a*a)_k` is convex in `a_i`, and we can find the minimum by bisection or ternary search!

This is a great coordinate descent approach. At each coordinate:
1. Compute the function `f(a_i) = max_k (a*a)_k` as a function of `a_i` (with all other coordinates fixed)
2. Find the minimizer by ternary search (since it's convex)
3. Update `a_i`

The catch: after updating `a_i`, the sum changes, so we need to renormalize. But if we're on the simplex, reducing `a_i` means the sum is < 1, and we need to add the difference somewhere. 

Actually, let's not normalize. Instead, work with the unnormalized objective:
`F(a) = 2n * max(conv(a,a)) / (sum(a))^2`

This is scale-invariant, so we can fix `sum(a) = 1` WLOG.

For coordinate descent on the simplex: when we change `a_i` from `v` to `v'`, we need to redistribute the difference `(v - v')` among the other coordinates. The simplest is to distribute proportionally:
`a_j' = a_j * (1 - v') / (1 - v)` for `j ≠ i`.

But this changes all other coordinates, which is expensive.

Alternative: Work with the homogeneous objective and use a different parametrization.

Actually, since the objective is scale-invariant, we can just work with any scaling. Let's fix `sum(a) = 1` and do coordinate descent where we:
1. Tentatively set `a_i = v'`
2. Scale all other coordinates to make the sum 1: `a_j *= (1-v')/(1-v)` for `j≠i`
3. Compute the new objective
4. If better, keep the change; otherwise, revert

The scaling step changes all coordinates, so the "coordinate descent" is really a "block descent" where we change one coordinate and proportionally scale the rest.

This is equivalent to working in the log-parametrization and doing a coordinate update there.

Let me think about the computational cost:
- Computing `conv(a,a)` takes O(n log n)
- For coordinate descent, at each step we need to evaluate the objective at a few values of `a_i` (say 10 for ternary search)
- Each evaluation requires computing the convolution, which is O(n log n)
- So each coordinate step is O(n log n * 10) = O(n log n)
- A full pass over all n coordinates is O(n^2 log n)

For n=4096, that's about 4096^2 * 12 ≈ 200M operations per pass. In numpy with FFT, this might take a few seconds per pass. So we could do maybe 100-200 passes in 400 seconds. That might be enough for convergence.

Actually, we can be smarter. Instead of computing the full convolution each time, we can use the fact that only O(1) terms change when we modify one coordinate.

When we change `a_i` from `v` to `v'` (and scale others proportionally), the change in `(a*a)_k` is:
- For the `a_i` terms: `a_i * a_{k-i}` changes
- For the scaling: all terms scale by `(1-v')/(1-v)`

This is still complicated. Let me just use the full convolution for simplicity.

Actually, for the ternary search, I don't need to compute the full convolution each time. I can precompute `(a_{-i} * a_{-i})` (the convolution with `a_i` set to 0), and then for each trial value of `a_i`, the convolution is:
`(a*a)_k = 2*a_{trial}*a_{k-i} + (a_{-i}*a_{-i})_k` (for `k ≠ 2i`)
`(a*a)_k = a_{trial}^2 + (a_{-i}*a_{-i})_k` (for `k = 2i`)

Wait, but I also need to account for the proportional scaling of the other elements. If I set `a_i = v'` and scale others by factor `s = (1-v')/(1-v)`, then:
`(a*a)_k = (s * a_{-i} + [v' at position i]) * (s * a_{-i} + [v' at position i])`
`= s^2 * (a_{-i}*a_{-i})_k + s*(a_{-i}*a_{-i})_k cross terms...`

This is getting messy. Let me just do the full convolution each time. For n=4096, one FFT convolution is very fast (maybe 0.1ms), so 10 evaluations per coordinate is 1ms, and a full pass is 4 seconds. In 400 seconds, I can do 100 passes. That should be enough.

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA, block QP descent, coordinate descent, and L-BFGS.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _fista_opt(a_init, beta, deadline, max_iter):
    """FISTA with softmax-smoothed max-convolution on simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 8.0 * n  # Lipschitz constant estimate
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(len(c_y))
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n-1:2*n-1]
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 20 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _face_descent(a_init, deadline, n_inner=30, beta_init=0.02):
    """Block coordinate descent: alternate weight update and face descent."""
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    beta = beta_init
    patience = 0
    
    while time.monotonic() < deadline:
        # Update weights
        c = _full_convolve(a, a)
        c_max = float(c.max())
        w = np.exp((c - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Inner loop: face descent on the QP
        for it in range(n_inner):
            if time.monotonic() > deadline:
                break
            # Gradient
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            grad = 2.0 * wc[n-1:2*n-1]
            
            # Steepest descent direction on current face
            support = np.where(a > 1e-14)[0]
            if len(support) < 2:
                break
            grad_s = grad[support]
            mean_g = grad_s.mean()
            d = np.zeros(n)
            d[support] = -(grad_s - mean_g)
            
            # Line search bounds
            t_max = 1.0
            neg_idx = support[d[support] < -1e-15]
            if len(neg_idx) > 0:
                t_cand = a[neg_idx] / (-d[neg_idx])
                t_max = min(t_max, float(t_cand.min()))
            
            if t_max < 1e-15:
                break
            
            # Quadratic coefficients for exact line search
            dd = _full_convolve(d, d)
            quad_coeff = float(np.dot(w, dd))
            ad = _full_convolve(a, d)
            lin_coeff = 2.0 * float(np.dot(w, ad))
            
            if quad_coeff > 1e-12:
                t_opt = -lin_coeff / (2.0 * quad_coeff)
            elif lin_coeff < 0:
                t_opt = t_max
            else:
                t_opt = 0.0
            
            t_opt = np.clip(t_opt, 0.0, t_max)
            
            if t_opt < 1e-15:
                break
            
            a = a + t_opt * d
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        
        # Check improvement
        sc = _score_norm(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Decrease beta
        beta *= 0.6
        if beta < 1e-12:
            beta = 1e-12
        
        # Early stop if stuck
        if patience > 20:
            break
    
    return best_a, best_score

def _coord_descent(a_init, deadline, max_rounds=200):
    """Coordinate descent with proportional scaling."""
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Randomize order
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            v = a[i]
            if v < 1e-10:
                # Try a small positive value
                trials = [1e-4, 5e-4, 2e-4]
            else:
                # Ternary search around current value
                lo = max(0.0, v * 0.3)
                hi = min(0.3, v * 3.0 + 1e-4)
                # Actually, let's use a wider range for exploration
                lo = 0.0
                hi = min(0.5, v + 0.1)
                trials = np.linspace(lo, hi, 7)
            
            best_trial = v
            best_trial_score = _score_norm(a)
            
            for tv in trials:
                if tv < 0:
                    continue
                if abs(tv - v) < 1e-15:
                    continue
                a_new = a.copy()
                a_new[i] = tv
                s_other = 1.0 - tv
                s_old = 1.0 - v
                if s_old < 1e-12:
                    continue
                scale = s_other / s_old
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_new[mask] *= scale
                a_new = np.maximum(a_new, 0.0)
                sc = _score_norm(a_new)
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = tv
                    best_trial_a = a_new.copy()
            
            if best_trial != v:
                a = best_trial_a
                improved = True
        
        sc = _score_norm(a)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    """Adam optimization in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 4.0 * h  # simplified gradient for softmax parametrization
        
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        
        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0001, current_lr * 0.9)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lbfgs_opt(a_init, deadline, beta):
    """L-BFGS-B refinement on log-parametrization."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()
    
    def objective(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))
    
    def gradient(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad_a = 2.0 * h
        # Transform to log-space: grad_z = a * (grad_a - dot(a, grad_a))
        return a * (grad_a - float(np.dot(a, grad_a)))
    
    remaining = deadline - time.monotonic() - 3.0
    if remaining <= 1.0:
        return a_init, best_score
    
    try:
        result = minimize(
            objective, z0, jac=gradient, method='L-BFGS-B',
            options={'maxiter': max(100, int(remaining * 50)), 'gtol': 1e-10, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    a_best = _softmax_compute(best_z)
    return a_best, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: FISTA + face descent + coord descent + Adam + L-BFGS."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
```

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines FISTA, face descent with exact line search, coordinate descent, Adam, and L-BFGS.
Key innovation: face descent exploiting the quadratic structure for exact 1D minimization,
plus multi-restart with diverse initial structures to escape local minima.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    # Better Lipschitz estimate: the Hessian of the smoothed max has eigenvalues bounded by ~4n*max(c)
    c_init = _full_convolve(a_curr, a_curr)
    L = 8.0 * n * max(1.0, float(c_init.max()))
    step = 1.0 / max(L, 1.0)
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(len(c_y))
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        a_next = _proj_simplex(y - step * grad)

            sc = _score_norm(a_next)
            if sc < best_score:

                best_a = a_next.copy()

        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:

        best_a = a_curr.copy()
    return best_a, best_score


def _face_descent(a_init, deadline, n_inner=25, beta_init=None):
    """Block coordinate descent: alternate weight update and exact face descent."""
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)

    if beta_init is None:
        c = _full_convolve(a, a)
        beta = 0.01 * float(c.max())
    else:
        beta = beta_init

    outer_iter = 0
    while time.monotonic() < deadline:
        outer_iter += 1
        # Update weights
        c = _full_convolve(a, a)
        c_max = float(c.max())
        w = np.exp((c - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        # Inner loop: face descent on the QP with fixed weights
        prev_qp_val = None
        for it in range(n_inner):
            if time.monotonic() > deadline:
                break

            # Gradient of sum_k w_k * (a*a)_k
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            grad = 2.0 * wc[n - 1:2 * n - 1]

            # Steepest descent direction on current face
            support = np.where(a > 1e-14)[0]
            if len(support) < 2:
                break
            grad_s = grad[support]
            mean_g = grad_s.mean()
            d = np.zeros(n)
            d[support] = -(grad_s - mean_g)

            # Line search bounds
            t_max = 1.0
            neg_mask = d[support] < -1e-14
            if np.any(neg_mask):
                neg_idx = support[neg_mask]
                t_cand = a[neg_idx] / (-d[neg_idx])
                t_max = min(t_max, float(t_cand.min()))

            if t_max < 1e-14:
                break

            # Exact quadratic line search
            dd = _full_convolve(d, d)
            quad_coeff = float(np.dot(w, dd))
            ad = _full_convolve(a, d)
            lin_coeff = 2.0 * float(np.dot(w, ad))

            if quad_coeff > 1e-14:
                t_opt = -lin_coeff / (2.0 * quad_coeff)
            elif lin_coeff < -1e-14:
                t_opt = t_max
            else:
                t_opt = 0.0

            t_opt = float(np.clip(t_opt, 0.0, t_max))

            if t_opt < 1e-14:
                break

            a = a + t_opt * d
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-15:
                break
            a = a / a_sum

            # Check inner convergence
            qp_val = float(np.dot(w, _full_convolve(a, a)))
            if prev_qp_val is not None and abs(qp_val - prev_qp_val) < 1e-14 * max(1.0, abs(prev_qp_val)):
                break
            prev_qp_val = qp_val

        # Check improvement
        sc = _score_norm(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()

        # Decrease beta adaptively
        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta_new = 0.5 * beta
        if beta_new < 1e-12 * c_max:
            beta_new = 1e-12 * c_max
        beta = beta_new

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=300, n_trials=7):
    """Coordinate descent with proportional rescaling of other coordinates."""
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(99999)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        round_improved = 0

        for i in order:
            if time.monotonic() > deadline:
                break

            v = a[i]
            s_other = 1.0 - v
            if s_other < 1e-10:
                continue

            # Generate trial values
            if v < 1e-6:
                trials = np.array([1e-5, 5e-5, 1e-4, 5e-4])
            else:
                # Search in a range around current value
                lo = max(0.0, v * 0.2)
                hi = min(0.4, v * 5.0 + 0.05)
                trials = np.linspace(lo, hi, n_trials)
                # Also try the current value as baseline
                if v < lo or v > hi:
                    trials = np.concatenate([trials, [v]])

            best_trial_score = _score_norm(a)
            best_trial_a = None

            for tv in trials:
                if tv < 0 or tv > 0.5:
                    continue
                if abs(tv - v) < 1e-8:
                    continue
                a_new = a.copy()
                a_new[i] = tv
                s_other_new = 1.0 - tv
                scale = s_other_new / s_other
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_new[mask] *= scale
                a_new = np.maximum(a_new, 0.0)
                sc = _score_norm(a_new)
                if sc < best_trial_score - 1e-12:
                    best_trial_score = sc
                    best_trial_a = a_new.copy()

            if best_trial_a is not None:
                a = best_trial_a
                round_improved += 1

        sc = _score_norm(a)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()

        # Early stop if no improvement in a round
        if round_improved == 0:
            # Try a few more rounds with different random order
            if rnd > 20 and rnd % 10 == 0:
                break

    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 40 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 4000 == 0:
            current_lr = max(0.0002, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def objective(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def gradient(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        return a * (grad_a - float(np.dot(a, grad_a)))

    remaining = deadline - time.monotonic() - 3.0
    if remaining <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            objective, z0, jac=gradient, method='L-BFGS-B',
            options={'maxiter': max(100, int(remaining * 40)), 'gtol': 1e-12, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "beta_dist":
        # Beta distribution with alpha=beta=2 (concave)
        t = np.linspace(0, 1, n)
        a = t * (1 - t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "piecewise":
        # Piecewise constant with smooth transitions
        t = np.linspace(0, 1, n)
        a = np.zeros(n)
        n_regions = 4
        for r in range(n_regions):
            start = r / n_regions
            end = (r + 1) / n_regions
            mask = (t >= start) & (t < end)
            height = 1.0 + 0.3 * np.sin(r * np.pi / 2)
            a[mask] = height
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _fista_adaptive(a_init, deadline, max_iter=100000):
    """FISTA with adaptive beta: start large, decrease when stalled."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    c = _full_convolve(a_curr, a_curr)
    max_c = float(c.max())
    beta = 0.1 * max_c
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    stall_count = 0
    prev_score = best_score

    while time.monotonic() < deadline and beta > 1e-12 * max_c:
        # Run FISTA for a batch
        batch_deadline = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) / 5.0))
        a_curr, _ = _fista_opt(a_curr, beta, batch_deadline, max_iter)
        a_curr = a_curr / a_curr.sum()
        sc = _score_norm(a_curr)

        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_curr.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Decrease beta
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())
        beta *= 0.4
        if beta < 1e-12 * max_c:
            beta = 1e-12 * max_c

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining FISTA, face descent, coordinate descent,
    Adam, and L-BFGS. Uses adaptive beta scheduling and multi-restart exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = a0.copy()

    # === Phase 1: Adaptive FISTA ===
    print("Phase 1: Adaptive FISTA")
    phase1_end = min(deadline, t_start + 300.0)
    a_work, _ = _fista_adaptive(a_work, phase1_end, max_iter=100000)
    a_work = a_work / a_work.sum()
    sc = _score(a_work)
    if sc < best_score:
        best_score = sc
        best_seq = a_work.copy()
        print(f"  Phase 1 done: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Face descent ===
    print("Phase 2: Face descent (exact line search)")
    phase2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        a_work, _ = _face_descent(a_work, phase2_end, n_inner=30, beta_init=0.005 * max_cc)
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            print(f"  Face descent: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: Coordinate descent ===
    print("Phase 3: Coordinate descent")
    phase3_end = min(deadline, t_start + 950.0)
    if time.monotonic() < phase3_end - 20:
        a_work, _ = _coord_descent(a_work, phase3_end, max_rounds=1000, n_trials=5)
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            print(f"  Coord descent: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 3: best={best_score:.10f}")

    # === Phase 4: Adam fine optimization ===
    print("Phase 4: Adam multi-beta")
    phase4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase4_end - 20:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.002 * max_cc, 0.0005 * max_cc, 0.0001 * max_cc, 0.00003 * max_cc]
        per_beta = (phase4_end - time.monotonic() - 10) / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase4_end - 10:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + max(5.0, per_beta),
                                  beta, lr_init=0.004, max_iter=500000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 4: best={best_score:.10f}")

    # === Phase 5: L-BFGS final polish ===
    print("Phase 5: L-BFGS polish")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        b_f = max(1e-10, 0.00002 * max_cp)
        a_pol, _ = _lbfgs_opt(a_pol, deadline - 15.0, b_f)
        sc_f = _score(a_pol)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_pol.copy()
            print(f"  L-BFGS: score={sc_f:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-restart with diverse structures ===
    remaining_time = deadline - time.monotonic() - 10
    if remaining_time > 80 and best_seq is not None:
        print(f"Phase 6: Multi-restart (remaining={remaining_time:.0f}s)")
        n_restarts = 4
        per_r = remaining_time / n_restarts
        structures = ["perturb_small", "perturb_large", "raised_cosine", "beta_dist"]

        for i in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            st = structures[i % len(structures)]
            if st == "perturb_small":
                p = _perturb(best_seq, 0.015, rng)
            elif st == "perturb_large":
                p = _perturb(best_seq, 0.04, rng)
            elif st == "raised_cosine":
                p = _make_profile(n_main, "raised_cosine")
                p = 0.7 * best_seq + 0.3 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif st == "beta_dist":
                p = _make_profile(n_main, "beta_dist")
                p = 0.6 * best_seq + 0.4 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            else:
                p = _perturb(best_seq, 0.02, rng)

            # Optimize: FISTA -> face descent -> Adam
            t_sub = time.monotonic() + per_r
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())

            p, _ = _fista_adaptive(p, t_sub * 0.4, max_iter=50000)
            p, _ = _face_descent(p, t_sub * 0.7, n_inner=20, beta_init=0.003 * max_cp)
            if time.monotonic() < t_sub * 0.95:
                p, _ = _adam_opt(p, t_sub * 0.95, 0.0001 * max_cp, 0.003, 100000)

            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                print(f"  Restart {i} ({st}): score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Ultra-fine final polish ===
    if best_seq is not None and time.monotonic() < deadline - 8:
        a_fin = best_seq.copy()
        a_fin = a_fin / a_fin.sum()
        c_fin = _full_convolve(a_fin, a_fin)
        max_cf = float(c_fin.max())
        # Very tight face descent
        a_fin, _ = _face_descent(a_fin, deadline - 5.0, n_inner=50, beta_init=0.0001 * max_cf)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"  Final polish: score={sc_fin:.10f}")

    # Final output
    print(f"\nDone: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
