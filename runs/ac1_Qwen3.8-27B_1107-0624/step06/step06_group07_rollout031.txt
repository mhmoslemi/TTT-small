"""
LP-alternating + L-BFGS hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key new idea: implement the state-of-art LP subproblem (max sum(b) s.t. conv(f,b)<=M, b>=0)
using scipy HiGHS with the structured sparse convolution matrix, combined with
adaptive-beta L-BFGS log-sum-exp smoothing and multi-n exploration.

import numpy as np
import time
import scipy.sparse
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 40, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _build_convolve_matrix_sparse(f, n):
    """Build sparse (2n-1) x n convolution matrix T where T[k,j]=f[k-j] if 0<=k-j<n."""
    m = 2 * n - 1
    # Diagonal d (0-indexed sub-diagonal) has value f[d] at n positions
    diags_data = [f[d] * np.ones(n) for d in range(n)]
    offsets = list(range(n))
    T = scipy.sparse.diags(diags_data, offsets, shape=(m, n), format='csc')
    return T


def _lp_alternate_step(f, n, max_lp_time=40.0):
    """
    One LP-based alternating step:
    1. Compute M = max(f*f)
    2. Solve LP: max sum(b) s.t. T_f * b <= M, b >= 0
    3. Return rescaled b (sum=1) and the LP optimal sum
    """
    c = _full_convolve(f, f)
    M = float(c.max())
    if M < 1e-30:
        return f.copy(), 0.0

    # Build sparse constraint matrix
    T = _build_convolve_matrix_sparse(f, n)

    # LP: maximize 1^T b  s.t.  T*b <= M*1,  b >= 0
    # In linprog: minimize -1^T b
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)

    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=T, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': max_lp_time,
                               'presolve': True})
    except Exception as e:
        print(f"  LP error: {e}")
        return f.copy(), 0.0

    elapsed = time.monotonic() - t0

    if res.status not in (0, 3):  # 0=optimal, 3=time limit with feasible solution
        if res.status != 3:
            print(f"  LP status {res.status}: {res.message} ({elapsed:.1f}s)")
            return f.copy(), 0.0

    b = np.maximum(res.x, 0.0)
    s_b = float(b.sum())
    if s_b < 1e-15:
        return f.copy(), 0.0

    b_rescaled = b / s_b
    return b_rescaled, s_b

        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + i * 0.0002)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 4:
                    lr *= 0.7
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.94

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 6)))
        amp = 0.003 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid LP-alternating + L-BFGS optimizer.
    Uses the structured LP subproblem from the state-of-art, combined with
    log-sum-exp smoothed L-BFGS, subgradient polishing, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(4096) / 4096.0
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    if best_seq is None:
        fallback = np.ones(256) / 256.0
        return [float(x) for x in fallback]

    n_best = len(best_seq)

    # === Phase 1: LP Alternating Minimization (NEW) ===
    print(f"\n=== Phase 1: LP Alternating Minimization (t={time.monotonic() - t_start:.1f}s) ===")
    lp_deadline = min(deadline - 600.0, t_start + 400.0)
    a_lp, sc_lp = _lp_alternating_minimization(best_seq, lp_deadline, max_iters=80)
    try_update(a_lp, "lp_alt")

    # === Phase 2: L-BFGS with beta continuation ===
    print(f"\n=== Phase 2: L-BFGS Beta Continuation (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 500:
        n = len(best_seq)
        c_cur = _full_convolve(best_seq, best_seq)
        max_c = float(c_cur.max())
        betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
                                  0.001, 0.0005, 0.0002, 0.0001])
        current_a = best_seq.copy()
        current_sc = _score(current_a)

        for i, beta in enumerate(betas):
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - 200 - time.monotonic()
            if remaining < 20:
                break
            est_time = 0.002 * (n / 2048.0) ** 1.3
            max_iter = min(6000, max(200, int(remaining / (est_time * 12))))
            a_new, _, _ = _lbfgs_run(current_a, beta, n, max_iter)
            sc = _score(a_new)
            if sc < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc
        try_update(current_a, "lbfgs_beta")

    # === Phase 3: Subgradient polish (z-space) ===
    print(f"\n=== Phase 3: Subgradient Polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg, sc_sg = _subgrad_z_polish(best_seq, deadline - 300,
                                         tol_frac=1e-4, max_steps=8000, lr_init=0.025)
        try_update(a_sg, "sg_z")

    # === Phase 4: Another LP round (with updated best) ===
    print(f"\n=== Phase 4: LP Round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        lp_deadline2 = min(deadline - 150.0, time.monotonic() + 120.0)
        a_lp2, sc_lp2 = _lp_alternating_minimization(best_seq, lp_deadline2, max_iters=40)
        try_update(a_lp2, "lp_alt_2")

    # === Phase 5: Fine L-BFGS ===
    print(f"\n=== Phase 5: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - 60 - time.monotonic()
            if remaining < 15:
                break
            max_iter = min(4000, max(300, int(remaining / 0.003)))
            a_new, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 6: Multi-restart with LP ===
    print(f"\n=== Phase 6: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        restart_configs = [
            ("g003", "gauss", 0.003),
            ("g008", "gauss", 0.008),
            ("g015", "gauss", 0.015),
            ("f2", "fourier", 0),
            ("f4", "fourier4", 0),
        ]

        for name, ptype, sigma in restart_configs:
            if time.monotonic() > deadline - 60:
                break

            if ptype == "gauss":
                p_start = _perturb_gaussian(best_seq, sigma, rng)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, rng, n_modes=2)
            else:
                p_start = _perturb_fourier(best_seq, rng, n_modes=4)

            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            # Quick LP step
            if time.monotonic() < deadline - 50:
                a_r, _ = _lp_alternating_minimization(p_start,
                                                       min(deadline - 40, time.monotonic() + 60),
                                                       max_iters=20, )
                try_update(a_r, f"restart_{name}_lp")

            # Quick L-BFGS
            if time.monotonic() < deadline - 40:
                n_r = len(best_seq)
                c_r = _full_convolve(best_seq, best_seq)
                mb_r = float(c_r.max())
                beta_r = mb_r * 1e-4
                max_iter = min(2000, max(200, int((deadline - 40 - time.monotonic()) / 0.003)))
                a_r2, _, _ = _lbfgs_run(p_start, beta_r, n_r, max_iter)
                try_update(a_r2, f"restart_{name}_lbfgs")

    # === Phase 7: Different lengths ===
    print(f"\n=== Phase 7: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for target_n in [2048, 3072, 6144, 8192]:
            if time.monotonic() > deadline - 50:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")

            if time.monotonic() < deadline - 40:
                remaining = deadline - 40 - time.monotonic()
                if remaining > 30:
                    c_i = _full_convolve(a_interp, a_interp)
                    mb_i = float(c_i.max())
                    beta_i = mb_i * 2e-4
                    max_iter = min(3000, max(200, int(remaining / 0.004)))
                    a_try, _, _ = _lbfgs_run(a_interp, beta_i, target_n, max_iter)
                    try_update(a_try, f"n_{target_n}")

    # === Phase 8: Final aggressive cycles ===
    print(f"\n=== Phase 8: Final Cycles (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        for rnd in range(6):
            if time.monotonic() > deadline - 10:
                break
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())

            # Tiny beta L-BFGS
            beta = mb * 5e-7
            remaining = deadline - 10 - time.monotonic()
            if remaining > 20:
                max_iter = min(3000, max(500, int(remaining / 0.004)))
                a_f, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
                try_update(a_f, f"final_{rnd}_lbfgs")

            # Tiny subgradient
            if time.monotonic() < deadline - 10:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 10,
                                            tol_frac=5e-5, max_steps=3000, lr_init=0.01)
                try_update(a_s, f"final_{rnd}_sg")

            # Micro LP
            if time.monotonic() < deadline - 10:
                a_l, _ = _lp_alternating_minimization(best_seq,
                                                       min(deadline - 5, time.monotonic() + 30),
                                                       max_iters=10)
                try_update(a_l, f"final_{rnd}_lp")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
