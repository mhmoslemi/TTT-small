import numpy as np

from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.12 * np.cos(2.0 * np.pi * 3.0 * t) * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.exp(-40.0 * (t - 0.3) ** 2) + 0.25 * np.exp(-40.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp_abs":
        t = np.linspace(0, 1, n)
        a = np.abs(np.sin(2.0 * np.pi * 5.0 * t * t + 1.0)) + 0.3
        return a / a.sum()
    else:

def _lbfgs_smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lbfgs_smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _lbfgs_smoobj(z, beta, n)

    def jac(z):
        return _lbfgs_smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 60,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init

def _compute_active_set(c, c_max, tol_frac):
    tol = tol_frac * c_max
    active = np.where(c >= (c_max - tol))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    return active


def _gradient_from_weights(a, weights_full, n):
    """Compute gradient: 2 * (weights * a_reversed)[n-1:2n-1]"""
    a_rev = a[::-1]
    wc = _full_convolve(weights_full, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(weights_full, _full_convolve(a, a)))
    return 2.0 * n * (h - W)


def _solve_dual_lp_small(a, active_idx, M, max_K=800):
    """
    Solve dual LP for small active set:
    min M * sum(lambda) s.t. sum_k lambda_k * a[k-m] >= 1 for all m, lambda >= 0
    Returns weights (normalized to sum to 1) or None.
    """
    n = len(a)
    K = len(active_idx)
    if K > max_K:
        return None
    if K == 0:
        return None

    # Build constraint matrix A[m, k] = a[active_idx[k] - m] if valid
    A = np.zeros((n, K))
    for k_idx in range(K):
        k = active_idx[k_idx]
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_end >= m_start:
            A[m_start:m_end + 1, k_idx] = a[k:m - m_start + 1 if False else k - m_end:k - m_start:-1] if m_end > m_start else np.array([a[0]])

    # Rebuild more carefully
    A = np.zeros((n, K))
    for k_idx in range(K):
        k = active_idx[k_idx]
        for m in range(max(0, k - n + 1), min(n - 1, k) + 1):
            A[m, k_idx] = a[k - m]

    # linprog: min c^T x s.t. A_ub x <= b_ub, bounds
    c_obj = M * np.ones(K)
    A_ub = -A
    b_ub = -np.ones(n)
    bounds = [(0, None)] * K

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.fun > 0:
            lam = res.x
            lam_sum = lam.sum()
            if lam_sum > 1e-15:
                return lam / lam_sum
        return None
    except Exception:
        return None

                best_a = a.copy()

def _coordinate_descent(a_init, deadline, max_rounds=2000):
    """
    Pairwise coordinate descent: adjust pairs of elements to reduce max convolution.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.003
    stall = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Determine which pairs to try
        idx = (rnd * 7919) % n
        jdx = (rnd * 7919 + 31) % n
        if idx == jdx:
            jdx = (jdx + 1) % n

        improved = False
        s_pair = best_a[idx] + best_a[jdx]
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
            delta = step * (frac - 0.5)
            new_i = s_pair * (frac + delta)
            new_j = s_pair * (1.0 - frac - delta)
            if new_i < 0 or new_j < 0:
                continue
            a_try = best_a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            a_try = a_try / a_try.sum()
            sc = _score(a_try)
            if sc < best_sc - 1e-14:
                best_a = a_try
                best_sc = sc
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 100:
                step *= 0.8
                if step < 1e-7:
                    break
        else:
            stall = 0

    return best_a, best_sc


def _perturb_freq(a, rng, n_modes=4, amp_scale=0.03):
    """Perturb in log-frequency domain."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 16)))
        amp = amp_scale * rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_gauss(a, rng, sigma=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()

        parent = a0

    # === Phase 1: L-BFGS warmup ===
    print(f"\n=== Phase 1: L-BFGS warmup (t={time.monotonic() - t_start:.1f}s) ===")
    a_p1 = _lbfgs_schedule(best_seq, deadline, min_time_remaining=900.0)
    try_update(a_p1, "lbfgs_warmup")

    # === Phase 2: LP-based active set ===
    print(f"\n=== Phase 2: LP active set (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 600:
        a_lp, sc_lp = _lp_based_optimize(best_seq, deadline - 600,
                                         tol_frac=5e-3, max_iter=3000)
        try_update(a_lp, "lp_active")

    # === Phase 3: Subgradient with averaging ===
    print(f"\n=== Phase 3: Subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_sg, sc_sg = _subgradient_optimize_safe(best_seq, deadline - 400)
        try_update(a_sg, "subgrad")

    # === Phase 4: Fine L-BFGS ===
    print(f"\n=== Phase 4: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([3e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - 200 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(4000, max(100, int(remaining / 0.003)))
            a_new = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord descent (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_cd, sc_cd = _coordinate_descent(best_seq, deadline - 200, max_rounds=3000)
        try_update(a_cd, "coord_desc")

    # === Phase 6: LP again with tighter tolerance ===
    print(f"\n=== Phase 6: LP tight (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_lp2, sc_lp2 = _lp_based_optimize(best_seq, deadline - 150,
                                           tol_frac=1e-4, max_iter=2000)
        try_update(a_lp2, "lp_tight")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_types = [
            ("freq_3", "freq", 3, 0.03),
            ("freq_5", "freq", 5, 0.02),
            ("freq_8", "freq", 8, 0.015),
            ("gauss_003", "gauss", 0, 0.03),
            ("gauss_008", "gauss", 0, 0.08),
            ("gauss_015", "gauss", 0, 0.15),
        ]
        for name, ptype, modes, sigma in restart_types:
            if time.monotonic() > deadline - 50:
                break
            if ptype == "freq":
                p_start = _perturb_freq(best_seq, rng, n_modes=modes, amp_scale=sigma)
            else:
                p_start = _perturb_gauss(best_seq, rng, sigma=sigma)

            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            # Quick optimization
            a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=30.0)
            try_update(a_r, f"r_{name}")

            if _score(a_r) < best_score_val + 0.003 and time.monotonic() < deadline - 25:
                a_r2, _ = _subgradient_optimize_safe(a_r, deadline - 25)
                try_update(a_r2, f"r_{name}_sg")

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 30:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_schedule(a_interp, deadline, min_time_remaining=20.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 9: Final polish cycle ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(6):
            if time.monotonic() > deadline - 5:
                break
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-6
            a_f = _lbfgs_run(best_seq, beta, n, 2000)
            try_update(a_f, f"fp_lbfgs_{rnd}")

            if time.monotonic() < deadline - 5:
                a_s, _ = _subgradient_optimize_safe(best_seq, deadline - 5)
                try_update(a_s, f"fp_sg_{rnd}")

            if time.monotonic() < deadline - 5:
                a_c, _ = _coordinate_descent(best_seq, deadline - 5, max_rounds=500)
                try_update(a_c, f"fp_cd_{rnd}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]


def _subgradient_optimize_safe(a_init, deadline):
    """Wrapper that handles exceptions gracefully."""
    try:
        return _subgradient_optimize(a_init, deadline, tol_frac=1e-4, max_iter=5000)
    except Exception as e:
        print(f"  Subgradient error: {e}")
        return a_init, _score(a_init)
