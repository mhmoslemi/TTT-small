Let me also consider: at the optimum, the KKT conditions tell us that the gradient of the max-conv should be parallel to the all-ones vector (for the sum constraint) plus the boundary terms. This means:
sum_{k in active} a[k-j] = constant (for all j where a[j] > 0)

This is a system of linear equations in a (given the active set). We can solve it!

Given an active set A (subset of {0, 1, ..., 2n-2}), the condition is:
sum_{k in A} a[k-j] = C (for all j in support(a))

This is: (P_A^T * a)[j] = C where (P_A^T * a)[j] = sum_{k in A} a[k-j].

But C is unknown, and we also have the constraint sum(a) = 1.

Actually, the KKT condition is:
2 * sum_{k in A} λ_k * a[k-j] = μ (for j in support)

where μ is the Lagrange multiplier for the sum constraint. If we set all λ_k = 1 (which is the case when all active constraints are "equally tight" in the max), then:
2 * sum_{k in A} a[k-j] = μ

So sum_{k in A} a[k-j] = μ/2 = C (constant for all j in support).

This is a linear system! If we know the active set A and the support of a, we can solve for a.

But we don't know A or the support a priori. This suggests an alternating algorithm:
1. Given a, compute the active set A (indices where conv[k] ≈ max).
2. Given A, solve the linear system to get the optimal a (with the constraint sum(a)=1, a>=0).
3. Repeat.

The linear system in step 2 is:
M * a = C * 1 (where M[j,k] = 1 if k-j is in A, 0 otherwise)
sum(a) = 1
a >= 0

This is a linear system with n unknowns and n+1 constraints (n from the KKT + 1 from the sum). It can be solved with standard linear algebra.

But wait, M is not square in general (it's n×n, but the system is M*a = C*1 with the additional constraint sum(a)=1). We have n+1 equations and n unknowns, so it's over-determined. We can solve it in the least-squares sense, or use active set methods (some of the a[j] = 0 constraints will be active).

```python
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _compute_cross(w, a):
    """Compute R[j] = sum_k w[k] * a[k-j] for j=0,...,n-1."""
    n = len(a)
    b = a[::-1]
    conv_wb = _full_convolve(w, b)
    return conv_wb[n-1 : 2*n-1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj_and_grad(z, beta, n):
    """Compute smoothed objective and gradient in z-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = c / beta - c_max / beta  # shift for numerical stability
    e = np.exp(shifted)
    S = e.sum()
    if S < 1e-300 or not np.isfinite(S):
        val = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        val = 2.0 * n * (c_max + beta * np.log(S))
        w = e / S
    # Gradient in a-space: grad_a[j] = 2n * beta / (sum(a))^2 * sum_k w[k] * a[k-j]
    # Since sum(a) = 1 (softmax), this simplifies to:
    # grad_a[j] = 2n * beta * sum_k w[k] * a[k-j] = 2n * beta * R[j]
    R = _compute_cross(w, a)
    grad_a = 2.0 * n * beta * R
    # Transform to z-space: grad_z[j] = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return val, grad_z

def _optimize_smoothed(a_init, deadline, max_time=None):
    """Optimize using L-BFGS with beta annealing."""
    n = len(a_init)
    a = a_init / a_init.sum()
    best_a = a.copy()
    best_score = _score(a)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    # Beta schedule: start large, decrease
    c_init = _full_convolve(a, a)
    c_max = c_init.max()
    c_sorted = np.sort(c_init)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted)-1)]
    base_beta = max(gap, c_max * 0.05)
    
    betas = [base_beta * f for f in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        est_time = 0.005 * max(1, n/2048.0)
        max_iter = min(3000, max(20, int(remaining * 0.7 / est_time)))
        
        def fun(z_local):
            val, _ = _smoobj_and_grad(z_local, beta, n)
            return val
        
        def jac(z_local):
            _, g = _smoobj_and_grad(z_local, beta, n)
            return g
        
        try:
            result = minimize(fun, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxcor': 50, 'maxls': 100})
            a_new = _softmax(result.x)
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    beta={beta:.6f}: score={sc:.10f}, improved")
        except Exception as e:
            print(f"    beta={beta:.6f}: error {e}")
    
    return best_a, best_score

def _projected_grad_descent(a_init, deadline, n_iter=100000):
    """Projected gradient descent with momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    
    v = np.zeros(n)  # momentum
    momentum = 0.9
    lr = 1.0
    
    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Identify active set (top values)
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Compute gradient
        R = _compute_cross(w, a)
        grad_a = 2.0 * R  # This is the subgradient of max conv (up to scaling)
        
        # Normalize gradient by n for step size
        grad_norm = np.linalg.norm(grad_a)
        if grad_norm < 1e-15:
            break
        
        # Update with momentum
        v = momentum * v - lr * grad_a / grad_norm
        a_new = a + v
        
        # Project: a >= 0, sum(a) = 1
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
            v = np.zeros(n)
            continue
        a_new = a_new / s
        
        # Check improvement
        if i % 100 == 0:
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            else:
                # Reduce learning rate
                lr *= 0.9
        
        a = a_new
    
    # Final check
    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

def _truncated_binomial(n):
    """Create the truncated binomial sequence."""
    a = np.zeros(n)
    for k in range(n):
        # binom(2k, k) / 4^k
        if k == 0:
            a[k] = 1.0
        else:
            # Recurrence: b[k] = b[k-1] * (2k-1)/(2k) * (2k-1)/k... 
            # Actually: binom(2k,k)/4^k = binom(2k-2,k-1)/4^{k-1} * (2k-1)/(2k) * (2k-1)/... 
            # Let's use the product form: binom(2k,k) = (2k)!/(k!)^2
            # binom(2k,k)/4^k = prod_{j=1}^{k} (2j-1)/(2j) * ... 
            # Recurrence: b[k] = b[k-1] * (2k-1)/k * k/(2k) = b[k-1] * (2k-1)/(2k)
            # Wait: b[k] = binom(2k,k)/4^k, b[k-1] = binom(2k-2,k-1)/4^{k-1}
            # b[k]/b[k-1] = [binom(2k,k)/binom(2k-2,k-1)] * 1/4
            # = [(2k)(2k-1)/(k^2)] * 1/4 = (2k-1)/(2k) * 2 = (2k-1)/k
            # Wait: binom(2k,k) = (2k)!/(k!)^2 = (2k)(2k-1)(2k-2)!/((k)(k-1) * (k-1)!)^2... 
            # = (2k)(2k-1)/k^2 * (2k-2)!/((k-1)!)^2 = (2k)(2k-1)/k^2 * binom(2k-2,k-1)
            # So b[k] = (2k)(2k-1)/k^2 * binom(2k-2,k-1)/4^k = (2k)(2k-1)/(4*k^2) * b[k-1]
            # = (2k-1)/(2k) * b[k-1]
            a[k] = a[k-1] * (2*k-1) / (2*k)
    a = a / a.sum()
    return a

def _ramp(n):
    """Create a linear ramp."""
    a = np.arange(1, n+1, dtype=np.float64)
    return a / a.sum()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        return _ramp(n)
    elif kind == "binomial":
        return _truncated_binomial(n)
    else:
        return np.ones(n) / n
