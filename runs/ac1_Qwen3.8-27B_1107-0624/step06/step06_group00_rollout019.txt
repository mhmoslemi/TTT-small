"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based convolution, adaptive active-set gradients with varying k,
simulated annealing for escape from local minima, diverse starting points,
backtracking line search, and multi-scale exploration (n=2048/4096/8192).
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, c, k, T):
    """
    Compute gradient using top-k active set with soft weighting.
    k: number of top convolution peaks to consider
    T: temperature for soft weighting (lower = sharper)
    """
    n = len(a)
    conv_len = 2 * n - 1
    # Find top-k indices of convolution
    if k >= conv_len:
        idx = np.arange(conv_len)
        weights = np.ones(conv_len)
    else:
        idx = np.argpartition(c, -k)[-k:]
        # Get weights based on values (soft-max over top-k)
        top_vals = c[idx]
        if T > 1e-15:
            shifted = (top_vals - top_vals.max()) / T
            weights = np.exp(shifted)
            weights = weights / weights.sum()
        else:
            weights = np.ones(k) / k
    
    # Build the probability distribution
    p = np.zeros(conv_len)
    p[idx] = weights

                   seed=42):
    """
    Adam optimizer in z-space with:
    - Adaptive active-set or two-scale gradient
    - Simulated annealing acceptance and periodic jumps
    - Adaptive restarts from best on stagnation
    """
    n = len(a_init)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0
    sa_T = sa_T_init

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)

        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        k = max(1, int(k_start * (k_end / k_start) ** frac))
        alpha = alpha_start + (alpha_end - alpha_start) * frac if use_two_scale else 1.0

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if use_two_scale:
            g = _two_scale_gradient(a, c, T, T * 0.1, alpha)
        elif T < T_start * 0.01:
            g = _active_gradient(a, c, k)
        else:
            g = _soft_gradient(a, c, T)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)

        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)

        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        a_new = _softmax_stable(z_new)
        sc_new = _score_norm(a_new)

        accepted = True
        if sa_enabled and sc_new > best_score:
            delta = sc_new - best_score
            if delta / max(sa_T, 1e-15) < 50.0:
                if rng.random() < np.exp(-delta / max(sa_T, 1e-15)):
                    accepted = True
                else:
                    accepted = False
            else:
                accepted = False
        sa_T *= sa_decay

        if accepted:
            z = z_new
        else:
            z = z + 0.01 * rng.standard_normal(n)
            z = np.clip(z, -80.0, 80.0)
            z = z - z.mean()
            a_new = _softmax_stable(z)
            sc_new = _score_norm(a_new)

        # Periodic SA jump
        if sa_enabled and sa_jump_interval > 0 and t % sa_jump_interval == 0:
            jump = sa_jump_size * rng.standard_normal(n)
            z = z + jump
            z = np.clip(z, -80.0, 80.0)
            z = z - z.mean()

        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        # Restart from best on stagnation
        if restart_interval > 0 and patience >= restart_interval:
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + restart_sigma * rng.standard_normal(n)
            z = np.clip(z, -80.0, 80.0)
            z = z - z.mean()
            m_buf = np.zeros(n)
            v_buf = np.zeros(n)
            patience = 0
            current_lr_mult = 1.0

        if patience > eval_interval * 10 and current_lr_mult > 0.1:
            current_lr_mult *= 0.6
            patience = 0

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _structured_perturb(a, rng, mode=0):
    """Apply a structured perturbation to a normalized sequence."""
    n = len(a)
    a_new = a.copy()
    if mode == 0:
        # Gaussian noise in log space
        z = np.log(np.maximum(a_new, 1e-15))
        z += 0.03 * rng.standard_normal(n)
        a_new = _softmax_stable(z)
    elif mode == 1:
        # Window perturbation: multiply by a smooth random window
        width = max(1, n // 8)
        center = rng.integers(0, n)
        window = 1.0 - 0.3 * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a_new = a_new * window
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
    elif mode == 2:
        # Subsequence flip: reverse a random contiguous block
        start = rng.integers(0, n - 1)
        length = rng.integers(10, max(11, n // 4))
        end = min(n, start + length)
        a_new[start:end] = a_new[start:end][::-1]
        a_new = a_new / a_new.sum()
    elif mode == 3:
        # Smooth sinusoidal perturbation
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        eps = rng.uniform(0.02, 0.08)
        window = 1.0 + eps * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a_new = a_new * window
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
    else:
        # Large random perturbation
        z = np.log(np.maximum(a_new, 1e-15))
        z += 0.1 * rng.standard_normal(n)
        a_new = _softmax_stable(z)
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse starts -> two-scale Adam -> active-set Adam
    -> multi-scale exploration -> structured perturbations -> fine polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n

    n0 = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n0}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: Quick optimization to improve starting point (0 - 100s)
    phase1_end = min(deadline, t_start + 100.0)
    if time.monotonic() < phase1_end:
        c0 = _fft_convolve(a, a)
        max_c0 = float(c0.max())
        a1, sc1 = _adam_optimize(
            a, phase1_end,
            T_start=0.01 * max_c0, T_end=0.001 * max_c0,
            lr_start=0.05, lr_end=0.02,
            k_start=20, k_end=3,
            max_iter=100000, eval_interval=150,
            use_two_scale=True, alpha_start=0.3, alpha_end=0.7,
            sa_enabled=False,
            restart_interval=500, restart_sigma=0.03,
            seed=100)
        full_sc1 = _score(a1)
        print(f"Phase 1: score={full_sc1:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc1) and full_sc1 < best_score:
            best_score = full_sc1
            best_a = a1.copy()

    # Phase 2: Two-scale gradient optimization (100s - 500s)
    phase2_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase2_end:
        c_b = _fft_convolve(best_a, best_a)
        max_cb = float(c_b.max())
        a2, sc2 = _adam_optimize(
            best_a, phase2_end - 3,
            T_start=0.005 * max_cb, T_end=0.0003 * max_cb,
            lr_start=0.04, lr_end=0.01,
            k_start=15, k_end=2,
            max_iter=300000, eval_interval=200,
            use_two_scale=True, alpha_start=0.5, alpha_end=0.85,
            sa_enabled=True, sa_T_init=0.003, sa_decay=0.9995,
            sa_jump_interval=800, sa_jump_size=0.15,
            restart_interval=600, restart_sigma=0.02,
            seed=200)
        full_sc2 = _score(a2)
        print(f"Phase 2 (two-scale): score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc2) and full_sc2 < best_score:
            best_score = full_sc2
            best_a = a2.copy()

    # Phase 3: Active-set gradient, low temperature (500s - 850s)
    phase3_end = min(deadline, t_start + 850.0)
    if time.monotonic() < phase3_end:
        c_b = _fft_convolve(best_a, best_a)
        max_cb = float(c_b.max())
        a3, sc3 = _adam_optimize(
            best_a, phase3_end - 3,
            T_start=0.002 * max_cb, T_end=0.0001 * max_cb,
            lr_start=0.03, lr_end=0.005,
            k_start=10, k_end=1,
            max_iter=300000, eval_interval=200,
            use_two_scale=False,
            sa_enabled=True, sa_T_init=0.002, sa_decay=0.9997,
            sa_jump_interval=1000, sa_jump_size=0.1,
            restart_interval=700, restart_sigma=0.015,
            seed=300)
        full_sc3 = _score(a3)
        print(f"Phase 3 (active): score={full_sc3:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc3) and full_sc3 < best_score:
            best_score = full_sc3
            best_a = a3.copy()

    # Phase 4: Multi-scale exploration - n=8192 (850s - 1150s)
    phase4_end = min(deadline, t_start + 1150.0)
    if time.monotonic() < phase4_end:
        n_large = 8192
        a_large = _interp_to_length(best_a, n_large)
        c_l = _fft_convolve(a_large, a_large)
        max_cl = float(c_l.max())
        a_large_opt, sc_large = _adam_optimize(
            a_large, phase4_end - 5,
            T_start=0.003 * max_cl, T_end=0.0002 * max_cl,
            lr_start=0.03, lr_end=0.008,
            k_start=10, k_end=2,
            max_iter=150000, eval_interval=200,
            use_two_scale=True, alpha_start=0.5, alpha_end=0.8,
            sa_enabled=True, sa_T_init=0.003, sa_decay=0.9995,
            sa_jump_interval=600, sa_jump_size=0.1,
            restart_interval=500, restart_sigma=0.02,
            seed=400)
        full_sc_large = _score(a_large_opt)
        print(f"Phase 4 (n=8192): score={full_sc_large:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_large) and full_sc_large < best_score:
            best_score = full_sc_large
            best_a = a_large_opt.copy()
            n_best = n_large
        else:
            n_best = len(best_a)
            # Try interpolating the large solution back
            if full_sc_large < 2.0:
                a_back = _interp_to_length(a_large_opt, n0)
                c_back = _fft_convolve(a_back, a_back)
                max_cback = float(c_back.max())
                a_back_opt, sc_back = _adam_optimize(
                    a_back, min(phase4_end - 3, t_start + 1100.0),
                    T_start=0.002 * max_cback, T_end=0.0002 * max_cback,
                    lr_start=0.025, lr_end=0.005,
                    k_start=8, k_end=2,
                    max_iter=80000, eval_interval=150,
                    use_two_scale=False,
                    sa_enabled=True, sa_T_init=0.002, sa_decay=0.9995,
                    seed=401)
                full_sc_back = _score(a_back_opt)
                print(f"  Interp back n={n0}: score={full_sc_back:.10f}")
                if np.isfinite(full_sc_back) and full_sc_back < best_score:
                    best_score = full_sc_back
                    best_a = a_back_opt.copy()
                    n_best = n0

    # Phase 5: Structured perturbation restarts (1150s - 1500s)
    phase5_end = min(deadline, t_start + 1500.0)
    n_best = len(best_a)
    if time.monotonic() < phase5_end:
        for attempt in range(8):
            if time.monotonic() > phase5_end - 30:
                break
            mode = attempt % 4
            a_pert = _structured_perturb(best_a, rng, mode=mode)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            a_pert_opt, sc_pert = _adam_optimize(
                a_pert, min(phase5_end - 5, t_start + 1500.0 - 40 * (8 - attempt)),
                T_start=0.001 * max_cp, T_end=0.0001 * max_cp,
                lr_start=0.025, lr_end=0.004,
                k_start=8, k_end=2,
                max_iter=60000, eval_interval=150,
                use_two_scale=(mode < 2),
                alpha_start=0.6, alpha_end=0.9,
                sa_enabled=True, sa_T_init=0.002, sa_decay=0.9997,
                sa_jump_interval=500, sa_jump_size=0.08,
                restart_interval=400, restart_sigma=0.02,
                seed=500 + attempt)
            full_sc = _score(a_pert_opt)
            print(f"  Pert[{attempt}] mode={mode}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()
                n_best = len(best_a)

    # Phase 6: Final ultra-fine polish (1500s - deadline)
    if time.monotonic() < deadline - 10:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_optimize(
            best_a, deadline - 8,
            T_start=0.0005 * max_cf, T_end=0.00002 * max_cf,
            lr_start=0.01, lr_end=0.001,
            k_start=5, k_end=1,
            max_iter=400000, eval_interval=100,
            use_two_scale=False,
            sa_enabled=True, sa_T_init=0.001, sa_decay=0.9998,
            sa_jump_interval=2000, sa_jump_size=0.03,
            restart_interval=1000, restart_sigma=0.008,
            seed=900)
        full_sc_final = _score(a_final)
        print(f"Phase 6 final: score={full_sc_final:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_final) and full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
