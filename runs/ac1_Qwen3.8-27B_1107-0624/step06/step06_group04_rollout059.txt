import numpy as np
import time

def _full_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    N = max(N, 2)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_c2(a):
    """Compute sum of squares of convolution (L2 norm squared of c)."""
    c = _full_convolve(a, a)
    return float(np.sum(c * c))

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses a two-stage approach: first minimize L2 norm of convolution (smooth),
then minimize smoothed max (logsumexp) with adaptive beta cooling.
Employs Adam with adaptive learning rates and multi-restart diversification.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_l2(a, c):
    """
    Gradient of sum_k c[k]^2 with respect to a[i], where c = a*a.
    Returns gradient of shape (n,).
    """
    n = len(a)
    a_rev = a[::-1]
    ca = _full_convolve(c, a_rev)  # length (2n-1) + n - 1 = 3n-2
    part1 = ca[n-1:2*n-1]  # indices n-1 to 2n-2
    ac = _full_convolve(a, c)  # length 3n-2
    part2 = ac[0:n]
    return 2.0 * (part1 + part2)


def _grad_smoothmax(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) with respect to a[i].
    Uses softmax weights for the max approximation.
    """
    n = len(a)
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    w_sum = float(e.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / w_sum
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)  # length 3n-2
    part1 = wc[n-1:2*n-1]
    aw = _full_convolve(a, w)
    part2 = aw[0:n]
    return 2.0 * (part1 + part2)

    grad_fn(a, c) should return the gradient of the objective.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:

    best_a = a.copy()
    c = _full_convolve(a, a)
    best_score = _score_norm(a)

        if un > clip:
            update = update * (clip / un)
        
        a_new = _proj_simplex(a - update)
        
        if t % eval_every == 0:
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 100:
                lr = max(0.001, lr * 0.9)
                no_improve = 0
            
            if t % lr_decay_every == 0:
                lr *= lr_decay_factor
        
        a = a_new
    
    c_final = _full_convolve(a, a)
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init=0.01, max_iter=500000,
                   eval_every=50, clip=0.5):
    """
    Adam in log-space for better handling of small values.
    """
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n
    
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()
    
    best_a = a_norm.copy()
    c_init = _full_convolve(a_norm, a_norm)
    best_score = _score_norm(a_norm)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    no_improve = 0
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1
            
            a = _softmax(z)
            c = _full_convolve(a, a)
            
            # Smoothing weights
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            ws = float(e.sum())
            if ws < 1e-300 or not np.isfinite(ws):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / ws
            
            # Gradient in log-space
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad * grad)
            mh = m / (1.0 - b1 ** t)
            vh = v / (1.0 - b2 ** t)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > clip:
                update = update * (clip / un)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % eval_every == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                
                if no_improve > 200 and t > 5000:
                    lr = max(0.001, lr * 0.95)
                    no_improve = 0
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _spectral_flatten(a_init, n_iter=300, deadline=None):
    """Iteratively flatten the spectrum of the sequence."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        # Blend: partially flatten
        A_flat = A / (mag + 1e-30) * avg_mag
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score


def _perturb_log(a, sigma, rng):
    """Perturb in log-space and renormalize."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
