            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_cosine(a_init, deadline, beta_schedule, lr_init, max_iter_total, warmup_frac=0.05):
    """Adam optimizer with cosine annealing learning rate."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    no_improve = 0
    warmup_iters = max(100, int(max_iter_total * warmup_frac))

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            # Cosine learning rate with warmup
            if t < warmup_iters:
                lr = lr_init * t / warmup_iters
            else:
                progress = (t - warmup_iters) / max(1, max_iter_total - warmup_iters)
                lr = lr_init * 0.5 * (1 + math.cos(math.pi * min(1.0, progress)))
                lr = max(lr, lr_init * 0.01)

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            if no_improve > 100 and t > 500:
                # Restart momentum
                m_buf *= 0.5
                no_improve = 0

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _multiplicative_weight_steps(a_init, n_steps, deadline, eta0=0.1, eta_min=0.001):
    """Multiplicative weight updates (mirror descent in log-space)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Use top-K peaks for the subgradient
        K = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:K]
        top_vals = c[top_idx] - cm
        weights = np.exp(top_vals / (0.01 * cm + 1e-15))
        weights = weights / weights.sum()

        # Weighted gradient
        w_full = np.zeros(len(c))
        w_full[top_idx] = weights

        a_rev = a[::-1]
        wc = _full_convolve(w_full, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w_full, c))
        grad = 2.0 * a * (h - W)

        # Adaptive step size
        eta = eta0 * (eta_min / eta0) ** (it / max(1, n_steps))
        z = z - eta * grad / (np.linalg.norm(grad) + 1e-15)
        z = z - z.mean()

        a_new = _softmax_compute(z)
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "two_peak":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _multi_scale_gradient(a, betas, n):
    """Compute gradient using multiple smoothing levels."""
    c = _full_convolve(a, a)
    cm = float(c.max())
    a_rev = a[::-1]
    
    total_grad = np.zeros(n)
    for beta in betas:
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / w_sum
        
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        total_grad += 2.0 * a * (h - W)
    
    return total_grad / len(betas), c, cm
