def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):

    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]

def _make_u_shape(n, sharpness=2.0):
    """Create a U-shaped (edge-heavy) profile."""
    t = np.linspace(0, 1, n)
    # U-shape: high at edges, low in middle
    a = 1.0 / (1.0 + sharpness * np.exp(-sharpness * (2*t - 1)**2))
    # Actually this gives a bump in the middle. Let me flip it.
    # U-shape: a = 1 / (1 + sharpness * exp(-sharpness * (2t-1)^2)) gives peak in middle
    # We want the opposite: high at edges
    a = 1.0 + sharpness * np.exp(-sharpness * (2*t - 1)**2)
    # This gives high in middle. We want high at edges.
    # Let's use: a = 1 + sharpness * (2t - 1)^2
    a = 1.0 + sharpness * (2*t - 1)**2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_v_shape(n, sharpness=1.0):
    """V-shaped: highest at edges, lowest at center."""
    t = np.linspace(0, 1, n)
    a = 1.0 + sharpness * np.abs(2*t - 1)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_gaussian_edges(n, width=0.3):
    """Gaussian bumps at both edges."""
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.0) / width)**2) + np.exp(-((t - 1.0) / width)**2) + 0.1
    return a / a.sum()


def _fourier_init(n, n_modes=16, rng=None):
    """Random Fourier series initialization (non-negative)."""
    if rng is None:
        rng = np.random.default_rng(42)
    t = np.arange(n) / n
    a = np.ones(n)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0, 2.0 / k)
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _resample(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

def _block_perturb(a, rng, n_blocks=4, block_frac=0.1):
    """Perturb by adding noise to random blocks."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, int(n * block_frac))
    for _ in range(n_blocks):
        start = rng.integers(0, n - block_size)
        sigma = rng.uniform(0.01, 0.1)
        z[start:start+block_size] += sigma * rng.standard_normal(block_size)
    z = np.clip(z, -50, 50)
    z = z - z.mean()
    return _softmax(z)
