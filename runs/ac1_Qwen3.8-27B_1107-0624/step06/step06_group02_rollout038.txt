Multi-scale optimizer for non-negative step functions minimizing 2n*max(a*a)/(sum a)^2.
Uses diverse initializations, Adam log-space optimization, FISTA with beta continuation,

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fftconv(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fftconv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1"""
    n = len(a)
    c = _fftconv(a, a)
    return float(2.0 * n * c.max())

def _softmax_weights(c, beta):
    """Compute softmax weights for smoothed max"""
    c_shift = (c - c.max()) / beta
    np.clip(c_shift, -500, 500, out=c_shift)
    e = np.exp(c_shift)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_smoothed_max(a, w):
    """
    Gradient of sum_k w[k] * c[k] where c = a*a (full convolution)
    c[k] = sum_i a[i]*a[k-i]
    dc/d a[j] = sum_k w[k] * (a[k-j] + a[j-k]) (with bounds)
    
    This can be computed as: 2 * (w * a_rev)[n-1:2n-1]
    where a_rev = a[::-1] and * is full convolution
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _fftconv(w, a_rev)
    # The n values from index n-1 to 2n-2 give dc/da
    grad = 2.0 * wc[n-1:2*n-1]
    return grad

def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x : x >= 0, sum(x) = target_sum}"""
    n = len(v)
    if n == 0:
        return v.copy()
    # Scale to target_sum
    v = v / target_sum  # work in simplex sum=1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0) * target_sum

                current_lr = max(0.0001, current_lr * 0.96)
        if verbose:
            print(f"    Adam beta={_beta:.2e}: score={best_score:.10f} iters={it_phase}")
    a_final = _softmax_norm(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _fista_simplex(a_init, beta, deadline, max_iter, ls_iters=8):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    step = 1.0 / n
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _fftconv(y, y)
        c_max = float(c_y.max())
        w = _softmax_weights(c_y, beta)
        grad = _grad_a(y, w)
        a_next = _proj_simplex(y - step * grad)
        # Backtracking
        for _ in range(ls_iters):
            c_trial = _fftconv(a_next, a_next)
            # If the max went up too much, reduce step
            if float(c_trial.max()) > c_max * 1.01:
                step *= 0.5
                a_next = _proj_simplex(y - step * grad)
            else:
                break
        if step < 1e-10:
            step = 1.0 / n
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
        if it % 1000 == 0:
            sc = _score_norm(a_curr)
            if sc < best_sc:
                best_sc = sc
                best_a = a_curr.copy()
    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _coord_descent(a_init, deadline, max_rounds=300, n_coords=1024):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(n_coords, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)
        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                a[j] = 1e-9
                sc = _score_norm(a / a.sum())
                if np.isfinite(sc) and sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = 0.0
                continue
            best_local = old_val
            best_local_sc = best_sc
            for factor in [0.90, 0.95, 0.98, 1.02, 1.05, 1.10]:
                a[j] = old_val * factor
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                sc = _score_norm(a / s)
                if np.isfinite(sc) and sc < best_local_sc:
                    best_local_sc = sc
                    best_local = old_val * factor
            a[j] = best_local
            if best_local != old_val:
                improved = True
                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    best_a = a.copy()
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved and r > 5:
            break
    return best_a, best_sc


def _make_hann(n):
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.12):
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x * x / (2.0 * sigma * sigma))
    return a / a.sum()


def _make_blackman(n):
    x = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _make_flat_top(n):
    x = np.arange(n, dtype=np.float64)
    N = float(n - 1)
    a = (0.21557895 - 0.41663158 * np.cos(2.0 * np.pi * x / N)
         + 0.277263158 * np.cos(4.0 * np.pi * x / N)
         - 0.083578947 * np.cos(6.0 * np.pi * x / N)
         + 0.006947368 * np.cos(8.0 * np.pi * x / N))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _make_dolphin(n):
    """Dolphin window (Kaiser with adaptive beta)"""
    x = np.arange(n, dtype=np.float64)
    N = n - 1
    a = np.ones(n)
    # Simple approximation: raised cosine with steeper edges
    t = 2.0 * x / N - 1.0
    a = np.cos(np.pi * t * 0.5) ** 2
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _short_adam(a_init, time_budget, beta_frac=0.001):
    """Quick Adam optimization with time budget in seconds"""
    deadline = time.monotonic() + time_budget
    c = _fftconv(a_init, a_init)
    max_c = float(c.max())
    betas = [beta_frac * max_c * (0.4 ** i) for i in range(6)]
    if betas[-1] < 1e-6 * max_c:
        betas[-1] = 1e-6 * max_c
    a_out, sc = _adam_logspace(a_init, deadline, betas, lr_init=0.008, max_iter_total=10**9)
    return a_out / a_out.sum(), _score(a_out / a_out.sum())

            candidates.append(("parent", parent.copy()))
        else:
            candidates.append(("parent_interp", _interp_profile(parent, n1)))

    candidates.append(("hann", _make_hann(n1)))
    candidates.append(("gauss", _make_gaussian(n1, 0.12)))
    candidates.append(("gauss_w", _make_gaussian(n1, 0.18)))
    candidates.append(("blackman", _make_blackman(n1)))

    candidates.append(("dolphin", _make_dolphin(n1)))
    # Random smooth
    z_rand = np.random.default_rng(seed).standard_normal(n1)
    # Smooth it

    a_rand = _softmax_norm(z_smooth)

    print("=== Phase 5: Multi-restart refinement ===")

                break
            if per_restart < 10:
                break
            # Diverse perturbations
            if i == 0:
                # Small Gaussian noise in log-space
                z = np.log(np.maximum(a_base, 1e-15))
                z += 0.01 * rng.standard_normal(n_best)
                z -= z.mean()
                p_start = _softmax_norm(z)
            elif i == 1:
                # Circular shift
                shift = rng.integers(1, n_best // 100 + 1)
                p_start = np.roll(a_base, shift).copy()
                p_start = p_start / p_start.sum()
            elif i == 2:
                # Frequency-domain perturbation

                A *= (1.0 + 0.02 * rng.standard_normal(len(A)) + 0.02j * rng.standard_normal(len(A)))
                p_start = np.fft.irfft(A, n_best)
                p_start = np.maximum(p_start, 0.0)
                p_start = p_start / p_start.sum()
            elif i == 3:
                # Block shuffle
                block = n_best // 8
                p_start = a_base.copy()
                for k in range(8):
                    s0 = k * block
                    s1 = (k + 1) * block
                    p_start[s0:s1] = np.roll(a_base[s0:s1], rng.integers(1, block // 4 + 1))
                p_start = p_start / p_start.sum()
            else:
                # Larger perturbation
                z = np.log(np.maximum(a_base, 1e-15))
                z += 0.05 * rng.standard_normal(n_best)
                z -= z.mean()
                p_start = _softmax_norm(z)

            sc_rs = _score(p_start)
            c_rs = _fftconv(p_start, p_start)
            max_rs = float(c_rs.max())
            betas_rs = [max_rs * f for f in [0.0002, 0.0001, 0.00005, 0.00002]]
            p_best_r, sc_best_r = _adam_logspace(
                p_start, time.monotonic() + per_restart * 0.8,
