    v_buf = np.zeros(n)

    patience = 0
    lr_mult = 1.0

def _next_pow2(n):

    Phase 4 (1050-1400s): Direct polish on best + eps-band gradients

    Phase 6 (1720s-end):  Ultra-fine final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Phase 1: Warm-to-cool Adam (0 - 350s)
    phase1_end = min(deadline, t_start + 350.0)
    if time.monotonic() < phase1_end:
        c_cur = _fft_convolve(a, a)
        max_c = float(c_cur.max())
        a1, sc1 = _adam_zspace(a, phase1_end,
                               T_start=0.005 * max_c, T_end=0.0005 * max_c,
                               lr_start=0.05, lr_end=0.008,
                               max_iter=500000, k_start=15, k_end=3,
                               eval_interval=200, clip_norm=0.6)
        full_sc = _score(a1)
        print(f"Phase 1: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a1.copy()

    # Phase 2: Cold Adam with small active set (350s - 700s)
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end:
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a2, sc2 = _adam_zspace(best_a, phase2_end,
                               T_start=0.001 * max_c, T_end=0.00005 * max_c,
                               lr_start=0.015, lr_end=0.002,
                               max_iter=500000, k_start=5, k_end=1,
                               eval_interval=100, clip_norm=0.4,
                               patience_limit=12)
        full_sc = _score(a2)
        print(f"Phase 2: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a2.copy()

    # Phase 3: Try n=8192 (700s - 1050s)
    phase3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase3_end:
        n_large = 8192
        a_large = _interp_to_length(best_a, n_large)
        c_large = _fft_convolve(a_large, a_large)
        max_cl = float(c_large.max())
        a3, sc3 = _adam_zspace(a_large, phase3_end,
                               T_start=0.003 * max_cl, T_end=0.00005 * max_cl,
                               lr_start=0.025, lr_end=0.003,
                               max_iter=300000, k_start=8, k_end=2,
                               eval_interval=200, clip_norm=0.5)
        full_sc = _score(a3)
        print(f"Phase 3 (n=8192): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a3.copy()
        # Try interpolating back to original n from large solution
        if len(a3) != n:
            a_int = _interp_to_length(a3, n)
            c_int = _fft_convolve(a_int, a_int)
            max_ci = float(c_int.max())
            a3b, sc3b = _adam_zspace(a_int, min(phase3_end, time.monotonic() + 60),
                                     T_start=0.001 * max_ci, T_end=0.00002 * max_ci,
                                     lr_start=0.01, lr_end=0.001,
                                     max_iter=100000, k_start=3, k_end=1,
                                     eval_interval=100, clip_norm=0.3)
            full_sc_b = _score(a3b)
            print(f"Phase 3b (interp back n={n}): score={full_sc_b:.10f}")
            if np.isfinite(full_sc_b) and full_sc_b < best_score:
                best_score = full_sc_b
                best_a = a3b.copy()

    # Phase 4: Epsilon-band gradient + direct polish (1050s - 1400s)
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end:
        n_best = len(best_a)
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        a4, sc4 = _adam_zspace(best_a, min(phase4_end - 60, time.monotonic() + 150),
                               T_start=0.0005 * max_cb, T_end=0.00002 * max_cb,
                               lr_start=0.008, lr_end=0.001,
                               max_iter=200000, k_start=3, k_end=1,
                               use_eps=True, eps_start=0.001 * max_cb, eps_end=0.0,
                               eval_interval=100, clip_norm=0.3, patience_limit=8)
        full_sc = _score(a4)
        print(f"Phase 4 eps-band: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a4.copy()

        # Direct polish
        if time.monotonic() < phase4_end - 10:
            a4b, sc4b = _try_direct_polish(best_a, phase4_end - 5, max_iter=100000)
            full_sc_b = _score(a4b)
            print(f"Phase 4b direct: score={full_sc_b:.10f}")
            if np.isfinite(full_sc_b) and full_sc_b < best_score:
                best_score = full_sc_b
                best_a = a4b.copy()

    # Phase 5: Diverse perturbation restarts (1400s - 1720s)
    phase5_end = min(deadline, t_start + 1720.0)
    n_best = len(best_a)
    restart_configs = [
        (0.008, 1),
        (0.015, 2),
        (0.025, 3),
        (0.01, 4),
        (0.02, 5),
        (0.005, 1),
    ]
    for attempt, (sigma, k_seed) in enumerate(restart_configs):
        if time.monotonic() > phase5_end - 45:
            break
        time_per_restart = 45.0
        z = np.log(np.maximum(best_a, 1e-15))
        z = z + sigma * rng.standard_normal(n_best)
        z = z - z.mean()
        a_pert = _softmax_stable(z)
        c_p = _fft_convolve(a_pert, a_pert)
        max_cp = float(c_p.max())
        sub_deadline = min(phase5_end - 5, time.monotonic() + time_per_restart)
        a_p, sc_p = _adam_zspace(a_pert, sub_deadline,
                                 T_start=0.001 * max_cp, T_end=0.00003 * max_cp,
                                 lr_start=0.012, lr_end=0.001,
                                 max_iter=100000, k_start=4, k_end=1,
                                 eval_interval=80, clip_norm=0.3,
                                 patience_limit=6)
        full_sc = _score(a_p)
        print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_p.copy()

    # Phase 6: Ultra-fine final polish (1720s - deadline)
    if time.monotonic() < deadline - 4:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a6, sc6 = _adam_zspace(best_a, deadline - 3,
                               T_start=0.00005 * max_cf, T_end=0.000002 * max_cf,
                               lr_start=0.002, lr_end=0.0001,
                               max_iter=200000, k_start=2, k_end=1,
                               eval_interval=50, clip_norm=0.2,
                               patience_limit=15)
        full_sc = _score(a6)
        print(f"Phase 6 final: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a6.copy()

    # Final validation and return
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
