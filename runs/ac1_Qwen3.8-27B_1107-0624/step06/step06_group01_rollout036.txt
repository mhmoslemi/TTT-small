def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _fista_optimize(a_init, deadline, n_iters=100000):
    """FISTA on the simplex with softmax-smoothed max."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    a_prev = a.copy()
    t = 1.0
    
    # Initial beta
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_mean = float(c.mean())
    beta = 0.05 * (c_max - c_mean) if c_max > c_mean else 0.01 * c_max
    
    best_a = a.copy()
    best_score = float(2.0 * n * c_max)
    patience = 0
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # FISTA extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + ((t - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        # Compute smoothed max and its gradient
        c_y = _full_convolve(y, y)
        c_max_y = float(c_y.max())
        shifted = (c_y - c_max_y) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient of smoothed max w.r.t. a
        # d/dy_i [sum_k w_k * (a*a)[k]] = 2 * sum_k w_k * a[k-i]
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n-1:2*n-1]
        
        # Step size: use Lipschitz estimate
        L = 4.0 * n * c_max_y / beta
        step = 1.0 / L
        
        # Project onto simplex (sum = 1, a >= 0)
        a_next = y - step * grad
        # Project to simplex
        a_next = _proj_simplex(a_next)
        
        a_prev = a.copy()
        a = a_next
        t = t_next
        
        # Periodic evaluation
        if it % 100 == 0:
            c_chk = _full_convolve(a, a)
            sc = float(2.0 * n * c_chk.max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 100
            
            # Adapt beta
            if patience > 2000:
                beta *= 0.5
                patience = 0
        
        # Gradually decrease beta
        if it % 5000 == 0 and it > 0:
            beta = max(1e-8, beta * 0.8)
    
    # Final check
    c_final = _full_convolve(a, a)
    sc_final = float(2.0 * n * c_final.max())
    if sc_final < best_score:
        best_a = a.copy()
        best_score = sc_final
    
    return best_a, best_score

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                # Too aggressive, reduce step
                step *= 0.5
                a_next = y - step * grad
                a_next = _proj_simplex(a_next)

        a_prev = a.copy()
        a = a_next
        t_val = t_next

        # Periodic evaluation
        if it % 100 == 0:
            c_chk = _full_convolve(a, a)
            sc = 2.0 * n * float(c_chk.max())
            if sc < best_score * (1.0 - 1e-8):
                best_score = sc
                best_cmax = float(c_chk.max())
                best_a = a.copy()
                iterations_without_improvement = 0
            else:
                iterations_without_improvement += 100

            # Warm restart: reset Nesterov momentum if stuck
            if iterations_without_improvement > 5000:
                t_val = 1.0
                a_prev = a.copy()
                iterations_without_improvement = 0

    c_final = _full_convolve(a, a)
    sc_final = 2.0 * n * float(c_final.max())
    if sc_final < best_score:
        best_a = a.copy()
        best_score = sc_final

    return best_a, best_score


def _fista_softmax(a_init, deadline, beta_init=0.02, max_iter=200000):
    """FISTA with softmax-smoothed max as alternative to top-K."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) / n
    else:
        a = a / s

    a_prev = a.copy()
    t_val = 1.0
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_mean = float(c.mean())
    beta = beta_init * (c_max - c_mean) if c_max > c_mean else 0.01 * c_max
    step = 1.0 / (4.0 * n * c_max / beta + 1e-15)

    best_a = a.copy()
    best_score = 2.0 * n * c_max
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a + ((t_val - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max_y = float(c_y.max())
        shifted = (c_y - c_max_y) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        L = 4.0 * n * c_max_y / beta + 1e-15
        step = 1.0 / L

        a_next = y - step * grad
        a_next = _proj_simplex(a_next)

        a_prev = a.copy()
        a = a_next
        t_val = t_next

        if it % 200 == 0:
            c_chk = _full_convolve(a, a)
            sc = 2.0 * n * float(c_chk.max())
            if sc < best_score * (1.0 - 1e-9):
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 200

            if patience > 5000:
                beta = max(1e-10, beta * 0.5)
                t_val = 1.0
                a_prev = a.copy()
                patience = 0

        if it % 10000 == 0 and it > 0:
            beta = max(1e-10, beta * 0.7)

    c_final = _full_convolve(a, a)
    sc_final = 2.0 * n * float(c_final.max())
    if sc_final < best_score:
        best_a = a.copy()
        best_score = sc_final

    return best_a, best_score


def _coordinate_descent(a_init, deadline, n_rounds=5000):
    """Targeted coordinate descent: perturb coordinates that contribute to peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_cmax = float(c.max())
    best_score = 2.0 * n * best_cmax
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find peak lag
        peak_lag = int(np.argmax(c))
        # Find coordinates that contribute to the peak: a[i]*a[peak_lag - i]
        i_min = max(0, peak_lag - n + 1)
        i_max = min(n - 1, peak_lag)
        if i_max < i_min:
            continue
        # The "pressure" on coordinate i is a[peak_lag - i]
        pressures = a[peak_lag - np.arange(i_min, i_max + 1)]
        # Pick the coordinate with highest pressure (most impactful to reduce)
        idx_in_range = int(np.argmax(pressures))
        i = i_min + idx_in_range

        # Try reducing this coordinate
        for factor in [0.95, 0.9, 0.85, 0.8, 1.05, 1.1]:
            a_try = a.copy()
            old_val = a_try[i]
            a_try[i] = old_val * factor
            if a_try[i] < 0:
                continue
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            c_try = _full_convolve(a_try, a_try)
            sc_try = float(c_try.max())
            if sc_try < best_cmax * (1.0 - 1e-10):
                best_cmax = sc_try
                best_score = 2.0 * n * best_cmax
                best_a = a_try.copy()
                a = a_try.copy()
                break

        # Also try random coordinate perturbations
        if rnd % 3 == 0:
            i_rand = int(rng.integers(0, n))
            for factor in [0.9, 1.1]:
                a_try = a.copy()
                a_try[i_rand] *= factor
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                c_try = _full_convolve(a_try, a_try)
                sc_try = float(c_try.max())
                if sc_try < best_cmax * (1.0 - 1e-10):
                    best_cmax = sc_try
                    best_score = 2.0 * n * best_cmax
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a, best_score


def _fourier_init(a_init, n_iters=100):
    """Quick Fourier shaping to get a better starting point."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = 2.0 * n * float(_full_convolve(a, a).max())

    for it in range(n_iters):
        b = _full_convolve(a, a)
        b = np.maximum(b, 0)
        b_max = b.max()
        if b_max < 1e-30:
            break
        b_norm = b / b_max
        p = 0.6 + 0.2 * (it / n_iters)  # start aggressive, become gentler
        b_shaped = b_norm ** p
        b_shaped_sum = b_shaped.sum()
        if b_shaped_sum < 1e-30:
            break
        b_target = b_shaped * (b.sum() / b_shaped_sum)

        B = np.fft.rfft(b_target, N)
        A = np.sqrt(B + 0j)
        a_new = np.fft.irfft(A, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s

        # Blend
        alpha = 0.7
        a_blend = (1.0 - alpha) * a + alpha * a_new
        a_blend = np.maximum(a_blend, 0)
        s_blend = a_blend.sum()
        if s_blend < 1e-30:
            break
        a_blend = a_blend / s_blend
        a = a_blend

        c = _full_convolve(a, a)
        sc = 2.0 * n * float(c.max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _perturb(a, sigma, seed=0):
    n = len(a)
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    z_max = z.max()
    e = np.exp(z - z_max)
    s = e.sum()
    return e / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                a_init = parent_raw / s
                print(f"Loaded parent: n={len(a_init)}, score={_score(a_init):.10f}")
            else:
                raise ValueError("sum too small")
        else:
            raise ValueError("empty")
    except Exception:
        a_init = np.ones(4096) / 4096
        print(f"No parent, using uniform n=4096, score={_score(a_init):.10f}")

    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score(a_init)
    print(f"Starting: n={n}, score={best_score:.10f}, time=0s")

    # Phase 1: Fourier shaping initialization (quick, ~60s)
    if time.monotonic() < deadline - 100:
        a_f, sc_f = _fourier_init(a_init, n_iters=200)
        sc_f_full = _score(a_f)
        print(f"After Fourier init: score={sc_f_full:.10f}")
        if sc_f_full < best_score:
            best_a = a_f.copy()
            best_score = sc_f_full
        # Use the better one for further optimization
        a_work = a_f if sc_f_full < best_score else a_init.copy()
    else:
        a_work = a_init.copy()

    # Phase 2: FISTA with top-K (main optimization, ~1000s)
    phase2_deadline = t_start + (deadline - t_start) * 0.65
    a_fista, sc_fista = _fista_k_schedule(a_work, phase2_deadline, K_start=10, K_end=1, max_iter=500000)
    sc_fista_full = _score(a_fista)
    print(f"After FISTA top-K: score={sc_fista_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc_fista_full < best_score:
        best_a = a_fista.copy()
        best_score = sc_fista_full

    # Phase 3: FISTA with softmax (different landscape, ~400s)
    if time.monotonic() < deadline - 200:
        phase3_deadline = min(deadline - 200, t_start + (deadline - t_start) * 0.85)
        a_sm, sc_sm = _fista_softmax(a_work, phase3_deadline, beta_init=0.015, max_iter=300000)
        sc_sm_full = _score(a_sm)
        print(f"After FISTA softmax: score={sc_sm_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_sm_full < best_score:
            best_a = a_sm.copy()
            best_score = sc_sm_full

    # Phase 4: Multi-restart from best (remaining time minus polish)
    if time.monotonic() < deadline - 120:
        n_restarts = 4
        remaining = deadline - time.monotonic() - 120
        per_restart = remaining / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > deadline - 120:
                break
            if i == 0:
                a_r = _perturb(best_a, 0.005, seed=100 + i)
            elif i == 1:
                a_r = _perturb(best_a, 0.02, seed=200 + i)
            elif i == 2:
                # Blend with uniform
                a_r = 0.8 * best_a + 0.2 * (np.ones(n) / n)
                a_r = a_r / a_r.sum()
            else:
                # Shift
                sz = max(1, n // 2048)
                shift = int(rng.integers(1, max(2, sz)))
                a_r = np.roll(best_a, shift).copy()
                a_r = a_r / a_r.sum()

            rd = time.monotonic() + max(10.0, per_restart)
            a_opt, sc_opt = _fista_k_schedule(a_r, rd, K_start=5, K_end=1, max_iter=200000)
            sc_opt_full = _score(a_opt)
            print(f"  Restart {i}: score={sc_opt_full:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_opt_full < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt_full

    # Phase 5: Coordinate descent polish
    if time.monotonic() < deadline - 30:
        cd_deadline = min(deadline - 10, time.monotonic() + 90)
        a_cd, sc_cd = _coordinate_descent(best_a, cd_deadline, n_rounds=20000)
        sc_cd_full = _score(a_cd)
        print(f"After coord descent: score={sc_cd_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_a = a_cd.copy()
            best_score = sc_cd_full

    # Phase 6: One more FISTA pass from the best
    if time.monotonic() < deadline - 20:
        a_final_f, sc_final_f = _fista_k_schedule(best_a, deadline - 5, K_start=3, K_end=1, max_iter=100000)
        sc_final_f_full = _score(a_final_f)
        if sc_final_f_full < best_score:
            best_a = a_final_f.copy()
            best_score = sc_final_f_full
            print(f"Final FISTA: score={best_score:.10f}")

    # Final validation and return
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(256) / 256
    # Ensure no NaN or inf
    final = np.nan_to_num(final, nan=0.0, posinf=1000.0, neginf=0.0)
    final = np.maximum(final, 0.0)

    sc_final = _score(final)
    print(f"\n=== FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s ===")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
