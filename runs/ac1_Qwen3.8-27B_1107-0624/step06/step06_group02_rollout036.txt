Hybrid optimizer combining smoothed gradient descent (Adam in log-space) with
exact 1D coordinate descent for minimizing 2*n*max(a*a)/(sum(a))^2 over
non-negative sequences. Includes multi-scale exploration (n=4096, 8192) and
random restarts.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1, so score = 2*n*max_c."""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())
