        else:
            y = np.ones(n) / n

        c = _full_conv(y, y)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        se = float(e.sum())
        if se < 1e-300 or not np.isfinite(se):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / se

        a_rev = y[::-1]
        wc = _full_conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 4.0 * n * beta * h

        a_new = _proj_simplex(y - step * grad)
        a_prev = a.copy()
        a = a_new
        t = t_next

        if it % 1000 == 0:
            sc = _score_nm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = _score_nm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a


def _adam_logspace(a_init, deadline, beta, lr_init=0.005, max_iter=300000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_nm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_conv(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        se = float(e.sum())
        if se < 1e-300 or not np.isfinite(se):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / se
        a_rev = a[::-1]
        wc = _full_conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 4.0 * n * a * (h - W)
        correction = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - correction)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 0.2:
            update *= 0.2 / un
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 100 == 0:
            a_chk = _softmax(z)
            sc = _score_nm(a_chk)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_chk.copy()

        if it % 5000 == 0:
            lr = max(0.0005, lr * 0.9)

    a_f = _softmax(z)
    sc_f = _score_nm(a_f)
    if np.isfinite(sc_f) and sc_f < best_sc:
        best_sc = sc_f
        best_a = a_f.copy()
    return best_a


def _cd_polish(a_init, deadline, max_rounds=30):
    """Direct coordinate descent on the simplex to reduce peak."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_nm(a)
    rng = np.random.default_rng(777)
    order = np.arange(n)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 5:
            break
        rng.shuffle(order)
        c = _full_conv(a, a)
        improved = False
        for i in order:
            if time.monotonic() > deadline - 5:
                break
            # Try adjusting a[i]
            base = a[i]
            a_rev = a[::-1]
            # g[k] = dc[k]/da[i] = a[k-i] + a[i-k] (with bounds, in shifted index)
            # In the shifted index (0..2n-2), position k corresponds to original lag k-(n-1)
            # a[k-i] is nonzero when 0 <= k-i < n, i.e., i <= k <= i+n-1
            # a[i-k] is nonzero when 0 <= i-k < n, i.e., i-n+1 <= k <= i
            # In shifted index: a[k-(n-1) - i] and a[i - (k-(n-1))]
            # Let's just compute g directly
            g = np.zeros(2 * n - 1)
            # a[k-i] for k = i..i+n-1 (shifted: k is 0..2n-2, a index is k-(n-1)-i... 
            # Actually let me just use the convolution approach
            ei = np.zeros(n)
            ei[i] = 1.0
            g = _full_conv(ei, a) + _full_conv(a, ei)
            # Now c_new[k] = c[k] + delta * g[k] + delta^2 * (1 if k == 2i-n+1 else 0)
            # where delta = new_a[i] - old_a[i] (approximately, ignoring renormalization)
            
            # Simple: try a few values
            best_local = best_sc
            best_delta = 0.0
            for frac in [0.7, 0.85, 0.5, 1.2, 1.5, 0.3]:
                new_val = base * frac
                if new_val < 0:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial = a_trial / a_trial.sum()
                sc_t = _score_nm(a_trial)
                if sc_t < best_local:
                    best_local = sc_t
                    best_delta = frac
            
            if best_delta != 0.0:
                a[i] = base * best_delta
                a = a / a.sum()
                improved = True
        
        c_new = _full_conv(a, a)
        sc_new = float(c_new.max()) * 2.0 * n
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
    
    return best_a


def _make_profile(n, kind="raised_cosine"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "raised_cosine":
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
    elif kind == "hamming":
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * t)
    elif kind == "blackman":
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if float(parent_raw.sum()) > 0.01:
                parent = parent_raw / float(parent_raw.sum())
    except Exception:
        parent = None

    n = 4096
    if parent is not None:
        n = len(parent)
        a = parent.copy()
        print(f"Parent: n={n}, score={_score(a):.10f}")
    else:
        a = _make_profile(n, "raised_cosine")
        print(f"Raised cosine: n={n}, score={_score(a):.10f}")

    best_a = a.copy()
    best_score = _score(a)

    # === Phase 1: FISTA warm-start (80s) ===
    print("Phase 1: FISTA warm-start")
    p1_end = min(deadline, t_start + 80.0)
    c0 = _full_conv(a, a)
    beta0 = 0.04 * float(c0.max())
    for b_frac in [1.0, 0.4, 0.15]:
        if time.monotonic() > p1_end - 3:
            break
        b = b_frac * beta0
        a = _fista_fast(a, b, min(p1_end - 2, time.monotonic() + 25.0), max_iter=100000)
        a = np.maximum(a, 0)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        print(f"  FISTA beta={b:.3e}: score={sc:.10f}")

    # === Phase 2: Adam log-space intermediate (150s) ===
    print("Phase 2: Adam log-space")
    p2_end = min(deadline, t_start + 230.0)
    if time.monotonic() < p2_end - 10:
        c_curr = _full_conv(a, a)
        max_c = float(c_curr.max())
        for b_frac in [0.01, 0.003, 0.001]:
            if time.monotonic() > p2_end - 5:
                break
            b = b_frac * max_c
            a = _adam_logspace(a, min(p2_end - 3, time.monotonic() + 45.0), b,
                               lr_init=0.004, max_iter=100000)
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            print(f"  Adam beta={b:.3e}: score={sc:.10f}")

    # === Phase 3: Progressive L-BFGS (main, ~1100s) ===
    print("Phase 3: Progressive L-BFGS (main)")
    p3_end = min(deadline, t_start + 1350.0)
    c_curr = _full_conv(a, a)
    max_c = float(c_curr.max())
    beta = 0.005 * max_c

    n_levels = 35
    ratio = 0.78
    total_time = p3_end - time.monotonic() - 20.0
    per_level = total_time / n_levels

    for lvl in range(n_levels):
        if time.monotonic() > p3_end - 15:
            break
        per = max(5.0, per_level)
        lvl_deadline = min(p3_end - 15, time.monotonic() + per)
        n_iter = max(200, int((lvl_deadline - time.monotonic()) * 200))
        n_iter = min(n_iter, 5000)

        a_new = _lbfgs_single(a, beta, lvl_deadline, max_iter=n_iter)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        sc = _score(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Keep trying from the better point
            a = a_new if _score_nm(a_new) < _score_nm(a) else a

        if lvl % 5 == 0:
            print(f"  L{lvl:2d} beta={beta:.3e}: score={sc:.10f} best={best_score:.10f} "
                  f"t={time.monotonic()-t_start:.0f}s")

        beta *= ratio
        if beta < 1e-10 * max_c:
            beta = 1e-10 * max_c

    # === Phase 4: Coordinate descent polish (150s) ===
    print("Phase 4: CD polish")
    p4_end = min(deadline, t_start + 1520.0)
    if time.monotonic() < p4_end - 10:
        a_cd = _cd_polish(best_a, p4_end, max_rounds=15)
        sc_cd = _score(a_cd)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
            print(f"  CD: score={sc_cd:.10f}")
        else:
            print(f"  CD: no improvement ({sc_cd:.10f} vs {best_score:.10f})")

    # === Phase 5: Final L-BFGS ultra-fine (120s) ===
    print("Phase 5: Final L-BFGS ultra-fine")
    p5_end = deadline - 8.0
    if time.monotonic() < p5_end - 5:
        c_f = _full_conv(best_a, best_a)
        max_cf = float(c_f.max())
        for b_frac in [2e-5, 5e-6, 1e-6]:
            if time.monotonic() > p5_end - 5:
                break
            b = b_frac * max_cf
            a_f = _lbfgs_single(best_a, b, min(p5_end - 3, time.monotonic() + 35.0),
                                max_iter=4000)
            a_f = np.maximum(a_f, 0)
            a_f = a_f / a_f.sum()
            sc_f = _score(a_f)
            if sc_f < best_score:
                best_score = sc_f
                best_a = a_f.copy()
                print(f"  Final beta={b:.2e}: score={sc_f:.10f}")
            else:
                print(f"  Final beta={b:.2e}: score={sc_f:.10f} (no improve)")

    # === Phase 6: Quick restart with different init (remaining) ===
    if time.monotonic() < deadline - 20:
        print("Phase 6: Quick restart")
        rem = deadline - time.monotonic() - 10
        if rem > 30:
            # Try a slightly different initial shape
            a_r = 0.7 * best_a + 0.3 * _make_profile(n, "hamming")
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            c_r = _full_conv(a_r, a_r)
            b_r = 0.002 * float(c_r.max())
            a_r = _lbfgs_single(a_r, b_r, min(deadline - 5, time.monotonic() + rem * 0.7),
                                max_iter=3000)
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_a = a_r.copy()
                print(f"  Restart: score={sc_r:.10f}")

    # Final validation
    print(f"\nDone: best_score={best_score:.10f}, n={len(best_a)}, "
          f"time={time.monotonic()-t_start:.0f}s")

    final = np.clip(best_a, 0.0, 1000.0)
    if float(final.sum()) < 0.01:
        final = np.ones(4096) / 4096
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
