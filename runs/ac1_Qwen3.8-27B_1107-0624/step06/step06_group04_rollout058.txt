def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _interp_profile(p, n_new):
