"""
Optimized sequence search using top-K smoothed FISTA, LP-guided directions,
and Adam log-space polishing. Targets score below 1.5027.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _topk_weights(c, K, beta):
    """Compute softmax weights only on top K values of c."""
    c_sorted = np.sort(c)[::-1]
    if K >= len(c):
        # Full softmax
        c_shifted = c - c.max()
        e = np.exp(c_shifted / beta)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        return w
    threshold = c_sorted[K - 1]
    mask = c >= threshold
    # Handle ties: include all values equal to threshold
    c_masked = np.where(mask, c, -np.inf)
    c_shifted = c_masked - c.max()
    e = np.exp(np.where(np.isfinite(c_shifted), c_shifted / beta, -np.inf))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        top_idx = np.argsort(c)[::-1][:K]
        w[top_idx] = 1.0 / K
    else:
        w = e / s
    return w

def _compute_grad(a, w):
    """Compute gradient of w @ conv(a,a) w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _fista_topk(a_init, deadline, max_iter, K, beta_init, beta_decay):
    """FISTA with top-K smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    beta = beta_init
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        w = _topk_weights(c_y, K, beta)
        
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 100 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
        
        # Reduce beta over time
        if it % 500 == 499:
            beta *= beta_decay
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _lp_direction(f, M, K):
    """Solve LP to find a direction that maximizes sum(g) s.t. conv(f,g) <= M at top K lags."""
    n = len(f)
    c = _full_convolve(f, f)
    # Get top K lag indices
    top_k_idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix
    # Constraint k: sum_j f[k-j] * g[j] <= M for j where 0 <= k-j <= n-1
    A = np.zeros((K, n))
    b = np.full(K, M)
    
    for idx, k in enumerate(top_k_idx):
        # g[j] appears with coefficient f[k-j] when 0 <= k-j <= n-1
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[idx, j] = f[k - j]
    
    # Solve: maximize sum(g) s.t. A @ g <= b, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

        if time.monotonic() > deadline:
            break

        if remain_iter <= 0:
            break

        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            else:

            if not np.all(np.isfinite(grad)):

            vh = v_buf / (1.0 - b2 ** t)

            update_norm = np.linalg.norm(update)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-K smoothed FISTA, LP-guided directions, SSE (Gerchberg-Saxton) initialization,
Adam log-space polishing, and coordinate descent. Targets score < 1.5027.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_weights(c, K, beta):
    n_c = len(c)
    if K >= n_c:
        c_shifted = (c - c.max()) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        return w
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[K - 1]
    mask = c >= threshold
    c_masked = np.where(mask, c - c.max(), -700.0)
    e = np.exp(c_masked / beta)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        top_idx = np.argsort(c)[::-1][:K]
        w[top_idx] = 1.0 / K
    else:
        w = e / s
    return w


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_topk(a_init, deadline, K, beta_init, beta_decay, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    beta = beta_init

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = float(y.sum())
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_weights(c_y, K, beta)
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 3:
            sc = _score_norm(a_next)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

        if it % 400 == 399:
            beta *= beta_decay

    sc_final = _score_norm(a_curr)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _ss_init(n, rng, n_iter=200):
    """Spectral synthesis (Gerchberg-Saxton) for flat-spectrum non-negative sequence."""
    a = rng.random(n) + 0.1
    a = a / a.sum()
    for _ in range(n_iter):
        F = np.fft.rfft(a)
        phase = np.angle(F)
        F_new = np.exp(1j * phase)
        F_new = np.concatenate([F_new, [0], np.conj(F_new[1:][::-1])]) if n % 2 == 0 else \
                np.concatenate([F_new, np.conj(F_new[1:][::-1])])
        N_fft = len(F_new)
        a_new = np.fft.irfft(F_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = 0.5 * a + 0.5 / n
            s = a_new.sum()
        a = a_new / s
    return a


def _chirp_init(n, rng):
    """Initialize with a chirp-modulated envelope for good spectral properties."""
    t = np.arange(n)
    freq = rng.uniform(0.1, 0.5)
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _lp_direction(f, M, K=30):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)[k]<=M for top K lags."""
    n = len(f)
    c = _full_convolve(f, f)
    top_k_idx = np.argsort(c)[::-1][:K]

    A = np.zeros((K, n))
    b = np.full(K, M)
    for idx, k in enumerate(top_k_idx):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        A[idx, j_min:j_max + 1] = f[k - j_max:k - j_min + 1]

    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': 30})
    if res.success:
        return res.x
    return None


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 40 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            if it_in_phase % 10000 == 0:
                current_lr = max(0.0003, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a, deadline, n_rounds=50):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(7)
    for r in range(n_rounds):
        if time.monotonic() > deadline:
            break
        for _ in range(8):
            i = int(rng.integers(0, n))
            for f in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a_try = best_a.copy()
                a_try[i] *= f
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    sc = _score_norm(a_try)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_try.copy()
                        break
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

                    best_for_n = a_opt.copy()

            print(f"  Init {init_idx}: best_sc_n={best_sc_n:.10f} t={time.monotonic()-t_start:.0f}s")

        if best_for_n is not None and best_sc_n < best_score:
            best_score = best_sc_n
            best_seq = best_for_n.copy()
            print(f"  New global best: {best_score:.10f}, n={n_target}")

    # Multi-restart diversification
    if best_seq is not None and time.monotonic() < deadline - 80:
        print(f"\n=== Multi-restart (best={best_score:.10f}) ===")
        n_best = len(best_seq)
        a_work = best_seq.copy() / best_seq.sum()
        strategies = [
            lambda: _perturb(a_work, 0.015, rng),
            lambda: _perturb(a_work, 0.04, rng),
            lambda: np.roll(a_work, int(rng.integers(1, max(2, n_best // 2048)))).copy(),
        ]
        for i, strat in enumerate(strategies):
            if time.monotonic() > deadline - 60:
                break
            p_start = strat()
            p_start = np.maximum(p_start, 0.0)
            ps = p_start.sum()
            if ps < 1e-15:
                continue
            p_start = p_start / ps
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas = [0.0005 * max_cs, 0.0001 * max_cs]
            p_best, sc_best = _adam_logspace(p_start, time.monotonic() + 30.0, betas, lr_init=0.008, max_iter_total=100000)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: {sc_full:.10f}")

    # Final ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Final polish ===")
        a_pol = np.maximum(best_seq, 0.0)
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000003 * max_cp]
        per_t = deadline - time.monotonic() - 8
        if per_t > 15:
            a_pol, _ = _adam_logspace(a_pol, time.monotonic() + per_t * 0.6, betas_ult, lr_init=0.002, max_iter_total=80000)
            sc_pol = _score(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"  Ultra-fine: {sc_pol:.10f}")

    # Coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_cd, _ = _coord_descent(best_seq, time.monotonic() + min(10.0, deadline - time.monotonic() - 3))
        sc_cd = _score(a_cd)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  Coord descent: {sc_cd:.10f}")

    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            s = 1.0
        sc_final = _score(final)
        print(f"Final: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
