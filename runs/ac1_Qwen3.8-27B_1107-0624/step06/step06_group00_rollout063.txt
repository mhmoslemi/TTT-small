def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)

    fb = np.fft.rfft(b, N)

"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Adaptive top-k gradient with gap-based k selection
- Gradient-aligned momentum (heavy ball)
- Asymmetry enforcement via z-space tilt
- Multi-scale portfolio
- Backtracking line search
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _adaptive_topk_gradient(a, c, max_k=None):
    """
    Compute gradient using adaptive top-k selection.
    k is chosen based on the gap between consecutive sorted values.
    """
    n = len(a)
    if max_k is None:
        max_k = max(1, n // 8)
    
    sorted_c = np.sort(c)[::-1]
    # Find the natural gap: where relative drop-off occurs
    # Use the "knees" of the sorted curve
    k = 1
    for i in range(1, min(max_k, len(c))):
        if sorted_c[i] / max(sorted_c[0], 1e-30) > 0.8:
            k = i + 1
        else:
            break
    
    k = max(1, min(k, max_k))
    
    top_idx = np.argsort(c)[::-1][:k]
    p = np.zeros_like(c)
    p[top_idx] = 1.0 / k
    
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n-1:2*n-1]
    return g, k

def _soft_gradient(a, c, T):
    """Standard soft-max gradient."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n-1:2*n-1]
    return g

def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=200000, use_adaptive_k=False, max_k=None,
                   eval_interval=100, clip_norm=1.0, tilt=0.0):
    """Adam optimizer in z-space with adaptive gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    if tilt != 0.0:
        z = z + tilt * np.linspace(-0.5, 0.5, n)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frag
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        if use_adaptive_k and T < T_start * 0.1:
            g, k_used = _adaptive_topk_gradient(a, c, max_k)
        else:
            g = _soft_gradient(a, c, T)
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:

            if patience > 8:

                m_buf = np.zeros(n)

    if np.isfinite(sc_final) and sc_final < best_score:

def _solve_qp_simplex(B_mul, init, eta=0.5, max_iter=1000):
    """
    Solve min a^T B a s.t. sum(a)=1, a>=0 using FISTA.
    B_mul is a function that computes B @ v for a given v.
    """
    n = len(init)
    a = init.copy()
    y = a.copy()
    t_k = 1.0
    
    for _ in range(max_iter):
        Ba = B_mul(a)
        grad = 2.0 * Ba
        a_new = y - eta * grad
        a_new = _project_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        a = a_new
        t_k = t_new
    
    return a

"""
Game-theoretic + Adam optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multiplicative weights for the column player (lag distribution),
FISTA for the row player's QP best response, adaptive top-k gradients,
asymmetry enforcement, and multi-scale portfolio exploration.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _Bq_matvec(q, v):
    """Compute B(q) @ v where B(q) = sum_k q_k M_k.
    (B(q)v)_j = sum_k q_k * v_{k-j}
    """
    n = len(v)
    q_rev = q[::-1]
    conv = _fft_convolve(v, q_rev)
    return conv[n - 1:2 * n - 1]


def _grad_soft(a, c, T):
    """Standard soft-max gradient of 2n*max(c)."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1]
    return g


def _grad_topk(a, c, k):
    """Gradient using top-k lags with equal weight."""
    n = len(a)
    k = max(1, min(k, len(c)))
    top_idx = np.argsort(c)[::-1][:k]
    p = np.zeros_like(c)
    p[top_idx] = 1.0 / k
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1]
    return g


def _grad_adaptive(a, c, max_k):
    """Adaptive top-k based on gap analysis."""
    n = len(a)
    sorted_c = np.sort(c)[::-1]
    top_val = sorted_c[0]
    if top_val < 1e-30:
        k = 1
    else:
        k = 1
        for i in range(1, min(max_k, len(c))):
            if sorted_c[i] / top_val > 0.7:
                k = i + 1
            else:
                break
    k = max(1, min(k, max_k))
    return _grad_topk(a, c, k), k


def _fista_qp(q, init_a, max_iter=200, eta=0.4):
    """Solve min a^T B(q) a s.t. a in simplex using FISTA."""
    n = len(init_a)
    a = init_a.copy()
    y = a.copy()
    t_k = 1.0
    for _ in range(max_iter):
        Ba = _Bq_matvec(q, y)
        grad = 2.0 * Ba
        a_new = y - eta * grad
        a_new = _project_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 0:
            y = y / y_sum
        else:
            y = a_new
        a = a_new
        t_k = t_new
    return a


def _game_iterative(a_init, deadline, max_rounds=500):
    """Game-theoretic optimization: alternate between QP best response and MW update."""
    n = len(a_init)
    n_lags = 2 * n - 1
    a = a_init.copy()
    q = np.ones(n_lags) / n_lags
    best_a = a.copy()
    best_score = _score_norm(a)
    eta_mw = 1.0 / (10.0 * n ** 0.5)
    fista_iters = 100

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        # Row player: best response to q
        a_new = _fista_qp(q, a, max_iter=fista_iters, eta=0.4)
        # Evaluate
        c = _fft_convolve(a_new, a_new)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        # Column player: multiplicative weights update on q
        c_shifted = (c - c.max()) * eta_mw
        q_new = q * np.exp(-c_shifted)
        q_sum = q_new.sum()
        if q_sum < 1e-300 or not np.isfinite(q_sum):
            q_new = np.ones(n_lags) / n_lags
        else:
            q_new = q_new / q_sum
        q = q_new
        a = a_new
        # Progressively increase MW aggressiveness
        eta_mw *= 1.02
        if r % 50 == 0:
            print(f"    Game round {r}: score={sc:.8f}, eta_mw={eta_mw:.4f}, t={time.monotonic():.0f}")

    return best_a, best_score


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=300000, use_adaptive=False, max_k=100,
                  eval_interval=100, clip_norm=1.0, tilt=0.0):
    """Adam in z-space with gradient options."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    if tilt != 0.0:
        z = z + tilt * np.linspace(-0.5, 0.5, n)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if use_adaptive and T < T_start * 0.05:
            g, _ = _grad_adaptive(a, c, max_k)
        else:
            g = _grad_soft(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 6 and lr_mult > 0.2:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

        print("Phase 5: Multi-scale exploration")
        for n_try in [4096, 8192]:
            if time.monotonic() > phase5_end - 60:
                break
            a_large = _interp_to_length(best_a, n_try)
            c_l = _fft_convolve(a_large, a_large)
            max_cl = float(c_l.max())
            a_opt, _ = _adam_z_space(a_large, min(phase5_end - 30, t_start + 1540.0),
                                      T_start=0.003 * max_cl, T_end=0.00001 * max_cl,
                                      lr_start=0.025, lr_end=0.002,
                                      max_iter=150000, use_adaptive=True, max_k=100,
                                      eval_interval=150)
            full_sc = _score(a_opt)
            print(f"  n={n_try}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()
            # Interpolate back
            a_back = _interp_to_length(a_opt, len(start_a) if best_a is not None else n_base)
            a_back_opt, _ = _active_set_adam(a_back, min(phase5_end - 5, t_start + 1545.0),
                                              k=3, lr_start=0.005, lr_end=0.0003,
                                              max_iter=80000, eval_interval=100)
            full_sc_b = _score(a_back_opt)
            print(f"  interp back: score={full_sc_b:.10f}")
            if np.isfinite(full_sc_b) and full_sc_b < best_score:
                best_score = full_sc_b
                best_a = a_back_opt.copy()

    # Phase 6: Diversified restarts (1550-1720s)
    phase6_end = min(deadline, t_start + 1720.0)
    if time.monotonic() < phase6_end and best_a is not None:
        print("Phase 6: Diversified restarts")
        n_best = len(best_a)
        for attempt in range(4):
            if time.monotonic() > phase6_end - 30:
                break
            # Different types of perturbations
            if attempt == 0:
                sigma = 0.02
                z = np.log(np.maximum(best_a, 1e-15)) + sigma * rng.standard_normal(n_best)
            elif attempt == 1:
                # Block perturbation
                z = np.log(np.maximum(best_a, 1e-15)).copy()
                block = n_best // 4
                z[block:2*block] += 0.05
                z[3*block:] -= 0.03
            elif attempt == 2:
                # Frequency-domain perturbation
                z = np.log(np.maximum(best_a, 1e-15)).copy()
                fz = np.fft.rfft(z)
                freq = np.arange(len(fz))
                mask = (freq > 2) & (freq < len(fz) // 3)
                fz[mask] += 0.1 * rng.standard_normal(np.sum(mask)) * 1j
                z = np.fft.irfft(fz, n_best)
            else:
                # Strong tilt
                z = np.log(np.maximum(best_a, 1e-15)) + 0.5 * np.linspace(-1, 1, n_best)

            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            a_pert_opt, _ = _active_set_adam(a_pert, min(phase6_end - 5, t_start + 1715.0),
                                              k=3, lr_start=0.006, lr_end=0.0004,
                                              max_iter=60000, eval_interval=80)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}]: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 7: Final ultra-fine polish (1720s - deadline)
    if time.monotonic() < deadline - 10 and best_a is not None:
        print("Phase 7: Final polish")
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        # Try different active set sizes
        for k_try in [2, 3, 5]:
            if time.monotonic() > deadline - 12:
                break
            a_polish, _ = _active_set_adam(best_a, deadline - 8, k=k_try,
                                            lr_start=0.003, lr_end=0.0002,
                                            max_iter=100000, eval_interval=50)
            full_sc = _score(a_polish)
            print(f"  Polish k={k_try}: score={full_sc:.10f}")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_polish.copy()

    # Final validation
    if best_a is None:
        best_a = np.ones(256) / 256
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
        s = 256.0
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
