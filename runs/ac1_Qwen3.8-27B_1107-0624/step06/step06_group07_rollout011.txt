import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)

    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max, a, c
    return 2.0 * n * (c_max + beta * np.log(s)), a, c

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA with simplex projection, adaptive log-sum-exp smoothing,
active-set QP refinement, multi-resolution search, and diverse restarts.
Key differences from prior approach: FISTA acceleration, projection-based
simplex optimization, active-set KKT refinement, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(y):
    """Project y onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(y)
    if n == 0:
        return y
    # Sort in descending order
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)


def _smoothed_objective(a, beta, n):
    """Compute smoothed objective value."""
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return 2.0 * n * c_max, c
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max, c
    return 2.0 * n * (c_max + beta * np.log(s)), c


def _smoothed_gradient(a, beta, n):
    """Compute gradient of smoothed objective w.r.t. a (unprojected)."""
    c = _full_convolve(a, a)
    c_max = c.max()

        return np.zeros_like(a), c
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * (h - W)
    return grad, c


def _subgradient_active(a, c, n, tol_frac=1e-4):
    """Compute subgradient using tight active set."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _fista_smoothed(a_init, beta, n, max_iter=5000, lr=None, deadline=None):
    """FISTA for the smoothed problem on the simplex."""
    a = a_init.copy()
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = _score(a)

    if lr is None:
        # Estimate Lipschitz constant
        g, _ = _smoothed_gradient(a, beta, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-30:
            return a, best_sc
        # Use gradient norm to estimate step
        lr = 1.0 / (2.0 * np.linalg.norm(_full_convolve(a, a)) + beta * n + 1e-10)
        lr = min(lr, 1.0 / (4.0 * n * np.max(a) + 1e-10))
        lr = max(lr, 1e-6 / (n ** 0.5))

    for i in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + ((t - 1.0) / t_new) * (a - a_prev)
        y = _project_simplex(y)

        g, c = _smoothed_gradient(y, beta, n)

        a_new = _project_simplex(y - lr * g)

        sc = _score(a_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_new.copy()

        a_prev = a
        a = a_new
        t = t_new

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

        c = _full_convolve(a, a)
        c_max = c.max()
        if c_max < 1e-30:
            return np.zeros_like(z)
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxcor': 60,
                                   'maxls': 60, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init


def _active_set_qp(a_init, n, max_iter=20, deadline=None):
    """
    Active-set refinement: identify active constraints, solve QP.
    min t s.t. c[k] <= t for k in active, sum(a) = 1, a >= 0
    Using scipy minimize with constraints.
    """
    a = a_init.copy()
    c = _full_convolve(a, a)
    c_max = c.max()
    best_a = a.copy()
    best_sc = _score(a)

    for iteration in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Identify active set (top peaks)
        tol = 1e-3 * c_max
        active = c >= (c_max - tol)
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])

        # Use L-BFGS with multiple constraints via penalty
        def obj_and_grad(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-30:
                return 1e10, np.zeros_like(x)
            x = x / s
            cx = _full_convolve(x, x)
            # Penalty for active constraints
            pen = 0.0
            grad_pen = np.zeros_like(x)
            for k in active_idx:
                if cx[k] > c_max:
                    viol = cx[k] - c_max
                    pen += viol ** 2
                    # Gradient of cx[k] w.r.t. x
                    a_rev = x[::-1]
                    delta = np.zeros(2 * n - 1)
                    delta[k] = 1.0
                    wc = _full_convolve(delta, a_rev)
                    h = wc[n - 1:2 * n - 1]
                    grad_pen += 2.0 * viol * 2.0 * n * h
            base_val = 2.0 * n * cx.max()
            base_g, _ = _subgradient_active(x, cx, n, tol_frac=1e-3), None
            # Simplified: just use subgradient
            sg = _subgradient_active(x, cx, n, tol_frac=1e-3)
            return base_val + 100.0 * pen, sg + grad_pen

        # Just do a few L-BFGS steps
        z0 = np.log(np.maximum(a, 1e-15))
        z0 = z0 - z0.mean()

        def fun_z(z):
            x = _softmax(z)
            cx = _full_convolve(x, x)
            pen = 0.0
            for k in active_idx:
                if cx[k] > c_max * (1 + 1e-4):
                    pen += (cx[k] - c_max) ** 2
            return 2.0 * n * cx.max() + 50.0 * pen

        def jac_z(z):
            x = _softmax(z)
            cx = _full_convolve(x, x)
            sg = _subgradient_active(x, cx, n, tol_frac=1e-3)
            pen_g = np.zeros_like(x)
            for k in active_idx:
                if cx[k] > c_max * (1 + 1e-4):
                    viol = cx[k] - c_max
                    delta = np.zeros(2 * n - 1)
                    delta[k] = 1.0
                    wc = _full_convolve(delta, x[::-1])
                    h = wc[n - 1:2 * n - 1]
                    pen_g += 200.0 * viol * 2.0 * n * h
            total = sg + pen_g
            return x * (total - np.dot(x, total))

        try:
            result = minimize(fun_z, z0, jac=jac_z, method='L-BFGS-B',
                              options={'maxiter': 500, 'ftol': 1e-16,
                                       'gtol': 1e-14, 'maxcor': 40, 'disp': False})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < best_sc - 1e-14:
                best_a = a_new.copy()
                best_sc = sc_new
                a = a_new
                c = _full_convolve(a, a)
                c_max = c.max()
        except Exception:
            break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ripple":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(0)
        a = np.abs(rng.standard_normal(n)) + 0.5
        # Smooth it
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
        a = a / a.sum()
        return a
    else:
        return np.ones(n) / n


def _perturb_fourier(a, rng, n_modes=5, amp=0.003):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        a_ = amp * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + a_ * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_swap(a, rng, n_swaps=10):
    n = len(a)
    a = a.copy()
    for _ in range(n_swaps):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        # Swirl: transfer mass from i to j
        frac = rng.uniform(0.05, 0.2)
        transfer = a[i] * frac
        a[i] -= transfer
        a[j] += transfer
        a[i] = max(a[i], 0)
    s = a.sum()
    if s < 1e-30:
        return a.copy()
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining FISTA, L-BFGS with adaptive smoothing,
    active-set QP refinement, and multi-resolution search to minimize
    2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p_norm = p / s  # normalize to sum=1 for our computations
        sc = _score(p_norm)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p_norm.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p_norm)}) [{label}] "
                  f"improvement={prev - best_score_val:.4e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: Multi-resolution initialization (0-200s) ===
    print(f"\n=== Phase 1: Multi-resolution (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 1400:
        n_target = len(best_seq)
        # Try a few different n values
        for trial_n in [n_target, n_target // 2, n_target * 2]:
            if trial_n < 256 or trial_n > 16384:
                continue
            if trial_n == n_target:
                continue
            if time.monotonic() > deadline - 1400:
                break
            a_interp = _interpolate_to_n(best_seq, trial_n)
            # Quick FISTA
            c0 = _full_convolve(a_interp, a_interp)
            beta = float(c0.max()) * 0.05
            n_tri = len(a_interp)
            a_r, sc_r = _fista_smoothed(a_interp, beta, n_tri,
                                        max_iter=2000, deadline=deadline - 1400)
            try_update(a_r, f"multi_n_{trial_n}")
            if len(a_r) != n_target:
                a_back = _interpolate_to_n(a_r, n_target)
                try_update(a_back, f"multi_n_{trial_n}_back")

    # === Phase 2: Progressive smoothing with FISTA (200-700s) ===
    print(f"\n=== Phase 2: FISTA progressive (t={time.monotonic() - t_start:.1f}s) ===")
    n = len(best_seq)
    c_cur = _full_convolve(best_seq, best_seq)
    max_c = float(c_cur.max())

    # FISTA with decreasing beta
    beta_fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for i, bf in enumerate(beta_fracs):
        if time.monotonic() > deadline - 700:
            break
        beta = max_c * bf
        remaining = deadline - 700 - time.monotonic()
        max_iter = min(8000, max(500, int(remaining / 0.002)))
        a_new, sc_new = _fista_smoothed(best_seq, beta, n,
                                        max_iter=max_iter, deadline=deadline - 700)
        try_update(a_new, f"fista_b{i}")
        # Update max_c estimate
        c_new = _full_convolve(best_seq, best_seq)
        max_c = float(c_new.max())

    # === Phase 3: L-BFGS fine polishing (700-1100s) ===
    print(f"\n=== Phase 3: L-BFGS fine (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 400:
        c_cur = _full_convolve(best_seq, best_seq)
        max_c = float(c_cur.max())
        lbfgs_betas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        for i, bf in enumerate(lbfgs_betas):
            if time.monotonic() > deadline - 400:
                break
            beta = max_c * bf
            remaining = deadline - 400 - time.monotonic()
            max_iter = min(5000, max(300, int(remaining / 0.003)))
            a_new = _lbfgs_softmax_run(best_seq, beta, n,
                                       max_iter=max_iter, deadline=deadline - 400)
            try_update(a_new, f"lbfgs_fine_{i}")

    # === Phase 4: Active set refinement (1100-1400s) ===
    print(f"\n=== Phase 4: Active set QP (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        a_as, sc_as = _active_set_qp(best_seq, n,
                                     max_iter=15, deadline=deadline - 200)
        try_update(a_as, "active_set_qp")

        # Second round of active set with tighter tolerance
        if time.monotonic() < deadline - 150:
            c_cur = _full_convolve(best_seq, best_seq)
            active = c_cur >= (c_cur.max() - 1e-5 * c_cur.max())
            print(f"  Active set size: {active.sum()} / {len(c_cur)}")
            a_as2, sc_as2 = _active_set_qp(best_seq, n,
                                           max_iter=10, deadline=deadline - 150)
            try_update(a_as2, "active_set_qp2")

    # === Phase 5: Diversification via perturbation (1400-1650s) ===
    print(f"\n=== Phase 5: Diversification (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        restart_configs = [
            ("gauss_001", "gauss", 0.005, 0),
            ("gauss_003", "gauss", 0.01, 0),
            ("fourier_3", "fourier", 0, 3),
            ("fourier_5", "fourier", 0, 5),
            ("fourier_8", "fourier", 0, 8),
            ("swap_5", "swap", 0, 5),
            ("swap_15", "swap", 0, 15),
        ]

        for name, ptype, sigma, modes in restart_configs:
            if time.monotonic() > deadline - 80:
                break

            if ptype == "gauss":
                p_start = _perturb_gaussian(best_seq, sigma, rng)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, rng, n_modes=modes)
            elif ptype == "swap":
                p_start = _perturb_swap(best_seq, rng, n_swaps=modes)
            else:
                continue

            sc_start = _score(p_start)
            if sc_start < best_score_val + 0.01:
                # Quick FISTA polish
                c_s = _full_convolve(p_start, p_start)
                beta_s = float(c_s.max()) * 0.001
                a_r, _ = _fista_smoothed(p_start, beta_s, n,
                                         max_iter=2000, deadline=deadline - 70)
                try_update(a_r, f"restart_{name}")

                # Quick L-BFGS
                if time.monotonic() < deadline - 65:
                    a_r2 = _lbfgs_softmax_run(a_r, beta_s, n,
                                              max_iter=1000, deadline=deadline - 60)
                    try_update(a_r2, f"restart_{name}_lbfgs")

    # === Phase 6: Final aggressive polish (1650-1775s) ===
    print(f"\n=== Phase 6: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        for rnd in range(5):
            if time.monotonic() > deadline - 5:
                break

            # Tiny-beta L-BFGS
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-7
            a_f = _lbfgs_softmax_run(best_seq, beta, n, max_iter=3000,
                                     deadline=deadline - 5)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight FISTA
            if time.monotonic() < deadline - 5:
                a_s, _ = _fista_smoothed(best_seq, beta, n,
                                         max_iter=5000, deadline=deadline - 5)
                try_update(a_s, f"final_fista_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
