import numpy as np
import time
import scipy.optimize
import scipy.sparse


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_simplex(y, w):
    n = len(y)
    a_rev = y[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_logspace(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _lp_direction(f, max_lp_time=5.0):
    """Compute LP-based direction: max sum(g) s.t. f*g <= M, g >= 0.
    Returns normalized g (sum=1) or None if LP fails."""
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    if M < 1e-30:
        return None
    
    # Build sparse constraint matrix A: (2n-1) x n
    # A[k, i] = f[k-i] if 0 <= k-i <= n-1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # For column i: A[k, i] = f[k-i] where 0 <= k-i <= n-1
        # So k ranges from i to i+n-1
        k_start = i
        k_end = i + n - 1
        k_range = np.arange(k_start, k_end + 1)
        valid = k_range <= (2 * n - 2)
        k_range = k_range[valid]
        f_indices = k_range - i  # these are 0..n-1
        rows.extend(k_range.tolist())
        cols.extend([i] * len(k_range))
        vals.extend(f[f_indices].tolist())
    
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Dual: min M * sum(y) s.t. A^T y >= 1, y >= 0
    # In scipy form: min c^T y s.t. A_ub y <= b_ub, y >= 0
    # A_ub = -A^T, b_ub = -1
    A_T = A.T.tocsr()
    c_dual = M * np.ones(2 * n - 1)
    
    t_lp_start = time.monotonic()
    try:
        result = scipy.optimize.linprog(
            c=c_dual,
            A_ub=-A_T,
            b_ub=-np.ones(n),
            bounds=[(0, None)] * (2 * n - 1),
            method='highs',
            options={'time_limit': max_lp_time, 'presolve': True}
        )
    except Exception:
        return None
    
    if time.monotonic() - t_lp_start > max_lp_time * 1.5:
        return None
    
    if not result.success:
        return None
    
    y = result.x
    # Recover primal g from complementary slackness
    # A^T y >= 1: if (A^T y)[i] > 1, then g[i] = 0
    # If (A^T y)[i] == 1, g[i] can be positive
    Aty = A.T @ y
    tol = 1e-6 * max(1.0, np.max(np.abs(Aty)))
    active = Aty <= (1.0 + tol)
    if not np.any(active):
        return None
    
    # For active constraints, we need to determine g values
    # Use the fact that at optimality, for active constraints:
    # sum_k A[k,i] * y[k] = 1
    # We can't directly recover g from y alone without more work
    # Instead, use y as a "weighting" for a subgradient-like direction
    
    # Alternative: use the dual solution to construct a direction
    # The subgradient of the LP value function gives us a direction
    # g is supported on coordinates where (A^T y)[i] ≈ 1
    g = np.zeros(n)
    g[active] = 1.0
    
    # Refine: use the weights from the dual to modulate
    # For each active coordinate i, the "slack" is (Aty[i] - 1)
    # We want to give more weight to coordinates where the constraint is tightest
    # Actually, let's just use uniform on active set for simplicity
    s = g.sum()
    if s < 1e-15:
        return None
    g = g / s
    return g


def _lp_direction_approx(f, top_k=50, n_iter=20):
    """Fast approximation of LP direction using iterative water-filling on top peaks."""
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    if M < 1e-30:
        return None
    
    # Start with uniform g
    g = np.ones(n) / n
    
    for it in range(n_iter):
        # Compute f * g
        fg = _full_convolve(f, g)
        # Scale g so that max(f*g) <= M
        fg_max = float(fg.max())
        if fg_max < 1e-30:
            break
        scale = M / fg_max
        # Multiply g by scale (this maintains sum(g) scaling)
        # But we want to maximize sum(g), so we should increase g where possible
        # Better: find the bottleneck and adjust
        # Simple approach: multiply g by min(1, M/fg_max)
        # This reduces g where it's too large
        
        # Actually, let's do: g_new = g * (M / max(f*g)) if max > M
        # This is a multiplicative update
        if fg_max > M * (1 + 1e-10):
            g = g * (M / fg_max)
        # Renormalize
        s = g.sum()
        if s < 1e-30:
            break
        g = g / s
    
    return g


def _fista_smoothed(a_init, beta, deadline, max_iter, lr_scale=1.0):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = lr_scale / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    stagnant = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
                stagnant = 0
            else:
                stagnant += 1
            if stagnant > 10:
                step *= 0.8
                stagnant = 0

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=5000, lr_decay_factor=0.95, grad_clip=1.0):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve_count > 80 and t > 500:
                current_lr = max(0.0003, current_lr * 0.92)
                no_improve_count = 0

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _spectral_flatten(a_init, n_iter=300, deadline=None):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        # Soft flattening: blend current and flat
        blend = 0.5 + 0.3 * (it / max(1, n_iter))
        target = blend * avg_mag + (1 - blend) * mag
        A_flat = A / (mag + 1e-30) * target
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _subgradient_steps(a_init, n_steps=3000, deadline=None):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    step_size = 0.3

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Use top-k peaks with quadratic weighting
        top_k = min(30, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        w_vals = c[top_idx] - cm + 1e-10
        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_vals ** 2
        w_sum = w_vec.sum()
        if w_sum < 1e-30:
            w_vec[int(np.argmax(c))] = 1.0
        else:
            w_vec = w_vec / w_sum

        grad = _compute_grad_simplex(a, w_vec)
        a_new = _proj_simplex(a - step_size * grad)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.9
            if step_size < 0.001:
                step_size = 0.3
                # Try a random perturbation to escape
                a_pert = a + 0.01 * np.random.randn(n)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    z = z - z.mean()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.4) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lp_iteration(f, t_step, deadline, max_lp_time=3.0):
    """One LP-based iteration: compute direction, update f."""
    n = len(f)
    g = _lp_direction(f, max_lp_time=max_lp_time)
    if g is None:
        return f, False
    
    # Try different step sizes
    best_f = f
    best_sc = _score_norm(f)
    for t in [t_step, t_step * 0.5, t_step * 2.0]:
        f_try = (1 - t) * f + t * g
        f_try = np.maximum(f_try, 0)
        s = f_try.sum()
        if s < 1e-10:
            continue
        f_try = f_try / s
        sc = _score_norm(f_try)
        if sc < best_sc:
            best_sc = sc
            best_f = f_try
    
    return best_f, best_sc < _score_norm(f)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # === Phase 1: FISTA with improved beta schedule on parent ===
    print("Phase 1: FISTA on n=" + str(n_main))
    phase1_end = min(deadline, t_start + 200.0)
    a_curr = a0.copy()
    c_curr = _full_convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    # Cosine beta schedule
    n_betas = 16
    beta_start = 0.05 * max_cc
    beta_end = 0.00001 * max_cc
    t_sched = np.linspace(0, 1, n_betas)
    betas_list = beta_start * (beta_end / beta_start) ** t_sched

    for i, b in enumerate(betas_list):
        if time.monotonic() > phase1_end - 3:
            break
        per_time = phase1_end - time.monotonic() - 2
        max_it = max(2000, int(per_time * 2000))
        phase_timeout = time.monotonic() + min(15.0, per_time)
        a_curr, sc = _fista_smoothed(a_curr, b, phase_timeout, max_it, lr_scale=1.0)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={b:.8f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 1: score={best_score:.10f}")

    # === Phase 2: LP-based iterations on best ===
    print("Phase 2: LP-based iterations")
    phase2_end = min(deadline, t_start + 350.0)
    a_work = best_seq.copy() / best_seq.sum()
    
    t_step = 0.15
    n_lp_iters = 0
    for it in range(50):
        if time.monotonic() > phase2_end - 5:
            break
        a_work, improved = _lp_iteration(a_work, t_step, phase2_end, max_lp_time=4.0)
        n_lp_iters += 1
        sc_lp = _score(a_work)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_work.copy()
            print(f"  LP iter {n_lp_iters}: score={sc_lp:.10f} t={time.monotonic()-t_start:.1f}s")
        if not improved and it > 5:
            t_step *= 0.8
            if t_step < 0.02:
                break

    print(f"End Phase 2: score={best_score:.10f}, LP iters={n_lp_iters}")

    # === Phase 3: Multi-length exploration ===
    print("Phase 3: Multi-length exploration")
    phase3_end = min(deadline, t_start + 550.0)
    
    # Try n=2048
    if time.monotonic() < phase3_end - 60:
        n_try = 2048
        a_try = _interp_profile(best_seq, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try} interpolated: score={sc_try:.10f}")
        
        c_try = _full_convolve(a_try, a_try)
        max_ct = float(c_try.max())
        betas_try = []
        b = 0.02 * max_ct
        while b > 0.00001 * max_ct:
            betas_try.append(b)
            b *= 0.5
        
        a_opt = a_try.copy()
        for b in betas_try[:8]:
            if time.monotonic() > phase3_end - 40:
                break
            per_t = min(20.0, phase3_end - time.monotonic() - 10)
            a_opt, sc8 = _fista_smoothed(a_opt, b, time.monotonic() + per_t, 5000)
            a_opt = a_opt / a_opt.sum()
            sc8_full = _score(a_opt)
            if sc8_full < best_score:
                best_score = sc8_full
                best_seq = a_opt.copy()
                print(f"  n={n_try} FISTA: score={sc8_full:.10f} t={time.monotonic()-t_start:.1f}s")
        
        # Adam on n=2048
        if time.monotonic() < phase3_end - 20:
            c_2048 = _full_convolve(a_opt, a_opt)
            max_c2 = float(c_2048.max())
            betas_fine = [0.001 * max_c2, 0.0003 * max_c2, 0.0001 * max_c2]
            a_opt, _ = _adam_logspace(a_opt, time.monotonic() + min(30.0, phase3_end - time.monotonic() - 10),
                                       betas_fine, lr_init=0.01, max_iter_total=50000)
            sc_full = _score(a_opt)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_opt.copy()
                print(f"  n={n_try} Adam: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Try n=8192
    if time.monotonic() < phase3_end - 40:
        n_try = 8192
        a_try = _interp_profile(best_seq, n_try)
        sc_try = _score(a_try)
        print(f"  n={n_try} interpolated: score={sc_try:.10f}")
        
        c_try = _full_convolve(a_try, a_try)
        max_ct = float(c_try.max())
        betas_try = []
        b = 0.01 * max_ct
        while b > 0.00001 * max_ct:
            betas_try.append(b)
            b *= 0.5
        
        a_opt = a_try.copy()
        for b in betas_try[:6]:
            if time.monotonic() > phase3_end - 20:
                break
            per_t = min(15.0, phase3_end - time.monotonic() - 10)
            a_opt, sc8 = _fista_smoothed(a_opt, b, time.monotonic() + per_t, 3000)
            a_opt = a_opt / a_opt.sum()
            sc8_full = _score(a_opt)
            if sc8_full < best_score:
                best_score = sc8_full
                best_seq = a_opt.copy()
                print(f"  n={n_try} FISTA: score={sc8_full:.10f} t={time.monotonic()-t_start:.1f}s")

    n_best = len(best_seq)
    print(f"End Phase 3: best n={n_best}, score={best_score:.10f}")

    # === Phase 4: Adam fine-tuning on best ===
    print(f"Phase 4: Adam fine-tuning on n={n_best}")
    phase4_end = min(deadline, t_start + 900.0)
    a_work = best_seq.copy() / best_seq.sum()
    
    if time.monotonic() < phase4_end - 10:
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())
        betas_fine = []
        b = 0.0003 * max_cw
        while b > 0.000001 * max_cw:
            betas_fine.append(b)
            b *= 0.5
        if len(betas_fine) > 15:
            betas_fine = betas_fine[:15]

        per_t = phase4_end - time.monotonic() - 5
        a_opt, sc_opt = _adam_logspace(a_work, time.monotonic() + per_t, betas_fine,
                                        lr_init=0.008, max_iter_total=500000,
                                        lr_decay_every=8000, lr_decay_factor=0.94)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Adam fine: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-restart with diverse initializations ===
    print("Phase 5: Multi-restart diversification")
    phase5_end = min(deadline, t_start + 1250.0)
    a_work = best_seq.copy() / best_seq.sum()
    
    if time.monotonic() < phase5_end - 30:
        n_restarts = 10
        per_restart = (phase5_end - time.monotonic() - 20) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break

            r = rng.random()
            if r < 0.25:
                p_start = _perturb(a_work, 0.015 + 0.005 * i, rng)
            elif r < 0.4:
                sz = n_best // 512
                shift = int(rng.integers(1, max(2, sz + 1)))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            elif r < 0.55:
                p_start = _perturb(a_work, 0.04, rng)
            elif r < 0.7:
                p_tmp = _perturb(a_work, 0.02, rng)
                p_start, _ = _spectral_flatten(p_tmp, n_iter=80)
            elif r < 0.85:
                p_struct = _make_profile(n_best, ["asymmetric", "two_hump", "cosine", "tapered"][i % 4])
                mix = 0.2 + 0.05 * i
                p_start = mix * p_struct + (1 - mix) * a_work
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                # LP-based perturbation
                g_lp = _lp_direction(a_work, max_lp_time=2.0)
                if g_lp is not None:
                    p_start = 0.7 * a_work + 0.3 * g_lp
                    p_start = np.maximum(p_start, 1e-10)
                    p_start = p_start / p_start.sum()
                else:
                    p_start = _perturb(a_work, 0.03, rng)

            # Quick FISTA
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            beta_fast = 0.001 * max_cs
            t_fista = min(5.0, per_restart * 0.3)
            p_opt, _ = _fista_smoothed(p_start, beta_fast, time.monotonic() + t_fista, 3000)
            p_opt = p_opt / p_opt.sum()

            # Adam
            if time.monotonic() < phase5_end - 15:
                betas_r = [0.0003 * max_cs, 0.0001 * max_cs, 0.00003 * max_cs]
                per_t_r = phase5_end - time.monotonic() - 10
                p_opt, sc_r = _adam_logspace(p_opt, time.monotonic() + per_t_r,
                                              betas_r, lr_init=0.006, max_iter_total=60000)
                sc_r_full = _score(p_opt)
                if sc_r_full < best_score:
                    best_score = sc_r_full
                    best_seq = p_opt.copy()
                    n_best = len(best_seq)
                    print(f"  Restart {i}: score={sc_r_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Subgradient + LP polish ===
    print("Phase 6: Subgradient + LP polish")
    phase6_end = min(deadline, t_start + 1500.0)
    a_work = best_seq.copy() / best_seq.sum()
    
    if time.monotonic() < phase6_end - 10:
        # Subgradient
        t_sg = min(60.0, phase6_end - time.monotonic() - 30)
        a_sg, sc_sg = _subgradient_steps(a_work, n_steps=3000, deadline=time.monotonic() + t_sg)
        sc_sg_full = _score(a_sg)
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg.copy()
            print(f"  Subgradient: score={sc_sg_full:.10f} t={time.monotonic()-t_start:.1f}s")
        
        # LP iterations
        a_work = best_seq.copy() / best_seq.sum()
        t_step = 0.1
        for it in range(20):
            if time.monotonic() > phase6_end - 5:
                break
            a_work, improved = _lp_iteration(a_work, t_step, phase6_end, max_lp_time=3.0)
            sc_lp = _score(a_work)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_work.copy()
                print(f"  LP polish {it}: score={sc_lp:.10f} t={time.monotonic()-t_start:.1f}s")
            if not improved:
                t_step *= 0.8
                if t_step < 0.01:
                    break

    # === Phase 7: Final ultra-fine Adam ===
    print("Phase 7: Ultra-fine final Adam")
    phase7_end = min(deadline, t_start + 1700.0)
    a_work = best_seq.copy() / best_seq.sum()
    
    if time.monotonic() < phase7_end - 10:
        c_pol = _full_convolve(a_work, a_work)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000005 * max_cp, 0.000002 * max_cp, 0.000001 * max_cp]
        per_t = phase7_end - time.monotonic() - 5
        a_pol, sc_pol = _adam_logspace(a_work, time.monotonic() + per_t, betas_ult,
                                        lr_init=0.002, max_iter_total=300000,
                                        lr_decay_every=10000, lr_decay_factor=0.93)
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Ultra-fine: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Spectral flatten attempt ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_sf, sc_sf = _spectral_flatten(best_seq, n_iter=1000, deadline=min(deadline - 25, time.monotonic() + 40))
        sc_sf_full = _score(a_sf)
        if sc_sf_full < best_score:
            best_score = sc_sf_full
            best_seq = a_sf.copy()
            print(f"  Spectral: score={sc_sf_full:.10f} t={time.monotonic()-t_start:.1f}s")
        
        if time.monotonic() < deadline - 15:
            c_sf = _full_convolve(a_sf, a_sf)
            max_cs = float(c_sf.max())
            betas_sf = [0.000005 * max_cs, 0.000002 * max_cs]
            per_t = deadline - time.monotonic() - 10
            if per_t > 10:
                a_sf2, _ = _adam_logspace(a_sf, time.monotonic() + per_t, betas_sf,
                                           lr_init=0.001, max_iter_total=80000)
                sc_sf2 = _score(a_sf2)
                if sc_sf2 < best_score:
                    best_score = sc_sf2
                    best_seq = a_sf2.copy()
                    print(f"  Spectral+Adam: score={sc_sf2:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            s = final.sum()
            final = final / s

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(512, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
