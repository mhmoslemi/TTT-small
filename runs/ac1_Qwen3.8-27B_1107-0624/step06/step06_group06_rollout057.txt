"""
Optimized step function search using top-k smoothed gradient descent
with Nesterov momentum, adaptive continuation, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _topk_gradient(a, c, k, N):
    """Compute gradient of top-k average of autocorrelation."""
    n = len(a)
    c_len = 2 * n - 1
    k = min(k, c_len)
    
    if k == 1:
        idx = np.argmax(c)
    else:
        idx = np.argpartition(c, -k)[-k:]
    
    w = np.zeros(c_len)
    w[idx] = 1.0 / k
    
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    
    # Cross-correlation: ifft(conj(fa) * fw)
    b1 = np.fft.irfft(np.conj(fa) * fw, N)
    # Convolution: ifft(fa * fw)
    b2 = np.fft.irfft(fa * fw, N)
    
    g = b1[:n] + b2[:n]
    return g

def _project(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _opt_topk(a_init, deadline, k_start=128, max_iters_per_k=200, lr_init=0.01):
    """Optimize using top-k smoothed projected gradient descent."""
    n = len(a_init)
    a = _project(a_init.copy())
    N = _next_pow2(3 * n)
    
    best_a = a.copy()
    best_max_c = float(_full_convolve(a, a).max())
    best_score = _score(a)
    
    # Continuation schedule
    ks = []
    k = min(k_start, 2 * n - 1)
    while k >= 1:
        ks.append(k)
        if k == 1:
            break
        k = max(1, k // 2)
    
    momentum = 0.9
    v = np.zeros(n)
    
    for k_val in ks:
        if time.monotonic() > deadline:
            break
        
        lr = lr_init * min(1.0, k_val / k_start)
        momentum = min(0.9, 0.5 + 0.4 * k_val / k_start)
        
        # Reset momentum at each k level
        v = np.zeros(n)
        
        for it in range(max_iters_per_k):
            if time.monotonic() > deadline:
                break
            
            c = _full_convolve(a, a)
            c_max = float(c.max())
            
            if c_max < best_max_c - 1e-14:
                best_max_c = c_max
                best_a = a.copy()
                best_score = _score(a)
            
            # Compute gradient
            g = _topk_gradient(a, c, k_val, N)
            
            # Project gradient onto sum=0 subspace
            g = g - np.mean(g)
            
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                break
            
            # Nesterov momentum step
            v = momentum * v + g / (gnorm + 1e-30)
            
            # Step
            a_new = a - lr * v
            
            # Project
            a_new = _project(a_new)
            
            # Check if we made progress
            c_new = _full_convolve(a_new, a_new)
            c_max_new = float(c_new.max())
            
            if c_max_new < c_max + 1e-12:
                a = a_new
            else:
                # Rollback and reduce learning rate
                v = 0.5 * v
                a = a_new  # still accept but with reduced momentum
                lr *= 0.9
            
            # Decay learning rate slowly
            lr *= 0.999
    
    # Final evaluation
    c_final = _full_convolve(best_a, best_a)
    sc_final = float(2.0 * n * c_final.max() / (best_a.sum() ** 2))
    if sc_final < best_score:
        best_score = sc_final
    
    return best_a, best_score
