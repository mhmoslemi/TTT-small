  - For `j` in `[i, i+n-1]` (i.e., `0 <= j-i < n`): `c_new[j] = c[j] + 2*(t-v)*a[j-i]` (if `j != 2i`) or `c[j] + t^2 - v^2` (if `j = 2i`)
  - For other `j`: `c_new[j] = c[j]`
- Denominator: `(S - v + t)^2`

The terms involving `t` linearly are: `c[j] + 2*(t-v)*a[j-i] = (c[j] - 2v*a[j-i]) + 2*a[j-i]*t` for `j` in the affected range, `j != 2i`.
The quadratic term is at `j = 2i`: `c[2i] - v^2 + t^2`.

So `g(t) = max(max_j (b_j + p_j * t), c_2i - v^2 + t^2) / (S - v + t)^2`

where `b_j = c[j] - 2v*a[j-i]`, `p_j = 2*a[j-i]` for affected `j`, and `b_j = c[j]`, `p_j = 0` for unaffected `j`.

The maximum of a set of linear functions and one quadratic, divided by a quadratic, is a piecewise-defined function. The minimum can be found by:
1. For each pair of "active" constraints, find the intersection point
2. At each such point, evaluate `g(t)`
3. Also check the stationary points of each individual term: `d/dt [(b_j + p_j * t) / (S-v+t)^2] = 0` gives `t* = S - v - (b_j + p_j * t*) / p_j`... this is circular. Let me solve it properly.

`h(t) = (b + p*t) / (S0 + t)^2` where `S0 = S - v`.
`h'(t) = [p*(S0+t)^2 - (b + p*t)*2*(S0+t)] / (S0+t)^4 = [p*(S0+t) - 2*(b + p*t)] / (S0+t)^3 = [p*S0 + p*t - 2b - 2p*t] / (S0+t)^3 = [p*S0 - 2b - p*t] / (S0+t)^3`

Setting to zero: `p*S0 - 2b - p*t = 0` → `t = S0 - 2b/p` (if `p > 0`).

For the quadratic term: `q(t) = (c2 - v^2 + t^2) / (S0 + t)^2`.
`q'(t) = [2t*(S0+t)^2 - (c2-v^2+t^2)*2*(S0+t)] / (S0+t)^4 = [2t*(S0+t) - 2*(c2-v^2+t^2)] / (S0+t)^3 = [2tS0 + 2t^2 - 2c2 + 2v^2 - 2t^2] / (S0+t)^3 = [2tS0 - 2(c2-v^2)] / (S0+t)^3`

Setting to zero: `t = (c2 - v^2) / S0`.

So the critical points are:

- `t_q = (c2 - v^2) / S0` for the quadratic term
- `t = 0` (boundary)

We should evaluate `g(t)` at all these points (that are `>= 0`) and take the minimum.

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses exact coordinate descent with 1D subproblem solving, gradient-based refinement,
and multi-phase search strategy.

import numpy as np
import time
import math

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

def _compute_grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: exact coordinate descent with O(n) incremental convolution updates,
combined with smoothed gradient methods (FISTA/Adam) and peak-reduction heuristics.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    eff_k = min(k, n_c)
    topk_thresh = np.partition(c, -eff_k)[-eff_k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _fista_smoothed(a_init, deadline, beta, max_iter, top_k=100):
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 16.0 * n / max(1e-10, max_c)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter_per_phase):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_per_phase:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad_a = _compute_grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad = 2.0 * n * a * (grad_a - mean_grad)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0002, current_lr * 0.95)
        m_buf *= 0.3
        v_buf *= 0.3
    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descend_exact(a, deadline, top_k=64, verbose=True):
    """
    Exact coordinate descent with incremental convolution update.
    For each coordinate i, solves the 1D subproblem exactly by evaluating
    at all critical points of the top-K active constraints.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    c = _full_convolve(a, a)
    S = 1.0
    best_score = float(2.0 * n * c.max())
    c_len = 2 * n - 1
    n_passes_done = 0
    while time.monotonic() < deadline:
        n_passes_done += 1
        improved = 0
        pass_start = time.monotonic()
        for i in range(n):
            if time.monotonic() > deadline:
                break
            v = a[i]
            if v < 1e-16:
                continue
            S0 = S - v
            j_start = i
            j_end = min(i + n, c_len)
            if j_end <= j_start:
                continue
            # d_vec[k] = a[j_start + k - i] = a[k] for k in [0, j_end - j_start)
            # Since j_start = i, d_vec[k] = a[k] for k in [0, n) but clipped
            d_len = j_end - j_start  # = n (since j_end - j_start = min(n, c_len - i))
            d_vec = a[:d_len].copy()
            c_aff = c[j_start:j_end].copy()
            # Precompute max of unaffected part
            if j_start > 0 and j_end < c_len:
                max_unaff = max(float(c[:j_start].max()), float(c[j_end:].max()))
            elif j_start > 0:
                max_unaff = float(c[:j_start].max()) if j_start > 0 else 0.0
            elif j_end < c_len:
                max_unaff = float(c[j_end:].max())
            else:
                max_unaff = 0.0
            # Find top-K indices in c_aff
            if d_len > top_k:
                topk_idx = np.argpartition(c_aff, -top_k)[-top_k:]
            else:
                topk_idx = np.arange(d_len)
            # Build candidate t values
            t_candidates = [0.0, v, v * 0.5, v * 1.5]
            for ti in topk_idx:
                j = j_start + ti
                d_val = d_vec[ti]
                if d_val < 1e-16:
                    continue
                if j == 2 * i:
                    if S0 > 1e-15:
                        tq = max(0.0, (c[j] - v * v) / S0)
                        t_candidates.extend([tq, tq * 0.7, tq * 1.3])
                else:
                    b_val = c[j] - 2.0 * v * d_val
                    p_val = 2.0 * d_val
                    if p_val > 1e-16:
                        tc = S0 - 2.0 * b_val / p_val
                        if tc >= 0 and np.isfinite(tc):
                            t_candidates.extend([tc, tc * 0.6, tc * 1.4])
            # Add some heuristic candidates
            t_candidates.append(S0 * 0.01)
            t_candidates = [max(0.0, min(tc, 2.0 * S)) for tc in t_candidates]
            # Deduplicate (rough)
            t_candidates = list(set([round(tc, 12) for tc in t_candidates]))
            # Evaluate g(t) at each candidate
            best_t = v
            best_g = best_score
            for tc in t_candidates:
                # Compute max of c_new_affected
                c_new_aff = c_aff + 2.0 * (tc - v) * d_vec
                # Fix quadratic term at j = 2i
                idx_2i = 2 * i - j_start
                if 0 <= idx_2i < d_len:
                    c_new_aff[idx_2i] = c[2 * i] - v * v + tc * tc
                max_aff = float(c_new_aff.max())
                max_c_new = max(max_aff, max_unaff)
                S_new = S0 + tc
                if S_new < 1e-15:
                    continue
                g_val = 2.0 * n * max_c_new / (S_new * S_new)
                if g_val < best_g:
                    best_g = g_val
                    best_t = tc
            if best_t != v and best_g < best_score:
                # Incremental update of c
                delta = best_t - v
                c[j_start:j_end] += 2.0 * delta * d_vec
                idx_2i = 2 * i - j_start
                if 0 <= idx_2i < c_len:
                    c[idx_2i] += delta * delta
                a[i] = best_t
                S = S0 + best_t
                best_score = best_g
                improved += 1
        if time.monotonic() > deadline:
            break
        pass_time = time.monotonic() - pass_start
        if verbose and (n_passes_done <= 5 or n_passes_done % 20 == 0):
            print(f"    CD pass {n_passes_done}: score={best_score:.10f}, improved={improved}, time={pass_time:.2f}s")
        # Early stopping if no improvement in a full pass
        if improved == 0 and n_passes_done > 5:
            # Try a few more passes in case of numerical issues
            if n_passes_done > 30:
                break
    # Renormalize
    s_final = a.sum()
    if s_final > 1e-15:
        a = a / s_final
    return a, best_score


def _peak_shaving(a, deadline, n_rounds=50):
    """
    Peak shaving: iteratively identify the peak of the convolution and
    reduce it by adjusting the contributing coordinates.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    c = _full_convolve(a, a)
    best_score = float(2.0 * n * c.max())
    for round_num in range(n_rounds):
        if time.monotonic() > deadline:
            break
        # Find the peak
        k_star = int(np.argmax(c))
        peak_val = c[k_star]
        # The contributing pairs are (i, k_star - i) for valid i
        # c[k_star] = sum_i a[i] * a[k_star - i]
        # Find the dominant contributors
        valid_i_start = max(0, k_star - n + 1)
        valid_i_end = min(n - 1, k_star)
        if valid_i_end < valid_i_start:
            continue
        idx_range = np.arange(valid_i_start, valid_i_end + 1)
        products = a[idx_range] * a[k_star - idx_range]
        # Find top contributors
        top_n_contrib = min(8, len(products))
        top_contrib_idx = np.argsort(products)[-top_n_contrib:]
        total_product = products.sum()
        if total_product < 1e-15:
            continue
        # For each top contributor pair (i, k_star-i), reduce the larger one
        for ci in top_contrib_idx:
            i1 = int(idx_range[ci])
            i2 = int(k_star - i1)
            if i1 == i2:
                # Single term: a[i]^2. Reduce a[i]
                old_val = a[i1]
                # We want to reduce a[i1]^2 by some amount
                # New value: a[i1] * (1 - fraction)
                new_val = old_val * 0.9
                a[i1] = new_val
            else:
                # Pair: reduce the larger one
                if a[i1] >= a[i2]:
                    a[i1] = a[i1] * 0.95
                else:
                    a[i2] = a[i2] * 0.95
        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a / s
            c = _full_convolve(a, a)
            sc = float(2.0 * n * c.max())
            if sc < best_score:
                best_score = sc
    return a, best_score


def _interpolate_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _structured_init(n, kind="mix"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos":
        t = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t / n) + 0.3 * np.cos(4.0 * np.pi * t / n + 0.7)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        # Mix of structures
        t = np.arange(n) / n
        a = 1.0
        a += 0.4 * np.cos(2 * np.pi * 1.0 * t)
        a += 0.2 * np.cos(2 * np.pi * 2.0 * t + 1.0)
        a += 0.1 * np.cos(2 * np.pi * 3.0 * t + 2.0)
        a *= np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "mix")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    n_target = len(best_seq)
    print(f"Target length: {n_target}")

    # Phase 1: FISTA (coarse)
    print(f"\n=== Phase 1: FISTA (t={time.monotonic()-t_start:.1f}s) ===")
    phase1_end = t_start + min(200.0, (deadline - t_start) * 0.2)
    a_work = best_seq.copy()
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())
    betas = []
    beta = 0.01 * max_cw
    while beta > 0.0001 * max_cw:
        betas.append(beta)
        beta *= 0.5
    betas = betas[:6]
    for bi, b in enumerate(betas):
        if time.monotonic() > phase1_end - 5:
            break
        per_phase = min(40.0, phase1_end - time.monotonic() - 1)
        a_work, sc = _fista_smoothed(a_work, time.monotonic() + per_phase, b,
                                      max_iter=100000, top_k=80)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  FISTA beta={b:.8f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 2: Adam log-space (medium)
    print(f"\n=== Phase 2: Adam log-space (t={time.monotonic()-t_start:.1f}s) ===")
    phase2_end = t_start + min(500.0, (deadline - t_start) * 0.45)
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())
    betas_fine = []
    beta = 0.003 * max_cw
    while beta > 0.00005 * max_cw:
        betas_fine.append(beta)
        beta *= 0.5
    betas_fine = betas_fine[:10]
    a_work, sc = _adam_logspace(a_work, phase2_end, betas_fine, lr_init=0.008,
                                 top_k=80, max_iter_per_phase=50000)
    sc_full = _score(a_work)
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_work.copy()
    print(f"  Adam: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 3: Exact coordinate descent (main polishing)
    print(f"\n=== Phase 3: Coordinate Descent (t={time.monotonic()-t_start:.1f}s) ===")
    phase3_end = deadline - 120.0  # Leave 120s for final phases
    a_cd, sc_cd = _coord_descend_exact(a_work, phase3_end, top_k=64, verbose=True)
    sc_cd_full = _score(a_cd)
    if sc_cd_full < best_score:
        best_score = sc_cd_full
        best_seq = a_cd.copy()
    print(f"  CD done: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 4: Short Adam refinement
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Adam refinement (t={time.monotonic()-t_start:.1f}s) ===")
        phase4_end = min(deadline - 60, time.monotonic() + 60)
        c_w = _full_convolve(best_seq, best_seq)
        max_cw = float(c_w.max())
        betas_ult = [0.0005 * max_cw, 0.0002 * max_cw, 0.0001 * max_cw]
        a_ult, sc_ult = _adam_logspace(best_seq, phase4_end, betas_ult,
                                        lr_init=0.003, top_k=50, max_iter_per_phase=30000)
        sc_ult_full = _score(a_ult)
        if sc_ult_full < best_score:
            best_score = sc_ult_full
            best_seq = a_ult.copy()
        print(f"  Adam ult: score={sc_ult_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Final coordinate descent polish
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 5: Final CD (t={time.monotonic()-t_start:.1f}s) ===")
        phase5_end = deadline - 20
        a_final, sc_final_cd = _coord_descend_exact(best_seq, phase5_end, top_k=48, verbose=True)
        sc_final_full = _score(a_final)
        if sc_final_full < best_score:
            best_score = sc_final_full
            best_seq = a_final.copy()
        print(f"  Final CD: score={sc_final_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq)}, time={time.monotonic()-t_start:.1f}s ===")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * (0.01 / 256)
    sc_verified = _score(final)
    print(f"VERIFIED: score={sc_verified:.10f}, n={len(final)}")
    return [float(x) for x in final]
