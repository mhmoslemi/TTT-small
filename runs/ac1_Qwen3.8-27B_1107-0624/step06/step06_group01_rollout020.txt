    # This is the "reverse convolution": h = ifft(fft(w) * conj(fft(a))) doesn't quite work
    # because of the indexing.
    #
    # Actually: h[i] = sum_j w[i+j] * a[j]
    # Let's flip w: w_rev[k] = w[len(w)-1-k] = w[2n-2-k]
    # Then h[i] = sum_j w[i+j]*a[j]
    # conv(w, a)[i] = sum_j w[j]*a[i-j]... not quite.
    #
    # Let me just compute it directly using the relationship:
    # h[i] = sum_j a[j] * w[i+j]
    # This is the cross-correlation <a, shift_i(w)> 
    # In FFT: ifft(conj(fft(a)) * fft(w)) gives sum_j a[j]*w[j-i] (correlation)
    # That's not the same.
    #
    # Alternative: h[i] = sum_j a[j] * w[i+j]
    # Let's define b[j] = a[j], c[j] = w[j]
    # h[i] = sum_j b[j] * c[i+j]
    # This is a "sliding dot product" which is related to convolution.
    # conv(b, c)[i] = sum_j b[j] * c[i-j]
    # So h[i] = conv(b, c)[i] with c replaced by c_flipped? 
    # h[i] = sum_j b[j] * c[i+j] = sum_j b[j] * c_flip[j-i] where c_flip[k] = c[-k]... 
    #
    # Simpler: just compute h as a correlation using FFT.
    # Cross-correlation of a and w: (a ⋆ w)[m] = sum_j a[j] * w[j+m]
    # This is exactly h[m]!
    # And (a ⋆ w) = ifft(conj(fft(a)) * fft(w)) -- wait, that gives sum_j a[j]*w[j-m]
    # Let me verify: ifft(fft(a) * conj(fft(w)))[m] = sum_j a[j+m] * conj(w[j])... 
    #
    # I'll just use the direct formula with a small FFT trick:
    # h[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Pad a to length 2n-1 (zero-pad on the right): ap = [a[0],...,a[n-1], 0,...,0]
    # Then h[i] = sum_j ap[j] * w[i+j] for j=0,...,n-1
    # = (ap * w)[centered]... 
    #
    # Actually, the simplest correct formula:
    # Define the cross-correlation: R[m] = sum_j a[j] * w[j-m] (for valid indices)
    # R[m] = ifft(conj(fft(a_padded)) * fft(w_padded))[m]
    # But we want h[i] = sum_j a[j] * w[i+j] = sum_j a[j] * conj(w[conj domain])... 
    #
    # Let me just do it with a straightforward FFT-based approach:
    # h[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Let's compute the convolution of a_rev and w:
    # conv(a_rev, w)[k] = sum_j a_rev[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
    # Let i = n-1-j, then j = n-1-i:
    # = sum_i a[i] * w[k-(n-1-i)] = sum_i a[i] * w[k-n+1+i]
    # So conv(a_rev, w)[k-n+1] = sum_i a[i] * w[i] ... no that's not right either.
    #
    # I'll just compute h directly using a loop over the FFT.
    # For efficiency, note that h is a "sliding sum" that can be computed as:
    # h = ifft(fft(w, N) * conj(fft(a, N)))[:n] -- let me verify this.
    #
    # ifft(fft(w) * conj(fft(a)))[m] = sum_j w[j] * conj(a[j-m]) (circular)
    # For real signals: = sum_j w[j] * a[j-m]
    # = sum_j w[j+m] * a[j] (substituting j -> j+m)
    # = h[m] (for the linear case, approximately, if we zero-pad enough)
    #
    # Yes! So h = ifft(fft(w, N) * conj(fft(a, N)))[:n] where N >= 3n-2.
    # Let me verify with the small example.
    # a = [a0, a1, a2], w = [w0, w1, w2, w3, w4] (length 5 = 2*3-1)
    # h[0] = a0*w0 + a1*w1 + a2*w2
    # h[1] = a0*w1 + a1*w2 + a2*w3
    # h[2] = a0*w2 + a1*w3 + a2*w4
    #
    # ifft(fft(w,N) * conj(fft(a,N)))[m] for N >= 7:
    # = sum_j w[j] * a[(j-m) mod N]
    # For m=0: sum_j w[j]*a[j mod N] = w0*a0 + w1*a1 + w2*a2 + w3*a0 + w4*a1 (circular!)
    # That's wrong for the linear case. We need N >= len(w) + len(a) - 1 = 5+3-1 = 7.
    # For N=7: a_padded = [a0, a1, a2, 0, 0, 0, 0], w_padded = [w0, w1, w2, w3, w4, 0, 0]
    # ifft(fft(w_p) * conj(fft(a_p)))[m] = sum_j w_p[j] * a_p[(j-m) mod 7]
    # For m=0: w0*a0 + w1*a1 + w2*a2 + w3*0 + w4*0 = h[0] ✓
    # For m=1: w0*a6 + w1*a0 + w2*a1 + w3*a2 + w4*0 = w1*a0 + w2*a1 + w3*a2 = h[1] ✓ (since a6=0 for N=7)
    # For m=2: w0*a5 + w1*a6 + w2*a0 + w3*a1 + w4*a2 = w2*a0 + w3*a1 + w4*a2 = h[2] ✓
    #
    # So the formula works: h[m] = ifft(fft(w, N) * conj(fft(a, N)))[m] for N >= 2n-1+n-1 = 3n-2.
    # Actually N >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2.
    
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(fw * np.conj(fa), N)[:n]
    return 2.0 * h


def _compute_grad_weighted_v2(a, w, n):
    """Compute gradient of weighted sum of autocorrelations using FFT."""
    # d(c[k])/da[i] = 2*a[k-i] for 0 <= k-i < n
    # grad[i] = 2 * sum_k w[k] * a[k-i] * I(0<=k-i<n)
    # = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]  (where w has length 2n-1)
    # This is the cross-correlation: h[i] = sum_j a[j] * w[i+j]
    # = ifft(fft(w, N) * conj(fft(a, N)))[i] for N >= 3n-2
    
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(fw * np.conj(fa), N)[:n]
    return 2.0 * h


def _fista_adaptive(a_init, deadline, max_iter):
    """FISTA with adaptive step size and decreasing beta (smoothing parameter)."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    mean_c = float(c_curr.sum()) / len(c_curr)
    
    # Beta schedule: start smooth, sharpen over time
    beta_start = 0.05 * max_c
    beta_end = 1e-6 * max_c
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    # Adaptive step size
    step = 1.0 / (4.0 * n * max_c)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Update beta (exponential decay)
        progress = min(1.0, it / max(1, max_iter * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Nesterov momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        # Compute smoothed gradient
        c_y = _full_convolve(y, y)
        w = _autocorr_grad_weights(y, c_y, beta)
        grad = _compute_grad_weighted_v2(y, w, n)
        
        # Backtracking line search
        x_trial = _project_simplex(y - step * grad)
        f_trial = _score_norm(x_trial)
        
        backtracks = 0
        while f_trial > best_score + 1e-10 and backtracks < 10:
            step *= 0.5
            x_trial = _project_simplex(y - step * grad)
            f_trial = _score_norm(x_trial)
            backtracks += 1
        
        # Accept if improved or first few iterations
        if f_trial <= _score_norm(a_curr) * 1.001 or it < 10:
            a_next = x_trial
        else:
            # Try smaller step
            step *= 0.7
            x_trial = _project_simplex(y - step * grad)
            a_next = x_trial
        
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
        
        if it % 50 == 0:
            sc = _score_norm(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        phase_iter = max(500, (max_iter_total // len(beta_schedule)))
        for it_in_phase in range(phase_iter):
            if time.monotonic() > deadline:
                break
            t += 1

            # Softmax parameterization
            z_shift = z - z.max()
            e = np.exp(z_shift)
            s = e.sum()
            if s < 1e-300:
                a = np.ones(n) / n
            else:
                a = e / s

            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e2 = np.exp(c_shifted)
            s2 = float(e2.sum())
            if s2 < 1e-300 or not np.isfinite(s2):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e2 / s2

            # Gradient in log-space
            # d(score)/dz[i] = d(score)/da[i] * da[i]/dz[i] = grad[i] * a[i]
            grad_a = _compute_grad_weighted_v2(a, w, n)
            grad_z = grad_a * a  # Jacobian of softmax: da_i/dz_i = a_i(1-a_i), but we use the diagonal approx

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = a  # already computed
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 10000 == 0:
                current_lr = max(0.0003, current_lr * 0.92)

    # Final evaluation
    z_shift = z - z.max()
    e = np.exp(z_shift)
    s = e.sum()
    a_final = e / s if s > 1e-300 else np.ones(n) / n
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _lp_direction(a, M, max_iter_lp=50):
    """
    Find the LP direction: maximize sum(b) s.t. (a*b)[k] <= M, b >= 0.
    Uses a simplified iterative approach (multiplicative weights) instead of full LP.
    """
    n = len(a)
    b = np.ones(n) / n  # start with uniform
    
    for it in range(max_iter_lp):
        # Compute a*b
        ab = _full_convolve(a, b)
        # Find violations
        viol = np.maximum(ab - M, 0)
        if viol.max() < 1e-12:
            break
        # Multiplicative update: reduce b where violation is high
        # grad[i] = sum_k viol[k] * a[k-i] (similar to before)
        w = viol / max(viol.sum(), 1e-30)
        grad = _compute_grad_weighted_v2(b, w, n)
        b = b * np.exp(-grad * b.sum() / max(grad.sum(), 1e-30))
        b = np.maximum(b, 1e-15)
        b = b / b.sum()
    
    return b


def _spectral_equalize(a, n_iter=20):
    """Try to equalize the power spectrum to flatten autocorrelation."""
    n = len(a)
    a_work = a.copy()
    N = _next_pow2(3 * n)
    
    for it in range(n_iter):
        fa = np.fft.rfft(a_work, N)
        # Power spectrum
        ps = np.abs(fa) ** 2
        # Target: flat spectrum (but DC component should be large for non-negative signal)
        target = np.ones_like(ps) * ps.mean()
        # Blend: keep some of the original spectrum
        alpha = 0.3
        new_ps = (1 - alpha) * ps + alpha * target
        new_ps = np.maximum(new_ps, 0)
        new_mag = np.sqrt(new_ps)
        # Preserve phase
        phase = np.angle(fa)
        new_fa = new_mag * np.exp(1j * phase)
        # IFFT
        a_new = np.fft.irfft(new_fa, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            # Check improvement
            if _score_norm(a_new) < _score_norm(a_work):
                a_work = a_new
    
    return a_work


def _make_init_ramp(n):
    """Linear ramp initialization - exploits anti-correlation between a[j] and a[n-1-j]."""
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * t  # ramps from 0.5 to 1.0
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_init_cosine(n, freq=1.0):
    """Cosine initialization with given frequency."""
    t = np.linspace(0, 1, n)
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_init_two_hump(n):
    """Two-hump profile: large on left, small on right (anti-symmetric-ish)."""
    t = np.linspace(0, 1, n)
    a = np.exp(-8.0 * (t - 0.3) ** 2) + 0.3 * np.exp(-8.0 * (t - 0.7) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_init_random_structured(n, rng):
    """Random with low-frequency structure."""
    a = np.ones(n)
    for k in range(1, 8):
        amp = 0.15 / k
        phase = rng.random() * 2 * np.pi
        a += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _local_refinement(a, deadline, n_steps=5000):
    """Local refinement using random coordinate pairs and small perturbations."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(123)
    
    step_size = 0.01
    improves_in_row = 0
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        if improves_in_row > 200:
            step_size *= 0.9
            improves_in_row = 0
        
        # Pick random pair of indices (to exploit the a[j]*a[n-1-j] structure)
        i = rng.integers(0, n)
        j_pair = n - 1 - i
        
        if i == j_pair:
            # Single coordinate
            for delta in [0.01, -0.01, 0.005, -0.005]:
                a_try = best_a.copy()
                a_try[i] = max(0, a_try[i] + delta * a_try[i])
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_try
                    improves_in_row += 1
                    break
        else:
            # Modify a pair (i, n-1-i)
            for di, dj in [(0.02, -0.02), (-0.02, 0.02), (0.01, -0.01), (-0.01, 0.01),
                          (0.03, -0.03), (-0.03, 0.03)]:
                a_try = best_a.copy()
                a_try[i] = max(0, a_try[i] * (1 + di))
                a_try[j_pair] = max(0, a_try[j_pair] * (1 + dj))
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_try
                    improves_in_row += 1
                    break
    
    return best_a, best_score


def _block_optimize(a, deadline, block_size=64):
    """Optimize blocks of coordinates in sequence."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    n_blocks = (n + block_size - 1) // block_size
    passes = 0
    
    while time.monotonic() < deadline:
        passes += 1
        for blk in range(n_blocks):
            if time.monotonic() > deadline:
                break
            start = blk * block_size
            end = min(start + block_size, n)
            
            # Get current gradient for this block
            c = _full_convolve(best_a, best_a)
            cm = float(c.max())
            w = _autocorr_grad_weights(best_a, c, 0.001 * cm)
            grad = _compute_grad_weighted_v2(best_a, w, n)
            
            # Update block with line search
            a_try = best_a.copy()
            for lr in [0.01, 0.005, 0.002, 0.001]:
                a_try[start:end] = np.maximum(0, a_try[start:end] - lr * grad[start:end])
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_try
                    break
    
    return best_a, best_score

            # Phase 1: FISTA
            if time.monotonic() < t_init_end - 5:
                t_fista = min(t_init_end - 20, time.monotonic() + time_per_init * 0.5)
                max_iter = max(500, int((t_fista - time.monotonic()) * 500))
                a_work, sc_fista = _fista_adaptive(a_work, t_fista, max_iter)
                sc_full = _score(a_work)
                print(f"    After FISTA: {sc_full:.8f}")
                if sc_full < best_score_for_n:
                    best_score_for_n = sc_full
                    best_for_n = a_work.copy()

            # Phase 2: Adam log-space
            if time.monotonic() < t_init_end - 5:
                c_curr = _full_convolve(a_work, a_work)
                max_cc = float(c_curr.max())
                betas_fine = [0.001 * max_cc, 0.0003 * max_cc, 0.0001 * max_cc, 0.00003 * max_cc]
                t_adam = min(t_init_end - 5, time.monotonic() + time_per_init * 0.4)
                a_opt, sc_opt = _adam_logspace(a_work, t_adam, betas_fine, lr_init=0.01, max_iter_total=100000)
                sc_full = _score(a_opt)
                print(f"    After Adam: {sc_full:.8f}")
                if sc_full < best_score_for_n:
                    best_score_for_n = sc_full
                    best_for_n = a_opt.copy()

            # Phase 3: Spectral equalization
            if time.monotonic() < t_init_end - 3 and best_for_n is not None:
                a_spec = _spectral_equalize(best_for_n, n_iter=30)
                sc_spec = _score(a_spec)
                if sc_spec < best_score_for_n:
                    best_score_for_n = sc_spec
                    best_for_n = a_spec.copy()
                    print(f"    After spectral: {sc_spec:.8f}")

        # Update global best
        if best_for_n is not None and best_score_for_n < best_score:
            best_score = best_score_for_n
            best_seq = best_for_n.copy()
            print(f"  New global best: n={n_target}, score={best_score:.10f}")

    # Local refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Local refinement (score={best_score:.10f}) ===")
        t_local_end = min(deadline - 10, time.monotonic() + local_time)
        a_ref, sc_ref = _local_refinement(best_seq, t_local_end, n_steps=20000)
        sc_ref_full = _score(a_ref)
        if sc_ref_full < best_score:
            best_score = sc_ref_full
            best_seq = a_ref.copy()
            print(f"  After local refinement: {sc_score:.10f}")

    # Block optimization
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Block optimization ===")
        t_block_end = min(deadline - 10, time.monotonic() + 30)
        a_blk, sc_blk = _block_optimize(best_seq, t_block_end, block_size=128)
        sc_blk_full = _score(a_blk)
        if sc_blk_full < best_score:
            best_score = sc_blk_full
            best_seq = a_blk.copy()
            print(f"  After block optimization: {sc_blk_full:.10f}")

    # Final ultra-fine Adam polish
    if best_seq is not None and time.monotonic() < deadline - 8:
        print(f"\n=== Final polish ===")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000003 * max_cp, 0.000001 * max_cp]
        t_final_end = min(deadline - 5, time.monotonic() + 30)
        a_pol, sc_pol = _adam_logspace(a_pol, t_final_end, betas_ult, lr_init=0.003, max_iter_total=50000)
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Final polish: {sc_pol_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.6f}")
    return [float(x) for x in fallback]
