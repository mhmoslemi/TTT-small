"""
Optimized step function coefficient search using spectral initialization,
L-BFGS with log-sum-exp smoothing, O(n) greedy peak-shaving local search,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _spectral_init(a, n_gs_iters=20):
    """GS-style spectral flattening: flatten |FFT| while maintaining simplex constraint."""
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    # Zero-pad to 2n-1 for linear autocorrelation
    N = 2 * n - 1
    for _ in range(n_gs_iters):
        # FFT of zero-padded sequence
        ap = np.zeros(N)
        ap[:n] = a
        F = np.fft.fft(ap)
        
        # Target: flat magnitude (geometric mean)
        mag = np.abs(F)
        target_mag = np.full(N, np.sqrt(np.mean(mag**2)))
        
        # Apply target magnitude, keep phase
        phase = np.angle(F)
        F_new = target_mag * np.exp(1j * phase)
        
        # Inverse FFT
        ap_new = np.real(np.fft.ifft(F_new))
        
        # Project to simplex: take first n, clip, normalize
        a_new = ap_new[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        a = a_new
    
    return a

        contrib = np.zeros(n)
        contrib[valid] = a[i_idx[valid]] * a[j_idx[valid]]
        
        i_star = int(np.argmax(contrib))
        
        # Find m_star: minimize the resulting max
        # Score for candidate m: max_k (conv_k - 2*delta*a[k-i_star] + 2*delta*a[k-m])
        # For small delta, approximate: max_k (conv_k + 2*delta*(a[k-m] - a[k-i_star]))
        # = max_k (conv_k - 2*delta*a[k-i_star]) + max_k (something involving a[k-m])
        # Simpler: just pick m that minimizes a[m]*a[k_star-m] (least contribution to peak)
        m_star = int(np.argmin(contrib))
        if m_star == i_star:
            # Pick the one with smallest value that's not i_star
            contrib_sorted = np.argsort(contrib)
            for m in contrib_sorted:
                if m != i_star:
                    m_star = int(m)
                    break
        
        max_delta = a[i_star] * 0.99
        
        # Compute coeff vector (change per unit delta)
        coeff = np.zeros(2 * n - 1)
        lo = m_star
        hi = min(m_star + n, 2 * n - 1)
        coeff[lo:hi] += 2.0 * a[max(0, lo - m_star):min(n, hi - m_star)]
        lo = i_star
        hi = min(i_star + n, 2 * n - 1)
        coeff[lo:hi] -= 2.0 * a[max(0, lo - i_star):min(n, hi - i_star)]
        
        # Find optimal delta via bisection on t
        pos_mask = coeff > 1e-15
        neg_mask = coeff < -1e-15
        
        if pos_mask.any() and neg_mask.any():
            # Bisection
            t_lo = peak_val * 0.95
            t_hi = peak_val
            
            for _ in range(15):
                t_mid = (t_lo + t_hi) / 2
                # Check feasibility
                ub = max_delta
                ub_vals = (t_mid - conv[pos_mask]) / coeff[pos_mask]
                ub = min(ub, float(np.min(ub_vals)))
                lb = 0.0
                lb_vals = (t_mid - conv[neg_mask]) / coeff[neg_mask]
                lb = max(lb, float(np.max(lb_vals)))
                if lb <= ub + 1e-15:
                    t_hi = t_mid
                else:
                    t_lo = t_mid
            
            t = t_hi
            ub = max_delta
            ub_vals = (t - conv[pos_mask]) / coeff[pos_mask]
            ub = min(ub, float(np.min(ub_vals)))
            lb = 0.0
            lb_vals = (t - conv[neg_mask]) / coeff[neg_mask]
            lb = max(lb, float(np.max(lb_vals)))
            
            if lb > ub:
                delta = 0.0
            else:
                delta = (lb + ub) / 2
        elif pos_mask.any():
            delta = min(max_delta, 1e-8)
        else:
            delta = max_delta
        
        if delta < 1e-14:
            stall += 1
            if stall > 50:
                break
        else:
            a[i_star] -= delta
            a[m_star] += delta
            conv += delta * coeff
            new_max = float(conv.max())
            if new_max < best_max + 1e-12:
                best_max = min(best_max, new_max)
                stall = 0
            else:
                # Undo
                a[i_star] += delta
                a[m_star] -= delta
                conv -= delta * coeff
                stall += 1
                if stall > 100:
                    break
    
    return a, best_max


def _multi_peak_shave(a, deadline, max_rounds=20, k_active=5):
    """Peak shaving that considers top-k active lags simultaneously."""
    n = len(a)
    a = a.copy()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a, _ = _peak_shave_vectorized(a, deadline, max_iters=20000, tol=1e-13)
    
    return a


def _lbfgs_schedule(a_start, deadline, min_time_remaining=100.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002, 0.0001])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.002 * (n / 2048.0) ** 1.3
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 20))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=10000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + i * 0.0005)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    lr *= 0.7
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.01 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(4, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + int(block_size * rng.uniform(0.3, 1.5)))
        z[start:end] += rng.uniform(-0.03, 0.03)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _local_pair_search(a, deadline, max_rounds=500):
    """Local search that tries pairwise adjustments guided by the subgradient."""
    n = len(a)
    a = a.copy()
    best_sc = _score(a)
    best_a = a.copy()
    conv = _full_convolve(a, a)
    
    step_frac = 0.1
    stall = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Find peak
        k_star = int(np.argmax(conv))
        
        # Compute gradient (subgradient at peak)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    grad[i] = 2.0 * a[i]
                else:
                    grad[i] = 2.0 * a[j]
        
        # Source: max gradient (contributes most to peak)
        # Destination: min gradient
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        
        if i_src == i_dst:
            second = np.argsort(grad)
            for idx in second:
                if idx != i_src:
                    i_dst = int(idx)
                    break
        
        # Try moving mass from src to dst
        max_move = a[i_src] * 0.5
        for frac in [0.1, 0.3, 0.5, 0.7]:
            delta = max_move * frac
            a_try = a.copy()
            a_try[i_src] -= delta
            a_try[i_dst] += delta
            
            if a_try[i_src] < 0 or a_try[i_dst] < 0:
                continue
            
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                a = a_try.copy()
                best_a = a.copy()
                best_sc = sc_try
                conv = _full_convolve(a, a)
                stall = 0
                break
        
        else:
            stall += 1
            if stall > 30:
                step_frac *= 0.8
                if step_frac < 0.01:
                    break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining spectral initialization, L-BFGS with
    log-sum-exp smoothing, O(n) greedy peak shaving, subgradient polishing,
    and multi-restart to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
        n_main = len(parent)
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        try_update(a0, "init_uniform")

    if best_seq is None:
        best_seq = np.ones(n_main) / n_main

    # === Phase 1: Spectral initialization ===
    print(f"\n=== Phase 1: Spectral init (t={time.monotonic() - t_start:.1f}s) ===")
    a_spec = _spectral_init(best_seq, n_gs_iters=30)
    try_update(a_spec, "spectral")

    # Use the best of parent and spectral as starting point
    a_work = best_seq.copy()

    # === Phase 2: L-BFGS coarse ===
    print(f"\n=== Phase 2: L-BFGS coarse (t={time.monotonic() - t_start:.1f}s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 200:
        a_p2 = _lbfgs_schedule(a_work, deadline, min_time_remaining=600.0)
        try_update(a_p2, "lbfgs_coarse")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak shaving (t={time.monotonic() - t_start:.1f}s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        a_ps, _ = _peak_shave(best_seq, deadline - 300, max_iters=100000)
        try_update(a_ps, "peak_shave_1")

    # === Phase 4: L-BFGS fine ===
    print(f"\n=== Phase 4: L-BFGS fine (t={time.monotonic() - t_start:.1f}s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 200:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 200:
                break
            remaining_t = deadline - 200 - time.monotonic()
            if remaining_t < 10:
                break
            max_iter = min(3000, max(100, int(remaining_t / 0.003)))
            a_new, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 5: More peak shaving (aggressive) ===
    print(f"\n=== Phase 5: Aggressive peak shaving (t={time.monotonic() - t_start:.1f}s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        a_ps2, _ = _peak_shave(best_seq, deadline - 50, max_iters=500000)
        try_update(a_ps2, "peak_shave_2")

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: Subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        a_sg, _ = _subgrad_z_polish(best_seq, deadline - 50,
                                     tol_frac=1e-4, max_steps=5000, lr_init=0.02)
        try_update(a_sg, "subgrad")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 150:
        restart_configs = [
            ("g_003", "gauss", 0.03, 0),
            ("g_007", "gauss", 0.07, 0),
            ("g_015", "gauss", 0.15, 0),
            ("f_3", "fourier", 0, 3),
            ("f_7", "fourier", 0, 7),
            ("b_4", "block", 0, 4),
            ("b_8", "block", 0, 8),
            ("b_16", "block", 0, 16),
        ]

        per_restart = max(10.0, (remaining - 100) / len(restart_configs))
        
        for name, ptype, sigma, modes in restart_configs:
            if time.monotonic() > deadline - 50:
                break

            if ptype == "gauss":
                p_start = _perturb_gaussian(best_seq, sigma, rng)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, rng, modes)
            else:
                p_start = _perturb_block(best_seq, rng, modes)

            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            # Quick L-BFGS
            rdl = min(deadline - 40, time.monotonic() + per_restart * 0.6)
            a_r = _lbfgs_schedule(p_start, rdl, min_time_remaining=5.0)
            try_update(a_r, f"rst_{name}_lbfgs")

            # Peak shave
            if time.monotonic() < deadline - 40:
                rdl2 = min(deadline - 30, time.monotonic() + per_restart * 0.4)
                a_ps, _ = _peak_shave(a_r, rdl2, max_iters=30000)
                try_update(a_ps, f"rst_{name}_ps")

    # === Phase 8: Cross-length ===
    print(f"\n=== Phase 8: Cross-length (t={time.monotonic() - t_start:.1f}s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 40:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            rdl = min(deadline - 30, time.monotonic() + 30.0)
            a_try = _lbfgs_schedule(a_interp, rdl, min_time_remaining=15.0)
            try_update(a_try, f"n_{target_n}")
            if time.monotonic() < deadline - 25:
                a_ps, _ = _peak_shave(a_try, deadline - 25, max_iters=20000)
                try_update(a_ps, f"n_{target_n}_ps")

    # === Phase 9: Final intensive peak shaving ===
    print(f"\n=== Phase 9: Final peak shaving (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 10:
        a_final_ps, _ = _peak_shave(best_seq, deadline - 5, max_iters=1000000)
        try_update(a_final_ps, "final_ps")

    # === Phase 10: Final L-BFGS with tiny beta ===
    print(f"\n=== Phase 10: Final L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta = mb * 1e-7
        max_iter = 2000
        a_f, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
        try_update(a_f, "final_lbfgs")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
