"""
Multi-phase optimizer using exact subgradient mirror descent with adaptive
step size, multi-initialization competition, multi-n exploration, and
post-processing flattening. Targets score < 1.5027.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1. Returns 2*n*max(conv)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, c, top_k=5):
    """Compute exact subgradient of 2n*max(c) using top-k active positions."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros(2 * n - 1)
    for k in k_indices:
        p[k] = 1.0 / top_k
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_soft_gradient(a, c, T):
    """Compute gradient using soft-max approximation to max."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_update(m_buf, v_buf, t, b1, b2, eps, grad):
    m_buf = b1 * m_buf + (1.0 - b1) * grad
    v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
    mh = m_buf / (1.0 - b1 ** t)
    vh = v_buf / (1.0 - b2 ** t)
    return m_buf, v_buf, mh / (np.sqrt(vh) + eps)


def _optimize_exact(a_init, deadline, top_k=5, lr=0.02, 
                    jump_interval=2000, jump_size=0.05,
                    eval_interval=200, clip_norm=0.5):
    """
    Mirror descent with exact subgradient and adaptive step size.
    Optimizes in z-space (log a) for automatic non-negativity.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr = lr
    
    while time.monotonic() < deadline:
        t += 1
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        g = _compute_gradient(a, c, top_k)
        
        # Center the gradient (project out the all-ones component for simplex optimization)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf, v_buf, update = _adam_update(m_buf, v_buf, t, b1, b2, eps, z_grad)
        
        # Clip gradient norm
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        
        z = z - current_lr * update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        # Periodic jump to escape local minima
        if t % jump_interval == 0:
            z += jump_size * np.random.standard_normal(n)
            z = z - z.mean()
            m_buf = np.zeros(n)
            v_buf = np.zeros(n)
        
        # Evaluate periodically
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            
            # Adaptive learning rate
            if patience > 5 and current_lr > lr * 0.05:
                current_lr *= 0.7
                patience = 0
                # Reset to best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    # Final evaluation
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _make_raised_cosine(n):
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = 0.5 * (1 + np.cos(x))
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2
    sigma = sigma_frac * n
    a = np.exp(-x**2 / (2 * sigma**2))
    return a / a.sum()


def _make_two_bump(n, frac=0.3):
    a = np.zeros(n)
    center1 = int(n * (1 - frac) / 2)
    center2 = int(n * (1 + frac) / 2)
    width = n // 10
    for i in range(n):
        d1 = abs(i - center1)
        d2 = abs(i - center2)
        a[i] = np.exp(-d1**2 / (2*width**2)) + np.exp(-d2**2 / (2*width**2))
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, c, top_k=5):
    n = len(a)
    k_eff = min(top_k, len(c))
    k_indices = np.argsort(c)[::-1][:k_eff]
    p = np.zeros(2 * n - 1)
    for k in k_indices:
        p[k] = 1.0 / k_eff
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_soft_gradient(a, c, T):

    c_shifted = (c - c.max()) / T

    phase3_end = t_start + (deadline - t_start) * 0.75
    if time.monotonic() < phase3_end:
        a_exact, _ = _mirror_descent(
            best_a, phase3_end - 3, use_soft=False,
            T_start=1.0, T_end=1.0,
            lr_start=0.012, lr_end=0.002,
            top_k=3, eval_interval=200,
            clip_norm=0.3, jump_interval=8000, jump_size=0.015,
            patience_limit=12)
        sc = _score(a_exact)
        print(f"Phase 3 (top_k=3): score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if sc < best_score:
            best_score = sc
            best_a = a_exact.copy()

    # Phase 4: Multi-n exploration (12% of time)
    phase4_end = t_start + (deadline - t_start) * 0.87
    if time.monotonic() < phase4_end:
        # n = 8192
        n_large = 8192
        a_large = _interp_to_length(best_a, n_large)
        half_time = (phase4_end - time.monotonic() - 5) / 2.0
        if half_time > 20:
            a_l_opt, _ = _mirror_descent(
                a_large, time.monotonic() + half_time, use_soft=False,
                T_start=1.0, T_end=1.0,
                lr_start=0.01, lr_end=0.002,
                top_k=5, eval_interval=400,
                clip_norm=0.4, jump_interval=5000, jump_size=0.02,
                patience_limit=8)
            sc = _score(a_l_opt)
            print(f"  n=8192: score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if sc < best_score:
                best_score = sc
                best_a = a_l_opt.copy()

        # n = 2048
        if time.monotonic() < phase4_end - 10:
            n_small = 2048
            a_small = _interp_to_length(best_a, n_small)
            a_s_opt, _ = _mirror_descent(
                a_small, min(phase4_end - 3, time.monotonic() + half_time),
                use_soft=False, T_start=1.0, T_end=1.0,
                lr_start=0.01, lr_end=0.002,
                top_k=5, eval_interval=200,
                clip_norm=0.4, jump_interval=5000, jump_size=0.02,
                patience_limit=8)
            sc = _score(a_s_opt)
            print(f"  n=2048: score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if sc < best_score:
                best_score = sc
                best_a = a_s_opt.copy()

    # Phase 5: Flattening + final ultra-fine polish (remaining time)
    if time.monotonic() < deadline - 12:
        # Try flattening
        a_flat = _flattening_steps(best_a.copy(), n_steps=30, reduce_frac=0.998)
        sc_flat = _score(a_flat)
        print(f"  Flattened: score={sc_flat:.10f}")
        if sc_flat < best_score:
            best_score = sc_flat
            best_a = a_flat.copy()

        # Final polish
        if time.monotonic() < deadline - 10:
            a_final, _ = _mirror_descent(
                best_a, deadline - 8, use_soft=False,
                T_start=1.0, T_end=1.0,
                lr_start=0.004, lr_end=0.0005,
                top_k=2, eval_interval=100,
                clip_norm=0.15, jump_interval=20000, jump_size=0.005,
                patience_limit=20)
            sc = _score(a_final)
            print(f"  Final polish: score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if sc < best_score:
                best_score = sc
                best_a = a_final.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
