So the constraint is: for k > 0, `sum_{i} a_i * a_{i+k} <= t`. This is a bilinear constraint (not a standard convex constraint), so CVXPY won't accept it directly.

However, we can use the **socpification** trick or reformulate. Actually, `sum a_i * a_{i+k} <= t` with `a >= 0` can be written as:
`||[a_0, ..., a_{n-1-k}]|| * ||[a_k, ..., a_{n-1}]|| <= t + sum(|a_i - a_{i+k}|)/2`... no, that's not right.

Actually, by AM-GM: `a_i * a_{i+k} <= (a_i^2 + a_{i+k}^2)/2`. So `sum a_i * a_{i+k} <= (sum a_i^2 + sum a_{i+k}^2)/2`. This gives an upper bound but it's not tight.

Let me abandon the direct convex formulation and stick with the sequential approach.

def _lp_direction_scipy(a, M, K=50):
    n = len(a)
    c = _full_convolve(a, a)
    
    # Top K active lags
    top_idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix: A[k, j] = a[k-j] if valid
    # We only need K rows
    A = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[i, j] = a[k - j]
    
    # LP: min -sum(b) s.t. A @ b <= M, b >= 0
    c_obj = -np.ones(n)
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A, b_ub=np.full(len(top_idx), M), 
                  bounds=[(0, None)]*n, method='highs')
    
    if not res.success:
        return a / a.sum()
    
    b_val = np.maximum(res.x, 0)
    s = b_val.sum()
    if s < 1e-15:
        return a / a.sum()
    return b_val / s

"""
LP-guided sequential optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set linear programming to find informed search directions,
combined with L-BFGS log-sum-exp polishing and subgradient refinement.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_tapered":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = int(0.05 * n)
        for i in range(taper):
            f = (i + 1) / (taper + 1)
            a[i] = f
            a[n - 1 - i] = f
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lp_direction(a, M, K=50):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for top K lags, b >= 0."""
    n = len(a)
    c = _full_convolve(a, a)
    top_idx = np.argsort(c)[::-1][:min(K, len(c))]

    A = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[i, j] = a[k - j]

    c_obj = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if not res.success or res.x is None:
            return a / a.sum()
        b_val = np.maximum(res.x, 0.0)
        s = b_val.sum()
        if s < 1e-15:
            return a / a.sum()
        return b_val / s
    except Exception:
        return a / a.sum()


def _lp_polish(a, deadline, max_iters=500, K_start=30, K_end=80):
    """Sequential LP-based optimization with increasing active set size."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a = a.copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        K = int(K_start + (K_end - K_start) * min(1.0, it / (max_iters * 0.3)))
        c = _full_convolve(a, a)
        M = float(c.max())

        b_dir = _lp_direction(a, M, K=K)

        sc_cur = _score(a)
        best_t = 0.0
        best_sc_t = sc_cur
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * b_dir
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc_t - 1e-15:
                best_sc_t = sc_try
                best_t = t

        if best_t == 0.0:
            break

        a = (1.0 - best_t) * a + best_t * b_dir
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()

        if it % 50 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_sc:.10f}, t={best_t:.2f}, K={K}")

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    wc = _full_convolve(w, a[::-1])
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init


def _lbfgs_schedule(a_start, deadline, betas=None):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    if betas is None:
        betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005,
                                  0.0025, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(3000, max(200, int(remaining / 0.003)))
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _subgrad_polish(a_init, deadline, max_steps=5000, tol_frac=1e-4):
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    a = a_init.copy()
    lr = 0.5 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        wc = _full_convolve(w, a[::-1])
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad = grad - grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_try = a - lr * grad
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            lr *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-14:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0
            lr = min(lr * 1.02, 0.1)
        else:
            stall += 1
            lr *= 0.9

        a = a_try
        if stall > 100:
            break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
