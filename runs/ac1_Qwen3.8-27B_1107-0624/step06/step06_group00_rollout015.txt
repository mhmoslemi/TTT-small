import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

"""
Coordinate descent + gradient hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses fast 1D coordinate descent with ternary search, combined with active-set gradient
refinement and multi-restart strategy. Operates on normalized sequences (sum=1).

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a, c=None):
    n = len(a)
    if c is None:
        c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_active_gradient(a, c, top_k=5):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _coord_descent_sweep(a, c, S, n, ters=40):
    """
    One full sweep of coordinate descent.
    a: length n, sum = S
    c: length 2n-1, convolution of a with itself
    Returns updated c (a and S modified in place).
    """
    for i in range(n):
        ai = a[i]
        
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        mod_len = k_end - k_start + 1
        
        # Unaffected max
        unaff_max = 0.0
        if k_start > 0:
            unaff_max = max(unaff_max, float(c[:k_start].max()))
        if k_end < 2 * n - 2:
            unaff_max = max(unaff_max, float(c[k_end + 1:].max()))
        
        # Modifier: mod[j] = 2*a[j] for j = 0..mod_len-1
        mod = 2.0 * a[:mod_len].copy()
        
        # k=2i handling
        has_2i = (2 * i >= k_start) and (2 * i <= k_end)
        idx_2i = 2 * i - k_start if has_2i else -1
        if has_2i:
            mod[idx_2i] = 0.0
        
        c_aff = c[k_start:k_end + 1]  # view, not copy
        
        # Search range
        lo = 0.0
        hi = max(3.0 * ai, 50.0 / n)
        
        # Ternary search
        for _ in range(ters):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            
            # F(m1)
            t1 = m1 - ai
            val1 = c_aff + t1 * mod
            if has_2i:
                val1 = val1.copy()  # need to modify idx_2i
                val1[idx_2i] = c_aff[idx_2i] - ai * ai + m1 * m1
            mx1 = max(float(val1.max()), unaff_max)
            S1 = S - ai + m1
            F1 = 2.0 * n * mx1 / (S1 * S1) if S1 > 1e-15 else np.inf
            
            # F(m2)
            t2 = m2 - ai
            val2 = c_aff + t2 * mod
            if has_2i:
                val2 = val2.copy()
                val2[idx_2i] = c_aff[idx_2i] - ai * ai + m2 * m2
            mx2 = max(float(val2.max()), unaff_max)
            S2 = S - ai + m2
            F2 = 2.0 * n * mx2 / (S2 * S2) if S2 > 1e-15 else np.inf
            
            if F1 < F2:
                hi = m2
            else:
                lo = m1
        
        x_opt = (lo + hi) / 2.0
        
        # Apply update
        delta = x_opt - ai
        if abs(delta) > 1e-12:
            c[k_start:k_end + 1] = c_aff + delta * mod
            if has_2i:
                c[2 * i] = c_aff[idx_2i] - ai * ai + x_opt * x_opt
            a[i] = x_opt
            S += delta
    
    return c
