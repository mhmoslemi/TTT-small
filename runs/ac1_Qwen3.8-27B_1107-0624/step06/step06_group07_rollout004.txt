        return _smograd(z, beta, n, topk)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _smoobj_lazy(z, beta, n, topk, cache):
    key = (beta, topk)
    if key not in cache:
        cache[key] = _smoobj(z, beta, n, topk)
    return cache[key]


def _beta_schedule(a_start, deadline, min_time_remaining, topk=1, n_betas=None):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    if topk > 1:
        betas = max_c * np.array([0.2, 0.1, 0.05, 0.03, 0.02, 0.01,
                                  0.005, 0.003, 0.002, 0.001,
                                  0.0005, 0.0003, 0.0002, 0.0001])
    else:
        betas = max_c * np.array([0.2, 0.1, 0.05, 0.03, 0.02, 0.01,
                                  0.005, 0.003, 0.002, 0.001,
                                  0.0005, 0.0003, 0.0002, 0.0001,
                                  0.00005, 0.00003, 0.00001])

    if n_betas is not None:
        betas = betas[:n_betas]

    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_iter_time = 0.002 * (n / 2048.0) ** 1.3
        max_iter = min(12000, max(200, int(remaining / (est_iter_time * 10))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter, topk)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=20000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        step = lr / (1.0 + i * 0.0002)
        z = z - step * grad
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    lr *= 0.75
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _coord_descent(a_init, deadline, max_rounds=50000):
    """Adaptive coordinate descent: pick best (i,j) pair for mass transfer."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Compute subgradient
        tol = 1e-4 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)

        # Find best coordinate to decrease (most negative grad means increasing reduces obj)
        # Actually, we want to decrease the objective. The objective increases with grad direction.
        # So we want to move in -grad direction, but maintain sum=1 and a>=0.
        # Best: decrease a where grad is most positive, increase a where grad is most negative.
        
        # Pick top few candidates for decrease and increase
        # For efficiency, sample a subset
        if rnd % 50 == 0:
            # Full scan every 50 rounds
            dec_candidates = np.argsort(grad)[-min(50, n):][::-1]
            inc_candidates = np.argsort(grad)[:min(50, n)]
        else:
            dec_candidates = np.random.choice(n, min(30, n), replace=False)
            inc_candidates = np.random.choice(n, min(30, n), replace=False)

        best_improvement = 0
        best_i = -1
        best_j = -1
        best_step = 0

        for i in dec_candidates:
            if grad[i] < 0:
                continue
            for j in inc_candidates:
                if j == i:
                    continue
                if grad[j] > 0:
                    continue
                # Transfer mass from i to j
                # a_i -> a_i - step, a_j -> a_j + step
                # Best step approx: minimize 2n*(new_max_c)/(sum)^2
                # Use quadratic approximation: step ~ a[i] * 0.1
                step = min(a[i] * 0.1, 0.01)
                
                a_try = a.copy()
                a_try[i] -= step
                a_try[j] += step
                if a_try[i] < 0:
                    step = a[i] * 0.5
                    a_try = a.copy()
                    a_try[i] -= step
                    a_try[j] += step
                
                sc_try = _score(a_try)
                improvement = best_sc - sc_try
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_j = j
                    best_step = step

        if best_i >= 0 and best_improvement > 1e-16:
            a[best_i] -= best_step
            a[best_j] += best_step
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            best_a = a.copy()
            best_sc = _score(a)
            c = _full_convolve(a, a)
            c_max = float(c.max())
        else:
            # Try a smaller random perturbation
            if rnd > 1000:
                i = int(np.random.randint(n))
                j = int(np.random.randint(n))
                if i != j:
                    step = min(a[i], 0.005) * np.random.uniform(0.1, 0.5)
                    a_try = a.copy()
                    a_try[i] -= step
                    a_try[j] += step
                    a_try = np.maximum(a_try, 0.0)
                    a_try = a_try / a_try.sum()
                    sc_try = _score(a_try)
                    if sc_try < best_sc:
                        a = a_try
                        best_a = a.copy()
                        best_sc = sc_try
                        c = _full_convolve(a, a)
                        c_max = float(c.max())

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quartic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.15 * (2.0 * t - 1.0) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _symmetrize(a):
    """Enforce palindromic symmetry."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return a_sym / a_sym.sum()


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    return a_interp / a_interp.sum()


def _perturb_fourier(a, rng, n_modes=4, amp_scale=0.008):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = amp_scale * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=6, block_frac=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(4, int(n * block_frac))
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + block_size)
        z[start:end] += rng.uniform(-0.02, 0.02)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: beta-continuation L-BFGS (top-k smoothing),
    subgradient polishing, coordinate descent, multi-profile/multi-length restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "tapered")
        print(f"No parent. Tapered n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_tapered")

    if best_seq is None:
        best_seq = _make_profile(4096, "uniform")
        best_score_val = _score(best_seq)

    # === Phase 1: Main L-BFGS with top-k=1 ===
    print(f"\n=== Phase 1: Main L-BFGS top-k=1 (t={time.monotonic()-t_start:.1f}s) ===")
    a_p1 = _beta_schedule(best_seq, deadline, min_time_remaining=400, topk=1)
    try_update(a_p1, "phase1_tk1")

    # === Phase 2: L-BFGS with top-k=5 ===
    print(f"\n=== Phase 2: L-BFGS top-k=5 (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 300:
        a_p2 = _beta_schedule(best_seq, deadline, min_time_remaining=250, topk=5)
        try_update(a_p2, "phase2_tk5")

    # === Phase 3: Symmetrize and re-optimize ===
    print(f"\n=== Phase 3: Symmetrize + re-optimize (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        a_sym = _symmetrize(best_seq)
        try_update(a_sym, "symmetrized")
        a_p3 = _beta_schedule(a_sym, deadline, min_time_remaining=150, topk=1)
        try_update(a_p3, "phase3_sym_lbfgs")

    # === Phase 4: Subgradient polishing ===
    print(f"\n=== Phase 4: Subgradient polish (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 120:
        a_sg, _ = _subgrad_z_polish(best_seq, deadline - 120,
                                    tol_frac=5e-5, max_steps=50000, lr_init=0.02)
        try_update(a_sg, "phase4_sg")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 60:
        a_cd, _ = _coord_descent(best_seq, deadline - 60, max_rounds=50000)
        try_update(a_cd, "phase5_cd")

    # === Phase 6: Very fine L-BFGS ===
    print(f"\n=== Phase 6: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 50:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([3e-5, 1e-5, 5e-6, 2e-6, 1e-6])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - 40 - time.monotonic()
            if remaining < 15:
                break
            max_iter = min(8000, max(300, int(remaining / 0.003)))
            a_new, _, _ = _lbfgs_run(best_seq, beta, n, max_iter, topk=1)
            try_update(a_new, f"fine_{i}")
            # Also try top-k=10 for smoothing
            if time.monotonic() < deadline - 40:
                a_new2, _, _ = _lbfgs_run(best_seq, beta * 2, n, max_iter, topk=10)
                try_update(a_new2, f"fine_tk10_{i}")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 80:
        restart_configs = [
            ("fourier_4", lambda: _perturb_fourier(best_seq, rng, 4, 0.006)),
            ("fourier_8", lambda: _perturb_fourier(best_seq, rng, 8, 0.004)),
            ("block_6", lambda: _perturb_block(best_seq, rng, 6, 0.04)),
            ("block_12", lambda: _perturb_block(best_seq, rng, 12, 0.03)),
            ("fourier_sym", lambda: _symmetrize(_perturb_fourier(best_seq, rng, 5, 0.008))),
        ]

        for name, gen_func in restart_configs:
            if time.monotonic() > deadline - 60:
                break
            p_start = gen_func()
            sc_start = _score(p_start)
            
            a_r = _beta_schedule(p_start, deadline, min_time_remaining=50, topk=1, n_betas=8)
            try_update(a_r, f"restart_{name}")

            if time.monotonic() < deadline - 45 and _score(a_r) < best_score_val + 0.003:
                a_r2, _ = _subgrad_z_polish(a_r, deadline - 45,
                                            tol_frac=5e-5, max_steps=5000, lr_init=0.015)
                try_update(a_r2, f"restart_{name}_sg")

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Different lengths (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 50:
        for target_n in [3072, 5120, 6144]:
            if time.monotonic() > deadline - 40:
                break
            if target_n == len(best_seq):
                continue
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _beta_schedule(a_interp, deadline, min_time_remaining=35,
                                   topk=1, n_betas=8)
            try_update(a_try, f"n_{target_n}")

    # === Phase 9: Final aggressive polish ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break

            n = len(best_seq)
            c_b = _full_convolve(best_seq, b.reverse if False else _full_convolve(best_seq, best_seq))
            mb = float(c_b.max())

            a_f, _, _ = _lbfgs_run(best_seq, mb * 5e-7, n, 5000, topk=1)
            try_update(a_f, f"final_lbfgs_{rnd}")

            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 5,
                                           tol_frac=3e-6, max_steps=10000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")

            if time.monotonic() < deadline - 5:
                a_cd, _ = _coord_descent(best_seq, deadline - 5, max_rounds=10000)
                try_update(a_cd, f"final_cd_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
