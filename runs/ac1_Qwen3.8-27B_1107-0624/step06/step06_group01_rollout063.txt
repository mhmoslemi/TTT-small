The multiplicative update rule for the "energy" minimization is:

where g_i is the gradient. After this update, we renormalize to maintain sum(a) = 1.

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses mirror descent (multiplicative updates), Adam in log-space, backtracking,
active set tracking, and multi-length exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _conv_self(a):
    return _full_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv)."""
    n = len(a)
    c = _conv_self(a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_from_weights(a, w):
    """Compute gradient of smoothed max given weights w over convolution lags.
    Uses FFT for efficiency.
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_weights(c, beta):
    """Compute softmax weights from convolution values c with temperature beta."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _mirror_descent(a, deadline, beta, max_iter, eta_init):
    """Mirror descent (multiplicative update) with adaptive step size."""
    n = len(a)
    a_curr = a.copy() / a.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    c_curr = _conv_self(a_curr)
    max_c = float(c_curr.max())
    eta = eta_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        beta_curr = beta * max_c
        w = _softmax_weights(c_curr, beta_curr)
        g = _grad_from_weights(a_curr, w)
        
        # Multiplicative update
        update = np.exp(-eta * g)
        a_next = a_curr * update
        s = a_next.sum()
        if s > 1e-30:
            a_next = a_next / s
        else:
            break
        
        # Check improvement (backtracking)
        sc = _score_norm(a_next)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_next.copy()
        
        if sc <= _score_norm(a_curr) + 1e-12:
            # Accept the step
            a_curr = a_next
            c_curr = _conv_self(a_curr)
            max_c = float(c_curr.max())
            # Decrease eta slowly
            if it % 1000 == 0:
                eta *= 0.98
        else:
            # Rejected step, decrease eta
            eta *= 0.8
        
        if it % 500 == 0:
            sc_check = _score_norm(a_curr)
            if sc_check < best_score:
                best_score = sc_check
                best_a = a_curr.copy()
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score
