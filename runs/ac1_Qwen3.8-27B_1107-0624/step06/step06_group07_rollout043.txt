def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project vector v onto the standard simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s > 0:
        w = w / s
    else:
        w = np.ones(n) / n
    return w

    p = np.fft.irfft(P, N)
    # p[j] = sum_i lam[i] * a[j-i] (circular, but N is large enough for linear)
    # We want (Q_lambda * a)[m] = sum_k lam[k] * a[k-m]
    # p[j] = sum_i lam[i] * a[j-i], so if we set j = k, p[k] = sum_i lam[i]*a[k-i]
    # That's (Q_lambda * a) evaluated at... hmm.
    # Actually p[j] = sum_{i=0}^{2n-2} lam[i] * a[(j-i) mod N]
    # Since N >= 2n and a is zero outside [0,n-1]:
    # p[j] = sum_{i: 0<=j-i<n} lam[i] * a[j-i] = sum_{m: 0<=m<n, j-m in [0,2n-2]} lam[j-m] * a[m]
    # Setting m as free: p[j] = sum_m a[m] * lam[j-m] (with zero padding)
    # We want: g[m] = sum_k lam[k] * a[k-m]
    # Let k = m + (j-m)... 
    # Actually: g[m] = sum_k lam[k] * a[k-m]. Let t = k-m, so k = t+m.
    # g[m] = sum_t lam[t+m] * a[t]
    # p[j] = sum_t lam[j-t] * a[t] (from above, with j replacing the index)
    # So g[m] = p[2n-1-m]? Let's check: p[2n-1-m] = sum_t lam[2n-1-m-t]*a[t]
    # We need lam[t+m]*a[t] = lam[2n-1-m-t]*a[t]? That requires t+m = 2n-1-m-t => 2t = 2n-1-2m => t = n-1-m
    # That's not generally true.
    
    # Let me just do it differently. Use the fact that:
    # g[m] = sum_k lam[k] * a[k-m]
    # This is the cross-correlation of (a, lam) at lag m.
    # corr(a, lam)[m] = sum_k conj(a[k]) * lam[k+m]... 
    # Or: it's the convolution of a with reversed lam.
    # conv(a, flip(lam))[m + (2n-2)] = sum_i a[i] * flip(lam)[m + (2n-2) - i]
    # flip(lam)[j] = lam[2n-2-j]
    # = sum_i a[i] * lam[2n-2-m-(2n-2)+i]... this is getting circular.
    
    # Simplest: just compute it with a small loop over lam or use the structure.
    # Actually, the fastest way: 
    # g = np.zeros(n)
    # for k in range(2*n-1):
    #     if lam[k] != 0:
    #         # a[k-m] for m=0..n-1 means: a[k], a[k-1], ..., a[k-(n-1)]
    #         # valid indices: max(0, k-(n-1)) <= i <= min(n-1, k)
    #         lo = max(0, k - (n - 1))
    #         hi = min(n - 1, k)
    #         m_lo = k - hi
    #         m_hi = k - lo
    #         g[m_lo:m_hi+1] += lam[k] * a[lo:hi+1][::-1]
    
    # For large n this loop is slow. Let's use FFT properly.
    # 
    # Define: extend a to length 2n-1 by zero-padding: A_ext[0..n-1] = a, A_ext[n..2n-2] = 0
    # Then g[m] = sum_k lam[k] * A_ext[k-m] for m = 0..n-1
    # This is the cross-correlation of lam and A_ext, sampled at lags 0..n-1.
    # cross_corr(lam, A_ext)[m] = sum_k lam[k] * A_ext[k+m] (convention varies)
    # 
    # Let me use: FFT-based circular cross-correlation.
    # corr[m] = sum_k lam[k] * A_ext[k+m] (circular)
    # We want g[m] = sum_k lam[k] * A_ext[k-m] = corr[-m] (circular)
    # 
    # Using FFT: corr = IFFT(FFT(lam) * conj(FFT(A_ext)))
    # Then g[m] = corr[(-m) mod N]
    
    N2 = _next_pow2(2 * n - 1)
    lam_ext = np.zeros(N2)
    lam_ext[:2*n-1] = lam
    A_ext = np.zeros(N2)
    A_ext[:n] = a
    
    L_f = np.fft.rfft(lam_ext, N2)
    A_f = np.fft.rfft(A_ext, N2)
    
    # Cross-correlation: corr[m] = sum_k lam[k] * A_ext[(k+m) mod N2]
    corr = np.fft.irfft(L_f * np.conj(A_f), N2)
    
    # g[m] = sum_k lam[k] * A_ext[k-m] = corr[(-m) mod N2] = corr[N2-m] (for m>0)
    # For m=0: corr[0]
    g = np.zeros(n)
    for m in range(n):
        idx = (N2 - m) % N2
        g[m] = corr[idx]
    
    # This loop is O(n) which is fine. But let's vectorize:
    indices = [0] + list(range(N2 - 1, N2 - n, -1))
    g = corr[indices]
    
    return 2.0 * g


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient of sum_k w[k]*c[k] w.r.t. a, then chain rule through softmax
    # grad_a[m] = 2 * sum_k w[k] * a[k-m]
    # grad_z = grad_a * a - (grad_a @ a) * a  (softmax Jacobian)
    # But for L-BFGS on z, we need d(obj)/dz
    # obj(z) = 2n * (c_max + beta * log(sum(exp((c-c_max)/beta))))
    # d(obj)/dz[i] = sum_j d(obj)/d(a[j]) * d(a[j])/d(z[i])
    # d(a[j])/d(z[i]) = a[j]*(delta(j,i) - a[i])
    # d(obj)/d(a[j]) = 2 * (2n) * sum_k w[k] * a[k-j] / (sum(a))^2 ... 
    
    # Actually since we work with normalized a (sum=1 via softmax), the gradient simplifies.
    # Let me just use the standard approach:
    # h[j] = 2 * sum_k w[k] * a[k-j] (gradient of sum_k w[k]*c[k] w.r.t. a[j])
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    # grad of (2n * W_smooth) w.r.t. a: 4n * a * (h - W) (in the simplex parameterization)
    # For softmax: grad_z = 4n * a * (h - W) - a * sum(a * 4n * (h - W))
    # Actually for softmax, the gradient of f(a(z)) w.r.t. z is: a * (grad_a - sum(a * grad_a))
    grad_a = 4.0 * n * a * (h - W)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return grad_z


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-1.5 * (t - 0.4) ** 2) + 0.5 * np.exp(-2.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ripple":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _minimax_iterate(a_init, deadline, max_iter=2000, tol=1e-7):
    """
    Minimax iteration:
    min_a max_k c[k] s.t. a >= 0, sum(a) = 1
    <-> max_lambda min_a sum_k lambda_k c[k]
    
    Inner: solve min_a a^T Q_lambda a on simplex (projected gradient)
    Outer: update lambda to concentrate on max c[k]
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    lam = np.ones(2 * n - 1) / (2 * n - 1)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute current convolution and max
        c = _full_convolve(a, a)
        T = float(c.max())
        
        # Update lambda: concentrate on active set
        frac = 0.005
        active = c >= T * (1.0 - frac)
        if active.sum() == 0:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(c))] = True
        lam_new = np.zeros(2 * n - 1)
        lam_new[active] = 1.0 / active.sum()
        # Smooth transition
        lam = 0.3 * lam + 0.7 * lam_new
        
        # Inner: minimize sum_k lam[k]*c[k] = a^T Q_lambda a on simplex
        # Using projected gradient with backtracking
        eta = 1.0 / (2.0 * np.linalg.norm(lam) * n) * 2.0
        for inner in range(30):
            g = _weighted_grad(a, lam, n)
            # Project gradient onto tangent space of simplex
            g_tan = g - np.mean(g)
            g_norm = np.linalg.norm(g_tan)
            if g_norm < 1e-14:
                break
            # Backtracking line search
            alpha = eta
            for bs in range(20):
                a_try = _project_simplex(a - alpha * g_tan)
                obj_try = float(np.dot(lam, _full_convolve(a_try, a_try)))
                obj_cur = float(np.dot(lam, _full_convolve(a, a)))
                if obj_try <= obj_cur - 1e-4 * alpha * g_norm ** 2:
                    a = a_try
                    break
                alpha *= 0.5
            else:
                a = _project_simplex(a - alpha * g_tan)
        
        # Evaluate
        if it % 10 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
        
        # Check convergence
        if it > 100 and it % 50 == 0:
            c_check = _full_convolve(a, a)
            if c_check.max() / c_check.mean() < 1.0 + tol:
                break
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc


def _newton_polish(a_init, deadline, max_iter=200):
    """
    Active-set Newton polishing.
    At the KKT optimum: for all i with a[i] > 0,
    sum_k lambda_k * (Q_k a)[i] = mu (constant)
    and sum_k lambda_k = 1, lambda_k >= 0, c[k] <= T, lambda_k*(c[k]-T) = 0.
    
    We iterate:
    1. Find active set
    2. Solve for lambda and a simultaneously
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        T = float(c.max())
        
        # Active set: indices where c[k] is within 1e-6 of T
        tol_active = 1e-6
        active = c >= T - tol_active
        if active.sum() == 0:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(c))] = True
        
        # Set lambda uniformly on active set (dual update)
        lam = np.zeros(2 * n - 1)
        lam[active] = 1.0 / active.sum()
        
        # Compute g = Q_lambda * a * 2
        g = _weighted_grad(a, lam, n)
        
        # Newton step: we want g[i] = mu for all i with a[i] > 0
        # Approximate: a_new = project(a - eta * (g - mean(g)))
        # With adaptive step size
        support = a > 1e-10
        if support.sum() == 0:
            break
        
        g_sup = g[support]
        mu = np.mean(g_sup)
        residual = np.abs(g_sup - mu).max()
        
        if residual < 1e-10:
            break
        
        # Step size: aim to reduce residual
        g_tan = g - np.mean(g)
        g_norm = np.linalg.norm(g_tan)
        if g_norm < 1e-14:
            break
        
        # Try a few step sizes
        best_step = 0
        best_obj = T
        for s in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_try = _project_simplex(a - s / g_norm * g_tan)
            sc_try = _score(a_try)
            if sc_try < best_obj:
                best_obj = sc_try
                best_step = s
        
        if best_step == 0:
            # Try smaller steps
            for s in [0.005, 0.001, 0.0005]:
                a_try = _project_simplex(a - s / g_norm * g_tan)
                sc_try = _score(a_try)
                if sc_try < best_obj:
                    best_obj = sc_try
                    best_step = s
        
        if best_step == 0:
            break
        
        a = _project_simplex(a - best_step / g_norm * g_tan)
        
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=10000, lr_init=0.03):
    """Subgradient in z-space with acceleration."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0
    y_moment = np.zeros_like(z)  # For heavy-ball momentum
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Heavy-ball momentum
        momentum = 0.9
        y_moment = momentum * y_moment + (1 - momentum) * grad
        eff_grad = y_moment / (1 - momentum ** (i + 1)) if i < 50 else y_moment
        
        step = lr / (1.0 + i * 0.0002)
        z = z - step * eff_grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.05
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    lr *= 0.7
                    y_moment *= 0.5
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.92
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _coordinate_descent(a_init, deadline, max_rounds=500):
    """
    Coordinate descent: cyclically try adjusting pairs of coordinates.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Try a few random pairs
        for trial in range(5):
            i = (rnd * 7919 + trial * 104729) % n
            j = (rnd * 104729 + trial * 7919) % n
            if i == j:
                j = (j + 1) % n
            
            # Try redistributing mass between i and j
            s_pair = best_a[i] + best_a[j]
            if s_pair < 1e-15:
                continue
            
            for frac in [0.3, 0.35, 0.4, 0.6, 0.65, 0.7]:
                a_try = best_a.copy()
                a_try[i] = s_pair * frac
                a_try[j] = s_pair * (1.0 - frac)
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_a = a_try
                    best_sc = sc_try
                    improved = True
        
        if not improved and rnd > 50:
            # Try single-coordinate changes
            i = (rnd * 31337) % n
            for delta in [0.01, -0.01, 0.005, -0.005]:
                a_try = best_a.copy()
                a_try[i] = max(0, a_try[i] + delta * best_sc * 1e-3)
                s = a_try.sum()
                if s > 1e-10:
                    a_try = a_try / s
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-14:
                        best_a = a_try
                        best_sc = sc_try
                        improved = True
    
    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS smoothing, minimax iteration,
    Newton active-set polishing, subgradient methods, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Using uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: L-BFGS with log-sum-exp smoothing ===
    print(f"\n=== Phase 1: L-BFGS smoothing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None:
        n = len(best_seq)
        c_cur = _full_convolve(best_seq, best_seq)
        max_c = float(c_cur.max())
        current_a = best_seq.copy()
        current_sc = _score(current_a)
        
        beta_start = max_c * 0.1
        betas = beta_start * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                       0.03125, 0.015625, 0.0078, 0.0039,
                                       0.00196, 0.00098, 0.00049, 0.000245])
        
        for i, beta in enumerate(betas):
            if time.monotonic() > deadline - 600:
                break
            remaining = deadline - 600 - time.monotonic()
            if remaining < 30:
                break
            max_iter = min(5000, max(200, int(remaining / (0.003 * (n/2048)**1.5 * 10))))
            a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
            sc = _score(a_new)
            if sc < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc
                try_update(a_new, f"lbfgs_beta_{i}")
        
        try_update(current_a, "phase1_final")

    # === Phase 2: Minimax iteration ===
    print(f"\n=== Phase 2: Minimax iteration (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_mm, sc_mm = _minimax_iterate(best_seq, deadline - 400, max_iter=3000)
        try_update(a_mm, "minimax")

    # === Phase 3: Newton active-set polishing ===
    print(f"\n=== Phase 3: Newton polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_nw, sc_nw = _newton_polish(best_seq, deadline - 250, max_iter=500)
        try_update(a_nw, "newton")

    # === Phase 4: Subgradient polishing ===
    print(f"\n=== Phase 4: Subgradient polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_z_polish(best_seq, deadline - 150,
                                         tol_frac=1e-4, max_steps=8000, lr_init=0.025)
        try_update(a_sg, "subgrad")

    # === Phase 5: Fine L-BFGS with tiny beta ===
    print(f"\n=== Phase 5: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - 60 - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(3000, max(100, int(remaining / 0.004)))
            a_new, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate descent (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_cd, sc_cd = _coordinate_descent(best_seq, deadline - 50, max_rounds=300)
        try_update(a_cd, "coord_descent")

    # === Phase 7: Multi-restart diversification ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n = len(best_seq)
        restarts = [
            ("gauss_003", 0.003),
            ("gauss_005", 0.005),
            ("gauss_008", 0.008),
            ("gauss_015", 0.015),
            ("gauss_025", 0.025),
        ]
        for name, sigma in restarts:
            if time.monotonic() > deadline - 30:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            p_start = _softmax(z)
            sc_start = _score(p_start)
            
            # Quick L-BFGS polish
            if time.monotonic() < deadline - 25:
                c_p = _full_convolve(p_start, p_start)
                mb_p = float(c_p.max())
                beta = mb_p * 0.01
                max_iter = min(1000, max(50, int((deadline - 25 - time.monotonic()) / 0.004)))
                a_r, _, _ = _lbfgs_run(p_start, beta, n, max_iter)
                try_update(a_r, f"restart_{name}")

    # === Phase 8: Final aggressive cycle ===
    print(f"\n=== Phase 8: Final polish cycle (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None:
        for rnd in range(5):
            if time.monotonic() > deadline - 3:
                break
            
            # Newton
            a_f, _ = _newton_polish(best_seq, deadline - 3, max_iter=100)
            try_update(a_f, f"final_newton_{rnd}")
            
            # Subgradient (tight)
            if time.monotonic() < deadline - 3:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 3,
                                           tol_frac=1e-5, max_steps=2000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")
            
            # Coordinate
            if time.monotonic() < deadline - 3:
                a_c, _ = _coordinate_descent(best_seq, deadline - 3, max_rounds=100)
                try_update(a_c, f"final_cd_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
