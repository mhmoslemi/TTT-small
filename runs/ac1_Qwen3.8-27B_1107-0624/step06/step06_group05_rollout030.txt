"""

2. Multiplicative coordinate descent with normalization-aware 1D optimization

4. Final L-BFGS polish

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1, returns 2n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_subgrad(a, w):
    """Compute subgradient of max_k conv(a,a)_k given weights w on the convolution.
    grad_j = 2 * sum_{k: 0<=k-j<n} w_k * a_{k-j}
    This is 2 * (cross-correlation of a with w, appropriately indexed)
    """
    n = len(a)
    # w has length 2n-1, a has length n
    # grad_j = 2 * sum_{m=0}^{n-1} a_m * w_{m+j}
    # This is like: for each j, dot(a, w[j:j+n])
    # We can compute this via FFT: it's the cross-correlation
    # Define v of length 2n-1, v[j] = sum_m a_m * w_{m+j} for j=0..n-1
    # This is the "anti-convolution" or cross-correlation
    # cross_corr[j] = sum_m a_m * w_{m+j}
    # = (a * reverse(w))[n-1+j] ... let me think
    # Actually: sum_m a_m * w_{m+j} for m=0..n-1
    # Let's define b = reverse(w) of length 2n-1... no that's messy.
    # Simpler: use FFT directly
    # v_j = sum_{m=0}^{n-1} a_m * w_{m+j}
    # Pad a to length 2n: a_pad = [a, 0, ..., 0]
    # Then v_j = sum_{m=0}^{n-1} a_pad[m] * w[m+j] for j = 0, ..., n-1
    # This is the cross-correlation of a_pad and w at lag j.
    # cross_corr[lag] = sum_m x[m] * y[m+lag]
    # = FFT approach: IFFT(FFT(x) * conj(FFT(y)))... 
    # Actually for cross-correlation: corr[k] = sum_n x[n] y[n+k]
    # DFT: corr = IFFT(FFT(x, N) * conj(FFT(y, N)))... 
    # No, that gives circular correlation. Let me just use the direct FFT approach.
    
    # v_j = sum_{m=0}^{n-1} a[m] * w[m+j]
    # Define x[m] = a[m] for m=0..n-1, 0 otherwise (length 2n-1)
    # Define y[k] = w[k] for k=0..2n-2
    # v_j = sum_m x[m] * y[m+j] = (x cross-correlate y)[j]
    # Using FFT: cross_corr = IFFT(conj(FFT(x)) * FFT(y))... 
    # Let me just use: v = irfft(conj(rfft(x, N)) * rfft(y, N))
    # This gives circular cross-correlation. For our lengths (x padded to 2n-1, y is 2n-1),
    # the linear cross-correlation equals circular if N >= len(x) + len(y) - 1 = 4n-3.
    # That's wasteful. Instead, let's use a different trick.
    
    # Alternative: v_j = sum_{m=0}^{n-1} a[m] * w[j+m]
    # Note that w[j+m] for j=0..n-1, m=0..n-1 gives indices j+m = 0..2n-2.
    # This is the "sliding dot product" of a with w.
    # We can compute it as: rev_a = a[::-1], then v = convolve(rev_a, w)[n-1 : 2n-1]
    # convolve(rev_a, w)[k] = sum_i rev_a[i] * w[k-i] = sum_i a[n-1-i] * w[k-i]
    # Let i' = n-1-i: = sum_{i'} a[i'] * w[k-n+1+i']
    # So convolve(rev_a, w)[k] = sum_{i'} a[i'] * w[k-n+1+i']
    # We want v_j = sum_m a[m] * w[j+m], so set k-n+1 = j, i.e., k = j+n-1.
    # v_j = convolve(rev_a, w)[j+n-1]
    
    a_rev = a[::-1]
    # Length of convolve output: n + (2n-1) - 1 = 3n-2
    N = _next_pow2(3 * n - 1)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    full = np.fft.irfft(fa * fw, N)
    v = full[n-1 : 2*n-1]  # length n
    return 2.0 * v


def _nesterov_pg(a_init, deadline, beta_init, beta_decay, decay_period, max_iter):
    """Nesterov accelerated projected gradient with adaptive beta."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    # Estimate Lipschitz constant
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n / beta_init  # rough estimate
    step = 1.0 / L
    
    it = 0
    no_improve_count = 0
    last_improve_score = best_score
    
    while it < max_iter:
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        # Compute smoothed gradient
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        # Compute weights with current beta
        shifted = (c_y - c_max) / beta_init
        shifted = np.clip(shifted, -50, 50)
        e = np.exp(shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum
        
        grad = _compute_subgrad(y, w)
        
        # Project gradient to tangent space
        grad = grad - np.mean(grad)
        
        # Gradient step
        a_next = _proj_simplex(y - step * grad)
        
        # Adapt step size based on function value
        sc_y = _score_norm(a_next)
        if sc_y > best_score * 1.05:
            # Bad step, reduce
            step *= 0.7
            a_next = _proj_simplex(0.5 * y + 0.5 * a_curr)
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
        it += 1
        
        # Periodic evaluation and beta update
        if it % decay_period == 0:
            beta_init *= beta_decay
            L = 4.0 * n / max(beta_init, 1e-15)
            step = 1.0 / L
            # Reset momentum
            a_prev = a_curr.copy()
            t_val = 1.0
            sc = _score_norm(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                no_improve_count = 0
                last_improve_score = sc
            else:
                no_improve_count += 1
                if no_improve_count > 3:
                    # Restart from best
                    a_curr = best_a.copy()
                    a_prev = best_a.copy()
                    t_val = 1.0
                    no_improve_count = 0
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _cd_pass(a, c, order):
    """One pass of coordinate descent. Returns updated a, c, and improvement."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-15:
        return a, c, 0.0
    a = a / S  # normalize
    S = 1.0
    c = _full_convolve(a, a)
    curr_max = float(c.max())
    
    for j in order:
        a_j = a[j]
        if a_j < 1e-15:
            # Try small positive value
            candidates = [1e-4, 1e-3]
        else:
            candidates = [0.0, 0.3 * a_j, 0.7 * a_j, 1.3 * a_j, 1.8 * a_j]
            # Also try the "balance" value
            # The optimal t satisfies: a_{k*} * S = M (approximately)
            # where k* is the argmax of the convolution
            k_star = int(np.argmax(c))
            # a_{k*-j} is the relevant value
            if 0 <= k_star - j < n:
                a_kj = a[k_star - j]
                if a_kj > 1e-15:
                    balance_t = curr_max / (2.0 * a_kj) if a_kj > 0 else 0
                    balance_t = min(balance_t, 0.5)  # cap it
                    balance_t = max(balance_t, 0.0)
                    candidates.append(balance_t)
        
        best_t = a_j
        best_val = curr_max  # current max (normalized)
        
        for t in candidates:
            t = max(0.0, t)
            # Compute new max without full conv
            delta = t - a_j
            if abs(delta) < 1e-15:
                val = curr_max
            else:
                # c'_k = c_k + 2*delta*a_{k-j} for k in [j, j+n-1]
                lo = j
                hi = min(j + n, 2 * n - 1)
                if lo < hi:
                    a_slice = a[lo - j : hi - j]  # a[0:n] effectively, but we need a[k-j] for k in [lo,hi)
                    # a[k-j] for k in [j, j+n): that's a[0], a[1], ..., a[n-1]
                    # Wait: k ranges from j to j+n-1, so k-j ranges from 0 to n-1
                    # So a[k-j] = a[0:n] = a itself!
                    # c'_k = c_k + 2*delta*a[k-j]
                    # max over k in [j, j+n-1] of (c_k + 2*delta*a[k-j])
                    c_slice = c[lo:hi]
                    new_affected_max = float(np.max(c_slice + 2.0 * delta * a))
                else:
                    new_affected_max = 0.0
                # Max outside affected range
                if lo > 0:
                    c_out_left = float(c[:lo].max())
                else:
                    c_out_left = 0.0
                if hi < 2 * n - 1:
                    c_out_right = float(c[hi:].max())
                else:
                    c_out_right = 0.0
                c_out = max(c_out_left, c_out_right)
                val = max(new_affected_max, c_out)
            
            if val < best_val:
                best_val = val
                best_t = t
        
        if best_t != a_j:
            delta = best_t - a_j
            lo = j
            hi = min(j + n, 2 * n - 1)
            if lo < hi and abs(delta) > 1e-15:
                c[lo:hi] += 2.0 * delta * a
            a[j] = best_t
            curr_max = best_val
    
    # Renormalize
    S = float(a.sum())
    if S > 1e-15:
        a = a / S
        c = c / (S * S)
    return a, c, 0.0

def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    """Adam optimizer in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp(np.clip((c - cm) / beta, -50, 50))
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad = _compute_subgrad(a, w)
        grad = grad - np.mean(grad)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -30.0, 30.0)
        z = z - z.mean()
        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 3000 == 0:
            current_lr = max(0.0001, current_lr * 0.93)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lbfgs_opt(a_init, deadline, beta):
    """L-BFGS-B refinement."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = np.clip((c - cm) / beta, -50, 50)
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = np.clip((c - cm) / beta, -50, 50)
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad_a = _compute_subgrad(a, w)
        grad_a = grad_a - np.mean(grad_a)
        return a * grad_a

    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 50)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score

    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangular":
        t = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a_work = parent.copy()
        sc0 = _score(a_work)
        print(f"Parent: n={n}, score={sc0:.10f}")
        best_seq = a_work.copy()
        best_score = sc0
    else:
        n = 4096
        a_work = _make_profile(n, "uniform")
        sc0 = _score(a_work)
        print(f"No parent, uniform n={n}, score={sc0:.10f}")
        best_seq = a_work.copy()
        best_score = sc0

    # === Phase 1: Nesterov PG with adaptive beta (40% of time) ===
    print("Phase 1: Nesterov PG adaptive beta")
    phase1_end = t_start + 0.40 * budget_s
    c_curr = _full_convolve(a_work, a_work)
    max_c0 = float(c_curr.max())
    
    beta = 0.05 * max_c0
    beta_decay = 0.55
    decay_period = 300
    
    while time.monotonic() < phase1_end - 3:
        a_work, _ = _nesterov_pg(a_work, time.monotonic() + min(30.0, phase1_end - time.monotonic() - 2),
                                  beta, beta_decay, decay_period, max_iter=100000)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            print(f"  PG: score={sc:.10f} beta={beta:.2e} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_work, a_work)
        beta = 0.4 * float(c_curr.max())
        if beta < 1e-10:
            beta = 1e-10

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Coordinate descent (30% of time) ===
    print("Phase 2: Coordinate descent")
    phase2_end = t_start + 0.70 * budget_s
    if time.monotonic() < phase2_end - 5:
        a_cd, _ = _cd_optimize(a_work, phase2_end, max_passes=50000, seed=int(seed)+1)
        sc_cd = _score(a_cd)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  CD: score={sc_cd:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  CD: score={sc_cd:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")
        # Use CD result for further optimization
        a_work = a_cd

    print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: Adam fine-tuning (15% of time) ===
    print("Phase 3: Adam fine-tuning")
    phase3_end = t_start + 0.85 * budget_s
    if time.monotonic() < phase3_end - 5:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.001 * max_cc, 0.0003 * max_cc, 0.0001 * max_cc, 0.00003 * max_cc]
        per_beta = (phase3_end - time.monotonic() - 5) / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase3_end - 5:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + max(3.0, per_beta),
                                  beta, lr_init=0.004, max_iter=200000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart exploration (on remaining time) ===
    print("Phase 4: Multi-restart")
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 30
        n_restarts = 3
        per_r = remaining / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p = _perturb(best_seq, 0.01, rng)
            elif i == 1:
                p = _perturb(best_seq, 0.03, rng)
            else:
                # Blend with a different profile
                p2 = _make_profile(n, "raised_cosine")
                p = 0.85 * best_seq + 0.15 * p2
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.001 * max_cp
            p, _ = _nesterov_pg(p, time.monotonic() + per_r * 0.4, b_r, 0.6, 200, max_iter=50000)
            p = p / p.sum()
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final CD + L-BFGS polish ===
    print("Phase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy() / best_seq.sum()
        
        # Quick CD pass
        cd_end = min(deadline - 8, time.monotonic() + 30.0)
        a_pol, _ = _cd_optimize(a_pol, cd_end, max_passes=5000, seed=int(seed)+100)
        sc = _score(a_pol)
        if sc < best_score:
            best_score = sc
            best_seq = a_pol.copy()
            print(f"  Final CD: score={sc:.10f}")
        
        # L-BFGS
        if time.monotonic() < deadline - 5:
            c_f = _full_convolve(a_pol, a_pol)
            b_f = 0.00005 * float(c_f.max())
            a_pol, _ = _lbfgs_opt(a_pol, deadline - 3.0, b_f)
            sc_f = _score(a_pol)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_pol.copy()
                print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
