Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS in log-space with geometric beta-annealing of the softmax-smoothed max.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _conv_ai(a, i):
    """conv(a,a)[i] for a given index i."""
    n = len(a)
    lo = max(0, i - n + 1)
    hi = min(n - 1, i)
    if lo > hi:
        return 0.0
    return np.dot(a[lo:hi+1], a[i-hi:i-lo+1][::-1])

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _objective_and_grad(z, beta, n):
    """Compute smoothed objective and gradient w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    w_sum = float(e.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / w_sum
    
    # Objective: 2*n*beta*log(sum exp(c/beta)) = 2*n*beta*(log(w_sum) + c_max/beta) = 2*n*beta*log(w_sum) + 2*n*c_max
    # But for optimization, we just need: 2*n*beta*log(sum exp((c-c_max)/beta)) + 2*n*c_max
    # The constant 2*n*c_max doesn't affect the gradient, so we can use:
    # f = 2*n*beta*log(sum exp((c-c_max)/beta))
    # But actually, let's just use f = 2*n*beta*log(w_sum) + 2*n*c_max for correctness
    obj = 2.0 * n * beta * np.log(w_sum) + 2.0 * n * c_max
    
    # Gradient w.r.t. a: grad_a = 4*n * conv(w, a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a_rev, N)
    conv_w_arev = np.fft.irfft(W * A_rev, N)[:len(w) + n - 1]
    grad_a = 4.0 * n * conv_w_arev[n-1:2*n-1]
    
    # Gradient w.r.t. z: grad_z[i] = grad_a[i] * a[i] - a[i] * sum_j grad_a[j]*a[j]
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z

    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    """Project v onto the simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_obj_grad_a(a, beta):
    """Compute smoothed objective value and gradient w.r.t. a (assuming sum(a)=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    w_sum = float(e.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / w_sum
    
    # Objective (constant offset removed for optimization)
    obj = 2.0 * n * beta * np.log(w_sum)
    
    # Gradient: 4*n * conv(w, a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a_rev, N)
    conv_wr = np.fft.irfft(W * Ar, N)[:len(w) + n - 1]
    grad_a = 4.0 * n * conv_wr[n-1:2*n-1]
    
    return obj, grad_a


def _smoothed_obj_grad_z(z, beta):
    """Compute smoothed objective and gradient w.r.t. z (log-space)."""
    a = _softmax(z)
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    w_sum = float(e.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / w_sum
    
    obj = 2.0 * n * beta * np.log(w_sum)
    
    a_rev = a[::-1]
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a_rev, N)
    conv_wr = np.fft.irfft(W * Ar, N)[:len(w) + n - 1]
    grad_a = 4.0 * n * conv_wr[n-1:2*n-1]
    
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    
    return obj, grad_z
