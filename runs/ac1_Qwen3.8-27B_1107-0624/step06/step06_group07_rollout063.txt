def _lp_direction(f, M, timeout_s=60):
    """Solve LP: max sum(b) s.t. (f*b)_i <= M, b >= 0.
    Returns optimal b (normalized to same sum as f) or None if failed."""
    n = len(f)
    t_start = time.monotonic()
    
    # Build constraint matrix T (2n-1) x n sparsely
    # T[i,j] = f[i-j] if 0 <= i-j < n, else 0
    rows = []
    cols = []
    data = []
    
    for j in range(n):
        # i ranges from j to min(2n-2, j+n-1)
        i_start = j
        i_end = min(2 * n - 2, j + n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        f_vals = f[i_vals - j]  # f[0], f[1], ..., f[i_end-j]
        rows.extend(i_vals.tolist())
        cols.extend([j] * len(i_vals))
        data.extend(f_vals.tolist())
    
    from scipy.sparse import coo_matrix
    T = coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsc()
    
    # LP: minimize -sum(b) s.t. T @ b <= M, b >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs')
        if res.status == 0:
            b = res.x
            s = float(b.sum())
            if s > 1e-15:
                return b / s  # Normalize to sum 1
            return None
        else:
            return None
    except Exception:
        return None

def _peak_shave(a, deadline, n_iters=500):
    """Directly shave peaks by adjusting contributing elements."""
    n = len(a)
    best = a.copy()
    best_sc = _score(a)
    current = a.copy()
    
    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(current, current)
        c_max = float(c.max())
        peak_idx = int(np.argmax(c))
        
        # Find top contributing pairs to the peak
        # (a*a)[peak_idx] = sum_j a[j]*a[peak_idx - j]
        j_range = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        contributions = current[j_range] * current[peak_idx - j_range]
        total = contributions.sum()
        
        if total < 1e-30:
            break
        
        # Reduce the top contributors
        top_k = min(10, len(j_range))
        top_indices = np.argsort(contributions)[-top_k:]
        
        # Reduce these elements proportionally
        reduce_frac = 0.01
        for j in top_indices:
            current[j] *= (1 - reduce_frac)
        
        # Renormalize
        s = float(current.sum())
        if s < 1e-15:
            break
        current = current / s
        
        sc = _score(current)
        if sc < best_sc:
            best_sc = sc
            best = current.copy()
    
    return best, best_sc

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. LP-based direction finding (solves max sum(b) s.t. (f*b)_i <= M, b>=0)
2. Adaptive top-k peak weighting in subgradient
3. Peak shaving heuristic
4. Two-scale optimization (coarse-to-fine)
5. Aggressive beta continuation with asymmetric schedule
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_topk_weighted(a, c, n, tol_frac=1e-4):
    """Subgradient with exponential weighting on near-peak values."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    active_vals = c[active]
    a_max = float(active_vals.max())
    spread = max(c_max * 1e-3, 1e-15)
    weights = np.exp((active_vals - a_max) / spread)
    w = np.zeros_like(c)
    w[active] = weights / weights.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    try:
        result = minimize(lambda z: _smoobj(z, beta, n),
                          z0,
                          jac=lambda z: _smograd(z, beta, n),
                          method='L-BFGS-B',

                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init.copy(), False, 0


def _build_conv_matrix(n, f):
    """Build sparse convolution matrix T where T[i,j] = f[i-j] if valid."""
    rows = []
    cols = []
    data = []
    for j in range(n):
        i_start = j
        i_end = min(2 * n - 2, j + n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        f_vals = f[i_vals - j]
        rows.extend(i_vals.tolist())
        cols.extend([j] * len(i_vals))
        data.extend(f_vals.tolist())
    T = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsc()
    return T


def _lp_direction(f, M, timeout_s=120):
    """Solve LP: max sum(b) s.t. (f*b)_i <= M, b >= 0. Returns normalized b or None."""
    n = len(f)
    if n > 8192:
        return None
    t0 = time.monotonic()
    try:
        T = _build_conv_matrix(n, f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout_s})
        if res.status == 0:
            b = res.x
            s = float(b.sum())
            if s > 1e-15:
                return b / s
        return None
    except Exception:
        return None

        peak_idx = int(np.argmax(c))
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(n - 1, peak_idx)
        j_range = np.arange(j_lo, j_hi + 1)
        contributions = current[j_range] * current[peak_idx - j_range]
        total = contributions.sum()
        if total < 1e-30:
            break
        top_k = min(20, len(j_range))
        top_idx_local = np.argsort(contributions)[-top_k:]
        top_idx_global = j_range[top_idx_local]
        for j in top_idx_global:
            current[j] *= (1.0 - reduce_frac)
        s = float(current.sum())
        if s < 1e-15:
            break
        current = current / s
        if i % 50 == 0:
            sc = _score(current)
            if sc < best_sc:
                best_sc = sc
                best = current.copy()
        if i % 200 == 0:
            reduce_frac *= 0.85
    sc = _score(current)
    if sc < best_sc:
        best_sc = sc
        best = current.copy()
    return best, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hamming":
        t = np.linspace(0, 1, n)
        a = 0.54 + 0.46 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quartic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.15 * (2.0 * t - 1.0) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _lbfgs_schedule(a_start, deadline, min_remaining, betas, n):
    """Run L-BFGS with a schedule of betas, adapting iteration counts."""
    current_a = a_start.copy()
    current_sc = _score(current_a)
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time_per_iter = 0.002 * (n / 2048.0) ** 1.5
        max_iter = min(6000, max(100, int(remaining / (est_time_per_iter * 20))))
        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS with log-sum-exp smoothing,
    LP-based direction finding, subgradient polishing, peak shaving,
    two-scale optimization, and multi-restart diversification.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"impr={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "hann")
        try_update(a0, "init_hann")

    n = len(best_seq)

    # === Phase 1: Two-scale coarse optimization ===
    print(f"\n=== Phase 1: Two-scale (t={time.monotonic()-t_start:.1f}s) ===")
    n_coarse = min(512, n)
    a_coarse = _interpolate_to_n(best_seq, n_coarse)
    c_coarse = _full_convolve(a_coarse, a_coarse)
    Mc = float(c_coarse.max())
    betas_c = Mc * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    a_coarse_opt, _ = _lbfgs_schedule(a_coarse, deadline, 1400, betas_c, n_coarse)
    a_up = _interpolate_to_n(a_coarse_opt, n)
    try_update(a_up, "two_scale_up")

    # === Phase 2: Quick L-BFGS on full size ===
    print(f"\n=== Phase 2: Quick L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    c0 = _full_convolve(best_seq, best_seq)
    M0 = float(c0.max())
    betas_q = M0 * np.array([0.08, 0.04, 0.02, 0.01, 0.005, 0.002])
    a_p2, _ = _lbfgs_schedule(best_seq, deadline, 1200, betas_q, n)
    try_update(a_p2, "phase2_lbfgs")

    # === Phase 3: LP-based direction finding ===
    print(f"\n=== Phase 3: LP direction (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 1000:
        a_lp, sc_lp = _lp_iteration(best_seq, deadline - 1000,
                                     n_steps=20, t_param=0.35)
        try_update(a_lp, "lp_dir")

    # === Phase 4: Fine L-BFGS ===
    print(f"\n=== Phase 4: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    c_f = _full_convolve(best_seq, best_seq)
    Mf = float(c_f.max())
    betas_fine = Mf * np.array([0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6])
    a_p4, _ = _lbfgs_schedule(best_seq, deadline, 700, betas_fine, n)
    try_update(a_p4, "fine_lbfgs")

    # === Phase 5: Subgradient polishing ===
    print(f"\n=== Phase 5: Subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 500:
        a_sg, sc_sg = _subgrad_z_polish(best_seq, deadline - 500,
                                         tol_frac=1e-4, max_steps=20000, lr_init=0.02)
        try_update(a_sg, "subgrad")

    # === Phase 6: Peak shaving ===
    print(f"\n=== Phase 6: Peak shaving (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 400:
        a_ps, sc_ps = _peak_shave(best_seq, deadline - 400, n_iters=3000)
        try_update(a_ps, "peak_shave")

    # === Phase 7: Second LP round ===
    print(f"\n=== Phase 7: LP round 2 (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 350:
        a_lp2, sc_lp2 = _lp_iteration(best_seq, deadline - 350,
                                       n_steps=12, t_param=0.25)
        try_update(a_lp2, "lp2")

    # === Phase 8: Multi-restart ===
    print(f"\n=== Phase 8: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    restart_sigs = [0.02, 0.04, 0.07, 0.12, 0.20]
    for sigma in restart_sigs:
        if time.monotonic() > deadline - 250:
            break
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        p_start = _softmax(z)
        c_r = _full_convolve(p_start, p_start)
        Mr = float(c_r.max())
        betas_r = Mr * np.array([0.005, 0.002, 0.001, 0.0005, 0.0002])
        a_r, _ = _lbfgs_schedule(p_start, deadline, 200, betas_r, n)
        try_update(a_r, f"restart_s{sigma}")

    # === Phase 9: Try different lengths ===
    print(f"\n=== Phase 9: Different lengths (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 150:
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 100:
                break
            if target_n == n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            c_t = _full_convolve(a_interp, a_interp)
            Mt = float(c_t.max())
            betas_t = Mt * np.array([0.01, 0.005, 0.002, 0.001])
            a_t, _ = _lbfgs_schedule(a_interp, deadline, 80, betas_t, target_n)
            try_update(a_t, f"n_{target_n}")

    # === Phase 10: Final aggressive polish ===
    print(f"\n=== Phase 10: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    for rnd in range(6):
        if time.monotonic() > deadline - 5:
            break
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta = mb * (5e-6 * (0.5 ** rnd))
        remaining = deadline - 5 - time.monotonic()
        max_iter = min(3000, max(100, int(remaining / 0.003)))
        a_f, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
        try_update(a_f, f"final_{rnd}")

        if time.monotonic() < deadline - 5:
            a_s, _ = _subgrad_z_polish(best_seq, deadline - 5,
                                        tol_frac=1e-5, max_steps=3000, lr_init=0.01)
            try_update(a_s, f"final_sg_{rnd}")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
