"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- No LP (saves time for more gradient iterations)
- Hybrid loss (log-sum-exp + top-k emphasis)
- Nesterov momentum
- Adaptive temperature based on progress
- Multiple initializations and n values
- Structured perturbations for exploration
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _hybrid_gradient(a, c, T, top_k, top_weight):
    """
    Compute gradient of a hybrid loss:
    L = (1-top_weight) * logsumexp(c/T) + top_weight * max(c)
    (approximately, using soft approximation for both)
    """
    n = len(a)
    c_max = c.max()
    
    # Log-sum-exp weights
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    p_soft = e / e.sum()
    
    # Top-k hard weights
    k = min(top_k, len(c))
    k_idx = np.argsort(c)[::-1][:k]
    p_hard = np.zeros_like(c)
    p_hard[k_idx] = 1.0 / k
    
    # Blend
    p = (1 - top_weight) * p_soft + top_weight * p_hard
    
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n-1:2*n-1][::-1]
    return g

def _adam_nesterov(a_init, deadline, T_start, T_end, lr_start, lr_end,
                    top_k=10, top_weight_start=0.0, top_weight_end=0.5,
                    max_iter=500000, eval_interval=200, clip_norm=0.8,
                    momentum=0.9):
    """
    Adam with Nesterov momentum in z-space, using hybrid gradient.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    v_nest = np.zeros(n)  # Nesterov velocity
    
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    no_improve_count = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        tw = top_weight_start + (top_weight_end - top_weight_start) * frac
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Ensure T doesn't go below a minimum relative to c scale
        T_eff = max(T, 1e-10 * c.max() + 1e-15)
        
        g = _hybrid_gradient(a, c, T_eff, top_k, tw)
        
        # Nesterov: look-ahead
        v_nest = momentum * v_nest + g
        z_nest = z + lr * v_nest
        a_nest = _softmax_stable(z_nest)
        c_nest = _fft_convolve(a_nest, a_nest)
        T_eff_nest = max(T, 1e-10 * c_nest.max() + 1e-15)
        g_nest = _hybrid_gradient(a_nest, c_nest, T_eff_nest, top_k, tw)
        
        W = float(np.dot(a_nest, g_nest))
        z_grad = a_nest * (g_nest - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1**t)
        vh = v_buf / (1.0 - b2**t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                no_improve_count = 0
            else:
                patience += 1
                no_improve_count += 1
            
            if patience > 8 and lr_mult > 0.1:
                lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15)) - np.log(np.maximum(best_a, 1e-15)).mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                v_nest = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _generate_u_shape(n, eps=0.3, p=1.0):
    """U-shaped initialization: more mass at edges."""
    x = np.linspace(0, 1, n)
    a = 1.0 + eps * np.abs(2*x - 1)**p
    a = np.maximum(a, 1e-6)
    return a / a.sum()

def _generate_cosine(n, freq=1, eps=0.3, p=1.0):
    """Cosine-modulated initialization."""
    x = np.arange(n)
    a = 1.0 + eps * np.cos(2*np.pi*freq*x/n)**p
    a = np.maximum(a, 1e-6)
    return a / a.sum()

def _generate_multi_cosine(n, freqs, eps=0.2):
    """Multi-frequency cosine initialization."""
    x = np.arange(n)
    a = np.ones(n)
    for f in freqs:
        a += eps * np.cos(2*np.pi*f*x/n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous version:
- No LP step (frees up time for gradient iterations)
- Nesterov-accelerated Adam in z-space
- Hybrid loss (log-sum-exp + top-k hard emphasis)
- Adaptive learning rate with progress-based reduction
- Multiple n values (4096, 8192) and structured initializations
- Extended active-set polishing with small learning rates
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_hybrid_grad(a, c, T, top_k, top_w):
    """Gradient of hybrid loss: (1-tw)*logsumexp(c/T) + tw*mean(top-k of c)."""
    n = len(a)
    c_max = c.max()
    T_eff = max(T, 1e-12 * c_max + 1e-15)

    c_shifted = (c - c_max) / T_eff
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p_soft = np.zeros_like(c)
        p_soft[np.argmax(c)] = 1.0
    else:
        p_soft = e / s

    k = min(top_k, len(c))
    k_idx = np.argsort(c)[::-1][:k]
    p_hard = np.zeros_like(c)
    p_hard[k_idx] = 1.0 / k

    p = (1.0 - top_w) * p_soft + top_w * p_hard
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_grad(a, c, active_k):
    """Gradient using exact top-k active set."""
    n = len(a)
    k = min(active_k, len(c))
    k_idx = np.argsort(c)[::-1][:k]
    p = np.zeros_like(c)
    p[k_idx] = 1.0 / k
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_nesterov(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   top_k=15, tw_start=0.0, tw_end=0.4,
                   max_iter=500000, eval_interval=200, clip_norm=0.7,
                   momentum=0.9):
    """Adam with Nesterov look-ahead in z-space, hybrid loss."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    v_nest = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        tw = tw_start + (tw_end - tw_start) * frac

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        T_eff = max(T, 1e-12 * c.max() + 1e-15)

        # Nesterov look-ahead
        v_nest = momentum * v_nest + _compute_hybrid_grad(a, c, T_eff, top_k, tw)
        z_nest = z + lr * v_nest
        a_nest = _softmax_stable(z_nest)
        c_nest = _fft_convolve(a_nest, a_nest)
        T_eff_n = max(T, 1e-12 * c_nest.max() + 1e-15)
        g_nest = _compute_hybrid_grad(a_nest, c_nest, T_eff_n, top_k, tw)

        W = float(np.dot(a_nest, g_nest))
        z_grad = a_nest * (g_nest - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)

        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:

                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                v_nest = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)

    return best_a, best_score

                   clip_norm=0.4):
    """Fine polishing with exact active set, small LR."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        g = _compute_active_grad(a, c, active_k)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)

        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _gen_u_shape(n, eps=0.3, p=1.0):
    x = np.linspace(0, 1, n)
    a = 1.0 + eps * np.abs(2.0 * x - 1.0) ** p
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def _gen_cosine(n, freq=1, eps=0.3):
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def _gen_multi_cos(n, freqs, eps=0.15):
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f in freqs:
        a += eps * np.cos(2.0 * np.pi * f * x / n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def _gen_asymmetric(n, eps=0.2):
    """Linearly tilted: more mass on one side."""
    x = np.linspace(0, 1, n)
    a = 1.0 + eps * (2.0 * x - 1.0)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: Nesterov Adam (hybrid loss) -> Active-set polish ->
    Multi-n exploration -> Perturbation restarts -> Final ultra-fine polish.
    Minimizes 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 18.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a_init = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a_init = np.ones(n) / n

    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score(a_init)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # === Phase 1: Nesterov Adam with hybrid loss on parent (0 - 400s) ===
    phase1_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase1_end:
        c = _fft_convolve(best_a, best_a)
        max_c = float(c.max())
        a1, s1 = _adam_nesterov(best_a, phase1_end,
                                 T_start=0.008 * max_c, T_end=0.00008 * max_c,
                                 lr_start=0.05, lr_end=0.004,
                                 top_k=15, tw_start=0.0, tw_end=0.35,
                                 max_iter=500000, eval_interval=200, clip_norm=0.7,
                                 momentum=0.9)
        sc = _score(a1)
        print(f"Phase 1 (Nesterov, n={n}): score={sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc < best_score:
            best_score = sc
            best_a = a1.copy()

    # === Phase 2: Active-set polish (400s - 700s) ===
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end:
        a2, s2 = _active_polish(best_a, phase2_end,
                                 lr_start=0.015, lr_end=0.001,
                                 max_iter=300000, active_k=5, eval_interval=100,
                                 clip_norm=0.4)
        sc = _score(a2)
        print(f"Phase 2 (active polish, n={len(best_a)}): score={sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc < best_score:
            best_score = sc
            best_a = a2.copy()

    # === Phase 3: Try structured initializations (700s - 1000s) ===
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end:
        inits = [
            ("U-shape", _gen_u_shape(n, 0.3, 1.0)),
            ("asym", _gen_asymmetric(n, 0.25)),
            ("cos1", _gen_cosine(n, 1, 0.25)),
            ("multi-cos", _gen_multi_cos(n, [1, 2, 3], 0.12)),
        ]
        time_per = max(40.0, (phase3_end - time.monotonic()) / len(inits))
        for name, a_i in inits:
            if time.monotonic() > phase3_end - 40:
                break
            c_i = _fft_convolve(a_i, a_i)
            max_ci = float(c_i.max())
            a_opt, s_opt = _adam_nesterov(a_i, min(phase3_end - 10, time.monotonic() + time_per),
                                           T_start=0.01 * max_ci, T_end=0.00005 * max_ci,
                                           lr_start=0.04, lr_end=0.003,
                                           top_k=15, tw_start=0.1, tw_end=0.4,
                                           max_iter=200000, eval_interval=200, clip_norm=0.7,
                                           momentum=0.9)
            sc = _score(a_opt)
            print(f"  Phase 3 ({name}, n={n}): score={sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
            # Quick active polish on the result
            if time.monotonic() < phase3_end - 20:
                a_pol, s_pol = _active_polish(a_opt, min(phase3_end - 5, time.monotonic() + 15),
                                               lr_start=0.01, lr_end=0.001,
                                               max_iter=50000, active_k=3, eval_interval=100,
                                               clip_norm=0.3)
                sc_pol = _score(a_pol)
                if sc_pol < best_score:
                    best_score = sc_pol
                    best_a = a_pol.copy()

    # === Phase 4: Try n=8192 (1000s - 1400s) ===
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end:
        n_large = 8192
        a_large = _interp_to_length(best_a, n_large)
        c_l = _fft_convolve(a_large, a_large)
        max_cl = float(c_l.max())
        a_l_opt, s_l = _adam_nesterov(a_large, phase4_end,
                                        T_start=0.005 * max_cl, T_end=0.00003 * max_cl,
                                        lr_start=0.03, lr_end=0.002,
                                        top_k=20, tw_start=0.1, tw_end=0.4,
                                        max_iter=300000, eval_interval=200, clip_norm=0.7,
                                        momentum=0.9)
        sc_l = _score(a_l_opt)
        print(f"Phase 4 (n=8192): score={sc_l:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc_l < best_score:
            best_score = sc_l
            best_a = a_l_opt.copy()

        # Active polish on n=8192 result
        if time.monotonic() < phase4_end - 30 and sc_l < 1.6:
            a_l_pol, s_lpol = _active_polish(a_l_opt, phase4_end,
                                              lr_start=0.008, lr_end=0.0005,
                                              max_iter=150000, active_k=4, eval_interval=100,
                                              clip_norm=0.3)
            sc_lpol = _score(a_l_pol)
            print(f"Phase 4 (n=8192 polished): score={sc_lpol:.10f}, t={time.monotonic()-t_start:.0f}s")
            if sc_lpol < best_score:
                best_score = sc_lpol
                best_a = a_l_pol.copy()

        # Try U-shape at n=8192
        if time.monotonic() < phase4_end - 60:
            a_u8k = _gen_u_shape(n_large, 0.25, 1.0)
            c_u = _fft_convolve(a_u8k, a_u8k)
            max_cu = float(c_u.max())
            a_u_opt, s_u = _adam_nesterov(a_u8k, phase4_end - 30,
                                            T_start=0.01 * max_cu, T_end=0.00005 * max_cu,
                                            lr_start=0.035, lr_end=0.003,
                                            top_k=20, tw_start=0.1, tw_end=0.35,
                                            max_iter=200000, eval_interval=200, clip_norm=0.7,
                                            momentum=0.9)
            sc_u = _score(a_u_opt)
            print(f"Phase 4 (U-shape, n=8192): score={sc_u:.10f}, t={time.monotonic()-t_start:.0f}s")
            if sc_u < best_score:
                best_score = sc_u
                best_a = a_u_opt.copy()

    # === Phase 5: Perturbation restarts (1400s - 1650s) ===
    phase5_end = min(deadline, t_start + 1650.0)
    n_best = len(best_a)
    if time.monotonic() < phase5_end:
        n_restarts = 6
        time_per_r = max(20.0, (phase5_end - time.monotonic()) / n_restarts)
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 20:
                break
            sigma = 0.008 + 0.004 * i
            z = np.log(np.maximum(best_a, 1e-15)) + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            r_end = min(phase5_end - 5, time.monotonic() + time_per_r)
            a_p_opt, s_p = _active_polish(a_pert, r_end,
                                            lr_start=0.012, lr_end=0.0008,
                                            max_iter=100000, active_k=3, eval_interval=100,
                                            clip_norm=0.3)
            sc_p = _score(a_p_opt)
            print(f"  Restart[{i}] (sigma={sigma:.4f}, n={n_best}): score={sc_p:.10f}, t={time.monotonic()-t_start:.0f}s")
            if sc_p < best_score:
                best_score = sc_p
                best_a = a_p_opt.copy()

    # === Phase 6: Final ultra-fine polish (1650s - deadline) ===
    if time.monotonic() < deadline - 10:
        n_fin = len(best_a)
        a_fin, s_fin = _active_polish(best_a, deadline - 5,
                                       lr_start=0.003, lr_end=0.0001,
                                       max_iter=200000, active_k=3, eval_interval=50,
                                       clip_norm=0.2)
        sc_fin = _score(a_fin)
        print(f"Phase 6 (final polish, n={n_fin}): score={sc_fin:.10f}, t={time.monotonic()-t_start:.0f}s")
        if sc_fin < best_score:
            best_score = sc_fin
            best_a = a_fin.copy()

        # One more Nesterov pass with very low temp
        if time.monotonic() < deadline - 10:
            c_f = _fft_convolve(best_a, best_a)
            max_cf = float(c_f.max())
            a_fin2, s_fin2 = _adam_nesterov(best_a, deadline - 3,
                                             T_start=0.0002 * max_cf, T_end=0.000005 * max_cf,
                                             lr_start=0.003, lr_end=0.0002,
                                             top_k=10, tw_start=0.3, tw_end=0.6,
                                             max_iter=100000, eval_interval=100, clip_norm=0.2,
                                             momentum=0.95)
            sc_fin2 = _score(a_fin2)
            print(f"Phase 6b (low-temp Nesterov): score={sc_fin2:.10f}, t={time.monotonic()-t_start:.0f}s")
            if sc_fin2 < best_score:
                best_score = sc_fin2
                best_a = a_fin2.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
