def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    # = grad_a[j] * a_j - a_j * sum_i grad_a[i] * a_i
    dot_val = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot_val)

    return float(obj), grad_z


def _lbfgs_optimize(z_init, beta, max_iter, tol):
    """Run L-BFGS-B in log-space with the smoothed objective."""
    n = len(z_init)

    def fun(z):
        val, _ = _smoothed_obj_and_grad(z, beta, n)
        return val

    def jac(z):
        _, g = _smoothed_obj_and_grad(z, beta, n)
        return g

    result = scipy.optimize.minimize(
        fun, z_init, jac=jac, method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol,
                 'iprint': -1, 'maxls': 50}
    )
    return result.x, result.fun


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on the simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        # Gradient of smoothed max w.r.t. a
        a_rev = y[::-1]
        N1 = _next_pow2(3 * n - 1)
        fw = np.fft.rfft(w, N1)
        fa_rev = np.fft.rfft(a_rev, N1)
        conv_w_ar = np.fft.irfft(fw * fa_rev, N1)[:3 * n - 2]
        fa_y = np.fft.rfft(y, N1)
        conv_y_w = np.fft.irfft(fa_y * fw, N1)[:3 * n - 2]
        G1 = conv_w_ar[n - 1:2 * n - 1]
        G2 = conv_y_w[0:n]
        grad = G1 + G2

        a_next = np.maximum(y - step * grad, 0.0)
        s_next = a_next.sum()
        if s_next > 1e-30:
            a_next = a_next / s_next
        else:
            a_next = np.ones(n) / n

        if it % 500 == 0 or it < 5:
            sc = _score_norm(a_next)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _pair_descent_polish(a, deadline, max_rounds=5000):
    """Polish by reducing contributions to the peak convolution lag."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(123)

    c = _full_convolve(best_a, best_a)
    peak_lag = int(np.argmax(c))

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Occasionally recompute peak
        if rnd % 200 == 0:
            c = _full_convolve(best_a, best_a)
            peak_lag = int(np.argmax(c))
            best_score = _score_norm(best_a)

        # Find top contributors to the peak
        # conv(a,a)[peak_lag] = sum_i a[i] * a[peak_lag - i]
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                contrib[i] = best_a[i] * best_a[j]

        # Pick a random contributor to reduce
        top_k = min(n, 50)
        top_idx = np.argsort(contrib)[-top_k:]
        i = int(rng.choice(top_idx))
        j = peak_lag - i
        if j < 0 or j >= n:
            continue

        # Try reducing a[i] or a[j]
        for target in [i, j]:
            for factor in [0.95, 0.9, 0.85, 1.05, 1.1]:
                if factor < 1.0:
                    # Move mass from target to a random other position
                    a_try = best_a.copy()
                    moved = a_try[target] * (1.0 - factor)
                    a_try[target] *= factor
                    # Distribute moved mass to positions that reduce the peak
                    other_positions = [p for p in range(n)
                                       if p != target and (peak_lag - p < 0 or peak_lag - p >= n)]
                    if not other_positions:
                        other_positions = list(range(n))
                        other_positions.remove(target)
                    dest = int(rng.choice(other_positions))
                    a_try[dest] += moved
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try = a_try / s
                        sc = _score_norm(a_try)
                        if sc < best_score - 1e-10:
                            best_score = sc
                            best_a = a_try.copy()
                else:
                    # Increase (to compensate)
                    a_try = best_a.copy()
                    diff = a_try[target] * (factor - 1.0)
                    a_try[target] *= factor
                    # Take from a position not in the peak
                    other = [p for p in range(n) if p != target]
                    low_idx = np.argsort(best_a)[:max(1, n // 4)]
                    low_in_peak = [p for p in low_idx
                                   if p != target and (peak_lag - p < 0 or peak_lag - p >= n)]
                    if not low_in_peak:
                        low_in_peak = other[:max(1, n // 10)]
                    src = int(rng.choice(low_in_peak))
                    if a_try[src] > diff:
                        a_try[src] -= diff
                    else:
                        continue
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try = a_try / s
                        sc = _score_norm(a_try)
                        if sc < best_score - 1e-10:
                            best_score = sc
                            best_a = a_try.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinusoidal_fast":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t) * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(99)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_vec(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # Main optimization: try the parent length and a few others
    lengths_to_try = [len(best_seq)]
    if len(best_seq) <= 4096:
        lengths_to_try.append(8192)
    if len(best_seq) >= 2048:
        lengths_to_try.append(2048)
    lengths_to_try = list(dict.fromkeys(lengths_to_try))

    n_lengths = len(lengths_to_try)
    time_per_length = (deadline - t_start - 60.0) / n_lengths

    for n_target in lengths_to_try:
        if time.monotonic() > deadline - 30:
            break

        elapsed = time.monotonic() - t_start
        remaining = deadline - time.monotonic()
        print(f"\n=== n={n_target} (remaining={remaining:.0f}s) ===")

        # Initializations to try
        inits = []
        if parent is not None and n_target == len(parent):
            inits.append(parent.copy())
        elif parent is not None:
            inits.append(_interp_profile(parent, n_target))
        inits.append(_make_profile(n_target, "uniform"))
        inits.append(_make_profile(n_target, "sinusoidal"))
        inits.append(_make_profile(n_target, "random"))

        best_for_n = None
        best_score_n = np.inf

        for init_idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break

            a_work = a_init.copy()
            a_work = a_work / a_work.sum()

            # Phase 1: FISTA with coarse beta
            c_curr = _full_convolve(a_work, a_work)
            max_cc = float(c_curr.max())

            betas_coarse = [0.05 * max_cc, 0.02 * max_cc, 0.01 * max_cc,
                            0.005 * max_cc, 0.002 * max_cc, 0.001 * max_cc]

            for beta in betas_coarse:
                if time.monotonic() > deadline - 30:
                    break
                phase_time = max(5.0, min(40.0, time_per_length / (len(betas_coarse) + 4)))
                a_work, sc = _fista_smoothed(a_work, beta,
                                             time.monotonic() + phase_time,
                                             max_iter=20000)
                sc_full = _score(a_work)
                if np.isfinite(sc_full) and sc_full < best_score_n:
                    best_score_n = sc_full
                    best_for_n = a_work.copy()

            # Phase 2: L-BFGS in log-space (coarse to fine)
            if time.monotonic() < deadline - 40 and best_for_n is not None:
                z = np.log(np.maximum(best_for_n, 1e-15))
                z = z - z.mean()

                c_curr = _full_convolve(best_for_n, best_for_n)
                max_cc = float(c_curr.max())

                lbfgs_betas = [0.003 * max_cc, 0.001 * max_cc,
                               0.0003 * max_cc, 0.0001 * max_cc,
                               0.00003 * max_cc, 0.00001 * max_cc]

                for beta in lbfgs_betas:
                    if time.monotonic() > deadline - 30:
                        break
                    remain = deadline - time.monotonic() - 20
                    if remain < 5:
                        break
                    max_it = max(50, int(remain * 2))
                    try:
                        z, obj_val = _lbfgs_optimize(z, beta, max_it, tol=1e-10)
                        a_try = _softmax_vec(z)
                        sc_full = _score(a_try)
                        if np.isfinite(sc_full) and sc_full < best_score_n:
                            best_score_n = sc_full
                            best_for_n = a_try.copy()
                            print(f"  L-BFGS beta={beta:.2e}: score={sc_full:.10f}")
                    except Exception as ex:
                        print(f"  L-BFGS error at beta={beta:.2e}: {ex}")

            # Check after each init
            sc_after = _score(a_work) if a_work is not None else np.inf
            print(f"  Init {init_idx}: best_score_n={best_score_n:.10f} "
                  f"t={time.monotonic()-t_start:.0f}s")

        # Update global best
        if best_for_n is not None and best_score_n < best_score:
            best_score = best_score_n
            best_seq = best_for_n.copy()
            print(f"  New global best: score={best_score:.10f}, n={n_target}")

    # Multi-restart on the best found
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Multi-restart (score={best_score:.10f}) ===")
        n_best = len(best_seq)
        a_work = best_seq.copy() / best_seq.sum()

        restarts = 5
        per_restart = (deadline - time.monotonic() - 30.0) / restarts

        for i in range(restarts):
            if time.monotonic() > deadline - 20:
                break

            if i == 0:
                p_start = _perturb(a_work, 0.01, rng)
            elif i == 1:
                p_start = _perturb(a_work, 0.05, rng)
            elif i == 2:
                # Blend with sinusoidal
                alt = _make_profile(n_best, "sinusoidal")
                p_start = 0.8 * a_work + 0.2 * alt
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif i == 3:
                # Random shift blend
                shift = rng.integers(1, max(2, n_best // 2048))
                p_shift = np.roll(a_work, shift) / np.roll(a_work, shift).sum()
                p_start = 0.7 * a_work + 0.3 * p_shift
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                p_start = _make_profile(n_best, "sinusoidal_fast")

            z = np.log(np.maximum(p_start, 1e-15))
            z = z - z.mean()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())

            lbfgs_betas_r = [0.001 * max_cs, 0.0003 * max_cs,
                             0.0001 * max_cs, 0.00003 * max_cs]

            for beta in lbfgs_betas_r:
                if time.monotonic() > deadline - 20:
                    break
                remain = min(per_restart, deadline - time.monotonic() - 15)
                if remain < 3:
                    break
                try:
                    z, obj_val = _lbfgs_optimize(z, beta,
                                                 max(30, int(remain * 3)),
                                                 tol=1e-10)
                    a_try = _softmax_vec(z)
                    sc_full = _score(a_try)
                    if np.isfinite(sc_full) and sc_full < best_score:
                        best_score = sc_full
                        best_seq = a_try.copy()
                        print(f"  Restart {i}, beta={beta:.2e}: score={sc_full:.10f}")
                except Exception:
                    pass

    # Ultra-fine L-BFGS polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Ultra-fine polish (score={best_score:.10f}) ===")
        a_pol = best_seq.copy() / best_seq.sum()
        z = np.log(np.maximum(a_pol, 1e-15))
        z = z - z.mean()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())

        betas_ult = [0.00001 * max_cp, 0.000003 * max_cp, 0.000001 * max_cp]
        per_t = (deadline - time.monotonic() - 5) / len(betas_ult)

        for beta in betas_ult:
            if time.monotonic() > deadline - 5:
                break
            try:
                z, obj_val = _lbfgs_optimize(z, beta,
                                             max(20, int(per_t * 3)),
                                             tol=1e-12)
                a_try = _softmax_vec(z)
                sc_full = _score(a_try)
                if np.isfinite(sc_full) and sc_full < best_score:
                    best_score = sc_full
                    best_seq = a_try.copy()
                    print(f"  Ultra-fine beta={beta:.2e}: score={sc_full:.10f}")
            except Exception:
                pass

    # Pair-descent polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Pair-descent polish (score={best_score:.10f}) ===")
        a_pd, sc_pd = _pair_descent_polish(best_seq, time.monotonic() + min(20.0, deadline - time.monotonic() - 5))
        sc_pd_full = _score(a_pd)
        if sc_pd_full < best_score:
            best_score = sc_pd_full
            best_seq = a_pd.copy()
            print(f"  Pair-descent: score={sc_pd_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
