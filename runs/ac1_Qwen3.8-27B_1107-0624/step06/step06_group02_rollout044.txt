Looking at this problem, I need to minimize `2*n*max(convolve(a,a)) / (sum(a))^2` for non-negative sequences.

**Key mathematical insight:** By Cauchy-Schwarz, for any non-negative $a$ with sum $S$: $\max(a*a) \geq (a*a)(0) = \sum a_i^2 \geq S^2/n$. Therefore the score is always $\geq 2n \cdot S^2/n / S^2 = 2$. The uniform sequence achieves exactly 2.

"""
Optimizer for score = 2n * max(a*a) / (sum(a))^2.
Theoretical minimum is 2 (achieved by uniform sequences), proven via Cauchy-Schwarz:
  max(a*a) >= (a*a)(0) = sum(a_i^2) >= (sum a_i)^2 / n
  => score >= 2n * (S^2/n) / S^2 = 2
Uses gradient optimization to verify and find the best sequence.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _subgrad_max_conv(a):
    """Compute subgradient of max_k (a*a)(k) with respect to a."""
    n = len(a)
    c = _convolve(a, a)
    k_max = int(np.argmax(c))
    grad = np.zeros(n)
    for j in range(n):
        val = a[j]
        k_minus_j = k_max - j
        if 0 <= k_minus_j < n:
            val += a[k_minus_j]
        grad[j] = 2.0 * val
    return grad, k_max, c


def _adam_logspace_polish(a_init, deadline, max_iter=200000):
    """Adam optimization in log-space to minimize the smoothed max correlation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.005

    c = _convolve(a_init, a_init)
    max_c = float(c.max())
    beta = 0.001 * max_c
    min_beta = 1e-12 * max_c

    while t < max_iter:
        if time.monotonic() > deadline:
            break

        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        np.clip(shifted, -400, 400, out=shifted)
        w = np.exp(shifted)
        s_w = float(w.sum())
        if s_w < 1e-300 or not np.isfinite(s_w):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / s_w

        # Gradient: 2n * d(max a*a)/da, approximated via weighted version
        # d(a*a)(k)/da_j = a_j + a_{k-j}
        # Weighted sum over k
        n_conv = len(c)
        grad = np.zeros(n)
        for j in range(n):
            total = 0.0
            for k in range(n_conv):
                wk = w[k]
                if wk < 1e-15:
                    continue
                val = a[j] if 0 <= j < n else 0.0
                kj = k - j
                if 0 <= kj < n:
                    val += a[kj]
                total += wk * val
            grad[j] = 2.0 * n * total

        # Simplify: use only the argmax for subgradient
        k_max = int(np.argmax(c))
        grad2 = np.zeros(n)
        for j in range(n):
            val = 2.0 * a[j]
            kj = k_max - j
            if 0 <= kj < n:
                val += 2.0 * a[kj]
            grad2[j] = val

        # Use a blend
        grad_final = 0.5 * grad + 0.5 * grad2

        t += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad_final
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_final * grad_final)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update *= 0.5 / update_norm

        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 200 == 0:
            a_chk = np.exp(z - z.max())
            a_chk = a_chk / a_chk.sum()
            sc = _score(a_chk)
            if sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if t % 5000 == 0:
            beta *= 0.5
            lr *= 0.9
            if beta < min_beta:
                beta = min_beta

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_subgrad(a_init, deadline, max_iter=100000):
    """FISTA with subgradient of the max correlation (on simplex)."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_sc = _score(a_curr)

    # Adaptive step size
    step = 0.01
    c_curr = _convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())
    step = 0.1 / (2.0 * n * max_cc + 1e-10)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        _, k_max, c_y = _subgrad_max_conv(y)
        grad = np.zeros(n)
        for j in range(n):
            val = 2.0 * y[j]
            kj = k_max - j
            if 0 <= kj < n:
                val += 2.0 * y[kj]
            grad[j] = val

        a_next = y - step * grad
        a_next = np.maximum(a_next, 0.0)
        s_n = a_next.sum()
        if s_n > 1e-30:
            a_next = a_next / s_n
        else:
            a_next = y.copy()

        sc = _score(a_next)
        if sc < best_sc:
            best_sc = sc
            best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

        # Decay step
        if it % 100 == 0:
            step *= 0.995

    return best_a, best_sc


def _coord_descent(a_init, deadline, max_rounds=100):
    """Simple coordinate descent to polish."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(0, n, max(1, n // 256)):
            if time.monotonic() > deadline:
                break
            old = a[j]
            for factor in [0.98, 0.99, 1.01, 1.02]:
                a[j] = old * factor
                sc = _score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
            else:
                a[j] = old
        if not improved and r > 10:
            break

    return best_a, best_sc
