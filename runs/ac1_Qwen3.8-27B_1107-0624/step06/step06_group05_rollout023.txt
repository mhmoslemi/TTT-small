"""
Hybrid optimizer: FISTA + spectral shaping + L-BFGS for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: spectral shaping (iterative DFT magnitude equalization)
combined with gradient-based methods.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1, returns 2n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    w is a distribution over the convolution output indices [0, 2n-2]."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter):
    """FISTA on smoothed objective with simplex constraint."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    c_max_init = float(c_init.max())
    L = 4.0 * n / max(c_max_init, 1e-15)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        if beta < 1e-15:
            # Use exact subgradient
            tol = 1e-10 * c_max + 1e-15
            active = np.where(c_y >= c_max - tol)[0]
            w = np.zeros_like(c_y)
            w[active] = 1.0 / len(active)
        else:
            w = np.exp((c_y - c_max) / beta)
            w_sum = float(w.sum())
            if w_sum < 1e-300 or not np.isfinite(w_sum):
                w = np.zeros_like(c_y)
                w[int(np.argmax(c_y))] = 1.0
            else:
                w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _spectral_shape(a, n_iters, alpha, deadline):
    """Iteratively shape the spectrum to equalize DFT magnitudes.
    This promotes flat autocorrelation."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    N_fft = _next_pow2(2 * n - 1)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute DFT (padded for linear convolution equivalent)
        A = np.fft.rfft(a, N_fft)
        
        # Target: equalize magnitudes (excluding DC)
        mags = np.abs(A)
        mags_non_dc = mags[1:]
        target = np.median(mags_non_dc[mags_non_dc > 1e-10])
        if target < 1e-15:
            target = mags_non_dc.mean()
        
        # Shape magnitudes toward target
        for k in range(1, N_fft // 2 + 1):
            if mags[k] > 1e-15:
                ratio = (target / mags[k]) ** alpha
                A[k] = A[k] * ratio
        
        # Inverse DFT
        a_new = np.fft.irfft(A, N_fft)[:n]
        
        # Project onto non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        a = a_new
        if it % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    """Adam optimization in log-softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a = np.exp(z)
        a = a / a.sum()
        c = _full_convolve(a, a)
        cm = float(c.max())
        if beta < 1e-15:
            tol = 1e-10 * cm
            active = np.where(c >= cm - tol)[0]
            w = np.zeros_like(c)
            w[active] = 1.0 / len(active)
        else:
            e = np.exp((c - cm) / beta)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        
        t = it + 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        
        if t % 50 == 0:
            a_chk = np.exp(z)
            a_chk = a_chk / a_chk.sum()
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = np.exp(z)
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    """L-BFGS-B on log-softmax parameterization."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = np.exp(z)
        a = a / a.sum()
        c = _full_convolve(a, a)
        cm = float(c.max())
        if beta < 1e-15:
            return 2.0 * n * cm
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = np.exp(z)
        a = a / a.sum()
        c = _full_convolve(a, a)
        cm = float(c.max())
        if beta < 1e-15:
            tol = 1e-10 * cm
            active = np.where(c >= cm - tol)[0]
            w = np.zeros_like(c)
            w[active] = 1.0 / len(active)
        else:
            c_sh = (c - cm) / beta
            e = np.exp(c_sh)
            s = float(e.sum())
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 5.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 50)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = np.exp(z_opt)
        a_opt = a_opt / a_opt.sum()
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = np.exp(best_z)
    a_best = a_best / a_best.sum()
    return a_best, best_score


def _nesterov_subgrad(a_init, deadline, max_iter):
    """Nesterov-accelerated subgradient method on the exact max objective."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    y = a_curr.copy()
    t_val = 1.0
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    c_init = _full_convolve(a_curr, a_curr)
    c_max_init = float(c_init.max())
    # Initial step size based on problem scale
    step0 = 0.5 / (n * c_max_init)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        
        # Nesterov step
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - y)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        tol = 1e-9 * c_max
        active = np.where(c_y >= c_max - tol)[0]
        w = np.zeros_like(c_y)
        w[active] = 1.0 / len(active)
        
        grad = _compute_grad(y, w)
        step = step0 / np.sqrt(it + 1)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return np.exp(z) / np.exp(z).sum()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n

Wait, I need to double-check the `_compute_grad` function. The gradient of `conv(a,a)[k]` with respect to `a[j]` is:

conv(a,a)[k] = sum_{i} a[i] * a[k-i] (where both i and k-i are in [0, n-1])

∂/∂a[j] conv(a,a)[k] = a[k-j] (if 0 ≤ k-j ≤ n-1) + a[j-k] (if 0 ≤ j-k ≤ n-1)

# Global state for L-BFGS callbacks (to avoid closures)
_LBFGS_N = 0
_LBFGS_BETA = 0.0


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _make_w(c, beta):
    cm = float(c.max())
    if beta < 1e-14:
        tol = 1e-9 * cm + 1e-15
        active = np.where(c >= cm - tol)[0]
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
    else:
        w = np.exp((c - cm) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
    return w


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    c_max_init = float(c_init.max())
    L = 4.0 * n / max(c_max_init, 1e-15)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        w = _make_w(c_y, beta)
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _spectral_shape(a, n_iters, alpha, deadline):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _score_norm(a)
    a = a / s
    best_score = _score_norm(a)
    best_a = a.copy()
    N_fft = _next_pow2(2 * n - 1)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mags = np.abs(A)
        mags_nc = mags[1:]
        valid = mags_nc[mags_nc > 1e-12]
        if len(valid) == 0:
            break
        target = np.median(valid)
        if target < 1e-12:
            target = valid.mean()
        # Shape magnitudes toward target
        ratios = np.ones(len(A))
        for k in range(1, len(A)):
            if mags[k] > 1e-12:
                ratios[k] = (target / mags[k]) ** alpha
        A = A * ratios
        a_new = np.fft.irfft(A, N_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score

def _lbfgs_obj(z):
    n = _LBFGS_N
    beta = _LBFGS_BETA
    a = np.exp(z)
    a = a / a.sum()
    c = _full_convolve(a, a)
    cm = float(c.max())
    if beta < 1e-14:
        return 2.0 * n * cm
    ratio = (c - cm) / beta
    e = np.exp(ratio)
    s = float(e.sum())
    if s < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _lbfgs_grad(z):
    n = _LBFGS_N
    beta = _LBFGS_BETA
    a = np.exp(z)
    a = a / a.sum()
    c = _full_convolve(a, a)
    w = _make_w(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad_a = 4.0 * n * h
    correction = float(np.dot(a, grad_a))
    return a * (grad_a - correction)


def _lbfgs_opt(a_init, deadline, beta):

            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = a0.copy()

    # === Phase 1: FISTA aggressive continuation ===
    print("Phase 1: FISTA continuation")
    phase1_end = min(deadline, t_start + 320.0)
    c_curr = _full_convolve(a_work, a_work)
    max_c0 = float(c_curr.max())
    beta = 0.025 * max_c0
    n_phases = 22

    for phase in range(n_phases):
        if time.monotonic() > phase1_end - 5:
            break
        remaining = phase1_end - time.monotonic() - 3
        if remaining < 2:
            break
        per_time = min(remaining / max(1, n_phases - phase), 22.0)
        max_it = max(3000, int(per_time * 5000))
        a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_time, max_it)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            if phase % 4 == 0:
                print(f"  FISTA p{phase} beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_work, a_work)
        beta = 0.32 * float(c_curr.max())
        if beta < 1e-13:
            beta = 1e-13

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Spectral shaping (NEW) ===
    print("Phase 2: Spectral shaping")
    phase2_end = min(deadline, t_start + 580.0)
    if time.monotonic() < phase2_end - 10:
        alphas = [0.25, 0.12, 0.06, 0.03]
        per_alpha = (phase2_end - time.monotonic() - 5) / len(alphas)
        for alpha in alphas:
            if time.monotonic() > phase2_end - 8:
                break
            a_shaped, _ = _spectral_shape(a_work, n_iters=30000, alpha=alpha,
                                          deadline=time.monotonic() + per_alpha)
            sc = _score(a_shaped)
            if sc < best_score:
                best_score = sc
                best_seq = a_shaped.copy()
                a_work = a_shaped
                print(f"  Spectral alpha={alpha}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: Adam fine ===
    print("Phase 3: Adam fine optimization")
    phase3_end = min(deadline, t_start + 950.0)
    if time.monotonic() < phase3_end - 20:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.0004 * max_cc, 0.00008 * max_cc, 0.00001 * max_cc]
        per_beta = (phase3_end - time.monotonic() - 10) / len(betas)
        for bi, b in enumerate(betas):
            if time.monotonic() > phase3_end - 15:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + max(5.0, per_beta),
                                  b, lr_init=0.005, max_iter=500000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={b:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 3: best={best_score:.10f}")

    # === Phase 4: Nesterov + Spectral interleaving (NEW) ===
    print("Phase 4: Nesterov + Spectral interleaving")
    phase4_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < phase4_end - 15:
        a_ns = a_work.copy()
        remaining = phase4_end - time.monotonic() - 10
        n_rounds = min(int(remaining / 12), 20)
        for r in range(n_rounds):
            if time.monotonic() > phase4_end - 10:
                break
            t_rem = phase4_end - time.monotonic() - 8
            t_per = max(3.0, t_rem / max(1, n_rounds - r))
            # Nesterov subgradient
            a_ns, _ = _nesterov_subgrad(a_ns, time.monotonic() + min(6.0, t_per * 0.5), max_iter=80000)
            sc = _score(a_ns)
            if sc < best_score:
                best_score = sc
                best_seq = a_ns.copy()
            # Spectral shaping
            a_ns, _ = _spectral_shape(a_ns, n_iters=8000, alpha=0.04,
                                      deadline=time.monotonic() + min(5.0, t_per * 0.4))
            sc = _score(a_ns)
            if sc < best_score:
                best_score = sc
                best_seq = a_ns.copy()
                print(f"  Nest+Spectral r{r}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_ns

    print(f"End Phase 4: best={best_score:.10f}")

    # === Phase 5: L-BFGS polish ===
    print("Phase 5: L-BFGS refinement")
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        for b_lb in [0.00003 * max_cc, 0.000002 * max_cc]:
            if time.monotonic() > phase5_end - 10:
                break
            t_alloc = min(120.0, (phase5_end - time.monotonic() - 8) / 2)
            a_lb, _ = _lbfgs_opt(a_work, time.monotonic() + t_alloc, b_lb)
            sc = _score(a_lb)
            if sc < best_score:
                best_score = sc
                best_seq = a_lb.copy()
                a_work = a_lb
                print(f"  L-BFGS beta={b_lb:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 5: best={best_score:.10f}")

    # === Phase 6: Multi-restart ===
    print("Phase 6: Multi-restart exploration")
    phase6_end = min(deadline, t_start + 1680.0)
    if time.monotonic() < phase6_end - 20:
        n_restarts = 5
        per_r = (phase6_end - time.monotonic() - 10) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            if i == 0:
                p = _perturb(a_work, 0.012, rng)
            elif i == 1:
                p = _make_profile(n_main, "raised_cosine")
                p = 0.75 * a_work + 0.25 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 2:
                p = _make_profile(n_main, "tapered")
                p = 0.6 * a_work + 0.4 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 3:
                p = a_work.copy()
                p, _ = _spectral_shape(p, n_iters=3000, alpha=0.15,
                                       deadline=time.monotonic() + per_r * 0.2)
            else:
                p = _perturb(a_work, 0.025, rng)

            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.0002 * max_cp
            t_fista = min(per_r * 0.4, 20.0)
            p, _ = _fista_opt(p, b_r, time.monotonic() + t_fista, 30000)
            t_adam = min(per_r * 0.4, 20.0)
            p, _ = _adam_opt(p, time.monotonic() + t_adam, b_r * 0.05, 0.003, 40000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-polish ===
    print("Phase 7: Final ultra-polish")
    if best_seq is not None and time.monotonic() < deadline - 12:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        per_t = deadline - time.monotonic() - 5
        if per_t > 15:
            c_pol = _full_convolve(a_pol, a_pol)
            max_cp = float(c_pol.max())

            # Very gentle spectral shaping
            a_pol, _ = _spectral_shape(a_pol, n_iters=50000, alpha=0.015,
                                       deadline=time.monotonic() + per_t * 0.3)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final spectral: score={sc_p:.10f}")

            # Final L-BFGS
            if time.monotonic() < deadline - 10:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000001 * float(c_f.max())
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 3.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

            # One more Adam pass
            if time.monotonic() < deadline - 8:
                a_pol, _ = _adam_opt(a_pol, deadline - 3.0, 1e-8 * max_cp, 0.001, 100000)
                sc_a = _score(a_pol)
                if sc_a < best_score:
                    best_score = sc_a
                    best_seq = a_pol.copy()
                    print(f"  Final Adam: score={sc_a:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
