- This is derived from the linearization: c ≈ c_0 + 2*Re(ifft(A_0 * delta_A))

"""
Advanced multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive Adam with spectral nudges, multi-resolution exploration,
diverse restarts, and scipy L-BFGS-B final polish.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_logspace(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _compute_grad_simplex(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _spectral_nudge(a, strength=0.1, deadline=None):
    """Apply a spectral equalization nudge to reduce convolution peaks."""
    n = len(a)
    a_work = a.copy()
    s = a_work.sum()
    if s < 1e-15:
        return a
    a_work = a_work / s

    N_fft = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a_work, N_fft)
    A2 = A * A
    c_full = np.fft.irfft(A2, N_fft)[:2 * n - 1]
    c_max = float(c_full.max())
    c_mean = float(c_full.mean())

    if c_max < c_mean * 1.01:
        return a

    target = np.full(2 * n - 1, c_mean)
    error = c_full - target
    error_fft = np.fft.rfft(error, N_fft)

    A_mag = np.abs(A)
    A_phase = np.angle(A)
    A_mod = A.copy()

    safe_mask = A_mag > 1e-10
    A_mod[safe_mask] -= strength * (error_fft[safe_mask] / (2.0 * A_mag[safe_mask] * A_mag[safe_mask])) * A[safe_mask]

    a_new = np.fft.irfft(A_mod, N_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
    a_new = a_new / s_new
    return a_new


def _adam_adaptive(a_init, deadline, beta_start, beta_end, max_iter_total, lr_init=0.01, lr_decay=0.97, lr_every=5000, grad_clip=1.0, nudge_interval=2000, nudge_strength=0.05):
    """Adam optimizer in log-space with adaptive beta and periodic spectral nudges."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0
    total_iter = max_iter_total
    beta_ratio = np.log(beta_end / beta_start) if beta_start > beta_end and beta_end > 0 else 0.0

    for it in range(total_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        beta = beta_start * np.exp(beta_ratio * t / total_iter)

        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / max(beta, 1e-12)
        e = np.exp(c_shifted)
        s_w = float(e.sum())
        if s_w < 1e-300 or not np.isfinite(s_w):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_w

        grad = _compute_grad_logspace(a, c, w, n)
        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > grad_clip:
            update = update * (grad_clip / update_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if it % nudge_interval == 0 and it > 0:
            a_tmp = _softmax_compute(z)
            a_nudged = _spectral_nudge(a_tmp, strength=nudge_strength * (1.0 - t / total_iter) * 0.5)
            z_new = np.log(np.maximum(a_nudged, 1e-15))
            z_new = z_new - z_new.mean()
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * 1.0)) / 2.0
            z = 0.7 * z + 0.3 * z_new
            z = z - z.mean()

        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                no_improve = 0
            else:
                no_improve += 1

        if t % lr_every == 0:
            current_lr = max(0.0003, current_lr * lr_decay)

        if no_improve > 30 and t > 500:
            current_lr = max(0.0003, current_lr * 0.92)
            no_improve = 0

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "two_peak":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.3) ** 2) + np.exp(-20.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _subgradient_polish(a_init, n_steps, deadline):
    """Polish using subgradient on the true max (not smoothed)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    step_size = 0.3

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        top_k = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        vals = c[top_idx] - cm + 1e-10
        w_vec = np.zeros(len(c))
        w_vec[top_idx] = vals ** 3
        w_sum = w_vec.sum()
        if w_sum < 1e-300:
            w_vec = np.zeros(len(c))
            w_vec[top_idx[0]] = 1.0
        else:
            w_vec /= w_sum

        grad = _compute_grad_simplex(a, w_vec)
        a_new = _proj_simplex(a - step_size * grad)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size * 1.02, 1.0)
        else:
            step_size *= 0.85

    return best_a, best_score
