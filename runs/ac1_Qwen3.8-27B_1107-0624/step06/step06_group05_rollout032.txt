    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find active set: top n_active values
        idx = np.argsort(c)[::-1][:n_active]
        # Weights: inverse gap (closer to max gets more weight)
        gaps = c_max - c[idx]
        w_active = np.exp(-gaps / (0.001 * c_max))
        w_active = w_active / w_active.sum()
        
        # Compute gradient with these weights
        w_full = np.zeros(len(c))
        w_full[idx] = w_active
        grad = _compute_grad(a, w_full)
        
        # Compute the "target level": we want grad to be constant on support
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break
        grad_on_support = grad[support]
        target = float(grad_on_support.mean())
        
        # Correction: push grad towards target
        # a_new = a - eta * (grad - target) * a (log-space update)

def _active_set_refine(a_init, deadline, n_top=8):
    """Refine using KKT-based active set linearization."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_a = a.copy()

        it += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set: top values with smooth weights
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:n_top]
        c_active = c[active]

        w_active = np.exp(-gap / (0.005 * c_max))
        w_active = w_active / w_active.sum()

        w_full = np.zeros(len(c))
        w_full[active] = w_active
        grad = _compute_grad(a, w_full)

            break
        g_s = grad[support]
        target = float(g_s.mean())

        # Only correct on support to maintain sparsity structure
        corr_masked = np.zeros_like(correction)
        corr_masked[support] = correction[support]

        if update_norm > max_update:
            update = update * (max_update / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if it % 3000 == 0 and it > 5000:
            current_lr = max(0.0001, current_lr * 0.90)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining FISTA, active-set refinement,
    Adam, L-BFGS, coordinate descent, and restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = a0.copy()

    # === Phase 1: FISTA aggressive continuation (50% of time) ===
    print("Phase 1: FISTA aggressive beta continuation")
    phase1_end = min(deadline, t_start + 0.45 * (deadline - t_start))
    c_curr = _full_convolve(a_work, a_work)
    max_c0 = float(c_curr.max())

    # Use a staircase beta schedule
    n_phases = 35
    for phase in range(n_phases):
        if time.monotonic() > phase1_end - 3:
            break
        remaining = phase1_end - time.monotonic() - 2
        if remaining < 1:
            break
        per_time = remaining / max(1, n_phases - phase)
        # Beta staircase: stay at same beta for a while, then drop
        beta = max_c0 * 0.04 * (0.28 ** (phase // 5))
        if beta < 1e-13:
            beta = 1e-13
        max_it = max(5000, int(per_time * 5000))
        a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_time, max_it)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            if phase % 7 == 0:
                print(f"  FISTA p{phase} beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_work, a_work)
        if float(c_curr.max()) < max_c0 * 0.5:
            max_c0 = float(c_curr.max())

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Active-set refinement (10% of time) ===
    print("Phase 2: Active-set KKT refinement")
    phase2_end = min(deadline, t_start + 0.55 * (deadline - t_start))
    if time.monotonic() < phase2_end - 5:
        a_as, _ = _active_set_refine(a_work, phase2_end, n_top=12)
        sc_as = _score(a_as)
        if sc_as < best_score:
            best_score = sc_as
            best_seq = a_as.copy()
            print(f"  Active-set: score={sc_as:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_as

    # === Phase 3: Adam fine (15% of time) ===
    print("Phase 3: Adam multi-beta fine optimization")
    phase3_end = min(deadline, t_start + 0.70 * (deadline - t_start))
    if time.monotonic() < phase3_end - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.0005 * max_cc, 0.0001 * max_cc, 0.00002 * max_cc, 0.000005 * max_cc]
        per_beta = (phase3_end - time.monotonic() - 8) / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase3_end - 8:
                break
            a_work, _ = _adam_fine(a_work, time.monotonic() + max(5.0, per_beta),
                                   beta, lr_init=0.008, max_iter=500000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS very fine (12% of time) ===
    print("Phase 4: L-BFGS high-precision refinement")
    phase4_end = min(deadline, t_start + 0.82 * (deadline - t_start))
    if time.monotonic() < phase4_end - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        beta_lb = 0.00001 * max_cc
        a_lb, _ = _lbfgs_opt(a_work, phase4_end, beta_lb)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lb

    # === Phase 5: Coordinate descent polish (7% of time) ===
    print("Phase 5: Biased coordinate descent polish")
    phase5_end = min(deadline, t_start + 0.89 * (deadline - t_start))
    if time.monotonic() < phase5_end - 8:
        a_cd, _ = _coord_descent_biased(a_work, phase5_end, rng)
        sc_cd = _score(a_cd)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  Coord descent: score={sc_cd:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_cd

    # === Phase 6: Multi-restart + final polish (11% of time) ===
    print("Phase 6: Multi-restart and final polish")
    if best_seq is not None and time.monotonic() < deadline - 25:
        n_restarts = 5
        per_r = (deadline - time.monotonic() - 20) / (n_restarts + 1)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p = _perturb(a_work, 0.008, rng)
            elif i == 1:
                p = _perturb(a_work, 0.02, rng)
            elif i == 2:
                # Frequency-domain perturbation
                N = _next_pow2(2 * n_main)
                fa = np.fft.rfft(a_work, N)
                # Small random phase perturbation
                phase_pert = 0.01 * rng.standard_normal(len(fa))
                fa_pert = fa * np.exp(1j * phase_pert)
                p = np.fft.irfft(fa_pert, N)[:n_main]
                p = np.maximum(p, 0.0)
                if p.sum() > 0:
                    p = p / p.sum()
                else:
                    p = _perturb(a_work, 0.01, rng)
            elif i == 3:
                # Smooth perturbation (low frequency)
                p = a_work.copy()
                smooth_noise = np.convolve(rng.standard_normal(n_main),
                                           np.ones(64) / 64, mode='same')
                z = np.log(np.maximum(p, 1e-15)) + 0.015 * smooth_noise
                z = z - z.mean()
                p = _softmax_compute(z)
            else:
                p = _perturb(a_work, 0.004, rng)

            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.0002 * max_cp
            p, _ = _fista_opt(p, b_r, time.monotonic() + per_r * 0.4, 50000)
            p, _ = _adam_fine(p, time.monotonic() + per_r * 0.3, b_r * 0.3, 0.005, 100000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

        # Final ultra-polish on the best
        if time.monotonic() < deadline - 10:
            a_pol = best_seq.copy()
            a_pol = a_pol / a_pol.sum()
            c_pol = _full_convolve(a_pol, a_pol)
            max_cp = float(c_pol.max())
            remaining = deadline - time.monotonic() - 5
            if remaining > 5:
                # Very tight L-BFGS
                b_f = 0.000001 * max_cp
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 3.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    # Final validation
    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
