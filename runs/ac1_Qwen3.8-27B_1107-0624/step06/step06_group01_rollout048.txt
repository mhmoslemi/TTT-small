def _next_pow2(n):
    k = 1
    while k < n:

    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

                delta = a_try[i_max] * (1 - frac)
                a_try[i_max] -= delta
                # Redistribute to a random position that doesn't contribute to this peak
                valid_j = np.where((np.arange(n) + k >= n) | (np.arange(n) + k < 0))[0]
                if len(valid_j) == 0:
                    valid_j = np.arange(n)
                j_target = rng.integers(0, n)
                a_try[j_target] += delta
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    sc = _score_norm(a_try)
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_a = a_try.copy()
                        improved = True
                        break
    
        if not improved and iteration > 100:
            # Add small random perturbation to escape
            if iteration % 50 == 0:
                z = np.log(np.maximum(best_a, 1e-15))
                z += 0.01 * rng.standard_normal(n)
                z -= z.mean()
                best_a = _softmax_compute(z)
                best_score = _score_norm(best_a)
    
    return best_a, best_score

def _block_optimize(a, deadline, block_size=64, n_rounds=20):
    """Block coordinate descent: optimize subsets of coordinates using scipy."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(77)
    
    for round_idx in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        # Select a random block of coordinates
        start = rng.integers(0, max(1, n - block_size))
        block = set(range(start, start + block_size))
        
        # Optimize this block
        a_frozen = best_a.copy()
        
        def obj(x):
            a_try = a_frozen.copy()
            a_try[list(block)] = np.maximum(x, 0)
            s = a_try.sum()
            if s < 1e-15:
                return 1e10
            return _score_norm(a_try)
        
        x0 = a_frozen[list(block)].copy()
        try:
            res = minimize(obj, x0, method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.fun < best_score - 1e-12:
                a_try = a_frozen.copy()
                a_try[list(block)] = np.maximum(res.x, 0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    sc = _score_norm(a_try)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_try.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

def _interleave_perturb(a, rng):
    """Perturb by interleaving with a different pattern."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    # Create a structured perturbation
    period = rng.integers(7, 23)
    z += 0.05 * np.sin(2 * np.pi * np.arange(n) / period)
    z -= z.mean()
    return _softmax_compute(z)

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    
    n = len(best_seq)
    
    # Phase 1: Refined Adam with peak focusing (50% of time)
    print(f"\n=== Phase 1: Refined Adam (n={n}) ===")
    phase1_deadline = t_start + 0.55 * (budget_s - 30)
    
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    
    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())
    
    # Multiple passes with decreasing beta
    for pass_idx in range(3):
        if time.monotonic() > phase1_deadline - 10:
            break
        beta_start = max_cc * (0.02, 0.005, 0.001)[pass_idx]
        beta_end = beta_start * 0.01
        phase_dl = min(phase1_deadline, time.monotonic() + 0.33 * (phase1_deadline - t_start - 60))
        
        a_opt, sc_opt = _adam_peak_focused(
            a_work, phase_dl,
            max_iter=500000, topK=5,
            beta_init=beta_start, beta_min=beta_end
        )
        sc_full = _score(a_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()
            print(f"  Pass {pass_idx}: score={sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        a_work = a_opt.copy()
        a_work = a_work / a_work.sum()
    
    # Phase 2: Peak equalization (20% of time)
    print(f"\n=== Phase 2: Peak Equalization ===")
    phase2_deadline = t_start + 0.75 * (budget_s - 30)
    
    a_eq, sc_eq = _peak_equalize(best_seq, phase2_deadline, n_iter=2000)
    sc_eq_full = _score(a_eq)
    if sc_eq_full < best_score:
        best_score = sc_eq_full
        best_seq = a_eq.copy()
        print(f"  Peak equalize: score={sc_eq_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    
    # Phase 3: More Adam with even smaller beta (15% of time)
    print(f"\n=== Phase 3: Ultra-fine Adam ===")
    phase3_deadline = t_start + 0.90 * (budget_s - 30)
    
    c_curr = _full_convolve(best_seq, best_seq)
    max_cc = float(c_curr.max())
    
    a_opt, sc_opt = _adam_peak_focused(
        best_seq, phase3_deadline,
        max_iter=800000, topK=3,
        beta_init=max_cc * 0.0005, beta_min=max_cc * 0.000005

        best_seq = a_opt.copy()

    phase4_deadline = t_start + 0.97 * (budget_s - 30)
    
    restarts = [
        ("perturb_small", lambda: _perturb(best_seq, 0.01, rng)),
        ("perturb_med", lambda: _perturb(best_seq, 0.03, rng)),

                max_iter=200000, topK=3,
                beta_init=max_cr * 0.001, beta_min=max_cr * 0.00001

    print(f"\n=== Phase 5: Block Coordinate Polish ===")
    phase5_deadline = deadline
    
    if time.monotonic() < phase5_deadline - 5:
        a_block, sc_block = _block_optimize(best_seq, phase5_deadline, block_size=32, n_rounds=50)

    # Final: One more ultra-fine Adam pass
    if time.monotonic() < deadline - 3:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        a_fin, sc_fin = _adam_peak_focused(
            best_seq, deadline,
            max_iter=300000, topK=3,
            beta_init=max_cf * 0.0001, beta_min=max_cf * 0.000001
        )
        sc_fin_full = _score(a_fin)
        if sc_fin_full < best_score:
            best_score = sc_fin_full
            best_seq = a_fin.copy()
            print(f"\n  Final Adam: score={sc_fin_full:.10f}")
    
    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq)} ===")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    
    return [float(x) for x in final]
