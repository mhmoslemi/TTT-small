"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses Adam in log-space with adaptive beta, followed by aggressive cyclic coordinate
descent with analytical 1D search and golden-section refinement. Tries n=4096 and
n=8192 with diverse initializations and random restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    topk_thresh = np.partition(c, -k)[-k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([np.argmax(c)])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter_per_phase):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_per_phase:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t_count += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_weights(c, beta, top_k)
            grad_a = _compute_grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad_z = 2.0 * n * a * (grad_a - mean_grad)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t_count)
            vh = v_buf / (1.0 - b2 ** t_count)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t_count % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 2000 == 0:
                current_lr = max(0.0003, current_lr * 0.98)

        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descend_aggressive(a, n_passes, deadline, top_k_for_crit=20):

            # If j!=2i: two terms a_i * a_{j-i} + a_{j-i} * a_i = 2*a_i*a_{j-i} (if 0<=j-i<n)

                        candidates.append(t_star * 0.7)

                    # This is different, but the critical point formula above is approximate.

                        fc = fd

                g_gold = _g_of_t(t_gold)
                if g_gold < best_g:
                    best_t = t_gold
                    best_g = g_gold

            if abs(best_t - a_i_old) > 1e-12:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:

            if sc < best_score:

    sc_final = _score_norm(a)

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum", rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        n_fords = rng.integers(2, 6)
        for _ in range(n_fords):
            freq = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.5)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        pwr = rng.uniform(0.5, 1.5)
        a = np.sin(np.pi * t) ** pwr
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        w = rng.uniform(3, 15)
        a = np.exp(-w * (t - c1) ** 2) + np.exp(-w * (t - c2) ** 2) + 0.05
        return a / a.sum()
    elif kind == "random_smooth":
        a = rng.random(n)
        # Smooth with a kernel
        k = max(3, n // 100)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

        a_work, sc = _adam_logspace(a_work, phase4_end, betas_ultra, lr_init=0.003, top_k=50, max_iter_per_phase=50000)
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  Phase4 (ultra Adam): score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Final CD polish
        phase5_end = min(time.monotonic() + 120, deadline - 30)
        a_cd2, sc_cd2 = _coord_descend_aggressive(a_work, 20, phase5_end, top_k_for_crit=30)
        sc_cd2_full = _score(a_cd2)
        if sc_cd2_full < best_score:
            best_score = sc_cd2_full
            best_seq = a_cd2.copy()
            print(f"  Phase5 (final CD): score={sc_cd2_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Try n=8192 if time allows
    if time.monotonic() < deadline - 120:
        print(f"\n=== Trying n=8192 (remaining: {deadline - time.monotonic():.1f}s) ===")
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big_init = _score(a_big)
        print(f"  Interpolated init: score={sc_big_init:.10f}")

        remaining = deadline - time.monotonic() - 30
        # Quick Adam + CD
        c_big = _full_convolve(a_big, a_big)
        max_cb = float(c_big.max())
        betas_big = [0.01 * max_cb, 0.003 * max_cb, 0.001 * max_cb, 0.0003 * max_cb]
        phase_big_end = time.monotonic() + max(30, remaining * 0.5)
        a_big, sc_b = _adam_logspace(a_big, phase_big_end, betas_big, lr_init=0.01, top_k=100, max_iter_per_phase=20000)
        sc_big = _score(a_big)
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()
            print(f"  n=8192 Adam: score={sc_big:.10f} t={time.monotonic()-t_start:.1f}s")

        # CD on n=8192
        if time.monotonic() < deadline - 60:
            phase_big_cd_end = time.monotonic() + max(20, (deadline - time.monotonic() - 30) * 0.5)
            a_big_cd, sc_bc = _coord_descend_aggressive(a_big, 10, phase_big_cd_end, top_k_for_crit=20)
            sc_big_cd = _score(a_big_cd)
            if sc_big_cd < best_score:
                best_score = sc_big_cd
                best_seq = a_big_cd.copy()
                print(f"  n=8192 CD: score={sc_big_cd:.10f} t={time.monotonic()-t_start:.1f}s")

    # Random restarts if time allows
    if time.monotonic() < deadline - 90 and best_seq is not None:
        print(f"\n=== Random restarts (remaining: {deadline - time.monotonic():.1f}s) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()
        n_restarts = 3
        per_restart = max(20, (deadline - time.monotonic() - 30) / n_restarts)

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if ri == 0:
                p_start = a_base + 0.01 * rng.standard_normal(n_best)
            elif ri == 1:
                shift = rng.integers(1, max(2, n_best // 512))
                p_start = np.roll(a_base, shift).copy()
            else:
                # Block perturbation
                p_start = a_base.copy()
                blk = n_best // 16
                s1 = rng.integers(0, n_best - blk)
                p_start[s1:s1+blk] *= (1.0 + 0.1 * rng.standard_normal())

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.002 * max_cs, 0.0005 * max_cs, 0.0001 * max_cs]
            p_best, sc_r = _adam_logspace(p_start, time.monotonic() + per_restart * 0.5,
                                          betas_r, lr_init=0.005, top_k=80, max_iter_per_phase=20000)
            # Quick CD
            if time.monotonic() < deadline - 30:
                p_best, sc_r2 = _coord_descend_aggressive(p_best, 5, time.monotonic() + per_restart * 0.4, top_k_for_crit=20)
            sc_r_full = _score(p_best)
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_seq = p_best.copy()
                print(f"  Restart {ri}: score={sc_r_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final output
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * 0.04
        sc_final = _score(final)
        print(f"Verified: score={sc_final:.10f}, n={len(final)}, sum={s:.4f}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) * 0.04
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
