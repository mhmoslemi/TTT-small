    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _grad_smoothed(a, c, beta, N_fft):
    """Gradient of 2n * beta * log(sum exp(c[k]/beta)) w.r.t. a"""
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    w = e / e.sum()  # softmax weights

    # = 2n * cross_corr(w, a)[j]

    A = np.fft.rfft(a, N_fft)
    g = np.fft.irfft(W * np.conj(A), N_fft)[:len(a)]
    return 2.0 * len(a) * g

    n = len(a_init)
    N_fft = _next_pow2(3 * n + 1)

    s = a.sum()

        grad = _grad_smoothed(y, c_y, beta, N_fft)

        # We want: f(proj(y - step*grad)) <= f(y) + grad^T * (proj(y - step*grad) - y) + (1/2*step_inv)*||proj(y - step*grad) - y||^2

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_softmax(a, c, beta, N_fft):
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    e_sum = e.sum()
    if e_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / e_sum
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    g = np.fft.irfft(W * np.conj(A), N_fft)[:len(a)]
    return 2.0 * len(a) * g


def _grad_topk(a, c, K, N_fft):
    top_idx = np.argsort(c)[::-1][:K]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / K
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    g = np.fft.irfft(W * np.conj(A), N_fft)[:len(a)]
    return 2.0 * len(a) * g


def _fista_softmax(a_init, beta_start, beta_end, deadline, max_iter):

    N_fft = _next_pow2(3 * n + 1)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    a_prev = a.copy()
    t_val = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    num_betas = 25
    betas = np.geomspace(beta_start, beta_end, num_betas)
    it_per_beta = max(100, max_iter // num_betas)

    for bi, beta in enumerate(betas):
        step = 1.0 / (4.0 * n / beta)
        for it in range(it_per_beta):
            if time.monotonic() > deadline:
                return best_a, best_score
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a + ((t_val - 1.0) / t_next) * (a - a_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-30:
                y = y / sy
            c_y = _full_convolve(y, y)
            grad = _grad_softmax(y, c_y, beta, N_fft)
            a_next = _proj_simplex(y - step * grad)
            a_prev = a.copy()
            a = a_next
            t_val = t_next
            if (bi * it_per_beta + it) % 50 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score

def _adam_logspace(a_init, deadline, betas, lr_init, max_iter_total):
    n = len(a_init)
    N_fft = _next_pow2(3 * n + 1)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()
    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1
            e_z = z - z.max()
            e = np.exp(e_z)
            a = e / e.sum()
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e_w = np.exp(c_shifted)
            s_w = float(e_w.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e_w / s_w
            W = np.fft.rfft(w, N_fft)
            A = np.fft.rfft(a, N_fft)
            cross = np.fft.irfft(W * np.conj(A), N_fft)[:n]
            grad = 2.0 * n * a * cross
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 50 == 0:
                e_chk = z - z.max()
                a_chk = np.exp(e_chk) / np.exp(e_chk).sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            if t % 10000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)
    e_final = z - z.max()
    a_final = np.exp(e_final) / np.exp(e_final).sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, n_steps, deadline):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(123)

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        top_idx = np.argsort(c)[::-1][:10]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / len(top_idx)
        N_fft = _next_pow2(3 * n + 1)
        W = np.fft.rfft(w, N_fft)
        A = np.fft.rfft(a, N_fft)
        cross = np.fft.irfft(W * np.conj(A), N_fft)[:n]
        grad = 2.0 * n * a * cross
        # Find coordinate with largest negative gradient (most promising to increase)
        # and largest positive gradient (most promising to decrease)
        j_dec = int(np.argmax(grad))
        # Pick a different coordinate to increase
        mask = np.ones(n, dtype=bool)
        mask[j_dec] = False
        j_inc = int(np.argmax(np.where(mask, -grad, -np.inf)))
        if j_inc == j_dec:
            continue
        delta = a[j_dec] * 0.01 * (1.0 + 0.5 * np.exp(-0.01 * it))
        delta = min(delta, 0.5)
        a_try = a.copy()
        a_try[j_dec] -= delta
        a_try[j_inc] += delta
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        sc_try = _score_norm(a_try)
        if sc_try < best_score:
            best_score = sc_try
            best_a = a_try.copy()
            a = a_try
    return best_a, best_score


def _phase_randomize(a_init, n_trials, deadline, rng):
    n = len(a_init)
    N_fft = _next_pow2(2 * n - 1)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        A = np.fft.rfft(a_init, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)
        noise_scale = 0.1 + 0.05 * trial
        phase_new = phase + noise_scale * rng.standard_normal(len(phase))
        A_new = mag * np.exp(1j * phase_new)
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
    return best_a, best_score


def _make_initialization(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        a = np.arange(1, n + 1, dtype=np.float64)
        return a / a.sum()
    elif kind == "ramp2":
        a = np.arange(n, 0, -1, dtype=np.float64)
        return a / a.sum()
    elif kind == "gaussian":
        center = n / 2.0
        width = n / 6.0
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
        return a / a.sum()
    elif kind == "two_gaussian":
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - n * 0.3) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - n * 0.7) / (n * 0.1)) ** 2)
        return a / a.sum()
    elif kind == "cosine_taper":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        a = np.abs(rng.standard_normal(n))
        # Smooth it
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    if sc_adam_full < best_score:
        best_score = sc_adam_full
        best_a = a_adam.copy()
        print(f"  Adam: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Coordinate descent polish ===
    print("Phase 5: Coordinate descent")
    phase5_end = min(deadline, t_start + 1450.0)
    a_work = best_a.copy()
    s = a_work.sum()
    if s > 0:
        a_work = a_work / s
    a_cd, sc_cd = _coord_descent(a_work, 50000, phase5_end)
    sc_cd_full = _score(a_cd)
    if sc_cd_full < best_score:
        best_score = sc_cd_full
        best_a = a_cd.copy()
        print(f"  Coord descent: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Phase randomization escape + re-optimize ===
    print("Phase 6: Phase randomization")
    phase6_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase6_end:
        a_pert, sc_pert = _phase_randomize(best_a, 20, phase6_end - 60, rng)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"  Phase randomize improved: score={best_score:.10f}")
        # Quick Adam after perturbation
        if time.monotonic() < phase6_end - 10:
            a_work = best_a.copy()
            s = a_work.sum()
            if s > 0:
                a_work = a_work / s
            c_w6 = _full_convolve(a_work, a_work)
            max_c_w6 = float(c_w6.max())
            a_adam2, sc_adam2 = _adam_logspace(a_work, phase6_end,
                                               [0.0001 * max_c_w6, 0.00005 * max_c_w6],
                                               lr_init=0.008, max_iter_total=50000)
            sc_adam2_full = _score(a_adam2)
            if sc_adam2_full < best_score:
                best_score = sc_adam2_full
                best_a = a_adam2.copy()
                print(f"  Adam after randomize: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Ultra-fine Adam polish ===
    print("Phase 7: Ultra-fine polish")
    phase7_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase7_end - 20:
        a_work = best_a.copy()
        s = a_work.sum()
        if s > 0:
            a_work = a_work / s
        c_w7 = _full_convolve(a_work, a_work)
        max_c_w7 = float(c_w7.max())
        betas_ult = [0.00001 * max_c_w7, 0.000005 * max_c_w7, 0.000002 * max_c_w7]
        a_ult, sc_ult = _adam_logspace(a_work, phase7_end, betas_ult,
                                        lr_init=0.005, max_iter_total=200000)
        sc_ult_full = _score(a_ult)
        if sc_ult_full < best_score:
            best_score = sc_ult_full
            best_a = a_ult.copy()
            print(f"  Ultra-fine: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final coordinate descent ===
    if best_a is not None and time.monotonic() < deadline - 30:
        a_work = best_a.copy()
        s = a_work.sum()
        if s > 0:
            a_work = a_work / s
        a_cd2, sc_cd2 = _coord_descent(a_work, 30000, deadline - 10)
        sc_cd2_full = _score(a_cd2)
        if sc_cd2_full < best_score:
            best_score = sc_cd2_full
            best_a = a_cd2.copy()
            print(f"  Final CD: score={best_score:.10f}")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_a is not None:
        final = np.clip(best_a, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = final.sum()
            final = final / s
        # Ensure non-negative
        final = np.maximum(final, 0.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(512)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
