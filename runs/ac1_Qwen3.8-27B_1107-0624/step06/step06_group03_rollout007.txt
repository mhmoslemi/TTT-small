"""
Optimized step function search using multi-scale Adam with exact subgradient annealing,
spectral shaping, and coordinate descent polishing. Targets large n (16384, 8192)
for better attack surface, with aggressive top-K annealing and periodic power-spectrum
flattening to drive the autocorrelation toward uniformity.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _topk_weights_hard(c, k):
    """Hard top-k: uniform weight on top-k elements."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    topk_thresh = np.partition(c, -k)[-k]
    topk_mask = c >= topk_thresh
    n_active = topk_mask.sum()
    if n_active == 0:
        idx = int(np.argmax(c))
        w = np.zeros(n_c)
        w[idx] = 1.0
        return w
    w = np.zeros(n_c)
    w[topk_mask] = 1.0 / n_active
    return w


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _spectral_shaping(a, floor_frac, n_iter=3):
    """Flatten the power spectrum iteratively to reduce autocorrelation peaks."""
    n = len(a)
    N = _next_pow2(2 * n)
    a_work = a.copy()
    for _ in range(n_iter):
        A = np.fft.fft(a_work, N)
        P = np.abs(A) ** 2
        P_max = float(P.max())
        P_mean = float(P.mean())
        if P_max < 1e-30:
            break
        # Raise the floor: push spectrum toward flat
        P_new = P + floor_frac * (P_max - P)
        # Preserve total power (roughly)
        P_new = P_new * (P.sum() / max(P_new.sum(), 1e-30))
        # Reconstruct with original phase
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = np.sqrt(P_new) * phase
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_work = a_new / s
    return a_work


def _adam_logspace_v2(a_init, deadline, beta_start, beta_end, top_k_start, top_k_end,
                      lr_init, max_iter, spectral_every=5000, spectral_floor=0.3):
    """Adam in log-space with annealed top-K and periodic spectral shaping."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    total_betas = 20  # number of annealing stages
    n_iter = 0

    while n_iter < max_iter:
        if time.monotonic() > deadline:
            break
        n_iter += 1
        t += 1

        # Annealing schedule
        frac = min(1.0, n_iter / max(n_iter, max_iter * total_betas / total_betas))
        # Actually use a smooth annealing based on n_iter
        progress = min(1.0, n_iter / max(1, max_iter))
        beta = beta_start * (beta_end / beta_start) ** progress
        top_k = max(1, int(top_k_start * (top_k_end / top_k_start) ** (progress ** 0.7)))

        a = _softmax_normalize(z)
        c = _full_convolve(a, a)
        w = _topk_weights(c, beta, top_k)
        grad = _logspace_grad(a, w, n)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        # Learning rate decay
        if n_iter % 5000 == 0:
            current_lr = max(0.0001, current_lr * 0.95)

        # Periodic spectral shaping
        if spectral_every > 0 and n_iter % spectral_every == 0:
            a_curr = _softmax_normalize(z)
            a_shaped = _spectral_shaping(a_curr, spectral_floor, n_iter=2)
            z = np.log(np.maximum(a_shaped, 1e-15))
            z = z - z.mean()
            m_buf *= 0.3
            v_buf *= 0.3

        # Score check
        if n_iter % 100 == 0:
            a_chk = _softmax_normalize(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_exact_subgrad(a_init, deadline, lr_init, max_iter, spectral_every=3000):
    """Adam using exact subgradient (argmax of convolution)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for n_iter in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_normalize(z)
        c = _full_convolve(a, a)

        # Use top-3 for stability
        k_eff = min(3, len(c))
        w = _topk_weights_hard(c, k_eff)

        grad = _logspace_grad(a, w, n)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if n_iter % 3000 == 0:
            current_lr = max(0.00005, current_lr * 0.93)

        if spectral_every > 0 and n_iter % spectral_every == 0:
            a_curr = _softmax_normalize(z)
            a_shaped = _spectral_shaping(a_curr, 0.2, n_iter=2)
            z = np.log(np.maximum(a_shaped, 1e-15))
            z = z - z.mean()
            m_buf *= 0.5
            v_buf *= 0.5

        if n_iter % 200 == 0:
            a_chk = _softmax_normalize(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Coordinate descent polishing with careful 1D optimization."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(99999)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break

            a_i_old = a[i]
            if a_i_old < 1e-15:
                a[i] = 0.5 * np.median(a)
                a = a / a.sum()
                continue

            # Build the dependency structure
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]

            c = _full_convolve(a, a)
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old  # assuming normalized

            # Find candidate t values
            idx_top = np.argsort(c)[-top_k_w:]
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5,
                           a_i_old * 0.7, a_i_old * 0.3, a_i_old * 1.2,
                           a_i_old * 1.8, a_i_old * 0.1]

            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = (S0 - 2.0 * c_base[j] / d[j])
                    if -a_i_old < t_star < 5.0 * max(a_i_old, 0.1):
                        t_candidates.append(max(0, t_star))
                        t_candidates.append(max(0, t_star * 0.8))
                        t_candidates.append(max(0, t_star * 1.2))

            t_candidates = [max(0.0, min(5.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial_max = float((c_base + d * tc).max())
                denom = (S0 + tc) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * c_trial_max / denom
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if abs(best_t - a_i_old) > 1e-12:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc

    return a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_spec":
        # Initialize with flat power spectrum (random phase)
        N = _next_pow2(2 * n)
        rng = np.random.default_rng(7)
        P = np.ones(N // 2 + 1)
        # Ensure Hermitian symmetry for real signal
        A = np.zeros(N, dtype=complex)
        for k in range(N // 2 + 1):
            A[k] = np.exp(1j * rng.uniform(0, 2 * np.pi))
        for k in range(1, N // 2 + 1):
            A[N - k] = np.conj(A[k - 1]) if k - 1 < N // 2 else A[N-k]
        a = np.fft.ifft(A).real[:n]
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_mixture":
        t = np.linspace(0, 1, n)
        centers = np.array([0.15, 0.4, 0.6, 0.85])
        widths = np.array([0.08, 0.12, 0.1, 0.08])
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-0.5 * ((t - c) / w) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        parent = _structured_init(4096, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n=4096, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Main search: try multiple lengths
    # Priority: 16384 (best attack surface), 8192, 4096
    lengths_to_try = [16384, 8192]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break

        remaining = deadline - time.monotonic()
        print(f"\n{'='*60}")
        print(f"=== n={n_try} (remaining: {remaining:.1f}s) ===")
        print(f"{'='*60}")

        # Initialize
        if n_try == len(parent):
            a0 = parent.copy()
        else:
            a0 = _interp_profile(parent, n_try)

        sc0 = _score(a0)
        print(f"  Interpolated init: score={sc0:.10f}")
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Also try structured inits
        inits = [
            ("interp", a0),
            ("cos_sum", _structured_init(n_try, "cos_sum")),
            ("tapered", _structured_init(n_try, "tapered")),
            ("flat_spec", _structured_init(n_try, "flat_spec")),
            ("gauss_mix", _structured_init(n_try, "gaussian_mixture")),
        ]

        # Pick the best initial point
        a_best_init = a0
        sc_best_init = sc0
        for name, a_init in inits[1:]:
            sc_i = _score(a_init)
            print(f"  Init '{name}': score={sc_i:.10f}")
            if sc_i < sc_best_init:
                sc_best_init = sc_i
                a_best_init = a_init
                best_score = sc_i
                best_seq = a_init.copy()

        print(f"  Best init: score={sc_best_init:.10f}")

        # Phase 1: Coarse Adam (large top-K, moderate beta)
        phase1_end = time.monotonic() + min(remaining * 0.3, 400.0)
        c_curr = _full_convolve(a_best_init, a_best_init)
        max_cc = float(c_curr.max())

        a_p1, sc_p1 = _adam_logspace_v2(
            a_best_init, phase1_end,
            beta_start=0.05 * max_cc,
            beta_end=0.001 * max_cc,
            top_k_start=max(100, n_try // 16),
            top_k_end=max(10, n_try // 256),
            lr_init=0.02,
            max_iter=500000,
            spectral_every=20000,
            spectral_floor=0.3
        )
        sc_p1_full = _score(a_p1)
        print(f"  Phase 1 (coarse Adam): score={sc_p1_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_p1_full < best_score:
            best_score = sc_p1_full
            best_seq = a_p1.copy()
            a_best_init = a_p1.copy()

        # Phase 2: Medium Adam (smaller top-K, smaller beta)
        if time.monotonic() > deadline - 90:
            break
        phase2_end = time.monotonic() + min((deadline - time.monotonic()) * 0.3, 300.0)
        c_curr = _full_convolve(a_best_init, a_best_init)
        max_cc = float(c_curr.max())

        a_p2, sc_p2 = _adam_logspace_v2(

        # Phase 3: Exact subgradient Adam (top-3, very small effective beta)
        if time.monotonic() > deadline - 60:

            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()

    # Phase 5: Diversity restarts from best
    if time.monotonic() < deadline - 30 and best_seq is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Diversity restarts (remaining: {remaining:.1f}s) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        n_restarts = max(2, int(remaining / 60))
        per_restart = (remaining - 10) / max(1, n_restarts)

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            if ri == 0:
                p_start = a_base + 0.01 * rng.standard_normal(n_best)
            elif ri == 1:

                p_start = np.roll(a_base, shift).copy()
            elif ri == 2:
                freq = rng.uniform(0.5, 5.0)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert
            elif ri == 3:
                # Spectral perturbation
                N = _next_pow2(2 * n_best)
                A = np.fft.fft(a_base, N)
                P = np.abs(A) ** 2
                rng.shuffle(P)
                phase = A / np.maximum(np.abs(A), 1e-30)
                A_new = np.sqrt(P) * phase
                p_start = np.fft.ifft(A_new).real[:n_best]
            elif ri == 4:
                # Block shuffle
                block = max(1, n_best // 16)
                p_start = np.zeros(n_best)
                indices = np.arange(0, n_best, block)
                rng.shuffle(indices)
                for idx in indices:
                    end = min(idx + block, n_best)
                    p_start[idx:end] = a_base[idx:end]
            else:

            p_start = np.maximum(p_start, 1e-12)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())

            # Quick Adam polish
            a_r, sc_r = _adam_logspace_v2(
                p_start, time.monotonic() + max(5, per_restart * 0.6),

        sc_final = _score(final)
        print(f"Verified final score: {sc_final:.10f}, n={len(final)}")

    # Fallback
    fallback = np.ones(1024)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
