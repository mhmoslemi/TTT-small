        a_rc_mix = np.maximum(a_rc_mix, 1e-15)
        a_rc_mix = a_rc_mix / a_rc_mix.sum()
        restarts[2] = ("raised_cosine_mix", np.log(a_rc_mix) - np.log(a_rc_mix).mean())

        a_ed = _make_profile(n_main, "exponent_decay")
        a_ed_mix = 0.6 * _softmax(z_main) + 0.4 * a_ed
        a_ed_mix = np.maximum(a_ed_mix, 1e-15)
        a_ed_mix = a_ed_mix / a_ed_mix.sum()
        restarts[3] = ("exp_decay_mix", np.log(a_ed_mix) - np.log(a_ed_mix).mean())

        for name, z_r in restarts:
            if time.monotonic() > phase4_end - 15:
                break
            remaining = phase4_end - time.monotonic() - 15
            per_r = max(20.0, remaining / 4)
            print(f"  Restart: {name}, t={time.monotonic()-t_start:.1f}s")
            z_r, a_r, sc_r = _continuation_opt(z_r, time.monotonic() + per_r * 0.6, n_betas=20, beta_decay=0.4, max_iter_per=2000)
            # Adam polish
            if time.monotonic() < phase4_end - 15:
                c_r = _full_convolve(a_r, a_r)
                bt_r = 0.0003 * float(c_r.max())
                z_r, a_r, sc_r = _adam_fine(z_r, time.monotonic() + per_r * 0.4, bt_r, lr_init=0.003, max_iter=50000)
            sc_r_actual = _score(a_r)
            if sc_r_actual < best_score:
                best_score = sc_r_actual
                best_seq = a_r.copy()
                z_main = z_r
                print(f"  Restart {name} IMPROVED: score={sc_r_actual:.10f}, t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {name}: score={sc_r_actual:.10f} (no improvement)")

    # === Phase 5: Try n=8192 ===
    print("Phase 5: Try n=8192")
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end - 60:
        n_big = 8192
        # Upsample parent to n_big
        from numpy import interp
        t_old = np.linspace(0, 1, n_main)
        t_new = np.linspace(0, 1, n_big)
        a_big = interp(t_new, t_old, _softmax(z_main))
        a_big = np.maximum(a_big, 1e-15)
        a_big = a_big / a_big.sum()
        z_big = np.log(a_big) - np.log(a_big).mean()
        sc_big = _score(a_big)
        print(f"  n=8192 upsampled: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()
            print(f"  IMPROVED with n=8192: {best_score:.10f}")

        remaining = phase5_end - time.monotonic() - 30
        if remaining > 30:
            z_big, a_big, sc_big = _continuation_opt(z_big, time.monotonic() + remaining * 0.7,
                                                      n_betas=25, beta_decay=0.4, max_iter_per=2500)
            if time.monotonic() < phase5_end - 10:
                c_b = _full_convolve(a_big, a_big)
                z_big, a_big, sc_big = _adam_fine(z_big, phase5_end - 5,
                                                    0.0002 * float(c_b.max()),
                                                    lr_init=0.003, max_iter=100000)
            sc_big_actual = _score(a_big)
            if sc_big_actual < best_score:
                best_score = sc_big_actual
                best_seq = a_big.copy()
                print(f"  n=8192 optimized: score={sc_big_actual:.10f} IMPROVED!")
            else:
                print(f"  n=8192 optimized: score={sc_big_actual:.10f} (no improvement)")

    # === Phase 6: Final ultra-polish on best ===
    print("Phase 6: Final ultra-polish")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        z_pol = np.log(np.maximum(a_pol, 1e-20))
        z_pol = z_pol - z_pol.mean()
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            # Adam with tiny beta
            t_split = remaining * 0.5
            z_pol, a_pol, sc_pol = _adam_fine(z_pol, time.monotonic() + t_split,
                                                5e-6 * max_cp, lr_init=0.001, max_iter=200000)
            sc_pol = _score(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_pol:.10f}")
            # L-BFGS
            if time.monotonic() < deadline - 8:
                c_p = _full_convolve(a_pol, a_pol)
                z_pol = _lbfgs_run(z_pol, 3e-6 * float(c_p.max()), deadline - 3.0, max_iter_per=10000)
                a_pol = _softmax(z_pol)
                sc_pol = _score(a_pol)
                if sc_pol < best_score:
                    best_score = sc_pol
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_pol:.10f}")

    # Final cleanup
    print(f"\nDone: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
