"""
LP-guided optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: LP-based active set direction finding with line search,
adaptive subgradient with momentum, multi-scale refinement, and
improved L-BFGS scheduling with log-sum-exp smoothing.

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_active(a, c, n, tol_frac):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 60,
                                   'maxls': 60, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False


def _lbfgs_schedule(a_start, deadline, min_remaining=30.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())

    betas = max_c * np.array([0.12, 0.06, 0.03, 0.015, 0.008,

                              0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
    current_a = a_start.copy()

    for i, beta in enumerate(betas):

def _lp_direction(a, active_idx, c_vals, n):
    k = len(active_idx)
    if k == 0:
        return np.zeros(n), 0.0, False

    A = np.zeros((k, n + 1))
    b = np.zeros(k)

    for r, j in enumerate(active_idx):
        i_start = max(0, j - n + 1)
        i_end = min(j, n - 1)
        if i_end >= i_start:
            idx = np.arange(i_start, i_end + 1)
            A[r, idx] = 2.0 * a[j - idx]
        A[r, n] = -1.0
        b[r] = -c_vals[j]

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    except Exception:
        pass

    return np.zeros(n), np.inf, False


def _line_search(a, direction, n, alpha_max, current_score, max_evals=30):
    if alpha_max <= 1e-15:
        return 0.0
    best_alpha = 0.0
    best_sc = current_score
    lo = 0.0
    hi = alpha_max

    for _ in range(max_evals):
        mid = 0.5 * (lo + hi)
        a_try = a + mid * direction
        if a_try.min() < -1e-12:
            hi = mid
            continue
        a_try = np.maximum(a_try, 0.0)
        sc = _score(a_try)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_alpha = mid
        if sc < current_score:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-8 * max(1.0, hi):
            break

    return best_alpha

def _subgrad_linesearch(a_init, deadline, tol_frac=1e-4, momentum_decay=0.9):

    z = z - z.mean()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()

            try_update(a_r, f"restart_{name}")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Length sweep (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40.0:
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 30:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_schedule(a_interp, deadline, min_remaining=15.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5.0:
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break
            a_l, _ = _local_search(best_seq, deadline - 5, max_rounds=200)
            try_update(a_l, f"final_ls_{rnd}")
            if time.monotonic() > deadline - 5:
                break
            a_s, _ = _subgrad_linesearch(best_seq, deadline - 5, tol_frac=5e-5)
            try_update(a_s, f"final_sg_{rnd}")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"sum={s:.6f}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
