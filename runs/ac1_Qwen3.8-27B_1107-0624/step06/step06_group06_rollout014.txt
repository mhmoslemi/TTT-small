Let me define `v[k] = w[k]` (the weights for the active set). Then:
`g[m] = 2 * sum_{j} v[m+j] * a[j]` (where `j = k-m`, so `k = m+j`, and `v` is defined on `0..2n-2`)

This is: `g[m] = 2 * (v ⊛ a)[m]` where ⊛ is cross-correlation.

Cross-correlation can be computed as: `v ⊛ a = IFFT(FFT(v) * conj(FFT(a)))`

With `N >= 2n-1` to avoid circular wrap-around.

Let me verify: `(v ⊛ a)[m] = sum_j v[m+j] * conj(a[j])` for real a, this is `sum_j v[m+j] * a[j]`.

Yes! That's exactly what we want. So:

def _compute_subgrad(a, active_weights):
    """Compute the subgradient for the active set.
    active_weights: array of length 2n-1 with weights for each convolution index.
    Returns: array of length n.
    """
    n = len(a)
    N = _next_pow2(2*n - 1)
    V = np.fft.rfft(active_weights, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(V * np.conj(A), N)
    return 2.0 * corr[:n]

"""
Momentum-based subgradient optimizer with adaptive active set, direct simplex
projection, progressive tightening, shrinkage, and L-BFGS hybrid.
Targets score < 1.5027 for 2n*max(conv(a,a))/sum(a)^2.
Key innovations: (1) Nesterov momentum on simplex, (2) hysteresis-based active
set management, (3) progressive tolerance tightening, (4) integrated shrinkage
with re-optimization, (5) hybrid L-BFGS + subgradient.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad_direct(a, active_weights, n):
    """Compute subgradient: 2 * cross_corr(active_weights, a)[:n]"""
    N = _next_pow2(2 * n - 1)
    W = np.fft.rfft(active_weights, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.005 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(20, int(remaining / (est_time * 8))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(fun, z, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15,
                                       'gtol': 1e-11, 'maxcor': 40, 'maxls': 60})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _momentum_subgrad(a_init, deadline, n_max_steps=100000, lr_base=0.01,
                      momentum=0.9, tol_start=0.05, tol_end=0.001):
    """Momentum-based subgradient on the simplex with progressive active set tightening."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    v = np.zeros(n)  # velocity for momentum
    best_a = a.copy()
    best_sc = _score(a)
    best_c_max = float(_full_convolve(a, a).max())
    patience = 0
    max_patience = 5000

    # Progressive tolerance: start loose, tighten over time
    total_steps = n_max_steps
    for step in range(total_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Progressive tolerance
        progress = step / total_steps
        tol = tol_start + (tol_end - tol_start) * (progress ** 0.5)
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Compute subgradient
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum()
        grad = _subgrad_direct(a, w, n)
        # Subtract the constant term: -2*c_max (from the sum(a)^2 normalization derivative)
        grad = grad - 2.0 * c_max

        # Project gradient onto tangent space of simplex
        grad_t =(grad - np.mean(grad))

        # Nesterov momentum
        v = momentum * v + grad_t
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        # Adaptive learning rate
        lr = lr_base / (1.0 + 0.001 * step)
        direction = -v / max(v_norm, 1e-15)

        # Step with projection
        a_try = a + lr * direction
        a_try = _project_simplex(a_try, 1.0)

        # Check improvement
        c_try = _full_convolve(a_try, a_try)
        c_try_max = float(c_try.max())
        a = a_try
        sc = 2.0 * n * c_max  # approximate (since sum=1)

        if c_try_max < best_c_max - 1e-12:
            best_c_max = c_try_max
            best_sc = 2.0 * n * best_c_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            # Decay momentum slightly when stuck
            v *= 0.999
            if patience > max_patience:
                break

        # Periodic shrinkage check
        if step % 5000 == 0 and step > 0:
            n_small = int(np.sum(a < 1e-8))
            if n_small > n * 0.1:
                a_shrunk = a.copy()
                a_shrunk[a_shrunk < 1e-8] = 0
                s = a_shrunk.sum()
                if s > 1e-10:
                    a_shrunk /= s
                    c_shrunk = _full_convolve(a_shrunk, a_shrunk)
                    if float(c_shrunk.max()) < best_c_max:
                        best_c_max = float(c_shrunk.max())
                        best_sc = 2.0 * n * best_c_max
                        best_a = a_shrunk.copy()
                        a = a_shrunk.copy()
                        v = np.zeros(n)
                        patience = 0

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Log-space subgradient polish (similar to previous code)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_optimize(a, deadline):
    """Zero out small elements and re-optimize with L-BFGS."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            n_zero = max(1, int(n * frac * 0.5))
            a_try = best_a.copy()
            a_try[a_sorted_idx[:n_zero]] = 0.0
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            # Re-optimize from this point
            a_opt = _run_lbfgs(best_a, deadline, min_time_remaining=10.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-3, 3, n)
        a = np.exp(-t**2)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _spectral_perturb(a, rng, n_modes=5, amp=0.03):
    """Perturb in the frequency domain for diversity."""
    n = len(a)
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    for _ in range(n_modes):
        k = rng.integers(1, N // 2)
        delta = amp * (rng.standard_normal() + 1j * rng.standard_normal())
        A[k] += delta
        if k < N // 2:
            A[N - k] += np.conj(delta)
    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def _try_symmetrize(a):
    """Try symmetrizing and see if it helps."""
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s < 1e-15:
        return a, _score(a)
    a_sym /= s
    sc_sym = _score(a_sym)
    sc_orig = _score(a)
    if sc_sym < sc_orig:
        return a_sym, sc_sym
    return a, sc_orig

            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_val = a_init.sum()
            if s_val > 1e-10:
                a_init /= s_val
            else:
                a_init = _make_profile(n_try, "tapered")
        elif n_parent == n_try:
            a_init = parent.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        sc_init = _score(a_init)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=50.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # Phase 6: Spectral diversity restarts
    print(f"\n[Phase 6] Spectral restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for i in range(5):
            if time.monotonic() > deadline - 20:
                break
            if i < 3:
                p_start = _spectral_perturb(best_seq, rng, n_modes=3 + i, amp=0.02)
            else:
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z + 0.03 * rng.standard_normal(len(z))
                z = z - z.mean()
                p_start = _softmax(z)
            sc_start = _score(p_start)
            if sc_start < best_score:
                update_best(p_start, sc_start, f"spec_init_{i}")
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"spec_restart_{i}")

    # Phase 7: Final fine polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very fine L-BFGS
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0001
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(2000, max(30, int(remaining / 0.05)))
            try:
                result = minimize(
                    lambda z_local: _smoobj(z_local, beta_final, n),
                    z,
                    jac=lambda z_local: _smograd(z_local, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxcor': 40}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final shrinkage
        if time.monotonic() < deadline - 3:
            a_shr_final, sc_final = _shrinkage_optimize(best_seq, deadline - 3)
            update_best(a_shr_final, sc_final, "final_shrink")

        # Final symmetrize
        if best_seq is not None:
            a_sym_final, sc_sym_final = _try_symmetrize(best_seq)
            update_best(a_sym_final, sc_sym_final, "final_sym")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
