"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA, Adam log-space, coordinate polishing, and multi-scale exploration.
Explores n=4096, 8192, 16384, and potentially 32768.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1: 2n * max(conv)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_simplex(a, w, N_fft=None):
    """Compute gradient of softmax-smoothed max-conv on simplex.
    a: sequence (length n), w: weight distribution (length 2n-1)
    Returns gradient of length n.
    """
    n = len(a)
    if N_fft is None:
        N_fft = _next_pow2(2 * n)
    A = np.fft.rfft(a, N_fft)
    W = np.fft.rfft(w, N_fft)
    # corr[k] = sum_j a[j] * w[j+k] (linear cross-correlation)
    corr = np.fft.irfft(W * np.conj(A), N_fft)[:n]
    return 2.0 * corr

def _fista(a_init, beta, deadline, max_iter=100000):
    """FISTA with softmax-smoothed max-conv on simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0
    N_fft = _next_pow2(2 * n)
    
    L = 4.0 * n / beta  # Lipschitz constant (upper bound)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(len(c_y))
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum
        
        grad = _compute_grad_simplex(y, w, N_fft)
        step_size = step
        a_next = _proj_simplex(y - step_size * grad)
        
        if it % 500 == 0 or it < 10:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = _next_pow2(2 * n)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute current convolution
        c = _full_convolve(a, a)
        
        # Random permutation of coordinates
        order = np.random.permutation(n)
        
        for i in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            
            old_ai = a[i]
            S = s - old_ai  # sum of all other elements
            
            # The max convolution value as a function of a_i:
            # For each k, c[k] changes by:
            #   2*(a[k-i] if 0<=k-i<n and k!=2i, or 1 if k==2i) * (new_ai - old_ai)
            # More precisely: new_c[k] = c[k] - 2*a[k-i]*old_ai + 2*a[k-i]*new_ai (for valid k-i, k!=2i)
            # For k=2i: new_c[2i] = c[2i] - old_ai^2 + new_ai^2
            
            # Simple approach: try a few candidate values and pick the best
            # Candidate 1: reduce if i is at a "hot" position
            # The "pressure" on coordinate i is proportional to sum_k w[k]*a[k-i]
            # where w is concentrated near the max
            
            cm = float(c.max())
            # Pressure on coordinate i: how much a[i] contributes to the max
            # d(c[k])/d(a_i) = 2*a[k-i] (for 0<=k-i<n)
            # So the "pressure" is sum_{k near max} 2*a[k-i] * (indicator k is max)
            
            # Simplified: the pressure is approximately 2 * a[i] * (something)
            # Let's just try: new_ai = old_ai * factor, for a few factors
            # Or: new_ai = old_ai - step (and project to >= 0)
            
            # Better: use the subgradient direction
            # The subgradient of max(c) at coordinate i is:
            # 2 * sum_{k in argmax} a[k-i] / |argmax|
            top_k = min(10, 2 * n - 1)
            idx = np.argsort(c)[::-1][:top_k]
            vals = c[idx] - cm + 1e-10
            weights = vals ** 2
            weights = weights / weights.sum()
            w = np.zeros(2 * n - 1)
            w[idx] = weights
            cross_i = 0.0
            for j in range(n):
                k = i + j
                if k < 2 * n - 1:
                    cross_i += w[k] * a[j]
            pressure = 2.0 * cross_i
            
            # Step in negative pressure direction
            step_size = 0.1 / (1.0 + 0.01 * sweep)
            new_ai = old_ai - step_size * pressure
            new_ai = max(0.0, new_ai)
            
            # Tentatively update and check
            a[i] = new_ai
            s_new = s - old_ai + new_ai
            # Quick check: does this reduce the max?
            # (Full convolution update would be expensive, so use approximate check)
            # The change in c[k] is approximately 2*a[k-i]*(new_ai - old_ai)
            delta = 2.0 * (new_ai - old_ai)
            # Max change is approximately max_{k} a[k-i] * delta (for the relevant range)
            
            # For efficiency, just accept and do a full check every 10 coordinates
            s = s_new
        
        # Full check and normalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy() / p.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s > 0:
        p_new = p_new / s
    else:
        p_new = np.ones(n_new) / n_new
    return p_new

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)
