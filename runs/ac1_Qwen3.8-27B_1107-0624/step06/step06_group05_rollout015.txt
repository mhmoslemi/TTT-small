"""
Convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: FISTA with aggressive beta continuation, mirror descent with Nesterov momentum,
projected subgradient, coordinate descent polish, and L-BFGS refinement.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_subgradient(a, w):
    """Compute subgradient of max_k conv(a,a)[k] given weights w over active set."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _get_active_weights(c, tol_rel=1e-8):
    """Get uniform weights over the active set (positions near max)."""
    c_max = float(c.max())
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, c_max
    threshold = c_max - tol_rel * c_max
    active = np.where(c >= threshold)[0]
    w = np.zeros_like(c)
    w[active] = 1.0 / len(active)
    return w, c_max


def _get_active_weights_soft(c, beta):
    """Get softmax weights over convolution values."""
    c_max = float(c.max())
    c_sh = (c - c_max) / beta
    e = np.exp(c_sh)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, c_max


def _fista_opt(a_init, beta, deadline, max_iter):
    """FISTA with smoothed objective on the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w, _ = _get_active_weights_soft(c_y, beta)
        grad = _compute_subgradient(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 60 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

    n = len(a_init)

    a = a / a.sum()

    best_a = a.copy()

        grad = _compute_subgradient(a, w)

        eta = eta_base / np.sqrt(t_val)
        a_candidate = _proj_simplex(a - eta * grad)

        sc = _score_norm(a_candidate)
        if sc < _score_norm(a):

def _coord_descent_polish(a_init, deadline, tol_rel=1e-7):
    """Coordinate descent: for each coordinate, find the optimal value via bisection
    while maintaining the simplex constraint by proportionally scaling others."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_score = _score_norm(a)
    best_a = a.copy()

    # Precompute the "contribution vector" for each coordinate
    # For coordinate j, changing a_j to x (and scaling others to maintain sum=1):
    # a_i_new = a_i * (1-x)/(1-a_j) for i != j, a_j_new = x
    # This is complex, so instead use a simpler approach:
    # Just do small perturbations and keep the best.

    # Simpler: for each coordinate j, try moving mass from j to the coordinate with
    # the most "room" (lowest gradient contribution).
    # Actually, let's do: for each j, find x in [0, min(3*a_j, 0.1)] that minimizes
    # the objective, where we adjust all other coordinates proportionally.

    # Even simpler: use the subgradient to determine which direction to move.
    # This is essentially gradient descent with coordinate-wise updates.

    c = _full_convolve(a, a)
    iterations = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        w, c_max = _get_active_weights(c, tol_rel)
        grad = _compute_subgradient(a, w)

        # Find the coordinate with the most negative gradient (most beneficial to increase)
        # and the one with positive gradient (most beneficial to decrease)
        # But we need to maintain the simplex constraint.
        # Use: move mass from argmax(grad) to argmin(grad)
        j_dec = int(np.argmax(grad))
        j_inc = int(np.argmin(grad))

        if j_dec == j_inc:
            break

        # Try a small move
        delta = min(a[j_dec], 0.05 * max(a).item())
        a_trial = a.copy()
        a_trial[j_dec] -= delta
        a_trial[j_inc] += delta

        # Check if this improves
        c_trial = _full_convolve(a_trial, a_trial)
        if c_trial.max() < c.max() * (1 - 1e-12):
            a = a_trial
            iterations += 1
        else:
            # Try smaller delta
            delta = delta * 0.3
            if delta < 1e-15:
                # Try all pairs? Too expensive. Just move on.
                break
            a_trial = a.copy()
            a_trial[j_dec] -= delta
            a_trial[j_inc] += delta
            c_trial = _full_convolve(a_trial, a_trial)
            if c_trial.max() < c.max() * (1 - 1e-12):
                a = a_trial
                iterations += 1
            else:
                break

        if iterations % 500 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    """Adam optimizer in log-space (softmax parameterization)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    """Scipy L-BFGS-B on log-space with fixed beta smoothing."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 40)), 'gtol': 1e-13, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _frank_wolfe(a_init, deadline, tol_rel=1e-7):
    """Frank-Wolfe (conditional gradient) with exact line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_score = _score_norm(a)
    best_a = a.copy()

    t_val = 1

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        w, c_max = _get_active_weights(c, tol_rel)
        grad = _compute_subgradient(a, w)

        # Linear minimization over simplex: pick the vertex with minimum gradient
        j_star = int(np.argmin(grad))
        e_j = np.zeros(n)
        e_j[j_star] = 1.0

        # Exact line search
        best_gamma = 0.0
        best_val = c_max

        # Try a few gamma values (geometric sequence)
        for g in [2.0 / (t_val + 2.0), 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - g) * a + g * e_j
            c_trial = _full_convolve(a_trial, a_trial)
            val = float(c_trial.max())
            if val < best_val:
                best_val = val
                best_gamma = g

        if best_gamma > 0:
            a = (1.0 - best_gamma) * a + best_gamma * e_j

        if it % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        t_val += 1

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score

        c_curr = _full_convolve(best_seq, a_work)  # just use best_seq
        max_cc = float(_full_convolve(best_seq, best_seq).max())
        beta_lb = 0.00002 * max_cc
        a_lb, _ = _lbfgs_opt(best_seq, time.monotonic() + min(phase6_end - time.monotonic() - 5, 180.0), beta_lb)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = best_seq

    # === Phase 7: Multi-restart exploration ===
    print("=== Phase 7: Multi-restart ===")
    phase7_end = min(deadline, t_start + 1740.0)
    if time.monotonic() < phase7_end - 20:
        n_restarts = 5
        per_r = (phase7_end - time.monotonic() - 15) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            if i == 0:
                p = _perturb(a_work, 0.01, rng)
            elif i == 1:
                p = _perturb(a_work, 0.02, rng)
            elif i == 2:
                # Shift
                shift = rng.integers(1, max(2, n_main // 256))
                p = np.roll(a_work, shift).copy()
                p = p / p.sum()
            elif i == 3:
                # Mix with raised cosine
                t = np.linspace(0, 1, n_main)
                rc = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
                rc = np.maximum(rc, 1e-10)
                rc = rc / rc.sum()
                p = 0.7 * a_work + 0.3 * rc
                p = p / p.sum()
            else:
                p = _perturb(a_work, 0.03, rng)

            # Quick FISTA + Adam
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.001 * max_cp
            time_per_stage = per_r * 0.4
            p, _ = _fista_opt(p, b_r, time.monotonic() + time_per_stage, 30000)
            b_r2 = 0.0001 * float(_full_convolve(p, p).max())
            p, _ = _adam_opt(p, time.monotonic() + time_per_stage, b_r2, 0.003, 60000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-polish ===
    print("=== Phase 8: Final ultra-polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            # Final Adam with very tiny beta
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + per_t * 0.5,
                                 0.000002 * max_cp, 0.001, 200000)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            # L-BFGS on remaining time
            if time.monotonic() < deadline - 8:
                c_f = _full_convolve(best_seq, best_seq)
                b_f = 0.000001 * float(c_f.max())
                a_pol, _ = _lbfgs_opt(best_seq, deadline - 3.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")
            # One more mirror descent pass
            if time.monotonic() < deadline - 5:
                a_md, _ = _mirror_descent_nesterov(best_seq, deadline - 3.0, tol_rel=1e-10)
                sc_md = _score(a_md)
                if sc_md < best_score:
                    best_score = sc_md
                    best_seq = a_md.copy()
                    print(f"  Final MirrorDescent: score={sc_md:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
