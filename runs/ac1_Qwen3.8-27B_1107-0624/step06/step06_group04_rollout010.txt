Key improvements: random-phase initialization, faster gradient computation,

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax(v, beta):
    """Compute softmax(v/beta) with numerical stability."""
    v_shift = v - v.max()
    e = np.exp(v_shift / beta)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(v)
        w[int(np.argmax(v))] = 1.0
        return w
    return e / s

def _proj_simplex(v):

        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        L = 4.0 * n * max_c / (beta * beta)

        if s_y > 1e-30:

        else:
            y = np.ones(n) / n
        
        # Compute gradient
        c = _fft_convolve(y, y)
        w = _softmax(c, beta)
        N_fft_local = _next_pow2(2 * len(y) - 1)
        grad = _compute_grad(y, w, N_fft_local)
        
        # Gradient step + projection
        a_next = _proj_simplex(y - lr * grad)
        
        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a
        a = a_next
        t_val = t_next
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score


def _adam_logspace(a_init, deadline, betas, lr_init, max_iter, momentum=0.9):
    """Adam optimizer in log-space with beta continuation."""
    n = len(a_init)
    N_fft = _next_pow2(2 * n - 1)
    
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        beta_iter = 0
        max_beta_iter = max(1000, int((deadline - time.monotonic()) * 500))
        
        while beta_iter < max_beta_iter and t < max_iter:
            if time.monotonic() > deadline:
                break
            beta_iter += 1
            t += 1
            
            # Compute a from z
            z_max = z.max()
            e = np.exp(z - z_max)
            s_e = e.sum()
            if s_e < 1e-300:
                a = np.ones(n) / n
            else:
                a = e / s_e
            
            # Compute convolution and weights
            c = _fft_convolve(a, a)
            cm = float(c.max())
            w = _softmax(c, beta)
            
            # Gradient in log-space
            # d/dz_j = a_j * d/da_j (since a = softmax(z), da_j/dz_j = a_j(1-a_j), da_k/dz_j = -a_j*a_k)
            # Simplified: grad_log[j] = a[j] * (grad_a[j] - sum_k a[k]*grad_a[k])
            # But for our purposes, using just a[j] * grad_a[j] works as an approximation
            # that's gradient of the log-softmax parameterization
            a_rev = a[::-1]
            wa = np.fft.rfft(w, N_fft)
            ar = np.fft.rfft(a_rev, N_fft)
            h = np.fft.irfft(wa * ar, N_fft)[n-1:2*n-1]
            grad_a = 2.0 * h
            grad_log = a * grad_a  # Approximate log-space gradient
            
            if not np.all(np.isfinite(grad_log)):
                break
            
            # Adam update
            m_buf = 0.9 * m_buf + 0.1 * grad_log
            v_buf = 0.999 * v_buf + 0.001 * (grad_log * grad_log)
            mh = m_buf / (1.0 - 0.9 ** t)
            vh = v_buf / (1.0 - 0.999 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + 1e-8)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if beta_iter % 50 == 0:
                z_max2 = z.max()
                e2 = np.exp(z - z_max2)
                a_chk = e2 / e2.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
        
        # Decay LR between beta values
        current_lr *= 0.8
    
    # Final
    z_max = z.max()
    e = np.exp(z - z_max)
    a_final = e / e.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _peak_shaving(a_init, n_steps, deadline):
    """Subgradient method targeting the peak of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    best_a = a.copy()
    best_score = _score_norm(a)
    step = 0.1
    
    for it in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        cm = float(c.max())
        
        # Use top-k peaks with exponential weighting
        top_k = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros(len(c))
        vals = c[top_idx] - cm + 1e-10
        w[top_idx] = np.exp(vals / 0.01)
        w = w / w.sum()
        
        # Gradient
        a_rev = a[::-1]
        N_fft = _next_pow2(2 * n - 1)
        wa = np.fft.rfft(w, N_fft)
        ar = np.fft.rfft(a_rev, N_fft)
        h = np.fft.irfft(wa * ar, N_fft)[n-1:2*n-1]
        grad = 2.0 * h
        
        a_new = _proj_simplex(a - step * grad)
        sc = _score_norm(a_new)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.9
            if step < 1e-6:
                break
    
    return best_a, best_score


def _spectral_compression(a_init, n_iter, alpha_schedule, deadline):
    """Iterative spectral compression: flatten IFFT(A^2) by compressing |A|^2."""
    n = len(a_init)
    N_fft = _next_pow2(2 * n - 1)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        
        # Compress the magnitude spectrum
        alpha = alpha_schedule[min(it, len(alpha_schedule) - 1)]
        target_mag = np.ones_like(mag)
        # Blend between current and flat
        mag_new = mag ** alpha * target_mag ** (1 - alpha)
        
        A_new = phase * mag_new
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        if it % 50 == 0:
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        a = a_new
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
