def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adaptive_active_set(c, tol_frac_base=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    sorted_c = np.sort(c)[::-1]
    n_active_est = max(3, int(0.02 * len(c)))
    if n_active_est >= len(c):
        n_active_est = len(c)
    threshold_from_rank = sorted_c[min(n_active_est - 1, len(c) - 1)]
    tol_rank = c_max - threshold_from_rank
    tol_fixed = tol_frac_base * c_max
    tol = max(tol_fixed, min(tol_rank, 0.5 * c_max))
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    return w


def _gradient_active_direct(a, c, n, w):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _smoobj_direct(a, beta, n):
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max / (a.sum() ** 2)
    return 2.0 * n * (c_max + beta * np.log(s)) / (a.sum() ** 2)


def _lbfgs_run_fine(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:

                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

        best_eta = eta

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)

    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):

def _perturb_for_restart(a, rng, mode="gauss", scale=0.03):

    elif mode == "fourier":
        t = np.arange(n)

        for _ in range(n_modes):
            freq = int(rng.integers(1, max(2, n // 10)))
            amp = 0.003 * rng.uniform(0.5, 2.5)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        block_size = max(8, n // 16)
        for _ in range(8):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-0.01, 0.01)
    z = z - z.mean()
    return _softmax(z)

    if best_seq is not None and time.monotonic() < deadline - 50:
        a_cd, _ = _coordinate_descent(best_seq, deadline - 50, max_rounds=10000)
        try_update(a_cd, "coord_desc")
    print(f"  Phase 6 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 7: Multi-restart with polish ===
    print(f"\n=== Phase 7: Multi-restart ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            ("gauss_s2", "gauss", 0.02),
            ("gauss_s5", "gauss", 0.05),
            ("gauss_s10", "gauss", 0.10),
            ("fourier", "fourier", 0.0),
            ("block", "block", 0.0),
        ]
        for name, mode, scale in restart_configs:
            if time.monotonic() > deadline - 30:
                break
            p_start = _perturb_for_restart(best_seq, rng, mode, scale)
            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            a_r = _lbfgs_fine_schedule(p_start, deadline - 20, min_time_remaining=15.0)
            try_update(a_r, f"restart_{name}")

            if time.monotonic() < deadline - 15:
                a_r2, _ = _peak_shaving(a_r, deadline - 15, max_iters=5000)
                try_update(a_r2, f"restart_{name}_ps")

    print(f"  Phase 7 done: {time.monotonic()-t_phase:.1f}s")

    # === Phase 8: Final intensive polish cycle ===
    print(f"\n=== Phase 8: Final polish ===")
    t_phase = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 5:
        for cycle in range(3):
            if time.monotonic() > deadline - 5:
                break

            # Peak shave
            if time.monotonic() < deadline - 5:
                a_f1, _ = _peak_shaving(best_seq, deadline - 5, max_iters=10000,
                                        tol_frac=1e-6)
                try_update(a_f1, f"final_ps_{cycle}")

            # Ultra L-BFGS
            if time.monotonic() < deadline - 5 and best_seq is not None:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 1e-7
                max_iter = min(5000, max(200, int((deadline - 5 - time.monotonic()) / 0.003)))
                a_f2, _, _ = _lbfgs_run_fine(best_seq, beta, n, max_iter)
                try_update(a_f2, f"final_lbfgs_{cycle}")

            # Coordinate descent
            if time.monotonic() < deadline - 5:
                a_f3, _ = _coordinate_descent(best_seq, deadline - 5, max_rounds=3000)
                try_update(a_f3, f"final_cd_{cycle}")

    print(f"  Phase 8 done: {time.monotonic()-t_phase:.1f}s")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
