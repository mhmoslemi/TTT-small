        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient of 2n * (c_max + beta * log(sum exp((c-kmax)/beta)))
    # = 2n * beta * sum_k w_k * (dc_k/da) / a  (in log-space)
    # dc_k/da_i = 2*a_{k-i} (with zero padding)
    # In log-space: d/dz_i = d/da_i * da/dz_i = d/da_i * a_i * (1 - a_i)
    # Wait, da/dz_i = a_i * (1 - a_i) for softmax... 
    # Actually, d(softmax(z))_i / dz_j = a_i * (delta_ij - a_j)
    # So the gradient in z-space is: sum_i (df/da_i) * a_i * (delta_ij - a_j)
    # = a_j * (sum_i df/da_i * a_i - df/da_j)
    # Hmm, let me just compute it directly.
    
    # df/da_i = 2n * sum_k w_k * 2*a_{k-i} (where a_m = 0 for invalid m)
    # = 4n * sum_k w_k * a_{k-i}
    # = 4n * (w * a_rev)[n-1-i]... let me think again.
    # sum_k w_k * a_{k-i} = (w * a_rev)[?] 
    # a_{k-i} = a_rev[i-k+n-1]... this is getting confusing.
    # Let me just compute it via convolution.
    # sum_k w_k * a_{k-i} = (conv(w, a))[?]... 
    # Actually: sum_k w_k * a_{k-i} where k ranges over 0..2n-2.
    # Let j = k-i, so k = j+i. Sum over j: w_{j+i} * a_j = (cross_corr)(i)
    # This is the cross-correlation of w and a at lag i.
    # cross_corr_w_a(i) = sum_j w_{j+i} * a_j
    # = (a * w_rev)[n-1+i]... hmm.
    # Let me use: a * w_rev where w_rev[k] = w[2n-2-k].
    # (a * w_rev)[m] = sum_j a_j * w_rev[m-j] = sum_j a_j * w[2n-2-m+j]
    # Set m = 2n-2-i: (a * w_rev)[2n-2-i] = sum_j a_j * w[2n-2-(2n-2-i)+j] = sum_j a_j * w[i+j]
    # That's not quite what we want either.
    
    # Let me just use FFT directly.
    # We want g_i = sum_k w_k * a_{k-i} for i = 0, ..., n-1.
    # This is the cross-correlation of w and a (with w as the "template").
    # In signal processing: (w ⋆ a)(i) = sum_k w(k) a(k-i)
    # = (w * flip(a))(i + n - 1) in full convolution.
    # Where flip(a)[j] = a[n-1-j].
    
    a_rev = a[::-1].copy()
    # conv(w, a_rev)[i + n - 1] = sum_k w_k * a_rev[i+n-1-k] = sum_k w_k * a[n-1-(i+n-1-k)] = sum_k w_k * a[k-i]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]  # h[i] = sum_k w_k * a_{k-i}
    
    # df/da_i = 4n * h[i]
    # df/dz_i = sum_j (df/da_j) * da_j/dz_i = sum_j (df/da_j) * a_j * (delta_ji - a_i)
    # Wait, da_j/dz_i = a_j * (delta_ji - a_i)
    # So df/dz_i = sum_j (df/da_j) * a_j * (delta_ji - a_i)
    # = a_i * df/da_i - a_i * sum_j (df/da_j) * a_j
    # = a_i * (df/da_i - sum_j (df/da_j) * a_j)
    
    df_da = 4.0 * n * h
    W = float(np.dot(w, c))  # sum_j df/da_j * a_j / (4n) ... wait
    # sum_j (df/da_j) * a_j = 4n * sum_j h_j * a_j = 4n * (h . a)
    # And h . a = sum_j h_j * a_j = sum_j a_j * sum_k w_k * a_{k-j} = sum_k w_k * sum_j a_j * a_{k-j} = sum_k w_k * c_k = w . c
    # So sum_j (df/da_j) * a_j = 4n * (w . c)
    
    grad_z = 4.0 * n * a * (h - W)
    return grad_z

def _active_grad(z, a, c, n, active_set=None, tol_frac=0.002):
    """Compute subgradient using an active set of peaks."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    if active_set is None:
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
    else:
        active = active_set
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1].copy()
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient method with adaptive active set and line search."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        
        # Adaptive active set: top peaks within tol
        c_max = float(c.max())
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        grad = _active_grad(z, a, c, n, active, tol_frac=0.005)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        # Backtracking line search
        step = lr
        success = False
        for ls in range(20):
            z_try = z - step * grad / grad_norm
            z_try = np.clip(z_try, -50.0, 50.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14 * step:
                z = z_try
                a = a_try
                if sc_try < best_sc - 1e-12:
                    best_sc = sc_try
                    best_a = a.copy()
                    patience = 0
                success = True
                break
            step *= 0.7
        
        if not success:
            patience += 1
            lr *= 0.8
            if patience > 20:
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                patience = 0
                lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _peeling(a_init, deadline, n_steps=2000):
    """Iterative peeling: reduce the peak by moving mass away from peak-contributing positions."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    # Work in log-space for stability
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top few peaks
        n_peaks = max(1, min(10, int(0.01 * len(c))))
        peak_indices = np.argsort(c)[-n_peaks:]
        w = np.zeros_like(c)
        for k in peak_indices:
            w[k] = 1.0 / n_peaks
        
        a_rev = a[::-1].copy()
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        
        # Direction: decrease a where h is large (contributing to peaks)
        # But maintain sum=1: project to tangent space
        direction = h - np.mean(h)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break
        
        # Try different step sizes
        best_z = z
        best_sc_local = best_sc
        for lr in [0.05, 0.03, 0.01, 0.005, 0.002]:
            z_try = z - lr * direction / dnorm
            z_try = np.clip(z_try, -50.0, 50.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            if sc_try < best_sc_local:
                best_sc_local = sc_try
                best_z = z_try.copy()
        
        if np.array_equal(best_z, z):
            break
        z = best_z
        if best_sc_local < best_sc - 1e-12:
            best_sc = best_sc_local
            best_a = _softmax(z).copy()
    
    return best_a, best_sc

def _fourier_refine(a_init, deadline, n_steps=100):

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 15))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2)
        a = a + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _resample(a, new_n):
    """Resample sequence a to new length new_n."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, active-set subgradient with line search,
peeling heuristic, Frank-Wolfe away steps, multi-resolution initialization,
and diverse perturbations. All helpers are top-level with no closures or lambdas.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1].copy()
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_subgrad(a, c, n, n_top=5, tol_frac=0.003):
    """Compute subgradient using top-n peaks or tolerance-based active set."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        active[int(np.argmax(c))] = True
        n_active = 1
    # Also ensure we include top n_top
    top_idx = np.argsort(c)[-n_top:]
    for k in top_idx:
        active[k] = True
    n_active = int(active.sum())
    w = np.zeros_like(c)
    w[active] = 1.0 / n_active
    a_rev = a[::-1].copy()
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_line_search(a_init, deadline, n_max_steps=5000, lr_init=0.025):
    """Subgradient with backtracking line search in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_subgrad(a, c, n, n_top=3, tol_frac=0.002)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr
        success = False
        for ls in range(15):
            z_try = z - step * grad / grad_norm
            z_try = np.clip(z_try, -50.0, 50.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14 * step:
                z = z_try
                if sc_try < best_sc - 1e-13:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    patience = 0
                success = True
                break
            step *= 0.6

        if not success:
            patience += 1
            lr *= 0.85
            if patience > 25:
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                patience = 0
                lr = lr_init * 0.4

    return best_a, best_sc


def _peeling(a_init, deadline, n_steps=2000):
    """Iterative peeling: reduce peaks by moving mass from contributing positions."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    stall = 0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Active set: top peaks
        n_peaks = max(1, min(5, int(0.005 * len(c))))
        peak_idx = np.argsort(c)[-n_peaks:]
        w = np.zeros_like(c)
        for k in peak_idx:
            w[k] = 1.0 / n_peaks

        a_rev = a[::-1].copy()
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]

        direction = h - np.mean(h)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break

        best_z = z.copy()
        best_sc_local = best_sc
        for lr in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            z_try = z - lr * direction / dnorm
            z_try = np.clip(z_try, -50.0, 50.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            if sc_try < best_sc_local - 1e-14:
                best_sc_local = sc_try
                best_z = z_try.copy()

        if np.allclose(best_z, z, atol=1e-15):
            stall += 1
            if stall > 20:
                break
        else:
            z = best_z
            stall = 0
            if best_sc_local < best_sc - 1e-12:
                best_sc = best_sc_local
                best_a = _softmax(z).copy()

    return best_a, best_sc


def _frank_wolfe_away(a_init, deadline, n_steps=300, lr_init=0.1):
    """Frank-Wolfe with away steps on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_act = int(active.sum())
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act

        a_rev = a[::-1].copy()
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        g = 2.0 * h  # subgradient in a-space (dropping the scale factor)

        # Towards step: min g_i
        i_towards = int(np.argmin(g))
        # Away step: max g_i / a_i for a_i > 0
        positive = a > 1e-10
        if positive.sum() > 1:
            g_away = np.where(positive, g / np.maximum(a, 1e-15), -np.inf)
            i_away = int(np.argmax(g_away))
        else:
            i_away = -1

        # Try towards
        v_towards = np.zeros(n)
        v_towards[i_towards] = 1.0
        # Try away
        if i_away >= 0 and i_away != i_towards:
            v_away = np.zeros(n)
            v_away[i_away] = -1.0
        else:
            v_away = None

        best_sc_local = best_sc
        best_a_local = best_a.copy()

        for v, label in [(v_towards, "towards"), (v_away, "away") if v_away is not None else (None, "")]:
            if v is None:
                continue
            for t in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
                a_try = a + t * (v - a * v.sum())
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc_local - 1e-13:
                    best_sc_local = sc_try
                    best_a_local = a_try.copy()

        if best_sc_local >= best_sc - 1e-14:
            lr *= 0.7
            if lr < 1e-5:
                break
        else:
            a = best_a_local.copy()
            best_sc = best_sc_local
            best_a = best_a_local.copy()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 12))))

        try:
            result = minimize(
                _make_smoo_fun(beta, n),
                z,
                jac=_make_smoo_jac(beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_smoo_fun(beta, n):
    """Return a function object for scipy (avoiding lambda)."""
    class _Fun:
        def __call__(self, z_local):
            return _smoobj(z_local, beta, n)
    return _Fun()


def _make_smoo_jac(beta, n):
    """Return a jac object for scipy (avoiding lambda)."""
    class _Jac:
        def __call__(self, z_local):
            return _smograd(z_local, beta, n)
    return _Jac()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 2.0 * t / n)
        a = a + 0.15 * np.cos(2.0 * np.pi * 5.0 * t / n + 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30 * (t - 0.3) ** 2) + np.exp(-30 * (t - 0.7) ** 2)
        a = a + 0.02
        return a / a.sum()
    elif kind == "ridged":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 1.0 * t / n) ** 2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_smooth(a, rng, n_modes=3):
    """Smooth Fourier perturbation."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = 0.01 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def _shrinkage(a, deadline):
    """Zero out smallest elements and re-score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    sorted_idx = np.argsort(a)

    for frac in [0.01, 0.02, 0.05, 0.10, 0.15, 0.25]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _update_best(state, p, sc=None, label=""):
    """Update best in a state dict (to avoid closures)."""
    if p is None:
        return
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 0.01:
        return
    if sc is None:
        sc = _score(p)
    if np.isfinite(sc) and sc < state['best_score']:
        prev = state['best_score']
        state['best_seq'] = p.copy()
        state['best_score'] = float(sc)
        t = time.monotonic() - state['t_start']
        print(f"  NEW BEST: {state['best_score']:.10f} (n={len(p)}) [{label}] "
              f"improve={prev - state['best_score']:.2e} t={t:.1f}s")
