import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is already normalized to sum=1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum}"""
    n = len(v)
    if n == 0:
        return v
    # Scale to sum=1 first, then we can adjust
    v_scaled = v / max(v.sum(), 1e-30) * target_sum
    # Actually, let's do proper projection
    # Project onto {x >= 0, sum(x) = target_sum}
    # This is equivalent to: find t such that sum(max(v - t, 0)) = target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad_max(a, w):
    """
    Compute gradient of sum(w * c) where c = conv(a,a).
    w has same length as c (2n-1).
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

(1/|A|) * sum_{k in A} grad_a(c[k])

where grad_a(c[k])[i] = 2*a[k-i] (with the appropriate boundary handling for the full convolution).

So the update is:
a_new = proj_simplex(a - step * (1/|A|) * sum_{k in A} grad_a(c[k]))

"""
Optimized multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses parent initialization, FISTA with adaptive steps, Adam in log-space,
active-set subgradient descent, and multi-restart exploration.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v, target=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_conv(a, w):
    """Gradient of dot(w, conv(a,a)) w.r.t. a. w has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_weights(c, beta):
    """Compute softmax weights over convolution values."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _fista_run(a_init, betas, deadline, iters_per_beta=5000):
    """FISTA with softmax-smoothed max, given a list of betas from coarse to fine."""
    n = len(a_init)
    a_curr = a_init.copy().astype(np.float64)
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        c_curr = _full_convolve(a_curr, a_curr)
        max_c = float(c_curr.max())
        if max_c < 1e-15:
            max_c = 1e-15
        L = 4.0 * n * max_c / beta
        step = 1.0 / L
        t_val = 1.0  # Reset momentum for each beta phase
        
        for it in range(iters_per_beta):
            if time.monotonic() > deadline:
                break
            
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = float(y.sum())
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            c_y = _full_convolve(y, y)
            w = _softmax_weights(c_y, beta)
            grad = _grad_conv(y, w)
            a_next = _proj_simplex(y - step * grad)
            
            # Periodic score check
            if it % 200 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
                # Adapt step size
                c_new = _full_convolve(a_next, a_next)
                max_new = float(c_new.max())
                if max_new > 1e-15:
                    L = 4.0 * n * max_new / beta
                    step = 1.0 / L
            
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _adam_logspace(a_init, deadline, betas, lr_init=0.01, iters_per_beta=100000):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        for it in range(iters_per_beta):
            if time.monotonic() > deadline:
                break
            t += 1
            
            # Compute softmax from z
            z_shifted = z - z.max()
            e = np.exp(z_shifted)
            s = float(e.sum())
            a = e / s
            
            # Compute convolution and weights
            c = _full_convolve(a, a)
            w = _softmax_weights(c, beta)
            
            # Gradient of log-sum-exp smoothed objective w.r.t. z
            # d/dz_i [sum_k w_k * c[k]] where c = conv(a,a), a = softmax(z)
            # This is complex, so we use the chain rule:
            # d/dz_i = sum_j (da_j/dz_i) * (dc/da_j)
            # da_j/dz_i = a_j * (delta(j,i) - a_i)
            # So d/dz_i = a_i * (sum_j dc/da_j * a_j - sum_j dc/da_j * a_i)
            # Hmm, this is getting complicated. Let me use a simpler approach.
            
            # Actually, the gradient w.r.t. a of the smoothed objective is:
            # grad_a = 2 * (W * a_rev)[n-1:2n-1] where W = full_convolve(w, a_rev)
            # Wait, that's _grad_conv(a, w) which gives grad w.r.t. a.
            
            # For gradient w.r.t. z: grad_z[i] = a[i] * (grad_a[i] - a[i] * dot(a, grad_a))
            # No wait. If a = softmax(z), then:
            # d L / d z_i = sum_j (dL/da_j) * (da_j/dz_i) = sum_j grad_a[j] * a[j]*(delta(j,i) - a[i])
            # = a[i] * grad_a[i] - a[i] * sum_j grad_a[j] * a[j]
            # = a[i] * (grad_a[i] - dot(a, grad_a))
            
            grad_a = _grad_conv(a, w)
            dot_a_grad = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - dot_a_grad)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = float(np.linalg.norm(update))
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            
            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                z_shifted2 = z - z.max()
                e2 = np.exp(z_shifted2)
                a_chk = e2 / e2.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it % 5000 == 0 and it > 0:
                current_lr = max(0.001, current_lr * 0.97)
    
    # Final
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    a_final = e / e.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _active_set_descent(a_init, deadline, tol=1e-8, max_iters=50000):
    """Subgradient descent with exact active set."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    alpha = 0.5  # initial step size
    prev_active = None
    active_count = 1
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        max_c = float(c.max())
        # Active set: positions where c[k] > max_c - tol * max_c
        threshold = max_c - max(tol * max_c, 1e-12)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # If active set is very large, use top-K
        K = max(1, min(len(active), 50))
        top_k = np.argsort(c)[-K:]
        
        # Compute subgradient (average over top-K)
        w = np.zeros_like(c)
        w[top_k] = 1.0 / K
        grad = _grad_conv(a, w)
        
        # Adaptive step size
        if prev_active is not None and np.array_equal(np.sort(active), np.sort(prev_active)):
            alpha *= 0.999  # Same active set, decrease step
        else:
            alpha = max(alpha * 1.01, 0.1)  # New active set, increase step
        alpha = min(alpha, 2.0)
        
        a_new = _proj_simplex(a - alpha * grad / max_c)
        
        sc_new = _score_norm(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        
        prev_active = np.sort(active)
        active_count = len(active)
        
        # Check convergence
        if it % 1000 == 0 and it > 0:
            if best_score < 2.0 * n * (1.0 / n + 1e-10):
                break
    
    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _interpolate_profile(src, n_new):
    n0 = len(src)
    if n0 == n_new:
        return src.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, src)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
