"""
Multi-strategy optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses p-norm smoothing, FISTA with backtracking, spectral initialization,
coordinate descent polish, and multi-restart with diverse initializations.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1, so just 2*n*max(c)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _grad_pnorm(a, c, p):
    """Gradient of (sum c_k^p)^(1/p) w.r.t. a, for simplex parameterization."""
    n = len(a)
    # c_k = sum_{i+j=k} a[i]*a[j]
    # dc_k/da[i] = sum_{j: i+j=k, 0<=j<n} a[j] + sum_{j: i+j=k, 0<=j<n} a[j] (for i != j)
    # Actually for full convolution: c_k = sum_i a[i]*a[k-i] (where valid)
    # dc_k/da[i] = a[k-i] + (a[i-k] if i-k != i else 0) ... complicated
    # Simpler: dc_k/da[i] = 2 * sum_{j: i+j=k, j != i} a[j] + a[i] (if 2i=k)
    # Even simpler: use the convolution formula
    # c = conv(a, a), so dc/da[i] = 2 * conv(a, e_i) restricted to [n-1, 2n-2]
    # = 2 * a reversed, shifted
    
    a_rev = a[::-1]
    # The gradient of c_k w.r.t. a_i for i in [0, n-1]:
    # dc_k/da_i = a[k-i] + a[i-k] (when indices valid, counting both terms)
    # For the "autocorrelation" part (k >= n-1):
    # Let l = k - (n-1), so l in [0, n-1]
    # dc_{n-1+l}/da_i = a[n-1+l-i] (if 0 <= n-1+l-i < n) + a[i-(n-1-l)] (if 0 <= i-(n-1-l) < n and i != n-1+l-i)
    
    # Actually the cleanest way: the gradient of sum_k w_k * c_k w.r.t. a
    # where c = conv(a,a) is: 2 * conv(w, a_flipped) (restricted to first n elements)
    # where a_flipped = a[::-1]
    
    # For p-norm: w_k = p * c_k^(p-1) / (sum c_j^p)^(1/p)
    
    c_pos = np.maximum(c, 0)  # ensure non-negative for fractional powers
    cp = c_pos ** p
    sum_cp = float(cp.sum())
    if sum_cp < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = p * c_pos ** (p - 1) / (sum_cp ** (1.0 / p))
        w = w / w.sum()  # normalize
    
    # Gradient of (sum w_k * c_k) w.r.t. a, restricted to first n coords:
    # = 2 * (w * a_rev) full conv, take elements [n-1: 2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    
    return grad

def _fista_pnorm(a_init, p, deadline, max_iter, lr_scale=1.0):
    """FISTA with p-norm smoothing."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    # Lipschitz constant estimate for p-norm: roughly p * max_c^(p-1) * 2n
    L = lr_scale * p * max_c_init ** (p - 1) * 2 * n
    if L < 1e-10:
        L = 1.0
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        grad = _grad_pnorm(y, c_y, p)
        
        # Adaptive step via backtracking (simplified)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
        
        # Decay step size slowly
        if it % 500 == 0 and it > 0:
            step *= 0.99
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _adam_pnorm(a_init, p, deadline, lr_init=0.01, max_iter=100000):
    """Adam optimizer in log-space with p-norm objective."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n
    
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()
    
    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        c = _full_convolve(a, a)
        
        # p-norm gradient in log-space
        # d/dz_i [p-norm] = d/d a_i [p-norm] * a_i
        grad_a = _grad_pnorm(a, c, p)
        grad_z = grad_a * a
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if t % 50 == 0:
            a_chk = np.exp(z - z.max())
            a_chk = a_chk / a_chk.sum()

def _subgradient_exact(a_init, n_steps=5000, deadline=None):
    """Subgradient method using exact max (top-k weighting)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score_norm(a)
    step_size = 0.1
    patience = 0
    
    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Top-k weighted subgradient
        top_k = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros(len(c))
        w[top_idx] = 1.0
        w = w / w.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        a_new = _proj_simplex(a - step_size * grad)
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience > 50:
                step_size *= 0.8
                patience = 0
            a_new2 = _proj_simplex(a - step_size * 0.5 * grad)
            if _score_norm(a_new2) < _score_norm(a):
                a = a_new2
    
    return best_a, best_score

def _spectral_flatten(a_init, n_iter=200, deadline=None):
    """Iteratively flatten the spectrum to reduce autocorrelation peaks."""
    n = len(a_init)

    s = a.sum()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.arange(n, dtype=np.float64)
        center = n / 2.0
        a = np.exp(-0.5 * ((t - center) / (n / 6.0)) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_peaks":
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((t - n/4) / (n/10)) ** 2) + np.exp(-0.5 * ((t - 3*n/4) / (n/10)) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovations over previous approach:

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s
