"""
Improved optimizer combining FISTA continuation, coordinate descent with equalization,
and multi-restart for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements: coordinate descent polish, equalization candidates, multiple n values,
enhanced beta schedule, and diverse structured initializations.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _coord_descent(a, passes=20, deadline=None):
    """Coordinate descent with equalization candidates."""
    n = len(a)
    conv = _full_convolve(a, a)
    S = float(a.sum())
    best_j = 2.0 * n * float(np.max(conv)) / (S * S)
    a_work = a.copy()

    for pas in range(passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        for i in range(n):
            t = a_work[i]
            if t < 1e-15:
                continue
            S_rest = S - t

            # Compute s[k] = a[k-i] if 0 <= k-i < n else 0
            s = np.zeros(2 * n - 1)
            lo_k = max(0, i)
            hi_k = min(2 * n - 1, i + n)
            if hi_k > lo_k:
                s[lo_k:hi_k] = a_work[lo_k - i:hi_k - i]

            # Quadratic term at k=2i
            has_quad = (2 * i < 2 * n - 1)

            # Generate candidates
            candidates = [0.0, t * 0.5, t * 0.75, t * 1.25, t * 1.5]

            # Fixed-point candidate
            k_star = int(np.argmax(conv))
            s_k = s[k_star]
            if s_k > 1e-15:
                t_fp = S_rest - float(conv[k_star]) / s_k + 2.0 * t
                if 0.0 < t_fp < 5.0 * t + 1.0:
                    candidates.append(t_fp)

            # Equalization candidate: equalize top 2 conv coefficients
            sorted_idx = np.argsort(conv)[::-1]
            k1 = sorted_idx[0]
            k2 = sorted_idx[1] if len(sorted_idx) > 1 else -1
            if k2 >= 0:
                s_k1 = s[k1]
                s_k2 = s[k2]
                denom = s_k2 - s_k1
                if abs(denom) > 1e-15:
                    t_eq = t + (float(conv[k1]) - float(conv[k2])) / (2.0 * denom)
                    if 0.0 <= t_eq < 5.0 * t + 1.0:
                        candidates.append(t_eq)

            # Evaluate candidates
            best_t = t
            best_sc = 2.0 * n * float(np.max(conv)) / (S * S)
            for t_prime in candidates:
                if t_prime < 0:
                    continue
                delta = t_prime - t
                q_val = (t_prime * t_prime - t * t) if has_quad else 0.0
                if q_val != 0.0 and has_quad:
                    new_conv = conv.copy()
                    new_conv += 2.0 * delta * s
                    new_conv[2 * i] += q_val
                else:
                    new_conv = conv + 2.0 * delta * s
                new_max = float(np.max(new_conv))
                new_S = S_rest + t_prime
                if new_S < 1e-10:
                    continue
                sc = 2.0 * n * new_max / (new_S * new_S)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_prime

            if best_t != t:
                delta = best_t - t
                a_work[i] = best_t
                S = S_rest + best_t
                if has_quad:
                    conv += 2.0 * delta * s
                    conv[2 * i] += (best_t * best_t - t * t)
                else:
                    conv = conv + 2.0 * delta * s
                new_j = 2.0 * n * float(np.max(conv)) / (S * S)
                if new_j < best_j:
                    best_j = new_j

    a_work = a_work / a_work.sum()
    return a_work, best_j


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 25)), 'gtol': 1e-11, 'maxls': 12}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_profile(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "double_hump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "noise_smooth":
        if rng is None:
            rng = np.random.default_rng(0)
        raw = rng.standard_normal(n)
        # Smooth with a window
        from scipy.ndimage import uniform_filter1d
        a = uniform_filter1d(raw, size=max(3, n // 32))
        a = np.abs(a) + 0.01
        return a / a.sum()
    elif kind == "prime_weighted":
        # Weight by primality-like pattern
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t * 7.0) + 0.2 * np.cos(np.pi * t * 13.0)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):

    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()

    max_c0 = float(c_curr.max())
    beta = 0.05 * max_c0

    phase1_end = min(deadline, deadline - (deadline - time.monotonic()) * 0.55)
    per_phase = (phase1_end - time.monotonic()) / max(1, n_phases_fista)

    for phase in range(n_phases_fista):
        if time.monotonic() > phase1_end - 2:
            break
        per_time = min(per_phase, 30.0)
        max_it = max(2000, int(per_time * 3000))
        a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_time, max_it)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
        c_curr = _full_convolve(a_work, a_work)
        beta = 0.4 * float(c_curr.max())
        if beta < 1e-12:
            beta = 1e-12

    # Phase 2: Coordinate descent
    if time.monotonic() < phase1_end + (deadline - phase1_end) * 0.35:
        cd_deadline = phase1_end + (deadline - phase1_end) * 0.35
        n_cd_passes = min(30, max(5, int((cd_deadline - time.monotonic()) / max(1, n * n / 5000000.0))))
        a_work, _ = _coord_descent(a_work, passes=n_cd_passes, deadline=cd_deadline)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            print(f"    CD improved: {sc:.10f}", flush=True)

    # Phase 3: Adam fine-tuning
    phase3_deadline = deadline - (deadline - time.monotonic()) * 0.25
    if time.monotonic() < phase3_deadline - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.001 * max_cc, 0.0003 * max_cc, 0.0001 * max_cc]
        per_beta = (phase3_deadline - time.monotonic() - 5) / len(betas)
        for beta in betas:
            if time.monotonic() > phase3_deadline - 5:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + max(5.0, per_beta),
                                  beta, lr_init=0.005, max_iter=200000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()

    # Phase 4: L-BFGS
    if time.monotonic() < deadline - 20:
        c_curr = _full_convolve(a_work, a_work)
        b_lb = 0.0001 * float(c_curr.max())
        a_work, _ = _lbfgs_opt(a_work, deadline - 10.0, b_lb)
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()

    # Phase 5: Final coordinate descent polish
    if time.monotonic() < deadline - 5:
        a_work, _ = _coord_descent(best_seq, passes=15, deadline=deadline - 3.0)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            print(f"    Final CD improved: {sc:.10f}", flush=True)

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}", flush=True)
        best_seq = parent.copy()
        best_score = _score(parent)
    else:
        n_parent = 4096
        best_seq = _make_profile(n_parent, "uniform")
        best_score = _score(best_seq)
        print(f"No parent, starting uniform n={n_parent}, score={best_score:.10f}", flush=True)

    # Define candidate configurations
    configs = []

    # Config 1: Optimize parent directly (most time)
    if parent is not None:
        configs.append(("parent", parent.copy(), 0.40))

    # Config 2: Different n with parent-like structure
    if parent is not None and len(parent) > 1024:
        for n_try in [2048, 8192]:
            if n_try != len(parent):
                # Resample parent to new length
                from scipy.ndimage import zoom
                factor = n_try / len(parent)
                a_resampled = zoom(parent, factor, order=3)
                a_resampled = np.maximum(a_resampled, 0.0)
                a_resampled = a_resampled / a_resampled.sum()
                configs.append((f"resample_n{n_try}", a_resampled, 0.15))

    # Config 3: Structured initializations
    n_main = len(parent) if parent is not None else 4096
    structured_kinds = ["asymmetric", "double_hump", "raised_cosine", "prime_weighted"]
    for kind in structured_kinds:
        a_struct = _make_profile(n_main, kind, rng)
        # Blend with parent for continuity
        if parent is not None and len(parent) == n_main:
            a_struct = 0.6 * parent + 0.4 * a_struct
            a_struct = np.maximum(a_struct, 1e-10)
            a_struct = a_struct / a_struct.sum()
        configs.append((f"struct_{kind}", a_struct, 0.10))

    # Config 4: Perturbed parent
    if parent is not None:
        a_pert = _perturb(parent, 0.05, rng)
        configs.append(("perturbed", a_pert, 0.10))

    # Allocate time
    total_alloc = sum(c[2] for c in configs)
    print(f"Configs: {[(c[0], c[2]) for c in configs]}, total_alloc={total_alloc:.2f}", flush=True)

    # Normalize allocations to fit deadline
    time_budget = deadline - t_start - 10.0
    for idx, (name, a_init, frac) in enumerate(configs):
        alloc_time = time_budget * frac / total_alloc
        if alloc_time < 30.0 and idx > 0:
            continue  # Skip if too little time
        if time.monotonic() > deadline - 20:
            break

        cfg_deadline = time.monotonic() + alloc_time - 2.0
        print(f"\n[{time.monotonic()-t_start:.1f}s] Config '{name}': n={len(a_init)}, "
              f"initial score={_score(a_init):.10f}, budget={alloc_time:.0f}s", flush=True)

        a_best, sc_best = _optimize_sequence(a_init.copy(), cfg_deadline)
        # Final coordinate descent on the best
        if time.monotonic() < cfg_deadline - 3:
            a_best, _ = _coord_descent(a_best, passes=10, deadline=cfg_deadline - 1.0)
            a_best = a_best / a_best.sum()
            sc_best = _score(a_best)

        if sc_best < best_score:
            best_score = sc_best
            best_seq = a_best.copy()
            print(f"  => NEW BEST: {sc_best:.10f} (config: {name})", flush=True)
        else:
            print(f"  => score={sc_best:.10f} (no improvement over {best_score:.10f})", flush=True)

    # Final global polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n[{time.monotonic()-t_start:.1f}s] Final global polish...", flush=True)
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()

        # One more round of CD + Adam + CD
        a_pol, _ = _coord_descent(a_pol, passes=15, deadline=time.monotonic() + (deadline - time.monotonic()) * 0.4)
        a_pol = a_pol / a_pol.sum()
        sc = _score(a_pol)
        if sc < best_score:
            best_score = sc
            best_seq = a_pol.copy()

        if time.monotonic() < deadline - 8:
            c_curr = _full_convolve(a_pol, a_pol)
            b_final = 0.00005 * float(c_curr.max())
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + (deadline - time.monotonic()) * 0.3,
                                 b_final, lr_init=0.002, max_iter=100000)
            a_pol = a_pol / a_pol.sum()
            sc = _score(a_pol)
            if sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()
                print(f"  Final Adam: {sc:.10f}", flush=True)

        if time.monotonic() < deadline - 5:
            a_pol, _ = _coord_descent(best_seq, passes=10, deadline=deadline - 3.0)
            a_pol = a_pol / a_pol.sum()
            sc = _score(a_pol)
            if sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()
                print(f"  Final CD: {sc:.10f}", flush=True)

    print(f"\n[{time.monotonic()-t_start:.1f}s] Done: best_score={best_score:.10f}", flush=True)

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}", flush=True)
        except Exception as ex:
            print(f"Eval error: {ex}", flush=True)
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
