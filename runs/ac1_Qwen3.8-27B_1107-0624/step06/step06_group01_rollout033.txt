```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses a ramp-like initialization (which asymptotically achieves 4/3) combined with
Adam optimization in log-space and soft-max smoothing of the max operator.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_unnormalized(a):
    """Score assuming sum(a) = 1, returns 2*n*max(conv)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_softmax_grad(z, n, beta):
    """
    Compute the gradient of 2n * soft_max(conv(a,a), beta) w.r.t. z,
    where a = softmax(z).
    Returns the gradient in z-space (n-dimensional).
    """
    a = _softmax_compute(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # grad_a[i] = sum_k w[k] * a[k-i] for valid indices
    # = cross-correlation of w and a
    # Using: grad[i] = sum_m a[m] * w[i+m]
    # Which is (a_rev * w)[i + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    full = _full_convolve(a_rev, w)  # length n + (2n-1) - 1 = 3n-2
    grad_a = full[n-1:2n-1]  # n elements
    
    # Convert to z-space: grad_z[i] = a[i] * (grad_a[i] - sum(a * grad_a))
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)
    return grad_z * 2.0 * n

def _adam_logspace(z_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with soft-max smoothing."""
    n = len(z_init)
    z = z_init.copy().astype(np.float64)
    
    best_z = z.copy()
    best_score = _score_unnormalized(_softmax_compute(z))
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max(0, max_iter_total - t)
        if remain_iter <= 0:
            break
        
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            # Compute gradient in z-space
            a_rev = a[::-1]
            full = _full_convolve(a_rev, w)
            grad_a = full[n-1:2n-1]
            dot = float(np.dot(a, grad_a))
            grad = a * (grad_a - dot) * 2.0 * n
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1**t)
            vh = v_buf / (1.0 - b2**t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 5.0:
                update = update * (5.0 / update_norm)
            
            z = z - update
            z = z - z.mean()  # Re-center for numerical stability
            
            # Periodic check
            if t % 100 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_unnormalized(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_z = z.copy()
            
            # Learning rate decay
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0001, current_lr * 0.9)
    
    a_final = _softmax_compute(z)
    sc_final = _score_unnormalized(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_z = z.copy()
    
    return _softmax_compute(best_z), best_score

def _make_ramp(n, alpha=1.0):
    """Create a ramp-like initialization: a[j] = (j+1)^alpha / sum"""
    t = np.arange(1, n+1, dtype=np.float64)
    a = t**alpha
    return a / a.sum()

def _make_smoothed_ramp(n, beta_shape=1.0):
    """S-curve shaped initialization"""
    t = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-beta_shape * (2.0*t - 1.0)))
    a = a - a[0] + 0.01
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _coordinate_descent_polish(a, deadline, n_steps=5000):
    """Coordinate descent to polish the solution."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_unnormalized(a)
    rng = np.random.default_rng(42)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        i = rng.integers(0, n)
        best_factor = 1.0
        best_sc = best_score
        for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 0.5, 1.5]:
            a_try = best_a.copy()
            a_try[i] *= factor
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_unnormalized(a_try)
                if sc < best_sc:
                    best_sc = sc
                    best_factor = factor
        if best_factor != 1.0:
            best_a[i] *= best_factor
            best_a = np.maximum(best_a, 0.0)
            best_a = best_a / best_a.sum()
```
