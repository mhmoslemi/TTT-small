    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _block_coordinate_descent(a, deadline, block_size=64, eval_interval=200):
    """Block coordinate descent with exact 1D line search within each block."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    n_blocks = n // block_size + 1
    t = 0
    
    while time.monotonic() < deadline:
        t += 1
        improved = False
        for b in range(n_blocks):
            if time.monotonic() > deadline:
                break
            start = b * block_size
            end = min(start + block_size, n)
            if start >= n:
                break
            
            a_fixed = a.copy()
            a_fixed[start:end] = 0  # zero out the block
            c_fixed = _fft_convolve(a_fixed, a_fixed)
            
            # The contribution of block to conv is:
            # conv_total = c_fixed + cross Terms + block self-conv
            # cross_terms[k] = sum_{i in block} a[i] * a_fixed[k-i] + a_fixed[k-i] * a[i]
            # block_self[k] = sum_{i,j in block} a[i] * a[j] where k = i-j (relative)
            
            # For simplicity, let's just do a few steps of gradient descent on the block
            block = a[start:end].copy()
            for _ in range(5):
                a_trial = a_fixed.copy()
                a_trial[start:end] = block
                c_trial = _fft_convolve(a_trial, a_trial)
                c_max = c_trial.max()
                # Gradient w.r.t. block elements
                k_indices = np.argsort(c_trial)[::-1][:5]
                p = np.zeros(2*n-1)
                p[k_indices] = 1.0 / len(k_indices)
                p_rev = p[::-1]
                conv_g = _fft_convolve(a_trial, p_rev)
                g_block = 2.0 * conv_g[n-1+start:n-1+end]
                
                # Simple update
                block = np.maximum(block - 0.01 * g_block, 0)
                # Renormalize to keep sum = 1
                total_sum = a_fixed.sum() + block.sum()
                if total_sum > 0:
                    a_fixed[start:end] = block
                    a = a_fixed / total_sum
        
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a, best_sc


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros(2 * n - 1)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _nesterov_descent(a, deadline, top_k=3, lr=0.05, momentum=0.9,
                      adapt_k=True, min_k=1, max_k=15,
                      T_soft=None, eval_interval=50,
                      cyclical_lr=False, lr_period=200):
    """Nesterov-accelerated gradient descent in z-space (softmax parameterization)."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    y = z.copy()  # Nesterov lookahead
    v = np.zeros(n)  # velocity
    t = 0
    best_a = a.copy()
    best_sc = _score_norm(a)
    patience = 0
    best_patience = 0
    current_lr_mult = 1.0

    while time.monotonic() < deadline:
        t += 1
        frac = t / max(t, 1)
        # Learning rate schedule
        lr_cur = lr * (0.5 ** (t / 500.0)) * current_lr_mult
        if cyclical_lr:
            lr_cur *= 0.5 + 0.5 * abs(np.sin(np.pi * t / lr_period))

        # Nesterov: evaluate gradient at lookahead point
        a_y = _softmax_stable(y)
        c = _fft_convolve(a_y, a_y)

        if adapt_k:
            c_max = c.max()
            threshold = c_max - 0.05 * c_max
            active = np.where(c >= threshold)[0]
            if len(active) > max_k:
                active = np.argsort(c)[::-1][:max_k]
            if len(active) < min_k:
                active = np.argsort(c)[::-1][:min_k]
            k_indices = active
        else:
            k_indices = np.argsort(c)[::-1][:top_k]

        if T_soft is not None and t % 5 == 0:
            g = _compute_grad_soft(a_y, c, T_soft)
        else:
            g = _compute_grad_active(a_y, c, k_indices)

        # Project gradient to tangent space of simplex
        a_z = _softmax_stable(z)
        g_proj = a_z * (g - np.dot(a_z, g))

        # Nesterov update
        v = momentum * v - lr_cur * g_proj
        y = z + momentum * v

        # Ensure z stays bounded
        y = np.clip(y, -60.0, 60.0)
        y = y - y.mean()
        z = y  # For simplicity, set z = y (heavy ball variant)

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 6 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                # Reset from best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                y = z.copy()
                v = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _backtracking_descent(a, deadline, top_k=3, lr=0.1, adapt_k=True,
                          min_k=1, max_k=10, eval_interval=30,
                          max_backtrack=20):
    """Gradient descent with backtracking line search in z-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = 0
    best_a = a.copy()
    best_sc = _score_norm(a)
    patience = 0
    current_lr = lr

    while time.monotonic() < deadline:
        t += 1
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)
        f_cur = _score_norm(a_cur)

        if adapt_k:
            c_max = c.max()
            threshold = c_max - 0.03 * c_max
            active = np.where(c >= threshold)[0]
            if len(active) > max_k:
                active = np.argsort(c)[::-1][:max_k]
            if len(active) < min_k:
                active = np.argsort(c)[::-1][:min_k]
            k_indices = active
        else:
            k_indices = np.argsort(c)[::-1][:top_k]

        g = _compute_grad_active(a_cur, c, k_indices)
        # Project to tangent space
        g_proj = a_cur * (g - np.dot(a_cur, g))

        # Backtracking line search in z-space
        alpha = current_lr
        success = False
        for _ in range(max_backtrack):
            z_trial = z - alpha * g_proj
            z_trial = np.clip(z_trial, -60.0, 60.0)
            z_trial = z_trial - z_trial.mean()
            a_trial = _softmax_stable(z_trial)
            f_trial = _score_norm(a_trial)
            if f_trial < f_cur * (1 - 1e-6):
                z = z_trial
                success = True
                break
            alpha *= 0.5

        if not success:
            # Accept anyway if very small step
            z_trial = z - alpha * g_proj
            z_trial = np.clip(z_trial, -60.0, 60.0)
            z_trial = z_trial - z_trial.mean()
            z = z_trial
            current_lr *= 0.5

        if t % eval_interval == 0:
            a_eval = _softmax_stable(z)
            sc = _score_norm(a_eval)
            if sc < best_sc:
                best_sc = sc
                best_a = a_eval.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10 and current_lr > 0.001:
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                patience = 0

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _simplex_pgdescent(a, deadline, top_k=3, step_init=0.5, eval_interval=20):
    """Projected subgradient descent directly on the simplex."""
    n = len(a)
    a_cur = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    step = step_init / n
    t = 0
    patience = 0
    avg_step = step

    while time.monotonic() < deadline:
        t += 1
        c = _fft_convolve(a_cur, a_cur)
        k_indices = np.argsort(c)[::-1][:top_k]
        g = _compute_grad_active(a_cur, c, k_indices)

        # Polyak step size if we have an upper bound
        # For now, use diminishing step
        step = avg_step / (1 + 0.01 * t)

        a_new = a_cur - step * g
        a_new = _project_simplex(a_new)

        sc = _score_norm(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a_cur = a_new
            patience = 0
            avg_step = min(avg_step * 1.05, 0.5)
        else:
            patience += 1
            if patience > 15:
                avg_step *= 0.7
                patience = 0
                a_cur = best_a.copy()

    return best_a, best_sc


def _adam_z_space(a, deadline, T_start, T_end, lr_start, lr_end,
                  use_active=False, active_k=3, eval_interval=100,
                  clip_norm=0.5):
    """Adam optimizer in z-space with temperature annealing."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    lr_mult = 1.0

    while t < 500000 and time.monotonic() < deadline:
        t += 1
        frac = min(1.0, t / 500000)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if use_active:
            k_idx = np.argsort(c)[::-1][:active_k]
            g = _compute_grad_active(a, c, k_idx)
        else:
            g = _compute_grad_soft(a, c, T)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -70.0, 70.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 5 and lr_mult > 0.1:
                lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _asymmetric_init(n, rng, n_peaks=5):
    """Create an asymmetric initial sequence with n_peaks gaussian bumps."""
    a = np.ones(n) / n
    weights = rng.dirichlet(np.ones(n_peaks))
    centers = rng.uniform(0.1, 0.9, n_peaks) * n
    widths = rng.uniform(n * 0.02, n * 0.1, n_peaks)
    x = np.arange(n)
    for w, c, s in zip(weights, centers, widths):
        a += w * np.exp(-0.5 * ((x - c) / s) ** 2) / (s * np.sqrt(2 * np.pi))
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

        if full_sc < best_score:

    phase7_end = min(deadline, t_start + 1650.0)
    n_best = len(best_a)
    if time.monotonic() < phase7_end:
        n_restarts = min(5, int((phase7_end - time.monotonic()) / 35))

            if full_sc < best_score:
