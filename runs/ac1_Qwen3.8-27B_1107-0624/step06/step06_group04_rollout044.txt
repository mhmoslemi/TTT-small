import numpy as np
import time

def _full_convolve(a, b):
    # FFT-based convolution
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _proj_simplex(v):
    # Project onto probability simplex
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_weights(c, beta):
    cm = c.max()
    cs = (c - cm) / beta
    e = np.exp(cs)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return w
    return e / s

def _autocorr_grad(a, w):
    # Gradient of sum_k w_k * (a*a)[k] with respect to a
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    # The relevant part is wc[n-1:2n-1]
    h = wc[n-1:2*n-1]
    return 2.0 * h

    # where w_k = softmax(c_k / beta)

    n = len(a_init)

    t_val = 1.0

    c = _full_convolve(a, a)

    L = 4.0 * n / beta  # Upper bound on Lipschitz constant

            break

def _peaks_reduction(a, n_steps=5000, timeout=60.0, top_k=50):

import numpy as np
import time

# Helper functions (top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _autocorr_grad_k(a, k):
    # Gradient of (a*a)[k] w.r.t. a
    # g[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    n = len(a)
    g = np.zeros(n)
    # For i in range(n): valid if 0 <= k-i < n, i.e., k-n+1 <= i <= k
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    if i_min <= i_max:

        g[indices] = 2.0 * a[k - indices]
    return g
