"""
Optimized step function generator targeting score < 1.5027.
Uses FISTA with continuous beta annealing, peak-shaving coordinate descent,
and multi-length exploration (up to n=16384).

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _grad_autocorr(a, w):
    """
    Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a.
    w has length 2n-1 (the convolution length).
    Returns array of length n.
    """
    n = len(a)
    P = _next_pow2(2 * n)
    FA = np.fft.rfft(a, P)
    FW = np.fft.rfft(w, P)
    corr = np.fft.irfft(FA * np.conj(FW), P)
    return 2.0 * corr[:n]

def _fista_optimize(a_init, deadline, beta_init, beta_min, lr_start=0.01):
    """FISTA with continuous beta annealing on the simplex {a >= 0, sum a = 1}."""
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    
    total_time = max(1.0, deadline - time.monotonic())
    iter_count = 0
    iter_start = time.monotonic()
    
    # Estimate iterations per second
    it_per_sec = 2000  # rough estimate, will adjust
    
    while time.monotonic() < deadline - 1:
        iter_count += 1
        
        # Adaptive beta: decrease over time
        frac = min(1.0, iter_count / max(1, int(total_time * it_per_sec)))
        beta = beta_init * (beta_min / beta_init) ** frac
        
        # FISTA extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + ((t - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
            t = 1.0
            a_prev = a.copy()
            continue
        
        # Compute smoothed max
        c = _convolve(y, y)
        c_max = c.max()
        z = (c - c_max) / beta
        np.clip(z, -500, 500, out=z)
        e = np.exp(z)
        w_sum = e.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / w_sum
        
        # Gradient
        grad = _grad_autocorr(y, w)
        
        # Step size (Lipschitz constant estimate)
        # The smooth part has Lipschitz constant ~ 4/beta * max(w) * max(a)
        # Being conservative:
        L = 4.0 / beta
        step = 1.0 / L
        
        # Projected gradient step
        a_next = _proj_simplex(y - step * grad)
        
        # Check improvement

            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a
        a = a_next
        t = t_next
        
        # Adjust iterations per second estimate
        if iter_count % 1000 == 0:

            if elapsed > 1:
                it_per_sec = iter_count / elapsed
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score

def _peak_shaving(a, deadline, max_rounds=1000):
    """
    Coordinate descent that specifically targets the top peaks of the autocorrelation.
    For each coordinate, try to reduce its contribution to the top M peaks.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    M = min(20, 2 * n - 1)  # number of top peaks to focus on
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a, a)
        c_max = c.max()
        
        # Find top M peaks
        top_idx = np.argpartition(c, -M)[-M:]
        top_vals = c[top_idx]
        
        # For each top peak k, the contribution of a[j] is a[k-j] (if 0<=k-j<n)
        # We want to reduce a[j] where it contributes most to peaks
        # and increase a[j] where it contributes least
        
        improved = False
        # Random order for diversity
        rng = np.random.default_rng(rnd * 7919 + 42)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() > deadline + 60:  # allow some overrun
                break
            
            old = a[j]
            if old < 1e-15:
                continue
            
            # Try reducing a[j]
            for factor in [0.95, 0.98, 0.99]:
                a[j] = old * factor
                s = a.sum()
                a_norm = a / s

                    best_score = sc
                    a = a_norm.copy()
                    best_a = a.copy()
                    improved = True
                    break
            a[j] = best_a[j]  # restore
        
        if not improved and rnd > 50:
            break
    
    return best_a, best_score

            t += 1

            else:

For a probability vector a of length n, the autocorrelation c[k] has:

If the autocorrelation were perfectly flat at value 1/(2n-1) ≈ 1/(2n), the score would be 2n/(2n) ≈ 1. But we can't achieve this with non-negative a.

The constraint a >= 0 forces the autocorrelation to be "peaked" in certain places. The minimum achievable max autocorrelation for non-negative a of length n is what we're optimizing.

**New idea: Use the fact that the problem has a "duality" structure.**

The primal problem: min max_k c_k subject to a >= 0, sum a = 1
Dual: max t subject to t <= c_k for all k, a >= 0, sum a = 1

At the optimum, the active constraints (c_k = t) form a "support set" S. The KKT conditions give:
- For j with a_j > 0: sum_{k in S} lambda_k * (a_{k-j} + a_{j-k}) = mu
  (simplified: 2 * sum_{k in S} lambda_k * a_{k-j} = mu, where we use the symmetry)

This suggests an iterative approach:
1. Given current a, find the active set S = {k: c_k ≈ max c}
2. Solve for the Lagrange multipliers lambda_k
3. Update a to satisfy the KKT conditions

This is essentially what the "active set" LP approach in the strategy description does, but done in a continuous way.

**Let me implement a "subgradient cutting plane" method:**

Maintain a set of "cuts" (k values where c_k is large). At each iteration:
1. Compute c = conv(a, a)
2. Add the top few k values to the active set
3. Solve: min t subject to t >= c_k for k in active_set, a >= 0, sum a = 1
   (This is a quadratic program since c_k is quadratic in a)
4. Update a

Step 3 is the tricky part - it's a non-convex QP (since c_k is not convex in a for non-negative a).

But we can linearize: at the current point a*, c_k ≈ c_k(a*) + grad_k(a*)^T (a - a*) (first order). Then the problem becomes:
min t subject to t >= c_k(a*) + grad_k(a*)^T (a - a*) for k in S, a >= 0, sum a = 1

This is a linear program! We can solve it efficiently.

Actually wait, this is just linearized gradient descent with multiple constraints. It's similar to what FISTA does but with an explicit active set.

Let me think about a simpler but effective approach:

**"Proportional adjustment" method:**

At each step:
1. Compute c = conv(a, a), find max c = M
2. Compute the "excess" at each lag: e_k = max(0, c_k - M_target) where M_target < M
3. For each coordinate j, compute its "pressure": p_j = sum_k e_k * a_{k-j}
4. Update: a_j = a_j * (1 - eta * p_j / max(p)) for some step size eta
5. Renormalize: a = a / sum(a)

This is a multiplicative update that reduces coordinates with high "pressure" (i.e., coordinates that contribute a lot to the peaks).

The key advantage: it's simple, fast, and naturally maintains non-negativity.

Let me combine this with the gradient-based approach.

"""
Step function optimizer targeting score < 1.5027.
Combines FISTA with continuous beta annealing, multiplicative peak-shaving,
Adam in log-space, and multi-length exploration (up to n=16384).
Key new element over previous: multiplicative pressure-based updates that
directly target top autocorrelation peaks.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_autocorr(a, w):
    n = len(a)
    P = _next_pow2(2 * n)
    FA = np.fft.rfft(a, P)
    FW = np.fft.rfft(w, P)
    corr = np.fft.irfft(FA * np.conj(FW), P)
    return 2.0 * corr[:n]


def _fista_optimize(a_init, deadline, beta_init, beta_min):
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    t_val = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    iter_count = 0
    t_begin = time.monotonic()
    total_budget = max(1.0, deadline - t_begin)

    while time.monotonic() < deadline - 0.5:
        iter_count += 1
        frac = min(1.0, (iter_count * 0.02) / total_budget)
        beta = beta_init * (beta_min / beta_init) ** frac

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a + ((t_val - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
            t_val = 1.0
            a_prev = a.copy()
            continue

        c = _convolve(y, y)
        c_max = float(c.max())
        z = (c - c_max) / beta
        np.clip(z, -500, 500, out=z)
        e = np.exp(z)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / w_sum

        grad = _grad_autocorr(y, w)
        L = 4.0 / beta
        step = 1.0 / L
        a_next = _proj_simplex(y - step * grad)

        if iter_count % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a
        a = a_next
        t_val = t_next

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _multiplicative_peak_shave(a, deadline, max_iters=50000):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    M_focus = min(30, 2 * n - 1)
    eta = 0.05
    iter_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 1:
            break
        iter_count += 1

        c = _convolve(a, a)
        c_max = float(c.max())
        target = c_max * 0.999

        # Focus on top peaks
        if 2 * n - 1 > M_focus:
            top_idx = np.argpartition(c, -M_focus)[-M_focus:]
        else:
            top_idx = np.arange(2 * n - 1)

        excess = np.maximum(c[top_idx] - target, 0.0)

        if excess.max() < 1e-15:
            # All peaks are close to flat; use full spectrum
            excess = np.maximum(c - c_max * 0.999, 0.0)
            top_idx = np.arange(2 * n - 1)

        # Compute pressure on each coordinate
        # pressure[j] = sum_k excess[k] * a[k-j] (for valid k-j)
        # This is a cross-correlation of 'a' with 'excess' (padded)
        w = np.zeros(2 * n - 1)
        w[top_idx] = excess
        pressure = _grad_autocorr(a, w) * 0.5  # remove the factor of 2 from _grad_autocorr

        # Multiplicative update: reduce high-pressure coords
        max_p = pressure.max()
        if max_p > 1e-30:
            factor = 1.0 - eta * (pressure / max_p)
            factor = np.maximum(factor, 0.7)
            a = a * factor

        # Renormalize
        s = a.sum()
        if s < 1e-30:
            break
        a = a / s

        if iter_count % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            elif sc > best_score * 1.01:
                # Degrading; revert and reduce eta
                a = best_a.copy()
                eta *= 0.9

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, betas, lr=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta in betas:
        if time.monotonic() > deadline - 3:
            break
        max_iter = max(500, int((deadline - time.monotonic()) * 300))
        for _ in range(max_iter):
            if time.monotonic() > deadline - 3:
                break
            t += 1
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _convolve(a, a)
            c_max = float(c.max())
            z_c = (c - c_max) / beta
            np.clip(z_c, -500, 500, out=z_c)
            e = np.exp(z_c)
            w_sum = float(e.sum())
            if w_sum < 1e-300 or not np.isfinite(w_sum):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / w_sum
            grad = _grad_autocorr(a, w)
            grad_log = grad * a
            if not np.all(np.isfinite(grad_log)):
                break
            m_buf = b1 * m_buf + (1 - b1) * grad_log
            v_buf = b2 * v_buf + (1 - b2) * (grad_log * grad_log)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)
            update = lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 0.3:
                update *= 0.3 / un
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 100 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent_top(a_init, deadline, max_rounds=500):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    M = min(15, 2 * n - 1)
    rng = np.random.default_rng(777)

    for r in range(max_rounds):
        if time.monotonic() > deadline - 2:
            break
        c = _convolve(a, a)
        if 2 * n - 1 > M:
            top_k = np.argpartition(c, -M)[-M:]
        else:
            top_k = np.arange(2 * n - 1)

        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = rng.permutation(n)

        improved = False
        for j in indices:
            if time.monotonic() > deadline - 2:
                break
            old = a[j]
            if old < 1e-18:
                continue
            for factor in [0.96, 0.98, 0.995, 1.005, 1.02, 1.04]:
                a[j] = old * factor
                s = a.sum()
                if s < 1e-30:
                    a[j] = old
                    continue
                a_norm = a / s
                sc = _score_norm(a_norm)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
            a[j] = best_a[j] if improved else old

        if not improved and r > 30:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    out = np.exp(z - z.max())
    return out / out.sum()

    for li, target_n in enumerate(lengths):

            break

        if target_n == base_n:

            a = _interp_profile(parent, target_n)
