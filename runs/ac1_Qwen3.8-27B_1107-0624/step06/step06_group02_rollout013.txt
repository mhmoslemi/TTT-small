def _lp_refine(f, M, max_iter=100, K_init=50, deadline=None):
    """LP-based refinement using active set method.
    f: current sequence (sum = 1)
    M: current max of f*f
    Returns: new sequence (sum = 1) with hopefully lower max.
    """
    n = len(f)
    c = _convolve(f, f)
    
    # Start with top K lags as active constraints
    active = np.argsort(c)[::-1][:K_init]
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Form the constraint matrix for active constraints
        # A[k, j] = f_{k-j} if 0 <= k-j < n, else 0
        K = len(active)
        A_ub = np.zeros((K, n))
        for i, k in enumerate(active):
            # c_k = sum_j f_j * g_{k-j} where 0 <= k-j < n
            # So A[k, j] = f_j if k-j is in [0, n-1], i.e., j is in [k-n+1, k]
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            A_ub[i, j_lo:j_hi+1] = f[j_lo:j_hi+1]
        
        # LP: maximize sum(g) s.t. A_ub @ g <= M (repeated for each active constraint)... 
        # Wait, each constraint is (f*g)_k <= M, which is a ROW of A_ub times g <= M.
        # So the constraint is A_ub @ g <= M * ones(K)
        b_ub = np.full(K, M)
        
        # Bounds: g >= 0
        bounds = [(0, None)] * n
        
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Check all constraints
        fg = _convolve(f, g)
        violations = np.where(fg > M + 1e-10)[0]
        
        if len(violations) == 0:
            # No violations, g is feasible
            break
        
        # Add violated constraints (up to a max)
        new_active = np.union1d(active, violations[:50])
        # Keep only the most violated + top K
        order = np.argsort(fg)[::-1]
        active = order[:max(K_init, len(new_active))]
    
    return g

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * np.max(c))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(y, w):
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _grad_subgrad(y, tol_frac=0.05):
    """Exact subgradient of max(y*y) w.r.t. y."""
    c = _convolve(y, y)
    cmax = float(np.max(c))
    n = len(y)
    thresh = cmax * (1.0 - tol_frac)
    active = np.where(c >= thresh)[0]
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    return _grad_smoothed(y, w), cmax

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_active_set(f, M, deadline, max_iters=30):
    """LP-based refinement using active set method.
    Find g >= 0 maximizing sum(g) s.t. (f*g)_k <= M for all k.
    Returns g normalized to sum 1, or None if failed.
    """
    n = len(f)
    c = _convolve(f, f)

    # Initial active set: top K lags
    K_init = 80
    active = np.argsort(c)[::-1][:K_init]

    g = None
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        K = len(active)
        # Build constraint matrix A_ub of shape (K, n)
        A_ub = np.zeros((K, n))
        for i, k in enumerate(active):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_hi >= j_lo:
                A_ub[i, j_lo:j_hi + 1] = f[j_lo:j_hi + 1]

        b_ub = np.full(K, M)
        bounds = [(0, None)] * n
        c_obj = -np.ones(n)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                    'time_limit': max(1.0, deadline - time.monotonic() - 5) if deadline else None})
        except Exception:
            break

        if res.status != 0:
            break

        g = res.x
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Check all constraints
        fg = _convolve(f, g)
        max_fg = float(np.max(fg))
        if max_fg <= M * (1 + 1e-8):
            break  # Feasible!

        # Add violated constraints
        violations = np.where(fg > M * (1 + 1e-8))[0]
        # Merge: keep top max(K_init, 200) by violation amount
        new_set = np.union1d(active, violations)
        if len(new_set) > 300:
            # Prioritize most violated
            fg_on_new = fg[new_set]
            order = np.argsort(fg_on_new)[::-1]
            new_set = new_set[order[:300]]
            active = new_set
        else:
            active = new_set

    if g is None:
        return None
    return g / g.sum()


def _lp_refine_iteration(f, deadline, num_iters=20):
    """Perform multiple LP-based refinement iterations. Returns best f found."""
    n = len(f)
    best_f = f.copy()
    best_sc = _score_norm(f)

    for it in range(num_iters):

        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:

            mix = np.maximum(mix, 0)
            s = mix.sum()
            if s < 1e-10:

            mix = mix / s
            sc = _score_norm(mix)
            if sc < best_mix_sc:

                best_t = t

        if best_t == 0.0:
            break

        f_new = (1.0 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()

        if best_mix_sc < best_sc:
            best_sc = best_mix_sc
            best_f = f_new.copy()

def _diverse_starts(n, rng):
    """Generate diverse initial sequences of length n."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    starts.append(np.ones(n) / n)

    # Linear ramp (non-palindromic - key insight!)
    ramp = x + 1.0
    starts.append(ramp / ramp.sum())

    # Inverted ramp
    inv_ramp = n - x
    starts.append(inv_ramp / inv_ramp.sum())

    # Quadratic ramp
    quad = (x + 1.0) ** 2
    starts.append(quad / quad.sum())

    # Concave (sqrt)
    sqrt_x = np.sqrt(x + 1.0)
    starts.append(sqrt_x / sqrt_x.sum())

    # Sine (one period)
    sine = 1.0 + np.sin(2.0 * np.pi * x / n)
    sine = np.maximum(sine, 0.01)
    starts.append(sine / sine.sum())

    # Super-Gaussian p=4
    center = (n - 1) / 2.0
    sigma = n / 4.0
    sg4 = np.exp(-(((x - center) / sigma) ** 4))
    starts.append(sg4 / sg4.sum())

    # Two-peak (asymmetric)
    c1 = n * 0.25
    c2 = n * 0.65
    two = np.exp(-((x - c1) ** 2) / (n / 8) ** 2) + np.exp(-((x - c2) ** 2) / (n / 6) ** 2)
    starts.append(two / two.sum())

    # Exponential decay
    exp_dec = np.exp(-x / (n / 3.0))
    starts.append(exp_dec / exp_dec.sum())

    # Random smooth
    z = rng.standard_normal(n)
    z = np.cumsum(z) * 0.1
    smooth = np.exp(z)
    starts.append(smooth / smooth.sum())

    return starts


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

    # Add parent to starts if available
    starts = _diverse_starts(n_main, rng)
    if parent is not None:
        starts.insert(0, parent.copy())
        # Also add perturbed parent
        z_p = np.log(np.maximum(parent, 1e-15))
        z_p += 0.03 * rng.standard_normal(n_main)
        starts.append(_softmax(z_p - z_p.mean()))

    # Quick FISTA from each start
    per_start_time = max(5.0, (phase1_end - t_start - 10) / len(starts))
    for i, start in enumerate(starts):
        if time.monotonic() > phase1_end - 2:
            break
        start = np.maximum(start, 0)
        s = start.sum()
        if s < 1e-10:
            continue
        start = start / s

        c_s = _convolve(start, start)
        max_cs = float(c_s.max())
        beta = 0.05 * max_cs

        a_opt, sc = _fista_smoothed(start, beta, time.monotonic() + per_start_time,
                                    max_iter=200000)
        a_opt = a_opt / a_opt.sum()
        sc_full = _score(a_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()
            print(f"  Start {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 1: best_score={best_score:.10f}")

    # === Phase 2: FISTA with aggressive beta continuation ===
    print("Phase 2: FISTA beta continuation")
    phase2_end = min(deadline, t_start + 600.0)
    a_curr = best_seq.copy()
    a_curr = a_curr / a_curr.sum()

    c_curr = _convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    betas_list = []
    beta = 0.02 * max_cc
    while beta > 1e-11 * max_c0:
        betas_list.append(beta)
        beta *= 0.4
    betas_list.append(1e-11 * max_c0)
    if len(betas_list) > 30:
        step = max(1, len(betas_list) // 30)
        betas_list = betas_list[::step] + [1e-11 * max_c0]

    for i, beta in enumerate(betas_list):
        if time.monotonic() > phase2_end - 5:
            break
        per_time = min(30.0, phase2_end - time.monotonic() - 3)
        if per_time < 2:
            break
        max_it = max(5000, int(per_time * 5000))
        a_curr, sc = _fista_smoothed(a_curr, beta, time.monotonic() + per_time,
                                     max_it, use_subgrad_switch=True)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 2: best_score={best_score:.10f}")

    # === Phase 3: Adam log-space fine optimization ===
    print("Phase 3: Adam log-space")
    phase3_end = min(deadline, t_start + 1050.0)
    a_work = best_seq.copy() / best_seq.sum()

    c_w = _convolve(a_work, a_work)
    max_cw = float(c_w.max())

    betas_fine = []
    beta = 0.001 * max_cw
    while beta > 1e-13 * max_c0:
        betas_fine.append(beta)
        beta *= 0.35
    if len(betas_fine) > 18:
        step = max(1, len(betas_fine) // 18)
        betas_fine = betas_fine[::step] + [1e-13 * max_c0]

    per_time = phase3_end - time.monotonic() - 15
    if per_time > 30:
        a_opt, sc_opt = _adam_logspace(
            a_work, time.monotonic() + per_time,
            betas_fine, lr_init=0.008, max_iter_total=5000000
        )
        sc_opt_full = _score(a_opt)
        print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()

    a_work = best_seq.copy() / best_seq.sum()
    print(f"End Phase 3: best_score={best_score:.10f}")

    # === Phase 4: LP-based refinement ===
    print("Phase 4: LP active-set refinement")
    phase4_end = min(deadline, t_start + 1350.0)
    lp_time = phase4_end - time.monotonic() - 10
    if lp_time > 30:
        a_lp, sc_lp = _lp_refine_iteration(a_work, time.monotonic() + lp_time,
                                           num_iters=15)
        sc_lp_full = _score(a_lp)
        print(f"  LP refine: score={sc_lp_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_lp_full < best_score:
            best_score = sc_lp_full
            best_seq = a_lp.copy()

    # === Phase 5: Second round of Adam after LP ===
    print("Phase 5: Post-LP Adam refinement")
    phase5_end = min(deadline, t_start + 1550.0)
    a_work = best_seq.copy() / best_seq.sum()
    c_w = _convolve(a_work, a_work)
    max_cw = float(c_w.max())

    betas_5 = [0.0005 * max_cw, 0.0002 * max_cw, 0.0001 * max_cw,
               0.00005 * max_cw, 0.00001 * max_cw, 0.000005 * max_cw]
    per_time = phase5_end - time.monotonic() - 10
    if per_time > 30:
        a_opt, sc_opt = _adam_logspace(
            a_work, time.monotonic() + per_time,
            betas_5, lr_init=0.005, max_iter_total=3000000
        )
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Post-LP Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Coordinate descent polish ===
    print("Phase 6: Coordinate descent")
    phase6_end = min(deadline, t_start + 1700.0)
    if best_seq is not None:
        a_cd = best_seq.copy() / best_seq.sum()
        cd_deadline = min(phase6_end, time.monotonic() + 100)
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=300)
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
        print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final FISTA with subgradient ===
    print("Phase 7: Final subgradient FISTA")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_fin = best_seq.copy() / best_seq.sum()
        c_fin = _convolve(a_fin, a_fin)
        max_cfin = float(c_fin.max())
        beta_fin = 0.00001 * max_cfin
        per_t = deadline - time.monotonic() - 10
        if per_t > 10:
            a_fin, sc_fin = _fista_smoothed(a_fin, beta_fin, time.monotonic() + per_t,
                                            max_iter=100000, use_subgrad_switch=True,
                                            subgrad_tol=0.01)
            a_fin = a_fin / a_fin.sum()
            sc_fin_full = _score(a_fin)
            if sc_fin_full < best_score:
                best_score = sc_fin_full
                best_seq = a_fin.copy()
            print(f"  Final FISTA: score={sc_fin_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
