**Approach: Spectral method**

The condition that a_i >= 0 and sum(a_i) = 1 means a is a probability distribution. The convolution conv(a,a) is the distribution of X+Y where X,Y are i.i.d. with distribution a.

The challenge is finding a non-negative distribution a on {0,...,n-1} such that a*a is as close to uniform on {0,...,2n-2} as possible.

        A = np.fft.rfft(a, n)

        if a_new.sum() > 0:

        else:
            a_new = a  # fallback
            
        a = a_new
    
    return a

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Quantile-thresholded gradient (smoother, more focused than top-k or softmax)
2. Spectral shaping initialization for diverse starting points
3. Robust Adam with warmup, gradient clipping, periodic state resets
4. Coordinate-level local search for final polishing
5. Multi-restart with diverse strategies within time budget
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_quantile(a, c, percentile=90, power=1.5):
    """
    Compute gradient using quantile-thresholded weights.
    w_k = max(0, c_k - threshold)^power, threshold = percentile of c.
    Gradient: g_j = 4n * sum_k w_k * a[k-j] (with a_i=0 outside [0,n-1])
    Computed via FFT cross-correlation.
    """
    n = len(a)
    threshold = np.percentile(c, percentile)
    excess = np.maximum(c - threshold, 0.0)
    if power != 1.0:
        excess = excess ** power
    s = excess.sum()
    if s < 1e-30:
        idx = np.argmax(c)
        excess = np.zeros_like(c)
        excess[idx] = 1.0
        s = 1.0
    w = excess / s

    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    g = 4.0 * n * corr[:n]
    return g


def _compute_gradient_topk(a, c, k=3):
    """Compute gradient using top-k peaks (uniform weight)."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    g = 4.0 * n * corr[:n]
    return g


def _compute_gradient_softmax(a, c, temperature=None):
    """Compute gradient using softmax-weighted peaks."""
    n = len(a)
    if temperature is None:
        temperature = 0.01 * c.max()
    c_shifted = (c - c.max()) / max(temperature, 1e-15)
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    N = _next_pow2(3 * n)
    W = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    g = 4.0 * n * corr[:n]
    return g


def _adam_z_space(a_init, deadline, percentile=90, power=1.5,
                  T_mode="quantile", lr_start=0.03, lr_end=0.002,
                  max_iter=300000, clip_norm=0.5,
                  warmup_frac=0.05, eval_interval=200,
                  patience_limit=8, seed=42):
    """
    Adam optimizer in z-space (a = softmax(z)) for minimizing 2n*max(conv(a,a)).
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    warmup_iters = int(max_iter * warmup_frac)

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)

        # Learning rate schedule with warmup
        if t < warmup_iters:
            lr_frac = t / warmup_iters
        else:
            lr_frac = (lr_start / lr_end) ** ((t - warmup_iters) / max(1, max_iter - warmup_iters))
        lr = min(lr_start, lr_frac) * lr_mult if t >= warmup_iters else lr_start * (t / warmup_iters) * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        # Compute gradient
        if T_mode == "quantile":
            g = _compute_gradient_quantile(a, c, percentile, power)
        elif T_mode == "topk":
            g = _compute_gradient_topk(a, c, k=5)
        else:
            temp = 0.01 * max(c.max(), 1e-15) * (1.0 + 0.1 * (1 - frac))
            g = _compute_gradient_softmax(a, c, temp)

        # Center gradient for simplex compatibility
        W_center = float(np.dot(a, g))
        z_grad = a * (g - W_center)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        # Gradient clipping
        gnorm = np.linalg.norm(z_grad)
        if gnorm > clip_norm:
            z_grad = z_grad * (clip_norm / gnorm)

        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        # Clip update
        unorm = np.linalg.norm(update)
        if unorm > 1.0:
            update = update * (1.0 / unorm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        # Evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1

            if patience >= patience_limit:
                # Reset: go back to best, reduce lr, fresh momentum
                if lr_mult > 0.05:
                    lr_mult *= 0.5
                else:
                    # Small perturbation
                    z = np.log(np.maximum(best_a, 1e-15)) + 0.01 * rng.standard_normal(n)
                    z = z - z.mean()
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _spectral_init(n, n_iters=100, seed=42):
    """
    Initialize a non-negative sequence with approximately flat spectrum.
    Alternates between time-domain non-negativity projection and
    frequency-domain spectral flattening.
    """
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()

    for it in range(n_iters):
        # Compute FFT
        A = np.fft.rfft(a, n)
        mag = np.abs(A)

        # Soft flattening: reduce peaks in |A[k]| for k >= 1
        target = np.median(mag[1:])
        alpha = 0.3 * (1.0 - it / n_iters) + 0.1
        # Scale down peaks, scale up valleys (gently)
        ratio = np.ones_like(mag)
        for k in range(1, len(mag)):
            if mag[k] > target:
                ratio[k] = (target / mag[k]) ** alpha
            elif mag[k] < target * 0.5:
                ratio[k] = (target / mag[k]) ** (alpha * 0.3)

        new_mag = mag * ratio
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, n)

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a
        a = a_new

    return a


def _local_search(a, deadline, max_rounds=50, seed=42):
    """
    Coordinate-level local search: try small perturbations to individual
    or small groups of coordinates and keep improvements.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a = a.copy()
    current_score = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Try random coordinate perturbations
        n_tries = 200
        for _ in range(n_tries):
            if time.monotonic() > deadline:
                break
            # Pick 2-5 random indices to perturb
            n_pert = rng.integers(2, 6)
            idx = rng.choice(n, size=n_pert, replace=False)

            # Random perturbation: move mass between selected coords
            pert = rng.standard_normal(n_pert)
            pert = pert - pert.mean()  # keep sum approximately the same
            scale = 0.005 * rng.random() + 0.001

            a_try = a.copy()
            a_try[idx] = a[idx] + scale * pert
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s

            sc = _score(a_try)
            if sc < current_score - 1e-10:
                a = a_try
                current_score = sc

        # Also try "smoothing" perturbations: blend with shifted version
        if rnd % 5 == 4:
            shift = rng.integers(1, 5)
            a_smooth = 0.95 * a + 0.05 * np.roll(a, shift)
            a_smooth = np.maximum(a_smooth, 0)
            a_smooth = a_smooth / a_smooth.sum()
            sc = _score(a_smooth)
            if sc < current_score - 1e-10:
                a = a_smooth
                current_score = sc

    return a, current_score


def _nesterov_step(a, deadline, max_iter=50000, lr_start=0.01, lr_end=0.001, seed=42):
    """
    Nesterov accelerated projected gradient for simplex-constrained problem.
    Works directly in a-space (not log-space).
    """
    n = len(a)
    a = a.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)  # momentum

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break

        lr = lr_start * (lr_end / lr_start) ** (t / max_iter)

        # Gradient at y
        c = _fft_convolve(y, y)
        g = _compute_gradient_quantile(y, c, percentile=85, power=1.0)

        # Nesterov: look-ahead
        # v_new = beta * v + (1-beta) * g
        # y_new = a_new + (t_k/(t_k+1)) * (a_new - a)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / (2.0 * t_k)
        t_k += 1

        v = 0.9 * v + 0.1 * g
        a_new = y - lr * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a

        y = a_new + ((t_k - 1.0) / t_k) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        a = a_new

        if t % 500 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
