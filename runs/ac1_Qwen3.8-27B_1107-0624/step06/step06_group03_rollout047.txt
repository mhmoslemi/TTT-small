import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_normalize(z):

    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:

    return e / s

    """Compute cross-correlation: out[i] = sum_j a[j] * b[j+i]"""

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(np.conj(fa) * fb, N)[:n_out]

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            
            # Top-K softmax weights
            if top_k < len(c):
                topk_thresh = np.partition(c, -top_k)[-top_k]
                topk_mask = c >= topk_thresh
            else:
                topk_mask = np.ones(len(c), dtype=bool)
            
            topk_indices = np.where(topk_mask)[0]
            if len(topk_indices) == 0:
                topk_indices = np.array([int(np.argmax(c))])
            
            c_topk = c[topk_indices]
            c_max = float(c_topk.max())
            exp_vals = np.exp((c_topk - c_max) / beta)
            w_sum = float(exp_vals.sum())
            
            w = np.zeros(len(c))
            if w_sum < 1e-300:
                w[topk_indices[0]] = 1.0
            else:
                w[topk_indices] = exp_vals / w_sum

            grad_a = 2.0 * _cross_correlation(a, w, n)
            
            # Log-space gradient
            mean_grad = float(np.dot(grad_a, a))
            grad_z = n * a * (grad_a - mean_grad)
            # Wait, for the normalized problem (sum a = 1), we minimize 2n * max(conv(a,a))
            # Using softmax parameterization a = softmax(z), the gradient is:
            # dF/dz_i = F * (a_i * grad_a_i - sum_j a_j * grad_a_j) * 2n
            # Actually: F = 2n * <w, conv(a,a)> (smoothed)
            # dF/dz_i = 2n * sum_j w_j * d(conv(a,a)_j)/dz_i
            # d(conv(a,a)_j)/dz_i = 2 * (a_i * (d a_{i+j}/dz_i + d a_{i-j}/dz_i))... this is complex
            #

            # dF/dz_i = sum_j (dF/da_j) * (da_j/dz_i) = sum_j (dF/da_j) * a_j * (delta_{ji} - a_i)
            #         = a_i * (dF/da_i - sum_k a_k * dF/da_k)
            
            grad_z = a * (grad_a - float(np.dot(grad_a, a)))
            
            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

        # Reset momentum at phase boundaries
        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

            # c_base[2i] = c[2i] - a_i * d[2i] = c[2i] - 0 = c[2i]
            # But we want: with a_i = t, conv_{2i} = c[2i] - a_i^2 + t^2 = c_base[2i] - a_i^2 + t^2
            # So I need: c_base[2i] = c[2i] - a_i^2
            
            c_base[2 * i] -= a_i * a_i  # Now c_base[2i] = c[2i] - a_i^2
            
            # Now: conv_j(t) = c_base[j] + t * d[j] for j != 2i
            #              conv_{2i}(t) = c_base[2i] + t^2
            
            # For optimization, let's define an "effective" value:
            # eff_j(t) = c_base[j] + t*d[j] for j != 2i
            # eff_{2i}(t) = c_base[2i] + t^2
            
            # The objective: f(t) = 2n * max_j eff_j(t) / (S0+t)^2
            
            # Find optimal t >= 0:
            # Candidates: t = 0, and critical points
            # For linear terms: t_j = S0 - c_base[j] / d[j] (if d[j] > 0)
            # For the quadratic term: derivative of (c_base[2i] + t^2) / (S0+t)^2 = 0
            #   => 2t(S0+t)^2 - (c_base[2i]+t^2)*2(S0+t) = 0
            #   => t(S0+t) - (c_base[2i]+t^2) = 0
            #   => t*S0 + t^2 - c_base[2i] - t^2 = 0
            #   => t*S0 = c_base[2i]
            #   => t = c_base[2i] / S0 (if S0 > 0)
            
            # For crossing points: (c_base[i1] + t*d[i1]) = (c_base[i2] + t*d[i2])
            # => t = (c_base[i2] - c_base[i1]) / (d[i1] - d[i2])
            
            # For efficiency, only consider top-K entries
            # Current max is at t = a_i, so find top-K by c_base[j] + a_i*d[j] (= c[j])
            top_idx = np.argsort(c)[-top_k_select:]
            
            t_candidates = [0.0, a_i]
            
            # Add critical points from top-K linear terms
            for j in top_idx:
                j_int = int(j)
                if j_int == 2 * i:
                    continue  # Handle quadratic separately
                if d[j_int] > 1e-15:
                    t_star = S0 - c_base[j_int] / d[j_int]
                    if 0 <= t_star <= 10.0 * max(a_i, 0.01):
                        t_candidates.append(t_star)
            
            # Quadratic critical point
            if S0 > 1e-15:
                t_quad = c_base[2 * i] / S0
                if 0 <= t_quad <= 10.0 * max(a_i, 0.01):
                    t_candidates.append(t_quad)
            
            # Crossing points among top few
            for idx_a in range(min(30, len(top_idx))):
                for idx_b in range(idx_a + 1, min(30, len(top_idx))):
                    ja, jb = int(top_idx[idx_a]), int(top_idx[idx_b])
                    if ja == 2 * i or jb == 2 * i:
                        continue
                    dd = d[ja] - d[jb]
                    if abs(dd) > 1e-15:
                        t_cross = (c_base[jb] - c_base[ja]) / dd
                        if 0 <= t_cross <= 10.0 * max(a_i, 0.01):
                            t_candidates.append(t_cross)
            
            t_candidates = [max(0.0, min(10.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i]
            
            # Evaluate f(t) at all candidates
            best_t = a_i
            best_f = np.inf
            for tc in t_candidates:
                # Compute max of all eff_j(tc)
                # For the top-K, compute exactly; for the rest, use an upper bound
                eff_top = c_base[top_idx] + tc * d[top_idx]
                # Fix the quadratic term
                if 2 * i in top_idx:
                    qi = np.where(top_idx == 2 * i)[0][0]
                    eff_top[qi] = c_base[2 * i] + tc * tc
                
                # For entries not in top_idx, use upper bound:
                # max of c_base[j] + tc*d[j] for j not in top_idx
                # Upper bound: max(c_base[j]) + tc * max(d[j]) for j not in top_idx
                # Or more tightly: we can skip this and just use the top-K max
                # (risk: a non-top-K entry might become the max at the new t)
                
                max_eff = float(eff_top.max())
                
                # Check a few other entries as a safeguard
                # (optional, for correctness)
                
                f_val = 2.0 * n * max_eff / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            
            if best_t != a_i:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                improved += 1
            
            if improved % 200 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score
