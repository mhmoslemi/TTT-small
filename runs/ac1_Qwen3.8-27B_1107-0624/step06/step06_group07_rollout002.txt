"""
Adam + L-BFGS hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses progressive log-sum-exp smoothing, Adam optimizer in log-space,
active-set subgradient polishing, and multi-restart diversification.
"""
import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def smooth_objective(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def smooth_gradient(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def active_gradient(a, c, n, tol_frac=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def adam_optimize(a_init, beta_schedule, deadline, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer in log-space with progressive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_a = a_init.copy()
    best_sc = score(a_init)
    
    current_beta = beta_schedule[0]
    beta_idx = 0
    
    # Determine how many total steps we can afford
    # Each step: compute gradient (FFT-based), update (O(n))
    # Estimate time per step
    t0 = time.monotonic()
    g = smooth_gradient(z, current_beta, n)
    t_step = time.monotonic() - t0
    est_steps = max(100, int((deadline - time.monotonic()) / t_step))
    
    # Split beta schedule across steps
    steps_per_beta = max(10, est_steps // len(beta_schedule))
    
    for step in range(est_steps):
        if time.monotonic() > deadline:
            break
        
        # Update beta
        if steps_per_beta > 0 and step > 0 and step % steps_per_beta == 0 and beta_idx < len(beta_schedule) - 1:
            beta_idx += 1
            current_beta = beta_schedule[beta_idx]
        
        t += 1
        
        # Compute gradient
        g = smooth_gradient(z, current_beta, n)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate with cosine decay
        progress = step / est_steps
        lr_t = lr * 0.5 * (1 + np.cos(np.pi * progress))
        
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        # Check score periodically
        if step % 50 == 0 or step == est_steps - 1:
            a = softmax(z)
            sc = score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a, best_sc


def subgradient_refine(a_init, deadline, tol_frac=1e-5, max_steps=50000):
    """Subgradient method with backtracking line search in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = score(a_init)
    lr = 0.02
    stall = 0
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = softmax(z)
        c = full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(20):
            z_try = z - step * grad
            z_try = np.clip(z_try, -80.0, 80.0)
            z_try = z_try - z_try.mean()
            a_try = softmax(z_try)
            sc_try = score(a_try)
            
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
            
            if sc_try < score(a) - 1e-15:
                z = z_try
                improved = True
                break
            step *= 0.5
        
        if not improved:
            stall += 1
            lr *= 0.9
            if stall > 50:
                break
        else:
            stall = 0
            lr = min(lr * 1.02, 0.1)
    
    return best_a, best_sc


def lbfgs_polish(a_init, beta_list, deadline):
    """L-BFGS polishing with progressively smaller beta."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = score(current_a)
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        
        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()
        
        def fun(z):
            return smooth_objective(z, beta, n)
        
        def jac(z):
            return smooth_gradient(z, beta, n)
        
        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                             options={'maxiter': 3000, 'ftol': 1e-18,
                                      'gtol': 1e-14, 'maxcor': 30,
                                      'maxls': 40, 'disp': False})
            a_new = softmax(result.x)
            sc_new = score(a_new)
            if sc_new < current_sc:
                current_a = a_new
                current_sc = sc_new
        except Exception:
            pass
    
    return current_a, current_sc


def perturb(a, rng, method='gauss', sigma=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    if method == 'gauss':
        z = z + sigma * rng.standard_normal(n)
    elif method == 'fourier':
        t = np.arange(n, dtype=float)
        n_modes = 3
        for _ in range(n_modes):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = sigma * 0.3 * rng.uniform(1.0, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif method == 'block':
        n_blocks = 6
        block_size = max(4, n // n_blocks)
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + int(block_size * rng.uniform(0.5, 1.5)))
            z[start:end] += rng.uniform(-sigma * 0.5, sigma * 0.5)
    z = z - z.mean()
    return softmax(z)


def make_profile(n, kind='uniform'):
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'tapered':
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'quadratic':
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()
