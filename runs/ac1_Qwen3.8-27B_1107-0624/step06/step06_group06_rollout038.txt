"""
Active-set QP optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines direct KKT linear system solves with L-BFGS smoothing,
symmetry exploitation, and multi-n search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _hess_mv(s_active, v, n):
    """Compute H_S @ v where (H_S)_{i,j} = s_active[i+j].
    s_active: array of length 2n-1 (0/1 values)
    v: array of length n
    Returns: array of length n
    """
    v_flip = v[::-1].copy()
    conv = _full_convolve(s_active, v_flip)
    return conv[n - 1: 2 * n - 1]


def _solve_kkt(s_active, P, n, max_cg_iter=200, tol=1e-10):
    """Solve the KKT system for the active set QP.
    H_S[P,P] @ a_P = (lambda/2) * 1
    sum(a_P) = 1
    
    Returns a_P (array of length |P|) and lambda, or None if failed.
    """
    k = len(P)
    if k == 0:
        return None, None
    
    if k == 1:
        # Only one positive index: a_P[0] = 1
        return np.array([1.0]), 2.0 * s_active[2 * P[0]]  # Hmm, this isn't right
    
    # For small k, form the matrix directly
    if k <= 2000:
        P_arr = np.asarray(P)
        idx_i = P_arr[:, None]
        idx_j = P_arr[None, :]
        H = s_active[idx_i + idx_j]
        # Solve [H  1][a]   [lambda/2 * 1]
        #       [1' 0][lam/2]= [1             ]
        # Augmented system
        A = np.zeros((k + 1, k + 1))
        A[:k, :k] = H
        A[:k, k] = 1.0
        A[k, :k] = 1.0
        rhs = np.zeros(k + 1)
        rhs[k] = 1.0
        # Wait, the system is:
        # H @ a = (lambda/2) * 1
        # 1' @ a = 1
        # So: H @ a - (lambda/2) * 1 = 0
        #     1' @ a = 1
        # Written as:
        # [H  -1/2] [a]     [0]
        # [1'  0   ] [lam] = [1]
        # Or equivalently:
        # [H  -1] [a]       [0]
        # [1'  0  ] [lam'] = [1]  where lam' = lambda/2
        A = np.zeros((k + 1, k + 1))
        A[:k, :k] = H
        A[:k, k] = -1.0
        A[k, :k] = 1.0
        rhs = np.zeros(k + 1)
        rhs[k] = 1.0
        try:
            sol = np.linalg.solve(A, rhs)
            a_P = sol[:k]
            lam_half = sol[k]
            # Check non-negativity
            if np.all(a_P >= -1e-12):
                a_P = np.maximum(a_P, 0)
                a_P = a_P / a_P.sum()
                return a_P, 2.0 * lam_half
        except np.linalg.LinAlgError:
            pass
    
    # For large k or if direct solve failed, use CG
    # System: H_S[P,P] @ a = (lambda/2) * 1, 1' @ a = 1
    # Use augmented system with CG
    # Define the operator: for x = [a; 0] (k+1 vector), 
    # M @ x = [H_PP @ x[:k] - x[k]*1; sum(x[:k])]
    # Target: M @ x = [0; 1]
    
    def matvec(x):
        a_part = x[:k]
        lam_part = x[k]
        # Compute H_PP @ a_part
        # H_PP[i,j] = s_active[P[i] + P[j]]
        # (H_PP @ a)[i] = sum_j s_active[P[i]+P[j]] * a[j]
        # = sum_j s_active[P[i]+P[j]] * a[j]
        # This is: for each i, sum over j of s_active[P[i]+P[j]] * a_P[j]
        # Let's compute H_full @ a_full where a_full is the full n-vector with a on P and 0 elsewhere
        a_full = np.zeros(n)
        a_full[P] = a_part
        h_full = _hess_mv(s_active, a_full, n)
        h_pp = h_full[P]
        out = np.zeros(k + 1)
        out[:k] = h_pp - lam_part  # H_PP @ a - lam * 1
        out[k] = a_part.sum()
        return out
    
    b = np.zeros(k + 1)
    b[k] = 1.0
    
    # Use scipy CG
    from scipy.sparse.linalg import cg, aslinearoperator
    op = aslinearoperator(np.zeros((1,)), matvec=matvec, shape=(k+1, k+1))
    x0 = np.zeros(k + 1)
    x0[:k] = 1.0 / k
    try:
        sol, info = cg(op, b, x0=x0, maxiter=max_cg_iter, tol=tol)
        if info >= 0:
            a_P = sol[:k]
            lam_half = sol[k]
            if np.all(a_P >= -1e-10):
                a_P = np.maximum(a_P, 0)
                a_P = a_P / a_P.sum()
                return a_P, 2.0 * lam_half
    except Exception:
        pass
    
    return None, None


def _active_set_qp(a_init, deadline, max_outer=50):
    """Active set QP solver for the flat autocorrelation problem."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Determine active set
        # Start with a moderate tolerance
        tol_frac = 0.01
        active = c >= (c_max - tol_frac * c_max)
        if active.sum() < 3:
            # At least include the top few
            top_k = np.argsort(c)[::-1][:max(3, n // 10)]
            active = np.zeros(2 * n - 1, dtype=bool)
            active[top_k] = True
        
        # Determine positive set
        P = np.where(a > 1e-10)[0]
        if len(P) < 2:
            P = np.where(a > 1e-15)[0]
        
        s_active = active.astype(np.float64)
        
        # Solve KKT
        a_P, lam = _solve_kkt(s_active, P, n)
        
        if a_P is not None:
            a_new = np.zeros(n)
            a_new[P] = a_P
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
            
            # Check KKT conditions
            c_new = _full_convolve(a_new, a_new)
            c_new_max = float(c_new.max())
            
            # Check if any previously inactive constraint is now violated
            inactive_violated = (~active) & (c_new > c_new_max - 1e-10)
            # Check positive set: are there indices where (H_S a) < lambda/2?
            h_new = _hess_mv(s_active, a_new, n)
            negative_violated = (a_new <= 1e-12) & (2 * h_new < lam - 1e-8)
            
            if not inactive_violated.any() and not negative_violated.any():
                # KKT satisfied

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

def _simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _hess_mv(s_active, v, n):
    """Compute H_S @ v where (H_S)_{i,j} = s_active[i+j]."""
    v_flip = v[::-1].copy()

    """Solve KKT system directly for small |P|. Returns (a_P, lambda) or (None, None)."""
    k = len(P)

        a_P = np.array([1.0])
        lam = float(2.0 * s_active[2 * P[0]])

    P_arr = np.asarray(P, dtype=np.int64)
    idx_i = P_arr[:, None]
    idx_j = P_arr[None, :]
    H = s_active[idx_i + idx_j]
    A = np.zeros((k + 1, k + 1))
    A[:k, :k] = 2.0 * H
    A[:k, k] = -1.0
    A[k, :k] = 1.0
    rhs = np.zeros(k + 1)
    rhs[k] = 1.0
    try:
        # Use Cholesky if possible, else LU
        try:
            sol = np.linalg.solve(A, rhs)
        except np.linalg.LinAlgError:
            sol, _ = np.linalg.lstsq(A, rhs, rcond=None)
        a_P = sol[:k]
        lam = sol[k]
        if np.all(a_P >= -1e-10) and np.isfinite(lam):
            a_P = np.maximum(a_P, 0.0)
            s = a_P.sum()
            if s > 1e-15:

            return a_P, lam
    except Exception:

    return None, None

        h_pp = h_full[P_arr]
        out = np.empty(k + 1)
        out[:k] = 2.0 * h_pp - lam_part
        out[k] = a_part.sum()
        return out

    b[k] = 1.0
    x0 = np.zeros(k + 1)

    dummy = aslinearoperator(np.zeros((1,)), matvec=matvec, shape=(k + 1, k + 1))

def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    """Subgradient polish with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1: 2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
