"""
Multi-phase optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses Nesterov-accelerated subgradient with adaptive smoothing, multi-length exploration,
and structured restarts to beat the 1.5027 target.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(a, c, beta):
    """Compute gradient of smoothed objective 2n*(cmax + beta*log(sum(exp((c-cmax)/beta))))."""
    n = len(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad[i] = 2 * sum_k w[k] * 2*a[k-i] = 4 * (w * flip(a))[n-1+i]... 
    # Actually: grad[i] = 2 * sum_k w[k] * (d conv[k]/d a[i])
    # d conv[k]/d a[i] = 2*a[k-i] if 0<=k-i<n, else 0 (with the caveat i=k-i gives 2*a[i] still)
    # So grad[i] = 4 * sum_k w[k] * a[k-i] (for valid k-i)
    # This is 4 * cross_correlate(w, a)[i] where cross_correlate(w,a)[i] = sum_k w[k]*a[k-i]
    n_conv = len(c)
    N = _next_pow2(n_conv + n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fw * np.conj(fa), N)[:n]
    grad = 4.0 * corr
    return grad


def _grad_exact(a, c):
    """Exact subgradient using uniform weights on active set (near-max positions)."""
    n = len(a)
    c_max = float(c.max())
    tol = max(1e-10, 1e-5 * c_max)
    active = np.where(c >= c_max - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    n_conv = len(c)
    N = _next_pow2(n_conv + n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fw * np.conj(fa), N)[:n]
    grad = 4.0 * corr
    return grad


def _fista_opt(a_init, deadline, max_iter, beta_start, beta_end, n_phases=20):
    """FISTA with adaptive beta decay."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a_prev = a.copy()
    t = 1.0
    best_score = _score_norm(a)
    best_a = a.copy()

    # Estimate Lipschitz constant
    c_init = _full_convolve(a, a)
    max_c = float(c_init.max())
    L = 8.0 * n  # conservative upper bound
    step = 1.0 / L

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + ((t - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / sy

        # Adaptive beta
        phase = it / max(1, max_iter)
        beta = beta_start * (beta_end / beta_start) ** phase
        if beta < 1e-12:
            beta = 1e-12

        c_y = _full_convolve(y, y)
        grad = _grad_smoothed(y, c_y, beta)

        # Backtracking line search (simplified: just use fixed step with safety)
        a_next = _proj_simplex(y - step * grad)

        a_prev = a.copy()
        a = a_next
        t = t_next

        if it % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _nesterov_subgrad(a_init, deadline, max_iter, f_lower=1.0):
    """Nesterov-accelerated subgradient with Polyak step size."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    a_prev = a.copy()
    t = 1.0
    best_score = _score_norm(a)
    best_a = a.copy()
    f_best = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + ((t - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / sy

        c_y = _full_convolve(y, y)
        c_max = float(c.max())

        # Adaptive active set: use positions within 0.1% of max
        tol = 0.001 * c_max
        active = np.where(c_y >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c_y))])
        w = np.zeros(len(c_y))
        w[active] = 1.0 / len(active)

        # Subgradient
        n_conv = len(c_y)
        N = _next_pow2(n_conv + n)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(y, N)
        corr = np.fft.irfft(fw * np.conj(fa), N)[:n]
        grad = 4.0 * corr

        # Polyak step size
        f_y = 2.0 * n * c_max
        gap = f_y - f_lower
        gnorm2 = float(np.dot(grad, grad))
        if gnorm2 < 1e-30:
            break
        step = gap / gnorm2
        # Cap the step to avoid instability
        step = min(step, 1.0 / (n * 0.5))
        step = max(step, 1e-8)

        a_next = _proj_simplex(y - step * grad)

        a_prev = a.copy()
        a = a_next
        t = t_next

        if it % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                f_best = sc

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _adam_log_space(a_init, deadline, beta, lr_init=0.005, max_iter=500000):
    """Adam optimizer in log-space with smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        # Softmax to get a from z
        z_shifted = z - np.max(z)
        e = np.exp(z_shifted)
        e_sum = e.sum()
        if e_sum < 1e-300:
            a = np.ones(n) / n
        else:
            a = e / e_sum

        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        ee = np.exp(shifted)
        ee_sum = float(ee.sum())
        if ee_sum < 1e-300 or not np.isfinite(ee_sum):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ee / ee_sum

        # Gradient
        n_conv = len(c)
        N = _next_pow2(n_conv + n)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a, N)
        corr = np.fft.irfft(fw * np.conj(fa), N)[:n]
        grad_a = 4.0 * corr
        # Project gradient to log-space: grad_z[i] = a[i] * (grad_a[i] - sum(a*grad_a))
        corr_dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - corr_dot)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.2:
            update = update * (0.2 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 50 == 0:
            a_chk = a  # already computed above
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    # Final
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    e_sum = e.sum()
    a_final = e / e_sum if e_sum > 1e-300 else np.ones(n) / n
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_refine(a_init, deadline, beta):
    """L-BFGS-B refinement in log-space."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        zs = z - np.max(z)
        e = np.exp(zs)
        s = e.sum()
        if s < 1e-300:
            return 2.0 * n * 1.0
        a = e / s
        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        ee = np.exp(shifted)
        es = float(ee.sum())
        if es < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(es))

    def grad_fn(z):
        zs = z - np.max(z)
        e = np.exp(zs)
        s = e.sum()
        if s < 1e-300:
            return np.zeros(n)
        a = e / s
        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        ee = np.exp(shifted)
        es = float(ee.sum())
        if es < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ee / es
        n_conv = len(c)
        N = _next_pow2(n_conv + n)
        fw = np.fft.rfft(w, N)
        fa = np.fft.rfft(a, N)
        corr = np.fft.irfft(fw * np.conj(fa), N)[:n]
        grad_a = 4.0 * corr
        corr_dot = float(np.dot(a, grad_a))
        return a * (grad_a - corr_dot)

    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 20)), 'gtol': 1e-10, 'maxls': 10}
        )
        z_opt = result.x
        zs = z_opt - np.max(z_opt)
        e = np.exp(zs)
        a_opt = e / e.sum()
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as ex:
        print(f"  L-BFGS error: {ex}")

    zs = best_z - np.max(best_z)
    e = np.exp(zs)
    a_best = e / e.sum()
    return a_best, best_score


def _make_bump(n, asym=0.3, power=2.0):
    """Create an asymmetric bump: higher on one side."""
    t = np.linspace(0, 1, n)
    # Asymmetric shape: peak shifted
    center = 0.5 + asym * 0.3
    a = np.exp(-((t - center) / (0.3 * (1.0 + asym * (t - 0.5)))) ** 2 * power)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_sine_bump(n, power=1.5):
    """Symmetric sine bump."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    zs = z - np.max(z)
    e = np.exp(zs)
    return e / e.sum()

def _try_length(a_best, n_target, deadline):
    """Try to optimize at a different length by interpolating the best sequence."""
    n_orig = len(a_best)
    if n_target == n_orig:

    # Interpolate to target length
    x_orig = np.linspace(0, 1, n_orig)
    x_new = np.linspace(0, 1, n_target)
    a_interp = np.interp(x_new, x_orig, a_best)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s < 1e-15:
        return a_best, _score(a_best)
    a_interp = a_interp / s

    # Optimize at new length
    c = _full_convolve(a_interp, a_interp)
    max_c = float(c.max())
    beta = 0.003 * max_c
    a_opt, _ = _fista_opt(a_interp, time.monotonic() + min(60, deadline - time.monotonic() - 10),

    # Quick Adam polish

    if time.monotonic() < p3_end - 30:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.0005 * max_cc, 0.00015 * max_cc, 0.00005 * max_cc, 0.000015 * max_cc]
        per_beta = (p3_end - time.monotonic() - 15) / len(betas)
        for beta in betas:
            if time.monotonic() > p3_end - 15:
                break
            a_work, _ = _adam_log_space(a_work, time.monotonic() + max(5.0, per_beta),
                                        beta, lr_init=0.005, max_iter=300000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS refinement ===
    print("Phase 4: L-BFGS refinement")
    p4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < p4_end - 20:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        a4, _ = _lbfgs_refine(a_work, p4_end, beta=0.00005 * max_cc)
        sc4 = _score(a4)
        if sc4 < best_score:
            best_score = sc4
            best_seq = a4.copy()
            print(f"  L-BFGS: score={sc4:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a4

    # === Phase 5: Try different lengths ===
    print("Phase 5: Multi-length exploration")
    p5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < p5_end - 30:
        for n_try in [8192, 2048]:
            if time.monotonic() > p5_end - 30:
                break
            if n_try == n_main:
                continue
            try:
                a_try, sc_try = _try_length(best_seq, n_try, p5_end)
                if sc_try < best_score:
                    best_score = sc_try
                    best_seq = a_try.copy()
                    a_work = a_try.copy()
                    print(f"  n={n_try}: score={sc_try:.10f} t={time.monotonic()-t_start:.1f}s")
                else:
                    print(f"  n={n_try}: score={sc_try:.10f} (not better) t={time.monotonic()-t_start:.1f}s")
            except Exception as ex:
                print(f"  n={n_try} error: {ex}")

    # === Phase 6: Structured restarts ===
    print("Phase 6: Structured restarts")
    p6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < p6_end - 30:
        per_restart = (p6_end - time.monotonic() - 10) / 3
        inits = []
        # Small perturbation
        inits.append(_perturb(a_work, 0.01, rng))
        # Asymmetric bump blend
        bump = _make_bump(n_main, asym=0.2, power=2.5)
        blend = 0.7 * a_work + 0.3 * bump
        blend = np.maximum(blend, 1e-12)
        inits.append(blend / blend.sum())
        # Roll/shift
        shift = max(1, n_main // 256)
        rolled = np.roll(a_work, shift)
        inits.append(rolled / rolled.sum())

        for i, init in enumerate(inits):
            if time.monotonic() > p6_end - 10:
                break
            a_r, _ = _two_phase_optimize(init, time.monotonic() + max(5.0, per_restart),
                                         n_phases_coarse=5, n_phases_fine=5)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                a_work = a_r.copy()
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-polish ===
    print("Phase 7: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy() / best_seq.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            # Very fine Adam
            a_pol, _ = _adam_log_space(a_pol, time.monotonic() + remaining * 0.5,
                                       0.000005 * max_cp, lr_init=0.001, max_iter=100000)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f} t={time.monotonic()-t_start:.1f}s")
            # L-BFGS on remaining
            if time.monotonic() < deadline - 10:
                c_f = _full_convolve(a_pol, a_pol)
                a_pol, _ = _lbfgs_refine(a_pol, deadline - 3.0, beta=0.000003 * float(c_f.max()))
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f} t={time.monotonic()-t_start:.1f}s}")

    # Final output
    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
