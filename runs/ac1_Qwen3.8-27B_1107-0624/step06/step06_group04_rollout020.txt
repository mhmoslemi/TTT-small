```python
"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses parent solution as starting point, FISTA initialization, Adam in log-space
for main optimization, and coordinate descent for final polishing.
Explores n=4096 and n=8192.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    return e / s


def _compute_grad_normal(a, w):
    """Gradient of sum_k w_k * C_k w.r.t. a (not log-space)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _compute_grad_logspace(a, w, c):
    """Gradient of sum_k w_k * C_k w.r.t. z (log-space, softmax parameterization)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _fista(a_init, beta, deadline, max_iter):
    """FISTA with softmax smoothing on the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    a_curr = a_curr / s if s > 0 else np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 4.0 * n / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        y = y / sy if sy > 1e-30 else np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = e.sum()
        w = e / w_sum if w_sum > 1e-300 else np.zeros_like(c_y)
        if w_sum <= 1e-300:
            w[int(np.argmax(c_y))] = 1.0

        grad = _compute_grad_normal(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc = _score_norm(a_curr)
    if sc < best_score:
        best_score = sc
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter, lr_decay_every=5000, lr_decay_factor=0.95, grad_clip=1.0):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score_norm(a)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = e.sum()
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            grad = _compute_grad_logspace(a, w, c)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > grad_clip:
                update = update * (grad_clip / un)

            z = z - update
            z = np.clip(z, -50, 50)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve > 100 and t > 2000:
                current_lr = max(0.0003, current_lr * 0.95)
                if current_lr <= 0.0003:
                    break

    a_final = _softmax(z)
    sc = _score_norm(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, n_sweeps, deadline):
    """Coordinate descent: for each coordinate, try a few values and keep the best."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    # Precompute: for coordinate i, the convolution values involving a[i]
    # C[k] = sum_j a[j] * a[k-j]
    # If we change a[i] to a[i]+da:
    #   C[k] += 2*da*a[k-i] - da^2 (if k=2i)
    #   (for valid k, i.e., max(0, i-n+1) <= k <= min(2n-2, i+n-1))

    c = _full_convolve(a, a)  # (2n-1,)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        # Random order for diversity
        order = np.random.permutation(n)

        for i in order:
            # Values to try: current * factors
            ai = a[i]
            if ai < 1e-15:
                candidates = [1e-8, 1e-7, 1e-6, 1e-5]
            else:
                candidates = [ai * f for f in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]]

            best_val = ai
            best_max_c = float(c.max())

            for cv in candidates:
                if cv < 0:
                    continue
                da = cv - ai
                # Update convolution
                # C[k] changes by 2*da*a[k-i] for k in [i, i+n-1] (where k-i is valid)
                # minus da*(cv+ai) at k=2i (correction for the a[i]^2 term)
                # Actually: C[k] += 2*da*a[k-i] (for k-i in [0,n-1])
                # But at k=2i: the term is da*cv + da*ai - da*ai - da*ai = da*(cv+ai-2*ai) = da*(cv-ai) = da^2
                # Wait, let me recompute.
                # Original: C[k] contains a[i]*a[k-i] (from j=i) and a[k-i]*a[i] (from k-j=i)
                # If k != 2i: both terms are present, each changes by da*a[k-i], total change = 2*da*a[k-i]
                # If k == 2i: only one term a[i]*a[i] = a[i]^2, changes to cv^2, change = cv^2 - ai^2 = da*(cv+ai)
                
                k_min = max(0, i)  # k-i >= 0 => k >= i
                k_max = min(2 * n - 2, i + n - 1)  # k-i <= n-1 => k <= i+n-1
                c_trial = c.copy()
                c_trial[k_min:k_max+1] += 2.0 * da * a[k_min-i:k_max+1-i]
                if 2 * i >= k_min and 2 * i <= k_max:
                    # Correction: we added 2*da*a[i] at k=2i, but the actual change is da*(cv+ai) = da^2 + 2*da*ai
                    # So we need to subtract da^2
                    c_trial[2 * i] -= da * da
                
                trial_max = float(c_trial.max())
                if trial_max < best_max_c:
                    best_max_c = trial_max
                    best_val = cv

            # Apply best
            if best_val != ai:
                da = best_val - ai
                k_min = max(0, i)
                k_max = min(2 * n - 2, i + n - 1)
                c[k_min:k_max+1] += 2.0 * da * a[k_min-i:k_max+1-i]
                if 2 * i >= k_min and 2 * i <= k_max:
                    c[2 * i] -= da * da
                a[i] = best_val

        # Renormalize
        a = a / a.sum()
        # Recompute convolution (since normalization changed all values)
        c = _full_convolve(a, a)
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score
```

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Start from parent solution, use Adam in log-space with fine beta continuation,
coordinate descent polishing, and multi-restart diversification. Explores n=4096 and n=8192.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_normal(a, w):
    n = len(a)
    wc = _full_convolve(w, a[::-1])
    return 2.0 * wc[n - 1:2 * n - 1]


def _grad_logspace(a, w, c):
    n = len(a)
    wc = _full_convolve(w, a[::-1])
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _make_w(c, beta):
    cm = float(c.max())
    cs = (c - cm) / beta
    e = np.exp(cs)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _optimize_fista(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        y = y / sy if sy > 1e-30 else np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _make_w(c_y, beta)
        grad = _grad_normal(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 300 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc = _score_norm(a_curr)
    if sc < best_score:
        best_score = sc
        best_a = a_curr.copy()
    return best_a, best_score

    a = a_init / a_init.sum()

    z = z - z.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    no_imp = 0

    for beta in betas:
        if time.monotonic() > deadline:
            break
        while t < max_it:
            if time.monotonic() > deadline:
                break
            t += 1
            a = _softmax(z)
            c = _full_convolve(a, a)

                    unaff_max = max(unaff_max, tmp)
                else:
                    unaff_max = max(unaff_max, 0.0)

                    best_mc = trial_max
                    best_cv = cv

                improved = True

        a = a / a.sum()
        c = _full_convolve(a, a)
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

def _interp(a, n_new):

    an = np.maximum(an, 1e-12)
    return an / an.sum()
```

    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_w(c, beta):
    cm = float(c.max())
    cs = (c - cm) / beta
    e = np.exp(cs)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad_normal(a, w):
    n = len(a)
    wc = _full_convolve(w, a[::-1])
    return 2.0 * wc[n - 1:2 * n - 1]


def _grad_logspace(a, w, c):
    n = len(a)
    wc = _full_convolve(w, a[::-1])
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)
