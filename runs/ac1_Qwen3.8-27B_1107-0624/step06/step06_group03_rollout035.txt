    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _cross_corr(a, w):
    """Compute v_i = sum_j a_j * w_{j+i} for i=0..n-1.
    a has length n, w has length 2n-1.
    """
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # cross-correlation: IFFT(FFT(a) * conj(FFT(w)))
    v = np.fft.irfft(fa * np.conj(fw), N)[:n]
    return v

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_full(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _convolve_full(a, a)
    return float(2.0 * n * c.max())

def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoothed_obj_and_grad(z, n, beta, top_k=None):
    """Compute smoothed objective and gradient in z-space.
    a = softmax(z), objective = 2n * beta * logsumexp(c/beta)
    Gradient: 4n * a_i * (v_i - a_i * <w, c>)
    where v = cross_corr(a, w), w = softmax(c/beta) (or top-k softmax)
    """
    a = _softmax_normalize(z)
    c = _convolve_full(a, a)
    
    if top_k is not None and top_k < len(c):
        # Top-K softmax
        thresh = np.partition(c, -top_k)[-top_k]
        mask = c >= thresh
        c_masked = np.where(mask, c, -np.inf)
    else:
        c_masked = c
        mask = np.ones(len(c), dtype=bool)
    
    c_max = float(c_masked[mask].max()) if mask.any() else 0.0
    exp_c = np.exp((c_masked - c_max) / beta)
    w_sum = float(exp_c.sum())
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c_masked))] = 1.0
    else:
        w = exp_c / w_sum
    
    # Objective: 2n * beta * log(w_sum) + 2n * c_max (since we subtracted c_max)
    # Actually: logsumexp(c/beta) = log(sum exp(c_k/beta)) = c_max/beta + log(sum exp((c_k-c_max)/beta))
    # So beta * logsumexp(c/beta) = c_max + beta * log(sum exp((c_k-c_max)/beta)) = c_max + beta * log(w_sum)
    obj = 2.0 * n * (c_max + beta * np.log(w_sum))
    
    # Gradient
    wc = float(np.dot(w, c))
    v = _cross_corr(a, w)
    grad = 4.0 * n * a * (v - a * wc)
    
    return obj, grad

def _lfgs_smooth(z_init, n, beta, top_k, max_iter, time_limit):
    """Run L-BFGS-B on the smoothed objective in z-space."""
    t0 = time.monotonic()
    
    def fun_grad(z):
        if time.monotonic() > time_limit:
            raise ValueError("Time limit exceeded")
        obj, grad = _smoothed_obj_and_grad(z, n, beta, top_k)
        return obj, grad
    
    result = minimize(fun_grad, z_init, method='L-BFGS-B', jac=True,
                      options={'maxiter': max_iter, 'maxfun': max_iter,
                               'gtol': 1e-8, 'ftol': 1e-10})
    
    z_opt = result.x
    a_opt = _softmax_normalize(z_opt)
    return a_opt

Multi-phase optimizer using L-BFGS with log-sum-exp smoothing,
incremental coordinate descent polishing, and multi-length exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_full(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _cross_corr_ab(a, w):
    """v[i] = sum_j a[j]*w[j+i] for i=0..n-1, via FFT."""
    n = len(a)
    len_w = len(w)
    a_flipped = a[::-1]
    N = _next_pow2(n + len_w - 1)
    fa = np.fft.rfft(a_flipped, N)
    fw = np.fft.rfft(w, N)
    conv = np.fft.irfft(fa * fw, N)
    return conv[n - 1: 2 * n - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_full(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve_full(a, a)
    return float(2.0 * n * c.max())


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj_grad(z, n, beta, top_k):
    a = _softmax_normalize(z)
    c = _convolve_full(a, a)
    if top_k is not None and top_k < len(c):
        thresh = np.partition(c, -top_k)[-top_k]
        mask = c >= thresh
        c_work = np.where(mask, c, -1e30)
    else:
        c_work = c
    c_max = float(np.max(c_work))
    exp_c = np.exp((c_work - c_max) / beta)
    w_sum = float(exp_c.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = exp_c / w_sum
    obj = 2.0 * n * (c_max + beta * np.log(max(w_sum, 1e-300)))
    wc = float(np.dot(w, c))
    v = _cross_corr_ab(a, w)
    grad = 4.0 * n * a * (v - a * wc)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(z)
    return obj, grad


def _lbfgs_optimize(z_init, n, beta, top_k, max_iter, time_limit):
    """Single L-BFGS call with time management."""
    call_count = [0]
    timed_out = [False]

    def fun_grad(z):
        call_count[0] += 1
        timed_out[0] = time.monotonic() > time_limit
        if timed_out[0]:
            obj, grad = _smoothed_obj_grad(z, n, beta, top_k)
            return obj, grad
        return _smoothed_obj_grad(z, n, beta, top_k)

    result = scipy_minimize(fun_grad, z_init.copy(), method='L-BFGS-B', jac=True,
                            options={'maxiter': max_iter, 'gtol': 1e-7, 'ftol': 1e-12})
    return result.x, result.fun, not timed_out[0]


def _lbfgs_schedule(a_init, deadline, n_max_iter):

def _coord_descend_polish(a, deadline, n_passes=10, top_k_w=100):
    """Incremental coordinate descent polishing."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = _convolve_full(a, a)  # current convolution
    best_score = float(2.0 * n * c.max())

    rng = np.random.default_rng(777)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improvements = 0

        for i in order:
            if time.monotonic() > deadline:
                break

            a_i = a[i]
            if a_i < 1e-12:
                continue

            # d[j] = a[j-i] for valid j (0 <= j-i < n)
            # conv_new[j] = c[j] + 2*(t - a_i) * d[j]
            d = np.zeros(2 * n - 1)
            offsets = np.arange(n)
            j_idx = i + offsets
            valid = (j_idx < 2 * n - 1)
            d[j_idx[valid]] = a[offsets[valid]]

            S0 = 1.0 - a_i

            # Find candidate t values
            # Use top-k positions of current c
            top_idx = np.argpartition(c, -top_k_w)[-top_k_w:]

            t_candidates = [0.0, a_i * 0.3, a_i * 0.7, a_i, a_i * 1.4, a_i * 2.0]

            # Switch points between top pairs
            top_sorted = top_idx[np.argsort(c[top_idx])[::-1]][:15]
            for ii in range(min(10, len(top_sorted))):
                for jj in range(ii + 1, min(15, len(top_sorted))):
                    j1, j2 = top_sorted[ii], top_sorted[jj]
                    dd = d[j1] - d[j2]
                    if abs(dd) > 1e-15:
                        t_sw = a_i + (c[j2] - c[j1]) / (2.0 * dd)
                        if 0 <= t_sw <= 5 * max(a_i, 0.005):
                            t_candidates.append(t_sw)

            t_candidates = [max(0.0, min(10.0 * max(a_i, 0.01), tc)) for tc in t_candidates]
            t_candidates = list(set(np.round(t_candidates, 10)))
            if not t_candidates:
                t_candidates = [a_i]

            best_t = a_i
            best_f = best_score

            for tc in t_candidates:
                if tc == a_i:
                    continue
                conv_trial = c + 2.0 * (tc - a_i) * d
                max_cv = float(conv_trial.max())
                S = S0 + tc
                if S < 1e-10:
                    continue
                f_val = 2.0 * n * max_cv / (S * S)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != a_i:
                # Update a and c incrementally
                delta = 2.0 * (best_t - a_i)
                c = c + delta * d
                a[i] = best_t
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                    c = c / (s_a * s_a)  # renormalize convolution
                    # Recompute a from normalized version
                improvements += 1

                if improvements % 500 == 0:
                    sc = float(2.0 * n * c.max())
                    if sc < best_score:
                        best_score = sc

        # Recompute exact score at end of pass
        c_exact = _convolve_full(a, a)
        sc_exact = float(2.0 * n * c_exact.max())
        if sc_exact < best_score:
            best_score = sc_exact
        c = c_exact  # resync

    return a, best_score


def _structured_init(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos1":
        j = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * j / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "cos2":
        j = np.arange(n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 2 * j / n) + 0.1 * np.cos(2.0 * np.pi * 3 * j / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "taper":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2) + 0.2
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        a[:n // 4] *= 0.5
        a[3 * n // 4:] *= 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:

    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
