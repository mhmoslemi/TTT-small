def _pad_to_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _self_convolve(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _pad_to_pow2(out_n)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:out_n]


def _convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _pad_to_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _self_convolve(a)
    c_max = float(c.max())
    if beta < 1e-15:
        return 2.0 * n * c_max
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return 2.0 * n * (c_max + beta * np.log(s))

def _sgd(a_init, dl, ms=20000, lr0=0.015):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z -= z.mean()
    ba = a_init.copy()
    bs = _score(ba)
    bz = z.copy()
    lr = lr0
    pat = 0
    for i in range(ms):
        if time.monotonic() > dl:
            break
        a = _softmax(z)
        c = _self_conv(a)
        g = _tb_grad(a, c, n, 5)
        step = lr / (1.0 + 5e-4 * i)
        z = z - step * g
        z = np.clip(z, -50, 50)
        z -= z.mean()
        if i % 50 == 0:
            ac = _softmax(z)
            sc = _score(ac)
            if sc < bs - 1e-13:
                bs = sc
                ba = ac.copy()
                bz = z.copy()
                pat = 0
                lr = min(lr * 1.03, lr0 * 3)
            else:
                pat += 1
                lr *= 0.92
                if pat > 40:
                    rg = np.random.default_rng(i * 7 + 3)
                    z = bz + 0.02 * rg.standard_normal(n)
                    z -= z.mean()
                    pat = 0
                    lr = lr0 * 0.4
    af = _softmax(z)
    sf = _score(af)
    if sf < bs:
        bs = sf
        ba = af.copy()
    return ba, bs


def _proj_grad(a_init, dl, mi=500):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s
    ba = a.copy()
    bs = _score(a)
    for it in range(mi):
        if time.monotonic() > dl:
            break
        c = _self_conv(a)
        cm = float(c.max())
        tol = max(0.002 * cm, 1e-12)
        act = c >= (cm - tol)
        na = max(1, int(act.sum()))
        w = act.astype(float) / na
        ar = a[::-1]
        wc = _conv(w, ar)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 4.0 * n * a * (h - W)
        gt = g - np.mean(g)
        gn = np.linalg.norm(gt)
        if gn < 1e-13:
            break
        al = 0.5
        an = a.copy()
        for ls in range(50):
            at = a - al * gt
            at = np.maximum(at, 0)
            st = at.sum()
            if st < 1e-15:
                al *= 0.5
                continue
            at = at / st
            ct = _self_conv(at)
            ft = float(ct.max())
            if ft < cm - 1e-12 * al * gn:
                an = at
                break
            al *= 0.5
        a = an
        sc = _score(a)
        if sc < bs - 1e-14:
            bs = sc
            ba = a.copy()
    return ba, bs


def _profile(n, kind):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sine":
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        a = (np.cos(np.pi * t / 2)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "zig1":
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "zig2":
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "zig4":
        a = 1.0 + 0.25 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        a = np.abs(2.0 * t - 1.0) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _pert(a, sig, rg):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z += sig * rg.standard_normal(n)
    z -= z.mean()
    return _softmax(z)


def _pert_fft(a, rg, nm=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(nm):
        f = rg.integers(1, max(2, n // 3))
        amp = rg.uniform(0.01, 0.08)
        ph = rg.uniform(0, 2 * np.pi)
        z += amp * np.cos(2.0 * np.pi * f * t / n + ph)
    z -= z.mean()
    return _softmax(z)
