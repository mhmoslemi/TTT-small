"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improved over previous SOTA with: better LP active sets, adaptive temperature
annealing, n=8192 exploration, convolution-aware local search, and
extended final polishing phase.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=5):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_percentile_gradient(a, c, pct=0.85):
    n = len(a)
    thresh = np.percentile(c, pct * 100)
    active = c >= thresh
    if not np.any(active):
        active = np.zeros_like(c, dtype=bool)
        active[np.argmax(c)] = True
    p = np.zeros_like(c)
    p[active] = 1.0 / active.sum()
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _build_lp_data(n, threshold_frac=0.7):
    """Build LP constraint matrix, optionally using only tight constraints."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def _lp_step(a, rows, cols, c_curr, active_frac=0.5, time_limit=12):
    """LP step: find direction b that maximizes sum(b) s.t. a*b <= M on active set."""
    n = len(a)
    M = float(c_curr.max())
    thresh = M * active_frac
    active_mask = c_curr >= thresh
    if active_mask.sum() < 10:
        top_k = 10
        top_idx = np.argsort(c_curr)[::-1][:top_k]
        active_mask = np.zeros_like(c_curr, dtype=bool)
        active_mask[top_idx] = True
    active_rows = rows[active_mask[rows]]
    active_cols = cols[active_mask[rows]]
    vals = a[active_rows - active_cols]
    A = sp.csc_matrix((vals, (active_rows, active_cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return a, c_curr
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, c_curr
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in np.linspace(0.05, 1.0, 20):
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            c_curr = _fft_convolve(a, a)
        return a, c_curr
    except Exception:
        return a, c_curr


def _adam_z_space(a, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, mode='soft', active_k=5, pct=0.80,
                  eval_interval=200, clip_norm=0.6, momentum_boost=1.0):
    """Adam optimizer in z-space with adaptive gradient mode switching."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    current_lr_mult = 1.0
    grad_mode = 'soft'

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if mode == 'mixed':
            if T > T_start * 0.1:
                g = _compute_soft_gradient(a, c, T)
            elif T > T_start * 0.01:
                g = 0.5 * _compute_soft_gradient(a, c, T) + 0.5 * _compute_percentile_gradient(a, c, pct)
            else:
                g = _compute_active_gradient(a, c, active_k)
        elif mode == 'active':
            g = _compute_active_gradient(a, c, active_k)
        elif mode == 'percentile':
            g = _compute_percentile_gradient(a, c, pct)
        else:
            g = _compute_soft_gradient(a, c, T)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent_local(a, deadline, n_passes=10, step_size=0.02):
    """Coordinate-wise local search in log-space with small perturbations."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    c_curr = _fft_convolve(best_a, best_a)
    top_idx = np.argsort(c_curr)[::-1][:20]

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                break
            if i % 500 == 0 and p > 0:
                sc_check = _score_norm(best_a)
                if sc_check < best_score:
                    best_score = sc_check
            z_i = np.log(max(best_a[i], 1e-15))
            for delta in [step_size, -step_size, step_size * 0.5, -step_size * 0.5]:
                trial = best_a.copy()
                trial[i] = np.exp(z_i + delta)
                s = trial.sum()
                if s > 0:
                    trial = trial / s
                    sc = _score_norm(trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = trial.copy()
                        break
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, seed=42):
    """Generate a structured initial guess."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    candidates = []
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(rc)
    # Plateau (trapezoid)
    trap = np.clip((x - 0.1) / 0.2, 0, 1) * np.clip((0.9 - x) / 0.2, 0, 1)
    candidates.append(trap)
    # Sine bump
    sb = np.sin(np.pi * x) ** 2
    candidates.append(sb)
    # Gaussian
    mu = 0.5
    sig = 0.2
    g = np.exp(-0.5 * ((x - mu) / sig) ** 2)
    candidates.append(g)
    # Linear ramp
    ramp = x + 0.01
    candidates.append(ramp)
    # Uniform with noise
    uni = np.ones(n) + 0.1 * rng.standard_normal(n)
    candidates.append(np.maximum(uni, 0.01))
    best_cand = None
    best_sc = np.inf
    for cand in candidates:
        cand = np.maximum(cand, 1e-10)
        cand = cand / cand.sum()
        sc = _score_norm(cand)
        if sc < best_sc:
            best_sc = sc
            best_cand = cand.copy()
    return best_cand


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP -> Adam(mixed) -> Adam(active) -> n=8192 ->
    coord descent -> perturbation restarts -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = _structured_init(n)

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Also try structured init
    a_struct = _structured_init(n, seed=99)
    sc_struct = _score(a_struct)
    print(f"Structured init score: {sc_struct:.10f}")
    if sc_struct < best_score:
        best_score = sc_struct
        best_a = a_struct.copy()
        a = a_struct.copy()

    # Build LP data
    rows, cols = _build_lp_data(n)
    print(f"LP data built: {len(rows)} nnz, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: LP iterations (0 - 150s)
    phase1_end = min(deadline, t_start + 150.0)
    lp_count = 0
    c_curr = _fft_convolve(a, a)
    prev_sc = _score_norm(a)
    stall = 0
    active_fracs = [0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]

    while time.monotonic() < phase1_end:
        af = active_fracs[min(lp_count // 3, len(active_fracs) - 1)]
        a, c_curr = _lp_step(a, rows, cols, c_curr, active_frac=af, time_limit=10)
        sc = _score_norm(a)
        lp_count += 1
        if sc < prev_sc - 1e-10:
            stall = 0
        else:
            stall += 1
        full_sc = _score(a)
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a.copy()
        if lp_count % 2 == 0:
            print(f"  LP[{lp_count}] active_frac={af:.2f} score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
        if stall > 20:
            break
        prev_sc = sc
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Adam z-space, mixed mode (warm to cool) (150s - 500s)
    phase2_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase2_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_opt, sc_opt = _adam_z_space(best_a, phase2_end - 3,
                                       T_start=0.01 * max_c, T_end=0.0001 * max_c,
                                       lr_start=0.04, lr_end=0.004,
                                       max_iter=500000, mode='mixed',
                                       active_k=5, pct=0.80,
                                       eval_interval=200, clip_norm=0.6)
        full_sc = _score(a_opt)
        print(f"Phase 2 Adam-mixed: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # Phase 3: Adam z-space, percentile gradient (500s - 800s)
    phase3_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase3_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_opt, sc_opt = _adam_z_space(best_a, phase3_end - 3,
                                       T_start=0.003 * max_c, T_end=0.00003 * max_c,
                                       lr_start=0.03, lr_end=0.003,
                                       max_iter=400000, mode='percentile',
                                       active_k=5, pct=0.75,
                                       eval_interval=200, clip_norm=0.5)
        full_sc = _score(a_opt)
        print(f"Phase 3 Adam-percentile: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # Phase 4: Adam z-space, active set (800s - 1050s)
    phase4_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase4_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_opt, sc_opt = _adam_z_space(best_a, phase4_end - 3,
                                       T_start=0.001 * max_c, T_end=0.00001 * max_c,
                                       lr_start=0.02, lr_end=0.002,
                                       max_iter=300000, mode='active',
                                       active_k=4,
                                       eval_interval=200, clip_norm=0.4)
        full_sc = _score(a_opt)
        print(f"Phase 4 Adam-active: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # Phase 5: n=8192 exploration (1050s - 1250s)
    phase5_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < phase5_end:
        n_large = 8192
        a_large = _interp_to_length(best_a, n_large)
        c_large = _fft_convolve(a_large, a_large)
        max_cl = float(c_large.max())
        a_large_opt, sc_large = _adam_z_space(a_large, phase5_end - 3,
                                               T_start=0.002 * max_cl, T_end=0.00002 * max_cl,
                                               lr_start=0.03, lr_end=0.003,
                                               max_iter=200000, mode='mixed',
                                               active_k=5, pct=0.80,
                                               eval_interval=300, clip_norm=0.5)
        full_sc = _score(a_large_opt)
        print(f"Phase 5 n=8192: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_large_opt.copy()
        # Interpolate back to original n and optimize
        if full_sc < 2.0 and time.monotonic() < phase5_end - 30:
            n_orig = len(a)
            a_back = _interp_to_length(a_large_opt, n_orig)
            c_back = _fft_convolve(a_back, a_back)
            max_cb = float(c_back.max())
            a_back_opt, sc_back = _adam_z_space(a_back, phase5_end - 3,
                                                  T_start=0.001 * max_cb, T_end=0.00001 * max_cb,
                                                  lr_start=0.02, lr_end=0.002,
                                                  max_iter=100000, mode='active',
                                                  active_k=3,
                                                  eval_interval=200)
            full_sc_back = _score(a_back_opt)
            print(f"Phase 5 back to n={n_orig}: score={full_sc_back:.10f}")
            if np.isfinite(full_sc_back) and full_sc_back < best_score:
                best_score = full_sc_back
                best_a = a_back_opt.copy()

    # Phase 6: Coordinate descent local search (1250s - 1400s)
    phase6_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase6_end:
        a_cd, sc_cd = _coord_descent_local(best_a, phase6_end - 3,
                                            n_passes=5, step_size=0.015)
        full_sc = _score(a_cd)
        print(f"Phase 6 coord descent: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_cd.copy()

    # Phase 7: Perturbation restarts (1400s - 1600s)
    phase7_end = min(deadline, t_start + 1600.0)
    n_best = len(best_a)
    if time.monotonic() < phase7_end:
        for attempt in range(6):
            if time.monotonic() > phase7_end - 30:
                break
            sigma = 0.01 + 0.006 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            remaining = max(5.0, phase7_end - time.monotonic() - 5)
            t_budget = remaining / max(6 - attempt, 1)
            end_t = min(phase7_end - 3, time.monotonic() + t_budget)
            a_pert_opt, sc_pert = _adam_z_space(a_pert, end_t,
                                                 T_start=0.001 * max_cp, T_end=0.00003 * max_cp,
                                                 lr_start=0.02, lr_end=0.002,
                                                 max_iter=60000, mode='active',
                                                 active_k=3,
                                                 eval_interval=150)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 8: Final ultra-fine polish (1600s - deadline)
    if time.monotonic() < deadline - 8:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_z_space(best_a, deadline - 5,
                                           T_start=0.0001 * max_cf, T_end=0.000002 * max_cf,
                                           lr_start=0.004, lr_end=0.0002,
                                           max_iter=300000, mode='active',
                                           active_k=3,
                                           eval_interval=100, clip_norm=0.3)
        full_sc = _score(a_final)
        print(f"Phase 8 final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

        # One last coordinate descent pass
        if time.monotonic() < deadline - 5:
            a_cd2, sc_cd2 = _coord_descent_local(best_a, deadline - 3,
                                                  n_passes=2, step_size=0.008)
            full_sc2 = _score(a_cd2)
            if np.isfinite(full_sc2) and full_sc2 < best_score:
                best_score = full_sc2
                best_a = a_cd2.copy()
            print(f"Final coord descent: score={full_sc2:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
        s = 256.0
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
