def _compute_grad_smooth_max(a, beta, top_k=0):

    Returns (gradient, c) where c = conv(a,a).
    """

        # Top-K softmax
        thresh = np.partition(c, -top_k)[-top_k]
        mask = c >= thresh
        c_masked = c.copy()
        c_masked[~mask] = -np.inf

        exp_vals[~mask] = 0
        w_sum = exp_vals.sum()
        if w_sum < 1e-300:

        else:

    else:
        c_max = c.max()
        w = np.exp((c - c_max) / beta)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum
    
    # grad = 2 * cross_correlate(w, a)
    # cross_correlate(w, a)_i = sum_j w_j * a_{j-i} (for valid j-i)
    # In FFT: ifft(conj(fft(a)) * fft(w)) gives sum_j a_j * w_{j-i}...

    # This is the cross-correlation of a with w, evaluated at lag i.
    # In terms of convolution: cross_corr(a, w)_i = sum_j a_j * w_{j+i}
    # = sum_j a_j * w_{j+i}
    # Using FFT: ifft(fft(a) * conj(fft(w_shifted)))... 
    
    # Simpler: use the formula grad_i = 2 * sum_j w_j * a_{j-i}

    # grad_i = 2 * sum_{j: 0<=j-i<n} w_j * a_{j-i}

    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    
    return 2.0 * h, c, w

def _mirror_descent_step(a, g, eta):
    """Multiplicative update: a_new_i = a_i * exp(-eta * g_i / a_i) normalized.
    Actually, in mirror descent with entropy: a_new_i proportional to a_i * exp(-eta * g_i)
    where g_i is the gradient in a-space.
    """
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * g
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def _backtrack_eta(a, g, eta_init, max_iter=50):
    """Find eta via backtracking."""
    current_score = _score_norm(a)
    eta = eta_init
    for _ in range(max_iter):
        a_new = _mirror_descent_step(a, g, eta)
        new_score = _score_norm(a_new)
        if new_score < current_score:
            return eta, a_new
        eta *= 0.5
    return 0, a

"""
Mirror descent optimizer with adaptive peak equalization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multiplicative updates with Nesterov momentum, adaptive soft-max smoothing,
backtracking step sizes, and multi-length exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_grad_and_weights(a, beta, top_k=0):
    """Compute soft-max weights and gradient of weighted convolution."""
    n = len(a)
    c = _full_convolve(a, a)
    n_c = len(c)

    if top_k > 0 and top_k < n_c:
        thresh = np.partition(c, -top_k)[-top_k]
        mask = c >= thresh
        c_for_exp = np.where(mask, c, -np.inf)
    else:
        c_for_exp = c
        mask = np.ones(n_c, dtype=bool)

    c_max = c_for_exp[mask].max()
    exp_vals = np.exp((c_for_exp - c_max) / beta)
    exp_vals[~mask] = 0.0
    w_sum = exp_vals.sum()
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros(n_c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exp_vals / w_sum

    # Gradient: grad_i = 2 * sum_j w_j * a_{j-i} for valid (j-i) in [0, n-1]
    # This equals 2 * cross_correlation(w, a)[i] = 2 * ifft(conj(fft(a)) * fft(w))[:n]
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    grad = 2.0 * h
    return grad, c, w


def _mult_update(a, g, eta):
    """Multiplicative (mirror descent) update: a_new_i ~ a_i * exp(-eta * g_i), then normalize."""
    log_a = np.log(np.maximum(a, 1e-300))
    log_new = log_a - eta * g
    mx = log_new.max()
    if not np.isfinite(mx):
        return a.copy()
    log_new -= mx
    new = np.exp(log_new)
    s = new.sum()
    if s < 1e-300:
        return a.copy()
    return new / s


def _mirror_descent_phase(a, deadline, beta_start, beta_end, top_k, lr_init, lr_decay, momentum=True):
    """Main mirror descent loop with adaptive beta and optional Nesterov momentum."""
    n = len(a)
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    # Momentum state (in log-space velocity)
    v = np.zeros(n)
    beta = beta_start
    lr = lr_init
    max_beta_ratio = beta_start / max(beta_end, 1e-300)
    total_iterations = int(np.log(max_beta_ratio) / np.log(0.7)) if max_beta_ratio > 1 else 10000

    for it in range(max(1, total_iterations * 2)):
        if time.monotonic() > deadline:
            break

        # Decay beta
        beta = max(beta_end, beta * 0.7 ** (it / max(1, total_iterations)))

        # Compute gradient
        grad, c, w = _compute_grad_and_weights(a_curr, beta, top_k)

        # Transform to log-space gradient: dF/dz_i = a_i * dF/da_i (for F(a) with a=softmax(z))
        # Actually for mirror descent: we want z_new = z - lr * grad_z
        # where grad_z = a * grad_a (chain rule: dF/dz_i = dF/da_i * da_i/dz_i = grad_a_i * a_i)
        # But we're doing multiplicative updates which is equivalent to:
        # a_new_i = a_i * exp(-lr * grad_a_i) / Z
        # So we use grad_a directly in the multiplicative update.

        # Nesterov: use extrapolated point for gradient computation
        if momentum and it > 0:
            v = 0.9 * v - lr * grad
            a_trial = _mult_update(a_curr, -v, 1.0)  # v already has negative sign
            grad_ext, _, _ = _compute_grad_and_weights(a_trial, beta, top_k)
            a_curr = _mult_update(a_curr, grad_ext, lr)
        else:
            v = -lr * grad
            a_curr = _mult_update(a_curr, grad, lr)

        # Ensure non-negativity and normalization
        a_curr = np.maximum(a_curr, 0.0)
        s = a_curr.sum()
        if s < 1e-300:
            a_curr = np.ones(n) / n
        else:
            a_curr = a_curr / s

        # Decay learning rate
        lr = max(lr_decay * lr_init, lr * 0.9995)

        # Check improvement (every few iterations to save time)
        if it % 50 == 0:
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

    # Final check
    sc = _score_norm(a_curr)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_curr.copy()

    return best_a, best_score


def _fine_polish(a, deadline, top_k=30):

def _coord_polish(a, deadline, n_passes=2, top_k_check=50):
    """Lightweight coordinate descent: try setting each coordinate to 0 or to optimal value."""
    n = len(a)
    a_work = a.copy()
    best_score = _score_norm(a_work)
    rng = np.random.default_rng(99)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a_work[i]
            if old_val < 1e-12:
                continue

            # Try setting to 0
            a_work[i] = 0.0
            sc0 = _score_norm(a_work)
            # Try reducing by half
            a_work[i] = old_val * 0.5
            sc_half = _score_norm(a_work)
            # Try increasing by half
            a_work[i] = old_val * 1.5
            a_work = a_work / a_work.sum()
            sc_inc = _score_norm(a_work)

            # Pick best
            best_sc = min(sc0, sc_half, sc_inc, best_score)
            if best_sc == sc0:
                a_work[i] = 0.0
            elif best_sc == sc_half:
                a_work[i] = old_val * 0.5
                a_work = a_work / a_work.sum()
            elif best_sc == sc_inc:
                pass  # already set
            else:
                # Restore
                a_work[i] = old_val

            if _score_norm(a_work) < best_score:
                best_score = _score_norm(a_work)

    return a_work, best_score


def _structured_init(n, kind="cos_sum"):

        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)

            a += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.005)
        return a / a.sum()
    elif kind == "sine_mod":
        t = np.arange(n) / n
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * t) + 0.3 * np.sin(2 * np.pi * 7 * t + 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "bimodal_cos":
        t = np.arange(n) / n
        a = np.exp(-15 * (t - 0.3) ** 2) + np.exp(-15 * (t - 0.7) ** 2) + 0.2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()

            a0 = _interp_to_length(parent, n_try)

        sc0 = _score(a0)
        print(f"  Initial score: {sc0:.10f}")
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Phase 1: Coarse mirror descent (large to medium beta)
        phase1_end = time.monotonic() + min(remaining * 0.4, 400.0)
        c0 = _full_convolve(a0, a0)
        max_c0 = float(c0.max())

        a1, sc1 = _mirror_descent_phase(
            a0, phase1_end,
            beta_start=0.01 * max_c0,
            beta_end=0.001 * max_c0,
            top_k=100,
            lr_init=0.05,
            lr_decay=0.3,
            momentum=True
        )
        sc1_full = _score(a1)
        print(f"  Phase 1 (coarse MD): score={sc1_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc1_full < best_score:
            best_score = sc1_full
            best_seq = a1.copy()

        # Phase 2: Medium mirror descent
        phase2_end = time.monotonic() + min(remaining * 0.25, 250.0)
        c1 = _full_convolve(a1, a1)
        max_c1 = float(c1.max())

        a2, sc2 = _mirror_descent_phase(
            a1, phase2_end,
            beta_start=0.003 * max_c1,
            beta_end=0.0003 * max_c1,
            top_k=60,
            lr_init=0.02,
            lr_decay=0.3,
            momentum=True
        )
        sc2_full = _score(a2)
        print(f"  Phase 2 (medium MD): score={sc2_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc2_full < best_score:
            best_score = sc2_full
            best_seq = a2.copy()

        # Use best of phase 1 and 2 for polishing
        a_work = a2 if sc2_full < sc1_full else a1

        # Phase 3: Fine polish
        phase3_end = time.monotonic() + min(remaining * 0.2, 150.0)
        a3, sc3 = _fine_polish(a_work, phase3_end, top_k=30)
        sc3_full = _score(a3)
        print(f"  Phase 3 (fine polish): score={sc3_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc3_full < best_score:
            best_score = sc3_full
            best_seq = a3.copy()

        # Phase 4: Ultra-fine with even smaller beta
        if time.monotonic() < deadline - 60:
            a_work2 = a3 if sc3_full < sc2_full else a2
            phase4_end = min(deadline - 30, time.monotonic() + 60.0)
            c_work = _full_convolve(a_work2, a_work2)
            max_cw = float(c_work.max())
            a4, sc4 = _mirror_descent_phase(
                a_work2, phase4_end,
                beta_start=0.0002 * max_cw,
                beta_end=0.00002 * max_cw,
                top_k=30,
                lr_init=0.005,
                lr_decay=0.5,
                momentum=False
            )
            sc4_full = _score(a4)
            print(f"  Phase 4 (ultra-fine): score={sc4_full:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc4_full < best_score:
                best_score = sc4_full
                best_seq = a4.copy()

    # Phase 5: Diversity restarts from best solution
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Diversity restarts, remaining={deadline - time.monotonic():.1f}s ===")
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()
        n_best = len(a_base)

        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 20) / n_restarts

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break

            if ri == 0:
                # Gaussian perturbation
                p_start = a_base + 0.01 * rng.standard_normal(n_best)
            elif ri == 1:
                # Frequency perturbation
                freq = rng.uniform(1.0, 5.0)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert
            else:
                # Shift + blend
                shift = int(rng.integers(1, max(2, n_best // 2048)))
                p_shift = np.roll(a_base, shift)
                p_start = 0.7 * a_base + 0.3 * p_shift

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())

            p_best, sc_best = _mirror_descent_phase(
                p_start, time.monotonic() + max(5.0, per_restart),
                beta_start=0.002 * max_cs,
                beta_end=0.0001 * max_cs,
                top_k=50,
                lr_init=0.01,
                lr_decay=0.4,
                momentum=True
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {ri}: score={sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {ri}: score={sc_full:.10f} (no improvement)")

    # Final validation
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
