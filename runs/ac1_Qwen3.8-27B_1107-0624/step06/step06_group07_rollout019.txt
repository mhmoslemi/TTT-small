"""
Hybrid optimizer combining L-BFGS with log-sum-exp smoothing,
spectral equalization, adaptive subgradient polishing, and local search
to minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _spectral_equalize(a, alpha, max_iter, deadline):
    """Spectral equalization: iteratively flatten the power spectrum."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_cur = a.copy().astype(np.float64)
    best_a = a_cur.copy()
    best_sc = _score(a_cur)
    
    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a_cur, N)
        power = np.abs(A)**2
        
        # Compute target: geometric interpolation toward flat
        # Flat spectrum would have power = (sum(a))^2 / N at each bin
        # But we only care about the first 2n-1 time samples
        # For simplicity, just compress the dynamic range
        p_max = power.max()
        p_min = power.min()
        if p_max < 1e-30:
            break
        
        # Compress: new_power = power^alpha (reduces dynamic range for alpha < 1)
        # Or: new_power = power^alpha / mean(power)^alpha
        p_mean = power.mean()
        if p_mean < 1e-30:
            break
        
        new_power = (power / p_mean) ** alpha
        # Keep the DC component (A[0]) unchanged to preserve the sum
        # Actually, let's just scale the magnitude
        A_new_mag = np.sqrt(new_power)
        # Preserve phase
        phase = np.angle(A)
        A_new = A_new_mag * np.exp(1j * phase)
        # Preserve DC
        A_new[0] = A[0]
        
        # Transform back
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a_cur = a_new
    
    return best_a, best_sc


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _beta_schedule(a_start, deadline, min_time_remaining):
    """Multi-scale L-BFGS with progressive beta decrease."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    
    # More aggressive schedule with more steps
    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 0.00005, 0.00002, 0.00001])
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        est_iter_time = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(10000, max(100, int(remaining / (est_iter_time * 15))))
        
        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
    
    return current_a


def _subgrad_polish(a_init, deadline, tol_frac, max_steps, lr_init):
    """Subgradient polishing in z-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0
    best_lr = lr
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        step = lr / (1.0 + i * 0.0002)
        z = z - step * grad
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                best_lr = lr
                lr *= 1.05
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    lr = max(lr * 0.7, best_lr * 0.5)
                    z = z + 0.5 * step * grad  # partial rollback
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.9
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _nelder_mead_polish(a_init, deadline, max_iter=5000):
    """Nelder-Mead derivative-free polish for final convergence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    # Only polish a subset of coordinates for efficiency
    # Use the coordinates with the largest gradient contributions
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    active = c >= c_max - 0.01 * c_max
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    
    # Select top coordinates by |gradient|
    n_sub = min(50, n // 4)
    indices = np.argsort(np.abs(grad))[-n_sub:]
    
    def obj(x):
        z_full = z.copy()
        z_full[indices] = x
        a_try = _softmax(z_full)
        return _score(a_try)
    
    x0 = z[indices].copy()
    try:
        result = minimize(obj, x0, method='Nelder-Mead',
                         options={'maxiter': max_iter, 'xatol': 1e-10,
                                  'fatol': 1e-12, 'adaptive': True})
        z_new = z.copy()
        z_new[indices] = result.x
        a_new = _softmax(z_new)
        return a_new
    except Exception:
        return a_init


def _perturb(a, sigma, rng, mode="gauss"):
    """Perturb the sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    if mode == "gauss":
        z = z + sigma * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        n_modes = max(1, min(5, n // 100))
        for _ in range(n_modes):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = sigma * 0.5 * rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        n_blocks = max(2, n // 512)
        block_size = max(4, n // n_blocks)
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-sigma, sigma)
    
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _local_coordinate_search(a, deadline, max_rounds=500):
    """Local search moving pairs of coordinates."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.003
    rng_local = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Pick random coordinate pairs
        for _ in range(3):
            idx = int(rng_local.integers(0, n))
            jdx = int(rng_local.integers(0, n))
            if idx == jdx:
                jdx = (jdx + 1) % n
            
            s_pair = best_a[idx] + best_a[jdx]
            for delta in [step, 2*step, 0.5*step]:
                new_i = s_pair * (0.5 + delta)
                new_j = s_pair * (0.5 - delta)
                if new_i < 0 or new_j < 0:
                    continue
                
                a_try = best_a.copy()
                a_try[idx] = new_i
                a_try[jdx] = new_j
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)
                
                if sc_try < best_sc - 1e-13:
                    best_a = a_try.copy()
                    best_sc = sc_try
                    improved = True
        
        if not improved:
            step *= 0.8
            if step < 1e-7:
                break
    
    return best_a, best_sc
