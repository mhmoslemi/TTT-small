"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA + Adam + L-BFGS with multi-scale exploration and peak-shaving heuristics.
Key improvements over previous: multi-n exploration, better beta schedule,
active-set equalization, and more time on Adam.
"""
import numpy as np
import time
import sys


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    """Compute gradient of weighted sum of conv terms.
    grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    = 2 * (w_conv_a_rev)[n-1 : 2n-1] where w_conv_a_rev = convolve(w, a_reversed)
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

    n = len(a)
    c_max = float(c.max())
    if smooth_beta is not None and smooth_beta > 1e-15:
        w = np.exp((c - c_max) / smooth_beta)
        s = w.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / s
    else:
        if n_active is not None and n_active > 0:
            idx = np.argsort(c)[::-1][:n_active]
            w = np.zeros_like(c)
            w[idx] = 1.0 / n_active
        else:
            tol = 1e-6 * c_max
            active = np.where(c >= c_max - tol)[0]
            w = np.zeros_like(c)
            if len(active) > 0:
                w[active] = 1.0 / len(active)
            else:
                w[int(np.argmax(c))] = 1.0
    return _compute_grad(a, w)


def _adam_opt(a_init, deadline, beta, lr_init, max_iter, use_exact_sg=False):
    """Adam optimizer on log-softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    patience = 0
    best_patience_score = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        if use_exact_sg and beta < 1e-10 * cm:
            # Use exact subgradient (top k active)
            k_active = max(3, min(n, int(0.02 * n)))
            grad = _subgradient_max(a, c, n_active=k_active)
        else:
            e = np.exp((c - cm) / max(beta, 1e-15))
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                patience = 0
            else:
                patience += 1
            # Adaptive learning rate
            if patience > 20:
                current_lr *= 0.85
                patience = 0
            if patience > 50:
                current_lr = max(current_lr * 0.7, lr_init * 0.01)
                patience = 0

        if it % 5000 == 0:
            current_lr = max(lr_init * 0.01, current_lr * 0.95)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _fista_opt(a_init, beta, deadline, max_iter):
    """FISTA with proximal step on simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    """L-BFGS-B on log-space with smoothed objective."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 50)), 'gtol': 1e-10, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _peak_shave(a, num_shaves=5):
    """Try to reduce the top peaks in the convolution by adjusting nearby a values."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    for _ in range(num_shaves):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find top peak lag
        k_peak = int(np.argmax(c))
        # The peak conv[k_peak] = sum_i a[i]*a[k_peak-i]
        # To reduce it, reduce a[i] for i where a[i]*a[k_peak-i] is large
        # and redistribute to positions that don't contribute to this peak
        contrib = np.zeros(n)
        valid_i = np.arange(max(0, k_peak - n + 1), min(n, k_peak + 1))
        for i in valid_i:
            j = k_peak - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        # Reduce the top contributors
        top_idx = np.argsort(contrib)[::-1][:max(1, n // 50)]
        total_reduce = contrib[top_idx].sum() * 0.05
        # Add back to positions far from the peak
        far_mask = np.ones(n, dtype=bool)
        far_mask[max(0, k_peak - n // 4):min(n, k_peak + n // 4 + 1)] = False
        if not np.any(far_mask):
            far_mask = np.ones(n, dtype=bool)
        far_weight = a[far_mask].copy()
        if far_weight.sum() > 0:
            far_weight = far_weight / far_weight.sum()
            a[top_idx] -= total_reduce * contrib[top_idx] / max(contrib[top_idx].sum(), 1e-15)
            a[far_mask] += total_reduce * far_weight
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
    return a


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n, alpha=1.0):
    t = np.arange(n) - n / 2.0
    a = np.exp(-alpha * t * t / (n * n / 4.0))
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    n = len(a)
    if new_n == n:
        return a.copy()
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a)
    return result / result.sum()


def _quick_adam(a_init, deadline, beta=None, lr=0.005):
    """Quick Adam optimization wrapper."""
    n = len(a_init)
    if beta is None:
        c = _full_convolve(a_init, a_init)
        beta = 0.01 * float(c.max())
    return _adam_opt(a_init, deadline, beta, lr, max_iter=50000)

    sc_gblend = _score(a_gblend)
