def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

"""
Active-set LP optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses threshold-based active set identification, binary search line search,
Nesterov-accelerated subgradient polishing, and multi-scale n exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _build_lp_structure(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def _lp_solve(a, rows, cols, time_limit=30):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-15:
                return b / b.sum(), M
    except Exception:
        pass
    return a.copy(), M


def _binary_line_search(a, b, n, lo=0.01, hi=1.0, tol=0.01, max_iters=10):
    """Binary search for best t in [lo, hi] minimizing score((1-t)*a + t*b)."""
    best_t = 0.0
    best_sc = _score_norm(a)
    
    # First try t=1.0
    sc_1 = _score_norm(b)
    if sc_1 < best_sc:
        best_sc = sc_1
        best_t = 1.0
        # Try to extend beyond 1.0 (extrapolation)
        sc_2 = _score_norm(2.0 * b - a)
        if sc_2 < best_sc and (2.0 * b - a).min() >= 0:
            # Can't extrapolate easily due to non-negativity
            pass
    
    if best_t == 1.0:
        return b.copy(), best_sc
    
    # Binary search in [lo, best_t]
    lo_bs = lo
    hi_bs = best_t if best_t > 0 else hi
    if hi_bs <= lo_bs:
        hi_bs = hi
    
    for _ in range(max_iters):
        if hi_bs - lo_bs < tol:
            break
        mid = 0.5 * (lo_bs + hi_bs)
        a_mid = (1.0 - mid) * a + mid * b
        a_mid = np.maximum(a_mid, 0.0)
        sc_mid = _score_norm(a_mid)
        if sc_mid < best_sc:
            best_sc = sc_mid
            best_t = mid
            lo_bs = mid
        else:
            hi_bs = mid
    
    if best_t == 0.0:
        return a.copy(), best_sc
    return (1.0 - best_t) * a + best_t * b, best_sc


def _nesterov_polish(a, deadline, top_k=5, max_iter=200000):
    """Nesterov-accelerated projected subgradient method."""
    n = len(a)
    a_cur = a.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    step = 1.0 / (2.0 * n)
    eval_interval = 100
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Subgradient of max conv
        c = _fft_convolve(y, y)
        c_max = float(c.max())
        k_indices = np.argpartition(c, -top_k)[-top_k:]
        p = np.zeros(2 * n - 1)
        p[k_indices] = 1.0 / len(k_indices)
        p_rev = p[::-1]
        conv_full = _fft_convolve(y, p_rev)
        g = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
        
        # Scale gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        a_new = _project_simplex(y - step * g)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a_cur)
        y = _project_simplex(y)
        t_k = t_new
        a_cur = a_new
        
        # Adaptive step
        if (it + 1) % eval_interval == 0:
            sc = _score_norm(a_cur)
            if sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                step *= min(1.1, 1.0 + 0.01 * (it // eval_interval))
            else:
                step *= 0.9
    
    sc = _score_norm(a_cur)
    if sc < best_score:
        best_score = sc
        best_a = a_cur.copy()
    
    return best_a, best_score


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

"""
Active-set LP + Nesterov hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP with full constraint set, binary line search, Nesterov subgradient polishing,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _build_lp_structure(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def _lp_solve(a, rows, cols, time_limit=25):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-15:
                return b / b.sum(), M
    except Exception:
        pass
    return a.copy(), M

                update_best(a_opt, _score(a_opt))

        print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 5: Perturbation restarts
    ph5_deadline = min(deadline, t_start + 1600.0)
    if time.monotonic() < ph5_deadline and best_a is not None:
        n_best = len(best_a)
        print(f"Phase 5: Perturbation restarts, n={n_best}, t={time.monotonic()-t_start:.0f}s")
        rows_best, cols_best = _build_lp_structure(n_best)

        for attempt in range(6):
            if time.monotonic() > ph5_deadline - 50:
                break
            sigma = 0.01 + 0.008 * attempt
            pert = best_a + sigma * rng.standard_normal(n_best)
            a_pert = _project_simplex(np.maximum(pert, 0))
            sub_dl = min(ph5_deadline - 10, time.monotonic() + 80)
            a_opt, _ = _lp_optimize(a_pert, rows_best, cols_best, sub_dl, max_stall=10)
            update_best(a_opt, _score(a_opt))
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={_score(a_opt):.10f} "
                  f"t={time.monotonic()-t_start:.0f}s")

    # Phase 6: Final polish with Adam (very low temp, active set)
    if time.monotonic() < deadline - 10 and best_a is not None:
        print(f"Phase 6: Final Adam polish, t={time.monotonic()-t_start:.0f}s")
        a_fin, _ = _adam_z_space(best_a, deadline - 5, top_k=3,
                                  lr_start=0.008, lr_end=0.0005,
                                  max_iter=200000,
                                  T_start=0.0001, T_end=0.000005)
        update_best(a_fin, _score(a_fin))

    # Final output
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(final))
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
