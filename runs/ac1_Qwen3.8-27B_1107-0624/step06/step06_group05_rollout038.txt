"""
Dual-ascent optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses block coordinate descent: outer loop on dual variable lambda (via power/temperature schedule),
inner loop solves QP via FISTA with FFT-based convolution.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes a is normalized (sum=1). Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_Qa(a, lambda_rev, N_cache=None):
    """Compute Q*a = reverse(conv(a, lambda_rev))[0:n]"""
    n = len(a)
    # conv(a, lambda_rev) has length n + len(lambda_rev) - 1
    c = _fft_convolve(a, lambda_rev)
    # (Q a)_i = c[2n-2-i] for i=0,...,n-1
    # c has indices 0 to n + len(lambda_rev) - 2
    # We need c[2n-2-i] for i=0,...,n-1, i.e., c[n-1], c[n], ..., c[2n-2]
    return c[n-1:2*n-1][::-1]


def _fista_qp(a_init, lambda_rev, deadline, max_iter, L=None):
    """Solve min a^T Q a s.t. sum(a)=1, a>=0 using FISTA.
    Q is defined by (Q*a)_i = sum_j a_j * lambda_{i+j}
    """
    n = len(a_init)
    if L is None:
        L = 2.5  # Slightly above the theoretical bound of 2
    
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    step = 1.0 / L
    best_a = a_curr.copy()
    best_val = float(a_curr @ _compute_Qa(a_curr, lambda_rev))
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
            a_curr = y
            a_prev = y
            t_val = t_next
            continue
        
        qy = _compute_Qa(y, lambda_rev)
        a_next = _proj_simplex(y - step * 2.0 * qy)
        
        # Check for non-convergence
        if not np.all(np.isfinite(a_next)):
            break
        
        val = float(a_next @ _compute_Qa(a_next, lambda_rev))
        if val < best_val:
            best_val = val
            best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    return best_a, best_val

def _make_profile(n, kind="uniform"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "geometric":
        a = np.exp(-0.5 * t)  # Mild geometric
        return a / a.sum()
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    """Adam on logit space with smooth approximation of max conv."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _fft_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        # Gradient of the smooth approximation w.r.t. a
        a_rev = a[::-1]
        wc = _fft_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)  # This is grad of 2n*smooth_max w.r.t. a
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 50 == 0:
            a_chk = _softmax(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if t % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s
