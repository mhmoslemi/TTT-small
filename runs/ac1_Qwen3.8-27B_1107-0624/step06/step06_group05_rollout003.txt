Key improvements over previous approach:

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wa = _full_convolve(w, a_rev)
    h = wa[n - 1:2 * n - 1]
    return 2.0 * h


def _active_set_weights(c, frac=0.001):
    """Compute weights for active set: uniform over positions within frac of max."""
    c_max = float(c.max())
    if c_max < 1e-30:
        n_c = len(c)
        w = np.zeros(n_c)
        w[0] = 1.0
        return w
    threshold = c_max * (1.0 - frac)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    return w


def _active_set_weights_soft(c, frac=0.001, temperature=0.01):
    """Softer active set: weight by exp(-alpha*(max-c)/temperature) over active set."""
    c_max = float(c.max())
    if c_max < 1e-30:
        w = np.zeros(len(c))
        w[0] = 1.0
        return w
    threshold = c_max * (1.0 - frac)
    active_mask = c >= threshold
    if not np.any(active_mask):
        active_mask = np.zeros(len(c), dtype=bool)
        active_mask[int(np.argmax(c))] = True
    alpha = 1.0 / (temperature * c_max + 1e-30)
    w = np.zeros(len(c))
    w[active_mask] = np.exp(-alpha * (c_max - c[active_mask]))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w[active_mask] = 1.0 / np.sum(active_mask)
        return w
    w = w / w_sum
    return w


def _nesterov_subgrad(a_init, deadline, active_frac=0.002, max_iter=500000):
    """Nesterov accelerated projected subgradient with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    a_prev = a.copy()
    t_val = 1.0
    best_score = _score_norm(a)
    best_a = a.copy()

    # Initial step size
    c0 = _full_convolve(a, a)
    grad0 = _compute_grad(a, _active_set_weights(c0, active_frac))
    gnorm = float(np.sum(grad0 * a))
    step = 1.0 / (max(1.0, n * 1.5))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a + ((t_val - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n

        # Compute subgradient at y
        c_y = _full_convolve(y, y)
        w = _active_set_weights(c_y, active_frac)
        grad = _compute_grad(y, w)

        # Projected update
        a_next = _proj_simplex(y - step * grad)

        # Periodic score check
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        # Adaptive step size: reduce if score increased, increase slowly otherwise
        if it % 500 == 0 and it > 0:
            sc_curr = _score_norm(a_next)
            if sc_curr > best_score * 1.01:
                step *= 0.7
            else:
                step *= 1.02
            step = min(step, 2.0 / n)

        a_prev = a.copy()
        a = a_next
        t_val = t_next

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score

def _adam_opt(a_init, deadline, beta, lr_init=0.005, max_iter=500000):
    """Adam optimizer in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)

        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:

        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    """L-BFGS-B on log-space with log-sum-exp smoothing."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 50)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_parabolic(n, sharpness=2.0):

            print(f"    '{name}' optimized: {sc_prof:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_prof < best_score:
                best_score = sc_prof
                best_seq = a_prof.copy()
                a_work = a_prof

        # Try different n values if time permits
        if time.monotonic() < phase5_end - 40:
            for n_try in n_explore:
                if time.monotonic() > phase5_end - 30:
                    break
                if n_try == n_parent:
                    continue
                print(f"  Trying n={n_try}")
                prof_try = _make_parabolic(n_try, 2.0)
                n_left = phase5_end - time.monotonic() - 15
                if n_left < 30:
                    break
                a_try, _ = _nesterov_subgrad(prof_try, time.monotonic() + n_left * 0.5,
                                             active_frac=0.003)
                c_t = _full_convolve(a_try, a_try)
                b_t = 0.0005 * float(c_t.max())
                a_try, _ = _adam_opt(a_try, time.monotonic() + n_left * 0.4,
                                     b_t, lr_init=0.003, max_iter=100000)
                sc_try = _score(a_try)
                print(f"    n={n_try}: score={sc_try:.10f}, t={time.monotonic()-t_start:.1f}s")
                if sc_try < best_score:
                    best_score = sc_try
                    best_seq = a_try.copy()
                    a_work = a_try

    # Phase 6: L-BFGS final polish
    print("\n=== Phase 6: L-BFGS final polish ===")
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end - 15:
        c_final = _full_convolve(a_work, a_work)
        max_cf = float(c_final.max())
        # Try a few beta values for L-BFGS
        lb_betas = [0.0001 * max_cf, 0.00003 * max_cf, 0.000008 * max_cf]
        per_lb = (phase6_end - time.monotonic() - 10) / len(lb_betas)
        for beta_lb in lb_betas:
            if time.monotonic() > phase6_end - 10:
                break
            a_lb, _ = _lbfgs_opt(a_work, time.monotonic() + max(3.0, per_lb), beta_lb)
            sc_lb = _score(a_lb)
            print(f"  L-BFGS beta={beta_lb:.2e}: score={sc_lb:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_lb < best_score:
                best_score = sc_lb
                best_seq = a_lb.copy()
            a_work = a_lb

    # Phase 7: Ultra-fine final Adam
    print("\n=== Phase 7: Ultra-fine final ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_ultra = best_seq.copy()
        a_ultra = a_ultra / a_ultra.sum()
        c_u = _full_convolve(a_ultra, a_ultra)
        max_cu = float(c_u.max())
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_ultra, _ = _adam_opt(a_ultra, time.monotonic() + remaining * 0.5,
                                   0.000002 * max_cu, lr_init=0.001, max_iter=200000)
            sc_u = _score(a_ultra)
            print(f"  Ultra Adam: score={sc_u:.10f}")
            if sc_u < best_score:
                best_score = sc_u
                best_seq = a_ultra.copy()
            if time.monotonic() < deadline - 5:
                c_f2 = _full_convolve(a_ultra, a_ultra)
                b_f2 = 0.000001 * float(c_f2.max())
                a_ultra, _ = _lbfgs_opt(a_ultra, deadline - 3.0, b_f2)
                sc_f2 = _score(a_ultra)
                print(f"  Final L-BFGS: score={sc_f2:.10f}")
                if sc_f2 < best_score:
                    best_score = sc_f2
                    best_seq = a_ultra.copy()

    print(f"\nDone: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_parabolic(256, 2.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_parabolic(4096, 2.0)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
