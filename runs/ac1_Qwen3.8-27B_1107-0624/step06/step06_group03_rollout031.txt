"""
Refined optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses exact subgradient with backtracking line search, spectral equalization,
coordinate descent with 1D golden-section, and diverse restarts.
Key improvements over previous: sharper gradients (top-3 exact), adaptive step size,
spectral flatness enforcement, and more diverse exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

    """Get exact subgradient weights: uniform over top-k positions, or sharp softmax."""
    n_c = len(c)
    k = min(k_top, n_c)
    if k == 1:
        idx = np.argmax(c)
        w = np.zeros(n_c)
        w[idx] = 1.0
        return w
    # Use sharp softmax over top-k
    topk_thresh = np.partition(c, -k)[-k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    beta = 0.001 * max(c_max, 1e-10)
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum > 1e-300:
        w[topk_indices] = exp_vals / w_sum
    else:
        w[topk_indices[0]] = 1.0
    return w


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_with_backtrack(a_init, deadline, beta_schedule, top_k_schedule, max_iter_per_phase=50000):
    """FISTA with backtracking line search and adaptive smoothing."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for phase_idx, (beta, top_k) in enumerate(zip(beta_schedule, top_k_schedule)):
        if time.monotonic() > deadline:
            break
        phase_end = min(deadline, time.monotonic() + max(10, (deadline - time.monotonic()) / (len(beta_schedule) - phase_idx)))

        for it in range(max_iter_per_phase):
            if time.monotonic() > phase_end:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y, y)
            w = _exact_subgrad_weights(c_y, top_k)
            grad = _compute_grad_conv(y, w)

            # Backtracking
            a_next = _proj_simplex(y - step * grad)
            for bt in range(20):
                sc_next = _score_norm(a_next)
                if sc_next <= _score_norm(y) + 1e-12:
                    break
                step *= 0.5
                a_next = _proj_simplex(y - step * grad)
            else:
                step = max(step, 1e-10)
                a_next = _proj_simplex(y - step * grad)

            if it % 100 == 0 or it < 10:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()

            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace_refined(a_init, deadline, beta_schedule, top_k, lr_init, max_iter=200000):
    """Adam in log-space with refined beta schedule and gradient clipping."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for phase_idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _exact_subgrad_weights(c, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e6:
                grad = grad / grad_norm * 1e6

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_norm = 0.5
            if update_norm > max_norm:
                update = update * (max_norm / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 2000 == 0:
                current_lr = max(0.0002, current_lr * 0.98)

        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descend_exact(a, n_passes, deadline):
    """Coordinate descent with exact 1D optimization per coordinate."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a, a)
    rng = np.random.default_rng(777)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline:
                break
            a_i_old = a[i]
            if a_i_old < 1e-15:
                a[i] = 1e-5
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    c = _full_convolve(a, a)
                continue

            # d[k] = a[k-i] if 0 <= k-i < n else 0
            d = np.zeros(2 * n - 1)
            lo = max(0, i)
            hi = min(2 * n - 1, i + n)
            d[lo:hi] = a[lo - i:hi - i]

            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old

            # Find top-K positions for evaluation
            # The function to minimize: max_k(c_base[k] + d[k]*t) / (S0+t)^2
            # Try candidate t values
            t_candidates = [0.0, a_i_old * 0.1, a_i_old * 0.3, a_i_old * 0.5,
                           a_i_old * 0.7, a_i_old, a_i_old * 1.3, a_i_old * 1.5,
                           a_i_old * 2.0]
            # Add kink points from top active constraints
            top_idx = np.argpartition(c_base, -20)[-20:]
            for j in top_idx:
                if d[j] > 1e-15:
                    # kink where c_base[j] + d[j]*t crosses with another
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if -0.5 <= t_star <= 2.0 * max(a_i_old, 0.5):
                        t_candidates.append(max(0.0, t_star))

            t_candidates = [max(0.0, min(2.0, tc)) for tc in t_candidates]
            # Remove duplicates
            t_candidates = sorted(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                max_c_trial = float((c_base + d * tc).max())
                denom = (S0 + tc) ** 2
                if denom < 1e-20:
                    continue
                f_val = 2.0 * n * max_c_trial / denom
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            # Golden section refinement around best_t
            if best_t != a_i_old:
                lo_g = max(0.0, best_t * 0.5)
                hi_g = min(2.0, best_t * 2.0)
                for _ in range(20):
                    m1 = hi_g - (hi_g - lo_g) * 0.382
                    m2 = lo_g + (hi_g - lo_g) * 0.382
                    f1 = 2.0 * n * float((c_base + d * m1).max()) / (S0 + m1) ** 2
                    f2 = 2.0 * n * float((c_base + d * m2).max()) / (S0 + m2) ** 2
                    if f1 < f2:
                        hi_g = m2
                    else:
                        lo_g = m1
                    if hi_g - lo_g < 1e-10:
                        break
                best_t = (lo_g + hi_g) / 2.0
                f_final = 2.0 * n * float((c_base + d * best_t).max()) / (S0 + best_t) ** 2
                if f_final < best_f:
                    best_f = f_final

            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                c = _full_convolve(a, a)
                improved = True

        if not improved:
            break

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _spectral_equalize(a, deadline, n_trials=20):
    """Try to flatten the convolution by adjusting the spectral content."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        Fa = np.fft.rfft(a, N)
        c = _full_convolve(a, a)

        # Compute the "peak contribution" at each frequency
        # c[p] = (1/N) * sum_k Fa[k]^2 * exp(2*pi*i*k*p/N)
        # To reduce peaks, reduce frequencies that constructively contribute to peaks
        peak_pos = np.argmax(c)
        # Phase alignment: exp(-2*pi*i*k*peak_pos/N)
        phases = np.exp(-2j * np.pi * np.arange(N // 2 + 1) * peak_pos / N)
        # Contribution of each freq to the peak (real part)
        contrib = np.real(np.conj(Fa) * Fa * phases)  # |Fa|^2 * phase alignment
        # Actually: c[peak] ~ (1/N) * sum_k |Fa[k]|^2 * cos(2*pi*k*peak/N + 2*angle(Fa[k]))
        # Simplified: just try to reduce high-power frequencies
        power = np.abs(Fa) ** 2
        mean_power = power.mean()

        # Apply tilt: boost low-power, reduce high-power
        alpha = 0.02 + 0.03 * rng.random()
        tilt = 1.0 + alpha * (1.0 - power / (mean_power + 1e-10))
        tilt[0] = 1.0  # Don't touch DC
        Fa_new = Fa * np.sqrt(np.maximum(tilt, 0.1))

        a_new = np.fft.irfft(Fa_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

    return best_a, best_score


def _subgradient_step(a, step_size, k_top):
    """Single subgradient step with exact top-k gradient."""
    n = len(a)
    c = _full_convolve(a, a)
    w = _exact_subgrad_weights(c, k_top)
    grad = _compute_grad_conv(a, w)
    a_new = _proj_simplex(a - step_size * grad)
    return a_new


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        x = np.arange(n) / n
        for freq, amp in [(1, 0.2), (2, 0.15), (3, 0.1), (5, 0.05), (7, 0.03)]:
            a += amp * np.cos(2.0 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "oscillatory":
        # High-frequency oscillation to flatten autocorrelation
        x = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * x / n) + 0.2 * np.cos(2 * np.pi * 7.3 * x / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "rand_smooth":
        rng = np.random.default_rng(123)
        a = np.ones(n) + 0.3 * rng.standard_normal(n)
        # Smooth
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Primary search at n=4096
    n_main = 4096
    if parent is not None and len(parent) == n_main:
        a_work = parent.copy()
    else:
        a_work = _interp_profile(parent, n_main)

    sc_start = _score(a_work)
    print(f"Working at n={n_main}, initial score={sc_start:.10f}")

    # === Phase 1: FISTA with coarse-to-fine beta ===
    phase1_budget = 250.0
    phase1_end = time.monotonic() + min(phase1_budget, deadline - time.monotonic() - 600)
    print(f"\n--- Phase 1: FISTA (budget={phase1_end - time.monotonic():.0f}s) ---")

    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    beta_schedule = []
    topk_schedule = []
    beta = 0.01 * max_cc
    top_k = 50
    while beta > 0.00001 * max_cc:
        beta_schedule.append(beta)
        topk_schedule.append(top_k)
        beta *= 0.6
        top_k = max(5, int(top_k * 0.7))
    if len(beta_schedule) > 8:
        beta_schedule = beta_schedule[:8]
        topk_schedule = topk_schedule[:8]

    a_work, sc = _fista_with_backtrack(a_work, phase1_end, beta_schedule, topk_schedule)
    a_work = a_work / a_work.sum()
    sc_full = _score(a_work)
    print(f"  Phase 1 done: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_work.copy()

    # === Phase 2: Adam log-space (fine) ===
    phase2_end = min(deadline - 400, time.monotonic() + 350)
    print(f"\n--- Phase 2: Adam fine (budget={phase2_end - time.monotonic():.0f}s) ---")

    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())
    beta_fine = []
    beta = 0.001 * max_cc
    while beta > 0.000001 * max_cc:
        beta_fine.append(beta)
        beta *= 0.7
    if len(beta_fine) > 10:
        beta_fine = beta_fine[:10]

    a_work, sc = _adam_logspace_refined(a_work, phase2_end, beta_fine, top_k=5,
                                         lr_init=0.02, max_iter=300000)
    a_work = a_work / a_work.sum()
    sc_full = _score(a_work)
    print(f"  Phase 2 done: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_work.copy()

    # === Phase 3: Coordinate descent ===
    phase3_end = min(deadline - 300, time.monotonic() + 150)
    print(f"\n--- Phase 3: Coord descent (budget={phase3_end - time.monotonic():.0f}s) ---")
    if phase3_end > time.monotonic() + 10:
        a_work, sc = _coord_descend_exact(a_work, n_passes=3, deadline=phase3_end)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        print(f"  Phase 3 done: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()

    # === Phase 4: Spectral equalization ===
    phase4_end = min(deadline - 200, time.monotonic() + 60)
    print(f"\n--- Phase 4: Spectral equalization ---")
    if phase4_end > time.monotonic() + 5:
        a_spec, sc_spec = _spectral_equalize(a_work, phase4_end, n_trials=30)
        a_spec = a_spec / a_spec.sum()
        sc_spec_full = _score(a_spec)
        print(f"  Phase 4 done: score={sc_spec_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_spec_full < best_score:
            best_score = sc_spec_full
            best_seq = a_spec.copy()
            a_work = a_spec

    # Phase 4b: Coord descent after spectral
    if time.monotonic() < deadline - 150:
        phase4b_end = min(deadline - 150, time.monotonic() + 60)
        a_work, sc = _coord_descend_exact(a_work, n_passes=2, deadline=phase4b_end)
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
        print(f"  Phase 4b coord: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Diversity restarts ===
    remaining = deadline - time.monotonic()
    print(f"\n--- Phase 5: Diversity restarts (remaining={remaining:.0f}s) ---")

    if best_seq is not None and remaining > 60:
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()

        perturbation_types = [
            "noise", "shift", "freq", "block", "spectral_flip", "anti_peak"
        ]
        n_restarts = min(len(perturbation_types), int(remaining / 50))

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            ptype = perturbation_types[ri % len(perturbation_types)]
            per_restart_time = min(45.0, (deadline - time.monotonic() - 30) / (n_restarts - ri))

            if ptype == "noise":
                p_start = a_base + 0.01 * rng.standard_normal(n_best)
            elif ptype == "shift":
                shift = rng.integers(1, max(2, n_best // 512))
                p_start = np.roll(a_base, shift).copy()
            elif ptype == "freq":
                freq = rng.uniform(2.0, 10.0)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert
            elif ptype == "block":
                p_start = a_base.copy()
                bsize = n_best // 16
                bstart = rng.integers(0, n_best - bsize)
                p_start[bstart:bstart+bsize] *= (0.7 + 0.6 * rng.random())
            elif ptype == "spectral_flip":
                N = _next_pow2(2 * n_best - 1)
                Fa = np.fft.rfft(a_base, N)
                # Flip phase of random frequency bin
                k_flip = rng.integers(1, len(Fa))
                Fa[k_flip] = -Fa[k_flip]
                if k_flip < len(Fa) - 1:
                    Fa[len(Fa) - 1 - k_flip] = -Fa[len(Fa) - 1 - k_flip]
                p_start = np.fft.irfft(Fa, N)[:n_best]
            else:  # anti_peak
                c_base = _full_convolve(a_base, a_base)
                peak = np.argmax(c_base)
                # Perturb to reduce the peak
                d = np.zeros(n_best)
                for i in range(n_best):
                    ki = peak - i
                    if 0 <= ki < 2 * n_best - 1:
                        d[i] = a_base[ki] if ki < n_best else 0
                p_start = a_base - 0.005 * d / (np.abs(d).max() + 1e-10)

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            # Quick optimization on the perturbation
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.002 * max_cs, 0.0005 * max_cs, 0.0001 * max_cs]
            p_best, sc_best = _adam_logspace_refined(p_start, time.monotonic() + per_restart_time,
                                                      betas_r, lr_init=0.01, top_k=5,
                                                      max_iter=100000)
            # Quick coord descent
            if time.monotonic() < deadline - 40:
                p_best, sc_cd = _coord_descend_exact(p_best, n_passes=1,
                                                      deadline=min(deadline - 40, time.monotonic() + 15))
            sc_full = _score(p_best)
            status = "IMPROVED" if sc_full < best_score else ""
            print(f"  Restart {ri} ({ptype}): score={sc_full:.10f} {status} t={time.monotonic()-t_start:.1f}s")
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()

    # === Final: try n=8192 briefly if time permits ===
    if time.monotonic() < deadline - 80 and best_seq is not None:
        print(f"\n--- Trying n=8192 (remaining={deadline - time.monotonic():.0f}s) ---")
        n8 = 8192
        a8 = _interp_profile(best_seq, n8)
        sc8 = _score(a8)
        print(f"  n=8192 interpolated: score={sc8:.10f}")

        # Quick Adam optimization
        c8 = _full_convolve(a8, a8)
        max_c8 = float(c8.max())
        betas8 = [0.001 * max_c8, 0.0003 * max_c8, 0.0001 * max_c8]
        a8_opt, _ = _adam_logspace_refined(a8, min(deadline - 40, time.monotonic() + 50),
                                            betas8, lr_init=0.01, top_k=5, max_iter=100000)
        sc8_opt = _score(a8_opt)
        print(f"  n=8192 optimized: score={sc8_opt:.10f}")
        if sc8_opt < best_score:
            best_score = sc8_opt
            best_seq = a8_opt.copy()

    # Final output
    print(f"\n{'='*50}")
    print(f"FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"OUTPUT: score={sc_final:.10f}, n={len(final)}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
