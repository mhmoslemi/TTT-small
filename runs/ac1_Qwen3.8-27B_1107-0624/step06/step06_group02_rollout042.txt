                    continue

                # Compute new max conv
                # Affected k range: [j, min(j+n-1, 2n-2)]
                k_lo = j
                k_hi = min(j + n - 1, 2 * n - 2)

                # Linear update for k in [k_lo, k_hi], k != 2j
                a_slice = a[max(0, k_lo - j):min(n, k_hi - j + 1)]
                # Actually a[k-j] for k in [k_lo, k_hi] is a[0], a[1], ..., a[k_hi-j]
                a_idx = np.arange(k_lo - j, k_hi - j + 1)
                a_val = a[a_idx]
                c_changed = c[k_lo:k_hi + 1] + 2.0 * dv * a_val

                # Fix the 2j term
                idx_2j = 2 * j - k_lo
                if 0 <= idx_2j < len(c_changed):
                    c_changed[idx_2j] = c[2 * j] + v * v - u * u

                max_in = float(c_changed.max())

                # Unchanged parts
                max_out = 0.0
                if k_lo > 0:
                    max_out = max(max_out, float(c[:k_lo].max()))
                if k_hi < 2 * n - 2:
                    max_out = max(max_out, float(c[k_hi + 1:].max()))

                new_max = max(max_in, max_out)
                new_S = S - u + v
                if new_S < 0.01:
                    continue
                new_sc = 2.0 * n * new_max / (new_S * new_S)

                if new_sc < best_local - 1e-15:
                    best_local = new_sc
                    best_v = v

            if best_v != u:
                # Apply update with incremental conv change
                dv = best_v - u
                k_lo = j
                k_hi = min(j + n - 1, 2 * n - 2)
                a_idx = np.arange(k_lo - j, k_hi - j + 1)
                c[k_lo:k_hi + 1] += 2.0 * dv * a[a_idx]
                if 2 * j >= k_lo and 2 * j <= k_hi:
                    idx_2j = 2 * j - k_lo
                    c[2 * j] = c[2 * j] - 2.0 * dv * a[2 * j - j] + (best_v * best_v - u * u)
                    # Wait, I need to be more careful:
                    # The linear update added 2*dv*a[j] to c[2j]
                    # But the correct update for c[2j] is v^2 - u^2
                    # So I need to subtract 2*dv*a[j] and add v^2-u^2
                    # But a[j] is still u at this point (I haven't updated it yet)
                    # c[2j] already has 2*dv*u added; I need c[2j] += v^2-u^2-2*dv*u
                    # = c[2j] - 2*dv*u + v^2 - u^2
                    # Wait, I already did c[k_lo:k_hi+1] += 2*dv*a[a_idx]
                    # This added 2*dv*a[j]=2*dv*u to c[2j]
                    # The correct delta for c[2j] is v^2 - u^2
                    # So I need to fix: c[2j] += (v^2-u^2) - 2*dv*u = v^2-u^2-2(v-u)*u = v^2-u^2-2vu+2u^2 = v^2-2vu+u^2 = (v-u)^2 = dv^2
                    # So c[2j] += dv^2
                    c[2 * j] += dv * dv

                a[j] = best_v
                S = S - u + best_v
                sweep_improved = True

                if best_local < best_score:
                    best_score = best_local
                    best_a = a.copy()

        # Renormalize (drift correction)
        S_actual = float(a.sum())
        if S_actual > 0 and abs(S_actual - 1.0) > 1e-10:
            a = a / S_actual
            c = _fft_conv(a, a)
            c = np.maximum(c, 0.0)
            S = 1.0
            sc = float(2.0 * n * c.max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if not sweep_improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweep % 200 == 0:
            sc_now = _score_norm(a)
            print(f"    CD sweep {sweep}: cur={sc_now:.10f} best={best_score:.10f}")

    a_final = a / a.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_solve(f, M, n):
    """
    Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns optimal g (unnormalized).
    """
    # Constraint matrix A: A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # Shape: (2n-1) x n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            val = f[k - j]
            if val > 0:
                rows.append(k)
                cols.append(j)
                vals.append(val)

    if len(rows) == 0:
        return np.ones(n)

    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': 300, 'presolve': True})

    if res.status == 0:
        return np.maximum(res.x, 0)
    # Fallback: return scaled f
    return f / max(f.max(), 1e-15)


def _lp_phase(a_init, deadline, max_steps=50, verbose=True):
    """
    LP-based refinement: iteratively find best g and move towards it.
    """
    n = len(a_init)
    f = a_init.copy().astype(np.float64)
    f = f / f.sum()
    best_a = f.copy()
    best_score = _score_norm(f)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(f, f)
        M = float(c.max())

        # Solve LP
        g = _lp_solve(f, M, n)
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Blend: try different t values
        best_t = 0
        best_blend_score = best_score
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 0.01:
                continue
            f_new = f_new / s
            sc = _score_norm(f_new)
            if sc < best_blend_score:
                best_blend_score = sc
                best_t = t

        if best_t > 0:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            sc = _score_norm(f)
            if sc < best_score:
                best_score = sc
                best_a = f.copy()
            if verbose:
                print(f"    LP step {step}: t={best_t}, score={sc:.10f}, best={best_score:.10f}")
        else:
            if verbose:
                print(f"    LP step {step}: no improvement, stopping")
            break

    return best_a, best_score


def _grad_polish(a_init, deadline, verbose=True):
    """
    Gradient-based polish using smoothed max with decreasing temperature.
    Works in log-space with Adam.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    # Temperature schedule
    c0 = _fft_conv(a_init, a_init)
    max_c0 = float(c0.max())
    betas = [max_c0 * 1e-3, max_c0 * 3e-4, max_c0 * 1e-4, max_c0 * 3e-5,
             max_c0 * 1e-5, max_c0 * 3e-6, max_c0 * 1e-6]

    lr = 0.005

    for beta in betas:
        if time.monotonic() > deadline:
            break
        phase_deadline = min(deadline, time.monotonic() + 120)
        lr = max(0.001, lr * 0.8)

        while time.monotonic() < phase_deadline:
            t += 1
            a = np.exp(z - np.max(z))
            a = a / a.sum()

            c = _fft_conv(a, a)
            cm = float(c.max())
            c_shift = (c - cm) / beta
            np.clip(c_shift, -500, 500, out=c_shift)
            w = np.exp(c_shift)
            w_sum = w.sum()
            if w_sum < 1e-300 or not np.isfinite(w_sum):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = w / w_sum

            # Gradient of smoothed max w.r.t. a
            # d/d a_j [sum_k w_k * c_k] where c = a*a
            # = w * (a * e_j + e_j * a) for each j
            # = w convolved with reversed a, appropriately sliced
            a_rev = a[::-1]
            wc = _fft_conv(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.1:
                update = update * (0.1 / update_norm)

            z = z - update
            z = np.clip(z, -50, 50)
            z = z - z.mean()

            if t % 200 == 0:
                a_chk = np.exp(z - np.max(z))
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    a_final = np.exp(z - np.max(z))
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    """Interpolate a profile to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Fast coordinate descent (incremental conv)
    2. Gradient polish (Adam log-space)
    3. LP-based refinement
    4. Multi-scale (n=8192) exploration
    5. Final coordinate descent polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score_norm(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score_norm(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # === Phase 1: Fast coordinate descent ===
    print(f"\nPhase 1: Coordinate descent (n={n_main})")
    p1_deadline = min(deadline, t_start + 600.0)
    a_cd, sc_cd = _coord_descent(a0, p1_deadline, max_sweeps=300000, verbose=True)
    sc_cd_full = _score_full(a_cd)
    print(f"  CD done: score={sc_cd_full:.10f}, best={best_score:.10f}")
    if sc_cd_full < best_score:
        best_score = sc_cd_full
        best_seq = a_cd.copy()

    # === Phase 2: Gradient polish ===
    print(f"\nPhase 2: Gradient polish (n={n_main})")
    p2_deadline = min(deadline, t_start + 900.0)
    a_gp, sc_gp = _grad_polish(best_seq, p2_deadline)
    sc_gp_full = _score_full(a_gp)
    print(f"  Grad done: score={sc_gp_full:.10f}, best={best_score:.10f}")
    if sc_gp_full < best_score:
        best_score = sc_gp_full
        best_seq = a_gp.copy()

    # Another CD after gradient
    if time.monotonic() < deadline - 300:
        p2b_deadline = min(deadline - 300, time.monotonic() + 150)
        a_cd2, sc_cd2 = _coord_descent(best_seq, p2b_deadline, max_sweeps=100000, verbose=False)
        sc_cd2_full = _score_full(a_cd2)
        print(f"  CD2 done: score={sc_cd2_full:.10f}, best={best_score:.10f}")
        if sc_cd2_full < best_score:
            best_score = sc_cd2_full
            best_seq = a_cd2.copy()

    # === Phase 3: LP refinement ===
    print(f"\nPhase 3: LP refinement (n={n_main})")
    p3_deadline = min(deadline, t_start + 1200.0)
    if time.monotonic() < p3_deadline - 30:
        a_lp, sc_lp = _lp_phase(best_seq, p3_deadline, max_steps=20, verbose=True)
        sc_lp_full = _score_full(a_lp)
        print(f"  LP done: score={sc_lp_full:.10f}, best={best_score:.10f}")
        if sc_lp_full < best_score:
            best_score = sc_lp_full
            best_seq = a_lp.copy()

        # CD after LP
        if time.monotonic() < deadline - 200:
            p3b_deadline = min(deadline - 200, time.monotonic() + 100)
            a_cd3, sc_cd3 = _coord_descent(best_seq, p3b_deadline, max_sweeps=50000, verbose=False)
            sc_cd3_full = _score_full(a_cd3)
            print(f"  CD3 done: score={sc_cd3_full:.10f}, best={best_score:.10f}")
            if sc_cd3_full < best_score:
                best_score = sc_cd3_full
                best_seq = a_cd3.copy()

    # === Phase 4: n=8192 exploration ===
    print(f"\nPhase 4: n=8192 exploration")
    p4_deadline = min(deadline, t_start + 1550.0)
    if time.monotonic() < p4_deadline - 60:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score_norm(a_big)
        print(f"  Interpolated n={n_big}: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        # CD on big
        if time.monotonic() < p4_deadline - 30:
            p4b_deadline = min(p4_deadline - 10, time.monotonic() + 180)
            a_big_cd, sc_big_cd = _coord_descent(a_big, p4b_deadline, max_sweeps=50000, verbose=True)
            sc_big_cd_full = _score_full(a_big_cd)
            print(f"  CD big: score={sc_big_cd_full:.10f}, best={best_score:.10f}")
            if sc_big_cd_full < best_score:
                best_score = sc_big_cd_full
                best_seq = a_big_cd.copy()

        # Grad on big
        if time.monotonic() < p4_deadline - 20 and best_seq is not None and len(best_seq) == n_big:
            p4c_deadline = min(p4_deadline - 5, time.monotonic() + 120)
            a_big_gp, sc_big_gp = _grad_polish(best_seq, p4c_deadline)
            sc_big_gp_full = _score_full(a_big_gp)
            print(f"  Grad big: score={sc_big_gp_full:.10f}, best={best_score:.10f}")
            if sc_big_gp_full < best_score:
                best_score = sc_big_gp_full
                best_seq = a_big_gp.copy()

    # === Phase 5: Final polish on best ===
    print(f"\nPhase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_final = len(best_seq)
        p5_deadline = deadline - 10
        a_final, sc_final = _coord_descent(best_seq, p5_deadline, max_sweeps=100000, verbose=False)
        sc_final_full = _score_full(a_final)
        print(f"  Final CD: score={sc_final_full:.10f}, best={best_score:.10f}")
        if sc_final_full < best_score:
            best_score = sc_final_full
            best_seq = a_final.copy()

    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        s = float(final.sum())
        if s > 0:
            final = final / s  # normalize to sum=1 for clean score
        sc_final = _score_full(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
