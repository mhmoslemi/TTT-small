    2. Updating p via multiplicative weights

Optimized step function search combining:
- Game-theoretic (minimax) initialization
- Adaptive temperature mirror descent
- Multi-scale exploration
- Active-set gradient polishing

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1, so score = 2n * max(conv)"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses game-theoretic initialization, adaptive mirror descent, 
active-set gradients, and multi-scale exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_active(a, c, k=1):
    """Compute gradient using top-k lags (active set)."""
    n = len(a)
    k = min(k, len(c))
    k_indices = np.argsort(c)[::-1][:k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / k
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_gradient_soft(a, c, T):
    """Compute gradient using soft (temperature) distribution over lags."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _game_init(a_init, n_rounds=200, T_p_start=100.0, T_p_end=10.0, eta_start=0.1, eta_end=0.01):
    """
    Game-theoretic initialization: alternate between updating a (via MW)
    and updating p (dual variable tracking the max lag distribution).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    L = 2 * n - 1
    p = np.ones(L) / L
    
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for t in range(n_rounds):
        frac = t / max(1, n_rounds - 1)
        T_p = T_p_start * (T_p_end / T_p_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac
        
        c = _fft_convolve(a, a)
        sc = c.max()
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Update p (dual)
        c_shifted = (c - c.max()) / T_p
        p = np.exp(c_shifted)
        p = p / p.sum()
        
        # Compute gradient g = K_p * a
        p_rev = p[::-1]
        g = _fft_convolve(a, p_rev)
        g = g[n-1:2*n-1][::-1]  # extract the relevant part
        
        # Update a via multiplicative weights
        a = a * np.exp(-eta * g / g.max() if g.max() > 0 else 0)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
    
    c_final = _fft_convolve(a, a)
    if c_final.max() < best_sc:
        best_sc = c_final.max()
        best_a = a.copy()
    
    return best_a, best_sc


def _mirror_descent(a_init, deadline, T_start, T_end, lr_start, lr_end,

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _spectral_perturb(a, freq, amplitude):
    """Perturb a sequence by adding a cosine wave in the frequency domain."""
    n = len(a)
    # Add a cosine at the specified frequency
    phase = np.random.default_rng(0).uniform(0, 2*np.pi)
    perturbation = amplitude * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a_new = a + perturbation
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()
