def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    # Find theta such that sum(max(v_i - theta, 0)) = target_sum
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cv - target_sum))[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_topk(a, c, n, K):
    """Gradient of sum of top K elements of c with respect to a."""
    c_sorted_idx = np.argsort(c)[::-1]
    if K > len(c):
        K = len(c)
    w = np.zeros_like(c)
    w[c_sorted_idx[:K]] = 1.0 / K

    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(fw * fa.conj(), N)
    return 2.0 * corr[:n]

def _active_set_projected_gradient(a_init, deadline, max_iter=2000, tol_frac=0.005):
    """Projected gradient on the active set subproblem."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = 1.0 / (2.0 * n)  # initial learning rate (1/L where L is Lipschitz constant)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Compute gradient
        N = _next_pow2(3 * n)
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w, N)
        corr = np.fft.irfft(fw * fa.conj(), N)
        grad = 2.0 * corr[:n]

        # Project gradient onto tangent space of simplex
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-12:
            break

        # Backtracking line search
        alpha = min(lr, 1.0 / (0.5 * gnorm + 1e-10))
        f_old = float(np.dot(w, c))
        a_new = a.copy()
        for ls in range(30):
            a_try = a - alpha * grad_t
            a_try = _project_simplex(a_try, 1.0)
            c_try = _full_convolve(a_try, a_try)
            f_try = float(np.dot(w, c_try))
            if f_try < f_old - 1e-14 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            break

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            lr = min(lr * 1.01, 1.0)
        else:
            lr *= 0.95

    return best_a, best_sc

def _shrinkage(a, deadline):
    """Try zeroing out small elements."""
    n = len(a)
    best_a = a.copy()

        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.6 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        return np.abs(np.random.default_rng(0).standard_normal(n)) + 0.1
    else:
        return np.ones(n) / n


def _spectral_init(n, rng, n_iter=20):
    """Initialize using spectral method: try to make autofunction flat."""
    a = np.abs(rng.standard_normal(n)) + 0.01
    a = a / a.sum()
    for _ in range(n_iter):
        c = _full_convolve(a, a)
        c_max = c.max()
        # We want c to be flat: push down the peaks, push up the valleys
        target = c.mean()
        # Gradient step: reduce where c > target, increase where c < target
        # This is in the "c-space", but we need to map back to a-space
        # Use the active set gradient for peaks
        w = (c > target).astype(float)
        if w.sum() == 0:
            w[int(np.argmax(c))] = 1.0
        w = w / w.sum()
        N = _next_pow2(3 * n)
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w, N)
        corr = np.fft.irfft(fw * fa.conj(), N)
        grad = 2.0 * corr[:n] - 2.0 * corr[:n].mean()
        a_new = a - 0.05 * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        if _score(a_new) < _score(a):
            a = a_new
    return a


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=3):
    """Swap random blocks."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = rng.integers(0, max(1, n - 2 * block_size))
        j = rng.integers(0, max(1, n - 2 * block_size))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i+block_size].copy()
        z[i:i+block_size] = z[j:j+block_size]
        z[j:j+block_size] = tmp
    return _softmax(z)


def _truncate_and_reopt(a, n_new, deadline):
    """Truncate to n_new and re-optimize briefly."""
    n_old = len(a)
    if n_new >= n_old:
        return a
    # Take the largest n_new elements and reorder
    idx = np.argsort(a)[::-1][:n_new]
    idx = np.sort(idx)
    a_trunc = a[idx].copy()
    a_trunc = a_trunc / a_trunc.sum()
    # Quick L-BFGS polish
    sc = _score(a_trunc)
    if sc < 1.6 and n_new >= 128:
        a_trunc = _run_lbfgs_adaptive(a_trunc, deadline, min_time_remaining=10.0)
    return a_trunc


def _coordinatedescent_polish(a, deadline, n_sweeps=10):
    """Try single-coordinate updates to reduce the max convolution."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Try random subset of coordinates
        n_try = min(n, 500)
        indices = np.random.default_rng(sweep).choice(n, n_try, replace=False)
        for j in indices:
            if time.monotonic() > deadline:
                break
            a = best_a.copy()
            orig = a[j]
            if orig < 1e-15:
                continue
            # Try reducing a[j] and redistributing
            for frac in [0.5, 0.7, 0.9]:
                a_try = a.copy()
                delta = a_try[j] * (1.0 - frac)
                a_try[j] *= frac
                # Redistribute to neighbors
                left = max(0, j - 1)
                right = min(n - 1, j + 1)
                a_try[left] += delta * 0.5
                a_try[right] += delta * 0.5
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-12:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    improved = True
                    break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining adaptive L-BFGS, active-set projected gradient,
    top-K smoothing, spectral initialization, multi-n search, and truncation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: Main L-BFGS with adaptive beta from parent ---
    print(f"\n[Phase 1] Adaptive L-BFGS from parent (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs_adaptive(parent, deadline, min_time_remaining=500.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # --- Phase 2: Top-K smoothing refinement ---
    print(f"\n[Phase 2] Top-K smoothing (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_topk = _run_lbfgs_topk(best_seq, deadline, K_values=(3, 5, 10), min_time_remaining=300.0)
        update_best(a_topk, _score(a_topk), "topk")

    # --- Phase 3: Active set projected gradient ---
    print(f"\n[Phase 3] Active set PG (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_as, sc_as = _active_set_projected_gradient(best_seq, deadline - 200, max_iter=3000, tol_frac=0.004)
        update_best(a_as, sc_as, "active_pg")

    # --- Phase 4: Subgradient polish ---
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_sg, sc_sg = _subgradient_polish(best_seq, deadline - 150, n_max_steps=5000, lr_init=0.025)
        update_best(a_sg, sc_sg, "sg_polish")

    # --- Phase 5: Shrinkage ---
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 120)
        update_best(a_shr, sc_shr, "shrinkage")
        if time.monotonic() < deadline - 100:
            a_shr2 = _run_lbfgs_adaptive(a_shr, deadline, min_time_remaining=80.0)
            update_best(a_shr2, _score(a_shr2), "shrink_reopt")

    # --- Phase 6: Multi-n search ---
    print(f"\n[Phase 6] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [1024, 1536, 2048, 3072, 4096, 6144]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        # Initialize from parent (resampled) or profile
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "tapered")
        elif n_parent == n_try:
            a_init = parent.copy()
        else:
            a_init = _make_profile(n_try, "cos2")

        a_opt = _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0)
        update_best(a_opt, _score(a_opt), f"n_{n_try}")

        if time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_projected_gradient(a_opt, deadline - 30, max_iter=1000, tol_frac=0.005)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_as")

    # --- Phase 7: Diverse restarts ---
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        perturbations = [
            ("gauss_05", lambda: _perturb(best_seq, 0.05, rng)),
            ("gauss_03", lambda: _perturb(best_seq, 0.03, rng)),
            ("fourier", lambda: _perturb_fourier(best_seq, rng, n_modes=5)),
            ("block", lambda: _perturb_block(best_seq, rng, n_blocks=4)),
        ]
        for label, pert_fn in perturbations:
            if time.monotonic() > deadline - 30:
                break
            p_start = pert_fn()
            a_r = _run_lbfgs_adaptive(p_start, deadline, min_time_remaining=25.0)
            update_best(a_r, _score(a_r), f"restart_{label}")

    # --- Phase 8: Spectral initialization and search ---
    print(f"\n[Phase 8] Spectral init (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        n_spec = len(best_seq) if best_seq is not None else 4096
        a_spec = _spectral_init(n_spec, rng, n_iter=30)
        update_best(a_spec, _score(a_spec), "spectral_init")
        if _score(a_spec) < best_score + 0.05:
            a_spec_opt = _run_lbfgs_adaptive(a_spec, deadline, min_time_remaining=20.0)
            update_best(a_spec_opt, _score(a_spec_opt), "spectral_opt")

    # --- Phase 9: Truncation exploration ---
    print(f"\n[Phase 9] Truncation (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        trunc_sizes = [n_best // 2, n_best * 3 // 4, int(n_best * 0.75)]
        for n_trunc in sorted(set(trunc_sizes)):
            if n_trunc < 128 or time.monotonic() > deadline - 10:
                continue
            a_trunc = _truncate_and_reopt(best_seq, n_trunc, deadline)
            update_best(a_trunc, _score(a_trunc), f"trunc_{n_trunc}")

    # --- Phase 10: Final fine polish ---
    print(f"\n[Phase 10] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very fine L-BFGS with tiny beta
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0003
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(3000, max(100, int(remaining / 0.03)))
            try:
                result = minimize(
                    lambda z_local: _smoobj_ls(z_local, beta_final, n),
                    z,
                    jac=lambda z_local: _smograd_ls(z_local, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final active set
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _active_set_projected_gradient(best_seq, deadline - 2, max_iter=500, tol_frac=0.002)
            update_best(a_fin2, sc_fin2, "final_as")

    # --- Phase 11: Coordinate descent polish ---
    print(f"\n[Phase 11] Coord descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_cd, sc_cd = _coordinatedescent_polish(best_seq, deadline - 1, n_sweeps=5)
        update_best(a_cd, sc_cd, "coord_descent")

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
