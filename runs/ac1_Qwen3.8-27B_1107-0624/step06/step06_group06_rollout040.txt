```python
    """Compute H @ v where H_{j,m} = 2 * 1_{j+m in S}"""
    N = 2 * n  # or next power of 2 >= 2n-1
```

```python
def solve_active_set(S, n, max_cg_iter=1000):
    """Solve H a = 1, then normalize. H_{j,m} = 2 * 1_{j+m in S}"""
    # Use CG to solve H x = 1
    x = np.ones(n) / n  # initial guess
    r = np.ones(n) - hessian_mv(x, S, n)
    p = r.copy()
    rs = np.dot(r, r)
    
    for _ in range(max_cg_iter):
        Hp = hessian_mv(p, S, n)
        pHp = np.dot(p, Hp)
        if pHp < 1e-15:
            break
        alpha = rs / pHp
        x += alpha * p
        r_new = r - alpha * Hp
        rs_new = np.dot(r_new, r_new)
        if np.sqrt(rs_new) < 1e-10:
            r = r_new
            break
        p = r_new + (rs_new / rs) * p
        r = r_new
        rs = rs_new
    
    # Normalize: a = x / (1^T x)
    s = x.sum()
    if s < 1e-15:
        return None
    a = x / s
    return a
```

```python
"""
Active-set CG optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact active set identification with FFT-based Hessian-vector products,
support identification, and iterative refinement.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hessian_mv(v, S_ind, n, N):
    """Compute H @ v where H_{j,m} = 2 * S_ind[j+m].
    S_ind is a boolean array of length 2n-1 (or a float array).
    Returns array of length n.
    """
    # (Hv)_j = 2 * sum_m v[m] * S_ind[j+m]
    # This is 2 * conv(v, S_flip)[2n-2-j]
    S_flip = S_ind[::-1]  # length 2n-1
    V = np.fft.rfft(v, N)
    Sf = np.fft.rfft(S_flip, N)
    conv = np.fft.irfft(V * Sf, N)
    # result[j] = 2 * conv[2n-2-j] for j = 0, ..., n-1
    result = 2.0 * conv[2*n-2 : n-1 : -1]
    return result

def _solve_hessian_one(S_ind, n, max_iter=5000, tol=1e-12):
    """Solve H x = 1 using CG, where H_{j,m} = 2 * S_ind[j+m].
    Returns x (unnormalized).
    """
    N = _next_pow2(2 * n - 1)
    x = np.ones(n) / n
    r = np.ones(n) - _hessian_mv(x, S_ind, n, N)
    p = r.copy()
    rs = np.dot(r, r)
    
    for it in range(max_iter):
        if rs < tol * tol:
            break
        Hp = _hessian_mv(p, S_ind, n, N)
        pHp = np.dot(p, Hp)
        if pHp < 1e-30:
            break
        alpha = rs / pHp
        x = x + alpha * p
        r = r - alpha * Hp
        rs_new = np.dot(r, r)
        if rs_new >= rs:
            # Breakdown, restart
            p = r.copy()
        else:
            p = r + (rs_new / rs) * p
        rs = rs_new
    
    return x

    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    # Determine initial active set
    b = _full_convolve(a, a)
    b_max = b.max()
    S = b >= (b_max * (1 - S_tol))

    support = np.ones(n, dtype=bool)
```

```python
"""
Active-set CG optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact active set identification with FFT-based Hessian-vector products,
iterative support identification, tolerance continuation, and gradient polish.
Key improvement: solves the KKT system directly rather than using approximate
gradient methods, leading to faster convergence to the true optimum.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _hvp(v, S_fft, N, n):
    """Hessian-vector product: (Hv)_j = 2 * sum_m v[m] * S_ind[j+m].
    S_fft is precomputed rfft of S_ind reversed, padded to N.
    v has length n. Returns array of length n.
    """
    V = np.fft.rfft(v, N)
    conv = np.fft.irfft(V * S_fft, N)
    return 2.0 * conv[2 * n - 2:n - 1: -1]


def _build_S_fft(S, n, N):
    """Build the FFT of reversed S for HVP computation."""
    S_flip = S[::-1].astype(np.float64)
    return np.fft.rfft(S_flip, N)


def _cg_solve_subspace(S_fft, support, n, N, x0=None, max_iter=5000, tol=1e-13):
    """Solve H[T,T] x = 1[T] via CG where x is supported on T.
    S_fft is precomputed. Returns x (length n, zero outside support).
    """

    if nsup < 1:
        return np.zeros(n)

        x = x0.copy()
        x[~support] = 0.0

    r = np.zeros(n)
    r[support] = 1.0 - Hx[support]
    p = r.copy()
    rs = float(np.dot(r[support], r[support]))

    if rs < tol * tol:
        return x

        pHp = float(np.dot(p[support], Hp[support]))

def _gradient_polish(a_init, deadline, n_steps=500, lr=0.01):
    """Projected gradient descent polish on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = np.exp(z - np.max(z))
        a = a / a.sum()
        b = _full_convolve(a, a)
        b_max = float(np.max(b))
        tol = 0.002 * b_max
        active = b >= (b_max - tol)
        n_active = active.sum()
        w = np.zeros(len(b))
        w[active] = 1.0 / n_active

        # Gradient of sum_{k in active} b_k w_k
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 4.0 * n * a * wc[n - 1:2 * n - 1]

        # Normalize gradient (on simplex)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Sherman-Morrison step
        step = lr * a / (1.0 + 0.01 * i)
        z = z - step * (grad / a + np.mean(grad / np.maximum(a, 1e-15)) - np.mean(grad / np.maximum(a, 1e-15)))
        # Simpler: just do z -= step * grad / a (log-space gradient)

        a_try = np.maximum(a_try, 0)

            a = best_a.copy()
            z = np.log(np.maximum(a, 1e-30))

def _shrinkage_optimize(a, deadline):

    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    sorted_idx = np.argsort(a)

    for frac in [0.01, 0.02, 0.05, 0.10]:
        if time.monotonic() > deadline - 30:

        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-8:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc

        if time.monotonic() > deadline - 80:
            break
        if n_try == n_main:
            continue  # Already tried

        # Resample parent to new length
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n_try)
        a_init = np.interp(x_new, x_old, parent)
        a_init = np.maximum(a_init, 0)
        s_init = a_init.sum()
        if s_init < 1e-10:
            continue
        a_init = a_init / s_init

        rem = deadline - 80 - time.monotonic()
        if rem < 30:
            break
        a_n = _active_set_optimize(a_init, t_start + time.monotonic() - t_start + rem,
                                   tol_continuation=[0.02, 0.01, 0.005, 0.002])
        _update(a_n, f"n_{n_try}")

    # --- Phase 6: Final tight active set + gradient ---
    print(f"\n[Phase 6] Final refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active set
        rem = deadline - 5 - time.monotonic()
        if rem > 20:
            a_final = _active_set_optimize(best_seq, deadline - 5,
                                           tol_continuation=[0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
            _update(a_final, "final_active")

        # Final gradient steps
        if time.monotonic() < deadline - 3:
            a_fin, sc_fin = _gradient_polish(best_seq, deadline - 3, n_steps=200, lr=0.002)
            _update(a_fin, "final_grad")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
