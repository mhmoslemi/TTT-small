import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _compute_grad_mp(a, w):
    """
    Compute gradient of smooth_max(conv(a,a)) w.r.t. a.
    w: softmax of conv(a,a)/beta, length 2n-1
    Returns gradient of shape (n,)
    """
    n = len(a)
    w_rev = w[::-1]
    # conv(a, w_rev) has length n + (2n-1) - 1 = 3n-2

def _multiplicative_descent(a_init, beta, n_iter, deadline, eta_init=0.1):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score_norm(a)
    eta = eta_init
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / w_sum
        
        grad = _compute_grad_mp(a, w)
        
        # Multiplicative update
        # Scale gradient for stability
        grad_scale = np.max(np.abs(grad))
        if grad_scale > 0:
            grad_normalized = grad / grad_scale
        
        log_update = -eta * grad_normalized
        log_update -= np.max(log_update)  # numerical stability
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-300:
            break
        a_new = a_new / s_new
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Adaptive step size
        if sc > _score_norm(a) * 1.01:
            eta *= 0.5
        elif it % 100 == 0 and sc < _score_norm(a):
            eta *= 1.1  # increase step if making progress
        
        a = a_new
    
    return best_a, best_score

Multi-phase optimizer using multiplicative descent (mirror descent), 
FISTA, targeted peak reduction, and Adam fine-tuning.
Focuses on the 4096-length parent sequence with beta-continuation.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _softmax(c, beta):
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    return e / s

            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        w = _softmax(c_y, beta)
        grad = _grad_smooth_max(y, w)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _multiplicative_descent(a_init, beta, deadline, eta_init=0.5, n_iter=10000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score_norm(a)
    eta = eta_init
    stagnation = 0
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        w = _softmax(c, beta)
        grad = _grad_smooth_max(a, w)
        
        # Normalize gradient for stable multiplicative updates
        g_max = float(np.max(np.abs(grad)))
        if g_max < 1e-15:
            break
        grad_norm = grad / g_max
        
        # Multiplicative update: a_new = a * exp(-eta * grad_norm)
        log_factor = -eta * grad_norm
        log_factor -= np.max(log_factor)  # stability
        a_new = a * np.exp(log_factor)
        s_new = a_new.sum()
        if s_new < 1e-300:
            break
        a_new = a_new / s_new
        
        sc_new = _score_norm(a_new)
        sc_curr = _score_norm(a)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            stagnation = 0
            # Increase step size if making good progress
            if sc_new < sc_curr * 0.99:
                eta = min(eta * 1.05, 5.0)
        else:
            stagnation += 1
            if stagnation > 50:
                eta *= 0.9
            if sc_new > sc_curr:
                # Reject the update
                a_new = a
                sc_new = sc_curr
        
        a = a_new
    
    return best_a, best_score

def _targeted_peak_reduction(a_init, deadline, n_top=50, n_rounds=100):
    """Target the top convolution coefficients for reduction."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        # Find top coefficients
        top_idx = np.argsort(c)[::-1][:n_top]
        c_max = float(c.max())
        
        # Weight for the top coefficients
        w = np.zeros_like(c)
        for idx in top_idx:
            w[idx] = (c[idx] / c_max) ** 2
        w = w / w.sum()
        
        # Gradient using these weights
        grad = _grad_smooth_max(a, w)
        
        # Small step in simplex
        step_size = 0.1 / (1.0 + 0.01 * rnd)
        a_new = _proj_simplex(a - step_size * grad)
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.5
            a_new = _proj_simplex(a - step_size * grad)
            if _score_norm(a_new) < _score_norm(a):
                a = a_new
    
    return best_a, best_score

def _adam_logspace(a_init, deadline, betas, lr_init=0.01, max_iter=200000):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n
    
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()
    
    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1
            
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            w = _softmax(c, beta)
            
            # Gradient in log-space
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1**t)
            vh = v_buf / (1 - b2**t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            z = z - update
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            
            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if t % 10000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _spectral_unfold(a, n_iter=200, deadline=None):
    """Try to flatten the convolution by adjusting the FFT phase."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    N = _next_pow2(2 * n - 1)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        B = A * A  # This is FFT of the convolution
        c = np.fft.irfft(B, N)[:2*n-1]
        
        # Try to make c flatter by adjusting B
        c_max = float(c.max())
        c_min = float(np.min(np.abs(c)))
        
        # Flatten B: reduce peaks
        mag_B = np.abs(B)
        avg_mag = mag_B.mean()
        # Soft-clipping of B magnitude
        B_flattened = B / (1.0 + 0.5 * (mag_B / avg_mag) ** 2)
        # Renormalize to keep B[0] = A[0]^2 = (sum(a))^2 = 1
        B_flattened = B_flattened * (1.0 / B_flattened[0])
        
        # Recover A from B (take square root)
        # B = A^2, so A = sqrt(B) (element-wise, complex square root)
        A_new = np.sqrt(B_flattened + 1j * 0)
        # Ensure A[0] is real and positive
        A_new[0] = np.abs(A_new[0])
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_score

def _perturb_logspace(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)
