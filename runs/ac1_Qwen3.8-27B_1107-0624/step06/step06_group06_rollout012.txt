Multi-strategy optimizer v2: L-BFGS + adaptive beta + coordinate descent + IPR + multi-n.
Improvements over v1: tighter beta schedule, coordinate descent polish,

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "blackman":
        t = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _coord_descent(z, a, c, n, deadline, max_steps=50000, rng=None):

            break
        if step % 100 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
                if patience > 500:
                    break
        
        # Compute subgradient at the max lag
        c_max = float(c.max())
        k_max = int(np.argmax(c))
        # Gradient of c_{k_max} w.r.t. a: 2 * a reversed and shifted
        # (grad c_k)_i = 2 * a_{k-i} (if 0 <= k-i <= n-1)
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Convert to z-space gradient (through softmax)
        # If a = softmax(z), then da_i/dz_j = a_i * (delta_{ij} - a_j)
        # So d(obj)/dz_j = sum_i (d obj/d a_i) * a_i * (delta_{ij} - a_j)
        # = a_j * (d obj/d a_j - sum_i (d obj/d a_i) * a_i)

        z_grad = a * (grad - W)

            i = int(np.argmax(np.abs(z_grad)))
        else:
            i = rng.integers(n)
        
        # Line search in coordinate i
        # Try a few step sizes
        best_z = z.copy()
        best_obj = float(c_max)  # we're minimizing max(c) approximately
        # Actually, let's use the smoothed objective for the line search
        # Or just try reducing the max directly
        
        # Try step in -z_grad[i] direction
        for lr in [0.1, 0.05, 0.02, 0.01, 0.005]:

    sc_final = _score(a_final)

def _shrinkage_v2(a, deadline):
    """Improved shrinkage: try multiple fractions and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
