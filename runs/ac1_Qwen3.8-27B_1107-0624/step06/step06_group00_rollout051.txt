"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over previous: (1) coordinate descent polish, (2) adaptive active-set,
(3) structured block perturbations, (4) n=2048 exploration, (5) smarter restarts.

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_adaptive_gradient(a, c, T_base):
    """Adaptive gradient: uses soft gradient with temperature based on local flatness."""
    n = len(a)
    c_max = c.max()
    c_sorted = np.sort(c)[::-1]
    n_total = len(c)
    # Determine effective temperature based on how "peaky" the convolution is
    ratio = c_sorted[min(10, n_total - 1)] / max(c_max, 1e-30)
    T = T_base * (1.0 - ratio) + T_base * 0.1
    T = max(T, T_base * 0.01)
    return _compute_soft_gradient(a, c, T)


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=20):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t == 0.0:
            for t_val in [0.01, 0.02, 0.05]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _adam_z_space(a, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, use_active=False, active_k=5,
                  eval_interval=200, clip_norm=0.8, adaptive=False):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)
        if adaptive:
            g = _compute_adaptive_gradient(a_cur, c, T)
        elif use_active and T < T_start * 0.01:
            k_eff = max(2, active_k - int(2 * frac))
            g = _compute_active_gradient(a_cur, c, k_eff)
        else:
            g = _compute_soft_gradient(a_cur, c, T)
        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        if t % eval_interval == 0:
            a_check = _softmax_stable(z)
            sc = _score_norm(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent(a, deadline, grid_size=12, max_passes=100,
                        random_mode=False, rng=None):
    """Coordinate descent to polish the solution."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    conv = _fft_convolve(best_a, best_a)
    s = float(best_a.sum())
    improved_any = False

    for phase in range(max_passes):
        if time.monotonic() > deadline:
            break
        if random_mode:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        improved_this_pass = 0
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = best_a[i]
            if old_val < 1e-15:
                continue
            # Compute affected range
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)
            # Try a grid of multipliers
            multipliers = np.concatenate([
                np.array([0.0, 0.25, 0.5, 0.75]),
                np.linspace(0.8, 1.3, grid_size),
                np.array([1.5, 2.0, 3.0])
            ])
            x_vals = old_val * multipliers
            x_vals = np.unique(x_vals)
            x_vals = x_vals[x_vals >= 0]

            best_x = old_val
            best_sc = best_score

            for x in x_vals:
                dx = x - old_val
                if abs(dx) < 1e-15:
                    continue
                # Compute new max efficiently
                # For k in [k_min, k_max], k != 2i:
                #   new_conv[k] = conv[k] + 2*dx*a[k-i]
                # For k = 2i (if in range): new_conv[2i] = conv[2i] + dx*(2*old_val + dx)
                k_range = np.arange(k_min, k_max + 1)
                ks = k_range
                offsets = ks - i  # range [0, min(n-1, k_max-i)]
                # Linear update
                new_vals = conv[ks].copy()
                new_vals += 2.0 * dx * best_a[offsets]
                # Quadratic correction at k = 2i
                k_2i = 2 * i
                if k_2i >= k_min and k_2i <= k_max:
                    idx_2i = k_2i - k_min
                    new_vals[idx_2i] = conv[k_2i] + dx * (2.0 * old_val + dx)
                # Unaffected positions
                max_affected = float(np.max(new_vals))
                max_unaffected = float(np.max(np.delete(conv, ks))) if len(ks) < len(conv) else 0.0
                new_max = max(max_affected, max_unaffected)
                new_sum = s - old_val + x
                sc_candidate = 2.0 * n * new_max / (new_sum * new_sum)
                if sc_candidate < best_sc:
                    best_sc = sc_candidate
                    best_x = x

            if best_x != old_val:
                best_a[i] = best_x
                s = s - old_val + best_x
                dx = best_x - old_val
                # Update convolution
                k_range = range(k_min, k_max + 1)
                offsets = np.array(list(range(k_min, k_max + 1))) - i
                conv[list(range(k_min, k_max + 1))] += 2.0 * dx * best_a[offsets]
                k_2i = 2 * i
                if k_2i >= k_min and k_2i <= k_max:
                    conv[k_2i] -= dx * (2.0 * old_val + dx)
                    conv[k_2i] += dx * (2.0 * best_x + dx)
                best_score = best_sc
                improved_this_pass += 1
                improved_any = True

        if improved_this_pass == 0:
            break

    _ = _fft_convolve(best_a, best_a)
    return best_a, _score_norm(best_a)


def _block_perturbation(a, rng, n_blocks=5, block_sizes=None):
    """Apply structured block perturbations."""
    n = len(a)
    if block_sizes is None:
        block_sizes = [64, 128, 256, 512]
    result = a.copy()
    for _ in range(n_blocks):
        bs = int(rng.choice(block_sizes))
        start = int(rng.integers(0, max(1, n - bs)))
        scale = rng.uniform(0.5, 2.0)
        result[start:start + bs] *= scale
        result = np.maximum(result, 0.0)
        if result.sum() > 0:
            result = result / result.sum()
    return result


def _fista_simplex(a_init, deadline, T_start, T_end, max_iter=100000):
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    L_init = 8.0 * n
    step = 1.0 / L_init
    eval_interval = 500

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        c = _fft_convolve(y, y)
        g = _compute_soft_gradient(y, c, T)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-10:
            step = step * min(2.0, max(0.5, 1.0 / (step * g_norm + 1e-10)))
        step = min(step, 2.0 / L_init)
        step = max(step, 1e-10)
        a_new = y - step * g
        a_new = _project_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        a_cd, sc_cd = _coordinate_descent(best_a, phase4_end - 3, grid_size=10, max_passes=50)
        full_sc = _score(a_cd)
        print(f"Phase 4 CoordDesc: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_cd.copy()

    # Phase 5: Random coordinate descent (850s - 1000s)
    phase5_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase5_end:
        a_rcd, sc_rcd = _coordinate_descent(best_a, phase5_end - 3, grid_size=8,
                                            max_passes=200, random_mode=True, rng=rng)
        full_sc = _score(a_rcd)
        print(f"Phase 5 RandCoordDesc: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_rcd.copy()

    # Phase 6: Diverse restarts with polish (1000s - 1400s)
    phase6_end = min(deadline, t_start + 1400.0)
    n_best = len(best_a)
    if time.monotonic() < phase6_end:
        restart_strategies = [
            # (type, params)
            ("gauss", 0.01),
            ("gauss", 0.02),
            ("gauss", 0.04),
            ("block", None),
            ("block", None),
            ("gauss", 0.005),
        ]
        for attempt, (rtype, rparam) in enumerate(restart_strategies):
            if time.monotonic() > phase6_end - 50:
                break
            if rtype == "gauss":
                sigma = rparam
                z = np.log(np.maximum(best_a, 1e-15))
                z = z + sigma * rng.standard_normal(n_best)
                z = z - z.mean()
                a_pert = _softmax_stable(z)
            else:
                a_pert = _block_perturbation(best_a, rng, n_blocks=3,
                                             block_sizes=[64, 128, 256, 512])
            # Quick Adam refinement
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            time_per_restart = max(30, (phase6_end - 50 - time.monotonic()) / max(1, len(restart_strategies) - attempt))
            a_pert_opt, sc_pert = _adam_z_space(a_pert, time.monotonic() + time_per_restart,
                                                 T_start=0.001 * max_cp, T_end=0.00003 * max_cp,
                                                 lr_start=0.02, lr_end=0.001,
                                                 max_iter=50000,
                                                 use_active=True, active_k=4,
                                                 eval_interval=100)
            # Coordinate descent polish
            cd_deadline = min(time.monotonic() + 15, phase6_end - 50)
            a_cd2, sc_cd2 = _coordinate_descent(a_pert_opt, cd_deadline, grid_size=6, max_passes=20)
            full_sc = _score(a_cd2)
            print(f"  Restart[{attempt}] ({rtype}, p={rparam}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_cd2.copy()

    # Phase 7: Try n=2048 if current n != 2048 (1400s - 1550s)
    phase7_end = min(deadline, t_start + 1550.0)
    if n != 2048 and time.monotonic() < phase7_end:
        n_small = 2048
        a_small = _interp_to_length(best_a, n_small)
        c_sm = _fft_convolve(a_small, a_small)
        max_csm = float(c_sm.max())
        a_small_opt, sc_sm = _adam_z_space(a_small, phase7_end - 5,
                                            T_start=0.002 * max_csm, T_end=0.00002 * max_csm,
                                            lr_start=0.03, lr_end=0.002,
                                            max_iter=100000, use_active=True, active_k=5,
                                            eval_interval=100)
        # CD polish on small
        cd_end = min(time.monotonic() + 20, phase7_end - 5)
        a_small_cd, _ = _coordinate_descent(a_small_opt, cd_end, grid_size=8, max_passes=30)
        full_sc_sm = _score(a_small_cd)
        print(f"Phase 7 n=2048: score={full_sc_sm:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_sm) and full_sc_sm < best_score:
            best_score = full_sc_sm
            best_a = a_small_cd.copy()

    # Phase 8: Final ultra-fine polish (1550s - deadline)
    if time.monotonic() < deadline - 10:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_fin, sc_fin = _adam_z_space(best_a, deadline - 30,
                                       T_start=0.0002 * max_cf, T_end=0.000005 * max_cf,
                                       lr_start=0.005, lr_end=0.0003,
                                       max_iter=200000, use_active=True, active_k=3,
                                       eval_interval=100)
        full_sc = _score(a_fin)
        print(f"Phase 8 FineAdam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_fin.copy()

        # Final CD
        a_fin2, _ = _coordinate_descent(best_a, deadline - 5, grid_size=10, max_passes=30)
        full_sc2 = _score(a_fin2)
        print(f"Phase 8 FinalCD: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc2) and full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_fin2.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
