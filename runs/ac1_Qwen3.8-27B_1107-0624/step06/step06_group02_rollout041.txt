    """Approximate max(c) by smoothing over the top-k values"""
    # Find top-k indices
    topk_idx = np.argsort(c)[-k:]

    w_full = np.zeros_like(c)
    w_full[topk_idx] = w
    return smoothed_val, w_full

1. Finding where the quadratic equals each linear function

"""
Multi-strategy optimizer for minimum peak autocorrelation sequences.
Combines Adam log-space optimization, coordinate descent with line search,
and multi-length exploration (n=4096, n=8192) to minimize the score.
"""
import numpy as np
import time

# Helper functions (top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_log(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _adam_logspace(a_init, deadline, beta_schedule, lr_init=0.01, max_iter_total=1000000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax_log(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):

                w[int(np.argmax(c))] = 1.0

            grad_log = a * (grad - a_dot_grad)
            
            if not np.all(np.isfinite(grad_log)):
                break
            
            m_buf = b1 * m_buf + (1 - b1) * grad_log
            v_buf = b2 * v_buf + (1 - b2) * (grad_log * grad_log)
            mh = m_buf / (1 - b1**t)
            vh = v_buf / (1 - b2**t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)
            
            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            
            if t % 200 == 0:
                a_chk = _softmax_log(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    if np.isfinite(sc_final) and sc_final < best_score:

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:

    return a_new / a_new.sum()

"""
Multi-phase optimizer targeting sub-1.5027 score for non-negative step functions.
Combines Adam log-space optimization with subgradient polish and coordinate descent,
focusing on n=8192 for more optimization freedom over the parent n=4096.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_log(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(a_init, deadline, beta_schedule, lr_init=0.01, max_iter_total=1000000):

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_log(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta

            s = float(e.sum())

                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            grad = 2.0 * wc[n - 1:2 * n - 1]

            v_buf = b2 * v_buf + (1 - b2) * (grad_log * grad_log)

            vh = v_buf / (1 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.25:

            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

                sc = _score_norm(a_chk)

                current_lr = max(0.0005, current_lr * 0.92)

    a_final = _softmax_log(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _subgradient_polish(a_init, deadline, max_steps=5000):
    """Direct subgradient steps targeting the peak of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    step_size = 0.01

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _convolve(a, a)
        cm = float(c.max())

        # Find top-3 peaks for averaging subgradient
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[-top_k:]
        # Weight by softmax over top peaks
        top_vals = c[top_idx]
        z = (top_vals - top_vals.max()) / (0.01 * cm + 1e-30)
        w_top = np.exp(z)
        w_top = w_top / w_top.sum()

        # Compute combined subgradient
        grad = np.zeros(n)
        for idx, wi in zip(top_idx, w_top):
            k = int(idx)
            # grad_i = a_{k-i} + a_{i-k} (with bounds)
            # a_{k-i} is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # a_{k-i} for i in [lo, hi] is a[k-hi : k-lo+1][::-1]
                grad[lo:hi + 1] += wi * a[k - hi:k - lo + 1][::-1]
            # a_{i-k} is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] += wi * a[lo2 - k:hi2 - k + 1]

        # Projected subgradient step
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s

        sc = _score_norm(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Reduce step size
            step_size *= 0.95
            if step_size < 1e-8:
                break

    return best_a, best_sc


def _coord_descent(a_init, deadline, max_rounds=300):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(2048, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_local = best_sc
            best_local_val = old_val
            # Try a range of multipliers
            for factor in [0.7, 0.85, 0.93, 0.97, 1.03, 1.07, 1.15, 1.3, 1.5]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                a_norm = a / s
                sc = _score_norm(a_norm)
                if sc < best_local:
                    best_local = sc
                    best_local_val = new_val
            a[j] = best_local_val
            s = a.sum()
            if s > 1e-30:
                a = a / s
            if best_local < best_sc - 1e-15:
                best_sc = best_local
                best_a = a.copy()
                improved = True
        if not improved and r > 20:
            break
    return best_a, best_sc


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a * np.sqrt(n0))
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_log(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        best_seq = a0.copy()
        best_score = _score(a0)
        print(f"No parent, uniform n={n_main}, score={best_score:.10f}")

    # === Phase 1: Adam at parent length (n=4096) ===
    print("Phase 1: Adam optimization at parent length")
    phase1_end = min(deadline, t_start + 250.0)
    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())
    betas1 = []
    beta = 0.005 * max_c0
    while beta > 1e-11 * max_c0:
        betas1.append(beta)
        beta *= 0.35
    if len(betas1) > 12:
        betas1 = betas1[::len(betas1) // 12] + [1e-11 * max_c0]

    a_opt1, sc_opt1 = _adam_logspace(a0, phase1_end, betas1, lr_init=0.012, max_iter_total=600000)
    sc_opt1_full = _score(a_opt1)
    print(f"  Adam n={n_main}: score={sc_opt1_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc_opt1_full < best_score:
        best_score = sc_opt1_full
        best_seq = a_opt1.copy()

    # Quick subgradient polish at n=4096
    if time.monotonic() < phase1_end + 30:
        a_sg = best_seq.copy()
        a_sg = a_sg / a_sg.sum()
        a_sg_opt, sc_sg = _subgradient_polish(a_sg, min(deadline, time.monotonic() + 40), max_steps=3000)
        sc_sg_full = _score(a_sg_opt)
        print(f"  Subgrad n={n_main}: score={sc_sg_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg_opt.copy()

    # === Phase 2: Expand to n=8192 and optimize ===
    print("Phase 2: Expand to n=8192 and optimize")
    phase2_end = min(deadline, t_start + 1100.0)
    n_big = 8192
    a_big = _interp_to_length(best_seq, n_big)
    sc_big = _score(a_big)
    print(f"  Interpolated n=8192: score={sc_big:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc_big < best_score:
        best_score = sc_big
        best_seq = a_big.copy()

    # Adam on n=8192 - this is the main phase
    c_big = _convolve(a_big, a_big)
    max_cb = float(c_big.max())
    betas2 = []
    beta = 0.003 * max_cb
    while beta > 1e-12 * max_cb:
        betas2.append(beta)
        beta *= 0.35
    if len(betas2) > 20:
        betas2 = betas2[::len(betas2) // 20] + [1e-12 * max_cb]

    per_time = phase2_end - time.monotonic() - 15
    if per_time > 50:
        a_big_opt, sc_big_opt = _adam_logspace(
            a_big, time.monotonic() + per_time,
            betas2, lr_init=0.01, max_iter_total=3000000
        )
        sc_big_opt_full = _score(a_big_opt)
        print(f"  Adam n=8192: score={sc_big_opt_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_big_opt_full < best_score:
            best_score = sc_big_opt_full
            best_seq = a_big_opt.copy()

    # === Phase 3: Subgradient polish on best ===
    print("Phase 3: Subgradient polish")
    phase3_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase3_end - 20:
        a_sg2 = best_seq.copy()
        a_sg2 = a_sg2 / a_sg2.sum()
        a_sg2_opt, sc_sg2 = _subgradient_polish(a_sg2, phase3_end, max_steps=8000)
        sc_sg2_full = _score(a_sg2_opt)
        print(f"  Subgrad polish: score={sc_sg2_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_sg2_full < best_score:
            best_score = sc_sg2_full
            best_seq = a_sg2_opt.copy()

    # === Phase 4: Coordinate descent ===
    print("Phase 4: Coordinate descent")
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end - 20:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        a_cd_opt, sc_cd = _coord_descent(a_cd, phase4_end, max_rounds=200)
        sc_cd_full = _score(a_cd_opt)
        print(f"  Coord descent: score={sc_cd_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd_opt.copy()

    # === Phase 5: Multi-restart with perturbations ===
    print("Phase 5: Multi-restart")
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end - 40:
        n_restarts = 4
        per_restart = (phase5_end - time.monotonic() - 20) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break
            if i == 0:
                p_start = _perturb_log(best_seq, 0.01, rng)
            elif i == 1:
                p_start = _perturb_log(best_seq, 0.03, rng)
            elif i == 2:
                # Shift-based perturbation
                block = len(best_seq) // 8
                shift = rng.integers(1, max(2, block // 2))
                p_start = best_seq.copy()
                p_start[:block] = np.roll(best_seq[:block], shift)
                p_start = p_start / p_start.sum()
            else:
                p_start = _perturb_log(best_seq, 0.005, rng)

            c_p = _convolve(p_start, p_start)
            max_cp = float(c_p.max())
            bs = [0.001 * max_cp, 0.0005 * max_cp, 0.0001 * max_cp, 0.00005 * max_cp]

            p_opt, sc_p = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart),
                bs, lr_init=0.006, max_iter_total=200000
            )
            sc_p_full = _score(p_opt)
            if sc_p_full < best_score:
                best_score = sc_p_full
                best_seq = p_opt.copy()
                print(f"  Restart {i}: score={sc_p_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polish ===
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()
        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.0001 * max_cp, 0.00005 * max_cp, 0.00001 * max_cp, 0.000005 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 15:
            a_pol_opt, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.003, max_iter_total=300000
            )
            sc_pol_full = _score(a_pol_opt)
            print(f"  Final Adam: score={sc_pol_full:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol_opt.copy()

            # One more subgradient pass
            if time.monotonic() < deadline - 5:
                a_sg_f = best_seq.copy()
                a_sg_f = a_sg_f / a_sg_f.sum()
                a_sg_f_opt, sc_sg_f = _subgradient_polish(a_sg_f, deadline, max_steps=2000)
                sc_sg_f_full = _score(a_sg_f_opt)
                if sc_sg_f_full < best_score:
                    best_score = sc_sg_f_full
                    best_seq = a_sg_f_opt.copy()
                    print(f"  Final subgrad: score={sc_sg_f_full:.10f}")

    # Final validation
    print(f"\nBest score achieved: {best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
