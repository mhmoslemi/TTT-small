def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):

def _gradient_active(a, c, n, tol_frac=0.002):

def _lp_direction(a, deadline, time_limit=5.0):
    """
    LP-based direction finding: find g >= 0, sum(g)=1, that minimizes 
    the maximum of conv(a, g) subject to the constraint.
    This gives a direction that, when blended with a, should not increase peaks.
    """
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    # Build constraint matrix for ||a * g||_inf <= M
    # Row k: sum_j A[k,j] * g[j] <= M
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    # For efficiency, only include rows where the constraint might be binding
    # (i.e., where conv(a,g) is likely to be close to M)
    
    # Actually, we want to find g that minimizes ||a*g||_inf
    # This is: min M' s.t. a*g <= M'*1, g >= 0, sum(g) = 1
    # Which is an LP!
    
    # Variables: g_0, ..., g_{n-1}, M'
    # Minimize M'
    # Subject to: for each k, sum_j A[k,j] * g_j <= M'
    #             sum(g) = 1
    #             g >= 0
    #             M' >= 0
    
    # The constraint matrix has 2n-1 rows for the conv constraint + 1 for sum = 1
    # For n=4096, that's 8191 constraints on 4097 variables.
    # This might be too large for linprog. Let's subsample the constraints.
    
    if n > 2048:
        # Subsample constraints - only keep every other one
        rows = np.arange(0, 2*n-1, 2)
    else:
        rows = np.arange(0, 2*n-1)
    
    n_constraints = len(rows)
    
    # Build A_ub (n_constraints * (n+1) matrix)
    # Row k: A[k,j] for j=0..n-1, then -1 for M' column
    # A_ub @ [g; M'] <= 0
    # i.e., sum_j A_orig[k,j] * g_j - M' <= 0
    
    A_ub = np.zeros((n_constraints, n + 1))
    for idx, k in enumerate(rows):
        # A_orig[k,j] = a[k-j] if 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[idx, j] = a[k - j]
        A_ub[idx, n] = -1.0  # -M' term
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M' (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * n + [(0, None)]
    
    t_start = time.monotonic()
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(n_constraints),
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                         method='highs')
        if result.success:
            g = result.x[:n]
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
            return g
    except Exception as e:
        print(f"  LP failed: {e}")
    
    return None

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5)**2 / (0.1**2))
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "binomial":
        # Approximate binomial coefficients
        k = np.arange(n)
        center = n / 2
        a = np.exp(-0.5 * ((k - center) / (center * 0.5))**2)
        return a / a.sum()
    elif kind == "log_uniform":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 10.0 * t)
        return a / a.sum()
    elif kind == "sinc":
        t = np.linspace(0, 1, n)
        x = np.pi * (t - 0.5)
        a = np.sinc(t - 0.5) ** 2  # sinc^2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):

def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):

        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)

def _perturb_block(a, rng, n_blocks=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i+block_size].copy()
        z[i:i+block_size] = z[j:j+block_size]
        z[j:j+block_size] = tmp
    return _softmax(z)

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.1, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 5))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxcor': 50, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _shrinkage(a, deadline):
    """Try zeroing out smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with adaptive log-sum-exp, Nesterov-accelerated subgradient,
LP-based direction finding, shrinkage, multi-n exploration, and diverse restarts.
Key improvements: adaptive active set growth, Nesterov momentum, LP kicks,
better time allocation, and progressive refinement.

import numpy as np
import time
from scipy.optimize import minimize, linprog

_lbfgs_params = {}


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.003):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(n)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _topk_gradient(a, c, n, k):
    """Gradient of sum of top-k autocorrelation values, scaled for objective."""
    if k >= len(c):
        indices = np.arange(len(c))
    else:
        indices = np.argsort(c)[::-1][:k]
    if len(indices) <= 200:
        grad = np.zeros(n)
        for j in indices:
            lo = max(0, j - n + 1)
            hi = min(n - 1, j)
            grad[lo:hi + 1] += a[j - hi:j - lo + 1][::-1]
        grad *= 2.0 * n
    else:
        w = np.zeros(2 * n - 1)
        w[indices] = 1.0
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * n * wc[n - 1:2 * n - 1]
    return grad


def _lbfgs_fun(z):
    return _smoobj(z, _lbfgs_params['beta'], _lbfgs_params['n'])


def _lbfgs_jac(z):
    return _smograd(z, _lbfgs_params['beta'], _lbfgs_params['n'])

        a = _softmax(z_nest)
        c = _full_convolve(a, a)

        grad = _topk_gradient(a, c, n, k_use)

        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-14:
            break

        step_count += 1

                    momentum = 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lp_direction(a, time_limit=5.0):
    """LP-based direction finding: minimize ||a*g||_inf over simplex."""
    n = len(a)
    if n > 3000:
        return None

    c = _full_convolve(a, a)
    c_max = float(c.max())
    threshold = 0.4 * c_max
    active_rows = np.where(c >= threshold)[0]

    if len(active_rows) > 800:
        rng_local = np.random.default_rng(0)
        indices = rng_local.choice(len(active_rows), 800, replace=False)
        active_rows = active_rows[indices]

    n_con = len(active_rows)
    A_ub = np.zeros((n_con, n + 1))
    for idx, k in enumerate(active_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_ub[idx, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
        A_ub[idx, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(n_con),
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                         method='highs',
                         options={'time_limit': time_limit})
        if result.success:
            g = result.x[:n]
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
            return g
    except Exception:
        pass
    return None


def _blended_direction(a, g, n_trials=12):
    """Try blending a with g at different t values."""
    best_a = a.copy()
    best_sc = _score(a)
    ts = np.linspace(0.05, 0.9, n_trials)
    for t in ts:
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2 / (0.08 ** 2))
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "binomial":
        k = np.arange(n, dtype=np.float64)
        center = n / 2.0
        a = np.exp(-0.5 * ((k - center) / (center * 0.4)) ** 2)
        return a / a.sum()
    elif kind == "log_decay":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 5.0 * t)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 2))
        amp = 0.02 * rng.uniform(0.5, 2.5)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.05, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02,
                                   0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(4000, max(30, int(remaining / (est_time_per_iter * 8))))

        _lbfgs_params['beta'] = float(beta)
        _lbfgs_params['n'] = n

        try:
            result = minimize(
                _lbfgs_fun, z, jac=_lbfgs_jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            current_a = a_new
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage(a, deadline):
    """Try zeroing out smallest elements and re-evaluating."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.05, 0.08, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = float(a_try.sum())
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _final_polish(a, deadline):
    """Very fine L-BFGS + subgradient on the best sequence found."""
    n = len(a)
    c_best = _full_convolve(a, a)
    mb = float(c_best.max())
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    beta_final = mb * 3e-5
    remaining = deadline - time.monotonic()
    if remaining > 5:
        max_iter = min(3000, max(50, int(remaining / 0.04)))
        _lbfgs_params['beta'] = float(beta_final)
        _lbfgs_params['n'] = n
        try:
            result = minimize(
                _lbfgs_fun, z, jac=_lbfgs_jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-13, 'maxcor': 60}
            )
            a_fin = _softmax(result.x)
            return a_fin
        except Exception:
            pass
    return a

        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS from parent
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=700.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Nesterov subgradient polish
    print(f"\n[Phase 2] Nesterov subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        sg_deadline = min(deadline - 400, time.monotonic() + 200)
        a_sg, sc_sg = _nesterov_subgrad(best_seq, sg_deadline, n_max_steps=15000, lr_init=0.012, tol_frac=0.003, max_k=80)
        update_best(a_sg, sc_sg, "nesterov_1")

    # Phase 3: LP direction finding
    print(f"\n[Phase 3] LP direction finding (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        n_best = len(best_seq)
        if n_best <= 2500:
            for attempt in range(3):
                if time.monotonic() > deadline - 250:
                    break
                g = _lp_direction(best_seq, time_limit=6.0)
                if g is not None:
                    a_blend, sc_blend = _blended_direction(best_seq, g)
                    update_best(a_blend, sc_blend, f"lp_dir_{attempt}")
                    if sc_blend < best_score and time.monotonic() < deadline - 100:
                        a_reopt = _run_lbfgs(a_blend, deadline, min_time_remaining=80.0)
                        update_best(a_reopt, _score(a_reopt), f"lp_reopt_{attempt}")
        else:
            print("  Skipping LP (n too large)")

    # Phase 4: Shrinkage
    print(f"\n[Phase 4] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 200)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score and time.monotonic() < deadline - 100:
            a_shr2 = _run_lbfgs(a_shr, deadline, min_time_remaining=80.0)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")
            if time.monotonic() < deadline - 50:
                a_shr3, sc_shr3 = _nesterov_subgrad(a_shr2, min(deadline - 40, time.monotonic() + 40), n_max_steps=5000, lr_init=0.01)
                update_best(a_shr3, sc_shr3, "sg_post_shrink")

    # Phase 5: Multi-n search
    print(f"\n[Phase 5] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144]
    if n_parent not in n_candidates and 512 <= n_parent <= 8192:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == len(best_seq) and best_seq is not None:
            continue

        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "tapered")
        else:
            a_init = _make_profile(n_try, "tapered")

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=50.0)
        update_best(a_opt, _score(a_opt), f"n_{n_try}")

        if time.monotonic() < deadline - 30:
            a_opt2, sc_opt2 = _nesterov_subgrad(a_opt, min(deadline - 20, time.monotonic() + 30), n_max_steps=3000, lr_init=0.01, max_k=50)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_sg")

    # Phase 6: Diverse restarts from best
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restarts = [
            ("perturb_small", lambda: _perturb(best_seq, 0.02, rng)),
            ("perturb_fourier", lambda: _perturb_fourier(best_seq, rng, n_modes=6)),
            ("perturb_block", lambda: _perturb_block(best_seq, rng, n_blocks=5)),
            ("tapered_init", lambda: _make_profile(len(best_seq), "tapered")),
            ("gaussian_init", lambda: _make_profile(len(best_seq), "gaussian")),
        ]
        for i, (name, gen_fn) in enumerate(restarts):
            if time.monotonic() > deadline - 15:
                break
            p_start = gen_fn()
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=10.0)
            update_best(a_r, _score(a_r), f"restart_{name}")
            if time.monotonic() < deadline - 8:
                a_r2, sc_r2 = _nesterov_subgrad(a_r, min(deadline - 5, time.monotonic() + 15), n_max_steps=2000, lr_init=0.008, max_k=30)
                update_best(a_r2, sc_r2, f"restart_{name}_sg")

    # Phase 7: Final intensive polish
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin = _final_polish(best_seq, deadline - 2)
        update_best(a_fin, _score(a_fin), "final_lbfgs")

        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _nesterov_subgrad(best_seq, deadline - 1, n_max_steps=5000, lr_init=0.004, tol_frac=0.002, max_k=40)
            update_best(a_fin2, sc_fin2, "final_nesterov")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
