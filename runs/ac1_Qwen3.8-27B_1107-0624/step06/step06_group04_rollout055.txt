"""
L-BFGS based optimizer with log-sum-exp smoothing for minimizing
2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-stage beta continuation, L-BFGS in log-space,
multi-restart diversification, and subgradient polishing.
Explores multiple sequence lengths (4096, 6144, 8192).
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _objective_and_grad(z, beta):
    a = _softmax(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    esum = float(e.sum())
    if esum < 1e-300 or not np.isfinite(esum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / esum
    obj = 2.0 * n * cm + 2.0 * n * beta * np.log(esum)
    a_rev = a[::-1]
    h = _full_convolve(w, a_rev)
    g_a = 4.0 * n * h[n - 1:2 * n - 1]
    mean_g = float(np.dot(a, g_a))
    grad_z = a * (g_a - mean_g)
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)
    return obj, grad_z


def _lbfgs_round(a_init, beta, max_iter, deadline, gtol=1e-8):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    state = {'calls': 0}

    def fun(z_inner):
        if time.monotonic() > deadline:
            raise RuntimeError("deadline")
        state['calls'] += 1
        obj, grad = _objective_and_grad(z_inner, beta)
        if not np.isfinite(obj):
            return 1e10, np.zeros_like(z_inner)
        return obj, grad

    try:
        result = minimize(fun, z, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                                  'gtol': gtol, 'maxls': 40, 'ftol': 1e-14})
    except (RuntimeError, Exception):
        a_out = _softmax(z)
        return a_out, _score_norm(a_out), state['calls']

    a_out = _softmax(result.x)
    sc = _score_norm(a_out)
    return a_out, sc, state['calls']


def _fista_fast(a_init, beta, deadline, max_iter=10000):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 8.0 * n / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(len(c_y))
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        y_rev = y[::-1]
        h = _full_convolve(w, y_rev)
        grad = 4.0 * n * h[n - 1:2 * n - 1]
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _subgradient_polish(a_init, deadline, n_steps=20000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    step_init = 0.1

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        top_k = min(50, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        w_vals = np.maximum(c[top_idx] - cm, 0) + 1e-12
        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_vals ** 2
        w_vec = w_vec / w_vec.sum()

        a_rev = a[::-1]
        h = _full_convolve(w_vec, a_rev)
        grad = 4.0 * n * h[n - 1:2 * n - 1]
        step = step_init / (1.0 + 0.05 * it)
        a_new = _proj_simplex(a - step * grad)
        sc = _score_norm(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.9
            a_new = _proj_simplex(a - step * grad)
            if _score_norm(a_new) < _score_norm(a):
                a = a_new

    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _run_lbfgs_schedule(a_init, deadline, n_betas=12, max_iter_per_beta=3000,
                        beta_frac_start=0.1, beta_frac_end=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    c = _full_convolve(a, a)
    max_c = float(c.max())
    if max_c < 1e-15:
        return a, _score_norm(a)

    best_a = a.copy()
    best_sc = _score_norm(a)

    beta_start = beta_frac_start * max_c
    beta_end = beta_frac_end * max_c

    fracs = np.geomspace(beta_frac_start, beta_frac_end, num=n_betas)

    for frac in fracs:
        if time.monotonic() > deadline:
            break
        beta = frac * max_c
        per_deadline = min(deadline, time.monotonic() + 60.0)
        a, sc, calls = _lbfgs_round(a, beta, max_iter_per_beta, per_deadline)
        sc_full = _score(a)
        if sc_full < best_sc:
            best_sc = sc_full
            best_a = a.copy()
        c = _full_convolve(a, a)
        max_c = float(c.max())
        beta = frac * max_c

    return best_a, best_sc
