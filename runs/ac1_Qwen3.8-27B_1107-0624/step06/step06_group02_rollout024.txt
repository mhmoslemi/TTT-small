```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_lp(a, M):
    n = len(a)
    # Variables: b[0], ..., b[n-1]
    # Maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (a*b)[i] <= M for i = 0, ..., 2n-2
    # (a*b)[i] = sum_j a[j] * b[i-j] (where 0 <= j < n and 0 <= i-j < n)
    # For fixed i: sum_{j=max(0,i-n+1)}^{min(n-1,i)} a[j] * b[i-j] <= M
    # Let k = i-j, so j = i-k:
    # sum_{k=max(0,i-n+1)}^{min(n-1,i)} a[i-k] * b[k] <= M
    
    # Build constraint matrix A_ub (shape: (2n-1, n))
    # A_ub[i, k] = a[i-k] if 0 <= i-k < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for i in range(2*n-1):
        k_min = max(0, i - n + 1)
        k_max = min(n - 1, i)
        for k in range(k_min, k_max + 1):
            A_ub[i, k] = a[i - k]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None
```

"""
Optimized step function search combining LP-based direction finding, 
gradient descent with adaptive active set, and 2-coordinate fine polishing.
Targets score < 1.5027.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assume sum(a) = 1."""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _lp_direction(a, M, max_time=10.0):
    """Solve LP to find direction that maximizes sum(b) subject to a*b <= M, b >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    
    # Build sparse constraint matrix
    # A_ub[i, k] = a[i-k] if 0 <= i-k < n, else 0
    rows = []
    cols = []
    data = []
    for k in range(n):
        i_start = k
        i_end = min(k + n, 2 * n - 1)
        length = i_end - i_start
        rows.extend(range(i_start, i_end))
        cols.extend([k] * length)
        data.extend(a[:length])
    
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    t0 = time.monotonic()
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and time.monotonic() - t0 < max_time:
        return result.x
    return None

def _lp_optimize(a_init, deadline, max_iters=200):
    """LP-based optimization: repeatedly solve LP and mix."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        c = _convolve(a, a)
        M = float(c.max())
        
        b = _lp_direction(a, M, max_time=15.0)
        if b is None:
            break
        
        b_sum = float(b.sum())
        if b_sum < 1e-10:
            break
        b = b / b_sum
        
        # Mixing step: a_new = (1-t)*a + t*b
        # Use decreasing step size
        t = max(0.01, 0.3 * (1 - it / max_iters))
        a_new = (1 - t) * a + t * b
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            if it % 10 == 0:
                print(f"  LP iter {it}: score={sc:.10f}")
        else:
            # Try smaller step
            t2 = t * 0.1
            a_new2 = (1 - t2) * a + t2 * b
            sc2 = _score_norm(a_new2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()
                a = a_new2.copy()
    
    return best_a, best_score

def _active_set_gradient(a, num_active=50):
    """Compute gradient using top-k active constraints."""
    n = len(a)
    c = _convolve(a, a)
    # Find top-k indices
    top_idx = np.argpartition(c, -num_active)[-num_active:]
    max_c = c.max()
    # Weight by proximity to max
    weights = np.exp(-(c[top_idx] - max_c) / (max_c * 0.01))
    weights /= weights.sum()
    
    # Gradient: grad[j] = sum_i w_i * 2 * a[i-j] (for valid i-j)
    # = 2 * convolve(w_padded, a)[n-1:2n-1] ... let me compute directly
    w = np.zeros(2 * n - 1)
    w[top_idx] = weights
    # grad[j] = sum_i w[i] * a[i-j] (for 0 <= i-j < n)
    # = cross-correlation
    grad = _convolve(w, a[::-1])
    grad = grad[n - 1:2 * n - 1]
    return 2.0 * grad, c

def _gradient_descent(a_init, deadline, max_iter=10000):
    """Projected gradient descent with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    step_size = 0.1
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline - 2:
            break
        
        if it % 100 == 0:
            num_active = min(50 + it // 50, 200)
            grad, c = _active_set_gradient(a, num_active)
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            # Adaptive step size
            step_size = max(0.001, 0.5 / (np.linalg.norm(grad) + 1e-10))
        
        v = momentum * v - step_size * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        else:
            a_new = np.ones(n) / n
        a = a_new
    
    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

def _two_coord_optimize(a_init, deadline):
    """2-coordinate optimization for fine polishing."""
    n = len(a_init)
    a = a_init.copy()
    c = _convolve(a, a)
    best_c_max = float(c.max())
    best_a = a.copy()
    rng = np.random.default_rng(42)
    
    # Delta grid - small values for fine tuning
    deltas = np.array([1e-5, 2e-5, 5e-5, 1e-4, 2e-4, 5e-4, 1e-3, 2e-3])
    
    updates_count = 0
    recompute_interval = 100
    no_improve_count = 0
    
    while time.monotonic() < deadline - 5:
        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Compute direction
        d = np.zeros(2 * n - 1)
        d[i:i+n] += 2.0 * a
        d[j:j+n] -= 2.0 * a
        
        # Find best delta
        best_delta = 0.0
        new_c = c + 0  # Current
        current_max = float(c.max())
        
        for delta in deltas:
            for sign in [1.0, -1.0]:
                dm = sign * delta
                if a[i] + dm < 0 or a[j] - dm < 0:
                    continue
                trial_max = float((c + dm * d).max())
                if trial_max < current_max:
                    current_max = trial_max
                    best_delta = dm
        
        if best_delta != 0:
            a[i] += best_delta
            a[j] -= best_delta
            c = c + best_delta * d
            updates_count += 1
            no_improve_count = 0
            
            if c.max() < best_c_max - 1e-15:
                best_c_max = float(c.max())
                best_a = a.copy()
        else:
            no_improve_count += 1
        
        if updates_count % recompute_interval == 0:
            c = _convolve(a, a)
        
        # If too many failures, try a different strategy
        if no_improve_count > 500:
            # Perturb slightly and restart
            mask = rng.random(n) < 0.01
            a[mask] *= (1 + 0.01 * rng.standard_normal(mask.sum()))
            a = np.maximum(a, 0)
            a = a / a.sum()
            c = _convolve(a, a)
            no_improve_count = 0
    
    return best_a, 2.0 * n * best_c_max

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _fista_smoothed(a_init, beta, deadline, max_iter=50000):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient
        wc = _convolve(w, y[::-1])
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()
        
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
