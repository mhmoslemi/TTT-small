"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: FISTA with adaptive softmax smoothing, Adam in log-space, frequency-domain equalization,
peak-targeting subgradient steps, and multi-restart diversification.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _softmax_compute(z):
    z_max = np.max(z)
    e = np.exp(z - z_max)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_adaptive(a_init, beta, deadline, max_iter, use_adaptive=True):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    stall_count = 0
    stall_threshold = 2000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        e = np.exp(shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad(y, w)

        if it % 100 == 0 or it < 3:

                stall_count = 0

                L = 4.0 * n * max_c_new / beta

            if stall_count > stall_threshold and it > 500:
                t_val = 1.0
                a_prev = a_next.copy()
                stall_count = 0

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total, use_freq_eq=True):

    v_buf = np.zeros(n)
    t = 0

    current_lr = lr_init
    stall_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        phase_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) / max(1, len(beta_schedule) - 0.01))
        while t < max_iter_total and time.monotonic() < phase_deadline:

                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            if s < 1e-300 or not np.isfinite(s):

                w[int(np.argmax(c))] = 1.0

                w = e / s

            if not np.all(np.isfinite(grad)):

            m_buf = b1 * m_buf + (1.0 - b1) * grad

            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:

            if t % 100 == 0:
                a_chk = _softmax_compute(z)

                else:
                    stall_count += 100

                A_new = new_power * np.exp(1j * phase)

                a_fe = np.maximum(a_fe, 1e-15)

                z_fe = z_fe - z_fe.mean()
                z = 0.7 * z + 0.3 * z_fe

    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:

        best_a = a_final.copy()

    return best_a, best_score

def _make_initialization(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "welch":
        # Welch-like construction: sum of squared magnitudes of a few chirps
        a = np.zeros(n)
        for k in range(3):
            t = np.arange(n)
            freq = (k + 1) / 4.0
            a += np.abs(np.exp(2j * np.pi * freq * t / n)) ** 2
        a += 0.01
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(999)
        a = rng.random(n) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    beta_schedule = np.exp(log_ratios)

    time_per_beta = (deadline - t_phase1_start - 500.0) / n_beta_levels

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline - 400:
            break

        phase_deadline = min(deadline - 400, time.monotonic() + max(30.0, time_per_beta))
        max_it = int((phase_deadline - time.monotonic()) * 5000)
        max_it = max(500, min(max_it, 200000))

        a_work, sc = _fista_adaptive(a_work, beta, phase_deadline, max_it, use_adaptive=True)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)

        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()

        print(f"  FISTA beta [{bi}/{n_beta_levels}]={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 2: Adam in log-space ---
    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    beta_fine_start = 0.003 * max_cc
    beta_fine_end = 3e-5 * max_cc
    n_fine = 12
    log_ratios_fine = np.linspace(np.log(beta_fine_start), np.log(beta_fine_end), n_fine)
    beta_fine_schedule = np.exp(log_ratios_fine)

    adam_deadline = min(deadline - 60, time.monotonic() + 350.0)
    print(f"\nPhase 2: Adam log-space, deadline={adam_deadline - time.monotonic():.1f}s")

    a_opt, sc_opt = _adam_logspace(a_work, adam_deadline, beta_fine_schedule,
                                   lr_init=0.012, max_iter_total=500000, use_freq_eq=True)
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
    print(f"  Adam done: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 3: Peak shaving ---
    if time.monotonic() < deadline - 30:
        shave_deadline = min(deadline - 10, time.monotonic() + 60.0)
        print(f"\nPhase 3: Peak shaving")
        a_shaved, sc_shaved = _peak_shaving(best_seq, shave_deadline, n_steps=3000)
        sc_shaved_full = _score(a_shaved)
        if sc_shaved_full < best_score:
            best_score = sc_shaved_full
            best_seq = a_shaved.copy()
        print(f"  Peak shaving: score={sc_shaved_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 4: Multi-restart diversification ---
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\nPhase 4: Multi-restart diversification (current best={best_score:.10f})")
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 30.0) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break

            if i == 0:
                p_start = _perturb(a_base, 0.015, rng)
            elif i == 1:
                sz = max(1, n // 2048)
                shift = rng.integers(1, max(2, sz))
                p_start = np.roll(a_base, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                # Blend with a different profile
                alt = _make_initialization(n, "tapered")
                p_start = 0.8 * a_base + 0.2 * alt
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.001 * max_cs, 0.0004 * max_cs, 0.00015 * max_cs, 0.00005 * max_cs]

            p_deadline = min(deadline - 20, time.monotonic() + max(10.0, per_restart))
            p_best, sc_best = _adam_logspace(p_start, p_deadline, bs,
                                             lr_init=0.006, max_iter_total=100000,
                                             use_freq_eq=True)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: NEW BEST score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {i}: score={sc_full:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # --- Phase 5: Ultra-fine final polish ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nPhase 5: Final ultra-fine polish")
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [3e-5 * max_cp, 1e-5 * max_cp, 3e-6 * max_cp]

        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol2, sc_pol = _adam_logspace(a_pol, time.monotonic() + per_t,
                                            betas_ult, lr_init=0.001, max_iter_total=80000,
                                            use_freq_eq=False)
            sc_pol_full = _score(a_pol2)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol2.copy()
                print(f"  Final polish: NEW BEST score={sc_pol_full:.10f}")
            else:
                print(f"  Final polish: score={sc_pol_full:.10f} (no improvement)")

    # --- Phase 6: Coordinate descent micro-adjustments ---
    if best_seq is not None and time.monotonic() < deadline - 5:
        print(f"\nPhase 6: Coordinate descent micro-adjustments")
        cd_deadline = min(deadline - 3, time.monotonic() + 10.0)
        a_cd, sc_cd = _peak_shaving(best_seq, cd_deadline, n_steps=500)
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  Coord descent: NEW BEST score={sc_cd_full:.10f}")
        else:
            print(f"  Coord descent: score={sc_cd_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initialization(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_initialization(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
