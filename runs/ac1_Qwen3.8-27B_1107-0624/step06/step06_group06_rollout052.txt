"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: Barzilai-Borwein adaptive step subgradient with momentum,
peak-targeted repulsion moves, adaptive L-BFGS continuation, multi-n search,
and diverse structured restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_conv(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * a * (h - W)


def _subgrad_bb(a_init, deadline, max_steps=20000, momentum=0.9):
    """Subgradient with Barzilai-Borwein step size and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_f = best_sc
    f = best_sc
    patience = 0
    max_patience = 200

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        # Adaptive tolerance: start loose, tighten over time
        tol_frac = max(0.0001, 0.01 * (0.999 ** i))
        g = _gradient_conv(a, c, n, tol_frac)
        g = g - np.mean(g)  # Project onto sum=0 tangent space

        # Barzilai-Borwein step
        da = a - prev_a
        df = f - prev_f
        gg = np.dot(da, g)
        if abs(df) > 1e-15 and abs(np.dot(da, da)) > 1e-15:
            step = np.dot(da, da) / max(abs(gg), 1e-10)
            step = min(step, 5.0 / (n * 0.01))
            step = max(step, 0.001)
        else:
            step = 1.0 / (n * 0.05)

        # Momentum
        v = momentum * v + (1.0 - momentum) * g
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-10:
            v = v / v_norm * min(step * np.linalg.norm(g), 2.0 / np.sqrt(n))

        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        f_new = float(_full_convolve(a_new, a_new).max()) * 2.0 * n
        # Acceptance: allow slight increase to escape local minima
        if f_new < f + 0.001 * f:
            prev_a = a.copy()
            prev_f = f
            a = a_new
            f = f_new
            if f < best_sc - 1e-12:
                best_sc = f
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            # Shrink momentum
            v *= 0.8

        if patience > max_patience:
            # Restart from best with small perturbation
            a = best_a + 0.005 * np.random.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            prev_a = a.copy()
            prev_f = _score(a)
            f = prev_f
            patience = 0
            max_patience = int(max_patience * 1.3)

    a_final = _score(a)
    if a_final < best_sc:
        best_sc = a_final
        best_a = a.copy()
    return best_a, best_sc


def _peak_repulsion(a_init, deadline, n_iters=5000):
    """Peak-targeted repulsion: move mass away from peak-contributing positions."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= (c_max - 0.01 * c_max)

        # Compute peak pressure on each position
        pressure = np.zeros(n)
        for k in np.where(active)[0]:
            for j in range(n):
                idx = int(k) - j
                if 0 <= idx < n:
                    pressure[j] += a[idx]

        # Find donor (high pressure, high value) and receiver (low pressure, low value)
        combined = pressure * a
        donor = int(np.argmax(combined))
        # Receiver: minimize pressure, prefer currently low values
        recv_score = pressure + 0.1 * a
        receiver = int(np.argmin(recv_score))

        if donor == receiver:
            continue

        # Determine transfer amount
        delta = 0.01 * a[donor]
        if delta < 1e-10:
            delta = a[donor] * 0.05

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try smaller delta
            delta2 = delta * 0.3
            a_try = a.copy()
            a_try[donor] -= delta2
            a_try[receiver] += delta2
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()

    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.006 * max(1, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15,
                         'gtol': 1e-12, 'maxcor': 50, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "alternating":
        a = np.ones(n)
        a[::2] = 1.5
        a[1::2] = 0.5
        return a / a.sum()
    elif kind == "quadratic_res":
        # Based on quadratic residue pattern
        p = max(3, int(np.sqrt(n) * 2 + 1))
        while True:
            if all((i * i) % p != (j * j) % p for i in range(1, p // 2) for j in range(i + 1, p // 2)):
                break
            p += 2
        a = np.ones(n) * 0.3
        for k in range(n):
            a[k] = 1.0 if (k * k) % p < p // 2 else 0.3
        return a / a.sum()
    elif kind == "gauss_sum":
        t = np.linspace(0, 1, n)
        a = 0.5
        for f in range(1, 5):
            a += np.exp(-f * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_structured(a, rng, mode="noise"):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    if mode == "noise":
        z += 0.03 * rng.standard_normal(n)
    elif mode == "fourier":
        t = np.arange(n)
        for _ in range(5):
            freq = rng.integers(1, max(2, n // 3))
            amp = 0.02 * rng.uniform(0.3, 1.5)
            phase = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "block":
        bs = max(1, n // 16)
        for _ in range(3):
            i = rng.integers(0, n - 2 * bs)
            j = rng.integers(0, n - 2 * bs)
            if abs(i - j) < bs:
                continue
            tmp = z[i:i + bs].copy()
            z[i:i + bs] = z[j:j + bs]
            z[j:j + bs] = tmp
    z = z - z.mean()
    return _softmax(z)


def _finalize_sequence(a):
    """Ensure the sequence is valid for the evaluator."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(256) / 256
    return a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _try_update(p, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = _finalize_sequence(p)
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _try_update(parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, uniform n={n_parent}, score={_score(parent):.10f}")
        _try_update(parent, "init")

    # === Phase 1: Subgradient BB on parent (300s) ===
    print(f"\n[Phase 1] Subgradient BB (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(deadline - 1200, t_start + 250)
    if best_seq is not None:
        a1, sc1 = _subgrad_bb(best_seq, p1_deadline, max_steps=50000, momentum=0.85)
        _try_update(a1, "subgrad_bb_1")

    # === Phase 2: L-BFGS polish (200s) ===
    print(f"\n[Phase 2] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        p2_deadline = min(deadline - 600, time.monotonic() + 180)
        a2 = _run_lbfgs(best_seq, p2_deadline, min_time_remaining=100.0)
        _try_update(a2, "lbfgs_p2")

    # === Phase 3: Peak repulsion (200s) ===
    print(f"\n[Phase 3] Peak repulsion (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        p3_deadline = min(deadline - 400, time.monotonic() + 180)
        a3, sc3 = _peak_repulsion(best_seq, p3_deadline, n_iters=3000)
        _try_update(a3, "peak_repulsion")
        if sc3 < best_score + 0.001:
            # Follow up with subgradient
            if time.monotonic() < deadline - 350:
                a3b, sc3b = _subgrad_bb(a3, min(deadline - 350, time.monotonic() + 80), max_steps=10000, momentum=0.9)
                _try_update(a3b, "peak_rep+sg")

    # === Phase 4: Multi-n search (400s) ===
    print(f"\n[Phase 4] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144, 8192]
    if n_parent not in n_candidates and 512 <= n_parent <= 16384:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 300:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue  # Already optimized

        # Initialize
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / a_init.sum()
        else:
            a_init = _make_profile(n_try, "uniform")

        sc_init = _score(a_init)
        print(f"  n={n_try}: init score={sc_init:.6f}")

        # L-BFGS optimization
        remaining_for_n = min(deadline - 250 - time.monotonic(), 120.0)
        if remaining_for_n > 10:
            a_opt = _run_lbfgs(a_init, time.monotonic() + remaining_for_n, min_time_remaining=30.0)
            _try_update(a_opt, f"lbfgs_n{n_try}")

            # Quick subgradient follow-up
            if time.monotonic() < deadline - 220:
                a_opt2, _ = _subgrad_bb(a_opt, min(deadline - 220, time.monotonic() + 60), max_steps=5000, momentum=0.9)
                _try_update(a_opt2, f"sg_n{n_try}")

    # === Phase 5: Diverse restarts (300s) ===
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        n_best = len(best_seq)
        modes = ["noise", "fourier", "block", "fourier", "noise"]
        for i, mode in enumerate(modes):
            if time.monotonic() > deadline - 100:
                break
            p_start = _perturb_structured(best_seq, rng, mode=mode)
            sc_start = _score(p_start)
            remaining_r = min(deadline - 80 - time.monotonic(), 60.0)
            if remaining_r > 5:
                a_r = _run_lbfgs(p_start, time.monotonic() + remaining_r, min_time_remaining=20.0)
                _try_update(a_r, f"restart_{mode}_{i}")
                if time.monotonic() < deadline - 60:
                    a_r2, _ = _subgrad_bb(a_r, min(deadline - 60, time.monotonic() + 30), max_steps=3000, momentum=0.9)
                    _try_update(a_r2, f"restart_{mode}_{i}_sg")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        # Iterative subgradient + L-BFGS cycle
        for cycle in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - 10 - time.monotonic()
            if remaining < 5:
                break

            # Subgradient phase
            sg_time = min(remaining * 0.4, 60.0)
            if sg_time > 3:
                a_s, _ = _subgrad_bb(best_seq, time.monotonic() + sg_time, max_steps=10000, momentum=0.88)
                _try_update(a_s, f"final_sg_{cycle}")

            # L-BFGS phase
            lbfgs_time = min(remaining * 0.4, 60.0)
            if lbfgs_time > 3 and time.monotonic() < deadline - 5:
                a_l = _run_lbfgs(best_seq, time.monotonic() + lbfgs_time, min_time_remaining=5.0)
                _try_update(a_l, f"final_lbfgs_{cycle}")

            # Peak repulsion touch-up
            if time.monotonic() < deadline - 5:
                a_p, _ = _peak_repulsion(best_seq, min(deadline - 5, time.monotonic() + 20), n_iters=500)
                _try_update(a_p, f"final_peak_{cycle}")

    # Final output
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = _finalize_sequence(best_seq)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(2048, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
