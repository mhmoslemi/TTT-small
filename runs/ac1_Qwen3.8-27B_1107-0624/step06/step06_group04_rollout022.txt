Key innovations over previous: Frank-Wolfe for local minimum escape,

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_conv_grad(a, w):
    """Compute gradient of weighted sum of convolution terms w.r.t. a.
    w is a weight array of length 2n-1 (same as convolution).
    Returns gradient of sum_k w[k] * (a*a)[k] w.r.t. a.
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    # The gradient of (a*a)[k] w.r.t. a[j] is a[j+k] + a[j-k] (with boundary handling)
    # This is equivalent to: grad = convolve(w, a) + convolve(w, a_rev) shifted appropriately
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_logspace(a, w):
    """Gradient in log-space: d/dz of objective where a = softmax(z)."""
    n = len(a)
    grad_simplex = _compute_conv_grad(a, w)
    # Chain rule: d/dz_j = (a_j * d/da_j) - (sum_i a_i * d/da_i) * a_j
    # Actually for softmax parameterization with normalization:
    # da_j/dz_j = a_j*(1-a_j), da_i/dz_j = -a_i*a_j for i != j
    # Gradient in z-space: g_z_j = sum_i (da_i/dz_j) * (df/da_i)
    # = a_j*(1-a_j)*g_j - a_j * sum_i a_i*g_i
    # = a_j * (g_j - sum_i a_i*g_i)
    W = float(np.dot(a, grad_simplex))
    return a * (grad_simplex - W)


def _softmax_weights(c, beta):
    """Compute softmax weights over convolution values for subgradient."""
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s_w = float(e.sum())
    if s_w < 1e-300 or not np.isfinite(s_w):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s_w


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on simplex with log-sum-exp smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _softmax_weights(c_y, beta)

        grad = _compute_conv_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=5000, lr_decay_factor=0.95, grad_clip=1.0):
    """Adam optimizer in log-space (softmax parameterization)."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            w = _softmax_weights(c, beta)

            grad = _compute_grad_logspace(a, w)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0002, current_lr * lr_decay_factor)

            if no_improve_count > 100 and t > 500:
                current_lr = max(0.0002, current_lr * 0.9)
                no_improve_count = 0

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _frank_wolfe(a_init, deadline, top_k=50, max_iter=50000):
    """Frank-Wolfe (conditional gradient) method for the non-smooth objective.
    Uses exact subgradient with top-k active set."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    t_fw = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_fw += 1

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Top-k active set for subgradient
        k_eff = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_eff]
        # Weights proportional to (c_k - cm + eps)^2 for active set
        c_active = c[top_idx]
        w_vals = (c_active - cm + 1e-10) ** 2
        w_sum = w_vals.sum()
        if w_sum < 1e-30:
            w_vals = np.ones(k_eff) / k_eff
        else:
            w_vals = w_vals / w_sum

        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_vals

        grad = _compute_conv_grad(a, w_vec)
        grad = np.maximum(grad, 0)  # Subgradient of max is non-negative combination

        # Frank-Wolfe: find s = argmin <grad, s> over simplex = e_j where j = argmin grad
        j_star = int(np.argmin(grad))

        # Step size: gamma_t = 2/(t+2) for standard FW, or use line search
        gamma = 2.0 / (t_fw + 2.0)
        # Cap the step to avoid too large jumps early on
        gamma = min(gamma, 0.1)

        s_vec = np.zeros(n)
        s_vec[j_star] = 1.0

        a_new = (1.0 - gamma) * a + gamma * s_vec
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 0:
            a_new = a_new / s_new

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _frank_wolfe_away(a_init, deadline, top_k=50, max_iter=50000):
    """Frank-Wolfe with away-step for faster convergence."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    support = set(np.where(a > 1e-10)[0].tolist())
    t_fw = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_fw += 1

        c = _full_convolve(a, a)
        cm = float(c.max())

        k_eff = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_eff]
        c_active = c[top_idx]
        w_vals = (c_active - cm + 1e-10) ** 2
        w_sum = w_vals.sum()
        if w_sum < 1e-30:
            w_vals = np.ones(k_eff) / k_eff
        else:
            w_vals = w_vals / w_sum

        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_vals

        grad = _compute_conv_grad(a, w_vec)
        grad = np.maximum(grad, 1e-15)

        # FW step: add direction
        j_add = int(np.argmin(grad))
        # Away step: remove direction (among support)
        if len(support) > 1:
            sup_list = list(support)
            grad_sup = np.array([grad[j] for j in sup_list])
            j_away_local = int(np.argmax(grad_sup))
            j_away = sup_list[j_away_local]
        else:
            j_away = -1

        gamma = 2.0 / (t_fw + 2.0)
        gamma = min(gamma, 0.05)

        # Try add step
        a_add = a.copy()
        a_add[j_add] += gamma
        a_add = np.maximum(a_add, 0)
        a_add = a_add / a_add.sum()
        sc_add = _score_norm(a_add)

        # Try away step
        if j_away >= 0:
            a_away = a.copy()
            a_away[j_away] -= gamma * a_away[j_away] * 2
            a_away = np.maximum(a_away, 0)
            s_away = a_away.sum()
            if s_away > 1e-10:
                a_away = a_away / s_away
                sc_away = _score_norm(a_away)
            else:
                sc_away = np.inf
        else:
            sc_away = np.inf

        # Try combine step
        a_comb = a.copy()
        a_comb[j_add] += gamma * 0.5
        a_comb[j_away] -= gamma * 0.5 * a_comb[j_away] * 2 if j_away >= 0 else 0
        a_comb = np.maximum(a_comb, 0)
        s_comb = a_comb.sum()
        if s_comb > 1e-10:
            a_comb = a_comb / s_comb
            sc_comb = _score_norm(a_comb)
        else:
            sc_comb = np.inf

        # Choose best
        candidates = [(sc_add, a_add), (sc_away, a_away if j_away >= 0 else a), (sc_comb, a_comb)]
        best_sc = min(candidates, key=lambda x: x[0])
        a = best_sc[1]

        # Update support
        support = set(np.where(a > 1e-10)[0].tolist())

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _spectral_flatten(a_init, n_iter=300, deadline=None):
    """Iteratively flatten the magnitude spectrum to reduce peak autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        # Blend: partially flatten
        blend = 0.3
        A_flat = A * ((1 - blend) + blend * avg_mag / (mag + 1e-30))
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _subgradient_steps(a_init, n_steps=5000, deadline=None, top_k=20):
    """Projected subgradient method with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    step_size = 0.3

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        k_eff = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_eff]
        w_vals = c[top_idx] - cm + 1e-10
        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_vals ** 2
        ws = w_vec.sum()
        if ws > 0:
            w_vec = w_vec / ws
        else:
            w_vec[int(np.argmax(c))] = 1.0

        grad = _compute_conv_grad(a, w_vec)
        a_new = _proj_simplex(a - step_size * grad)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size * 1.02, 1.0)
        else:
            step_size *= 0.9
            a_new2 = _proj_simplex(a - step_size * grad)
            if _score_norm(a_new2) < _score_norm(a):
                a = a_new2

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * t + 0.2 * np.sin(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bipolar_smooth":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(np.pi * t) * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    except Exception:
        parent = None

            sc_large = _score(a_large)
            print(f"  Interpolated n={n_large}: score={sc_large:.10f}")
            if sc_large < best_score + 0.1:  # Only worth trying if close
                c_l = _full_convolve(a_large, a_large)
                max_cl = float(c_l.max())
                beta_l = 0.005 * max_cl
                a_l_opt, _ = _fista_smoothed(a_large, beta_l, time.monotonic() + 30, 10000)
                a_l_opt = a_l_opt / a_l_opt.sum()
                sc_l = _score(a_l_opt)
                if sc_l < best_score:
                    best_score = sc_l
                    best_seq = a_l_opt.copy()
                    print(f"  n={n_large} FISTA: score={sc_l:.10f}")

                # Quick Adam
                if time.monotonic() < phase5_end - 20:
                    betas_l = [0.001 * max_cl, 0.0005 * max_cl, 0.0002 * max_cl]
                    per_tl = phase5_end - time.monotonic() - 10
                    a_l2, _ = _adam_logspace(a_l_opt, time.monotonic() + per_tl, betas_l,
                                             lr_init=0.005, max_iter_total=50000)
                    sc_l2 = _score(a_l2)
                    if sc_l2 < best_score:
                        best_score = sc_l2
                        best_seq = a_l2.copy()
                        print(f"  n={n_large} Adam: score={sc_l2:.10f}")

    print(f"End Phase 5: score={best_score:.10f}")

    # === Phase 6: Multi-restart diversification (300s) ===
    print("Phase 6: Multi-restart")
    phase6_end = min(deadline, t_start + 970.0)
    n_best = len(best_seq)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    if time.monotonic() < phase6_end - 30:
        n_restarts = 10
        per_restart = (phase6_end - time.monotonic() - 20) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break

            r = rng.random()
            if r < 0.15:
                p_start = _perturb(a_work, 0.01 + 0.005 * i, rng)
            elif r < 0.3:
                # Spectral flatten perturbed
                p_tmp = _perturb(a_work, 0.02, rng)
                p_start, _ = _spectral_flatten(p_tmp, n_iter=50, deadline=time.monotonic() + 5)
            elif r < 0.45:
                # Mix with structured profile
                profile_type = ["cosine", "tapered", "asymmetric", "bipolar_smooth"][i % 4]
                p_struct = _make_profile(n_best, profile_type)
                mix = 0.2 + 0.05 * i
                p_start = mix * p_struct + (1 - mix) * a_work
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif r < 0.6:
                # Circular shift in frequency domain
                A = np.fft.rfft(a_work, 2 * n_best - 1)
                # Random phase perturbation
                phase_pert = rng.standard_normal(len(A)) * 0.1
                A = A * np.exp(1j * phase_pert)
                p_time = np.fft.irfft(A, 2 * n_best - 1)[:n_best]
                p_start = np.maximum(p_time, 0)
                ps = p_start.sum()
                if ps > 1e-10:
                    p_start = p_start / ps
                else:
                    p_start = _perturb(a_work, 0.02, rng)
            elif r < 0.75:
                # Coordinate-wise random perturbation
                p_start = a_work.copy()
                mask = rng.random(n_best) < 0.1
                p_start[mask] *= np.exp(rng.standard_normal(mask.sum()) * 0.05)
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                p_start = _perturb(a_work, 0.03, rng)

            # Quick FISTA then Adam
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            beta_fast = 0.002 * max_cs
            p_opt, _ = _fista_smoothed(p_start, beta_fast, time.monotonic() + max(2.0, per_restart * 0.3), 3000)
            p_opt = p_opt / p_opt.sum()

            if time.monotonic() < phase6_end - 15:
                betas_r = [0.0005 * max_cs, 0.0002 * max_cs, 0.0001 * max_cs]
                per_t_r = phase6_end - time.monotonic() - 5
                p_opt, sc_r = _adam_logspace(p_opt, time.monotonic() + per_t_r, betas_r,
                                             lr_init=0.005, max_iter_total=30000)
                sc_r_full = _score(p_opt)
                if sc_r_full < best_score:
                    best_score = sc_r_full
                    best_seq = p_opt.copy()
                    n_best = len(best_seq)
                    a_work = best_seq.copy() / best_seq.sum()
                    print(f"  Restart {i}: score={sc_r_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 6: score={best_score:.10f}")

    # === Phase 7: Subgradient polish (100s) ===
    print("Phase 7: Subgradient polish")
    phase7_end = min(deadline, t_start + 1070.0)
    a_work = best_seq.copy() / best_seq.sum()
    if time.monotonic() < phase7_end - 10:
        a_sg, sc_sg = _subgradient_steps(a_work, n_steps=10000, deadline=phase7_end, top_k=30)
        sc_sg_full = _score(a_sg)
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg.copy()
            print(f"  Subgradient: score={sc_sg_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 7: score={best_score:.10f}")

    # === Phase 8: Ultra-fine Adam polish (400s) ===
    print("Phase 8: Ultra-fine Adam polish")
    phase8_end = min(deadline, t_start + 1470.0)
    a_work = best_seq.copy() / best_seq.sum()
    if time.monotonic() < phase8_end - 10:
        c_pol = _full_convolve(a_work, a_work)
        max_cp = float(c_pol.max())
        betas_ult = []
        b = 0.00001 * max_cp
        while b > 0.0000001 * max_cp:
            betas_ult.append(b)
            b *= 0.5
        if len(betas_ult) > 8:
            betas_ult = betas_ult[:8]

        per_t = phase8_end - time.monotonic() - 5
        a_pol, sc_pol = _adam_logspace(a_work, time.monotonic() + per_t, betas_ult,
                                       lr_init=0.002, max_iter_total=500000,
                                       lr_decay_every=10000, lr_decay_factor=0.93)
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Ultra-fine: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 8: score={best_score:.10f}")

    # === Phase 9: Final Adam with adaptive restart (200s) ===
    print("Phase 9: Final adaptive Adam")
    phase9_end = min(deadline, t_start + 1670.0)
    a_work = best_seq.copy() / best_seq.sum()
    if time.monotonic() < phase9_end - 20:
        c_fin = _full_convolve(a_work, a_work)
        max_cf = float(c_fin.max())
        betas_final = [0.000005 * max_cf, 0.000002 * max_cf, 0.000001 * max_cf, 0.0000005 * max_cf]

        # Run Adam in two halves with a small perturbation restart in between
        per_t1 = (phase9_end - time.monotonic() - 10) * 0.5
        a_f1, _ = _adam_logspace(a_work, time.monotonic() + per_t1, betas_final[:2],
                                 lr_init=0.001, max_iter_total=200000)
        sc_f1 = _score(a_f1)
        if sc_f1 < best_score:
            best_score = sc_f1
            best_seq = a_f1.copy()

        # Small perturbation and continue
        a_work2 = _perturb(a_f1, 0.005, rng)
        per_t2 = phase9_end - time.monotonic() - 5
        if per_t2 > 10:
            a_f2, _ = _adam_logspace(a_work2, time.monotonic() + per_t2, betas_final[2:],
                                     lr_init=0.0008, max_iter_total=200000)
            sc_f2 = _score(a_f2)
            if sc_f2 < best_score:
                best_score = sc_f2
                best_seq = a_f2.copy()
                print(f"  Final Adam half2: score={sc_f2:.10f}")

    print(f"End Phase 9: score={best_score:.10f}")

    # === Phase 10: One last spectral + Adam attempt (100s) ===
    print("Phase 10: Final spectral attempt")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_sf_final, _ = _spectral_flatten(best_seq, n_iter=500, deadline=min(deadline - 40, time.monotonic() + 40))
        sc_sf = _score(a_sf_final)
        if sc_sf < best_score:
            best_score = sc_sf
            best_seq = a_sf_final.copy()
            print(f"  Final spectral: score={sc_sf:.10f}")

        if time.monotonic() < deadline - 20:
            c_fs = _full_convolve(a_sf_final, a_sf_final)
            max_cfs = float(c_fs.max())
            betas_fs = [0.000001 * max_cfs, 0.0000005 * max_cfs]
            per_t_fs = deadline - time.monotonic() - 10
            if per_t_fs > 15:
                a_fs2, _ = _adam_logspace(a_sf_final, time.monotonic() + per_t_fs, betas_fs,
                                          lr_init=0.001, max_iter_total=100000)
                sc_fs2 = _score(a_fs2)
                if sc_fs2 < best_score:
                    best_score = sc_fs2
                    best_seq = a_fs2.copy()
                    print(f"  Final spectral+Adam: score={sc_fs2:.10f}")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            s = final.sum()
            final = final / s

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(512, "uniform")
    sc_fb = _score(fallback)
    print(f"FALLBACK: score={sc_fb:.10f}")
    return [float(x) for x in fallback]
