def _next_pow2(n):

def _proj_simplex(v):
    n = len(v)
    if n == 0:

def _pressure_flatten(a_init, deadline, alpha_schedule=None, max_iter=200000):
    """Iterative flattening using pressure-based gradient with target level."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    if alpha_schedule is None:
        alpha_schedule = [0.99, 0.98, 0.97, 0.95, 0.93, 0.90, 0.85, 0.80, 0.75]
    
    c_curr = _full_convolve(a, a)
    max_c = float(c_curr.max())
    
    eta = 0.1
    prev_grad = None
    prev_a = None
    prev_obj = None
    prev_grad_norm = None
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute target level
        alpha_idx = min(int(it / 2000), len(alpha_schedule) - 1)
        alpha = alpha_schedule[alpha_idx]
        T = alpha * max_c
        
        # Compute weights: w[k] = (c[k] - T)^+ squared, normalized
        excess = np.maximum(c_curr - T, 0.0)
        w_sum = float(excess.sum())
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c_curr))] = 1.0
        else:
            w = excess / w_sum
        
        # Compute gradient
        grad = _compute_gradient(a, w)
        
        # Barzilai-Borwein step size
        if prev_grad is not None and prev_a is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            s_norm2 = float(np.dot(s_vec, s_vec))
            y_norm2 = float(np.dot(y_vec, y_vec))
            sy = float(np.dot(s_vec, y_vec))
            if sy > 1e-10 and s_norm2 > 1e-10 and y_norm2 > 1e-10:
                eta_new = sy / y_norm2
                eta_new = np.clip(eta_new, 0.0001, 0.5)
                # Damped BB
                eta = 0.5 * (eta + eta_new)
            elif sy > 1e-10 and s_norm2 > 1e-10:
                eta_new = s_norm2 / sy
                eta_new = np.clip(eta_new, 0.0001, 0.5)
                eta = 0.5 * (eta + eta_new)
        
        # Update
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new
        
        # Evaluate
        c_new = _full_convolve(a_new, a_new)
        max_new = float(c_new.max())
        sc = _score_norm(a_new)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Store for BB
        prev_grad = grad.copy()
        prev_a = a.copy()
        a = a_new
        c_curr = c_new
        max_c = max_new
        
        # Decay eta slowly
        if it % 5000 == 0:
            eta *= 0.99
    
    return best_a, best_score

            w_vals = (c_top - c_min_top + 1e-10) ** 2
            w_vals = w_vals / w_vals.sum()
            w[top_idx] = w_vals
            
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            
            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)
            if no_improve_count > 100 and t > 2000:
                current_lr = max(0.0003, current_lr * 0.95)
                no_improve_count = 0
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _spectral_concentrate(a_init, n_iter=500, deadline=None):
    """Low-pass filter in frequency domain to concentrate power spectrum."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a / s
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    N_fft = _next_pow2(2 * n - 1)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        
        # Apply a "concentration" filter: boost low frequencies, suppress high
        freqs = np.arange(len(A))
        # Gentle low-pass: multiply by 1/(1 + (f/f0)^2)
        f0 = n / 4
        filt = 1.0 / (1.0 + (freqs / f0) ** 2)
        A_new = A * filt
        
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Blend
        a = 0.3 * a + 0.7 * a_new
    
    return best_a, best_score
