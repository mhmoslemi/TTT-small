"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Frank-Wolfe, active-set equalization, and adaptive smoothed L-BFGS.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _grad_qk(a, k, n):
    """Gradient of q_k(a) = sum_{i+j=k} a[i]*a[j] with respect to a."""
    # q_k(a) = sum_i a[i]*a[k-i] (valid indices)
    # dq_k/da[j] = sum_i [delta(i,j)*a[k-i] + delta(k-i,j)*a[i]]
    # = a[k-j] + a[k-j] = 2*a[k-j] if 0 <= k-j <= n-1
    # plus a[k/2] if k = 2j (the self-term)
    # Actually: q_k = sum_{i: 0<=i<=n-1, 0<=k-i<=n-1} a[i]*a[k-i]
    # If k is even and j = k/2 is valid, the term a[j]^2 appears once, so its derivative is 2*a[j].
    # For j != k/2, the terms a[j]*a[k-j] and a[k-j]*a[j] both appear, giving derivative 2*a[k-j].
    # Wait, let me be more careful.
    # q_k(a) = sum_{i: max(0,k-n+1) <= i <= min(k, n-1)} a[i] * a[k-i]
    # dq_k/da[j] = sum_{i in range} [delta(i,j) * a[k-i] + a[i] * delta(k-i, j)]
    # = a[k-j] (if j is in the valid range for i) + a[j] (if k-j is in the valid range for i)
    # The first term: j is valid for i iff max(0,k-n+1) <= j <= min(k, n-1) and 0 <= k-j <= n-1
    #   i.e., max(0, k-n+1) <= j <= min(k, n-1) and max(0, k-n+1) <= k-j <= min(k, n-1)
    #   The second condition: k-j >= max(0,k-n+1) => j <= k - max(0,k-n+1) = min(k, n-1)
    #   and k-j <= min(k, n-1) => j >= k - min(k, n-1) = max(0, k-n+1)
    #   So both terms have the same valid range!
    # Therefore: dq_k/da[j] = 2*a[k-j] if max(0,k-n+1) <= j <= min(k,n-1) and 0 <= k-j <= n-1
    #              = 2*a[k-j] if max(0, 2k-n+1) <= j <= min(k, n-1-k) ... hmm this is getting complicated.
    
    # Simpler: dq_k/da[j] = a[k-j] + a[k-j] = 2*a[k-j] if 0 <= k-j <= n-1, else 0.
    # But wait, we also need j to be in [0, n-1] (which it always is since a has length n).
    # And we need k-j to be in [0, n-1].
    # So: grad[j] = 2*a[k-j] if 0 <= k-j <= n-1, else 0.
    
    # Special case: if k = 2j (i.e., j = k/2 and k is even), then the term a[j]*a[j] appears
    # only once in the sum (not twice), so the derivative is a[j] (not 2*a[j]).
    # Wait no: in the sum sum_i a[i]*a[k-i], when i=j=k/2, the term is a[j]*a[k-j] = a[j]*a[j].
    # The derivative w.r.t. a[j] of this single term is 2*a[j] (since both factors are a[j]).
    # But the term a[j]*a[k-j] with j=k/2 is the same as the term with i=j, so it appears only once.
    # However, in the sum over i, when i=j=k/2, we get a[j]*a[k-j] = a[j]^2.
    # The derivative of a[j]^2 w.r.t. a[j] is 2*a[j].
    # But in my formula above, I get 2*a[k-j] = 2*a[j]. So it's consistent.
    # 
    # Wait, I think I'm overcomplicating this. Let me just use:
    # grad[j] = 2*a[k-j] if 0 <= k-j <= n-1, else 0.
    # This handles all cases correctly.
    
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo > hi:
        return np.zeros(n)
    j = np.arange(lo, hi + 1)
    k_minus_j = k - j
    grad = np.zeros(n)
    grad[j] = 2.0 * a[k_minus_j]
    return grad

def _grad_max_conv(a, n, top_k=1):
    """Subgradient of max_k conv(a,a)[k] w.r.t. a."""
    c = _full_convolve(a, a)
    # Find top_k peaks
    idx = np.argsort(c)[::-1][:top_k]
    # Subgradient: average of gradients at active peaks
    grad = np.zeros(n)
    for k in idx:
        grad += _grad_qk(a, int(k), n)
    grad /= len(idx)
    return grad, c

def _frank_wolfe(a_init, deadline, max_iter=10000, min_time_remaining=30.0):
    """Frank-Wolfe (conditional gradient) method for minimizing max conv on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() + min_time_remaining > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Compute subgradient
        # Use top few peaks for stability
        sorted_idx = np.argsort(c)[::-1]
        # Adaptive: use more peaks if they're close to max
        tol = 0.01 * c_max
        active = sorted_idx[c[sorted_idx] >= c_max - tol]
        if len(active) == 0:
            active = sorted_idx[:1]
        
        grad = np.zeros(n)
        for k in active:
            grad += _grad_qk(a, int(k), n)
        grad /= len(active)
        
        # Linearization: min_{s in simplex} grad^T s = grad[argmin(grad)]
        j_min = int(np.argmin(grad))
        s = np.zeros(n)
        s[j_min] = 1.0
        
        # Frank-Wolfe gap
        gap = float(grad @ (a - s))
        if gap < 1e-14:
            break
        
        # Step size: use armijo line search or standard FW step
        # Standard: t = 2/(it+2)
        t = 2.0 / (it + 2)
        
        # Line search
        f_old = c_max
        best_t = t
        for ls in range(10):
            a_try = (1 - t) * a + t * s
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old:
                best_t = t
                break
            t *= 0.5
        
        a = (1 - best_t) * a + best_t * s
        
        if it % 100 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc

def _active_set_equalize(a_init, deadline, n_peaks=5, max_iter=1000):
    """
    Active set method: find direction that reduces the max peak while
    trying to equalize the top peaks.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Get top peaks
        sorted_idx = np.argsort(c)[::-1][:n_peaks]
        tol = 0.005 * c_max
        active = sorted_idx[c[sorted_idx] >= c_max - tol]
        if len(active) < 2:
            active = sorted_idx[:min(n_peaks, len(sorted_idx))]
        
        n_active = len(active)
        
        # We want to find d such that:
        # 1. sum(d) = 0 (stay on simplex)
        # 2. a + d >= 0 (feasibility)
        # 3. grad_qk(a)^T d = 0 for k in active (equalize peaks) - relax this
        # 4. grad_qk(a)^T d < 0 for k in active (decrease peaks)
        #
        # Simplified: minimize sum_{k in active} grad_qk(a)^T d subject to sum(d)=0, ||d||_2 <= 1, a+td >= 0
        # This gives the steepest descent direction in the active peaks.
        
        # Build the combined gradient
        grad = np.zeros(n)
        for k in active:
            grad += _grad_qk(a, int(k), n)
        grad /= n_active
        
        # Project gradient to sum=0 subspace
        grad_t = grad - np.mean(grad)
        
        # Ensure a + t*d >= 0: limit step size
        # d = -grad_t
        d = -grad_t
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        # Max step before hitting zero
        neg_mask = d < 0
        if neg_mask.any():
            t_max = np.min(a[neg_mask] / (-d[neg_mask]))
        else:
            t_max = 1.0
        t_max = min(t_max, 0.1)  # Cap step size
        
        # Line search
        f_old = c_max
        t = t_max
        found = False
        for ls in range(20):
            a_try = a + t * d
            if a_try.min() < -1e-10:
                t *= 0.5
                continue
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12:
                found = True
                break
            t *= 0.5
        
        if not found:
            # Try smaller peaks set
            break
        
        a = a + t * d
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        if it % 50 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad_a = 4n * a * (h - W) where h = conv(w, a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _run_smoothed_lbfgs(a_init, deadline, beta_start=None, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)
    
    if beta_start is None:
        c_sorted = np.sort(c_cur)[::-1]
        gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_sorted[0]
        beta_start = max(gap, max_c * 0.01)
    
    betas = beta_start * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.0075, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 8))))
        
        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            else:
                # No improvement, try next beta
                pass
        except Exception:
            pass
    
    return current_a, current_sc

def _symmetric_optimize(a_init, deadline, min_time_remaining=30.0):
    """Optimize under the constraint that a is symmetric: a[i] = a[n-1-i]."""
    n = len(a_init)
    # Extract the first half (plus middle if odd)
    half = (n + 1) // 2
    # For symmetric a: a[i] = a[n-1-i]
    # First half determines everything
    
    a = a_init.copy()
    # Symmetrize
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    a_sym = a_sym / a_sym.sum()
    
    best_a = a_sym.copy()
    best_sc = _score(best_a)
    
    # Optimize using smoothed L-BFGS but with symmetry constraint
    # Parameterize: z[0:half] are free, a = [z_expand, z_expand_reversed]
    
    def sym_expand(z_half):
        if n % 2 == 0:
            full = np.concatenate([z_half, z_half[::-1]])
        else:
            full = np.concatenate([z_half, z_half[-2::-1]])
        return full
    
    # Use log-parameterization for the half
    z_half = np.log(np.maximum(a_sym[:half], 1e-30))
    
    def fun_sym(z_h):
        h = np.exp(np.clip(z_h - np.max(z_h), -700, 700))
        full = sym_expand(h)
        full = full / full.sum()
        c = _full_convolve(full, full)
        return c.max()
    
    # Actually, let's just use the main smoothed L-BFGS but project onto symmetric cone after each step
    # Simpler: just run the main optimizer and symmetrize the result
    
    a_opt, sc_opt = _run_smoothed_lbfgs(a_sym, deadline, min_time_remaining=min_time_remaining)
    a_opt_sym = 0.5 * (a_opt + a_opt[::-1])
    a_opt_sym = np.maximum(a_opt_sym, 0)
    a_opt_sym = a_opt_sym / a_opt_sym.sum()
    sc_opt_sym = _score(a_opt_sym)
    
    if sc_opt_sym < best_sc:
        best_a = a_opt_sym
        best_sc = sc_opt_sym
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03, top_k=3):
    """Subgradient method with top-k peak tracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set: top_k peaks
        sorted_idx = np.argsort(c)[::-1][:top_k]
        tol = 0.005 * c_max
        active = sorted_idx[c[sorted_idx] >= c_max - tol]
        if len(active) == 0:
            active = sorted_idx[:1]
        
        # Gradient of smoothed max
        grad = np.zeros(n)
        for k in active:
            grad += _grad_qk(a, int(k), n)
        grad /= len(active)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.9
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.3
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

def _softmax(z):

def _grad_qk(a, k, n):
    """Gradient of q_k(a) = sum_{i+j=k} a[i]*a[j] with respect to a."""
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo > hi:
        return np.zeros(n)
    j = np.arange(lo, hi + 1)
    k_minus_j = k - j
    grad = np.zeros(n)
    grad[j] = 2.0 * a[k_minus_j]
    return grad


def _active_gradient(a, c, n, top_k=3):

def _frank_wolfe(a_init, deadline, max_iter=20000, min_time_remaining=1.0):
    """Frank-Wolfe (conditional gradient) for minimizing max conv on simplex."""

            grad += _grad_qk(a, int(k), n)
        grad /= len(active)

        # Frank-Wolfe direction: vertex of simplex with min gradient
        j_min = int(np.argmin(grad))
        s = np.zeros(n)
        s[j_min] = 1.0

        # Gap check
        gap = float(grad @ a) - float(grad[j_min])
        if gap < 1e-14 * c_max:
            break

        # Step size with line search

        for ls in range(8):
            candidate = (1.0 - t) * a + t * s
            c_try = _full_convolve(candidate, candidate)
            if float(c_try.max()) < c_max - 1e-15:
                a_new = candidate
                break
            t *= 0.5

        if a_new is None:

                break
            continue

        a = a_new

        if it % 200 == 0 or it < 10:

                best_sc = sc
                best_a = a.copy()

    sc_final = _score(a)
    if sc_final < best_sc:

        best_a = a.copy()

def _active_set_equalize(a_init, deadline, n_peaks=5, max_iter=2000):
    """Active set method: reduce max peak by projected gradient on active set."""
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

        z = z + amp * np.cos(2.0 * np.pi * freq * t_arr / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    """Resample sequence to new length using linear interpolation."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _multi_scale_optimize(a_init, deadline, min_time_remaining=100.0):
    """Coarse-to-fine optimization: optimize at lower resolution, then refine."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Try coarser versions
    for factor in [4, 2]:
        n_coarse = max(64, n // factor)
        if n_coarse >= n:
            continue

        # Downsample
        a_coarse = _resample(a_init, n_coarse)
        sc_coarse = _score(a_coarse)

        # Optimize coarse
        if time.monotonic() + min_time_remaining > deadline:
            break
        a_coarse_opt, sc_coarse_opt = _run_smoothed_lbfgs(a_coarse, deadline, min_time_remaining=min_time_remaining + 30)

        # Upsample and refine
        a_up = _resample(a_coarse_opt, n)
        sc_up = _score(a_up)

        if sc_up < best_sc:
            # Refine at full resolution
            a_refined, sc_refined = _run_smoothed_lbfgs(a_up, deadline, min_time_remaining=max(30, min_time_remaining - 30))
            if sc_refined < best_sc:
                best_a = a_refined
                best_sc = sc_refined

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: Frank-Wolfe + active-set + smoothed L-BFGS +
    subgradient + symmetry + multi-scale + diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={parent_sc:.10f}")
        update_best(parent, parent_sc, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: Main smoothed L-BFGS ---
    print(f"\n[Phase 1] Smoothed L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main, sc_main = _run_smoothed_lbfgs(parent, deadline, min_time_remaining=500.0, beta_schedule="geometric")
    update_best(a_main, sc_main, "lbfgs_1")

    if best_seq is not None:
        # Subgradient polish
        if time.monotonic() < deadline - 400:
            fw_dl = min(deadline - 400, time.monotonic() + 100)
            a_sg, sc_sg = _subgrad_polish(best_seq, fw_dl, n_max_steps=3000, lr_init=0.02, top_k=3)
            update_best(a_sg, sc_sg, "sg_1")

        # Active set equalization
        if time.monotonic() < deadline - 350:
            aes_dl = min(deadline - 350, time.monotonic() + 80)
            a_aes, sc_aes = _active_set_equalize(best_seq, aes_dl, n_peaks=5, max_iter=1000)
            update_best(a_aes, sc_aes, "aes_1")

        # Second L-BFGS round
        if time.monotonic() < deadline - 300:
            a_main2, sc_main2 = _run_smoothed_lbfgs(best_seq, deadline, min_time_remaining=280.0, beta_schedule="adaptive")
            update_best(a_main2, sc_main2, "lbfgs_2")

    # --- Phase 2: Frank-Wolfe ---
    print(f"\n[Phase 2] Frank-Wolfe (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        fw_dl = min(deadline - 250, time.monotonic() + 120)
        a_fw, sc_fw = _frank_wolfe(best_seq, fw_dl, max_iter=10000)
        update_best(a_fw, sc_fw, "frank_wolfe")

        if sc_fw < best_score + 0.005 and time.monotonic() < deadline - 200:
            a_fw2, sc_fw2 = _run_smoothed_lbfgs(a_fw, deadline, min_time_remaining=180.0)
            update_best(a_fw2, sc_fw2, "fw_lbfgs")

    # --- Phase 3: Symmetric optimization ---
    print(f"\n[Phase 3] Symmetric (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sym, sc_sym = _symmetric_optimize(best_seq, deadline, min_time_remaining=130.0)
        update_best(a_sym, sc_sym, "symmetric")

    # --- Phase 4: Multi-scale ---
    print(f"\n[Phase 4] Multi-scale (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a_ms, sc_ms = _multi_scale_optimize(best_seq, deadline, min_time_remaining=100.0)
        update_best(a_ms, sc_ms, "multi_scale")

    # --- Phase 5: Multi-n exploration ---
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        if parent is not None and n_parent != n_try:
            a_init = _resample(parent, n_try)
        else:
            a_init = parent.copy()

        a_opt, sc_opt = _run_smoothed_lbfgs(a_init, deadline, min_time_remaining=40.0)
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # --- Phase 6: Diverse restarts ---
    print(f"\n[Phase 6] Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            (0.03, "perturb_small"),
            (0.07, "perturb_med"),
            ("fourier", "perturb_fourier"),
            (0.12, "perturb_large"),
        ]
        for i, cfg in enumerate(restart_configs):
            if time.monotonic() > deadline - 20:
                break
            if cfg[0] == "fourier":
                p = _perturb_fourier(best_seq, rng, n_modes=5)
            else:
                p = _perturb(best_seq, cfg[0], rng)

            a_r, sc_r = _run_smoothed_lbfgs(p, deadline, min_time_remaining=max(10, 20 - i * 3))
            update_best(a_r, sc_r, f"restart_{cfg[1]}")

            # Quick subgradient
            if sc_r < best_score + 0.003 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _subgrad_polish(a_r, min(deadline - 8, time.monotonic() + 15), n_max_steps=500, lr_init=0.008, top_k=5)
                update_best(a_r2, sc_r2, f"restart_{cfg[1]}_sg")

    # --- Phase 7: Final intensive polish ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        remaining = deadline - 3 - time.monotonic()
        n = len(best_seq)

        # Very fine L-BFGS with tiny beta
        if remaining > 5:
            c_best = _full_convolve(best_seq, best_seq)
            max_c = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = max_c * 0.00005
            max_iter = min(3000, max(100, int(remaining / 0.02)))
            try:
                result = minimize(
                    _smoobj, z, args=(beta_final, n),
                    jac=_smograd,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50, 'maxls': 100}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final subgradient with many peaks
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2, n_max_steps=1000, lr_init=0.003, top_k=8)
            update_best(a_fin2, sc_fin2, "final_sg")

        # Final active set with many peaks
        if time.monotonic() < deadline - 1:
            a_fin3, sc_fin3 = _active_set_equalize(best_seq, deadline - 1, n_peaks=10, max_iter=500)
            update_best(a_fin3, sc_fin3, "final_aes")

    # --- Return best ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
