import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    return 4.0 * n * a * (h - W)

                else:
                    w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_step = 0.5
            if update_norm > max_step:
                update = update * (max_step / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 80 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 15000 == 0:
                current_lr = max(0.00005, current_lr * 0.92)

        current_lr = max(0.00005, current_lr * lr_decay_per_phase)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _block_coordinate_descent(a, deadline, block_size=4, n_steps=10000):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    factors = [0.88, 0.93, 1.07, 1.12, 0.85, 1.15, 0.95, 1.05]

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        start = rng.integers(0, max(1, n - block_size))
        fidx = rng.integers(0, len(factors))
        factor = factors[fidx]
        a_try = best_a.copy()
        a_try[start:start + block_size] *= factor
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s > 1e-15:
            a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a_try.copy()

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _zadoff_chu_init(n, root=1):
    t = np.arange(n, dtype=np.float64)
    phase = np.pi * root * t * (t + 1) / (n + 1)
    a = np.cos(phase)
    a = np.maximum(a, 0.0) + 0.01
    s = a.sum()
    if s > 1e-15:
        return a / s
    return np.ones(n) / n


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_suppressed":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _anneal_step(a, z, temperature, n, beta, rng):
    """One simulated annealing step in log-space."""
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    grad = _compute_grad_z(a, c, w, n)
    z_new = z - temperature * grad + 0.01 * rng.standard_normal(n)
    z_new = np.clip(z_new, -60.0, 60.0)
    z_new = z_new - z_new.mean()
    return z_new


def _sa_then_adam(a_init, deadline, n, rng):
    """Simulated annealing followed by Adam polish."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c_init = _full_convolve(a_init, a_init)
    max_cc = float(c_init.max())
    beta = 0.001 * max_cc
    temperature = 0.002

    sa_iters = 200
    for i in range(sa_iters):
        if time.monotonic() > deadline - 60.0:
            break
        a = _softmax_compute(z)
        z = _anneal_step(a, z, temperature, n, beta, rng)
        temperature *= 0.995
        if i % 20 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    a_sa = _softmax_compute(z)
    sc = _score_norm(a_sa)
    if sc < best_score:
        best_score = sc
        best_a = a_sa.copy()

    # Adam polish
    c_now = _full_convolve(best_a, best_a)
    max_cn = float(c_now.max())
    betas = [0.0001 * max_cn, 0.00005 * max_cn, 0.00002 * max_cn]
    a_final, sc_final = _adam_logspace(
        best_a, deadline, betas, lr_init=0.003, max_iter_total=50000
    )
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        a0 = _make_profile(n_parent, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_parent}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    n_main = n_parent
    c_curr = _full_convolve(best_seq, best_seq)
    max_cc = float(c_curr.max())

    # Phase 1: Core fine Adam optimization
    print(f"\n=== Phase 1: Fine Adam (n={n_main}, max_cc={max_cc:.8f}) ===")
    phase1_deadline = t_start + 400.0

    betas_fine = []
    beta = 0.0003 * max_cc
    while beta > 2e-8 * max_cc:
        betas_fine.append(beta)
        beta *= 0.5
    if len(betas_fine) > 12:
        betas_fine = betas_fine[:12]

    a_opt, sc_opt = _adam_logspace(
        best_seq, phase1_deadline,
        betas_fine, lr_init=0.004, max_iter_total=600000,
        lr_decay_per_phase=0.8
    )
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        c_curr = _full_convolve(best_seq, best_seq)
        max_cc = float(c_curr.max())
        print(f"  Phase 1: NEW BEST score={sc_opt_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Phase 1: no improvement, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 2: Diverse restarts
    print(f"\n=== Phase 2: Diverse restarts ===")
    n_restarts = 10
    remaining = deadline - time.monotonic() - 150.0
    per_restart = max(25.0, min(remaining / n_restarts, 120.0))

    restart_strategies = [
        ("tiny_perturb", 0.003),
        ("small_perturb", 0.008),
        ("med_perturb", 0.015),
        ("zc_init", None),
        ("edge_blend", None),
        ("selective_perturb", None),
        ("antisym_perturb", None),
        ("roll_shift", None),
        ("large_perturb", 0.03),
        ("bimodal_blend", None),
    ]

    for i, (name, sigma) in enumerate(restart_strategies):
        if time.monotonic() > deadline - 100.0:
            break

        n = n_main
        if name == "tiny_perturb" or name == "small_perturb" or name == "med_perturb" or name == "large_perturb":
            p_start = _perturb(best_seq, sigma, rng)
        elif name == "zc_init":
            p_start = _zadoff_chu_init(n, root=rng.integers(1, n - 1))
        elif name == "edge_blend":
            alt = _make_profile(n, "edge_suppressed")
            p_start = 0.97 * best_seq + 0.03 * alt
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        elif name == "selective_perturb":
            mask = rng.random(n) < 0.05
            z = np.log(np.maximum(best_seq, 1e-15))
            z[mask] += 0.08 * rng.standard_normal(int(mask.sum()))
            z = z - z.mean()
            p_start = _softmax_compute(z)
        elif name == "antisym_perturb":
            half = n // 2
            pert = 0.02 * rng.standard_normal(half)
            z = np.zeros(n)
            z[:half] = pert
            z[half:] = -pert
            z_log = np.log(np.maximum(best_seq, 1e-15))
            z_log = z_log + z
            z_log = z_log - z_log.mean()
            p_start = _softmax_compute(z_log)
        elif name == "roll_shift":
            shift = rng.integers(1, max(2, n // 8192 + 1))
            p_start = np.roll(best_seq, shift).copy()
            p_start = p_start / p_start.sum()
        elif name == "bimodal_blend":
            alt = _make_profile(n, "bimodal")
            p_start = 0.9 * best_seq + 0.1 * alt
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        else:
            p_start = _perturb(best_seq, 0.01, rng)

        c_s = _full_convolve(p_start, p_start)
        max_cs = float(c_s.max())
        bs = [0.0001 * max_cs, 0.00004 * max_cs, 0.000015 * max_cs]

        r_deadline = time.monotonic() + per_restart
        p_best, sc_best = _adam_logspace(
            p_start, r_deadline,
            bs, lr_init=0.003, max_iter_total=80000,
            lr_decay_per_phase=0.85
        )
        sc_full = _score(p_best)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = p_best.copy()
            c_curr = _full_convolve(best_seq, best_seq)
            max_cc = float(c_curr.max())
            print(f"  Restart {i} ({name}): NEW BEST score={sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Restart {i} ({name}): score={sc_full:.10f} (best={best_score:.10f}), t={time.monotonic()-t_start:.1f}s")

    # Phase 3: SA + Adam hybrid
    print(f"\n=== Phase 3: SA + Adam hybrid ===")
    if time.monotonic() < deadline - 100.0:
        n_sa = 3
        sa_time = min(80.0, (deadline - time.monotonic() - 60.0) / n_sa)
        for i in range(n_sa):
            if time.monotonic() > deadline - 50.0:
                break
            p_start = _perturb(best_seq, 0.005 + 0.005 * i, rng)
            sa_deadline = time.monotonic() + sa_time
            sa_best, sa_sc = _sa_then_adam(p_start, sa_deadline, n_main, rng)
            sa_sc_full = _score(sa_best)
            if sa_sc_full < best_score:
                best_score = sa_sc_full
                best_seq = sa_best.copy()
                print(f"  SA {i}: NEW BEST score={sa_sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  SA {i}: score={sa_sc_full:.10f} (best={best_score:.10f}), t={time.monotonic()-t_start:.1f}s")

    # Phase 4: Top-k penalty optimization
    print(f"\n=== Phase 4: Top-k penalty ===")
    if time.monotonic() < deadline - 80.0:
        a_topk, sc_topk = _adam_logspace(
            best_seq, time.monotonic() + min(100.0, deadline - time.monotonic() - 50.0),
            [0.00005 * max_cc, 0.00002 * max_cc, 0.000008 * max_cc],
            lr_init=0.002, max_iter_total=80000,
            use_topk=True, topk_frac=0.003,
            lr_decay_per_phase=0.8
        )
        sc_topk_full = _score(a_topk)
        if sc_topk_full < best_score:
            best_score = sc_topk_full
            best_seq = a_topk.copy()
            print(f"  Top-k: NEW BEST score={sc_topk_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Top-k: score={sc_topk_full:.10f} (best={best_score:.10f}), t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Block coordinate descent
    print(f"\n=== Phase 5: Block coordinate descent ===")
    if time.monotonic() < deadline - 40.0:
        bcd_deadline = time.monotonic() + min(50.0, deadline - time.monotonic() - 30.0)
        a_bcd, sc_bcd = _block_coordinate_descent(best_seq, bcd_deadline, block_size=3, n_steps=15000)
        sc_bcd_full = _score(a_bcd)
        if sc_bcd_full < best_score:
            best_score = sc_bcd_full
            best_seq = a_bcd.copy()
            print(f"  BCD: NEW BEST score={sc_bcd_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 6: Final ultra-fine polish with smallest possible beta
    print(f"\n=== Phase 6: Ultra-fine final polish ===")
    if time.monotonic() < deadline - 15.0:
        c_pol = _full_convolve(best_seq, best_seq)
        max_cp = float(c_pol.max())
        betas_ult = [1e-6 * max_cp, 5e-7 * max_cp, 2e-7 * max_cp, 1e-7 * max_cp]
        a_ultra, sc_ultra = _adam_logspace(
            best_seq, deadline - 10.0,
            betas_ult, lr_init=0.0008, max_iter_total=40000,
            lr_decay_per_phase=0.7
        )
        sc_ultra_full = _score(a_ultra)
        if sc_ultra_full < best_score:
            best_score = sc_ultra_full
            best_seq = a_ultra.copy()
            print(f"  Ultra-fine: NEW BEST score={sc_ultra_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Ultra-fine: score={sc_ultra_full:.10f} (best={best_score:.10f}), t={time.monotonic()-t_start:.1f}s")

    # Phase 7: One last tiny BCD pass
    if time.monotonic() < deadline - 10.0:
        a_last, sc_last = _block_coordinate_descent(
            best_seq, deadline - 5.0, block_size=2, n_steps=5000
        )
        sc_last_full = _score(a_last)
        if sc_last_full < best_score:
            best_score = sc_last_full
            best_seq = a_last.copy()
            print(f"  Final BCD: NEW BEST score={sc_last_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq)} ===")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
