import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    # Assumes sum(a) = 1
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    # Project onto {x : x >= 0, sum(x) = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _grad_conv_max_smoothed(a, beta):
    # Gradient of the smoothed max of a*a
    n = len(a)
    c = _convolve(a, a)
    cm = c.max()
    shifted = (c - cm) / beta
    np.clip(shifted, -500, 500, out=shifted)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    # Gradient of max_k (a*a)_k w.r.t. a
    # (a*a)_k = sum_j a_j * a_{j+k}
    # d/dx_i (a*a)_k = a_{i+k} + a_{i-k} (for valid indices)
    # So grad_i = sum_k w_k * (a_{i+k} + a_{i-k})
    # This is 2 * (w * a_reversed) convolved with a, evaluated at the right indices
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    grad = 2.0 * wc[n-1:2*n-1]
    return grad, w, c

def _fista(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_init.copy()
    t_val = 1.0
    L = 4.0 / beta  # Lipschitz constant bound (rough)
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n
        
        grad, _, _ = _grad_conv_max_smoothed(y, beta)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()
        
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

        if time.monotonic() > deadline:
            break
        
        c = _convolve(a, a)
        peak_lag = int(np.argmax(c))
        peak_val = c[peak_lag]
        
        # Find coordinates that contribute most to the peak lag
        # (a*a)_peak = sum_j a_j * a_{j+peak}
        # Contribution of a_j to peak: a_{j+peak} + a_{|j-peak|} (roughly)
        
        if n > 2048:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)
        
        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]

                    a[j] = old_val
                    break
        if not improved and r > 10:

    return best_a, best_sc

Multi-phase optimizer for minimizing peak autocorrelation of non-negative sequences.
Phases: FISTA with backtracking, Adam in log-space, Nesterov subgradient,
coordinate line search, and n-refinement. Targets sub-1.5027 score.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    cm = c.max()
    shifted = (c - cm) / beta
    np.clip(shifted, -500, 500, out=shifted)
    e = np.exp(shifted)

        w[int(np.argmax(c))] = 1.0
    else:

    return w

    n = len(a)
    a_rev = a[::-1]

        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        w = _compute_weights(c_y, beta)
        grad = _grad_smoothed(y, w)

        # Objective: smoothed max of a*a
        # f(a) = (1/beta) * log(sum(exp((c - c_max)/beta))) + c_max
        # For projection, we use the gradient step with backtracking
        step_attempt = step
        a_next = None
        for _bt in range(20):
            a_cand = _proj_simplex(y - step_attempt * grad)
            c_cand = _convolve(a_cand, a_cand)
            obj_cand = float(c_cand.max())
            obj_y = float(c_y.max())
            # Lipschitz condition: f(a_cand) <= f(y) + <grad, a_cand - y> + (L/2)||a_cand - y||^2
            linear = float(np.dot(grad, a_cand - y))
            quad = 0.5 * L * step_attempt * step_attempt * float(np.dot(grad, grad))
            if obj_cand <= obj_y + linear + quad + 1e-12:
                a_next = a_cand
                break
            step_attempt *= 0.5
        if a_next is None:
            a_next = _proj_simplex(y - step_attempt * grad)

        # Update best
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_log(a_init, deadline, betas, lr_init=0.01, max_iter_total=1000000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            w = _compute_weights(c, beta)

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 0.4:
                update = update * (0.4 / un)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.0002, current_lr * 0.92)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _nesterov_subgrad(a_init, deadline, max_iter=50000):
    n = len(a_init)
    x_curr = a_init.copy()
    x_prev = a_init.copy()
    t_val = 1.0
    c_init = 0.5
    best_a = x_curr.copy()
    best_sc = _score_norm(x_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x_curr + ((t_val - 1.0) / t_next) * (x_curr - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        k_star = int(np.argmax(c_y))
        # Subgradient at the exact max
        a_rev = y[::-1]
        e_k = np.zeros(2 * n - 1)
        e_k[k_star] = 1.0
        grad = _grad_smoothed(y, e_k)

        alpha = c_init / (1.0 + it ** 0.5)
        x_next = _proj_simplex(y - alpha * grad * c_y.max() * 2.0 / (t_next * t_next))

        if it % 200 == 0:
            sc = _score_norm(x_next)
            if sc < best_sc:
                best_sc = sc
                best_a = x_next.copy()

        x_prev = x_curr
        x_curr = x_next
        t_val = t_next

    sc_final = _score_norm(x_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = x_curr.copy()
    return best_a, best_sc


def _coord_line_search(a_init, deadline, max_rounds=300):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _convolve(a, a)
        peak_val = c.max()

        if n > 2048:
            size = min(768, n)
            # Bias selection toward high-amplitude coordinates
            probs = a / a.sum()
            probs = 0.5 * probs + 0.5 / n
            indices = rng.choice(n, size=size, replace=False, p=probs)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-12:
                # Try increasing from zero
                for factor in [1e-6, 1e-5, 1e-4]:
                    a[j] = factor
                    sc = _score_norm(a)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                        old_val = factor
                    else:
                        a[j] = 0.0
                        break
                continue
            # Binary search for best multiplier
            best_factor = 1.0
            best_local_sc = best_sc
            lo, hi = 0.5, 1.5
            a[j] = old_val
            for _ in range(8):
                mid = (lo + hi) / 2.0
                a[j] = old_val * mid
                sc = _score_norm(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_factor = mid
                    best_a_local = a.copy()
                if sc < peak_val * 2.0 * n:
                    lo = mid
                else:
                    hi = mid
            a[j] = old_val * best_factor
            if best_local_sc < best_sc - 1e-12:
                best_sc = best_local_sc
                best_a = a.copy()
                improved = True
            else:
                a[j] = old_val

        if not improved and r > 15:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals():
            hs = globals()['height_sequence_1']
            if hs is not None and len(hs) > 0:
                parent = np.asarray(hs, dtype=np.float64).ravel()
                parent = np.clip(parent, 0.0, 1000.0)
                s = float(parent.sum())
                if s > 0.01:
                    parent = parent / s
    except Exception:
        parent = None

    if parent is not None:
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={len(a0)}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n0 = 4096
        a0 = np.ones(n0) / n0
        sc0 = _score(a0)
        print(f"Uniform n={n0}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # === Phase 1: FISTA with backtracking ===
    print("Phase 1: FISTA backtracking")
    phase1_end = t_start + 350.0
    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())

    # Beta continuation
    n_betas = 12
    beta_start = 0.04 * max_c0
    beta_end = 1e-9 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for i, beta in enumerate(betas):
        if time.monotonic() > phase1_end - 3:
            break
        remaining = phase1_end - time.monotonic() - 2
        per_time = min(25.0, remaining)
        if per_time < 3:
            break
        a0, sc = _fista_bt(a0, beta, time.monotonic() + per_time, 80000)
        a0 = a0 / a0.sum()
        sc_full = _score(a0)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a0.copy()
            print(f"  FISTA beta={beta:.2e} (i={i}): score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    # === Phase 2: Nesterov subgradient (exact max) ===
    print("Phase 2: Nesterov subgradient")
    phase2_end = t_start + 550.0
    if time.monotonic() < phase2_end - 10:
        per_t = phase2_end - time.monotonic() - 5
        a0, sc = _nesterov_subgrad(a0, time.monotonic() + per_t, max_iter=100000)
        a0 = a0 / a0.sum()
        sc_full = _score(a0)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a0.copy()
            print(f"  Subgrad: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    # === Phase 3: Adam log-space fine optimization ===
    print("Phase 3: Adam log-space")
    phase3_end = t_start + 1100.0
    c3 = _convolve(a0, a0)
    max_c3 = float(c3.max())
    betas_fine = np.geomspace(0.001 * max_c3, 1e-11 * max_c0, 14)
    per_t = phase3_end - time.monotonic() - 15
    if per_t > 30:
        a_opt, sc_opt = _adam_log(a0, time.monotonic() + per_t, betas_fine, lr_init=0.008)
        sc_full = _score(a_opt)
        print(f"  Adam: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()

    # === Phase 4: Coordinate line search ===
    print("Phase 4: Coordinate line search")
    phase4_end = t_start + 1400.0
    if time.monotonic() < phase4_end - 30:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        per_t = phase4_end - time.monotonic() - 10
        a_cd, sc_cd = _coord_line_search(a_cd, time.monotonic() + per_t, max_rounds=400)
        sc_cd_full = _score(a_cd)
        print(f"  Coord: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.0f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Phase 5: Multi-restart with perturbations ===
    print("Phase 5: Multi-restart")
    phase5_end = t_start + 1600.0
    if time.monotonic() < phase5_end - 60:
        n_restarts = 5
        per_restart = (phase5_end - time.monotonic() - 30) / n_restarts
        a_work = best_seq.copy() / best_seq.sum()
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 20:
                break
            sigma = 0.01 + 0.008 * i
            z = np.log(np.maximum(a_work, 1e-15)) + sigma * rng.standard_normal(len(a_work))
            z = z - z.mean()
            p_start = _softmax(z)

            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = np.geomspace(0.0003 * max_cs, 1e-11 * max_cs, 8)
            per_t = max(5.0, per_restart * 0.6)
            p_best, _ = _adam_log(p_start, time.monotonic() + per_t, bs, lr_init=0.005)
            # Quick FISTA polish
            c_pb = _convolve(p_best, p_best)
            max_cp = float(c_pb.max())
            per_t2 = max(3.0, per_restart * 0.3)
            p_best, _ = _fista_bt(p_best, 1e-6 * max_cp, time.monotonic() + per_t2, 20000)
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                a_work = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    # === Phase 6: Try n=8192 ===
    print("Phase 6: n=8192 exploration")
    phase6_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase6_end - 50 and best_seq is not None:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interp n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < phase6_end - 30:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            # Quick FISTA
            per_t = min(30.0, phase6_end - time.monotonic() - 20)
            if per_t > 10:
                a_big, _ = _fista_bt(a_big, 0.005 * max_cb, time.monotonic() + per_t, 30000)
                a_big = a_big / a_big.sum()
                sc = _score(a_big)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_big.copy()
                    print(f"  n=8192 FISTA: score={sc:.10f}")
            # Adam
            if time.monotonic() < phase6_end - 15:
                c_b2 = _convolve(a_big, a_big)
                max_cb2 = float(c_b2.max())
                bs_big = np.geomspace(0.001 * max_cb2, 1e-10 * max_cb2, 8)
                per_t2 = phase6_end - time.monotonic() - 10
                if per_t2 > 15:
                    a_big, _ = _adam_log(a_big, time.monotonic() + per_t2, bs_big, lr_init=0.004)
                    sc = _score(a_big)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a_big.copy()
                        print(f"  n=8192 Adam: score={sc:.10f}")

    # === Phase 7: Final ultra-fine Adam ===
    print("Phase 7: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()
        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = np.geomspace(5e-6 * max_cp, 1e-12 * max_cp, 10)
        per_t = deadline - time.monotonic() - 3
        if per_t > 10:
            a_pol, _ = _adam_log(a_pol, time.monotonic() + per_t, betas_ult, lr_init=0.001)
            sc_pol = _score(a_pol)
            print(f"  Final: score={sc_pol:.10f} t={time.monotonic()-t_start:.0f}s")
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()

    # Final validation
    print(f"All done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
