def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = v - v.max()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]

            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _peak_directed_cd(a_init, deadline, max_rounds=500, n_peaks=3):
    """Coordinate descent that directly attacks the highest autocorrelation peaks.
    For each peak, identifies contributing coordinates and uses gradient-based
    updates to reduce the peak while minimizing damage to other peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(99)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _convolve(a, a)
        # Identify top peaks
        sorted_idx = np.argsort(c)[::-1]
        top_peaks = sorted_idx[:n_peaks]

        round_improved = False
        for k in top_peaks:
            if time.monotonic() > deadline:
                break

            # c[k] = sum_j a[j]*a[k-j] for valid j
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min >= j_max:
                continue

            js = np.arange(j_min, j_max + 1)
            products = a[js] * a[k - js]

            # Get top contributing pairs
            n_top = min(30, len(js))
            top_j = js[np.argsort(products)[-n_top:]]

            for j in top_j:
                partner = k - j
                if partner == j:
                    # a[j]^2 term: reduce a[j]
                    for factor in [0.92, 0.96, 0.99]:
                        old = a[j]
                        a[j] = old * factor
                        s = a.sum()
                        if s > 0.01:
                            sc = _score_norm(a / s)
                            if sc < best_sc - 1e-12:
                                best_sc = sc
                                best_a = a.copy()
                                round_improved = True
                                a = a / s
                                break
                        a[j] = old
                else:
                    # a[j]*a[partner] term: reduce one of them
                    for target in [j, partner]:
                        for factor in [0.94, 0.97, 0.99]:
                            old = a[target]
                            a[target] = old * factor
                            s = a.sum()
                            if s > 0.01:
                                a_norm = a / s
                                sc = _score_norm(a_norm)
                                if sc < best_sc - 1e-12:
                                    best_sc = sc
                                    best_a = a_norm.copy()
                                    round_improved = True
                                    a = a_norm
                                    break
                            a[target] = old

        # Periodic re-randomization to escape local minima
        if r % 50 == 49 and r > 10:
            perturb = 1.0 + 0.01 * rng.standard_normal(n)
            a = a * np.abs(perturb)
            a = a / a.sum()
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        if not round_improved and r > 30:
            # Try a larger perturbation
            perturb = 1.0 + 0.05 * rng.standard_normal(n)
            a = a * np.abs(perturb)
            a = a / a.sum()
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            else:
                a = best_a.copy()
                if r > 80:
                    break

    return best_a, best_sc


def _sinusoidal_init(n, n_components=5, rng=None):
    """Initialize with a sum of rectified sinusoids."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = np.ones(n) * 0.1
    for _ in range(n_components):
        freq = rng.integers(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 2.0)
        a += amp * np.cos(freq * x + phase)
    a = np.maximum(a, 0.0)
    a = a + 0.01
    a = a / a.sum()
    return a


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _multi_start_optimize(a_base, deadline, n_restarts, rng):
    """Try multiple perturbations of a_base with Adam + L-BFGS polish."""
    n = len(a_base)
    best_a = a_base.copy()
    best_sc = _score_norm(a_base)

    per_restart = (deadline - time.monotonic() - 10) / max(1, n_restarts)

    for i in range(n_restarts):
        if time.monotonic() > deadline - 10:
            break
        if time.monotonic() + 5 > deadline:
            break

        phase_type = i % 5
        if phase_type == 0:
            a_start = _perturb(a_base, 0.015 + 0.005 * i, rng)
        elif phase_type == 1:
            sz = max(1, n // 2048)
            shift = rng.integers(1, max(2, sz))
            a_start = np.roll(a_base, shift).copy()
            a_start = a_start / a_start.sum()
        elif phase_type == 2:
            a_start = _perturb(a_base, 0.04, rng)
        elif phase_type == 3:
            # Block shuffle
            block = max(1, n // 8)
            a_start = a_base.copy()
            blocks = np.array_split(np.arange(n), block)
            rng.shuffle(blocks)
            a_start = a_base[blocks]
            a_start = a_start / a_start.sum()
        else:
            # Mirror + perturb
            a_start = np.concatenate([a_base[:n//2][::-1], a_base[n//2:]])
            a_start = _perturb(a_start, 0.02, rng)

        c_s = _convolve(a_start, a_start)
        max_cs = float(c_s.max())

        # Adam phase
        t_adam_end = time.monotonic() + max(3.0, per_restart * 0.5)
        betas = [0.001 * max_cs, 0.0003 * max_cs, 0.0001 * max_cs]
        a_opt, sc_opt = _adam_logspace(
            a_start, t_adam_end, betas, lr_init=0.008, max_iter_total=100000
        )

        # L-BFGS polish
        if time.monotonic() < deadline - 15:
            z_log = np.log(np.maximum(a_opt, 1e-15))
            z_log = z_log - z_log.mean()
            c_opt = _convolve(a_opt, a_opt)
            max_co = float(c_opt.max())
            t_lb_end = min(deadline - 10, time.monotonic() + max(3.0, per_restart * 0.3))
            z_pol, f_pol = _lbfgs_optimize(z_log, 0.0001 * max_co, 200,
                                            t_lb_end - time.monotonic())
            a_pol = _softmax(z_pol)
            a_pol = a_pol / a_pol.sum()
            sc_pol = _score_norm(a_pol)
            if sc_pol < sc_opt:
                a_opt = a_pol
                sc_opt = sc_pol

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"    Multi-start {i}: score={_score(best_a):.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}")

    # === Phase 1: FISTA with beta continuation (300s) ===
    print("Phase 1: FISTA beta continuation")
    phase1_end = min(deadline, t_start + 300.0)
    a_curr = a0.copy()

    betas_fista = []
    beta = 0.04 * max_c0
    while beta > 1e-10 * max_c0:
        betas_fista.append(beta)
        beta *= 0.35
    betas_fista.append(1e-10 * max_c0)
    if len(betas_fista) > 20:
        step_b = max(1, len(betas_fista) // 20)
        betas_fista = betas_fista[::step_b] + [1e-10 * max_c0]

    for i, beta in enumerate(betas_fista):
        if time.monotonic() > phase1_end - 5:
            break
        per_time = min(30.0, phase1_end - time.monotonic() - 3)
        if per_time < 2:
            break
        a_curr, sc = _fista_smoothed(a_curr, beta, time.monotonic() + per_time, 100000)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _convolve(a_curr, a_curr)
        max_c_curr = float(c_curr.max())

    print(f"End Phase 1: score={best_score:.10f}")

    # === Phase 2: L-BFGS with continuation (400s) ===
    print("Phase 2: L-BFGS continuation")
    phase2_end = min(deadline, t_start + 700.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    c_work = _convolve(a_work, a_work)
    max_cw = float(c_work.max())

    z_work = np.log(np.maximum(a_work, 1e-15))
    z_work = z_work - z_work.mean()

    lbfgs_betas = []
    beta = 0.005 * max_cw
    while beta > 5e-14 * max_c0:
        lbfgs_betas.append(beta)
        beta *= 0.3
    lbfgs_betas.append(5e-14 * max_c0)
    if len(lbfgs_betas) > 15:
        step_l = max(1, len(lbfgs_betas) // 15)
        lbfgs_betas = lbfgs_betas[::step_l] + [5e-14 * max_c0]

    for i, beta in enumerate(lbfgs_betas):
        if time.monotonic() > phase2_end - 10:
            break
        remaining = phase2_end - time.monotonic() - 5
        per_time = min(remaining / (len(lbfgs_betas) - i), 40.0)
        if per_time < 3:
            break

        z_work, f_val = _lbfgs_optimize(z_work, beta, 500, per_time)
        a_lb = _softmax(z_work)
        a_lb = a_lb / a_lb.sum()
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS beta={beta:.2e}: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")

        # Also try power-mean for robustness
        if i % 3 == 1 and time.monotonic() < phase2_end - 15:
            z_pw, f_pw = _lbfgs_power(z_work, 8, 200, 15.0)
            a_pw = _softmax(z_pw)
            a_pw = a_pw / a_pw.sum()
            sc_pw = _score(a_pw)
            if sc_pw < best_score:
                best_score = sc_pw
                best_seq = a_pw.copy()
                z_work = np.log(np.maximum(a_pw, 1e-15))
                z_work = z_work - z_work.mean()
                print(f"  L-BFGS power beta={beta:.2e}: score={sc_pw:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 2: score={best_score:.10f}")

    # === Phase 3: Peak-directed coordinate descent (200s) ===
    print("Phase 3: Peak-directed CD")
    phase3_end = min(deadline, t_start + 900.0)
    a_cd = best_seq.copy()
    a_cd = a_cd / a_cd.sum()
    if time.monotonic() < phase3_end - 30:
        a_cd, sc_cd = _peak_directed_cd(a_cd, phase3_end, max_rounds=1000, n_peaks=5)
        sc_cd_full = _score(a_cd)
        print(f"  Peak CD: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Phase 4: Multi-start exploration (250s) ===
    print("Phase 4: Multi-start")
    phase4_end = min(deadline, t_start + 1150.0)
    if time.monotonic() < phase4_end - 30:
        a_ms, sc_ms = _multi_start_optimize(best_seq, phase4_end, n_restarts=8, rng=rng)
        sc_ms_full = _score(a_ms)
        print(f"  Multi-start: score={sc_ms_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_ms_full < best_score:
            best_score = sc_ms_full
            best_seq = a_ms.copy()

    # === Phase 5: Fine L-BFGS + Adam polish (200s) ===
    print("Phase 5: Fine polish")
    phase5_end = min(deadline, t_start + 1350.0)
    a_fin = best_seq.copy()
    a_fin = a_fin / a_fin.sum()

    c_fin = _convolve(a_fin, a_fin)
    max_cf = float(c_fin.max())

    # Adam with very fine beta
    if time.monotonic() < phase5_end - 30:
        t_adam = min(phase5_end - 20, time.monotonic() + 80.0)
        betas_fine = [0.0001 * max_cf, 0.00003 * max_cf, 0.00001 * max_cf,
                      0.000003 * max_cf, 0.000001 * max_cf]
        a_fin, sc_fin = _adam_logspace(a_fin, t_adam, betas_fine,
                                       lr_init=0.003, max_iter_total=500000)
        sc_fin_full = _score(a_fin)
        print(f"  Fine Adam: score={sc_fin_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_fin_full < best_score:
            best_score = sc_fin_full
            best_seq = a_fin.copy()

    # L-BFGS ultra-fine
    if time.monotonic() < phase5_end - 10:
        z_fin = np.log(np.maximum(best_seq, 1e-15))
        z_fin = z_fin - z_fin.mean()
        c_b = _convolve(best_seq, best_seq)
        max_cb = float(c_b.max())
        z_fin, f_fin = _lbfgs_optimize(z_fin, 1e-6 * max_cb, 300,
                                       min(60.0, phase5_end - time.monotonic() - 5))
        a_fin2 = _softmax(z_fin)
        a_fin2 = a_fin2 / a_fin2.sum()
        sc_fin2 = _score(a_fin2)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_seq = a_fin2.copy()
            print(f"  Fine L-BFGS: score={sc_fin2:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: n=8192 exploration (200s) ===
    print("Phase 6: n=8192 exploration")
    phase6_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase6_end - 30 and best_seq is not None:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interpolated n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < phase6_end - 20:
            c_big = _convolve(a_big, a_big)
            max_cb2 = float(c_big.max())
            # FISTA on big
            t_fista_big = min(phase6_end - 15, time.monotonic() + 40.0)
            betas_big = [0.01 * max_cb2, 0.003 * max_cb2, 0.001 * max_cb2]
            for beta_b in betas_big:
                if time.monotonic() > t_fista_big - 5:
                    break
                a_big, sc_b = _fista_smoothed(a_big, beta_b, t_fista_big, 50000)
                a_big = a_big / a_big.sum()
                sc_b_full = _score(a_big)
                if sc_b_full < best_score:
                    best_score = sc_b_full
                    best_seq = a_big.copy()
                    print(f"  n=8192 FISTA beta={beta_b:.2e}: score={sc_b_full:.10f}")

            # Adam on big
            if time.monotonic() < phase6_end - 10:
                c_bo = _convolve(a_big, a_big)
                max_cbo = float(c_bo.max())
                bs_big = [0.001 * max_cbo, 0.0003 * max_cbo, 0.0001 * max_cbo,
                          0.00003 * max_cbo]
                per_t = min(60.0, phase6_end - time.monotonic() - 5)
                if per_t > 10:

                                                       bs_big, lr_init=0.005,
                                                       max_iter_total=200000)

                    if sc_big_full < best_score:

                        best_seq = a_big.copy()
                        print(f"  n=8192 Adam: score={sc_big_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine coordinate descent (150s) ===
    print("Phase 7: Final CD polish")
    if best_seq is not None and time.monotonic() < deadline - 20:

        a_final = a_final / a_final.sum()

        if sc_final_full < best_score:

            best_seq = a_final.copy()

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:

        if s < 0.01:

        try:

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
