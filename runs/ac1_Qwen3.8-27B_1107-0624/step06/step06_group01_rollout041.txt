"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using asymmetric Beta profile initialization,
FISTA with softmax-smoothed max, Adam log-space refinement, active-set polish,
and coordinate descent. Key insight: non-palindromic sequences with zero endpoints
minimize the self-convolution peak by avoiding palindromic alignment at the center.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_conv(a, w, n):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on softmax-smoothed max of convolution, projected to simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta + 1e-10
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad_conv(y, w, n)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta + 1e-10
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 2.0:
                update = update * (2.0 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 10000 == 0:
                current_lr = max(0.0003, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _active_set_polish(a, deadline, n_active=50):
    """Targeted optimization on active (near-max) convolution positions."""
    n = len(a)
    best_a = a.copy()
    c = _full_convolve(best_a, best_a)
    best_score = float(c.max()) * 2.0 * n

    for step in range(200):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        cm = float(c.max())
        threshold = cm * 0.95
        active_mask = c >= threshold
        n_act = int(active_mask.sum())
        if n_act == 0:
            active_mask[int(np.argmax(c))] = True
            n_act = 1

        # Limit active set size
        if n_act > n_active:
            top_k = np.argsort(c)[::-1][:n_active]
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[top_k] = True

        grad = _compute_grad_conv(best_a, active_mask.astype(np.float64), n)
        # Feature normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break
        grad = grad / gnorm

        # Backtracking line search
        lr = 0.01
        a_curr = best_a
        f_curr = _score_norm(a_curr)
        for _ in range(20):
            a_try = _proj_simplex(a_curr - lr * grad)
            f_try = _score_norm(a_try)
            if f_try < f_curr - 1e-12 * f_curr:
                best_a = a_try
                break
            lr *= 0.5
        else:
            break

    return best_a


def _coord_descent(a, deadline, n_trials=5000):
    """Random coordinate perturbation polish."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    factors = [0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 0.6, 1.4]

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        i = int(rng.integers(0, n))
        f = float(rng.choice(factors))
        a_try = best_a.copy()
        a_try[i] *= f
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        sc = _score_norm(a_try)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_try.copy()

    return best_a, best_score


def _make_beta_profile(n, p, q):
    """Asymmetric Beta profile: a[i] = (i+1)^p * (n-i)^q, zero at endpoints."""
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t**p * (n + 1 - t)**q
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_hann_tilted(n, alpha):
    """Hann window with exponential tilt for asymmetry."""
    i = np.arange(n, dtype=np.float64)
    hann = np.sin(np.pi * (i + 0.5) / n) ** 2
    tilt = np.exp(alpha * i / n)
    a = hann * tilt
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                sc_p = _score(parent)
                print(f"Parent loaded: n={len(parent)}, score={sc_p:.10f}")
                best_seq = parent.copy()
                best_score = sc_p
    except Exception:
        pass

    # Main optimization at n=8192
    n_main = 8192
    print(f"\n=== Main optimization: n={n_main} ===")
    t = time.monotonic()

    # Generate diverse initializations
    profiles = []
    # Asymmetric Beta profiles (key: p != q for non-palindromic structure)
    for p, q in [(2, 1), (3, 1), (1, 2), (3, 2), (4, 1), (2, 3), (5, 1), (1, 3)]:
        profiles.append(_make_beta_profile(n_main, p, q))
    # Tilted Hann windows
    for alpha in [1.0, 2.0, 3.0, -1.0, -2.0, 4.0]:
        profiles.append(_make_hann_tilted(n_main, alpha))
    # Parent (if available, interpolated)
    if parent is not None:
        n_p = len(parent)
        if n_p != n_main:
            x0 = np.linspace(0, 1, n_p)
            x1 = np.linspace(0, 1, n_main)
            p_interp = np.interp(x1, x0, parent)
            p_interp = np.maximum(p_interp, 1e-12)
            p_interp = p_interp / p_interp.sum()
            profiles.append(p_interp)
        else:
            profiles.append(parent.copy())

    # Score all profiles and pick top 4
    profile_scores = []
    for i, prof in enumerate(profiles):
        sc = _score(prof)
        profile_scores.append((sc, i))
        print(f"  Profile {i}: score={sc:.8f}")
    profile_scores.sort()
    top_profiles = [profiles[idx] for _, idx in profile_scores[:4]]
    print(f"  Top 4 profiles selected, best initial: {profile_scores[0][0]:.8f}")

    # Optimize each top profile with FISTA (coarse to fine)
    time_per_profile = min(300.0, (deadline - t_start - 120.0) / len(top_profiles))

    for pi, a_init in enumerate(top_profiles):
        if time.monotonic() > deadline - 60:
            break
        print(f"\n  FISTA profile {pi} (initial score={_score(a_init):.8f})")
        a_work = a_init.copy()

        # Coarse-to-fine FISTA
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = []
        beta = 0.05 * max_cc
        while beta > 0.0001 * max_cc:
            betas.append(beta)
            beta *= 0.4
        betas = betas[:8]

        for bi, beta in enumerate(betas):
            if time.monotonic() > deadline - 45:
                break
            phase_time = min(time_per_profile / max(1, len(betas) - bi), 60.0)
            phase_deadline = min(deadline - 40, time.monotonic() + phase_time)
            max_it = max(1000, int((phase_deadline - time.monotonic()) * 2000))
            a_work, sc = _fista_smoothed(a_work, beta, phase_deadline, max_it)
            a_work = a_work / a_work.sum()
            sc_full = _score(a_work)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_work.copy()
            if bi % 2 == 0:
                print(f"    beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

        # Adam fine-tuning
        if time.monotonic() < deadline - 60:
            c_curr = _full_convolve(a_work, a_work)
            max_cc = float(c_curr.max())
            betas_fine = [0.002 * max_cc, 0.0005 * max_cc, 0.0001 * max_cc, 0.00002 * max_cc]
            adam_dl = min(deadline - 30, time.monotonic() + time_per_profile * 0.4)
            a_opt, sc_opt = _adam_logspace(a_work, adam_dl, betas_fine, lr_init=0.01, max_iter_total=200000)
            sc_full = _score(a_opt)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_opt.copy()
            print(f"    After Adam: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    print(f"\n  Best after FISTA+Adam: {best_score:.10f}")

    # Active set polish on best
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Active set polish ===")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        a_pol = _active_set_polish(a_work, min(deadline - 20, time.monotonic() + 60))
        sc_pol = _score(a_pol)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
        print(f"  After active set: {best_score:.10f}")

    # Coordinate descent polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Coordinate descent polish ===")
        cd_dl = min(deadline - 5, time.monotonic() + 30)
        a_cd, _ = _coord_descent(best_seq, cd_dl, n_trials=3000)
        sc_cd = _score(a_cd)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
        print(f"  After coord descent: {best_score:.10f}")

    # Try n=4096 as well if time remains (parent length)
    if time.monotonic() < deadline - 60:
        print(f"\n=== Trying n=4096 ===")
        n_alt = 4096
        if parent is not None and len(parent) == n_alt:
            a_alt = parent.copy()
        else:
            a_alt = _make_beta_profile(n_alt, 2, 1)

        c_alt = _full_convolve(a_alt, a_alt)
        max_ca = float(c_alt.max())
        betas_alt = [0.05 * max_ca, 0.01 * max_ca, 0.002 * max_ca, 0.0003 * max_ca]
        for beta in betas_alt:
            if time.monotonic() > deadline - 30:
                break
            a_alt, sc = _fista_smoothed(a_alt, beta, min(deadline - 25, time.monotonic() + 40), 5000)
            a_alt = a_alt / a_alt.sum()
            sc_full = _score(a_alt)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_alt.copy()
                print(f"  n=4096 beta={beta:.2e}: score={sc_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"Final score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
