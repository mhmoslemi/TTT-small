Multi-phase optimizer for minimizing 2n*max(conv(p,p)) over probability vectors p.
Uses FISTA with log-sum-exp smoothing, top-K peak chasing, Frank-Wolfe hybrids,
multi-restart, and length exploration. Targets sub-1.5027 score.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming a is already normalized to sum=1."""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_weights(w, p):
    """
    Compute grad[j] = 2 * sum_k w[k] * p[k-j] for j=0..n-1.
    w has length 2n-1, p has length n.
    Uses FFT cross-correlation.
    """
    n = len(p)
    N = _next_pow2(2 * n + len(w))
    W = np.fft.rfft(w, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(W * np.conj(P), N)
    return 2.0 * corr[:n]


def _est_lipschitz(w, p, iters=20):
    """Estimate Lipschitz constant of gradient via power iteration on T(w)."""
    n = len(p)
    # T(w) is the matrix such that (T(w)*p)[j] = sum_k w[k]*p[k-j]
    # This is a symmetric Toeplitz-like matrix. Estimate ||T(w)||_2.
    x = np.ones(n) / np.sqrt(n)
    x = x - x.mean()
    x = x / (np.linalg.norm(x) + 1e-30)
    for _ in range(iters):
        y = _grad_weights(w, x) / 2.0  # T(w) * x
        y = y - y.mean()  # Project to tangent space of simplex
        ny = np.linalg.norm(y)
        if ny < 1e-15:
            return 2.0
        x = y / ny
    # Rayleigh quotient
    y = _grad_weights(w, x) / 2.0
    y = y - y.mean()
    lam = float(np.dot(x, y))
    return max(2.0 * lam, 1.0)


def _softmax_weights(c, beta):
    """Compute softmax weights for smoothing."""
    c_shifted = (c - c.max()) / beta
    np.clip(c_shifted, -500, 500, out=c_shifted)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _fista_lse(p_init, beta, deadline, max_iter, use_fw=False):

    n = len(p_init)
    p_curr = p_init.copy()
    p_prev = p_curr.copy()
    t_val = 1.0
    L = 4.0
    step = 1.0 / L
    best_p = p_curr.copy()
    best_sc = _score_norm(p_curr)
    iters_since_石狮 = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Momentum update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = p_curr + ((t_val - 1.0) / t_next) * (p_curr - p_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        # Compute gradient
        c_y = _convolve(y, y)
        w = _softmax_weights(c_y, beta)
        grad = _grad_weights(w, y)

        # Gradient step + projection
        p_next = _proj_simplex(y - step * grad)

        # Optional: Frank-Wolfe step every 500 iterations
        if use_fw and it % 500 == 499:

            fw_p = (1 - 2.0 / (it + 2)) * p_prev + (2.0 / (it + 2)) * np.zeros(n)
            # FW: move towards e_{j*}

            L = L_new

    sc_final = _score_norm(p_curr)
    if sc_final < best_sc:

def _fista_peakhinge(p_init, deadline, max_iter, K=5):
    """FISTA targeting top-K peaks with hinge loss."""
    n = len(p_init)
    p_curr = p_init.copy()
    p_prev = p_curr.copy()
    t_val = 1.0
    L = 4.0
    step = 1.0 / L
    best_p = p_curr.copy()
    best_sc = _score_norm(p_curr)

    # Initialize top-K
    c_curr = _convolve(p_curr, p_curr)
    topK = np.argsort(c_curr)[-K:]
    t_target = float(c_curr[topK].min())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Update top-K every 30 iterations
        if it % 30 == 0:
            c_curr = _convolve(p_curr, p_curr)
            topK = np.argsort(c_curr)[-K:]
            t_target = float(c_curr[topK].min())

        # Momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = p_curr + ((t_val - 1.0) / t_next) * (p_curr - p_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        # Hinge weights: alpha[k] = max(0, c[k] - t_target) for k in topK
        c_y = _convolve(y, y)
        w = np.zeros(2 * n - 1)
        for k in topK:
            hinge = c_y[k] - t_target
            if hinge > 0:
                w[k] = hinge
        w_sum = w.sum()
        if w_sum < 1e-30:
            # Fallback to uniform on topK
            w[topK] = 1.0 / K
        else:
            w = w / w_sum

        grad = _grad_weights(w, y)
        p_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(p_next)
            if sc < best_sc:
                best_sc = sc
                best_p = p_next.copy()

        if it % 500 == 250:
            L_new = _est_lipschitz(w, y, iters=15)
            L = L_new
            step = 1.0 / L

        p_prev = p_curr
        p_curr = p_next
        t_val = t_next

    sc_final = _score_norm(p_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_p = p_curr.copy()
    return best_p, best_sc


def _perturb_logspace(p, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(p, 1e-15))
    z = z + sigma * rng.standard_normal(len(p))
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()


def _sinusoidal_init(n, freq, alpha, rng):
    """Create a sinusoidally-modulated initialization."""
    j = np.arange(n)
    base = 1.0 + alpha * np.cos(2.0 * np.pi * freq * j / n)
    base = np.maximum(base, 0.01)
    base += 0.01 * rng.standard_normal(n)
    base = np.maximum(base, 0.001)
    return base / base.sum()


def _two_cluster_init(n, gap, rng):
    """Two-cluster initialization."""
    p = np.zeros(n)
    half = n // 2
    p[:half] = 1.0 + 0.3 * rng.standard_normal(half)
    p[half:] = 1.0 + 0.3 * rng.standard_normal(n - half)
    p = np.maximum(p, 0.01)
    return p / p.sum()


def _interpolate_profile(p, n_new):
    """Interpolate a profile to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _coord_descent_top(p_init, deadline, top_indices=None, max_rounds=200):
    """Coordinate descent on a subset of coordinates (most impactful)."""
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_sc = _score_norm(p)
    rng = np.random.default_rng(777)

    # Determine which coordinates to optimize
    if top_indices is None:
        c = _convolve(p, p)
        # Find coordinates that contribute most to the peak
        peak_idx = int(np.argmax(c))
        top_indices = np.arange(n)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        indices = top_indices.copy()
        rng.shuffle(indices)
        # Limit per round for large n
        if len(indices) > 256:
            indices = indices[:256]

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = p[j]
            for factor in [0.95, 0.98, 1.0, 1.02, 1.05]:
                p[j] = old_val * factor
                s = p.sum()
                if s < 1e-30:
                    p[j] = old_val
                    continue
                p_n = p / s
                sc = _score_norm(p_n)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_p = p_n.copy()
                    p = p_n.copy()
                    old_val = p[j]
                    improved = True
                else:
                    p[j] = old_val
        p = p / p.sum()
        if not improved and r > 10:
            break
    return best_p, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main search function. Returns best sequence found within budget."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # --- Initialization ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_main = 4096
    if parent is not None:
        n_main = len(parent)
        p0 = parent.copy()
    else:
        p0 = np.ones(n_main) / n_main

    sc0 = _score_norm(p0)
    print(f"Initial: n={n_main}, score={sc0:.10f}")
    best_seq = p0.copy()
    best_score = sc0

    c0 = _convolve(p0, p0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.8f}")

    # === Phase 1: FISTA with coarse beta continuation (0 - 450s) ===
    print("\n=== Phase 1: Coarse FISTA ===")
    phase1_end = min(deadline, t_start + 450.0)
    p_curr = p0.copy()

    # Beta schedule: geometric from 0.1*max_c to 0.005*max_c
    n_betas = 8
    beta_start = 0.1 * max_c0
    beta_end = 0.005 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for i, beta in enumerate(betas):
        remaining = phase1_end - time.monotonic()
        if remaining < 10:
            break
        per_beta_time = min(remaining, 50.0)
        max_it = int(per_beta_time * 2000)  # ~2000 iters/sec estimate
        p_curr, sc = _fista_lse(p_curr, beta, time.monotonic() + per_beta_time, max_it)
        p_curr = p_curr / p_curr.sum()
        sc_full = _score_norm(p_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = p_curr.copy()
        elapsed = time.monotonic() - t_start
        print(f"  P1 beta={beta:.2e} ({i+1}/{n_betas}): score={sc_full:.10f} t={elapsed:.0f}s")

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: FISTA with fine beta + FW (450s - 900s) ===
    print("\n=== Phase 2: Fine FISTA + FW ===")
    phase2_end = min(deadline, t_start + 900.0)
    c_curr = _convolve(p_curr, p_curr)
    max_cc = float(c_curr.max())

    n_betas2 = 8
    beta_start2 = 0.005 * max_cc
    beta_end2 = 0.0005 * max_cc
    betas2 = np.geomspace(beta_start2, beta_end2, n_betas2)

    for i, beta in enumerate(betas2):
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        per_beta_time = min(remaining, 50.0)
        max_it = int(per_beta_time * 2000)
        p_curr, sc = _fista_lse(p_curr, beta, time.monotonic() + per_beta_time, max_it, use_fw=True)
        p_curr = p_curr / p_curr.sum()
        sc_full = _score_norm(p_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = p_curr.copy()
        elapsed = time.monotonic() - t_start
        print(f"  P2 beta={beta:.2e} ({i+1}/{n_betas2}): score={sc_full:.10f} t={elapsed:.0f}s")

    print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: Peak-chasing with hinge (900s - 1200s) ===
    print("\n=== Phase 3: Peak chasing (hinge) ===")
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end - 30:
        p_work = best_seq.copy()
        p_work = p_work / p_work.sum()
        # Try different K values
        for K in [5, 8, 3]:
            remaining = phase3_end - time.monotonic()
            if remaining < 20:
                break
            per_K = min(remaining - 10, 80.0)
            max_it = int(per_K * 1500)
            p_work, sc = _fista_peakhinge(p_work, time.monotonic() + per_K, max_it, K=K)
            p_work = p_work / p_work.sum()
            sc_full = _score_norm(p_work)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_work.copy()
                print(f"  P3 K={K}: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    print(f"End Phase 3: best={best_score:.10f}")

    # === Phase 4: Multi-restart with diverse initializations (1200s - 1500s) ===
    print("\n=== Phase 4: Multi-restart ===")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end - 30:
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic() - 20) / n_restarts
        p_ref = best_seq.copy()

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            # Different initializations
            if i == 0:
                p_start = _perturb_logspace(p_ref, 0.01, rng)
            elif i == 1:
                p_start = _perturb_logspace(p_ref, 0.03, rng)
            elif i == 2:
                # Sinusoidal modulation
                p_start = _sinusoidal_init(n_main, freq=7, alpha=0.3, rng=rng)
                # Blend with reference
                p_start = 0.7 * p_ref + 0.3 * p_start
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()
            elif i == 3:
                # Roll
                shift = rng.integers(1, n_main // 4)
                p_start = np.roll(p_ref, shift).astype(float)
                p_start = p_start / p_start.sum()
            elif i == 4:
                # Two-cluster blend
                p_cluster = _two_cluster_init(n_main, n_main // 2, rng)
                p_start = 0.8 * p_ref + 0.2 * p_cluster
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()
            else:
                # Stronger perturbation
                p_start = _perturb_logspace(p_ref, 0.05, rng)

            # Short FISTA optimization
            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = np.geomspace(0.002 * max_cs, 0.0002 * max_cs, 4)

            p_cur = p_start.copy()
            for beta in betas_r:
                if time.monotonic() > phase4_end - 5:
                    break
                per_b = min(per_restart * 0.2, 20.0)
                p_cur, _ = _fista_lse(p_cur, beta, time.monotonic() + per_b,
                                       int(per_b * 2000))
                p_cur = p_cur / p_cur.sum()

            # Peak chase polish
            per_pc = min(per_restart * 0.3, 20.0)
            if per_pc > 5:
                p_cur, _ = _fista_peakhinge(p_cur, time.monotonic() + per_pc,
                                            int(per_pc * 1500), K=5)
                p_cur = p_cur / p_cur.sum()

            sc_r = _score_norm(p_cur)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p_cur.copy()
            print(f"  P4 restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.0f}s")

    print(f"End Phase 4: best={best_score:.10f}")

    # === Phase 5: Larger n exploration (1500s - 1700s) ===
    print("\n=== Phase 5: n=8192 exploration ===")
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end - 30 and best_seq is not None:
        for n_try in [8192]:
            if time.monotonic() > phase5_end - 30:
                break
            p_big = _interpolate_profile(best_seq, n_try)
            sc_big = _score_norm(p_big)
            print(f"  Interpolated n={n_try}: score={sc_big:.10f}")
            if sc_big < best_score:
                best_score = sc_big
                best_seq = p_big.copy()

            # Optimize the larger version
            if time.monotonic() < phase5_end - 20:
                c_big = _convolve(p_big, p_big)
                max_cb = float(c_big.max())
                betas_big = np.geomspace(0.005 * max_cb, 0.0005 * max_cb, 4)
                p_cur = p_big.copy()
                for beta in betas_big:
                    if time.monotonic() > phase5_end - 10:
                        break
                    per_b = min(25.0, phase5_end - time.monotonic() - 5)
                    if per_b < 5:
                        break
                    p_cur, _ = _fista_lse(p_cur, beta, time.monotonic() + per_b,
                                          int(per_b * 1500))
                    p_cur = p_cur / p_cur.sum()

                sc_big_opt = _score_norm(p_cur)
                if sc_big_opt < best_score:
                    best_score = sc_big_opt
                    best_seq = p_cur.copy()
                    print(f"  Optimized n={n_try}: score={sc_big_opt:.10f}")

    # === Phase 6: Final ultra-fine polish (1700s - deadline) ===
    print("\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        p_pol = best_seq.copy()
        p_pol = p_pol / p_pol.sum()
        c_pol = _convolve(p_pol, p_pol)
        max_cp = float(c_pol.max())

        # Ultra-fine beta FISTA
        betas_ult = np.geomspace(0.0001 * max_cp, 0.00001 * max_cp, 5)
        p_cur = p_pol.copy()
        per_beta = max(5.0, (deadline - time.monotonic() - 30) / len(betas_ult))
        for beta in betas_ult:
            if time.monotonic() > deadline - 20:
                break
            p_cur, _ = _fista_lse(p_cur, beta, time.monotonic() + per_beta,
                                  int(per_beta * 2000))
            p_cur = p_cur / p_cur.sum()
            sc_cur = _score_norm(p_cur)
            if sc_cur < best_score:
                best_score = sc_cur
                best_seq = p_cur.copy()

        # Short peak-chase
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic() - 10
            if remaining > 10:
                p_cur, _ = _fista_peakhinge(p_cur, time.monotonic() + remaining,
                                            int(remaining * 1500), K=3)
                p_cur = p_cur / p_cur.sum()
                sc_cur = _score_norm(p_cur)
                if sc_cur < best_score:
                    best_score = sc_cur
                    best_seq = p_cur.copy()

    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
