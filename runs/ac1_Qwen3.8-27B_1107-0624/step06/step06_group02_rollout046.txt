"""
Multi-phase optimizer targeting sub-1.5027 score for non-negative step functions.
Uses FISTA with aggressive beta continuation, Adam log-space optimization,
top-k smoothed max, block coordinate descent, and multi-scale exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1, so score = 2*n*max(conv)"""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(y, w):
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _grad_topk(y, c, k):
    """Gradient using top-k mask instead of softmax."""
    n = len(y)
    if k >= len(c):
        w = np.ones(len(c)) / len(c)
    else:
        w = np.zeros(len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        w[top_idx] = 1.0 / k
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _fista_smoothed(a_init, beta, deadline, max_iter, use_topk=False, topk=1):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        c_max = float(c_y.max())

        if use_topk and topk < len(c_y):
            w = np.zeros(len(c_y))
            top_idx = np.argpartition(c_y, -topk)[-topk:]
            w[top_idx] = 1.0 / topk
        else:
            exponents = (c_y - c_max) / beta
            np.clip(exponents, -500, 500, out=exponents)
            w = np.exp(exponents)
            w_sum = float(w.sum())
            if w_sum < 1e-300 or not np.isfinite(w_sum):
                w = np.zeros_like(c_y)
                w[int(np.argmax(c_y))] = 1.0
            else:
                w = w / w_sum

        grad = _grad_smoothed(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total, use_topk_frac=None):
    """Adam optimizer in log-space with adaptive beta.
    use_topk_frac: if set, use top-k smoothing where k = int(frac * len(c))
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        no_improve_count = 0
        phase_best = best_score

        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = np.exp(z - np.max(z))
            a = a / a.sum()
            c = _convolve(a, a)
            cm = float(c.max())

            if use_topk_frac is not None:
                k = max(1, int(use_topk_frac * len(c)))
                w = np.zeros(len(c))
                top_idx = np.argpartition(c, -k)[-k:]
                w[top_idx] = 1.0 / k
            else:
                c_shifted = (c - cm) / beta
                np.clip(c_shifted, -5000, 5000, out=c_shifted)
                e = np.exp(c_shifted)
                s = float(e.sum())
                if s < 1e-300 or not np.isfinite(s):
                    w = np.zeros_like(c)
                    w[int(np.argmax(c))] = 1.0
                else:
                    w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = np.exp(z - np.max(z))
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.92)

            # Early stop if no improvement for too long
            if no_improve_count > 200:
                break

    a_final = np.exp(z - np.max(z))
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=300, block_size=1):
    """Block coordinate descent with line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096 and block_size == 1:
            num_coords = min(256, n)
        else:
            num_coords = n

        improved = False
        for _ in range(num_coords):
            if time.monotonic() > deadline:
                break
            if block_size == 1:
                j = rng.integers(0, n)
                old_val = a[j]
                for factor in [0.95, 0.98, 1.02, 1.05]:
                    new_val = old_val * factor
                    if new_val < 0:
                        continue
                    a[j] = new_val
                    sc = _score_norm(a / a.sum())
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = old_val
                        break
            else:
                j = rng.integers(0, n - block_size + 1)
                block = a[j:j+block_size].copy()
                a_norm = a.sum()
                for _ in range(4):
                    pert = 1.0 + 0.05 * rng.standard_normal(block_size)
                    pert = np.maximum(pert, 0.1)
                    a[j:j+block_size] = block * pert
                    sc = _score_norm(a / a.sum())
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j:j+block_size] = block
                        break

        a = best_a.copy() if improved else a / a.sum()

        if not improved and r > 30:
            break

    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - np.max(z))
    return e / e.sum()


def _anneal_polish(a_init, deadline, n_iter=5000):
    """Simple simulated annealing polish."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(999)
    T = 0.001 * best_sc / (2.0 * n)
    T_min = T * 1e-6

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        T = max(T_min, T * (1.0 - 1.0 / n_iter))

        j = rng.integers(0, n)
        old = a[j]
        factor = 1.0 + T * 50.0 * rng.standard_normal()
        new_val = max(0.0, old * factor)
        a[j] = new_val
        s = a.sum()
        if s < 1e-30:
            a[j] = old
            continue
        a_n = a / s
        sc = _score_norm(a_n)

        if sc < best_sc or rng.random() < np.exp(-(sc - best_sc) / (2.0 * n * T)):
            if sc < best_sc:
                best_sc = sc
                best_a = a_n.copy()
            a = a_n
        else:
            a[j] = old

    return best_a, best_sc

                    time.monotonic() + 5, 10000
                )
                a_pol = a_pol / a_pol.sum()
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()

    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        # Ensure valid
        final = [float(max(0.0, x)) for x in final]
        sc_final = _score(np.array(final))
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return final

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
