"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous version:
1. Nesterov averaging in FISTA (averaging iterates gives better convergence for convex problems)
2. Piecewise-constant temperature schedule instead of continuous cooling
3. Larger active sets (top-20 peaks) for more stable gradients
4. Periodic reheat to escape local minima
5. Multi-n exploration (try n=8192 alongside n=4096)
6. Direct peak-flow interpretation for gradient computation
7. Better learning rate adaptation based on gradient magnitude
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=20):
    n = len(a)
    c_len = len(c)
    k = min(top_k, c_len)
    k_indices = np.argsort(c)[::-1][:k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / k
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_exact_subgradient(a, c, top_k=1):
    n = len(a)
    k = min(top_k, len(c))
    k_indices = np.argsort(c)[::-1][:k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / k
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=30):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t == 0.0:
            for t_val in [0.02, 0.05, 0.08, 0.12]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _fista_nesterov_avg(a_init, deadline, T_levels, steps_per_level=20000,
                        L_init=None, eval_interval=100):
    """FISTA with Nesterov averaging and piecewise-constant temperature.
    T_levels: list of temperatures to use, one per phase.
    Returns best_a (from averaging) and best_score.
    """
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    if L_init is None:
        L_init = 8.0 * n
    step = 1.0 / L_init

    total_steps = steps_per_level * len(T_levels)
    # Nesterov averaging: accumulate a weighted sum
    a_avg = np.zeros(n)
    avg_weight_total = 0.0
    patience = 0
    best_patience = best_score

    for level_idx, T in enumerate(T_levels):
        for t in range(steps_per_level):
            global_t = level_idx * steps_per_level + t
            if time.monotonic() > deadline:
                break

            c = _fft_convolve(y, y)
            g = _compute_soft_gradient(y, c, T)
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-10:
                # Backtracking line search
                local_step = step
                while local_step > 1e-12:
                    a_trial = y - local_step * g
                    a_trial = _project_simplex(a_trial)
                    sc_trial = _score_norm(a_trial)
                    sc_y = _score_norm(y)
                    if sc_trial <= sc_y + 0.5 * local_step * (g_norm ** 2):
                        break
                    local_step *= 0.5
                step = local_step
            step = max(step, 1e-12)
            step = min(step, 2.0 / L_init)

            a_new = y - step * g
            a_new = _project_simplex(a_new)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
            y = _project_simplex(y)
            t_k = t_new
            a = a_new

            # Nesterov averaging: weight = 1/(t+1) for recent, 1 for older
            w = 1.0 / (global_t + 1)
            a_avg += w * a
            avg_weight_total += w
            a_avg_norm = a_avg / avg_weight_total

            if global_t % eval_interval == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    patience = 0
                else:
                    patience += 1

                sc_avg = _score_norm(a_avg_norm)
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = a_avg_norm.copy()
                    patience = 0
                else:
                    patience += 1

                if patience > 20:
                    patience = 0

        # After each level, check average
        a_avg_norm = a_avg / avg_weight_total
        sc_avg = _score_norm(a_avg_norm)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg_norm.copy()
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _adam_z_nesterov_avg(a_init, deadline, T_start, T_end, lr_start, lr_end,
                         max_iter=500000, use_active_set=False, active_k=20,
                         eval_interval=200, clip_norm=0.8, reheat_interval=2000,
                         reheat_factor=2.0):
    """Adam in z-space with Nesterov averaging."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    # Nesterov averaging
    a_avg = np.zeros(n)
    avg_weight_total = 0.0

    # Reheat schedule
    current_T_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        base_T = T_start * (T_end / T_start) ** frac
        T = base_T * current_T_mult
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if use_active_set and frac > 0.3:
            g = _compute_active_gradient(a, c, active_k)
        else:
            g = _compute_soft_gradient(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Nesterov averaging
        w = 1.0 / (t + 1)
        a_avg += w * a
        avg_weight_total += w

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            a_avg_norm = a_avg / avg_weight_total
            sc_avg = _score_norm(a_avg_norm)

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            elif np.isfinite(sc_avg) and sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg_norm.copy()
                patience = 0
            else:
                patience += 1

            if patience > 8:
                if current_lr_mult > 0.1:
                    current_lr_mult *= 0.6
                patience = 0
                # Don't reset z - keep exploring

        # Periodic reheat
        if reheat_interval > 0 and t % reheat_interval == 0:
            current_T_mult = max(1.0, current_T_mult * reheat_factor)
        elif t % (reheat_interval // 2) == 0:
            current_T_mult = max(1.0, current_T_mult * 0.7)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    a_avg_norm = a_avg / max(avg_weight_total, 1e-15)
    sc_avg_final = _score_norm(a_avg_norm)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    if np.isfinite(sc_avg_final) and sc_avg_final < best_score:
        best_score = sc_avg_final
        best_a = a_avg_norm.copy()
    return best_a, best_score


def _peak_flow_descent(a, deadline, max_iter=50000, top_k=5, step_init=0.02):
    """Direct peak-flow descent: identify peak lags and move mass to reduce them."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    step = step_init

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        M = float(c.max())
        # Top k peaks
        k = min(top_k, len(c))
        k_indices = np.argsort(c)[::-1][:k]
        # Gradient: sum of shifted a for each peak
        g = np.zeros(n)
        for ki in k_indices:
            # c_ki = sum_j a_j * a_{ki-j}, dc_ki/da_j = a_{ki-j}
            # The shifted version: for each j, the contribution is a_{ki-j} if in range
            if 0 <= ki < n:
                g[:ki+1] += a[ki::-1]
                if ki+1 < n:
                    g[ki+1:] += a[n-1:ki-1:-1][:n-ki-1]
            elif ki >= n:
                shift = ki - (n - 1)
                g[shift:] += a[:n-shift]
        g /= k
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Step with projection
        a_new = a - step * g
        a_new = _project_simplex(a_new)
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        step *= 0.9995

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            a_pert_opt, sc_pert = _adam_z_nesterov_avg(a_pert, min(phase7_end - 5, t_start + 1760.0),
                                                       T_start=0.001 * max_cp, T_end=0.00002 * max_cp,
                                                       lr_start=0.02, lr_end=0.0015,
                                                       max_iter=50000,
                                                       use_active_set=True, active_k=10,
                                                       eval_interval=100,
                                                       reheat_interval=3000,
                                                       reheat_factor=1.0)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # === Phase 8: Final ultra-fine polish (1760s - deadline) ===
    if time.monotonic() < deadline - 10:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_z_nesterov_avg(best_a, deadline - 8,
                                                 T_start=0.0001 * max_cf, T_end=0.000001 * max_cf,
                                                 lr_start=0.003, lr_end=0.0002,
                                                 max_iter=200000,
                                                 use_active_set=True, active_k=5,
                                                 eval_interval=100,
                                                 reheat_interval=10000,
                                                 reheat_factor=1.0)
        full_sc = _score(a_final)
        print(f"Phase 8 final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.1f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
