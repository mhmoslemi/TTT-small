    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _spectral_flatten(a, alpha, n_iters=10):
    """Flatten the power spectrum to reduce conv peaks."""
    n = len(a)
    a = a.copy()
    N = _next_pow2(2 * n - 1)

    for _ in range(n_iters):
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P_mean = float(P.mean())
        # Geometric interpolation towards flat
        P_new = np.maximum(P, 1e-30) ** (1.0 - alpha) * max(P_mean, 1e-30) ** alpha
        # Reconstruct with modified magnitude, same phase
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = np.sqrt(P_new) * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s

    return a / a.sum()


def _peak_targeted_polish(a, n_passes, deadline, top_k_peaks=10):
    """Target the peak positions directly with 1D coordinate search."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(999)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        peak_idx = np.argsort(c)[-top_k_peaks:]
        for j in peak_idx:
            if time.monotonic() > deadline:
                break
            # conv[j] = sum_i a[i]*a[j-i] (valid indices)
            # Find the dominant contributors
            contribs = {}
            for i in range(n):
                k = j - i
                if 0 <= k < n:
                    contribs[i] = a[i] * a[k]
            if not contribs:
                continue
            # Sort by contribution
            sorted_items = sorted(contribs.items(), key=lambda x: -x[1])
            # Try reducing the top contributors
            for i, val in sorted_items[:5]:
                k = j - i
                if not (0 <= k < n):
                    continue
                # Try scaling a[i] down and a[k] down
                for scale in [0.7, 0.5, 0.3, 1.3, 1.5]:
                    a_trial = a.copy()
                    a_trial[i] *= scale
                    if i != k:
                        # Compensate by slightly increasing another element
                        a_trial[-1] += (a[i] - a_trial[i]) / max(1, n - 1)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        a = a_trial.copy()
                        break

        # Also try increasing small elements to spread mass
        q = np.quantile(a, 0.1)
        small_idx = np.where(a <= q)[0]
        if len(small_idx) > 0:
            for idx in rng.choice(small_idx, size=min(5, len(small_idx)), replace=False):
                a_trial = a.copy()
                a_trial[idx] *= 2.0
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-15:
                    a_trial = a_trial / s
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        a = a_trial.copy()

    return a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _structured_init(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        i = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * (i + 0.5) / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos2":
        i = np.arange(n)
        a = (0.5 * (1 + np.cos(2 * np.pi * i / n))) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal_cos":
        i = np.arange(n)
        a = 0.5 * (1 + np.cos(4 * np.pi * i / n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "rectified_sin":
        i = np.arange(n)
        a = np.maximum(np.sin(2 * np.pi * i / n), 0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangular":
        i = np.arange(n)
        mid = n // 2
        a = 1.0 - np.abs(i - mid) / mid
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _adam_direct(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    """Adam optimizer in direct space with non-negativity clamping."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            c, w, grad = _conv_and_grad(a, beta, top_k)
            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                grad = grad / gnorm

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            a_new = a - current_lr * mh / (np.sqrt(vh) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            a = a_new

            if t % 100 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if it_in_phase % 2000 == 0:
                current_lr = max(0.001, current_lr * 0.95)

        m_buf *= 0.5
        v_buf *= 0.5

    sc_final = _score_norm(a)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _simulated_annealing(a_init, deadline, n_trials=2000, temp_init=0.01, temp_decay=0.999):
    """Simulated annealing with structured perturbations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)
    temp = temp_init

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Generate perturbation
        pert_type = rng.integers(0, 4)
        a_trial = a.copy()

        if pert_type == 0:
            # Gaussian noise
            noise_scale = temp * a.mean()
            a_trial += noise_scale * rng.standard_normal(n)
        elif pert_type == 1:
            # Block perturbation
            block_size = rng.integers(n // 32, max(2, n // 4))
            start = rng.integers(0, n - block_size)
            a_trial[start:start + block_size] *= (1 + temp * rng.standard_normal())
        elif pert_type == 2:
            # Frequency perturbation
            freq = rng.uniform(0.5, 5.0)
            amp = temp * a.mean()
            a_trial += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        else:
            # Swap perturbation
            i1, i2 = rng.integers(0, n, 2)
            delta = a[i1] * temp
            a_trial[i1] -= delta
            a_trial[i2] += delta

        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-15:
            temp *= temp_decay
            continue
        a_trial = a_trial / s

        sc_trial = _score_norm(a_trial)
        sc_cur = _score_norm(a)

        if sc_trial < sc_cur or rng.random() < np.exp(-(sc_trial - sc_cur) / max(temp, 1e-10)):
            a = a_trial
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a_trial.copy()

        temp *= temp_decay

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_base = 4096

    if parent is not None and len(parent) == n_base:
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Parent: n={n_base}, score={best_score:.10f}")
    elif parent is not None:
        best_seq = _interp_to_length(parent, n_base)
        best_score = _score(best_seq)
        print(f"Parent interpolated: n={n_base}, score={best_score:.10f}")
    else:
        best_seq = _structured_init(n_base, "hann")
        best_score = _score(best_seq)
        print(f"No parent. Hann init: score={best_score:.10f}")

    # ===== Phase 1: Diverse initializations with quick optimization =====
    print(f"\n=== Phase 1: Diverse initializations (budget: 120s) ===")
    inits = {
        "parent": best_seq.copy(),
        "hann": _structured_init(n_base, "hann"),
        "raised_cos2": _structured_init(n_base, "raised_cos2"),
        "bimodal_cos": _structured_init(n_base, "bimodal_cos"),
        "rectified_sin": _structured_init(n_base, "rectified_sin"),
        "triangular": _structured_init(n_base, "triangular"),
        "uniform": _structured_init(n_base, "uniform"),
    }

    # Quick Adam on each initialization
    quick_deadline = t_start + 120
    quick_betas = [0.5, 0.2, 0.1]

    for name, a0 in inits.items():
        if time.monotonic() > quick_deadline:
            break
        a0 = np.maximum(a0, 0)
        a0 = a0 / a0.sum()
        sc0 = _score(a0)
        print(f"  {name}: initial score={sc0:.10f}")
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Quick optimization
        a_opt, sc_opt = _adam_logspace(a0, min(quick_deadline, time.monotonic() + 15),
                                       quick_betas, lr_init=0.01, top_k=100,
                                       max_iter=50000)
        sc_full = _score(a_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()
            print(f"  {name} after quick opt: score={sc_full:.10f}")

    print(f"  Best after Phase 1: {best_score:.10f}")

    # ===== Phase 2: Main Adam optimization from best =====
    print(f"\n=== Phase 2: Main Adam optimization (budget: 500s) ===")
    a_work = best_seq.copy()
    phase2_deadline = min(deadline - 600, t_start + 650)

    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())

    # Fine beta schedule
    betas_fine = []
    beta = 0.005 * max_cw
    while beta > 0.00001 * max_cw:
        betas_fine.append(beta)
        beta *= 0.6
    if len(betas_fine) > 15:
        betas_fine = betas_fine[:15]

    a_opt, sc_opt = _adam_logspace(a_work, phase2_deadline, betas_fine,
                                   lr_init=0.008, top_k=80, max_iter=1000000)
    sc_full = _score(a_opt)
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_opt.copy()
        print(f"  Phase 2 Adam: score={sc_full:.10f}")

    a_work = a_opt if sc_opt < _score_norm(best_seq) else best_seq.copy()
    a_work = np.maximum(a_work, 0)
    a_work = a_work / a_work.sum()

    # ===== Phase 3: Spectral flattening =====
    print(f"\n=== Phase 3: Spectral flattening (budget: 150s) ===")
    phase3_deadline = min(deadline - 400, time.monotonic() + 150)

    for alpha in [0.02, 0.05, 0.1, 0.05, 0.02]:
        if time.monotonic() > phase3_deadline:
            break
        a_spec = _spectral_flatten(a_work, alpha, n_iters=20)
        sc_spec = _score(a_spec)
        if sc_spec < best_score:
            best_score = sc_spec
            best_seq = a_spec.copy()
            a_work = a_spec.copy()
            print(f"  Spectral alpha={alpha}: score={sc_spec:.10f}")

        # Follow up with a few Adam steps
        remaining = phase3_deadline - time.monotonic()
        if remaining > 10:
            c_s = _full_convolve(a_spec, a_spec)
            mc = float(c_s.max())
            betas_s = [0.002 * mc, 0.001 * mc]
            a_sf, sc_sf = _adam_logspace(a_spec, time.monotonic() + min(20, remaining),
                                         betas_s, lr_init=0.005, top_k=60, max_iter=20000)
            sc_sf_full = _score(a_sf)
            if sc_sf_full < best_score:
                best_score = sc_sf_full
                best_seq = a_sf.copy()
                a_work = a_sf.copy()
                print(f"  Spectral+Adam alpha={alpha}: score={sc_sf_full:.10f}")

    # ===== Phase 4: Peak-targeted polish =====
    print(f"\n=== Phase 4: Peak-targeted polish (budget: 150s) ===")
    phase4_deadline = min(deadline - 250, time.monotonic() + 150)
    a_pt, sc_pt = _peak_targeted_polish(a_work, n_passes=20, deadline=phase4_deadline,
                                        top_k_peaks=15)
    sc_pt_full = _score(a_pt)
    if sc_pt_full < best_score:
        best_score = sc_pt_full
        best_seq = a_pt.copy()
        print(f"  Peak-targeted: score={sc_pt_full:.10f}")
    a_work = a_pt if sc_pt < _score_norm(a_work) else a_work

    # ===== Phase 5: Simulated annealing =====
    print(f"\n=== Phase 5: Simulated annealing (budget: 100s) ===")
    phase5_deadline = min(deadline - 150, time.monotonic() + 100)
    a_sa, sc_sa = _simulated_annealing(a_work, phase5_deadline,
                                       n_trials=5000, temp_init=0.005, temp_decay=0.9995)
    sc_sa_full = _score(a_sa)
    if sc_sa_full < best_score:
        best_score = sc_sa_full
        best_seq = a_sa.copy()
        print(f"  Simulated annealing: score={sc_sa_full:.10f}")
    a_work = a_sa if sc_sa < _score_norm(a_work) else a_work

    # ===== Phase 6: Diverse restarts =====
    print(f"\n=== Phase 6: Diverse restarts (remaining time) ===")
    n_restarts = 5
    remaining = deadline - time.monotonic() - 30
    per_restart = max(10, remaining / n_restarts)

    a_base = best_seq.copy()
    a_base = np.maximum(a_base, 0)
    a_base = a_base / a_base.sum()

    for ri in range(n_restarts):
        if time.monotonic() > deadline - 30:
            break
        r_deadline = time.monotonic() + per_restart

        if ri == 0:
            pert = 0.01 * rng.standard_normal(n_base)
            p_start = a_base + pert
        elif ri == 1:
            shift = rng.integers(1, max(2, n_base // 256))
            p_start = np.roll(a_base, shift).copy()
        elif ri == 2:
            # Modulate with a different frequency
            freq = rng.uniform(0.3, 3.0)
            amp = 0.02
            pert = amp * np.cos(2 * np.pi * freq * np.arange(n_base) / n_base)
            p_start = a_base + pert
        elif ri == 3:
            # Spectral perturbation starting point
            p_start = _spectral_flatten(a_base, alpha=0.15, n_iters=5)
        else:
            # Block swap
            p_start = a_base.copy()
            b1_start = rng.integers(0, n_base // 2)
            b2_start = rng.integers(n_base // 2, n_base)
            blk = n_base // 16
            p_start[b1_start:b1_start + blk], p_start[b2_start:b2_start + blk] = \
                p_start[b2_start:b2_start + blk].copy(), p_start[b1_start:b1_start + blk].copy()

        p_start = np.maximum(p_start, 1e-10)
        p_start = p_start / p_start.sum()

        c_p = _full_convolve(p_start, p_start)
        mc_p = float(c_p.max())
        betas_r = [0.003 * mc_p, 0.001 * mc_p, 0.0005 * mc_p]

        p_best, sc_best_r = _adam_logspace(p_start, r_deadline, betas_r,
                                           lr_init=0.006, top_k=70, max_iter=100000)
        sc_full = _score(p_best)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = p_best.copy()
            a_base = p_best.copy()
            print(f"  Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # ===== Phase 7: Final ultra-fine polish =====
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 7: Final polish ===")
        final_deadline = deadline - 10
        a_fin = best_seq.copy()
        a_fin = np.maximum(a_fin, 0)
        a_fin = a_fin / a_fin.sum()
        c_fin = _full_convolve(a_fin, a_fin)
        mc_fin = float(c_fin.max())
        betas_fin = [0.0003 * mc_fin, 0.0001 * mc_fin, 0.00005 * mc_fin]
        a_fin_opt, sc_fin = _adam_logspace(a_fin, final_deadline, betas_fin,
                                           lr_init=0.003, top_k=40, max_iter=200000)
        sc_fin_full = _score(a_fin_opt)
        if sc_fin_full < best_score:
            best_score = sc_fin_full
            best_seq = a_fin_opt.copy()
            print(f"  Final polish: score={sc_fin_full:.10f}")

    # Ensure valid output
    print(f"\n=== FINAL: best_score={best_score:.10f}, n={len(best_seq)}, time={time.monotonic()-t_start:.1f}s ===")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * (0.01 / 256)
    sc_final = _score(final)
    print(f"Output: score={sc_final:.10f}, n={len(final)}")
    return [float(x) for x in final]
